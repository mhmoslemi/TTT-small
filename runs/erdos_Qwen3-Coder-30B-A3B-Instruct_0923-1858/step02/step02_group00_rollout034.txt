import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()

        n_points = 45
        x = np.linspace(0, 2, n_points)
        # Use a piecewise linear function to start
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i]
            else:
                h[i] = 2 - x[i]
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Enhanced optimization procedure using locally weighted random search
    def stochastic_local_search(h_vec, n_iter=100):
        h_current = h_vec.copy()
        current_C5 = compute_C5(h_current)
        
        for _ in range(n_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Create neighbor by perturbing some components
            neighbor = h_current.copy()
            # Select random indices to perturb
            n_perturb = max(1, n_points // 10)
            perturb_indices = np.random.choice(n_points, size=n_perturb, replace=False)
            
            for idx in perturb_indices:
                # Add small random change
                delta = np.random.normal(0, 0.05)
                neighbor[idx] += delta
            
            neighbor = project_feasible(neighbor)
            candidate_C5 = compute_C5(neighbor)
            
            if candidate_C5 < current_C5:
                h_current = neighbor.copy()
                current_C5 = candidate_C5
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h_current.copy()
        
        return h_current

    # Starting with initial solution
    h = project_feasible(h)
    current_C5 = compute_C5(h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h.copy()
    
    # Main loop with local search
    max_patience = 50
    patience = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        iter_count += 1
        
        # Perform local search around current solution
        h = stochastic_local_search(h, n_iter=50)
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Random perturbation to escape local minima
            h = best_h.copy()
            idxs = np.random.choice(n_points, size=n_points//10, replace=False)
            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
            h = project_feasible(h)
            patience = 0
            
        # Occasionally re-evaluate best solution
        if iter_count % 20 == 0:
            test_C5 = compute_C5(best_h)
            if test_C5 < best_C5:
                best_C5 = test_C5

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
