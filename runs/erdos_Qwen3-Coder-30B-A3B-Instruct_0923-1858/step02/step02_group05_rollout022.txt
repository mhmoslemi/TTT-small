import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_particles = [initial_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    else:
        t = np.linspace(0, 2, n_points)
        h_particles = [0.5 + 0.3 * np.sin(np.pi * t) + np.random.uniform(-0.1, 0.1, n_points) 
                       for _ in range(n_particles)]
    
    # Project particles to feasible region
    for i in range(n_particles):
        h_particles[i] = project(h_particles[i], dx)
    
    # Initialize best positions
    p_best = [h_particles[i].copy() for i in range(n_particles)]
    p_best_C = [evaluate(h_particles[i], dx) for i in range(n_particles)]
    g_best_idx = np.argmin(p_best_C)
    g_best = p_best[g_best_idx].copy()
    g_best_C = p_best_C[g_best_idx]
    
    # PSO parameters
    c1, c2, w = 2.0, 2.0, 0.7
    max_iter_pso = 10000
    velocity_limit = 0.2
    
    t0 = time.time()
    max_time_pso = 0.7 * budget_s
    
    # PSO Exploration
    for iter_pso in range(max_iter_pso):
        if time.time() - t0 > max_time_pso:
            break
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            v = (w * (p_best[i] - h_particles[i]) + 
                 c1 * r1 * (g_best - h_particles[i]) + 
                 c2 * r2 * (g_best - h_particles[i]))
            
            # Clip velocities
            v = np.clip(v, -velocity_limit, velocity_limit)
            
            # Update position
            h_particles[i] += v
            h_particles[i] = project(h_particles[i], dx)
            
            # Evaluate
            current_C = evaluate(h_particles[i], dx)
            if current_C < p_best_C[i]:
                p_best[i] = h_particles[i].copy()
                p_best_C[i] = current_C
                if current_C < g_best_C:
                    g_best = h_particles[i].copy()
                    g_best_C = current_C
    
    # Gradient Descent Exploitation
    current_h = g_best.copy()
    lr = 0.005
    max_iter_gd = 10000
    max_time_gd = budget_s - (time.time() - t0)
    
    best_h = g_best.copy()
    best_C = g_best_C
    
    for iter_gd in range(max_iter_gd):
        if time.time() - t0 > budget_s:
            break
        if iter_gd % 1000 == 0 and iter_gd > 0:
            lr *= 0.98
            
        # Compute numerical gradient
        grad = np.zeros_like(current_h)
        eps = 1e-6
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        current_h -= lr * grad
        current_h = project(current_h, dx)
        
        # Check for improvement
        current_C = evaluate(current_h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
    
    return (best_h, best_C, n_points)
