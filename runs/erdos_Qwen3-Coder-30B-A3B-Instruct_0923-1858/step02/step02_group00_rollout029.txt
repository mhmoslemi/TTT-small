import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a better initial solution using a piecewise linear function
        n_points = 40
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a function that starts low, rises to 1, then drops
        for i in range(n_points):
            if x[i] < 0.5:
                h[i] = 2 * x[i]
            elif x[i] < 1.5:
                h[i] = 1
            else:
                h[i] = 2 - 2 * x[i]
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Quick helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Projection to feasibility set
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Solve a convex relaxation to get a good starting point
    def solve_convex_relaxation():
        # Perform a convex relaxation to get a good starting point
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Allow a few sample shifts to prevent numerical issues
        # Create a reasonable set of constraints for (x, x+k)
        # We will use a very simple relaxation by setting a few linear constraints
        # We could use a subset of shift constraints, but we stick to one main quadratic bound
        obj = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=500)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return np.array(h_var.value)
        except:
            pass
        return h.copy()

    # A more efficient approach for gradient evaluation
    def compute_gradient(h_vec, k_max):
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        
        # Only compute for a subset of relevant indices
        # Determine which elements might influence the maximum
        eps = 1e-5
        h_plus = h_vec.copy()
        h_minus = h_vec.copy()
        
        for i in range(n_points):
            h_plus[i] = h_vec[i] + eps
            h_minus[i] = h_vec[i] - eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[k_max + n_points//2] - corr_minus[k_max + n_points//2]) / (2 * eps)
            
            # Reset the changes
            h_plus[i] = h_vec[i]
            h_minus[i] = h_vec[i]
            
        return grad

    # SGD steps
    def sgd_step(h_vec, lr=0.01, iters=5):
        h_vec = h_vec.copy()
        for _ in range(iters):
            # Compute max shift
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            k_max = np.argmax(corr) - n_points//2
            
            # Compute gradients
            grad = compute_gradient(h_vec, k_max)
            h_vec = h_vec - lr * grad
            h_vec = project_feasible(h_vec)
        return h_vec
    
    def perturb_solution(h_vec, prob=0.1):
        h_vec = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_vec[i] = np.clip(h_vec[i] + np.random.normal(0, 0.03), 0, 1)
        return project_feasible(h_vec)
    
    # Main loop
    max_patience = 100
    patience = 0
    eta = 0.01
    iter_count = 0
    last_update = 0
    
    # Get a better initial solution with relaxation
    h = solve_convex_relaxation()
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # New random perturbation
                h = perturb_solution(h, prob=0.15)
                patience = 0

        # Adapt step size
        if iter_count % 20 == 0:
            eta *= 0.98
            if eta < 1e-5:
                eta = 1e-5

        # Optimizer step
        h = sgd_step(h, lr=eta, iters=10)
        iter_count += 1
        
        # Occasionally recompute using a better estimation
        if iter_count % 30 == 0:
            if patience > 0:
                h = sgd_step(h, lr=eta, iters=1)
            else:
                h = sgd_step(h, lr=eta, iters=3)

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
