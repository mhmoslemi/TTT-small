import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    
    try:

    except NameError:
        # Create a base starting point
        t = np.linspace(0, 2, n_points)
        base_h = 0.5 + 0.3 * np.sin(np.pi * t)
    
    # Initialize particles with small perturbations
    for i in range(n_particles):
        pert = np.random.uniform(-0.1, 0.1, n_points)
        particles[i] = base_h + pert
        particles[i] = project(particles[i], dx)
    
    # PSO parameters
    c1, c2 = 2.0, 2.0
    w = 0.7
    v_max = 0.2
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Initialize personal best
    p_best = particles.copy()
    p_best_C = np.array([evaluate(p, dx) for p in p_best])
    
    # Global best
    best_idx = np.argmin(p_best_C)
    g_best = p_best[best_idx].copy()
    best_C = p_best_C[best_idx]
    
    # PSO exploration phase (70% budget)
    t0 = time.time()
    max_time = 0.7 * budget_s
    iteration = 0
    
    while time.time() - t0 < max_time:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_C = evaluate(particles[i], dx)
            
            # Update personal best
            if current_C < p_best_C[i]:
                p_best_C[i] = current_C
                p_best[i] = particles[i].copy()
                
                # Update global best
                if current_C < best_C:
                    best_C = current_C
                    g_best = particles[i].copy()
        
        iteration += 1
    
    # Gradient descent exploitation phase (30% budget)
    current_h = g_best.copy()
    lr = 0.005
    lr_decay = 0.98
    decay_interval = 1000
    
    max_time_exploit = 0.3 * budget_s
    start_time = time.time()
    
    while time.time() - start_time < max_time_exploit:
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update
        current_h -= lr * grad
        current_h = project(current_h, dx)
        
        # Evaluate
        current_C = evaluate(current_h, dx)
        
        if current_C < best_C:
            best_C = current_C
            g_best = current_h.copy()
        
        # Decay learning rate
        if iteration % decay_interval == 0:
            lr *= lr_decay
            
        iteration += 1
    
    # Final validation
    final_h = project(g_best, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
