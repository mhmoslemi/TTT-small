def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""

    max_corr = np.max(corr)
    return max_corr * dx

    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h
