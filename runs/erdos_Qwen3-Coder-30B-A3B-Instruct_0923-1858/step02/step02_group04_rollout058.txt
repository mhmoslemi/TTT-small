import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing
    T = 1.0
    cooling_rate = 0.99
    min_T = 1e-6
    step_size = 0.1
    iter_count = 0
    max_iter = 100000

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit or iter_count >= max_iter:
            break
            
        # Perturb the current solution
        h_new = h.copy()
        num_perturb = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for idx in indices:
            perturbation = np.random.normal(0, step_size)
            h_new[idx] += perturbation
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Compute energy difference
        C5_current = compute_C5(h)
        C5_new = compute_C5(h_new)
        delta = C5_new - C5_current
        
        # Accept or reject
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        
        # Cool down
        T = max(min_T, T * cooling_rate)
        step_size = max(0.001, step_size * cooling_rate)
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
