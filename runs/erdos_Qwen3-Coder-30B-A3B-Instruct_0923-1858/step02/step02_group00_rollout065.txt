import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a better initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        # Use a piecewise linear function
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i]
            else:
                h[i] = 2 - x[i]
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Particle Swarm Optimization
    def pso_optimize():
        # PSO parameters
        n_particles = 50
        max_iter = 100
        w_start, w_end = 0.9, 0.4
        c1, c2 = 2.0, 2.0
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        
        for _ in range(n_particles):
            # Initialize with small random perturbations of best_h
            particle = np.random.normal(best_h, 0.1)
            particle = project_feasible(particle)
            particles.append(particle.copy())
            velocities.append(np.zeros(n_points))
            pbest.append(particle.copy())
            pbest_vals.append(compute_C5(particle))
        
        # Global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        for iter_num in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            w = w_start - (w_start - w_end) * iter_num / max_iter
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] = particles[i] + velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate fitness
                fitness = compute_C5(particles[i])
                
                # Update personal best
                if fitness < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = fitness
                    
                    # Update global best
                    if fitness < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = fitness
                        
        return gbest, gbest_val
    
    # Gradient-based refinement
    def gradient_refinement(h_init, max_iter=50):
        h = h_init.copy()
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            # Compute current correlation
            corr = correlate(h, 1 - h, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h) - 1)
            
            # Gradient approximation
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_feasible(h_minus)
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update with learning rate decay
            eta = 0.02 / (1 + 0.01 * _)
            h = h - eta * grad
            h = project_feasible(h)
            
        return h
    
    # Main optimization loop
    pso_time = 0
    refine_time = 0
    
    while elapsed() < budget_s * 0.95:
        # PSO phase
        start_pso = time.time()
        h_pso, C5_pso = pso_optimize()
        pso_time += time.time() - start_pso
        
        # Refinement phase
        start_refine = time.time()
        h_refined = gradient_refinement(h_pso)
        C5_refined = compute_C5(h_refined)
        refine_time += time.time() - start_refine
        
        # Update best
        if C5_refined < best_C5:
            best_C5 = C5_refined
            best_h = h_refined.copy()
            
        # Exit if staying time for final processing
        if elapsed() > budget_s * 0.95:
            break
    
    # Final evaluation and return
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
