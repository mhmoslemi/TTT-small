import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available
    try:

        if len(initial_h) != n_points:

        initial_h = np.zeros(n_points)
        # First 15 points: increasing linearly from 0 to 1
        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))
    
    def objective(h):
        return evaluate(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Setup constraints
    bounds = [(0, 1) for _ in range(n_points)]
    con = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = initial_h.copy()
    best_C = evaluate(initial_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave headroom
    
    # Run optimization
    try:
        res = minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=con,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
            tol=1e-9
        )
        
        if res.success:
            h_opt = np.clip(res.x, 0, 1)
            h_opt = h_opt * (1.0 / (np.sum(h_opt) * dx))
            current_C = evaluate(h_opt, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h_opt
    except Exception as e:
        pass
    
    # Post-process with local refinement
    h = best_h.copy()
    lr = 0.01
    for i in range(5000):
        if time.time() - t0 > max_time:
            break
            
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (1.0 / (np.sum(h_plus) * dx))

            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (1.0 / (np.sum(h_minus) * dx))

            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)

        # Update step
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (1.0 / (np.sum(h) * dx))
        
        # Update best if improved
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
            
        if i % 100 == 0:
            lr *= 0.995  # Decay learning rate

    return (best_h, best_C, n_points)
