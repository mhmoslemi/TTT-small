import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, segments=10):
    """Generate a piecewise constant individual with given number of segments."""
    h = np.zeros(n_points)
    segment_size = n_points // segments
    for i in range(segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < segments - 1 else n_points
        h[start:end] = np.random.uniform(0.1, 0.9, size=end-start)
    return h

def crossover(parent1, parent2, dx):
    """Single-point crossover between two individuals."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return project(child1, dx), project(child2, dx)

def mutate(individual, dx, mutation_rate=0.1, sigma=0.1):
    """Mutate an individual by adjusting segment heights."""
    n_points = len(individual)
    segments = 10
    segment_size = n_points // segments
    for i in range(segments):
        if np.random.random() < mutation_rate:
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            # Adjust segment height with normal distribution
            adjustment = np.random.normal(0, sigma, size=end-start)
            individual[start:end] += adjustment
    return project(individual, dx)

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), size=k, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmax(tournament_fitnesses)]
    return population[winner_index]
