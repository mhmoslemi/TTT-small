import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Enhanced optimization with momentum and multiple strategies
    def compute_gradient(h_vec, k_idx):
        # Compute finite difference gradient for the specific k index
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        h_plus = h_vec.copy()
        h_minus = h_vec.copy()
        
        for i in range(len(h_vec)):
            h_plus[i] = h_vec[i] + eps
            h_minus[i] = h_vec[i] - eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")[k_idx]
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")[k_idx]
            
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
            
            # Reset for next iteration
            h_plus[i] = h_vec[i]
            h_minus[i] = h_vec[i]
            
        return grad

    # Mini-batch gradient descent approach
    def run_minibatch_descent(h_vec, num_steps=100, batch_size=10):
        h_current = h_vec.copy() 
        # Momentum term
        velocity = np.zeros_like(h_current)
        momentum = 0.9
        learning_rate = 0.01
        
        for step in range(num_steps):
            # Sample random shifts for gradient estimation
            k_indices = np.random.randint(0, len(h_current) * 2 - 1, batch_size)
            
            # Compute average gradient over batch
            avg_grad = np.zeros_like(h_current)
            for k_idx in k_indices:
                grad = compute_gradient(h_current, k_idx)
                avg_grad += grad
            
            if batch_size > 0:
                avg_grad /= batch_size
            
            # Update with momentum
            velocity = momentum * velocity - learning_rate * avg_grad
            h_current = h_current + velocity
            
            # Project back to feasible set
            h_current = project_feasible(h_current)
            
            # Decay learning rate
            if step % 20 == 0 and step > 0:
                learning_rate *= 0.95
                
        return h_current

            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)

            patience = 0

    # Final validation

    if final_C5 < best_C5:

        best_h = best_h.copy()
