import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Try to use the initial construction if available
    try:
        h0 = initial_h_values
        if len(h0) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a reasonable initial guess
        t = np.linspace(0, 2, n_points)
        h0 = np.exp(-((t - 1)**2) / 0.1)  # Gaussian peak at center
        h0 = h0 / (np.sum(h0) * dx)  # Normalize
    
    start_time = time.time()
    
    # Try convex optimization approach
    try:
        # Define variables
        h = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1, sum(h) * dx == 1, t >= 0]

        # Approximate with a simpler approach: just compute a few key overlaps
        
        # Try a simpler approach with direct optimization with a warm start
        h_current = h0.copy()
        best_C = np.inf
        
        # Reuse some idea from the previous approach
        # But now use a better optimization method
        
        # Use a basic gradient descent approach with fixed step size
        max_iter = 10000
        lr = 0.01
        patience = 0
        max_patience = 50
        
        h_prev = h_current.copy()
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient (finite difference)
            grad = np.zeros_like(h_current)
            eps = 1e-4
            
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / (np.sum(h_plus) * dx)
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus / (np.sum(h_minus) * dx)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            # Update
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / (np.sum(h_new) * dx)
            
            # Evaluate
            current_C = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
            
            if current_C < best_C:
                h_current = h_new
                best_C = current_C
                patience = 0
                # Adaptive learning rate
                lr = min(0.1, lr * 1.02)
            else:
                h_current = h_new
                patience += 1
                if patience > max_patience:
                    lr = max(1e-6, lr * 0.9)
                    patience = 0
            
            # Occasionally noise injection
            if i % 30 == 0 and np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.005, len(h_current))
                h_current = np.clip(h_current + noise, 0, 1)
                h_current = h_current / (np.sum(h_current) * dx)
        
        # Return final result
        return (h_current, best_C, n_points)
        
    except Exception as e:
        # Fallback to simple approach if optimization fails
        h_current = h0.copy()

        lr = 0.01

            if time.time() - start_time > budget_s - 5:

            eps = 1e-5

            val_current = np.max(corr_current) * dx
            
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / (np.sum(h_plus) * dx)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (np.max(corr_plus) * dx - val_current) / eps
            
            # Update
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / (np.sum(h_new) * dx)
            
            # Evaluate
            current_C = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
            
            if current_C < best_C:
                h_current = h_new
                best_C = current_C
                lr = min(0.1, lr * 1.01)
            else:
                lr = max(1e-5, lr * 0.99)
            
            if i % 100 == 0:
                pass
        
        return (h_current, best_C, n_points)
