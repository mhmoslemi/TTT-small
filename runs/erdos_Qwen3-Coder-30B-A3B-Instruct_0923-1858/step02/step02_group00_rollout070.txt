import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Exponential smoothing approach
    def smooth_objective(h_vec, beta=1000.0, gamma=1e-4):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Normalize correlation values to avoid numerical issues
        corr_max = np.max(corr)
        if corr_max > 0:
            corr_norm = corr / corr_max
        else:
            corr_norm = corr
        penalty = np.sum(np.exp(-beta * corr_norm))
        # Barrier terms for bounds
        barrier = -gamma * np.sum(np.log(h_vec + 1e-10) + np.log(1 - h_vec + 1e-10))
        return penalty + barrier

    def compute_gradient(h_vec, beta=1000.0, gamma=1e-4, eps=1e-5):
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        
        # Compute base correlation for reference
        base_corr = correlate(h_vec, 1 - h_vec, mode="full")
        base_corr_max = np.max(base_corr)
        if base_corr_max > 0:
            base_corr_norm = base_corr / base_corr_max
        else:
            base_corr_norm = base_corr
            
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Project
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            # Evaluate at difference points
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")

            if base_corr_max > 0:
                corr_plus_norm = corr_plus / base_corr_max
                corr_minus_norm = corr_minus / base_corr_max
            else:
                corr_plus_norm = corr_plus
                corr_minus_norm = corr_minus
                
            penalty_plus = np.sum(np.exp(-beta * corr_plus_norm))
            penalty_minus = np.sum(np.exp(-beta * corr_minus_norm))
            
            # Gradient of penalty term
            grad_penalty = (penalty_plus - penalty_minus) / (2 * eps)
            
            # Gradient of barrier term
            grad_barrier = -gamma / (h_vec[i] + 1e-10) + gamma / (1 - h_vec[i] + 1e-10)
            
            grad[i] = grad_penalty + grad_barrier
            
        return grad

    # Optimization loop
    beta = 1000.0
    gamma = 1e-4
    eta = 1e-5
    patience = 0
    max_patience = 100
    
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb the solution
                idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
                
        # Update with gradient descent
        gradient = compute_gradient(h, beta, gamma)
        h = h - eta * gradient
        h = project_feasible(h)
        
        # Adaptive parameters
        if iter_count % 50 == 0:
            beta *= 0.99
            gamma *= 1.1

        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.5
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
