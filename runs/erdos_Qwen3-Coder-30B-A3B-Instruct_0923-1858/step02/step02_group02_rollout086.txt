import numpy as np

def project_h(h, dx):
    """Project h to satisfy ∫h dx = 1 with 0 ≤ h[i] ≤ 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate maximum overlap integral C5."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def init_segments(n_points, n_segments=10):
    """Initialize piecewise constant segments."""
    h = np.zeros(n_points)
    segment_size = n_points // n_segments
    for i in range(n_segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < n_segments - 1 else n_points
        h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
    return project_h(h, 2.0 / n_points)

def optimize_heights(h_segments, positions, n_points, dx, n_segments):
    """Optimize segment heights via convex relaxation."""
    # Simplified constraint: use least squares approximation
    # Since full optimization is complex, we use a coarse projection
    h_new = np.zeros(n_points)
    for i in range(n_segments):
        start = int(positions[i] / 2 * n_points)
        end = int(positions[i+1] / 2 * n_points)
        h_new[start:end] = h_segments[i]
    return project_h(h_new, dx)

def optimize_positions(h_full, positions, n_points, dx, n_segments, lr=0.05):
    """Optimize segment positions using gradient approximation."""
    grad = np.zeros(n_segments - 1)
    epsilon = 1e-4

        # Perturb position

        pos_plus[i] += epsilon
        pos_plus = np.clip(pos_plus, 0, 2)
        h_plus = reconstruct_h(pos_plus, h_full, n_points, n_segments)
        cost_plus = evaluate(h_plus, dx)

        pos_minus = positions.copy()
        pos_minus[i] -= epsilon
        pos_minus = np.clip(pos_minus, 0, 2)
        h_minus = reconstruct_h(pos_minus, h_full, n_points, n_segments)
        cost_minus = evaluate(h_minus, dx)

        grad[i] = (cost_plus - cost_minus) / (2 * epsilon)
        
    # Update positions
    new_positions = positions - lr * grad
    new_positions = np.clip(new_positions, 0, 2)
    return new_positions

def reconstruct_h(positions, h_full, n_points, n_segments):
    """Reconstruct h from segment positions and heights."""
    h = np.zeros(n_points)
    for i in range(n_segments):
        start_idx = int(positions[i] / 2 * n_points)
        end_idx = int(positions[i+1] / 2 * n_points)
        if i < n_segments - 1:
            h[start_idx:end_idx] = h_full[i]
        else:
            # Last segment
            h[start_idx:] = h_full[i]
    return project_h(h, 2.0 / n_points)
