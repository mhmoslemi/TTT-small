import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))

    mutation_rate = 0.1

    def create_individual():
        h = np.zeros(n_points)
        segments = 10
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        h = project(h, dx)
        return h

    population = [create_individual() for _ in range(population_size)]
    
    # If initial_h_values exists, inject into population
    try:
        init_h = initial_h_values.copy()
        init_h = project(init_h, dx)
        population[0] = init_h

    # Evaluate initial population
    fitness = [1.0 / (evaluate(ind, dx) + 1e-10) for ind in population]

        individual_evals.append(evaluate(population[i], dx))
    
    for gen in range(generations):

        def select():

        # Decrease mutation rate over time
        mutation_rate = max(0.02, mutation_rate * 0.999)
        
        # Early stopping
        if gen - last_improvement > no_improvement_limit:
            break
    
    return (best_h, best_C, n_points)
