import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Initialize from initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Use simulated annealing with local search
    def local_search(h_vec, temp):
        h_copy = h_vec.copy()
        idx = np.random.randint(0, n_points)
        # Perturb one element
        h_copy[idx] += np.random.normal(0, temp)
        h_copy = project_feasible(h_copy)
        return h_copy
    
    # Parameters for the approach
    T0 = 0.1
    decay = 0.995
    min_temp = 1e-6
    patience = 0
    max_patience = 100
    
    h_current = h.copy()
    current_C5 = compute_C5(h_current)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h_current.copy()
    
    # Optimization loop
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        T = T0 * (decay ** it)
        if T < min_temp:
            T = min_temp
            
        h_new = local_search(h_current, T)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < current_C5:
            h_current = h_new
            current_C5 = new_C5
            patience = 0
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()
        else:
            # Accept with probability based on temperature
            delta = new_C5 - current_C5
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new
                current_C5 = new_C5
            patience += 1
            
        if patience > max_patience:
            # Random restart
            h_current = np.random.rand(n_points)
            h_current = project_feasible(h_current)
            current_C5 = compute_C5(h_current)
            patience = 0
            
        it += 1
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
