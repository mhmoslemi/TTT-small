import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_initial_h(n_points, dx):
    """Create a symmetric initial guess for h."""
    h = np.zeros(n_points)
    peak = 1.0
    decay_points = n_points // 2
    h[:decay_points] = np.linspace(0, peak, decay_points)
    h[decay_points:] = np.linspace(peak, 0, decay_points + 1)[1:]
    h = project(h, dx)
    return h

def compute_gradient(h, dx):
    """Compute gradient using finite differences."""
    eps = 1e-5
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        h_plus = project(h_plus, dx)
        h_minus = project(h_minus, dx)
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def time_expired(start_time, budget):
    """Check if time budget has expired."""
    return time.time() - start_time > budget * 0.95

    h = initial_h.copy()

    t0 = 0.5

    velocity = np.zeros_like(h)

    max_iter_coarse = 10000  # Coarse exploration
    max_iter_fine = 5000     # Fine refinement
    iter_count = 0
    
    # Optimization loop
    for i in range(max_iter_total):
        if time_expired(start_time, budget_s):
            break
            
        # Compute gradient
        grad = compute_gradient(h, 2.0 / n_points)
        
        # Update velocities and position using FIRE
        velocity = alpha * velocity + beta * grad
        h_trial = h + velocity
        
        # Enforce constraints
        h_trial = project(h_trial, 2.0 / n_points)
        
        # Compute trial objective
        trial_C = evaluate(h_trial, 2.0 / n_points)
        
        # Check trial improvement
        if trial_C < best_C:

            best_C = trial_C
            current_step = 1.0

            velocity = velocity * 0.9
