import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        h = project(h, dx)
        
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Slightly perturb the starting point
        h_pert = h + np.random.uniform(-0.1, 0.1, n_points)
        h_pert = project(h_pert, dx)
        particles.append(h_pert.copy())
        velocities.append(np.zeros(n_points))
        
        # Evaluate the particle
        val = evaluate(h_pert, dx)
        p_best.append(h_pert.copy())
        p_best_vals.append(val)
    
    # Find global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # Begin PSO iterations
    t0 = time.time()
    max_time = budget_s - 2
    
    iteration = 0
    while time.time() - t0 < max_time:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (p_best[i] - particles[i])
                           + c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            val = evaluate(particles[i], dx)
            
            # Update personal best
            if val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = val
                
                # Update global best
                if val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = val
        
        # Break if no improvement for a while
        if iteration % 100 == 0:
            if iteration > 1000:
                break

    return (g_best, g_best_val, n_points)
