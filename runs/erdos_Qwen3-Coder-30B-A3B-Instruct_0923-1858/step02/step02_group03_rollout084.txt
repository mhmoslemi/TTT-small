import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values at points t."""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip h to [0,1] and normalize so that integral is 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate C5 for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial if available

        if len(initial_h) != n_points:

    except NameError:

        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)
        initial_h = clip_and_normalize(initial_h, dx)
    
    # PSO parameters
    N = 10  # Number of Fourier coefficients
    P = 50  # Number of particles
    max_iter = 1000
    omega = 0.7
    phi1 = phi2 = 1.5
    
    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N+1))  # a0, a1..aN, b1..bN
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_values = np.full(P, np.inf)
    
    # Global best
    gbest_idx = 0
    gbest_value = np.inf
    
    start_time = time.time()
    iter_count = 0
    time_limit = budget_s - 10
    
    # Mapping from particles to h
    def particle_to_h(particle):
        a = particle[:N+1]
        b = particle[N+1:]
        t = np.linspace(0, 2, n_points)
        h = fourier_to_h(a, b, t)
        h = clip_and_normalize(h, dx)
        return h
    
    while iter_count < max_iter and time.time() - start_time < time_limit:
        # Evaluate all particles
        for i in range(P):
            h = particle_to_h(particles[i])
            C = evaluate(h, dx)
            
            if C < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = C
                
            if C < gbest_value:
                gbest_value = C
                gbest_idx = i
        
        # Update velocities and positions
        r1 = np.random.random((P, 2*N+1))
        r2 = np.random.random((P, 2*N+1))
        
        for i in range(P):
            velocities[i] = (omega * velocities[i] +
                             phi1 * r1[i] * (pbest[i] - particles[i]) +
                             phi2 * r2[i] * (pbest[gbest_idx] - particles[i]))
            
            particles[i] += velocities[i]
        
        iter_count += 1
    
    # Get best solution
    best_h = particle_to_h(pbest[gbest_idx])
    
    # Final check
    final_C = evaluate(best_h, dx)
    
    return (best_h, final_C, n_points)
