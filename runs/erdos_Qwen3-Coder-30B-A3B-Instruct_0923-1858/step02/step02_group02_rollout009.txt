import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    population_size = 50

    def create_individual():
        segments = np.random.uniform(0, 1, num_segments)

        segments = segments * (n_points / (num_segments * np.sum(segments)))
        # Create piecewise constant function
        individual = np.repeat(segments, n_points // num_segments)

        return project(individual, dx)

    population = []
    if 'initial_h_values' in globals():

        for i in range(population_size // 5, population_size):

        best_idx = np.random.randint(0, len(pop))

        return pop[best_idx]

        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()

        crossover_point = np.random.randint(1, len(parent1))
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])

            old_best_C = best_C

    return (best_h, best_C, n_points)
