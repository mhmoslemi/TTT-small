import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    w = 0.7
    c1 = 2.0
    c2 = 2.0
    n_particles = 50
    max_iter = 1000
    velocity_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Slight perturbation of the initial solution
        h_particle = h + np.random.uniform(-0.1, 0.1, n_points)
        h_particle = project(h_particle, dx)
        particles.append(h_particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h_particle.copy())
        p_best_vals.append(evaluate(h_particle, dx))
    
    # Find the global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    t0 = time.time()
    iter_count = 0
    
    while time.time() - t0 < budget_s - 2 and iter_count < max_iter:
        iter_count += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate and update personal best
            current_val = evaluate(particles[i], dx)
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best if needed
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
    
    return (g_best, g_best_val, n_points)
