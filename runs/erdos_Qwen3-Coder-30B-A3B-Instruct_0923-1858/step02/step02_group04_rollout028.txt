import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    time_limit = budget_s * 0.95
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Create CVXPY variables
    h_vars = cp.Variable(n_points)
    
    # Create constraints
    constraints = [h_vars >= 0, h_vars <= 1]
    constraints.append(cp.sum(h_vars) == n_points / 2)
    
    # For each shift k, we want sum(h[i] * (1 - h[i+k])) <= t
    # This can be done explicitly by constructing the correlation constraints
    t = cp.Variable()
    # We'll solve a feasibility problem directly or use a max approach
    
    # Try to better structure constraints:
    # For each possible shift k (from -n_points+1 to n_points-1)
    # overlap[k] = sum_{i=0}^{n_points-1} h[i] * (1 - h[i + k]) 
    # where h[i + k] is 0 if index out of bounds
    # We'll model this as a linear program with a max added constraint
    
    # Let's try a simpler convex approach: minimize the maximum overlap
    # This can be badly handled by the original formulation due to complexity.
    # Instead, we'll solve it with a more structured constraint setup via CVXPY
    
    # Formulate the convex optimization:
    # Variables: h_vars, t
    # Objective: minimize t
    # Constraints:
    #   h_vars[i] in [0,1] for all i
    #   sum(h_vars) = n_points / 2
    #   for each k: sum(h[i] * (1 - h[i+k])) <= t
    
    # Generate all correlations explicitly
    # For k = 0: overlap = sum(h[i] * (1-h[i])) = sum(h[i] - h[i]^2)  
    # For k>0: overlap = sum(h[i] * (1-h[i+k])) for valid indices
    
    def construct_overlap_constraint(k):
        if k > 0:
            # sum h[i] * (1 - h[i+k]) for i = 0 to n_points-1-k
            expr = 0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i+k])
        elif k < 0:
            # same with negative index, equivalent sum h[i+|k|] * (1-h[i])
            expr = 0
            for i in range(n_points + k):
                expr += h_vars[i - k] * (1 - h_vars[i])
        else:
            # k == 0: sum h[i] * (1 - h[i]) = sum(h[i] - h[i]^2)
            expr = 0
            for i in range(n_points):
                expr += h_vars[i] * (1 - h_vars[i])
        return expr
    
    # Create constraints for shift over all valid values
    max_shift = n_points - 1
    # Since max overlap is limited in number of terms, let's generate as many as needed
    
    # Sample a subset of shifts for computational tractability, but make it wide enough
    # to cover essential features
    shift_range = range(-max_shift, max_shift + 1) 
    # We only use a few shifts from our key domain to make this numerically efficient
    # But also make sure it's dense in key points using 13 shifts
    shift_sample = list(range(-15, 16))  # covers from -15 to 15, about 30 shifts
    
    for k in shift_sample:
        # Subsample shift index to make problem manageable
        # Using more dense coverage over [-10,10]
        if k >= 0:
            expr = 0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i+k])
        else:
            expr = 0
            for i in range(n_points + k):
                expr += h_vars[i - k] * (1 - h_vars[i])
        constraints.append(expr <= t)
    
    # Define the optimization problem
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    best_h = h.copy()
    best_C5 = float('inf')
    best_t = float('inf')
    
    # Set aside the solver for efficiency
    solver = 'MOSEK'  # or 'SCS', 'ECOS', 'CVXOPT'
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status == cp.OPTIMAL:
            optimized_h = h_vars.value
            # Also, let's manually calculate C5 using the original function.
            # This is much more robust across solvers and doubt not pixellated
        
            # Manually evaluate C5 using correlate
            corr = correlate(optimized_h, 1 - optimized_h, mode="full")
            computed_C5 = np.max(corr) * dx
            
            if computed_C5 < best_C5:
                best_C5 = computed_C5
                best_h = optimized_h.copy()
                best_t = t.value
        else:
            # fallback to original values if infeasible or error
            corr = correlate(h, 1 - h, mode="full")
            computed_C5 = np.max(corr) * dx
            best_C5 = computed_C5
            best_h = h.copy()
            best_t = computed_C5
    
    except Exception as e:
        # In case of any solver failure, just go with the initial
        corr = correlate(h, 1 - h, mode="full")
        computed_C5 = np.max(corr) * dx
        best_C5 = computed_C5
        best_h = h.copy()
        best_t = computed_C5
    
    # If still defying the bounds, do final normalization and projection
    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = np.clip(best_h, 0, 1)
        current_sum = np.sum(best_h)
        if current_sum > 0:
            best_h = best_h * (n_points / 2) / current_sum
    
    # Final compute
    final_corr = correlate(best_h, 1 - best_h, mode="full")
    final_C5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_C5, n_points)
