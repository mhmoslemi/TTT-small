import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    n_dimensions = n_points
    max_iter = 1000
    w = 0.7
    c1 = c2 = 1.5
    max_time = budget_s - 10
    start_time = time.time()
    
    # Initialize particles and velocities
    particles = np.array([initial_h.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_dimensions))
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate(p, dx) for p in particles])
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_scores)
    gbest = pbest[global_best_idx].copy()
    gbest_score = pbest_scores[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            score = evaluate(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    
    return (gbest, gbest_score, n_points)
