import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Exponential smoothing + barrier method approach
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def compute_objective(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Exponential penalty
        penalty = np.sum(np.exp(-beta * corr))
        # Barrier term
        barrier = -gamma * np.sum(np.log(np.clip(h_vec, 1e-8, 1-1e-8)) + 
                                  np.log(np.clip(1 - h_vec, 1e-8, 1-1e-8)))
        return penalty + barrier

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-8, 1-1e-8)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Stochastic gradient descent with exponential smoothing
    beta = 1000.0  # Temperature for exponential penalty
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    patience = 0
    max_patience = 500
    iter_count = 0

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 1e-8, 1-1e-8)
                h = project_feasible(h)
                patience = 0

        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        orig_obj = compute_objective(h, beta, gamma)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            
            obj_plus = compute_objective(h_plus, beta, gamma)
            grad[i] = (obj_plus - orig_obj) / eps
        
        # Update step
        h = h - eta * grad
        h = project_feasible(h)
        
        # Adjust hyperparameters
        if iter_count % 50 == 0:
            beta *= 0.99
            gamma *= 1.10
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.5
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
