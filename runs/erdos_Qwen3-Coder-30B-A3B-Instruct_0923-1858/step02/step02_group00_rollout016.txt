import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization
    def pso_optimize():
        n_particles = 50
        max_iter = 100
        inertia_w = 0.9
        c1 = 2.0
        c2 = 2.0
        
        # Initialize particles
        particles = []
        velocities = []
        pbest_positions = []
        pbest_fitness = []
        
        for _ in range(n_particles):
            # Create random feasible particle
            h_p = np.random.rand(n_points)
            h_p = project_feasible(h_p)
            particles.append(h_p.copy())
            velocities.append(np.random.normal(0, 0.01, n_points))
            pbest_positions.append(h_p.copy())
            pbest_fitness.append(compute_C5(h_p))
        
        # Global best
        gbest_idx = np.argmin(pbest_fitness)
        gbest_position = pbest_positions[gbest_idx].copy()
        gbest_fitness = pbest_fitness[gbest_idx]
        
        for iter_num in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia_w * velocities[i] + 
                                c1 * r1 * (pbest_positions[i] - particles[i]) +
                                c2 * r2 * (gbest_position - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate fitness
                fitness = compute_C5(particles[i])
                
                # Update personal best
                if fitness < pbest_fitness[i]:
                    pbest_fitness[i] = fitness
                    pbest_positions[i] = particles[i].copy()
                    
                    # Update global best
                    if fitness < gbest_fitness:
                        gbest_fitness = fitness
                        gbest_position = particles[i].copy()
            
            # Decrease inertia weight
            inertia_w = 0.9 + (0.4 - 0.9) * (iter_num / max_iter)
            
        return gbest_position, gbest_fitness
    
    # Gradient-based refinement 
    def refine_with_gradient(h_vec):
        # Simple gradient ascent approach: for each update, compute gradient
        eta = 0.01
        max_iter = 50
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            h_vec = np.array(h_vec)
            
            # Compute cross-correlation
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)
            
            # Estimate gradient via finite differences for each component
            grad = np.zeros_like(h_vec)
            eps = 1e-5
            for j in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update step
            h_vec = h_vec - eta * grad
            h_vec = project_feasible(h_vec)
            
            # Adjust eta based on improvement
            if i % 10 == 0:
                eta *= 0.95
                
        return h_vec
    
    # Main optimization loop
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        # PSO global search
        h_new, C5_new = pso_optimize()
        
        # Gradient refinement
        h_new = refine_with_gradient(h_new)
        C5_new = compute_C5(h_new)
        
        # Update best solution
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
            
        iter_count += 1
        
        # Early exit condition if improvement is very small
        if iter_count % 10 == 0 and C5_new < 0.38085:
            break
    
    # Final refinement
    final_h = refine_with_gradient(best_h)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
