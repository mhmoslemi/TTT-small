import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    offspring1 = np.where(mask, parent1, parent2)
    offspring2 = np.where(mask, parent2, parent1)
    return offspring1, offspring2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    selected_indices = np.random.choice(len(population), tournament_size)
    best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_idx]

    dx = 2.0 / n_points

        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = project(initial_h, dx)
    
    # GA Parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    tournament_size = 3

    reset_limit = 100
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):

    # Evaluate initial population
    fitnesses = [evaluate(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    
    best_h = population[best_idx].copy()
    best_C = fitnesses[best_idx]
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom for cleanup
    
    stagnation_count = 0
    reset_count = 0
    
    for generation in range(100000):  # High generation count
        if time.time() - t0 > max_time:

        sorted_indices = np.argsort(fitnesses)

        fitnesses = [fitnesses[i] for i in sorted_indices]

        current_best = fitnesses[0]

            mutation_rate = 0.2

        while len(new_population) < population_size:

                parent = tournament_selection(population, fitnesses, tournament_size)

        # Update population
        population = new_population
        fitnesses = [evaluate(ind, dx) for ind in population]
