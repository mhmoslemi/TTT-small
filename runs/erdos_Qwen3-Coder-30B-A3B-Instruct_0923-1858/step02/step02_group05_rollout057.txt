import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    n = len(h)
    # Compute correlation manually for efficiency
    max_corr = 0.0
    for lag in range(2*n - 1):
        corr = 0.0
        if lag < n:
            # h[0:n-lag] * (1 - h[lag:n])
            for i in range(n - lag):
                corr += h[i] * (1 - h[lag + i])
        else:
            # h[lag-n+1:n] * (1 - h[0:n-(lag-n+1)])
            lag_offset = lag - n + 1
            for i in range(n - lag_offset):
                corr += h[lag_offset + i] * (1 - h[i])
        
        if corr > max_corr:
            max_corr = corr
    
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    np.random.seed(seed)

    dx = 2.0 / n_points

        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(n_particles):
        # Start from initial
        h = initial_h.copy()
        # Add small random perturbation
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        h += perturb
        h = project(h, dx)
        
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        p_best_vals.append(evaluate(h, dx))
    
    # Global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2  # Leave headroom
    
    iteration = 0
    while time.time() - t0 < max_time:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * velocities[i] + 
                   c1 * r1 * (p_best[i] - particles[i]) + 
                   c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            vel = np.clip(vel, -v_max, v_max)
            velocities[i] = vel
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible space
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
        
        # Early termination if improvement has stalled
        if iteration % 20 == 0 and iteration > 100:
            # Simple check for stagnation
            if iteration % 100 == 0:
                improved = False
                for i in range(n_particles):
                    if abs(p_best_vals[i] - g_best_val) < 1e-6:
                        improved = True
                        break
                if not improved:
                    break
    
    # Return final result
    final_C = evaluate(g_best, dx)
    return (g_best, final_C, n_points)
