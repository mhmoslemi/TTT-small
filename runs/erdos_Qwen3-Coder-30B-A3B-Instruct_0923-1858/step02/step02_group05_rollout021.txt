import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    # Initialize particles for PSO
    n_particles = 50
    particles = []
    velocities = []
    
    # Initialize particles using the starting h with small perturbations
    for i in range(n_particles):

    p_best = [p.copy() for p in particles]
    p_best_C = [evaluate(p, dx) for p in particles]
    g_best_idx = np.argmin(p_best_C)
    g_best = p_best[g_best_idx].copy()
    g_best_C = p_best_C[g_best_idx]
    
    # PSO parameters
    c1, c2, w = 2.0, 2.0, 0.7

            for i in range(n_particles):

            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to constraints
            particles[i] = project(particles[i], dx)

            C = evaluate(particles[i], dx)
