import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):

    phi1 = 1.5

            particle_h = h.copy()
        else:  # Others random perturbations

            v = omega * (particle_h - particle_h) + \

            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            particle_h = particle_h + v
            
            # Project to feasible region
            particle_h = project(particle_h, dx)
            
            # Evaluate new C5
            new_C = evaluate(particle_h, dx)
            
            # Update personal best
            if new_C < p_best_C[i]:
                p_best[i] = particle_h.copy()
                p_best_C[i] = new_C
            
            # Update global best
            if new_C < best_C:
                best_h = particle_h.copy()
                best_C = new_C
                
            # Update swarm position
            swarm[i] = particle_h
            
    return (best_h, best_C, n_points)
