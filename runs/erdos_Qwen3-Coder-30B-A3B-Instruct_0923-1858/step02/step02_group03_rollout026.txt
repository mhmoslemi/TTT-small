import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    n_dimensions = n_points
    max_iter = 1000
    
    # PSO hyperparameters
    w = 0.7
    c1 = c2 = 1.5
    inertia = w
    cognitive_weight = c1
    social_weight = c2
    
    # Initialize particles and velocities
    particles = np.array([initial_h + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)])
    particles = np.array([project(p, dx) for p in particles])
    velocities = np.zeros((n_particles, n_dimensions))
    
    # Individual and global bests
    pbest = particles.copy()
    pbest_C = np.array([evaluate(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_C)
    gbest = pbest[gbest_idx].copy()
    gbest_C = pbest_C[gbest_idx]
    
    start_time = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_dimensions), np.random.rand(n_dimensions)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive_weight * r1 * (pbest[i] - particles[i]) + 
                           social_weight * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate current particle
            current_C = evaluate(particles[i], dx)
            
            # Update personal best
            if current_C < pbest_C[i]:
                pbest_C[i] = current_C
                pbest[i] = particles[i].copy()
                
                # Update global best
                if current_C < gbest_C:
                    gbest_C = current_C
                    gbest = particles[i].copy()
        
        # Decay inertia
        inertia = 0.95 ** iteration
    
    return (gbest, gbest_C, n_points)
