import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    T0 = 100.0

    max_iter = 10000

    max_time = budget_s - 10  # Leave some headroom
    
    # Perturb 5% of points at each iteration
    perturb_frac = 0.05

        h_proposal = h.copy()
        n_perturb = max(1, int(perturb_frac * n_points))

        # Project the proposal to feasible region
        h_proposal = project(h_proposal, dx)
        
        # Evaluate
        E_proposal = evaluate(h_proposal, dx)
        E_current = evaluate(h, dx)
        
        # Compute energy difference
        delta_E = E_proposal - E_current
        
        # Accept or reject based on simulated annealing criteria
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
