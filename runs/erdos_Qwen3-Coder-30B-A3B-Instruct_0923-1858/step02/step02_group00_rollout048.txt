import time
from scipy.signal import correlate

    start_time = time.time()

        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Initialize optimization parameters
    beta = 1000.0  # Temperature for exponential smoothing
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    decay_factor = 0.99
    barrier_increase = 1.1
    patience = 0
    max_patience = 50
    
    def compute_gradient(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        max_corr = corr[max_k_idx]
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Add barrier gradient
        barrier_grad = -gamma * (1/h_vec + 1/(1-h_vec))
        return grad + barrier_grad
    
    def update_step(h_vec, beta, gamma, eta):
        grad = compute_gradient(h_vec, beta, gamma)
        h_new = h_vec - eta * grad
        h_new = project_feasible(h_new)
        return h_new
    
    # Main optimization loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Add noise to escape local minima
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = project_feasible(h)
                patience = 0
        
        # Update h with exponential smoothing + barrier
        h = update_step(h, beta, gamma, eta)
        
        # Update hyperparameters
        if iter_count % 50 == 0 and iter_count > 0:
            beta *= decay_factor
            gamma *= barrier_increase
            eta *= 0.5
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
