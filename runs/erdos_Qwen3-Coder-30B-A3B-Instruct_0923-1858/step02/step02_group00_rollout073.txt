import time

        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Exponential smoothing approach with barrier method
    def compute_objective_and_gradient(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(corr)
        if max_corr > 0:
            corr_normalized = corr / max_corr
        else:
            corr_normalized = corr

        # Exponential penalty
        penalty = np.sum(np.exp(-beta * corr_normalized))
        
        # Barrier
        barrier = -gamma * np.sum(np.log(np.maximum(h_vec, 1e-8)) + np.log(np.maximum(1 - h_vec, 1e-8)))
        
        # Combine
        F = penalty + barrier
        
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            if max_corr > 0:
                corr_plus_norm = corr_plus / max_corr
                corr_minus_norm = corr_minus / max_corr
            else:
                corr_plus_norm = corr_plus
                corr_minus_norm = corr_minus
                
            grad[i] = (np.sum(np.exp(-beta * corr_plus_norm)) -

            # Add barrier gradient

                grad[i] += -gamma * (1/h_vec[i] - 1/(1-h_vec[i]))
                
        return F, grad

    gamma = 1e-4
    eta = 1e-5

            patience = 0

            beta *= 0.99
            gamma *= 1.1
            if eta > 1e-6:

        F, grad = compute_objective_and_gradient(h, beta, gamma)

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
