import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50

    velocities = []

        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        initial_h = project(initial_h, dx)

    # Initialize each particle
    for _ in range(n_particles):
        h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))

    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Track personal bests and global best
    p_best = [p.copy() for p in particles]
    p_best_vals = [evaluate(p, dx) for p in particles]
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2  # Leave headroom
    
    for iteration in range(1000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best if needed
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val

    return (g_best, g_best_val, n_points)
