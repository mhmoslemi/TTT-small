import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        segments = 10
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # Parameters for genetic algorithm
    population_size = 50
    generations = 10000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'h' in locals():
            # Perturb initial solution
            mut_h = h + np.random.normal(0, 0.05, n_points)
            mut_h = project(mut_h, dx)
            population.append(mut_h)
        else:
            # Random initialization
            new_h = np.random.rand(n_points)
            new_h = project(new_h, dx)
            population.append(new_h)
    
    best_solution = None
    best_fitness = float('inf')
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # Evaluate initial population
    fitness_scores = []
    for i, individual in enumerate(population):
        fitness = evaluate(individual, dx)
        fitness_scores.append(fitness)
        if fitness < best_fitness:
            best_fitness = fitness
            best_solution = individual.copy()
    
    for gen in range(generations):
        if time.time() - t0 > max_time:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create next generation
        new_population = []
        
        # Elitism: keep best solutions
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
                parent1 = selected[parent1_idx]
                parent2 = selected[parent2_idx]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points-1)

                if np.random.rand() < mutation_rate:

        # Update population
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness_scores = []
        for individual in population:
            fitness = evaluate(individual, dx)
            fitness_scores.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_solution = individual.copy()
    
    return (best_solution, best_fitness, n_points)
