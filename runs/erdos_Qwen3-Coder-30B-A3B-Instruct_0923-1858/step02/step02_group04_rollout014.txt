import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Use a more sophisticated search: simulated annealing with local refinement
    def neighbor(h_vec, temperature):
        # Make a small perturbation
        h_new = h_vec.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, temperature * 0.1)
        h_new[idx] += delta
        return project_feasible(h_new)
    
    # Simulated annealing loop
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    
    # Parameters
    T_init = 0.1
    T_min = 1e-6
    alpha = 0.995
    max_iter = 100000
    iter_count = 0
    
    T = T_init
    while T > T_min and iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Generate neighbor
        new_h = neighbor(current_h, T)
        new_C5 = compute_C5(new_h)
        
        # Accept or reject
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            current_h = new_h
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        T *= alpha
        iter_count += 1
    
    # Local refinement with gradient-based optimization
    def local_optimize(h_vec):
        def obj(h_vec):
            return compute_C5(h_vec)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Use trust-constr with bounds
        try:
            res = minimize(
                obj,
                h_vec,
                method='trust-constr',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'disp': False, 'maxiter': 50}
            )
            if res.success:
                return project_feasible(res.x)
        except:
            pass
        return h_vec
    
    # Run final local refinement
    refined_h = local_optimize(best_h)
    final_C5 = compute_C5(refined_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = refined_h
    
    return (best_h.tolist(), best_C5, len(best_h))
