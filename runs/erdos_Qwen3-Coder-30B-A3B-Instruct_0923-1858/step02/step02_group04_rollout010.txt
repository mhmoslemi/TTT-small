import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Use a different optimization approach
    # Try to use a smoother optimization with regularization
    
    # We will use a randomized search with local refinement
    iter_count = 0
    patience = 0
    max_patience = 50
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        
        # Random perturbation
        h_new = h.copy()
        idx = np.random.choice(n_points, size=5, replace=False)
        for i in idx:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.02), 0, 1)
        h_new = project_feasible(h_new)
        
        current_C5 = compute_C5(h)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < current_C5:
            h = h_new
            patience = 0
        else:
            patience += 1
        
        # Update best solution
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
        
        # If no improvement for a while, add some noise
        if patience > max_patience:
            noise_scale = 0.005 * (1 - elapsed / time_limit)
            h = np.clip(h + np.random.normal(0, noise_scale, n_points), 0, 1)
            h = project_feasible(h)
            patience = 0
        
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
