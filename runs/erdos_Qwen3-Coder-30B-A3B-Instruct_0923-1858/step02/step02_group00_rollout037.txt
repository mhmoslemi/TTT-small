import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available, otherwise create a pyramid
    try:

        n_points = len(h)
    except NameError:

        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Optimizer using CVXPY with warm start
    def optimize_with_cvxpy(h_init, max_iter=50):
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Soft constraint on maximizing overlap
        obj = cp.Minimize(cp.sum_squares(h_var - h_init))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_result = h_var.value
            else:
                h_result = h_init
        except:
            h_result = h_init
            
        return project_feasible(h_result)

    # Find the maximum correlation
    def find_max_correlation(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.argmax(corr), np.max(corr)

    # Gradient estimation using finite differences
    def estimate_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        max_idx, _ = find_max_correlation(h_vec)
        max_k = max_idx - (n_points - 1)
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * eps)
            
        return grad

    # Main optimization loop with trust region update and gradient-based steps
    iter_count = 0
    eta = 0.01
    patience = 0
    max_patience = 50
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=max(1, int(0.05 * n_points)), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Gradient step
        grad = estimate_gradient(h)
        h_new = h - eta * grad
        h_new = project_feasible(h_new)
        
        # CVXPY projection for better feasibility and improvement
        h = optimize_with_cvxpy(h_new)
        
        # Adaptive learning rate
        if iter_count % 10 == 0:
            eta *= 0.96
        
        iter_count += 1
        if iter_count % 100 == 0 and elapsed() > budget_s * 0.90:
            break

    return (best_h.tolist(), best_C5, len(best_h))
