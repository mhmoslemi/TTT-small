import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_qp_update(h_vec):
        # Quadratically regularized projection onto feasible set
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        objective = cp.Minimize(cp.sum_squares(h_var - h_vec))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.array(h_var.value).reshape(-1)
        except:
            return project_feasible(h_vec)

    def identify_worst_shifts(h_vec, num_shifts=10):
        # Identify shifts that maximize overlaps
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Shifts from -n_points to n_points
        shift_indices = np.arange(len(corr))
        max_shifts = np.argsort(corr)[-num_shifts:][::-1]
        return [s - (len(h_vec) - 1) for s in max_shifts]
        
    def solve_subproblem(h_vec, k):
        # Solve one subproblem for shift k
        # Minimize overlap for shift k
        h_vec = np.array(h_vec)
        h_new = h_vec.copy()
        # Add penalty for k-shift overlap
        # Use gradient descent on shift-k overlap only
        def objective(h_new):
            # This is the overlap for shift k
            corr_k = 0.0
            for i in range(len(h_new)):
                if 0 <= i + k < len(h_new):
                    corr_k += h_new[i] * (1 - h_new[i + k])
            return -corr_k  # Minimize negative overlap to maximize overlap 
        
        # We can also do a direct projection or simple updates
        # For now, simple gradient step (with k-dependent adjustment)
        step_size = 0.01
        # Create perturbation that decreases overlap at shift k
        pert = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if 0 <= i + k < len(h_vec):
                # Grad w.r.t. h[i] of (h[i] * (1-h[i+k])) is (1 - h[i+k] - h[i])
                pert[i] = -1.0 * (1 - h_vec[i + k] - h_vec[i])
        # Project to feasible set
        h_new = h_vec + step_size * pert
        h_new = project_feasible(h_new)
        return h_new

    # Column generation with update/iterations
    max_iters = 1000
    patience = 0
    max_patience = 50
    iter_count = 0
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iters:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation for escape from local minimum
                idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
                
        # Identify worst shifts every 10 iterations
        if iter_count % 10 == 0:
            worst_shifts = identify_worst_shifts(h, num_shifts=5)
        else:
            worst_shifts = worst_shifts[:3]  # Keep top 3 recent shifts for updates
            
        # Solve subproblems for top shifts
        for k in worst_shifts:
            if abs(k) <= n_points // 2:
                h = solve_subproblem(h, k)
        
        # Final QP projection to maintain feasibility (increase quality)
        h = solve_qp_update(h)
        
        # Occasionally increase solution quality with heavy QP
        if iter_count % 20 == 0:
            h = solve_qp_update(h)
        
        iter_count += 1

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    # Debug validation checks
    assert np.all(best_h <= 1) and np.all(best_h >= 0), "h not in [0,1]"
    assert abs(np.sum(best_h) - n_points / 2) < 1e-6, "Sum not correct"
    
    return (best_h.tolist(), best_C5, len(best_h))
