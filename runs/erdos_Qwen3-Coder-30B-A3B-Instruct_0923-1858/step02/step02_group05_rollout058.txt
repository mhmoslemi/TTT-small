import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    dx = 2.0 / n_points

        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        # Create slight variations from initial point
        h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Track best solutions
    p_best = [p.copy() for p in particles]
    p_best_c = [evaluate(p, dx) for p in particles]
    g_best_idx = np.argmin(p_best_c)
    g_best = p_best[g_best_idx].copy()
    g_best_c = p_best_c[g_best_idx]
    
    # PSO iterations
    t0 = time.time()
    max_time = budget_s - 2
    
    iteration = 0
    while time.time() - t0 < max_time:
        if iteration % 100 == 0 and iteration > 0:

            pass
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to valid constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_c = evaluate(particles[i], dx)
            
            # Update particle best
            if current_c < p_best_c[i]:
                p_best[i] = particles[i].copy()
                p_best_c[i] = current_c
                
                # Update global best
                if current_c < g_best_c:
                    g_best = particles[i].copy()
                    g_best_c = current_c
                    
        iteration += 1
    
    # Final evaluation
    final_c = evaluate(g_best, dx)
    
    return (g_best, final_c, n_points)
