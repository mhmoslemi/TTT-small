import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing with local search
    T = 1.0
    cooling_rate = 0.99
    mutation_step = 0.1
    max_iter = 100000
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Perturb the solution
        h_new = h.copy()
        n_mutate = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, size=n_mutate, replace=False)
        for i in indices:
            perturbation = np.random.normal(0, mutation_step)
            h_new[i] += perturbation
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Evaluate objective
        C5_new = compute_C5(h_new)
        C5_current = compute_C5(h)
        
        # Accept or reject
        delta = C5_new - C5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        
        # Update temperature and mutation step
        if iter_count % 100 == 0:
            T *= cooling_rate
            mutation_step = max(0.001, mutation_step * cooling_rate)
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
