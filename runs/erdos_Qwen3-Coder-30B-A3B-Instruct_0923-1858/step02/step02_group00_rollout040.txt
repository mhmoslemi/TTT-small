import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a piecewise linear initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_gradient_analytical(h_vec, k_opt):
        # For the optimal shift k_opt, compute gradient w.r.t. h[i]
        # This is an approximation: for shift k, get contribution to corr[i] to (1-h[i+k])
        # We use finite differences for robustness
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        h_vec = np.array(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[k_opt + len(h_vec) - 1] - corr_minus[k_opt + len(h_vec) - 1]) / (2 * eps)
            
        return grad

    def trust_region_update(h_vec, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Find current max correlation index
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_new) - 1)
            
            # Compute approximate gradient
            grad = compute_gradient_analytical(h_new, max_k)
            
            # Gradient descent step with line search component
            step_size = 0.01
            h_candidate = h_new - step_size * grad
            h_candidate = project_feasible(h_candidate)
            
            # Accept if better
            if compute_C5(h_candidate) < compute_C5(h_new):
                h_new = h_candidate
            else:
                step_size *= 0.5
                if step_size < 1e-6:
                    break
        return h_new

    # PSO parameters
    n_particles = 50
    max_pso_iter = 50
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # PSO initialization
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    for _ in range(n_particles):
        h_init = h.copy()
        # Small random perturbation
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in perturb_indices:
            h_init[i] = np.clip(h_init[i] + np.random.normal(0, 0.05), 0, 1)
        h_init = project_feasible(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(compute_C5(h_init))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    best_C5 = gbest_val
    best_h = gbest.copy()

    # PSO iterations
    for iter in range(max_pso_iter):
        if elapsed() >= budget_s * 0.8:
            break
            
        w = w_start - (w_start - w_end) * iter / max_pso_iter

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate particle
            current_val = compute_C5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
            
            # Update global best
            if current_val < gbest_val:
                gbest = particles[i].copy()
                gbest_val = current_val
                if current_val < best_C5:
                    best_C5 = current_val
                    best_h = gbest.copy()

    # Gradient-based refinement of the best solution
    refine_h = gbest.copy()
    refine_iterations = 50
    for i in range(refine_iterations):
        if elapsed() >= budget_s * 0.95:
            break
        refine_h = trust_region_update(refine_h, max_iter=10)
        refined_val = compute_C5(refine_h)
        if refined_val < best_C5:
            best_C5 = refined_val
            best_h = refine_h.copy()

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
