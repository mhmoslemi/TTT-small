import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

            particles.append(h)

    c1, c2, w = 2.0, 2.0, 0.7

    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Evaluate initial particles
    costs = [evaluate(p, dx) for p in particles]
    p_best = particles.copy()
    p_best_costs = costs.copy()
    g_best_idx = np.argmin(p_best_costs)
    g_best = p_best[g_best_idx].copy()
    g_best_cost = p_best_costs[g_best_idx]
    
    # PSO exploration phase
    t0 = time.time()
    max_time = budget_s * 0.7
    
    iteration = 0
    while time.time() - t0 < max_time:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] +

            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate and update personal best
            cost = evaluate(particles[i], dx)
            if cost < p_best_costs[i]:
                p_best[i] = particles[i].copy()
                p_best_costs[i] = cost
            
            # Update global best
            if cost < g_best_cost:
                g_best = particles[i].copy()
                g_best_cost = cost
            
            iteration += 1
            if time.time() - t0 > max_time:
                break

    # Gradient descent exploitation phase
    max_time = budget_s * 0.3

        # Update step
        current_h = current_h - lr * grad
        current_h = project(current_h, dx)
        
        # Check for improvement
        current_C = evaluate(current_h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
