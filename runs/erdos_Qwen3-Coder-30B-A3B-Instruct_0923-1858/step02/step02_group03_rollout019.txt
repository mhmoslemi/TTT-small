import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_points = 45
    dx = 2.0 / n_points

        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    start_time = time.time()
    max_time = budget_s - 5
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Cutting-plane method approach
    # Variables
    h_vars = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints from initial point
    constraints = []
    for k in range(n_points):
        # Constraint for shift k: int h(x)(1 - h(x+k)) dx <= t
        # This translates to: 1 - int h(x) h(x+k) dx <= t
        # or: int h(x) h(x+k) dx >= 1 - t
        # We'll add all constraints that are currently active and tighten the problem
        h_shifted = np.roll(h, k)
        corr_val = np.sum(h * h_shifted) 
        # Save this in a list
        constraints.append(cp.sum(h_vars * np.roll(h_vars, k)) >= 1 - t)
    
    # Set up objective and problem  
    constraints.append(cp.sum(h_vars) * dx == 1)
    for i in range(n_points):
        constraints.append(h_vars[i] >= 0)
        constraints.append(h_vars[i] <= 1)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Iteratively improve with cutting planes
    for i in range(1000):

        h_vars.value = h
        
        # Solve
        try:

        # Get new h from solution
        h_new = np.array(h_vars.value).reshape(-1)
        
        # Handle numerical issues
        h_new = project(h_new, dx)
        
        C_new = evaluate(h_new, dx)
        if C_new < best_C:
            # Accept new solution
            h = h_new.copy()
            best_C = C_new
            best_h = h.copy()
            
            # In case of success, reset problem with more constraints
            constraints = []

            if C_pert < best_C:

                best_C = C_pert
                best_h = h.copy()
    
    return (best_h, best_C, n_points)
