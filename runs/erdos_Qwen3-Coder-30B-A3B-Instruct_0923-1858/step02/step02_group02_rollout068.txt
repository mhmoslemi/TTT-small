import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

            if i + 4 < n_points:
                h[i:i+5] = np.linspace(0.2, 0.8, 5)
    
    # Initialize
    h_current = project(h, dx)
    E_current = evaluate(h_current, dx)
    
    best_h = h_current.copy()
    best_C = E_current
    
    # Simulated Annealing parameters
    T0 = 100.0

    max_iter = 10000

    t0 = time.time()
    max_time = budget_s - 10
    
    T = T0
    for i in range(max_iter):
        if time.time() - t0 > max_time:

        # Generate a proposal
        h_proposal = h_current.copy()
        # Perturb 5% of elements
        n_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            h_proposal[idx] += np.random.normal(0, 0.05)
        
        # Project to feasible region
        h_proposal = project(h_proposal, dx)
        
        # Evaluate proposal
        E_proposal = evaluate(h_proposal, dx)
        
        # Accept or reject
        delta_E = E_proposal - E_current

        # Cool down
