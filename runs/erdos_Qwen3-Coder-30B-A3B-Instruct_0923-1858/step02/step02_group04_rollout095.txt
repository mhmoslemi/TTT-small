        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Add perturbations to find better solutions
    def perturb(h_vec, rate=0.05):
        h_new = h_vec.copy()
        indices = np.random.choice(n_points, int(n_points * rate), replace=False)
        for i in indices:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.03), 0, 1)
        return project_feasible(h_new)
    
    # Gradient-based refinement
    def compute_gradient(h_vec, eps=1e-6):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)

            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad

            h_new = project_feasible(h_new)
            new_val = compute_C5(h_new)
            if new_val <= current_val - c * alpha * np.dot(grad, grad):

            alpha *= 0.5
        return h_vec, alpha_max

        best_h = best_h.copy()
