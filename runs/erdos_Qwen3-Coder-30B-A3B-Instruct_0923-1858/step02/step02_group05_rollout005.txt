import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    n_lags = 2 * n_points - 1
    
    # Initialize particle swarm
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.2
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        if len(h_init) != n_points:
            h_init = np.ones(n_points) / 2
    else:
        t = np.linspace(0, 2, n_points)
        h_init = 0.5 + 0.3 * np.sin(np.pi * t)
        h_init = project(h_init, dx)
    
    particles = []
    for _ in range(n_particles):
        h = h_init + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track personal and global best
    p_best = [particles[i].copy() for i in range(n_particles)]
    p_best_scores = [evaluate(particles[i], dx) for i in range(n_particles)]
    
    g_best = p_best[np.argmin(p_best_scores)].copy()
    g_best_score = np.min(p_best_scores)
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # PSO iterations
    for iter_num in range(10000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vmax, vmax)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_score = evaluate(particles[i], dx)
            
            # Update personal best
            if current_score < p_best_scores[i]:
                p_best_scores[i] = current_score
                p_best[i] = particles[i].copy()
                
                # Update global best
                if current_score < g_best_score:
                    g_best_score = current_score
                    g_best = particles[i].copy()
    
    return (g_best, g_best_score, n_points)
