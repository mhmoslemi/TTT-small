import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a random feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    offspring1 = np.where(mask, parent1, parent2)
    offspring2 = np.where(mask, parent2, parent1)
    return offspring1, offspring2

def mutate(h, dx, mutation_rate=0.1):
    """Apply mutation to an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def tournament_selection(population, fitness, tournament_size=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

    max_generations = 10000

    # Evaluate initial population
    fitness = [evaluate(ind, dx) for ind in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C = fitness[best_idx]
    
    t0 = time.time()
    generation = 0
    stagnation_count = 0
    last_improvement = 0
    
    while generation < max_generations and (time.time() - t0) < budget_s - 10:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        else:
            best_C = fitness[0]
            best_h = population[0].copy()

                offspring1 = mutate(offspring1, dx, mutation_rate)
