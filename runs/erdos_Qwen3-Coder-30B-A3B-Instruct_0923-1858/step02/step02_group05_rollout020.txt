import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_points = 45

        initial_h = project(initial_h, dx)
    
    # Particle Swarm Optimization parameters
    n_particles = 50
    c1, c2 = 2.0, 2.0
    w = 0.7
    max_vel = 0.2
    max_ps_iterations = 20000
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(n_particles):
        if 'initial_h' in locals():
            # Initialize with small perturbations
            pert = np.random.uniform(-0.1, 0.1, n_points)
            h = initial_h + pert
        else:
            h = np.random.uniform(0, 1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        p_best_vals.append(evaluate(h, dx))
    
    # Global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    # PSO Phase
    t0 = time.time()
    max_time_ps = budget_s * 0.7
    
    iteration = 0
    for iteration in range(max_ps_iterations):

            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)

            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and rescale
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
    
    # Gradient Descent Exploitation Phase

            best_h = current_h.copy()
