import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO Parameters
    n_particles = 50
    max_iter_pso = 100
    inertia_w = 0.9
    inertia_decay = 0.99
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.05
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        h_init = h.copy()
        # Add some noise
        noise = np.random.normal(0, 0.05, n_points)
        h_init += noise
        h_init = project_feasible(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(compute_C5(h_init))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iteration
    for iter_num in range(max_iter_pso):
        if elapsed() > budget_s * 0.95:
            break
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_val = compute_C5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
        
        # Mutation
        if iter_num % 10 == 0:
            for i in range(n_particles):
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.02, n_points)
                    particles[i] += noise
                    particles[i] = project_feasible(particles[i])
        
        # Decay inertia
        inertia_w *= inertia_decay
    
    # Now gradient descent on best result from PSO
    h = gbest.copy()
    learning_rate = 0.01
    max_iter = 1000
    patience = 0
    max_patience = 50
    
    for i in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Reset to best if no improvement
                h = best_h.copy()
                patience = 0
                
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        base_corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(base_corr)
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[j] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * eps)
            
        # Gradient descent step
        h = h - learning_rate * grad
        h = project_feasible(h)
        
        # Adaptive learning rate
        if i % 10 == 0 and i > 0:
            learning_rate *= 0.95
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
