import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 50
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    def adaptive_line_search(h_vec, grad, eta=0.01):
        # Perform backtracking line search
        alpha = 1.0
        beta = 0.5
        c = 1e-4
        max_iter = 10
        f0 = compute_C5(h_vec)
        for _ in range(max_iter):
            h_new = project_feasible(h_vec - alpha * grad)
            f_new = compute_C5(h_new)
            if f_new <= f0 - c * alpha * np.dot(grad, grad):
                return h_new, alpha
            alpha *= beta
        return h_vec, alpha
    
    # Multi-resolution strategy: coarse to fine
    resolutions = [50, 75, 100, 150, 200]
    current_res = 0

    population_scores = []

        population_scores.append(compute_C5(individual))

    max_patience = 150
    eta = 0.01
    
    iter_count = 0

            break

        current_best_idx = np.argmin(population_scores)
        current_best = population[current_best_idx]

            best_C5 = current_C5

                best_C5 = compute_C5(best_h)
                patience = 0

            if np.random.random() < 0.5 and i == 0:

                    new_h, step = adaptive_line_search(population[i], grad, eta)
                    if step > 1e-6:
                        population[i] = new_h
                except:
                    pass
                    
            # Evaluate
            population_scores[i] = compute_C5(population[i])
            
        # Refinement at higher resolution if needed
        if current_res < len(resolutions) - 1:
            # Only increase resolution if we have good progress
            if iter_count % 50 == 0 and population_scores[current_best_idx] < 0.4:
                n_points = resolutions[current_res+1]
                dx = 2.0 / n_points
                
                interpolated_h = np.interp(
                    np.linspace(0, 2, n_points),
                    np.linspace(0, 2, len(current_best)),
                    current_best
                )
                h = project_feasible(interpolated_h)
                
                # Update population with new resolution
                for i in range(pop_size):
                    if i == 0:
                        population[i] = h.copy()
                    else:
                        # Perturb
                        population[i] = h + np.random.normal(0, 0.05, n_points)
                        population[i] = project_feasible(population[i])
                        
                population_scores = [compute_C5(p) for p in population]
                current_res += 1
                
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
