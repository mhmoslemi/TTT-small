import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def perturb(h, dx, magnitude=0.01):
    """Generate neighbor solution by perturbing h."""
    perturbation = np.random.uniform(-magnitude, magnitude, size=h.shape)
    h_candidate = h + perturbation
    return project(h_candidate, dx)

def simulated_annealing(h, dx, budget_s):
    """Run simulated annealing optimization."""
    T_initial = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    T = T_initial
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    start_time = time.time()
    
    while T > T_min and time.time() - start_time < budget_s - 10:
        # Generate neighbor solution
        h_candidate = perturb(h, dx)
        current_C = evaluate(h_candidate, dx)
        
        # Compute acceptance probability
        delta = current_C - best_C
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_candidate.copy()
            best_C = current_C
            best_h = h_candidate.copy()
            
        # Adaptive cooling
        if current_C < best_C:
            T = T * cooling_rate
        else:
            T = T * 1.01
            cooling_rate += 0.001
            cooling_rate = min(cooling_rate, 0.999)
    
    return best_h, best_C
