import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing with local search
    T = 1.0
    cooling_rate = 0.99
    mutation_step = 0.1
    max_iter = 100000
    iter_count = 0
    patience = 0
    max_patience = 50
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit or iter_count >= max_iter:
            break

        # Create new candidate by perturbing some points
        h_new = h.copy()
        num_perturb = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            delta = np.random.normal(0, mutation_step)
            h_new[idx] += delta
        h_new = project_feasible(h_new)
        
        # Compute objective values
        current_C5 = compute_C5(h)
        new_C5 = compute_C5(h_new)
        
        # Accept or reject the new solution
        delta = new_C5 - current_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
        
        # Update temperature and mutation step
        if iter_count % 100 == 0:
            T *= cooling_rate
            mutation_step = max(0.001, mutation_step * 0.995)
        
        # Reset if too many iterations without improvement
        if patience > max_patience:
            # Perturb more aggressively
            idx = np.random.choice(n_points)
            h[idx] = np.clip(h[idx] + np.random.normal(0, 0.1), 0, 1)
            h = project_feasible(h)
            patience = 0
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
