import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 200
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.95
    
    # Initialize particles
    particles = []
    particle_best = []
    particle_best_C5 = []
    for _ in range(n_particles):
        h_p = np.random.rand(n_points)
        h_p = project_feasible(h_p)
        particles.append(h_p)
        particle_best.append(h_p.copy())
        particle_best_C5.append(compute_C5(h_p))
    
    # Global best
    global_best_idx = np.argmin(particle_best_C5)
    global_best = particle_best[global_best_idx].copy()
    global_best_C5 = particle_best_C5[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.9:
            break
            
        # Update each particle
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            # Update velocity
            v = (w * (particles[i] - particle_best[i]) + 
                 c1 * r1 * (particle_best[i] - particles[i]) + 
                 c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += v
            
            # Apply constraints
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_C5 = compute_C5(particles[i])
            
            # Update personal best
            if current_C5 < particle_best_C5[i]:
                particle_best[i] = particles[i].copy()
                particle_best_C5[i] = current_C5
                
                # Update global best
                if current_C5 < global_best_C5:
                    global_best = particles[i].copy()
                    global_best_C5 = current_C5

        # Update inertia
        w *= inertia_decay

    # Gradient refinement
    h = global_best.copy()
    eta = 0.01
    max_iter_refine = 300
    patience = 0
    max_patience = 10
    
    for i in range(max_iter_refine):
        if elapsed() > budget_s * 0.99:
            break
            
        # Estimate gradient
        h_vec = np.array(h)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        
        for j in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        # Update
        h -= eta * grad
        h = project_feasible(h)
        
        # Evaluate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb
                noise = np.random.normal(0, 0.01, n_points)
                h += noise
                h = project_feasible(h)
                patience = 0

        if i % 10 == 0:
            eta *= 0.95

    # Final computation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
