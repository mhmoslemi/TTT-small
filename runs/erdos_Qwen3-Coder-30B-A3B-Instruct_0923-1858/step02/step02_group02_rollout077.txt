import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available

        if len(h) != n_points:

    except NameError:

        h = np.zeros(n_points)
        segments = 3
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # PSO parameters
    n_particles = 50
    max_velocity = 0.2
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Initialize with some particles from the initial guess
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            particle = np.random.rand(n_points) * 0.5 + 0.25  # Random except -10%
        
        particle = project(particle, dx)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_vals.append(evaluate(particle, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 10
    
    # PSO loop
    for iter in range(100000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (p_best[i] - particles[i]) + 
                            social * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate current particle
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best if needed
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
        
        # Occasionally do random perturbation
        if iter % 100 == 0:
            # Randomly perturb some particles
            for i in range(n_particles):
                if np.random.rand() < 0.1:
                    particles[i] = project(particles[i] + np.random.normal(0, 0.01, n_points), dx)

    # Return the best solution found
    return (g_best, g_best_val, n_points)
