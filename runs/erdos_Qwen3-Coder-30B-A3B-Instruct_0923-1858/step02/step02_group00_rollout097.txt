        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1-1e-6)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Exponential smoothing with barrier method
    beta = 1000.0  # Temperature parameter
    gamma = 1e-4   # Barrier parameter
    eta = 1e-5     # Learning rate
    
    patience = 0
    max_patience = 50
    
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 1e-6, 1-1e-6)
                h = project_feasible(h)
                patience = 0

        # Compute full correlation
        corr = correlate(h, 1 - h, mode="full")

        if max_corr <= 1e-10:

        # Exponential smoothing of the objective
        exp_corr = np.exp(-beta * corr / max_corr)
        penalty = np.sum(exp_corr)

        barrier = -gamma * np.sum(np.log(h) + np.log(1 - h))

        grad = np.zeros_like(h)

            h_plus[i] += eps
            h_minus[i] -= eps

            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")

            exp_plus = np.exp(-beta * corr_plus / max_corr)
            exp_minus = np.exp(-beta * corr_minus / max_corr)

            # Combine gradients
            grad[i] = (np.sum(exp_plus - exp_minus) / (2 * eps * max_corr) * (-beta) +

        # Update

        h = project_feasible(h)

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
