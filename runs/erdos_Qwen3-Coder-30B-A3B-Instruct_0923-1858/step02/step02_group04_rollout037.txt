import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create triangular initial solution
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Use a simpler gradient-free approach with random search
    # to try to escape local optima
    patience = 0
    max_patience = 50
    iter_count = 0
    eta = 0.01
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Random perturbations
        if patience > max_patience:
            # Mix of small random perturbations and some large ones
            idx = np.random.choice(n_points, size=min(5, n_points//5), replace=False)
            h[idx] += np.random.normal(0, 0.02, len(idx))
            
            # Add small random noise to some points
            idx2 = np.random.choice(n_points, size=n_points//10, replace=False)
            h[idx2] += np.random.normal(0, 0.01, len(idx2))
            
            h = project_feasible(h)
            patience = 0
        else:
            # Local update with small random changes
            idx = np.random.choice(n_points, size=3, replace=False)
            h[idx] += np.random.normal(0, 0.005, len(idx))
            h = project_feasible(h)
            
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
