import time
from scipy.signal import correlate
import cvxpy as cp

        # Define the function to evaluate overlap
        def compute_C(h_vals):
            corr = correlate(h_vals, 1 - h_vals, mode='full')
            return np.max(corr) * dx

        h_vals = initial_h.copy()
        C = compute_C(h_vals)
        best_C = min(best_C, C)
        best_h = h_vals.copy()
        
        # Simple local search using gradient descent on a subset of points
        def project(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_sum = np.sum(h_vec)
            if h_sum > 0:
                h_vec = h_vec * (1.0 / (h_sum * dx))
            return h_vec

        max_iter = 50000

                h_plus[j] += eps

                h_minus[j] -= eps

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h
