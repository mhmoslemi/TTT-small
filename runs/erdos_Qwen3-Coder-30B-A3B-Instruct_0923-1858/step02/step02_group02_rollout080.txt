import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    dx = 2.0 / n_points
    
    # Parameters

    generations = 10000

        else:

        population.append(h)

    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom

    # Track history for adaptive mutation rate
    last_improvement = 0
    adaptive_mutation_rate = mutation_rate

    for gen in range(generations):
        if time.time() - t0 > max_time:
            break

        # Evaluate population
        fitness_scores = []
        for h in population:
            C = evaluate(h, dx)
            fitness_scores.append(1.0 / (C + 1e-10))  # higher fitness = better solution

        # Selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            winner = tournament[np.argmax([fitness_scores[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            if np.random.rand() < crossover_rate:
                # Simple uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < adaptive_mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < adaptive_mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
                    
            child1 = project(child1, dx)
            child2 = project(child2, dx)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Update best solution
        for h in population:
            current_C = evaluate(h, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h.copy()
                last_improvement = gen
                adaptive_mutation_rate = mutation_rate  # Reset mutation rate
            elif gen - last_improvement > 100:
                # Increase mutation rate if no improvement in 100 generations
                adaptive_mutation_rate = min(0.3, adaptive_mutation_rate + 0.005)
            elif gen - last_improvement > 500:
                # Decrease mutation rate if no improvement in 500 generations
                adaptive_mutation_rate = max(0.01, adaptive_mutation_rate - 0.005)
    
    return (best_h, best_C, n_points)
