import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        initial_h[15:] = np.linspace(1, 0, 30)

        initial_h = np.clip(initial_h, 0, 1)

        return evaluate(h, dx)

    def constraint_sum(h):
        return np.sum(h) * dx - 1.0

    bounds = [(0, 1) for _ in range(n_points)]

    t0 = time.time()
