import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

    except NameError:
        # Create a triangular initial solution
        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, k=None):
        h_vec = np.array(h_vec)
        if k is None:
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            k = max_k_idx - (len(h_vec) - 1)
        
        # Compute gradient efficiently by finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[len(h_vec) - 1 + k] - corr_minus[len(h_vec) - 1 + k]) / (2 * eps)
            
        return grad

    def trust_region_update(h_vec, eta=0.01, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Get gradient with respect to the current max overlap
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Randomized local search with adaptive step size
    def randomized_local_search():
        current_h = h.copy()
        current_C5 = compute_C5(current_h)
        best_C5_local = current_C5
        best_h_local = current_h.copy()
        eta = 0.01
        iter_count = 0
        
        while elapsed() < budget_s and iter_count < 2000:
            # Occasionally sample the gradient with respect to a random shift
            if np.random.rand() < 0.7:
                # Fix max shift computed on this iteration
                corr = correlate(current_h, 1 - current_h, mode="full")
                max_k_idx = np.argmax(corr)
                k = max_k_idx - (len(current_h) - 1)
                grad = get_gradient(current_h, k)
            else:
                grad = get_gradient(current_h)
                
            # Update step
            current_h = current_h - eta * grad
            current_h = project_feasible(current_h)
            current_C5 = compute_C5(current_h)
            
            if current_C5 < best_C5_local:
                best_C5_local = current_C5
                best_h_local = current_h.copy()
            
            # Adjust step size
            if iter_count % 50 == 0 and iter_count > 0:
                eta *= 0.95
                
            iter_count += 1
            
        return best_h_local, best_C5_local

        # Maximize max_k <h, 1 - h_k> 
        # we try a simple linearized version of the shifted construction
        # First get the current best case
        try:

            max_k_idx = np.argmax(corr)
            k = max_k_idx - (len(h) - 1)
        except:
            k = 0
            
        # For now returning original.  This is just for getting a more feasible solution
        return h.copy()
    
    # Main optimization loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        # Try local search
        local_h, local_C5 = randomized_local_search()
        
        if local_C5 < best_C5:
            best_C5 = local_C5
            best_h = local_h.copy()
        
        # Occasionally perturb the current solution to escape local minima
        if np.random.rand() < 0.1:
            idxs = np.random.choice(n_points, size=max(1, int(n_points * 0.05)), replace=False)
            for i in idxs:
                local_h[i] = np.clip(local_h[i] + np.random.normal(0, 0.02), 0, 1)
            local_h = project_feasible(local_h)
            current_C5 = compute_C5(local_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = local_h.copy()
                
        iter_count += 1
        
        # Occasionally reinitialize from random
        if iter_count % 20 == 0 and np.random.rand() < 0.3:
            h = np.random.rand(n_points)
            h = project_feasible(h)
            
        if iter_count % 10 == 0:
            # Update h to be the best h from this iteration
            h = best_h.copy()
            # Adaptive step size
            if best_C5 < 0.381:

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
