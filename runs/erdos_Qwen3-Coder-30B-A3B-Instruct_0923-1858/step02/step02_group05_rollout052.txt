import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        h = project(h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Slightly perturb the initial solution
        pert = np.random.uniform(-0.1, 0.1, n_points)
        h_part = h + pert
        h_part = project(h_part, dx)
        particles.append(h_part.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best.append(h_part.copy())
        p_best_vals.append(evaluate(h_part, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate new position
            new_val = evaluate(particles[i], dx)
            
            # Update personal best
            if new_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = new_val
                
                # Update global best
                if new_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = new_val
    
    # Final evaluation
    final_C = evaluate(g_best, dx)
    
    return (g_best, final_C, n_points)
