import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, sample_shifts=None):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Sample a subset of shifts
        if sample_shifts is None:
            # Use a smaller sample to reduce computation
            sample_shifts = np.random.choice(len(h_vec), min(15, len(h_vec)), replace=False)
        
        # Compute a direction with partial gradients
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in sample_shifts:
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 30
    perturb_prob = 0.1
    iter_count = 0
    max_iter = int(budget_s * 0.8 / 0.01)  # Estimate iterations

    # Initialize with a better starting guess
    h = project_feasible(h)
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=int(perturb_prob * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Gradient estimation
        grad = get_gradient(h)
        
        # SGD step
        h = h - eta * grad
        h = project_feasible(h)
        
        # Adaptive learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.95
            # Occasionally resample shifts for diversity
            if np.random.random() < 0.5:
                sample_shifts = None
            else:
                sample_shifts = np.random.choice(len(h), min(15, len(h)), replace=False)
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
