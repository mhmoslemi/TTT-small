import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with provided construction or create a reasonable starting point
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        # Create a piecewise linear function that is symmetric and well-behaved
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        # Use a combination of linear segments
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i] * 2
            else:
                h[i] = (2 - x[i]) * 2
        h = h * (n_points / 2) / np.sum(h)  # Normalize

    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation approach to initialize
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        # Solve a slight relaxation of the problem
        objective = cp.Minimize(cp.sum(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                result = h_var.value
            else:
                result = np.ones(n_points) * 0.5
        except:
            result = np.ones(n_points) * 0.5
        return np.clip(result, 0, 1)

    # Get a better initial solution
    h = solve_convex_relaxation()
    h = project_feasible(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Optimization using L-BFGS-B
    def objective(h_vec):
        h_vec = np.array(h_vec)
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec)
        
    def grad(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        eps = 1e-5
        g = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return g
    
    # Alternating approach: optimize h while tracking worst shift
    patience = 0
    max_patience = 50
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        # Optimize h using minimization
        try:
            res = minimize(objective, h, method='L-BFGS-B', jac=grad, options={'maxiter': 100})
            if res.success:
                h = res.x
                h = project_feasible(h)
            else:
                # fallback random perturbation
                h = project_feasible(h + np.random.normal(0, 0.01, n_points))
        except:
            h = project_feasible(h + np.random.normal(0, 0.01, n_points))
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb more significantly
                eps = 0.1
                noise = np.random.normal(0, eps, n_points)
                h = project_feasible(h + noise)
                patience = 0
                
        iter_count += 1
        
        # Occasionally re-evaluate the best
        if iter_count % 50 == 0:
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
