import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        # Create a triangular initial solution
        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Enhanced gradient estimation
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

    # Trust region with adaptive step size
    def trust_region_update(h_vec, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            # Adaptive step size
            step_size = 1e-3 / (1 + i * 0.01)
            h_new = h_new - step_size * grad
            h_new = project_feasible(h_new)
        return h_new

    # Particle Swarm Optimization approach
    def pso_search():
        swarm_size = 50
        max_iter = 100
        w = 0.9
        c1 = 2.0
        c2 = 2.0
        
        # Initialize swarm
        swarm = []
        best_positions = []
        best_values = []
        
        for _ in range(swarm_size):
            h_sample = h.copy()
            # Add small random noise
            h_sample += np.random.normal(0, 0.01, n_points)
            h_sample = project_feasible(h_sample)
            swarm.append(h_sample)
            best_positions.append(h_sample.copy())
            best_values.append(compute_C5(h_sample))
            
        global_best_idx = np.argmin(best_values)
        global_best = best_positions[global_best_idx].copy()
        global_best_value = best_values[global_best_idx]
        
        for iter_count in range(max_iter):
            if elapsed() > budget_s * 0.7:
                break
                
            for i in range(swarm_size):
                r1, r2 = np.random.random(2)
                # Update velocity and position
                # Note: here we handle the constraint more carefully for PSO
                swarm[i] += w * (best_positions[i] - swarm[i]) * r1 + \
                           c2 * (global_best - swarm[i]) * r2
                swarm[i] = project_feasible(swarm[i])
                
                val = compute_C5(swarm[i])
                if val < best_values[i]:
                    best_values[i] = val
                    best_positions[i] = swarm[i].copy()
                    
                if val < global_best_value:
                    global_best_value = val
                    global_best = swarm[i].copy()
                    
            # Reduce inertia weight over time
            w = 0.9 - (0.5 * iter_count / max_iter)
            
        return global_best, global_best_value

    # Main optimization loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        # PSO search phase
        if iter_count % 20 == 0 and iter_count > 0:
            pso_h, pso_C5 = pso_search()
            if pso_C5 < compute_C5(h):
                h = pso_h.copy()
        
        # Trust region update phase
        h = trust_region_update(h)
        
        # Evaluate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        iter_count += 1
        if elapsed() > budget_s * 0.9:
            break
            
        # Occasionally perturb to avoid local minima
        if iter_count % 50 == 0:
            idxs = np.random.choice(n_points, size=max(1, int(0.1 * n_points)), replace=False)
            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)

    # Final refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
