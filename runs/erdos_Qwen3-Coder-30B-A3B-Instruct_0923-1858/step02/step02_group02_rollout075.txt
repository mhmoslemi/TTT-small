import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    phi2 = 1.5
    max_velocity = 0.2

            # 10% initialized with perturbed initial_h_values
            h = h0 + np.random.normal(0, 0.05, n_points)
        else:
            # Random initialization
            h = np.random.rand(n_points)
        h = project(h, dx)
        swarm.append(h.copy())
        p_best.append(h.copy())
        p_best_values.append(evaluate(h, dx))
    
    # Global best
    global_best_idx = np.argmin(p_best_values)
    global_best = p_best[global_best_idx].copy()
    global_best_value = p_best_values[global_best_idx]
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    iteration = 0
    while time.time() - t0 < max_time:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            phi1 * r1 * (p_best[i] - swarm[i]) + 
                            phi2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project(swarm[i], dx)
            
            # Evaluate
            current_value = evaluate(swarm[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = swarm[i].copy()
                p_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
        
        # Occasionally perturb swarm to prevent stagnation
        if iteration % 100 == 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:
                    swarm[i] = project(swarm[i] + np.random.normal(0, 0.01, n_points), dx)
                    
    return (global_best, global_best_value, n_points)
