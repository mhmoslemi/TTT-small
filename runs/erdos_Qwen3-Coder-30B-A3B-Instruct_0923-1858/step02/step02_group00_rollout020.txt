import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Exponential smoothing and barrier method
    beta = 1000.0  # Temperature parameter for exponential penalty
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    beta_decay = 0.99
    gamma_increase = 1.1
    
    def compute_objective_and_gradient(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(corr)
        if max_corr == 0:
            max_corr = 1.0
            
        # Normalize correlations to avoid overflow
        corr_norm = corr / max_corr
        
        # Compute exponential penalty
        penalty = np.sum(np.exp(-beta * corr_norm))
        
        # Compute barrier
        barrier = -gamma * np.sum(np.log(np.clip(h_vec, 1e-10, 1-1e-10)) + 
                                  np.log(np.clip(1 - h_vec, 1e-10, 1-1e-10)))
        
        # Total objective
        obj = penalty + barrier
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            # Compute gradients for both
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            # Normalize correlations
            corr_plus_norm = corr_plus / max_corr
            corr_minus_norm = corr_minus / max_corr
            
            penalty_plus = np.sum(np.exp(-beta * corr_plus_norm))
            penalty_minus = np.sum(np.exp(-beta * corr_minus_norm))
            
            grad[i] = (penalty_plus - penalty_minus) / (2 * eps) + \
                      (-gamma * (1/h_vec[i] - 1/(1-h_vec[i])) if h_vec[i] > 1e-10 and h_vec[i] < 1-1e-10 else 0)
        
        return obj, grad

    last_improvement = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)

            best_h = h.copy()
            last_improvement = iter_count

        obj, grad = compute_objective_and_gradient(h)
        h = h - eta * grad
        
        # Project to feasible region
        h = project_feasible(h)
        
        # Adapt parameters
        if iter_count % 50 == 0:
            beta *= beta_decay
            gamma *= gamma_increase
            if eta > 1e-8:
                eta *= 0.5
        
        # Early stopping if no improvement in a while
        if iter_count - last_improvement > 5000:
            break
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
