import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_initial_h(n_points, dx):
    """Create a symmetric initial guess for h."""
    h = np.zeros(n_points)
    mid = n_points // 2
    h[:mid] = np.linspace(0, 1, mid)
    h[mid:] = np.linspace(1, 0, n_points - mid)
    h = project(h, dx)
    return h

def compute_gradient(h, dx, eps=1e-5):
    """Compute gradient using finite differences."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def upsample_h(h_low, n_points_high):
    """Upsample a low-resolution h to higher resolution."""
    x_low = np.linspace(0, 2, len(h_low))
    f = interp1d(x_low, h_low, kind='linear', fill_value='extrapolate')
    x_high = np.linspace(0, 2, n_points_high)
    h_high = f(x_high)
    return project(h_high, 2.0 / n_points_high)

def time_expired(start_time, budget):
    """Check if time budget has been exceeded."""
    elapsed = time.time() - start_time
    return elapsed > budget * 0.95

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Coarse optimization (30 points)
    n_points_coarse = 30
    dx_coarse = 2.0 / n_points_coarse
    h = create_initial_h(n_points_coarse, dx_coarse)
    best_h = h.copy()
    best_C = evaluate(h, dx_coarse)
    
    # FIRE algorithm parameters
    alpha = 0.99
    beta = 0.95
    t0 = 0.5
    t1 = 0.8
    current_t = t0
    velocity = np.zeros_like(h)
    
    for iteration in range(10000):
        if time_expired(start_time, budget_s):
            break
            
        gradient = compute_gradient(h, dx_coarse)
        velocity = alpha * velocity + beta * gradient
        h_trial = h + velocity
        h_trial = project(h_trial, dx_coarse)
        
        trial_C = evaluate(h_trial, dx_coarse)
        
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            current_t = t0
        else:
            velocity = velocity * 0.9
            current_t = current_t * 0.9
        
        current_t = max(current_t, t1)
        velocity = velocity * current_t

    # Stage 2: Refinement at higher resolution (60 points)
    n_points_fine = 60
    dx_fine = 2.0 / n_points_fine
    h = upsample_h(best_h, n_points_fine)
    best_h = h.copy()
    best_C = evaluate(h, dx_fine)
    
    # Reset FIRE algorithm parameters for fine tuning
    velocity = np.zeros_like(h)
    current_t = t0
    
    for iteration in range(5000):
        if time_expired(start_time, budget_s):
            break
            
        gradient = compute_gradient(h, dx_fine)
        velocity = alpha * velocity + beta * gradient
        h_trial = h + velocity
        h_trial = project(h_trial, dx_fine)
        
        trial_C = evaluate(h_trial, dx_fine)
        
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            current_t = t0
        else:
            velocity = velocity * 0.9
            current_t = current_t * 0.9
        
        current_t = max(current_t, t1)
        velocity = velocity * current_t

    return (best_h, best_C, n_points_fine)
