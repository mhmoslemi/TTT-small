import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx):
    """Perform uniform crossover."""
    mask = np.random.rand(len(parent1)) < 0.5
    offspring1 = np.where(mask, parent1, parent2)
    offspring2 = np.where(mask, parent2, parent1)
    return project(offspring1, dx), project(offspring2, dx)

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

    np.random.seed(seed)

        initial_h[15:] = np.linspace(1, 0, 30)
        initial_h = project(initial_h, dx)
    
    # GA parameters
    pop_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    max_generations = 5000
    stagnation_limit = 100
    enhancement_limit = 50

    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        population.append(generate_individual(n_points, dx))
    
    # Evaluate initial population
    fitnesses = [evaluate(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_C = fitnesses[best_idx]
    
    t0 = time.time()
    generation = 0
    stagnation_count = 0
    enhancement_count = 0
    
    while time.time() - t0 < budget_s - 5 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
