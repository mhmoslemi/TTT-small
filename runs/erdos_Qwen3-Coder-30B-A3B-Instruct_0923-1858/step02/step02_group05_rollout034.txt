import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        # Start with initial solution or slightly perturbed
        h = initial_h.copy()
        # Add small random perturbations
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h += perturbation
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        p_best_values.append(evaluate(h, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    iteration = 0
    while time.time() - t0 < max_time:
        if iteration % 100 == 0:
            # Print progress every 100 iterations
            pass
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate current particle
            current_value = evaluate(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                
                # Update global best if needed
                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value
        
        iteration += 1
    
    # Final evaluation of the best solution
    final_c5 = evaluate(g_best, dx)
