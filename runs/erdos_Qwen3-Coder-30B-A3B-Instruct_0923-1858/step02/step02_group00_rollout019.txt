import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a "tent" shape that's slightly more optimized for the problem
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1)) * 0.7 + 0.3
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization approach
    def pso_optimization():
        # Parameters
        n_particles = 50
        n_iter = 100
        inertia = 0.9
        c1 = 2.0
        c2 = 2.0
        max_vel = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        
        for _ in range(n_particles):
            # Create random initial solutions and project them
            h_init = np.random.rand(n_points) * 0.5 + 0.25  # Between 0.25 and 0.75
            h_init = project_feasible(h_init)
            particles.append(h_init.copy())
            velocities.append(np.random.rand(n_points) * 0.001)
            pbest.append(h_init.copy())
            pbest_vals.append(compute_C5(h_init))
            
        # Global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        for iter_no in range(n_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (inertia * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) +
                                c2 * r2 * (gbest - particles[i]))
                
                # Clip velocities
                velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate fitness
                current_val = compute_C5(particles[i])
                
                # Update personal best
                if current_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = current_val
                    
                    if current_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = current_val
        
        return gbest, gbest_val

    # Gradient-based refinement using trust region
    def trust_region_refinement(h_vec, max_iter=50):
        h_current = h_vec.copy()
        current_C5 = compute_C5(h_current)
        
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute gradient of the objective at current point
            # Use finite differences to estimate gradient of the maximum correlation
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_current) - 1)
            
            grad = np.zeros_like(h_current)
            eps = 1e-5
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update with trust region
            h_new = h_current - 0.01 * grad
            h_new = project_feasible(h_new)
            new_C5 = compute_C5(h_new)
            
            if new_C5 < current_C5:
                h_current = h_new
                current_C5 = new_C5
            else:
                break
                
        return h_current, current_C5
    
    # Main optimization loop
    iter_count = 0
    while elapsed() < budget_s:
        # PSO stage
        psotop_h, psotop_val = pso_optimization()
        
        # Refinement stage
        refined_h, refined_C5 = trust_region_refinement(psotop_h)
        
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
        
        iter_count += 1

    return (best_h.tolist(), best_C5, len(best_h))
