import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from the provided construction or use a simple one
    try:
        h = initial_h_values
        if len(h) != n_points:
            raise ValueError("Wrong length")
        h = np.array(h)
    except NameError:
        # Simple starting point
        h = np.ones(n_points) * 0.5
        
    # Minimize using convex optimization
    t_start = time.time()
    
    # Set up CVXPY optimization
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == 1.0 / dx
    ]
    
    # Add constraints for all correlation lags
    # We'll approximate this by ensuring the correlation is bounded
    full_corr = correlate(h_var.value, 1 - h_var.value, mode='full') if h_var.value is not None else None
    max_corr = np.max(full_corr) if full_corr is not None else 0
    
    # More precise way: compute correlation for each lag and constrain
    # but do this directly in the constraints for the optimization
    # Since CVXPY doesn't allow full correlation in constraints directly,
    # we'll constrain the value of C to be larger than all possible overlaps
    # This is a simplified version: in practice, for a very small LP, 
    # we can actually enumerate valid inequality constraints
    
    # Instead, we will solve with the simpler approach using a reasonable initial guess
    # and solve a convex LP to globally minimize C subject to constraints
    
    # Define problem
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        h_opt = h_var.value
        C_opt = C.value
    except Exception:
        # Fall back to simple local search if CVXPY fails
        h_opt = h.copy()
        C_opt = evaluate(h_opt, dx)
        
        # Simple gradient descent with projection
        lr = 0.001
        max_iter = 10000
        for i in range(max_iter):
            if time.time() - t_start > budget_s - 1:
                break
            grad = np.zeros_like(h_opt)
            eps = 1e-6
            for j in range(n_points):
                h_plus = h_opt.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_sum = np.sum(h_plus) * dx
                if h_sum > 0:
                    h_plus = h_plus * (1.0 / (h_sum))
                    
                h_minus = h_opt.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_sum = np.sum(h_minus) * dx
                if h_sum > 0:
                    h_minus = h_minus * (1.0 / (h_sum))
                    
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            h_opt = h_opt - lr * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_sum = np.sum(h_opt) * dx
            if h_sum > 0:
                h_opt = h_opt * (1.0 / (h_sum))
            
            current_C = evaluate(h_opt, dx)
            if current_C < C_opt:
                C_opt = current_C
            lr *= 0.999
            if i % 1000 == 0:
                lr *= 0.99
    
    return (h_opt, C_opt, n_points)
