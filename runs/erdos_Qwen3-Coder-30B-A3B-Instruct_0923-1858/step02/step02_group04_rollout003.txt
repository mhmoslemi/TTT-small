import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Initialize with initial_h_values or create triangular
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    # Prepare the optimization problem
    dx = 2.0 / n_points
    h_vars = cp.Variable(n_points)
    
    # Define constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2
    ]
    
    # Define the objective using a max over all shifts
    t = cp.Variable()
    max_overlaps = []
    for k in range(n_points * 2 - 1):
        # Compute overlap for shift k
        if k < n_points:
            overlap = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
        else:
            overlap = cp.sum(cp.multiply(h_vars[k-n_points+1:], 1 - h_vars[:n_points-(k-n_points+1)]))
        max_overlaps.append(overlap)
    
    # Add constraint that all overlaps are bounded by t
    for overlap in max_overlaps:
        constraints.append(overlap <= t)
    
    # Minimize t (upper bound on maximum overlap)
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Solve
    try:
        problem.solve(solver=cp.MOSEK, verbose=False, time_limit=time_limit - (time.time() - start_time))
        h_opt = h_vars.value
        c5_bound = problem.value
        if c5_bound is None or np.isnan(c5_bound):
            c5_bound = float('inf')
    except Exception as e:
        # Fallback to original approach if CVXPY fails
        h_opt = h.copy()
        c5_bound = float('inf')
        # Compute overlap directly as fallback
        corr = correlate(h_opt, 1 - h_opt, mode="full")
        c5_bound = np.max(corr) * dx
    
    # Final evaluation - ensure feasibility and compute true C5
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    if current_sum > 0:
        h_opt = h_opt * (n_points / 2) / current_sum
    
    corr = correlate(h_opt, 1 - h_opt, mode="full")
    actual_c5 = np.max(corr) * dx
    
    return (h_opt.tolist(), actual_c5, n_points)
