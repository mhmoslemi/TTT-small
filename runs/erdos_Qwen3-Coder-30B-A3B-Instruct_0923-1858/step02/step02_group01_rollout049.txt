import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def perturb(h, dx):
    """Perturb h using a random change."""
    perturbation = np.random.uniform(-0.01, 0.01, size=h.shape)
    h_candidate = h + perturbation
    return project(h_candidate, dx)
