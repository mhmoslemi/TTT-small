import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Modified Nelder-Mead optimization approach
    patience = 0
    max_patience = 200
    iter_count = 0
    adaptive_eta = 0.01
    
    # Create a list of perturbation strategies
    strategies = [
        lambda h, idx: np.random.normal(0, 0.03),
        lambda h, idx: np.random.normal(0, 0.02),
        lambda h, idx: np.random.normal(0, 0.01),
        lambda h, idx: np.random.uniform(-0.05, 0.05),
    ]
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Compute current value
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Escape local minimum
            idx = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for i in idx:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
            patience = 0
            adaptive_eta = max(0.001, adaptive_eta * 0.95)
        
        # Perturb some points
        num_perturb = max(1, n_points // 20)
        idxs = np.random.choice(n_points, size=num_perturb, replace=False)
        strategy = np.random.choice(strategies)
        for i in idxs:
            h[i] = np.clip(h[i] + strategy(h, i), 0, 1)
            
        h = project_feasible(h)
        iter_count += 1
    
    # Final verification
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
