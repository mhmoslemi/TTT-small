import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a smooth triangular initial solution
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # CVXPY optimization approach
    dx = 2.0 / n_points
    
    # Define CVXPY variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Overlap constraints for all possible shifts
    max_k = n_points - 1
    for k in range(max_k):
        # Compute overlap for shift k
        # We compute sum h[i] * (1 - h[i+k]) for valid indices
        overlap = 0
        for i in range(n_points - k):
            overlap += h_var[i] * (1 - h_var[i + k])
        constraints.append(overlap <= t)
    
    # Objective function
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Try solving with MOSEK or fallback to ECOS
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Fallback to gradient descent if convex optimization fails
            h = h.copy()
            def compute_C5(h_vec):
                corr = correlate(h_vec, 1 - h_vec, mode="full")
                return np.max(corr) * dx
            
            def project_feasible(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                current_sum = np.sum(h_vec)
                if current_sum > 0:
                    h_vec = h_vec * (n_points / 2) / current_sum
                return h_vec
            
            # Quick gradient descent
            eta = 0.01
            for i in range(2000):
                elapsed = time.time() - start_time
                if elapsed >= budget_s * 0.98:
                    break
                current_C5 = compute_C5(h)
                # Simple update using finite differences
                eps = 1e-4
                grad = np.zeros_like(h)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = project_feasible(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    h_minus = h.copy()
                    h_minus[j] -= eps
                    h_minus = project_feasible(h_minus)
                    corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    grad[j] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
                h = h - eta * grad
                h = project_feasible(h)
                eta *= 0.9995
            
            # Return best solution found
            final_C5 = compute_C5(h)
            return (h.tolist(), float(final_C5), n_points)
    
    # Extract solution if successful
    if prob.status == cp.OPTIMAL:
        h_opt = h_var.value
        # Clip to feasible range
        h_opt = np.clip(h_opt, 0, 1)
        # Normalize to sum to n_points / 2
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Compute final C5
        corr = correlate(h_opt, 1 - h_opt, mode="full")
        final_C5 = np.max(corr) * dx
    else:
        # Use previous solution as fallback
        h_opt = h.copy()
        corr = correlate(h_opt, 1 - h_opt, mode="full")
        final_C5 = np.max(corr) * dx
    
    return (h_opt.tolist(), float(final_C5), n_points)
