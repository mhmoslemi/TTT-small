import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_particles = 50
    c1, c2 = 2.0, 2.0

    p_best = [p.copy() for p in particles]

    best_idx = np.argmin(p_best_C)
    g_best = p_best[best_idx].copy()

    # PSO exploration phase (70% of time)
    t0 = time.time()

            if current_C < p_best_C[i]:

                if current_C < best_C:

                    g_best = particles[i].copy()
