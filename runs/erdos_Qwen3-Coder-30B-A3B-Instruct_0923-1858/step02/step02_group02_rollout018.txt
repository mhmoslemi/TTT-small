import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Start with a simple uniform distribution
        h = np.ones(n_points) * 0.5

    h = project(h, dx)
    best_h = h.copy()
    best_C = evaluate(h, dx)

    # Simulated Annealing parameters
    T0 = 100.0
    alpha = 0.999
    max_iter = 10000
    T = T0
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom

    # Perturbation parameters
    noise_std = 0.1
    perturb_fraction = 0.05  # 5% of points

    for i in range(max_iter):
        if time.time() - t0 > max_time:
            break

        # Create proposal by perturbing some points
        h_proposal = h.copy()
        num_perturbed = max(1, int(n_points * perturb_fraction))
        indices = np.random.choice(n_points, num_perturbed, replace=False)
        for idx in indices:
            h_proposal[idx] += np.random.normal(0, noise_std)
        
        # Project to feasible region
        h_proposal = project(h_proposal, dx)
        
        # Evaluate
        E_proposal = evaluate(h_proposal, dx)
        E_current = evaluate(h, dx)
        delta_E = E_proposal - E_current

        # Accept or reject
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_proposal
            if E_proposal < best_C:
                best_C = E_proposal
                best_h = h_proposal.copy()

        T *= alpha

    return (best_h, best_C, n_points)
