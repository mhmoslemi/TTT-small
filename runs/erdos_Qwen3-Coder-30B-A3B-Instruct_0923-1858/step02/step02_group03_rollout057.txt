import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    # Construct h from Fourier coefficients
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    # Ensure h is in [0,1] and integrate to 1
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    P = 50  # number of particles
    max_iter = 1000
    
    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))  # a_coeffs, b_coeffs
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_scores = np.full(P, np.inf)
    
    gbest_idx = 0
    gbest_score = np.inf
    
    omega = 0.7
    phi1 = phi2 = 1.5
    
    # PSO iterations
    for iter in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Evaluate all particles
        for i in range(P):
            # Reconstruct h from current particle
            a_coeffs = particles[i, :N]
            b_coeffs = particles[i, N:]
            
            t = np.linspace(0, 2, n_points)
            h_reconstructed = fourier_to_h(a_coeffs, b_coeffs, t)
            h_projected = clip_and_normalize(h_reconstructed, dx)
            
            score = evaluate(h_projected, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest_idx = i
                
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                             phi1 * r1 * (pbest[i] - particles[i]) +
                             phi2 * r2 * (pbest[gbest_idx] - particles[i]))
            particles[i] += velocities[i]
    
    # Final reconstruction with the best particle
    best_a = pbest[gbest_idx, :N]
    best_b = pbest[gbest_idx, N:]
    t_final = np.linspace(0, 2, n_points)
    best_h = fourier_to_h(best_a, best_b, t_final)
    best_h = clip_and_normalize(best_h, dx)
    final_c = evaluate(best_h, dx)
