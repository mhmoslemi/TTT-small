import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def perturb(h, dx):
    """Generate a perturbed version of h."""
    perturbation = np.random.uniform(-0.01, 0.01, size=h.shape)
    h_candidate = h + perturbation
    return project(h_candidate, dx)

def simulated_annealing(h, dx, budget_s):
    T_initial = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    T = T_initial
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    start_time = time.time()
    
    iterations = 0
    while T > T_min and time.time() - start_time < budget_s - 10:
        h_candidate = perturb(h, dx)
        current_C = evaluate(h_candidate, dx)
        
        delta = current_C - best_C
        
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_candidate.copy()
            best_C = current_C
            best_h = h.copy()
            
        if delta < 0:
            T = T * cooling_rate
        else:
            T = T * 1.01  # Short-term increase if stuck
            
        cooling_rate = min(cooling_rate, 0.999)
        iterations += 1
        
    return best_h, best_C
