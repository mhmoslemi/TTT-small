import time
import cvxpy as cp
from scipy.signal import correlate

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    try:
        # Set up variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = []
        # Feasibility constraints
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        # For each shift k, compute the overlap integral
        # We compute the full correlation with mode="full", then extract valid shifts
        # But we can simplify to just adding the constraints for all valid shifts
        # The maximum overlap occurs when we consider the correlation of h with 1-h
        # For each shift k, the correlation is sum(h[i] * (1 - h[i + k]))
        # We add constraint that all these are bounded by t
        
        # Manually construct constraint for each valid shift
        dx = 2.0 / n_points
        # Since the correlation is in full mode, we get valid shifts from -n_points+1 to n_points-1
        # But we only care about shifts where both arrays have valid entries 
        # Shifts range from 0 to n_points-1 with 1-h shifted by k positions
        shifts = range(-n_points + 1, n_points)  # All possible shifts in full correlation
        
        # In Python's correlate with mode="full", the output is length 2*n_points - 1 
        # Shifts go from 0 to 2*n_points - 2, corresponding to shifts from -n_points+1 to n_points-1 
        for k in shifts:
            # k=0 corresponds to shift -n_points+1, etc.
            corr_sum = 0
            if k >= 0:
                # Horizontal shift of h to left by k
                for i in range(n_points - k):
                    corr_sum += h_vars[i] * (1 - h_vars[i + k])
                # Add constraint that overlap for this shift <= t
                constraints.append(corr_sum <= t)
            else:
                # Horizontal shift of 1-h to left by |k|, so shift h right by |k|
                k_abs = abs(k)
                for i in range(n_points - k_abs):
                    corr_sum += h_vars[i + k_abs] * (1 - h_vars[i])
                constraints.append(corr_sum <= t)
                
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Try solving (with fallback if solver fails or timeout near)
        elapsed = time.time() - start_time
        if elapsed < budget_s * 0.95:
            prob.solve(solver=cp.MOSEK, verbose=False)

            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:

        def compute_C5(h_vec):
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        def project_feasible(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            current_sum = np.sum(h_vec)

            eps = 1e-5
            grad = np.zeros_like(h)
            h_base = h.copy()
            
            c_base = compute_C5(h_base)
            for j in range(n_points):
                h_plus = h_base.copy()
                h_plus[j] += eps
                h_plus = project_feasible(h_plus)
                c_plus = compute_C5(h_plus)
                grad[j] = (c_plus - c_base) / eps
                
            # Update
            h = h - eta * grad
            h = project_feasible(h)
            
            c_current = compute_C5(h)
            if c_current < best_C5:
                best_C5 = c_current
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            if patience > max_patience:
                break
            eta *= 0.995  # decay gradient step size

        h = best_h
        best_C5 = compute_C5(h)
    
    final_C5 = compute_C5(h)
    return (h.tolist(), final_C5, n_points)
