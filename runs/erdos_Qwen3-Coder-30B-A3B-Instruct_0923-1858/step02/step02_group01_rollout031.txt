import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_symmetric_initial_h(n_points, dx):
    """Create a symmetric initial function h."""
    h = np.zeros(n_points)
    half = n_points // 2
    # Linear increase to peak
    h[:half] = np.linspace(0, 1, half)
    # Linear decrease from peak
    h[half:] = np.linspace(1, 0, n_points - half + 1)[1:]
    return project(h, dx)

def compute_gradient(h, dx, eps=1e-5):
    """Compute gradient using finite differences."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def time_expired(start_time, budget):
    return time.time() - start_time > budget * 0.95

def fire_optimize(h, dx, max_iter, start_time, budget):
    """FIRE optimization algorithm."""
    # FIRE parameters
    alpha = 0.99
    beta = 0.95
    t0 = 0.5
    t1 = 0.8
    current_t = t0
    
    # Initialize velocity
    velocity = np.zeros_like(h)
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    for iteration in range(max_iter):
        if time_expired(start_time, budget):
            break
            
        # Compute gradient
        gradient = compute_gradient(h, dx)
        
        # Update velocity and position
        velocity = alpha * velocity + beta * gradient
        h_trial = h + velocity
        
        # Project to constraints
        h_trial = project(h_trial, dx)
        
        # Evaluate trial
        trial_C = evaluate(h_trial, dx)
        
        # Update if better
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            current_t = t0
            # Reset velocity for good step
            velocity = np.zeros_like(velocity)
        else:
            # Bad step, apply damping
            velocity = velocity * 0.5
            current_t = current_t * 0.9
        
        # Keep current_t bounded
        current_t = min(current_t, t1)
        velocity = velocity * current_t
        
        # Occasionally restart velocity when stuck
        if iteration % 100 == 0 and iteration > 0:
            velocity = np.zeros_like(velocity)
    
    return h, best_C

def upsample_h(h_low, n_points_high):
    """Upsample h from lower resolution to higher."""
    x_low = np.linspace(0, 2, len(h_low))
    x_high = np.linspace(0, 2, n_points_high)
    f = interp1d(x_low, h_low, kind='linear', fill_value='extrap')
    h_high = f(x_high)
    dx_high = 2.0 / n_points_high
    return project(h_high, dx_high)
