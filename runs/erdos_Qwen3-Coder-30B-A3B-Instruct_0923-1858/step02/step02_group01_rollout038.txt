import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx, crossover_prob=0.8):
    """Uniform crossover."""
    if np.random.rand() < crossover_prob:
        mask = np.random.rand(len(parent1)) < 0.5
        offspring1 = np.where(mask, parent1, parent2)
        offspring2 = np.where(mask, parent2, parent1)
        return project(offspring1, dx), project(offspring2, dx)
    else:
        return parent1, parent2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Tournament selection."""
    selected_indices = np.random.choice(len(population), size=tournament_size)
    best_index = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_index]

    np.random.seed(seed)

    dx = 2.0 / n_points

        # First 15 points: increasing linearly from 0 to 1
        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = project(initial_h, dx)
    
    # GA parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    tournament_size = 3
    max_generations = 10000
    stagnation_threshold = 50
    reset_threshold = 100
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points, dx))
    
    best_h = initial_h.copy()
    best_C = evaluate(initial_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom for cleanup
    generation = 0
    stagnant_count = 0
    
    while time.time() - t0 < max_time and generation < max_generations:
        if stagnant_count > reset_threshold:
            # Reset population
            population = [initial_h.copy()]
            for _ in range(population_size - 1):
                population.append(generate_individual(n_points, dx))
            stagnant_count = 0
            mutation_rate = 0.2
        elif stagnant_count > stagnation_threshold:

        # Evaluate fitness
        fitnesses = [evaluate(ind, dx) for ind in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)

        new_population = []

        new_population.append(population[min_fitness_idx].copy())
        
        # Generate offspring
        for _ in range(num_offspring):
            parent1 = tournament_selection(population, fitnesses, tournament_size)

            child1 = mutate(child1, dx, mutation_rate)
            child2 = mutate(child2, dx, mutation_rate)
