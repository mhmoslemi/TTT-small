import numpy as np
import time
from scipy.signal import correlate
from scipy import signal

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

        # Create a smooth piecewise structure
        x = np.linspace(0, 2, n_points)
        # Create a "bump" centered around middle of interval
        h = 0.5 * (1 + np.sin(2 * np.pi * x))  # Smooth oscillation
        h = project(h, dx)

    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    lr = 0.01
    decay_steps = 100
    decay_factor = 0.99
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # Optimizer parameters
    num_optimization_steps = 50000
    
    for i in range(num_optimization_steps):
        if time.time() - t0 > max_time:
            break
            
        # Estimate gradient using finite differences
        epsilon = 1e-4
        grad = np.zeros_like(h)
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
        
        # Update step
        h = h - lr * grad
        
        # Project to constraints
        h = project(h, dx)
        
        # Occasionally add noise to escape local minima
        if i % 50 == 0 and np.random.rand() < 0.3:
            noise = np.random.normal(0, 0.005, len(h))
            h += noise
            h = project(h, dx)
        
        # Occasionally decay learning rate
        if i % decay_steps == 0 and i > 0:
            lr *= decay_factor
            
        # Check if current solution is better
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
    
    return (best_h, best_C, n_points)
