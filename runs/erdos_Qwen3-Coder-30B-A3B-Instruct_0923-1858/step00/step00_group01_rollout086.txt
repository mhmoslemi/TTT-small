import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from available construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Simple configuration as fallback
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.6
        h[-1] = 0.4
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure correct integral constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    def compute_C5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def project_onto_feasible(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2.0) / np.sum(h_vals)
        return h_vals
    
    def perturb_h(h_vals, num_attempts=10):
        best_h = h_vals.copy()
        best_C5 = compute_C5(h_vals)
        
        corr_full = correlate(h_vals, 1 - h_vals, mode='full')
        shift_idx = np.argmax(corr_full)
        max_k = shift_idx - (len(h_vals) - 1)
        
        # Determine regions contributing to the max overlap
        overlap_start = max(0, -max_k)
        overlap_end = min(len(h_vals), len(h_vals) - max_k)
        
        for _ in range(num_attempts):
            h_new = h_vals.copy()
            
            # Pick two points to redistribute
            i, j = np.random.choice(range(len(h_vals)), 2, replace=False)
            
            # Prioritize overlap regions if possible
            if overlap_start <= i < overlap_end and overlap_start <= j < overlap_end:
                # Shift more from i to j when beneficial
                transfer_max = min(h_new[i], 1 - h_new[j])
                if transfer_max > 0:
                    amount = np.random.uniform(0, transfer_max)
                    h_new[i] -= amount
                    h_new[j] += amount
            else:
                # General random transfer
                transfer_max = min(h_new[i], 1 - h_new[j])
                if transfer_max > 0:
                    amount = np.random.uniform(0, transfer_max)
                    h_new[i] -= amount
                    h_new[j] += amount
            
            # Project and recompute
            h_new = project_onto_feasible(h_new)
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5:
                best_h = h_new
                best_C5 = new_C5
                
        return best_h, best_C5
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    iterations_without_improvement = 0
    
    max_iterations = 1000
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_h, new_C5 = perturb_h(current_h)
        
        if new_C5 < current_C5:
            current_h = new_h
            current_C5 = new_C5
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
            
        if iterations_without_improvement >= 100:
            break
    
    # Interpolate to double resolution if beneficial
    if len(current_h) < 100:
        h_dbl = np.interp(np.linspace(0, 2, 2 * len(current_h)), 
                          np.linspace(0, 2, len(current_h)), 
                          current_h)
        h_dbl = project_onto_feasible(h_dbl)
        test_C5 = compute_C5(h_dbl)
        if test_C5 < current_C5:
            current_h = h_dbl
            current_C5 = test_C5
    
    final_h = project_onto_feasible(current_h)
    final_C5 = compute_C5(final_h)
    return (final_h.tolist(), final_C5, len(final_h))
