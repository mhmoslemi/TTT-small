import numpy as np
import time
from scipy import optimize

def evaluate_overlap(h, dx):
    # Compute the overlap integral for all shifts
    n = len(h)
    h_aug = np.concatenate([h, np.zeros(n)])
    overlaps = np.zeros(2*n - 1)
    
    for k in range(2*n - 1):
        if k < n:
            overlaps[k] = np.sum(h[:n-k] * (1 - h_aug[k:k+n]))
        else:
            overlaps[k] = np.sum(h[k-n+1:] * (1 - h_aug[:n-k+1]))
    
    return np.max(overlaps) * dx

def project_h(h, target_sum):
    # Project h to satisfy sum constraint while keeping values in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scatter or gather delta proportionally based on current values
    delta = target_sum - current_sum
    if abs(delta) < 1e-10:
        return h
    
    # Adjust by redistributing the delta proportionally to remaining capacity
    remaining_capacity = 1 - h
    total_capacity = np.sum(remaining_capacity)
    if total_capacity > 1e-10:
        scale = min(delta / total_capacity, 1.0)
        h = np.clip(h + scale * remaining_capacity, 0, 1)
    
    # Re-adjust if over/under-corrected
    if abs(np.sum(h) - target_sum) > 1e-10:
        # linear adjustment to meet exact constraint
        adjustment = (target_sum - np.sum(h)) / len(h)
        h = np.clip(h + adjustment, 0, 1)
        
    return h
