import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial values if available, otherwise create a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    step_size = 0.01
    
    # Projection function to enforce constraints
    def project(h):
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h) * dx
        if current_sum > 0:
            h = h * (n_points / 2.0) / current_sum
        return np.clip(h, 0, 1)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        corr *= dx
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        # Gradient direction for the dominant lag
        # Cross correlation convolves h with (1-h) at k = max_k
        # Gradient w.r.t h[i]: contribution to max_k overlap
        grad = np.zeros_like(h)
        
        # Calculate gradient for dominant lag k* (full correlation index)
        # For indices such that i + k* = j (j in 0..2n-2)
        # where k* = max_k - (n-1) is the actual lag
        
        actual_lag = max_k - (n_points - 1)
        # Weights need to be from 1-h for specific positions
        
        # More accurate gradient calculation:
        # we compute h[i] -> contributes to overlap at lag k*
        # (h[i] * (1-h[i+k*)) when k* = i - j, and vice versa
        # To be precise, correlation of h with (1-h) at k* 
        # gives contribution to i from coefficient (1 - h[i+k*])
        # and to (i+k*) from h[i]
        
        # We will handle this in an efficient way.
        # Find all i where gradient contributes
        for i_idx in range(n_points):
            j_idx = i_idx + actual_lag
            if 0 <= j_idx < n_points:
                grad[i_idx] += -1 * (1 - h[j_idx])  # derivative of h[i]*
                # Also add contribution from (1-h)
                grad[j_idx] += -1 * h[i_idx] # derivative of h[j_idx]
        
        # Apply gradient step
        h = h - step_size * grad
        h = project(h)
        
        # Adaptive step size
        if i % 100 == 0:
            step_size *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
