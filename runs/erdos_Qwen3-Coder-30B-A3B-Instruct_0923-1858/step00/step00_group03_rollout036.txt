import numpy as np
import time
import random
from scipy import correlate

def binary_convert(h, n_points):
    """Convert h to binary with exact sum constraint."""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h
    # Adjust to meet sum constraint
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones_indices = np.where(h == 1)[0]
        to_flip = current_sum - target_sum
        flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(h == 0)[0]
        to_flip = target_sum - current_sum
        flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
        h[flip_indices] = 1
    return h

def compute_C5(h, dx):
    """Compute C5 from binary h."""
    # Compute overlap for all shifts
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def get_neighbors(h):
    """Get a neighbor by flipping two bits."""
    h_copy = h.copy()
    ones = np.where(h_copy == 1)[0]
    zeros = np.where(h_copy == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h_copy
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h_copy[i] = 0
    h_copy[j] = 1
    return h_copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = binary_convert(np.array(initial_h_values), 70)
    else:
        # Initialize binary h with 1s in first half
        h = np.zeros(70, dtype=int)
        h[:35] = 1

    # Parameters
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    T = 1.0
    alpha = 0.999
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    improvement_threshold = 1e-6

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_h = get_neighbors(h)
        new_C5 = compute_C5(new_h, dx)
        
        delta = new_C5 - best_C5
        
        # Accept or reject based on Metropolis criterion
        if delta < 0 or random.random() < np.exp(-delta / T):
            h = new_h
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        
        # Update temperature
        T *= alpha
        
        # Check for termination based on small improvement
        if abs(delta) < improvement_threshold:
            break
            
    return (best_h.tolist(), best_C5, n_points)
