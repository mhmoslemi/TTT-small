import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a starting point
        h = np.full(n_points, 0.5)
        # Add small random perturbation
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    def compute_c5(h):
        """Compute C5 as the maximum overlap."""
        # Compute cross-correlation of h with (1 - h)
        corr = np.correlate(h, 1 - h, mode='full')
        # The maximum value is at center
        max_corr = np.max(corr)
        return max_corr * dx

    def project_h(h):
        """Project h to feasible region."""
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to enforce sum = n_points/2
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h

    def compute_gradient(h):
        """Approximate gradient of C5 with respect to h."""
        # This is a simple approximation
        # For a better implementation, would compute gradient of max correlation
        # Here we use finite differences for simplicity
        eps = 1e-5
        grad = np.zeros_like(h)
        baseline = compute_c5(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            val_plus = compute_c5(h_plus)
            grad[i] = (val_plus - baseline) / eps
        return grad

    # Optimization routine
    max_iterations = 10000
    learning_rate = 1e-2
    last_improvement = 0
    last_c5 = compute_c5(h)
    
    for i in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Compute gradient and update
        grad = compute_gradient(h)
        h_new = h - learning_rate * grad
        
        # Project back to feasible region
        h_new = project_h(h_new)
        
        # Check improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 20:
                learning_rate *= 0.9
                last_improvement = 0
                
        if last_improvement > 100 or (last_improvement > 10 and learning_rate < 1e-6):
            break
    
    # Final evaluation
    final_c5 = compute_c5(h)
    return (h.tolist(), final_c5, n_points)
