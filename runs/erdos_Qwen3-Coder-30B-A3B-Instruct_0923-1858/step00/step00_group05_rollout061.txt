def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = signal.correlate(h_values, 1 - h_values, mode="full")
    # Scale by dx to get integral
    scaled_correlations = correlations * dx
    # Return maximum overlap
    return np.max(scaled_correlations)

    pert = np.random.uniform(-eps, eps, len(h))

    h_new = np.clip(h_new, 0, 1)
    return h_new
