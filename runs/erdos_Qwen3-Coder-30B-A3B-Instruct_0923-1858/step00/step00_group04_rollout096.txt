import numpy as np
import time
from scipy import correlate

def evaluate_h(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(np.abs(corr)) * dx
    return c5

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / 2)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    alpha = np.random.random()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return project_to_feasible(h, len(h))

def tournament_selection(population, fitnesses, k=5):
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42
    pop_size = 30
    max_generations = 1000
    elite_rate = 0.1
    mutation_rate = 0.1
    crossover_rate = 0.7
    tournament_size = 5
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 1.0 / 2)
    
    population = []
    for _ in range(pop_size):
        if len(population) == 0:
            h = initial_h.copy()
        else:
            h = np.random.random(n_points)
            h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s and generation < max_generations:
        fitnesses = [evaluate_h(h, n_points) for h in population]
        
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitnesses.index(current_best)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= 200:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elite selection
        elite_count = int(elite_rate * pop_size)
        new_population = sorted_population[:elite_count]
        
        # Tournament selection and reproduction
        while len(new_population) < pop_size:
            parent1 = tournament_selection(sorted_population, sorted_fitnesses, tournament_size)
            parent2 = tournament_selection(sorted_population, sorted_fitnesses, tournament_size)
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:pop_size]
        generation += 1
        
        # Periodic print of progress
        if generation % 50 == 0:
            print(f"Generation {generation}: Best C5 = {best_c5:.6f}")
    
    # Final check of the best solution
    if best_h is not None:
        best_c5 = evaluate_h(best_h, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
