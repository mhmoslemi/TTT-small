import numpy as np
from scipy import signal
import time

def evaluate_overlap(h):
    """Evaluate the overlap integral C5 for a given step function h."""
    dx = 2.0 / len(h)
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Maximum value scaled by dx
    return np.max(conv) * dx

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    # Scale to meet the constraint sum(h) = n_points / 2
    scale_factor = (n_points / 2) / np.sum(h)
    return h * scale_factor

    reset_counter = 0
    max_iters = 5000

    best_c5 = evaluate_overlap(best_h)
    # Initialize simplex
    simplex = [initial_h.copy()]
    for i in range(n_points):
        vertex = initial_h.copy()
        vertex[i] += simplex_size * np.random.uniform(-1, 1)

        simplex.append(vertex)

        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()

        if iter_num > 50:

                else:

        centroid = np.mean(simplex[:-1], axis=0)

        alpha = 1.0

            gamma = 2.0

            expanded = project_h(expanded, n_points)
            expanded_value = evaluate_overlap(expanded)
            if expanded_value < reflected_value:
                simplex[-1] = expanded
                values[-1] = expanded_value
            else:
                simplex[-1] = reflected
                values[-1] = reflected_value
        elif reflected_value < values[-2]:
            simplex[-1] = reflected
            values[-1] = reflected_value
        else:
            # Contraction
            beta = 0.5
            if reflected_value < values[-1]:
                contracted = centroid + beta * (reflected - centroid)
            else:
                contracted = centroid + beta * (simplex[-1] - centroid)
            contracted = project_h(contracted, n_points)
            contracted_value = evaluate_overlap(contracted)

                values[-1] = contracted_value

                best = simplex[0]
                for i in range(1, n_points + 1):
                    simplex[i] = best + rho * (simplex[i] - best)
                    simplex[i] = project_h(simplex[i], n_points)
                    values[i] = evaluate_overlap(simplex[i])
