import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:

        n_points = 42
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[1::2] = 0.0
        h *= n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    perturb_freq = 100
    epsilon = 0.02
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Find indices where convolution is maximal
        max_indices = np.where(conv == np.max(conv))[0]
        k_max = max_indices[len(max_indices)//2] - (n_points - 1)
        
        # Adjust h to reduce overlap at k_max
        new_h = h.copy()
        for j in range(n_points):
            idx = (j + k_max) % n_points
            if h[j] > 0 and h[idx] < 1:
                delta = min(epsilon, h[j], 1 - h[idx])
                new_h[j] -= delta
                new_h[idx] += delta
        
        # Project to maintain constraint sum(h) * dx = 1
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h) * dx
        correction = (n_points / 2) / current_sum
        new_h *= correction
        new_h = np.clip(new_h, 0, 1)
        
        # Ensure sum constraint is satisfied
        actual_sum = np.sum(new_h) * dx
        if abs(actual_sum - 1.0) > 1e-6:
            diff = (1.0 - actual_sum) / n_points
            new_h += diff
            new_h = np.clip(new_h, 0, 1)
        
        h = new_h
        
        if i % perturb_freq == 0:
            # Apply random perturbation to escape local optima
            perturb_indices = np.random.choice(n_points, size=n_points//4, replace=False)
            tot_move = 0
            for idx in perturb_indices:
                if h[idx] > 0:
                    move = min(epsilon, h[idx])
                    tot_move += move
                    h[idx] -= move
                    
            # Distribute the moved mass
            if tot_move > 0:
                empty_indices = np.where(h < 1)[0]
                if len(empty_indices) > 0:
                    n_empty = len(empty_indices)
                    to_add = tot_move / n_empty
                    for idx in empty_indices:
                        h[idx] = min(1.0, h[idx] + to_add)
            
            # Ensure sum constraint again
            h *= (n_points / 2) / (np.sum(h) * dx)
            h = np.clip(h, 0, 1)
    
    # Final computation of best
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
