import numpy as np
from scipy import signal
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute overlap integral for all lags
    correlation = signal.correlate(h, 1 - h, mode='full')
    # The full correlation has 2*n_points - 1 elements
    # The lags range from -(n_points-1) to (n_points-1) 
    # We want the lags from 0 to 2*n_points - 2 (length 2*n_points - 1)
    # But we only care about overlaps where the second function is shifted
    # So for shift k from 0 to n_points - 1, we get correlation[n_points - 1 + k]
    # So Lags are from 0 to n_points - 1.
    # To get all valid correlations: 
    # we're iterating over k from 0 to 2*n_points - 2
    # But we don't want negative indices
    # We compute s = k - (n_points - 1) -> 0 to 2*n_points - 2
    # The values of correlation at s are
    # s = 0 to n_points - 1: correl should be from index n_points - 1 to 2*n_points - 2  
    # s = n_points to 2*n_points - 2: correl from index 0 to n_points - 1
    
    # Actually, note that only the valid lag ranges are 0 to n_points - 1 shifts, 
    # but only considering 1st parts and 2nd parts overlapping in the middle.
    # We need to be careful of the definition of correl...

    # Double-checking: 
    # signal.correlate(h, 1-h, mode='full') gives size 2*n_points - 1 
    # at indices 0 to 2*n_points - 2.
    # For sequence h of length n_points:
    # correlation[k] is sum over i: h[i] * (1 - h[i + k - n_points + 1]) 
    #   where actual index i + k - n_points + 1 should be in [0, n_points - 1] 
    # and index i is in [0, n_points - 1]
    
    # So the lags that describe the "overlap" or
    # cross-correlation with shift k = 0 to n_points - 1 correspond to:
    # correlation[n_points - 1 + k], for k = 0 to n_points - 1  
    # The zero shift k is at index n_points - 1.
    correlations_for_positive_shifts = correlation[n_points - 1 : 2 * n_points - 1] 
    return np.max(correlations_for_positive_shifts) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    h = h / np.sum(h) * (n_points / 2)
    
    best_c5 = float('inf')
    best_h = h.copy()
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    
    # Max iterations or something - good enough for 600s
    max_iter = 2000
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute all correlations to know where the maximum is
        correlation = signal.correlate(h, 1 - h, mode='full')  
        correlations_shifts = correlation[n_points - 1 : 2 * n_points - 1] # shifts = 0 to n_points - 1
        max_correlation_idx = np.argmax(correlations_shifts)
        max_correlation_value = correlations_shifts[max_correlation_idx] * dx
        
        # Gather all correlations within epsilon of max
        epsilon = 1e-3
        close_indices = np.where(np.abs(correlations_shifts - max_correlation_value) < epsilon)[0]
        
        # This part is tricky for gradient-based approach.
        # Using a convex linear program instead: try to reduce the max, using the gradient. 
        # Step 1: Set up a new optimization problem where we try to lower current C5 value
        # let's instead do: let's take only the max correlation (and surrounding ones if it has flatness)
        
        # Here we are just solving a constrained optimization problem  
        # that tries to minimize the current max overlap, subject to Constraints
        # On feasible new h that satisfy bounds and sum constraint
        
        # Build a simple fitting procedure that minimizes max overlap, but with a simplified model 
        # Let's try coordinate descent or some efficient iterative method
        
        # Frequent substraction works because we just rearrange it coherently
        # With l ago 0 to n_points - 1 we can get 
        # correlation index n_points - 1 + l
        # so overlap integral = integral over [0,2] of h(x) * (1-h(x + l*dx))
        # Maximum for l = 0, ..., n_points - 1
        
        # Since we have focus, we'll add a constraint that pushes C5 lower. 
        # Try this: make a CVXPY problem to minimize max overlap, keeping sum (h) = 1
        # And individual h[i] within [0,1]
        
        change_direction = 0.01 * (np.random.rand(n_points) - 0.5) # Random small perturbation

        c5_new = evaluate_erdos_solution(h_new)

            best_h = h_new.copy()
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
