import numpy as np
from scipy.optimize import minimize
import time
import sys

def evaluate_c5(h):
    # h is array of length n
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum value
    return np.max(corr) * dx

def project_h(h, n_points):
    # Project h onto feasible region: 0 <= h[i] <= 1, sum(h) = n/2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

def genetic_algorithm(seed, budget_s, n_points=45):
    np.random.seed(seed)
    population_size = 50
    generations = 200
    mutation_rate = 0.95
    elite_fraction = 0.2
    tournament_size = 5
    n = n_points

    # Initialize population
    if 'initial_h_values' in globals():
        pop = [np.array(initial_h_values)]
        pop.extend([np.random.rand(n) for _ in range(population_size - 1)])
    else:
        # Symmetric starting point
        pop = [project_h(np.ones(n) * 0.5, n)]
        pop.extend([np.random.rand(n) for _ in range(population_size - 1)])
    
    best_solution = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for individual in pop:
            c5 = evaluate_c5(individual)
            fitness.append(c5)
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_solution = pop[fitness.index(min_fitness)].copy()
        
        # Selection
        elite_count = int(elite_fraction * population_size)
        elite_indices = np.argsort(fitness)[:elite_count]
        elites = [pop[i].copy() for i in elite_indices]
        
        # Create new generation
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            # Tournament selection
            tourn = np.random.choice(population_size, tournament_size, replace=False)
            winner = tourn[np.argmin([fitness[i] for i in tourn])]
            
            # Crossover with random parent
            parent2 = np.random.choice(population_size)
            alpha = 0.5
            child = alpha * pop[winner] + (1 - alpha) * pop[parent2]
            
            # Mutation
            if np.random.rand() < 0.8:  # 80% chance to mutate
                noise = np.random.normal(0, 1e-3, n)
                child += noise
                child = project_h(child, n)
            
            new_pop.append(child)
        
        pop = new_pop
        mutation_rate *= 0.95  # Decay mutation rate
        
    return best_solution, best_c5

def gradient_descent_step(h, n_points, lr=0.01):
    # Compute gradient using finite differences
    dx = 2.0 / n_points
    n = n_points
    h = h.copy()
    eps = 1e-5
    orig_c5 = evaluate_c5(h)
    
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - orig_c5) / eps
    
    # Update with gradient
    h_new = h - lr * grad
    h_new = project_h(h_new, n)
    
    return h_new

def refine_with_gradient(h, n_points, budget_s):
    start_time = time.time()
    lr = 0.01
    max_iter = 1000
    
    h_best = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.1:
            break
            
        h_new = gradient_descent_step(h_best, n_points, lr)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new
        else:
            lr *= 0.995  # Reduce learning rate on plateau
        
    return h_best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_total = time.time()
    
    n_points = 45  # Slightly lower resolution to increase speed
    if 'initial_h_values' in globals():
        # Start from the given initial construction
        h_start = np.array(initial_h_values)
    else:

    # Use population-based approach to explore
    h_ga, c5_ga = genetic_algorithm(seed, budget_s * 0.95, n_points)
    
    # Refine with gradient descent
    h_final, c5_final = refine_with_gradient(h_ga, n_points, budget_s * 0.05)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
