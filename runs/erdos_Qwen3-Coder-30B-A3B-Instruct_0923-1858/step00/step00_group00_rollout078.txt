import numpy as np
from scipy import optimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, else create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Add a simple asymmetric initialization if no parent exists
        # A basic piecewise symmetric function
        mid = n_points // 2
        for i in range(mid):
            h[i] = min(1.0, 0.5 + 0.3 * (i / (mid-1)))
        for i in range(mid, n_points):
            h[i] = min(1.0, 0.5 + 0.3 * ((n_points-1-i) / (mid-1)))
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)  # Rescale sum to n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Annealing schedule
    T = 0.1
    alpha = 0.99
    sigma = 0.1
    no_improvement_count = 0
    max_no_improvement = 1000
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Perturb h
        delta_h = np.random.normal(0, sigma, n_points)
        h_candidate = h + delta_h
        
        # Project onto feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        C5_candidate = compute_C5(h_candidate, dx)
        
        # Accept or reject
        if C5_candidate < best_C5:
            h = h_candidate.copy()
            best_C5 = C5_candidate
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Accept with probability based on temperature
            if np.random.rand() < math.exp(-(C5_candidate - best_C5) / T):
                h = h_candidate.copy()
        
        # Update temperature and sigma
        T *= alpha
        if iteration % 100 == 0:
            sigma *= 0.9
        
        # Check termination conditions
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final cleanup to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / np.sum(best_h)
    final_C5 = compute_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)

def compute_C5(h, dx):
    """Compute C5 using convolution as in the problem statement."""
    n = len(h)
    # We use full convolution
    conv_result = np.correlate(h, 1-h, mode='full')
    # Extract the relevant part (corresponding to k from -1 to 1, but actually k from 0 to 2 in shifted form)
    # Get the maximum over all valid k, but convolution gives us indices on [0, 2(n-1)]
    # The indices that correspond to meaningful k: from -1 up to 1 in terms of original problem
    # But we have to be careful with the indexing
    # The full convolution has length 2*n - 1
    # Index i corresponds to shift k = i - (n-1)
    # We want to consider all positive k, i.e., i = n-1 to 2*(n-1) corresponding to k = 0 to n-1
    # Still, we only need to consider up to 2 in normalized domain, so typically just the middle section
    # For convergence analysis, we consider all indices for correctness
    max_val = np.max(conv_result) * dx
    return max_val
