import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate C5 for a given step function h."""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def feasible_h(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def mutate(h, mutation_rate=0.1, epsilon=0.1):
    """Mutate h with small random perturbations."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, epsilon)
    return feasible_h(h, len(h))

def crossover(h1, h2, crossover_rate=0.7):
    """Blend crossover of two individuals."""
    if np.random.rand() > crossover_rate:
        return h1.copy(), h2.copy()
    
    n = len(h1)
    alpha = np.random.rand()
    h1_new = h1 * alpha + h2 * (1 - alpha)
    h2_new = h1 * (1 - alpha) + h2 * alpha
    return feasible_h(h1_new, n), feasible_h(h2_new, n)

def tournament_selection(population, fitnesses, k=5):
    """Select parent using tournament selection."""
    selected_idx = np.random.choice(len(population), size=k)
    winner_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    max_generations = 1000
    population_size = 30
    elite_size = int(0.1 * population_size)
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = feasible_h(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        h = feasible_h(h, n_points)
        population.append(h)
    
    # Fill remaining population
    for i in range(population_size - 1):
        h = np.random.rand(n_points)
        h = feasible_h(h, n_points)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Time tracking
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute fitness for each individual in population
        fitnesses = []
        for h in population:
            c5 = evaluate(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        elites = population[:elite_size]
        
        # Generate new population through selection, crossover, and mutation
        new_population = elites[:]
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    if best_h is None:
        # Fallback in case of early termination
        fitnesses = [evaluate(h, dx) for h in population]
        best_idx = np.argmin(fitnesses)
        best_h = population[best_idx].copy()
        best_c5 = fitnesses[best_idx]
    
    return (list(best_h), best_c5, n_points)
