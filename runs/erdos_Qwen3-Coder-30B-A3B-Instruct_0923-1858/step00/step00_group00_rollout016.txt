import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, else create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    current_h = h.copy()
    best_c5 = evaluate_erdos_solution(current_h, dx)
    start_time = time.time()
    max_iter = 10000
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    
    iter_since_improvement = 0
    max_iter_no_improvement = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        perturbation = np.random.normal(0, perturb_scale, n_points)
        candidate_h = current_h + perturbation
        
        # Project to feasible region
        candidate_h = np.clip(candidate_h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(candidate_h)
        candidate_h = candidate_h * scale_factor
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = candidate_h.copy()
            current_h = candidate_h.copy()
            iter_since_improvement = 0
        else:
            if np.random.rand() < np.exp(-(candidate_c5 - best_c5) / T):
                current_h = candidate_h.copy()
            iter_since_improvement += 1
            
        # Cool down
        T *= alpha
        if i % 100 == 0:
            perturb_scale *= 0.99
            
        if iter_since_improvement > max_iter_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_values, dx):
    h = np.array(h_values)
    # Compute correlations for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Multiply by dx to get the actual integral
    c5 = np.max(corr) * dx
    return c5
