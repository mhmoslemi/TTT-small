import time
from scipy.signal import correlate

    dx = 2.0 / n_points
    
    # Ensure valid starting point
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute current correlation
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        C5_current = corr[k_max] * dx
        
        # Determine which regions contribute most to max cross-correlation
        # For full mode, the indices range from 0 to 2*n_points-2
        # The exact max shift we need to target
        shifts = np.arange(-n_points + 1, n_points)
        k_shift = shifts[k_max]
        
        # Identify overlapping regions that contribute to this max cross-correlation
        overlap_indices = []
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                overlap_indices.append((i, j))
        
        # Perturb by redistributing some mass
        h_new = h.copy()
        C5_new = C5_current
        
        # Pick two indices for transfer
        if overlap_indices:
            i, j = overlap_indices[np.random.randint(len(overlap_indices))]
            
            # Compute transfer amount
            max_transfer = min(h_new[i], 1 - h_new[j])
            if max_transfer > 1e-8:
                transfer = np.random.uniform(0, max_transfer)
                h_new[i] -= transfer
                h_new[j] += transfer

                h_new = h_new / np.sum(h_new) * (n_points / 2)

                C5_new = evaluate_C5(h_new, dx)

            best_C5 = C5_new

    final_C5 = evaluate_C5(best_h, dx)
    
    return (best_h, final_C5, n_points)


def evaluate_C5(h, dx):
    # Compute cross-correlation and find maximum
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def upsample_h(h):
    # Simple linear interpolation to double resolution
    n = len(h)
    h_new = np.zeros(2 * n)
    for i in range(n):
        h_new[2 * i] = h[i]
        if i < n - 1:
            h_new[2 * i + 1] = (h[i] + h[i + 1]) / 2
    return h_new
