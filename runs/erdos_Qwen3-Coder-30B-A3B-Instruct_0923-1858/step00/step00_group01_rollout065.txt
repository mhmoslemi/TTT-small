import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h_vals, dx):
    """Evaluate the overlap integral C5 for given h values."""
    h = np.array(h_vals)
    # Compute cross-correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part and compute maximum
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(h_vals, dx):
    """Repair h to satisfy constraints: 0 <= h <= 1 and sum(h)*dx = 1."""
    h = np.array(h_vals)
    # Clip values
    h = np.clip(h, 0, 1)
    # Rescale to sum to 1/dx
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return np.clip(h, 0, 1)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize the population with feasible individuals."""
    population = []
    if initial_h is not None:
        population.append(repair_h(initial_h, 2.0 / n_points))
    
    # Generate others randomly
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, 2.0 / n_points)
        population.append(h)
    
    return population

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point1 = random.randint(1, n // 2)
    point2 = random.randint(point1, n - 1)
    
    child1 = parent1.copy()
    child2 = parent2.copy()
    
    child1[point1:point2] = parent2[point1:point2]
    child2[point1:point2] = parent1[point1:point2]
    
    return child1, child2

def mutate(h, mutation_rate=0.05, strength=0.1):
    """Mutate an individual."""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += random.gauss(0, strength)
    return h

def selection(population, fitnesses, elite_ratio=0.2):
    """Select top individuals based on fitness."""
    sorted_indices = np.argsort(fitnesses)
    n_elite = int(len(population) * elite_ratio)
    return [population[i] for i in sorted_indices[:n_elite]]

    """Evolutionary optimization of h."""
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            repaired_h = repair_h(h, dx)
            c5 = evaluate_h(repaired_h, dx)
            fitnesses.append(c5)
            
        # Update best
        min_fitness = np.min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
        
        # Select elites
        elites = selection(population, fitnesses, elite_ratio=0.2)
        new_population = elites.copy()
        
        # Generate children
        while len(new_population) < pop_size:
            # Select two parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutate
            child1 = mutate(child1, mutation_rate=0.05, strength=0.05)
            child2 = mutate(child2, mutation_rate=0.05, strength=0.05)
            
            # Repair
            child1 = repair_h(child1, dx)
            child2 = repair_h(child2, dx)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
    
    return best_h, best_c5
