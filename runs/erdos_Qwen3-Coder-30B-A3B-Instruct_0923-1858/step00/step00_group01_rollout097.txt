import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 47
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct initial sparse ruler
        m = floor(np.sqrt(n_points)) + 1
        base_positions = [i for i in range(n_points) if i % m == 0]
        offset_positions = []
        for k in range(1, m):
            for pos in base_positions:
                offset_positions.append((pos + k) % n_points)
        all_positions = list(set(base_positions + offset_positions))
        h = np.zeros(n_points)
        s = (n_points / 2) / len(all_positions)
        for i in all_positions:
            h[i] = s
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_conv = np.max(conv)
        return max_conv * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Gradient descent optimization
    max_iter = 5000
    lr = 0.005
    patience = 100
    patience_counter = 0
    best_loss = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current gradient numerically
        eps = 1e-4
        loss_current = objective(h)
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            loss_plus = objective(h_plus)
            grad[j] = (loss_plus - loss_current) / eps
            
        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        loss_new = objective(h_new)
        
        if loss_new < best_loss:
            best_loss = loss_new
            best_h = h_new.copy()
            best_C5 = loss_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        h = h_new
        
        if patience_counter >= patience:
            lr *= 0.9
            patience_counter = 0
    
    # Final refinement
    def refined_objective(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_conv = np.max(conv)
        return max_conv * dx
    
    # Use scipy optimization to refine
    result = minimize(
        refined_objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
    )
    
    if result.success:
        best_h = result.x
        best_C5 = refined_objective(best_h)
    
    # Ensure constraint is met
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), float(best_C5), n_points)
