import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    n = len(h)
    dx = 2.0 / n
    # Correlation of h with (1-h) with full mode
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral for each k
    overlap = conv * dx
    # C5 is the maximum of these values
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if not (np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) * dx - 1.0) < 1e-10):
            h = np.random.uniform(0, 1, n_points)
            h *= (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h *= (n_points / 2) / np.sum(h)
    
    # Functional to minimize
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    # Equality constraint: sum(h) * dx = 1, i.e. sum(h) = n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for h: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for scipy.optimize
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Run optimization
    try:
        # Use SLSQP method which handles constraints well
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        
        if result.success:
            h_opt = result.x
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
    except Exception as e:
        pass  # continue with best found so far
    
    # Final evaluation with updated best
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
