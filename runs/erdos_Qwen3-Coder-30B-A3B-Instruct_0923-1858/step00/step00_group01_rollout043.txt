import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate C5 for given h function."""
    dx = 2.0 / len(h)
    # Compute the overlap integral
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h = h * scale
    return np.clip(h, 0, 1)

def crossover(parent1, parent2, n_points):
    """Create a child through crossover of two parents."""
    # Randomly select crossover points
    n = len(parent1)
    p1 = np.random.randint(1, n)
    p2 = np.random.randint(1, n)
    start, end = min(p1, p2), max(p1, p2)
    
    child = parent1.copy()
    child[start:end] = parent2[start:end]
    
    # Repair to maintain constraints
    child = repair_h(child, n_points)
    return child

def mutate(h, mutation_rate=0.05):
    """Mutate h with given rate."""
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    if np.any(mask):
        # Add noise to mutated positions
        noise = np.random.normal(0, 0.05, n)
        h = h + noise * mask
        # Repair
        h = repair_h(h, len(h))
    return h

def create_initial_population(n_points, pop_size, initial_h=None):
    """Create initial population."""
    population = []
    if initial_h is not None:
        population.append(repair_h(np.array(initial_h), n_points))

    for _ in range(pop_size):
        h = np.random.random(n_points)
        h =repair_h(h, n_points)
        population.append(h)
    
    return population
