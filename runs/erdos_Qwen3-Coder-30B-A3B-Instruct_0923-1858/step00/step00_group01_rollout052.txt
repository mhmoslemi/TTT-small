def evaluate_erdos_solution(h_values):

    n = len(h_values)

    corr = correlate(h_values, 1 - h_values, mode='full')
    # Get the maximum value (over all shifts k) and scale by dx
    max_corr = np.max(corr)
    return max_corr * dx
