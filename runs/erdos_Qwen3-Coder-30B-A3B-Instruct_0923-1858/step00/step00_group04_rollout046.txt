import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    """Evaluate overlap integral for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, n_points):
    """Ensure h is within [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover"""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mut_rate=0.1, noise_std=0.01):
    """Gaussian mutation"""
    mask = np.random.random(len(h)) < mut_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return h

def tournament_select(population, fitnesses, k=5):
    """Tournament selection of size k"""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 30
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0)
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Fill rest of population with random feasible candidates
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Track best
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Generational GA
    gen = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < 200:
        gen += 1
        
        # Evaluate fitness
        fitnesses = [evaluate(h, dx) for h in population]
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            last_improvement = gen
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Track intermediate results every 50 gens
        if gen % 50 == 0:
            pass  # No action needed unless special tracking desired
        
        # Selection
        new_population = []
        
        # Elite
        elite_count = max(1, pop_size // 10)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            p1 = tournament_select(population, fitnesses)
            p2 = tournament_select(population, fitnesses)
            
            if np.random.random() < 0.7:
                child = crossover(p1, p2)
            else:
                child = p1.copy()
            
            child = mutate(child)
            child = project_to_feasible(child, n_points)
            new_population.append(child)
        
        population = new_population
    
    # Final validation
    if best_h is not None:
        best_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), best_c5, n_points)
    else:
        h0 = population[0]
        c5 = evaluate(h0, dx)
        return (h0.tolist(), c5, n_points)
