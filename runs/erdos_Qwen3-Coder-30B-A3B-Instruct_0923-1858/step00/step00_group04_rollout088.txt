import numpy as np
from scipy import signal
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1 - h)
    correlation = signal.correlate(h, 1 - h, mode='full')
    # Find the maximum value and scale by dx
    c5 = np.max(correlation) * dx
    return c5

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize a population of step functions."""
    population = []
    if initial_h is not None:
        population.append(initial_h)
    # Fill the rest with random feasible solutions
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        # Scale to meet the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select an individual via tournament selection."""
    indices = random.sample(range(len(population)), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Perform blend crossover between two parents."""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child
