import numpy as np
from scipy import signal
import time
import math

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value of cross-correlation is the C5 constant
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to target sum
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h *= target_sum / current_sum
    return h

def gradient_descent_step(h, n_points, learning_rate):
    """Perform one gradient descent step on h"""
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum correlation and its index
    max_idx = np.argmax(corr)
    max_corr = corr[max_idx]
    # Compute gradient w.r.t. h
    grad_h = np.zeros_like(h)
    # Gradient of h[i] w.r.t. correlation
    # We have that (h ⋆ (1-h))[k] = sum(h[i] * (1-h[i+k]))
    # So partial derivative of corr[k] w.r.t. h[i] is (1-h[i+k]) or (1-h[i-k]) depending on sign
    # We only care about the maximum
    # For simplicity of implementation, we can use finite differences to estimate gradient
    # But a more accurate approach:
    # if k = i - j, then contribution of h[i] to corr[k] is (1-h[j])
    # hence, derivative w.r.t. h[i] is relevant when k = i - j for varying j
    # actually, simpler approach: compute gradient numerically for the max element
    # compute the derivative which maximizes the objective
    # for max_idx, assume it comes from i and j such that i - j = max_idx - (len-1)
    # i.e., j = i - (max_idx - (len-1))
    # so for a fixed i, the derivative of the corr with i-k is non-zero when k = i-j
    # That is: derivative of (h ⋆ (1-h))[i-j] w.r.t. h[i] = (1-h[i-j])
    # Let's compute in a more algorithmic way
    
    # Find the shift k that achieves the max
    shift = max_idx - (n_points - 1)
    # Don't do slow loop - just use direct formula
    # At shift k, the derivative w.r.t. h[i] is (1 - h[i-k]) if i-k >= 0 and i-k < n
    # Let's compute the full derivative vector:
    # To compute accurate gradient for max value of cross-correlation
    # If max_idx is the index of maximum in correlation, consider:
    # corr[max_idx] = sum_{i} h[i] * (1 - h[i-shift])
    # The gradient of this w.r.t. h[i] is (1 - h[i-shift]) 
    # But the max might be pushed in different directions
    # Instead, we work with the entire gradient
    # Each increase in h[i] causes increase of correlation of sum_{k} h[i+k](1-h[k]) when
    # k = shift = i - (max_idx - (n_points-1)) <=> i = k + shift
    # So, the derivative with respect to each h[i]
    # To avoid computational complexity, we take a different step:
    # We compute the gradient direction as: 
    # When we increase h[i], we increase correlation by sum of (1-h[i+k]) over all k
    # Except for the actual shift k corresponding to maximum, which needs to be treated.
    # We'll do this as an approximation:
    # Approximate gradient at each point based on its influence
    grad = np.zeros_like(h)
    for i in range(n_points):
        # When h[i] increases by 1, how does the correlation change?
        # Add (1-h[i-shift]) if shift is non-negative and within bounds
        # Also add term from the reverse side
        # This derivative is not exact analytic but is good for optimization
        # At correlation max_index, the gradient in that direction explains it
        # so we approximate: grad[i] = 1 - h[i-shift] 
        # scaler correction based on position in correlation
        # Use more accurate finite diff
        idx = i - shift
        if 0 <= idx < n_points:
            grad[i] = 1 - h[idx]
    # Project gradient into feasible direction
    # Step decreases objective
    # Gradient step: h_new = h - lr * grad
    # Project it back
    h_new = h - learning_rate * grad
    h_new = project_h(h_new, n_points)
    # residual gradient magnitude
    return h_new, np.linalg.norm(grad)
