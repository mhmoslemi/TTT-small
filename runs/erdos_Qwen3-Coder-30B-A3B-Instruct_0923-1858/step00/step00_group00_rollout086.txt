import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Construct symmetric starting point
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        # Make some local adjustments
        h0[:n_points//3] = 0.8
        h0[2*n_points//3:] = 0.2
    
    # Make sure sum(h) * dx = 1
    dx = 2.0 / n_points
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = float('inf')
    
    # Start time tracking
    start_time = time.time()
    
    # Binary search on C_5
    c_min, c_max = 0.0, 0.5
    tolerance = 1e-4
    
    # Store the largest feasible C that still works
    feasible_c = 0.0
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        c_candidate = (c_min + c_max) / 2
        
        # Formulate and solve optimization problem
        h = Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
        
        # Bilinear constraints for overlap integral
        for k in range(n_points):
            # Create a larger vector for convolution
            shift_idx = min(k, n_points - 1)
            # Construct an approximate bilinear constraint
            if k < n_points:
                # We use a relaxed model here with a coarse approximation
                # For performance, we simplify: just ensure max convolution is bounded
                conv_k = 0
                for i in range(n_points):
                    if i + k < n_points:
                        conv_k += h[i] * (1 - h[i+k]) * dx
                constraints.append(conv_k <= c_candidate)

        # Objective (minimize C is implicit in the feasibility check)
        prob = Problem(Minimize(0), constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=100)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_val = np.array(h.value).flatten()
                if np.all(h_val >= 0) and np.all(h_val <= 1):
                    # Check if this is really feasible by computing result directly
                    h_copy = h_val.copy()
                    corr = np.correlate(h_copy, 1 - h_copy, mode="full")
                    c5_computed = np.max(corr) * dx
                    if c5_computed <= c_candidate:
                        feasible_c = c_candidate
                        best_h = h_copy.copy()
                        best_c5 = c5_computed
                        c_max = c_candidate
                    else:
                        c_min = c_candidate
                else:
                    c_min = c_candidate
            else:
                c_min = c_candidate
        except:
            # Problem could not be solved; try a hand-crafted optimization
            # At least try refining h
            def eval_h(h):
                corr = np.correlate(h, 1 - h, mode="full")
                return np.max(corr) * dx
            
            def obj(h):
                return eval_h(h)
                
            def constraint_eq(h):
                return np.sum(h) - n_points / 2
                
            def constraint_ineq(h):
                corr = np.correlate(h, 1 - h, mode="full")
                return c_candidate - np.max(corr) * dx
                
            # Optimize using scipy
            bounds = [(0, 1)] * n_points
            cons = [
                {'type': 'eq', 'fun': constraint_eq},
                {'type': 'ineq', 'fun': constraint_ineq}
            ]
            try:
                res = minimize(
                    obj,
                    h0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-6}
                )
                if res.success:
                    h_val = res.x
                    c_val = eval_h(h_val)
                    if c_val <= c_candidate:
                        feasible_c = c_candidate
                        best_h = h_val.copy()
                        best_c5 = c_val
                        c_max = c_candidate
                    else:
                        c_min = c_candidate
                else:
                    c_min = c_candidate
            except:
                c_min = c_candidate
    
    # Final check using most recent solution
    if np.sum(best_h) * dx - 1 > 1e-6:
        # Project onto feasible set
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    # Validate bounds
    eval_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), eval_c5, n_points)
