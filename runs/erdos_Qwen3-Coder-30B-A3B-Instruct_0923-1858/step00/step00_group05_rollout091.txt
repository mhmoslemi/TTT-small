```python
import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_overlap(h, dx):
    """Evaluate the overlap integral for given h and dx"""
    # h is a step function, we want max_k int h(x)(1-h(x+k)) dx
    # This is equivalent to max_k (h * (1-h))[k] * dx
    # Using numpy's correlate with mode='full' gives us all lags
    conv = np.correlate(h, 1 - h, mode='full') * dx
    # For a function defined on [0,2] with n samples,
    # the cross-correlation has length 2*n-1
    # The maximum overlap is at the center
    max_k = len(conv) // 2
    return np.max(conv)

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0<=h<=1, sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    # Ensure sum is exactly n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_fitness(h, dx):
    """Evaluate the fitness = C5 value of given h"""
    return evaluate_overlap(h, dx)

def local_optimize(h, dx, maxiter=50):
    """Use scipy.optimize to refine h"""
    n_points = len(h)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective to minimize, and constraint for sum
    def objective(h_vec):
        return evaluate_overlap(h_vec, dx)
        
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Use SLSQP, which can handle bounds and constraints
    result = minimize(
        objective, 
        h, 
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': maxiter, 'ftol': 1e-8, 'gtol': 1e-8},
        tol=1e-8
    )
    
    # If optimization failed or we get a better solution, return the optimized one
    if result.success:
        h_opt = result.x
        return project_to_feasible(h_opt, n_points)
    else:
        return project_to_feasible(h, n_points)

def create_initial_population(pop_size, n_points):
    """Create initial population"""
    population = []
    for _ in range(pop_size):
        # Use the existing solution if available, otherwise create something reasonable
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
        else:
            # Start with uniform distribution
            h = np.full(n_points, 0.5)
        # Slightly randomize it
        h += np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=10):
    """Select an individual using tournament selection"""
    # Select random individuals for tournament
    selected_indices = np.random.choice(len(population), tournament_size)
    selected_fitness = [fitnesses[i] for i in selected_indices]
    # Return index of best in tournament
    best_idx = selected_indices[np.argmin(selected_fitness)]
    return population[best_idx]

def crossover(parent1, parent2):
    """Single-point crossover"""
    n_points = len(parent1)
    if n_points <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
    child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
    return child1, child2

def mutate(h, mutation_rate=0.1):
    """Add noise with probability mutation_rate"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.2)
    return project_to_feasible(h, len(h))

def run(seed=42, budget_s=600):
    """Main function to run the optimization"""
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Get initial population
    pop_size = 100
    population = create_initial_population(pop_size, n_points)
    
    best_h = None
    best_c5 = float('inf')
    
    # GA loop
    generations = 50
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness of each individual
        fitnesses = []
        for i, h in enumerate(population):
            fit = evaluate_fitness(h, dx)
            fitnesses.append(fit)
            if fit < best_c5:
                best_c5 = fit
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Select parents, perform crossover and mutation, generate new population
        new_population = []
        # Elitism: keep top 10%
        elite_size = pop_size // 10
        new_population.extend(population[:elite_size])
        
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        population = new_population[:pop_size]
    
    # Local optimization of best individual
    if best_h is not None:
        # Do some local refinement
        refined_h = local_optimize(best_h, dx)
        refined_c5 = evaluate_fitness(refined_h, dx)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            
    # Final check of the best solution
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_fitness(final_h, dx)
    
    # Ensure the solution satisfies the constraint exactly
    assert abs(np.sum(final_h) - n_points / 2) < 1e-10, "Sum constraint violated!"
    
    return (final_h.tolist(), final_c5, n_points)
```
