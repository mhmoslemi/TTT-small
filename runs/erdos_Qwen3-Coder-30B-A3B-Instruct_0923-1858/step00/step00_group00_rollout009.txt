import numpy as np
from scipy import signal
import time

def evaluate_h(h):
    """Evaluate the overlap integral C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode="full")
    # The maximum value scaled by dx
    c5 = np.max(conv) * dx
    return c5

def project_h(h, n_points):
    """Project h onto constraints: h[i] in [0,1], sum(h) = n/2."""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    scale = (n_points / 2.0) / np.sum(h)
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default parameters
    max_iter = 5000
    n_points = 45
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        # Create symmetric initial h
        initial_h = np.ones(n_points) * 0.5
        # Create piecewise constant step function
        mid = n_points // 2
        initial_h[:mid] = 0
        initial_h[mid:] = 1.0

    # Initialize best solution  
    best_h = initial_h.copy()
    best_c5 = evaluate_h(best_h)
    
    # Nelder-Mead parameters
    alpha = 1.0  # Reflection
    gamma = 2.0  # Expansion
    beta = 0.5   # Contraction
    rho = 0.5    # Shrink
    
    # Adaptive simplex parameters
    init_size = 0.1
    min_size = 0.01
    reset_threshold = 50
    max_reset_attempts = 10
    
    # Begin optimization loop
    start_time = time.time()
    iteration = 0
    reset_count = 0
    last_improvement = 0
    
    simplex_size = init_size
    h = project_h(initial_h, n_points)
    
    # Create initial simplex
    simplex = np.tile(h, (n_points + 1, 1))
    for i in range(n_points + 1):
        if i > 0:
            # Perturb along a random direction
            direction = np.random.normal(0, simplex_size, n_points)
            simplex[i] = h + direction
    
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        # Project each vertex to constraints
        for i in range(n_points + 1):
            simplex[i] = project_h(simplex[i], n_points)
        
        # Evaluate objective for all vertices
        obj_values = np.array([evaluate_h(simplex[i]) for i in range(n_points + 1)])
        
        # Sort vertices by objective (ascending order)
        idxs = np.argsort(obj_values)
        simplex = simplex[idxs]
        obj_values = obj_values[idxs]
        
        # Update best if improved
        if obj_values[0] < best_c5:
            best_c5 = obj_values[0]
            best_h = simplex[0].copy()
            last_improvement = iteration
            
        # If no improvement in a while, reset simplex
        if iteration - last_improvement > reset_threshold:
            if reset_count < max_reset_attempts and simplex_size > min_size:
                simplex_size *= 0.5
                reset_count += 1
                # Reinitialize simplex around best vertex
                simplex = np.tile(best_h, (n_points + 1, 1))
                for i in range(n_points + 1):
                    if i > 0:
                        direction = np.random.normal(0, simplex_size, n_points)
                        simplex[i] = best_h + direction
                last_improvement = iteration
            else:
                break
        
        # Compute centroid (excluding worst vertex)
        centroid = np.mean(simplex[:-1], axis=0)
 
        # Reflect worst vertex
        worst = simplex[-1]
        h_reflect = centroid + alpha * (centroid - worst)
        h_reflect = project_h(h_reflect, n_points)
        c_reflect = evaluate_h(h_reflect)
        
        if obj_values[0] <= c_reflect < obj_values[-2]:
            # Accept reflection
            simplex[-1] = h_reflect
        elif c_reflect < obj_values[0]:
            # Try expansion
            h_expand = centroid + gamma * (h_reflect - centroid)
            h_expand = project_h(h_expand, n_points)
            c_expand = evaluate_h(h_expand)
            if c_expand < c_reflect:
                simplex[-1] = h_expand
            else:
                simplex[-1] = h_reflect
        elif c_reflect >= obj_values[-2]:
            # Try contraction
            h_contract = centroid + beta * (worst - centroid)
            h_contract = project_h(h_contract, n_points)
            c_contract = evaluate_h(h_contract)
            if c_contract <= obj_values[-1]:
                simplex[-1] = h_contract
            else:
                # Shrink toward best vertex
                for i in range(1, n_points + 1):
                    simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
        
        iteration += 1

    # Final evaluation
    final_c5 = evaluate_h(best_h)
    return (best_h.tolist(), final_c5, n_points)
