import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Compute C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximum value scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto feasible set: 0<=h<=1, sum(h)=n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to have correct sum
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, n_points):
    """Approximate gradient of C5 w.r.t. h using finite differences"""
    # This is a simplified gradient approximation
    # We compute direction of steepest ascent by sampling small perturbations
    dx = 2.0 / n_points
    # Correlation of h with 1-h
    corr_h = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr_h)
    
    # Arange indices for full correlation
    indices = np.arange(-n_points+1, n_points)
    # Compute overlap for current h, then perturb h[i] by 1e-3 
    # to estimate gradient direction
    grad = np.zeros_like(h)
    eps = 1e-6
    
    for i in range(n_points):
        h_pert = h.copy()
        h_pert[i] += eps
        h_pert = project_h(h_pert, n_points)
        corr_pert = correlate(h_pert, 1-h_pert, mode='full')
        grad[i] = (corr_pert[max_idx] - corr_h[max_idx]) / eps
    
    return grad

        h += noise

    learning_rate = 1e-2

        new_c5 = evaluate_h(h_new)
