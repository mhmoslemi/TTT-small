import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time


def evaluate_h(h, dx):
    conv = correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val


def project_onto_constraints(h, n_points):
    # Project h onto [0,1] constraints
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

    if initial_h is not None and len(initial_h) == n_points:
        return np.array(initial_h)
    else:
        return np.ones(n_points) * 0.5
