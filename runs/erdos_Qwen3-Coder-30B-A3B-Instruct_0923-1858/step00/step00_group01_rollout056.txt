import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h_values):
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum value corresponds to C5
    return np.max(corr) * 2.0 / len(h_values)

def project_h(h, dx):
    # Project h to satisfy the integral constraint
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0  # scale to make integral = 1
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    return h
