import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 bound for the given step function h."""
    n = len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum correlation value determines C5
    max_corr = np.max(corr)
    # Scale by dx = 2/n
    dx = 2.0 / n
    return max_corr * dx

def create_feasible_h(n_points):
    """Create a feasible h vector with sum = n_points/2 and values in [0,1]."""
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    return np.clip(h, 0, 1)

def mutate_h(h, mutation_rate=0.1, epsilon=0.1):
    """Mutate the h vector."""
    h_new = h.copy()
    n = len(h)
    # Apply small random perturbations
    for i in range(n):
        if random.random() < mutation_rate:
            delta = random.uniform(-epsilon, epsilon)
            h_new[i] += delta
    # Ensure sum constraint
    h_new = h_new * (n / 2) / np.sum(h_new)
    return np.clip(h_new, 0, 1)

def crossover_h(h1, h2):
    """Blend crossover of two h vectors."""
    alpha = random.random()
    h_new = alpha * h1 + (1 - alpha) * h2
    return np.clip(h_new, 0, 1)

def tournament_selection(population, fitnesses, k=5):
    """Select one parent using tournament selection."""
    n = len(population)
    candidates = random.sample(range(n), min(k, n))
    best_idx = candidates[np.argmin([fitnesses[i] for i in candidates])]
    return population[best_idx]

    population_size = 30
    generations = 1000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_rate = 0.1
    elite_count = int(elitism_rate * population_size)
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_feasible_h(n_points))
    
    # Evaluation buffer
    fitnesses = [evaluate_c5(h) for h in population]
    global_best = min(fitnesses)
    global_best_h = population[np.argmin(fitnesses)].copy()
    
    start_time = time.time()
    gen = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 200:
        # Evaluate population
        fitnesses = [evaluate_c5(h) for h in population]
        
        # Update global best
        min_fitness = min(fitnesses)
        if min_fitness < global_best:
            global_best = min_fitness
            global_best_h = population[np.argmin(fitnesses)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        gen += 1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_population = sorted_population[:elite_count]
        
        # Fill rest with offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(sorted_population, sorted_fitnesses)
            parent2 = tournament_selection(sorted_population, sorted_fitnesses)
            
            if random.random() < crossover_prob:
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
            
            child = mutate_h(child, mutation_rate)
            new_population.append(child)
        
        population = new_population[:population_size]
    
    final_c5 = evaluate_c5(global_best_h)
    return (global_best_h.tolist(), final_c5, len(global_best_h))
