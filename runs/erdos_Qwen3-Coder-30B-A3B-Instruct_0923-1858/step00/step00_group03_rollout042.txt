import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute overlaps for all shifts
    overlap = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize the sum to be exactly 1 (i.e., n_points/2)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-6:
        # Adjust by making the rightmost 0s become 1s or vice versa
        diff = int(target_sum - current_sum)
        if diff > 0:
            # Need to make some 0s to 1s
            indices = np.argsort(h)[:diff]
            h[indices] = 1
        else:
            # Need to make some 1s to 0s
            indices = np.argsort(h)[diff:]
            h[indices] = 0
    
    # Convert to binary (grid search will optimize to binary)
    h = (h > 0.5).astype(int)
    
    total_time = 0
    best_h = h.copy()
    best_C5 = evaluate_h(h, dx)
    
    start_time = time.time()
    
    for iteration in range(10000):
        if total_time > budget_s:
            break
            
        # Recompute current maximum overlap and shift
        overlap = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        if max_overlap <= best_C5:
            # No improvement, check if we are in a plateau
            pass
        
        # Start with a flip to improve:
        max_shift = max_idx - (n_points - 1)  # Convert from full correlation index to shift
        
        # Compute how much flipping each position changes overlap at max_shift
        deltas = np.zeros(n_points)
        
        # For each position, compute the effect of flipping it
        for i in range(n_points):
            # Compute the change in overlap at max_shift if we flip h[i]
            # Only the correlation terms that depend on h[i] and h[i+max_shift] 
            # and h[i-max_shift] (if relevant) change.
            # Neighboring contributions to this specific correlation term are:
            if 0 <= i - max_shift < n_points:
                # From negative shift correlation (h[i], h[i-max_shift])
                deltas[i] += h[i] - (1 - h[i])  # Change in contribution
            if 0 <= i + max_shift < n_points:
                # From positive shift correlation (h[i], h[i+max_shift])
                deltas[i] += h[i] - (1 - h[i])  # Change in contribution
            
            # Actually, we want to properly compute the correct overlaps
            # Instead, correct approach: determine change in overlap
            # when we change h[i] from 1 to 0, or 0 to 1
            temp_h = h.copy()
            temp_h[i] = 1 - temp_h[i]
            temp_overlap = correlate(temp_h, 1 - temp_h, mode='full')
            temp_max_overlap = temp_overlap[max_idx] * dx
            deltas[i] = temp_max_overlap - max_overlap
            
        # Find the best flip
        best_delta = 0
        flip_i = -1
        flip_val = 0
        flip_j = -1
        
        one_indices = np.where(h == 1)[0]
        zero_indices = np.where(h == 0)[0]
        
        # Try flipping each 1 -> 0 and find some other 0 -> 1 to keep count
        for i in one_indices:
            delta = deltas[i]
            if delta < best_delta:
                best_delta = delta
                flip_i = i
                flip_val = 0  # flip from 1 to 0
                # now we must find a 0 to flip to 1 to keep count
                for j in zero_indices:
                    if j != flip_i:
                        new_h = h.copy()
                        new_h[i] = 0
                        new_h[j] = 1
                        new_overlap = correlate(new_h, 1 - new_h, mode='full')
                        new_max_overlap = new_overlap[max_idx] * dx
                        actual_delta = new_max_overlap - max_overlap
                        if actual_delta == delta:
                            flip_j = j
                            break
        
        if flip_i != -1 and flip_j != -1:
            # Apply that flip
            h[flip_i] = 0
            h[flip_j] = 1
            current_C5 = evaluate_h(h, dx)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        total_time = time.time() - start_time
        
        # Periodically reset from a random restart to explore more
        if iteration % 500 == 0 and iteration > 0:
            h = best_h.copy()
            # Slightly randomize it to escape local optima
            for _ in range(5):
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points)
                h[idx1], h[idx2] = h[idx2], h[idx1]
            h = (h > 0.5).astype(int)

    return (best_h.tolist(), best_C5, n_points)
