import numpy as np
import time
from scipy import signal

def evaluate(h, n_points):
    """Evaluate C5 for given h"""
    dx = 2.0 / n_points
    # Compute full convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Find maximum correlation
    max_corr = np.max(conv)
    # Scale by dx to get C5
    c5 = max_corr * dx
    return c5

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with feasible solutions"""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        pop_size -= 1
    
    # Fill remaining population with random feasible solutions
    for _ in range(pop_size):
        h = np.random.uniform(0, 1, n_points)
        # Scale to meet constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        # Ensure bounds
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def mutation(h, mutation_rate=0.1, epsilon=0.05):
    """Apply small random perturbations to h"""
    h_new = h.copy()
    n_points = len(h)
    
    # Select indices to mutate
    n_mutate = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    
    for idx in indices:
        # Add small random change
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[idx] += delta
        # Clamp to [0,1]
        h_new[idx] = np.clip(h_new[idx], 0, 1)
    
    # Adjust to meet sum constraint
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    return h_new

def crossover(parent1, parent2, crossover_prob=0.7):
    """Blend crossover between two parents"""
    if np.random.random() > crossover_prob:
        return parent1.copy()
    
    n_points = len(parent1)
    alpha = np.random.random()
    
    # Blend: h = alpha * parent1 + (1 - alpha) * parent2
    h = alpha * parent1 + (1 - alpha) * parent2
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select a parent via tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def genetic_algorithm(n_points, pop_size=30, generations=1000, 
                      mutation_rate=0.1, initial_h=None, budget_s=600):
    """Main genetic algorithm"""
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    # Evaluate initial population
    fitnesses = [evaluate(h, n_points) for h in population]
    
    # Track global best
    best_idx = np.argmin(fitnesses)
    global_best = population[best_idx].copy()
    global_best_fitness = fitnesses[best_idx]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection, crossover, mutation
        new_population = [global_best.copy()]  # Elitism: keep best
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutation(offspring, mutation_rate)
            
            new_population.append(offspring)
        
        # Replace population
        population = new_population
        
        # Evaluate new population
        fitnesses = [evaluate(h, n_points) for h in population]
        
        # Update global best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < global_best_fitness:
            global_best = population[min_fitness_idx].copy()
            global_best_fitness = fitnesses[min_fitness_idx]
    
    return global_best, global_best_fitness

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run optimizer"""
    np.random.seed(seed)
    
    # Given information
    n_points = 42 if 'initial_h_values' in globals() else 100
    
    # Get initial construction if available
    initial_h = globals().get('initial_h_values', None) if n_points == 42 else None
    
    # Run search
    try:
        h_opt, c5_bound = genetic_algorithm(
            n_points=n_points,
            pop_size=30,
            generations=1000,
            mutation_rate=0.15,
            initial_h=initial_h,
            budget_s=budget_s
        )
    except Exception:
        # Fallback: just return the initial construction if something goes wrong
        h_opt = initial_h if initial_h is not None else np.full(n_points, 0.5) 
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5_bound = evaluate(h_opt, n_points)
    
    return (h_opt, c5_bound, n_points)
