def evaluate_c5(h):
    """Evaluate the C5 constant for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value in the correlation is the C5 bound
    c5 = np.max(corr) * dx
    return c5

def fitness(h):
    """Fitness function to minimize (negative of C5)."""
    return -evaluate_c5(h)

def mutate_h(h, mutation_rate=0.1):
    """Mutate h with given rate."""
    n = len(h)
    mask = np.random.rand(n) < mutation_rate
    if np.any(mask):
        h[mask] = np.random.rand(np.sum(mask)) * 2  # Random values in [0,2]
        # Ensure values are within [0,1] and sum constraint is still approximately met
        h = np.clip(h, 0, 1)
        # Normalize to keep correct sum
        h = h / np.sum(h) * (n / 2)
    return h

def crossover_h(h1, h2):
    """Perform crossover between two h vectors."""
    n = len(h1)
    point = np.random.randint(1, n-1)
    h_new = np.concatenate([h1[:point], h2[point:]])
    # Renormalize to meet sum constraint
    h_new = h_new / np.sum(h_new) * (n / 2)
    return h_new

def genetic_algorithm(n_points=70, population_size=100, generations=50, 
                      mutation_rate=0.1, crossover_rate=0.8, 
                      time_budget=600):
    """Perform a genetic algorithm to find a good h."""
    start_time = time.time()
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h = create_initial_h()
        h = mutate_h(h, mutation_rate)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > time_budget - 5:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        sorted_indices = np.argsort(fitnesses)
        
        # Track best
        if -fitnesses[sorted_indices[0]] < best_c5:
            best_c5 = -fitnesses[sorted_indices[0]]
            best_h = population[sorted_indices[0]].copy()
        
        # Selection (tournament)
        new_population = []
        tournament_size = 5 if gen < generations/2 else 3  # Decrease with generations
        
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                h1 = new_population[i].copy()
                h2 = new_population[i+1].copy()
                offspring1 = crossover_h(h1, h2)
                offspring2 = crossover_h(h2, h1)
                new_population[i] = offspring1
                new_population[i+1] = offspring2
            else:
                new_population[i] = new_population[i].copy()
                new_population[i+1] = new_population[i+1].copy()
            
            # Mutate
            new_population[i] = mutate_h(new_population[i], mutation_rate)
            new_population[i+1] = mutate_h(new_population[i+1], mutation_rate)
        
        population = new_population
    
    return best_h, best_c5
