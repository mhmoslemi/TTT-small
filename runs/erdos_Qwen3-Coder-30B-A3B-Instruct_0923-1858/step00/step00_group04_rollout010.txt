import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """Evaluate the overlap integral C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation h ⋆ (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum correlation and scale by dx
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def create_initial_population(n_points, population_size, initial_h=None):
    """Create initial population of feasible solutions."""
    population = []
    
    if initial_h is not None:
        # Use provided initial solution
        population.append(initial_h.copy())
    
    # Fill the rest with random feasible solutions
    for _ in range(population_size - len(population)):
        # Generate random values in [0,1]
        h = np.random.random(n_points)
        # Scale to ensure sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
    return population

def mutate(h, mutation_rate=0.1, epsilon=0.1):
    """Mutate a solution by adding small random perturbations."""
    h_new = h.copy()
    n = len(h)
    
    # Decide which indices to mutate
    indices_to_mutate = np.random.choice(n, size=int(n * mutation_rate), replace=False)
    
    for i in indices_to_mutate:
        # Add small random perturbation
        perturbation = np.random.uniform(-epsilon, epsilon)
        h_new[i] += perturbation
        # Clip to [0,1]
        h_new[i] = np.clip(h_new[i], 0, 1)
        
    # Re-scale to meet sum constraint
    h_new = h_new * (n / 2) / np.sum(h_new)
    
    return h_new

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform blend crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy()
        
    n = len(parent1)
    child = np.zeros(n)
    
    # Blend crossover: for each position, take a weighted average of parent values
    alpha = np.random.random()
    for i in range(n):
        child[i] = alpha * parent1[i] + (1 - alpha) * parent2[i]
    
    # Ensure valid range
    child = np.clip(child, 0, 1)
    
    # Re-scale to meet sum constraint
    child = child * (n / 2) / np.sum(child)
    
    return child

def select_parents(population, fitnesses, tournament_size=5):
    """Select two parents using tournament selection."""
    # Select first parent
    tournament_indices = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = tournament_indices[np.argmin([fitnesses[i] for i in tournament_indices])]
    parent1 = population[best_idx]
    
    # Select second parent
    tournament_indices = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = tournament_indices[np.argmin([fitnesses[i] for i in tournament_indices])]
    parent2 = population[best_idx]
    
    return parent1, parent2

def genetic_algorithm(n_points, initial_h=None, population_size=30, generations=1000, 
                      mutation_rate=0.1, crossover_rate=0.7, elitism_rate=0.1, 
                      budget_s=600):
    """Run genetic algorithm to optimize the step function h."""
    start_time = time.time()
    
    # Initialize population
    population = create_initial_population(n_points, population_size, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        n_elite = int(elitism_rate * population_size)
        elite = population[:n_elite]
        
        # Generate new population
        new_population = elite[:]
        
        # Create offspring
        while len(new_population) < population_size:
            parent1, parent2 = select_parents(population, fitnesses)
            child = crossover(parent1, parent2, crossover_rate)
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = new_population
        
    return best_h, best_c5
