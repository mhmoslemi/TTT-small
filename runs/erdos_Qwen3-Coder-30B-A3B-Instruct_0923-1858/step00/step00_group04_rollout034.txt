import numpy as np
from scipy.optimize import minimize
from scipy.sparse.linalg import eigs
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise create a balanced one
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 42
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 0.5
            else:
                h[i] = 0.0
        # Rescale to sum to n_points / 2
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    iterations = 0
    last_improvement = 0
    
    # Precompute correlation matrix
    def compute_correlation_matrix(h):
        # Create correlation vector for each k
        n = len(h)
        corr = np.zeros((2 * n - 1,))
        for k in range(-n + 1, n):
            tmp = 0
            for i in range(n):
                if 0 <= i + k < n:
                    tmp += h[i] * (1 - h[i + k])
            corr[k + n - 1] = tmp
        return corr
    
    def evaluate_c5(h):
        corr = compute_correlation_matrix(h)
        return np.max(corr) * dx
    
    def project_h(h):
        # Clip to [0,1] and adjust sum to 1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h *= (n_points / 2) / h_sum
        return h
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and iterations - last_improvement < 100:
        iterations += 1
        
        # Evaluate current state
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iterations
            
        # Compute gradient estimation via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_copy = h.copy()
            h_copy[i] += eps
            h_copy = project_h(h_copy)
            c5_plus = evaluate_c5(h_copy)
            
            h_copy = h.copy()
            h_copy[i] -= eps
            h_copy = project_h(h_copy)
            c5_minus = evaluate_c5(h_copy)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Frank-Wolfe step
        # Solve argmin_d ⟨grad, d⟩ st 0 ≤ d ≤ 1, sum(d) = 0
        # This is a simple projection to feasible region of d
        d = -grad
        # Project to sum=0: subtract mean
        d = d - np.mean(d)
        # Project to box [0,1]: clip
        d = np.clip(d, 0, 1)
        
        # Line search for optimal step size
        eta = 0.01
        best_eta = eta
        best_h_trial = h.copy()
        best_c5_trial = current_c5
        
        for _ in range(10):  # limited line search
            h_trial = h + eta * d
            h_trial = project_h(h_trial)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5_trial:
                best_c5_trial = c5_trial
                best_h_trial = h_trial.copy()
                best_eta = eta
            eta *= 0.5
        
        h = best_h_trial
        h = project_h(h)
        
        # Random perturbation every 50 iterations
        if iterations % 50 == 0:
            perturb_mag = 0.05
            h += np.random.normal(0, perturb_mag, n_points)
            h = project_h(h)
    
    return (best_h.tolist(), best_c5, n_points)
