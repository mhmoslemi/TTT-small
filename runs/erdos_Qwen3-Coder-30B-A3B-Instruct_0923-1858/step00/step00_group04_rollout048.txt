import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral for all k values
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(np.abs(corr)) * dx
    return c5

def normalize_h(h, n_points):
    # Ensure sum of h equals n_points / 2
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    return h

def project_feasible(h, n_points):
    # Clip and normalize
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points)
    return h

def crossover(parent1, parent2, alpha=0.5):
    # Blend crossover with alpha blending
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    # Random mutation with normal noise
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return h

def tournament_selection(population, fitnesses, k=5):
    # Select the best from k random individuals
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def compute_fitness(h_values, n_points, dx):
    return evaluate_h(h_values, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    max_generations = 1000
    population_size = 30
    mutation_rate = 0.1
    elite_size = int(0.1 * population_size)
    replacement_size = int(0.2 * population_size)
    max_no_improvement = 200
    
    # Population initialization
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_feasible(h_init, n_points)
        population.append(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init, n_points)
        population.append(h_init)
    
    # Fill the rest of the population with random feasible individuals
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Fitness tracking
    fitnesses = [compute_fitness(h, n_points, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    global_best = population[best_idx].copy()
    global_best_fitness = fitnesses[best_idx]
    
    start_time = time.time()
    no_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s and no_improvement < max_no_improvement:
        generation += 1
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if fitnesses[0] < global_best_fitness:
            global_best = population[0].copy()
            global_best_fitness = fitnesses[0]
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Generate offspring
        new_population = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            if np.random.rand() < 0.7:  # crossover probability
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mutation_rate)
            
            # Project to feasible region
            child = project_feasible(child, n_points)
            new_population.append(child)
        
        # Replace population
        population = new_population[:population_size]
        fitnesses = [compute_fitness(h, n_points, dx) for h in population]
    
    final_fitness = compute_fitness(global_best, n_points, dx)
    return (global_best.tolist(), final_fitness, n_points)
