import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial projection to satisfy sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Evaluate initial value
    C5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    best_C5 = C5
    best_h = h.copy()
    
    # Parameters
    T0 = 0.1
    T = T0
    alpha = 0.99
    sigma = 0.1
    patience = 0
    max_patience = 1000
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 0.1 and patience < max_patience:
        # Perturb
        delta = np.random.normal(0, sigma, n_points)
        h_candidate = h + delta
        
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
        
        # Evaluate
        C5_candidate = np.max(correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
        
        # Accept/reject
        delta_C5 = C5_candidate - C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_candidate
            C5 = C5_candidate
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
        
        # Update temperature and perturbation
        T *= alpha
        if iterations % 100 == 0:
            sigma *= 0.9
        
        iterations += 1
    
    return (best_h.tolist(), best_C5, n_points)
