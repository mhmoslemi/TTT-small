import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction from parent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Symmetric construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h[1] = 0.5
        h[-2] = 0.5
        
    n = len(h)
    dx = 2.0 / n
    
    # Normalize initial h to sum = n/2
    h = h * (n / 2) / np.sum(h)
    
    def compute_C5(h_vals):
        # Compute all correlations
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        # Multiply by dx and find maximum
        return np.max(corr) * dx
    
    def project_h(h_vals):
        # Clamp to [0, 1]
        h_clamped = np.clip(h_vals, 0, 1)
        # Rescale to maintain sum = n/2
        h_scaled = h_clamped * (n / 2) / np.sum(h_clamped)
        return h_scaled
    
    # Initialize best solution
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    patience = 1000
    no_improve_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        h_new = h_best + np.random.normal(0, perturb_scale, n)
        h_new = project_h(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        delta = C5_new - C5_best
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_best = h_new
            C5_best = C5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Cooling
        T *= alpha
        if i % 100 == 0:
            perturb_scale *= 0.99
            
        if no_improve_count > patience:
            break
            
    return (h_best.tolist(), C5_best, n)
