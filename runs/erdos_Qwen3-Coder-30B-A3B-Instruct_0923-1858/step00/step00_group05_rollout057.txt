import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Optimization parameters
    learning_rate = 1e-2
    max_iter = 1000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_overlap(h):
        """Compute the overlap integral C5"""
        h_rev = np.flip(1 - h)
        corr = signal.correlate(h, h_rev, mode='full')
        # The maximum correlation is the one we want
        k_max = len(corr) // 2
        c5 = np.max(corr) * dx
        return c5
    
    def compute_gradient(h):
        """Compute gradient of the overlap with respect to h"""
        # This is a simplified gradient approximation approach
        # Direct computation of gradient is complex for this problem
        # so we approximate using finite difference
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            c5_plus = compute_overlap(h_plus)
            c5_minus = compute_overlap(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2*eps)
        return grad
    
    def project_h(h):
        """Project h onto feasible region"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        grad = compute_gradient(h)
        h -= learning_rate * grad
        h = project_h(h)
        
        # Adapt learning rate
        if iteration % 10 == 0:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
    
    # Final computation of C5
    final_c5 = compute_overlap(best_h)
    
    return (list(best_h), final_c5, n_points)
