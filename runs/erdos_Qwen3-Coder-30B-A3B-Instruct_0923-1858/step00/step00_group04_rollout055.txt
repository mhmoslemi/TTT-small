import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[1::2] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Frank-Wolfe with DC programming
    max_iter = 10000
    lr = 0.1
    pert_freq = 100
    
    # Precompute constant terms
    ones = np.ones(n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        h_correlation = np.correlate(h, 1-h, mode="full")
        # Only consider valid shifts for the max
        corr_values = h_correlation[n_points-1:]
        
        # Max correlation index
        max_corr_idx = np.argmax(corr_values)
        max_corr_value = corr_values[max_corr_idx]
        
        # Gradient step
        # we want to minimize max_k integral(h(x)(1-h(x+k))dx)
        # equivalent to minimize max_k <h, 1-h_k>
        # for the gradient we consider the jacobian of the max_k term
        # Modifying the j-th component of h changes h_j(1-h_{j+k})
        # The gradient is the modification that maximizes h_j(1-h_{j+k})
        # This is complex; we use a direct approach
        
        # Compute gradient for individual positions
        grad = np.zeros(n_points)
        k = max_corr_idx
        for j in range(n_points):
            # How changing h[j] affects the integral with shift k
            # i -> h[i] * (1 - h[i-k])
            # when i = j, we get h[j] * (1 - h[j-k])
            # Thus: d/dh_j = 1 - h[j-k] - h[j] * (0) = 1 - h[j-k]
            idx = (j - k) % n_points
            grad[j] = 1 - h[idx]
        
        # Line search for step size
        # Frank-Wolfe: minimize <grad, d> subject to 0 <= d <= 1, sum(d) = 0
        # This is d_j = 1 if grad[j] < 0, 0 if grad[j] > 0, .5 if grad[j] = 0
        step_dir = np.zeros_like(h)
        step_dir[grad < 0] = 1.0
        step_dir[grad > 0] = 0.0
        
        # Project to sum=0 with feasible bounds
        # d = clip(d, 0, 1) then shift to make sum=0
        step_dir = np.clip(step_dir, 0, 1)
        step_dir -= np.mean(step_dir)
        
        # Line search for the step size
        best_step = 0.0
        best_step_h = h.copy()
        best_step_c5 = best_c5
        
        alpha_max = 1.0
        alpha_vals = [0.0, 0.001, 0.01, 0.05, 0.1, 0.2, 0.5]
        for alpha in alpha_vals:
            if alpha > alpha_max:
                continue
            test_h = h + alpha * step_dir
            test_h = np.clip(test_h, 0, 1)
            test_h = test_h - np.mean(test_h) + (n_points / 2) / n_points
            
            test_c5 = evaluate_erdos_solution(test_h, n_points)
            if test_c5 < best_step_c5:
                best_step = alpha
                best_step_h = test_h.copy()
                best_step_c5 = test_c5
                
        h = best_step_h
        if best_step_c5 < best_c5:
            best_c5 = best_step_c5
            best_h = h.copy()
            
        if i % pert_freq == 0 and i > 0:
            # Perturb solution to escape local minima
            pert = np.random.normal(0, 0.02, n_points)
            h += pert
            h = np.clip(h, 0, 1)
            h = h - np.mean(h) + (n_points / 2) / n_points
            h = np.clip(h, 0, 1)  # Trim any boundary violation
            
    return (best_h.tolist(), float(best_c5), int(n_points))
    
def evaluate_erdos_solution(h, n_points):
    """Evaluate the C5 constant using the integral formula"""
    dx = 2.0 / n_points
    # Compute all correlation terms
    inv_h = 1.0 - h
    correlations = np.correlate(h, inv_h, mode="full")
    max_corr = np.max(correlations[n_points-1:])
    return max_corr * dx
