import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular waveform as initial guess
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        # Scale to integral = 1
        h = h * 2.0 / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Optimization parameters
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        old_c5 = evaluate_h(h, dx)
        
        # Compute gradient numerically using finite differences
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1.0, h[i] + epsilon)
            h_minus[i] = max(0.0, h[i] - epsilon)
            
            # Ensure sum constraint
            delta = (h_plus[i] - h[i]) - (h[i] - h_minus[i])
            if i == 0:
                h_plus[1] += delta
                h_minus[1] -= delta
            else:
                h_plus[0] += delta
                h_minus[0] -= delta
            
            # Clamp to [0, 1]
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Fix sum
            h_plus = h_plus * 2.0 / np.sum(h_plus)
            h_minus = h_minus * 2.0 / np.sum(h_minus)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Compute trial step
        trial_h = h - trust_radius * grad
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = trial_h * 2.0 / np.sum(trial_h)
        
        new_c5 = evaluate_h(trial_h, dx)
        
        # Accept step if improvement
        if new_c5 < old_c5:
            h = trial_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            trust_radius = min(0.5, trust_radius * 1.2)
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            trust_radius *= 0.5
            
            if no_improvement_count > max_no_improvement:
                # Perturb the solution
                perturbation = np.random.normal(0, 0.01, n_points)
                h += perturbation
                h = np.clip(h, 0, 1)
                h = h * 2.0 / np.sum(h)
                no_improvement_count = 0

    return (list(best_h), best_c5, n_points)

def evaluate_h(h, dx):
    # Compute 1-h
    h_complement = 1 - h
    # Compute cross-correlation
    corr = correlate(h, h_complement, mode="full")
    # The relevant part is from index (n_points-1) to 2*n_points-2
    # The max gives us C5
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx
