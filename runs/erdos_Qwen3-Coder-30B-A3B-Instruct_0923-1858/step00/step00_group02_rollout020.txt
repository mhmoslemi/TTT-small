import numpy as np
from scipy import optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        n_points = 50
        h0 = np.full(n_points, 0.5)
    
    # Binary search bounds
    C_low = 0.0
    C_high = 0.38085
    best_C = C_high
    best_h = h0.copy()
    
    # Precompute shifts
    shifts = list(range(1-n_points, n_points))
    
    # Helper function to compute overlaps
    def compute_overlaps(h):
        # Compute all overlaps for different shifts
        overlaps = []
        for k in shifts:
            if k >= 0:
                h1 = h[:-k] if k > 0 else h
                h2 = h[k:] if k > 0 else h
            else:
                h2 = h[-k:] if k < 0 else h
                h1 = h[:len(h2)]
            overlap = np.sum(h1 * (1 - h2))
            overlaps.append(overlap)
        return np.array(overlaps)
    
    # Run binary search
    while C_high - C_low > 1e-6 and time.time() - start_time < budget_s - 1:
        C_mid = (C_low + C_high) / 2.0
        
        # Try to find a feasible solution for this C
        try:
            # Define variables
            h_vars = cp.Variable(n_points)
            
            # Define constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2.0]
            
            # Add overlap constraints for each shift k
            for k in shifts:
                if k >= 0:
                    # h[i] * (1 - h[i+k]) for i in range(n_points-k)
                    indices = np.arange(n_points - k)
                    constraint = cp.sum(cp.multiply(h_vars[indices], 1 - h_vars[indices + k])) <= C_mid * n_points
                else:
                    # h[i+k] * (1 - h[i]) for i in range(-k, n_points)
                    indices = np.arange(-k, n_points)
                    constraint = cp.sum(cp.multiply(h_vars[indices + k], 1 - h_vars[indices])) <= C_mid * n_points
                constraints.append(constraint)
            
            # Create and solve the problem
            obj = cp.Minimize(0)  # dummy objective
            prob = cp.Problem(obj, constraints)
            
            # Set solver and solve
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_opt = h_vars.value
                best_h = h_opt
                best_C = C_mid
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid

    # Final evaluation
    overlaps = compute_overlaps(best_h)
    final_C = np.max(overlaps) / n_points
    
    return (best_h, final_C, n_points)
