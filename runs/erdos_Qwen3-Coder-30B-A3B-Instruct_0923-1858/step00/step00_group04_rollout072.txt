import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s > 0:
        h = h * (n_points / 2) / s
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with constraint satisfaction"""
    pop = []
    if initial_h is not None:
        pop.append(project_h(initial_h, n_points))
    while len(pop) < pop_size:
        h = np.random.random(n_points)
        h = project_h(h, n_points)
        pop.append(h)
    return pop

def tournament_selection(pop, fitnesses, k=5):
    """Select parent using tournament selection"""
    idxs = np.random.choice(len(pop), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return pop[best_idx]

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    """Mutate h with Gaussian noise"""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return project_h(h, len(h))
