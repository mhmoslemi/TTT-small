import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the maximum overlap integral C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value of the overlap integral
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum=h_sum."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def gradient_step(h, alpha, n_points):
    """Compute gradient step and project."""
    dx = 2.0 / n_points
    # Compute current C5
    corr = correlate(h, 1 - h, mode='full')
    k_max = len(corr) - 1
    max_idx = np.argmax(corr)
    max_k = max_idx - (k_max // 2)
    # Gradient computation based on derivative of correlation at max_k
    # We compute gradient w.r.t. h that increases the max overlap
    grad = np.zeros_like(h)
    # Efficient gradient computation for a specific lag
    # derivative of (h_k * (1 - h_{k+max_k})) w.r.t h_k
    if max_k > 0:
        for i in range(len(h) - max_k):
            grad[i] += -1 + 2 * h[i + max_k]
    else:
        max_k = -max_k
        for i in range(len(h) - max_k):
            grad[i + max_k] += -1 + 2 * h[i]
    grad *= dx
    h_new = h - alpha * grad
    h_new = project_h(h_new, n_points)
    return h_new
