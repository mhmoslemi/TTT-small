import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    
    def compute_C5(h):
        # Compute correlation: correlate(h, 1-h, mode="full")
        corr = np.correlate(h, 1-h, mode="full")
        # Find maximum
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient(h):
        # Numerically compute gradient
        eps = 1e-6
        grad = np.zeros_like(h)
        C0 = compute_C5(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            # Renormalize
            h_plus = h_plus * target_sum / np.sum(h_plus)
            C_plus = compute_C5(h_plus)
            grad[i] = (C_plus - C0) / eps
        return grad
    
    # Gradient descent with line search
    max_iter = 1000
    alpha = 0.01
    patience = 0
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        g = gradient(h)
        # Line search
        alpha_ls = alpha
        h_new = h - alpha_ls * g
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale
        h_new = h_new * target_sum / np.sum(h_new)
        
        C_new = compute_C5(h_new)
        
        if C_new < best_C5:
            best_C5 = C_new
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Decrease learning rate
        if i % 100 == 0 and i > 0:
            alpha *= 0.5
    
    # Refinement at higher resolution
    if time.time() - start_time < budget_s - 5:
        n_points_high = n_points * 2
        dx_high = 2.0 / n_points_high
        # Interpolate
        h_high = np.interp(np.linspace(0, 2, n_points_high), 
                           np.linspace(0, 2, n_points), 
                           best_h)
        h_high = h_high * target_sum / np.sum(h_high)
        
        # Gradient descent at higher resolution
        for j in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            g = gradient(h_high)
            alpha_ls = alpha * 0.5
            h_new = h_high - alpha_ls * g
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            
            C_new = compute_C5(h_new)
            if C_new < best_C5:
                best_C5 = C_new
                best_h = h_new.copy()
            h_high = h_new.copy()
            
        h = best_h
        n_points = n_points_high
    
    # Final validation
    final_C5 = compute_C5(h)
    
    return (h.tolist(), final_C5, n_points)
