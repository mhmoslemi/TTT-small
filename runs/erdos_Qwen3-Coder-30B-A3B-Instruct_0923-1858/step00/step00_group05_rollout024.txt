import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = correlate(h, 1 - h, mode="full")
    max_correlation = np.max(np.abs(correlations))
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 70
    max_iter = 50000
    alpha = 0.01
    perturbation_rate = 0.1
    perturbation_freq = 100
    
    # initialize
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # scale to correct sum
    h = h * (0.5 / np.mean(h))
    
    # track best
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # current evaluation
        C5_candidate = evaluate_erdos_solution(h)
        if C5_candidate < best_C5:
            best_h = h.copy()
            best_C5 = C5_candidate
        
        # compute gradient approximation
        gradient = np.zeros(n_points)
        dx = 2.0 / n_points
        
        # gradient based on cross-correlation structure
        # direct computation of gradient is complex, 
        # so we use finite differences for approximation purposes
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus = h.copy()
            h_minus[i] = max(0.0, h_minus[i] - eps)
            C5_plus = evaluate_erdos_solution(h_plus)
            C5_minus = evaluate_erdos_solution(h_minus)
            gradient[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # update step
        h = h - alpha * gradient
        
        # enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.mean(h))
        
        # perturb occasionally
        if iteration % perturbation_freq == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.mean(h))
        
        # adjust learning rate
        if iteration % 1000 == 0:
            alpha *= 0.95
    
    # final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    return (best_h, final_C5, n_points)
