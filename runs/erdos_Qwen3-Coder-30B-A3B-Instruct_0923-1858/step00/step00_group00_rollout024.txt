import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using full convolution and taking max
    corr = np.correlate(h, 1 - h, mode='full')
    # Trim to valid range of shifts (0 to 2)
    # convolution length is 2*n - 1, valid indices for shift k=0 to k=2 are n-1 to 2*n-2
    mid = len(corr) // 2
    valid_corr = corr[mid:mid + len(h)]
    return np.max(valid_corr) * dx

def initial_h():
    # Try to start from initial_h_values if available.
    # If not available, create a symmetric function
    try:
        return np.array(initial_h_values)
    except:
        # Simple symmetric initial construction
        n = 45
        h = np.ones(n)
        h[:n//2] = 0
        return h

def project_to_feasible(h, n_points):
    # Ensure h is in [0,1] and sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to target sum
        h = h * target_sum / current_sum
    return h

def genetic_algorithm(seed, budget_s):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    P = 50  # population size
    generations = 500
    mutation_rate = 0.95
    tournament_size = 5
    max_stagnation = 20

    h0 = initial_h()
    if len(h0) != n_points:
        h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)

    # Initialize population
    population = [h0.copy()]
    for _ in range(P - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = h0.copy()
    stagnation_count = 0
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.98:
            break
            
        # Evaluate population
        fitness = [evaluate_h(h, dx) for h in population]
        best_idx = np.argmin(fitness)
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > max_stagnation:
            break
            
        # Selection using tournament
        selected = []
        for _ in range(P):
            tournament = np.random.choice(P, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner])
            
        # Crossover and mutation
        new_population = [best_h.copy()]
        for i in range(1, P, 2):
            parent1 = selected[i-1]
            parent2 = selected[i]
            
            # Blend crossover
            alpha = 0.5
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            std = 0.05 * mutation_rate ** gen
            child1 += np.random.normal(0, std, n_points)
            child2 += np.random.normal(0, std, n_points)
            
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
            
        population = new_population[:P]
    
    return best_h, best_c5

def gradient_descent(h_start, budget_s):
    n_points = len(h_start)
    dx = 2.0 / n_points
    start_time = time.time()
    learning_rate = 0.01
    epsilon = 1e-5
    max_iter = 1000
    
    h = h_start.copy()
    
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    # Keep track of previous values for early stopping
    prev_values = []
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Calculate gradient using finite differences
        grad = np.zeros(n_points)
        
        for k in range(n_points):
            h_plus = h.copy()
            h_plus[k] += epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[k] -= epsilon
            h_minus = project_to_feasible(h_minus, n_points)
            
            c_minus = evaluate_h(h_minus, dx)
            
            grad[k] = (c_plus - c_minus) / (2 * epsilon)
        
        # Update using projected gradient descent
        h_new = h - learning_rate * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Also check if projection improved within bounds
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            
        # Check for convergence (if no significant improvement)
        prev_values.append(c5_new)
        if len(prev_values) > 10:
            last_10 = prev_values[-10:]
            if max(last_10) - min(last_10) < 1e-7:
                break
                
        # Update learning rate
        learning_rate *= 0.995
        if learning_rate < 1e-6:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Genetic algorithm to get good starting point
    h_ga, c5_ga = genetic_algorithm(seed, budget_s/2)
    
    # Refine with gradient descent
    h_final, c5_final = gradient_descent(h_ga, budget_s/2)
    
    return (h_final, c5_final, n_points)
