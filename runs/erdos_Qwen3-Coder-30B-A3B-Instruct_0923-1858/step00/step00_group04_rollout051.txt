import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Compute the maximum overlap for given h"""
    # Cross-correlation with 1-h
    cross_corr = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx to match integral definition
    max_overlap = np.max(cross_corr) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to satisfy sum constraint, keeping in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Linear interpolation to adjust all values
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.minimum(1.0, h + diff / len(h))
    else:
        # Need to decrease values
        excess = -diff
        h = np.maximum(0.0, h - excess / len(h))
    # Scale if needed to hit exact target
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h.copy()
    else:
        h = np.ones(n_points) * 0.5
        h[::2] = 1.0
        h[1::2] = 0.0
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iter_count = 0
    perturb_interval = 100
    perturb_strength = 0.05
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute cross-correlation
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        # Find max overlap index
        max_idx = np.argmax(cross_corr)
        # Get the actual overlap value
        max_overlap = cross_corr[max_idx] * dx
        
        # Find the indices contributing to this max overlap
        # We need to be careful with indexing here
        # The correlation has length 2*n_points - 1
        # The center (n_points - 1) corresponds to k=0
        k = max_idx - (n_points - 1)
        
        # Now identify where to make changes
        # Compute for each position, how much h[i] contributes to max overlap
        # This is when we look at h[i] compared to h[(i + k) % n_points]
        # i on left side and (i+k) on right side
        
        # Create a modification vector
        modification = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            # Increase modification at i to decrease contribution of that pair
            # Small adjustment to reduce overlap
            modification[i] = -0.01 * (h[i] * (1 - h[j]))
        
        # Apply modifications
        h = h + modification
        h = project_h(h, target_sum)
        
        # Random perturbation occasionally
        if iter_count % perturb_interval == 0:
            # Select some random indices
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            # Add some random mass to decrease overlap
            delta = np.random.normal(0, perturb_strength, size=len(indices))
            delta = np.clip(delta, -np.min(h[indices]), np.max(1 - h[indices]))
            h[indices] += delta
            h = project_h(h, target_sum)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
