        corr = correlate(h, 1 - h, mode='full')

        # Compute the contribution regions
        overlap_indices = []
        if max_k < len(h):
            # Regular case: h[i] and 1-h[i+k] where k = shift
            k = max_k
            for i in range(len(h)):
                if i + k < len(h):
                    overlap_indices.append(i)
        else:
            # Anti-diagonal case
            k = max_k - len(h) + 1
            for i in range(len(h) - k):
                overlap_indices.append(i)

        candidates = []

def evaluate_C5(h_values, dx):
    h = np.array(h_values)
    # Compute the cross-correlation
    corr = correlate(h, 1-h, mode='full')
    # Return the maximum value multiplied by dx
    return np.max(corr) * dx
