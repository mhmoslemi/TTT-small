import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values or create a triangular waveform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a triangular waveform as starting point
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 1.0 - np.abs(x - 1.0)  # Triangular waveform
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Rescale to sum to n_points/2
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Feasibility check and rescaling
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to meet sum constraint
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def compute_gradient(h_vals, epsilon=1e-4):
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_vals.copy()
            h_minus[i] -= epsilon
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        return grad
    
    def trust_region_step(h_vals, grad, trust_radius):
        # Simple trust region step: project gradient step into trust ball
        step = -grad
        norm_step = np.linalg.norm(step)
        if norm_step > trust_radius and norm_step > 1e-10:
            step = step * trust_radius / norm_step
        new_h = h_vals + step
        new_h = project_h(new_h)
        return new_h
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Take trust region step
        new_h = trust_region_step(h, grad, trust_radius)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5:
            h = new_h
            trust_radius = min(2.0 * trust_radius, 1.0)  # Expand trust region
        else:
            trust_radius = trust_radius * 0.5  # Shrink trust region
            if trust_radius < 1e-6:
                # Perturb
                perturbation = np.random.normal(0, 0.01, n_points)
                h = project_h(h + perturbation)
                trust_radius = 0.1
        
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
