def evaluate_h(h_values, dx):

    return max_corr * dx

def project_to_feasible(h):
    """Project h to satisfy constraints: 0<=h<=1, sum(h)=n/2."""
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

    # Sort population by fitness
    fitness = [evaluate_h(h, dx) for h in h_pop]

    h_pop = [h_pop[i] for i in sorted_indices]

    # Keep elites
    new_pop = h_pop[:elites]

        parent1 = h_pop[np.random.choice(elites)]
        parent2 = h_pop[np.random.choice(elites)]
        
        # Crossover
        alpha = 0.5
        child = alpha * parent1 + (1 - alpha) * parent2
        
        # Mutation
        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, 0.05, n)
            child += noise
            child = project_to_feasible(child)
        
        new_pop.append(child)
    
    return new_pop

        h = h_new
