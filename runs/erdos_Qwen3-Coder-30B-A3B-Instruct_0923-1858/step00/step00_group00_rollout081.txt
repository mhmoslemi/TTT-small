import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    return h * scale

def evaluate_c5(h, n_points):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / n_points
    # Compute correlation using numpy
    corr = correlate(h, 1 - h, mode='full')
    # Max over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def nelder_mead_simplex(h0, n_points, budget_s):
    """Modified Nelder-Mead with adaptive simplex size"""
    # Nelder-Mead parameters
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink
    
    # Initial simplex size
    simplex_size = 0.1
    max_reset_attempts = 10
    reset_counter = 0
    stagnation_threshold = 50
    
    num_vertices = len(h0) + 1
    vertices = np.tile(h0, (num_vertices, 1))
    
    # Create initial simplex
    for i in range(num_vertices - 1):
        vertices[i+1, i] += simplex_size
    
    # Ensure feasible vertices
    for i in range(num_vertices):
        vertices[i] = project_h(vertices[i], n_points)
    
    func_values = np.array([evaluate_c5(v, n_points) for v in vertices])
    
    best_idx = np.argmin(func_values)
    best_h = vertices[best_idx].copy()
    best_c5 = func_values[best_idx]
    
    start_time = time.time()
    iteration = 0
    stagnant_count = 0
    prev_best = float('inf')
    
    while time.time() - start_time < budget_s and iteration < 5000:
        iteration += 1
        
        # Sort vertices by function value
        sorted_indices = np.argsort(func_values)
        vertices = vertices[sorted_indices]
        func_values = func_values[sorted_indices]
        
        # Update best if improved
        if func_values[0] < best_c5:
            best_c5 = func_values[0]
            best_h = vertices[0].copy()
            stagnant_count = 0
            prev_best = best_c5
        else:
            stagnant_count += 1
            
        # Check for stagnation
        if stagnant_count >= stagnation_threshold:
            reset_counter += 1
            if reset_counter >= max_reset_attempts:
                break
            # Reset simplex size
            simplex_size *= 0.5
            if simplex_size < 0.01:
                simplex_size = 0.01
            # Reinitialize simplex
            vertices = np.tile(vertices[0], (num_vertices, 1))
            for i in range(num_vertices - 1):
                vertices[i+1, i] += simplex_size
            stagnant_count = 0
        else:
            # Compute centroid (excluding worst)
            centroid = np.mean(vertices[:-1], axis=0)
            
            # Reflect
            h_r = centroid + alpha * (centroid - vertices[-1])
            h_r = project_h(h_r, n_points)
            c_r = evaluate_c5(h_r, n_points)
            
            if func_values[0] <= c_r < func_values[-2]:
                vertices[-1] = h_r
                func_values[-1] = c_r
            elif c_r < func_values[0]:
                # Expand (expansion)
                h_e = centroid + gamma * (h_r - centroid)
                h_e = project_h(h_e, n_points)
                c_e = evaluate_c5(h_e, n_points)
                if c_e < c_r:
                    vertices[-1] = h_e
                    func_values[-1] = c_e
                else:
                    vertices[-1] = h_r
                    func_values[-1] = c_r
            else:
                # Contract or shrink
                if c_r < func_values[-1]:
                    # Outside contraction
                    h_c = centroid + beta * (h_r - centroid)
                    h_c = project_h(h_c, n_points)
                    c_c = evaluate_c5(h_c, n_points)
                    if c_c <= c_r:
                        vertices[-1] = h_c
                        func_values[-1] = c_c
                    else:
                        # Shrink
                        for i in range(1, num_vertices):
                            vertices[i] = vertices[0] + rho * (vertices[i] - vertices[0])
                            vertices[i] = project_h(vertices[i], n_points)
                else:
                    # Inside contraction
                    h_c = centroid + beta * (vertices[-1] - centroid)
                    h_c = project_h(h_c, n_points)
                    c_c = evaluate_c5(h_c, n_points)
                    if c_c < func_values[-1]:
                        vertices[-1] = h_c
                        func_values[-1] = c_c
                    else:
                        # Shrink
                        for i in range(1, num_vertices):
                            vertices[i] = vertices[0] + rho * (vertices[i] - vertices[0])
                            vertices[i] = project_h(vertices[i], n_points)
    
    return best_h, best_c5
