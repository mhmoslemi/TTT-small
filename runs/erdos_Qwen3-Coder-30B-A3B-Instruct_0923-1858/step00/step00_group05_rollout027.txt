import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral C5
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for k = 0 to 2*dx
    correlations = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (k from 0 to 2)
    # The full correlation array is of size 2*n - 1
    # We want the part where k >= 0, which is from index n-1 to 2*n-2
    # But we need k from 0 to 2, not from -2 to 2
    # So we need to pick from k=0 to k=2 in the correlation array
    # correlation index offset is n-1 and we want correlations from 0 to 2 dx,
    # which corresponds to indices n-1 to n+1
    # However, we can compute as "full" and then extract
    # Indices from 0 to 2*n-2
    # For k = 0, index = n - 1, k = 2, index = n + 1 
    max_corr = correlations[n-1:n+2] 
    # These are for k: 0, 1, 2 
    # but we need indices to find k from 0 to 2 in dx steps
    # correlation index i gives k = i - (n-1) 
    # so index n-1 gives k=0, n gives k=1, n+1 gives k=2 
    # so k=0,1,2 corresponds to indices n-1, n, n+1 respectively
    # hence max_corr is correlations[n-1:n+2] 
    # the values are already scaled by dx so no need to multiply again
    return np.max(max_corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial if available
    # If initial_h_values is an array, we must retain only its first n_points elements.
    # So we construct the full length for internal calculus, but use a reduced chunk if needed.
    # Don't directly modify initial_h_values even if it exists
    try:
        initial_h_values
    except NameError:
        initial_h_values = None

    n_points = 100  # Start with a modest point count
    dx = 2.0 / n_points

    # Initialize h
    if initial_h_values is not None and len(initial_h_values) >= n_points:
        h = initial_h_values[:n_points].copy()
        h = h / np.sum(h) * (n_points / 2)  # normalize so sum = n_points/2
    else:
        # If initial_h_values is not available, create a reasonable starting point
        h = np.ones(n_points) * 0.5  # Start with a constant halfway value
        h = h / np.sum(h) * (n_points / 2)  # normalize so sum = n_points/2

    # Ensure h is in valid range [0, 1]
    h = np.clip(h, 0, 1)

    # Define objective
    def obj(h_vec):
        # Return negative of C5 (because minimize wants to minimize)
        return -evaluate_erdos_solution(h_vec)

    # Flatten constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Equality constraint for sum(h) = n_points/2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    constraints = {'type': 'eq', 'fun': constraint_sum}

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Use too many retries and early breaking
    attempts = 0
    max_attempts = 5
    opt_method = 'SLSQP'
    while attempts < max_attempts and time.time() - start_time < budget_s - 1:
        # Try optimization starting from current best h
        try:
            res = minimize(
                obj,
                best_h,
                method=opt_method,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            
            if res.success:
                new_h = res.x
                new_c5 = -obj(new_h)  # obj returns -C5
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
            else:
                # If optimization failed, maybe try another initialization
                pass
        except:
            pass

        # Try other random starting points if time permits
        if time.time() - start_time < budget_s - 5:
            # Construct an alternate feasible solution
            # Random perturbations of the best
            h_alt = np.random.rand(n_points) * 0.1 + 0.45  # Near 0.5 but random
            h_alt = np.clip(h_alt, 0, 1)
            h_alt = h_alt / np.sum(h_alt) * (n_points / 2)
            # Reoptimize from this point
            try:
                res = minimize(
                    obj,
                    h_alt,
                    method=opt_method,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 500, 'ftol': 1e-5, 'gtol': 1e-5}
                )
                if res.success:
                    new_h = res.x
                    new_c5 = -obj(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
            except:
                pass

        attempts += 1

    # Final check
    if np.sum(best_h) < n_points / 2 - 1e-6 or np.sum(best_h) > n_points / 2 + 1e-6:
        # Re-norm
        best_h = best_h / np.sum(best_h) * (n_points / 2)

    final_c5 = evaluate_erdos_solution(best_h)
    final_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(final_h)

    return (final_h, final_c5, n_points)
