import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with provided construction, or create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Projection function to keep h in [0,1] and maintain sum constraint
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        return h
    
    # Objective function
    def compute_c5(h):
        h = np.array(h)
        corr = signal.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initial C5
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Annealing parameters
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_patience = 1000
    patience_counter = 0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Perturb
        pert = np.random.normal(0, perturb_scale, n_points)
        candidate_h = h + pert
        
        # Project
        candidate_h = project_h(candidate_h)
        
        # Compute objective
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            h = candidate_h
            best_c5 = candidate_c5
            patience_counter = 0
        else:
            # Accept probabilistically
            delta = candidate_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = candidate_h
                patience_counter = 0
            else:
                patience_counter += 1
                
        # Cooling
        T *= alpha
        if i % 100 == 0:
            perturb_scale *= 0.99
            
        if patience_counter > no_improve_patience:
            break
    
    # Final projection and verification
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
