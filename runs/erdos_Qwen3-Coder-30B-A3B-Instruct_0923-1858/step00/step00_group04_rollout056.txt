import numpy as np
from scipy.optimize import minimize
from math import sqrt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
        h[1::2] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(np.abs(corr))
        return max_corr * dx
    
    # Project h to feasible region
    def project_h(h_vals):
        h = np.array(h_vals)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Frank-Wolfe optimization
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Parameters
    max_iter = 10000
    eta = 0.1
    pert_freq = 100
    pert_strength = 0.05

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient estimate (approximate)
        h_current = best_h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(np.abs(corr))
        
        # Construct gradient-like direction
        # gradient direction: (1 - h) shifted left by max_k
        grad_direction = np.zeros(n_points)
        if max_k < n_points:
            grad_direction[max_k] = 1.0
        else:
            grad_direction[max_k - n_points] = 1.0
            
        # Backward shift to get actual gradient approx (approximate cross-correlation gradient)
        # Direct finite difference for gradient direction proxy
        direction = np.zeros(n_points)
        # Approximate gradient: in direction that decreases max correlation
        direction = (1 - h_current) if max_k < n_points else np.roll(1 - h_current, max_k - n_points)
        direction = direction / (np.linalg.norm(direction) + 1e-12)

        # Solve linearized subproblem (solve direction)
        d = direction * eta
        h_new = h_current + d
        h_new = project_h(h_new)
        
        # Update h
        h_current = h_new
        h_current = project_h(h_current)
        
        # Evaluate
        current_c5 = compute_c5(h_current)
        
        # Check improvements
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
        
        # Perturbation every few iterations
        if iteration % pert_freq == 0 and iteration > 0:
            perturbation = np.random.normal(0, pert_strength, n_points)
            h_current = h_current + perturbation
            h_current = project_h(h_current)
            
    return (best_h.tolist(), best_c5, n_points)
