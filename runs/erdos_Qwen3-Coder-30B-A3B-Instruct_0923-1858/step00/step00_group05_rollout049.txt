from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = correlate(h, 1 - h, mode="full")
    max_correlation = np.max(np.abs(correlations))
    return max_correlation * dx
