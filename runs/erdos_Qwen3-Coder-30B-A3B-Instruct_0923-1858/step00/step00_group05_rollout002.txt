import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    # Compute the overlap integral for all shifts
    n = len(h)
    overlap = np.correlate(h, 1 - h, mode="full")
    # Take only the valid shifts (not the padding)
    valid_overlap = overlap[n-1:2*n-1]
    return np.max(valid_overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if np.abs(np.sum(h) * dx - 1.0) > 1e-10:
            # Renormalize to sum to 1
            h = h * (1.0 / (np.sum(h) * dx)) * dx
    else:
        # Create a random feasible h
        h = np.random.rand(n_points)
        h = h * (1.0 / (np.sum(h) * dx)) * dx
        h = np.clip(h, 0, 1)
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(x):
        return np.sum(x) * dx - 1.0
    
    # Use scipy.optimize to minimize C5
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Use a simple gradient-based approach
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def gradient(h_vec):
        # Simple finite difference gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (evaluate_erdos_solution(h_plus, dx) - evaluate_erdos_solution(h_minus, dx)) / (2 * eps)
        return grad
    
    # Precompute the constraints for SLSQP
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    # Add a time budget
    start_time = time.time()
    
    # Try using multiple initializations to escape local minima
    for attempt in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Slightly different initializations
        if attempt == 0:
            h_current = h.copy()
        else:
            # Perturb current h a bit
            h_current = h.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            # Renormalize if needed
            if np.abs(np.sum(h_current) * dx - 1.0) > 1e-10:
                h_current = h_current * (1.0 / (np.sum(h_current) * dx)) * dx
        
        # Optimization step
        try:
            result = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6}
            )
            
            if result.success:
                h_opt = result.x
                C5_opt = evaluate_erdos_solution(h_opt, dx)
                if C5_opt < best_C5:
                    best_h = h_opt
                    best_C5 = C5_opt
                
        except Exception as e:
            # Skip but continue
            pass
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final adjustment to ensure sum constraint
    best_h = best_h * (1.0 / (np.sum(best_h) * dx)) * dx
    
    return (best_h, best_C5, n_points)
