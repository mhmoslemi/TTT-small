import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: simple step function
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    n_points = len(h)
    dx = 2.0 / n_points

    # Ensure sum constraint is met
    h = h * (n_points / 2.0) / np.sum(h)
    
    start_time = time.time()
    
    C5_best = np.inf
    best_h = h.copy()
    iterations_without_improvement = 0
    max_no_improve = 100
    
    # Iteration loop
    while time.time() - start_time < budget_s:
        # Compute current cross-correlation and C5
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        C5_current = corr[max_corr_idx] * dx
        
        if C5_current < C5_best:
            C5_best = C5_current
            best_h = h.copy()
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
        
        if iterations_without_improvement > max_no_improve:
            break

        # Find indices where max appears to shift
        max_corr_shift = max_corr_idx - (len(h) - 1)
        # Find overlapping regions
        overlap_start = max(0, -max_corr_shift)
        overlap_end = min(len(h), len(h) - max_corr_shift)
        
        # Create candidate perturbations
        candidate_h = h.copy()
        
        # Select a few pairs to shift mass
        num_tries = min(50, len(h))
        best_delta = 0.0
        best_candidate = None

        for _ in range(num_tries):
            i = np.random.randint(0, len(h))
            j = np.random.randint(0, len(h))

            # Ensure we don't go out of bounds
            if i == j:
                continue
                
            # Calculate maximal transfer
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
                
            fraction = np.random.uniform(0.0, 1.0) * 0.1  # Small, random shift
            transfer = fraction * max_transfer
            
            # Apply transfer
            candidate_h[i] -= transfer
            candidate_h[j] += transfer
            
            # Ensure feasibility
            if candidate_h[i] < 0 or candidate_h[j] > 1:
                # Reset if infeasible
                candidate_h[i] += transfer
                candidate_h[j] -= transfer
                continue
                
            # Recompute C5 for the new candidate
            corr_new = correlate(candidate_h, 1 - candidate_h, mode='full')
            new_max = corr_new[np.argmax(corr_new)] * dx
            
            if new_max < C5_current:
                if best_candidate is None or new_max < best_delta:
                    best_candidate = candidate_h.copy()
                    best_delta = new_max
            
            # Reset for next trial
            candidate_h[i] += transfer
            candidate_h[j] -= transfer

        # If we found an improvement, update h
        if best_candidate is not None:
            h = best_candidate.copy()
        
        # Occasionally rescale to maintain constraint
        if np.abs(np.sum(h) * dx - 1.0) > 1e-5:
            h = h * (n_points / 2.0) / np.sum(h)
        
        # Optional: Double resolution after some iterations
        if (time.time() - start_time) > budget_s * 0.5 and len(h) < 500:
            n_points_old = len(h)
            h_interp = np.interp(np.linspace(0, 2, 2 * n_points_old), 
                                np.linspace(0, 2, n_points_old), h)
            h = h_interp
            dx = 2.0 / len(h)
            n_points = len(h)

    # Final validation of h
    assert(np.all(h >= 0) and np.all(h <= 1))
    assert(np.abs(np.sum(h) * dx - 1.0) < 1e-6)
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_C5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_C5, len(best_h))
