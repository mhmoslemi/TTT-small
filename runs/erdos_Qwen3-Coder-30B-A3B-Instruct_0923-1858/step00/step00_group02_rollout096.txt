import numpy as np
import cvxpy as cp
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create initial construction
        n_points = 45
        h0 = np.full(n_points, 0.5)
    
    # Normalize to sum to 1
    h0 = h0 * 2.0 / np.sum(h0)
    
    # Binary search setup
    C_low = 0.0
    C_high = 0.38085
    best_h = h0.copy()
    best_C = C_high
    
    # Precompute all shifts
    shifts = list(range(1 - n_points, n_points))
    
    def compute_overlap(h_vec):
        # Computes all overlaps using circular correlation
        h_vec = np.array(h_vec)
        # For each shift k, compute sum h[i] * (1 - h[(i+k)%n])
        overlaps = []
        for k in shifts:
            if k >= 0:
                h_shifted = np.concatenate([h_vec[k:], h_vec[:k]])
            else:
                h_shifted = np.concatenate([h_vec[k:], h_vec[:k]])
            overlap = np.sum(h_vec * (1 - h_shifted))
            overlaps.append(overlap)
        return np.max(overlaps)
    
    def solve_feasibility(C):
        # Formulate and solve the LP feasibility problem
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Overlap constraints for all shifts
        for k in shifts:
            # Handle circular convolution efficiently
            if k >= 0:
                h_shifted = cp.hstack([h[k:], h[:k]])
            else:
                h_shifted = cp.hstack([h[k:], h[:k]])
            
            # Sum h[i] * (1 - h[i+k]) over all i
            overlap = cp.sum(cp.multiply(h, 1 - h_shifted))
            constraints.append(overlap <= C * n_points)
        
        # Create problem (dummy objective to satisfy CVXPY)
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
            else:
                return None
        except:
            return None
            
    # Binary search loop
    while (time.time() - start_time < budget_s - 1) and (C_high - C_low > 1e-6):
        C_mid = (C_low + C_high) / 2.0
        result = solve_feasibility(C_mid)
        if result is not None:
            C_high = C_mid
            best_h = result.copy()
            best_C = C_mid
        else:
            C_low = C_mid
            
    # Final validation
    final_overlap = compute_overlap(best_h)
    
    # Ensure the result is within bounds
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * 2.0 / np.sum(h_final)
    
    # Return h values, C5 bound, and points used
    return h_final.tolist(), float(min(best_C, final_overlap)), len(h_final)
