import numpy as np
import time
from scipy import signal

def evaluate(h):
    """Evaluate the overlap integral C5 for given h."""
    # Compute the overlap integral using convolution
    h = np.asarray(h)
    dx = 2.0 / len(h)
    # Compute correlation of h with (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum value
    max_corr = np.max(corr)
    # Account for scaling
    c5 = max_corr * dx
    return c5

def repair_h(h):
    """Ensure h satisfies constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""

    n_points = len(h)

    h = np.clip(h, 0, 1)
    # Normalize sum to exactly n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum != 0:
        h *= target_sum / current_sum
    return h

def mutate(h, rate=0.03):
    """Apply small random mutations to h."""
    h = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.rand() < rate:
            # Add small Gaussian noise
            h[i] += np.random.normal(0, 0.05)
    return repair_h(h)

    """Create offspring via crossover of two parents."""

            h[i] = h2[i]
