import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a symmetric starting point
        n_points = 45
        h_start = np.ones(n_points) * 0.5
        h_start[10:35] = 1.0  # High values in the middle
        h_start[:10] = 0.0   # Low at start
        h_start[35:] = 0.0   # Low at end
    
    # Normalize to desired integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Take the maximum (excluding trivial zero cases)
        c5 = np.max(corr) * dx
        return c5
    
    # Modified Nelder-Mead optimization
    def optimize_h(h_guess):
        # Transform to unconstrained space for optimizer
        def obj_transformed(x):
            # Inverse sigmoid to map [-inf, inf] to [0, 1]
            h_vals = 1.0 / (1.0 + np.exp(-x))
            # Adjust for sum constraint
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            h_vals = np.clip(h_vals, 0, 1)
            return compute_c5(h_vals)
        
        # Minimize over unconstrained variables
        try:
            res = minimize(obj_transformed, np.log(h_guess / (1 - h_guess)), method='Nelder-Mead', 
                           options={'maxiter': 1000, 'xtol': 1e-6, 'ftol': 1e-6})
            h_final = 1.0 / (1.0 + np.exp(-res.x))
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            h_final = np.clip(h_final, 0, 1)
            return h_final
        except:
            return h_guess
    
    # Run multiple local optimizations with different starts
    for iter_idx in range(50):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_start = np.random.rand(n_points) * 0.5 + 0.25  # Random normalized start
        h_opt = optimize_h(h_start)
        c5_val = compute_c5(h_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
    
    # Binary search to tighten bounds more precisely
    c_min, c_max = 0.0, 0.5
    for _ in range(15):
        if time.time() - start_time > budget_s - 1:
            break
        c_mid = (c_min + c_max) / 2.0
        
        # Use a fixed heuristic construction to check feasibility
        h_test = best_h.copy()
        # Slightly tweak to see if it works
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        c5_test = compute_c5(h_test)
        if c5_test <= c_mid:
            c_max = c_mid
            best_h = h_test.copy()
        else:
            c_min = c_mid
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
