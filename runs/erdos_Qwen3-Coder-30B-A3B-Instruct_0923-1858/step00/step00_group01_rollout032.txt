import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdos constant C5 from a step function h"""
    # Compute overlap integral
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 20

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_best_h = h.copy()
    current_best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    max_iter = 1000
    lr = 0.01
    lr_decay = 0.1
    
    def project_h(h, dx):
        """Project h to satisfy the sum constraint and bounds"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-10:
            scale = 1.0 / h_sum
            h *= scale
        return h
    
    def compute_gradient(h, dx):
        """Compute gradient of C5 w.r.t h using finite differences"""
        # Evaluate current value
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient by perturbing each element
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, dx)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (corr_plus[max_idx] - max_val) / eps
        
        return grad
    
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        # Compute gradient
        grad = compute_gradient(h, dx)
        # Update with projected gradient descent
        h = h - lr * grad
        h = project_h(h, dx)
        
        # Evaluate and potentially update the best solution
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < current_best_c5:
            current_best_c5 = c5
            current_best_h = h.copy()
        
        # Decrease learning rate at intervals
        if iteration % 100 == 0 and iteration > 0:
            lr *= lr_decay
            
        iteration += 1
        
        # Optionally increase resolution
        if iteration % 200 == 0 and iteration > 0 and n_points * 2 <= 1000:
            # Double resolution
            n_points *= 2
            dx = 2.0 / n_points
            # Interpolate h on finer grid
            old_x = np.linspace(0, 2, len(h))
            new_x = np.linspace(0, 2, n_points)
            h = np.interp(new_x, old_x, h)
            h = project_h(h, dx)
            # Reset learning rate for new resolution
            lr = 0.01
    
    # Final check of constraints
    assert abs(np.sum(current_best_h) * dx - 1.0) < 1e-6, "Sum constraint violated"
    assert np.all(current_best_h >= 0) and np.all(current_best_h <= 1), "Bounds constraint violated"
    
    return (current_best_h, current_best_c5, len(current_best_h))
