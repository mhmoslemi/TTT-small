import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """ Evaluate C5 for given h """
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to C5
    c5 = np.max(corr) * dx
    return c5

def create_initial_population(n_points, pop_size, initial_h):
    """ Create initial population starting with initial_h if available """
    population = []
    if initial_h is not None and len(initial_h) == n_points:
        population.append(initial_h.copy())
    
    # Fill the rest with random feasible individuals
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Scale to meet sum constraint
        h = np.clip(h, 0, 1)  # Clip to [0,1]
        population.append(h)
    
    return population

def select_parents(population, fitnesses, tournament_size=5):
    """ Tournament selection """
    parents = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        parents.append(population[winner_idx])
    return parents

def blend_crossover(parent1, parent2, alpha=0.5):
    """ Blend crossover """
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(h, mutation_rate=0.1, epsilon=0.1):
    """ Mutate h """
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, epsilon)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n / 2
    scale_factor = target_sum / current_sum
    h *= scale_factor
    return h

def genetic_algorithm(n_points, initial_h, budget_s=600):
    """ Main genetic algorithm to minimize C5 """
    # Parameters
    pop_size = 30
    max_generations = 1000
    mutation_rate = 0.1
    elite_size = pop_size // 10  # Top 10% as elite

    # Initialize population
    population = create_initial_population(n_points, pop_size, initial_h)
    
    # Track best solution
    global_best_h = None
    global_best_c5 = float('inf')
    
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h) for h in population]
        
        # Update global best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < global_best_c5:
            global_best_c5 = fitnesses[best_idx]
            global_best_h = population[best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Create offspring (new population)
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and Mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            offspring1, offspring2 = blend_crossover(parent1, parent2)
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            new_population.extend([offspring1, offspring2])
        
        # Trim population to exact size
        population = new_population[:pop_size]
    
    # Return final solution
    return global_best_h, global_best_c5

        h0 = h0 * (n_points / (2 * np.sum(h0)))  # Scale
    
    # Run genetic algorithm
    best_h, c5 = genetic_algorithm(n_points, h0, budget_s)
    
    return best_h, c5, n_points
