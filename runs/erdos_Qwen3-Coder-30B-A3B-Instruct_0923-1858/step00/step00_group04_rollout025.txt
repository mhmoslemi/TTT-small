import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.full(n_points, 0.5)
        h[::2] = 1.0
        h[1::2] = 0.0
        # Scale to sum = n_points / 2
        h *= (n_points / 2) / np.sum(h)

    # Ensure it sums to exactly n_points / 2
    h *= (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    perturb_interval = 100
    epsilon = 0.01
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.99:
            break
            
        # Compute correlations
        cor = np.correlate(h, 1-h, mode="full")
        # Normalize by dx
        C5 = np.max(cor) * dx
        
        # Early exit if we hit target
        if C5 <= 0.3807:
            return h.tolist(), C5, n_points
            
        # Find where maximum overlap occurs
        max_idx = np.argmax(cor)
        max_k = max_idx - (len(h) - 1)
        
        # Adjust h to decrease overlap at k
        new_h = h.copy()
        
        # Effectively shift indices that cause high overlap
        for i in range(n_points):
            # When we have h[i] of mass and we add shift k, we reduce that
            # To reduce this, we decrease h[i] where it contributes to the overlap
            # The contribution to overlap at position k is h[i] * (1 - h[(i+k)%n_points])
            j = (i + max_k) % n_points
            new_h[i] = max(0, new_h[i] - epsilon * h[i])
            new_h[i] = min(1, new_h[i])
            
        # Ensure the sum constraint and clamp
        new_h = np.clip(new_h, 0, 1)
        # Project to sum = n_points / 2
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h *= (n_points / 2) / current_sum
            
        # Sometimes do a global perturbation
        if iteration % perturb_interval == 0:
            # Redistribute mass to reduce global overlap
            # Find indices with highest local overlap effects  
            # Smear current to help convergence
            perturb_amount = 0.01 * np.random.rand(n_points)
            perturb_amount = perturb_amount - np.mean(perturb_amount)
            new_h += perturb_amount
            
            # Clamp and reproject
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h)
            if current_sum > 0:
                new_h *= (n_points / 2) / current_sum
            
        h = new_h
        C5 = evaluate_C5(h, dx)
        
        if C5 < best_C5:
            best_h = h.copy()
            best_C5 = C5
    
    return best_h.tolist(), best_C5, n_points

def evaluate_C5(h, dx):
    """Helper to compute C5 from h"""
    cor = np.correlate(h, 1-h, mode="full")
    return np.max(cor) * dx
