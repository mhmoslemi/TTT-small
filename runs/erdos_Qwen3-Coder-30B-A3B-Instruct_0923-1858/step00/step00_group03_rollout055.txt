import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute all overlaps using full correlation
    corr = correlate(h, 1 - h, mode='full') * dx
    # We only consider valid overlaps (k from -n+1 to n-1)
    # but correlate returns len(h)*2 - 1 values
    n = len(h)
    # The overlaps are from k=-n+1 to k=n-1
    # For each k, the overlap is computed as convolution sum
    # k=0 is the maximum of the correlation, which is always the biggest
    # but we go through all valid k to get the maximum
    max_overlap = -np.inf
    for k in range(-n+1, n): # This covers all valid shifts
        # For k >= 0, we compute overlap for h[i] * (1 - h[i+k])
        # For k < 0, we compute overlap for h[i+k] * (1 - h[i])
        if k >= 0:
            # Sum h[i]*(1-h[i+k]) for i such that i+k < n, i < n
            # This becomes sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]
            # But we want to compute directly
            # For k = 0: h[i]*(1-h[i]) = h[i]-h[i]^2 -> overlap = sum(h) - sum(h^2)
            # These can be computed in a loop over i, but much more efficiently as standard correlation
            # Let's just get the max directly
            overlap = corr[n - 1 + k] # here n-1 is the index for k=0
            if overlap > max_overlap:
                max_overlap = overlap
        else:
            # k < 0
            overlap = corr[n - 1 + k]  # k should be negative, so n-1+k might be 0 or more (but < n-1)
            if overlap > max_overlap:
                max_overlap = overlap
    return max_overlap

def project_h(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    # Clamp to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum requirement
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    if abs(current_sum - target_sum) > 1e-10:
        # Compute deviation
        diff = (target_sum - current_sum) / n_points
        h += diff
        # Clamp again after adjustment
        h = np.clip(h, 0, 1)
    return h

        # Project to valid sum
        dx = 2.0 / n_points
        h = project_h(h, n_points, dx)  # This ensures sum(h)*dx == 1.0

    # Evaluate initial value
    dx = 2.0 / n_points
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    start_time = time.time()
    
    # Parameters for optimization
    lr = 0.1
    decay = 0.9
    tolerance = 1e-5  # Stopping condition on improvement
    last_c5 = best_c5 + 10*tolerance  # To enter the loop
    
    iterations = 0
    while time.time() - start_time < budget_s - 1.0:  # Leave 1s for cleanup
        iterations += 1
        # We compute the maximum overlap and its corresponding k
        # Efficiently compute correlation
        corr_full = correlate(h, 1 - h, mode='full') * dx
        # corr_full has length 2*n_points - 1
        # k = 0 is at index n-1
        # The overlap values for k = -n+1 to n-1 are at indices 0 to 2*n-2
        k_range = range(-n_points+1, n_points)
        # Find maximum overlap index
        max_idx = np.argmax(corr_full)
        k_best = k_range[max_idx]
        # Compute gradient using 'manual' finite difference
        # For each h[i], gradient is (1 - h[i+k_best]) if i+k_best is valid, else 0
        # For h[i+k_best], gradient is -h[i]
        # But to compute derivative of max -> we only take the derivative of that max-overlap term
        # For all i, the contribution to max overlap (at k=k_best) is h[i]*(1-h[i+k_best])
        # thus gradient w.r.t h[i] is (1 - h[i+k_best]) and w.r.t h[i+k_best] is -h[i]
        # But since derivative of max over i is not straightforward, we use a modified trick
        
        # Let's simplify and compute an explicit gradient approximation
        # Let's compute for each i the gradient WRT that entry to the max overlap term
        grad = np.zeros_like(h)
        if k_best >= 0:
            # h[i] * (1 - h[i + k_best]) 
            for i in range(n_points - k_best):
                grad[i] += (1 - h[i + k_best])
            for i in range(k_best, n_points):
                grad[i] -= h[i - k_best]
        else:   # k_best < 0
            # h[i - (-k_best)] * (1 - h[i]) = h[i + |k_best|] * (1 - h[i])
            for i in range(-k_best, n_points):
                grad[i] -= h[i + k_best]
            for i in range(n_points + k_best):
                grad[i] += (1 - h[i - k_best])                    
                
        # Apply gradient step
        h = h - lr * grad
        # Project to feasible region
        h = project_h(h, n_points, dx)
        
        # Evaluate new value
        new_c5 = evaluate_h(h, dx)
        # Update best if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Check for convergence
        if abs(last_c5 - new_c5) < tolerance:
            break
        last_c5 = new_c5
        
        # Decay learning rate
        if iterations % 100 == 0:
            lr *= decay

    # Final check, also return the best solution found
    final_c5 = evaluate_h(best_h, dx)
    
    # Now let's do a small binary search-like operation by rounding the solution to binary values
    # Iterate over rounds
    refine_h = best_h.copy()
    for _ in range(5):  # A few rounds of projection to binary
        # Use a simple greedy approach: round optimally
        # if uniform heuristic gives good results:
        # create a binary rough solution
        # Let's do slightly better by rounding towards a binary structure
        sum_target = n_points / 2
        
        # Simple projection to binary (majority rule)
        # Compute the sum so we know how many 1s we should have
        # First get a binary version
        h_binary = np.round(refine_h)
        diff = sum_target - np.sum(h_binary)
        if abs(diff) > 0.5:  # if difference > 0.5
            # Move one 1 to 0 or 0 to 1
            indices = np.argsort(np.abs(refine_h - 0.5))
            if diff > 0:
                # change 0s to 1s
                for i in indices:
                    if h_binary[i] == 0 and diff > 0:
                        h_binary[i] = 1
                        diff -= 1
                    if diff <= 0:
                        break
            else:
                # change 1s to 0s
                for i in indices:
                    if h_binary[i] == 1 and diff < 0:
                        h_binary[i] = 0
                        diff += 1
                    if diff >= 0:
                        break
        refine_h = h_binary
        
    # Final evaluation after rounding
    final_c5_rounded = evaluate_h(refine_h, dx)
    if final_c5_rounded < final_c5:
        final_c5 = final_c5_rounded
        best_h = refine_h
    return (best_h.tolist(), final_c5, n_points)
