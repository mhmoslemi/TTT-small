def evaluate_erdos_solution(h, n_points):
    # Compute the overlap integral C5
    dx = 2.0 / n_points
    # Using convolution with flipped h
    overlap = signal.correlate(h, 1 - h, mode='full')
    # The full convolution result is 2*n_points - 1 elements
    # We want the maximum value (excluding the trivial overlaps due to signal length)
    max_overlap = np.max(overlap) * dx
    return max_overlap
