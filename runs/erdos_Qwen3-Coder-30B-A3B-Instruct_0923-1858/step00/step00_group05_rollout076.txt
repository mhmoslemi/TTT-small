from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value of the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx
