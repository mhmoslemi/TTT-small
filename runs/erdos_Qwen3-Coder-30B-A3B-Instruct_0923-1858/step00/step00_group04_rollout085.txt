import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum correlation value divided by n gives the C5 bound
    c5 = np.max(corr) * dx
    return c5

def rescale_h(h, target_sum):
    """Rescale h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * target_sum / current_sum

def make_feasible(h, target_sum):
    """Ensure h is feasible with bounds and sum constraint."""
    h = np.clip(h, 0, 1)
    h = rescale_h(h, target_sum)
    return h

def crossover_h(p1, p2, rate=0.7):
    """Perform blend crossover between two parents."""
    n = len(p1)
    if random.random() > rate:
        return np.copy(p1)
    # Blend some segments
    mask = np.random.rand(n) < 0.5
    offspring = np.where(mask, p1, p2)
    return offspring

def mutate_h(h, rate=0.1, epsilon=0.1):
    """Mutate h by small random perturbations."""
    n = len(h)
    if random.random() > rate:
        return np.copy(h)
    # Add noise to some positions
    mask = np.random.rand(n) < 0.2
    noise = np.random.normal(0, epsilon, n)
    h = h + noise * mask
    return make_feasible(h, np.sum(h))

def genetic_algorithm(initial_h, n_points, budget_s):
    """Run genetic algorithm to minimize C5."""
    start_time = time.time()
    population_size = 30
    num_generations = 1000
    elite_size = int(population_size * 0.1)
    tournament_size = 5
    
    # Initialize population
    population = []
    target_sum = n_points / 2.0
    if initial_h is not None:
        population.append(make_feasible(initial_h, target_sum))
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = make_feasible(h, target_sum)
            population.append(h)
    else:
        # Create starting population with alternating pattern
        h = np.tile([0.5, 0.5], n_points // 2 + 1)[:n_points]
        h = make_feasible(h, target_sum)
        population.append(h)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = make_feasible(h, target_sum)
            population.append(h)
    
    global_best = None
    global_best_c5 = float('inf')
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_h(h) for h in population]
        ranked = sorted(zip(population, fitness), key=lambda x: x[1])
        population = [x[0] for x in ranked]
        fitness = [x[1] for x in ranked]
        
        # Track best
        if fitness[0] < global_best_c5:
            global_best_c5 = fitness[0]
            global_best = np.copy(population[0])
        
        # Selection and reproduction - tournament selection
        new_pop = []
        # Elite
        for i in range(elite_size):
            new_pop.append(np.copy(population[i]))
        
        # Generate offspring
        while len(new_pop) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            parent2 = population[np.random.randint(population_size)]
            
            offspring = crossover_h(parent1, parent2)
            offspring = mutate_h(offspring)
            new_pop.append(offspring)
        
        population = new_pop
        
    return global_best, global_best_c5

    np.random.seed(seed)
    n_points = 42  # Use the provided initial construction size if available
    
    # Check if initial_h_values is available
    initial_h = None

    # Run GA
    best_h, c5_bound = genetic_algorithm(initial_h, n_points, budget_s)
    
    # Final validation
    if best_h is not None:
        c5_bound = evaluate_h(best_h)
    
    if best_h is None:
        # Fallback if GA failed
        best_h = np.full(n_points, 0.5)
        best_h = make_feasible(best_h, n_points / 2)
        c5_bound = evaluate_h(best_h)
        
    return (best_h.tolist(), c5_bound, n_points)
