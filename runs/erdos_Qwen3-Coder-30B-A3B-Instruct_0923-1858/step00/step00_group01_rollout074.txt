        # Compute cross-correlation with 1-h
        corr = np.correlate(h, 1 - h, mode="full")
        # Returns maximum of all correlations (shifts)
        return np.max(corr) * dx
    
    def project_h(h):
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Scale to satisfy sum(h) * dx = 1
        total = np.sum(h) * dx
        if total > 0:
            h = h / total * 1.0
        return h

        best_h = h.copy()

        # Simple gradient-based optimization (suboptimal but effective)
        for i in range(max_iter):
            if time() - start_time > budget_s - 1:

            # Estimate gradient with finite differences
            epsilon = 1e-4
            grad = np.zeros_like(h)

                h_plus = h.copy()
                h_plus[j] += epsilon
                h_plus = project_h(h_plus)
                c5_plus = compute_c5(h_plus)

                h_minus = project_h(h_minus)

            # Gradient descent step
            lr = 0.01 * (0.99 ** i)
