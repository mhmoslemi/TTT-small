import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Compute the maximum overlap integral for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integral for each k
    overlaps = corr * dx
    # Return the maximum of those overlaps
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values.copy())
        n_points = len(h)
    else:
        # Default construction
        n_points = 42
        h = np.ones(n_points) * 0.5
        # Make sure sum is correct (n_points / 2)
        h[::2] = 1.0
        h[1::2] = 0.0
        # Scale to correct sum
        h = h * (n_points / 2) / np.sum(h)
    
    # Required so that sum(h) * dx = 1
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters for optimization
    max_iter = 10000
    perturb_freq = 100
    epsilon = 0.01  # Perturbation strength

    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlaps and find the maximum one
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        
        # Get index of maximum overlap
        max_idx = np.argmax(overlaps)
        max_k = max_idx - (n_points - 1)  # Convert from full correlation index to lag
        
        # Save previous h
        h_old = h.copy()
        
        # Adjust h: reduce h where k causes max overlap
        # We adjust around k such that h[i] * (1 - h[(i + k) % n_points]) contributes to max overlap
        # Identify indices where this contributes the most
        
        # Compute the contribution to overlap at max_k
        influence_indices = []
        for j in range(n_points):
            if j + max_k >= 0 and j + max_k < n_points:
                influence_indices.append((j, (j + max_k) % n_points))
        
        # For these position pairs, reduce h[j] by a small amount
        # and increase h[(j+k)%n_points] by same amount
        # We do this in a balanced way to minimize effect on sum
        delta = epsilon
        for j, k_idx in influence_indices[:len(influence_indices)//4]:
            if h[j] > 0 and h[k_idx] < 1:
                h[j] = max(0, h[j] - delta)
                h[k_idx] = min(1, h[k_idx] + delta)
        
        # Project to maintain sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6:
            # Adjust all elements scale-wise to fix sum
            scale_factor = target_sum / current_sum
            h = h * scale_factor
            
        # Ensure bounds
        h = np.clip(h, 0, 1)
        
        # Occasionally apply a random redistribution to escape local minima
        if i % perturb_freq == 0:
            # Randomly choose indices to redistribute from
            num_to_move = min(5, n_points // 10)
            from_indices = np.random.choice(n_points, size=num_to_move, replace=False)
            to_indices = np.random.choice(n_points, size=num_to_move, replace=False)
            
            # Select a small amount to move
            delta = epsilon / 2
            for i_from, i_to in zip(from_indices, to_indices):
                if h[i_from] > 0 and h[i_to] < 1:
                    move = min(delta, h[i_from], 1 - h[i_to])
                    h[i_from] -= move
                    h[i_to] += move
                    
            # Re-project to meet sum again
            current_sum = np.sum(h)
            if abs(current_sum - target_sum) > 1e-6:
                scale_factor = target_sum / current_sum
                h = h * scale_factor
            h = np.clip(h, 0, 1)
        
        # Evaluate and update best
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
