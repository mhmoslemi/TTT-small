import numpy as np
from scipy import optimize
import time
import math

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    if 'initial_h_values' in globals():

        for i in range(n_points):
            h[i] = 0.5 + 0.1 * np.sin(2 * np.pi * i / n_points)
        # Normalize
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    current_c5 = evaluate_erdos_solution(h, n_points)
    best_c5 = current_c5
    T = 0.1
    alpha = 0.99
    sigma = 0.1
    patience = 0
    max_patience = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s and patience < max_patience:
        iteration += 1
        # Perturb h
        delta_h = np.random.normal(0, sigma, n_points)
        h_candidate = h + delta_h
        # Project onto feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= (n_points / 2) / np.sum(h_candidate)
        # Compute new C5 value
        new_c5 = evaluate_erdos_solution(h_candidate, n_points)
        # Accept or reject
        if new_c5 < current_c5:
            h = h_candidate
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            patience = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(- (new_c5 - current_c5) / T):
                h = h_candidate
                current_c5 = new_c5
                patience = 0
            else:
                patience += 1
        # Update temperature and sigma
        T *= alpha
        if iteration % 100 == 0:
            sigma *= 0.9
    return (best_h.tolist(), best_c5, n_points)
