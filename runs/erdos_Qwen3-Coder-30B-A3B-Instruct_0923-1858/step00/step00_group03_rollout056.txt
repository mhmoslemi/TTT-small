import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Default construction: uniform step function with 70 points
        n_points = 70
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Adjust to make sum = 1
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Normalize to sum to n_points/2 for correct integration
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    # Gradient descent loop with adaptive step size
    max_iter = 10000
    alpha = 0.1
    tolerance = 1e-6
    
    current_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlap sums
        overlap = np.correlate(h, 1-h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient: ∂C5/∂h[i] = (1 - h[i+k_max]) for i in valid range
        grad_h = np.zeros_like(h)
        if k_max < n_points:
            # For h[i] term: partial derivative is (1 - h[i+k_max])
            for i in range(n_points):
                j = i + k_max
                if j < n_points:
                    grad_h[i] += (1 - h[j])
        elif k_max >= n_points:
            # For h[i+k_max-n_points] contribution
            shift = k_max - n_points
            for i in range(n_points - shift):
                grad_h[i] += (1 - h[i + shift])
        
        # Update h
        h = h - alpha * grad_h
        
        # Project onto [0,1] and sum constraint
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
            
        # Check for convergence
        new_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
        if abs(new_c5 - current_c5) < tolerance:
            break
        current_c5 = new_c5
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Adaptive step size
        if iteration % 100 == 0 and iteration > 0:
            alpha *= 0.9
            
    return (best_h, best_c5, n_points)
