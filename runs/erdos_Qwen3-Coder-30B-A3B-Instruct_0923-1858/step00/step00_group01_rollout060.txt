import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h, dx):
        h_rev = np.flip(1 - h)
        corr = correlate(h, h_rev, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def perturb_h(h, dx):
        # Evaluate current state
        corr = correlate(h, np.flip(1 - h), mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Identify the shift k where max occurs
        k = max_idx - (len(h) - 1)
        
        # Define overlapping regions
        left_start = max(0, k)
        left_end = min(len(h), len(h) + k)
        right_start = max(0, -k)
        right_end = min(len(h), len(h) - k)
        
        # Heuristic: focus on regions around max overlap
        overlap_indices = []
        if k >= 0:
            overlap_indices = list(range(left_start, left_end))
        else:
            overlap_indices = list(range(right_start, right_end))
            
        if not overlap_indices:
            # If no overlap, do a randomized perturbation
            i = np.random.randint(0, len(h))
            j = np.random.randint(0, len(h))
            if i == j:
                return h.copy()
            # Transfer from i to j
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer <= 0:
                return h.copy()
            fraction = np.random.uniform(0, 1)
            transfer = fraction * max_transfer
            new_h = h.copy()
            new_h[i] -= transfer
            new_h[j] += transfer
        else:
            # Distribute over overlap regions
            # Randomly select a pair of indices in overlap
            i = np.random.choice(overlap_indices)
            j = np.random.choice(overlap_indices)
            if i == j:
                return h.copy()
            # Transfer from i to j
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer <= 0:
                return h.copy()
            fraction = np.random.uniform(0, 1)
            transfer = fraction * max_transfer
            new_h = h.copy()
            new_h[i] -= transfer
            new_h[j] += transfer
        
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        
        # Renormalize to maintain constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        return new_h

    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Memory for stopping condition
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 2:
        iteration += 1
        
        new_h = perturb_h(best_h, dx)
        new_C5 = compute_C5(new_h, dx)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if last_improvement > 100:
            break

    if len(best_h) < 100:
        n_points_doubled = len(best_h) * 2
        dx_doubled = 2.0 / n_points_doubled
        new_h_doubled = np.interp(np.linspace(0, 2, n_points_doubled),

        new_h_doubled = np.clip(new_h_doubled, 0, 1)
        new_h_doubled = new_h_doubled * (n_points_doubled / 2) / np.sum(new_h_doubled)
        final_C5 = compute_C5(new_h_doubled, dx_doubled)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = new_h_doubled.copy()
    
    # Final validation
    final_C5 = compute_C5(best_h, dx)
    return (best_h.tolist(), final_C5, len(best_h))
