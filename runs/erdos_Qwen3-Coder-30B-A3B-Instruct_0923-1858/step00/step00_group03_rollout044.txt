import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial h_values if available, otherwise create a binary sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.round(h).astype(int)
        n_points = len(h)
    else:
        n_points = 70
        # Create a reasonable starting binary vector
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure sum is exactly n_points/2
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum > target_sum:
        # Flip some 1s to 0s
        flip_indices = np.where(h == 1)[0]
        np.random.shuffle(flip_indices)
        for idx in flip_indices[:current_sum - target_sum]:
            h[idx] = 0
    elif current_sum < target_sum:
        # Flip some 0s to 1s
        flip_indices = np.where(h == 0)[0]
        np.random.shuffle(flip_indices)
        for idx in flip_indices[:target_sum - current_sum]:
            h[idx] = 1
            
    dx = 2.0 / n_points
    max_overlap = -1
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        if iter_count % 500 == 0:
            # Check time remaining
            elapsed = time.time() - start_time
            if elapsed > 0.9 * budget_s:
                break
                
        # Compute all overlaps at once
        shifted = 1 - h
        overlaps = correlate(h, shifted, mode='full')
        overlaps = overlaps * dx  # scale by dx
        overlap_indices = np.arange(len(overlaps))
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        shift_k = overlap_indices[max_overlap_idx] - (n_points - 1)  # compute actual shift
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Reset if no improvement for too long
            no_improvement_count = 0
            h = best_h.copy()
            continue
        
        # Compute contribution of each position to the maximum overlap
        contributions = np.zeros(n_points)
        for i in range(n_points):
            if shift_k >= 0:
                if i + shift_k < n_points:
                    contributions[i] = h[i] * (1 - h[i + shift_k])
            else:
                if i < n_points + shift_k:
                    contributions[i] = h[i] * (1 - h[i - shift_k])
                    
        # Flatten the overlaps over all possible shifts for better targeting
        # Compute impact of flipping any bit
        best_flip = None
        best_impact = 0
        
        for i in range(n_points):
            old_val = h[i]
            new_val = 1 - old_val
            if old_val == new_val:
                continue
            temp_h = h.copy()
            temp_h[i] = new_val
            new_sum = np.sum(temp_h)
            if new_sum != target_sum:
                continue  # skip if not maintaining constraint
            # Compute impact on max overlap
            shifted_new = 1 - temp_h
            overlaps_new = correlate(temp_h, shifted_new, mode='full')
            overlaps_new = overlaps_new * dx
            new_max = np.max(overlaps_new)
            
            if max_overlap - new_max > best_impact:
                best_impact = max_overlap - new_max
                best_flip = (i, new_val)
        
        # Also consider flipping a 0 to 1 to maintain sum if needed
        if best_flip is None:
            for i in range(n_points):
                if h[i] == 0:  # we can flip this 0 to 1
                    temp_h = h.copy()
                    temp_h[i] = 1
                    # Find an index j to flip 1 to 0
                    ones_idx = np.where(temp_h == 1)[0]
                    if len(ones_idx) > 0:
                        j = ones_idx[0]  # take any
                        temp_h[j] = 0
                        new_sum = np.sum(temp_h)
                        if new_sum == target_sum:
                            shifted_new = 1 - temp_h
                            overlaps_new = correlate(temp_h, shifted_new, mode='full')
                            overlaps_new = overlaps_new * dx
                            new_max = np.max(overlaps_new)
                            
                            if max_overlap - new_max > best_impact:
                                best_impact = max_overlap - new_max
                                best_flip = (i, 1, j, 0)
                                
        if best_flip is not None:
            if len(best_flip) == 2:
                i, new_val = best_flip
                h[i] = new_val
            else:  # 4-tuple
                i, new_val, j, new_val2 = best_flip
                h[i] = new_val
                h[j] = new_val2
                
    return (best_h.tolist(), best_C5, n_points)
