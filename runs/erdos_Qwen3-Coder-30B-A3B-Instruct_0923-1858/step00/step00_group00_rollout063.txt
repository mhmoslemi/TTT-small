def evaluate_erdos_solution(h_values):
    """Compute C5 for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum value over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def solve_feasibility(h0, C_target, n_points, dx):
    """Try to construct a feasible solution subject to constraint C_target"""
    # Use simplified gradient-based method for speed
    # Initialize with a copy of the starting point
    h = h0.copy()
    
    # Objective: minimize max_k int h(x) * (1 - h(x + k)) dx, subject to:
    # - 0 <= h[i] <= 1  
    # - sum(h) = n/2
    # - all overlaps <= C_target

    h = h * (n_points / 2) / np.sum(h)

            break
        # Evaluate current overlap (max convolved value)
        conv = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv) * dx
        
        if max_overlap <= C_target:
            return h.copy()
        
        # Gradient descent step
        # The gradient is complex, so we linearize around the current point
        # Use simple descent in direction of largest negative gradient
        # This is not exact but should get us closer
        h = h + 0.01 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        # Re-normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally, project to valid overlaps
        if _ % 5 == 0:
            # Relax to meet C_target
            # Clamp the overlap with a simple adjustment
            # The negative part of convolution reflects if max_overlap is too high
            conv_full = np.correlate(h, 1 - h, mode='full')
            if np.max(conv_full) * dx > C_target:
                # If overlap too high, try to reduce it
                # Use simple gradient of overlap w.r.t. h
                # Approximate the gradient based on average behavior
                # A simple projection: reduce h where max overlap occurs
                # This is not exact but gets us near feasible solutions
                max_i = np.argmax(conv_full)
                h = h.copy()
                # Clamp values near adjustment zone
                adjust_zone = 5  # Around max_i, change around it
                # We're doing a greedy approach to seeing what changes would reduce overlap
                # We already have too much overlap, so (for computing) use relax policy
                h = np.clip(h + 0.01 * (0.5 - h), 0, 1)
                # Re-normalize
                h = h * (n_points / 2) / np.sum(h)
                
    return h if evaluate_erdos_solution(h) <= C_target else None
