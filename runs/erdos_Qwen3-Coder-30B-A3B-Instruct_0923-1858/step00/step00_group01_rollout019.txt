import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the C5 bound for a given h."""
    # Compute the full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is over all lags k, we want the maximum value
    # Since we're looking at all k, just use the maximum correlation value in the full result
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(h, dx):
    """Ensure h satisfies the constraints."""
    # Ensure all h values are in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust so that sum(h) * dx == 1
    target_sum = 1.0 / dx
    current_sum = np.sum(h)
    if np.abs(current_sum - target_sum) > 1e-10:
        # Scale h
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, n_population, initial_h=None):
    """Initialize a population of h vectors."""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        n_population -= 1
    # Fill with random vectors
    for _ in range(n_population):
        h = np.random.rand(n_points)
        h = repair_h(h, 2.0 / n_points)
        population.append(h)
    return population

def select_elite(population, fitnesses, elite_ratio=0.2):
    """Return the elite individuals."""
    # Sort by fitness (ascending, since lower is better)
    sorted_indices = np.argsort(fitnesses)
    elite_count = max(1, int(len(population) * elite_ratio))
    return [population[i] for i in sorted_indices[:elite_count]]

def crossover(parent1, parent2, crossover_rate=0.5):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Simple uniform crossover
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, mutation_rate=0.05, delta=0.1):
    """Mutate a h vector."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            # Add noise with bounded perturbation
            h[i] += np.random.normal(0, delta)
            # ReClip to [0,1] and re-adjust
    h = repair_h(h, 2.0 / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # If we have an initial solution, we can use it
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = repair_h(initial_h, 2.0 / len(initial_h))
    else:
        initial_h = None
    
    # Start with not too many points for fast search
    n_points = 47
    dx = 2.0 / n_points
    n_population = 50
    
    # Initialize population
    population = initialize_population(n_points, n_population, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Main optimization loop
    gen = 0
    no_improvement = 0
    while True:
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate(h, dx)
            fitnesses.append(c5)
        fitnesses = np.array(fitnesses)
        
        # Find best
        min_idx = np.argmin(fitnesses)
        best_in_gen = fitnesses[min_idx]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[min_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Terminate if no improvement
        if no_improvement > 50:
            break

        # Selection
        elites = select_elite(population, fitnesses, 0.2)
        
        # Generate new population
        new_population = elites[:]
        while len(new_population) < n_population:
            # Pick two parents
            p1, p2 = np.random.choice(elites, 2, replace=False)
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            new_population.extend([c1, c2])
            
        # Trim to proper population size
        population = new_population[:n_population]
        gen += 1

    if best_h is None:
        # Fallback solution
        h = np.array(initial_h_values) if initial_h is not None else np.random.rand(n_points)
        h = repair_h(h, dx)
        best_h = h
        best_c5 = evaluate(h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
