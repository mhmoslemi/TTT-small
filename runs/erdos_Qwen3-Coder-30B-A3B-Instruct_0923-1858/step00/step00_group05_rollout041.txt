def evaluate_h(h_values, dx):
    """Evaluate the maximum overlap integral for given h values"""
    # Compute cross-correlation h * (1 - h) with full mode

    # The maximum is at the center (corresponding to lag k=0)
    # But we need to find the maximum over all lags

    return max_corr * dx
