import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    return h

def crossover(parent1, parent2, alpha=0.5):
    child = alpha * parent1 + (1 - alpha) * parent2
    return project_to_feasible(child, len(parent1))

def mutate(h, mutation_rate=0.1, noise_scale=0.01):
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_scale, len(h))
    h = h + noise * mask
    return project_to_feasible(h, len(h))

def tournament_selection(population, fitnesses, k=5):
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 30
    population = []
    fitnesses = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Fill rest of population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    for h in population:
        fitnesses.append(evaluate(h, dx))
    
    # Track best
    best_idx = np.argmin(fitnesses)
    global_best = population[best_idx].copy()
    global_best_fitness = fitnesses[best_idx]
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 200:
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if fitnesses[0] < global_best_fitness:
            global_best = population[0].copy()
            global_best_fitness = fitnesses[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Create new population
        new_population = [population[0]]  # Elite
        new_fitnesses = [fitnesses[0]]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < 0.7:  # crossover probability
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            child = mutate(child)
            
            new_population.append(child)
            new_fitnesses.append(evaluate(child, dx))
            
        population = new_population
        fitnesses = new_fitnesses
        
        generation += 1
        
        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_fitness = evaluate(global_best, dx)
    
    return (global_best.tolist(), final_fitness, n_points)
