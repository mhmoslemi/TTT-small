import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute the overlap integral for all shifts
    conv = correlate(h, 1 - h, mode='full')
    # Only the valid shifts (from -n+1 to n-1)
    n = len(h)
    valid_shifts = np.arange(-n + 1, n)
    # The valid range of indices in the cross-correlation for our purposes
    start_idx = n - 1
    end_idx = 2 * n - 2
    c5_vals = conv[start_idx:end_idx] * dx
    # Return max over all valid shifts
    return np.max(c5_vals)

def binary_conversion(h, n_points):
    # Convert to binary by rounding and adjust sum
    h_rounded = np.round(h).astype(int)
    current_sum = np.sum(h_rounded)
    target_sum = n_points // 2
    
    if current_sum == target_sum:
        return h_rounded
    
    diff = target_sum - current_sum
    
    if diff > 0:
        # Need to flip some 0s to 1s
        # Flip the ones that increase the overlap the least
        # For simplicity, flip the first diff zeros to ones
        zeros_idx = np.where(h_rounded == 0)[0]
        for i in range(min(diff, len(zeros_idx))):
            h_rounded[zeros_idx[i]] = 1
    else:
        # Need to flip some 1s to 0s
        # Flip the ones that decrease overlap the least - just the first ones
        ones_idx = np.where(h_rounded == 1)[0]
        for i in range(min(-diff, len(ones_idx))):
            h_rounded[ones_idx[i]] = 0

    return h_rounded

def neighbor_flip(h, n_points):
    # Find indices where h[i] = 1 and h[j] = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy(), 0.0
    
    # Select a random 1 and 0
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    
    # Create new h
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    
    return h_new, (i, j)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Start from initial if available, else generate simple one
    if 'initial_h_values' in globals():

        h = binary_conversion(h, n_points)
    else:
        # Start with uniform 1s in the first half
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    current_iter = 0
    
    while time.time() - start_time < budget_s - 1 and current_iter < max_iter:
        # New candidate by flipping a 1 to 0 and a 0 to 1
        h_new, changes = neighbor_flip(h, n_points)
        
        if len(changes) == 0:
            break
            
        c5_new = evaluate_c5(h_new, dx)
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final binary adjustment for exact constraint
    best_h = binary_conversion(best_h, n_points)
    final_c5 = evaluate_c5(best_h, dx)
    
    # Return (h_values, c5_bound, n_points)
    return (best_h.tolist(), final_c5, n_points)
