        corr = correlate(h, 1 - h, mode="full")

        # Check for convergence
        if abs(c5 - last_c5) < tolerance:
            break
        last_c5 = c5

        grad = np.zeros(n_points)

            eps = 1e-4
            h_plus = h.copy()
            h_minus = h.copy()
            
            # Clip to bounds before updating
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus[i] = max(0.0, h_minus[i] - eps)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) * dx / (2 * eps)

        # Update with gradient descent
        h = h - step_size * grad

        h_sum = np.sum(h)

    # Final evaluation
    corr_final = correlate(h, 1 - h, mode="full")
    c5_final = np.max(corr_final) * dx
