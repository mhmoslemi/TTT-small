import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Returns the C5 value for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1 - np.array(h_values), mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h):
    """Projects h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total > 0:
        h = h * (len(h) / 2) / total
    return h

def initialize_population(n_points, pop_size, initial=None):
    """Initialize population with feasible candidates."""
    population = []
    if initial is not None:
        population.append(project_to_feasible(np.array(initial)))
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = project_to_feasible(h)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    idxs = np.random.choice(len(population), k)
    best = np.argmin([fitnesses[i] for i in idxs])
    return population[idxs[best]]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Perform blend crossover (BLX-0) on two parents."""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        a, b = parent1[i], parent2[i]
        range_ = abs(a - b) * alpha
        lower, upper = min(a, b) - range_, max(a, b) + range_
        child[i] = np.random.uniform(lower, upper)
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    """Mutate h with Gaussian noise."""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return project_to_feasible(h)

    gen = 0
    while time.time() - start_time < budget_s - 1 and no_improve_count < max_no_improve:

        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind) for ind in population]
        
        # Update best
        min_fitness = np.min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()

        else:

        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
