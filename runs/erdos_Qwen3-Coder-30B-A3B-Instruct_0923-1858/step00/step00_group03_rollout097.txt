import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    h *= (target_sum / current_sum)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        # The shifts go from -(n_points-1) to (n_points-1)
        # But we're only interested in shifts from 1 to (n_points-1)
        # The maximum overlap is for shift k in [1, n_points-1], 
        # with indices i in [0, n_points-1-k]
        # So we index the full correlation from n_points to 2*n_points-2
        overlaps = corr[n_points-1:2*n_points-1] * dx  # scale by dx
        max_overlap_idx = np.argmax(overlaps)
        k_star = max_overlap_idx + 1  # k.star from 1 to n_points-1
        
        # Find indices that contribute most to this overlap
        # for all i: h[i] * (1 - h[i + k_star])
        # So we want i such that both h[i] and h[i + k_star] 
        # are not too close to 0 or 1 to avoid degenerate cases
        contribs = h[:-k_star] * (1 - h[k_star:])
        max_contrib_idx = np.argmax(contribs)
        
        # Try to improve by modifying h[max_contrib_idx] and h[max_contrib_idx + k_star]
        # Increase the first, decrease the second (if possible)
        # But be careful not to make h too far from 0 or 1 to allow changes
        
        delta = 0.01
        
        # Try to move h[max_contrib_idx] up and h[max_contrib_idx+k_star] down
        new_h = h.copy()
        i = max_contrib_idx
        j = i + k_star
        
        # Check what changes are possible
        if new_h[i] < 1.0 and new_h[j] > 0.0:
            # Can increase i and decrease j
            new_h[i] = min(1.0, new_h[i] + delta)
            new_h[j] = max(0.0, new_h[j] - delta)
            
            # Normalize
            new_sum = np.sum(new_h)
            if abs(new_sum - target_sum) > 1e-10:
                # Redistribute
                diff = new_sum - target_sum
                adjustments = -diff / n_points
                new_h += adjustments
                if np.min(new_h) < 0:
                    # Rescale
                    new_h = np.clip(new_h, 0, 1)
                    new_sum = np.sum(new_h)
                    adjustments = (target_sum - new_sum) / n_points
                    new_h += adjustments
                elif np.max(new_h) > 1:
                    # Rescale
                    new_h = np.clip(new_h, 0, 1)
                    new_sum = np.sum(new_h)
                    adjustments = (target_sum - new_sum) / n_points
                    new_h += adjustments
            
            # Evaluate
            new_c5 = max(correlate(new_h, 1-new_h, mode='full')[n_points-1:2*n_points-1] * dx)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                h = new_h.copy()
            else:
                h = h.copy()  # Finish early
        else:
            # Otherwise, just try to maintain current best
            h = best_h.copy()
        
        iter_count += 1
        if iter_count % 1000 == 0:
            # Check if we've stagnated recently
            if time.time() - start_time > budget_s * 0.9:
                break
    
    # Final computation of the best solution
    final_c5 = max(correlate(best_h, 1-best_h, mode='full')[n_points-1:2*n_points-1] * dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
