import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 45  # Start with the provided initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 2.0 * i / (mid - 1) if mid > 1 else 0.0
        for i in range(mid, n_points):
            h[i] = 2.0 * (n_points - 1 - i) / (n_points - 1 - mid) if mid < n_points - 1 else 0.0
    
    # Ensure sum is correct
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Optimization parameters
    eta = 0.01
    epsilon = 0.05
    decay_every = 100
    perturb_every = 50
    
    iteration = 0
    last_improved = 0
    while time.time() - start_time < budget_s:
        # Compute current C5
        C5 = compute_C5(h, dx)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            last_improved = iteration
        
        # Compute gradient using correlation approach
        # Specifically, for each shift k, we compute:
        # d/dh[i] (h[i] * (1 - h[i+k])) = 1 - 2*h[i+k] for h[i] and h[i+k]
        # So we need to handle all such contributions
        
        # Simpler gradient: use correlations
        # For objective C5 = max_k (sum(h[i] * (1 - h[i+k])))  
        
        # Approximate gradient with finite differences for small steps
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * target_sum / h_plus_sum
            C5_plus = compute_C5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum > 0:
                h_minus = h_minus * target_sum / h_minus_sum
            C5_minus = compute_C5(h_minus, dx)
            
            grad[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Gradient step
        h = h - eta * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        
        # Apply perturbation occasionally 
        if iteration % perturb_every == 0 and iteration > 0:
            # Add small random perturbation to escape local minima
            perturb = np.random.normal(0, epsilon, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * target_sum / h_sum
        
        # Reduce learning rate occasionally
        if iteration % decay_every == 0 and iteration > 0:
            eta *= 0.95
        
        iteration += 1
        
        # Early termination check
        if iteration - last_improved > 500 and iteration > 1000:
            break
    
    # Final check
    final_C5 = compute_C5(best_h, dx)
    return (best_h.tolist(), final_C5, len(best_h))

def compute_C5(h, dx):
    """Compute the maximum overlap integral."""
    # Using full mode correlation to handle all shifts
    corr = correlate(h, 1 - h, mode="full")
    # Extract relevant portion (from -n+1 to n-1) which corresponds to shifts from -1 to 1
    mid = len(corr) // 2
    corr_vals = corr[mid - len(h) + 1 : mid + len(h)] 
    C5 = np.max(corr_vals) * dx
    return C5
