        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_values):
        # Ensure values are in [0,1]
        h_values = np.clip(h_values, 0, 1)
        # Rescale to sum(h) * dx = 1
        current_sum = np.sum(h_values) * dx
        if abs(current_sum) > 1e-10:
            h_values = h_values * (1.0 / current_sum)
        return h_values
    
    def gradient(h_values):
        # Compute correlation for all lags
        corr = np.correlate(h_values, 1 - h_values, mode='full')

        # Simplified gradient estimation using finite differences
