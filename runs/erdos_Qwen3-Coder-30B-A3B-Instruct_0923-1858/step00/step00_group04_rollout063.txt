import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute full cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def feasible_h(h, n_points):
    """Ensure h is feasible: values in [0,1] and sum equals n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population of step functions."""
    population = []
    if initial_h is not None:
        population.append(feasible_h(initial_h, n_points))
    else:
        # Initialize with alternating binary pattern
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[1::2] = 0.5
        h = feasible_h(h, n_points)
        population.append(h)
    
    # Fill remaining population with random feasible functions
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = feasible_h(h, n_points)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select one parent using tournament selection."""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover to create offspring."""
    offspring = alpha * parent1 + (1 - alpha) * parent2
    return offspring

def mutate(h, mut_rate=0.1, eps=0.05):
    """Apply small random mutations."""
    mask = np.random.rand(len(h)) < mut_rate
    delta = np.random.normal(0, eps, len(h))
    h = h + delta * mask
    return feasible_h(h, len(h))

    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    generation = 0

    while time.time() - start_time < budget_s and no_improvement_count < 200:
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_c5(h) for h in population]
        
        # Track best solution so far
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Selection
        new_population = []
        elite_count = max(1, pop_size // 10)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = blend_crossover(parent1, parent2)
            child = mutate(child, mut_rate=0.1, eps=0.05)
            new_population.append(child)

        population = new_population
        generation += 1

    return best_h, best_c5
