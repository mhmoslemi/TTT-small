import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / len(h)
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value is the C5 bound
    c5 = np.max(corr) * dx
    return c5

def repair_h(h):
    """Ensure h satisfies constraints: 0≤h≤1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    n_points = len(h)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        scale = target_sum / current_sum
        h *= scale
    return h

def crossover_h(h1, h2):
    """Create offspring by blending two parent sequences."""
    n = len(h1)
    # Simple uniform crossover - random point
    point = np.random.randint(1, n)
    child = np.empty_like(h1)
    child[:point] = h1[:point]
    child[point:] = h2[point:]
    # Repair to satisfy constraints
    child = repair_h(child)
    return child

def mutate_h(h, rate=0.05, mutation_strength=0.1):
    """Apply mutation to a sequence."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < rate:
            # Add Gaussian noise
            noise = np.random.normal(0, mutation_strength)
            h[i] += noise
    # Repair to satisfy constraints
    h = repair_h(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use the initial solution if available, otherwise create random
    n_points = 47
    if 'initial_h_values' in globals():

        h_init = repair_h(h_init)
    else:
        # Initial guess: uniform distribution scaled to sum = n_points/2
        h_init = np.ones(n_points) * 0.5
        h_init = repair_h(h_init)

    # Parameters
    population_size = 60
    generations = 200
    elite_fraction = 0.2
    mutation_rate = 0.05
    mutation_strength = 0.1

    # Initialize population
    population = [h_init]
    for _ in range(population_size - 1):
        h = h_init.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = repair_h(h)
        population.append(h)

    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h)

    # Evolutionary loop
    for gen in range(generations):
        # Check for timeout
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h)
            fitness.append(c5)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()

        # Select elite
        elite_size = max(2, int(elite_fraction * population_size))
        elites = population[:elite_size]

        # Generate new population through crossover and mutation
        new_population = elites[:]  # Keep elites
        while len(new_population) < population_size:
            # Randomly select two parents from elites
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1 = elites[idx1]
            parent2 = elites[idx2]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutation
            child = mutate_h(child, mutation_rate, mutation_strength)
            
            new_population.append(child)

        population = new_population

    return (best_h.tolist(), best_c5, n_points)
