import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise construct a symmetric initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
        # Add a small perturbation to make it non-trivial
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        # Normalize to sum to n_points / 2
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    T = 0.1  # Initial temperature
    alpha = 0.99  # Annealing factor
    sigma = 0.1  # Initial noise level
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s:
        # Perturb
        h_new = h + np.random.normal(0, sigma, n_points)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new, dx)
        
        # Accept or reject
        delta_C5 = C5_new - best_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Update temperature and noise level
        T *= alpha
        if no_improvement_count % 100 == 0:
            sigma *= 0.9
        
        # Termination condition
        if no_improvement_count > max_no_improvement:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Compute the maximum overlap using correlation
    conv = correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx
