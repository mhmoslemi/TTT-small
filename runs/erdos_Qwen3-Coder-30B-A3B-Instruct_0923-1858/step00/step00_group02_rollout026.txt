import numpy as np

    start_time = time.time()

        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2 / np.sum(h0))  # rescale
    
    h = h0.copy()
    dx = 2.0 / n_points
    max_iter = 20000
    learning_rate = 0.01
    decay_rate = 0.95
    perturb_freq = 50
    perturb_magnitude = 0.05
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute C5
        c5_val = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute gradient approximation
        # We take finite differences to estimate ∂C5/∂h[i]
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            if h_plus[j] > 1:
                h_plus[j] = 1
            h_plus = h_plus * (n_points / 2 / np.sum(h_plus))  # rescale
            c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode="full")) * dx
            
            h_minus = h.copy()
            h_minus[j] -= eps
            if h_minus[j] < 0:
                h_minus[j] = 0
            h_minus = h_minus * (n_points / 2 / np.sum(h_minus))  # rescale
            c5_minus = np.max(np.correlate(h_minus, 1 - h_minus, mode="full")) * dx
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2 / np.sum(h))  # rescale
        
        # Occasionally perturb to escape local minima
        if i % perturb_freq == 0:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2 / np.sum(h))  # rescale
        
        # Reduce learning rate occasionally
        if i % 100 == 0 and i > 0:
            learning_rate *= decay_rate
    
    # Final evaluation
    if best_h is None:
        best_h = h
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2 / np.sum(best_h))
    
    return (best_h.tolist(), float(final_c5), n_points)
