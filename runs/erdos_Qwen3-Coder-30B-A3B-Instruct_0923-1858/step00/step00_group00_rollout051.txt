import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from provided initial_h_values or create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure starting point satisfies constraint
    h = h * target_sum / np.sum(h)
    
    # Objective function to compute C5
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        # Only consider the relevant part of the correlation for max
        # The maximum is over all shifts from -n+1 to n-1
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Project h to feasible domain [0,1] and rescale to sum to target_sum
    def project(h_vals):
        h_proj = np.clip(h_vals, 0, 1)
        h_proj = h_proj * target_sum / np.sum(h_proj)
        return h_proj
    
    # Annealing schedule
    T0 = 0.1
    alpha = 0.99
    T = T0
    sigma = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve_count = 0
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        # Generate perturbation
        delta = np.random.normal(0, sigma, n_points)
        h_candidate = h + delta
        h_candidate = project(h_candidate)
        
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject
        dC = c5_candidate - best_c5
        if dC < 0 or np.random.rand() < np.exp(-dC / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        # Reduce temperature and sigma periodically
        if iteration % 100 == 0:
            T *= alpha
            sigma *= 0.9
        
        # Early exit if no improvement in many iterations
        if no_improve_count > 1000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
