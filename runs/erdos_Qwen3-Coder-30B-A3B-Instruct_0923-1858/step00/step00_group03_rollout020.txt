import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial guess
        n_points = 70
        h = np.ones(n_points, dtype=float) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to meet sum constraint
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative projection approach
    iterations = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        # Take only the valid overlaps (non-negative shifts)
        valid_overlaps = overlaps[len(h)-1:]
        k_star = np.argmax(valid_overlaps)
        max_overlap = valid_overlaps[k_star]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Find positions contributing most to this overlap
        # These are pairs (i, i+k_star) that contribute to max_overlap
        best_pairs = []
        max_contribution = -1
        
        for i in range(len(h) - k_star):
            pair_contribution = h[i] * (1 - h[i + k_star])
            if pair_contribution > max_contribution:
                max_contribution = pair_contribution
                best_pairs = [(i, i + k_star)]
            elif pair_contribution == max_contribution:
                best_pairs.append((i, i + k_star))
        
        # Pick one of the best pairs to adjust
        if best_pairs:
            i, j = best_pairs[0]
            
            # Attempt to reduce overlap by adjusting both values
            # Prioritize limiting the value that contributes most
            adjust_amount = min(h[i], 1.0 - h[j], 0.1)
            
            # Make adjustments
            h[i] -= adjust_amount
            h[j] += adjust_amount
            
            # Ensure all values stay within [0,1]
            h = np.clip(h, 0, 1)
            
            # Adjust to maintain sum constraint
            current_sum = np.sum(h)
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                # Distribute adjustment across all points proportionally
                h += diff / len(h)
      
        iterations += 1
    
    # Final evaluation
    final_overlaps = correlate(best_h, 1 - best_h, mode="full")
    valid_overlaps = final_overlaps[len(best_h)-1:]
    final_c5 = np.max(valid_overlaps) * dx
    
    # Re-check feasibility for final result
    if not np.all((best_h >= 0) & (best_h <= 1)):
        # This shouldn't happen but for safety
        best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-5:
        # Ensure correct sum
        best_h = best_h * target_sum / final_sum
    
    return (best_h.tolist(), final_c5, n_points)
