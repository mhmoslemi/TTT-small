import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute all correlations and return maximum
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points, dx):
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust sum to match constraint
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    if abs(current_sum - target_sum) > 1e-10:
        scale = target_sum / current_sum
        h *= scale
    return h

    best_c5 = evaluate_h(h, dx)
    
    # Optimizer parameters
    max_iter = 10000
    step_size = 0.001
    decay_rate = 0.9999
    tolerance = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Find dominant lag k* by computing all correlations
        c = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(c)
        k_star = k_star - (n_points - 1)  # Convert to lag
        
        # Compute gradient w.r.t. h for lag k*
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k_star < n_points:
                grad[j] = -1  # gradient of h[j] * (1 - h[j+k_star])
            if 0 <= j - k_star < n_points:
                grad[j] += 1  # gradient of h[j-k_star] * (1 - h[j])
                
        # Apply gradient
        h_new = h - step_size * grad
        
        # Project onto feasible region
        h_new = project_h(h_new, n_points, dx)
        
        # Check for convergence
        if np.allclose(h, h_new, atol=tolerance):
            break
            
        h = h_new
        step_size *= decay_rate

    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))
