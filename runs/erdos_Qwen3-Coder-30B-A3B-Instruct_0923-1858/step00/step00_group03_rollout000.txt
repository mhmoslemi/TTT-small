import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided construction or create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Simple starting point: uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    
    start_time = time.time()
    
    # Ensure the constraint ∫h = 1
    h_sum = np.sum(h) * dx
    h = h * (1.0 / h_sum)
    
    # Gradient descent
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals for all lags
        # Since h is step function, compute correlation with 1-h
        h1 = 1.0 - h
        corr = signal.correlate(h, h1, mode='full')
        corr = corr * dx  # Apply dx scaling
        
        # Find the maximum lag (domain [0,2] maps to indices [0, 2*n_points-1])
        # The maximum corresponds to the largest value in extent [-n_points+1, n_points-1]
        # But since we look at max of [0, 2], we consider only positive lags properly
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        # If lag is negative or too large, bring to valid range
        # Actual lags considered for [0,2]: max_lag in [0, 2]
        # If max_lag < 0, set to 0, or adjust it properly
        # The full convolution computed from 0 to 2*n_points - 2 indices
        # Each index k corresponds to lag_k = k - (n_points - 1)  
        # i.e. lag in [-n_points+1, n_points-1]
        # So we're looking for appropriate positive lag
        
        assert max_lag >= -n_points + 1 and max_lag <= n_points - 1
        
        # But practically we should find maximum where overlap is large,
        # which in common sense corresponds to some translation in [0, 2] interval 
        # This is input to direct optimization; we'll take the maximum which defines the dominant overlap

        # Get the gradient of overlap w.r.t. h for the dominant lag
        # This is a gradient of a correlation over a shifted pattern

        n_corr = len(corr)
        # Correlation result has size 2*n_points - 1
        # Align: for k = 0 to n_points - 1, lag = k - (n_points - 1) = k - n_points + 1
        # So, for a lag value l in range [-n_points+1, n_points-1],
        # we find index l + n_points - 1 for correlation output
        
        # pt k (lag) maps to index k + n_points - 1 in full correlation
        corr_max_idx = max_idx
        # Get the gradient contribution from this lag
        
        grad = np.zeros(n_points)
        
        # For a convolution h * (1-h) at lag k:
        # We have (h * f)(k) = sum h[i] * f[i-k] for all i 
        # f = 1 - h
        # At lag k = k_orig - (n_points - 1) => k_orig = k + n_points - 1
        # we derive d/dh[i] of sum_j h[j]*f[j-k] 
        # For j = i + k, this gives f[i] = 1 - h[i]
        
        # Alternative: symmetric approach using convolution derivatives
        # The gradient w.r.t. h for each term 
        # Correlation outputs: corr[i] = sum_j h[j] * (1-h)[j - (i - (n_points-1))] 
        # For each position j and corresponding correlation index i: 
        # corr[i] = sum_j h[j] * f[j - (i - (n_points-1))] = sum_j h[j] * (1 - h[j - (i - (n_points-1))])
        
        # To get gradient d/dh[l] of corr[i]:
        # There is a term h[l] * (1-h[l - (i - (n_points-1))])  
        # The derivative is (1 - h[l - (i - (n_points-1))]) + h[l] * (-df/dh[l])  
        # This is 1 - h[l - (i - (n_points-1))] when that term refers to h[l] itself
        # Let's implement more carefully:
        
        # Use direct derivation from definition of convolution
        # Take corr[k] = sum_j h[j] * (1 - h[j-k])  
        # For a fixed j = i, derivative w.r.t. h[i] = 1 - h[i-k]
        # i-k >= 0 so h[i-k] exists 
        # i-k <= n_points-1 so h[i-k] is within bounds
        # So d/dh[i] corr[k] = 1 - h[i-k] if i-k >= 0 and i-k < n_points
        # Since h indices can be viewed convolved with f = 1-h, we compute as part of top-down approach
        
        # Perhaps generalize: Fuer eine gegebene convolution c_k = sum_j h[j]*f[j-k]
        # d/dh[i] c_k = f[i-k] = 1 - h[i-k]  
        # k index points to lag = k - (n_points - 1) so lag_lag_index = k
        # Gradient is just 1- h[i-k] where that h[i-k] is valid

        # This just computes:
        # For each position i in h, compute d/dh[i] (max corr)
        # = sum over k (d/dh[i] corr[k]) = (1 - h[i-k]) where i-k >=0 and i-k < n_points 
        # We use vectorized gradient at logical index where the position contributes:
        
        # let's do direct gradient from one term: h[i]*f[i-k], where i = l_start, f = 1-h 
        # The derivative is like: for each i in 0..n_points-1,
        # find all k such that h[i] is in term k = i + k_shift => contribute to corr[k]
        # In full corr, if we have indices 0 to 2*n_points-2, index k corresponds to lag k-(n-1)
        # To find gradient of corr at index max_idx, we track: 
        #啥是i-k在h的边界？ 是译码时 ||
        # Practical gradient computation = 
        # grad[i] = sum_{k such that h[i] contributes to corr[k]} d/dh[i] corr[k]
        # Contribution from h[i] at term corr[...]:
        # If h[j]*f[j-k] contributes h[i] term, then j=i, so f[j-k] = f[i-k] = 1 - h[i-k]
        # point contributes to term corr[i+k], when  i+k = max_idx
        # So k = max_idx - i, and requires i+k >= 0, i+k < 2n-1  => 80% of points
            
        for i in range(n_points):
            if max_idx - i >= 0 and max_idx - i < n_points:
                grad[i] += -(1.0 - h[max_idx - i])  # Derivative of (1 - h[i-k]) wrt h[i]
            # Also need with wrt to a shifted version
            # h[i] can be a factor in corr: h[j] * f[j - k] where j = i 
            # This contributes to term at index = i + (i-k) = 2*i - k
            # So k = i + (index - 2*i) = index - i
            # Therefore, h[i] contributes to term with index = i + (j - i) = j
            # etc.
            # Better: just use central logic from signal.
            # mathematically: the gradient 
            # for correlation conv(h, 1-h), clearly:
            # for one outer index k=0..(2n-2), there is fixed contribution from some specific h[i]
            # For fixed k, rightmost 1 element in h contributes to h_k, whose value is f[i] = 1 - h[i]
            # So contribution to corr[k] is similar to a derivative at each element.
            # For each element h[i], we weight each instance of -h[i]
            # By number of times it contributes.
            # Recall grad[i] = d/dh[i] max corr taken over all lags
            # Since gradient computed only w.r.t. current maximizer, just 1 significant gradient component
            # We compute based on 
            # corr[k] = sum nof h[j] * (1-h[j-k]) 
            # d/dh[i] corr[k] = 1 - h[i-k]
            # We are taking derivative of the maximum among these.
            # Let it be max_k = k*
            # In this case, grad = grad_1 = (1-h[i-k*]) if i-k* >= 0 else 0

        # Apply gradient step
        h = h - step_size * grad
        
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to sum to 1
        h_sum = np.sum(h) * dx
        h = h * (1.0 / h_sum)
        
        # Optional: reduce step size occasionally
        if iteration % 500 == 0:
            step_size *= 0.95
            
    # Final evaluation
    h1 = 1.0 - h
    corr = signal.correlate(h, h1, mode='full')
    corr = corr * dx
    best_c5 = np.max(corr)
    
    return (h.tolist(), best_c5, n_points)
