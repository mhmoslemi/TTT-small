import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    # Constraints enforcement
    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Evaluate current solution
    def evaluate(h):
        correlations = correlate(h, 1 - h, mode='full')
        max_correlation = np.max(np.abs(correlations))
        return max_correlation * dx
    
    best_h = h.copy()
    best_C5 = evaluate(h)
    
    # Hyperparameters
    learning_rate = 0.01
    perturbation_rate = 0.1
    perturbation_freq = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Get current overlap
        correlations = correlate(h, 1 - h, mode='full')
        current_C5 = np.max(np.abs(correlations)) * dx
        
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
        
        # Compute gradient using finite difference
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
        
        # Normalize and apply gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
        step = learning_rate * grad
        h = h - step
        
        # Apply constraints
        h = enforce_constraints(h)
        
        # Perturbation
        if iteration % perturbation_freq == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = enforce_constraints(h)
        
        # Adjust learning rate
        if iteration % 50 == 0:
            learning_rate *= 0.99
            if learning_rate < 1e-6:
                learning_rate = 1e-6
                
        iteration += 1
    
    final_C5 = evaluate(best_h)
    
    return (best_h, final_C5, n_points)
