def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value in the correlation is the C5
    max_corr = np.max(corr)
    return max_corr * dx
