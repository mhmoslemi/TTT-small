import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h *= scale
    return h

def evaluate_c5(h, dx):
    # Cross-correlation of h and (1-h) with full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Maximum value
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        h = clip_and_rescale(h, 2.0 / n_points)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    lr = 0.01
    max_iter = 1000
    iter_count = 0
    
    # Gradient descent
    for i in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Evaluate current objective and gradient
        c5 = evaluate_c5(h, dx)
        
        # Use finite differences for gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = clip_and_rescale(h_plus, dx)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = clip_and_rescale(h_minus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient descent
        h = h - lr * grad
        h = clip_and_rescale(h, dx)
        
        # Update best solution if better
        c5_candidate = evaluate_c5(h, dx)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h.copy()
        
        # Adapt learning rate
        iter_count += 1
        if iter_count % 100 == 0:
            lr *= 0.9
        
    return (best_h.tolist(), best_c5, n_points)
