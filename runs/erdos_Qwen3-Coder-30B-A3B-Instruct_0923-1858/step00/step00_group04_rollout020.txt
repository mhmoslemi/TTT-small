import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(np.abs(corr)) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy sum constraint while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / 2)
    return h

def create_initial_population(n_points, pop_size, initial_h):
    """Create initial population, using initial_h if available."""
    population = []
    if initial_h is not None and len(initial_h) == n_points:
        population.append(initial_h.copy())
    else:
        population.append(np.full(n_points, 1.0 / 2))
    
    # Fill the rest with random candidates
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents."""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_scale=0.01):
    """Mutate h with Gaussian noise."""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_scale, len(h))
    h += noise * mask
    h = np.clip(h, 0, 1)
    return h
