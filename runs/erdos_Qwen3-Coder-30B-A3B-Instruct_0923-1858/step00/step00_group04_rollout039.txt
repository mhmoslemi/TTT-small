import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum correlation and multiply by dx
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_population(n_points, population_size, initial_h=None):
    """Initialize population of feasible step functions."""
    population = []
    
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Fill remaining population with random feasible functions
    for i in range(population_size - len(population)):
        # Create random vector
        h = np.random.rand(n_points)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = random.sample(range(len(population)), tournament_size)
    best_idx = indices[0]
    for idx in indices[1:]:
        if fitnesses[idx] < fitnesses[best_idx]:
            best_idx = idx
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Perform blend crossover between two parents."""
    n = len(parent1)
    child = np.zeros(n)
    for i in range(n):
        low = min(parent1[i], parent2[i])
        high = max(parent1[i], parent2[i])
        range_val = high - low
        child[i] = low - alpha * range_val + random.random() * (1 + 2 * alpha) * range_val
        # Clip to [0, 1]
        child[i] = max(0, min(1, child[i]))
    return child

def mutate(h, mut_rate=0.1, epsilon=0.1):
    """Mutate a step function with small random perturbations."""
    n = len(h)
    new_h = h.copy()
    
    for i in range(n):
        if random.random() < mut_rate:
            new_h[i] += random.uniform(-epsilon, epsilon)
    
    # Ensure values are in [0,1]
    new_h = np.clip(new_h, 0, 1)
    
    # Rescale to correct sum
    new_sum = np.sum(new_h)
    target_sum = n / 2
    if abs(new_sum - target_sum) > 1e-8:
        scale_factor = target_sum / new_sum
        new_h = new_h * scale_factor
    
    return new_h

def optimize_c5(n_points=42, population_size=30, generations=1000, 
               mut_rate=0.1, cross_prob=0.7, elitism_rate=0.1, 
               seed=42, budget_s=600):
    """Optimize the C5 value using a genetic algorithm."""
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Initialize population
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        population = initialize_population(n_points, population_size, initial_h)
    else:
        initial_h = np.ones(n_points) * 0.5
        # Ensure sum is correct
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        population = initialize_population(n_points, population_size, initial_h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        c5 = evaluate_h(h)
        fitnesses.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Track best so far
    global_best_c5 = best_c5
    global_best_h = best_h.copy()
    
    # Manage in time budget
    gen = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s and no_improve_count < 200:
        gen += 1
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism: keep best individuals
        elite_size = max(1, int(elitism_rate * population_size))
        new_population = sorted_population[:elite_size]
        new_fitness = sorted_fitnesses[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if random.random() < cross_prob:
                child = blend_crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mut_rate)
            
            new_population.append(child)
            new_fitness.append(evaluate_h(child))  # Evaluate fitness of offspring
            
            # Track if we improved
            if new_fitness[-1] < best_c5:
                best_c5 = new_fitness[-1]
                best_h = child.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        # Update population and restart
        population = new_population
        fitnesses = new_fitness
        
        if best_c5 < global_best_c5:
            global_best_c5 = best_c5
            global_best_h = best_h.copy()
    
    return global_best_h, global_best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    """Main run function for the optimization."""
    # Optimize with given parameters
    h, c5, n_points = optimize_c5(
        n_points=42,
        population_size=30,
        generations=1000,
        mut_rate=0.1,
        cross_prob=0.7,
        elitism_rate=0.1,
        seed=seed,
        budget_s=budget_s
    )
    
    return h, c5, n_points
