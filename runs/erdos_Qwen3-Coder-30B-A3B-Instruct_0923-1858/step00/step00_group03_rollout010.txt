import numpy as np
import time
from scipy.signal import correlate

def binary_convert(h, n_points):
    """Convert h to binary, ensuring sum(h) = n_points / 2"""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h
    diff = current_sum - target_sum
    if diff > 0:
        # Need to flip some 1s to 0s
        indices = np.where(h == 1)[0]
        flip_indices = indices[:diff]
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        indices = np.where(h == 0)[0]
        flip_indices = indices[:abs(diff)]
        h[flip_indices] = 1
    return h

def compute_c5(h, dx):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    h_full = np.concatenate([h, np.zeros(len(h))])
    h_conj = np.concatenate([np.zeros(len(h)), 1 - h])
    cross_corr = correlate(h_full, h_conj, mode='valid')
    max_corr = np.max(cross_corr)
    return max_corr * dx

def get_neighbor(h, n_points):
    """Get a neighbor by flipping one 1 to 0 and one 0 to 1"""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    else:

        h[:n_points//2] = 1
    
    h = binary_convert(h, n_points)
    current_c5 = compute_c5(h, dx)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        h_new = get_neighbor(h, n_points)
        new_c5 = compute_c5(h_new, dx)
        
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        T *= alpha
        iter_count += 1
        
        if last_improvement > 1000:
            break
            
    return (best_h.tolist(), best_c5, n_points)
