import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral for each shift k
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: adjust by a constant
    adjustment = (target_sum - current_sum) / len(h)
    h = h + adjustment
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.zeros(n_points)
        # Even indices get 0.5, odd indices get 0, then scale to sum=1
        h[::2] = 0.5
        h[1::2] = 0
        h = h * n_points / (2 * np.sum(h))
    
    target_sum = n_points / 2.0
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    iteration = 0
    perturb_interval = 100
    max_perturb = 0.1
    
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % perturb_interval == 0:
            # Perturb h to explore different regions
            perturbation = np.random.uniform(-max_perturb, max_perturb, n_points)
            h += perturbation
            h = project_to_constraint(h, target_sum)
        
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        # Find maximum overlap index
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx
        
        # Perturb around max correlation point to reduce it
        # Adjust by reducing values where overlapping occurs
        adjustment = np.zeros(n_points)
        # For simplicity, zoom into peak area and reduce those
        # Use a range around the max shift index
        peak_window = min(5, n_points // 4)
        for i in range(max(0, max_idx - peak_window), min(n_points + peak_window, len(corr))):
            # Convert correlation index back to actual shift k
            if i < n_points:
                k = i  # For h, 1-h cross-correlation (shifts can be 0 to n-1)
            else:
                k = i - n_points
            if k == 0: continue  # Skip self overlap
            # Redistribute some mass to decrease difficult correlation
            if k > 0:
                # h sliding to the right, we penalize h[i] and (1-h[i+k])
                for j in range(n_points):
                    if (j + k) < n_points:
                        adjustment[j] -= 0.0001 * h[j]  # rate_based
                        adjustment[(j + k) % n_points] -= 0.0001 * (1 - h[j])
            else:
                k = -k
                for j in range(n_points):
                    if (j - k) >= 0:
                        adjustment[j] -= 0.0001 * h[j]
                        adjustment[(j - k) % n_points] -= 0.0001 * (1 - h[j])
        
        # Apply adjustments and re-project
        h += adjustment
        h = project_to_constraint(h, target_sum)
        
        # Reeval C5
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Occasionally increase perturbation to escape local minima
        if iteration % 500 == 0:
            max_perturb = min(0.2, max_perturb * 1.1)
    
    return (best_h.tolist(), best_c5, n_points)
