import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    # Compute cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation scaled by dx
    return np.max(corr) * dx

def project_h(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to target sum
    target_sum = len(h) / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Linear adjustment to meet sum constraint
        delta = (target_sum - current_sum) / len(h)
        h += delta
        # Re-clip to maintain bounds
        h = np.clip(h, 0, 1)
    return h

        h_new = project_h(h_new)

        return evaluate_h(individual)

    population = [create_individual() for _ in range(POPSIZE)]

        offspring = []
