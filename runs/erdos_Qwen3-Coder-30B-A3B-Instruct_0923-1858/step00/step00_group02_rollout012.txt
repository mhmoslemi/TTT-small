        # Compute current overlap integral
        overlaps = np.correlate(h, 1 - h, mode='full')

        # Compute expected values
        # Use a quadratic penalty for the non-convex term
        # Instead of a full convex relaxation, we use a quasi-Newton inspired update

        grad = np.zeros_like(h)

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute overlap integral over all possible shifts
    overlaps = np.correlate(h, 1 - h, mode='full')
    return np.max(overlaps) * dx
