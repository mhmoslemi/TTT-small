import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    # Compute overlaps for all shifts
    h_rev = np.flip(h)
    # correlate with 1-h to get h * (1-h) shifts
    overlap = correlate(h, 1 - h, mode='full')
    # Convert to integral form (scale by dx)
    overlap *= dx
    # Take max (the C_5 value)
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize with the provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # construct a basic one if none is present
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
        
    # Precompute overlap estimate to reduce overhead
    max_iter = 1000  # max iterations
    iteration = 0
    
    # Initialize
    c5_bound = evaluate_overlap(h, dx)
    
    # Start optimizing with a simple block-based approach.
    # Repeating refines using cutting plane method.
    
    while iteration < max_iter:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_c5 = evaluate_overlap(h, dx)
        
        # Let's run a CVXPY optimization step with identified constraints
        # assuming simple knot points with sufficient precision
        
        # This is a heuristic to converge more efficiently.
        # Generate shifts of interest
        all_shifts = list(range(-(n_points-1), n_points))
        # Use full set for better coverage in initial passes
        shifts_interest = sorted(all_shifts, key=lambda k:
                                 correlate(h, 1-h, mode='full')[k+n_points-1], reverse=True)
        # Select top fraction for optimization
        num_cons = min(max(50, len(shifts_interest) // 4), len(shifts_interest))
        shifts_to_use = shifts_interest[:num_cons]
        
        # Build and solve convex optimization problem to find improved h
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Objective: minimize C5
        objective = cp.Minimize(c5_var)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2,
            c5_var >= 0,
        ]
        
        # Add overlap constraints for each shift
        for k in shifts_to_use:
            # This is a tricky part: CVXPY doesn't allow auto-correlations directly
            # So we will decode the constraint manually
            
            # For overlap[k], we compute the sums of h(i) * (1 - h(i + k)) 
            # This is equivalent to: sum over all i where i+k is valid:
            # h[i] * (1 - h[i+k])
            
            # Adjust indices to avoid out of bounds
            if k >= 0:
                # sum(h[i] * (1 - h[i+k])) over valid i
                # i ranges from 0 to n_points - k - 1
                valid_indices = range(n_points - k)
                temp = sum(h_vars[i] * (1 - h_vars[i + k]) for i in valid_indices)
            else:
                # i ranges from -k to n_points - 1
                valid_indices = range(-k, n_points)
                temp = sum(h_vars[i] * (1 - h_vars[i + k]) for i in valid_indices)
            
            constraints.append(temp <= c5_var * dx)
        
        # Formulate problem and solve
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                c5_new = c5_var.value
                
                if h_new is not None and c5_new is not None:
                    h = np.clip(h_new, 0, 1)
                    c5_bound = min(c5_bound, c5_new)
        except:
            pass  # Avoiding errors
            
        iteration += 1
        
    # Final validation
    c5_bound = evaluate_overlap(h, dx)
    
    # Normalize h to the exact integral constraint
    h = h / np.sum(h) * (n_points / 2)
    c5_bound = evaluate_overlap(h, dx)
    
    return list(h), c5_bound, n_points
