import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    dx = 2.0 / len(h)
    
    # Ensure the constraint ∫ h(x) dx = 1
    h_sum = np.sum(h) * dx
    h = h * (1.0 / h_sum)
    
    # Gradient descent parameters
    max_iter = 10000
    step_size = 0.01
    tol = 1e-6
    time_start = time.time()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Correct for the bias due to np.correlate's mode
        # Also convert back to integral scale
        c5_vals = corr * dx
        
        # Find the maximum
        max_k = np.argmax(c5_vals)
        max_val = c5_vals[max_k]
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        # Compute gradient w.r.t. dominant lag
        # Look at how a small change in h[i] affects the overlap at max_k
        # (Actually we can directly construct this via convolution)
        grad = np.zeros_like(h)
        
        # For gradient of C₅ = max_k ∫ h(x)(1 - h(x+k)) dx
        # We take gradient with respect to the term that achieves max_k
        # The full convolution gives ∫ h(x)(1 - h(x+k)) dx = conv(h, 1 - h)
        # Gradient at each i with respect to max_k term
        # ∂/∂h[i] ∫ h(x)(1 - h(x+k)) dx, for fixed k
        
        # Direct method for gradient:
        # Consider effect of changing h[i] on the i+k term in convolution
        # We apply chain rule: ∂ max_k f_k / ∂ h_i = (df_k/dh_i) if k=k*
        # Derivative of integral h(x)(1 - h(x + k)) w.r.t h_i
        # if x=i, x + k = i + k
        # d/dh_i [h(x)(1 - h(x+k))] = 1_{x=i} * (1 - h(i+k)) + 1_{x+k=i} * (- h(i)) 
        # But with convolution, we want to compute this in gradient of full conv
        
        # Let's approach this numerically using finite differences for more stability
        # or else directly compute:
        # For k = max_k, the derivative ∂ ∫ h(x)(1 - h(x+k)) dx / ∂ h[i]
        # Note: this derivative exists in general since h*(x+k) with x index  
        # corresponds to h*(i-k) if we're in the domain [0, 2], and current h[i] modifies 
        # term h*(i-k) when k is the index such that x+k = i => k = i-x => k = i-i = 0  
        # But we're looking at that i-k is fixed... need to rethink e.g. direct indexing 
        # Let’s use the cross-correlation formula
        # The function to minimize is conv(h, 1-h) indexed by lag k
        # We compute shift of h by k, multiply by (1-h), integrate,
        # finding maximum k. Then we compute the Jacobian
    
        # So to get gradient ∂C₅/∂h[i]: 
        # look position i in our vector and see how h[i] contributes to overlapping terms
        # specifically, ∂∫ h(x)(1-h(x+k))dx / ∂h[i] = 
        # ∫ ∂h(x)/∂h[i] (1 - h(x+k)) dx + ∫ h(x) ∂(1-h(x+k))/∂h[i] dx
        # = ∫ δ(x-i) (1 - h(x+k)) dx = 1 - h(i+k)
        # = ∫ h(x) (-δ(x-k)) dx = -h(k) 
        
        # Gradient for specific k (dominant one)
        # ∂ ∫ h(x)(1-h(x+k)) dx / ∂ h[i] = (1 - h[i+k]) * δ_i^x + h[i-k] * (-δ_k^x)
        # We extract the i-th component of gradient w.r.t. the overlap at lag k
    
        # Let's compute direction of gradient for the maximum overlapping lag  
        # see how h[i] contributes to integral over lag k=0 to max_k (corresponding to max_k)
        # derived appropriately:
        shifted = np.roll(1 - h, max_k - (len(h) - 1))  # shift to align with lag
        # This is a bit tricky — since we don't know exactly which positions
        # contribute gradients due to h(i) in integral over lag k, let's just take
        # derivative as if derivative from lag k_0 is ∂/∂h_i ∫ h(x)(1-h(x+k)) dx = - (1 - h(i+k))
        # as explained in output of this tool
        
        # Actually retrace: 
        # The function C₅ at lag k = ∫ h(x)(1 - h(x + k)) dx 
        # ∂ C₅_k / ∂ h[i] = ∫ δ(x-i)(1 - h(x+k)) dx = 1 - h(i+k) (supposing x = i, x+k = i+k)
        #             = ∫ h(x)(-δ(x+k-i)) dx = - h(i-k) (reverse case)
        
        # So gradient for the k with max overlap is:
        # For each i in h, we compute ∂ C₅_{k*} / ∂ h[i]  
        # i -> evaluation: 1 - h[i+k*], and also -h[i-k*]
        # But at boundary it's not well-defined, so we'll use numerical conversion around h[i]
        # Very useful: in cyclic context f = 1-h previous value (behind) + self value*1
        # Handle all gradient computation for lag k*  
        
        # Another approach: 
        # What we'll be doing via full correlation:
        # we compute C₅(k) = ∫ h(x)(1-h(x+k)) dx
        # weight a small change in h[i] to C₅[k*]  (w.r.t k*)
        # This is: ∫ [h(x) d(1-h(x+k))dx + (1-h(x+k)) dh(x)dx] dx  
        # = ∫ [h(x) * (- δ(x+k-i)) dx + (1-h(x+k)) δ(x-i) dx]
        # = -h(i+k) + (1-h(i+k)) = 1-2*h(i+k)
        # Correct as above one again:
        # let’s use discrete conv technique directly.
        # Differentiate C₅(k) = sum h[i](1-h[i+k])  
        # Now total ∂ C₅(k)/∂h[i] = 1-h[i+k], for some offset
        # but also we must express in correct indices.
        # There's circulant property of convolution, so if we compute:
        # to get core gradient ∂ C₅(k*) / ∂ h[i]: 
        # draw h from element i. gives contribution if:
        # h[i]*[1-h[i+k]] -> drop term from zero because of derivative etc.
        # you see when evaluator (cf. integrate h(x)(1-h(x+k))) evaluates this with index i,
        # it's using h[i] * (1-h[i+k]) so we want derivative w.r.t. to h[i].
        
        # Let's build this correctly with indices
        # i-th element contributes to h[i]*mu term in the integral 
        # if x=i then h(x+k) = h[i+k]
        # Contribution to x=i term in: h(x) * (1-h(x+k))  
        # gets derivative in this via 1* (but (1-h[i+k]) is constant wrt h[i]) →  0
        # or -1* (1-h[x+k]) = -1*(1-h[i+k])
        # So divergent terms...
        
        # The proper idea:
        # To what index is h[i] involved in h(x) (1-h(x+k)) for k = k*?
        # That occurs when x = i, then h[i+k] is the contribution to coefficient.  
        # it is ∫ h(x)(1-h(x+k)) dx. Consider ∂ / ∂ h[i]
        # ∫   {  ... + h[i](1-h[i+k]) + ... }  = 1 - h[i+k]  
        
        # So [∂ C₅_{k*} / ∂ h[i]] = 1 - h[i+k*]  [just from definition of derivation]
        # the actual difference when allowing h to shift around. 
        # No! Correct index alignment:
        # We convert full convolution to discrete difference
        # d/dy Int h(x)(1 - h(x+y)) dx = ∑_i 1_{x=i} (1 - h[i+y])
        # 
        # ∂ / ∂h[i] Int h(x) (1 - h(x+k)) dx  = ∫ [h(x) (1 - h(x+k)) dx] / ∂ h(i)
        #                                 = 1 - h(i+k) , with proper index dealing
		
        # We can construct this as convolution with 1's, but better approach:
        # Compute h(i+k) as convolution: h * 1-shifted (with computed shift)
        # Actually say for computing weight gradient for each shift:
        # update is: ∇ = - (1 - 2 h[i+k*]) for i=0 to 2/dx, weighted by stepsize  
        
        grad = np.zeros_like(h)
        offset = max_k - (len(h) - 1)  # positive lag means subsequent index in corr
        for i in range(len(h)):
            # gradient for lag k* of ∫ h(x)(1-h(x+k*)) dx w.r.t h[i]
            # It's: h[x] -> its h[i], we have contribution to h[i] * (1 - h[i+k*])
            # Gradient is: 1 - h[i+k*]
            k_idx = (i + offset) % len(h)  # modular? over index k in lags
            grad[i] = 1 - h[k_idx]
        
        # Now follow descent direction
        h = h - step_size * grad
        
        # Projection to [0,1] then normalizing again
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        h = h * (1.0 / h_sum)
        
        # Convergence checks
        if i % 10 == 0:
            if abs(h_sum - 1) < tol or abs(max_val - best_c5) < tol:
                break
                
    return (best_h.tolist(), float(best_c5), len(best_h))
