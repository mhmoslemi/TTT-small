import numpy as np
import time
from scipy import correlate

def evaluate_solution(h, dx):
    """Evaluate the overlap integral C5 for given h."""
    # Compute the overlap integral for all shifts
    overlaps = correlate(h, 1 - h, mode="full")
    # Scale by dx to get the actual integral value
    overlaps *= dx
    # Return the maximum value
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction, simple step function
        n_points = 70
        h = np.zeros(n_points)
        # Set first half as 1s and second half as 0s (will be adjusted to satisfy sum)
        h[:n_points//2] = 1.0
        h = h.astype(float)
        # Ensure sum is exactly n_points/2
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if current_sum != target_sum:
            # Adjust by flipping some bits to make sum correct
            diff = int(target_sum - current_sum)
            if diff > 0:
                # Need to flip diff 0s to 1s
                zero_indices = np.where(h == 0)[0]
                flip_indices = zero_indices[:diff]
                h[flip_indices] = 1.0
            else:
                # Need to flip abs(diff) 1s to 0s
                one_indices = np.where(h == 1)[0]
                flip_indices = one_indices[:abs(diff)]
                h[flip_indices] = 0.0
        assert abs(np.sum(h) - target_sum) < 1e-10, "Sum constraint violated"
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_solution(h, dx)
    
    # Precompute overlap contributions for all positions and shifts
    # each overlap is just sum over i of h[i] * (1-h[i+k])
    # which is sum over i of h[i] - h[i]*h[i+k] = sum h[i] - sum h[i]*h[i+k]
    # This means we're computing sums over all possible shifts for each pair of positions
    
    # Optimization loop
    max_iter = 50000
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        # Find the current maximum overlap and its shift index
        overlaps = correlate(h, 1 - h, mode="full")
        overlaps *= dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = np.max(overlaps)
        
        # For each position in h, compute change in overlaps if that position is flipped
        # We need to build an influence matrix of how each h[i] affects overlaps at shift k
        # It's easier to compute the effect for all positions efficiently at once:
        # H[i] contributes to overlap at shift k when the overlap calculation involves
        # h[i] * (1 - h[i+k]) or h[i+k] * (1 - h[i]) depending on how the indices align.
        # For overlap shift k, sum for i from 0 to n_points-1-k:
        # h[i] * (1 - h[i+k]) contribution to overlap at that shift.
        # Therefore, for a flip at i:
        # If h[i] was originally 1, then we take 1 - h[i+k] from overlap, 
        # If h[i] was originally 0, then we take 0 - (1 - h[i+k]) = h[i+k] from overlap.

        # Quick way to compute effect of flipping a single bit:
        # It's the contribution from index i to overlaps at all shifts
        # Let's compute how flipping h[i] affects overlap at k if that is part of the max_overlap
        # but first, do it for all positions to see which are most impactful
        
        # Give up if we don't find anything to improve  
        if max_overlap < best_c5 * 0.9999:
            if iter_count - last_improvement > 1000:
                break
            last_improvement = iter_count
        
        # Compute the best position to flip to reduce max_overlap
        best_flip_i = -1
        best_flip_delta = 0.0
        best_flip_is_1_to_0 = False  # whether we flip 1 to 0 or 0 to 1
        
        # Current sum constraint is current_sum == n_points / 2
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        
        # For optimization, we'll compute all contributions of flipping each '1'
        # to see if it helps reduce the outlier peg in the overlap function
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        flip_cand_indices = list(ones) + list(zeros)
        # phase 1: flip those that are currently 1, to reduce overlap (1->0)
        # phase 2: flip those that are currently 0 to 1 (may increase overlap but has a trade-off)
        
        # heuristic: try to reduce max overlap without breaking sum
        for i in flip_cand_indices:
            # What would the effect be to flip h[i] from value v (0 or 1) to 1-v
            current_val = h[i]
            new_val = 1.0 - current_val
            
            # Compute current 'contribution' to # of overlaps based on this bit at the max overlap shift
            max_k = max_overlap_idx - (n_points - 1)  # shift for the max overlap
            
            # Compute how flip h[i] affects overlap at shift k where it contributes positively
            # for index j, h[j] * (1 - h[j+k]) contributes
            # to overlap at shift k; the net change due to flipping h[i] 
            # is 0 if i not in matched set
            # otherwise so compute change for all 
            change_sum = 0.0
            for k in range(n_points):
                # how many (i,j) pairs contribute to overlap at shift k where both are valid
                # overlap k contributors = sum of h[j] * (1 - h[j+k])
                # where for fixed j, h[j] * (1 - h[j+k]) is only a meaningful term when we have j and (j+k) in range
                j = i
                # Only when j and j+k are both valid indexes and when we're flipping j
                if j + k < n_points:
                    # Compute contribution of h[j] to overlap at shift k
                    # originally = h[j] * (1 - h[j+k])
                    # changed = new_val * (1 - h[j+k]) 
                    # change = (new_val - h[j]) * (1 - h[j+k])
                    
                    if h[j] == new_val:
                        continue
                    contrib = (new_val - h[j]) * (1 - h[j + k])
                    change_sum += contrib
                    
            # Very crude: for all shifts, the effect is just sum for all valid j+k = i+k that is within range
            # situation better for direct, but just encode approximation for a start:
            if current_val == 1 and max_k >= 0:
                # Looking specifically at max_k, i=1 now becomes 0
                if i + max_k < n_points:
                    delta = - (1 - h[i + max_k])  # old minus new
                    if delta < best_flip_delta:
                        best_flip_delta = delta
                        best_flip_i = i
                        best_flip_is_1_to_0 = True
            elif current_val == 0 and max_k >= 0:
                # i flip 0 to 1
                if i + max_k < n_points:
                    delta = h[i + max_k]  # new - old
                    if delta > best_flip_delta:
                        best_flip_delta = delta
                        best_flip_i = i
                        best_flip_is_1_to_0 = False
                        
            # Simplified - just iterate over particular shifts where max_overlap arises
            # This leads to trivial optimization, so see better approximation
            # For all shifts, efficiently calculate effect of true flip:
            # Instead, just try to find next best couple of positions to flip 
            # to modify one overlap:
            # Make simple heuristic that works  better
            if current_sum - current_val + new_val == target_sum:
                # we can afford to move a bit if sum is balanced
                # But check that this move does not break other things
                
                # Estimate effect in neighborhood, but do several direct flips:
                # If flip 1 to 0 and 0 to 1, then impact of overlap at max_shift can be estimated by doing it directly.
                if i in ones:
                    # This is a 1 being flipped to 0
                    # Estimate how all shifts would change; just focus on shifts that are impacted
                    if i + max_k < n_points:
                        delta = - (1 - h[i + max_k])  # from 1 to 0
                        if delta < best_flip_delta:
                            best_flip_delta = delta
                            best_flip_i = i
                            best_flip_is_1_to_0 = True
                elif i in zeros:
                    # flip 0 to 1
                    if i + max_k < n_points:
                        delta = h[i + max_k]  # from 0 to 1
                        if delta > best_flip_delta:
                            best_flip_delta = delta
                            best_flip_i = i
                            best_flip_is_1_to_0 = False

        if best_flip_i != -1:
            # Perform the flip
            # If trying to flip 1 to 0, ensure it's a valid 1
            if best_flip_is_1_to_0 and h[best_flip_i] != 1:
                continue
            if not best_flip_is_1_to_0 and h[best_flip_i] != 0:
                continue
                
            # Check sum constraint still holds:
            new_sum = current_sum - 1 + 1  # This is just a simple flip
            if abs(new_sum - target_sum) < 1e-10:
                # Flip the hit
                h[best_flip_i] = 1.0 - h[best_flip_i]
                c5_current = evaluate_solution(h, dx)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h.copy()
        else:
            # Try hard search in case best didn't find, run often enough to avoid locals
            # Simple version: try random flips
            i = np.random.randint(0, n_points)
            # Only flip if it keeps sum constraint
            current_sum = np.sum(h)
            if current_sum == target_sum:
                h[i] = 1.0 - h[i]
                c5_current = evaluate_solution(h, dx)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h.copy()
    
    # Final validation
    final_sum = np.sum(best_h)
    dx = 2.0 / n_points
    C5_final = evaluate_solution(best_h, dx)
    return (best_h.tolist(), C5_final, n_points)
