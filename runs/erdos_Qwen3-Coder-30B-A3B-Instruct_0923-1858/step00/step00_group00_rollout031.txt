import numpy as np
from scipy import signal
from time import time

def evaluate(h):
    """Evaluate C5 bound for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_val = np.max(conv) * dx
    return max_val

def project(h, n_points):
    """Project h onto feasibility constraints."""
    # Clamp to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to have sum = n_points / 2
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    return h

def nelder_mead_step(vertices, func_values, alpha=1.0, gamma=2.0, beta=0.5, rho=0.5):
    """Perform one step of Nelder-Mead algorithm."""
    n = len(vertices) - 1
    # Order vertices by function value (ascending)
    indices = np.argsort(func_values)
    vertices = [vertices[i] for i in indices]
    func_values = [func_values[i] for i in indices]
    
    # Best and worst vertices
    x_best = vertices[0]
    x_worst = vertices[-1]
    
    # Compute centroid of all but worst
    centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    x_reflect = centroid + alpha * (centroid - x_worst)
    x_reflect = project(x_reflect, len(x_reflect))
    f_reflect = func_values[-1]
    
    # Try to improve - if better than best vertex
    if func_values[0] > func_values[-2]:
        if func_values[0] > func_values[-2] > func_values[-1]:
            # Try expansion
            x_expand = centroid + gamma * (centroid - x_worst)
            x_expand = project(x_expand, len(x_expand))

                f_expand = func(x_expand)

            if f_expand < func_values[-1]:

        # Else, try outside contraction
        x_contract = centroid + beta * (centroid - x_worst)
        x_contract = project(x_contract, len(x_contract))
        f_contract = func_values[-1]  # Placeholder
        try:
            f_contract = func(x_contract)
        except:
            pass
        if f_contract < func_values[-1]:
            return x_contract, f_contract
        # Else, shrink simplex
        new_vertices = [x_best]
        for i in range(1, n+1):
            new_x = x_best + rho * (vertices[i] - x_best)
            new_x = project(new_x, len(new_x))
            new_vertices.append(new_x)
        return new_vertices, None
    
    # If reflected point is better than worst, replace
    if func_values[0] > func_values[-2]:
        return x_reflect, func_values[-2]
    
    return [x_best, x_reflect, *vertices[1:-1]], None

def func(h):
    """Objective function to minimize."""
    return evaluate(h)

    reset_attempts = 0

                vertices[-1] = x_e

                simplex.append(h_v)
            func_values = [func(v) for v in simplex]
            iter_since_improvement = 0
        
        j += 1
    
    # Final evaluation
    best_c5 = evaluate(best_h)
    
    return (list(best_h), best_c5, n_points)
