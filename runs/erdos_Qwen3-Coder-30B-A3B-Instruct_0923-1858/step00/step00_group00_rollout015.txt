    h = h * (n_points / 2.0) / np.sum(h)
    
    # Function to compute C5
    def compute_C5(h_vals):
        conv = correlate(h_vals, 1 - h_vals, mode='full')
        max_conv = np.max(conv)
        return max_conv * dx
    
    # Initial evaluation
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.995
    perturbation_scale = 0.1
    patience = 100
    no_improvement = 0
    
    # Main optimization loop
    max_iterations = 10000
    iteration = 0
    
    while iteration < max_iterations:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Perturb the solution
        h_new = h + np.random.normal(0, perturbation_scale, n_points)
        
        # Clamp values to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet the constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Compute new C5
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            h = h_new
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1
            
        # Cool temperature
        T *= alpha
        
        # Decrease perturbation scale periodically
        if iteration % 100 == 0 and perturbation_scale > 0.001:
            perturbation_scale *= 0.99
            
        # Early stopping
        if no_improvement > patience:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_C5, n_points)
