import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.01
    perturb_freq = 100
    perturb_rate = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum = n_points / 2
    h = h * (0.5 / np.mean(h))
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        correlations = correlate(h, 1-h, mode="full")
        current_C5 = np.max(np.abs(correlations)) * dx
        
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        # Compute gradient using finite differences for cross-correlation
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Compute gradient w.r.t. h[i]
            for k in range(-n_points + 1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j])
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Take gradient step
        h = h - lr * gradient
        
        # Project onto constraints
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.mean(h))
        
        # Apply perturbations periodically
        if iteration % perturb_freq == 0:
            perturb = np.random.uniform(-perturb_rate, perturb_rate, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.mean(h))
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, n_points)
