import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[1::2] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iter = int(1e4)
    perturb_freq = 100
    epsilon = 0.05  # perturbation strength
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = signal.correlate(h, 1 - h, mode='full')
        corr = corr * dx
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify max overlap indices
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)  # offset to get actual k
        
        # Adjust h: reduce in regions causing max overlap
        # Create a small adjustment array
        adj = np.zeros(n_points)
        for k in range(n_points):
            if (k - max_k) % n_points == 0:
                adj[k] = epsilon * 0.1
            
        # Apply adjustment and clip
        h -= adj
        h = np.clip(h, 0, 1)
        
        # Project to satisfy sum constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        
        if abs(diff) > 1e-10:
            # Distribute mass evenly
            h += diff / n_points
            
        # Make sure we maintain the sum constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        
        # Periodic perturbation
        if i % perturb_freq == 0:
            # Identify high overlap regions 
            corr = signal.correlate(h, 1 - h, mode='full')
            corr = corr * dx
            
            # Create redistribution: move mass from overrepresented regions
            # Scale with maximum overlap
            max_overlap_val = np.max(corr)
            if max_overlap_val > 0:
                # Modify h to reduce max overlap
                # Use a simple method: add random noise to avoid local minima
                h += np.random.normal(0, epsilon / 10, n_points)
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                h = h * target_sum / h_sum
    
    return (best_h, best_c5, n_points)
