    h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps using correlate
        h_1h = np.correlate(h, 1 - h, mode="full")
        overlaps = h_1h * dx
        c5_bound = np.max(overlaps)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
            
        # Find indices with max overlap
        max_overlaps = np.where(np.abs(overlaps - c5_bound) < 1e-3)[0]
        
        # Set up LP problem
        C = cp.Variable()
        h_var = cp.Variable(n_points)
        
        # Constraints for significant overlaps
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Add overlap constraints
        for k_idx in max_overlaps:
            # Define the lag k
            lag = k_idx - (n_points - 1)
            # Compute kernel for this lag
            kernel = np.concatenate([np.zeros(-lag if lag < 0 else 0), 
                                    1 - h, 
                                    np.zeros(lag if lag > 0 else 0)])
            # Get the correct portion for correlation
            kernel = kernel[:n_points]
            # This is not straightforward in CVXPY, so compute explicitly for a few important lags
            
        # For simplicity and performance, modify one component at a time
        # We'll make small adjustments to h, guided by numerical gradient estimation
        
        # Estimate gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(n_points):
            h_plus[i] = min(1, h[i] + eps)
            h_minus[i] = max(0, h[i] - eps)
            
            # Recompute overlaps
            overlap_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode="full") * dx)
            overlap_minus = np.max(np.correlate(h_minus, 1 - h_minus, mode="full") * dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
            
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Take a step in negative gradient direction
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final verification of best solution
    final_overlaps = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(final_overlaps)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
