import numpy as np
from time import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap is the maximum of the correlation values
    C5 = np.max(corr) * dx
    return C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a smooth initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Slightly perturb to avoid symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        # Normalize to sum to 1
        h *= (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time()
    max_iter = 10000
    iteration = 0
    
    while time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        k_vals = np.arange(-n_points + 1, n_points)
        # Avoid negative indices
        valid_k = np.abs(k_vals) < n_points
        C_vals = corr[valid_k] * dx
        
        # Find the shift with max overlap
        max_idx = np.argmax(C_vals)
        k_max = k_vals[max_idx]
        
        # Now find the best pair of indices to adjust
        if k_max >= 0:
            # k_max is positive, so we're shifting by +k_max
            # h[i] contributes to h[i+k_max] with the term h[i](1-h[i+k_max])
            # So we look at positions i and i+k_max
            # Adjust the pair that contributes most to this overlap
            best_reduction = -1
            best_i = -1
            
            # We need to iterate through valid i values for which both i and i+k_max exist
            for i in range(n_points - k_max):
                # Compute the current contribution
                contrib = h[i] * (1 - h[i + k_max])
                # How much can we reduce this?
                # Pick one of the two in a way to reduce the contribution most
                # Try to reduce value at i or h[i+k_max]
                # Reduce contribution by reducing h[i] or h[i+k_max] (but maintain values in [0,1])
                # Let's reduce h[i] - we want to minimize h[i]*(1-(h[i+k_max]))
                # So to minimize h[i]*(1-h[i+k_max]), we want to increase h[i]
                # or reduce h[i+k_max] -- adjust minimum of both
                new_i = min(1.0, h[i] + 0.01)
                new_j = max(0.0, h[i + k_max] - 0.01)
                new_contrib = new_i * (1 - new_j)
                reduction = contrib - new_contrib
                if reduction > best_reduction:
                    best_reduction = reduction
                    best_i = i
        else:
            # k_max is negative, h[i] contributes less when we look at h[i+k_max]
            k_abs = -k_max  # k_abs > 0
            best_reduction = -1
            best_i = -1
            
            for i in range(n_points - k_abs):
                contrib = h[i] * (1 - h[i + k_abs])
                new_i = min(1.0, h[i] + 0.01)
                new_j = max(0.0, h[i + k_abs] - 0.01)
                new_contrib = new_i * (1 - new_j)
                reduction = contrib - new_contrib
                if reduction > best_reduction:
                    best_reduction = reduction
                    best_i = i
                    k_max_ = k_abs  # Renaming for clarity
        
        if best_i >= 0 and best_reduction > 1e-6:
            # Make the adjustment
            # Clamp to valid range
            h[best_i] = min(1.0, max(0.0, h[best_i] + 0.01))
            h[best_i + abs(k_max)] = max(0.0, min(1.0, h[best_i + abs(k_max)] - 0.01))
            
            # Adjust other entries to preserve sum
            total_diff = np.sum(h) - (n_points / 2)
            if np.abs(total_diff) > 1e-3:
                # Redistribute the difference
                avg_adjust = total_diff / n_points
                h -= avg_adjust
                
            # Check if this improved
            new_C5 = evaluate_erdos_solution(h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        else:
            # Deteriorate rate, modify based on expanding search
            # Try a more random approach
            i1 = np.random.randint(0, n_points)
            i2 = np.random.randint(0, n_points)
            # Perturb both
            h[i1] = np.clip(h[i1] + np.random.normal(0, 0.01), 0, 1)
            h[i2] = np.clip(h[i2] + np.random.normal(0, 0.01), 0, 1)
            
            total_diff = np.sum(h) - (n_points / 2)
            if np.abs(total_diff) > 1e-3:
                avg_adjust = total_diff / n_points
                h -= avg_adjust
                
            new_C5 = evaluate_erdos_solution(h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        
        iteration += 1
    
    return (best_h.tolist(), best_C5, n_points)
