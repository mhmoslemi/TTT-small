import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a balanced construction
        n_points = 42
        h = np.ones(n_points) * 0.5
        # Ensure sum = n_points / 2
        h[::2] = 1.0
        h[1::2] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute constants
    dx = 2.0 / n_points
    
    # Frank-Wolfe with adaptive learning rate and perturbation
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    max_iter = 10000
    lr_min = 0.01
    lr_max = 0.2
    perturb_freq = 100
    epsilon = 0.05
    
    def evaluate_c5(h_vals):
        # Vectorized evaluation using full correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(np.abs(corr))
        return max_corr * dx
        
    # Initial evaluation
    current_c5 = evaluate_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Main optimization loop
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:
            break
            
        # Compute subgradient via finite differences
        eps = 1e-3
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] = min(1.0, h_plus[j] + eps)
            h_minus = h.copy()
            h_minus[j] = max(0.0, h_minus[j] - eps)
            
            val_plus = evaluate_c5(h_plus)
            val_minus = evaluate_c5(h_minus)
            grad[j] = (val_plus - val_minus) / (2 * eps)
        
        # Frank-Wolfe linear subproblem
        # minimize <grad, d> subject to 0 <= d[i] <= 1 and sum(d) = 0
        # Equivalent to: minimize sum(grad * d) subject to 0 <= d[i] <= 1, sum(d) = 0
        
        # Simple heuristic approach: sort and project
        d = -grad  # descent direction
        d = d - np.mean(d)  # make sum zero
        
        # Project to box constraints
        d = np.clip(d, -h, 1 - h)
        
        # Adjust step size
        lr = lr_max - (lr_max - lr_min) * (i / max_iter)
        h_new = h + lr * d
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new - np.mean(h_new) + 0.5  # enforce sum constraint
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Accept with probability based on exponential decay
            accept_prob = np.exp(-(current_c5 - new_c5) / (current_c5 + 1e-8))
            if np.random.rand() < accept_prob:
                h = h_new
                current_c5 = new_c5
        
        # Periodic perturbation
        if i % perturb_freq == 0 and i > 0:
            perturb = np.random.normal(0, epsilon, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h - np.mean(h) + 0.5
            
    return (best_h, best_c5, n_points)
