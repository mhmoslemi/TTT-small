import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = 1.0 - (i - mid) / (n_points - 1 - mid)
    
    # Normalize to exact sum
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Hyperparameters
    eta = 0.01
    perturb_rate = 0.05
    perturb_freq = 50
    decay_freq = 100
    decay_factor = 0.95
    
    # Gradient descent loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        
        # Compute correlation for gradient estimation
        corr_full = correlate(h, 1 - h, mode="full")
        corr_full = corr_full * dx  # Scale by dx
        
        # Compute current C5
        current_c5 = np.max(corr_full)
        
        # Update with gradient step
        # For gradient, we want to compute ∂C5/∂h[i]
        # For simplicity, approximate with finite differences
        
        # Approximate gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_plus = corr_plus * dx
            
            grad[i] = (np.max(corr_plus) - current_c5) / eps
        
        # Update h
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Check improvement
        new_c5 = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            last_improvement = iter_count
        else:
            # Apply perturbation occasionally or if no recent improvement
            if iter_count % perturb_freq == 0 or iter_count - last_improvement > 50:
                h_perturbed = h + np.random.normal(0, perturb_rate, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
                perturbed_c5 = np.max(correlate(h_perturbed, 1 - h_perturbed, mode="full")) * dx
                if perturbed_c5 < best_c5:
                    h = h_perturbed
                    best_c5 = perturbed_c5
                    best_h = h.copy()
                    last_improvement = iter_count
        
        # Decay learning rate
        if iter_count % decay_freq == 0:
            eta *= decay_factor
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)
