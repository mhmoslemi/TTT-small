import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    # Normalize h to have sum equal to n_points / 2
    h = np.array(h)
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    # Compute correlation
    corr = correlate(h, 1 - h, mode="full")
    # Scale by dx
    corr *= dx
    # Return maximum value
    return np.max(corr)

def project_to_constraints(h, n_points):
    # Ensure h is within [0, 1]
    h = np.clip(h, 0, 1)
    # Project to correct sum
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42  # Start with the given initial construction

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values.copy())
    else:
        h = np.ones(n_points) * 0.5
        h[1::2] = 0
        h = project_to_constraints(h, n_points)

    # Variables for tracking best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Parameters
    max_iter = 10000
    perturbation_strength = 0.05
    perturbation_interval = 100
    dx = 2.0 / n_points
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h, n_points)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        
        # Find index of maximum overlap
        max_idx = np.argmax(corr)
        # Find the actual k value from the correlation index
        k = max_idx - (n_points - 1)
        
        # Adjust h to reduce this overlap
        h = h.copy()
        # Create a displacement vector to decrease h at positions causing overlap
        # We modify h[i] where h[i] * (1 - h[i+k]) contributes heavily
        
        # First get indices causing the overlap
        indices_to_decrease = []
        indices_to_increase = []
        
        # Since the overlap is at k, we consider h[i] with h[i+k]
        for j in range(n_points):
            i_k = (j + k) % n_points
            if h[j] > 0 and h[i_k] < 1:
                indices_to_decrease.append(j)
                indices_to_increase.append(i_k)
        
        # Now adjust according to slack
        if len(indices_to_decrease) > 0:
            # Decrease a small portion from one index and add to another
            idx1 = np.random.choice(indices_to_decrease)
            idx2 = np.random.choice(indices_to_increase)
            
            # Try small adjustments in both directions
            adjustment = min(perturbation_strength * h[idx1], 
                             (1 - h[idx2]) * perturbation_strength)
            
            h[idx1] -= adjustment
            h[idx2] += adjustment
            
            # Apply projections
            h = project_to_constraints(h, n_points)
            
        # Occasionally perform random perturbation to explore
        if i % perturbation_interval == 0:
            # Randomly redistribute some mass from high values to low
            h = h.copy()
            sort_indices = np.argsort(h)
            # Move mass from top 10% to bottom 10%
            num_to_move = max(1, n_points // 10)
            high_indices = sort_indices[-num_to_move:]
            low_indices = sort_indices[:num_to_move]
            
            # Skip if not enough variability
            if len(high_indices) > 0 and len(low_indices) > 0:
                # Move small amounts of mass from high to low
                move_total = h[high_indices].sum() / 10
                if move_total > 0:
                    to_move = np.minimum(move_total, h[high_indices].sum())
                    # Distribute evenly
                    move_each = to_move / len(low_indices)
                    for idx in high_indices:
                        if to_move <= 0:
                            break
                        dec = min(move_each, h[idx])
                        h[idx] -= dec
                        to_move -= dec
                    for idx in low_indices:
                        if to_move <= 0:
                            break
                        inc = min(move_each, 1 - h[idx])
                        h[idx] += inc
                        to_move -= inc
                    h = project_to_constraints(h, n_points)

    # Final check
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
