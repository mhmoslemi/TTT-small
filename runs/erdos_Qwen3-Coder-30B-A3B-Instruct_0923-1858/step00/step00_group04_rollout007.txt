    def c5_objective(h_vec):

        max_correlation = np.max(np.abs(correlation))
        return max_correlation * dx

    def gradient_step(h_vec, step_size=0.01):
        h_vec = np.array(h_vec)
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_k = np.argmax(np.abs(corr))
        max_val = corr[max_k]
        
        # Approximate gradient direction based on how changing h[i] affects max correlation
        # For simplicity, we'll sample a few directions and take the best improvement
        best_h = h_vec.copy()
        best_c5 = c5_objective(h_vec)
        for _ in range(10):
            # Small random perturbation to explore
            pert = np.random.normal(0, 0.01, n_points)
