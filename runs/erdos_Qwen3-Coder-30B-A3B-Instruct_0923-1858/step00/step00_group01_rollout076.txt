import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h = h * scale
    return h

def evaluate_c5(h, dx):
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # Max of cross-correlation values (scaled by dx)
    return np.max(corr) * dx
