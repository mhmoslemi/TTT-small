import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Generate triangular waveform
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h /= np.sum(h) * 2.0 / n_points  # Rescale to sum = n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    def evaluate_C5(h_vals, dx):
        # Correlation of h with (1 - h) 
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The maximum value over all lags
        return np.max(corr) * dx
    
    def project_h(h_vals):
        # Clip to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to satisfy sum constraint
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals = h_vals * (n_points / 2) / current_sum
        return h_vals
    
    def finite_diff_grad(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_h(h_plus)
            h_minus = project_h(h_minus)
            grad[i] = (evaluate_C5(h_plus, dx) - evaluate_C5(h_minus, dx)) / (2 * eps)
        return grad
    
    # Optimization loop
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        # Compute gradient
        grad = finite_diff_grad(h)
        
        # Compute candidate step
        step = -grad * trust_radius / (np.linalg.norm(grad) + 1e-10)
        
        # Apply step
        h_new = h + step
        h_new = project_h(h_new)
        
        # Evaluate new solution
        new_C5 = evaluate_C5(h_new, dx)
        
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            best_h = h.copy()
            no_improvement_count = 0
            trust_radius *= 1.2  # Increase trust radius
        else:
            no_improvement_count += 1
            trust_radius *= 0.5  # Decrease trust radius
            
        # Perturb if no improvement for a while
        if no_improvement_count > max_no_improvement:
            no_improvement_count = 0
            h = project_h(h + np.random.normal(0, 0.05, n_points))
    
    return (best_h.tolist(), best_C5, n_points)
