import numpy as np
from scipy import optimize
import time
import math

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap integral is the maximum correlation value, scaled by dx
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45
    
    # Use initial_h_values if available

    except:
        # Fallback to symmetric construction
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        h[mid-2:mid+2] = 1.0
    
    # Ensure sum equals n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    # Keep track of last improvement time
    last_improvement = 0
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.9995
    sigma = 0.1
    sigma_decay = 0.99
    
    dx = 2.0 / n_points
    max_iter = 1000000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Perturb with Gaussian noise
        delta = np.random.normal(0, sigma, n_points)
        h_new = h + delta
        
        # Project to feasible region [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = i
        else:
            # Accept with probability based on temperature
            delta_c5 = c5_new - best_c5
            if np.random.rand() < math.exp(-delta_c5 / temp):
                best_c5 = c5_new
                best_h = h_new.copy()
                last_improvement = i
        
        # Update temperature and noise level
        temp *= cooling_rate
        if i % 100 == 0:
            sigma *= sigma_decay
            
        if i - last_improvement > 1000:
            break
    
    # Return the best solution found
    if best_h is None:
        # Fallback to initial solution
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        h *= (n_points / 2) / np.sum(h)
        best_c5 = evaluate_erdos_solution(h)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
