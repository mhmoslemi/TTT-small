import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from parent construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2.0) / h_sum
    
    # Define function to evaluate C5
    def evaluate_c5(h_vec):
        h_vec = np.array(h_vec)
        n_points = len(h_vec)
        dx = 2.0 / n_points
        # Compute correlation for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Apply dx scaling
        corr = corr * dx
        # Extract valid overlaps for lags in [0, n_points-1] in terms of x-shift
        # The max overlap for a valid lag k should be max of overlap at shift k
        max_overlap = np.max(corr[n_points-1:])
        return max_overlap
    
    min_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Project h to satisfy constraints
    def project_h(h_vec):
        # Clamp to [0, 1]
        h_vec = np.clip(h_vec, 0.0, 1.0)
        # Adjust to satisfy sum constraint
        current_sum = np.sum(h_vec)
        if abs(current_sum - n_points / 2.0) > 1e-10:
            diff = (n_points / 2.0) - current_sum
            # Distribute the difference equally
            h_vec += diff / n_points
            # Clamp again after adjustment
            h_vec = np.clip(h_vec, 0.0, 1.0)
        return h_vec
    
    start_time = time.time()
    iteration = 0
    alpha = 0.05
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute overlap for all shifts
        h_vec = np.array(h)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        corr = corr * dx
        # Extract the positive lags (0 to n_points-1, representing shifts from 0 to n_points-1)
        overlaps = corr[n_points-1:]
        k_star = np.argmax(overlaps)
        
        # Compute gradient w.r.t h: more costly way
        # At each i, gradient for h[i] = (1 - h[i+k_star]) for valid i and k_star
        # At each i +k_star, gradient for h[i+k_star] = -h[i]
        # Create gradient vector (0 at positions where it doesn't contribute)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                grad[i] += (1 - h_vec[i + k_star])
            if i - k_star >= 0:
                grad[i - k_star] -= h_vec[i]
        
        # Update
        h = h - alpha * grad
        
        # Project
        h = project_h(h)
        
        # Evaluate
        new_c5 = evaluate_c5(h)
        if new_c5 < min_c5:
            min_c5 = new_c5
            best_h = h.copy()
        
        # Decrease step size
        if iteration % 100 == 0 and alpha > 1e-6:
            alpha *= 0.9
    
    # Final rounding for tightness
    final_h = project_h(best_h)
    return (final_h.tolist(), float(min_c5), n_points)
