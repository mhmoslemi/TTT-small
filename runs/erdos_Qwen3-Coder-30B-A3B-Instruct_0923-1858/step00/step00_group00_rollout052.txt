import numpy as np
from scipy import signal
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create symmetric construction centered at 1.0 if no initial provided
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[n_points//2:] = 1.0  # Half of the points at 1
        h[:n_points//2] = 0.0  # First half at 0
        # Adjust to make sum = n_points/2
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Start timing
    start_time = time.time()
    
    # Annealing parameters
    T = 0.1  # Initial temperature
    alpha = 0.99  # Annealing rate
    sigma = 0.1  # Initial noise scale
    no_improve_count = 0
    max_no_improve = 1000
    cooling_freq = 100
    
    def compute_c5(h_vals):
        # Use valid convolution mode
        corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
        # Multiply by dx to account for step size
        corr *= dx
        return np.max(corr)
    
    def project(h_vals):
        # Clamp values to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to satisfy constraint sum(h) = n_points / 2
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals *= (n_points / 2) / current_sum
        return h_vals
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Generate perturbation
        delta_h = np.random.normal(0, sigma, n_points)
        h_candidate = h + delta_h
        
        # Project to feasible region
        h_candidate = project(h_candidate)
        
        # Evaluate cost function
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject based on simulated annealing
        if c5_candidate < current_c5:
            h = h_candidate
            current_c5 = c5_candidate
            no_improve_count = 0
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        else:
            # Accept worse solution with probability
            if np.random.rand() < np.exp(-(c5_candidate - current_c5) / T):
                h = h_candidate
                current_c5 = c5_candidate
            no_improve_count += 1
            
        # Update temperature and noise
        if iteration % cooling_freq == 0:
            T *= alpha
            sigma *= 0.99
            
        if no_improve_count >= max_no_improve:
            break
            
        iteration += 1
    
    # Final check and return
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
