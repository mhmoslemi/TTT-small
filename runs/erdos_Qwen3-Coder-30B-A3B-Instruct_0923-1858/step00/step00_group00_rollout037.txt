import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except:
        # Default construction
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    # Ensure initial h satisfies the constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init)
    
    # Binary search for better bound
    c_min, c_max = 0.0, 0.5
    start_time = time.time()
    
    for iteration in range(20):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        c_mid = (c_min + c_max) / 2.0
        
        # Try to find feasible h with C5 <= c_mid using optimization
        h_opt = optimize_feasible_h(h_init, n_points, c_mid, dx)
        
        if h_opt is not None:
            c5_val = evaluate_erdos_solution(h_opt)
            if c5_val <= c_mid and c5_val < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_val
            c_max = c_mid
        else:
            c_min = c_mid

    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral of h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value
    c5 = np.max(conv) * dx
    return c5

def optimize_feasible_h(h_init, n_points, c_target, dx):
    """Attempt to optimize h to satisfy C5 <= c_target"""
    # Variables
    h = Variable(n_points)
    
    # Problem constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
    
    # Add bilinear constraints for each shift
    # This is approximate, rather than exact bilinear,
    # we relax bilinear constraints with McCormick envelopes
    # Let's note that exact optimization with bilinear terms is hard,
    # so use simpler heuristic approach
    
    # Use scipy for simpler local optimization approach
    def obj_func(h_vec):
        return evaluate_erdos_solution(h_vec)
        
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start from h_init
    h0 = h_init.copy()
    
    # Use a simple local solver to improve it
    try:
        # Use L-BFGS-B for bounded optimization
        result = minimize(
            obj_func, 
            h0, 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if result.success and result.fun <= c_target:
            return result.x
        elif result.success:
            # even if not within target, may be useful for next iter
            return result.x
        else:
            return None
    except:
        return None
