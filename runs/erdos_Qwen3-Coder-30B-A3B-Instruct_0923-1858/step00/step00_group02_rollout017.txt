import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given step function h"""
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum over all shifts (k from -1 to 1)
    max_val = np.max(corr)
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define problem parameters
    n_points = 45  # Use the provided initial construction
    dx = 2.0 / n_points
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)  # Normalize to correct integral
    else:
        # Create a simple initial guess
        h0 = np.ones(n_points) * 0.5
    
    # Objective function to minimize
    def objective(h):
        # Normalize to correct integral
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_h(h, dx)
    
    # Bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    def integral_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Coerce h to be within bounds
    def enforce_bounds(h):
        h = np.clip(h, 0, 1)
        return h
    
    # Use differential evolution for global optimization
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Optimization loop
    remaining_time = budget_s
    max_iter = 200
    iter_count = 0
    
    while iter_count < max_iter and remaining_time > 1.0:
        # Run differential evolution
        try:
            res = differential_evolution(
                objective,
                bounds,
                seed=seed + iter_count,
                maxiter=int(remaining_time * 2),
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                atol=1e-6,
                ftol=1e-6,
                callback=lambda x, convergence: None,
                disp=False
            )
            
            if not res.success:
                break
                
            # Post-process current solution
            h = res.x.copy()
            h = h * (n_points / 2) / np.sum(h)  # Normalize integral
            h = enforce_bounds(h)
            current_c5 = evaluate_h(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
        except Exception as e:
            break
            
        iter_count += 1
        remaining_time = budget_s - (time.time() - start_time)
        
        if remaining_time < 1.0:
            break
            
    # Final refinement if needed
    if iter_count > 0:
        # Try gradient-based optimization on the best solution
        try:
            from scipy.optimize import minimize
            
            def obj_to_minimize(h):
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                return evaluate_h(h, dx)
                
            # Add small noise to break symmetry for local optimization if needed
            h_refined = best_h.copy()
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            h_refined = np.clip(h_refined, 0, 1)
            
            # Perform local optimization
            options = {'maxiter': 1000, 'ftol': 1e-8}
            result = minimize(obj_to_minimize, h_refined, 
                            method='L-BFGS-B', 
                            bounds=[(0, 1) for _ in range(n_points)],
                            options=options)
            if result.success:
                h_final = result.x.copy()
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                h_final = np.clip(h_final, 0, 1)
                c5_final = evaluate_h(h_final, dx)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
