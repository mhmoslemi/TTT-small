import numpy as np
from scipy import signal
import time
import random

def evaluate_C5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation: h * (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum value and scale by dx
    max_corr = np.max(corr)
    return max_corr * dx

def normalize_h(h):
    """Ensure h sums to n/2"""
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def is_valid_h(h):
    """Check if h is valid: values in [0,1], sum = n/2"""
    n = len(h)
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n/2) < 1e-10)

def create_random_h(n):
    """Create a random valid h"""
    h = np.random.rand(n)
    h = normalize_h(h)
    return np.clip(h, 0, 1)

def crossover(parent1, parent2, crossover_prob=0.7):
    """Blend crossover between two parents"""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
        
    n = len(parent1)
    # Select crossover point
    point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Ensure valid sum
    child1 = normalize_h(child1)
    child2 = normalize_h(child2)
    
    return np.clip(child1, 0, 1), np.clip(child2, 0, 1)

def mutate(h, mutation_rate=0.1):
    """Mutation: perturb some elements"""
    h = h.copy()
    n = len(h)
    
    for i in range(n):
        if random.random() < mutation_rate:
            # Small random change, ensure bounded
            delta = random.gauss(0, 0.05)
            h[i] += delta
            # Rebalance if needed
            if h[i] < 0 or h[i] > 1:
                h[i] = np.clip(h[i], 0, 1)
    
    # Ensure sum constraint
    h = normalize_h(h)
    return np.clip(h, 0, 1)

def init_population(n, pop_size=30):
    """Initialize population of candidates"""
    pop = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_init = normalize_h(np.array(initial_h_values))
        if is_valid_h(h_init):
            pop.append(h_init)
    
    # Fill rest with random valid solutions
    while len(pop) < pop_size:
        h = create_random_h(n)
        if is_valid_h(h):
            pop.append(h)
    
    return pop

def select_parents(pop, fitnesses, tournament_size=5):
    """Tournament selection"""
    n = len(pop)
    parent1 = None
    parent2 = None
    
    # Select first parent
    tournament_indices = random.sample(range(n), min(tournament_size, n))
    best_idx = min(tournament_indices, key=lambda i: fitnesses[i])
    parent1 = pop[best_idx]
    
    # Select second parent
    tournament_indices = random.sample(range(n), min(tournament_size, n))
    best_idx = min(tournament_indices, key=lambda i: fitnesses[i])
    parent2 = pop[best_idx]
    
    return parent1, parent2

def genetic_algorithm(n, budget_s=600, pop_size=30, max_generations=1000):
    """Main genetic algorithm"""
    
    start_time = time.time()
    
    # Initialize population
    pop = init_population(n, pop_size)
    global_best = None
    global_best_C5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitnesses = []
        for h in pop:
            c5 = evaluate_C5(h)
            fitnesses.append(c5)
        
        # Track global best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < global_best_C5:
            global_best_C5 = fitnesses[min_idx]
            global_best = pop[min_idx].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        pop = [pop[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism: keep top 10%
        elite_count = max(1, pop_size // 10)
        elite = pop[:elite_count]
        
        # Generate new population
        new_pop = elite.copy()
        
        while len(new_pop) < pop_size:
            parent1, parent2 = select_parents(pop, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        pop = new_pop[:pop_size]
    
    return global_best, global_best_C5

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization"""
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a reasonable size, adjust if needed
    n = 42
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    
    # Run GA
    best_h, best_C5 = genetic_algorithm(n, budget_s=budget_s, pop_size=30)
    
    # Final validation
    if not is_valid_h(best_h):
        # Re-initialize to a valid one
        best_h = create_random_h(n)
        best_C5 = evaluate_C5(best_h)
    
    return (best_h, best_C5, len(best_h))
