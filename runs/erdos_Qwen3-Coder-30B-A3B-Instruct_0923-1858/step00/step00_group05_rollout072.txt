    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        # Compute current overlaps for all lags k
        correlations = np.correlate(h, 1-h, mode="full")
        C5_candidate = np.max(np.abs(correlations)) * dx
        
        # Update best solution
        if C5_candidate < best_C5:
            best_h = h.copy()
            best_C5 = C5_candidate
        
        # Compute gradient of max correlation with respect to h
        # Gradient computation: for each position i, we need to compute how 
        # changing h[i] affects the maximum value in the cross-correlation
        gradient = np.zeros(n_points)
        
        # Find which lag k achieves the maximum correlation
        max_corr_index = np.argmax(np.abs(correlations))
        max_k = max_corr_index - (n_points - 1)  # Convert full index to lag
        
        # Compute gradient for that specific lag
        # For a fixed k, d/dh[i] (h[i] * (1 - h[i+k])) = (1 - h[i+k]) if i+k valid
        for i in range(n_points):
            if 0 <= i + max_k < n_points:
                gradient[i] += (1 - h[i + max_k])
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Take gradient step
        h = h - alpha * gradient
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.mean(h))
        
        # Perturb periodically to escape local minima
        if iteration % perturbation_freq == 0:

            h = np.clip(h, 0, 1)
