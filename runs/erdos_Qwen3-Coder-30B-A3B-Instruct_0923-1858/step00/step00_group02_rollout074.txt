import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.full(n_points, 0.5)

    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-6
    best_C = C_high
    best_h = h0.copy()
    
    # Prepare shift indices for all k in [-n+1, n-1]
    shift_indices = {}
    for k in range(1 - n_points, n_points):
        shift_indices[k] = np.arange(n_points)
        if k >= 0:
            shift_indices[k] = np.arange(k, n_points)
        else:
            shift_indices[k] = np.arange(0, n_points + k)
    
    # Optimizer
    solver = cp.ECOS  # Fast enough for the task
    
    # Binary search loop
    while C_high - C_low > tolerance and (time.time() - start_time) < budget_s - 1:
        C = (C_low + C_high) / 2.0
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Problem setup
        h = cp.Variable(n_points, name="h")
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Generate constraints for all shifts
        for k in range(1 - n_points, n_points):
            if k == 0:
                continue
            i_indices = shift_indices[k]
            j_indices = np.roll(shift_indices[k], -k)
            # Overlap = sum(h[i] * (1 - h[j])) <= C * n_points
            overlap = cp.sum(cp.multiply(h[i_indices], 1 - h[j_indices]))
            constraints.append(overlap <= C * n_points)
            
        # Optimization problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_val = h.value
                if h_val is not None:
                    current_C = C
                    C_high = current_C
                    best_h = h_val.copy()
                    best_C = current_C
                else:
                    C_low = C
            else:
                C_low = C
        except Exception:
            C_low = C
    
    # Validation
    h_final = best_h
    if np.any(h_final < 0) or np.any(h_final > 1) or abs(np.sum(h_final) - n_points / 2) > 1e-5:

        best_C = min(best_C, current_C)
    except:
        pass
