import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Initialize h from initial_h_values if available, otherwise use midpoint
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
    else:
        h = np.full(n_points, 0.5)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    start_time = time.time()
    trust_radius = 0.1
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for current h
        overlap = correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlap)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute expected values for convex approximation (simplified for convexity)
        expected_h = np.roll(h, 1)  # Approximate expectation as shifted values
        Hessian_approx = np.eye(n_points, dtype=float)
        gradient = np.zeros(n_points)
        for k in range(n_points):
            if k == 0:
                gradient += h - h * expected_h
            else:
                gradient += h - h * np.roll(expected_h, k)
        gradient = gradient - np.mean(gradient)
        
        # Solve convex subproblem using CVXPY
        try:
            d = cp.Variable(n_points)
            objective = cp.Minimize(cp.quad_form(d, Hessian_approx) + 
                                  gradient.T @ d)
            constraints = [
                d >= -h,
                d <= 1 - h,
                cp.sum(d) == 0
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                d_val = d.value
            else:
                d_val = -gradient  # fallback to gradient descent step
        except:
            d_val = -gradient  # fallback
        
        # Line search
        alpha = 1.0
        step_size = 0.5
        max_backtrack = 20
        for _ in range(max_backtrack):
            h_new = h + alpha * d_val
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum > 0:
                h_new = h_new * target_sum / h_new_sum
            
            # Evaluate new overlap
            overlap_new = correlate(h_new, 1 - h_new, mode='full') * dx
            new_c5 = np.max(overlap_new)
            
            if new_c5 < current_c5:
                h = h_new
                break
            alpha *= step_size

        # Maintain constraint by rescaling if necessary
        h_sum = np.sum(h)
        if h_sum != target_sum and h_sum > 0:
            h = h * target_sum / h_sum
            
    # Final check
    final_overlap = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_overlap)
    
    return (list(best_h), float(final_c5), int(n_points))


def evaluate_erdos_solution(h_values, n_points):
    # Convert to numpy array
    h = np.array(h_values)
    dx = 2.0 / n_points
    
    # Compute overlap integral
    overlap = correlate(h, 1 - h, mode='full') * dx
    c5 = np.max(overlap)
    
    return c5
