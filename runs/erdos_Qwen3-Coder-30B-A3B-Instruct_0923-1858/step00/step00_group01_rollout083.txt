import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize learning rate and iteration count
    lr = 0.01
    iterations = 0
    max_iterations = 1000
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vec):
        # Compute convolution of h with (1-h) 
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h_vec):
        # Compute the gradient numerically using finite differences for crossover
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_c5 = compute_c5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Keep integral constraint
            h_plus = h_plus * target_sum / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    def project_and_normalize(h_vec):
        # Project to [0,1] and normalize to sum = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        # Compute current C5
        current_c5 = compute_c5(h)
        if iterations == 0 or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Line search for step size
        alpha = lr
        h_new = h - alpha * grad
        h_new = project_and_normalize(h_new)
        new_c5 = compute_c5(h_new)
        
        # Backtracking line search
        while new_c5 >= current_c5 and alpha > 1e-10:
            alpha *= 0.5
            h_new = h - alpha * grad
            h_new = project_and_normalize(h_new)
            new_c5 = compute_c5(h_new)
        
        # Update
        h = h_new
        iterations += 1
        
        # Decrease learning rate
        if iterations % 100 == 0:
            lr *= 0.1
            if lr < 1e-6:
                lr = 1e-6
        
        # Resolution increase
        if iterations >= 100 and n_points <= 100:
            n_points *= 2
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            dx = 2.0 / n_points
            h = project_and_normalize(h)
            last_improvement = 0
    
    return (best_h.tolist(), best_c5, n_points)
