import numpy as np
from scipy import optimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) * dx - 1.0) > 1e-10:
        h = h * (1.0 / (np.sum(h) * dx))  # Rescale to sum to 1
    return h

def evaluate_C5(h, dx):
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum correlation scaled by dx
    return np.max(corr) * dx

def gradient_C5(h, dx, eps=1e-8):
    n = len(h)
    grad = np.zeros(n)
    base_C5 = evaluate_C5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        C5_plus = evaluate_C5(h_plus, dx)
        grad[i] = (C5_plus - base_C5) / eps
    return grad

def line_search(h, grad, dx, alpha_max=1.0, factor=0.5):
    alpha = alpha_max
    while alpha > 1e-6:
        h_new = h - alpha * grad
        h_new = clip_and_rescale(h_new, dx)
        if evaluate_C5(h_new, dx) < evaluate_C5(h, dx):
            return alpha, h_new
        alpha *= factor
    return alpha, h

    best_C5 = evaluate_C5(h, dx)
    
    # Parameters
    max_iter = 1000
    lr = 0.01
    decay_every = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        if i % decay_every == 0 and i > 0:
            lr *= 0.1
            
        # Compute gradient and update
        grad = gradient_C5(h, dx)
        alpha, h_new = line_search(h, grad, dx, alpha_max=lr)

        current_C5 = evaluate_C5(h, dx)
        if current_C5 < best_C5:

            best_h = h.copy()
