import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.full(n_points, 0.5)
    
    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-6
    best_C = C_high
    best_h = h0.copy()
    
    start_time = time.time()
    
    # Function to evaluate overlaps
    def compute_overlaps(h, n_points):
        overlaps = []
        dx = 2.0 / n_points
        # Compute correlation-like overlaps
        h_extended = np.concatenate([h, h])  # For circular shifts
        for k in range(1, n_points):  # All non-trivial shifts
            shift = h_extended[k:k+n_points]
            overlap = np.sum(h * (1 - shift)) * dx
            overlaps.append(overlap)
        return overlaps
    
    # Function to solve LP feasibility for given C
    def solve_feasibility(h0, C, n_points):
        # Variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Compute overlap constraints for all shifts (k from -n+1 to n-1)
        # Using discrete convolution effectively
        dx = 2.0 / n_points
        h_extended = np.concatenate([h0, h0])  # Broadcast to avoid modulo
        
        for k in range(n_points):  # This covers k from 0 to n-1 due to circular
            # Overlap for shift k: sum(h[i] * (1 - h[i+k]))
            idx_left = np.arange(n_points)
            idx_right = (idx_left + k) % n_points
            # Build the constraint: sum(h[i] * (1 - h[i+k])) <= C * dx * n_points
            # which can be written as: h^T * (1 - h_shifted) <= C * n 
            # but avoiding matrix inversion if possible:
            # 1 - h[idx_right] -> to compute x_i = h[i] * (1 - h[i+k])
            product = h[idx_left] * (1 - h[idx_right])
            overlap_constraint = cp.sum(product) <= C * n_points * dx
            constraints.append(overlap_constraint)
            
        # Objective (any since we only check feasibility):
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
            else:
                return None
        except:
            return None
    
    # Optimization using binary search
    while C_high - C_low > tolerance and time.time() - start_time < budget_s - 1:
        C_mid = (C_low + C_high) / 2.0
        
        # Try to construct a feasible h
        h_sol = solve_feasibility(h0, C_mid, n_points)
        
        if h_sol is not None and np.all(h_sol >= 0) and np.all(h_sol <= 1):
            # Check if it actually satisfies all constraints
            overlaps = compute_overlaps(h_sol, n_points)
            max_overlap = max(overlaps) if overlaps else 0
            if max_overlap <= C_mid:
                best_h = h_sol.copy()
                best_C = C_mid
                C_high = C_mid
            else:
                C_low = C_mid
        else:
            C_low = C_mid
            
    # Final validation of solution
    final_overlaps = compute_overlaps(best_h, n_points)
    final_C = max(final_overlaps) if final_overlaps else 0
    
    # Return best feasible solution within time limit
    return (best_h, final_C, n_points)
