import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    """Clamp h to [0,1] and rescale to maintain integral constraint."""
    h = np.clip(h, 0, 1)
    integral = np.sum(h) * dx
    if abs(integral - 1.0) > 1e-10:
        h = h / integral * 1.0
    return h

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral C5."""
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation is the desired C5
    C5 = np.max(corr) * dx
    return C5

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h."""
    # We'll use finite differences to estimate gradient
    eps = 1e-6
    n = len(h)
    grad = np.zeros(n)
    base_C5 = evaluate_C5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        C5_plus = evaluate_C5(h_plus, dx)
        grad[i] = (C5_plus - base_C5) / eps
    return grad

def line_search(h, grad, dx, alpha_max=1.0):
    """Simple line search to find an appropriate step size."""
    alpha = alpha_max
    while alpha > 1e-8:
        h_new = h - alpha * grad
        h_new = clip_and_rescale(h_new, dx)
        new_C5 = evaluate_C5(h_new, dx)
        old_C5 = evaluate_C5(h, dx)
        if new_C5 <= old_C5:
            return alpha
        alpha *= 0.5
    return alpha
