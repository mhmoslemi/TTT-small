import numpy as np
import time
import random
from scipy.signal import correlate

def to_binary(h, n_points):
    """Convert h to binary array with exact sum n_points/2"""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h
    
    # Adjust to get exact sum
    diff = target_sum - current_sum
    if diff > 0:  # Need to add 1s
        indices = np.argsort(h)
        for i in range(diff):
            h[indices[i]] = 1
    else:  # Need to subtract 1s
        indices = np.argsort(h)[::-1]
        for i in range(-diff):
            h[indices[i]] = 0
    return h

def compute_C5(h, dx):
    """Compute C5 using full cross-correlation"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def local_search(h, n_points, dx, max_iter=100000, temp_start=1.0, cooling=0.999, time_budget_s=600):
    """Perform simulated annealing to optimize h"""
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    current_h = h.copy()
    current_C5 = best_C5
    
    temp = temp_start
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s - 1:
            break
            
        # Pick one 1 and one 0 to flip
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = random.choice(ones)
        j = random.choice(zeros)
        
        # Memory of current values
        old_i = current_h[i]
        old_j = current_h[j]
        
        # Flip
        current_h[i] = 0
        current_h[j] = 1
        
        # Compute new C5
        new_C5 = compute_C5(current_h, dx)
        
        # Accept or reject
        delta = new_C5 - current_C5
        if delta < 0 or random.random() < np.exp(-delta / temp):
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        else:
            # Revert
            current_h[i] = old_i
            current_h[j] = old_j
            
        temp *= cooling
        
    return best_h, best_C5
