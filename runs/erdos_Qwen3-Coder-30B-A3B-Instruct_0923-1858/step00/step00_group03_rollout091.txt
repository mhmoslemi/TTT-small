import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    # Compute overlap integral for all shifts
    conv = correlate(h, 1-h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def adjust_sum(h, target_sum):
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h.copy()
    
    diff = int(current_sum - target_sum)
    h = h.copy()
    
    if diff > 0:  # Too many 1s, need to flip some to 0
        ones_indices = np.where(h == 1)[0]
        flip_indices = ones_indices[:diff]
        h[flip_indices] = 0
    else:  # Too few 1s, need to flip some to 1
        zeros_indices = np.where(h == 0)[0]
        flip_indices = zeros_indices[:abs(diff)]
        h[flip_indices] = 1
    
    return h
