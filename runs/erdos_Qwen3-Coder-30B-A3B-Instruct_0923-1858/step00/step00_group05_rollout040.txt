import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_overlap(h, dx):
    # Compute cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The max of this gives us the C5 value
    return np.max(corr) * dx

def satisfy_constraints(h, n_points):
    # Ensure h values are in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points / 2
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h
