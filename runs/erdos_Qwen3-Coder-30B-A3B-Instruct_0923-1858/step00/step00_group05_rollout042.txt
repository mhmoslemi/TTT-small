import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value among all lags
    max_corr = np.max(corr)
    return max_corr * dx

def ensure_constraints(h, n_points):
    """Ensures h satisfies all constraints"""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to make sum = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Simple adjustment: rescale
        h = h * target_sum / current_sum
    return h

def fitness(h):
    """Fitness function to minimize"""
    h = ensure_constraints(h, len(h))
    return evaluate_c5(h)

def mutate_individual(h, mutation_rate=0.1):
    """Basic mutation of an individual"""
    n = len(h)
    h = h.copy()
    # Randomly mutate some elements
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Add Gaussian noise, clip to [0,1]
            h[i] += np.random.normal(0, 0.1)
            h[i] = np.clip(h[i], 0, 1)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    n = len(parent1)
    point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def local_optimize(h, n_points):
    """Use scipy.optimize to refine a candidate"""
    h = ensure_constraints(h, n_points)
    def objective(x):
        x = ensure_constraints(x, n_points)
        return evaluate_c5(x)
    
    # Use differential evolution as a robust local optimizer
    result = differential_evolution(objective, 
        bounds=[(0, 1)] * n_points,
        maxiter=50, 
        popsize=10,
        seed=42)
    best_x = result.x
    return best_x

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization algorithm"""
    # Start timing
    t_start = time.time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():

        n_points = len(h_init)
    else:
        # Create a baseline starting solution
        n_points = 70
        h_init = np.ones(n_points) * 0.5
        h_init = ensure_constraints(h_init, n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # GA population size
    pop_size = 100
    max_gens = 50
    n_generations = 0
    
    population = [h_init.copy()]
    
    while time.time() - t_start < 0.9 * budget_s and n_generations < max_gens:
        # Evaluate the population
        fitness_vals = []
        for i in range(len(population)):
            h = ensure_constraints(population[i], n_points)
            f = evaluate_c5(h)
            fitness_vals.append(f)
        
        # Update best solution found so far
        best_idx = np.argmin(fitness_vals)
        if fitness_vals[best_idx] < best_c5:
            best_c5 = fitness_vals[best_idx]
            best_h = population[best_idx].copy()
        
        # Create offspring
        offspring = []
        sorted_indices = np.argsort(fitness_vals)
        elite = [population[i] for i in sorted_indices[:10]]  # Top 10
        
        # Select parents by tournament
        while len(offspring) < pop_size:
            # Select two random parents
            parent1 = population[np.random.choice(sorted_indices[:50])]
            parent2 = population[np.random.choice(sorted_indices[:50])]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate_individual(child1)
            child2 = mutate_individual(child2)
            offspring.extend([child1, child2])
        
        # Keep only pop_size individuals
        population = offspring[:pop_size]
        
        # Keep elites from previous generation
        population[:len(elite)] = elite
        
        n_generations += 1

    # Perform final local optimizations on the best individuals
    for _ in range(5):
        if time.time() - t_start > 0.95 * budget_s:
            break
        # Create a fresh candidate from best_h
        test_h = best_h + np.random.normal(0, 0.01, n_points)
        test_h = ensure_constraints(test_h, n_points)
        refined = local_optimize(test_h, n_points)
        test_c5 = evaluate_c5(refined)
        if test_c5 < best_c5:
            best_c5 = test_c5
            best_h = refined.copy()
    
    # Final check
    best_h = ensure_constraints(best_h, n_points)
    best_c5 = evaluate_c5(best_h)
    
    return (list(best_h), best_c5, n_points)
