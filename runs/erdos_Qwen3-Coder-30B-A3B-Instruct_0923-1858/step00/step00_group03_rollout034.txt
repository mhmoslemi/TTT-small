import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Construct starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Adjust to meet sum constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        h *= target_sum / h_sum
        
    dx = 2.0 / n_points
    start_time = time.time()
    
    def compute_C5(h_vals):
        # Use scipy's correlate for efficiency
        h_vals = np.array(h_vals)
        # Compute all convolutions for all shifts
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Only take the relevant part
        result = np.max(corr) * dx
        return result
    
    def project_h(h_vals):
        # Clamp to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to meet sum constraint
        h_sum = np.sum(h_vals)
        target_sum = n_points / 2
        diff = target_sum - h_sum
        if abs(diff) > 1e-10:
            # Distribute difference uniformly
            h_vals += diff / n_points
            h_vals = np.clip(h_vals, 0, 1)
        return h_vals
    
    # Gradient-based minimization with projected updates
    max_iter = 2000
    alpha = 0.1
    tol = 1e-6
    best_C5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Compute gradient w.r.t max overlap
        # Find k that maximizes overlap
        corr = np.correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Gradient w.r.t h_j
            if 0 <= j + max_k < n_points:
                grad[j] += (1 - h[j + max_k])
            # Gradient w.r.t h_{j + max_k}
            if 0 <= j - max_k < n_points:
                grad[j - max_k] -= h[j]
                
        # Update with gradient descent
        h = h - alpha * grad
        h = project_h(h)
        
        # Reduce step size periodically
        if i % 100 == 0 and i > 0:
            alpha *= 0.9
            
        # Early stopping
        if abs(current_C5 - compute_C5(h)) < tol:
            break
        
    # Final project
    final_h = project_h(best_h)
    final_C5 = compute_C5(final_h)
    
    # Try binary rounding to potentially improve
    binary_h = np.round(final_h).astype(float)
    binary_h = project_h(binary_h)
    binary_C5 = compute_C5(binary_h)
    
    if binary_C5 < final_C5:
        final_h = binary_h
        final_C5 = binary_C5
    
    return (final_h.tolist(), final_C5, n_points)
