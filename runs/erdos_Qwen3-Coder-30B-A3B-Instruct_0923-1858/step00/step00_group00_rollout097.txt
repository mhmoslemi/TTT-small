import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    # Compute the full convolution of h and (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value among all shifts
    max_val = np.max(conv)
    return max_val * dx

def project_feasible(h, n_points):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def create_initial_h(n_points):
    """Create initial step function h."""
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Create a symmetric initial solution
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        h[mid-2:mid+2] = 1.0  # Spike in the middle
        return project_feasible(h, n_points)

def genetic_algorithm_step(h_pop, n_points, dx, pop_size=50, elite_ratio=0.2, mutation_rate=0.05):
    """Perform one generation of genetic algorithm."""
    n_elite = int(elite_ratio * pop_size)
    # Evaluate current population
    fitness = []
    for h in h_pop:
        val = evaluate_overlap(h, dx)
        fitness.append(val)
    fitness = np.array(fitness)
    # Sort by fitness
    idxs = np.argsort(fitness)
    elite_idx = idxs[:n_elite]
    # Create next generation
    next_pop = [h_pop[i] for i in elite_idx]
    # Fill with new individuals
    while len(next_pop) < pop_size:
        # Tournament selection
        idx1 = np.random.choice(elite_idx)
        idx2 = np.random.choice(elite_idx)
        h1, h2 = h_pop[idx1], h_pop[idx2]
        # Crossover
        alpha = 0.5
        child = alpha * h1 + (1 - alpha) * h2
        # Mutation
        if np.random.random() < mutation_rate:
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
        child = project_feasible(child, n_points)
        next_pop.append(child)
    return next_pop

def gradient_descent_step(h0, dx, max_iter=100, learning_rate=0.01):
    """Refine a solution with gradient descent."""
    h = h0.copy()
    n_points = len(h)
    prev_val = evaluate_overlap(h, dx)
    
    for i in range(max_iter):
        # Gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_feasible(h_plus, n_points)
            val_plus = evaluate_overlap(h_plus, dx)
            grad[j] = (val_plus - prev_val) / eps
        
        # Update step
        h -= learning_rate * grad
        h = project_feasible(h, n_points)
        
        new_val = evaluate_overlap(h, dx)
        if abs(new_val - prev_val) < 1e-8:
            break
        prev_val = new_val
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound on C5."""
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45  # using 45 for a faster evaluation
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    h_pop = [create_initial_h(n_points) for _ in range(pop_size)]
    
    best_c5 = float('inf')
    best_h = None
    
    max_generations = 100
    gen = 0
    
    while time.time() - start_time < budget_s - 5 and gen < max_generations:
        gen += 1
        # Genetic algorithm step
        h_pop = genetic_algorithm_step(h_pop, n_points, dx, pop_size=50, elite_ratio=0.2, mutation_rate=0.05)
        
        # Evaluate population
        fitness = []
        for h in h_pop:
            val = evaluate_overlap(h, dx)
            fitness.append(val)
        fitness = np.array(fitness)
        
        # Keep the best individuals for gradient descent
        best_indices = np.argsort(fitness)[:3]
        for i in best_indices:
            h = h_pop[i]
            h_refined = gradient_descent_step(h, dx)
            val = evaluate_overlap(h_refined, dx)
            if val < best_c5:
                best_c5 = val
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
