import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h *= scale
    return h

def evaluate_C5(h, dx):
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum value, scaled by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a uniform distribution
        n_points = 47
        h = np.ones(n_points, dtype=float) / n_points
        h *= 2.0 / n_points  # scale to integral = 1
    
    dx = 2.0 / n_points
    
    # Normalize h to have integral = 1
    h = clip_and_rescale(h, dx)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Gradient descent parameters
    max_iter = 1000
    lr = 0.01
    decay_every = 100
    decay_factor = 0.1
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        if i > 0 and i % decay_every == 0:
            lr *= decay_factor
            
        # Estimate gradient via finite difference
        eps = 1e-5
        grad = np.zeros_like(h)
        base_C5 = evaluate_C5(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = clip_and_rescale(h_plus, dx)
            C5_plus = evaluate_C5(h_plus, dx)
            grad[j] = (C5_plus - base_C5) / eps
        
        # Update h
        h = h - lr * grad
        h = clip_and_rescale(h, dx)
        
        # Check for improvement
        current_C5 = evaluate_C5(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
    
    # Possibly increase resolution and refine
    if n_points < 1000:
        new_n = min(1000, 2 * n_points)
        new_h = np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, n_points), best_h)
        new_h = clip_and_rescale(new_h, 2.0 / new_n)
        new_C5 = evaluate_C5(new_h, 2.0 / new_n)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h
    
    return (best_h.tolist(), best_C5, len(best_h))
