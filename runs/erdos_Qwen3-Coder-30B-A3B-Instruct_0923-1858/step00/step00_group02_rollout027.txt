import numpy as np
from scipy import optimize
import time

def evaluate_c5(h_values):
    """Evaluate the C5 constant given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # The maximum overlap is the maximum of these correlations
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n / 2) / h_sum
    return h

    """Run optimization to find a better C5 bound."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine number of points based on time budget
    n_points = 100
    dx = 2.0 / n_points

        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.zeros(n_points)
        mid = n_points // 2
        left = mid
        right = n_points - mid
        h[:left] = np.linspace(0, 1, left)
        h[left:left+right] = np.linspace(1, 0, right)
        h = project_h(h, n_points)
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Learning rate and parameters
    eta = 0.01
    perturb_rate = 0.05
    decay_every = 100
    
    step = 0
    while time.time() - start_time < budget_s * 0.95:
        step += 1
        
        # Compute current C5
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient using finite difference of correlation
        # We compute partial derivatives of objective w.r.t. each h[i]
        # Using finite differences of the correlation
        
        # Create a corresponding gradient vector
        grad = np.zeros(n_points)
        # Correlation matrix approach for computing gradient
        # We compute dC5/dh[i] for all i = 0..n-1
        # Using the definition of correlation
        # Correlation is a convolution with reversed kernel
        
        # Simplified gradient estimation by finite differences on the full correlation
        epsilon = 1e-4
        c5_base = evaluate_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            grad[i] = (c5_plus - c5_base) / epsilon
            
        # Update h
        h -= eta * grad
        h = project_h(h, n_points)

            # Add some random perturbations
            perturb = np.random.uniform(-perturb_rate, perturb_rate, n_points)
            h += perturb
            h = project_h(h, n_points)
        
        # Decrease learning rate periodically
        if step % decay_every == 0:
            eta *= 0.95
    
    # Final evaluation of best
    final_c5 = evaluate_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
