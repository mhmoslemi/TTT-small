import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Start with a symmetric construction
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Ensure integral constraint
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Helper function to evaluate C5
    def compute_C5(h_vals):
        dx = 2.0 / len(h_vals)
        # Compute full convolution of h with (1-h)
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Take maximum value and scale by dx
        max_val = np.max(correlation) * dx
        return max_val
    
    # Main optimization loop with binary search
    C_min = 0.0
    C_max = 0.5
    
    best_h = h_init.copy()
    best_C5 = compute_C5(h_init)
    
    # Binary search on C5
    for iteration in range(20):
        if time.time() - start_time > budget_s - 2:
            break
        C_mid = (C_min + C_max) / 2
        
        # Try to find a feasible solution with C5 <= C_mid
        try:
            # Convex relaxation approach:
            # Use a simpler local search approach for this project
            # We optimize all h in [0,1] with sum(h)*dx = 1
            opt_h = scipy_optimize_h(n_points, C_mid, dx)
            new_C5 = compute_C5(opt_h)
            if new_C5 <= C_mid and new_C5 < best_C5:
                best_h = opt_h.copy()
                best_C5 = new_C5
                C_max = new_C5
            else:
                C_min = C_mid
        except Exception:
            C_min = C_mid
    
    # Final validation check
    final_C5 = compute_C5(best_h)
    if final_C5 > best_C5:
        best_C5 = final_C5
    return (best_h.tolist(), float(best_C5), int(n_points))

def scipy_optimize_h(n_points, target_C, dx):
    # Create objective: minimize overlap
    def objective(h):
        # Maximize overlap in backward direction
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_lb(h):
        return h - 0.0  # h >= 0
    
    def constraint_ub(h):
        return 1.0 - h  # h <= 1
    
    # Optimizer initialization
    h0 = np.array([1.0 / n_points] * n_points)
    cons = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_lb},
        {'type': 'ineq', 'fun': constraint_ub}
    ]
    
    # We'll use a grid search with local improvements
    # due to tight budget, we'll not get too complex
    
    def obj_func(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    # Iterate to improve solution
    best_h = h0.copy()
    best_val = obj_func(best_h)
    
    # Simple gradient-based optimization attempt
    for i in range(50):
        if i % 10 == 0:
            # Add some noise if stuck
            h_noisy = np.clip(best_h + (np.random.rand(n_points) - 0.5) * 0.01, 0, 1)
            h_val = obj_func(h_noisy)
            if h_val < best_val:
                best_h = h_noisy.copy()
                best_val = h_val
        # Simple update step
        grad = np.zeros(n_points)
        # Approximate gradient
        eps = 1e-4
        for k in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[k] += eps
            h_minus[k] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            grad[k] = (obj_func(h_plus) - obj_func(h_minus)) / (2*eps)
        
        step = 0.005
        new_h = best_h - step * grad
        # Project back to constraints
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h) * dx
        new_h = new_h * (1.0 / new_sum) if new_sum > 0 else new_h
        new_val = obj_func(new_h)
        if new_val < best_val:
            best_h = new_h.copy()
            best_val = new_val
        if best_val < target_C:
            break

    return best_h
