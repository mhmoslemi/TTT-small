import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Initialize with triangular waveform
        n_points = 45
        h = np.linspace(0, 1, n_points)
        h = np.minimum(h, 1-h)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Gradient descent with perturbation
    eta = 0.01
    perturbation_rate = 0.05
    perturbation_interval = 50
    convergence_patience = 100
    patience = 0
    
    for iteration in range(int(budget_s * 10)):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5 and gradient
        c5 = evaluate_c5(h, n_points)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience = 0
        else:
            patience += 1
            
        # Compute gradient via correlation
        # For each shift k, compute ∂C5/∂h[i] = (1-h[i+k]) * dx
        # We approximate with finite differences
        grad = np.zeros(n_points)
        dx = 2.0 / n_points
        
        # Compute correlate of h and 1-h
        h_1_minus_h = np.correlate(h, 1-h, mode="full")
        # Gradients per position
        for i in range(n_points):
            grad[i] = h_1_minus_h[n_points-1+i] if i < n_points else 0.0

        # Update with gradient and clip
        h -= eta * grad
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally add random perturbation
        if iteration % perturbation_interval == 0:
            perturb = np.random.normal(0, perturbation_rate, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if converged
        if patience > convergence_patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode="full") * dx
    return np.max(corr)
