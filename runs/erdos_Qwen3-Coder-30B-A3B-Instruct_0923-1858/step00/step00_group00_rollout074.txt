import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_c5(h_values, dx):
    # Compute correlation with 1-h
    corr = correlate(h_values, 1 - np.array(h_values), mode="full")
    # Take maximum and scale by dx
    max_val = np.max(corr)
    return max_val * dx

def project_feasible(h_vals, n_points):
    # Ensure values are in [0,1]
    h_vals = np.clip(h_vals, 0, 1)
    # Scale to correct integral
    current_sum = np.sum(h_vals)
    if current_sum > 0:
        h_vals = h_vals * (n_points / 2) / current_sum
    return h_vals

def mutate_solution(h_vals, sigma):
    noise = np.random.normal(0, sigma, len(h_vals))
    new_h = h_vals + noise
    return project_feasible(new_h, len(h_vals))

def crossover(parent1, parent2, alpha=0.5):
    child = alpha * parent1 + (1 - alpha) * parent2
    return project_feasible(child, len(parent1))

def tournament_selection(population, fitnesses, tournament_size=5):
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def genetic_algorithm(initial_h, n_points, n_generations=200, population_size=50):
    # Initialize population
    population = [initial_h]
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        population.append(project_feasible(h, n_points))
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    dx = 2.0 / n_points
    mutation_sigma = 0.1
    
    for gen in range(n_generations):
        # Evaluate all individuals
        fitnesses = [evaluate_c5(h, dx) for h in population]
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Selection
        elites = [population[i] for i in np.argsort(fitnesses)[:population_size // 5]]
        
        # Generate new population
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate_solution(child, mutation_sigma)
            new_pop.append(child)
        
        population = new_pop
        mutation_sigma *= 0.95
        
        if gen % 20 == 0:
            print(f"Generation {gen}: Best C5 = {best_c5:.6f}")
    
    return best_h, best_c5

def gradient_descent(h0, max_iter=150, learning_rate=0.01):
    h = h0.copy()
    dx = 2.0 / len(h)
    epsilon = 1e-5
    
    for i in range(max_iter):
        c5_old = evaluate_c5(h, dx)
        if i % 50 == 0:
            print(f"GD Iter {i}: C5 = {c5_old:.6f}")
        
        # Gradient approximation with finite differences
        grad = np.zeros_like(h)
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project_feasible(h_minus, len(h))
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update
        h = h - learning_rate * grad
        h = project_feasible(h, len(h))
        
        # Adaptive learning rate
        learning_rate *= 0.995
        
        # Early stopping if no improvement
        c5_new = evaluate_c5(h, dx)
        if c5_new > c5_old * 0.9999:
            break
    
    return h, evaluate_c5(h, dx)
