import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Compute C5 for a given h"""
    dx = 2.0 / len(h)
    # Cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto feasible set: [0,1] and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to have sum = n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, n_points):
    """Compute approximate gradient of C5 with respect to h"""
    dx = 2.0 / n_points
    # We will compute gradient for max of cross-correlation
    # This is a simplified heuristic version, since full gradient is complex
    # We compute correlations for all shifts, and pick the one that maximizes
    # We then estimate a gradient with finite-differences for the maximizing shift
    # For simplicity, we use a Gumbel-type approach for estimation
    # Better approach: apply implicit differentiation technique
    # but we will do simpler version here with finite differences

    # For now, a crude approximation: since overlap involves convolution,
    # we compute the gradient by finite differences on that.
    # This is approximate but enough for decent optimization

    # Instead, we'll compute a simplified gradient via finite difference
    # for the top entry in the correlation
    h_copy = h.copy()
    eps = 1e-6
    max_val = -np.inf
    best_shift = 0

    # Evaluate all correlations to find max shift
    corr = correlate(h_copy, 1 - h_copy, mode="full")
    max_loc = np.argmax(corr)
    best_shift = max_loc - (len(h) - 1)
    max_val = corr[max_loc]

    # Now use finite difference to estimate gradient
    g = np.zeros_like(h)
    h_pert = h_copy.copy()
    for i in range(len(h)):
        h_pert[i] += eps
        corr_pert = correlate(h_pert, 1 - h_pert, mode="full")
        max_loc_pert = np.argmax(corr_pert)
        val_pert = corr_pert[max_loc_pert]
        g[i] = (val_pert - max_val) / eps
        h_pert[i] = h_copy[i]
    
    # Normalized gradient
    return g / (np.max(np.abs(g)) + 1e-12) if np.max(np.abs(g)) > 0 else g

def optimize_step(h, n_points, alpha=1e-3):
    """Perform one optimization step using projected gradient descent"""
    # Compute gradient of maximum correlation
    g = compute_gradient(h, n_points)
    
    # Update h
    h_new = h - alpha * g
    
    # Project back to feasible set
    h_new = project_h(h_new, n_points)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 50  # Start with lower resolution
    dx = 2.0 / n_points
    
    # Initialize h either from provided initial or generate one

        h = h[:n_points]

        h = project_h(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points)
    
    # Optimization loop
    alpha = 1e-2
    prev_c5 = evaluate_h(h)
    stagnation_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        h_new = optimize_step(h, n_points, alpha)
        c5_new = evaluate_h(h_new)
        if c5_new < prev_c5:
            h = h_new
            prev_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count % 10 == 0:
                alpha *= 0.9  # Reduce learning rate
        
        # Stop if improvement is small
        if stagnation_count > 100 and abs(prev_c5 - c5_new) < 1e-5:
            break
            
        if iteration > 5000:
            break
    
    # Now upscale to higher resolution
    high_res_points = 70
    h_high_res = np.interp(np.linspace(0, 2, high_res_points), 
                           np.linspace(0, 2, n_points), h)
    h_high_res = project_h(h_high_res, high_res_points)
    
    # Fine-tune at higher resolution
    alpha = 1e-3
    prev_c5 = evaluate_h(h_high_res)
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        h_new = optimize_step(h_high_res, high_res_points, alpha)
        c5_new = evaluate_h(h_new)
        if c5_new < prev_c5:
            h_high_res = h_new
            prev_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count % 10 == 0:
                alpha *= 0.9
                
        if stagnation_count > 100 and abs(prev_c5 - c5_new) < 1e-5:
            break
            
        if time.time() - start_time > budget_s - 3:
            break

    return (h_high_res.tolist(), prev_c5, high_res_points)
