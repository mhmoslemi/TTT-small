        overlap = signal.correlate(h_arr, 1 - h_arr, mode='full')
        # Only the relevant shifts
        shift_range = len(h_arr) - 1

        return overlap * dx
    
    def compute_C5(h):
        return np.max(eval_overlap(h))
    
    current_h = h_init.copy()

        # Overlap constraints
        h_arr = np.array(h_vals)
        overlap_full = signal.correlate(h_arr, 1 - h_arr, mode='full')
        overlap_full = overlap_full[len(h_arr)-1:]

        # For large-scale constraint inclusion, we'll use iterative approximation
        return constraints
    
    # Iterative cutting plane approach
    max_iter = 200
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Construct overlap problem small enough for CVXPY
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraint matrix approach to limit memory and computation
        # Compute current overlap values
        current_overlap = eval_overlap(current_h)
        # Find max violation
        max_overlaps = np.argsort(current_overlap)[::-1][:n_points//3]
        
        # Build a minimal constraint set
        constraints = [cp.sum(h_vars) == n_points / 2,
                       h_vars >= 0,
                       h_vars <= 1]
        
        # Add only significant constraints to avoid full dimensionality
        for k in max_overlaps:
            # For shift k, overlapping indices in range [0, n_points-k-1]
            # overlap[k] is convolution result
            left_total = 0
            for i in range(min(k+1, n_points)):
                if i + k < n_points:
                    left_total += h_vars[i] * (1.0 - h_vars[i + k])
            constraints.append(left_total <= C5_var * dx)
        
        # Add full problem formulation constraints
        # Actually, we're doing approximate constraints
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve with different tolerances or solvers
            prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, verbose=False)
        except:
            # Fallback solution - force return of best found
            break
            
        if prob.status != cp.OPTIMAL:
            break
            
        new_h = h_vars.value
        new_C5 = C5_var.value

        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Check improvement
        if new_C5 < current_C5 - 1e-6:
            current_C5 = new_C5
            current_h = new_h
        else:
            # Step size decreasing or stagnant
            break
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(current_h)
    
    return (current_h.tolist(), final_C5, n_points)
