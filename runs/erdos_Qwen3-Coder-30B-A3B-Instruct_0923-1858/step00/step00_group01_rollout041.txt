import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral C_5 for given h."""
    dx = 2.0 / len(h)
    # Compute the correlation of h with (1-h)
    corr = correlate(h, 1-h, mode='full')
    # Take maximum correlation value and multiply by dx
    c5 = np.max(corr) * dx
    return c5

def repair_h(h):
    """Ensure h satisfies constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    n_points = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n = len(parent1)
    # Random crossover point
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a single h vector."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Small Gaussian perturbation
            h[i] += np.random.normal(0, 0.05)
    return repair_h(h)

def evolve_population(pop, fitnesses, pop_size, generation):
    """Evolve a population for one generation."""
    # Selection: keep top 20%
    sorted_indices = np.argsort(fitnesses)
    elites = [pop[i] for i in sorted_indices[:pop_size//5]]
    
    # Generate new children
    new_pop = elites.copy()
    while len(new_pop) < pop_size:
        # Sample two parents for crossover
        p1 = np.random.choice(elites)
        p2 = np.random.choice(elites)
        c1, c2 = crossover(p1, p2)
        c1 = mutate(c1)
        c2 = mutate(c2)
        new_pop.extend([c1, c2])
    
    # Keep population size
    new_pop = new_pop[:pop_size]
    
    return new_pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 47
    pop_size = 80

    # Initial population
    population = []

        h0 = repair_h(h0)
        population.append(h0)
    
    # Add random samples to reach population size
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Evolution loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all members
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(c5)
            
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Evolve population
        population = evolve_population(population, fitnesses, pop_size, gen)
    
    return best_h, best_c5, n_points
