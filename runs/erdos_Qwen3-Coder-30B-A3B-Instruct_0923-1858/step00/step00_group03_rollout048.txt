import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize to sum to 1
    h = h * (n_points / 2.0) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Iteration counter
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute overlaps
        conv = correlate(h, 1 - h, mode='full')
        # Only consider valid shifts (k from 0 to n_points - 1)
        overlaps = conv[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        if max_overlap < 1e-10:
            break
            
        # Find indices that contribute to this overlap
        max_contributors = []
        for i in range(n_points - k_max):
            if h[i] > 0 and h[i + k_max] < 1:
                contrib = h[i] * (1 - h[i + k_max])
                max_contributors.append((i, contrib))
                
        if not max_contributors:
            # No beneficial adjustments
            break
            
        # Sort by contribution
        max_contributors.sort(key=lambda x: x[1], reverse=True)
        
        # Take the top few for adjustment
        candidates = max_contributors[:min(5, len(max_contributors))]
        
        # Try adjusting one candidate
        best_adjustment = None
        best_reduced = 0
        
        for i, _ in candidates:
            if i + k_max >= n_points:
                continue
            # Adjust h[i] and h[i + k_max] to reduce the overlap
            # Reduce h[i] by some small amount, increase h[i + k_max] by corresponding amount
            delta = min(h[i], 1 - h[i + k_max], 0.1)
            if delta < 1e-8:
                continue
            # Compute effect of the adjustment
            old_val = h[i] * (1 - h[i + k_max])
            h_new = h.copy()
            h_new[i] -= delta
            h_new[i + k_max] += delta
            # Clamp
            h_new = np.clip(h_new, 0, 1)
            # Re-normalize
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Evaluate new overlap
            conv_new = correlate(h_new, 1 - h_new, mode='full')
            overlaps_new = conv_new[n_points-1:] * dx
            new_max = np.max(overlaps_new)
            
            reduced = max_overlap - new_max
            if reduced > best_reduced:
                best_reduced = reduced
                best_adjustment = (i, delta)
                
        if best_adjustment is None:
            # No beneficial adjustment found
            break
            
        i, delta = best_adjustment
        h[i] -= delta
        h[i + k_max] += delta
        # Clamp
        h = np.clip(h, 0, 1)
        # Re-normalize
        h = h * (n_points / 2.0) / np.sum(h)
                
    # Final evaluation
    conv = correlate(best_h, 1 - best_h, mode='full')
    overlaps = conv[len(best_h)-1:] * dx
    final_c5 = np.max(overlaps)
    
    return (best_h.tolist(), float(final_c5), n_points)
