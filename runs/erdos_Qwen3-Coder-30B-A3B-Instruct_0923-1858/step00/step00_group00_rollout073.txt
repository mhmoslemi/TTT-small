import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, dx):
    """Evaluate C5 for given h values"""
    n = len(h_values)
    # Compute correlation of h with (1-h) 
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum value is the maximum of all correlations
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    population_size = 50

    population = [initial_h.copy()]
    dx = 2.0 / n_points
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    start_time = time.time()
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            
        # Update best solution
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_solution = population[min_idx].copy()
            
        # Tournament selection
        new_population = []
        elite_size = population_size // 5
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Crossover with another random individual
            parent2_idx = np.random.choice(population_size)
            parent1 = population[winner_idx]
            parent2 = population[parent2_idx]
            
            # Blend crossover
            child = crossover_alpha * parent1 + (1 - crossover_alpha) * parent2
            
            # Add Gaussian noise
            child += np.random.normal(0, mutation_sigma, n_points)
            
            # Project to feasible region
            child = project_to_feasible(child, n_points)
            
            new_population.append(child)
            
        population = new_population
        
        # Decay mutation
        mutation_sigma *= 0.98
        
    return best_solution, best_c5

def gradient_descent(h0, n_points, budget_s):
    """Gradient descent refinement"""
    dx = 2.0 / n_points
    epsilon = 1e-5
    lr = 0.01
    max_iter = 200
    start_time = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project_to_feasible(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)

        h = h - lr * grad

        # Check improvement
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive learning rate
        lr *= 0.995
    
    return best_h, best_c5
