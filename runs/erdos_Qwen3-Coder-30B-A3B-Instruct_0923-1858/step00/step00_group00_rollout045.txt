import numpy as np

import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral C_5 for given h"""
    n = len(h)
    # Compute all correlations of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of these values
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm(initial_h, n_points, budget_s, pop_size=50, generations=100):

    dx = 2.0 / n_points

    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):

        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    
    # Evolution parameters
    alpha = 0.5
    mutation_sigma = 0.1
    tournament_size = 5
    stagnation_count = 0
    prev_best = best_c5
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_overlap(h, dx)
            fitness.append(c5)
        
        # Selection - tournament
        new_population = []
        elites = np.argsort(fitness)[:pop_size // 5]
        for i in elites:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(range(pop_size), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Crossover with another random individual
            other_idx = np.random.choice(range(pop_size))
            if other_idx != winner_idx:
                child = alpha * population[winner_idx] + (1 - alpha) * population[other_idx]
            else:
                child = population[winner_idx].copy()
                
            # Mutation
            if np.random.rand() < 0.8:  # 80% chance of mutation
                noise = np.random.normal(0, mutation_sigma, n_points)
                child = child + noise
            
            # Project to feasible region
            child = project_to_feasible(child, n_points)
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update mutation rate
        mutation_sigma *= 0.95
        
        # Check for stagnation
        if stagnation_count > 20:
            break
            
    return best_h, best_c5

def gradient_descent(h0, n_points, budget_s):
    """Refine solution using gradient descent"""
    dx = 2.0 / n_points
    start_time = time.time()
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    eta = 0.01  # Initial learning rate
    
    # Finite difference gradient estimation
    epsilon = 1e-5
    
    for i in range(1000):  # Max iterations
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Estimate gradient using finite differences for each parameter
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            h_minus = project_to_feasible(h_minus, n_points)
            f_plus = evaluate_overlap(h_plus, dx)
            f_minus = evaluate_overlap(h_minus, dx)
            grad[j] = (f_plus - f_minus) / (2 * epsilon)
        
        # Update with gradient descent
        h = h - eta * grad
        h = project_to_feasible(h, n_points)
        
        # Evaluate new solution
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adapt learning rate
        eta = 0.01 / np.sqrt(i + 1)
    
    return best_h, best_c5
