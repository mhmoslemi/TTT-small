import numpy as np
import time
from scipy import signal
import random

def evaluate_solution(h, dx):
    # Compute the overlap integral for all shifts
    # h is a binary array
    # we compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # This is equivalent to max_k sum h[i] * (1 - h[i+k]) * dx
    # which is same as max_k sum h[i] - h[i]*h[i+k] * dx
    # which is max_k sum h[i] * dx - sum h[i] * h[i+k] * dx
    
    # The maximum overlap is over all valid k
    # This is the same as the maximum value of the cross-correlation of h with (1-h)
    # with mode='full', but we only want the valid shifts [0, n_points-1] for a valid overlap
    
    # Efficient way with scipy
    # The cross-correlation of h with 1-h
    h1 = 1 - h  # this is what the overlap integral uses
    corr = signal.correlate(h, h1, mode='full')
    # corr[0] corresponds to shift -(n_points-1)
    # corr[2*n_points-2] corresponds to shift (n_points-1)  
    # Valid shifts are 0, 1, ..., n_points-1 (i.e. indices n_points-1 to 2*n_points-2)
    max_correlation = np.max(corr[n_points-1:2*n_points-1])
    return max_correlation * dx

def make_binary(h_values):
    """convert h to binary array with exactly half 1s"""
    h = np.array(h_values)
    h = np.round(h).astype(int)
    ones = np.sum(h)
    needed = n_points // 2
    if ones == needed:
        return h
    if ones > needed:
        # find indices of 1s to flip to 0
        ones_indices = np.where(h == 1)[0]
        # flip the first (ones - needed) ones to 0
        flip_indices = ones_indices[:ones - needed]
        h[flip_indices] = 0
    else:
        # find indices of 0s to flip to 1
        zeros_indices = np.where(h == 0)[0]
        # flip the first (needed - ones) zeros to 1
        flip_indices = zeros_indices[:needed - ones]
        h[flip_indices] = 1
    return h

def get_initial_h():
    """use initial_h_values if available, otherwise construct one"""
    try:
        global initial_h_values
        h_0 = np.array(initial_h_values)
        return make_binary(h_0)
    except NameError:
        # otherwise construct a uniform binary array 
        h_0 = np.zeros(n_points)
        h_0[:n_points//2] = 1
        return h_0.astype(int)

def local_search_simulated_annealing(seed, time_budget_s):
    """employs a local search with simulated annealing to minimize the maximum overlap"""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # initialization
    h = get_initial_h()
    dx = 2.0 / n_points
    current_C5 = evaluate_solution(h, dx)
    best_C5 = current_C5
    best_h = h.copy()
    
    # parameters for annealing
    temp = 1.0
    alpha = 0.999
    max_iter = 100000
    t0 = time.time()
    
    for i in range(max_iter):
        if time.time() - t0 > time_budget_s - 1:
            break
            
        # try to improve by flipping two bits
        # find locations that are 1 and 0 to consider swapping
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) < 1 or len(zeros_idx) < 1:
            continue
            
        i1 = np.random.choice(ones_idx)
        i0 = np.random.choice(zeros_idx)
        
        # make a temporary copy
        h_temp = h.copy()
        h_temp[i1] = 0
        h_temp[i0] = 1
        
        new_C5 = evaluate_solution(h_temp, dx)
        delta = new_C5 - current_C5
        
        # accept condition
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_temp
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                
        temp *= alpha
        
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    global n_points
    n_points = 70
    
    h_best, c5_value = local_search_simulated_annealing(seed, budget_s)
    
    # normalize to satisfy constraint sum(h) * dx = 1
    # not really needed here since we enforce exact 1s/0s with binary conversion
    # but safe to do just in case
    h_best = h_best.astype(float)
    h_best = h_best / n_points * 2
        
    # recompute c5 with correct dx and binary form
    h_binary = make_binary(h_best)
    dx = 2.0 / n_points
    c5_value = evaluate_solution(h_binary, dx)
    
    return (h_binary.tolist(), c5_value, n_points)
