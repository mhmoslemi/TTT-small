import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a triangular shape
        h = np.linspace(0, 1, n_points)
        h = np.minimum(h, 1 - h)
        h = h / np.sum(h) * n_points / 2
    
    # Rescale to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_bound(h, n_points)
    
    # Gradient descent parameters
    learning_rate = 0.01
    decay_rate = 0.95
    perturb_rate = 0.05
    perturb_freq = 50
    grad_freq = 100
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5
        current_c5 = evaluate_bound(h, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute gradient (using finite differences for direction)
        grad = compute_gradient(h, n_points)
        
        # Update with gradient
        h = h - learning_rate * grad
        
        # Project to feasible domain [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturbation
        if iteration % perturb_freq == 0:
            noise = np.random.normal(0, perturb_rate, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Learning rate decay
        if iteration % grad_freq == 0 and iteration > 0:
            learning_rate *= decay_rate
    
    # Final evaluation of best solution
    final_c5 = evaluate_bound(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_bound(h, n_points):
    """Evaluate the maximum overlap integral C5."""
    dx = 2.0 / n_points
    # Compute the full correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value of correlation (over all shifts)
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, n_points):
    """Compute approximate gradient of the objective function."""
    dx = 2.0 / n_points
    
    # Compute correlation with 1-h 
    corr_h_1mh = np.correlate(h, 1 - h, mode="full")
    
    # Find the shift index that maximizes the correlation
    max_shift_idx = np.argmax(corr_h_1mh)
    
    # Compute approximate gradient 
    # We consider how changing h[i] affects the maximum correlation
    # This is a rough approximation using the correlation at max shift
    
    # Find the local derivative using finite differences
    grad = np.zeros_like(h)
    
    h_plus = h.copy()
    h_minus = h.copy()
    
    for i in range(len(h)):
        eps = 1e-4
        h_plus[i] += eps
        h_minus[i] -= eps
        
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Reset for next iteration
        h_plus[i] = h[i]
        h_minus[i] = h[i]
        
    return grad
