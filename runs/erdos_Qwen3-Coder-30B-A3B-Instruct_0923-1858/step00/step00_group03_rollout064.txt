import numpy as np
import time
from scipy import signal

def binary_convert(h):
    """Converts h to binary array with exact sum constraint."""
    h = np.round(h).astype(int)
    target_sum = len(h) // 2
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    elif current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = current_sum - target_sum
        indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(indices, size=diff, replace=False)
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        diff = target_sum - current_sum
        indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(indices, size=diff, replace=False)
        h[flip_indices] = 1
    return h

def evaluate(h, dx):
    """Computes C5 = max of overlap integral."""
    # Compute cross-correlation: h * (1 - h) in full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take only the valid part (max shift = len(h) - 1)
    corr = corr[len(h)-1:]
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Default initialization - semi-uniform binary pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Randomize a bit
        indices = np.random.permutation(n_points)[:n_points//10]
        for idx in indices:
            h[idx] = 1 - h[idx]
    
    h = binary_convert(h)
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Create neighbor solution by flipping two bits
        # Find 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Create new solution
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        
        # Rebalance to exact constraint
        new_h = binary_convert(new_h)
        
        # Evaluate
        new_c5 = evaluate(new_h, dx)
        
        # Accept or reject
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        
        T *= alpha
    
    return (best_h.tolist(), best_c5, n_points)
