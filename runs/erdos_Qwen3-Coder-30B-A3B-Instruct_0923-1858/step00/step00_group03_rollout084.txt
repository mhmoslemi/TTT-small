import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial solution if available
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialize variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    start_time = time.time()
    
    # Gradient descent parameters
    alpha = 0.1
    decay_rate = 0.9
    decay_interval = 100
    tolerance = 1e-6
    
    # Precompute all possible correlations
    def compute_overlap(h_vec):
        # Using scipy's correlate for full correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return corr * dx
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % decay_interval == 0 and iteration > 0:
            alpha *= decay_rate
            
        # Compute current overlaps
        overlaps = compute_overlap(h)
        
        # Find the maximum overlap
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute the gradient for this maximum overlap
        gradient = np.zeros_like(h)
        k = max_k - (n_points - 1)
        
        # The gradient is partial derivative of max_overlap w.r.t. h[i]
        # For shift k, the contribution to overlap is h[i] * (1 - h[i+k])
        # So partial derivative w.r.t. h[i] is (1 - h[i+k])
        # And w.r.t. h[i+k] is -h[i]
        
        for i in range(n_points):
            if 0 <= i + k < n_points:
                gradient[i] += (1 - h[i + k])
            if 0 <= i - k < n_points:
                gradient[i - k] -= h[i]
        
        # Update h
        h = h - alpha * gradient
        
        # Project h to feasible region
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust sum to exactly n_points/2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        
        if abs(current_sum - target_sum) > 1e-8:
            diff = target_sum - current_sum
            # Distribute the difference
            h += diff / n_points  # Equal distribution
        
        # Ensure bounds after adjustment
        h = np.clip(h, 0, 1)
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        iteration += 1
        
        # Check for convergence
        if abs(best_c5 - new_c5) < tolerance:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
