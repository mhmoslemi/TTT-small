import numpy as np
from scipy.signal import correlate
import time

def evaluate_C5(h, dx):
    # Compute the overlap integral
    conv = correlate(h, 1 - h, mode="full")
    max_conv = np.max(conv) * dx
    return max_conv

def project_h(h, n_points):
    # Project h to [0,1] and rescale to sum to n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def nelder_mead_step(vertices, values, alpha=1.0, gamma=2.0, beta=0.5, rho=0.5):
    # Sort vertices by function value
    indices = np.argsort(values)
    vertices = vertices[indices]
    values = values[indices]
    
    # Best, worst, second worst
    best = vertices[0]
    worst = vertices[-1]
    second_worst = vertices[-2]
    
    # Compute centroid without worst
    centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    reflected = centroid + alpha * (centroid - worst)
    reflected = project_h(reflected, len(reflected))
    reflected_value = evaluate_C5(reflected, 2.0 / len(reflected))
    
    # Check if reflected is better than best
    if reflected_value < values[0]:
        # Expansion
        expanded = centroid + gamma * (reflected - centroid)
        expanded = project_h(expanded, len(expanded))
        expanded_value = evaluate_C5(expanded, 2.0 / len(expanded))
        if expanded_value < reflected_value:
            vertices[-1] = expanded
            values[-1] = expanded_value
        else:
            vertices[-1] = reflected
            values[-1] = reflected_value
    elif reflected_value < values[-2]:
        # Accept reflected
        vertices[-1] = reflected
        values[-1] = reflected_value
    else:
        # Contraction
        if reflected_value < values[-1]:
            # Outside contraction
            contracted = centroid + beta * (reflected - centroid)
        else:
            # Inside contraction
            contracted = centroid + beta * (worst - centroid)
        contracted = project_h(contracted, len(contracted))
        contracted_value = evaluate_C5(contracted, 2.0 / len(contracted))
        if contracted_value < values[-1]:
            vertices[-1] = contracted
            values[-1] = contracted_value
        else:
            # Shrink
            for i in range(1, len(vertices)):
                vertices[i] = vertices[0] + rho * (vertices[i] - vertices[0])
                vertices[i] = project_h(vertices[i], len(vertices[i]))
            values = [evaluate_C5(v, 2.0 / len(v)) for v in vertices]
    
    return vertices, values
