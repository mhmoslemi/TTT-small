import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral C5"""
    # Compute all overlaps for each shift k
    corr = signal.correlate(h, 1 - h, mode='full')
    # Only consider valid shifts (corresponding to the original range)
    n = len(h)
    overlap = corr[n-1:] * dx  # Take the valid portion
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_original = np.array(initial_h_values)
        n_points = len(h_original)
        h = np.round(h_original).astype(int)
    else:
        # Default construction if initial_h_values is not available
        n_points = 70
        h = np.ones(n_points // 2, dtype=int)
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    
    # Ensure correct sum
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum != target_sum:
        # Change some 1s to 0s to match the target
        diff = current_sum - target_sum
        if diff > 0:
            # Need to flip some 1s to 0s
            ones_indices = np.where(h == 1)[0]
            flip_indices = np.random.choice(ones_indices, size=diff, replace=False)
            h[flip_indices] = 0
        else:
            # Need to flip some 0s to 1s
            zeros_indices = np.where(h == 0)[0]
            flip_indices = np.random.choice(zeros_indices, size=-diff, replace=False)
            h[flip_indices] = 1
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Compute initial C5
    best_C5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:  # Leave some margin
        iteration += 1
        
        # Find max overlap and its k
        corr_full = signal.correlate(h, 1 - h, mode='full')
        overlap_vals = corr_full[n_points-1:] * dx
        max_k_idx = np.argmax(overlap_vals)
        max_overlap = overlap_vals[max_k_idx]
        
        # If no improvement can be hoped for, break early
        if max_overlap < 1e-6:
            break
            
        # For each position, calculate change in overlap if we flip it,
        # and how that changes contributions to max_k_idx
        best_change = 0
        best_pos = -1
        best_from = -1
        
        # We will try flipping each element
        for i in range(n_points):
            # Calculate current contribution to overlap at max_k_idx
            # Original contribution to overlap[0] if h[i]=1 and h[i+k]=1
            # i.e., 1 if h[i]=1 and h[i+k]=1, else 0
            # But the correlation result we compute is for the kernel (1-h)
            
            # Switch from original direction to more direct approach:
            # For each flip, measure how it changes the overlap at max_k_idx
            current_contribution = h[i] * (1 - h[(i + max_k_idx) % n_points])
            # Maybe flip it (i.e., from 1 to 0, or 0 to 1)
            flipped_val = 1 - h[i]
            # Calculate new contribution
            new_contribution = flipped_val * (1 - h[(i + max_k_idx) % n_points])
            # The change in overlap for this flip is
            delta = new_contribution - current_contribution
            
            # Is this useful? only if we can flip without violating constraint
            # we need to save a 0 to make it 1, or we have a 1 to make it 0
            
            if h[i] == 1:
                # Try to flip it to 0
                # We need to find a 0 to make 1, else this move is invalid
                zeros = np.where(h == 0)[0]
                if len(zeros) == 0:
                    continue  # Can't flip this one
                # Find a best zero among those that help in max_k_idx
                # We can do slightly better by looking more globally
                
                # Just early stop if valid target is found that increases overlap
                if delta < best_change:
                    best_change = delta
                    best_pos = i
                    best_from = 1  # flip from 1 to 0
            else:
                # Try to flip it to 1
                # Potentially useful only if it matches a 0 to convert in other place
                # for now, just evaluate positive change (if delta is more negative)
                # Threshold to reduce noise: just focus on improving this specific shift
                if delta < best_change:
                    best_change = delta
                    best_pos = i
                    best_from = 0  # flip from 0 to 1
        
        # If we found a better shift and possible flip
        if best_change < -1e-10:
            # Apply the flip
            if best_from == 1:
                # Flip 1 -> 0
                h[best_pos] = 0
                # Find another 0 to flip to 1
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    flip2 = zeros[0]
                    h[flip2] = 1
            else:
                h[best_pos] = 1
                # Find a 1 to flip to 0
                ones = np.where(h == 1)[0]
                if len(ones) > 0:
                    flip2 = ones[0]
                    h[flip2] = 0
            
            # Recalculate max overlap and update best solution
            new_C5 = evaluate_overlap(h, dx)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                
        # Occasionally reset to avoid getting stuck
        if iteration % 100 == 0:
            # Reset to a slight random variation of current best
            # Not that regular restarts improve results much here
            pass
            
    return (list(best_h), best_C5, n_points)
