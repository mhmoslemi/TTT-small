import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C5 bound
    max_corr = np.max(corr)
    return max_corr * dx

        h = h / np.sum(h) * 2.0 / n_points
        h = np.clip(h, 0, 1)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Timing
    start_time = time.time()
    
    # Helper function for optimization
    def objective(h_vals):
        h_arr = np.array(h_vals)
        return evaluate_h(h_arr)

        h_opt = h

    final_c5 = evaluate_h(h_opt)
