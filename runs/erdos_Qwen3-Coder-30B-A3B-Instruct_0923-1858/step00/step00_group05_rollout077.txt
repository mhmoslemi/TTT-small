import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of these correlations
    max_overlap = np.max(corr)
    return max_overlap * dx
