import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    n = len(h)
    overlaps = []
    for k in range(-n+1, n):
        # Compute correlation for shift k
        corr = correlate(h, np.ones_like(h), mode="full")
        # Adjust for h(x)(1-h(x+k)) -- complex to do directly
        # Instead: compute all valid pairs
        # This is a math-heavy step, simplified by using a formulation
        # Direct evaluation of sum(h[i] * (1 - h[(i+k)%n]))
        if k >= 0:
            shifted = np.concatenate([h[k:], h[:k]])
        else:
            shifted = np.concatenate([h[k:], h[:k]])
        overlap = np.sum(h * (1 - shifted))
        overlaps.append(overlap * dx)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    global initial_h_values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = len(h0)
    else:
        # Initialize with 45 points, based on problem description
        n_points = 45
        h0 = np.full(n_points, 0.5)
    
    # Rescale to satisfy constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Binary search parameters
    low = 0.0
    high = 0.38085
    best_solution = h0.copy()
    best_c5 = high
    tolerance = 1e-6
    
    dx = 2.0 / n_points
    max_iter = 30
    iter_count = 0
    
    # Phase 1: binary search with LP feasibility checks
    while low < high - tolerance and time.time() - start_time < budget_s - 1:
        iter_count += 1
        mid = (low + high) / 2.0
        
        # Construct LP feasibility problem
        h_vars = cp.Variable(n_points, name='h')
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2.0,
        ]
        
        # Add overlap constraints
        # For each shift k, compute sum h[i] * (1 - h[(i+k)%n])
        valid_k_values = list(range(-n_points+1, n_points))
        for k in valid_k_values:
            # Create indices and sum efficiently
            if k >= 0:
                shifted = cp.hstack([h_vars[k:], h_vars[:k]])
            else:  # k < 0
                shifted = cp.hstack([h_vars[k:], h_vars[:k]])
            # Overlap constraint: h[i] * (1 - h[(i+k)%n]) <= mid * dx
            # Sum of products
            overlap_sum = cp.sum(cp.multiply(h_vars, 1-shifted))
            # We want to ensure that sum <= mid * dx * n
            # Actually, here we interpret mid as max allowed value
            # Our formula is: h(x)(1-h(x+k)) dx
            # Full overlapping integral is the correlation scaled
            constraints.append(overlap_sum <= mid * n_points * dx)
        
        # Define objective (trivial as we are only checking feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                sol = h_vars.value
                if sol is not None:
                    best_solution = sol.copy()
                    best_c5 = mid
                    high = mid
            else:
                low = mid
        except:
            low = mid
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation to check feasibility
    best_overlap = evaluate_overlap(best_solution, dx)
    
    # Ensure result is within bounds
    final_h = np.clip(best_solution, 0, 1)
    
    # Normalize to correct integral
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Final check of constructed solution
    final_overlap = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_overlap, len(final_h))
