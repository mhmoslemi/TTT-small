import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitnesses, k=5):
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.3):
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, sigma=0.01):
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, sigma, len(h))
    h = h + noise * mask
    return project_to_feasible(h, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    pop_size = 30
    elite_rate = 0.1
    crossover_rate = 0.7
    mutation_rate = 0.1
    tournament_size = 5
    max_generations = 1000
    
    # Initial population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = [np.random.rand(n_points) for _ in range(pop_size)]
    population = [project_to_feasible(h, n_points) for h in population]
    
    # Fitness tracking
    fitnesses = [evaluate(h, 2.0/n_points) for h in population]
    best_idx = np.argmin(fitnesses)
    global_best = population[best_idx]
    global_best_fitness = fitnesses[best_idx]
    generation = 0
    
    # Main loop
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update global best
        if fitnesses[0] < global_best_fitness:
            global_best = population[0]
            global_best_fitness = fitnesses[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 200:
            break
            
        # Elitism
        new_population = population[:int(elite_rate * pop_size)]
        
        # Generate offspring
        while len(new_population) < pop_size:
            if np.random.random() < crossover_rate:
                parent1 = tournament_selection(population, fitnesses, tournament_size)
                parent2 = tournament_selection(population, fitnesses, tournament_size)
                child = blend_crossover(parent1, parent2)
                child = mutate(child, mutation_rate)
            else:
                child = tournament_selection(population, fitnesses, tournament_size)
                child = mutate(child, mutation_rate)
            new_population.append(child)
            
        population = new_population
        fitnesses = [evaluate(h, 2.0/n_points) for h in population]
        generation += 1
        
    return (global_best, global_best_fitness, n_points)
