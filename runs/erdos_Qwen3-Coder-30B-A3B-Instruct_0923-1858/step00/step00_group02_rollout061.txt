    # Parameters
    n_points = 100  # Lower resolution for faster optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    if 'initial_h_values' in globals():

        h = h * (target_sum / np.sum(h))
    
    # Helper functions
    def compute_C5(h):
        # Use full correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        # Take the maximum value (properly scaled)
        return np.max(corr) * dx
    
    def project_h(h):
        # Clip to [0,1] and rescale to sum=target_sum
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h
    
    # Main optimization loop with trust region
    best_h = h.copy()
    best_C5 = compute_C5(h)
    trust_radius = 0.1
    max_iter = 1000
    iter_count = 0

        iter_count += 1
        current_C5 = compute_C5(h)
        
        # Updated strategy: use gradient ascent in a simplified manner
        # Approximate the gradient numerically
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus)
            C5_plus = compute_C5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_h(h_minus)
            C5_minus = compute_C5(h_minus)
            
            grad[i] = (C5_plus - C5_minus) / (2 * epsilon)
        
        # Compute descent direction via projected gradient descent
        d = -grad
        # Project d to satisfy sum constraint
        mean_d = np.mean(d)
        d = d - mean_d  # This ensures sum(d) = 0
        
        # Line search with backtracking
        alpha = 1.0
        max_backtracks = 20
        for _ in range(max_backtracks):
            h_new = h + alpha * d

            new_C5 = compute_C5(h_new)
            if new_C5 < current_C5 or alpha < 1e-8:
                break
            alpha *= 0.5
        
        h = h_new
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
            
    # Final validation
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
