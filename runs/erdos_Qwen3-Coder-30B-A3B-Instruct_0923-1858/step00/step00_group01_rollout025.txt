import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def clip_and_rescale(h):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = 1.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

    # Gradient-based optimization loop
    iter_count = 0
    learning_rate = 0.01
    max_iter = 1000
    
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h)
        
        # Approximate gradient using finite differences
        grad = np.zeros_like(h)

        # Update with gradient descent
        h = h - learning_rate * grad
