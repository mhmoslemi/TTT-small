import numpy as np
from scipy.signal import correlate
import time

def evaluate(h, dx):
    """Evaluate the overlap integral C5 for given step function h."""
    h = np.array(h)
    conv = correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val

def project(h, n_points):
    """Project h onto feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.maximum(0, np.minimum(1, h))
    h = h * (n_points / 2) / np.sum(h)
    return h

def nelder_mead_step(vertices, f_values, alpha=1.0, gamma=2.0, beta=0.5, rho=0.5):
    """Perform one Nelder-Mead step on simplex"""
    n = len(vertices) - 1
    # Sort vertices by function value
    idx = np.argsort(f_values)
    vertices = [vertices[i] for i in idx]
    f_values = [f_values[i] for i in idx]
    
    # Compute centroid of best n vertices
    centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    h_r = centroid + alpha * (centroid - vertices[-1])
    h_r = project(h_r, len(h_r))
    f_r = evaluate(h_r, 2.0 / len(h_r))
    
    if f_values[0] <= f_r < f_values[-2]:
        # Accept reflection
        vertices[-1] = h_r
        f_values[-1] = f_r
    elif f_r < f_values[0]:
        # Expansion
        h_e = centroid + gamma * (centroid - vertices[-1])
        h_e = project(h_e, len(h_e))
        f_e = evaluate(h_e, 2.0 / len(h_e))
        if f_e < f_r:
            vertices[-1] = h_e
            f_values[-1] = f_e
        else:
            vertices[-1] = h_r
            f_values[-1] = f_r
    else:
        # Contraction
        h_c = centroid + beta * (centroid - vertices[-1])
        h_c = project(h_c, len(h_c))
        f_c = evaluate(h_c, 2.0 / len(h_c))
        if f_c < f_values[-1]:
            vertices[-1] = h_c
            f_values[-1] = f_c
        else:
            # Shrink
            for i in range(1, len(vertices)):
                vertices[i] = vertices[0] + rho * (vertices[i] - vertices[0])
                vertices[i] = project(vertices[i], len(vertices[i]))
                f_values[i] = evaluate(vertices[i], 2.0 / len(vertices[i]))
                
    return vertices, f_values
