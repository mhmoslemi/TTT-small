import numpy as np
from scipy import optimize
import time

def evaluate_c5(h_vals):
    # Compute C5 using numpy correlation
    n = len(h_vals)
    dx = 2.0 / n
    corr = np.correlate(h_vals, 1 - h_vals, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, dx):
    # Ensure h is in [0,1] and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    scale = (0.5 / np.sum(h)) * dx
    h *= scale
    return h

        c5_plus = evaluate_c5(h_plus)
        
        h_minus = h_copy.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, dx)
        c5_minus = evaluate_c5(h_minus)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
    return grad
