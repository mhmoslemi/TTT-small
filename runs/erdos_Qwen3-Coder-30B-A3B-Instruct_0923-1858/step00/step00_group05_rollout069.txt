import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Find maximum value scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def generate_initial_h(n_points):
    """Generate initial step function h."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Uniform distribution scaled to meet constraint
        h = np.full(n_points, 0.5)
        # Scale to satisfy sum(h) = n_points/2
        h *= (n_points / 2) / np.sum(h)
    return h

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def fitness(h, dx):
    """Fitness function for optimization: negative of C5 (minimize)."""
    return -evaluate_overlap(h, dx)

def local_optimization(h0, dx, maxiter=1000):
    """Refine the current solution using local optimization."""
    def obj(h):
        return -evaluate_overlap(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - (len(h) / 2)
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(len(h0))]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': maxiter, 'ftol': 1e-8})
        if res.success:
            h = res.x
        else:
            h = h0
    except:
        h = h0
    return project_to_feasible(h, len(h0))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Use genetic algorithm inspired approach locally
    def ga_step():
        nonlocal best_h, best_c5
        # Initialize population
        population_size = 100
        population = []
        for _ in range(population_size):
            h = generate_initial_h(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        
        # Evolution loop
        for gen in range(50):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            fitness_values = [evaluate_overlap(h, dx) for h in population]
            # Sort by fitness (lower is better)
            sorted_indices = np.argsort(fitness_values)
            
            # Track best solution
            current_best_idx = sorted_indices[0]
            current_best_h = population[current_best_idx]
            current_best_c5 = fitness_values[current_best_idx]
            
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = current_best_h.copy()
            
            # Select top 20% for reproduction
            top_indices = sorted_indices[:population_size // 5]
            selected = [population[i] for i in top_indices]
            
            # Generate offspring via crossover and mutation
            new_population = []
            new_population.extend(selected)
            
            # Tournament selection + crossover + mutation
            for _ in range(population_size - len(selected)):
                if np.random.rand() < 0.8: # Crossover
                    parent1, parent2 = np.random.choice(selected, 2, replace=False)
                    cross_point = np.random.randint(1, n_points-1)
                    child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
                else: # Mutation
                    parent = np.random.choice(selected)
                    child = parent.copy()
                    mutate_idx = np.random.randint(n_points)
                    child[mutate_idx] = np.clip(child[mutate_idx] + np.random.normal(0, 0.1), 0, 1)
                # Ensure sum constraint holds after mutation
                child = project_to_feasible(child, n_points)
                new_population.append(child)
            
            population = new_population
        
        return best_h, best_c5
    
    # Run GA
    final_h, final_c5 = ga_step()
    
    # Refine the best solution found with local optimization
    refined_h = local_optimization(final_h, dx)
    refined_c5 = evaluate_overlap(refined_h, dx)
    
    if refined_c5 < final_c5:
        final_h = refined_h
        final_c5 = refined_c5
    
    # Final check
    final_h = project_to_feasible(final_h, n_points)
    return (final_h.tolist(), final_c5, n_points)
