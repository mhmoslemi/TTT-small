from scipy.optimize import minimize
import time

        h0[2*n_points//3:] = 0.0
        # Normalize to sum to 1 with dx=2/n_points
        h0 *= n_points / (2 * np.sum(h0))  # Adjust to make sum = n_points/2
    
    dx = 2.0 / n_points
    
    def compute_C5(h):
        # Correlation using numpy's correlate
        h_padded = np.concatenate([h, np.zeros(n_points-1)])
        h_rev = h[::-1]
        conv = np.correlate(h_padded, 1 - h_rev, mode='full')
        # The overlap integral values
        overlaps = conv[:n_points] * dx
        return np.max(overlaps)
    
    def obj(h):
        return compute_C5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Use optimization with bounds
    # Start with a slightly modified version of h0
    h0 = h0.copy()
    
    # Run optimization with BFGS
    result = minimize(
        obj,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 5000}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_C5(best_h)

    elapsed = time.time() - start_time
