import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a starting point
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 1000
    tol = 1e-6
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient step parameters
    alpha = 0.01
    decay_factor = 0.1
    decay_interval = 100
    
    # State for optimization
    iter_count = 0
    last_improvement = 0
    
    def compute_c5(h_vals):
        # Compute cross correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_gradient(h_vals):
        # Compute the gradient of max correlation with respect to h
        # For numerical stability, use a small epsilon
        h_copy = h_vals.copy()
        corr_full = np.correlate(h_copy, 1 - h_copy, mode="full")
        max_k = np.argmax(corr_full)
        
        # Approximate gradient by finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (corr_plus[max_k] - corr_full[max_k]) / eps
        return grad
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current state
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        try:
            grad = compute_gradient(h)
        except Exception:
            # If gradient computation fails, fallback to random perturbation
            h = np.clip(h + np.random.normal(0, 0.001, len(h)), 0, 1)
            h *= target_sum / np.sum(h)
            iter_count += 1
            continue
        
        # Perform gradient step
        h_new = h - alpha * grad
        
        # Clip to allowable range
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale
        h_new *= target_sum / np.sum(h_new)
        
        h = h_new
        
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count % decay_interval == 0:
            alpha *= decay_factor
        
        # Early stopping
        if iter_count > 200 and abs(current_c5 - best_c5) < tol:
            last_improvement += 1
            if last_improvement > 5:
                break
        else:
            last_improvement = 0
    
    # Final validation
    if compute_c5(best_h) < best_c5:
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, len(best_h))
