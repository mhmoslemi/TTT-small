import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[-1] = 0.0
        h0 = h0 / np.sum(h0) * 2.0  # adjust so integral = 1

    dx = 2.0 / n_points
    if abs(np.sum(h0) * dx - 1.0) > 1e-6:
        scale = 1.0 / (np.sum(h0) * dx)
        h0 = h0 * scale
        
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    
    # Binary search on the overlap integral bound C5
    c_min = 0.0
    c_max = 0.5
    tolerance = 1e-4
    max_iterations = 20
    iters = 0
    
    while iters < max_iterations and (c_max - c_min) > tolerance:
        if time.time() - start_time > budget_s * 0.95:
            break
        c_mid = (c_min + c_max) / 2.0
        h_candidate = optimize_step(h0, c_mid, dx, n_points)
        c5_candidate = compute_c5(h_candidate, dx)
        
        if c5_candidate <= c_mid + 1e-6:
            c_max = c_mid
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
            h0 = h_candidate
        else:
            c_min = c_mid
            
        iters += 1
            
    # Return the best constructed solution
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    """Compute the maximum overlap integral for the step function h"""
    n = len(h)
    # Compute overlap integral for all shifts
    h_folded = np.concatenate([h, np.zeros(n)])
    h1_minus_h = np.concatenate([1 - h, np.zeros(n)])
    
    # Correlation with 1-h
    overlap = np.correlate(h_folded, h1_minus_h, mode='full')
    overlap = overlap[n-1:2*n-1]  # Get the valid overlaps
    return np.max(overlap) * dx

def optimize_step(h0, c_target, dx, n_points):
    """Optimize a step function subject to overlap constraint"""
    # Use a simple descent approach with projections
    h = h0.copy()
    max_iter = 50
    lr = 0.01
    
    for _ in range(max_iter):
        if time.time() - time.time() > 580:  # Prevent timeout
            break
        c5 = compute_c5(h, dx)
        if c5 <= c_target + 1e-6:
            break
        # Gradient-like update with projections
        grad = gradient_step(h, dx)
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) / dx * 2.0  # fix integral constraint
    return h

def gradient_step(h, dx):
    """Approximate gradient for overlap minimization"""
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n)])
    h1_minus_h = np.concatenate([1 - h, np.zeros(n)])
    
    # Compute correlation of h with 1-h
    overlap = np.correlate(h_extended, h1_minus_h, mode='full')
    overlap = overlap[n-1:2*n-1]
    max_shift = np.argmax(overlap)
    
    # Construct gradient direction using the difference operator
    # for current max overlap shift
    grad = np.zeros(n)
    for i in range(n):
        if i + max_shift < n:
            grad[i] = 1.0 - h[i + max_shift]
        else:
            grad[i] = 0.0
    
    # The gradient direction aims to reduce overlap
    # This is a simplified estimate, not exact
    return grad
