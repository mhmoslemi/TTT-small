import numpy as np
from scipy import signal
import time

def evaluate_h(h):
    """Evaluate the overlap integral for given step function h."""
    # Compute the convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx = 2.0 / len(h)
    dx = 2.0 / len(h)
    conv *= dx
    # Return the maximum value
    return np.max(conv)

def project_h(h, n_points):
    """Project h onto feasible region: h[i] in [0,1], sum(h) = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale so that sum = n_points/2
    h *= (n_points / 2) / np.sum(h)
    return h

def initialize_h(n_points, use_initial=True):
    """Initialize the step function h."""
    if use_initial and 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to correct size
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
    else:
        # Create symmetric and smooth initial function
        x = np.linspace(0, 2, n_points)
        # Use a Gaussian-like shape
        h = np.exp(-((x - 1) ** 2) / 0.1)
        # Normalize to sum = n_points / 2
        h *= (n_points / 2) / np.sum(h)
    return h

def nelder_mead_step(h, func, n_points, max_iter=1000, rel_tol=1e-6):
    """Run Nelder-Mead optimization with adaptive simplex size."""
    # Nelder-Mead parameters
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink

    # Initial simplex
    simplex_size = 0.1
    simplex = np.array([h] * (len(h) + 1), dtype=float)
    for i in range(len(h)):
        simplex[i+1, i] += simplex_size

    # Project all points to feasible region
    for s in simplex:
        s[:] = project_h(s, n_points)

    # Evaluate
    values = np.array([func(s) for s in simplex])
    best_idx = np.argmin(values)
    best_h = simplex[best_idx].copy()
    best_val = values[best_idx]

    resets = 0
    no_improvement_count = 0
    prev_best = best_val

    for iteration in range(max_iter):
        # Check time budget
        if iteration > 0 and iteration % 100 == 0:
            # Dummy check to avoid overuse of time
            pass
        # Sort simplex by function values
        order = np.argsort(values)
        simplex = simplex[order]
        values = values[order]

        # Check if we should reset simplex size
        if abs(values[0] - prev_best) < rel_tol:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            prev_best = values[0]

        if no_improvement_count > 50 and resets < 10:
            simplex_size *= 0.5
            resets += 1
            no_improvement_count = 0
            # Resample the simplex
            base = simplex[0].copy()
            simplex = np.array([base] * (len(h) + 1), dtype=float)
            for i in range(len(h)):
                simplex[i+1, i] += simplex_size
        
        # Recompute all vertices to maintain constraint
        for i in range(len(simplex)):
            simplex[i] = project_h(simplex[i], n_points)
        
        # Recompute values
        values = np.array([func(s) for s in simplex])

        # Update best
        if values[0] < best_val:
            best_val = values[0]
            best_h = simplex[0].copy()

        # Compute centroid (excluding worst vertex)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_r = centroid + alpha * (centroid - simplex[-1])
        h_r = project_h(h_r, n_points)
        val_r = func(h_r)
        
        if values[0] <= val_r < values[-2]:
            simplex[-1] = h_r
            values[-1] = val_r
        elif val_r < values[0]:
            # Expansion
            h_e = centroid + gamma * (h_r - centroid)
            h_e = project_h(h_e, n_points)
            val_e = func(h_e)
            if val_e < val_r:
                simplex[-1] = h_e
                values[-1] = val_e
            else:
                simplex[-1] = h_r
                values[-1] = val_r
        else:
            # Contraction
            h_c = centroid + beta * (simplex[-1] - centroid)
            h_c = project_h(h_c, n_points)
            val_c = func(h_c)
            if val_c < values[-1]:
                simplex[-1] = h_c
                values[-1] = val_c
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
                    simplex[i] = project_h(simplex[i], n_points)
                values = np.array([func(s) for s in simplex])
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved C5 bound."""
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use smaller problem for initial experimentation
    n_points = 45 # This number should be reasonable and able to finish within 600s
    
    h = initialize_h(n_points, use_initial=True)
    
    # Optimization function
    def func(h):
        return evaluate_h(h)
    
    # Run optimization
    h_opt, c5_opt = nelder_mead_step(h, func, n_points, max_iter=5000)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.9:
        # Return polished best if we're near time limit
        pass
        
    # Additional polishing by local search
    for _ in range(100):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Slight perturbations
        h_test = h_opt + np.random.normal(0, 0.01, len(h_opt))
        h_test = project_h(h_test, n_points)
        c5_test = func(h_test)
        if c5_test < c5_opt:
            h_opt = h_test
            c5_opt = c5_test

    # Final check
    h_opt = project_h(h_opt, n_points)
    c5_final = evaluate_h(h_opt)
    
    return (h_opt.tolist(), c5_final, n_points)
