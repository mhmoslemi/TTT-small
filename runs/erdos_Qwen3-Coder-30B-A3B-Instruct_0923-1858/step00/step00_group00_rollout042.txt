import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # Start with flat distribution
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = max(np.correlate(h, 1-h, mode="full")) * dx
    
    # Annealing parameters
    T = 1.0
    alpha = 0.995
    perturb_decay = 0.99
    min_temp = 1e-6
    max_iter = 10000
    no_improve_limit = 1000
    no_improve_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb h
        perturbation = np.random.normal(0, T, n_points)
        h_new = h + perturbation
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
            
        # Compute new C5
        c5_new = max(np.correlate(h_new, 1-h_new, mode="full")) * dx
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve_count = 0
            else:
                no_improve_count += 1
                
        # Cooling and perturbation adjustment
        if i % 100 == 0:
            T *= alpha
            T = max(T, min_temp)
            
        if no_improve_count >= no_improve_limit:
            T *= perturb_decay
            no_improve_count = 0
            
    return (best_h.tolist(), best_c5, len(best_h))
