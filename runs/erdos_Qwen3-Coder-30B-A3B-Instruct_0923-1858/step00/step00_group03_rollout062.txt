import numpy as np
import time
from scipy import signal

def binary_convert(h, n_points):
    """Convert h to binary and ensure sum equals n_points/2"""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        indices = np.where(h == 1)[0]
        flip_count = current_sum - target_sum
        h[indices[:flip_count]] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        indices = np.where(h == 0)[0]
        flip_count = target_sum - current_sum
        h[indices[:flip_count]] = 1
    
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    # Compute the full cross-correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap occurs in the middle of the correlation array
    max_overlap = np.max(corr)
    return max_overlap * dx

def local_search(h, n_points, dx, time_limit):
    """Run local search with simulated annealing"""
    start_time = time.time()
    
    # Convert to binary and make sure sum is correct
    h = binary_convert(h, n_points)
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    
    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Select two indices to flip: one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i_idx = np.random.choice(ones)
        j_idx = np.random.choice(zeros)
        
        # Compute change in C5 if we flip these
        old_c5 = current_c5
        
        # Temporary copy
        new_h = h.copy()
        new_h[i_idx] = 0
        new_h[j_idx] = 1
        
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept if better or by probability
        delta = new_c5 - old_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = new_h
            current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        T *= alpha
    
    return best_h, best_c5
