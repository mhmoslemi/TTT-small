def evaluate_erdos_solution(h, dx):
    """Evaluate C5 = max_k integral h(x)(1-h(x+k) dx"""
    h = np.array(h)
    # Compute the auto-correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode='full')
    # Take the maximum (excluding the trivial 0 shift)
    max_corr = np.max(corr[-len(h)+1:len(h)-1])
    return max_corr * dx
