import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_h(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using numpy's correlate with mode="full"
    # This computes all possible overlaps
    h_rev = h[::-1]
    correlation = np.correlate(h, h_rev, mode="full")
    # Only consider the relevant part (overlaps)
    # The full correlation has len(h)*2 - 1 elements
    # The valid shifts are from -len(h)+1 to len(h)-1
    # Take the middle part, with the maximum value
    mid = len(correlation) // 2
    max_corr = np.max(correlation[mid - len(h) + 1 : mid + len(h)])
    # But we want h(x)(1-h(x+k)), NOT h(x)h(x+k)
    # So we need to compute: int h(x)(1-h(x+k))dx = int h(x)dx - int h(x)h(x+k)dx
    # For the given constraints that sum(h) = n/2, int h(x)dx = 1
    # So we want to maximize int h(x)h(x+k)dx among all shifts k
    # And minimize the maximum of this over k. Thus C5 = 1 - min(int h(x)h(x+k)dx)
    # This is wrong approach to update.
    # Let's compute it more directly:
    # h * (1 - h.shifted) = h - h * h.shifted
    # So int h * (1 - h.shifted) should be int h - int h * h.shifted
    # where int h = 1 (constraint) and the integral int h * h.shifted 
    # should be minimized, so that int h * (1-h.shifted) = 1 - (int h * h.shifted) is maximized.
    # But actually we want to minimize the maximum of int h(x) * (1 - h(x+k) dx):
    # unsimplified = int( h(x) * (1 - h(x+k)) dx )
    # -> int h(x) dx - int h(x) * h(x + k) dx
    # -> 1 - int h(x) * h(x + k) dx
    # So we want to minimize the max of (1 - int h(x) * h(x + k) dx)
    # Which is same as maximizing the min of (int h(x) * h(x + k) dx)
    # So we find的最大 h(x)h(x+k)积分 over all k, then subtract this from 1.
    # When k = 0: h(x)h(x) = h(x)^2
    # When k>0: h(x)h(x+k) 
    # Full correlation of h with h_rev is sum over all k:
    # sum_i h[i] * h[i+k] for k = -n+1 ... n-1
    # This is correlation of h with h reversed to compute h(i) * h(i+k)
    # But since we actually want sum of h[i]*h[i+k] for k=0,1,...n-1 only
    # We should align and compute correlation for all valid k.
    # Let's be more precise.
    # So we compute for each k = 0,1,...,n-1
    # int h(x)*h(x+k) dx
    # Actually we are doing discrete computation:
    # sum_{i=0}^{n-1} h[i]*h[i+k] for each k
    # This includes:
    # k=0: sum h[i]^2
    # k=1: sum h[i]*h[i+1]
    # ...
    # k=n-1: sum h[i]*h[i+n-1] = 0 (since h[i+n-1] = h[i+n-1] = undefined)
    # We want sum of h[i]*h[i+k] for valid k = 0,...,n-1
    # But we might need k=0 to k=n-1, so the correlation in python gives:
    # correlations for shifted k = -n+1, ... n-1
    # for k = 0 it's like sum h[i]*h[i]
    # For k=1, it's h[0]*h[1] + h[1]*h[2] + ... 
    # We only care for k = 0 to n-1
    # So correlation from mode='full' is 2*n-1 long. 
    # Take middle: 
    # index for k = -n+1 is 0
    # for k = 0 is n-1
    # for k = n-1 is 2*n-2
    # We want sum from k=0 to n-1 = n indices, starting from index of k=0 which is n-1.
    # The indices for k=0,1,...,n-1 are n-1 to 2*n-2
    # We have to add dx back
    valid_correlations = correlation[n-1 : 2*n-1]
    # But this is the sum of h[i]*h[i+k] in the discrete case
    # That is what we want
    # So, our goal is minimize max_k [ sum h[i]*h[i+k] ] 
    # The overlap integral = 1 - sum_h[i]*h[i+k]
    # So minimize max_k [ 1 - sum h[i]*h[i+k] ] = max_k [ sum h[i]*h[i+k] ] 
    # So want to *maximize* the min of this.
    # But weighing with dx after
    integral_values = valid_correlations * dx
    # Overlap Integral = 1 - integral values, but the goal is to minimize max of this:
    # goal = minimize max_k [ 1 - sum h[i]*h[i+k] * dx ] = minimize max_k [ 1 - integral_values ]
    # which = 1 - min integral_values
    # So goal = minimize (1 - min integral_values) 
    # That is minimize 1 - min correlation -> maximize min correlation.
    # The value to return is the max of the overlaps, which is 1 - min(integral_values)
    return 1.0 - min(integral_values)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 45
    dx = 2.0 / n
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Default symmetric construction
        h = np.ones(n) * 0.5

    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters for GA
    P = 50  # Population
    generations = 200
    tournament_size = 5
    mutation_std = 0.05
    mutation_decay = 0.95
    
    # Initialize population
    population = [h.copy()]
    for _ in range(P-1):
        new_h = h + np.random.normal(0, 0.05, n)
        new_h = np.clip(new_h, 0, 1)
        # Ensure sum(h) = n/2 (i.e., integral = 1)
        total = np.sum(new_h)
        new_h = new_h * (n/2) / total
        population.append(new_h)
    
    def tournament_select(pool, k):
        selected = np.random.choice(pool, k, replace=False)
        scores = [evaluate_h(h, dx) for h in selected]
        return selected[np.argmin(scores)]
        
    def crossover(p1, p2, alpha=0.5):
        child = alpha * p1 + (1 - alpha) * p2
        # Keep in [0, 1]
        child = np.clip(child, 0, 1)
        # Correct integral 
        total = np.sum(child)
        child = child * (n/2) / total
        return child

    loop_start = time.time()
    for gen in range(generations):
        if time.time() - loop_start > budget_s:
            break
            
        # Evaluate population
        scores = [evaluate_h(h, dx) for h in population]
        
        # Sort and get elites
        elite_indices = np.argsort(scores)[:P//5]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate rest of the population
        while len(new_population) < P:
            parent1 = tournament_select(population, tournament_size)
            parent2 = tournament_select(population, tournament_size)
            child = crossover(parent1, parent2, 0.5)
            # Mutation
            if np.random.random() < 0.8:
                child += np.random.normal(0, mutation_std, n)
                child = np.clip(child, 0, 1)
                total = np.sum(child)
                child = child * (n/2) / total
            new_population.append(child)
        
        population = new_population
        # Update mutation std
        mutation_std *= mutation_decay
        
        # Keep best
        current_scores = [evaluate_h(h, dx) for h in population]
        min_idx = np.argmin(current_scores)
        if current_scores[min_idx] < best_c5:
            best_c5 = current_scores[min_idx]
            best_h = population[min_idx].copy()
    
    # Gradient descent refinement of top 3 individuals
    top_h = []
    scores = [evaluate_h(h, dx) for h in population]
    top_indices = np.argsort(scores)[:3]
    for i in top_indices:
        top_h.append(population[i].copy())

    def finite_diff_gradient(h, dx):
        epsilon = 1e-6
        grad = np.zeros_like(h)
        base_c = evaluate_h(h, dx)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            # Re-normalize to keep integral = 1
            total = np.sum(h_plus)
            h_plus = h_plus * (n/2) / total
            c_plus = evaluate_h(h_plus, dx)
            grad[i] = (c_plus - base_c) / epsilon
        return grad

    for h_refine in top_h:
        if time.time() - loop_start > budget_s:
            break
    
        h_current = h_refine.copy()
        lr = 0.01
        for step in range(150):
            if time.time() - loop_start > budget_s:
                break
            grad = finite_diff_gradient(h_current, dx)
            h_current -= lr * grad
            # Project back to feasible region
            h_current = np.clip(h_current, 0, 1)
            total = np.sum(h_current)
            h_current = h_current * (n/2) / total
            c_curr = evaluate_h(h_current, dx)
            if c_curr < best_c5:
                best_c5 = c_curr
                best_h = h_current.copy()

    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n)
