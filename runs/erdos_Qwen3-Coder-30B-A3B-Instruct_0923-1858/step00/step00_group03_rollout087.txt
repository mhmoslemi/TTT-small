import numpy as np
import time
from scipy.signal import correlate

def binary_convert(h, n_points):
    """Convert h to binary array with exactly n_points/2 ones."""
    h_bin = np.round(h).astype(int)
    ones_count = np.sum(h_bin)
    target = n_points // 2
    if ones_count > target:
        # Need to flip some 1s to 0s
        flip_indices = np.argsort(h_bin)[::-1][:ones_count - target]
        h_bin[flip_indices] = 0
    elif ones_count < target:
        # Need to flip some 0s to 1s
        flip_indices = np.argsort(h_bin)[:target - ones_count]
        h_bin[flip_indices] = 1
    return h_bin

def compute_c5(h, dx):
    """Compute C5 = max_k integral of h(x)(1 - h(x+k)) dx."""
    # We compute the cross-correlation of h with (1-h)
    h1 = 1 - h
    corr = correlate(h, h1, mode="full")
    # The maximum is taken over the valid lags (with overlap)
    # Due to symmetry and how correlate works, the valid range is:
    # for lag k in [-n+1, n-1] where n is length of h
    # For our shift, indices are taken care of by correlate, so we use 
    # full correlation from scipy and extract valid range properly
    # correlation at lag k corresponds to id = n-1+k for k in [-n+1, n-1]
    # max is over all actual valid lags: we just need maximum from full convolution
    return np.max(corr) * dx

def simulated_annealing_binary(initial_h, n_points, dx, budget_s):
    """Optimize binary h using simulated annealing."""
    best_h = binary_convert(initial_h.copy(), n_points)
    best_c5 = compute_c5(best_h, dx)
    current_h = best_h.copy()
    current_c5 = best_c5
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    time_start = time.time()
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s - 1:  # leave some margin
            break
        # Pick two indices to flip
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Create new configuration
        new_h = current_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        
        # Check if it's still valid (exactly half ones)
        ones_count = np.sum(new_h)
        if ones_count != n_points // 2:
            continue
            
        new_c5 = compute_c5(new_h, dx)
        
        # Accept if better or with probability
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        T *= alpha
    
    return best_h, best_c5
