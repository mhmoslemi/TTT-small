        h = cp.Variable(n_points)

        constraints = [
            h >= 0,
            h <= 1,

                constraints.append(overlap_expr <= C * n_points)
        
        # Construct and solve LP
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
