import time

    no_improvement_count = 0

        iterations += 1
        
        # Evaluate current C5
        conv = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(conv) * dx
        
        # Find the shift k where max occurs
        max_idx = np.argmax(conv)
        k = max_idx - (len(h) - 1)
        
        # Determine overlapping regions
        overlap_range = range(max(0, -k), min(len(h), len(h) - k))
        
        # Generate candidates by redistributing
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        num_attempts = 10
        for _ in range(num_attempts):
            candidate = h.copy()
            
            # Select a pair of indices to trade between
            i = np.random.choice(overlap_range)
            j = np.random.choice(range(len(h)))
            
            # Compute max transfer
            max_transfer = min(candidate[i], 1 - candidate[j])
            if max_transfer > 0:
                transfer_amount = np.random.uniform(0, max_transfer)
                candidate[i] -= transfer_amount
                candidate[j] += transfer_amount
                
                # Maintain bounds
                candidate = np.clip(candidate, 0, 1)
                
                # Ensure constraint sum(h) * dx = 1
                current_sum = np.sum(candidate)
                target_sum = n_points / 2
                diff = target_sum - current_sum
                if abs(diff) > 1e-10:
                    # Adjust last element
                    candidate[-1] += diff
                
                # Evaluate
                conv_cand = correlate(candidate, 1 - candidate, mode="full")
                candidate_c5 = np.max(conv_cand) * dx
                
                if candidate_c5 < best_candidate_c5:
                    best_candidate = candidate
                    best_candidate_c5 = candidate_c5
                    
        # Accept if better
        if best_candidate_c5 < current_c5:
            h = best_candidate
            no_improvement_count = 0
            if best_candidate_c5 < best_c5:
                best_h = h.copy()
                best_c5 = best_candidate_c5
        else:
            no_improvement_count += 1

    if iterations > 100:

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5
