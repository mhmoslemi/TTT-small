import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h):
    """Evaluate the overlap integral C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h) using full mode
    conv = correlate(h, 1 - h, mode="full")
    # The overlap integral C5 is the maximum of these convolutions, scaled by dx
    C5 = np.max(conv) * dx
    return C5

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum=h/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h *= (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial h if available

        n_points = len(h0)
    else:
        n_points = 45
        # Construct initial symmetric h
        h0 = np.ones(n_points) * 0.5
        # Make it symmetric with a triangular shape (more likely to be good)
        half = n_points // 2
        for i in range(half):
            h0[i] = h0[n_points-1-i] = (i + 1) / (half + 1) * 0.5
        h0 = project_h(h0, n_points)
    
    # Nelder-Mead with adaptive simplex
    max_iter = 5000
    dim = n_points
    simplex_size = 0.1
    max_reset_attempts = 10
    stagnation_limit = 50
    
    # Initialize simplex: one vertex is h0, others are perturbed
    simplex = [h0.copy()]
    for i in range(dim):
        h = h0.copy()
        h[i] += np.random.uniform(-simplex_size, simplex_size)
        h = project_h(h, n_points)
        simplex.append(h)
    
    # Evaluate simplex
    values = [evaluate_h(v) for v in simplex]
    best_idx = np.argmin(values)
    best_h = simplex[best_idx].copy()
    best_C5 = values[best_idx]
    
    # Iteration counter and stagnation tracking
    iterations = 0
    reset_attempts = 0
    best_improvement = 0
    last_improvement = 0
    
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink
    
    # Main loop
    while iterations < max_iter and time.time() - start_time < budget_s:
        # Order vertices
        idxs = np.argsort(values)
        simplex = [simplex[i] for i in idxs]
        values = [values[i] for i in idxs]
        
        best_idx = 0
        best_h = simplex[0].copy()
        best_C5 = values[0]
        
        # Check stagnation
        if values[0] < best_improvement - 1e-8:
            best_improvement = values[0]
            last_improvement = iterations
        elif iterations - last_improvement > stagnation_limit:
            if reset_attempts < max_reset_attempts:
                simplex_size *= 0.5
                reset_attempts += 1
                # Redo simplex initialization
                simplex = [best_h.copy()]
                for i in range(dim):
                    h = best_h.copy()
                    h[i] += np.random.uniform(-simplex_size, simplex_size)
                    h = project_h(h, n_points)
                    simplex.append(h)
                values = [evaluate_h(v) for v in simplex]
                continue
            else:
                break
        
        # Compute centroid (excluding worst vertex)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_r = centroid + alpha * (centroid - simplex[-1])
        h_r = project_h(h_r, n_points)
        C_r = evaluate_h(h_r)
        
        # Accept reflection if it's better than second worst
        if values[0] <= C_r < values[-2]:
            simplex[-1] = h_r
            values[-1] = C_r
        else:
            # Otherwise try expansion or contraction
            if C_r < values[0]:
                # Expansion
                h_e = centroid + gamma * (h_r - centroid)
                h_e = project_h(h_e, n_points)
                C_e = evaluate_h(h_e)
                if C_e < C_r:
                    simplex[-1] = h_e
                    values[-1] = C_e
                else:
                    simplex[-1] = h_r
                    values[-1] = C_r
            else:
                # Contraction
                h_c = centroid + beta * (simplex[-1] - centroid)
                h_c = project_h(h_c, n_points)
                C_c = evaluate_h(h_c)
                if C_c < values[-1]:
                    simplex[-1] = h_c
                    values[-1] = C_c
                else:
                    # Shrink
                    for i in range(1, dim+1):
                        simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
                        simplex[i] = project_h(simplex[i], n_points)
                    values = [evaluate_h(v) for v in simplex]
        
        iterations += 1
    
    # Ensure final answer is correct
    final_h = project_h(best_h, n_points)
    final_C5 = evaluate_h(final_h)
    
    # Return result as requested
    return (final_h.tolist(), final_C5, n_points)
