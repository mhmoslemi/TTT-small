import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.01
    perturbation_rate = 0.1
    perturbation_frequency = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to 1
    h = h * (0.5 / np.mean(h))
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (0.5 / np.mean(h))
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        correlations = np.correlate(h, 1 - h, mode='full')
        C5_candidate = np.max(np.abs(correlations)) * dx
        
        if C5_candidate < best_C5:
            best_h = h.copy()
            best_C5 = C5_candidate
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                grad_slice = 1 - h[k:]
                grad_indices = np.arange(len(grad_slice))
            else:
                grad_slice = 1 - h[:k]
                grad_indices = np.arange(-k, n_points)
            
            if len(grad_slice) > 0:
                gradient[grad_indices] += grad_slice
                
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Gradient step
        h = h - learning_rate * gradient
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.mean(h))
        
        # Perturbation
        if iteration % perturbation_frequency == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.mean(h))
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, n_points)
