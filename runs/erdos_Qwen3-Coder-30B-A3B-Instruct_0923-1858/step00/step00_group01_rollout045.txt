    def compute_C5(h):
        # Compute the correlation sum using full convolution
        h_reflect = np.flip(1 - h)
        correlation = signal.correlate(h, h_reflect, mode="full")
        # The correlation result has length 2*n_points -1, but the max is:
        # indices k from 0 to 2*n_points - 2
        # For k=0 to n_points-1, correlation has valid values
        # For our problem, we want max over all shifts k
        return np.max(correlation) * dx

    def objective(h):
        return compute_C5(h)

    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
        
    def project_h(h):
        # Project onto constraint and bounds
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h) * dx
        if abs(current_sum) > 1e-10:
            h = h * (1.0 / current_sum)
        return h

    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    best_h = current_h.copy()
    best_C5 = current_C5

    # Simple stochastic hill climbing + line search + projection
    max_iter = 10000
    lr = 0.01
    decay_factor = 0.9999
    iter_since_improve = 0
