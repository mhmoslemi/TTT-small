import time
from scipy import optimize

    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.01
    perturbation_rate = 0.1
    perturbation_freq = 100
    best_C5 = float('inf')
    best_h = None

    else:

    h = h / np.mean(h) * 0.5  # Enforce sum(h) = n_points / 2
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlations = np.correlate(h, 1 - h, mode="full")
        current_C5 = np.max(np.abs(correlations)) * dx
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (0.5 / np.mean(h_plus))
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (0.5 / np.mean(h_minus))
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            cor_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (np.max(np.abs(cor_plus)) - np.max(np.abs(cor_minus))) / (2 * eps)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Take gradient step
        h_new = h - learning_rate * grad
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (0.5 / np.mean(h_new))
        
        h = h_new
        
        # Perturb occasionally to escape local optima
        if iteration % perturbation_freq == 0:
            perturb = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.mean(h))
    
    # Ensure final solution is valid
    if best_h is None:
        best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (0.5 / np.mean(best_h))
    
    # Final evaluation
    correlations = np.correlate(best_h, 1 - best_h, mode="full")
    final_C5 = np.max(np.abs(correlations)) * dx
    
    return (best_h.tolist(), final_C5, n_points)
