import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        # Cut to valid shift range for max
        shifts = np.arange(-n_points + 1, n_points)
        valid_corr = corr[n_points - 1:]  # Just the positive shifts
        k_star = shifts[np.argmax(valid_corr)]
        max_overlap = np.max(valid_corr)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Adjust the pair that contributes most to max_overlap
        # Determine where this maximum is achieved
        max_pos = np.argmax(valid_corr)
        
        # We need to determine the x_0 position that contributed to this
        # This is where we look at correlation between h[i] and 1-h[i+k]
        i_pos = max_pos
        
        # We want to decrease h[i] and h[i+k] to reduce the overlap
        # However, for stability, we don't change the absolute values too much
        step_size = min(0.05, 0.5 * h[i_pos], 0.5 * h[i_pos + k_star])
        if step_size <= 0:
            continue
            
        # Decrease both h[i] and h[i+k] by step_size
        h[i_pos] -= step_size
        h[i_pos + k_star] -= step_size
        
        # Adjust to keep in [0,1]
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint is maintained
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-8:
            diff = (n_points / 2 - current_sum)
            # Distribute the difference evenly
            h += diff / n_points
        
        # Normalize to ensure sum target
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # h is a list of values
    h = np.array(h)
    n_points = len(h)
    corr = correlate(h, 1 - h, mode='full')
    shifts = np.arange(-n_points + 1, n_points)
    valid_corr = corr[n_points - 1:]  # Only positive shifts
    max_overlap = np.max(valid_corr)
    return max_overlap * dx
