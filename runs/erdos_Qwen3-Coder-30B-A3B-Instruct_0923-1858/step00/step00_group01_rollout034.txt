import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5  # Uniform distribution
    
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.01
    iter_count = 0
    last_improvement = 0
    
    # Optimizer state
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current c5 and gradient
        c5_val = evaluate_c5(h, dx)
        
        if iter_count > 0 and iter_count % 100 == 0:
            # Increase resolution
            h = interpolate_h(h, n_points * 2)
            n_points *= 2
            dx = 2.0 / n_points
        
        # Compute gradient approximation
        grad = compute_gradient(h, dx)
        
        # Line search
        step_size = alpha
        while True:
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)  # Rescale to meet constraint
            
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 <= c5_val or step_size < 1e-8:
                h = h_new
                break
            step_size *= 0.5
        
        # Update best solution
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
            last_improvement = iter_count
        
        iter_count += 1

    return best_h.tolist(), best_c5, n_points

def evaluate_c5(h, dx):
    """Calculate C5 value using the full cross-correlation."""
    # compute correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # maximum over all lags
    c5 = np.max(corr) * dx
    return c5

def compute_gradient(h, dx):
    """Approximate gradient w.r.t. h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    
    # For each point in h, compute finite difference on the c5
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = np.clip(h_plus, 0, 1)
        h_minus = np.clip(h_minus, 0, 1)
        # Rescale to maintain area constraint
        h_plus = h_plus * (n / 2) / np.sum(h_plus)
        h_minus = h_minus * (n / 2) / np.sum(h_minus)
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def interpolate_h(h, new_n):
    """Interpolate h to a new resolution."""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n)
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_old, h)
    # Preserve integral
    h_new = h_new * (new_n / 2) / np.sum(h_new)
    return h_new
