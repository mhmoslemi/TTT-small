import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum value is at the center (shift 0)
    max_corr = np.max(corr)
    return max_corr * dx

def clip_and_rescale(h, dx):
    """Ensure h is in [0,1] and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, dx):
    """Compute gradient of C5 w.r.t. h using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    # Evaluate at current point
    current_C5 = evaluate_erdos_solution(h)
    
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        C5_plus = evaluate_erdos_solution(h_plus)
        grad[i] = (C5_plus - current_C5) / eps
    return grad

def line_search(h, grad, dx, alpha_max=0.1, max_iter=10):
    """Simple backtracking line search"""
    alpha = alpha_max
    current_C5 = evaluate_erdos_solution(h)
    for _ in range(max_iter):
        h_new = clip_and_rescale(h - alpha * grad, dx)
        new_C5 = evaluate_erdos_solution(h_new)
        if new_C5 < current_C5:
            return alpha, h_new
        alpha *= 0.5
    return alpha, h_new
