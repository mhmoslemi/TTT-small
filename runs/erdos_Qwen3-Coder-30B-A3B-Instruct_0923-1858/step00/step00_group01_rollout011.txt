import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        # Compute current cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        # Find maximum correlation index
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(h) - 1)
        max_corr = corr[max_idx]
        
        # Get indices that contribute to max correlation
        # These are elements that align at shift k
        if max_k >= 0:
            # Correlation at k = max_k is h[i] * (1 - h[i + k])
            # Pairing indices i: 0 to len(h) - 1 - k, j: k to len(h) - 1
            i_indices = np.arange(len(h) - max_k)
            j_indices = np.arange(max_k, len(h))
        else:
            # k < 0, swap roles
            # Correlation at k = max_k is h[i] * (1 - h[i - k])
            i_indices = np.arange(-max_k, len(h))
            j_indices = np.arange(len(h) + max_k)
        
        # Select random regions for perturbation
        # We focus on the most critical indices
        i_idx = np.random.choice(i_indices)
        j_idx = np.random.choice(j_indices)
        
        # Determine amount to transfer
        max_transfer = min(h[i_idx], 1 - h[j_idx])
        
        if max_transfer > 1e-8:
            # Transfer amount
            transfer_amount = np.random.uniform(0, max_transfer)
            # Apply transfer
            h[i_idx] -= transfer_amount
            h[j_idx] += transfer_amount
            
            # Ensure feasibility
            h[i_idx] = np.clip(h[i_idx], 0, 1)
            h[j_idx] = np.clip(h[j_idx], 0, 1)
            
            # Normalize to sum to 1
            h_sum = np.sum(h)
            h *= (n_points / 2) / h_sum
            
            # Evaluate new C5
            new_c5 = evaluate_erdos_solution(h, dx)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                # Revert the change
                h[i_idx] += transfer_amount
                h[j_idx] -= transfer_amount
                no_improvement_count += 1
        else:
            no_improvement_count += 1

    # Interpolate if beneficial
    if iter_count > 100:
        higher_res_h = np.interp(np.linspace(0, 2, 2 * n_points), 
                                 np.linspace(0, 2, n_points), 
                                 best_h)
        dh = 2.0 / (2 * n_points)
        new_c5 = evaluate_erdos_solution(higher_res_h, dh)
        if new_c5 < best_c5:
            best_h = higher_res_h
            best_c5 = new_c5
            n_points = 2 * n_points
            dx = dh

    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
