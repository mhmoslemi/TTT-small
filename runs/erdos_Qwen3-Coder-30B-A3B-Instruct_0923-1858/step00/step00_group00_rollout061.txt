import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # symmetric construction
        h_init = np.ones(n_points) * 0.5
        h_init[0] = 0.0
        h_init[-1] = 0.0
        h_init[1] = 0.0
        h_init[-2] = 0.0
        
    # Normalize
    h_init = h_init * (n_points / 2) / np.sum(h_init)

    def compute_c5(h):
        h = np.array(h)
        conv = np.correlate(h, 1-h, mode="full")
        return np.max(conv) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.concatenate([h, 1-h])

    # Optimizer
    best_c5 = float('inf')
    best_h = None
    
    options = {'maxiter': 5000, 'disp': False}
    
    # Random restarts
    num_restarts = 10
    for i in range(num_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_start = h_init.copy()
        
        # Random perturbation
        noise = np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start + noise, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Scipy minimize
        try:
            res = minimize(objective, h_start, method='L-BFGS-B',
                           bounds=[(0, 1) for _ in range(n_points)],
                           constraints={'type': 'eq', 'fun': constraint_sum},
                           options=options)
            
            if res.success:
                c5_val = objective(res.x)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = res.x.copy()
                    
        except:
            pass

    # Final verification and refinement
    if best_h is None:
        best_h = h_init.copy()
        best_c5 = compute_c5(best_h)
        
    # Binary search to sharpen bound
    c_min, c_max = 0.0, best_c5
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        c_mid = (c_min + c_max) / 2
        
        # Simple feasibility check
        def check_feasible(h):
            return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h)*dx - 1) < 1e-6
        
        test_h = best_h.copy()
        if check_feasible(test_h):
            # Perform local refinement
            try:
                res = minimize(objective, test_h, method='L-BFGS-B',
                               bounds=[(0, 1) for _ in range(n_points)],
                               constraints={'type': 'eq', 'fun': constraint_sum},
                               options={'maxiter': 50})
                if res.success:
                    c5_val = objective(res.x)
                    if c5_val < c_mid:
                        c_max = c5_val
                        best_c5 = c5_val
                        best_h = res.x.copy()
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
