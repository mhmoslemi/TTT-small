import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h_values, n_points):
    """Evaluate the C5 constant for given h values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode='full')
    # The maximum overlap (scaled by dx)
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * 2.0  # normalize to sum to 2
    else:
        # Construct initial h as a simple piecewise constant function
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        # little relaxation to bias toward a good-looking start
        h0[0] = 0.6
        h0[1] = 0.4
        h0[-1] = 0.4
        h0[-2] = 0.6
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Start timing
    start_time = time.time()
    
    # Direct optimization attempt using CVXPY
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Support constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Sum constraint
    constraints.append(cp.sum(h) == 1.0/dx)
    
    # Initial set of overlap constraints (sample a few)
    overlaps = correlate(h0, 1 - h0, mode='full')
    max_k = len(overlaps) - 1
    # Sorted by descending overlap
    indices = np.argsort(overlaps)[::-1]
    # Use top 50-80% of constraints to make problem more manageable
    constraint_indices = indices[:max(10, len(indices)//2)]
    
    # Add initial constraints
    for k in constraint_indices:
        # Get proper index for the overlap array
        if k < len(overlaps):
            # Adjust overlap index for the actual U[0,2] based correlation
            # See how correlate handles overlaps on its own
            pass  # we will compute everything using a more direct approach
    
    # Refine work - We'll progressively tighten constraints
    C5_best = evaluate_c5(h0, n_points)
    h_best = h0.copy()
    
    # Cutting plane method
    iteration = 0
    max_iter = 100
    not_improving = 0
    min_improvement = 1e-5
    last_c5 = float('inf')
    loop_calls = 0
    
    # Collect constraints to build up system
    constraints_existing = []
    while time.time() - start_time < budget_s - 2 and iteration < max_iter:
        iteration += 1
        h_current = h_best.copy()
        dx = 2.0 / n_points
        
        # Redefine variable instances for new loop
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Rebuild constraints
        constraints = []
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == 1.0/dx)
        
        # Compute overlaps for checking constraints
        h_vals = np.array(h_current)
        max_k = 2 * n_points - 1
        # Doing the computation with numpy for validation
        h_full = np.concatenate([h_vals, np.zeros(n_points - 1)])
        ones_h = np.concatenate([1 - h_vals, np.zeros(n_points - 1)]) 
        overlaps_full = correlate(h_full, ones_h, mode='full') 
        overlaps_full = overlaps_full[:max_k] * dx
        
        # Create constraint set of top violations for this solution
        violations = np.arange(len(overlaps_full))
        violations = violations[np.argsort(overlaps_full)[::-1]][:max(10, n_points // 4)]
        violations = sorted(violations)
        for k in violations:
            if k < max_k:
                constraints.append(overlaps_full[k] <= C5)
        
        # Objective: minimize C5
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_val_new = h.value
                C5_val = C5.value
                if C5_val is not None and not np.isnan(C5_val):
                    h_best = h_val_new
                    C5_best = min(C5_best, C5_val)
                    if np.abs(last_c5 - C5_best) < min_improvement:
                        not_improving += 1
                    else:
                        not_improving = 0
                    last_c5 = C5_best
                else:
                    h_best = h_vals
                    C5_best = evaluate_c5(h_vals, n_points)
        except Exception as e:
            # Revert to previous
            pass
        
        # Double-check final evaluation
        C5_final = evaluate_c5(h_best, n_points)
        C5_best = min(C5_best, C5_final)
        
        if not_improving > 5:
            break
        
        loop_calls += 1
    
    # Final evaluation
    C5_final = evaluate_c5(h_best, n_points)
    C5_best = min(C5_best, C5_final)
    
    return (list(h_best), C5_best, n_points)
