        h = cp.Variable(n_points)

                shifted_indices = np.roll(np.arange(n_points), -k)
                # Create overlap constraint
                overlap = cp.sum(cp.multiply(h, 1 - h[shifted_indices]))
                # Add constraint
                constraints.append(overlap <= C_mid * n_points)
        
        # Minimize a dummy objective (feasibility check)
        objective = cp.Minimize(0)
