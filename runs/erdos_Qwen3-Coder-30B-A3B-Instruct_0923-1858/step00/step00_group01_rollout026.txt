import numpy as np
from scipy.signal import correlate
import time

def evaluate_C5(h, dx):
    # Compute cross-correlation of h and (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def clip_and_rescale(h, dx):
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to satisfy integral constraint
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0
    return h

def compute_gradient(h, dx):
    # This computes a finite difference approximation of the gradient
    # Multiple correlation terms contribute to gradient
    n = len(h)
    grad = np.zeros_like(h)
    h1 = 1 - h
    
    # For each position i, compute how much changing h[i] affects C5
    corr_h1 = correlate(h1, h, mode='full')
    corr_h = correlate(h, h1, mode='full')
    
    # Gradient approximation via finite difference
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        c_plus = evaluate_C5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = clip_and_rescale(h_minus, dx)
        c_minus = evaluate_C5(h_minus, dx)
        
        grad[i] = (c_plus - c_minus) / (2 * eps) 
    
    return grad

def line_search(h, grad, dx, alpha_max=1.0):
    # Try to find a good step size alpha
    alpha = alpha_max
    for _ in range(10):  # Max line search steps
        h_new = clip_and_rescale(h - alpha * grad, dx)
        c_new = evaluate_C5(h_new, dx)
        if c_new < evaluate_C5(h, dx):
            return alpha
        alpha *= 0.5
    return alpha
