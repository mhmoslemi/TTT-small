import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute all overlap integrals for different shifts
    # We need to compute: int h(x)(1 - h(x+k)) dx for all valid k
    # This is equivalent to correlate(h, 1-h, mode="full")
    h_rev = 1 - h
    corr = correlate(h, h_rev, mode="full")
    # The overlap integral for shift k is given by corr[n_points-1+k] * dx
    # Only consider shifts from -n_points+1 to n_points-1 
    # But just take the maximum of all shifts from 0 to 2*n_points-2
    # Since h is defined on [0,2], shift k ranges from 0 to 2 
    # The full correlation gives us shift indices from -n+1 to n-1
    # But we only care about the valid range, which is from 0 to n-1
    # Actually, we want max_{k} int h(x) * (1 - h(x+k)) dx which is different
    # Let's compute all correlations and take the max over all positive shifts
    
    # Let's recompute this properly:
    # For shift k (where 0 <= k <= 2), the integral is:
    # integral_{0}^{2-k} h(x)(1-h(x+k)) dx
    # But in discretized form, we compute the max of all those correlation values
    # corr[i] corresponds to shift i - (n-1) = i - (n-1) (so -n+1 to n-1)
    
    # So we compute from index n-1 to 2n-2 = 2*(n-1) + (n-1) = 3*(n-1) ... No just look: 
    # We have shift indices from -n+1 to n-1 (total of 2n-1 entries).
    # We look at overlaps at various shifts.
    # We want: maximum overlap for k from 0 to n-1 at shift k = n-1 - k or similar logic
    
    n = len(h)
    # max index of correlation is 2*n - 2 (so correlation from k=-n+1 to k=n-1)
    # Positive shifts and the full range makes it confusing.
    
    # Re-evaluate: when h on [0,2] we shift by k with 0<=k<=2 each shift is valid.
    # to go from index to actual shift value: if original index is i, we go to 
    # i -> i+k for correlation核算
    
    # simplification: just use max correlation value in valid range
    # The furthest right shift we can get with h on [0,2] and k up to 2 is when 
    # h(x) and h(x+k) are both active (if h starts at 0 also k max 2)
    # correlation contributes see when shift i, we have correlation at (i+n-1)
    
    # Get all the correlations and find the maximum
    # Hit with diff-view: maximum valid shift is 2 - 0 = 2 (two full intervals)

    # We actually want the max value in [n-1, 2*n-2]  (that is shifts k from 0 to n-1)
    # as we compare i and i+k, with k from 0 to n-1 => corr index from n-1 to 2n-2
    max_val = np.max(corr[n-1:2*n-1]) * dx
    return max_val

def project_h(h, n_points, dx):
    """Project h onto feasible region: [0,1], sum = 1"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to have sum = n_points / 2
    current_sum = np.sum(h) * dx
    scale = (n_points / 2) / current_sum
    h = h * scale
    return h
