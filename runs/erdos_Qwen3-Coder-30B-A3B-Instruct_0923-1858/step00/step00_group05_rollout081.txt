import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral C5"""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximum value over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to enforce sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, dx):
    """Approximate gradient of C5 with respect to h"""
    # Use finite differences to approximate gradient
    eps = 1e-6
    grad = np.zeros_like(h)
    baseline_c5 = evaluate_overlap(h, dx)
    
    for i in range(len(h)):
        h_eps = h.copy()
        h_eps[i] += eps
        h_eps = project_h(h_eps, len(h))
        c5_eps = evaluate_overlap(h_eps, dx)
        grad[i] = (c5_eps - baseline_c5) / eps
    
    return grad

        n_points = 50

    lr = 1e-2

        current_c5 = evaluate_overlap(h, dx)
