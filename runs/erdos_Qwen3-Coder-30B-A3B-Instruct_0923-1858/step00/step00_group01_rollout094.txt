import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h_values):
    """Evaluate the C5 constant for given h values"""
    dx = 2.0 / len(h_values)
    # Use np.correlate with mode='full' to compute all overlaps
    correlation = np.correlate(h_values, 1 - h_values, mode='full')
    # The maximum correlation value is the one we want
    max_overlap = np.max(correlation) * dx
    return max_overlap

def repair_h(h_values):
    """Ensure h_values respects constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    dx = 2.0 / len(h_values)
    h_values = np.clip(h_values, 0, 1)
    current_sum = np.sum(h_values)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h_values = h_values * target_sum / current_sum
        h_values = np.clip(h_values, 0, 1)
    return h_values

def crossover(h1, h2):
    """Perform single-point crossover on two h vectors"""
    n = len(h1)
    crossover_point = random.randint(1, n - 1)
    child = np.zeros(n)
    child[:crossover_point] = h1[:crossover_point]
    child[crossover_point:] = h2[crossover_point:]
    return repair_h(child)

def mutate(h, mutation_rate=0.05):
    """Mutate h by adding small random noise"""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            # Add small random perturbation and repair
            h[i] += random.gauss(0, 0.05)
    return repair_h(h)

def create_initial_population(n_points, population_size, use_initial=True):
    """Create initial population of h values"""
    population = []
    if use_initial and 'initial_h_values' in globals():
        # Use the provided initial construction
        h0 = repair_h(np.array(initial_h_values))
        population.append(h0)
    # Add random individuals to reach desired population size
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = repair_h(h)
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization loop"""
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with 47 points as per initial construction
    n_points = 47
    dx = 2.0 / n_points
    
    # Define parameters
    population_size = 75
    generations = 100
    elite_fraction = 0.2
    crossover_rate = 0.7
    mutation_rate = 0.03
    
    # Initialize population
    population = create_initial_population(n_points, population_size, use_initial=True)
    
    best_h = None
    best_c5 = float('inf')
    time_limit = time.time() + budget_s
    no_improvement_count = 0
    last_best_c5 = float('inf')
    
    for generation in range(generations):
        if time.time() > time_limit:
            break
            
        # Evaluate all individuals in current population
        fitness_scores = []
        for h in population:
            c5 = evaluate_h(h)
            fitness_scores.append(c5)
        
        # Sort population by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update global best
        if sorted_scores[0] < best_c5:
            best_c5 = sorted_scores[0]
            best_h = sorted_population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 50:
            break
        
        # Elite selection (keep top 20%)
        elite_size = max(2, int(elite_fraction * population_size))
        elites = sorted_population[:elite_size]
        
        # Generate new population through crossover and mutation
        new_population = elites[:]  # Start with elites
        
        while len(new_population) < population_size:
            if random.random() < crossover_rate and len(elites) >= 2:
                # Select two parents from elite group
                parent1, parent2 = random.sample(elites, 2)
                child = crossover(parent1, parent2)
                child = mutate(child, mutation_rate)
            else:
                # Just mutate a random elite
                parent = random.choice(elites)
                child = mutate(parent, mutation_rate)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Default fallback if no progress: use initial solution
        try:
            final_h = repair_h(np.array(initial_h_values))
            final_c5 = evaluate_h(final_h)
            return (final_h.tolist(), final_c5, n_points)
        except:
            # Generate default solution
            h = np.full(n_points, 1.0 / n_points)
            h = repair_h(h)
            c5 = evaluate_h(h)
            return (h.tolist(), c5, n_points)
