import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    # Compute the overlap integral using correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum correlation value times dx
    return np.max(corr) * dx

def project_h(h, n_points):
    # Ensure values are in [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale so that sum is exactly n_points / 2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)
    else:
        # Create symmetric starting point
        n_points = 45
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points)
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 1000
    no_improve_count = 0
    
    # Compute initial C5
    dx = 2.0 / n_points
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Iteration counter
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        # Perturb the function
        perturbation = np.random.normal(0, perturb_scale, n_points)
        candidate_h = h + perturbation
        
        # Project back onto feasible set
        candidate_h = project_h(candidate_h, n_points)
        
        # Evaluate the candidate
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject based on temperature
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = candidate_h
            current_c5 = candidate_c5
            
            # Update best solution if needed
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        # Cooling schedule
        T *= alpha
        if iteration % 100 == 0:
            perturb_scale *= 0.99
        
        iteration += 1
        
        # Check for termination conditions
        if no_improve_count >= no_improve_limit:
            break
    
    return (best_h.tolist(), best_c5, n_points)
