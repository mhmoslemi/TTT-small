import numpy as np
from time import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize to sum to 1
    h *= (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time()
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode="full")
        c5_vals = corr * dx
        max_k = np.argmax(c5_vals)
        max_c5 = c5_vals[max_k]
        
        if max_c5 < best_c5:
            best_h = h.copy()
            best_c5 = max_c5
        
        # Early termination if no improvement in a while
        if time() - start_time - last_improvement > 5:
            break
            
        # Identify the most contributing overlap and target adjustments
        # Find indices where this overlap comes from
        max_k_shift = max_k - (n_points - 1)
        
        # Compute the adjustment candidates
        # For k shift, try to adjust pairs (i, i+k) that contribute to overlap
        best_change = -np.inf
        best_i = -1
        best_j = -1
        best_delta = 0
        
        # Loop through all possible adjustments
        for i in range(n_points):
            j = i + max_k_shift
            if j < 0 or j >= n_points:
                continue
                
            # Try small increases and decreases
            for delta in [-0.01, -0.005, 0.005, 0.01]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                    
                # Estimate change in overlap
                est_change = delta * (1 - h[j]) + (h[i] + delta) * (-1)  # Simplified
                if est_change < best_change:
                    best_change = est_change
                    best_i = i
                    best_j = j
                    best_delta = delta

        # Apply the adjustment if it's beneficial
        if best_i >= 0 and best_delta != 0:
            h[best_i] += best_delta
            h[best_j] -= best_delta
            
            # Project back to feasible region
            h = np.clip(h, 0, 1)
            
            # Re-normalize
            h *= (n_points / 2) / np.sum(h)
            
            last_improvement = time() - start_time
        else:
            # Random jitter to escape local minima
            noise = np.random.normal(0, 0.005, n_points)
            h += noise
            h = np.clip(h, 0, 1)  
            h *= (n_points / 2) / np.sum(h) 
                
    return (best_h.tolist(), best_c5, n_points)
