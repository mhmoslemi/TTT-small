def evaluate_erdos_solution(h_values):
    # This is a function to compute C5 from code
    n_points = len(h_values)
    dx = 2.0 / n_points
    corr = np.correlate(h_values, 1-h_values, mode='full') * dx
    return np.max(np.abs(corr))
