import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Fallback: create symmetric construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[-1] = 0.0
        h0[1] = 1.0
        h0[-2] = 1.0
        h0[2] = 0.0
        h0[-3] = 0.0
        
    # Normalize to sum = n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Parameters
    T = 1.0
    alpha = 0.995
    perturb_decay = 0.99
    max_iter = 10000
    no_improve_limit = 500
    no_improve_count = 0
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_C5 = compute_C5(best_h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        perturb_scale = T
        h_candidate = best_h + np.random.normal(0, perturb_scale, n_points)
        
        # Project back to [0,1] and normalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute C5
        C5 = compute_C5(h_candidate, dx)
        
        # Accept or reject
        if C5 < best_C5 or np.random.rand() < np.exp(-(C5 - best_C5) / T):
            best_h = h_candidate.copy()
            best_C5 = C5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Cool temperature
        T *= alpha
        
        # Occasionally decay perturbation scale
        if i % 100 == 0:
            perturb_scale *= perturb_decay
            
        # Check for early termination
        if no_improve_count > no_improve_limit:
            break
            
    return (best_h, best_C5, n_points)

def compute_C5(h, dx):
    # Correlate h with (1-h) using full mode
    correlation = correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the correlation
    max_overlap = np.max(correlation)
    return max_overlap * dx
