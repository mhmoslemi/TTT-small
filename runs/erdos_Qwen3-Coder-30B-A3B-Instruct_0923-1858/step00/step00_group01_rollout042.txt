import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h, dx):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        h = h * (1.0 / total)
    return h

def crossover(h1, h2, dx):
    """Create child via crossover, maintain feasibility"""
    n = len(h1)
    point = np.random.randint(1, n)
    child = np.concatenate([h1[:point], h2[point:]])
    return repair(child, dx)

def mutate(h, dx, mutation_rate=0.05):
    """Mutate h while preserving constraints"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            delta = np.random.normal(0, 0.1)
            h[i] += delta
    return repair(h, dx)

def generate_population(n_points, pop_size, dx, initial=None):
    """Generate initial population"""
    pop = []
    if initial is not None:
        pop.append(repair(initial, dx))
    
    for _ in range(pop_size - len(pop)):
        h = np.random.uniform(0, 1, n_points)
        h = repair(h, dx)
        pop.append(h)
    return pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Starting resolution
    pop_size = 60
    dx = 2.0 / n_points
    
    # Initialize population
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    population = generate_population(n_points, pop_size, dx, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    elite_size = max(5, pop_size // 5)
    generations = 0
    
    while time.time() - start_time < budget_s * 0.95:
        generations += 1
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Elitism
        elites = population[:elite_size]
        
        # Generate new population
        new_pop = elites[:]  # Start with elites
        
        while len(new_pop) < pop_size:
            if len(elites) >= 2:
                parent1, parent2 = np.random.choice(elites, 2, replace=False)
                child = crossover(parent1, parent2, dx)
                child = mutate(child, dx)
                new_pop.append(child)
            else:
                # Generate random
                h = np.random.uniform(0, 1, n_points)
                h = repair(h, dx)
                new_pop.append(h)
        
        population = new_pop[:pop_size]
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h, final_c5, n_points)
