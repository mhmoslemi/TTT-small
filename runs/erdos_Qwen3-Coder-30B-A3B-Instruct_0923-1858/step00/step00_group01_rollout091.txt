```python
import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for a given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, dx):
    """Ensure h is feasible with bounds and integral constraint"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    target_sum = 1.0 / dx
    if h_sum > 0:
        h = h * target_sum / h_sum
    return np.clip(h, 0, 1)

def crossover_h(h1, h2):
    """Create offspring via crossover"""
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return child

def mutate_h(h, mutation_rate=0.05, noise_std=0.1):
    """Mutate h with small random changes"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, noise_std)
    return h

def initial_population(n_points, pop_size, dx):
    """Initialize population with feasible solutions"""
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = repair_h(h0, dx)
        population.append(h0)
    
    # Fill remaining with random feasible individuals
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, dx)
        population.append(h)
        
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Common parameters
    n_points = 47
    dx = 2.0 / n_points
    pop_size = 50
    generations = 1000
    elite_ratio = 0.2
    mutation_rate = 0.05
    noise_std = 0.1
    
    # Initial population
    population = initial_population(n_points, pop_size, dx)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 50:
            break
            
        # Select elite
        elite_size = max(1, int(elite_ratio * pop_size))
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite[:]  # Keep elites
        
        while len(new_population) < pop_size:
            # Select parents
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutate
            child = mutate_h(child, mutation_rate, noise_std)
            
            # Repair
            child = repair_h(child, dx)
            
            new_population.append(child)
            
        population = new_population
        
    return (best_h, best_c5, n_points)
```
