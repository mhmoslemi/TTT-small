import numpy as np
from scipy import signal
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    start_time = time()
    
    # Helper to compute overlap integral and bound
    def compute_c5(h_vals):
        # Normalize
        h_vals = h_vals * target_sum / np.sum(h_vals)
        # Compute cross-correlation
        corr = signal.correlate(h_vals, 1-h_vals, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap, h_vals
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    n_iter = 0
    
    while time() - start_time < budget_s - 1:
        n_iter += 1
        
        # Compute overlaps
        overl = signal.correlate(h, 1-h, mode='full')
        overl *= dx
        c5 = np.max(overl)
        max_indices = np.where(np.abs(overl - c5) < 1e-3)[0]
        
        if not max_indices.size:
            max_indices = [np.argmax(overl)]
        
        # Select a few indices for optimization, avoid too many for performance
        if len(max_indices) > 5:
            max_indices = max_indices[:5]
        
        # Formulate as a feasibility problem to tighten C
        # Introduce variable C and add constraints that all overlaps are <= C
        # But we will solve for a minimal C s.t. all overlap constraints hold
        # So solve: minimize C subject to max_k (h * (1-h))[k] <= C
        
        # Let's keep a simplified approach: project gradient descent
        # Gradients for integration and constraints
        eps = 1e-6
        grad_h = np.zeros_like(h)
        
        # Compute empirical gradient using finite differences for approx
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(n_points):
            h_plus[i] += eps
            h_minus[i] -= eps
            c_plus, _ = compute_c5(h_plus)
            c_minus, _ = compute_c5(h_minus)
            grad_h[i] = (c_plus - c_minus) / (2 * eps)
            h_plus[i] -= eps
            h_minus[i] += eps
        
        # Gradient step: update h to reduce gradient
        lr = 0.005
        h_try = h - lr * grad_h
        
        # Project back into valid range [0,1]
        h_try = np.clip(h_try, 0, 1)
        
        # Renormalize
        h_try = h_try * target_sum / np.sum(h_try)
        
        # Check if this helps
        c5_new, _ = compute_c5(h_try)
        
        if c5_new < c5:
            h = h_try.copy()
            c5 = c5_new
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Occasionally recompute best
        if n_iter % 20 == 0:
            c5_current, _ = compute_c5(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
            
        # Early stopping criterion
        if best_c5 < 0.3807:
            break
            
    # Final computation
    final_c5, final_h = compute_c5(best_h)
    
    return (final_h, final_c5, n_points)
