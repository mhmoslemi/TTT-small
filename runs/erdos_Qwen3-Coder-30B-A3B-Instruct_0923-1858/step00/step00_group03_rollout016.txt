import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    # Compute overlaps for all shifts
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # Return the maximum overlap
    return np.max(overlaps)
