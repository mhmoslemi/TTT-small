import numpy as np
from time import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction or create a default one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a basic starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    dx = 2.0 / len(h)
    target_sum = len(h) / 2.0
    max_iter = 10000
    time_limit = time() + budget_s
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time() > time_limit:
            break
            
        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        c5_vals = corr * dx
        
        k_star = np.argmax(c5_vals)
        max_c5 = c5_vals[k_star]
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find the best pair to modify
        best_reduction = -1
        best_i = -1
        best_j = -1
        
        # Only consider useful positions
        for i in range(len(h)):
            j = i + k_star
            if j < len(h):
                # Estimate effect: reduce both h[i] and h[j]
                # Approximate c5 contribution: h[i] * (1 - h[j])
                contribution = h[i] * (1 - h[j])
                
                # Try increasing h[i] by small amount (but keeping within [0,1])
                new_h_i = min(1.0, h[i] + 0.01)
                new_h_j = max(0.0, h[j] - 0.01)
                # Check if feasible
                if new_h_i <= 1 and new_h_j >= 0:
                    # Estimate new contribution
                    new_contribution = new_h_i * (1 - new_h_j)
                    reduction = contribution - new_contribution
                    if reduction > best_reduction:
                        best_reduction = reduction
                        best_i = i
                        best_j = j
                    
        if best_reduction > 0 and best_i >= 0:
            # Perform adjustment
            delta = 0.01
            if h[best_i] + delta <= 1 and h[best_j] - delta >= 0:
                # Adjust values
                h[best_i] += delta
                h[best_j] -= delta
                
                # Project back to valid sum
                current_sum = np.sum(h)
                diff = current_sum - target_sum
                if abs(diff) > 1e-10:
                    # Distribute difference
                    h = h - diff / len(h)
                    
                    # Clamp to [0,1]
                    h = np.clip(h, 0, 1)
                    
                    # Adjust again to make sum correct
                    current_sum = np.sum(h)
                    diff = current_sum - target_sum
                    if abs(diff) > 1e-10:
                        h = h - diff / len(h)
        
        # Ensure feasibility of all constraints at end of the step
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * (target_sum / current_sum)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr * dx)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
