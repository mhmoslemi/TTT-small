        # We treat each h[i] as a variable and in the objective expression
        # we compute a first-order approximation of the constraint

        grad_h = np.zeros(n_points)

            h_plus = h.copy()

            h_plus = np.clip(h_plus, 0, 1)

        mean_grad = np.mean(grad_h)
