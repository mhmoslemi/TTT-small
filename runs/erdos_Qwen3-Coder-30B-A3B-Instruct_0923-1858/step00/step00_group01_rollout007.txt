import numpy as np
from scipy import linalg
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Start with a coarse construction
        n_points = 20
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[1] = 1.0
        h[2] = 0.0
        h[3] = 1.0
        h[4] = 0.0
        h[5] = 1.0

    dx = 2.0 / n_points
    
    # Ensure the constraint sum(h) * dx = 1 (equivalently sum(h) = n_points/2)
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute 1-h for efficiency
    h_rev = 1 - h
    
    # Optimization parameters
    max_iter = 1000
    lr = 0.01
    decay_every = 100
    min_lr = 1e-6
    
    # Helper functions
    def compute_c5(h, dx):
        # Compute the full cross-correlation h * (1-h)
        correlations = np.correlate(h, h_rev, mode='full')
        # The max of all k values (k = 0, ..., 2*n_points-2)
        max_corr = np.max(correlations)
        return max_corr * dx
    
    def compute_gradient(h, dx):
        # Compute correlations
        correlations = np.correlate(h, h_rev, mode='full')
        # Find where maximum occurs
        max_idx = np.argmax(correlations)
        # The cross-correlation that matters is h[i] * (1-h[i+k]) for k index of max
        # For the full cross-correlation, the value at max_index corresponds to k = max_index - n
        k = max_idx - (n_points - 1)
        # Since we compute correlations at all k's, we derive gradient from that matrix
        # Let's compute the exact gradient more carefully by differentiating the cross-correlation for k = max_k
        # The derivate of each component w.r.t. h[i] is the sum over the elements of the other vector that
        # would contribute to that correlate:
        # e.g for correlation with k=0, ∂corr/∂h[i] = (1-h[i]) = 1-h[i]
        # i.e. for "i" in h and "i+k" in (1-h)
        # For a moment, compute it manually on the corrmatrix:
        dC_dh = np.zeros_like(h)
        # Note for cross-correlation, if the window is from 0 to 2 [0,2] with 2n nodes then
        # Here is an efficient way to compute the gradient w.r.t. h using the two arrays h and 1-h
        shifted_h_rev = np.array(h_rev)
        for i in range(n_points):
            # We need to compute the derivative of correlation wrt h[i]
            for k in range(2 * n_points - 1):
                # This can be derived from the next formula:
                # correlation = Σ_i h[i] * (1-h[i+k])
                # So dC/dh[i] = (1-h[i+k]) if some k exists, else 0
                # For the corresponding state 
                if 0 <= i + k - (n_points - 1) < n_points:
                    j = i + k - (n_points - 1)
                    if 0 <= j < n_points:
                        dC_dh[i] += shifted_h_rev[j]
        
        # The relative scaling between the x-coordinates of the two arrays is aligned
        # So we do this differently:
        # crosscorr[i] = sum(h[j] * (1 - h[j+k]) ) over j such that j+k == i (this is a misleading interpretation)
        # Better approach:
        # crosscorr[k] = sum_{j} h[j] * h_rev[j + k - (n-1)] 
        # interpreted: correlated sum for offset k, let's call it sum_h_rev
        # Correction: for index, we align:
        # Opt: bars = np.correlate(h, 1-h, mode="full"), let's find k of max
        # And compute the derivative.
        # Let's get compute gradient from scratch:
        # mult = np.zeros(n_points) 
        # for i in range(n_points):
        #     for j in range(n_points):
        #         if j + k == i: # k = i - j
        #             mult[i] += h_rev[j]
        # Not efficient.
        # So we compute cross-correlation, extract derivative with respect to max chosen parameter
        # Manually compute it:
        corr_matrix = np.zeros((n_points, 2*n_points - 1))
        for i in range(n_points):
            for k in range(2*n_points - 1):
                j = i + k - (n_points - 1)
                if 0 <= j < n_points:
                    corr_matrix[i, k] = h_rev[j]
                    
        # Priorities for k = max_idx
        grad = np.zeros(n_points)
        if max_idx < n_points:
            for j in range(max_idx + 1):
                if j < n_points:
                    grad[j] += h_rev[max_idx - j]
        else:
            for j in range(max_idx - (n_points - 1), n_points):
                if j >= 0:
                    grad[j] += h_rev[max_idx - (j - (n_points - 1))]
                    
        return grad
    
    def project_h(h, dx):
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to maintain sum = n_points/2
        cr = np.sum(h) * dx
        if cr > 1e-10:
            h = h * (n_points / 2) / cr
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # Main optimization loop
    c5_best = float('inf')
    h_best = h.copy()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute current contribution
        c5_current = compute_c5(h, dx)
        if c5_current < c5_best:
            c5_best = c5_current
            h_best = h.copy()
            
        # Compute gradient
        gradient = compute_gradient(h, dx)
        
        # Gradient descent with learning rate decay
        lr = max(min_lr, lr * 0.995)
        h -= lr * gradient
        # Project to valid range and maintain constraint
        h = project_h(h, dx)
        
        # Occasionally increase resolution
        if iter % 200 == 0 and n_points < 1000:
            # Double resolution
            old_n = n_points
            n_points *= 2
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, old_n), h)
            h_rev = 1 - h 
            h = project_h(h, dx)
        
    # Final evaluation
    final_c5 = compute_c5(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
