import numpy as np
import time
from scipy.signal import correlate

def binary_convert(h, n_points):
    """Convert h to binary with exact sum constraint"""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    
    if current_sum == target_sum:
        return h
    
    # Adjust to meet exact constraint
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = current_sum - target_sum
        ones_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(ones_indices, diff, replace=False)
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        diff = target_sum - current_sum
        zeros_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zeros_indices, diff, replace=False)
        h[flip_indices] = 1
        
    return h

def compute_c5(h, dx):
    """Compute C5 value efficiently"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take the maximum (excluding trivial shifts)
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx

def get_neighbor(h, n_points):
    """Get a neighbor by flipping a 1 to 0 and 0 to 1"""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
        
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    
    new_h = h.copy()
    new_h[i] = 0
    new_h[j] = 1
    return new_h

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999

        new_h = get_neighbor(h, n_points)

            current_c5 = new_c5
            
            if current_c5 < best_c5:

                best_h = h.copy()
        
        # Cool down
        T *= alpha
