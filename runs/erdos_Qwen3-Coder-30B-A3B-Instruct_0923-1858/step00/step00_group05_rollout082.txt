def evaluate_overlap(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Get the maximum value of the convolution
    max_val = np.max(corr)
    return max_val * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
