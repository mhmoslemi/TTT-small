import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h0)
    except:
        n_points = 42
        h0 = np.ones(n_points) * 0.5  # Initial guess
        h0[0] = 0.0
        h0[-1] = 0.0
        h0[1] = 1.0
        h0[-2] = 1.0
        h0 = h0 / np.sum(h0) * 2.0  # Ensure sum is 2.0 after dx=2/n_points
    
    # Ensure it's a valid starting vector (h[i] in [0,1] and sum = 1.0 with dx=2/n_points)
    h0 = np.clip(h0, 0, 1)
    dx = 2.0 / n_points
    h0 = h0 * (1.0 / np.sum(h0) * 2.0)  # normalize sum to 1.0

    # Define bounds and number of points
    max_shift = n_points - 1

    # Find all the possible correlations in the full spectrum
    def compute_overlap(h_vec):
        hvec = np.array(h_vec)
        # Create full correlation (overlap) of h with (1-h)
        corr = signal.correlate(hvec, 1 - hvec, mode='full')
        # Take only double-length parts (index from n_points-1 to 2*n_points-2)
        return corr[n_points-1 : 2*n_points-1]

    # Constraint function
    def solve_subproblem(h_init, max_shift, current_C):
        # Define the optimization variables
        h = cp.Variable(n_points)
        C = cp.Variable()

        # Objective: minimize C
        objective = cp.Minimize(C)

        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == 2.0 / dx,  # equivalent to sum(h) * dx = 1
            C >= 0  # For numerical stability
        ]

        # Derive constraints from recent overlap computation
        overlap_vec = compute_overlap(h_init)
        # We only consider first max_shift + 1 of them (from 0 to max_shift)
        shift_indices = np.arange(0, min(len(overlap_vec), max_shift + 1))
        for i, k in enumerate(shift_indices):
            if i < len(overlap_vec):
                # The k-th shift overlaps should be less than or equal to C * dx 
                # Since overlap[k] = sum(h[i] * (1-h[i+k])), and we scale by dx
                # np.correlate gives us larger overlaps so the scaling factor dx and matching
                # full correction just multiply indices for consistency, so we use meshgrid and
                # extract appropriate values, not vectorized yet fully
                # But in fact the current approach approximates the full computation already
                # because we use signal.correlate with valid → compute it over all starts of shifts 
                # and so our overlap is actually the full value at all overlapping lags up to max_shift.
                # So just: enforce a constraint for each shift that overlap[k] <= C / dx -> duplication of dx
                # And later double-check that we summarize overlap values appropriately.

        # Let's make a better version: 
        # use the (actual full or approximate) overlaps as known
        # however, we still build constraints in loop:
        # Only use top approximately top 10% most violating shifts (at most max_shift)
        
        # Get indices of maximum overlaps
        overlap_indices = np.argsort(overlap_vec)[::-1][:min(max_shift, 30)]  # Top 30 if many
        overlap_indices = overlap_indices[overlap_indices >= 0]  # Positive shift indices

        for k in overlap_indices:
            # k: correlation index (shift) that causes problem
            # we want to construct a constraint for this overlap using variables for h.
            # But this gets complicated in SciPy due to its dual nature.

            # Let's switch to symbolic solver (CVXPY). Let's compute the correlation
            # and get constraints that meet the relationship then built.
            if k < len(overlap_vec):
                constraints.append(cp.sum(h * np.roll(1 - h, k)) * dx <= C)

        # Finally, create problem
        prob = cp.Problem(objective, constraints)
        try:
            # Try solvers, likely MOSEK or ECOS - ECOS is definitely available
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                new_h = h.value
                new_C = C.value
                return new_h, new_C
        except:
            pass
        return h_init, current_C

    best_h = h0.copy()
    best_C = float('inf')

    # Optimization loop
    elapsed = time.time() - start_time
    iteration = 0
    while elapsed < budget_s * 0.95:
        iteration += 1
        # Compute overlap vector based on current h (approximate for now)
        try:
            curr_overlap = compute_overlap(best_h)
            # Take the max
            max_overlap = np.max(curr_overlap)
            current_C = max_overlap / dx
        except Exception:
            current_C = best_C + 1.0
        if current_C < best_C:
            best_C = current_C
            best_h = best_h.copy()

        # Try solving with tight constraint on high overlaps
        new_h, new_C = solve_subproblem(best_h, max_shift, current_C)
        if new_C < best_C:
            best_C = new_C
            best_h = new_h.copy()

        elapsed = time.time() - start_time

    # Final evaluation with true full overlap w.r.t. all shifts  
    overlap_full = compute_overlap(best_h)
    final_C = np.max(overlap_full) / dx

    return (best_h.tolist(), final_C, n_points)
