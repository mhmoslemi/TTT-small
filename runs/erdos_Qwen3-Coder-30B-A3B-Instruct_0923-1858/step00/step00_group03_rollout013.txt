import numpy as np
import time
from scipy.signal import correlate
import math

def binary_convert(h, n_points):
    """Convert h to binary array with exactly n_points/2 ones."""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    
    if current_sum == target_sum:
        return h
    
    # Adjust to meet the sum constraint
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones_indices = np.where(h == 1)[0]
        num_to_flip = current_sum - target_sum
        flip_indices = ones_indices[:num_to_flip]
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(h == 0)[0]
        num_to_flip = target_sum - current_sum
        flip_indices = zeros_indices[:num_to_flip]
        h[flip_indices] = 1
    
    return h

def compute_c5(h, dx):
    """Compute C5 value for given h."""
    # Use numpy's correlate for efficiency
    corr = correlate(h, 1 - h, mode="full")
    # Only take the valid overlaps (not the padded parts)
    middle_start = len(h) - 1
    middle_end = len(h) + len(h) - 1
    corr = corr[middle_start:middle_end]
    
    max_correlation = np.max(corr)
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize as all 0s, then randomize
        h = np.zeros(n_points)
        # Place 1s in roughly half the positions
        indices = np.random.choice(n_points, n_points//2, replace=False)
        h[indices] = 1
    
    h = binary_convert(h, n_points)
    current_c5 = compute_c5(h, dx)
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    improvement_threshold = 1e-6
    best_c5 = current_c5
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Pick two indices to flip
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            continue
            
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Create candidate solution
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Accept with probability
            delta = new_c5 - current_c5
            prob = math.exp(-delta / T)
            if np.random.rand() < prob:
                h = h_new
                current_c5 = new_c5
        
        # Cool down
        T *= alpha
        
        # Small chance to restart if stagnating
        if iteration % 100 == 0 and iteration > 0:
            if abs(current_c5 - best_c5) < improvement_threshold:
                T = max(0.01, T * 0.5)
    
    return (best_h.tolist(), best_c5, n_points)
