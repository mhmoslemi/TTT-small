def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 constant for given h"""
    n = len(h)
    # Compute cross-correlation with 1-h
    corr = np.correlate(h, 1-h, mode="full")
    # Take maximum over all possible shifts
    max_corr = np.max(corr)
    return max_corr * dx

def project_h_to_constraints(h, dx):
    """Project h to satisfy integral constraint and bounds"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to have correct integral
    old_sum = np.sum(h) * dx
    if old_sum > 0:
        h = h * (1.0 / old_sum)
    return h

def construct_sparse_ruler(n_points):
    """Construct a sparse ruler-like initial point"""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Modular form approach
    m = max(2, floor(sqrt(n_points)) + 1)
    s = 1.0  # step value
    
    # Fill positions with modular spacing
    count = 0
    for i in range(n_points):
        if i % m == 0:
            h[i] = s
            count += 1
    
    # Adjust step value to enforce integral constraint
    if count > 0:
        total = s * count
        s = (n_points / 2) / total
        h = h * s
    
    return h

            h_plus[i] += eps
            h_plus = project_h_to_constraints(h_plus, dx)

        return grad
    
    # Use provided initial point
    h = initial_h.copy()

        h_new = project_h_to_constraints(h_new, dx)

        if c5_new < best_c5:

        h = h_new.copy()
        
        # Adaptive step size
        if i % 50 == 0:
            step *= 0.9
            
    return best_h, best_c5
