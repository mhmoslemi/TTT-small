from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

        center = n_points // 2
        h[max(0, center - 5):min(n_points, center + 5)] = 0.8
        h[center - 2:center + 2] = 1.0
        # Normalize
        h = h * (n_points / 2.0) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        # Compute the full cross-correlation with (1-h)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Take the maximum lagged correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vec):
        # Clamp constraints
        h_vec = np.clip(h_vec, 0, 1)
        # Adjust sum to be exactly n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2.0) / current_sum
        return h_vec
    
    # Initialize the best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Gradient descent with adaptive learning rate
    lr = 0.05
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute current C5 and find dominant overlap
        C5 = compute_C5(best_h)
        # Compute full correlation
        corr = correlate(best_h, 1 - best_h, mode='full')
        # Find the lag that gives max correlation
        max_lag_idx = np.argmax(corr)
        # The actual lag is from -n+1 to n-1, so adjust

        grad = np.zeros_like(best_h)

        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] = min(1, h_plus[j] + eps)
            h_plus = project_h(h_plus)
            C5_plus = compute_C5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[j] = max(0, h_minus[j] - eps)
            h_minus = project_h(h_minus)
            C5_minus = compute_C5(h_minus)

        # Take negative gradient (since we want min)

            lr *= 0.5
            
    # Final evaluation

    # Final validation and normalization
    best_h = project_h(best_h)
    final_C5 = compute_C5(best_h)
