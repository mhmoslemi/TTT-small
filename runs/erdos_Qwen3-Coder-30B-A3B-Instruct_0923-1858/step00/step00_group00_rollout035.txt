import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_start = initial_h_values.copy()
        n_points = len(h_start)
    except:
        # Create symmetric initial construction
        n_points = 45
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        mid = n_points // 2
        h_start[mid-1:mid+2] = 1.0
        h_start /= np.sum(h_start) * dx  # normalize
    
    # Refine using optimization
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Binary search parameters
    c_min, c_max = 0.0, 0.5
    iterations = 0
    max_iterations = 20
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        c_mid = (c_min + c_max) / 2.0
        h_candidate = optimize_h_for_c(h_start, c_mid, n_points, dx)
        
        if h_candidate is not None:
            c_test = evaluate_erdos_solution(h_candidate)
            if c_test <= c_mid and c_test <= best_c5:
                best_h = h_candidate.copy()
                best_c5 = c_test
                c_max = c_mid
            else:
                c_min = c_mid
        else:
            c_min = c_mid
            
        iterations += 1
        
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    dx = 2.0 / len(h)
    shifted = np.concatenate([h, np.zeros(len(h))])
    correlations = np.correlate(shifted, 1 - h, mode='full')
    max_correlation = np.max(correlations[:len(h)])
    return max_correlation * dx

def optimize_h_for_c(h_start, c_target, n_points, dx):
    # Use a simple local optimization approach
    def objective(h):
        # Minimize sum of squared differences from h_start
        # While trying to keep the overlap constraint satisfied
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)  # enforce sum
        c_val = evaluate_erdos_solution(h)
        return (c_val - c_target)**2 + 0.01 * np.sum((h - h_start)**2)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    try:
        # Use scipy minimize with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 100})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final / np.sum(h_final) * (n_points / 2)
            return h_final
        return None
    except:
        return None
