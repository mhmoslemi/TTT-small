import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h):
    """Evaluate the overlap integral for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution (divided by n*dx) gives C5
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine number of points
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 45
        # Start with a symmetric function that's peaked in the center
        x = np.linspace(0, 2, n_points)
        h = np.exp(-((x - 1)**2) / 0.1)
        h = project_h(h, n_points)
    
    # Initial evaluation
    best_c5 = evaluate_h(h)
    best_h = h.copy()
    
    # Parameters for Nelder-Mead
    max_reset_attempts = 10
    reset_iter = 50
    n = len(h)
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink
    
    # Initialize simplex around initial point
    simplex_size = 0.1
    simplex = [h.copy()]
    for _ in range(n):
        pert = np.random.normal(0, simplex_size, n)
        pert = project_h(h + pert, n_points)
        simplex.append(pert)
    
    iter_count = 0
    last_improvement = 0
    reset_attempts = 0
    simplex_size = 0.1
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        if iter_count > 5000:
            break
            
        # Evaluate all simplex points
        evals = []
        for point in simplex:
            proj_point = project_h(point, n_points)
            c5 = evaluate_h(proj_point)
            evals.append(c5)
            
        # Sort simplex by evaluation (best first)
        indices = np.argsort(evals)
        simplex = [simplex[i] for i in indices]
        evals = [evals[i] for i in indices]
        
        # Check if we improved
        current_best = evals[0]
        if current_best < best_c5 - 1e-10:
            best_c5 = current_best
            best_h = simplex[0].copy()
            last_improvement = iter_count
            
        # Check for stagnation
        if iter_count - last_improvement > reset_iter:
            reset_attempts += 1
            if reset_attempts > max_reset_attempts:
                break
            simplex_size *= 0.5
            last_improvement = iter_count
            
        # Compute centroid of best points
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        worst = simplex[-1]
        h_r = centroid + alpha * (centroid - worst)
        h_r = project_h(h_r, n_points)
        c5_r = evaluate_h(h_r)
        
        # Accept reflection if it's better than worst
        if c5_r < evals[-2]:
            if c5_r < evals[0]:
                # Expansion
                h_e = centroid + gamma * (centroid - worst)
                h_e = project_h(h_e, n_points)
                c5_e = evaluate_h(h_e)
                if c5_e < c5_r:
                    simplex[-1] = h_e
                else:
                    simplex[-1] = h_r
            else:
                simplex[-1] = h_r
        else:
            # Contraction
            h_c = centroid + beta * (centroid - worst)
            h_c = project_h(h_c, n_points)
            c5_c = evaluate_h(h_c)
            if c5_c < evals[-1]:
                simplex[-1] = h_c
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
                simplex = [project_h(s, n_points) for s in simplex]
    
    # Final check
    best_c5 = evaluate_h(best_h)
    if best_c5 > 0.3807:
        # Try a few more times to see if we can improve it
        for _ in range(100):
            if time.time() - start_time >= budget_s:
                break
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            new_h = project_h(best_h + noise, n_points)
            new_c5 = evaluate_h(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
