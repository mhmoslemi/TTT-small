import numpy as np
from scipy import signal
import time
import math

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx for actual integral
    corr *= dx
    # Return the maximum value
    return np.max(corr)

def project_h(h, dx):
    """Project h to satisfy constraints"""
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to make sum = 1
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / h_sum)
    return h
