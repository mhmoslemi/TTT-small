def evaluate_erdos_solution(h_vals, n_points, dx):
    h = np.array(h_vals)
    # Compute correlations as done in original function
    corr = np.correlate(h, 1-h, mode='full')
    # External indices only
    corr = corr[n_points-1:]
    c5 = np.max(corr) * dx
    return c5
