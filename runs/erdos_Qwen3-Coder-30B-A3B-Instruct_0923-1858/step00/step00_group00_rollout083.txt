import numpy as np
from scipy import signal
import time

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.convolve(h, 1 - h, mode='full')
    # Find maximum value and scale by dx
    max_val = np.max(conv) * dx
    return max_val

def project_h(h, n_points):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n/2."""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def nelder_mead_step(vertices, values, alpha=1.0, gamma=2.0, beta=0.5, rho=0.5):
    """Perform one step of Nelder-Mead optimization."""
    n = len(vertices)
    # Sort vertices by value
    indices = np.argsort(values)
    vertices = [vertices[i] for i in indices]
    values = [values[i] for i in indices]
    
    # Best, worst, second worst
    best = vertices[0]
    worst = vertices[-1]
    second_worst = vertices[-2]
    
    # Compute centroid of all but worst
    centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    h_reflect = centroid + alpha * (centroid - worst)
    h_reflect = project_h(h_reflect, len(h_reflect))
    
    # Reflection value
    val_reflect = evaluate_overlap(h_reflect)
    
    # If better than best, try expansion
    if val_reflect < values[0]:
        h_expand = centroid + gamma * (h_reflect - centroid)
        h_expand = project_h(h_expand, len(h_expand))
        val_expand = evaluate_overlap(h_expand)
        if val_expand < val_reflect:
            return h_expand, val_expand
        else:
            return h_reflect, val_reflect
    # If better than second worst
    elif val_reflect < values[-2]:
        return h_reflect, val_reflect
    # Else contraction
    else:
        # Outside contraction
        if val_reflect < values[-1]:
            h_contract = centroid + beta * (h_reflect - centroid)
        else:
            # Inside contraction
            h_contract = centroid - beta * (worst - centroid)
        h_contract = project_h(h_contract, len(h_contract))
        val_contract = evaluate_overlap(h_contract)
        return h_contract, val_contract
