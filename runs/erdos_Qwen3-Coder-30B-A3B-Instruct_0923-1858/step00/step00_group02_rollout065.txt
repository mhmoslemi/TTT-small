def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral for given h."""
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode='full')
    # The maximum value is the maximum correlation
    max_corr = np.max(corr)
    return max_corr * dx
