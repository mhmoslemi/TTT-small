import numpy as np
import cvxpy as cp
import time

def evaluate_overlap(h, n_points):
    """Compute the maximum overlap for given h."""
    dx = 2.0 / n_points
    h_full = np.concatenate([h, h])
    overlaps = []
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            shifted = h_full[k:k + n_points]
        else:
            shifted = h_full[n_points + k:n_points]
        overlap = np.sum(h * (1 - shifted)) * dx
        overlaps.append(overlap)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default construction
        n_points = 45
        h0 = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Binary search for minimum C_5
    C_low = 0.0
    C_high = 0.38085
    best_C = C_high
    best_h = h0.copy()
    
    # Grid for shifts
    shift_range = range(-n_points + 1, n_points)
    
    def create_overlap_constraint(h, k):
        # Convert shift k to indices for convolution-like operation
        n = len(h)
        if k >= 0:
            # h[i] * (1 - h[i+k]) where i+k < n
            indices1 = np.arange(n - k)
            indices2 = np.arange(k, n)
        else:
            indices1 = np.arange(-k, n)
            indices2 = np.arange(n + k)
        return np.sum(h[indices1] * (1 - h[indices2]))
    
    while time.time() - start_time < budget_s - 1:
        C_mid = (C_low + C_high) / 2.0
        # Construct and solve linear program
        
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2.0
        ]
        
        # Add overlap constraints
        for k in shift_range:
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0 .. n-k-1
                indices1 = np.arange(n_points - k)
                indices2 = np.arange(k, n_points)
                expr = cp.sum(cp.multiply(h_vars[indices1], 1 - h_vars[indices2]))
            else:
                # h[i] * (1 - h[i+k]) for i = -k .. n-1
                indices1 = np.arange(-k, n_points)
                indices2 = np.arange(n_points + k)
                expr = cp.sum(cp.multiply(h_vars[indices1], 1 - h_vars[indices2]))
            constraints.append(expr <= C_mid * n_points)
        
        # Create problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h_vars.value
                if h_solution is not None:
                    if np.all(h_solution >= 0) and np.all(h_solution <= 1):
                        if abs(np.sum(h_solution) - n_points / 2) < 1e-6:
                            # Validate max overlap
                            max_overlap = evaluate_overlap(h_solution, n_points)
                            if max_overlap <= C_mid + 1e-6:
                                best_h = h_solution.copy()
                                best_C = C_mid
                                C_high = C_mid
                            else:
                                C_low = C_mid
                        else:
                            C_low = C_mid
                    else:
                        C_low = C_mid
                else:
                    C_low = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
            
        if C_high - C_low < 1e-6:
            break
    
    # Final validation
    final_max_overlap = evaluate_overlap(best_h, n_points)
    return (best_h.tolist(), final_max_overlap, n_points)
