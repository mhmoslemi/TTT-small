import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute C5 as max of correlation values
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, target_sum):
    # Clamp to [0,1] and then scale to target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def init_population(n_points, pop_size, initial_h=None):
    population = []
    
    if initial_h is not None:
        population.append(initial_h.copy())
    else:
        # Use a simple construction
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points / 2)
        population.append(h)
    
    # Fill rest of population with random valid points
    np.random.seed(42)
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points / 2)
        population.append(h)
    
    return population

def tournament_select(population, fitnesses, k=5):
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    # Blend crossover between two parents
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, rate=0.1, noise_std=0.01):
    # Mutate with Gaussian noise
    mask = np.random.rand(len(h)) < rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + noise * mask
    h = project_to_feasible(h, len(h) / 2)
    return h

def ga_optimize_h(initial_h, n_points, pop_size=30, max_gen=1000, budget_s=600):
    start_time = time.time()
    
    # Set up GA parameters
    dx = 2.0 / n_points
    
    # Initialize population
    population = init_population(n_points, pop_size, initial_h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    global_best_idx = np.argmin(fitnesses)
    global_best = population[global_best_idx].copy()
    best_fitness = fitnesses[global_best_idx]
    
    elite_size = max(1, int(pop_size * 0.1))
    replace_size = max(1, int(pop_size * 0.2))
    
    generation = 0
    
    while time.time() - start_time < budget_s and generation < max_gen:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update global best
        if fitnesses[0] < best_fitness:
            best_fitness = fitnesses[0]
            global_best = population[0].copy()
        
        # Create new population
        new_population = population[:elite_size].copy()  # Keep the elite
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.7:
                child = blend_crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child)
            
            new_population.append(child)
        
        # Truncate to exact size if over
        new_population = new_population[:pop_size]
        population = new_population
        
        # Evaluate new population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        generation += 1
        
        # Check for early termination (no improvement for many steps)
        if generation % 100 == 0 and generation > 0:
            if abs(fitnesses[0] - best_fitness) < 1e-6:
                break
    
    return global_best, best_fitness, n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    pop_size = 30
    
    # Get initial solution if available
    initial_h = getattr(__builtins__, 'initial_h_values', None)
    
    # Optimize
    best_h, best_c5, n_points = ga_optimize_h(initial_h, n_points, pop_size, budget_s=budget_s)
    
    return (best_h.tolist(), best_c5, n_points)
