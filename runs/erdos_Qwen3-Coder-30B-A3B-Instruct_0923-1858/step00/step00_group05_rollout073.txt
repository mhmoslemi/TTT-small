import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = correlate(h, 1 - h, mode="full")
    max_corr = np.max(np.abs(correlations))
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    max_iter = 100000
    start_time = time.time()
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Adjust h to satisfy constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Training parameters
    learning_rate = 0.01
    perturbation_rate = 0.1
    iter_count = 0
    stagnation_count = 0
    min_improvement = 1e-6

    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute current correlations
        correlations = correlate(h, 1 - h, mode="full")
        C5_current = np.max(np.abs(correlations)) * dx

        # Update best solution
        if C5_current < best_C5:
            best_h = h.copy()
            best_C5 = C5_current
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Gradient computation
        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j])

        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm

        # Gradient step
        h = h - learning_rate * gradient

        # Project to valid range [0, 1]
        h = np.clip(h, 0, 1)

        # Project to correct sum
        h = h * (n_points / 2) / np.sum(h)

        # Periodic perturbation to escape local optima
        if iter_count % 200 == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Adapt learning rate
        if stagnation_count > 100:
            learning_rate *= 0.9
            stagnation_count = 0
        if learning_rate < 1e-6:
            break

        iter_count += 1

    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    return (best_h, final_C5, n_points)
