import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    T = 1.0  # Initial temperature
    alpha = 0.995  # Cooling rate
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_count = 0
    max_no_improve = 500
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        
        # Project back to [0,1] and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        C5_new = compute_C5(h_new, dx)
        
        # Accept or reject
        delta_C5 = C5_new - best_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cooling schedule
        T *= alpha
        
        if i % 100 == 0:
            perturb_scale *= 0.99
            
        if no_improve_count > max_no_improve:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # This assumes h is already normalized to sum to n_points/2
    # Using scipy's correlate with mode="full"
    # The correlation of h with (1-h) gives us the overlap integral values
    # We want max over all lags
    corr = correlate(h, 1 - h, mode="full")
    # Since correlate in scipy multiplies by dx, we need to scale properly
    # Actually, we just want max of this
    return np.max(corr) * dx
