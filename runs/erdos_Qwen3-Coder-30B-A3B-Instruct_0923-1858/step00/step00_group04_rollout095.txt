import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    corr = signal.correlate(h, 1 - h, mode="full")
    max_corr = np.max(np.abs(corr))
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto the feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total > 0:
        h = h * (n_points / 2) / total
    return h

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover with alpha blending."""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_scale=0.01):
    """Apply Gaussian noise mutation."""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_scale, len(h))
    h = h + mask * noise
    return h

def tournament_selection(population, fitnesses, k=5):
    """Select an individual via tournament selection."""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    dx = 2.0 / n_points
    pop_size = 30
    max_gens = 1000
    elite_rate = 0.1
    mutation_rate = 0.1
    crossover_rate = 0.7
    tournament_size = 5
    no_improve_limit = 200

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = project_h(initial_h, n_points)

    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = np.inf
    no_improve_count = 0

    for gen in range(max_gens):
        if time.time() - start_time > budget_s:
            break

        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve_count = 0
        no_improve_count += 1

        # Elitism
        elite_size = int(elite_rate * pop_size)
        elites = [population[i] for i in np.argsort(fitnesses)[:elite_size]]

        # Selection, crossover, mutation
        new_pop = elites[:]
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1 = tournament_selection(population, fitnesses, tournament_size)
                parent2 = tournament_selection(population, fitnesses, tournament_size)
                child = crossover(parent1, parent2)
            else:
                parent = tournament_selection(population, fitnesses, tournament_size)
                child = parent.copy()
            
            child = mutate(child, mutation_rate)
            child = project_h(child, n_points)
            new_pop.append(child)

        population = new_pop

        if no_improve_count > no_improve_limit:
            break

    return (best_h, best_c5, n_points)
