import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from global initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial vector
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.7
        h[-1] = 0.3
    
    dx = 2.0 / n_points
    max_iter = 1000
    patience = 100
    no_improve_count = 0
    best_C5 = float('inf')
    best_h = h.copy()
    
    def compute_C5(h_vals):
        # Compute the max overlap integral
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        return max_val * dx
    
    def project_h(h_vals):
        # Project h to feasible region
        h_vals = np.clip(h_vals, 0, 1)
        total = np.sum(h_vals) * dx
        if abs(total - 1.0) > 1e-10:
            scale = 1.0 / total
            h_vals *= scale
        return h_vals
    
    def perturb_h(h_vals, k_max, overlap_indices):
        h_new = h_vals.copy()
        # Get cross-correlation
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Do small perturbations in critical regions
        if len(overlap_indices) < 2:
            return h_vals
        cand_pairs = []
        # Try small moves between points in overlap regions
        for _ in range(100):  # limit iterations per call
            if time.time() - start_time > budget_s - 1:
                break
            i, j = np.random.choice(overlap_indices, 2, replace=False)
            transfer = min(h_new[i], 1 - h_new[j])
            if transfer <= 1e-10:
                continue
            delta = np.random.uniform(0, transfer) * 0.1
            h_new[i] -= delta
            h_new[j] += delta
            if np.all((h_new >= 0) & (h_new <= 1)):
                cand_pairs.append((h_new.copy(), delta, i, j))
            else:
                h_new[i] += delta
                h_new[j] -= delta
        if not cand_pairs:
            return h_vals
        # Evaluate each candidate
        best_candidate = h_vals.copy()
        best_C5_val = compute_C5(h_vals)
        for h_cand, delta, i, j in cand_pairs:
            new_C5 = compute_C5(h_cand)
            if new_C5 < best_C5_val:
                best_C5_val = new_C5
                best_candidate = h_cand
        return best_candidate
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
        C5_val = compute_C5(h)
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        if no_improve_count > patience:
            break
            
        # Identify the critical shift k_max
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        # For large resolution, analyze most affected areas
        n_full = 2 * n_points - 1
        overlap_indices = []
        # Indices around the peak shift
        for i in range(max(0, k_max - 5), min(n_full, k_max + 5)):
            if corr[i] > 0.9 * np.max(corr):  # consider values within 10% of max
                # Map to original h indices
                if i < n_points:
                    overlap_indices.append(i)
                else:
                    overlap_indices.append(i - n_points)
        # Apply perturbation only in overlap regions
        h = project_h(perturb_h(h, k_max, overlap_indices))
    
    # Interpolation after refinement
    if n_points > 20:
        new_n = 2 * n_points
        h_up = np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, n_points), h)
        h_up = project_h(h_up)
        C5_up = compute_C5(h_up)
        if C5_up < best_C5:
            best_h = h_up
            best_C5 = C5_up

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
