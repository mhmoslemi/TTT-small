import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for a given h."""
    # Compute the overlap integral for all shifts
    # h is assumed to be normalized so that sum(h)*dx = 1
    # We use correlation to compute the overlap efficiently
    corr = correlate(h, 1 - h, mode='full')
    # Only consider the relevant part of the correlation
    # The overlap for shift k is corr[n_points-1+k]
    # We want max over k in [-n_points+1, n_points-1]
    # But since h is 0 outside [0, n_points-1], we only consider valid shifts
    overlaps = np.zeros(2 * len(h) - 1)
    for k in range(-len(h)+1, len(h)):
        idx = len(h)-1+k
        if 0 <= idx < len(corr):
            overlaps[k] = corr[idx]
    # Multiply by dx to get the actual value
    overlaps *= dx
    # Return the maximum overlap
    return np.max(overlaps)

def binary_convert(h, n_points):
    """Convert a continuous h to a binary one, obeying the sum constraint."""
    # Round first
    h_binary = np.round(h)
    # Adjust to meet sum constraint
    current_sum = np.sum(h_binary)
    target_sum = n_points / 2
    if current_sum == target_sum:
        return h_binary
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        # Find indices where h_binary is 1 and decrease them
        ones_idx = np.where(h_binary == 1)[0]
        to_flip = int(current_sum - target_sum)
        # Flip the last few ones to zeros (least significant bits)
        h_binary[ones_idx[-to_flip:]] = 0
    else:
        # Need to flip some 0s to 1s
        zeros_idx = np.where(h_binary == 0)[0]
        to_flip = int(target_sum - current_sum)
        # Flip the first few zeros to ones
        h_binary[zeros_idx[:to_flip]] = 1
    return h_binary

def local_search_binary(h_initial, n_points, dx, budget_s, seed=42):
    """Perform simulated annealing on binary configurations."""
    np.random.seed(seed)
    # Initial solution
    h = binary_convert(h_initial, n_points)
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 1.0
    alpha = 0.999
    max_iter = 100000
    iter_count = 0
    start_time = time.time()
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        # Select two indices at random
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Save current state
        old_h = h.copy()
        old_c5 = current_c5
        # Swap 1 and 0
        h[i] = 0
        h[j] = 1
        # Compute new C5
        new_c5 = evaluate_h(h, dx)
        # Accept or reject
        delta = new_c5 - old_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Restore old state
            h[:] = old_h
        # Decrease temperature
        T *= alpha
        iter_count += 1
        # Early stop if improvement is very slow
        if iter_count % 1000 == 0:
            if T < 1e-10:
                break
    return best_h, best_c5

            h_start[:n_points//2] = 1
            # Normalize sum to be 1 (but later binary_convert handles correction)
        # Adjust it so sum is exactly correct
        h_start = h_start / np.sum(h_start) * n_points / 2
        # Run optimization
        h_opt, c5_opt = local_search_binary(h_start, n_points, dx, budget_s, seed)
