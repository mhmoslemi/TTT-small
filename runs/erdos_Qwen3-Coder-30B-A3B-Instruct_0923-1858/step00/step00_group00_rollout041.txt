import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate C5 for given h."""
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # Max value over all lags, scaled by dx
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to meet constraints: h[i] in [0,1], sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def perturb_h(h, temperature, n_points):
    """Perturb h with noise scaled by temperature."""
    noise = np.random.normal(0, temperature, len(h))
    h_new = h + noise
    return project_h(h_new, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get starting construction if available, otherwise create symmetric one.
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a symmetric starting vector
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_h(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 500
    no_improve_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_new = perturb_h(h, perturb_scale, n_points)
        C5_new = evaluate_h(h_new, dx)
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Accept worse solutions with probability
        if C5_new > best_C5:
            prob = np.exp(-(C5_new - best_C5) / T)
            if np.random.rand() < prob:
                h = h_new
        else:
            h = h_new
            
        # Cooling and perturbation scale adaptation
        T *= alpha
        if i % 100 == 0 and perturb_scale > 0.001:
            perturb_scale *= 0.99
            
        if no_improve_count > no_improve_limit:
            break
            
    return (best_h.tolist(), best_C5, n_points)
