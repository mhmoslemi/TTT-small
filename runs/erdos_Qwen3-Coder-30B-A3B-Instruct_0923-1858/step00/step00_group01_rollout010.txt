import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initialization if no parent construction exists
        n_points = 47
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Ensure sum(h) * dx = 1
        h *= 1.0 / (np.sum(h) * dx) * 1.0
        dx = 2.0 / len(h)
    
    n_points = len(h)
    dx = 2.0 / n_points

    # Optimization parameters
    max_iter = 1000
    patience = 100
    best_c5 = float('inf')
    patience_count = 0
    
    def compute_c5(h_vals):
        # Compute cross-correlation: correlate(h, 1-h, mode='full')
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # The full correlation has length 2*n_points - 1
        # The maximum shift is from -n_points+1 to n_points-1
        # But we are interested in positive shifts k from 0 to n_points-1
        # The first n_points elements of corr correspond to negative lags
        # The middle element (idx=n_points-1) corresponds to lag 0
        # The next n_points-1 elements correspond to lags 1 to n_points-1
        # So we look at corr[n_points-1:] for shifts 0 to n_points-1
        c5 = np.max(corr[n_points-1:]) * dx
        return c5
    
    # Check initial solution
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute cross-correlation to find current maximum
        corr = correlate(h, 1 - h, mode='full')
        corr_max = np.max(corr[n_points-1:])
        # Find indices where maximum occurs
        max_indices = np.where(corr[n_points-1:] == corr_max)[0]  # shifts 0 to n_points-1
        if len(max_indices) == 0:
            # Should not happen if corr is computed properly
            continue
            
        # Use first maximum shift
        k = max_indices[0]  # this gives shift 0 to n_points-1

        # Determine regions that influence the maximum
        # Forward shift k, so we're looking at h[i] * (1 - h[i+k])
        # Overlap: where both terms exist
        # i from 0 to n_points - k - 1
        # From shift k=0, also latest k points might be relevant but we do not want to overcomplicate
        # Let's just focus on active region for influence

        # This is a greedy approach
        # Try making small changes where the maximum occurs
        h_new = h.copy()

        # Identify candidate indices that contribute the most to the cross-correlation at shift k
        # The term h[i] * (1 - h[i+k]) contributes to the correlation
        # We want to make changes that reduce this as much as possible
        # So we want to keep h[i] and h[i+k] away from extremes if they're contributing

        # Focus on a moving window around where correlation is maximal
        # Compute gradients or local impact
        # For now, make simple local perturbations to i and i+k

        # Choose a pair of indices to modify for potential reduction
        idx1 = np.random.randint(0, n_points - k)  # ensure i + k < n_points
        idx2 = idx1 + k

        # Determine magnitude of possible change
        delta = 0.01 * np.random.random()  # small random change
        # Ensure h remains in [0, 1]
        # How much can we trade between idx1 and idx2?
        delta1 = min(delta, h[idx1])  # How much can we reduce idx1?
        delta2 = min(delta, 1 - h[idx2])  # How much can we increase idx2?
        max_transfer = min(delta1, delta2)
        if max_transfer <= 0:
            continue

        # Make a transfer
        transfer = max_transfer * (0.5 + 0.5 * np.random.random())
        h_new[idx1] -= transfer
        h_new[idx2] += transfer

        # Check boundsafety
        if np.any(h_new < 0) or np.any(h_new > 1):
            continue

        # Check sum constraint
        sum_check = np.sum(h_new) * dx
        if abs(sum_check - 1.0) > 1e-6:
            # Adjust sum to maintain constraint
            adjust = (1.0 - sum_check) / n_points
            h_new += adjust
            if np.any(h_new < 0) or np.any(h_new > 1):
                continue

        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept improvement or acceptance criterion
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1

        # Early termination if no improvement for too long
        if patience_count >= patience:
            break

    # Final validation and return
    computed_c5 = compute_c5(best_h)
    
    # Ensure we return exactly what is asked
    # (h_values, c5_bound, n_points)
    return (best_h.tolist(), float(computed_c5), len(best_h))
