import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum of the correlation (scaled)
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction if available, otherwise generate
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        # Start with a triangular waveform
        n = 45
        h = np.zeros(n)
        x = np.linspace(0, 2, n)
        for i in range(n):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum = n/2
        h = h * (n/2) / np.sum(h)
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n/2) / np.sum(h)
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    last_improvement = 0
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0

    # Gradient approximation helper
    def finite_diff_grad(h, eps=1e-4):
        grad = np.zeros_like(h)
        h_copy = h.copy()
        for i in range(n):
            h_plus = h_copy.copy()
            h_minus = h_copy.copy()
            h_plus[i] = min(1, h_copy[i] + eps)
            h_minus[i] = max(0, h_copy[i] - eps)
            # Normalize the modified vectors to sum to n/2
            h_plus = h_plus * (n/2) / np.sum(h_plus)
            h_minus = h_minus * (n/2) / np.sum(h_minus)
            f_plus = evaluate_erdos_solution(h_plus)
            f_minus = evaluate_erdos_solution(h_minus)
            grad[i] = (f_plus - f_minus) / (2 * eps)
        return grad

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        grad = finite_diff_grad(h)
        # Trust region update
        update = -grad * trust_radius
        
        # Try the update
        h_new = h + update
        # Clip to [0,1] and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            trust_radius *= 1.2  # Expand trust region
        else:
            trust_radius *= 0.5  # Shrink trust region
            # Add some random perturbation
            if iter_count - last_improvement > 100:
                noise = np.random.normal(0, 0.05, n)
                h_pert = h + noise
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n/2) / np.sum(h_pert)
                c5_pert = evaluate_erdos_solution(h_pert)
                if c5_pert < current_c5:
                    h = h_pert
                    last_improvement = iter_count

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
