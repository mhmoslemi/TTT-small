import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    correlations = np.correlate(h, 1-h, mode="full")
    max_corr = np.max(np.abs(correlations))
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    max_iter = 10000
    learning_rate = 0.01
    perturbation_rate = 0.1
    perturbation_freq = 100
    
    # Initialize
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize sum to be 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h)
        
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j])
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Gradient descent step
        h = h - learning_rate * gradient
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturbation
        if iteration % perturbation_freq == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h, best_C5, n_points)
