import numpy as np
import time
from scipy import optimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create initial construction with 47 points
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure constraint ∫ h dx = 1
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    
    # Variables to track progress
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        
        # Compute current cross-correlation
        ccorr = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(ccorr)
        max_k = max_idx - (len(h) - 1)
        current_c5 = np.max(ccorr) * dx
        
        # Identify overlapping regions contributing to maximum
        overlap_regions = []
        if max_k >= 0:
            # h(x) overlaps with (1-h)(x + max_k)
            for i in range(len(h) - max_k):
                overlap_regions.append((i, i + max_k))
        else:
            # h(x + |max_k|) overlaps with (1-h)(x)
            for i in range(len(h) + max_k):
                overlap_regions.append((i - max_k, i))
        
        # Perturb a few indices based on possible overlaps
        candidates = []
        for _ in range(10):  # Make 10 candidates per iteration
            h_new = h.copy()
            # Randomly sample from overlap regions
            if not overlap_regions:
                i = np.random.randint(0, len(h))
                j = np.random.randint(0, len(h))
            else:
                i, j = overlap_regions[np.random.randint(len(overlap_regions))]
                # Ensure i and j are valid indices (to avoid index-out-of-bounds)
                i = max(0, min(i, len(h)-1))
                j = max(0, min(j, len(h)-1))
            # Perturb h[i] and h[j]
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer > 1e-8:
                transfer = np.random.uniform(0, max_transfer)
                h_new[i] -= transfer
                h_new[j] += transfer
                # Ensure bounds
                h_new[i] = max(0, min(1, h_new[i]))
                h_new[j] = max(0, min(1, h_new[j]))
                
                # Check the new C5
                new_c5 = compute_c5(h_new, dx)
                candidates.append((h_new.copy(), new_c5))
        
        if not candidates:
            no_improvement_count += 1
            continue
            
        # Pick the best candidate
        best_candidate = min(candidates, key=lambda x: x[1])
        h_new, new_c5 = best_candidate
        
        # Accept the move if it's better
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
            best_h = h.copy()
        else:
            no_improvement_count += 1

    # Final evaluation with double resolution for comparison
    if n_points < 100:  # Only upsample if not already high resolution
        h_upsampled = np.interp(np.linspace(0, 2, 2*n_points), np.linspace(0, 2, n_points), h)
        final_c5 = compute_c5(h_upsampled, 2.0/(2*n_points))
    else:
        final_c5 = best_c5
    
    return list(best_h), final_c5, len(best_h)

def compute_c5(h, dx):
    """Compute the C5 constant using discrete cross-correlation."""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx
