import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Convolution of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum value over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    n_points = len(h)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def genetic_algorithm(n_points, budget_s):
    """Use genetic algorithm to find good initial h"""
    pop_size = 100
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, size=10, replace=False)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        next_pop = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                cx_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
                child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.1)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.1)
            
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            next_pop.extend([child1, child2])
        
        population = next_pop[:pop_size]
    
    return best_h, best_c5

def refine_with_local_search(h, budget_s):
    """Refine the best solution using local search"""
    start_time = time.time()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Ensure sum constraint (project)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2.0) / current_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx  # Negative because we want to minimize
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2.0
    
    # Optimize with scipy
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Try both Nelder-Mead and differential evolution
    try:
        # Try Nelder-Mead (faster)
        result = minimize(objective, h, method='Nelder-Mead', 
                          bounds=[(0, 1) for _ in range(n_points)], 
                          options={'maxiter': 1000})
        h_new = result.x.reshape(n_points)
        h_new = project_h(h_new)
        c5_new = evaluate_h(h_new)
        if c5_new < evaluate_h(h):
            return h_new, c5_new
    except:
        pass
    
    # Try differential evolution
    try:
        result = differential_evolution(objective, bounds, maxiter=100, 
                                       popsize=15, seed=42)
        h_new = result.x.reshape(n_points)
        h_new = project_h(h_new)
        c5_new = evaluate_h(h_new)
        if c5_new < evaluate_h(h):
            return h_new, c5_new
    except:
        pass
    
    return h, evaluate_h(h)
