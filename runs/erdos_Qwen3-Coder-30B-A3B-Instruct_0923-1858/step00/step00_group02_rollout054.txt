import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Triangular initialization
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * n_points / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    best_C5 = min(best_C5, 0.38085)  # safety limit
    
    # Gradient descent with perturbations
    eta = 0.01
    perturb_rate = 0.05
    patience = 0
    patience_limit = 100
    
    for i in range(100000):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5
        current_C5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            C5_plus = np.max(np.correlate(h_plus, 1-h_plus, mode="full")) * dx
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            C5_minus = np.max(np.correlate(h_minus, 1-h_minus, mode="full")) * dx
            
            grad[j] = (C5_plus - C5_minus) / (2 * eps)
            
        # Update
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Random perturbations
        if i % 50 == 0 and patience > 0:
            mask = np.random.random(n_points) < perturb_rate
            h[mask] = np.random.uniform(0, 1, size=np.sum(mask))
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate
        if i % 100 == 0:
            eta *= 0.95
            
    # Final check of best solution
    final_C5 = np.max(np.correlate(best_h, 1-best_h, mode="full")) * dx
    return (best_h.tolist(), final_C5, len(best_h))
