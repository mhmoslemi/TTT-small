import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Symmetric initialization
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 0.7
        h[n_points//2:] = 0.3
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h_vals):
        h = np.array(h_vals)
        # For each shift k, compute the overlap integral
        # We'll compute correlations with 1-h
        h_padded = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1 - h
        one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n_points)])
        correlation = np.correlate(h_padded, one_minus_h_padded, mode='full')
        # Take only the full convolution with the shifts that make sense
        shift_range = range(n_points)
        overlap_values = np.array([correlation[n_points + k] for k in shift_range])
        return np.max(overlap_values) * dx
    
    def solve_feasibility(h_guess, C_target):
        n = len(h_guess)
        h = Variable(n)
        constraints = [h >= 0, h <= 1]
        # Add integral constraint
        constraints.append(sum(h) == n/2)
        
        # Add bilinear constraints using McCormick envelopes
        C = C_target
        # We want max_k int h(x)(1-h(x+k)) dx <= C
        # This means for all k, sum(h[i] * (1-h[i+k]) * dx) <= C
        # This is the same as sum(h[i]*1 - h[i]*h[i+k]) <= C/dx
        # which is sum(h[i]) - sum(h[i] * h[i+k]) <= C/dx
        # but we compute this by avoiding bilinear terms
        
        # Alternative direct approach: let's just loop the constraints in optimization
        # The optimizer handles this
        # But since we're using convex optimization explicitly, build with CVXPY
        
        # Actually for consistency with cvxpy approach:
        # We must relax bilinear terms: h[i] * h[i+k] for all valid k
        # Use McCormick envelopes: always introduce slacks
        
        # Instead, let's solve this through an iterative process:
        # We'll minimize C such that the system is feasible
        # For this, let's just recomputing and enforcing C as a constraint
        # But constraint can be added by penalizing when correlation exceeds C
        
        # Let's take a slightly different path:
        # The constraint max_k (int h(x)(1-h(x+k)) dx) <= C 
        # is handled as: for each k compute sum(h[i]*1 - h[i]*h[i+k]) <= C*dx
        
        # We'll write a specific approach to relax:
        # Include these cons in an optimization framework
        
        # But simpler: iterative steepest descent with projection back to feasible
        # Let's use scipy minimize directly with soft constraints or trust region method
        
        # Given constrained, just mathematically formulate proxy constraint
        # But we're treating this as a compact problem:
        
        return h, constraints
    
    # We'll implement a search where we first bin-search on the optimal C
    C_min = 0.0
    C_max = 0.5
    best_h = h.copy()
    best_C = compute_C5(h)
    
    # Search for better C₅
    for _ in range(20):  # binary search iterations
        if time.time() - start_time > budget_s - 5:  # Time buffer
            break
        C_mid = (C_min + C_max) / 2
        # Try to find h such that C₅ <= C_mid
        # We want to find the min of C₅ subject to constraints
        # Let's use a scaled approach:
        # The strategy is to define an MPF ( McCormick-based) constraint
        # We'll simulate solving:
        
        # For that, a proxy approach:
        # Denoting C₅ as performance measure, we want min C₅.
        # So we'll MIP-style solve:
        # Minimize C₅ s.t. the overlap constraints, h in [0,1]
        # Let's just constrain using a scaled problem
        
        # Alternative approach:
        # Solve using a smoothing/penalty approach to avoid bilinear issues
        # We can define a simple gradient-based solver approach and check result
        
        # We solve the hard version: instead add explicit smoothing penalty:
        # Let's solve using gradients/list of constraints
        
        # Let me take direct approach:
        # Since I can't easily set up CVXPY constraints with full correlation for huge form,
        # I will iterate by gradient-based optimization
        
        def objective(h_candidate):
            return compute_C5(h_candidate)
        
        # Constraint: sum(h) * dx = 1
        def constraint_sum(h_candidate):
            return np.sum(h_candidate) * dx - 1
        
        # Constraint: 0 <= h[i] <= 1
        # This already handled by bounds in optimizer
        
        # We'll solve a local optimization problem each iteration
        result = minimize(objective, h, method='SLSQP', constraints={'type': 'eq', 'fun': constraint_sum},
                             bounds=[(0, 1)] * n_points, options={'maxiter': 500, 'ftol':1e-6})
        
        if result.success:
            h = result.x
            current_C = compute_C5(h)
            if current_C < best_C:
                best_C = current_C
                best_h = h.copy()
            if current_C <= C_mid:
                C_max = C_mid
            else:
                C_min = C_mid
        else:
            # If not successful, continue
            C_min = C_mid

    # Final check after optimization
    final_C = compute_C5(best_h)
    
    return (best_h.tolist(), float(final_C), n_points)
