import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using scipy.signal.correlate with mode='full'
    # The integral is approximated by sum(h * (1 - h_shifted)) * dx
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def repair_h(h):
    # Ensure 0 <= h[i] <= 1 and sum(h) = n_points / 2
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = target_sum / current_sum
        h *= scale
    return h

def crossover_h(h1, h2):
    # Uniform crossover with blending
    n = len(h1)
    # Select crossover point
    cp = np.random.randint(1, n-1)
    # Combine parts
    child_h = np.concatenate([h1[:cp], h2[cp:]])
    return repair_h(child_h)

def mutate_h(h, rate=0.05):
    # Add small random noise
    n = len(h)
    mask = np.random.rand(n) < rate
    noise = np.random.normal(0, 0.05, n)
    h = h + noise * mask
    return repair_h(h)

def create_population(n_points, pop_size):
    # If initial_h_values exists, use it
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = repair_h(h0)
    else:
        h0 = np.full(n_points, 0.5)  # Simple initial guess

    population = [h0]
    for i in range(pop_size - 1):
        h = np.random.rand(n_points) * 0.5
        h = repair_h(h)
        population.append(h)
    return population

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 47  # Start with same resolution as initial
    dx = 2.0 / n_points
    pop_size = 70
    max_gen = 1000
    elite_size = max(1, pop_size // 5)
    mutation_rate = 0.03
    crossover_rate = 0.7
    start_time = time.time()
    
    # Initialize population
    population = create_population(n_points, pop_size)
    
    best_h = None
    best_c5 = np.inf
    no_improvement_count = 0
    
    for generation in range(max_gen):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_h(h)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 50:
            break
            
        # Selection: keep top elite
        elites = population[:elite_size]
        
        # Generate new population
        new_population = elites[:]
        
        # Crossover and mutate
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(elites) >= 2:
                p1, p2 = np.random.choice(elites, 2, replace=False)
                child = crossover_h(p1, p2)
            else:
                child = np.random.choice(elites).copy()
                child = mutate_h(child, mutation_rate)
                
            new_population.append(child)
            
        population = new_population

    if best_h is None:
        # Fallback: return a decent candidate
        best_h = np.ones(n_points) * 0.5
        best_h = repair_h(best_h)
        best_c5 = evaluate_h(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
