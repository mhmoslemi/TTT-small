import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h"""
    # Compute the overlap integral
    n = len(h)
    dx = 2.0 / n
    # Reflect h to compute 1-h
    h_reflect = 1 - h
    # Compute full cross-correlation
    correlation = np.correlate(h, h_reflect, mode="full")
    # The maximum value is what we want for C5
    c5 = np.max(correlation) * dx
    return c5

def objective(h, target=None):
    """Objective function to minimize: C5 value"""
    return evaluate_c5(h)

def constraint_eq(h):
    """Equality constraint: integral of h equals 1"""
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def constraint_lb(h):
    """Lower bound constraint for each h[i]"""
    return h

def constraint_ub(h):
    """Upper bound constraint for each h[i]"""
    return 1.0 - h

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved C5 bound"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Initialize with a simple piecewise constant function
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define constraints
    constraints = [
        {'type': 'eq', 'fun': constraint_eq},
    ]
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize with different algorithms
    current_h = h0.copy()
    best_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    
    # Method 1: L-BFGS with scipy
    try:
        result = minimize(
            objective, 
            current_h, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-5},
            callback=lambda x: None
        )
        
        if result.success and evaluate_c5(result.x) < best_c5:
            best_c5 = evaluate_c5(result.x)
            best_h = result.x.copy()
            
    except Exception as e:
        pass  # Continue to other methods if needed
    
    # Method 2: Random sampling with local refinement
    for i in range(50):  # Limited iterations for budget
        if time.time() - start_time > budget_s * 0.9:  # Leave some time for final steps
            break
            
        # Random perturbation
        perturbed_h = current_h + np.random.normal(0, 0.01, n_points)
        # Clip to bounds
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Renormalize
        perturbed_h = perturbed_h * (1.0 / np.sum(perturbed_h) * dx)
        
        # Simple constraint enforcement
        # Ensure sum equals 1
        perturbed_h = perturbed_h * (1.0 / (np.sum(perturbed_h) * dx)) * dx
        
        # Evaluate
        c5_new = evaluate_c5(perturbed_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = perturbed_h.copy()
            
        current_h = perturbed_h.copy()
    
    # Final refinement on best solution
    try:
        result = minimize(
            objective,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500}
        )
        
        if result.success:
            final_c5 = evaluate_c5(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()
    except Exception as e:
        pass  # If final optimization fails, continue with current best
    
    # Make sure the result meets constraints
    final_h = best_h.copy()
    
    return (final_h.tolist(), float(best_c5), n_points)
