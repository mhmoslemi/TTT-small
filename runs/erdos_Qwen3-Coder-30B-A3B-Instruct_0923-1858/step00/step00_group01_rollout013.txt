import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial guess: piecewise constant
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Perturbation loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Compute current C5
        current_c5 = max_corr * dx
        
        # Find the region that contributes to max correlation
        # The maximum occurs at shift max_k - (n_points - 1)
        k_shift = max_k - (n_points - 1)
        # We will work on indices that cause this max
        # For simplicity, just modify a subset of elements
        
        # Create a perturbation
        new_h = h.copy()
        # Identify the supporting interval
        # We'll modify elements that contribute to the correlation peak
        # by transferring small amounts
        
        # Select affected indices approximately
        range_start = max(0, k_shift)
        range_end = min(n_points, n_points + k_shift)
        
        # Try a random pair of indices to redistribute
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Calculate how much we can shift
        max_transfer = min(new_h[i], 1 - new_h[j])
        if max_transfer > 0:
            amount = np.random.uniform(0, max_transfer)
            new_h[i] -= amount
            new_h[j] += amount
            
            # Check feasibility
            if np.all((new_h >= 0) & (new_h <= 1)):
                # Check if the sum constraint is roughly maintained
                if abs(np.sum(new_h) * dx - 1.0) < 1e-3:
                    # Re-compute C5 for new_h
                    new_c5 = compute_c5(new_h, dx)
                    if new_c5 < current_c5:
                        h = new_h
                        no_improvement_count = 0
                        if new_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = new_c5
                    else:
                        no_improvement_count += 1
                else:
                    # Restore and try again
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1

        # Occasionally reinterpolate to higher resolution
        if iteration % 100 == 0:
            # Interpolate to double resolution
            n_double = 2 * n_points
            x_new = np.linspace(0, 2, n_double)
            x_old = np.linspace(0, 2, n_points)
            # Simple linear interpolation
            h_new = np.interp(x_new, x_old, h)
            # Project back to constraints
            h_new = np.clip(h_new, 0, 1)
            total = np.sum(h_new) * (2.0 / n_double)
            h_new = h_new * (1.0 / total) if total > 0 else h_new
            n_points = n_double
            dx = 2.0 / n_points
            h = h_new
            
            # Re-evaluate current best
            if compute_c5(h, dx) < best_c5:
                best_h = h.copy()
                best_c5 = compute_c5(h, dx)

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (list(best_h), final_c5, len(best_h))

def compute_c5(h, dx):
    """Compute the maximum overlap integral"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
