import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h):
    """Evaluate the C5 bound for a given h"""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation (multiplied by dx) gives C5
    c5 = np.max(corr) * dx
    return c5

def enforce_constraints(h):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    n_points = len(h)
    if h_sum != 0:
        h *= (n_points / 2) / h_sum
    return h

        return evaluate_h(h)

    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Perform optimization
    result = differential_evolution(
        objective, 
        bounds, 
        constraints=constraints, 
        seed=42, 
        maxiter=500,
        popsize=15,
        recombination=0.7,
        mutation=(0.5, 1.0)

    # If successful, return optimized h, otherwise return initial
    if result.success:
        h_opt = enforce_constraints(result.x)
