import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    # Compute the overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # Using full convolution with proper normalization
    conv = correlate(h, 1 - h, mode='full')
    # The full convolution gives us results for all lags from -(n-1) to (n-1)
    # But we want the maximum of positive lags only (k = 1 to n-1 in our context)
    # Actually, we interpret it as computing overlap for all shifts, which aligns
    # with the full convolution output ordering
    c5 = np.max(conv) * dx
    return c5

def project_h(h, n_points):
    # Ensure h[i] ∈ [0,1] and sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def nelder_mead_step_simplex(h, dx, n_points):
    # Simple Nelder-Mead with adaptive simplex size and projection
    # Start with a simple simplex - base point h and n perturbations of size 0.1
    n = len(h)
    simplex = np.tile(h, (n+1, 1))
    # Initialize simplex vertices by adding small random perturbations
    np.random.seed(123)  # Deterministic for reproducibility
    for i in range(1, n+1):
        # Add small noise with direction
        noise = np.random.randn(n) * 0.1
        simplex[i] = h + noise
    # Project back into feasible region
    for i in range(n+1):
        simplex[i] = project_h(simplex[i], n_points)
    
    # Evaluate all points
    f_vals = np.array([evaluate_overlap(s, dx) for s in simplex])
    
    # Nr of iterations
    max_iter = 2000
    iter_count = 0
    best_f = np.min(f_vals)
    best_h = simplex[np.argmin(f_vals)].copy()
    last_improvement = 0
    simplex_size = 0.1
    
    # Simplex coefficients
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink
    
    # Main loop
    while iter_count < max_iter:
        # Sort vertices by function value
        idx = np.argsort(f_vals)
        simplex = simplex[idx]
        f_vals = f_vals[idx]
        
        # Check if we improved
        if f_vals[0] < best_f:
            best_f = f_vals[0]
            best_h = simplex[0].copy()
            last_improvement = 0
        else:
            last_improvement += 1

        # Check for stagnation and reset simplex if needed
        if last_improvement > 50:
            simplex_size *= 0.5
            if simplex_size < 0.01:
                break  # halt if simplex too small
            # Recreate simplex around best point
            simplex = np.tile(best_h, (n+1, 1))
            for i in range(1, n+1):
                simplex[i] = best_h + np.random.randn(n) * simplex_size
            for i in range(n+1):
                simplex[i] = project_h(simplex[i], n_points)
            f_vals = np.array([evaluate_overlap(s, dx) for s in simplex])
            last_improvement = 0

        # Compute centroid excluding worst point
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_r = centroid + alpha * (centroid - simplex[-1])
        h_r = project_h(h_r, n_points)
        f_r = evaluate_overlap(h_r, dx)
        
        # Check if we improved by reflection
        if f_vals[0] <= f_r < f_vals[-2]:
            # Reflection accepted
            simplex[-1] = h_r
            f_vals[-1] = f_r
            iter_count += 1
            continue

        # If reflection is better than worst, but not better than the best, try expansion
        if f_r < f_vals[0]:
            h_e = centroid + gamma * (h_r - centroid)
            h_e = project_h(h_e, n_points)
            f_e = evaluate_overlap(h_e, dx)
            if f_e < f_r:
                simplex[-1] = h_e
                f_vals[-1] = f_e
            else:
                simplex[-1] = h_r
                f_vals[-1] = f_r
            iter_count += 1
            continue

        # If reflection was not good enough, do contraction
        h_c = centroid + beta * (simplex[-1] - centroid)
        h_c = project_h(h_c, n_points)
        f_c = evaluate_overlap(h_c, dx)
        if f_c < f_vals[-1]:
            simplex[-1] = h_c
            f_vals[-1] = f_c
            iter_count += 1
            continue

        # Shrink simplex towards best point
        for i in range(1, n+1):
            simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
            simplex[i] = project_h(simplex[i], n_points)
        f_vals = np.array([evaluate_overlap(s, dx) for s in simplex])
        iter_count += 1

    return best_h, best_f

        for i in range(n_points):
            if i < n_points / 2:
                h[i] += offset * (i - n_points/4) / (n_points/4)
            else:
                h[i] -= offset * (i - n_points*3/4) / (n_points/4)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / float(n_points)

    # Try Nelder-Mead
    best_h, min_c5 = nelder_mead_step_simplex(h, dx, n_points)
    best_h = project_h(best_h, n_points)
    min_c5 = evaluate_overlap(best_h, dx)

    # Check time budget to see if we can do more optimization
    elapsed = time.time() - start_time
    if elapsed < budget_s - 1 and min_c5 > 0.37:
        # Try a few more runs from random points to explore
        current_best_h = best_h.copy()
        current_best_c5 = min_c5
        for _ in range(5):
            if time.time() - start_time > budget_s - 2:
                break
            # Perturb best solution and start optimization
            perturbation = np.random.normal(0, 0.05, n_points)
            h_test = current_best_h + perturbation
            # Recompute
            h_test = project_h(h_test, n_points)
            test_c5 = evaluate_overlap(h_test, dx)
            if test_c5 < current_best_c5:
                current_best_c5 = test_c5
                current_best_h = h_test
        best_h = current_best_h
        min_c5 = current_best_c5
