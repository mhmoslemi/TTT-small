import numpy as np
from scipy import optimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Function to compute C5
    def compute_c5(h_vals):
        h_vals = np.array(h_vals)
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_val = np.max(conv)
        return max_val * dx
    
    # Function to project h to feasible region
    def project_h(h_vals):
        h_proj = np.clip(h_vals, 0, 1)
        actual_sum = np.sum(h_proj)
        if actual_sum > 0:
            h_proj = h_proj * target_sum / actual_sum
        return h_proj
    
    # Function to perturb h
    def perturb_h(h_vals, sigma):
        noise = np.random.normal(0, sigma, len(h_vals))
        h_new = h_vals + noise
        return project_h(h_new)
    
    # Annealing parameters
    T0 = 0.1
    alpha = 0.99
    sigma = 0.1
    T = T0
    
    # Tracking best solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Time tracking
    start_time = time.time()
    
    # Iteration parameters
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Perturb
        h_candidate = perturb_h(h, sigma)
        
        # Compute new C5
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            h = h_candidate.copy()
            best_c5 = c5_candidate
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Accept with probability
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()
                best_c5 = c5_candidate
                no_improvement_count = 0
        
        # Update temperature
        T *= alpha
        
        # Adapt sigma
        if iter_count % 100 == 0:
            sigma *= 0.9
        
        # Termination check
        if no_improvement_count > max_no_improvement:
            break
            
        # Occasionally print status
        if iter_count % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {iter_count}, Time {elapsed:.1f}s, Best C5 {best_c5:.6f}")
    
    # Return final solution
    return (best_h.tolist(), best_c5, n_points)
