import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define problem parameters
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create an initial triangular waveform
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 2.0 * i / (n_points - 1)
            h[n_points - 1 - i] = h[i]
        h[mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute C5
    def compute_C5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    trust_radius = 0.1
    max_iter = 5000
    iter_count = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Gradient estimation using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Rescale to meet sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute gradient approximation
            C5_plus = compute_C5(h_plus)
            C5_minus = compute_C5(h_minus)
            grad[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Trust region step
        step = -grad * trust_radius
        h_new = h + step
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_C5 = compute_C5(h_new)
        
        # Accept the step if it improves the objective
        if new_C5 <= current_C5:
            h = h_new
            # Increase trust radius
            trust_radius = min(0.5, trust_radius * 1.2)
        else:
            # Decrease trust radius
            trust_radius *= 0.5
            # Perturb if trust radius becomes too small
            if trust_radius < 1e-6:
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                trust_radius = 0.1
    
    # Final check
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)
