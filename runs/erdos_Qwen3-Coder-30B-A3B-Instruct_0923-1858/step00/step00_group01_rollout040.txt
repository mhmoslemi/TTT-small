import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    """Evaluate the overlap integral C_5 for given h"""
    # Compute the correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Return the maximum value scaled by dx
    return np.max(corr) * dx

def repair_h(h, dx):
    """Ensure h satisfies constraints: 0<=h<=1 and sum(h)*dx=1"""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to make sum(h) = 1/dx (which means sum(h)*dx = 1)
    target_sum = 1.0 / dx
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clamp again after rescaling
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2, dx):
    """Create a child from two parents via uniform crossover"""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child = np.copy(parent1)
    child[crossover_point:] = parent2[crossover_point:]
    # Repair to satisfy constraints
    child = repair_h(child, dx)
    return child

def mutate(h, dx, mutation_rate=0.05):
    """Mutate h with small random changes"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    # Repair
    h = repair_h(h, dx)
    return h

def initialize_population(n_points, pop_size, dx):
    """Initialize population with repaired individuals"""
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values)
        initial = repair_h(initial, dx)
        population = [initial]
    else:
        population = []
    # Fill with random individuals
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, dx)
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    pop_size = 70
    generations = 100
    elite_ratio = 0.2
    mutation_rate = 0.05
    
    # Initialize population
    population = initialize_population(n_points, pop_size, dx)
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate fitness for everyone
        fitness = []
        for h in population:
            c5 = evaluate(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sort population by fitness (ascending for minimization)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elites
        elite_count = max(1, int(elite_ratio * pop_size))
        elite = population[:elite_count]
        
        # Generate offspring
        new_population = elite[:]
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s - 2:
                break
            # Select two parents
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            # Crossover
            child = crossover(parent1, parent2, dx)
            # Mutate
            child = mutate(child, dx, mutation_rate)
            new_population.append(child)
            
        population = new_population
    
    # Final evaluation of best
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h, final_c5, n_points)
    else:
        # Fallback in case no improvement found
        final_h = population[0] if population else np.ones(n_points)*0.5
        final_h = repair_h(final_h, dx)
        final_c5 = evaluate(final_h, dx)
        return (final_h, final_c5, n_points)
