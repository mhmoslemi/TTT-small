import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[10:20] = 1.0
        h[50:60] = 0.0
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to sum constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    last_improvement_time = start_time
    
    while time.time() - start_time < budget_s * 0.95:
        # Compute all overlaps for all shifts
        h_rev = np.flip(h)
        overlaps = correlate(h, h_rev, mode="full")
        overlaps = overlaps[:n_points]  # Only positive shifts
        overlaps = overlaps * dx  # Scale by dx
        overlaps[0] = 0  # Shift 0 is irrelevant for our C5 definition
        
        k_star = np.argmax(overlaps)
        max_overlap = overlaps[k_star]
        
        if max_overlap <= best_c5 * 0.999999:  # Significant improvement?
            last_improvement_time = time.time()
        
        # If no improvement recently, break early
        if time.time() - last_improvement_time > 30.0:
            break
        
        # Find indices that contribute to maximum overlap
        max_positions = []
        for i in range(n_points - k_star):
            if h[i] > 0 and h[i + k_star] < 1:
                max_positions.append((i, i + k_star))
        
        if not max_positions:
            continue
            
        # Sort by impact (product of values)
        max_positions.sort(key=lambda x: h[x[0]] * (1 - h[x[1]]), reverse=True)
        
        # Take a few top candidates for adjustment
        adjustments = max_positions[:min(10, len(max_positions))]
        
        # Make small adjustments
        for i, j in adjustments:
            if i >= n_points or j >= n_points:
                continue
                
            # Compute influence (how much this contributes)
            influence = h[i] * (1 - h[j])
            if influence < 1e-8:
                continue
                
            # Adjust to reduce impact
            step_size = min(0.02, influence * 0.5)
            
            if h[i] > 0 and h[j] < 1:
                # Reduce both values to decrease overlap
                h[i] = max(0, h[i] - step_size)
                h[j] = min(1, h[j] + step_size)
            
            # Adjust the sum constraint
            current_sum = np.sum(h)
            diff = current_sum - target_sum
            if abs(diff) > 1e-6:
                # Distribute difference
                h -= diff / n_points
                
                # Clamp and re-normalize if needed
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
        
        # Evaluate new objective
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (list(best_h), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    n = len(h)
    # Compute all overlaps
    h_rev = np.flip(h)
    overlaps = correlate(h, h_rev, mode="full")
    overlaps = overlaps[:n]  # Only consider shifts 0 to n-1
    overlaps = overlaps * dx
    # Shift 0 doesn't contribute to overlap
    overlaps[0] = 0
    return np.max(overlaps)
