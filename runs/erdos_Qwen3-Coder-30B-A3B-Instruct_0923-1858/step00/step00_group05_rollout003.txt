import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    corr = np.correlate(h, 1-h, mode="full")
    # Extract the relevant part (lags from -n+1 to n-1)
    max_corr = np.max(corr)
    return max_corr * dx

        # We maximize the overlap integral

        corr = np.correlate(h, 1-h, mode="full")
        return -np.max(corr) * dx
    
    def constraint_eq(h):
        return np.sum(h) - n_points / 2

        return h

        return 1 - h

    cons = [

        {'type': 'ineq', 'fun': constraint_lb},
        {'type': 'ineq', 'fun': constraint_ub}
    ]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-6},
                callback=None
            )

                h_opt = np.clip(h_opt, 0, 1)
