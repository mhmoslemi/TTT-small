import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create symmetric initial construction
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        h[:mid] = 0.0
        h[mid:] = 1.0
    
    dx = 2.0 / n_points
    max_iter = 100000
    start_time = time.time()
    
    # Annealing parameters
    T0 = 0.1
    alpha = 0.99
    T = T0
    sigma = 0.1
    sigma_decay = 0.9
    no_improvement_count = 0
    
    # Initial evaluation
    h_proj = h.copy()
    h_proj = np.clip(h_proj, 0, 1)
    h_proj *= (n_points / 2) / np.sum(h_proj)
    C5_current = np.max(signal.correlate(h_proj, 1-h_proj, mode='full')) * dx
    best_C5 = C5_current
    best_h = h_proj.copy()
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s:
        # Generate perturbation
        delta = np.random.normal(0, sigma, n_points)
        
        # Apply perturbation
        h_candidate = h_proj + delta
        
        # Project onto feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        C5_candidate = np.max(signal.correlate(h_candidate, 1-h_candidate, mode='full')) * dx
        
        # Accept or reject
        if C5_candidate < C5_current:
            h_proj = h_candidate
            C5_current = C5_candidate
            if C5_candidate < best_C5:
                best_C5 = C5_candidate
                best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            # Accept with probability
            delta_C = C5_candidate - C5_current
            if np.random.rand() < np.exp(-delta_C / T):
                h_proj = h_candidate
                C5_current = C5_candidate
            no_improvement_count += 1
            
        # Update annealing parameters
        T *= alpha
        if iter_count % 100 == 0:
            sigma *= sigma_decay
            
        iter_count += 1
        
        if no_improvement_count > 1000:
            break
            
    return (best_h, best_C5, n_points)
