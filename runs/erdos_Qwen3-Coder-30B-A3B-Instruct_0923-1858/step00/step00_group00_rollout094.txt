import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    except:
        # Create symmetric initial guess
        n_points = 45
        h0 = np.full(n_points, 1.0 / n_points * 2.0)  # sum = 2.0, will be rescaled
        h0[0] = 0.0
        h0[-1] = 0.0
        h0[1:-1] = np.linspace(0.0, 1.0, len(h0[1:-1]))
        h0 = h0 / np.sum(h0) * (n_points / 2.0)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Initialize current best
    h_current = h0.copy()
    current_C5 = max(correlate(h_current, 1 - h_current, mode="full")) * dx
    best_h = h_current.copy()
    best_C5 = current_C5
    
    # Annealing parameters
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 1000
    no_improve_count = 0
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Perturb
        h_new = h_current + np.random.normal(0, perturb_scale, n_points)
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate new solution
        new_C5 = max(correlate(h_new, 1 - h_new, mode="full")) * dx
        
        # Accept or reject
        if new_C5 < current_C5 or np.random.random() < np.exp(-(new_C5 - current_C5) / T):
            h_current = h_new
            current_C5 = new_C5
            
            # Update best if improved
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cooling schedule
        T *= alpha
        if iter_count % 100 == 0:
            perturb_scale *= 0.99
            
        iter_count += 1
        if no_improve_count >= no_improve_limit:
            break
            
    return (best_h, best_C5, n_points)
