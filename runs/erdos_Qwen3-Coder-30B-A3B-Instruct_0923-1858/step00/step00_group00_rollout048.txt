import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    # Compute the correlation: h(x) * (1 - h(x+k))
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is over all shifts k from -n+1 to n-1
    # We take the maximum over the valid indices (excluding padding)
    max_k = len(corr) - n + 1
    c5 = np.max(corr[n-1:n-1+max_k]) * dx
    return c5

def project_to_feasible(h, dx):
    """Project h to a feasible solution: 0 <= h <= 1, sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * (1.0 / dx)  # scale to sum = 1/dx => sum(h)*dx = 1
    return h

def multi_start_optimize(initial_h, dx, budget_s):
    """Main optimization loop using a hybrid approach"""
    start_time = time.time()
    
    # Parameters for the algorithm
    n_points = len(initial_h)
    max_generations = 1000
    pop_size = 50
    elite_ratio = 0.2
    mutation_rate = 0.05
    crossover_alpha = 0.5
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, dx)
        population.append(h.copy())
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    no_improvement_count = 0
    
    # GA loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_c5(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Update best
        if fitness[sorted_indices[0]] < best_c5:
            best_c5 = fitness[sorted_indices[0]]
            best_h = population[sorted_indices[0]].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 50:
            break
            
        # Select elites
        elite_count = int(elite_ratio * pop_size)
        elites = [population[i] for i in sorted_indices[:elite_count]]
        
        # Generate new population
        new_population = elites.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, elite_count)]
            parent2_idx = sorted_indices[np.random.randint(0, elite_count)]
            
            # Crossover
            child = crossover_alpha * population[parent1_idx] + (1 - crossover_alpha) * population[parent2_idx]
            # Add mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            child = project_to_feasible(child, dx)
            new_population.append(child)
            
        population = new_population
        
        # Adaptive mutation rate
        if gen % 10 == 0:
            mutation_rate *= 0.99
    
    # Gradient-based refinement of best solutions
    refining_solutions = [best_h.copy()]
    for i in range(2):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, dx)
        refining_solutions.append(h.copy())
    
    for h in refining_solutions:
        if time.time() - start_time > budget_s - 1:
            break
        # Gradient descent with projection
        for iter in range(150):
            if time.time() - start_time > budget_s - 1:
                break
            
            current_c5 = evaluate_c5(h, dx)
            # Finite difference gradient estimation
            eps = 1e-5
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, dx)
                c5_plus = evaluate_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, dx)
                c5_minus = evaluate_c5(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
            # Update step with adaptive learning rate
            eta = 0.01 / (1.0 + iter/100)
            h_new = h - eta * grad
            h = project_to_feasible(h_new, dx)
            
            # Break early if no improvement
            new_c5 = evaluate_c5(h, dx)
            if new_c5 > current_c5:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return best_h, best_c5, n_points
