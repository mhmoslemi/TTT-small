import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    # The overlap integral C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using correlation: correlate(h, 1-h, mode="full") gives the integral values
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h  # flip the function
    # Compute the full correlation (equivalent to overlap integrals for all k)
    corr = correlate(h, h1, mode="full")
    # Since correlation is symmetric and we want maximum overlap,
    # we take the maximum value (divided by dx for normalization)
    c5 = np.max(corr) * dx
    return c5

def objective(h):
    """Objective function to minimize (negative because we minimize C5)."""
    return -evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Fallback to a simple construction
        n_points = 45
        h_start = np.ones(n_points) * 0.5
        h_start[20:30] = 1.0  # Add some structure
    
    # Define bounds for the optimization (h[i] must be between 0 and 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # But we also need to ensure the constraint that integral equals 1
    # We will handle that with a custom constraint in optimization

    def constraint(h):
        """Ensure that ∫₀² h(x) dx = 1"""
        dx = 2.0 / len(h)
        return np.sum(h) * dx - 1.0

    # Use scipy's differential evolution for global optimization
    # It's suitable for expensive black-box problems like this
    start_time = time.time()
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        seed=seed,
        maxiter=int(5000),  # reasonable limit
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        tol=1e-6,
        atol=1e-6,
        callback=lambda x, convergence: time.time() - start_time > budget_s - 1,
    )

    # Recover the best solution
    h_opt = result.x
    
    # Enforce that sum(h) * dx = 1 exactly by rescaling
    dx = 2.0 / n_points
    scale_factor = 1.0 / (np.sum(h_opt) * dx)
    h_opt = h_opt * scale_factor
    
    # Ensure all values are within [0, 1]
    h_opt = np.clip(h_opt, 0.0, 1.0)
    
    # Final check: ensure constraint is satisfied exactly
    final_integral = np.sum(h_opt) * dx
    h_opt = h_opt / final_integral  # Normalize to correct integral
    
    # Evaluate the final bound
    c5_bound = evaluate_c5(h_opt)
    
    # Clamp to valid range
    h_opt = np.clip(h_opt, 0.0, 1.0)
    
    return (h_opt, c5_bound, n_points)
