import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular waveform
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    trust_radius = 0.1
    max_iterations = 10000
    iteration = 0
    
    def evaluate_c5(h_vals, dx_val):
        # Compute full correlation with 1-h
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Only take the relevant part of correlation
        mid = len(corr) // 2
        # C5 is the maximum correlation value (excluding zero-lag which should be peak)
        max_corr = -1
        for k in range(len(corr)):
            if k != mid:
                max_corr = max(max_corr, corr[k] * dx_val)
        return max_corr
    
    def project_h_to_feasible(h_vals):
        # Clip to [0,1]
        h_new = np.clip(h_vals, 0, 1)
        # Rescale to sum to n_points/2
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (n_points / 2) / current_sum
        return h_new
    
    def compute_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        baseline = evaluate_c5(h_vals, dx)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_h_to_feasible(h_plus)
            h_minus = project_h_to_feasible(h_minus)
            
            grad[i] = (evaluate_c5(h_plus, dx) - evaluate_c5(h_minus, dx)) / (2 * eps)
        return grad

    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        grad = compute_gradient(h)
        
        # Trust region step
        step = -grad * trust_radius
        h_trial = h + step
        h_trial = project_h_to_feasible(h_trial)
        
        trial_c5 = evaluate_c5(h_trial, dx)
        current_c5 = evaluate_c5(h, dx)
        
        if trial_c5 < current_c5:
            h = h_trial
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = h.copy()
            trust_radius = min(0.5, trust_radius * 1.5)  # Increase trust radius on improvement
        else:
            trust_radius = trust_radius * 0.5  # Decrease trust radius on no improvement
            
            # Random perturbation
            if iteration % 100 == 0:
                noise = np.random.uniform(-0.1, 0.1, n_points)
                # Clip and rescale
                h += noise
                h = project_h_to_feasible(h)
                
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
