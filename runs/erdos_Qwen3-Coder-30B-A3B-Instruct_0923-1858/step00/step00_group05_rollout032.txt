import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_low = 50
    n_high = 70
    max_iters = 1000
    tol = 1e-5
    lr_low = 1e-2
    lr_high = 1e-3
    
    # Initialize h_low from initial_h_values or create fresh start
    if 'initial_h_values' in globals():
        h_low = np.array(initial_h_values)
        h_low = h_low[:n_low]
        h_low = h_low / np.sum(h_low) * (n_low / 2)
    else:
        h_low = np.full(n_low, 0.5)
        noise = np.random.normal(0, 0.02, n_low)
        h_low += noise
        h_low = np.clip(h_low, 0, 1)
        h_low = h_low / np.sum(h_low) * (n_low / 2)
    
    # Optimization at low resolution
    def compute_C5(h):
        h_pad = np.pad(h, (0, len(h) - 1), 'constant')
        conv = signal.convolve(h_pad, 1 - h, mode='valid')
        return np.max(conv) * (2.0 / len(h))
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_low / 2) / h_sum
        return h
    
    # Gradient descent for low-resolution
    last_improvement_iter = 0
    last_C5 = compute_C5(h_low)
    lr = lr_low
    
    for i in range(max_iters):
        if time.time() - start_time > budget_s * 0.65:
            break
            
        # Forward pass
        C5_current = compute_C5(h_low)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_low)
        for j in range(len(h_low)):
            h_plus = h_low.copy()
            h_minus = h_low.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus)
            h_minus = project_h(h_minus)
            grad[j] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
        
        # Update step
        h_low = h_low - lr * grad
        h_low = project_h(h_low)
        
        # Adaptive learning rate
        if C5_current < last_C5 - tol:
            last_C5 = C5_current
            last_improvement_iter = i
        else:
            if i - last_improvement_iter > 10:
                lr *= 0.9
                last_improvement_iter = i
        
        if C5_current < last_C5 - tol:
            last_C5 = C5_current
            
    # Upsample
    h_high = np.interp(np.linspace(0, 2, n_high), 
                       np.linspace(0, 2, n_low), 
                       h_low)
    h_high = h_high / np.sum(h_high) * (n_high / 2)
    
    # High resolution optimization
    def compute_C5_high(h):
        h_pad = np.pad(h, (0, len(h) - 1), 'constant')
        conv = signal.convolve(h_pad, 1 - h, mode='valid')
        return np.max(conv) * (2.0 / len(h))
    
    def project_h_high(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_high / 2) / h_sum
        return h
    
    lr = lr_high
    last_C5 = compute_C5_high(h_high)
    last_improvement_iter = 0
    
    for i in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        C5_current = compute_C5_high(h_high)
        eps = 1e-5
        grad = np.zeros_like(h_high)
        for j in range(len(h_high)):
            h_plus = h_high.copy()
            h_minus = h_high.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h_high(h_plus)
            h_minus = project_h_high(h_minus)
            grad[j] = (compute_C5_high(h_plus) - compute_C5_high(h_minus)) / (2 * eps)
            
        h_high = h_high - lr * grad
        h_high = project_h_high(h_high)
        
        if C5_current < last_C5 - tol:
            last_C5 = C5_current
            last_improvement_iter = i
        else:
            if i - last_improvement_iter > 10:
                lr *= 0.9
                last_improvement_iter = i
    
    # Final check and return
    final_C5 = compute_C5_high(h_high)
    return (h_high.tolist(), final_C5, n_high)
