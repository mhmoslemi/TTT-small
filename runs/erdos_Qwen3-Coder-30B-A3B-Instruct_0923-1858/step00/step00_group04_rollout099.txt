import numpy as np
import time
from scipy import signal

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = signal.correlate(h, 1 - h, mode="full")
    return np.max(np.abs(corr)) * dx

def project_onto_feasible(h, target_sum):
    """Project h onto feasible region: 0 <= h <= 1, sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def init_population(n_points, pop_size, initial_h):
    """Initialize a population with a given initial solution and random members."""
    population = [initial_h.copy()]
    target_sum = n_points / 2.0
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_onto_feasible(h, target_sum)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents."""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    """Mutate a solution with Gaussian noise."""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + noise * mask
    return h
