import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for given h with proper discretization."""
    n = len(h)
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from -n+1 to n-1)
    # The maximum is for positive shifts only due to symmetry of problem
    max_corr = max(corr[n-1:])  # Part from shift 0 to n-1 (positive shifts)
    return max_corr * dx

def project_onto_constraints(h, dx, target_sum):
    """Project h onto the constraint sum(h)*dx == target_sum"""
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-12:
        return h
    scale = target_sum / current_sum
    return h * scale

def compute_gradient(h, dx, k_max):
    """Compute approximate gradient based on finite differences."""
    n = len(h)
    grad = np.zeros_like(h)
    eps = 1e-5
    base_c5 = evaluate_erdos_solution(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + eps)
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - eps)
        c5_plus = evaluate_erdos_solution(h_plus, dx)
        c5_minus = evaluate_erdos_solution(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define problem parameters
    n_points = 47  # Based on initial value provided
    
    # Initial construction based on proposal
    dx = 2.0 / n_points
    m = floor(np.sqrt(n_points)) + 1  # Select modular base
    
    h = np.zeros(n_points)
    # Create a sparse pattern with modular positions
    positions = []
    for k in range(m):
        for i in range(n_points):
            if i % m == k:
                positions.append(i)
    # Fill with a step height that satisfies sum(h)*dx = 1
    num_positions = len(positions)
    step_height = 1.0 / (num_positions * dx) if num_positions > 0 else 0.0
    step_height = min(1.0, max(0.0, step_height))
    for idx in positions:
        h[idx] = step_height
    
    # Project to satisfy integral constraint
    h = project_onto_constraints(h, dx, 1.0)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    current_time = time.time()
    
    # Simple gradient descent with backtracking line search
    max_iter = 5000
    alpha = 0.005
    decay = 0.9
    tolerance = 1e-6
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if current_time - start_time > budget_s - 0.1:
            break
            
        # Compute gradient (approximate)
        grad = compute_gradient(h, dx, n_points)
        
        # Apply gradient step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to integral constraint
        h_new = project_onto_constraints(h_new, dx, 1.0)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # If improved, accept and continue
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new.copy()
        prev_c5 = new_c5
        
        current_time = time.time()
        if abs(prev_c5 - new_c5) < tolerance:
            break
        if (i + 1) % 200 == 0:
            # Reduce learning rate
            alpha *= decay
    
    return (best_h.tolist(), best_c5, n_points)
