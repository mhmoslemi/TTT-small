import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.mean(h)
    
    # Hyperparameters
    learning_rate = 0.01
    perturbation_rate = 0.1
    max_iter = 10000
    start_time = time.time()
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        correlations = correlate(h, 1 - h, mode='full')
        C5_candidate = np.max(np.abs(correlations)) * dx
        
        if C5_candidate < best_C5:
            best_h = h.copy()
            best_C5 = C5_candidate
            
        # Compute gradient approximately via finite differences
        gradient = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.mean(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            C5_plus = np.max(np.abs(corr_plus)) * dx
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.mean(h_minus)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            C5_minus = np.max(np.abs(corr_minus)) * dx
            
            gradient[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Take step
        h = h - learning_rate * gradient
        
        # Project onto constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.mean(h)
        
        # Occasionally perturb
        if iteration % 100 == 0:
            perturb = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.mean(h)
            
        # Adaptive learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.9
            
    # Final validations and re-evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)
