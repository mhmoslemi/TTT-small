import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 100000
    initial_temp = 0.1
    anneal_factor = 0.99
    sigma_decay_interval = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Symmetric initialization
        h = np.ones(n_points) * 0.5
        h[0] = 0.2
        h[-1] = 0.2
    
    # Current best
    h_best = h.copy()
    h_current = h.copy()
    
    # Compute initial C5
    corr = signal.correlate(h_current, 1 - h_current, mode='full')
    C5_current = np.max(corr) * dx
    C5_best = C5_current
    
    # Annealing parameters
    T = initial_temp
    sigma = 0.1
    
    # Timer
    start_time = time.time()
    
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb
        delta_h = np.random.normal(0, sigma, n_points)
        h_candidate = h_current + delta_h
        
        # Project onto feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        corr = signal.correlate(h_candidate, 1 - h_candidate, mode='full')
        C5_candidate = np.max(corr) * dx
        
        # Accept or reject
        if C5_candidate < C5_current:
            h_current = h_candidate
            C5_current = C5_candidate
            if C5_candidate < C5_best:
                h_best = h_candidate.copy()
                C5_best = C5_candidate
                last_improvement = iteration
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(C5_candidate - C5_current) / T):
                h_current = h_candidate
                C5_current = C5_candidate
        
        # Update temperature and sigma
        if iteration % sigma_decay_interval == 0:
            sigma *= 0.9
        T *= anneal_factor
        
        # Check for timeout or stagnation
        if time.time() - start_time > budget_s - 1:
            break
        if iteration - last_improvement > 1000:
            break
    
    return (h_best.tolist(), C5_best, n_points)
