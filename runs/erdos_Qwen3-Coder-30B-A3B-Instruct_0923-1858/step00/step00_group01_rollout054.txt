import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h = h * scale
    return h

def evaluate_C5(h, dx):
    h_flip = 1 - h
    correlation = np.correlate(h, h_flip, mode='full')
    max_correlation = np.max(correlation)
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 47
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    h = clip_and_rescale(h, dx)
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Optimization loop
    iteration = 0
    max_iter = 1000
    learning_rate = 0.01
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute gradient estimation
        epsilon = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = clip_and_rescale(h_plus, dx)
            C5_plus = evaluate_C5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = clip_and_rescale(h_minus, dx)
            C5_minus = evaluate_C5(h_minus, dx)
            
            grad[i] = (C5_plus - C5_minus) / (2 * epsilon)
        
        # Update h
        h = h - learning_rate * grad
        h = clip_and_rescale(h, dx)
        
        # Evaluate solution
        C5 = evaluate_C5(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.1
        
        iteration += 1
        
        # Increase resolution if needed
        if iteration % 300 == 0 and n_points < 1000:
            n_points *= 2
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            h = clip_and_rescale(h, dx)
            best_h = h.copy()
            best_C5 = evaluate_C5(h, dx)
    
    return (best_h.tolist(), best_C5, n_points)
