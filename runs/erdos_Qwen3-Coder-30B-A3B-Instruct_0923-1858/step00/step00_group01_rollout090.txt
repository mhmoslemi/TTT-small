import numpy as np
from scipy import optimize
import time
import random

def evaluate_h(h):
    """Evaluate the overlap integral C_5 for given h"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum != 0:
        h = h * target_sum / current_sum
    return h

def crossover_h(h1, h2):
    """Single-point crossover between two h vectors"""
    n = len(h1)
    if n <= 1:
        return h1.copy()
    crossover_point = random.randint(1, n - 1)
    child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return child

def mutate_h(h, mutation_rate=0.05):
    """Mutate h with given rate"""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] = random.uniform(0, 1)
    return h

def evolve_population(population, fitnesses, elite_ratio=0.2, crossover_rate=0.5, mutation_rate=0.05):
    """Evolve a new population from current one"""
    n = len(population)
    elite_count = int(n * elite_ratio)
    sorted_indices = np.argsort(fitnesses)
    elite = [population[i] for i in sorted_indices[:elite_count]]
    new_population = elite[:]
    
    # Generate children via crossover and mutation
    while len(new_population) < n:
        if random.random() < crossover_rate and len(elite) >= 2:
            parent1, parent2 = random.sample(elite, 2)
            child = crossover_h(parent1, parent2)
        else:
            parent = random.choice(elite)
            child = parent.copy()
        child = mutate_h(child, mutation_rate)
        child = repair_h(child, len(child))
        new_population.append(child)
    
    return new_population[:n]

def fitness_function(h):
    """Fitness function to minimize"""
    return evaluate_h(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(47)
    
    # Set discretization parameters
    n_points = 47
    dx = 2.0 / n_points
    
    # Make sure initial solution is feasible
    initial_h = repair_h(initial_h, n_points)
    
    # Population-based optimization
    population_size = 50
    population = [initial_h.copy()]
    
    # Generate additional random starting candidates
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_h = initial_h.copy()
    best_fitness = evaluate_h(best_h)
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate fitness for all population
        fitnesses = [fitness_function(h) for h in population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        current_best = population[min_idx]
        current_fitness = fitnesses[min_idx]
        
        if current_fitness < best_fitness:
            best_h = current_best.copy()
            best_fitness = current_fitness
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Evolve population
        population = evolve_population(population, fitnesses)
        
        generation += 1
        
    # Final repair and evaluation
    best_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
