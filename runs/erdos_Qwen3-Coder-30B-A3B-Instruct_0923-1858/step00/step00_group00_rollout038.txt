import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise construct symmetric one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        # Construct a more structured initial point
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    def C5_func(h):
        h = np.array(h)
        # compute overlap integral for all shifts
        conv = np.correlate(h, 1 - h, mode='full')
        # origin is at index n_points - 1
        conv = conv[n_points-1:]  # only positive shifts
        return np.max(conv) * dx
    
    # Constraint: ∫ h(x) dx = 1 
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # constraint functions
    def constraint_bounds(h):
        return np.concatenate([h, 1-h])
    
    # binary search for the minimum C5
    C_min, C_max = 0.0, 0.5
    best_C = C_max
    best_h = h0
    
    # Some bounds checks
    while C_max - C_min > 1e-4 and time.time() - start_time < budget_s * 0.95:
        C_guess = (C_min + C_max) / 2.0
        
        # Flatten problem and solve using scipy minimize
        def objective(h):
            return C5_func(h)
            
        # Define constraints
        cons = [{'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': lambda h: 1 - h},  # h <= 1
                {'type': 'ineq', 'fun': lambda h: h}]     # h >= 0
        # Add max_overlap constraint using penalty
        def max_constraint(h):
            return C_guess - C5_func(h)
        cons.append({'type': 'ineq', 'fun': max_constraint})
        
        # Bound for bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # solve with different starting points
        try:
            # randomness in initial point
            h_start = h0.copy()
            h_start += np.random.normal(0, 0.01, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
            result = minimize(objective, h_start, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100}, 
                              callback=lambda x: None)
            
            if result.success and max_constraint(result.x) >= -1e-4:
                best_C = C_guess
                best_h = result.x
                C_max = C_guess
            else:
                C_min = C_guess
        except:
            C_min = C_guess
            
    # Final validation check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final compute C5
    c5_bound = C5_func(best_h)
    
    return (best_h.tolist(), c5_bound, n_points)
