import numpy as np
from time import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Simple construction as fallback
        n_points = 70
        h = np.ones(n_points, dtype=float) * 0.5

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to meet sum constraint
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    
    start_time = time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute overlap for all shifts
        # Correlate h with (1-h)
        convolved = correlate(h, 1-h, mode="full")
        # Extract only valid shifts (0 to n_points-1)
        shifts = np.arange(0, n_points)
        overlap_values = convolved[len(h)-1:] * dx
        max_overlap_idx = np.argmax(overlap_values)
        max_overlap = overlap_values[max_overlap_idx]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = 0
        
        # Stop if no improvement in a while
        last_improvement += 1
        if last_improvement > 50:
            break
            
        # Pick the shift that contributes most
        k_star = max_overlap_idx
        if k_star >= n_points:
            continue
            
        # Find the best adjustment: element pair (i, i+k_star)
        best_delta = 0.0
        best_i = -1
        best_update = (0, 0)
        
        # Check all possible adjustment pairs
        for i in range(n_points - k_star):
            # How much can we reduce overlap at k_star by modifying h[i] and h[i+k_star]?
            # Current contribution: h[i] * (1 - h[i+k_star])
            # Let's make a small change
            delta_i = 0.01
            # Try adjusting h[i] up and h[i+k_star] down
            test_h = h.copy()
            test_h[i] = min(1, test_h[i] + delta_i)
            test_h[i+k_star] = max(0, test_h[i+k_star] - delta_i)
            
            # Rebalance to keep sum
            old_sum = np.sum(h)
            new_sum = np.sum(test_h)
            diff = target_sum - new_sum
            if abs(diff) > 1e-10:
                # Simple redistribution
                total_diff = diff
                h_adj = test_h.copy()
                for j in range(n_points):
                    if abs(total_diff) < 1e-12:
                        break
                    change = min(total_diff, 1 - h_adj[j]) if total_diff > 0 else max(total_diff, -h_adj[j])
                    h_adj[j] += change
                    total_diff -= change
                test_h = h_adj
            
            convolved_test = correlate(test_h, 1-test_h, mode="full")
            overlap_test = convolved_test[len(test_h)-1:] * dx
            new_max_overlap = np.max(overlap_test)
            
            improvement = max_overlap - new_max_overlap
            if improvement > best_delta:
                best_delta = improvement
                best_i = i
                best_update = (delta_i, -delta_i)
                
        # Apply the best update
        if best_i >= 0 and best_delta > 1e-6:
            h[best_i] = min(1, h[best_i] + best_update[0])  
            h[best_i + k_star] = max(0, h[best_i + k_star] + best_update[1])
            
            # Adjust to maintain the target sum
            current_sum = np.sum(h)
            diff = current_sum - target_sum
            if abs(diff) > 1e-10:
                # Redistribute diff evenly
                change_per_element = diff / n_points
                h = np.clip(h - change_per_element, 0, 1)
                
                # Adjust to hit target sum exactly
                final_sum = np.sum(h)
                diff = target_sum - final_sum
                if abs(diff) > 1e-10 and n_points > 0:
                    h[0] += diff
                    
    # Final evaluation
    convolved = correlate(best_h, 1-best_h, mode="full")
    overlap_values = convolved[len(best_h)-1:] * dx
    final_c5 = np.max(overlap_values)
    
    return (best_h.tolist(), final_c5, n_points)
