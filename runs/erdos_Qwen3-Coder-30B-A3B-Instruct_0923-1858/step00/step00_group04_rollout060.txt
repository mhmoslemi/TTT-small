import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # Find the maximum value and scale by dx
    c5 = np.max(corr) * dx
    return c5

def create_initial_h(n_points):
    """Create an initial feasible h vector."""
    # Use initial_h_values if available, otherwise create a simple balanced construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Resample to desired length
            h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
    else:
        # Create a symmetric construction
        h = np.ones(n_points) * 0.5
    # Ensure correct sum
    h = h * (n_points / 2) / np.sum(h)
    return h

def mutate_h(h, rate=0.1, epsilon=0.1):
    """Apply mutation to h."""
    h_new = h.copy()
    n = len(h)
    # Random indices to mutate
    indices = random.sample(range(n), max(1, int(rate * n)))
    for i in indices:
        # Add small perturbation
        delta = random.uniform(-epsilon, epsilon)
        h_new[i] = h_new[i] + delta
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Renormalize to sum to n/2
    h_new = h_new * (n / 2) / np.sum(h_new)
    return h_new

def crossover_h(h1, h2, alpha=0.5):
    """Blend crossover between two h vectors."""
    h_new = alpha * h1 + (1 - alpha) * h2
    # Clip to [0, 1]
    h_new = np.clip(h_new, 0, 1)
    # Renormalize to sum to n/2
    n = len(h1)
    h_new = h_new * (n / 2) / np.sum(h_new)
    return h_new

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    tournament_indices = random.sample(range(len(population)), k)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    """Run genetic algorithm to optimize h and find lower C5 bound."""
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 42  # Using a small size that's efficient
    pop_size = 30
    generations = 1000
    mutation_rate = 0.1
    crossover_prob = 0.7
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(np.array(initial_h_values))
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
    else:
        population.append(np.ones(n_points) * 0.5)
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
        
    # Fill the rest with random feasible vectors
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
    # Track best individual
    best_h = None
    best_c5 = float('inf')
    
    # Start evolution
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all population
        fitnesses = [evaluate_h(h) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Select parents and create offspring
        new_population = []
        # Elitism: keep the best 10%
        elite_size = max(1, pop_size // 10)
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            if random.random() < crossover_prob:
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                child = crossover_h(parent1, parent2)
            else:
                parent = tournament_selection(population, fitnesses)
                child = mutate_h(parent, mutation_rate)
                
            new_population.append(child)
            
        population = new_population[:pop_size]
        
    # Final evaluation of best individual
    final_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
