import numpy as np
from scipy import signal
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
    else:
        h = np.full(n_points, 0.5)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
    
    # Optimize with projected gradient descent
    max_iter = 10000
    best_h = None
    best_c5 = np.inf
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
        
        # Calculate overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient approximation
        # We want to minimize max_k \int h(x)(1 - h(x+k)) dx
        # For each k, define L_k(h) = \int h(x)(1 - h(x+k)) dx
        # Its gradient wrt h[i] = (1 - h[(i+k)%n]) - h[(i-k)%n]
        # So we compute approximate gradient from current values
        
        # Simplified gradient estimation via finite differences
        # For simplicity, here we do a coordinate descent approach
        # Perturb each point and compute gradient sign:
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus = h.copy()
            h_minus[i] = max(0.0, h_minus[i] - eps)
            
            corr_plus = signal.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = signal.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient descent step
        step_size = 1e-3
        h_new = h - step_size * grad
        
        # Project to [0,1] and correct sum
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * target_sum / h_sum
        
        h = h_new
    
    # Final evaluation
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), float(final_c5), int(n_points))
