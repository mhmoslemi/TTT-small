import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values or create symmetric version
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2.0)

    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        # Compute correlation of h with (1 - h)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return the maximum value scaled by dx
        return np.max(corr) * dx
    
    def project_h(h_vec):
        # Clip to [0, 1] and rescale to sum to n_points / 2
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2.0) / current_sum
        return h_vec
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.2
    max_iter = 10000
    patience = 1000
    patience_counter = 0
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb current h
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        h_new = project_h(h_new)
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            h = h_new
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_h = h.copy()
                best_C5 = current_C5
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        # Cooling with adaptive perturbation scaling
        if iteration % 100 == 0:
            perturb_scale *= 0.99
        T *= alpha
        
        # Early stopping based on patience
        if patience_counter > patience:
            break
            
    return (best_h.tolist(), best_C5, n_points)
