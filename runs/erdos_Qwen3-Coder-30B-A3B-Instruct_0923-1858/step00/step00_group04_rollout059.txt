        n = len(h)
        max_corr = 0
        for k in range(-n+1, n):
            if k < 0:
                corr = np.sum(h[:n+k] * (1 - h[-k:]))
            else:
                corr = np.sum(h[k:] * (1 - h[:n-k]))
            max_corr = max(max_corr, corr)
        return max_corr * dx

    best_h = h.copy()
    best_C5 = evaluate_C5(h)

    lr = 0.1

    h_opt = best_h
    c5_bound = best_C5
