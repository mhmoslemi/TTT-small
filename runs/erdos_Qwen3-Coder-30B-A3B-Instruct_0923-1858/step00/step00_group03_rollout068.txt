import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute all overlaps for each shift k
    # Using correlate with mode='full' to get all correlations
    # h and 1-h are the two sequences
    corr = correlate(h, 1 - h, mode='full')
    # Shift correlation to be from -n+1 to n-1
    # Since we're working with [0, 2] of length 2, and h of length n
    # The correlation output should be 2*n - 1 in length
    # But we only care about overlapping parts, so k from 0 to 2*len(h) - 2
    # Which corresponds to k from 0 to 2*n - 2
    # The overlaps are indexed from -n+1 to n-1
    # But in practice, we get the full range with indices 0 to 2*n-2
    # And we want the shift indices k = 0 to n-1
    # The full correlation: [0,...,n-1, n ,n+1,...,2n-2], 
    # we take k=0 to k=n-1 inclusive, so indices 0 to n-1
    # But, a simpler interpretation:
    # For each valid shift k from 0 to len(h)-1, we want 
    # sum over i of h[i] * (1 - h[i+k]) for valid i
    # But when k=0, it's just sum(h*(1-h)), which is also 
    # something we can compute directly
    
    # Let's rather directly compute all the overlaps
    # Overlaps will be for k from 0 to len(h)-1 (only those, not full)
    # But actually we need to compute as in the full correlation.
    max_overlap = 0.0
    
    # Recall that the correlate with mode='full' yields n + n - 1 = 2*n - 1 values.
    # The relevant shifts are from k=0 to k=n-1. Those correspond to indices 
    # from n-1 to 2*n-2 in the result.
    # But wait, actual shifts go from 0 to n-1, and overlapping range:
    # For shift k (0 to n-1): [0 to n-k-1] 
    # So the start of the valid indices is at n-1 for shift=0
    # The index for shift k is at (n-1+k)=n+k-1.
    # So the overlap for shift k is at position n+k-1 of the correlate result.
    # That gives a range from n-1 (for k=0) to 2*n-2 (for k=n-1).
    # But since the indexing goes from 0 to 2*n-2, we have to slice correctly.
    # overlap_k = correlate_result[n-1 + k] * dx
    # But they're weighted with dx = 2.0 / n_points
    
    # To align properly, take relevant slices:
    # correlate returns 2*n - 1
    # We want overlaps for shift k from 0 to n-1
    # so indices n-1 to 2*n-2 inclusive
    # This gives us the exact slice: n-1:2*n-1 
    # (end not inclusive in Python slice)
    # But correlate gives full result including both sides, we only care about positive shifts
    
    # Actually, the overlaps are properly around the center at k=0,
    # and by symmetry, the sorted indices that matter are 0 to n-1
    # on the right. The indices n-1 to 2*n-2 are the are related to kernels [0, n),
    # so for k = 0 to n-1: it's index k + n - 1 in correlate result
    # so for uniform increase, we can do:
    # take correlate_result[n-1:2*n-1] (start from n-1 where shift 0 is, and go up to 2*n-2 inclusive)
    # But it is clearer to take just the first n;
    # Because it says shift k is correlate result at index k + n - 1
    # i.e., for shift 0: np.correlate(h, 1-h)[n-1]
    # shift 1: np.correlate(h, 1-h)[n]
    # ... 
    # shift n-1: np.correlate(h, 1-h)[2*n-2]

    overlaps = corr[n-1:]
    valid_overlaps = overlaps[:n]
    max_overlap = np.max(valid_overlaps) * dx
    return max_overlap


def flip_h(h, i, new_h_i):
    h = h.copy()
    h[i] = new_h_i
    return h


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 70
    dx = 2.0 / n
    # Initialize using initial_h_values (binary)
    h = initial_h_values.copy()
    h = (h > 0.5).astype(int)
    h = h.astype(float)
    current_sum = np.sum(h)
    target_sum = n / 2
    
    # Adjust if needed
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        indices_1 = np.where(h == 1)[0]
        n_flip_down = int(current_sum - target_sum)
        old_indices = np.random.choice(indices_1, size=n_flip_down, replace=False)
        for idx in old_indices:
            h[idx] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        indices_0 = np.where(h == 0)[0]
        n_flip_up = int(target_sum - current_sum)
        old_indices = np.random.choice(indices_0, size=n_flip_up, replace=False)
        for idx in old_indices:
            h[idx] = 1
    
    # Compute initial overlaps
    C5 = evaluate_h(h, dx)
    best_C5 = C5
    best_h = h.copy()
    
    max_iter = 10000
    iter = 0
    
    while time.time() - start_time < budget_s - 0.1 and iter < max_iter:
        iter += 1
        # Find the shift k with maximum overlap
        corr = correlate(h, 1-h, mode='full')
        overlaps_full = corr[n-1:]
        overlaps = overlaps_full[:n]
        
        k_star = np.argmax(overlaps)
        max_overlap = overlaps[k_star] * dx
        
        # We want to flip positions i to minimize overlap at k_star
        # For finding the best flip, we check all i and see if it improves the k_star overlap
        
        best_improvement = 0
        best_i = -1
        flip_type = None  # 1_to_0 or 0_to_1

        # Compute changes in overlap if we flip each position
        for i in range(n):
            if h[i] == 1:
                # Flipping 1 to 0: compute how much the max overlap at k_star changes
                # Check which windows will be affected by flipping h[i] = 1 to 0
                # At shift k_star, this affects terms:
                # h[i] * (1 - h[i + k_star])
                # and h[i - k_star] * (1 - h[i])
                # But h[i] * (1 - h[i + k_star]) = 1 * (1 - h[i + k_star])
                # So it changes by - (1 - h[i + k_star]) = h[i + k_star] - 1
                
                # Also, for the indices that are exactly indexed by i, 
                # in the shift k_star computation, for h[i] (which is 1) we have a 1 * (1 - h[i + k_star])
                # Then, in the computation of h[i]* (1 - h[i + k_star]), 
                # when h[i] = 1, it’s 1 * (1 - h[i + k_star]) = 1 - h[i + k_star]
                
                # Perturb this term. 
                # For shift k_star, current term is: h[i] * (1 - h[i + k_star]) = (1) * (1 - h[i + k_star]) = 1 - h[i + k_star]
                # If h[i] is now 0, contribution = 0 * (1 - h[i + k_star]) = 0
                # So total change = 0 - (1 - h[i + k_star]) = h[i + k_star] - 1

                # If h[i + k_star] == 1, change is 0 
                # If h[i + k_star] == 0, change is -1
                
                # And the pair h[i - k_star] * (1 - h[i]) when i >= k_star 
                # where h[i] = 1 => 1 - h[i] = 0 => 0 * (1 - h[i]) = 0
                # So no change to the pair
                
                # But wait, the h[i - k_star] contrib -> when we flipped h[i], we lose
                # contribution from term h[i - k_star] * (1 - h[i]) to shift k_star
                # (i - k_star) could be out of bounds, so need to be careful:
                # 
                
                # In a standard correlation, we are summing h[j] * (1 - h[j + k]) over valid indices
                # So for shift k, we consider valid j where:
                # j in [0, n-k) if k < n
                # For each j, it contributes to overlap k value if j+k < n 
                # i.e., j from [0, n-k)  
                # so total is for j from 0 to n-k-1, sum h[j] * (1 - h[j + k]) 

                # Conceptually:
                # When we change h[i] from 1 to 0, 
                # in part of the overlap (h[i] * (1 - h[i + k_star]) = 1*(1-h[i+k_star]) = 1-h[i+k_star])
                # It changes to 0*(1-h[i+k_star]) = 0
                # => net change is: 0 - (1 - h[i+k_star]) = h[i+k_star] - 1
                
                # Then also, the corresponding h[i-k_star] contributes in shift k_star 
                # h[i - k_star] (if valid) * (1 - h[i]) =  h[i-k_star] * (1-1) = 0
                # But previously it was h[i-k_star] * (1-h[i]) = h[i-k_star] * 0 = 0
                # So no change
                
                # But when doing full correlation (moving subsequence, negative shifts also)
                # probably engulfs more influence
                
                if i + k_star < n:  # bound check
                    to_flip = 1 - h[i + k_star]
                    improvement = to_flip - 1  # Bug? We want to reduce h[i+k_star] * 1 
                    # Let's double-check: 
                    # Contribution is h[i] * (1 - h[i + k_star]).
                    # Original: 1 * (1 - h[i + k_star]) 
                    # After flip: 0 * 1 = 0
                    # Net change = 0 - (1 - h[i + k_star]) = h[i + k_star] - 1
                    
                    # Greater SPAN of error check for robustness:
                    # The Net change to the objective (for the k_star shift) = contribution of term changes  
                    # Currently, contribution at shift k_star for term i is h[i] * (1 - h[i + k_star]) =  1 * (1 - h[i + k_star])
                    # After flipping h[i] to 0 → dp = 0 - (1 - h[i + k_star]) = h[i + k_star] - 1.
                    # s.t. if h[i+k_star] = 0 → change = -1  
                    # If h[i+k_star] = 1 → change = 0
                    # So improvement = change
                    # We want maximization of how the net loss is reduced, so we want {change < 0}.
                    # Our goal is to reduce min h by changing a 1 to 0 or a 0 to 1.
                    
                    if improvement < best_improvement:
                        best_improvement = improvement
                        best_i = i
                        flip_type = '1_to_0'
                
                # Next lets also consider pairs (switches)
            else:
                # h[i] = 0. Let's consider flip it to 1:
                # Original contribution at shift k_star = 0 * (1 - h[i + k_star]) = 0
                # New contribution = 1 * (1 - h[i + k_star]) = 1 - h[i + k_star]  
                # So change = (1 - h[i + k_star]) - 0 = 1 - h[i + k_star]
                # We want robustness, take best change if can reduce the overlapping k_star 
                # So we want reduced overlap = negative change
                # So change = 1 - h[i + k_star]  
                # To *minimize* overlap at k_star, we want a *positive* change in the contribution,
                # i.e. we want large 1 - h[i + k_star], i.e. small h[i + k_star]
                # Hence, try 1 - h[i + k_star] as positive gain if small
                
                # But here we're exporting value from current state
                # If h[i + k_star] = 0, then change = 1 
                # If h[i + k_star] = 1, then change = 0
                
                if i + k_star < n: 
                    change = 1 - h[i + k_star] 
                    # Because we want to maximize reduction of the overlap term
                    # The function is: h_(j) * (1 - h_(j + k_star))  
                    # If I set h[i] = 1 (from previously 0), 
                    # - That increases overlaps at shift k_star involving i 
                    # But more importantly, there is a partner element
                    #  Let's keep just the flip that decreases overlap most at k_star, 
                    #  and it is a change of adjust +1 if we have h[i + k_star] = 0
                    #  This is best we can do w.r.t. contribution.
                
                # More robustly:
                # If we flip a 0 to 1, how "bad" is it for the max overlap at shift k_star? 
                # shift k_star is sum over j: h[j] * (1 - h[j + k_star]) where j in [0, n-k_star)
                # For this, we have contribution at term j = h[j] * (1 - h[j + k_star])
                # If we flip h[i] to 1, it only changes the term j = i in that set if:
                # i is valid range for shift k_star, i.e. i + k_star < n and i >= 0
                # The change in sum at that shift is h[i] * (1 - h[i + k_star]) = 1 * (1 - h[i + k_star])
                # [before] h[i] = 0, so it contributed 0
                # [after] h[i] = 1, so it contributes 1 - h[i + k_star]  
                # Change = 1 - h[i + k_star]
                
                # But we can't just change it randomly. To maintain sum = n/2,  
                # we must flip one 1 to 0 and one 0 to 1.
                # So it's a bit complex. Things we can do:
                # (1) Flip a 1 to 0 (reducing h[i] by 1)
                # (2) Flip the smallest 0 in k_star + offset to 1 (i.e., a 0 to 1, increases h by 1)
                # This is paired or something.
                
                # The principle remains:
                # Must delete one 1, add one 0.
                # I want to expect maximal impact, such that these combined changes
                # reduce overlap k_star as much as possible.
                # We can try to optimize the two-step flip for max benefit.
                if i + k_star < n:
                    new_contribution = 1 - h[i + k_star]
                    # Target: reduced principal.
                    if new_contribution > best_improvement:
                        best_improvement = new_contribution
                        best_i = i
                        flip_type = '0_to_1'
        
        # If no improvement found, skip
        if best_improvement >= 0 and best_i == -1:
            continue
            
        # Apply the best flip
        old_h = h.copy()
        if flip_type == '1_to_0':
            h[best_i] = 0
            # Loss: we must gain a 1 back to keep size constant
            # Look for 0, add 1
            # Try to find i' such that h[i'] = 0 
            # and it is not best_i and flip to 1
            bad_indices = set()
            for check_i in range(n):
                if h[check_i] == 0 and check_i != best_i:
                    flip_idx = check_i
                    break
            else:
                # try finding any 0 
                zero_indices = np.where(h == 0)[0]
                if len(zero_indices) > 0:
                    flip_idx = zero_indices[0] 
                else:
                    continue  # No 0 found to flip up!!

            h[flip_idx] = 1
        else:  # flip_type == '0_to_1'
            h[best_i] = 1
            # And flip a 1 to 0 instead to balance sum. Find one.
            # Find i' that h[i'] = 1 and i' != best_i
            one_indices = np.where(h == 1)[0]
            if len(one_indices) > 0:
                flip_idx = one_indices[0]
                h[flip_idx] = 0
            else:
                continue  # should not happen: sum issue
            
        # Recompute overlaps
        new_C5 = evaluate_h(h, dx) 
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
        else:
            h = old_h  # revert
        
        # Occasionally slightly cycle reset
        if iter % 100 == 99:
            h = best_h.copy()
            
    return (best_h.tolist(), best_C5, n)
