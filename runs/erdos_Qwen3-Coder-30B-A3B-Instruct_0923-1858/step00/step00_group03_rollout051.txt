import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute all correlations and return maximum
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, dx, target_sum):
    # Clamp values to [0,1] and adjust sum to match target
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to match sum
        scale = target_sum / current_sum
        h = h * scale
    return h
