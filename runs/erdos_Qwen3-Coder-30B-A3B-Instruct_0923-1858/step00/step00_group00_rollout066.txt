import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create symmetric construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 500
    no_improve_count = 0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new C5
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        # Annealing schedule
        T *= alpha
        if i % 100 == 0:
            perturb_scale *= 0.99
            
        # Early stopping if no improvement for too long
        if no_improve_count > no_improve_limit:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the overlap integral: max_k int h(x)(1 - h(x+k)) dx
    # This is equivalent to max of correlation of h with (1-h)
    correlation = signal.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(correlation)
    return max_correlation * dx
