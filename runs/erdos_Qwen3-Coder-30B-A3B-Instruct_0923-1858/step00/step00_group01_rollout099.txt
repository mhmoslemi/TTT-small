import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

                h0[i] = 1.0
        # Normalize
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Optimization parameters
    max_iter = 5000
    lr = 0.005
    tolerance = 1e-6
    time_limit = budget_s
    
    # Precompute indices
    positions = np.arange(n_points)
    
    def compute_c5(h_vec):
        # This replicates the behavior of evaluate_erdos_solution
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Gradient-based optimization
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit - 1:
            break
            
        # Approximate gradient using finite differences
        epsilon = 1e-4
        grad = np.zeros_like(best_h)
        base_c5 = compute_c5(best_h)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            # Clamp to bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            # Rescale to maintain constraint
            h_plus *= (n_points / 2) / np.sum(h_plus)
            h_minus *= (n_points / 2) / np.sum(h_minus)
            
            # Compute directional derivative
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Apply gradient descent
        new_h = best_h - lr * grad
        # Clamp values
        new_h = np.clip(new_h, 0, 1)

        new_h *= (n_points / 2) / np.sum(new_h)

        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        
        # Check for convergence
        if abs(new_c5 - base_c5) < tolerance:
            lr *= 0.9
            
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    """Compute C5 from the paper"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
