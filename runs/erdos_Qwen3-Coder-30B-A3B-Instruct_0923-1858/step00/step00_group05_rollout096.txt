import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    perturbation_rate = 0.1
    perturbation_freq = 100
    
    # Initialize
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Gradient descent loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5

        # Compute gradients
        # Cross-correlation: h * (1 - h) 
        conv_result = signal.correlate(h, 1-h, mode='full')
        # Max correlation value * dx
        max_correlation = np.max(np.abs(conv_result))
        # Correlation index of max_value
        max_idx = np.argmax(np.abs(conv_result))
        
        # Objective is max correlation, compute gradient 
        # We need to compute d/dh_i of max_k h[i] * (1 - h[i+k])
        # This is a bit tricky with 2d dependence. Let's do numerical gradient
        # using finite differences for stability
        gradient = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c_plus = evaluate_erdos_solution(h_plus)
            
            gradient[i] = (c_plus - current_C5) / eps
            
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Update step
        h = h - alpha * gradient
        
        # Projection onto feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturb periodically
        if iteration % perturbation_freq == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if iteration > 0 and iteration % 100 == 0:
            # Simple decay
            alpha *= 0.95
            if alpha < 1e-6:
                alpha = 1e-6
    
    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)
    
    # Ensure constraints
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2)
    
    return (best_h, final_C5, n_points)

def evaluate_erdos_solution(h):
    # Compute the overlap integral C_5
    # Using the discrete cross-correlation approach
    # The overlap integral is max_k int h(x)(1 - h(x+k)) dx
    # Which corresponds to max_k sum_i h[i](1 - h[i+k]) * dx
    
    n = len(h)
    dx = 2.0 / n
    
    # numpy.correlate computes sum_i h[i] * (1 - h[i+k]) for all shifts k
    # Normalized by dx
    correlations = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(np.abs(correlations)) * dx
    
    return C5
