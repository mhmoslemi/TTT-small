import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Convolution with (1-h)
    conv = correlate(h, 1 - h, mode="full")
    # Take maximum and scale by dx
    return np.max(conv) * dx

def project_h(h, n_points):
    """Project h to feasible region"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

def nelder_mead_simplex(h0, evaluate, n_points, budget_s):
    """Nelder-Mead with adaptive simplex size"""
    # Nelder-Mead parameters
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink
    
    max_iter = 5000
    n = len(h0)
    dx = 2.0 / n_points
    
    # Initial simplex around h0
    simplex_size = 0.1
    simplex = [h0.copy()]
    for i in range(n):
        h_new = h0.copy()
        h_new[i] += simplex_size
        simplex.append(project_h(h_new, n_points))
    
    # Main loop
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h, dx)
    no_improvement = 0
    reset_attempts = 0
    max_reset_attempts = 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate all vertices
        values = []
        for h in simplex:
            values.append(evaluate_h(h, dx))
        
        # Order by value (ascending)
        sort_idx = np.argsort(values)
        simplex = [simplex[i] for i in sort_idx]
        values = [values[i] for i in sort_idx]
        
        # Check best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Reset simplex if stuck
        if no_improvement > 50:
            if reset_attempts >= max_reset_attempts:
                break
            reset_attempts += 1
            simplex_size *= 0.5  # Reduce simplex size
            # Rebuild simplex
            simplex = [best_h.copy()]
            for i in range(n):
                h_new = best_h.copy()
                h_new[i] += simplex_size
                simplex.append(project_h(h_new, n_points))
            no_improvement = 0
            continue
            
        # Centroid excluding worst
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_r = centroid + alpha * (centroid - simplex[-1])
        h_r = project_h(h_r, n_points)
        c_r = evaluate_h(h_r, dx)
        
        # Accept reflection if better than worst but not best
        if values[0] <= c_r < values[-2]:
            simplex[-1] = h_r
            continue
            
        # Expansion
        if c_r < values[0]:
            h_e = centroid + gamma * (centroid - simplex[-1])
            h_e = project_h(h_e, n_points)
            c_e = evaluate_h(h_e, dx)
            if c_e < c_r:
                simplex[-1] = h_e
            else:
                simplex[-1] = h_r
            continue
            
        # Contraction
        if c_r >= values[-2]:
            if c_r < values[-1]:  # outside contraction
                h_c = centroid + beta * (simplex[-1] - centroid)
            else:  # inside contraction
                h_c = centroid + beta * (centroid - simplex[-1])
            h_c = project_h(h_c, n_points)
            c_c = evaluate_h(h_c, dx)
            if c_c < values[-1]:
                simplex[-1] = h_c
            else:
                # Shrink
                best = simplex[0]
                for i in range(1, n+1):
                    simplex[i] = best + rho * (simplex[i] - best)
                simplex = [project_h(h, n_points) for h in simplex]
    
    return best_h, best_c5
