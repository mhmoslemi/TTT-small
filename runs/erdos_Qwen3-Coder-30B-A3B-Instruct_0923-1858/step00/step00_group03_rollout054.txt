import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the constraint ∫₀² h(x) dx = 1
    h_sum = np.sum(h) * dx
    h = h * (1.0 / h_sum)
    
    # Gradient step parameters
    step_size = 0.001
    max_iter = 10000
    tol = 1e-6
    
    def compute_c5(h_vec):
        # Compute correlation with 1-h
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The full correlation result has length 2*n_points-1
        # We are interested in lags from -(n_points-1) to (n_points-1)
        # For our problem, we need max over all possible lags
        # The overlap is the integral
        corr *= dx
        max_corr = np.max(corr)
        return max_corr
    
    def compute_gradient(h_vec, k_star):
        """Compute gradient w.r.t. h for a given lag k_star"""
        # correlation for a specific lag k is sum_i h[i] * (1 - h[i+k])
        # For a given lag k, we compute gradients for h[i]
        # But in the full correlation, we are interested in max over k
        # This is a bit tricky, we approximate by focusing on the dominant k
        
        # Instead, we compute an approximate direction
        # Let's recompute the full correlation and look at the max
        corr_vals = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Find the index of max value
        max_idx = np.argmax(corr_vals)
        k_star = max_idx - (n_points - 1)
        
        # Get the gradient of the shifted version (exact)
        # correlation for lag k is sum_{i=0}^{n-1} h[i] (1 - h[i+k])
        # Gradient w.r.t. h[i] is (1 - h[i+k]) if k >= 0 and i+k < n
        # or (1 - h[i+k]) if k <= 0 and i+k >= 0
        # But we are looking at *all* lags, which makes it more complex
        # We will compute the gradient manually for the correlated term

        grad_h = np.zeros_like(h_vec)
        for i in range(n_points):
            for j in range(n_points):
                k = j - i
                if 0 <= i + k < n_points:
                    # We're counting contribution of h[i] to lag k
                    grad_h[i] += (1 - h_vec[i + k]) * dx

        # Normalize gradient
        norm = np.linalg.norm(grad_h)
        if norm > 0:
            grad_h = grad_h / norm
        return grad_h
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        # Compute correlation for all lags
        corr_vals = signal.correlate(h, 1 - h, mode='full')
        corr_vals *= dx
        max_idx = np.argmax(corr_vals)
        max_lag = max_idx - (n_points - 1)
        max_corr = corr_vals[max_idx]
        
        # Compute gradient direction for the dominant lag
        # approximate gradient
        k = max_lag
        # Gradient within one correlation kernel:
        # For lag k in the correlation corr[i] = sum_j h[j]*h[i+k-j]
        # So gradient w.r.t. h[i] is sum_{j between 0 and n-1}
        # s.t. j+k-i is in [0,n-1] of (1-h[j])
        # This is complex. we take a gradient direction for the major term
        
        # Simplified direction adjustment:
        # For the max correlation lag k, we want minimizing h[i] where it contributes to that overlap
        # We can derive an artificial gradient by estimating which h[i] to decrease/increase
        # We observe: at lag k, value h[i] contributes to term (1-h[i+k])
        # For the maximum to decrease, we should improve factors that make h[i+k] small
        # But we'll use a forward-difference type approach:

        grad_dir = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad_dir[i] = -(1 - h[j]) * dx
        # Normalize
        norm_grad = np.linalg.norm(grad_dir)
        if norm_grad > 0:
            grad_dir /= norm_grad
            
        # Take gradient step
        h_new = h - step_size * grad_dir
        # Clamp values
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum
        h_sum_new = np.sum(h_new) * dx
        h_new = h_new * (1.0 / h_sum_new)
        
        # Compute result
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new
        
        # Adjust step size
        if iter_count % 100 == 0:
            step_size *= 0.99
            
    return (best_h.tolist(), best_c5, n_points)
