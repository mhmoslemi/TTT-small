import numpy as np

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h *= scale
    return h

def compute_C5(h, dx):
    # Compute the cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (corresponds to the maximum overlap integral)
    return np.max(corr) * dx
