import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2, prob=0.7):
    """Blend crossover between two parents."""
    if np.random.rand() > prob:
        return parent1.copy(), parent2.copy()
    
    alpha = np.random.rand()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, rate=0.1, std=0.01):
    """Add Gaussian noise to h."""
    mask = np.random.rand(len(h)) < rate
    noise = np.random.normal(0, std, len(h))
    h = h + mask * noise
    return project_to_feasible(h, len(h))

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    n_pop = 30
    elite_rate = 0.1
    mutation_rate = 0.1
    crossover_prob = 0.7
    tournament_size = 5
    max_no_improve = 200
    
    # Generate initial population
    if 'initial_h_values' in globals():
        pop = [np.array(initial_h_values)]
        pop.extend([np.random.rand(n_points) for _ in range(n_pop - 1)])
    else:
        pop = [np.random.rand(n_points) for _ in range(n_pop)]
    
    # Normalize initial population
    for i in range(len(pop)):
        pop[i] = project_to_feasible(pop[i], n_points)
        
    dx = 2.0 / n_points
    
    # Track best
    global_best_h = None
    global_best_c5 = float('inf')
    no_improve_count = 0
    start_time = time.time()
    
    for gen in range(1000):  # Max generations
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in pop:
            c5 = evaluate(h, dx)
            fitnesses.append(c5)
            
        # Update global best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < global_best_c5:
            global_best_c5 = fitnesses[min_idx]
            global_best_h = pop[min_idx].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > max_no_improve:
            break

        # Selection and reproduction
        sorted_indices = np.argsort(fitnesses)
        elites = [pop[i] for i in sorted_indices[:int(elite_rate * n_pop)]]
        
        new_pop = elites[:]
        
        # Generate offspring
        while len(new_pop) < n_pop:
            parent1 = tournament_selection(pop, fitnesses, tournament_size)
            parent2 = tournament_selection(pop, fitnesses, tournament_size)
            
            child1, child2 = crossover(parent1, parent2, crossover_prob)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_pop.extend([child1, child2])
            
        # Trim to population size if needed
        pop = new_pop[:n_pop]
    
    # Final evaluation of the global best
    if global_best_h is not None:
        final_c5 = evaluate(global_best_h, dx)
        return (global_best_h.tolist(), final_c5, n_points)
    else:
        # Fallback evaluation
        final_pop = pop[0] if pop else np.random.rand(n_points)
        final_pop = project_to_feasible(final_pop, n_points)
        final_c5 = evaluate(final_pop, dx)
        return (final_pop.tolist(), final_c5, n_points)
