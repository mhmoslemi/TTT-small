import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create simple symmetric construction
        n_points = 45
        h = np.ones(n_points) * 0.5  # Simple uniform
        # Slightly perturb to make it non-trivial
        h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = max(np.correlate(h, 1-h, mode="full")) * dx
    last_improvement = 0
    iteration = 0
    start_time = time.time()
    
    # Parameters
    temp = 1.0
    temp_decay = 0.99
    sigma = 0.1
    sigma_decay = 0.995
    patience = 1000
    
    while time.time() - start_time < budget_s and last_improvement < patience:
        # Perturb
        pert = np.random.normal(0, sigma, n_points)
        h_candidate = h + pert
        
        # Project onto feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        c5_candidate = max(np.correlate(h_candidate, 1-h_candidate, mode="full")) * dx
        
        # Accept with probability
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
            last_improvement = 0
        else:
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = h_candidate
                best_c5 = c5_candidate
                last_improvement = 0
            else:
                last_improvement += 1
                
        # Update parameters
        temp *= temp_decay
        sigma *= sigma_decay
        iteration += 1
    
    # Final correction to ensure sum is exact
    h_corrected = h * (n_points / 2) / np.sum(h)
    final_c5 = max(np.correlate(h_corrected, 1-h_corrected, mode="full")) * dx
    
    return (h_corrected.tolist(), final_c5, n_points)
