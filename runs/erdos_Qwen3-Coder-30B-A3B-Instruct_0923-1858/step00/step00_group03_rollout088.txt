import numpy as np
import time
import random
from scipy.signal import correlate

def binary_convert(h_values):
    """Convert to binary values maintaining sum constraint"""
    h = np.array(h_values)
    h_rounded = np.round(h).astype(int)
    current_sum = np.sum(h_rounded)
    target_sum = len(h) // 2
    
    if current_sum == target_sum:
        return h_rounded
    
    # Adjust to target sum
    diff = target_sum - current_sum
    if diff > 0:  # Need to add 1s
        indices = np.argsort(h_rounded)
        for i in range(diff):
            if h_rounded[indices[-(i+1)]] == 0:
                h_rounded[indices[-(i+1)]] = 1
    else:  # Need to subtract 1s
        indices = np.argsort(h_rounded)
        for i in range(-diff):
            if h_rounded[indices[i]] == 1:
                h_rounded[indices[i]] = 0
    
    return h_rounded

def compute_C5(h, dx):
    """Compute C5 using correlation"""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Simulated Annealing with binary search
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    time_start = time.time()
    
    for iteration in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
            
        # Modify two positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = random.choice(ones)
        j = random.choice(zeros)
        
        # Try flip
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        new_C5 = compute_C5(h_new, dx)
        old_C5 = compute_C5(h, dx)
        
        delta = new_C5 - old_C5
        
        if delta < 0 or random.random() < np.exp(-delta / T):
            h = h_new
            if new_C5 < best_C5:
                best_h = h_new.copy()
                best_C5 = new_C5
        
        T *= alpha
        
    # Final check
    final_C5 = compute_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)
