import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute cross-correlation of h and (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Maximum value over all lags
    c5 = np.max(corr) * dx
    return c5

def project_h(h, dx):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to ensure integral = 1
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, else start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(47) * 0.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    h = project_h(h, dx)
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Gradient descent parameters
    lr = 0.01
    decay_every = 100
    max_iter = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and gradient approximation
        c5 = evaluate_c5(h)
        # Compute a gradient approximation by finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, dx)
            c5_plus = evaluate_c5(h_plus)
            grad[j] = (c5_plus - c5) / eps
        
        # Gradient descent step
        h = h - lr * grad
        # Project to feasible region
        h = project_h(h, dx)
        
        # Check for improvement
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Decay learning rate
        if (i + 1) % decay_every == 0:
            lr *= 0.1
            
    # Final check
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
