import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[1::2] = 0.0
        h *= n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    eps = 0.01
    perturb_freq = 100
    max_iter = int(1e5)
    
    def project_h(h_vec):
        # Ensure h is in [0,1] and sum to target_sum
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec *= target_sum / current_sum
        return h_vec
    
    def evaluate_c5(h_vec, dx):
        # Cross-correlation for full mode
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def adjust_h(h_vec, eps):
        # Only modify if we're still within budget
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
            return h_vec.copy()
        
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_k = np.argmax(corr)
        n = len(h_vec)
        max_corr = corr[max_k]
        
        # Find the indices causing the maximum correlation
        # Since we're in full mode, corr[k] corresponds to shift k - (n-1)
        shift = max_k - (n - 1)
        
        # Modify h to decrease overlap
        h_new = h_vec.copy()
        for i in range(n):
            k = (i + shift) % n
            if k < n:  # Valid index
                # Reduce h[i] and increase around k
                h_new[i] -= eps * 0.1
                h_new[k] += eps * 0.1
                
        return project_h(h_new)
    
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Periodically perturb
        if iter_count % perturb_freq == 0:
            # Redistribution: move mass from high to low overlap areas
            h_pert = h.copy()
            # Select indices with high values and move to low ones
            sorted_indices = np.argsort(h_pert)[::-1]
            # Move small amounts from top to bottom quartile
            n_move = max(2, n_points // 10)
            for i in range(n_move):
                from_idx = sorted_indices[i]
                to_idx = sorted_indices[-i-1]
                if h_pert[from_idx] > eps:
                    move_amount = min(h_pert[from_idx], eps * 0.5)
                    h_pert[from_idx] -= move_amount
                    h_pert[to_idx] += move_amount
            h = project_h(h_pert)
        else:
            h = adjust_h(h, eps)
        
        # Evaluate current solution
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
