import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode='full')
    # Take only the relevant part (shifts from -n+1 to n-1)
    n = len(h)
    overlaps = overlaps[n-1:2*n-1]
    return np.max(overlaps) * dx

def project_h(h, target_sum, dx):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to target sum
    current_sum = np.sum(h) * dx
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Proportional adjustment
    adjustment = diff / (np.sum(h > 0) if np.sum(h > 0) > 0 else 1)
    h = h + adjustment
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load or initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial guess if no parent construction
        h = np.ones(70) * 0.5
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = 1.0  # sum(h) * dx = 1
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 20000
    iter_per_log = 100
    
    step_size = 0.1
    iteration = 0
    
    while iteration < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Evaluate current state
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute overlaps for all shifts
        overlaps = correlate(h, 1 - h, mode='full')
        n = len(h)
        overlaps = overlaps[n-1:2*n-1]
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if max_overlap < 1e-10:
            break
            
        # Compute gradients
        # For the maximum overlap shift k*:
        # grad_i = (1 - h[i + k*]) if valid
        # grad_{i+k*} = - h[i] if valid
        
        grad = np.zeros(n)
        k_star = max_k - (n-1)
        
        # Construct the gradient efficiently
        h_shifted = np.roll(h, k_star)
        # Derivative w.r.t. h[i]: (1 - h[i + k_star])
        grad = 1 - h_shifted
        
        # Adjust for rolling
        if k_star > 0:
            # h[i + k_star] element in h is at h[i] in the shifted vector
            # So we have to subtract h[i - k_star] from the gradient at i
            # But gradient at i = 1 - h[i + k_star] 
            # which is 1 - h[i + k_star] from the original h array
            # So for i = [0, ..., n-1]
            # gradient at i = 1 - h[(i + k_star) % n]
            # But we're looking at i where i + k_star < n for valid index
            for i in range(n):
                idx = (i + k_star) % n
                # We adjust the gradient
                if grad[i] > 0:
                    grad[i] = 1 - h[idx]
                else:
                    grad[i] = 0
        else:
            # Loop through valid indices
            for i in range(n):
                if i - k_star < n:
                    grad[i] = 1 - h[i - k_star]
                else:
                    grad[i] = 1
        
        # Update h
        h = h - step_size * grad
        h = project_h(h, target_sum, dx)
        
        # Adaptive step size
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.95
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    best_h = project_h(best_h, target_sum, dx)
    
    return (best_h, final_c5, len(best_h))
