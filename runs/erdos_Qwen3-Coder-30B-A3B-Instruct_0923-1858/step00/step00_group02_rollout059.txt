        h *= (n_points / 2) / np.sum(h)

    # Define objective function for minimization
    def objective(h_vec):
        h = h_vec.copy()
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Rescale to satisfy sum constraint
        h *= (n_points / 2) / np.sum(h)
        
        # Compute correlation with 1-h
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define constraint for sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Set up optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    start_time = time.time()

    initial_c5 = objective(h)
    
    # Initialize best solution and best C5
    best_h = h.copy()
    best_c5 = initial_c5
    
    # Trust region size
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    
    # Gradient approximation function
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Trust region optimization loop
    while time.time() - start_time < budget_s and iter_count < max_iter:

        # Compute gradient
        try:
            grad = compute_gradient(best_h)
        except:
            break
        
        # Update using trust region step
        step = -grad.copy()
        step_norm = np.linalg.norm(step)
        
        # Normalize step
        if step_norm > 0:
            step = step / step_norm
        
        # Try step within trust radius
        step *= min(trust_radius, 1.0)
        
        # Apply step
        new_h = best_h + step
        
        # Project to feasible region
        new_h = np.clip(new_h, 0, 1)
        # Rescale to meet sum constraint
        new_h *= (n_points / 2) / np.sum(new_h)
        
        # Evaluate new objective
        new_c5 = objective(new_h)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_h = new_h.copy()

        # Check if we are stuck

            break
    
    # Final evaluation of best solution
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
