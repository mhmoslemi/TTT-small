```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute overlaps for all shifts
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap (scaled by dx)
    c5 = np.max(overlaps) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        
    # Normalize to sum to 1
    dx = 2.0 / n_points
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_current = h_init.copy()
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    # Binary search parameters
    C_min, C_max = 0.0, 0.5
    precision = 1e-4
    iterations = 0
    max_iterations = 15
    
    # Iterative refinement
    while iterations < max_iterations and (C_max - C_min) > precision:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        C_candidate = (C_min + C_max) / 2.0
        
        # Try optimization
        h_refined = optimize_h(h_current, n_points, C_candidate, dx)
        
        if h_refined is not None:
            c5_candidate = evaluate_erdos_solution(h_refined, n_points)
            if c5_candidate <= C_candidate:
                C_max = C_candidate
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_refined.copy()
            else:
                C_min = C_candidate
        else:
            C_min = C_candidate
            
        h_current = best_h.copy()
        iterations += 1
    
    # Final check and refinement if needed
    if time.time() - start_time < budget_s * 0.95:
        final_h = optimize_h(best_h, n_points, best_c5, dx)
        if final_h is not None:
            final_c5 = evaluate_erdos_solution(final_h, n_points)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)

def optimize_h(h_current, n_points, C_target, dx):
    h = cp.Variable(n_points)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    
    # For binary search stage, we avoid explicit convolution calculation
    # Instead, one convex relaxation approach is to add auxiliary variables
    # and use McCormick envelopes for bilinear terms.
    # Here, we do not compute all overlap terms with CVXPY directly due to complexity.
    
    # Given computational constraints and the fact that we're working with
    # a heuristic approach plus our estimator, we'll do a small iterative
    # improvement using scipy optimized: gradient-based local search
    
    def objective(h_vec):
        # This is just a placeholder for demonstration
        h_vec = np.array(h_vec)
        h_vec = np.maximum(0, np.minimum(1, h_vec))
        overlaps = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(overlaps) * dx
    
    # For the optimization, we use a simple but effective procedure
    # perturb the solution and try gradient descent steps
    h_refined = h_current.copy()
    
    # Simple local search using scipy (convex relaxation is too heavy)
    t_start = time.time()
    while time.time() - t_start < 2:
        def loss(h_vec):
            h_vec = np.maximum(0, np.minimum(1, h_vec))
            # Compute overlaps
            overlaps = np.correlate(h_vec, 1 - h_vec, mode="full")
            c5_val = np.max(overlaps) * dx
            penalty = 0
            if np.sum(h_vec) < n_points / 2 - 1e-6:
                penalty += 1e6 * (n_points / 2 - np.sum(h_vec))
            if np.sum(h_vec) > n_points / 2 + 1e-6:
                penalty += 1e6 * (np.sum(h_vec) - n_points / 2)
            return c5_val + penalty
        
        # Optimize with L-BFGS-B
        res = minimize(loss, h_refined, method='L-BFGS-B', 
                       bounds=[(0, 1)] * n_points,
                       options={'maxiter': 20, 'disp': False})
        
        if res.success and res.fun < C_target:
            h_refined = res.x
        else:
            break
        
        if res.fun < C_target:
            break
    
    if np.sum(h_refined) < n_points / 2 - 1e-6:
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    elif np.sum(h_refined) > n_points / 2 + 1e-6:
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
    return h_refined
```
