    h = h * target_sum / np.sum(h)

        # Compute all overlaps 
        conv = np.correlate(h, 1 - h, mode='full')
        conv = conv * dx
        k_values = np.arange(-n_points + 1, n_points)
        c5 = np.max(conv)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find k that gives maximum overlap
        max_idx = np.argmax(conv)
        k_star = k_values[max_idx]
        
        if k_star >= 0:
            # We need to check all i such that i + k_star < n_points
            ranges = range(n_points - k_star)
        else:
            # k_star is negative, so shift is to the left
            ranges = range(-k_star, n_points)
            
        # Compute gradient or adjust parameters to reduce overlap
        best_reduction = 0
        best_i = -1
        best_j = -1
        
        # Try adjusting each position
        for i in ranges:
            j = i + k_star
            if j < n_points and j >= 0:
                # Gradient direction
                # h[i] and h[j] contribute to overlap
                # Try adjusting both
                if h[i] > 0 and h[j] < 1:
                    # Reduce contribution of this pair
                    # Trying to lower h[i] a bit and increase h[j] a bit
                    # Which one gives better reduction ?
                    reduction1 = h[i] * (1 - h[j]) * dx

                    new_h1 = h.copy()

                        new_h1[i] -= step
                        new_h1[j] += step
                        new_c5 = evaluate_erdos_solution(new_h1, dx)
                        reduction = c5 - new_c5
                        if reduction > best_reduction:
                            best_reduction = reduction
                            best_i = i
                            best_j = j

            if step > 0:
                h[best_i] -= step
                h[best_j] += step
                
                # Project back to feasible domain
                h = np.clip(h, 0, 1)
                # Ensure sum constraint
                current_sum = np.sum(h)
                diff = target_sum - current_sum
                if abs(diff) > 1e-8:  
                    # Distribute correction evenly
                    h += diff / n_points
                h = np.clip(h, 0, 1)

                    current_sum = np.sum(h)
                    diff = target_sum - current_sum
                    if abs(diff) > 1e-8:
                        h += diff / n_points
                    h = np.clip(h, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
