import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    N = len(h)
    dx = 2.0 / N
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value and scale by dx
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to enforce sum(h) = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def optimize_step(h, dx, max_iter=100, learning_rate=1e-3, tol=1e-5):
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    for i in range(max_iter):
        # Compute the overlap integral manually to get gradient
        # Center of the cross-correlation
        N = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        # Compute gradient of the max element w.r.t. h
        # This is quite approximate but good enough for gradient step
        grad = np.zeros_like(h)
        # Compute finite difference gradient
        eps = 1e-6
        for k in range(N):
            h_plus = h.copy()
            h_plus[k] += eps
            h_plus = project_h(h_plus, N)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_corr_plus = np.max(corr_plus)
            grad[k] = (max_corr_plus - max_corr) / eps
        # Apply gradient step
        h_new = h - learning_rate * grad
        # Project
        h_new = project_h(h_new, N)
        # Check improvement
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        # Early termination
        if i > 10 and abs(new_c5 - best_c5) < tol:
            break
        h = h_new
    return best_h, best_c5

def interpolate_to_higher_resolution(h_low, n_high):
    h_high = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, len(h_low)), h_low)
    h_high = project_h(h_high, n_high)
    return h_high

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_low = 50
    n_high = 70
    dx_low = 2.0 / n_low
    dx_high = 2.0 / n_high

        h_init = project_h(h_init, len(h_init))
    else:
        h_init = np.ones(n_low) * 0.5
        # Add noise to get out of symmetric local optima
        h_init += np.random.normal(0, 0.02, n_low)
        h_init = project_h(h_init, n_low)
    
    h = h_init.copy()
    c5 = evaluate_erdos_solution(h)

    # Low res optimization
    h, c5 = optimize_step(h, dx_low, max_iter=200, learning_rate=1e-2, tol=1e-6)
    elapsed = time.time() - start_time 
    if elapsed > budget_s * 0.7:
        return (list(h), c5, len(h))
    
    # Up sample
    h = interpolate_to_higher_resolution(h, n_high)
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.9:
        return (list(h), c5, len(h))

    # High res optimization
    h, c5 = optimize_step(h, dx_high, max_iter=1000, learning_rate=1e-3, tol=1e-6)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.9:
        return (list(h), c5, len(h))
    
    return (list(h), c5, len(h))
