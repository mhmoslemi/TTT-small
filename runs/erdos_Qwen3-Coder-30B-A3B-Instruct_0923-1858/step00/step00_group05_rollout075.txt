import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    # This is the evaluation function that's pre-imported
    dx = 2.0 / n_points
    # Compute the full correlation between h and (1-h)
    correlation = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum overlap value
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h with the given initial construction if available
    if 'initial_h_values' in globals():

        if np.abs(np.sum(h_initial) - target_sum) < 1e-10:
            h = h_initial.copy()
        else:
            # Normalize
            h = h_initial * target_sum / np.sum(h_initial)
    else:
        # Create a feasible starting point
        h = np.ones(n_points) * 0.5
        # Adjust to make the sum correct
        h[0] += (target_sum - np.sum(h)) / n_points
    
    # Define Objective Function
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, n_points)
    
    # Constraint functions
    def sum_constraint(h_vec):
        return np.sum(h_vec) - target_sum

    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Optimization settings
    method = 'SLSQP'
    options = {'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Start with the initial solution
    h_current = h.copy()
    c5_current = objective(h_current)
    
    if c5_current < best_c5:
        best_h = h_current.copy()
        best_c5 = c5_current
    
    # Simple local optimization approach using SLSQP
    # First, ensure feasibility
    h_normalized = np.clip(h_current, 0, 1)
    h_normalized = h_normalized * target_sum / np.sum(h_normalized)
    
    try:
        result = minimize(
            objective,
            h_normalized,
            method=method,
            bounds=bounds,
            constraints=constraints,
            options=options,
            tol=1e-8
        )
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * target_sum / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
        else:
            pass  # Keeping best_h from earlier iterations
    except Exception:
        # Keep best_h as is
        pass
    
    # Now, we try a few randomized restarts to escape local minima
    n_restarts = 5
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1.0:
            break

        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        try:
            # Restart optimization from this point
            result = minimize(

                h_rand,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options=options,
                tol=1e-8

                h_test = result.x
                h_test = np.clip(h_test, 0, 1)

    # Final check

    best_h = best_h * target_sum / np.sum(best_h)

    return (best_h.tolist(), best_c5, n_points)
