import numpy as np
from scipy import optimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h *= scale
    return h

def evaluate_C5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def gradient_step(h, dx, lr=0.01):
    n = len(h)
    # Compute current correlation
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    
    # Compute gradient element-wise
    # We use finite differences to estimate gradient
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    best_C5 = float('inf')

    current_C5 = evaluate_C5(h, dx)

    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h.copy()

    # Gradient descent loop
    lr = 0.01

        if time.time() - start_time > budget_s - 1:

        grad = gradient_step(h, dx, lr)
