import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_c5(h_values, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    h = np.array(h_values)
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum is over all shifts - the full correlation has length 2*n-1
    # The indices for shifts 0 to 2*n-2 
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h):
    """Project h to feasible region: [0,1] and sum = n/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    n = len(h)
    if h_sum > 0:
        h = h * (n/2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction or create symmetric one
    if 'initial_h_values' in globals():
        h_values = np.array(initial_h_values)
        n = len(h_values)
    else:
        n = 45
        h_values = np.ones(n) * 0.5  # symmetric start
    
    dx = 2.0 / n
    best_h = h_values.copy() 
    best_c5 = evaluate_c5(best_h, dx)
    
    # Parameters
    pop_size = 50
    generations = 200
    elite_size = pop_size // 5
    tournament_size = 5
    mutation_rate = 0.1
    crossover_alpha = 0.5
    
    population = [project_to_feasible(np.random.rand(n)) for _ in range(pop_size-1)]
    population.insert(0, best_h.copy())
    
    # GA Main Loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_c5(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection + crossover + mutation
        new_pop = population[:elite_size]  # Keep elites
        
        # Tournament selection
        def tournament_selection():
            idxs = np.random.choice(pop_size, tournament_size)
            best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            return population[best_idx]
        
        # Generate new population
        while len(new_pop) < pop_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            child = parent1 * crossover_alpha + parent2 * (1 - crossover_alpha)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n)
                child += noise
                child = project_to_feasible(child)
                
            new_pop.append(child)
        
        # Environment selection (keep top 70%)
        population = new_pop[:int(0.7 * pop_size)]
    
    # Gradient refinement of best solutions
    for _ in range(150):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Take top 3 solutions
        top_solutions = population[:3]
        
        for h in top_solutions:
            # Simple gradient descent step
            eps = 1e-5
            h_old = h.copy()
            h_grad = np.zeros_like(h)
            
            for i in range(n):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus)
                c5_plus = evaluate_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus)
                c5_minus = evaluate_c5(h_minus, dx)
                
                h_grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            lr = 0.01 / sqrt(1 + _)  # decaying learning rate
            h = h - lr * h_grad
            h = project_to_feasible(h)
            
            # Evaluate update
            c5_new = evaluate_c5(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), int(n))
