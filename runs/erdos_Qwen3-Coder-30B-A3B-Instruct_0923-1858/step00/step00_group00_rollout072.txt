import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Only take the overlap part (shifts from -n+1 to n-1)
    mid = len(corr) // 2
    overlaps = corr[mid-(len(h)-1):mid+(len(h)-1)+1]
    return np.max(overlaps) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def eval_and_project(h, n_points, dx):
    """Evaluate and ensure feasibility"""
    h = project_to_feasible(h, n_points)
    c5 = evaluate_h(h, dx)
    return h, c5

def genetic_algorithm(initial_h_values, n_points, dx, budget_s):
    """Genetic algorithm optimization"""
    start_time = time.time()
    
    # Parameters
    population_size = 50
    generations = 200
    elite_ratio = 0.2
    mutation_rate = 0.05
    crossover_alpha = 0.5
    
    # Initialize population
    population = [initial_h_values.copy()]
    for i in range(population_size - 1):
        # Random perturbation ensuring feasibility
        h = initial_h_values + np.random.normal(0, 0.1, n_points)
        h, _ = eval_and_project(h, n_points, dx)
        population.append(h)
    
    # Main evolution loop
    best_c5 = float('inf')
    best_h = None
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            h, c5 = eval_and_project(h, n_points, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Elitism
        elite_count = int(elite_ratio * population_size)
        new_population = population[:elite_count]
        
        # Generate children
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_select(population, fitness, 5)
            parent2 = tournament_select(population, fitness, 5)
            
            # Crossover
            child = blend_crossover(parent1, parent2, crossover_alpha)
            
            # Mutation
            if np.random.random() < mutation_rate:
                child += np.random.normal(0, 0.05, n_points)
            
            # Add to new population
            new_population.append(child)
        
        # Truncate to population size
        population = new_population[:population_size]
        
    return best_h, best_c5

def tournament_select(population, fitness, tournament_size):
    """Select individual with tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def blend_crossover(h1, h2, alpha):
    """Blend crossover"""
    return alpha * h1 + (1 - alpha) * h2

def gradient_descent(initial_h, n_points, dx, budget_s):
    """Refine using gradient-based method"""
    start_time = time.time()
    
    h = initial_h.copy()
    learning_rate = 0.01
    eps = 1e-5
    max_iter = 500
    
    def loss_fn(h):
        h, c5 = eval_and_project(h, n_points, dx)
        return c5
    
    def gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_loss = loss_fn(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            grad[i] = (loss_fn(h_plus) - base_loss) / eps
        return grad
    
    best_h = h.copy()
    best_c5 = loss_fn(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = gradient(h, eps)
        
        # Update with learning rate
        h = h - learning_rate * grad
        
        # Project to feasible region
        h, c5 = eval_and_project(h, n_points, dx)
        
        # Update best if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute adaptive learning rate
        learning_rate = 0.01 / np.sqrt(i + 1)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction or create a sensible one
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    else:
        # Fallback construction: a simple symmetric function
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    
    # Initial evaluation
    h, c5_bound = eval_and_project(h, n_points, dx)
    
    # Phase 1: Genetic Algorithm
    if time.time() - start_time < budget_s * 0.7:
        ga_h, ga_c5 = genetic_algorithm(initial_h_values if 'initial_h_values' in globals() else h, n_points, dx, budget_s * 0.7)
        if ga_c5 < c5_bound:
            h, c5_bound = ga_h, ga_c5
    
    # Phase 2: Gradient Descent on top solutions
    if time.time() - start_time < budget_s - 5:
        gd_h, gd_c5 = gradient_descent(h, n_points, dx, budget_s * 0.3)
        if gd_c5 < c5_bound:
            h, c5_bound = gd_h, gd_c5
            
    # Final evaluation
    _, final_c5 = eval_and_project(h, n_points, dx)
    
    return (h, final_c5, n_points)
