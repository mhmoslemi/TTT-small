import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)  # normalize to sum = n_points/2
    
    # Parameters
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 500
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    no_improve_count = 0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Perturb
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        C5_new = evaluate_C5(h_new, dx)
        
        # Accept or reject
        if C5_new < best_C5 or np.random.random() < np.exp(-(C5_new - best_C5) / T):
            h = h_new
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cooling
        T *= alpha
        if iteration % 100 == 0:
            perturb_scale *= 0.99
            
        iteration += 1
        
        if no_improve_count > no_improve_limit:
            break
    
    return (best_h, best_C5, n_points)

def evaluate_C5(h, dx):
    # Compute convolution of h and (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Use the maximum value from the full convolution
    max_val = np.max(corr)
    return max_val * dx
