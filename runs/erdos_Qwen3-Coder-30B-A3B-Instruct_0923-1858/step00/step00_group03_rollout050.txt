import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: create a reasonable initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h * 2.0 / np.sum(h)  # Normalize to sum = 2.0/2.0 = 1.0
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Set up the optimization parameters
    max_iter = 10000
    lr = 0.01
    tol = 1e-6
    best_c5 = float('inf')
    best_h = None
    
    # Precompute shifted versions for efficiency
    h_full = np.concatenate([h, np.zeros(n_points-1)])
    h_flip = np.flip(1-h)
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break
            
        # Compute all correlations
        corr = correlate(h_full, h_flip, mode="full")
        k_indices = np.arange(len(corr))
        k_shifts = k_indices - (n_points - 1)
        
        # Find the dominant peak
        max_idx = np.argmax(corr)
        k_star = k_shifts[max_idx]
        max_corr = corr[max_idx] * dx
        
        # Early stopping for low objective values
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
            
        if max_corr < 0.3807:
            break
            
        # Compute gradient for the dominant overlap
        gradient = np.zeros_like(h)
        h_shifted = np.roll(h, -k_star)
        if k_star >= 0:
            gradient[:len(h)-k_star] += h_shifted[k_star:]
        else:
            gradient[-k_star:] += h_shifted[:-k_star]
        
        # Update h
        h = h - lr * gradient
        
        # Project onto [0, 1] box constraints
        h = np.clip(h, 0, 1)
        
        # Adjust the sum to exactly 1.0
        h_sum = np.sum(h)
        h = h * (1.0 / h_sum)
        
        # Early stopping
        if abs(max_corr - best_c5) < tol:
            lr *= 0.95
        if lr < 1e-6:
            break
    
    # Final evaluation
    if best_h is None:
        best_h = h
        
    h_full = np.concatenate([best_h, np.zeros(n_points-1)])
    corr = correlate(h_full, np.flip(1-best_h), mode="full")
    final_c5 = np.max(corr) * dx
    
    return (best_h, final_c5, n_points)
