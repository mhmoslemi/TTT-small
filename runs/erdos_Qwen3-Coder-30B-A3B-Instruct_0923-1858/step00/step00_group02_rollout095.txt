import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Default initialization, set n_points to 45 as in the note
        n_points = 45
        h0 = np.full(n_points, 0.5)
        
    # Set bounds for binary search
    C_low = 0.0
    C_high = 0.38085
    best_C = C_high
    best_h = h0.copy()
    
    # Binary search for minimal C
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 30:
        iteration += 1
        C = (C_low + C_high) / 2.0
        
        # Formulate the LP
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # Compute shift index
            idx = np.arange(n_points)
            shift_idx = (idx + k) % n_points
            # Dot product corresponds to overlap integral
            overlap = cp.sum(cp.multiply(h, 1 - h[shift_idx]))
            constraints.append(overlap <= C * n_points)
            
        # Objective is to minimize C (implicit in the binary search)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Solve the LP
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            # Check feasibility
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h.value
                # Validate solution
                if (np.all(h_sol >= 0) and np.all(h_sol <= 1) and 
                    abs(np.sum(h_sol) - n_points / 2.0) < 1e-4):
                    # Update best solution
                    C_high = C
                    best_C = C
                    best_h = h_sol.copy()
                else:
                    C_low = C
            else:
                C_low = C
        except:
            C_low = C
            
    # Double-check final solution
    dx = 2.0 / n_points
    overlaps = np.correlate(best_h, 1 - best_h, mode="full")
    computed_C = np.max(overlaps) * dx
    if computed_C < best_C:
        best_C = computed_C
    
    return (best_h, best_C, n_points)
