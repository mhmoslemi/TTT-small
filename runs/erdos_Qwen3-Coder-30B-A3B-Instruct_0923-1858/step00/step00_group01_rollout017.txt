import numpy as np

from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def repair_h(h, dx):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    target_sum = len(h) / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def init_population(n_points, pop_size, initial_h=None):
    """Initialize population"""
    population = []

        population.append(repair_h(initial_h.copy(), 2.0 / n_points))

        h = np.random.random(n_points)
        h = repair_h(h, 2.0 / n_points)
        population.append(h)
    return population

def crossover(parent1, parent2):
    """Blend crossover with repair"""
    alpha = np.random.random()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.05):
    """Mutate with repair"""
    mask = np.random.random(len(h)) < mutation_rate
    h = h.copy()
    h[mask] += np.random.normal(0, 0.1, np.sum(mask))
    return repair_h(h, 2.0 / len(h))
