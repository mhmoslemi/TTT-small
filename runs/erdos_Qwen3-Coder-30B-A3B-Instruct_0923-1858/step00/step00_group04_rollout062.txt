def evaluate_h(h):
    """Evaluate C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # Max correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def is_feasible(h):
    """Check if h is feasible"""
    n = len(h)
    return (np.all(h >= 0) and np.all(h <= 1) and
            abs(np.sum(h) - n / 2.0) < 1e-10)

def generate_random_h(n):
    """Generate a random feasible h"""
    h = np.random.rand(n)
    # Scale to meet sum constraint
    h = h * (n / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2, crossover_prob=0.7):
    """Blend crossover with constraint awareness"""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    # Blend crossover
    alpha = random.random()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    
    # Ensure feasibility
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    # Rescale to satisfy sum constraint
    child1 = child1 * (n / 2.0) / np.sum(child1)
    child2 = child2 * (n / 2.0) / np.sum(child2)
    
    return child1, child2
