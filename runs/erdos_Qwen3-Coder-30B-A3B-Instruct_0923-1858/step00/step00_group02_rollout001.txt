import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, else create triangular
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create triangular function
        x = np.linspace(0, 2, n_points)
        h = 1.0 - np.abs(x - 1.0)
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    perturbation_rate = 0.05
    perturbation_interval = 50
    lr_decay = 0.95
    decay_interval = 100
    
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute correlation for gradient approximation
        # We need to optimize max_k int h(x)(1-h(x+k)) dx
        # Since we're trying to minimize max_k, we can look at gradients w.r.t each h[i]
        
        # Compute all correlations with 1-h for each shift
        ones_minus_h = 1.0 - h
        full_corr = signal.correlate(h, ones_minus_h, mode='full')
        # Corresponding shifts
        shifts = np.arange(-n_points + 1, n_points)
        # Find maximum
        max_k_idx = np.argmax(full_corr)
        max_k = shifts[max_k_idx]
        c5_current = full_corr[max_k_idx] * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Simple gradient approximation
        # Using finite differences, we compute gradient:
        # d/dh_i of max_k int h(x)(1-h(x+k)) dx
        # This is approximated using correlation of h and 1-h
        # For each position i, the gradient contribution comes from 
        # where 1-h contributes to the position (i+k shifted)
        # This is essentially the derivative of the correlation
        
        # To get gradient, we compute average correlation derivative
        # Most efficient way: use the standard correlation difference
        # We approximate gradient as the derivative of the max correlation
        # A more direct way is to compute finite difference gradient
        
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - c5_current) / eps

        # Update step
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally perturb
        if iter_count % perturbation_interval == 0:
            noise = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally reduce learning rate
        if iter_count % decay_interval == 0:
            learning_rate *= lr_decay
            
    return (best_h.tolist(), best_c5, len(best_h))

def compute_c5(h, dx):
    # Equivalent to computing max_k int h(x)(1-h(x+k)) dx  
    # So we compute all correlations for all k
    ones_minus_h = 1.0 - h
    full_corr = np.correlate(h, ones_minus_h, mode="full")
    max_correlation = np.max(full_corr)
    return max_correlation * dx
