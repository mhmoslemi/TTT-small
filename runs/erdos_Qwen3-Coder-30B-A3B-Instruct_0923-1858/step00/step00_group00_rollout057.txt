import numpy as np
from scipy import signal
import time

def evaluate_h(h):
    """Evaluate the overlap integral C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value gives the C_5 bound
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h onto constraints: [0,1] and sum(h) = n/2"""
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def nelder_mead_optimize(initial_h, budget_s):
    """Optimize using Nelder-Mead simplex algorithm"""
    start_time = time.time()
    n_points = len(initial_h)
    h = initial_h.copy()
    
    # Parameters for Nelder-Mead
    reflection = 1.0
    expansion = 2.0
    contraction = 0.5
    shrink = 0.5
    initial_simplex_size = 0.1
    
    # Initialize simplex
    simplex = [h.copy()]
    for i in range(n_points):
        pert = np.zeros(n_points)
        pert[i] = initial_simplex_size
        simplex.append(project_h(h + pert, n_points))
    
    # Compute function values for all simplex points
    values = [evaluate_h(vertex) for vertex in simplex]
    
    # Keep track of best solution
    best_idx = np.argmin(values)
    best_h = simplex[best_idx].copy()
    best_c5 = values[best_idx]
    
    reset_count = 0
    max_reset_attempts = 10
    stagnation_counter = 0
    stagnation_threshold = 50
    
    iterations = 0
    max_iterations = 5000
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        # Sort vertices by function value (ascending)
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Check if we improve
        current_best = values[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = simplex[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Reset simplex if stagnation
        if stagnation_counter >= stagnation_threshold and reset_count < max_reset_attempts:
            # Shrink simplex size
            initial_simplex_size *= 0.5
            simplex = [best_h.copy()]
            for i in range(n_points):
                pert = np.zeros(n_points)
                pert[i] = initial_simplex_size
                simplex.append(project_h(best_h + pert, n_points))
            values = [evaluate_h(vertex) for vertex in simplex]
            reset_count += 1
            stagnation_counter = 0
            
        if reset_count >= max_reset_attempts:
            break
            
        # Calculate centroid (excluding the worst point)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        reflected_point = centroid + reflection * (centroid - simplex[-1])
        reflected_point = project_h(reflected_point, n_points)
        reflected_value = evaluate_h(reflected_point)
        
        # Check if reflected point is better
        if values[0] <= reflected_value < values[-2]:
            simplex[-1] = reflected_point
            values[-1] = reflected_value
        # Expansion if better
        elif reflected_value < values[0]:
            expanded_point = centroid + expansion * (reflected_point - centroid)
            expanded_point = project_h(expanded_point, n_points)
            expanded_value = evaluate_h(expanded_point)
            if expanded_value < reflected_value:
                simplex[-1] = expanded_point
                values[-1] = expanded_value
            else:
                simplex[-1] = reflected_point
                values[-1] = reflected_value
        # Contraction
        else:
            contracted_point = centroid + contraction * (simplex[-1] - centroid)
            contracted_point = project_h(contracted_point, n_points)
            contracted_value = evaluate_h(contracted_point)
            if contracted_value < values[-1]:
                simplex[-1] = contracted_point
                values[-1] = contracted_value
            # Shrink
            else:
                for i in range(1, n_points + 1):
                    simplex[i] = simplex[0] + shrink * (simplex[i] - simplex[0])
                    simplex[i] = project_h(simplex[i], n_points)
                values = [evaluate_h(vertex) for vertex in simplex]
        
        iterations += 1
    
    return best_h, best_c5
