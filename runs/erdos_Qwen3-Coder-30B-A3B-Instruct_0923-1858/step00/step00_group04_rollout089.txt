import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation: h * (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value, scaled by dx
    c5 = np.max(corr) * dx
    return c5

def normalize_h(h):
    """Ensure h sums to n/2 while staying in [0,1]."""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-adjust to target sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find the largest value and decrease it by diff
        max_idx = np.argmax(h)
        h[max_idx] = max(0, h[max_idx] - diff)
    return h

def mutate_h(h, mutation_rate=0.1, epsilon=0.1):
    """Mutate a function h."""
    n = len(h)
    h_new = h.copy()
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Perturb by small amount
            delta = np.random.normal(0, epsilon)
            h_new[i] += delta
    # Normalize to constraints
    return normalize_h(h_new)

def crossover_h(h1, h2, crossover_rate=0.7):
    """Blend crossover of two functions."""
    if np.random.random() > crossover_rate:
        return h1.copy()
    n = len(h1)
    mask = np.random.random(n) > 0.5
    h_new = np.where(mask, h1, h2)
    return normalize_h(h_new)

def initialize_population(n_points, initial_h=None, pop_size=30):
    """Initialize a population with feasible step functions."""
    population = []
    if initial_h is not None:
        population.append(initial_h)
    
    # Fill the rest with random feasible points
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = normalize_h(h)
        population.append(h)
    
    return population

        # Track best
