import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_overlap(h):
    # Compute the overlap integral using convolution
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value and multiply by dx
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h = np.clip(h, 0, 1)

    dx = 2.0 / n_points

    def objective(h_vec):

        # Ensure constraints
        h_local = np.clip(h_local, 0, 1)
        # Normalize
        h_local = h_local * (1.0 / np.sum(h_local) * dx)
