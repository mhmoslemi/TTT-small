import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a reasonable starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[-1] = 0.4
        h[-2] = 0.6
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy sum constraint
    h = h * target_sum / np.sum(h)
    
    # Early stopping threshold
    min_improvement = 1e-6
    last_c5 = float('inf')
    
    # Optimization parameters
    alpha = 0.1
    decay_factor = 0.9
    decay_interval = 100
    
    max_iter = 10000
    for itr in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlap integrals
        # Correlate h with (1 - h)
        full_corr = correlate(h, 1 - h, mode='full')
        # Extract the relevant part (shifts from -n+1 to n-1) = range of positions
        mid = len(full_corr) // 2
        overlaps = full_corr[mid - n_points + 1:mid + n_points - 1]
        
        # Compute the maximum overlap (multiply by dx)
        c5 = np.max(overlaps) * dx
        
        if np.isnan(c5) or np.isinf(c5):
            break
            
        # Check for convergence
        if abs(last_c5 - c5) < min_improvement:
            break
        last_c5 = c5
        
        # Compute gradient corresponding to max overlap
        max_k = np.argmax(overlaps)
        # max_k represents shift, in indices: shift_k = max_k - (n_points - 1)
        shift_k = max_k - (n_points - 1)
        # Gradient of C5 w.r.t. h[i]: 
        # When i + shift_k is valid: (1 - h[i + shift_k])
        # Overlap is sum over i of h[i] * (1 - h[i + shift_k])
        # So_grad = (1 - h[i + shift_k]) for valid i 
        # and for i + shift_k: -h[i] (but needs adjustment)
        # But actually, to build the gradient with full overlap being max_k:
        # We are optimizing: sum over all i: h[i] * (1 - h[i + k*])
        # Grad for h[i] = (1 - h[i + k*])  (if i + k* >= 0 and i + k* < n)
        #                 0                (otherwise)
        # Grad for h[i + k*] = -h[i] 
        # For each k*, find the valid overlaps
    
        # Build gradient vector
        grad = np.zeros(n_points)
        if shift_k >= 0:
            # h[i] contributes to h[i + shift_k] term
            grad[:n_points-shift_k] += (1 - h[shift_k:])
            # h[i + shift_k] contributes to h[i] term
            grad[shift_k:] -= h[:-shift_k]
        else:
            # h[i] contributes to term: h[i + shift_k] where shift_k < 0, so we are looking at negative shift
            grad[-shift_k:] += (1 - h[:n_points+shift_k])
            grad[:n_points+shift_k] -= h[-shift_k:]
        
        # Update h
        h = h - alpha * grad
        
        # Project h back
        # Clamp to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Adjust sum to be target_sum
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Simple adjustment to keep sum
            adjustment = (target_sum - current_sum) / n_points
            h = h + adjustment
            h = np.clip(h, 0.0, 1.0) 
            
        # Apply step size decay every 100 iterations
        if (itr + 1) % decay_interval == 0:
            alpha *= decay_factor
            if alpha < 1e-6:
                alpha = 1e-6
    
    # Final evaluation
    full_corr = correlate(h, 1 - h, mode='full')
    mid = len(full_corr) // 2
    overlaps = full_corr[mid - n_points + 1:mid + n_points - 1]
    final_c5 = np.max(overlaps) * dx
    
    # Return h as a list for compatibility
    return (h.tolist(), float(final_c5), n_points)
