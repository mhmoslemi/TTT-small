import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the known construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a reasonable starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    start_time = time.time()
    
    # Gradient descent parameters
    alpha = 0.1
    decay_rate = 0.9
    min_alpha = 1e-6
    iter_decay = 100
    tol = 1e-5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        if iteration % iter_decay == 0:
            alpha = max(alpha * decay_rate, min_alpha)
        
        # Compute all correlations
        corr_full = signal.correlate(h, 1 - h, mode="full")
        k_shifts = np.arange(-n_points + 1, n_points)
        max_idx = np.argmax(corr_full)
        k_max = k_shifts[max_idx]
        
        # Compute gradient for the max correlation case
        grad = np.zeros_like(h)
        if k_max >= 0:
            # h[i] contributes (1 - h[i + k_max])  
            # h[i + k_max] contributes -h[i]
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] += (1 - h[i + k_max])
                if i - k_max >= 0:
                    grad[i - k_max] -= h[i]
        else:
            # k_max < 0, so h[i] contributes (1 - h[i + k_max])
            # h[i + k_max] contributes -h[i]
            for i in range(n_points):
                if i - k_max < n_points:
                    grad[i] += (1 - h[i - k_max])
                if i + k_max >= 0:
                    grad[i + k_max] -= h[i]
        
        # Update
        h_new = h - alpha * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to meet sum constraint
        current_sum = np.sum(h_new) * dx
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Distribute the adjustment evenly in proportion to current values
            adjust_factor = diff / np.sum(h_new)
            h_new = h_new + adjust_factor * h_new
            h_new = np.clip(h_new, 0, 1)
            
        # Re-adjust sum
        current_sum = np.sum(h_new) * dx
        if abs(current_sum - target_sum) > 1e-10:
            # Simple redistribution
            scale = target_sum / current_sum
            h_new = h_new * scale
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early stopping
        if iteration > 100 and abs(c5 - best_c5) < tol:
            break
    
    # Final rounding step (optional) for binary construction
    # but no need since we keep continuous values for the metric
    
    return (best_h.tolist(), best_c5, n_points)
    
def evaluate_erdos_solution(h, dx):
    # Compute overlap integrals for all shifts
    h_arr = np.array(h)
    corr_full = signal.correlate(h_arr, 1 - h_arr, mode="full")
    # Get all valid overlaps (k from -n+1 to n-1)
    k_shifts = np.arange(-len(h_arr) + 1, len(h_arr))
    # We want the maximum among valid correlations:
    # valid k range is [-n+1, n-1], excluding 0 where correlation is zero
    valid_correlations = corr_full[len(h_arr)-1:]
    # The valid ones are indices from len(h_arr) - 1 to 2*len(h_arr) - 2 
    max_corr = np.max(valid_correlations)
    return max_corr * dx
