def evaluate_erdos_solution(h_values):
    """Returns C5 bound for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take max and multiply by dx
    c5 = np.max(corr) * dx
    return c5
