import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant C5 for given step function h"""
    n = len(h)
    # Create the convolution with 1-h
    h_flipped = np.flip(1 - h)
    # Full convolution (this computes all k values)
    corr = np.convolve(h, h_flipped, mode='full')
    # The full convolution is of length 2*n-1
    # The valid range for k is from -(n-1) to n-1
    # We extract the overlap values for each shift k
    # Specifically, we want for k from 0 to 2*n-2 (but k=0 to n-1 is sufficient due to symmetry)
    # Actually, overlap corresponds to the values where we have valid overlaps
    # We compute correlation for all shifts
    # But to minimize max over k, we focus on the maximum correlation term
    
    # Convert back to equivalent of what's computed in the problem
    # For shift k = 0, the correlation is sum(h*(1-h)) for k = 1, etc.
    # Blocking with full mode, the result length is 2*n-1
    # Indices 0 through 2*n-2
    # The correlation (sum of overlaps) for k = 0 to 2*n-2 (shifts)
    # The number of such shifts is 2*n - 1
    # We want to take the maximum of all overlaps for any k ≥ 0  
    # So extract only valid k shifts (0 to n-1)
    # But the integral is over [0,2] with dx step
    # and we're considering all possible k shifts from 0 to 2 
    max_overlap = 0
    # For k from 0 to 2, we have samples every dx
    # The full convolution gives the overlaps for all possible positions
    # From the problem: return max_k of integral over overlap
    # We compute the overlap for all possible lags and return max
    for k in range(len(corr)):
        corr_k = corr[k]
        if corr_k > max_overlap:
            max_overlap = corr_k
    
    return max_overlap * dx

def compute_C5(h, dx):
    """Compute C5 directly as the sup over k of the overlap integral"""
    n = len(h)
    # Compute the overlaps for shift k in [0, n-1]
    # Use full convolution
    h_flip = np.flip(1 - h)
    # Full convolution of h with (1-h)
    overlaps = np.convolve(h, h_flip, mode='full')
    # Now overlaps contains correlation values for all lags from -(n-1) to n-1
    # The total overlap integral for each k is overlaps[2*(n-1) - k] 
    # since indices go from 0 to 2*(n-1)
    # So the overlap for shift k in [0,2*n-2] should be at index k+ (n-1) in full mode
    # Let's just take index for k=0 to n-1, which includes all relevant overlaps
    max_overlap = 0
    for k in range(len(overlaps)):
        max_overlap = max(max_overlap, overlaps[k])
    return max_overlap * dx

def setup_initial_h(n_points):
    """Initialize a feasible h with sum(h)*dx = 1"""
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0[:n_points]
        # renormalize if needed
        current_sum = np.sum(h0)
        h0 *= (n_points/2) / current_sum

    else:
        # Create a random starting point that sums to n_points/2
        h0 = np.random.rand(n_points)
        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        if current_sum > 0:
            h0 *= (n_points / 2) / current_sum
        return h0

def objective(h):
    """Objective to minimize (C5)"""
    dx = 2.0 / len(h)
    # Compute directly via correlation
    h_flip = np.flip(1 - h)
    overlaps = np.convolve(h, h_flip, mode='full')
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def constraint_sum(h):
    """Constraint that sum(h) = n_points / 2"""
    return np.sum(h) - (len(h) / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70  # BALANCE: resolution vs speed
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or creating a random valid one
    h = setup_initial_h(n_points)
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Now build the constraint dictionary for SLSQP
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]

    # Initial objective 
    best_C5 = objective(h)
    best_h = h.copy()
    
    # Use scipy.optimize.minimize with SLSQP
    # In case of tight limits, use callback to stop early
    last_opt_time = time.time()
    
    def callback(x):
        nonlocal last_opt_time
        if time.time() - start_time > budget_s * 0.95:  # early exit check
            return True
        # Update best solution
        current_C5 = objective(x)
        if current_C5 < best_C5:
            best_C5 = current_C5
            global best_h
            best_h = x.copy()
        return False
    
    options = {'maxiter': 10000, 'disp': False}
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s * 0.98:
        try:
            # Get a random restart for diversity
            h_start = setup_initial_h(n_points)
            
            res = minimize(
                objective, 
                h_start, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options=options,
                callback=callback
            )
            
            if res.success or (time.time() - start_time < budget_s * 0.9):
                # Evaluate again to get current value
                current_C5 = objective(res.x)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = res.x.copy()
            
            iter_count += 1
            if iter_count > 50 or time.time() - start_time > budget_s * 0.95:
                break
                
        except Exception as e:
            # If optimization fails, maybe continue with previous best
            break
    
    # Final validation
    final_C5 = compute_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)
