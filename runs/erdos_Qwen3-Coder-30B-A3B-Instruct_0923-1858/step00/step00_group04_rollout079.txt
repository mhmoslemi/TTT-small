import numpy as np
from scipy import signal
import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 constant for given h"""
    # Full correlation: h * (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx and take maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_constraint(h, n_points):
    """Project h to satisfy sum constraint"""
    h_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(h_sum - target_sum) < 1e-10:
        return h
    # Adjust to get target sum
    scale = target_sum / h_sum
    h = h * scale
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Fallback construction
        n_points = 42
        h = np.ones(n_points) * 0.5
        # Set every other index to 0 to maintain balance
        for i in range(0, n_points, 2):
            h[i] = 0
        h = project_to_constraint(h, n_points)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization loop
    iter_count = 0
    perturb_freq = 100
    ε = 0.02
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        corr_scaled = corr * dx
        max_idx = np.argmax(corr_scaled)
        max_val = np.max(corr_scaled)
        
        # Indices in the full correlation that produce max overlap
        # The peak corresponds to some lag k
        lag = max_idx - (n_points - 1)
        
        # Create a smooth adjustment based on where this peak occurs
        new_h = h.copy()
        
        # Identify where this peak occurs to reduce it
        # Reduce mass near the peak region
        if lag >= 0:
            start_idx = lag
            end_idx = min(start_idx + 10, n_points)
        else:
            start_idx = 0
            end_idx = max(0, -lag + 10)
            
        # Perturb around the area where maximum occurs
        for i in range(start_idx, min(end_idx, n_points)):
            new_h[i] = max(0, h[i] - ε)
        
        # Add the removed mass to low-overlap regions
        mass_to_add = ε * (end_idx - start_idx)
        # Find low-overlap regions
        for i in range(n_points):
            if i < start_idx or i >= end_idx:
                new_h[i] = min(1.0, new_h[i] + mass_to_add / (n_points - (end_idx - start_idx)))
        
        # Ensure sum constraint
        new_h = project_to_constraint(new_h, n_points)
        
        # Possibly do more global perturbations
        if iter_count % perturb_freq == 0:
            # Randomly redistribute mass to explore more
            idxs = np.random.choice(n_points, size=n_points//4, replace=False)
            # For simplicity, we just redistribute
            for i in idxs:
                if h[i] > 0.01:
                    # Move small amount away from high values
                    move_amount = min(0.01, h[i])
                    new_h[i] -= move_amount
                    # Add to a random index
                    j = np.random.randint(0, n_points)
                    new_h[j] = min(1.0, new_h[j] + move_amount)
        
        # Final validation and update best
        new_h = np.clip(new_h, 0, 1)
        new_h = project_to_constraint(new_h, n_points)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        h = new_h.copy()
        
        if iter_count % 1000 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break
                
    return (list(best_h), best_c5, n_points)
