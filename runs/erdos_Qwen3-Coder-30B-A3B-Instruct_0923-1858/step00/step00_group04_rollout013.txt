import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def feasibility_check(h):
    """Ensure h is feasible: values in [0,1] and sum = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - 1.0) > 1e-10:
        h = h * (1.0 / h_sum)
    return h

def genetic_search(initial_h_values, n_points, budget_s):
    """Genetic algorithm optimization for C5."""
    np.random.seed(42)
    pop_size = 30
    generations = 1000
    mutation_rate = 0.1
    tournament_size = 5
    elitism_rate = 0.1
    
    # Initialize population
    population = []
    if initial_h_values is not None:
        population.append(initial_h_values.copy())
        population[0] = feasibility_check(population[0])
    
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = feasibility_check(h)
        population.append(h)
    
    global_best = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
            
        # Track best individual
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            global_best = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 200:
            break
            
        # Selection via tournament
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            best_in_tour = min(tournament, key=lambda idx: fitness[idx])
            selected.append(population[best_in_tour].copy())
            
        # Elitism
        elite_count = int(elitism_rate * pop_size)
        elite_indices = np.argsort(fitness)[:elite_count]
        new_pop = [population[i].copy() for i in elite_indices]
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            p1 = np.random.choice(pop_size)
            p2 = np.random.choice(pop_size)
            if p1 != p2 and np.random.rand() < 0.7:
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * selected[p1] + (1 - alpha) * selected[p2]
            else:
                child = selected[np.random.choice(pop_size)].copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.05)
                        
            child = feasibility_check(child)
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
    return global_best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run optimization."""
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 42
        initial_h = np.ones(n_points) * 0.5
        initial_h = feasibility_check(initial_h)
        
    # Genetic optimization
    best_h, best_c5 = genetic_search(initial_h, n_points, budget_s)
    
    # Final evaluation
    dx = 2.0 / n_points
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
