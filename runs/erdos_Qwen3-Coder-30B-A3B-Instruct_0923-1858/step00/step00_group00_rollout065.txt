import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vec):
        # Ensure h in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to ensure sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 500
    no_improve_count = 0
    i = 0
    
    while i < max_iter and time.time() - start_time < budget_s - 1:
        # Perturb
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        h_new = project_h(h_new)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / T):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Cooling schedule
        T *= alpha
        if i % 100 == 0:
            perturb_scale *= 0.99
        
        i += 1
        
        # Early termination if no improvement for a while
        if no_improve_count > no_improve_limit:
            break
    
    return (best_h.tolist(), best_c5, n_points)
