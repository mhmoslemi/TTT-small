import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    """Clip h to [0,1] and rescale to satisfy integral constraint."""
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if abs(total) > 1e-10:
        h = h / total * (1.0 / dx)
    return h

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral C5."""
    n = len(h)
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Correlation is of length 2*n-1, but we only need the relevant part
    # C5 = max_k int h(x) * (1 - h(x+k)) dx
    # For the full correlation, the maximum index is n-1, and the peak 
    # for C5 corresponds to overlapping regions when shifted
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_descent_step(h, dx, lr=0.01):
    """Perform one step of gradient descent on C5."""
    n = len(h)
    # Evaluate cross-correlation and its gradient
    corr = np.correlate(h, 1 - h, mode='full')
    # Gradient w.r.t. h
    # The derivative of correlation with respect to h[i] 
    # depends on convolution with 1-h (shifted)
    # For compactness, we'll do a finite difference approximation
    epsilon = 1e-6
    grad = np.zeros_like(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr)) / epsilon
    
    h_new = h - lr * grad
    h_new = clip_and_rescale(h_new, dx)
    return h_new
