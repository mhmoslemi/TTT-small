import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        # Fallback to a simple construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h_vals):
        # Compute all overlaps
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vals):
        # Clamp to [0,1]
        h_clamped = np.clip(h_vals, 0, 1)
        # Adjust sum constraint
        current_sum = np.sum(h_clamped)
        diff = (n_points / 2) - current_sum
        if abs(diff) > 1e-10:
            # Distribute diff evenly
            h_clamped += diff / n_points
            h_clamped = np.clip(h_clamped, 0, 1)
        return h_clamped
    
    def get_max_overlap_k(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        max_idx = np.argmax(corr)
        # Offset for shift k
        k = max_idx - (n_points - 1)
        return k, corr[max_idx]
    
    current_c5 = evaluate_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Check if we're stuck
        if time.time() - start_time > budget_s * 0.9 and last_improvement > 100:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        # Find shift with maximum overlap
        max_idx = np.argmax(corr)
        k_star = max_idx - (n_points - 1)
        
        # Compute the indices that contribute to this overlap
        # For shift k_star, h[i] and h[i+k_star] contribute to the overlap
        # We want to maximize reduction in overlap for this k*
        max_reduction = -1e10
        best_pair = None
        best_new_h = h.copy()
        
        for i in range(n_points - abs(k_star)):
            if k_star > 0:
                idx1, idx2 = i, i + k_star
            else:
                idx1, idx2 = i - k_star, i
                
            if idx1 >= n_points or idx2 >= n_points:
                continue
                
            # Try changing h[i] and h[i+k_star]
            h_copy = h.copy()
            change = 0.01
            
            # Try small adjustments
            for dh1 in [-change, change]:
                for dh2 in [-change, change]:
                    h1 = h[i] + dh1
                    h2 = h[i + k_star] + dh2
                    if 0 <= h1 <= 1 and 0 <= h2 <= 1:
                        h_copy[i] = h1
                        h_copy[i + k_star] = h2
                        new_c5 = evaluate_c5(h_copy)
                        reduction = current_c5 - new_c5
                        if reduction > max_reduction:
                            max_reduction = reduction
                            best_pair = (i, i + k_star)
                            best_new_h = h_copy.copy()
        
        if max_reduction > 0 and max_reduction > 1e-8:
            h = project_h(best_new_h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            # Small random perturbations to avoid local minima
            h_per = h.copy()
            idxs = np.random.choice(n_points, size=min(5, n_points), replace=False)
            for idx in idxs:
                h_per[idx] += np.random.normal(0, 0.005)
            h_per = project_h(h_per)
            new_c5 = evaluate_c5(h_per)
            if new_c5 < current_c5:
                h = h_per
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    last_improvement = 0
            last_improvement += 1
        
    return list(best_h), best_c5, n_points
