import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Evaluate the maximum overlap integral
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = correlate(h, 1-h, mode="full")
    max_correlation = np.max(np.abs(correlations))
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    max_iter = 10000
    alpha = 0.01
    perturbation_rate = 0.1
    perturbation_interval = 100
    best_C5 = float('inf')
    best_h = None
    
    # Initialize solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values.copy())
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization loop
    for iteration in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Compute gradient
        # This is an approximation; exact gradient is complex for cross-correlation
        # We work with numerical gradient through small perturbations
        C5_ref = evaluate_erdos_solution(h)
        gradient = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            C5_plus = evaluate_erdos_solution(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            C5_minus = evaluate_erdos_solution(h_minus)
            
            gradient[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Update step
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Apply random perturbation periodically
        if iteration % perturbation_interval == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if iteration > 0 and iteration % 1000 == 0:
            alpha *= 0.95
        
        # Ensure reasonable step size
        alpha = max(alpha, 1e-6)
    
    # Final evaluation of best solution
    if best_h is not None:
        final_C5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), final_C5, n_points)
    else:
        # Return current if no better found, but also evaluate
        final_C5 = evaluate_erdos_solution(h)
        return (h.tolist(), final_C5, n_points)
