import numpy as np
from scipy import linalg
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.1
    pert_freq = 100
    pert_strength = 0.05
    rank_approx = 10
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Create a balanced start: 0.5 for even indices, 0 for odd, then rescale
        h = np.array([0.5 if i % 2 == 0 else 0.0 for i in range(n_points)])
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    # Precompute the full correlation matrix for optimization
    # We'll approximate this efficiently
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        # Using full correlation (equivalent to what's expected by evaluator)
        corr = np.correlate(h, 1-h, mode="full")
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # DC-Frank-Wolfe step
        # Linearization around current h: 
        # We want dir_deriv = max_k ∫ h[i] * (1 - h[(i+k) % n_points]) dx = max_k [A(k) - B(k)]
        # A(k) = ∑ h[i] (constant) 
        # B(k) = ∑ h[i] * h[(i+k) % n_points]
        # In fact, we can compute -d/dh B(k) = -h[(i+k) % n_points]
        # But we need to think of this in terms of gradients for direction search
        
        # Compute expected gradient: ∇_h B(k) = h[(i-k) % n_points]
        # So ∇_h [sum(h[i]) - sum(h[i]*h[i+k])] 
        # = 1 - h[(i-k) % n_points], max_k over all k
        # Since B = sum h[i] h[i+k], we compute ∇B with respect to h[j] 
        # When j = i → ∇B/h[i] = h[i+k]
        # But we look over all k, so this is more complex
        
        # Instead, compute grad_a = 1 (constant or sum of 1) = 1 for all i  
        # grad_b = solve max_k sum h[i] h[i+k] by variational approach:
        # ∇_h (sum_i h[i] (sum_k h[i+k])) = sum_k h[i+k]  
        # (but need to be careful of indexing; for optimizing sum_i h[i] h[i+k])
        # So gradient of B with respect to h[i] = sum_k h[i+k] = convolution h * 1
        # This insight helps us approximate:
        
        # For efficiency, we'll use samples from correlation matrix to compute gradient
        
        # Gradient approx for current point
        # We sample one k (say 0 for now) to estimate gradient:
        # ∇C5 approximated by a subset of all.
        # So we'll compute for a more clever approach:
        # We approximate the gradient by computing the worst-case correlation one
        
        # Sample a few directions for gradient estimation
        grad_sum = np.zeros_like(h)
        
        # Sample to estimate gradient directions, by using correlation matrix with random shifts
        for _ in range(5):
            # Choose a random k
            k = np.random.randint(0, n_points)
            shifted_h = np.roll(h, k) 
            grad_sum += shifted_h
    
        # Note: we're making approximation - linearization takes ∇B = h[k] so
        # Actually the direction d that minimizes (c_5 -  gradient^T d) 
        # With constraints that d is set to meet boundary conditions.
        # However, since we are iterating, we can take a step in the direction  
        # that better satisfies k = argmax_k int h * (1 - h_{+k})
        
        # Compute closest adverse correlation direction by brute force on small subset
        # Ref: find the direction that reduces max correlation the most in one step
        
        # Try guided step:  
        current_corr = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(current_corr)
        best_k = max_idx - (n_points - 1)
        
        # Simple gradient ascent step on one k direction
        # Stop if two extremes are close or no improvement
        
        # compute∇h[sum h[i] * h[i+k]] = h[i+k] (positive gradient w.r.t. h[i]) 
        # Gradient of sum h[i](1-h[i+k]) = 1 - h[i+k]  
        # we have ∇_h (sum h[i] - h[i]*h[i+k])  
        # = ∇ ( sum h[i] - (sum h[i] h[i+k]) ) = 1 - [h[i+k]]
        # This is J^T q, and q = h, so ∇ = 1 - h_k (conveniently)
        
        # Let's make d = 1 - h_k (only force computation), but define direction carefully
        d = np.zeros_like(h)
        # Look for worst k, and help fix that gradient direction
        # Direction of decrease is one such that d[i] = 1 - h[i + best_k] ?
        k_ind = best_k % n_points
        shifted_h = np.roll(h, k_ind)
        d = 1.0 - shifted_h
        
        # Frank-Wolfe style step
        step_size = lr
        h_new = h - step_size * d
        h_new = np.clip(h_new, 0, 1)  # Projection
        
        # Adjust to ensure sum constraint:  
        mean_h = np.sum(h_new) / n_points
        h_new = h_new - mean_h + 0.5  # Ensures sum = n_points/2
    
        h = h_new
        
        # Periodic perturbation
        if iteration % pert_freq == 0 and iteration > 0:
            noise = np.random.normal(0, pert_strength, n_points)
            h = np.clip(h + noise, 0, 1)
            mean_h = np.sum(h) / n_points
            h = h - mean_h + 0.5
                
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
