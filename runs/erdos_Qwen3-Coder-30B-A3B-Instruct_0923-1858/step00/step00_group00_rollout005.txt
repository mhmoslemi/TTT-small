import numpy as np
from scipy import signal
import time

def evaluate_h(h):
    """Evaluate the overlap integral C5 for a given step function h."""
    # Compute the correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode="full")
    # Scale by dx to get actual integral value
    dx = 2.0 / len(h)
    c5_val = np.max(corr) * dx
    return c5_val

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum(h) = n_points/2."""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def nelder_mead_optimize(initial_h, max_iter=5000, budget_s=600):
    """Optimize h using Nelder-Mead with adaptive simplex"""
    n_points = len(initial_h)

    simplex_size = 0.1

    alpha = 1.0    # reflection
    gamma = 2.0    # expansion
    beta = 0.5     # contraction
    rho = 0.5      # shrink

    best_h = initial_h.copy()

    # Create initial simplex
    simplex = [initial_h.copy()]
    for i in range(n_points):
        v = initial_h.copy()
        v[i] += simplex_size * (np.random.random() - 0.5)
        simplex.append(v)
    
    start_time = time.time()
    reset_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Project all vertices
        for i in range(len(simplex)):
            simplex[i] = project_h(simplex[i], n_points)
        
        # Evaluate all
        values = [evaluate_h(v) for v in simplex]
        order = np.argsort(values)

        # Track best
        if values[0] < best_c5:

        if iteration - last_improvement > 50 and reset_count < 10:

            # Resample simplex around best
            simplex = [best_h.copy()]

                simplex.append(v)
            continue
        
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_r = centroid + alpha * (centroid - simplex[-1])
        h_r = project_h(h_r, n_points)
        c_r = evaluate_h(h_r)
        
        if values[0] <= c_r < values[-2]:
            simplex[-1] = h_r
            continue
        elif c_r < values[0]:
            # Expansion
            h_e = centroid + gamma * (h_r - centroid)
            h_e = project_h(h_e, n_points)
            c_e = evaluate_h(h_e)
            if c_e < c_r:
                simplex[-1] = h_e
            else:
                simplex[-1] = h_r
        else:
            # Contraction
            if c_r < values[-1]:
                h_c = centroid + beta * (h_r - centroid)
                h_c = project_h(h_c, n_points)
                c_c = evaluate_h(h_c)
                if c_c <= c_r:
                    simplex[-1] = h_c
                    continue
            # Shrink
            for i in range(1, len(simplex)):
                simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
                simplex[i] = project_h(simplex[i], n_points)
    
    return best_h, best_c5
