import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to meet constraint
    h_sum = np.sum(h)
    h = h * (target_sum / h_sum)
    
    # Precompute 1-h for efficiency
    one_minus_h = 1.0 - h
    
    # Gradient descent parameters
    max_iter = 10000
    lr = 0.01
    tol = 1e-6
    
    best_h = h.copy()
    best_c5 = np.inf
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        conv = np.correlate(h, one_minus_h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        c5_vals = conv * dx
        
        # Find the maximum
        max_idx = np.argmax(c5_vals)
        max_c5 = c5_vals[max_idx]
        
        if max_c5 < best_c5:
            best_h = h.copy()
            best_c5 = max_c5
        
        # Compute gradient wrt to h for the maximizing lag
        k = lags[max_idx]
        
        # Correlation gradient: d/dh_i (sum over j of h[j] * (1 - h[j+k]))
        grad = np.zeros_like(h)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1.0 - h[j + k]) * dx
            if 0 <= j - k < n_points and k != 0:
                grad[j - k] += (-h[j]) * dx
                
        # Update h
        h = h - lr * grad
        
        # Projection: clamp and scale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (target_sum / h_sum)
            
    # Final evaluation
    final_conv = np.correlate(best_h, 1.0 - best_h, mode='full')
    final_lags = np.arange(-n_points + 1, n_points)
    final_c5_vals = final_conv * dx
    final_c5 = np.max(final_c5_vals)
    
    return (best_h.tolist(), final_c5, n_points)
