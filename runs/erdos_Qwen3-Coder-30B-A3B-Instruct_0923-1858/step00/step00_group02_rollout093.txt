import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    conv = np.correlate(h, 1-h, mode="full")
    # The maximum overlap integral is what we want to minimize
    c5 = np.max(conv) * dx
    return c5

    dx = 2.0 / n_points
    h0 = h0 * (1.0 / np.sum(h0)) / dx

    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Ensure the constraint ∫h = 1
        h_normalized = h * (1.0 / np.sum(h)) / dx
        return evaluate_h(h_normalized)
