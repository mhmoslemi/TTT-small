import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # So, for i, partial wrt h[i] involves terms where h[i] appears.
        # Now this is tricky in full setup; for now just implement with 
        # primary gradient of correlation
        # For a single shift that contributes: 
        # If i -> i+k contributes, then dC5/dh[i] is 1 - h[i+k]

        # Approximate gradient at indices by looking at forward/backward difference
