import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap value
    max_overlap = np.max(corr)
    c5 = max_overlap * dx
    return c5

def initialize_population(n_points, n_pop=30):
    """Initialize population of step functions."""
    pop = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Normalize to have correct sum
        h0 = h0 * (n_points / 2) / np.sum(h0)
        pop.append(h0)
    else:
        # Fallback to simple pattern
        h0 = np.ones(n_points) * 0.5
        pop.append(h0)
    
    # Fill with random feasible solutions
    for i in range(n_pop - len(pop)):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        pop.append(h_rand)
    
    return pop

def mutate_individual(h, mutation_rate=0.1, epsilon=0.1):
    """Mutate an individual by adding small random changes."""
    h_new = h.copy()
    n = len(h)
    n_mutate = max(1, int(n * mutation_rate))
    indices = np.random.choice(n, n_mutate, replace=False)
    
    for i in indices:
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[i] += delta
        # Ensure h stays within [0,1]
        h_new[i] = np.clip(h_new[i], 0, 1)
    
    # Rescale to maintain sum = n/2
    h_new = h_new * (n / 2) / np.sum(h_new)
    return h_new

def crossover_individuals(h1, h2, crossover_rate=0.7):
    """Blend crossover between two individuals."""
    if np.random.rand() > crossover_rate:
        return h1.copy(), h2.copy()
        
    n = len(h1)
    alpha = np.random.rand()
    h1_new = alpha * h1 + (1 - alpha) * h2
    h2_new = (1 - alpha) * h1 + alpha * h2
    
    # Ensure h stays within [0,1] and rescale sum
    h1_new = np.clip(h1_new, 0, 1)
    h2_new = np.clip(h2_new, 0, 1)
    h1_new = h1_new * (n / 2) / np.sum(h1_new)
    h2_new = h2_new * (n / 2) / np.sum(h2_new)
    
    return h1_new, h2_new

def select_parents(population, fitnesses, tournament_size=5):
    """Select parents using tournament selection."""
    n = len(population)
    parents = []
    
    # Get indices sorted by fitness
    sorted_indices = np.argsort(fitnesses)
    
    # Add best individuals (elite)
    elite_size = max(1, n // 10)
    parents.extend([population[i] for i in sorted_indices[:elite_size]])
    
    # Tournament selection
    for _ in range(n - elite_size):
        tournament = np.random.choice(n, tournament_size, replace=False)
        winner_idx = tournament[np.argmin([fitnesses[i] for i in tournament])]
        parents.append(population[winner_idx])
    
    return parents

def genetic_algorithm(n_points, n_pop=30, max_generations=1000, 
                      mutation_rate=0.1, crossover_rate=0.7, 
                      time_budget=600):
    """Run genetic algorithm optimization."""
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, n_pop)
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > time_budget - 5:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_h(h) for h in population]
        
        # Update global best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Create new population
        new_population = parents[:n_pop]  # Keep elites
        
        # Fill rest via crossover and mutation
        while len(new_population) < n_pop:
            p1, p2 = np.random.choice(len(parents), 2, replace=False)
            h1, h2 = crossover_individuals(parents[p1], parents[p2], crossover_rate)
            h1 = mutate_individual(h1, mutation_rate)
            h2 = mutate_individual(h2, mutation_rate)
            new_population.extend([h1, h2])
        
        population = new_population[:n_pop]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Entry point for optimization."""
    np.random.seed(seed)
    n_points = 100  # A moderate value to keep things within budget
    
    h_values, c5_bound = genetic_algorithm(n_points, time_budget=budget_s)
    
    # Final validation
    dx = 2.0 / len(h_values)
    sum_h = np.sum(h_values) * dx
    assert abs(sum_h - 1.0) < 1e-10, "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_values), "Domain constraint violated"
    
    return h_values, c5_bound, n_points
