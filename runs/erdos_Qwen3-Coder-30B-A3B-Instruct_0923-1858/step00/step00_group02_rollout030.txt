import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular construction
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * n_points / 2  # Adjust to sum to n_points/2
    
    dx = 2.0 / n_points
    
    # Initialize trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def evaluate_c5(h_vals):
        # Compute overlap integral
        correlation = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Only take the valid part
        c5 = np.max(correlation) * dx
        return c5
    
    def project_h(h_vals):
        # Ensure h is in [0,1]
        h_proj = np.clip(h_vals, 0, 1)
        # Rescale to meet sum constraint
        h_sum = np.sum(h_proj)
        if h_sum > 0:
            h_proj = h_proj * (n_points / 2) / h_sum
        return h_proj
    
    def compute_gradient(h_vals, epsilon=1e-4):
        grad = np.zeros_like(h_vals)
        c5_base = evaluate_c5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = project_h(h_plus)
            h_minus = project_h(h_minus)
            grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * epsilon)
        return grad
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate objective
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Perturb if no improvement
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = project_h(h)
            trust_radius = 0.1  # Reset trust radius
            patience_counter = 0
            continue
            
        # Compute gradient
        try:
            grad = compute_gradient(h)
        except Exception:
            # Revert to a simpler strategy if gradient computation fails
            h = project_h(h + 0.01 * (np.random.rand(n_points) - 0.5))
            continue
        
        # Compute step using trust region
        # Approximate Hessian with identity for simplicity
        step = -grad / (np.linalg.norm(grad) + 1e-8)
        step_norm = np.linalg.norm(step)
        
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        # Try the step
        h_trial = h + step
        h_trial = project_h(h_trial)
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < c5:
            h = h_trial
            trust_radius = min(trust_radius * 1.5, 0.5)
        else:
            trust_radius = max(trust_radius * 0.5, 1e-5)
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
