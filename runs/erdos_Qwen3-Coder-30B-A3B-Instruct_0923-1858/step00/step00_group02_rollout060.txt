import numpy as np
from scipy import signal
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up problem parameters
    n_points = 45  # Standard resolution
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
        
    # Initialize best solution
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    trust_radius = 0.1
    rho = 0.1
    max_backtrack = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h, dx)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        # Compute expected values for linearization
        expected_h = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h, k)
            expected_h += shifted / n_points
            
        # Create approximate objective using first-order Taylor
        # We'll approximate using linearized cross-correlation
        # Using correlation of h and (1 - h)
        
        # Compute approximate gradients
        grad = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h, k)
            grad += (1 - 2*h) * shifted
        
        # Convex solve subproblem using CVXPY
        # Variables
        d = cp.Variable(n_points)
        
        # Objective: minimize gradient term + regularization
        obj = cp.Minimize(
            cp.sum(cp.multiply(grad, d)) + 0.5 * cp.sum_squares(d)
        )
        
        # Constraints
        constraints = [
            h + d >= 0,
            h + d <= 1,
            cp.sum(d) == 0  # Keep sum constraint
        ]
        
        # Form and solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                step = d.value
            else:
                # Fallback: simple projection
                step = -grad
                step = np.clip(step, -trust_radius, trust_radius)
                # Ensure sum constraint
                step = step - np.mean(step)
        except:
            # Fallback: simple gradient descent step
            step = -grad
            step = np.clip(step, -trust_radius, trust_radius)
            step = step - np.mean(step)
            
        # Linesearch
        alpha = 1.0
        for _ in range(max_backtrack):
            h_new = h + alpha * step
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_C5 = compute_C5(h_new, dx)
            if new_C5 <= current_C5 - rho * alpha * np.dot(grad, step):
                h = h_new
                break
            else:
                alpha *= 0.5

        # Update trust region
        trust_radius = max(1e-4, min(1.0, trust_radius * 1.2))

    # Final validation
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_C5 = compute_C5(h, dx)
    
    return (h.tolist(), final_C5, n_points)

def compute_C5(h, dx):
    # Use fast correlation computation for overlap
    h_conv = np.concatenate([h, np.zeros(len(h))])
    one_minus_h = 1 - h
    one_minus_h_conc = np.concatenate([one_minus_h, np.zeros(len(one_minus_h))])
    correlation = signal.correlate(h_conv, one_minus_h_conc, mode='full')
    max_overlap = np.max(correlation[:len(h)]) / len(h)  # Normalize by length
    return max_overlap * dx
