import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from initial_h_values if available
    try:

    except NameError:
        # Fallback: create a simple step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Make it a step function with some variation
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 0.3
            else:
                h[i] = 0.7
                
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all overlaps and find max
        correlations = correlate(h, 1 - h, mode="full")
        # Cut off the mode="full" to only look at valid overlaps in [0, 2]
        valid_correlations = correlations[len(h)-1:]
        k_star = np.argmax(valid_correlations)
        max_overlap = valid_correlations[k_star]
        
        if max_overlap <= 0.00001:
            break

        # Find the indices that contribute most to this overlap
        best_i = -1
        best_reduction = -1e-9
        best_j = -1
        
        # Search for the best i to modify to reduce overlap
        for i in range(len(h) - k_star):
            j = i + k_star
            if h[i] > 0 and h[j] < 1:
                # How much would reducing h[i] and increasing h[j] help?
                # Current contribution to overlap: h[i] * (1 - h[j])
                # New contribution: (h[i]-eps) * (1 - (h[j]+eps)) = h[i]*(1-h[j]) - eps*(1-h[j]) - eps*h[i]
                # Reduction = eps*(1 - h[j] + h[i])
                # If h[i] is small and h[j] is large, we want to lower h[i]
                # We can reduce the overlap by 2 * eps * h[i] * (1 - h[j])
                # But more safely, we can only reduce h[i] up to its value
                # and increase h[j] up to 1
                available_h_i = min(h[i], 1 - h[j])
                if available_h_i > 0:
                    reduction = available_h_i * h[i] * (1 - h[j])
                    if reduction > best_reduction:
                        best_reduction = reduction
                        best_i = i
                        best_j = j
                        
        # Now try to increase h[i] (and thus decrease h[j])
        if best_i != -1:
            # Compute how much we can change
            # We want to move up to min(h[best_i], 1 - h[best_j])
            max_change = min(h[best_i], 1 - h[best_j])
            if max_change > 0.0001 and best_j >= 0:
                # Small step adjustment
                step = max_change * 0.1
                h[best_i] -= step
                h[best_j] += step
                
                # Maintain the sum
                current_sum = np.sum(h)
                diff = current_sum - target_sum
                if abs(diff) > 1e-6:
                    # Redistribute appropriately
                    idxs = np.argsort(h)
                    redistrib = diff / len(h)
                    for i in range(len(h)):
                        h[idxs[i]] -= redistrib
                        
                # Clamp to [0, 1]
                h = np.clip(h, 0, 1)
                
                # Renormalize to sum = target_sum
                current_sum = np.sum(h)
                h = h * (target_sum / current_sum)
                
                # Evaluate new C5
                c5 = evaluate_c5(h, dx)
                if c5 < best_c5 - 1e-8:
                    best_c5 = c5
                    best_h = h.copy()
                    
            # If no improvement and we're very close to target
            if iteration > 500 and abs(c5 - best_c5) < 1e-6:
                break
                
        # If no benefit from this single pair, try improving globally
        if best_i == -1:
            # Try to smooth
            h_new = h.copy()
            for i in range(1, n_points - 1):
                h_new[i] = 0.5 * (h[i-1] + h[i+1])
            h = h_new
            
            # Clamp and renormalize
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (target_sum / current_sum)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5 - 1e-8:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    # h is the function array; compute overlaps for all shifts
    correlations = correlate(h, 1 - h, mode="full")
    # Only consider valid shifts: k from 0 to 2-length
    valid_correlations = correlations[len(h)-1:]
    return np.max(valid_correlations) * dx
