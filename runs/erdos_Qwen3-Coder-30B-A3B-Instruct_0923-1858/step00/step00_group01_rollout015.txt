```python
import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    correlation = correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx

def repair_h(h, n_points, dx):
    """Repair h to satisfy constraints"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet integral constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        scale = 1.0 / current_sum
        h = h * scale
        # Make sure we enforce sum(h)*dx == 1
        h = h * (1.0 / (np.sum(h) * dx)) * (1.0 / dx) * dx
        h = np.clip(h, 0, 1)
    return h

def initialize_population(n_points, n_pop, initial_h=None):
    """Initialize population with feasible h vectors"""
    population = []
    if initial_h is not None:
        # Use initial solution as first member
        h = repair_h(initial_h.copy(), n_points, 2.0 / n_points)
        population.append(h)
    # Fill up with random populations
    for i in range(n_pop - len(population)):
        h = np.random.random(n_points)
        h = repair_h(h, n_points, 2.0 / n_points)
        population.append(h)
    return population

def crossover(parent1, parent2, dx):
    """Crossover two h vectors"""
    n_points = len(parent1)
    # Select crossover point
    point = random.randint(1, n_points - 1)
    # Create offspring
    child1 = np.concatenate((parent1[:point], parent2[point:]))
    child2 = np.concatenate((parent2[:point], parent1[point:]))
    # Repair children
    child1 = repair_h(child1, n_points, dx)
    child2 = repair_h(child2, n_points, dx)
    return child1, child2

def mutate(h, mutation_rate, dx):
    """Mutate a h vector"""
    n_points = len(h)
    h = h.copy()
    for i in range(n_points):
        if random.random() < mutation_rate:
            # Add small random change
            delta = np.random.normal(0, 0.05)
            h[i] += delta
    return repair_h(h, n_points, dx)

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Start with middle resolution
    n_points = 47
    dx = 2.0 / n_points
    
    # Initialize population
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    n_pop = 50
    population = initialize_population(n_points, n_pop, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    # Limit generations for the current resolution
    max_generations = 1000
    
    while time.time() - start_time < budget_s * 0.95:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Select top 20%
        sorted_indices = np.argsort(fitness)
        elite_count = max(1, n_pop // 5)
        elites = [population[i] for i in sorted_indices[:elite_count]]
        
        # Create offspring
        new_population = elites.copy()
        while len(new_population) < n_pop:
            # Selection: pick two parents randomly from elites
            parent1 = elites[random.randint(0, len(elites)-1)]
            parent2 = elites[random.randint(0, len(elites)-1)]
            
            # Crossover
            if random.random() < 0.7:
                child1, child2 = crossover(parent1, parent2, dx)
                new_population.append(child1)
                if len(new_population) < n_pop:
                    new_population.append(child2)
            else:
                # Just mutate rather than crossover
                child = mutate(parent1, 0.02, dx)
                new_population.append(child)
        
        # Mutate some additional individuals
        while len(new_population) < n_pop:
            parent = elites[random.randint(0, len(elites)-1)]
            child = mutate(parent, 0.05, dx)
            new_population.append(child)
        
        # Trim to right size
        population = new_population[:n_pop]
        
        generation += 1
        if generation > max_generations:
            # Try to refine at higher resolution if we haven't changed much
            if no_improvement_count > 50:
                # Refine to higher resolution if we have a good solution
                if n_points < 200:
                    n_points = min(200, n_points * 2)
                    dx = 2.0 / n_points
                    # Reinitialize with better solution interpolated
                    new_pop = []
                    if best_h is not None:
                        # Interpolate best solution to higher resolution
                        h_upsampled = np.interp(
                            np.linspace(0, 2, n_points),
                            np.linspace(0, 2, len(best_h)),
                            best_h
                        )
                        # Repair to be feasible
                        h_upsampled = repair_h(h_upsampled, n_points, dx)
                        new_pop.append(h_upsampled)
                    # Fill with random
                    for _ in range(n_pop - len(new_pop)):
                        h = np.random.random(n_points)
                        h = repair_h(h, n_points, dx)
                        new_pop.append(h)
                    population = new_pop
                    max_generations = 500
                    no_improvement_count = 0
                else:
                    break
            else:
                no_improvement_count += 1
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h, final_c5, n_points)
    
    # Try one last effort with the current best candidate
    candidate_h = population[np.argmin(fitness)]
    final_c5 = evaluate_h(candidate_h, dx)
    return (candidate_h, final_c5, n_points)
```
