import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create symmetric initial guess
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        # Add some structure for better convergence
        h0[:n_points//3] = 0.8
        h0[n_points//3:2*n_points//3] = 0.2
        h0[2*n_points//3:] = 0.6
        # Normalize to sum to n/2
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Binary search on C5
    c_min, c_max = 0.0, 0.5
    for iteration in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        c_mid = (c_min + c_max) / 2.0
        
        # Try to find a feasible solution for c_mid
        # Projected gradient descent approach
        h = h0.copy()
        # Constraints: 0 <= h <= 1, sum(h) = n/2
        
        # Simple gradient descent for a few steps
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            # Gradient of C5 w.r.t h (approximate)
            # Derivative of overlap integral w.r.t h[i]
            grad = np.zeros_like(h)
            # Compute full convolution
            conv_h = np.correlate(h, np.ones_like(h), mode='full')
            conv_1h = np.correlate(h, 1-h, mode='full')
            # Take gradients properly with respect to h
            # We'll be approximate but viable
            # Larger values in h reduce overlap more
            for i in range(n_points):
                # Approximation of gradient 
                grad[i] = conv_1h[n_points-1+i] - 0.5 * conv_h[n_points-1+i]
            
            # Update with gradient
            h_new = h - 0.01 * grad
            
            # Clip and renormalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update h
            h = h_new
            
            # Evaluate current
            c5_current = evaluate_erdos_solution(h)
            if c5_current < best_c5:
                best_h = h.copy()
                best_c5 = c5_current
                
        # Check feasibility with current c_mid
        c5_test = evaluate_erdos_solution(h)
        if c5_test <= c_mid:
            c_max = c_mid
            if c5_test < best_c5:
                best_h = h.copy()
                best_c5 = c5_test
        else:
            c_min = c_mid
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    # Make sure it's within bounds and feasible
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6
    
    return (best_h, final_c5, n_points)

def evaluate_erdos_solution(h):
    # Computing overlap integral with assumed dx = 2/n
    n = len(h)
    dx = 2.0 / n
    
    # Compute full correlation of h(x) with (1 - h(x+k))
    # This is equivalent to the overlap integral
    tail = np.concatenate([h, np.zeros(n)])
    head = np.ones_like(h)
    conv_result = np.correlate(tail, head - h, mode='full') 
    conv = conv_result[n-1:2*n-1]  # Relevant part
    max_c5 = np.max(conv) * dx
    
    return max_c5
