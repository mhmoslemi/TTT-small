import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

    max_iter = 1000

    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.01
    decay_every = 100
    max_steps = 50
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current C5 and gradients
        current_c5 = compute_c5(h, dx)
        grad = compute_gradient(h, dx)
        
        # Line search
        for step in range(max_steps):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain integral constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
                break
            alpha *= 0.5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count

                h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h, dx)

def compute_c5(h, dx):
    # Compute cross-correlation between h and (1-h) in full mode
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_gradient(h, dx):
    # Derivative of C5 w.r.t h using finite differences (simplified)
    # For this problem, we'll use an approximate gradient by computing
    # perturbations and estimating directional derivatives
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # Evaluate current C5
    c5_base = compute_c5(h, dx)
    eps = 1e-6
    
    # Compute directional derivatives for each dimension
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        
        c5_plus = compute_c5(h_plus, dx)
        
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad

def interp_to_new_res(h_old, n_new):
    """Interpolate h to a new resolution while preserving integral"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    # Rescale to preserve integral constraint
    h_new = h_new * (n_new / 2) / np.sum(h_new)
    return h_new
