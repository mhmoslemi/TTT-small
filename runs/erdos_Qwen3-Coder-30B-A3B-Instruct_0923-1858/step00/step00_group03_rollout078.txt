from scipy.signal import correlate
import time

    h *= (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 10000
    eta = 0.01
    tolerance = 1e-6
    decay_factor = 0.999
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        lag_indices = np.arange(-n_points + 1, n_points)
        max_idx = np.argmax(corr)
        max_lag = lag_indices[max_idx]
        max_val = corr[max_idx] * dx
        
        # Early termination if optimal enough
        if max_val < 0.3807:
            break
            
        # Compute gradient w.r.t. h for the maximum lag
        grad = np.zeros_like(h)
        if max_lag >= 0:
            # h[i] contributes to correlation with (1-h[i+max_lag])
            grad[max_lag:] += -(1 - h[:-max_lag])
            # h[i+max_lag] contributes to correlation with (1-h[i])
            grad[:-max_lag] += -h[max_lag:]
        else:
            # h[i] contributes to correlation with (1-h[i-max_lag])
            grad[:max_lag] += -(1 - h[-max_lag:])
            # h[i-max_lag] contributes to correlation with (1-h[i])
            grad[-max_lag:] += -h[:-max_lag]
        
        # Update h
        h = h - eta * grad
        
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Adjust sum to exactly n_points / 2
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Adaptive learning rate
        eta *= decay_factor
        
        # Check convergence
        if np.abs(eta) < 1e-8:
            break
    
    # Final computation
    final_corr = correlate(h, 1 - h, mode="full")
    lag_indices = np.arange(-n_points + 1, n_points)
    final_max = np.max(final_corr) * dx
    
    return (h.tolist(), float(final_max), n_points)
