import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h, dx=2.0/45):
    """Evaluate the C5 constant for given h values"""
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n-1)])
    h_shifted = np.concatenate([np.zeros(n-1), h])
    # Compute all correlations: h(x) * (1 - h(x+k)) for all k
    correlations = np.correlate(h_extended, 1 - h_shifted, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr

def project_to_feasible(h):
    """Ensure h is in [0,1] and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    n = len(h)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) > 1e-10:
        # Rescale to match target sum
        if current_sum > 0:
            h = h * (n/2) / current_sum
        else:
            h = np.ones(n) * 0.5
    return h

def create_population(n_points, pop_size=50):
    """Create initial population of h values"""
    pop = []
    base_h = create_initial_h()
    
    for i in range(pop_size):
        h = base_h.copy()
        # Random perturbation
        if i > 0:
            # rand is between [0,1]
            noise = np.random.RandomState(i).normal(0, 0.05, n_points)
            h = h + noise
            h = project_to_feasible(h)
        pop.append(h)
    return pop

def calculate_c5(h):
    """Fast calculation using np.correlate"""
    n = len(h)
    dx = 2.0 / n
    h_ext = np.concatenate([h, np.zeros(n-1)])
    h_shift = np.concatenate([np.zeros(n-1), h])
    correlations = np.correlate(h_ext, 1 - h_shift, mode="full")
    max_corr = np.max(correlations)
    return max_corr * dx

def mutate_h(h, sigma):
    """Mutate h with Gaussian noise"""
    noise = np.random.normal(0, sigma, len(h))
    h_new = h + noise
    return project_to_feasible(h_new)

def crossover(h1, h2, alpha=0.5):
    """Blend two h values"""
    h_new = alpha * h1 + (1 - alpha) * h2
    return project_to_feasible(h_new)

def genetic_algorithm(n_points, pop_size=50, max_generations=200):
    """Run genetic algorithm to optimize h"""
    pop = create_population(n_points, pop_size)
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for gen in range(max_generations):
        # Evaluate current population
        fitness = [calculate_c5(h) for h in pop]
        sorted_indices = np.argsort(fitness)
        current_best = fitness[sorted_indices[0]]
        
        if current_best < best_c5:
            best_c5 = current_best
            best_h = pop[sorted_indices[0]].copy()
        
        # Selection and reproduction
        elite_size = pop_size // 5
        elites = [pop[i] for i in sorted_indices[:elite_size]]
        
        new_pop = elites.copy()
        remaining = pop_size - elite_size
        
        # Generate offspring
        for _ in range(remaining):
            parent1 = elites[np.random.randint(0, len(elites))]
            parent2 = elites[np.random.randint(0, len(elites))]
            child = crossover(parent1, parent2, 0.5)
            child = mutate_h(child, 0.05 * (0.95 ** gen))
            new_pop.append(child)
            
        pop = new_pop
        
        # Early stopping if time is nearly exhausted
        if time.time() - start_time > 550:
            break
    
    return best_h, best_c5

def refine_with_gradient_descent(h0, max_iter=150):
    """Refine using gradient-based method"""
    n = len(h0)
    dx = 2.0 / n
    
    def objective(h):
        # Need to calculate full correlations
        h_ext = np.concatenate([h, np.zeros(n-1)])
        h_shift = np.concatenate([np.zeros(n-1), h])
        correlations = np.correlate(h_ext, 1 - h_shift, mode="full")
        return np.max(correlations) * dx
    
    def gradient(h):
        """Estimate gradient using finite differences"""
        eps = 1e-5
        grad = np.zeros_like(h)
        base_obj = objective(h)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad
    
    h = h0.copy()
    best_h = h.copy()
    min_c5 = objective(h)
    lr = 0.01
    last_improvement = 0
    
    for i in range(max_iter):
        grad = gradient(h)
        h = h - lr * grad
        h = project_to_feasible(h)
        
        new_c5 = objective(h)
        if new_c5 < min_c5:
            min_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 20:
                break
        
        lr = 0.01 / (1 + i/50)  # decay learning rate
        
        if i % 50 == 0:
            pass  # debug info
            
    return best_h, min_c5
