import numpy as np
from scipy import optimize
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    # Compute cross-correlation: h convolved with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum and multiply by dx
    return np.max(corr) * dx

def project_h(h, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to satisfy sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def finite_diff_grad(h, dx, eps=1e-5):
    """Compute finite difference gradient of C5"""
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    
    for i in range(n):
        # Perturb h[i] slightly
        h_plus = h_base.copy()
        h_minus = h_base.copy()
        
        h_plus[i] += eps
        h_minus[i] -= eps
        
        # Project to feasible region
        h_plus = project_h(h_plus, dx)
        h_minus = project_h(h_minus, dx)
        
        # Evaluate C5 at perturbed points
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def trust_region_update(h, grad, trust_radius, dx):
    """Compute trust region update"""
    # Simple projected gradient step
    h_new = h - trust_radius * grad
    # Project to feasible region
    h_new = project_h(h_new, dx)
    return h_new
