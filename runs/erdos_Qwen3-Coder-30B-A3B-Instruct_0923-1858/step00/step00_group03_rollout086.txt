import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):

    # Compute full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # The overlap is the max of the correlation (excluding the zero-shift case)
    # We only consider the shifts that make sense (the correlation for shift k
    # corresponds to the overlap of h with 1-h shifted by k)
    # The full correlation has length 2*n - 1
    # The shift values go from -(n-1) to (n-1)
    # For our problem, we want shifts from 1 to n-1 (positive shifts)
    # We want to exclude the 0 shift if it exists
    max_overlap = max(corr[len(h):])  # Take from the positive shifts
    # However, we also want to make sure we're comparing it correctly
    # But let's think about what correlate does
    # correlate(h, 1-h, mode='full') gives us sum over i of h[i] * (1-h[i+k])
    # for k from -(n-1) to (n-1)
    # we want the max value among those shifts that are valid
    # i.e. valid shifts are for which i+k is valid in the range [0, 2)
    # or with discretization they are valid i and i+k in [0, n)
    # for positive shifts k = 1 to n-1, and for negative shifts k = -n+1 to -1
    # we want to take the maximum across all valid k
    # but here we want the maximum across all possible k, so it's just the maximum
    # of all but the middle element (corresponding to k=0)
    # Actually, let's see if we can recompute naively if needed
    max_corr = -1.0
    n = len(h)
    for k in range(-n + 1, n):  # This covers all valid shifts
        # For shift k, we need to compute sum of h[i] * (1 - h[i + k]) 
        # i.e., sum over all i where i + k is also valid, i being in [0, n)
        # This means i in [0, n) and i + k in [0, n)
        # Which means i in [0, n) and i in [-k, n - k]
        # So i in [max(0, -k), min(n - 1, n-1-k)]  
        # Which is [max(0, -k), min(n - 1, n - 1 - k)]
        # If k is positive, -k < 0, so max(0, -k) = 0
        # min(n - 1, n - 1 - k) = n - 1 - k (since k > 0)
        # If k is negative, n - 1 - k > n - 1
        # Does it work that we want valid pairs?
        # z = 0 if k is negative (i >= -k means i >= 0)

        # which simplifies to k >= 0 or 0 <= n-1, both of which are true in range.
        z = 0
        u = n - 1
        if k > 0:

        if k < 0:

                valid_count += 1
            # average_over_valid = total / valid_count if valid_count > 0 else 0.0
            # Actually no, the integral depends on the discretization dx as well.
            # The discrete sum is what we compute, and we want it normalized
            # The anchors are spacings of dx in the continuous case
            # But we already formalized this in instruction as integral approximated by sum
            # Perhaps because we are integrating over [0,2], the integral is approximated
            # by sum h(x_i) * dx , so: 
            # When we compute the overlap of h and 1-h shifted by k, we look at individual
            # contributions. Each such contribution is h[i] * (1 - h[i + k]) 
            # multiplied by dx (the width) for the appropriate ranges of i.
            # But we have already done that implicitly via the cross-correlation:
            # instead of explicitly writing range bounds, we let the correlation do it.
            # so make sure we only take the part that corresponds to valid overlaps 
            # (that don't have out-of-bound indices) when computing the maximum.
            # but note that array indices are shifted by n-1 when comparing
            # So the correlation index corresponds to k =  index - (n - 1) = index - n + 1
            # so k from -n+1 to n-1 
            # We might have shifted the indices, let's recompute properly...
            # The computation in scipy.signal.correlate with integration as

            # correspond to .full mode = length 2*n-1

            # while  
            if corr[n - 1 + k] > max_corr:

def binary_rounding(h):
    """Convert h to binary, ensuring sum is n/2."""
    n = len(h)
    h_int = np.round(h).astype(int)
    current_sum = np.sum(h_int)
    diff = n // 2 - current_sum
    if diff == 0:
        return h_int
    # Adjust the difference by flipping elements
    if diff > 0:  # Need to set more 1s
        # Flip 0s to 1s, pick ones that give minimal change in value
        # Simple: just choose lowest among 0s
        indices = np.where(h_int == 0)[0]
        active = indices[:diff]
        h_int[active] = 1
    else:  # Need to set less 1s
        # Flip 1s to 0s, flip smallest ones
        indices = np.where(h_int == 1)[0]
        # flip the ones where h[i] is closest to 0.5, but we already have all 1s/0s
        # so just flip those with certain indices, but which?
        # a simple pick with lowest indices to make sum decrease appropriately
        active = indices[:abs(diff)]
        h_int[active] = 0
    return h_int
