def evaluate_erdos_solution(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum over all shifts
    max_corr = np.max(corr)
    return max_corr * dx
