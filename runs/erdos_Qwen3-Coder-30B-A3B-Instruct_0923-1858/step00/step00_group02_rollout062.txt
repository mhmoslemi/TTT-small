import numpy as np
from scipy import signal
import time
import cvxpy as cp
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
    else:
        h = np.ones(n_points) * 0.5
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    max_iter = 1000
    iter_count = 0
    
    # Trust region parameters
    trust_radius = 0.1
    min_radius = 1e-4
    max_radius = 1.0
    rho = 0.1
    alpha = 1.0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1.0:
        iter_count += 1
        
        # Compute current overlap values
        overlap_values = np.correlate(h, 1-h, mode='full') * dx
        current_c5 = np.max(overlap_values)
        
        # Solve convex approximation subproblem
        try:
            # Compute expected values for linear approximation
            expected_h = np.zeros(n_points)
            for i in range(n_points):
                forward_indices = [(i+j) % n_points for j in range(1, n_points)]
                expected_h[i] = np.mean([h[j] for j in forward_indices])
            
            # Build gradient vector
            grad = np.zeros(n_points)
            for i in range(n_points):
                grad[i] = 1.0 - 2.0 * h[i] - expected_h[i]
            
            # Set up convex quadratic program
            d = cp.Variable(n_points)
            obj = cp.Minimize(cp.sum_squares(d) / 2.0 + grad.T @ d)
            constr = [cp.norm(d, 2) <= trust_radius,
                      d >= -h,
                      d <= 1-h,
                      cp.sum(d) == 0]
            prob = cp.Problem(obj, constr)
            
            # Solve
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                step = d.value
            else:
                # Fallback to simple line search
                step = -grad
        except:
            # Fallback to gradient descent
            step = -grad
        
        # Line search
        alpha = 1.0
        while alpha > 1e-6:
            h_new = h + alpha * step
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * target_sum / h_new_sum
            
            # Evaluate new C5
            overlap_new = np.correlate(h_new, 1-h_new, mode='full') * dx
            new_c5 = np.max(overlap_new)
            
            # Accept if improvement
            if new_c5 < current_c5 - rho * alpha * np.dot(grad, step):
                h = h_new
                break
            alpha *= 0.5
        
        # Update trust radius
        if alpha < 0.5:
            trust_radius = max(min_radius, trust_radius * 0.5)
        else:
            trust_radius = min(max_radius, trust_radius * 2.0)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h = best_h * target_sum / best_h_sum
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
