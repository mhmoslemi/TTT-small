import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    # Compute convolution h * (1 - h) with full mode
    conv = correlate(h, 1 - h, mode="full")
    # Multiply by dx to get integral scale
    conv *= dx
    # Return the maximum value
    return np.max(conv)

def project_h(h, n_points):
    # Project onto [0,1] constraints and rescale to sum to n/2
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h *= (n_points / 2) / np.sum(h)
    return h

    if values[0] <= f_r < values[-2]:
        # Accept reflection
        simplex[-1] = x_r

        # Expansion
        x_e = x_centroid + gamma * (x_r - x_centroid)

            values[-1] = f_e

            simplex[-1] = x_r
            values[-1] = f_r
    else:

        x_c = project_h(x_c, len(x_c))
        f_c = evaluate_h(x_c)

    return simplex, values
