def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Maximum value rescaled by dx
    max_corr = np.max(corr) * dx
    return max_corr

def clip_and_rescale(h, dx):
    """Ensure h is in [0,1] and sum(h)*dx = 1"""
    # Ensure values are in [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum(h)*dx = 1
    current_sum = np.sum(h) * dx
    if abs(current_sum) > 1e-10:
        h = h / current_sum * (1.0 / dx)
    return h
