import numpy as np
from scipy.optimize import minimize
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

        n_points = len(h0)
    except NameError:
        # fallback to a simple construction
        n_points = 47
        h0 = np.zeros(n_points)
        # place a few steps in a sparse pattern
        step_positions = [i * n_points // 10 for i in range(10)]
        for pos in step_positions:
            if pos < n_points:
                h0[pos] = 1
        # normalize
        h0 = h0 * (n_points / 2) / np.sum(h0)

    dx = 2.0 / n_points

    def compute_C5(h):
        # compute the full cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # take max over all lags
        max_corr = np.max(corr)
        return max_corr * dx

    def project_h(h):
        # project h to [0,1] and rescale to sum(h)*dx = 1
        h = np.clip(h, 0, 1)
        s = np.sum(h) * dx
        if s > 0:
            h = h * (n_points / 2) / s
        return h

    def obj_and_grad(h_vec):
        h = h_vec.copy()
        h = project_h(h)
        c5 = compute_C5(h)
        
        # numerical gradient evaluation via finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_C5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_C5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return c5, grad

    # initialize
    best_h = h0.copy()
    best_c5 = compute_C5(best_h)
    current_h = h0.copy()
    
    # optimize using L-BFGS
    iter_count = 0
    last_improvement = time()
    while time() - start_time < budget_s - 0.1:
        if iter_count % 20 == 0:
            # periodic evaluation
            c = compute_C5(current_h)
            if c < best_c5:
                best_c5 = c
                best_h = current_h.copy()
            # stop if not improving
            if time() - last_improvement > 10:
                break
            last_improvement = time()
        
        try:
            res = minimize(
                lambda x: obj_and_grad(x)[0],
                current_h,
                method='L-BFGS-B',
                jac=lambda x: obj_and_grad(x)[1],
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            current_h = res.x
            current_h = project_h(current_h)
            iter_count += 1
        except:
            break

    c5_bound = compute_C5(best_h)
    return (best_h, c5_bound, n_points)
