import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute correlation of h with (1-h) for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx to match the integral
    corr *= dx
    # Return maximum value
    return np.max(corr)

def project_h(h, dx, target_sum):
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust sum to match constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) > 1e-10:
        diff = (target_sum - current_sum) / dx
        # Distribute difference evenly
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial
    if 'initial_h_values' in globals():

    else:
        # Create a basic symmetric construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Add some variation near center
        start = n_points // 4
        end = 3 * n_points // 4
        h[start:end] = 1.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Initial evaluation
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 2000
    lr = 0.01
    tolerance = 1e-6
    start_time = time.time()
    
    for iter_num in range(max_iter):
        # Check timeout
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Find maximum contribution lag
        corr = correlate(h, 1 - h, mode='full')
        corr *= dx
        max_lag_idx = np.argmax(corr)
        # Get the corresponding lag
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h
        grad = np.zeros(n_points)
        # The derivative of overlap w.r.t. h[i] is the contribution from
        # the correlation at lag max_lag
        lag = max_lag
        # loop over all positions and accumulate gradient contribution
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] += (1 - h[i + lag]) if (i + lag) >= 0 else 0
            if 0 <= i - lag < n_points:
                grad[i] += h[i - lag] if (i - lag) >= 0 else 0
                
        # Also consider gradient w.r.t. 1 - h[i], this is the change in another term
        # But a better approximation using symmetry: 
        # correlation g(i,k) = h[i] * (1 - h[i+k])
        # So d/dh[i] g(i,k) = 1 - h[i+k] (if k >= 0) or h[i-k] (if k < 0)
        # This leads to the previous formula up to sign.
        grad *= dx
        
        # Update steps according to gradient (but with sign change for maximization)
        h -= lr * grad
        
        # Project back onto constraints
        h = project_h(h, dx, target_sum)
        
        # Check for improvement
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    # Final check and return best solution
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))
