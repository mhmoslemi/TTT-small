import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """Evaluate C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value is the worst-case overlap
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def normalize_h(h):
    """Ensure sum(h) = n/2 and clip to [0,1]."""
    n = len(h)
    h = np.clip(h, 0, 1)
    scale = (n / 2) / np.sum(h)
    h = h * scale
    return h

def crossover(parent1, parent2, crossover_prob=0.7):
    """Blend crossover of two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    alpha = np.random.rand()
    child1 = parent1 * alpha + parent2 * (1 - alpha)
    child2 = parent2 * alpha + parent1 * (1 - alpha)
    return normalize_h(child1), normalize_h(child2)

def mutate(h, mutation_rate=0.1, epsilon=0.1):
    """Apply small mutations."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, epsilon)
    return normalize_h(h)

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def genetic_algorithm(budget_s, n_points=42):
    """Genetic algorithm for minimizing C5."""
    np.random.seed(42)
    
    # Initialize population
    pop_size = 30
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = normalize_h(h0)
        population.append(h0)
    else:
        # Create a simple initial population
        h0 = np.ones(n_points) * 0.5
        h0 = normalize_h(h0)
        population.append(h0)
    
    # Add random members
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = normalize_h(h)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    gen_no_improvement = 0
    max_no_improvement = 200
    
    for generation in range(1000):  # Max generations
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for termination conditions
        if gen_no_improvement >= max_no_improvement:
            break
        gen_no_improvement += 1
        if max_no_improvement - gen_no_improvement >= 50:
            # Reset if no progress for long time
            gen_no_improvement = 0
        
        # Select parents and generate children
        new_population = []
        # Elitism: keep 10% best
        elite_count = max(1, pop_size // 10)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offsprings
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:pop_size]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization."""
    np.random.seed(seed)
    h, c5 = genetic_algorithm(budget_s, n_points=42)
    
    # Final check
    h = normalize_h(h)
    c5 = evaluate_h(h)
    
    return h.tolist(), c5, len(h)
