import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise create a balanced one
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 42
        h = np.full(n_points, 0.5)
        h[1::2] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # Perturbation strength and period
    epsilon = 0.02
    perturb_period = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral (cross-correlation with 1-h)
        correlation = signal.correlate(h, 1 - h, mode='full')
        correlation *= dx
        c5 = np.max(correlation)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration

        # If stalled, perturb
        if iteration - last_improvement > 2000:
            # Disturb h by redistributing small amounts of mass
            perturbation = np.random.normal(0, epsilon, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            # Project to satisfy constraint sum(h) = n_points / 2
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            continue
            
        # Adjust h to reduce maximum overlap
        # Identify where overlapping happens most
        max_idx = np.argmax(correlation)
        k_max = max_idx - (len(correlation) - 1) // 2  # k where overlap is max
        k_max = k_max % n_points  # ensure k_max is in [0, n_points-1]
        
        # Create perturbation based on overlapping pattern
        delta = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            # Try to reduce h[i] and increase h[j] if that lowers the overlap
            delta[i] -= epsilon
            delta[j] += epsilon
            
        h_new = h + delta
        
        # Ensure bounds [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to satisfy sum constraint
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        h = h_new
        
        if iteration % perturb_period == 0:
            # For diversity, perturb the solution slightly
            perturbation = np.random.normal(0, epsilon/2, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum

    # Final check
    correlation = signal.correlate(best_h, 1 - best_h, mode='full')
    correlation *= dx
    final_c5 = np.max(correlation)
    
    return (best_h, final_c5, n_points)
