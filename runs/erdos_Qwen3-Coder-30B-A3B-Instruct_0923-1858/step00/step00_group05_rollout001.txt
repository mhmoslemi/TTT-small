def evaluate_erdos_solution(h):
    """Compute the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (from k=0 to k=n-1)
    max_k = n - 1
    max_overlap = max(corr[n-1:n-1+max_k+1]) * dx
    return max_overlap
