import numpy as np
from scipy.optimize import minimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        h = h * (1.0 / current_sum)
    return h

def evaluate_C5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available

        h = np.ones(n_points) * 0.5
        h = clip_and_rescale(h, 2.0 / n_points)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    tol = 1e-6
    alpha = 0.01
    decay_every = 100
    decay_factor = 0.1
    
    # Gradient descent
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        base_C5 = evaluate_C5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = clip_and_rescale(h_plus, dx)
            C5_plus = evaluate_C5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = clip_and_rescale(h_minus, dx)
            C5_minus = evaluate_C5(h_minus, dx)
            
            grad[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Update h
        h = h - alpha * grad
        h = clip_and_rescale(h, dx)
        
        # Evaluate new C5
        current_C5 = evaluate_C5(h, dx)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
        
        # Adaptive learning rate
        if iteration > 0 and iteration % decay_every == 0:

        h_high = np.interp(np.linspace(0, 2, n_points_high), np.linspace(0, 2, n_points), best_h)

        # Fine optimization at higher resolution

            eps = 1e-4
            grad = np.zeros_like(h_high)
