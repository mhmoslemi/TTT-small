import numpy as np
import time
from scipy.signal import correlate

def binary_convert(h, n_points):
    """Convert h to binary with exact sum constraint."""
    h_bin = np.round(h).astype(int)
    current_sum = np.sum(h_bin)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h_bin
    # Adjust to meet sum constraint
    diff = target_sum - current_sum
    if diff > 0:
        # Need to set more 0s to 1s
        zero_indices = np.where(h_bin == 0)[0]
        for i in range(diff):
            if i < len(zero_indices):
                h_bin[zero_indices[i]] = 1
    else:
        # Need to set more 1s to 0s
        one_indices = np.where(h_bin == 1)[0]
        for i in range(-diff):
            if i < len(one_indices):
                h_bin[one_indices[i]] = 0
    return h_bin

def compute_c5(h, dx):
    """Compute C5 from binary h."""
    # Compute overlap for all shifts
    h_flipped = 1 - h
    corr = correlate(h, h_flipped, mode='full')
    # Only valid shifts (full correlation)
    c5_vals = corr[len(h)-1:] * dx
    return np.max(c5_vals)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent or create a simple binary distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    h = binary_convert(h, n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simulated Annealing with binary flips
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Randomly select two indices: one with 1 and one with 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Tentative new state
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the move
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        T *= alpha
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
