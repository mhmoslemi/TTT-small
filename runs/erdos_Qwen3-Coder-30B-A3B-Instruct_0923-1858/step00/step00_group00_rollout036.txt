import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create symmetric initial construction
        n_points = 45
        h_start = np.ones(n_points) * 0.5
        # Add some structure to improve optimization
        half = n_points // 2
        for i in range(half):
            h_start[i] = i / half * 0.5
        for i in range(half, n_points):
            h_start[i] = (n_points - i) / (n_points - half) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    
    def compute_C5(h_vals):
        # convolution of h with (1-h)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def objective(h_vals):
        return compute_C5(h_vals)
    
    # Constraints
    def constraint_sum(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    def constraint_bounds(h_vals):
        return np.concatenate([h_vals, 1 - h_vals])
    
    # Optimization
    best_h = h.copy()
    best_C5 = compute_C5(h)
    n_eval = 0
    
    # Simple optimization
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation with constraints
        h_new = h.copy()
        # Randomly perturb a few points
        indices = np.random.choice(n_points, size=min(5, n_points//3), replace=False)
        for idx in indices:
            # Add small random change
            pert = np.random.normal(0, 0.01)
            h_new[idx] += pert
            # Keep in bounds
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Project onto constraint
        h_new = h_new * (1.0 / (np.sum(h_new) * dx))
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (1.0 / (np.sum(h_new) * dx))
        h_new = np.clip(h_new, 0, 1)
        
        C5_new = compute_C5(h_new)
        n_eval += 1
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
        
        h = h_new
        
        if n_eval % 50 == 0:
            # Early stopping if improving very little
            if best_C5 < 0.38085 and time.time() - start_time > budget_s * 0.8:
                break
    
    return (best_h.tolist(), best_C5, n_points)
