import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Triangular initialization
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Normalize initial h
    h = h.copy()
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Optimization parameters
    lr = 0.01
    decay = 0.95
    perturb_freq = 50
    perturb_mag = 0.05
    max_iter = 10000
    
    dx = 2.0 / n_points
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            grad[j] = (evaluate_c5(h_plus, n_points) - evaluate_c5(h_minus, n_points)) / (2 * eps)
        
        # Update with gradient
        h = h - lr * grad
        
        # Perturbation
        if i % perturb_freq == 0:
            noise = np.random.normal(0, perturb_mag, n_points)
            h = h + noise
            
        # Projection to feasible region
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Learning rate decay
        if i % 100 == 0 and i > 0:
            lr *= decay
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode='full')
    # Trim to correct size, avoiding padding issue
    start_idx = n_points - 1
    corr = corr[start_idx:start_idx + n_points]
    return np.max(corr) * dx
