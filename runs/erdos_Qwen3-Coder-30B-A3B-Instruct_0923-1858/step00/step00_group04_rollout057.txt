        # Take maximum over all k
        max_correlation = np.max(correlation)

    # Frank-Wolfe style optimization helper
    def project_to_feasible(x):
        # Clip to bounds
        x = np.clip(x, 0, 1)
        # Adjust sum to match constraint
        current_sum = np.sum(x)
        target_sum = n_points / 2
        adjustment = (target_sum - current_sum) / n_points
        x = x + adjustment
        return x

    # Gradient approximation via finite differences
    def estimate_gradient(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        base_c5 = compute_c5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad

    # Main optimization loop
    iteration = 0
    last_improvement = 0
    lr = 0.05
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:

        grad = estimate_gradient(h)
        
        # Frank-Wolfe step direction: minimize linear approximation
        # We want: minimize <grad, d> subject to 0 <= d <= 1, sum(d) = 0
        # This is minimized by taking d = sign(-grad) with proper normalization
        d = -grad
        d = np.clip(d, 0, 1)
        d = d - np.mean(d)  # zero mean for feasibility
        
        # Step size using line search
        best_lr = lr

            test_c5 = compute_c5(new_h)

            h = project_to_feasible(h)
            
        # Early termination if no progress
        if iteration - last_improvement > 1000:
            break
    
    # Final check
    final_c5 = compute_c5(best_h)
