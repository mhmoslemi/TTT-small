import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_h(h, dx):
    # Compute the overlap integral C5
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def objective(h):
    dx = 2.0 / len(h)
    return evaluate_h(h, dx)

def project_feasible(h, n_points):
    # Ensure h is within [0, 1] and sum equals n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def genetic_algorithm(n_points, budget_s):
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 50
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness (lower is better)
        fitness = []
        for h in population:
            try:
                c5 = objective(h)
                fitness.append(c5)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except:
                fitness.append(float('inf'))
        
        # Selection (tournament)
        tournament_size = 10
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size] if i+1 < population_size else selected[0]
            
            if random.random() < crossover_rate:
                # Single point crossover
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child1[j] += random.gauss(0, 0.05)
                if random.random() < mutation_rate:
                    child2[j] += random.gauss(0, 0.05)
            
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    return best_h, best_c5

def local_optimization(h_start, n_points, budget_s):
    start_time = time.time()
    
    # Set bounds (0,1) for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simple local refinement with scipy.optimize.minimize
    # We will do gradient-based refinement around the best GA solution
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Callback to update best solution
    best_sol = [h_start.copy()]
    best_c5 = [float('inf')]
    
    def callback(h):
        dx = 2.0 / len(h)
        c5 = evaluate_h(h, dx)
        if c5 < best_c5[0]:
            best_c5[0] = c5
            best_sol[0] = h.copy()
        if time.time() - start_time > budget_s - 5:
            raise KeyboardInterrupt
    
    # Use minimize with SLSQP method
    try:
        result = minimize(
            objective, 
            h_start, 
            method='SLSQP', 
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            callback=callback,
            options={'maxiter': 1000}
        )
        if result.success:
            return result.x, objective(result.x)
    except:
        pass
    
    return best_sol[0], best_c5[0]
