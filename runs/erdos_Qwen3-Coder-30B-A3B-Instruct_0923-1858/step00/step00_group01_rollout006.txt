import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Start with resolution 20
        n_points = 20
        h = np.ones(n_points) * 0.5
        h[10:15] = 1.0  # Slightly higher in center
    
    dx = 2.0 / n_points
    
    # Ensure constraint ∫h dx = 1
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_C5(h):
        # Compute cross-correlation of h with (1-h)
        corr = np.correlate(h, 1 - h, mode="full")
        # Find maximum value
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_gradient(h):
        # This approximates gradient w.r.t h for a fixed k which contributes to max
        # For a simple approach we compute correlated gradient over all shifts
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            grad[i] = 0.0
            for j in range(n):
                k = i - j + n - 1  # offset for full mode
                if 0 <= k < 2*n - 1:
                    grad[i] += (1 - h[j]) * (1 if k >= n - 1 else 0)  # rough estimate
        return grad * dx
    
    def project_h(h, dx):
        # Ensure h is in [0,1] and sum(h)*dx = 1
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        if abs(total) > 1e-10:
            h = h / total * (n_points / 2)
        return h
    
    def line_search(h, grad, alpha_max=1.0, lr=0.01):
        best_h = h.copy()
        best_C5 = compute_C5(h)
        for _ in range(20):  # try some shrink steps
            new_h = h - lr * grad
            new_h = project_h(new_h, dx)
            new_C5 = compute_C5(new_h)
            if new_C5 < best_C5:
                best_h = new_h
                best_C5 = new_C5
                lr *= 0.5  # adaptive step size
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
        return best_h, best_C5
    
    # Main iterative loop
    iterations = 0
    max_iter = 1500
    lr = 0.01
    improvement_count = 0
    prev_C5 = float('inf')
    
    while time.time() - start_time < budget_s:
        if iterations >= max_iter:
            break
        
        C5 = compute_C5(h)
        # If improved significantly, reset counter
        if prev_C5 - C5 > 1e-6:
            improvement_count = 0
            prev_C5 = C5
        else:
            improvement_count += 1
        
        if improvement_count > 200:
            # Possibly re-start from a better point
            # Or increase resolution to 47 if not already there
            if n_points < 47:  # double resolution
                n_points *= 2
                h = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h)), h)
                dx = 2.0 / n_points
                h = project_h(h, dx)
            else:
                break  # Push to final
        else:
            # Compute gradients via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            base_C5 = compute_C5(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, dx)
                if abs(h_plus[i] - h[i]) < 1e-5:  # Didn't change due to clipping
                    grad[i] = 0.0
                else:
                    c5_plus = compute_C5(h_plus)
                    grad[i] = (c5_plus - base_C5) / eps
            
            # Gradient descent step
            h_new, _ = line_search(h, grad, lr=lr)
            h = h_new
            
            # Adaptively reduce learning rate
            if iterations % 50 == 0 and iterations > 0:
                lr *= 0.9
        
        iterations += 1
    
    # Final validation
    C5 = compute_C5(h)
    h = project_h(h, dx)
    return (h.tolist(), C5, n_points)
