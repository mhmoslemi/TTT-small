import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h, dx):
    # Compute the overlap integral using correlation
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Only take the relevant part (shifts from -n+1 to n-1)
    # For our case, we want the maximum of the correlation results
    # shifted by 0 to 2*n-2. But since we want shift in [0,2], 
    # and h is of length n, shifts are 0 to n-1
    # The full correlation is 2*n-1 values, center at n-1
    # We are interested in shifts k=0 to n-1, shifted by n-1
    # So we take indices from n-1 to 2*n-2 = 0 to n-1 shifted by n-1 = n-1 to 2*n-2
    # But for k=0 to n-1 we look at shifts: 0 to n-1, which corresponds to 
    # indices: n-1 to n-1 + (n-1) = 2*n-2, so we take 
    # corr[n-1:2*n-1]
    cor_values = corr[n-1:]
    c5_val = np.max(cor_values) * dx
    return c5_val

def project_to_feasible(h, n_points, dx):
    # project h to [0,1]^n and enforce sum(h*dx) = 1
    # Clip values
    h = np.clip(h, 0, 1)
    # Rescale to satisfy sum = 1
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h * (1.0 / h_sum)
    return h

def gradient_descent_step(h0, dx, max_iter=100):
    h = h0.copy()
    n = len(h0)
    epsilon = 1e-5
    lr = 0.01
    for i in range(max_iter):
        c5_old = evaluate_c5(h, dx)
        # Compute gradient numerically via finite differences
        grad = np.zeros(n)
        for j in range(n):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_to_feasible(h_plus, n, dx)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[j] = (c5_plus - c5_old) / epsilon
        
        # Update step
        h = h - lr * grad
        # Project back to feasible region
        h = project_to_feasible(h, n, dx)
        c5_new = evaluate_c5(h, dx)
        if c5_new >= c5_old:
            break
        lr *= 0.99
    return h

def genetic_algorithm(initial_h, n_points, dx, budget_s, pop_size=50, num_generations=100):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Create initial population
    population = []
    population.append(initial_h)
    for i in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points, dx)
        population.append(h)
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate all individuals
        fitness = []
        for i, h in enumerate(population):
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, i))
            
        # Sort by fitness (ascending)
        fitness.sort()
        
        # Keep best 20% as elites
        num_elites = max(1, pop_size // 5)
        elites = [population[fitness[i][1]] for i in range(num_elites)]
        
        # Update best solution
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = population[fitness[0][1]].copy()
        
        # Create new population via crossover and mutation
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            # Tournament selection
            parent1 = elites[np.random.randint(0, len(elites))]
            parent2 = elites[np.random.randint(0, len(elites))]
            
            # Blend crossover
            alpha = 0.5
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation with decreasing standard deviation
            sigma = 0.1 * (0.95 ** gen)
            noise = np.random.normal(0, sigma, n_points)
            child += noise
            
            # Project to feasible region
            child = project_to_feasible(child, n_points, dx)
            new_pop.append(child)
        
        population = new_pop
    
    return best_h, best_c5

    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise initialize manually
    if 'initial_h_values' in globals():

    else:

        n_points = 45
        initial_h = np.ones(n_points) * 0.5  # Start with uniform
        initial_h[:n_points//2] = 1.0  # Make first half 1
        initial_h[n_points//2:] = 0.0  # Make second half 0
        initial_h = project_to_feasible(initial_h, n_points, 2.0/n_points)
    
    # Perform optimization
    dx = 2.0 / len(initial_h)
    
    # First, run genetic algorithm for global search
    ga_h, ga_c5 = genetic_algorithm(initial_h, len(initial_h), dx, budget_s * 0.5)
    
    # Then, refine with gradient descent on the best solutions
    best_h = ga_h.copy()
    best_c5 = ga_c5
    
    # Refine the very best solution
    refined_h = gradient_descent_step(best_h, dx, max_iter=150)
    refined_c5 = evaluate_c5(refined_h, dx)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    # Also, try a few more points for gradient descent
    for _ in range(3):
        if time.time() - time.time() > budget_s:
            break
        h_trial = np.random.rand(len(initial_h))
        h_trial = project_to_feasible(h_trial, len(initial_h), dx)
        h_refined = gradient_descent_step(h_trial, dx, max_iter=100)
        c5_trial = evaluate_c5(h_refined, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_refined
    
    # Prevent returning malformed solutions
    final_h = np.clip(best_h, 0, 1)
    final_h = project_to_feasible(final_h, len(initial_h), dx)
    return (final_h.tolist(), best_c5, len(initial_h))
