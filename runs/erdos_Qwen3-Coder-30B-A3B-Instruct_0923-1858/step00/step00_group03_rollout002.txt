import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        # Fallback: construct a simple initial function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.6
        h[-1] = 0.4
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure integral constraint
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    
    # Optimization parameters
    max_iter = 10000
    learning_rate = 0.01
    tol = 1e-6
    
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate all correlations
        h_folded = np.concatenate([h, np.zeros(n_points-1)])
        h_reflected = np.flip(1 - h)
        correlations = np.convolve(h_folded, h_reflected, mode='valid')
        correlations = correlations[:n_points] * dx
        
        # Find dominant lag
        k_star = np.argmax(correlations)
        c5 = correlations[k_star]
        
        if c5 < last_c5 and last_c5 - c5 < tol:
            break
        last_c5 = c5
        
        # Compute gradient w.r.t. dominant lag
        grad = np.zeros(n_points)
        k = k_star
        for j in range(n_points):
            if j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if j - k >= 0:
                grad[j] += h[j - k] * dx
                
        # Update h
        h = h - learning_rate * grad
        
        # Clamp h between 0 and 1
        h = np.clip(h, 0, 1)
        
        # Adjust sum to meet constraint
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
    
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    h_folded = np.concatenate([h, np.zeros(n_points-1)])
    h_reflected = np.flip(1 - h)
    correlations = np.convolve(h_folded, h_reflected, mode='valid')
    correlations = correlations[:n_points] * dx
    c5_bound = np.max(correlations)
    
    return (h, c5_bound, n_points)
