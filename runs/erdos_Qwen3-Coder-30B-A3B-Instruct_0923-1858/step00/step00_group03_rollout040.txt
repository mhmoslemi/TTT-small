import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5"""
    # Compute all overlaps for different shifts
    overlaps = correlate(h, 1 - h, mode='full') * dx
    # The valid range of shifts is from 0 to len(h)-1
    # The full correlation includes negative shifts, but we only care about forward shifts (0 to len(h)-1)
    # Actually, we want shifts from 0 to len(h)-1 (inclusive) corresponding to overlap at shift k.
    # The full correlation returns 2*len(h)-1 values, centered at k=0.
    # The overlap values corresponding to shifts k=0 to len(h)-1 are at indices len(h)-1 to 2*len(h)-2
    valid_overlaps = overlaps[len(h)-1:]  # This is actually correct because we're dealing with auto-correlation of (1-h)
    # But we should compute overlap as ∫ h(x)(1-h(x+k)) dx
    # Which corresponds to correlation of h with (1-h), with convolution properly indexed.
    # Go with correct indexing for s = 0, 1, 2, ..., n-2, n-1 => we want overlap at each shift k = 0 to n-1
    # (we already centered the full result for the first array's start in convolution)
    # For the overlap at shift k, we compute the convolution with second argument = 1-h
    # correlate(h, 1 - h, mode='full') n = len(h) returns a result of size 2*n - 1
    # Let's make sure we pick the right segment
    # At shift k=0: h[0]*1-(h[0]) + h[1]*1-(h[1]) + ... = sum(h * (1-h)) -> index n-1 in full correlation
    # At shift k=1: h[0]*1-(h[1]) + h[1]*1-(h[2]) + ... -> index n-2 in full correlation
    # ...
    # So we extract from index n-1 to 2*n-2 which corresponds to shifts 0 to n-1
    return np.max(valid_overlaps)
