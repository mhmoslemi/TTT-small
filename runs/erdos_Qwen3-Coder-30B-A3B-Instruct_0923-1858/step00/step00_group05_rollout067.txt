def evaluate(h, dx):
    """Evaluate the maximum overlap integral."""
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def constrain_h(h, n_points):
    """Ensure h satisfies constraints."""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)

    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h
