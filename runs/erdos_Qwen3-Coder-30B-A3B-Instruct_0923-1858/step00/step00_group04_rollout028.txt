    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Perturbation parameters
    epsilon = 0.02
    perturb_freq = 100
    max_iter = 10000
    
    for iteration in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break

        # Compute correlation for all lags
        corr = signal.correlate(h, 1 - h, mode='full')
        C5 = np.max(corr) * dx

        # Update best solution
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()

        # Find indices where maximum overlap occurs (could be multiple)
        max_indices = np.where(corr == np.max(corr))[0]
        max_k = max_indices[0] - (n_points - 1)  # Convert index to lag k

        # Apply adjustment
        # Create a small adjustment to decrease h where k overlap occurs
        # and increase where it doesn't interfere
        new_h = h.copy()
        
        # For simplicity, adjust values around the peak correlation lag
        for i in range(n_points):
            if (i + max_k) % n_points < n_points:
                # Reduce h[i] slightly, adjust others to maintain sum
                new_h[i] = max(0, new_h[i] - epsilon)
        
        # Ensure sum constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        
        if abs(diff) > 1e-8:
            # Adjust values to keep the sum
            increase_factor = diff / n_points  # Approximate increase needed per point
            # Distribute the sum difference evenly
            new_h = np.clip(new_h + increase_factor, 0, 1)
        
        # Final projection to maintain sum exactly
        adjusted_sum = np.sum(new_h)
        if abs(adjusted_sum) > 1e-10:
            new_h = new_h * target_sum / adjusted_sum
        
        h = new_h

        # Periodically apply random perturbation to escape local minima
        if iteration % perturb_freq == 0:
            # Reinforce the sum constraint before perturbation
            h_sum = np.sum(h)
            h = h * (target_sum / h_sum)
            
            # Create a smoother redistribution of mass
            perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
