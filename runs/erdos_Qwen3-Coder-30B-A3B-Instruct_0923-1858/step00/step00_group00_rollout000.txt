import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initialization
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h[n_points//2] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial sum matches target
    h = h * (target_sum / np.sum(h))
    
    # Cache current best
    h_best = h.copy()
    h_current = h.copy()
    
    # Evaluate current C5
    conv = correlate(h_current, 1 - h_current, mode="full")
    c5_current = np.max(conv) * dx
    c5_best = c5_current
    
    # Annealing parameters
    temp = 0.1
    alpha = 0.99
    sigma = 0.1
    patience = 0
    max_patience = 1000
    
    # Iterate until time budget is exhausted
    while time.time() - start_time < budget_s:
        # Perturb current solution
        h_candidate = h_current + np.random.normal(0, sigma, n_points)
        
        # Project to feasible region [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to maintain integral constraint
        scale = target_sum / np.sum(h_candidate)
        h_candidate *= scale
        
        # Evaluate new C5
        conv = correlate(h_candidate, 1 - h_candidate, mode="full")
        c5_candidate = np.max(conv) * dx
        
        # Accept with probability
        if c5_candidate < c5_current:
            h_current = h_candidate
            c5_current = c5_candidate
            patience = 0
        else:
            if np.random.rand() < np.exp(-(c5_candidate - c5_current) / temp):
                h_current = h_candidate
                c5_current = c5_candidate
            patience += 1
            
        # Update best solution
        if c5_current < c5_best:
            c5_best = c5_current
            h_best = h_current.copy()
            
        # Anneal temperature and sigma
        temp *= alpha
        if patience % 100 == 0 and sigma > 0.001:
            sigma *= 0.9
            
        # Check for early termination
        if patience > max_patience:
            break
            
    return (h_best.tolist(), c5_best, n_points)
