import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except NameError:
        # Create symmetric initialization if no parent construction
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        # Make it symmetric and normalize
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
        # Normalize sum to n/2
        h_init *= (n_points / 2) / np.sum(h_init)

    # Parameters
    dx = 2.0 / n_points
    max_iter = 100000
    T0 = 0.1
    alpha = 0.99
    sigma_start = 0.1
    sigma_decay = 0.9
    min_iter_for_accept = 500
    patience = 1000
    
    # Initialize
    h_current = h_init.copy()
    best_h = h_current.copy()
    
    # Compute initial C5
    def compute_C5(h):
        conv = correlate(h, 1-h, mode='full')
        return np.max(conv) * dx
    
    C5_current = compute_C5(h_current)
    best_C5 = C5_current
    
    start_time = time.time()
    
    # Annealing schedule with adaptive sigma
    T = T0
    sigma = sigma_start
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Perturb
        delta_h = np.random.normal(0, sigma, n_points)
        h_candidate = h_current + delta_h
        
        # Project to feasible region [0,1] and rescale sum to n/2
        h_candidate = np.clip(h_candidate, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_candidate)
        h_candidate *= scale_factor
        
        # Compute new C5
        C5_candidate = compute_C5(h_candidate)
        
        # Accept or reject
        if C5_candidate < C5_current:
            h_current = h_candidate
            C5_current = C5_candidate
            if C5_candidate < best_C5:
                best_C5 = C5_candidate
                best_h = h_candidate.copy()
                last_improvement = i
        else:
            if i > min_iter_for_accept:
                prob = np.exp(-(C5_candidate - C5_current) / T)
                if np.random.rand() < prob:
                    h_current = h_candidate
                    C5_current = C5_candidate
                    
        # Update temperature and sigma
        T *= alpha
        if i % 100 == 0:
            sigma *= sigma_decay
            
        if i - last_improvement > patience:
            break
    
    return (best_h, best_C5, n_points)
