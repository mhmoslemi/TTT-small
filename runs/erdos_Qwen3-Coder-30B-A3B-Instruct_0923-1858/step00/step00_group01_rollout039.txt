import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    max_iter = 1000
    no_improve_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Helper to compute C5
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(corr)
    
    # Compute initial C5
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Determine regions of overlap
        shift = max_k - (len(h) - 1)
        overlap_start = max(0, -shift)
        overlap_end = min(len(h), len(h) - shift)
        
        # Generate candidates by local perturbations
        candidates = []
        for _ in range(20):  # 20 candidates per iteration
            h_new = h.copy()
            i = np.random.randint(0, len(h))
            j = np.random.randint(0, len(h))
            
            # Simple redistribution
            transfer = min(h_new[i], 1 - h_new[j])
            if transfer > 1e-6:
                amount = np.random.uniform(0, transfer)
                h_new[i] -= amount
                h_new[j] += amount
                
                # Ensure bounds
                h_new[i] = max(0, min(1, h_new[i]))
                h_new[j] = max(0, min(1, h_new[j]))
                
                # Ensure sum constraint approximately
                current_sum = np.sum(h_new) * dx
                if abs(current_sum - 1.0) > 1e-5:
                    # Scale to meet constraint
                    scale = 1.0 / current_sum
                    h_new *= scale
                
                # Add to candidates
                c5_new = compute_c5(h_new)
                if c5_new < max_val:
                    candidates.append((h_new, c5_new))
                    
        if candidates:
            # Select best candidate
            best_candidate = min(candidates, key=lambda x: x[1])
            h = best_candidate[0]
            current_c5 = best_candidate[1]
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        if no_improve_count > 100:
            break
    
    # Interpolate if length < 100 (to increase resolution)
    if len(best_h) < 100:
        n_new = min(100, max(20, len(best_h)*2))
        eps = 1e-10
        h_new = np.zeros(n_new)
        dx_new = 2.0 / n_new
        for i in range(len(best_h)):
            x_start = i * dx
            x_end = (i+1) * dx
            for j in range(n_new):
                xj_start = j * dx_new
                xj_end = (j+1) * dx_new
                overlap_start = max(x_start, xj_start)
                overlap_end = min(x_end, xj_end)
                if overlap_end > overlap_start:
                    overlap = overlap_end - overlap_start
                    h_new[j] += best_h[i] * overlap / dx
                    
        # Ensure sum constraint
        h_new *= 1.0 / np.sum(h_new) / dx_new
        best_h = h_new
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
