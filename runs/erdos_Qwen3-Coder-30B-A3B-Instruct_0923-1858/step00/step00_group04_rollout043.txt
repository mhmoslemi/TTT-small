import numpy as np
import cvxpy as cp
from time import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Default construction if no initial is available
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
    
    dx = 2.0 / n_points
    max_k = n_points - 1
    start_time = time()
    
    # Initial evaluation
    h = h0.copy()
    h = h / np.sum(h) * (n_points / 2)  # ensure sum is exactly n_points/2
    overlaps = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(overlaps) * dx
    
    # Constraint-based optimization approach
    best_h = h.copy()
    best_c5 = current_c5
    
    # Iteratively add the most violating constraints
    max_iterations = 50
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get all overlaps
        overlaps = correlate(h, 1 - h, mode='full')
        cur_max = np.max(overlaps) * dx
        
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h.copy()
            
        # Collect constraints for the most violated ones
        # Use 30% of the largest overlaps
        cutoff = max(1, n_points//3)
        top_indices = np.argsort(overlaps)[-cutoff:]
        top_shifts = np.arange(-max_k, max_k+1)[top_indices]
        
        # Set up the optimization problem
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Build constraints
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2, C5 >= 0]
        
        # Add overlap constraints
        for shift in top_shifts:
            k = shift + max_k
            constraints.append(overlaps[k] * dx <= C5)
            
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h = h_new / np.sum(h_new) * (n_points / 2)
            else:
                break  # No improvement
        except Exception:
            break
            
    # Final evaluation
    overlaps = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(overlaps) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h, best_c5, n_points)
