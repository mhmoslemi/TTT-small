import numpy as np
import time
from scipy import correlate

def evaluate_C5(h):
    """Evaluate the C5 constant for a given step function h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    C5 = np.max(np.abs(corr)) * dx
    return C5

def project_to_feasible(h):
    """Project h to satisfy the constraint sum(h) = n/2"""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    # Ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize the population of candidates"""
    population = []
    if initial_h is not None:
        # Use the initial construction as the first individual
        population.append(initial_h.copy())
    else:
        # Generate a random individual with correct sum
        h = np.random.rand(n_points)
        h = project_to_feasible(h)
        population.append(h)
        
    # Fill the rest of the population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), k, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Perform blend crossover (BLX-0) between two parents"""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        a, b = parent1[i], parent2[i]
        range_ = abs(a - b) * alpha
        lower = min(a, b) - range_
        upper = max(a, b) + range_
        child[i] = np.random.uniform(lower, upper)
    return child

def mutate(individual, mutation_rate=0.1, noise_std=0.01):
    """Mutate an individual"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, noise_std)
    return project_to_feasible(mutated)

def ga_optimize(n_points, initial_h=None, pop_size=30, max_generations=1000, 
                mutation_rate=0.1, crossover_prob=0.7, elite_rate=0.1,

    """Genetic algorithm for optimizing C5"""
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_overall = float('inf')
    best_individual = None
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > time_budget_s:
            break
            
        # Evaluate fitness of population
        fitnesses = [evaluate_C5(ind) for ind in population]
        
        # Track best individual
        best_idx = np.argmin(fitnesses)
        best_fitness = fitnesses[best_idx]
        if best_fitness < best_overall:
            best_overall = best_fitness
            best_individual = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination if no improvement for many generations
        if no_improvement_count > 200:
            break
            
        # Sort population by fitness
        sort_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sort_indices]
        sorted_fitnesses = [fitnesses[i] for i in sort_indices]
        
        # Keep elite
        elite_size = int(elite_rate * pop_size)
        new_population = sorted_pop[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_prob:
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                child = blend_crossover(parent1, parent2)
                child = mutate(child, mutation_rate)
            else:
                parent = tournament_selection(population, fitnesses)
                child = mutate(parent, mutation_rate)
            new_population.append(child)
            
        population = new_population[:pop_size]
        
    return best_individual, best_overall
