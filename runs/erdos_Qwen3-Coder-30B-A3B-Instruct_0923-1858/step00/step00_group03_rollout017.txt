import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations (overlaps)
    overlaps = correlate(h, 1 - h, mode='full')
    # Only valid overlaps (from shift 0 to n-1)
    overlaps = overlaps[n-1:]
    C5 = np.max(overlaps) * dx
    return C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Using initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.round(h).astype(int)
    else:
        # Initialize with all 1's first, then set half to 0
        n = 70
        h = np.ones(n, dtype=int)
        # Flip half of them to 0
        flip_indices = np.random.choice(n, n//2, replace=False)
        h[flip_indices] = 0

    # Check and fix sum constraint
    current_sum = np.sum(h)
    target_sum = len(h) // 2
    if current_sum != target_sum:
        # Find 1s and 0s to flip
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            need_flip = current_sum - target_sum
            flip_ones = np.random.choice(ones, need_flip, replace=False)
            h[flip_ones] = 0
        else:
            # Need to flip some 0s to 1s
            need_flip = target_sum - current_sum
            flip_zeros = np.random.choice(zeros, need_flip, replace=False)
            h[flip_zeros] = 1

    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    current_time = time.time()
    iteration = 0

    while current_time - start_time < budget_s:
        # Identify the most problematic shift
        n = len(h)
        dx = 2.0 / n
        overlaps = correlate(h, 1 - h, mode='full')
        valid_overlaps = overlaps[n-1:]
        k_max = np.argmax(valid_overlaps)
        max_overlap = valid_overlaps[k_max]
        if max_overlap <= best_C5 * 0.9999:  # prevent precision issues
            break

        # For each position, evaluate change in max overlap if flipped
        best_delta = 0
        best_flip_i = -1
        best_flip_to = -1
        
        for i in range(n):
            # Try flipping h[i]
            old_val = h[i]
            new_val = 1 - old_val
            h[i] = new_val
            
            # Evaluate impact on overlap
            new_overlaps = correlate(h, 1 - h, mode='full')
            new_valid_overlaps = new_overlaps[n-1:]
            new_max_overlap = np.max(new_valid_overlaps)
            
            if new_max_overlap < max_overlap:
                delta = max_overlap - new_max_overlap
                if delta > best_delta:
                    best_delta = delta
                    best_flip_i = i
                    best_flip_to = new_val

            h[i] = old_val  # revert

        if best_delta <= 0:
            # No improvement - try to randomize periodically
            if iteration % 100 == 0 and iteration > 0:
                # Randomly perturb solution
                idx = np.random.randint(n)
                h[idx] = 1 - h[idx]
            iteration += 1
            current_time = time.time()
            continue

        # Apply best flip
        # find the other side if needed
        if best_flip_to == 0:
            # need to flip a 0 to a 1 to maintain constraint
            zeros = np.where(h == 0)[0]
            if len(zeros) < 1:
                iteration += 1
                current_time = time.time()
                continue
            flip_other_idx = np.random.choice(zeros)
            h[flip_other_idx] = 1
        else:
            # flip a 1 to a 0
            ones = np.where(h == 1)[0]
            if len(ones) < 1:
                iteration += 1
                current_time = time.time()
                continue
            flip_other_idx = np.random.choice(ones)
            h[flip_other_idx] = 0

        h[best_flip_i] = best_flip_to
        iteration += 1
        current_time = time.time()
        
        # Update best if improved
        new_C5 = evaluate_erdos_solution(h)
        if new_C5 < best_C5:
            best_h = h.copy()
            best_C5 = new_C5

    if np.sum(best_h) != len(best_h) // 2:
        # Fix sum by adjusting
        current_sum = np.sum(best_h)
        target_sum = len(best_h) // 2
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            need_flip = current_sum - target_sum
            ones = np.where(best_h == 1)[0]
            flip_indices = np.random.choice(ones, need_flip, replace=False)
            for i in flip_indices:
                best_h[i] = 0
        else:
            # Need to flip some 0s to 1s
            need_flip = target_sum - current_sum
            zeros = np.where(best_h == 0)[0]
            flip_indices = np.random.choice(zeros, need_flip, replace=False)
            for i in flip_indices:
                best_h[i] = 1
    
    return (best_h.tolist(), best_C5, len(best_h))
