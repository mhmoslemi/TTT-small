import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """Compute the overlap integral C_5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value in the correlation is the C_5 bound
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n/2."""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to enforce sum(h) = n/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

def optimize_h(h0, n, max_iter=2000, lr=1e-3, tol=1e-5):

            break
            
        c5 = evaluate_h(h)

            best_c5 = c5
            best_h = h.copy()

        grad = np.zeros_like(h)

            h_temp = project_h(h_temp, n)
            c5_minus = evaluate_h(h_temp)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient descent step
        h = h - lr * grad
        h = project_h(h, n)

            break

    return best_h, best_c5

def upsample_h(h_low, n_high):
    """Upsample a lower resolution h to higher resolution."""

    h_high = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n_low), h_low)
    return project_h(h_high, n_high)
