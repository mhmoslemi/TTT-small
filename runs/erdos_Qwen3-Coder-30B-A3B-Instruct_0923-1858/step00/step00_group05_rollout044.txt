import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial sum is correct
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    # Evaluate initial C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # GA parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 10
    
    # Encode constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # GA implementation
    population = []
    fitness_scores = []
    
    # Generate initial population
    for _ in range(population_size):
        h_new = np.random.rand(n_points)
        h_new = h_new * target_sum / np.sum(h_new)
        population.append(h_new)
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute fitness (C5) for all individuals
        fitness_scores = []
        for i, individual in enumerate(population):
            c5 = compute_c5(individual)
            fitness_scores.append(c5)
        
        # Find best individuals
        sorted_indices = np.argsort(fitness_scores)
        best_individuals = [population[i] for i in sorted_indices[:10]]
        best_fitness = fitness_scores[sorted_indices[0]]
        
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[sorted_indices[0]].copy()
        
        # Selection and reproduction
        new_population = []
        
        # Elitism (keep best 10)
        for i in range(10):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Tournament selection and crossover/mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # 80% chance of crossover
            if np.random.rand() < crossover_rate:
                parent2_idx = np.random.choice(len(population), tournament_size)
                parent2_fitness = [fitness_scores[i] for i in parent2_idx]
                parent2_idx = parent2_idx[np.argmin(parent2_fitness)]
                parent2 = population[parent2_idx]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Apply mutation
                if np.random.rand() < mutation_rate:
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child1[i] = np.clip(child1[i] + np.random.normal(0, 0.05), 0, 1)
                if np.random.rand() < mutation_rate:
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child2[i] = np.clip(child2[i] + np.random.normal(0, 0.05), 0, 1)
                
                # Renormalize to target sum
                child1 = child1 * target_sum / np.sum(child1)
                child2 = child2 * target_sum / np.sum(child2)
                
                new_population.extend([child1, child2])
            else:
                # Clone and mutate
                child = parent1.copy()
                if np.random.rand() < mutation_rate:
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
        
        # Truncate to exact size
        population = new_population[:population_size]
    
    # Local refinement phase on best individuals
    local_best_h = best_h.copy()
    local_best_c5 = best_c5
    
    local_candidates = []
    # Add the best from GA to candidates and a few others too
    local_candidates.append(best_h.copy())
    
    for i in range(5):
        if i < len(population):
            local_candidates.append(population[i].copy())
    
    # Gradient-based refinement of top candidates with slightly different algorithm
    for candidate in local_candidates:
        # Construct a function to optimize
        def obj(h_vec):
            # Make sure vector is in bounds [0,1]
            h_vec = np.clip(h_vec, 0, 1)
            # Re-normalize if needed (not strictly necessary since GA should keep it in balance but let's be safe)
            h_vec = h_vec * target_sum / np.sum(h_vec)
            return compute_c5(h_vec)
        
        # Use differential evolution as a robust local search
        try:
            res = differential_evolution(obj, bounds, seed=seed, maxiter=200, popsize=15, 
                                        mutation=(0.5, 1), recombination=0.7, atol=1e-6)
            refined_h = np.clip(res.x, 0, 1)
            refined_h = refined_h * target_sum / np.sum(refined_h)
            refined_c5 = compute_c5(refined_h)
            
            if refined_c5 < local_best_c5:
                local_best_c5 = refined_c5
                local_best_h = refined_h.copy()
        except:
            pass  # If it fails, keep current best

    # Final validation
    final_h = local_best_h.copy()
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
