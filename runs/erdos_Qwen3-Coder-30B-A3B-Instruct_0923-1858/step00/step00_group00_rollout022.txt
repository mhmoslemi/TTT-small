import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, dx):
    n = len(h)
    # Compute the correlation
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum is over all shifts from - (n-1) to n-1
    # We want the maximum difference at any shift
    max_val = 0.0
    for i in range(len(conv)):
        max_val = max(max_val, conv[i] * dx)
    return max_val

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints"""
    # Ensure bounds [0,1]
    h = np.clip(h, 0, 1)
    # Scale so that sum = n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm(initial_h, n_points, budget_s):
    start_time = time.time()
    n_gen = 200
    pop_size = 50
    mutation_rate = 0.1
    crossover_alpha = 0.5
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        # Create random feasible solution
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
        
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_fitness = fitness[sorted_indices[0]]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[sorted_indices[0]].copy()
            
        # Selection (tournament)
        new_pop = []
        elite_count = pop_size // 5
        for i in range(elite_count):
            new_pop.append(population[sorted_indices[i]].copy())
        
        # Create offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            tourn_size = 5
            idx1 = min(random.sample(range(pop_size), tourn_size), 
                      key=lambda i: fitness[i])
            idx2 = min(random.sample(range(pop_size), tourn_size), 
                      key=lambda i: fitness[i])
            
            # Crossover
            h1 = population[idx1]
            h2 = population[idx2]
            child = h1 * crossover_alpha + h2 * (1 - crossover_alpha)
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_to_feasible(child, n_points)
                
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
    return best_h, best_c5

def gradient_descent_step(h, dx, max_iter=100):
    # Simple gradient descent using finite differences
    epsilon = 1e-5
    eta = 0.01
    
    for t in range(max_iter):
        # Compute C5 value
        c5_current = evaluate_c5(h, dx)
        
        # We want to minimize C5, so we compute gradient with respect to h
        # Approximate gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Perturb h[i]
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, len(h))
            
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[i] = (c5_plus - c5_current) / epsilon
            
        # Update h
        h = h - eta * grad
        
        # Project to feasible region
        h = project_to_feasible(h, len(h))
        
        # Update learning rate
        eta *= 0.995
    
    return h
