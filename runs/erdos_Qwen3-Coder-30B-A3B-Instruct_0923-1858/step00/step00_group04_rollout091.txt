import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction or a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5  # Simple midpoint initialization
        
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate overlap integral
    def compute_C5(h_vals):
        # Full correlation
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        # Normalize by dx
        corr *= dx
        # Return maximum value
        return np.max(corr)
    
    # Iterate to improve solution
    start_time = time.time()
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        current_C5 = compute_C5(h)
        
        # Create a simplified problem using CVXPY
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = []
        
        # Box constraints
        constraints += [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Overlap constraints
        # For full correlation calculation, we'll use a subset of the constraints
        # to make it computationally feasible
        full_corr = correlate(h, 1 - h, mode="full")
        shifts = list(range(-n_points + 1, n_points))
        # Select top violations
        violations = list(zip(shifts, full_corr))
        violations.sort(key=lambda x: x[1], reverse=True)
        top_violations = violations[:max(10, n_points // 4)]
        
        for k, _ in top_violations:
            # Compute only necessary correlations
            if k >= 0:
                corr = np.sum(h_var[k:] * (1 - h_var[:n_points-k]))
            else:
                corr = np.sum(h_var[:n_points+k] * (1 - h_var[-k:]))
            constraints.append(corr <= C5_var / dx)
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Formulate and solve
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                new_h = np.clip(new_h, 0, 1)
                h = new_h
                continue
        except:
            pass
            
        # Fallback to gradient descent on a random subset
        # Get current overlaps
        full_corr = correlate(h, 1 - h, mode="full")
        
        # Try a perturbation-style local search (simplest for time)
        perturbation = np.random.normal(0, 0.01, n_points)
        # Keep within bounds
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        # Normalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept if better
        new_C5 = compute_C5(new_h)
        if new_C5 < current_C5:
            h = new_h
        
        # Early termination if no improvement in several iterations
        if iteration % 20 == 0:
            if compute_C5(h) >= current_C5:
                break

    final_C5 = compute_C5(h)
    return (h.tolist(), float(final_C5), n_points)
