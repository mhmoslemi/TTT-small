import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from the provided construction
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the constraint ∫₀² h(x) dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization
    learning_rate = 0.01
    max_iter = 100000
    tol = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap integral for all lags
        corr = signal.correlate(h, 1 - h, mode='full')
        # The full correlation has 2*n_points - 1 elements
        # The actual overlap is scaled by dx
        overlaps = corr * dx
        
        # Find the maximum overlap (C5)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Gradient w.r.t. h (for the dominating lag)
        # Use finite differences for computational efficiency
        # Shift the derivative computation to account for the correlation
        
        # Approximate gradient: contribution to overlap at max_k lag
        grad = np.zeros_like(h)
        # For the shift k=max_k - (n_points - 1), we proceed carefully:
        # The full correlation runs from lag -(n_points-1) to (n_points-1)
        # So max_k = k + n_points -1 where k is actual lag
        actual_lag = max_k - (n_points - 1)
        
        for i in range(n_points):
            # Shift h by actual_lag in the negative direction
            # So we compute d/dh[i] of sum over j: h[j]* (1- h[j+actual_lag])
            j = i - actual_lag
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx  # Weighted by dx
                
        # Apply gradient update
        h = h - learning_rate * grad
        
        # Project onto feasible region: 0 <= h[i] <= 1
        h = np.clip(h, 0, 1)
        
        # Adjust sum to satisfy ∫ h(x) dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping based on objective improvement
        if max_overlap < tol:
            break
    
    # Final refinement
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # return the best feasible solution
    return (best_h.tolist(), final_c5, n_points)
