def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximize over all lags
    max_corr = np.max(corr)
    # Return the bound
    return max_corr * dx

def clip_and_rescale(h):
    """Clip h to [0,1] and rescale to satisfy integral constraint"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale
    n = len(h)
    dx = 2.0 / n
    integral = np.sum(h) * dx
    if integral > 0:
        h = h / integral * 1.0  # scale to make integral = 1
    return h
