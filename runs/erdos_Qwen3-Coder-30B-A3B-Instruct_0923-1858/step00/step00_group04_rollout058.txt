import numpy as np

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Frank-Wolfe with perturbations
    max_iter = 2000
    pert_freq = 100
    eta = 0.1
    epsilon = 0.02
    
    for i in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        # Compute gradients
        # Using finite difference approach for gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            # Rescale to sum constraint
            h_plus = h_plus - np.mean(h_plus) + (n_points / 2) / n_points
            h_minus = h_minus - np.mean(h_minus) + (n_points / 2) / n_points
            val_plus = evaluate_erdos_solution(h_plus, dx)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            grad[j] = (val_plus - val_minus) / (2 * eps)
        
        # Linearized subproblem
        # Minimize <grad, d> subject to 0 <= d <= 1, sum(d) = 0
        # which is equivalent to minimizing <grad, d> subject to sum(d) = 0
        d = -grad  # direction of descent
        d = d - np.mean(d)  # make zero-sum
        d = np.clip(d, -1, 1)
        d = d - np.mean(d)
        
        # Line search
        best_step = 0
        best_val = best_c5
        for step in [0.05, 0.1, 0.2, 0.3, 0.5]:
            h_new = h + step * d
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new - np.mean(h_new) + (n_points / 2) / n_points
            val = evaluate_erdos_solution(h_new, dx)
            if val < best_val:
                best_val = val
                best_step = step
        
        if best_step > 0:
            h = h + best_step * d
            h = np.clip(h, 0, 1)
            h = h - np.mean(h) + (n_points / 2) / n_points
        else:
            # No progress, try perturbation
            h = h + np.random.normal(0, epsilon, n_points)
            h = np.clip(h, 0, 1)
            h = h - np.mean(h) + (n_points / 2) / n_points
        
        # Periodic evaluation
        if i % 10 == 0:
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Perturb periodically
        if i % pert_freq == 0 and i > 0:
            h = h + np.random.normal(0, epsilon, n_points)
            h = np.clip(h, 0, 1)
            h = h - np.mean(h) + (n_points / 2) / n_points
    
    return (best_h, best_c5, n_points)
