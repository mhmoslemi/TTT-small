import numpy as np
import time
from scipy import signal

def evaluate_h(h_values):
    """Evaluate the overlap integral C5 for given h values."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute all overlaps using full convolution
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of these correlations
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction or create a binary version
    if 'initial_h_values' in globals():

    else:

        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    # Ensure the sum constraint
    current_sum = np.sum(h)
    target_sum = 35.0  # n_points/2
    if current_sum != target_sum:
        # Find positions to flip
        diff = int(target_sum - current_sum)
        if diff > 0:
            # Need to flip some 0s to 1s
            zero_indices = np.where(h == 0)[0]
            flip_indices = zero_indices[:diff]
            h[flip_indices] = 1
        else:
            # Need to flip some 1s to 0s
            one_indices = np.where(h == 1)[0]
            flip_indices = one_indices[:abs(diff)]
            h[flip_indices] = 0
    
    # Evaluate initial solution
    best_c5 = evaluate_h(h)
    best_h = h.copy()
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Precompute the convolution for each shift
    def compute_all_overlaps(h_vec):
        # This is an efficient way to compute all overlaps for all shifts
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        return corr * dx
    
    def compute_impact(h_vec, k):
        """Compute the impact of flipping a single bit at k."""
        # For a single bit at position i, compute change in overlap at k
        # The key insight: only the shifts where h[i] or h[i+k] contribute
        # Note: changing h[i] affects terms: h[i](1-h[i+k]) and h[i+k](1-h[i])
        # Compute the net effect
        return 0  # Implements later
    
    iterations = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < 20000:
        iterations += 1
        
        # Compute current overlaps
        corr_full = compute_all_overlaps(h)
        max_k = np.argmax(corr_full)
        max_overlap = corr_full[max_k]
        
        # Check if improvement is significant
        if max_overlap < best_c5 - 1e-8:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = iterations
        
        # If no improvement in last 500 iterations, reset
        if iterations - last_improvement > 500:
            # Reset to the best solution so far
            h = best_h.copy()
            last_improvement = iterations
        
        # Find the worst (maximum) overlap
        # Shift character: 
        # For a shift k, the shift index in full convolution is sliced 
        # at positions [k : n_points - k] when valid. 
        k_indices = np.arange(len(corr_full))
        k_of_max = k_indices[max_k]

        # Now find which positions are critical for this shift
        # Use a greedy strategy:
        # Look at each position i, and estimate the effect of changing h[i] 
        # on the current max overlap (or at least the k of max overlap)
        delta_max = None
        best_flip = (-1, -1)  # (from_pos, to_pos) or (pos, None) if flipping to 0
        
        # For a change in position i
        # Effect on correlation at shift k: 
        # old_term = h[i] * (1 - h[i+k])
        # new_term = new_h[i] * (1 - new_h[i+k])
        # Change is (new_term - old_term)
        # Do this for changing h[i] from 0 to 1, or 1 to 0
        
        for i in range(n_points):
            old_h_i = h[i]
            new_h_i = 1 - old_h_i
            
            # Check change on the k-th shift due to flipping h[i]
            # Noting that shift k corresponds to indices such that 
            # the overlap term is from h[i] * (1 - h[i+k])
            # and also h[i+k] * (1 - h[i])
            # But since we only know i, and fixed k, 
            # we compute flip effect for knowing change to one overlap
        
            # When shifting is k from our full correlation:
            # actual_positions in correlation = [0, n_points-1]
            # If k = 0, position 0 corresponds to h[0] * (1 - h[0])  
            # If k = 1, h[0]*1-h[1] etc.
            # We are in the middle so shift_k_index = k + n_points - 1 
            # Hence, correlation at index k+shift gives term h[shift]* (1 - h[shift+k])
            if i + k_of_max < n_points: 
                # Check effect if we changed i to new_h_i
                old_contribution = h[i] * (1 - h[i+k_of_max]) 
                new_contribution = new_h_i * (1 - h[i+k_of_max])
                delta = new_contribution - old_contribution
                
                # Apply to the cost function
                if delta_max is None or abs(delta) > abs(delta_max):
                    delta_max = delta
                    best_flip = (i, None)
            
            # Also check if flip from 1 to 0 is beneficial by also flipping i+k to 1
            # This maintains sum if we flip both bits at i and at i+k
            # Only if i+k < n_points and i+k is the shift index 
            # and the following is useful
            # If i+k is not the shift, we still can look at how flip i and i+k 
            # would affect this particular correlation term
        
        # To keep sum constraint, we must flip 2 bits if flipping 1 to 0
        # Flip most beneficial flip seen
        
        if best_flip[0] >= 0:
            i = best_flip[0]
            # Now, flip the bit if beneficial in that overlap
            # Also make sure to maintain the sum constraint
            # Try: accept the change if benefits the overlap
            if h[i] == 1:
                # Flip it to 0, keep sum constraint by flipping a 0 to 1
                # Find an index with h[j] = 0
                j_indices = np.where(h == 0)[0]
                if len(j_indices) > 0:
                    j = j_indices[0]  # Take first 0
                    # Update to flip both
                    h[i] = 0
                    h[j] = 1
                    # Check if this improves the situation
                    # Or keep checking the impact of previous
            else:
                # Flip 0 to 1, we will also flip a 1 to 0 
                # to maintain sum
                j_indices = np.where(h == 1)[0]
                if len(j_indices) > 0:
                    j = j_indices[0]
                    h[i] = 1
                    h[j] = 0

        # Reevaluate after the change to determine if it was beneficial
        current_c5 = evaluate_h(h)
        if current_c5 < best_c5 - 1e-10:
            best_c5 = current_c5
            best_h = h.copy()
        # else: might revert but don't strictly value it
    
        if iterations % 50 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break

    # To be safe, final validation
    final_c5 = evaluate_h(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), float(final_c5), len(best_h))
