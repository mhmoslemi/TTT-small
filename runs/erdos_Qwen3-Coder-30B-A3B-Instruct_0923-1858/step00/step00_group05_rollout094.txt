import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h):
    # Compute overlap integral using full cross-correlation
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) == n_points:
            # Normalize to sum = n_points/2
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points) * 0.5  # fallback
    else:
        h = np.ones(n_points) * 0.5
    return h

def constraint_sum(h, target_sum):
    return np.sum(h) - target_sum

def optimize_h_ga(n_points, budget_s):
    start_time = time.time()
    # Genetic algorithm parameters
    pop_size = 100
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 10
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = initialize_h(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure correct sum
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
        
        # Create offspring via crossover and mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
                    
            # Enforce bounds
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            
            # Renormalize
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_pop.extend([child1, child2])
        
        # Replace population
        population = new_pop[:pop_size]
    
    return best_h, best_c5

def refine_local(h, n_points):
    # Use scipy's minimize with constraints to refine the solution
    def objective(h_vec):
        h = h_vec.reshape(-1)
        return evaluate_h(h)
    
    def constraint_sum_eq(h_vec):
        h = h_vec.reshape(-1)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum_eq}
    
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 500, 'ftol': 1e-8})
    
    if res.success:
        h_refined = res.x.reshape(-1)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        c5_refined = min(evaluate_h(h_refined), evaluate_h(h))
        return h_refined, c5_refined
    else:
        return h, evaluate_h(h)

def run(seed=42, budget_s=600, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    h = initialize_h(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5_bound = evaluate_h(h)
    
    # Run genetic algorithm phase
    best_h, best_c5 = optimize_h_ga(n_points, budget_s / 2)
    
    # Refine best with local search
    final_h, final_c5 = refine_local(best_h, n_points)
    final_c5 = evaluate_h(final_h)
    
    # Ensure constraints hold
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_h(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
