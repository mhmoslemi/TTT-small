import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the overlap integral"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, n_points):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def gradient(h, dx):
    """Compute gradient with respect to overlap integral"""
    n = len(h)
    # compute correlation and gradient for each time shift
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # for simplicity, we'll compute an approximate gradient
    # by finite difference
    grad = np.zeros_like(h)
    eps = 1e-5
    base_val = evaluate(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, n)
        val_plus = evaluate(h_plus, dx)
        grad[i] = (val_plus - base_val) / eps
    # normalize
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    return grad
