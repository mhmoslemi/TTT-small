import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from typing import Tuple, List

def evaluate_h(h: np.ndarray, dx: float) -> float:
    """Evaluate the maximum overlap integral for given h."""
    # Compute cross-correlation h * (1 - h)
    correlation = np.correlate(h, 1 - h, mode='full')
    # Return the maximum value scaled by dx
    return np.max(correlation) * dx

def fitness(h: np.ndarray, dx: float) -> float:
    """Fitness function to minimize (negative of max overlap)."""
    return -evaluate_h(h, dx)

def project_to_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project h to satisfy sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def genetic_algorithm_step(h: np.ndarray, dx: float, n_points: int,

    population = []
    for _ in range(population_size):
        new_h = np.random.random(n_points) * 0.5 + 0.25  # Start with values around 0.375
        new_h = project_to_feasible(new_h, n_points / 2)
        population.append(new_h)
    
    best_h = None
    best_fitness = float('inf')
    
    for _ in range(generations):
        # Evaluate population
        fitnesses = []
        for individual in population:
            fit = fitness(individual, dx)
            fitnesses.append(fit)
            if fit < best_fitness:
                best_fitness = fit
                best_h = individual.copy()
        
        # Selection (tournament selection)
        selected = []
        tournament_size = 10
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1, parent2 = selected[i], selected[i+1] if i+1 < len(selected) else selected[i]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            mutation_rate = 0.1
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[j] = np.random.random()
                if np.random.random() < mutation_rate:
                    child2[j] = np.random.random()
            
            # Project to feasible set
            child1 = project_to_feasible(child1, n_points / 2)
            child2 = project_to_feasible(child2, n_points / 2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population[:population_size]
    
    return best_h, -best_fitness

    """Refine solution using local optimization."""
    def objective(x):
        return -evaluate_h(x, dx)
    
    # Box constraints for bounds (0,1) for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Fix the sum constraint by removing one degree of freedom

    def constraint_func(x):
        return np.sum(x) - n_points / 2
    
    # Create a projection function to ensure the sum constraint
    def project_h(x):
        x = np.array(x)
        return project_to_feasible(x, n_points / 2)
    
    best_h = h.copy()
    best_value = -objective(h)
    
    # Try different optimization methods
    for _ in range(5):  # Try several local optimizations
        x0 = np.random.random(n_points) * 0.5 + 0.25  # Random start
        x0 = project_to_feasible(x0, n_points / 2)
        try:
            # Use minimize for constrained optimization
            result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': max_iter}, tol=1e-8)
            if result.success:
                optimized_h = project_h(result.x)
                value = -objective(optimized_h)
                if value < best_value:
                    best_value = value
                    best_h = optimized_h.copy()
        except Exception:
            pass  # Skip if optimization failed
    
    return best_h, best_value
