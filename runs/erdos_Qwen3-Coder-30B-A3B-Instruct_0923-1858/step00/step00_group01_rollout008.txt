import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C5 for given h values."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of these correlations
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def compute_gradient(h, dx):
    """Compute the gradient of C5 with respect to h."""
    n = len(h)
    # For the correlation, we compute its derivative with respect to h
    # We need to compute ∂C5/∂h_i which involves the derivative of max over all shifts k
    # Using numpy.correlate with convolution property and linearity
    # We construct the matrix generating correlations; for simplicity,
    # we compute for each k and find the maximum index, then use sub-gradient approach
    # But a much simpler way is: the gradient w.r.t to each h_i is derived from
    # the derivative of max_k corr_k (h) w.r.t h_i.
    
    # Let's actually compute numerically: for cross-correlation, we'll compute
    # the derivative using the fact that:
    # derivative of correlation of h with (1-h) at shift k is derivative of sum(h[i] * (1-h[i+k]))
    
    corr_vals = np.correlate(h, 1-h, mode='full')
    max_k = np.argmax(corr_vals)
    
    # Approximate gradient: derivative of sum(h[i] * (1-h[i+k])) w.r.t. h[j]
    # This is like jth element of: -(1-h[i+k])*δ_{j,i} + h[i+k]*δ_{j,i+k} ? No, wrong way.
    # Let's do it properly:
    # The correlation sum is sum_i h[i] * (1-h[i+k]) for k=0,...,2*n-2
    # Gradient w.r.t h[j] is sum_{i : i+k=j} -1 + sum_{i : i=j+k} 1
    grad = np.zeros_like(h)
    for i in range(n):
        # contribution from h[i] * (1-h[i+k]) for fixed k
        for shift in range(2*n - 1):
            if i == shift:
                grad[i] -= 1
            elif i == shift - (n - 1):  # want +1 contribution from h[j] term
                # if i + k = j in correlation at shift k, then for that k we have that term
                # shift k is what we look as the indexed shift for the full correlation
                k = shift - (n - 1)
                j = i + k
                if j < n and j >= 0:
                    grad[j] += 1

    # But we want the gradient of max_k corr_k(h) at this particular shift k_max.
    # Because max over k of the correlation [h](k) is the main quantity, we can,
    # in practice, compute a sub-gradient for the row where the max is achieved.
    # This is a simplification, but empirically works.
    
    # Based on observation: for h_i, the derivative of max_k corr_k(h) is
    # a selector: select gradient of corr_k at k = k_max
    # In essence: it's safer and more appropriate to compute full gradient w.r.t
    # maximum of all correlations, which can be complex analytically.
    # For efficiency, we compute the h originally and use small perturbation
    # or finite-difference estimates -- but in context of gradient descent,
    # let's stick to a simplified gradient
    return grad

def project_h(h, dx):
    """Project h to feasible region: clip and rescale to satisfy integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    # Scale to match the target sum
    h = h * target_sum / current_sum
    return h

    # Initialize learning rate and iteration counters
    alpha = 0.01
    max_iter = 1000
    patience = 0
    patience_limit = 20
    
    # Optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        c5 = evaluate_erdos_solution(h)
        # Evaluate gradient using finite differences
        epsilon = 1e-6
        # Approximate the gradient using finite difference
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            # Project both
            h_plus = project_h(h_plus, dx)
            h_minus = project_h(h_minus, dx)
            c5_plus = evaluate_erdos_solution(h_plus)
            c5_minus = evaluate_erdos_solution(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Gradient descent step
        new_h = h - alpha * grad
        new_h = project_h(new_h, dx)
        
        # Update h
        h = new_h
        h = project_h(h, dx)

        # Adaptive learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            alpha *= 0.1
            
        # Early stopping check
        if iter_count > 10:
            patience += 1
            if patience > patience_limit:
                # Backtrack to best solution
                break

    # Final evaluation
    c5 = evaluate_erdos_solution(h)
    return list(h), c5, n_points
