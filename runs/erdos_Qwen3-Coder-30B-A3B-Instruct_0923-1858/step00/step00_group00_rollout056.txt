import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    # Project h onto constraints: h[i] in [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, n_points):
    # Compute C5 = max_k integral of h(x)(1-h(x+k))dx
    dx = 2.0 / n_points
    # Correlate h with (1-h) with full mode
    corr = correlate(h, 1 - h, mode="full")
    # Get the maximum value (corresponds to max overlap)
    max_overlap = np.max(corr)
    return max_overlap * dx

def nelder_mead_simplex_optimize(h0, n_points, budget_s):
    # Nelder-Mead implementation for minimizing C5
    # Initialize simplex around h0
    n = len(h0)
    simplex = np.array([h0] * (n + 1))
    # Set initial step size (small to allow fine search)
    step_size = 0.1
    for i in range(1, n + 1):
        simplex[i] = h0.copy()
        simplex[i][i-1] += step_size
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, n_points)
    
    max_iter = 5000  # Max iterations
    iteration = 0
    no_improvement_count = 0
    max_reset_attempts = 10
    reset_attempts = 0
    
    # Nelder-Mead parameters
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    beta = 0.5   # contraction
    rho = 0.5    # shrink
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Project all vertices to feasible region
        for i in range(n + 1):
            simplex[i] = project_h(simplex[i], n_points)
        
        # Evaluate C5 for all vertices
        values = []
        for i in range(n + 1):
            c5_val = evaluate_c5(simplex[i], n_points)
            values.append(c5_val)
        values = np.array(values)
        
        # Order vertices by evaluation
        indices = np.argsort(values)
        simplex = simplex[indices]
        values = values[indices]
        
        # Check for improvement
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Reset simplex if no improvement over 50 iterations
        if no_improvement_count >= 50 and reset_attempts < max_reset_attempts:
            step_size /= 2
            if step_size < 0.01:
                break  # Give up if too small
            for i in range(1, n+1):
                simplex[i] = best_h.copy()
                simplex[i][i-1] += step_size
            no_improvement_count = 0
            reset_attempts += 1
        
        # Centroid excluding the worst vertex
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        worst = simplex[-1]
        h_r = centroid + alpha * (centroid - worst)
        h_r = project_h(h_r, n_points)
        c5_r = evaluate_c5(h_r, n_points)
        
        # Check if reflection is better than worst
        if values[0] <= c5_r < values[-2]:
            simplex[-1] = h_r
            continue
        
        # Expansion
        if c5_r < values[0]:
            h_e = centroid + gamma * (h_r - centroid)
            h_e = project_h(h_e, n_points)
            c5_e = evaluate_c5(h_e, n_points)
            if c5_e < c5_r:
                simplex[-1] = h_e
            else:
                simplex[-1] = h_r
            continue
        
        # Contraction
        h_c = centroid + beta * (worst - centroid)
        h_c = project_h(h_c, n_points)
        c5_c = evaluate_c5(h_c, n_points)
        
        if c5_c < values[-1]:
            simplex[-1] = h_c
            continue
        
        # Shrink
        for i in range(1, n + 1):
            simplex[i] = simplex[0] + rho * (simplex[i] - simplex[0])
    
    return best_h, best_c5

    h_opt, c5_bound = nelder_mead_simplex_optimize(h, n_points, budget_s)
    
    # Ensure final solution meets requirements
    h_opt = project_h(h_opt, n_points)
