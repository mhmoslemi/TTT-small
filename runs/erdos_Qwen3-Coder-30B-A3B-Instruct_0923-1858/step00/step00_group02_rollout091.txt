def evaluate_h(h_vals, n_points):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h)
    conv = np.correlate(h_vals, 1 - h_vals, mode="full")
    # The maximum value over all shifts is our bound
    max_overlap = np.max(conv) * dx
    return max_overlap
