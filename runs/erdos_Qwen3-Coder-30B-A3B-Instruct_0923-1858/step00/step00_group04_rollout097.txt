import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    # Calculate correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximum correlation value scaled by dx
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, n_points):
    # Ensure all values in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, rate=0.7):
    # Blend crossover: create child by blending parents
    if np.random.random() > rate:
        return parent1.copy()
    mask = np.random.random(len(parent1)) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    # Gaussian mutation
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + noise * mask
    return h

def tournament_selection(population, fitness, k=5):
    # Select the best individual from k random individuals
    selected = np.random.choice(len(population), k)
    best_idx = selected[np.argmax([fitness[i] for i in selected])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    # Start from initial if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Random initialization
        initial_h = np.random.rand(n_points)
        # Clip and project to sum constraint
        initial_h = project_to_feasible(initial_h, n_points)
    population_size = 30
    generations = 0
    max_generations = 1000
    max_no_improve = 200
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate population
    fitness = []
    for h in population:
        fit = evaluate_h(h, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_fitness = fitness[best_idx]
    no_improve_count = 0
    start_time = time.time()
    # For checkpointing
    checkpoint_interval = 50
    last_checkpoint = 0
    
    while time.time() - start_time < budget_s and no_improve_count < max_no_improve:
        generations += 1
        # Sort by fitness (low to high)
        sorted_idx = np.argsort(fitness)
        if generations % checkpoint_interval == 0:
            # Save best so far
            print(f"Gen {generations}: Best C5 = {best_fitness:.6f}")
        
        # Generate offspring
        new_population = [population[i] for i in sorted_idx[:population_size // 10]]  # Elite
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = project_to_feasible(child, n_points)
            new_population.append(child)
        
        # Replace old population
        population = new_population
        
        # Evaluate new population
        fitness = []
        for h in population:
            fit = evaluate_h(h, dx)
            fitness.append(fit)
        
        # Update best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_fitness:
            best_fitness = fitness[new_best_idx]
            best_h = population[new_best_idx]
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    return (best_h, best_fitness, n_points)
