import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters

    dx = 2.0 / n_points

        assert len(h0) == n_points
    except:
        # Generate initial guess
        h0 = np.ones(n_points) * 0.5

    # Optimization parameters
    max_iter = 100
    time_limit = time() + budget_s
    
    # Variables
    h = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C >= 0
    ]

    # Initialize with current best
    best_C = float('inf')
    best_h = h0.copy()

    # Optimization loop
    for iteration in range(max_iter):
        if time() > time_limit:
            break
            
        # Approximate overlap constraints
        overlap_constraints = []
        h_current = best_h.copy()
        
        # Compute overlaps for all shifts
        corr = np.correlate(h_current, 1-h_current, mode='full')
        shifts = np.arange(-n_points+1, n_points)
        overlap_values = corr * dx
        
        # Select the worst violations for tighter constraints
        violations = np.where(overlap_values > best_C * 0.99)[0]
        if len(violations) == 0:
            break
        # Use top 10% violations
        top_violations = np.argsort(overlap_values[violations])[::-1][:max(1, len(violations)//10)]
        selected_shifts = violations[top_violations]
        for k in selected_shifts:
            if k >= 0:
                shift_overlap = np.sum(h_current[k:] * (1 - h_current[:n_points-k]))
            else:
                shift_overlap = np.sum(h_current[:n_points+k] * (1 - h_current[-k:]))
            shift_overlap *= dx
            overlap_constraints.append(shift_overlap <= C)
        
        # Add new constraints
        constraints.extend(overlap_constraints)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                new_C = C.value
                new_h = h.value
                if new_C < best_C:
                    best_C = new_C
                    best_h = new_h.copy()
                # Revert constraints for next loop, but keep only tight constraints
                constraints = constraints[:-len(overlap_constraints)]
            else:
                # If infeasible or error, reduce problem size
                best_h = h_current
                break
        except:
            # If optimization fails, use current incumbent
            break
    
    # Final check
    if best_C > target:
        # Try slightly more global optimizations
        h_global = best_h.copy()
        for i in range(20):
            # Gradient-based perturbation
            pert = np.random.normal(0, 0.02, size=n_points)
            h_test = h_global + pert
            h_test = np.clip(h_test, 0, 1)
            h_test *= (n_points / 2) / np.sum(h_test)
            
            # Compute current C5
            corr = np.correlate(h_test, 1-h_test, mode='full')
            shifts = np.arange(-n_points+1, n_points)
            overlap_values = corr * dx
            max_C = np.max(overlap_values)
            if max_C < best_C:
                best_C = max_C
                best_h = h_test
        
        # Final verify
        corr = np.correlate(best_h, 1-best_h, mode='full')
        overlap_values = corr * dx
        final_C = np.max(overlap_values)
        assert final_C <= best_C + 1e-6
        
    # Enforce sum constraint
    best_h *= (n_points / 2) / np.sum(best_h)
    
    # Final check
    corr = np.correlate(best_h, 1-best_h, mode='full')
    overlap_values = corr * dx
    final_C = np.max(overlap_values)
    
    # Return results
    return (best_h.tolist(), float(final_C), n_points)
