import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Default construction: piecewise constant
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        # Make it feasible: sum = n/2 = 21
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    max_shift = n_points - 1
    
    def compute_overlap(h):
        # Compute full correlation (overlap) for all shifts
        # h is indexed from 0 to n_points-1
        # needs to be padded for correlation
        h_pad = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1 - h
        one_minus_h_pad = np.concatenate([one_minus_h, np.zeros(n_points)])
        # Full cross-correlation
        corr = correlate(h_pad, one_minus_h_pad, mode='full')
        # Extract relevant part: indices from n_points-1 to 2*n_points-2
        overlap = corr[n_points-1:2*n_points-1] * dx
        return overlap
    
    def evaluate(h):
        overlap = compute_overlap(h)
        return np.max(overlap)
    
    # Try the initial solution first
    current_h = h0.copy()
    current_c5 = evaluate(current_h)
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Iteration vars
    iterations = 0
    max_iter = 100
    tol = 1e-5
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute current overlaps
        overlap = compute_overlap(current_h)
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Assemble constraints
        # We'll add some most-violated constraints to refine improvements
        # Only consider shifts k s.t. overlap[k] > current_c5 / dx or k close to max_idx
        
        # Sort indices by overlap
        sort_order = np.argsort(overlap)[::-1]
        # Scores based on how much they deviate from current estimate
        violations = np.where(overlap > current_c5 / dx)[0]
        if len(violations) > 0:
            # For these, we keep them in constraints
            active_indices = violations[:10] # Limit number to avoid over-constraining
        else:
            # If no violations, add top 10
            active_indices = sort_order[:10]
            
        # Set up constraints using those k values
        try:
            h_var = Variable(n_points)
            C5_var = Variable()
            
            # Box constraints
            constrs = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
            
            # Add overlap constraints for active k
            for k in active_indices:
                shift = k - (n_points - 1)
                # Due to convolution structure, overlap about index k from left of correlation
                # For left-part of h, overlap[k] = sum(h[i] * (1 - h[i+shift])) for valid i
                if shift >= 0: # h shifts right
                    h1 = h_var[:n_points-shift]
                    h2 = h_var[shift:]
                    term = sum(h1 * (1 - h2)) * dx
                else: # negative shift
                    h1 = h_var[-shift:]
                    h2 = h_var[:n_points+shift]
                    term = sum(h1 * (1 - h2)) * dx
                constrs.append(term <= C5_var / dx)
            
            # Solve
            objective = Minimize(C5_var)
            prob = Problem(objective, constrs)
            # Use a fast solver to reduce overhead
            prob.solve(solver=ECOS, verbose=False, max_iters=100)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value.copy()
                new_c5 = C5_var.value
                # Validate feasibility
                if np.all(0 <= new_h) and np.all(new_h <= 1) and abs(np.sum(new_h) - n_points/2) < 1e-3:
                    current_h = new_h
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
                        
        except Exception:
            pass
            
        iterations += 1
    
    # Final check
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
