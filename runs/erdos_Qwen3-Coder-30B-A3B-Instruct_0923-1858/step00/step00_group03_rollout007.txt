import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C_5 for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute all overlaps using full convolution
    correlation = signal.correlate(h, 1 - h, mode="full")
    # The maximum overlap corresponds to the maximum value in the correlation
    max_overlap = np.max(correlation)
    # Return the scaled maximum overlap which is C_5
    return max_overlap * dx

def project_h(h, n_points):
    """Project h onto feasible set: [0,1] and sum=h_sum"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to hit the sum constraint
    h_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - h_sum
    if abs(diff) > 1e-10:
        # Distribute the difference uniformly
        h += diff / n_points
        # Clamp again after redistribution
        h = np.clip(h, 0, 1)
    return h
