import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute the overlap integral C5
    conv = correlate(h, 1 - h, mode='full')
    max_conv = np.max(conv) * dx
    return max_conv

def project_h(h, n_points):
    # Project h to [0,1] and normalize to sum = n_points / 2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def nelder_mead_step(vertices, values, alpha=1.0, gamma=2.0, beta=0.5, rho=0.5):
    # Sort by function values (ascending)
    idxs = np.argsort(values)
    vertices = vertices[idxs]
    values = values[idxs]
    
    n = len(vertices) - 1
    best = vertices[0]
    worst = vertices[-1]
    second_worst = vertices[-2]
    
    # Compute centroid (excluding worst)
    centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    r = centroid + alpha * (centroid - worst)
    r = project_h(r, len(r))
    r_value = evaluate_h(r, 2.0 / len(r))
    
    if values[0] <= r_value < values[-2]:
        vertices[-1] = r
        values[-1] = r_value
    elif r_value < values[0]:
        # Expansion
        e = centroid + gamma * (r - centroid)
        e = project_h(e, len(e))
        e_value = evaluate_h(e, 2.0 / len(e))
        if e_value < r_value:
            vertices[-1] = e
            values[-1] = e_value
        else:
            vertices[-1] = r
            values[-1] = r_value
    else:
        # Contraction
        if r_value < values[-1]:
            c = centroid + beta * (r - centroid)
        else:
            c = centroid + beta * (worst - centroid)
        c = project_h(c, len(c))
        c_value = evaluate_h(c, 2.0 / len(c))
        if c_value < values[-1]:
            vertices[-1] = c
            values[-1] = c_value
        else:
            # Shrink
            for i in range(1, len(vertices)):
                vertices[i] = best + rho * (vertices[i] - best)
                vertices[i] = project_h(vertices[i], len(vertices[i]))
            values = np.array([evaluate_h(v, 2.0 / len(v)) for v in vertices])
    
    return vertices, values

    n_points = len(h)

    h = project_h(h, n_points)
    
    # Initialize simplex
    n_dim = len(h)
    vertices = np.tile(h, (n_dim + 1, 1))
    for i in range(n_dim):
        vertices[i+1, i] += 0.1
    
    # Evaluate initial values
    values = np.array([evaluate_h(v, dx) for v in vertices])
