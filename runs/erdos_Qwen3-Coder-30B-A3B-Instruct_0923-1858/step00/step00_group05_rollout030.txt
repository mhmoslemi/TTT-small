import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        # Compute the correlation: h * (1 - h) with full mode
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The maximum value is what we want
        return np.max(corr) * dx

    def project_h(h_vec):
        # Ensure values are in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so that sum equals target_sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    def compute_gradient(h_vec, c5_val):
        # linear approximation of gradient for small perturbations
        # This is heuristic gradient; in real cases may be more complex
        dx = 2.0 / len(h_vec)
        # Estimate gradient using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_val) / eps
        # Gradient descent update
        return grad

    # Optimize at low resolution first
    if n_points > 50:
        # Downsample to 50 points
        h_high = h.copy()
        h = np.interp(np.linspace(0, 2, 50), np.linspace(0, 2, n_points), h)
        h = project_h(h)
        n_points = 50

    # Optimization loop
    best_c5 = compute_c5(h)
    best_h = h.copy()
    initial_c5 = best_c5
    alpha = 1e-2
    patience = 0
    max_patience = 10
    iter_count = 0

    while time.time() - start_time < budget_s - 5 and iter_count < 1000:
        c5_val = compute_c5(h)
        grad = compute_gradient(h, c5_val)
        # Gradient descent
        h = h - alpha * grad
        h = project_h(h)
        
        # Track improvements
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                alpha *= 0.9
                patience = 0
                
        iter_count += 1

    # Try upsampling to higher resolution
    upsampled_h = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, len(best_h)), best_h)
    upsampled_h = project_h(upsampled_h)

    h = upsampled_h.copy()
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Reduce learning rate for more stable fine-tuning
    alpha = 1e-3
    patience = 0
    max_patience = 20  # less patience because we're fine-tuning
    old_c5 = compute_c5(h)
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        c5_val = compute_c5(h)
        grad = compute_gradient(h, c5_val)
        h = h - alpha * grad
        h = project_h(h)
        new_c5 = compute_c5(h)
        if new_c5 < old_c5:
            old_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                alpha *= 0.9
                patience = 0

    c5_final = compute_c5(h)
    
    # Final validation and return
    h_final = h.copy()
    c5_final = compute_c5(h_final)
    
    return (h_final.tolist(), c5_final, len(h_final))
