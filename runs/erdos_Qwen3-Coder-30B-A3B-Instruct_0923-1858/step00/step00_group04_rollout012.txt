import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    # Compute overlap integral C5 = max_k integral h(x)(1-h(x+k))dx
    # Using convolution to efficiently compute cross-correlation
    n = len(h)
    dx = 2.0 / n
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    C5 = max_corr * dx
    return C5

def initialize_population(n_points, pop_size, initial_h=None):
    # Create a population of feasible step functions
    pop = []
    if initial_h is not None:
        pop.append(initial_h.copy())
    # Fill the rest with random feasible functions
    for _ in range(pop_size - len(pop)):
        h = np.random.random(n_points)
        h = h * (n_points / 2.0 / np.sum(h))  # rescale to sum = n_points/2
        h = np.clip(h, 0, 1)
        pop.append(h)
    return pop

def selection_tournament(pop, fitnesses, k=5):
    # Tournament selection
    selected = []
    for _ in range(len(pop)):
        indices = np.random.choice(len(pop), k)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(pop[best_idx].copy())
    return selected

def crossover_blend(parent1, parent2, alpha=0.3):
    # Blend crossover with alpha parameter
    n = len(parent1)
    child = np.zeros(n)
    for i in range(n):
        lb = min(parent1[i], parent2[i]) - alpha * abs(parent1[i] - parent2[i])
        ub = max(parent1[i], parent2[i]) + alpha * abs(parent1[i] - parent2[i])
        child[i] = np.random.uniform(lb, ub)
    return child

def mutate(h, mutation_rate=0.1, epsilon=0.1):
    # Mutate a step function by adding small random perturbations
    h_new = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            h_new[i] += np.random.normal(0, epsilon)
    # Normalize and clip to [0,1]
    total = np.sum(h_new)
    h_new = h_new * (n / 2.0) / total
    h_new = np.clip(h_new, 0, 1)
    return h_new

def genetic_algorithm(n_points, pop_size=30, max_generations=1000, 
                      mutation_rate=0.1, crossover_prob=0.7, 
                      initial_h=None, budget_s=600):
    start_time = time.time()
    pop = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_C5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in pop:
            C5 = evaluate_h(h)
            fitnesses.append(C5)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
        
        # Check for early termination (no improvement for many generations)
        if generation > 200 and generation % 100 == 0:
            if all(f > best_C5 for f in fitnesses):
                break
                
        # Selection
        selected = selection_tournament(pop, fitnesses)
        
        # Elitism: keep top 10%
        elite_size = max(1, pop_size // 10)
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elite = [pop[i] for i in elite_indices]
        
        # Generate offspring
        new_pop = elite.copy()
        while len(new_pop) < pop_size:
            if np.random.random() < crossover_prob:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child = crossover_blend(parent1, parent2)
                child = mutate(child, mutation_rate)
                new_pop.append(child)
            else:
                # Just mutate
                parent = np.random.choice(selected)
                child = mutate(parent, mutation_rate)
                new_pop.append(child)
                
        pop = new_pop[:pop_size]
    
    return best_h, best_C5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42  # small enough to be fast, large enough for good resolution
    
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    
    best_h, best_C5 = genetic_algorithm(
        n_points, pop_size=30, max_generations=1000,
        mutation_rate=0.1, crossover_prob=0.7,
        initial_h=initial_h, budget_s=budget_s
    )
    
    # Final validation
    if best_h is None:
        if initial_h is not None:
            best_h = initial_h.copy()
        else:
            best_h = np.ones(n_points) * 0.5
    
    return (best_h, best_C5, n_points)
