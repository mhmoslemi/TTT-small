import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise create a symmetric start
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 1.0)
        h[:n_points//2] = 0.0
        h[n_points//2:] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_c5(h):
        h1 = 1 - h
        conv = np.correlate(h, h1, mode="full")
        max_corr = np.max(conv)
        return max_corr * dx
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    def perturb_h(h, sigma):
        noise = np.random.normal(0, sigma, len(h))
        h_new = h + noise
        return project_h(h_new)
    
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    # Parameters
    sigma = 0.1
    alpha = 0.99
    temp = 1.0
    patience = 0
    max_patience = 1000
    
    # Annealing schedule
    for iter_count in range(1000000):
        if time.time() - start_time > budget_s:
            break
            
        # Perturb
        h_candidate = perturb_h(best_h, sigma)
        
        # Compute new C5
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = h_candidate
            best_c5 = c5_candidate
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            break
            
        # Update sigma and temp
        if iter_count % 100 == 0:
            sigma *= 0.9
        temp *= alpha
    
    # Final verification
    final_h = project_h(best_h)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
