import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h[1] = 0.1
        h[-2] = 0.1
        h = h / np.sum(h) * (n_points / 2.0)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 500
    no_improve_count = 0
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        # Perturb the current solution
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet sum constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2.0)
        
        # Compute new C5
        C5_new = evaluate_C5(h_new, dx)
        
        # Accept or reject the new solution
        delta_C5 = C5_new - best_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cooling schedule
        T *= alpha
        if iter_count % 100 == 0:
            perturb_scale *= 0.99
            
        # Early stopping if no improvement for a while
        if no_improve_count > no_improve_limit:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx):
    # Compute the overlap integral
    correlation = signal.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx
