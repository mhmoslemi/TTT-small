import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the constraint ∫h(x)dx = 1 (h.sum() * dx = 1)
    h_sum = h.sum()
    h *= (n_points / 2) / h_sum

    # Optimization parameters
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-6
    last_c5 = float('inf')
    
    for i in range(max_iter):
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # The correlation is indexed from -n+1 to n-1
        # The maximal overlap C5 is the maximum of these
        c5_val = np.max(corr) * dx
        if c5_val < last_c5 - tolerance:
            last_c5 = c5_val
        else:
            # Simple adaptive step size backoff
            step_size *= 0.99
        
        # Check time budget
        if time.time() - start_time > budget_s - 1.0:
            break
        
        # Find dominant lag
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        
        # Gradient: partial derivative of correlation w.r.t h[i]
        # For h[i] contribution to overlap at lag k, we look at the indices where
        # h[j] * (1 - h[j+k]) contributes to the dot product. This is non-trivial.
        # Instead, use gradient of the correlation computed using finite differences
        # or compute derivative explicitly.
        # The correct derivative of correlation at lag k w.r.t. h[i] is:
        # if (i + k >= 0 and i + k < n_points), then derivative is -1 * (1 - h[i+k])
        # but this doesn't give the right full picture for all lags.
        # So we use the gradient w.r.t. to the peak lag.
        
        # For now, compute gradient at the peak lag explicitly
        # We want to reduce overlap at lag 'lag'
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + lag < n_points:
                grad[j] = -(1 - h[j + lag])
            if 0 <= j - lag < n_points:
                grad[j] += -(1 - h[j - lag])
        
        # Apply gradient descent: step in direction that reduces overlap
        h_new = h - step_size * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust for integral constraint
        h_sum_new = h_new.sum()
        h_new *= (n_points / 2) / h_sum_new
        
        # Update h
        h = h_new
        
    # Compute final value of C5
    corr_final = correlate(h, 1 - h, mode='full')
    c5_final = np.max(corr_final) * dx
    
    return (h.tolist(), c5_final, n_points)
