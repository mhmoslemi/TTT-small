import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral C_5
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(h, dx):
    # Ensure h is in [0,1] and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def crossover_h(p1, p2, dx):
    # Blend two parents to create offspring
    if len(p1) != len(p2):
        raise ValueError("Parents must have same length")
    n = len(p1)
    crossover_point = np.random.randint(1, n-1)
    child = np.zeros(n)
    
    # Take first part from p1, second part from p2
    child[:crossover_point] = p1[:crossover_point]
    child[crossover_point:] = p2[crossover_point:]
    
    # Repair to meet constraints
    child = repair_h(child, dx)
    return child

def mutate_h(h, dx, mutation_rate=0.05):
    # Apply small random perturbations
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            # Add small noise
            delta = np.random.normal(0, 0.05)
            h[i] += delta
    h = repair_h(h, dx)
    return h

def genetic_search(n_points, dx, initial_h=None, max_time=600):

    population_size = 60
    elite_size = int(0.2 * population_size)

        population.append(initial_h.copy())
    else:
        # Create initial random individuals
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)

            population.append(h)

        sorted_indices = np.argsort(fitness)

        # Update best solution
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        
        # Selection: keep top elite_size individuals
        elites = population[:elite_size]

        new_pop = elites[:]

                child = crossover_h(p1, p2, dx)
                child = mutate_h(child, dx, mutation_rate)
                new_pop.append(child)
            else:
                # If we don't have enough elites, create random individual
                h = np.random.rand(n_points)
                h = repair_h(h, dx)
                new_pop.append(h)
        
        population = new_pop
    
    return best_h, best_c5
