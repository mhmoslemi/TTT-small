def evaluate_overlap(h, dx):
    # Compute all overlaps
    overlaps = np.correlate(h, 1 - h, mode="full")
    # Select only the relevant ones (maximum overlap)
    max_overlap = np.max(overlaps)
    return max_overlap * dx
