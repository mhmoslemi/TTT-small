import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a basic construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
        dx = 2.0 / n_points
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize if needed
    current_sum = np.sum(h)
    h = h * (n_points / 2.0) / current_sum
    
    # Helper function to compute C5
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        # Compute convolution of h with (1-h)
        conv = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Adjust for indexing and return max
        return np.max(conv) * dx
    
    # Project h to feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        h_vec = h_vec * (n_points / 2.0) / current_sum
        return np.clip(h_vec, 0, 1)
    
    # Gradient descent step
    def optimize_step(h_vec, step_size=0.01):
        # Get max overlap index to compute gradients
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_k = np.argmax(conv)
        max_val = conv[max_k]
        
        # Create gradient
        gradient = np.zeros_like(h_vec)
        # For overlap with shift k, gradient of h[i] = (1 - h[i+k]) for i+k<n
        # and gradient of h[i+k] = -h[i] for i<n-k
        h_shift = np.concatenate([h_vec, [0]*max_k])  # Concatenate to match size
        for i in range(n_points):
            if i + max_k < n_points:
                gradient[i] += (1 - h_vec[i + max_k])
                if i < n_points - max_k:
                    gradient[i + max_k] -= h_vec[i]
        # Apply step (and scale by dx implicitly)
        h_new = h_vec - step_size * gradient
        h_new = project_h(h_new)
        return h_new
    
    # Initial C5
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    # Gradient descent loop
    step_size = 0.05
    iteration = 0
    convergence_threshold = 1e-6
    last_improvement = best_C5
    patience = 0
    
    while time.time() - start_time < budget_s:
        if patience > 50:
            break
            
        h_new = optimize_step(h, step_size) 
        C5 = compute_C5(h_new)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_new.copy()
            patience = 0
            last_improvement = C5
        else:
            patience += 1
            
        h = h_new
        
        # Adjust step size
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.9
            
        iteration += 1
        
        if np.abs(last_improvement - C5) < convergence_threshold:
            patience += 1
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final projection
    best_h = project_h(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
