import numpy as np
import time
from scipy import optimize
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the C5 constant for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value (in terms of correlation indices) gives the C5 bound
    C5 = np.max(corr) * dx
    return C5

def project_h(h, n_points):
    """Project h to the feasible set [0,1] and maintain sum constraint."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    return h

def trust_region_update(h, grad, trust_radius, n_points):
    """Update h using a trust region approach."""
    # Compute trust region step
    step = -grad
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    h_new = h + step
    # Project back to feasible region
    h_new = project_h(h_new, n_points)
    return h_new

def finite_diff_gradient(h, n_points, dx):
    """Numerically compute gradient using finite differences."""
    grad = np.zeros_like(h)
    eps = 1e-5
    h_eval = evaluate_C5(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n_points)
        grad[i] = (evaluate_C5(h_plus) - evaluate_C5(h_minus)) / (2 * eps)
    return grad

def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)
    start_time = time.time()

        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2)  # Adjust for sum constraint
    
    # Ensure feasible start
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    trust_radius = 0.1
    max_iter = 10000
    last_improvement = 0
    iter_count = 0
    
    dx = 2.0 / n_points
    
    while True:
        if time.time() - start_time > budget_s - 1:
            break
            
        iter_count += 1
        # Evaluate current gradient
        grad = finite_diff_gradient(h, n_points, dx)
        
        # Update using trust region
        h_new = trust_region_update(h, grad, trust_radius, n_points)
        
        # Evaluate new objective
        new_C5 = evaluate_C5(h_new)
        
        # Accept update if improvement
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            last_improvement = iter_count
            trust_radius = min(0.1, trust_radius * 2.0)
        else:
            trust_radius = max(1e-6, trust_radius * 0.5)
            
            # Occasionally perturb if no improvement
            if iter_count - last_improvement > 100:
                noise = np.random.uniform(-0.05, 0.05, n_points)
                h_noisy = h + noise
                h_noisy = project_h(h_noisy, n_points)

                if noisy_C5 < best_C5:
                    h = h_noisy
                    best_C5 = noisy_C5
                    last_improvement = iter_count
        
        # Terminate if converged or time limit
