import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create initial triangular waveform
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    
    dx = 2.0 / n_points
    
    def evaluate_c5(h_vals):
        # Using full correlation to capture all k values
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # We only want the overlapping parts: from 0 to n_points-1
        corr = corr[n_points-1:]
        return np.max(corr) * dx
    
    def compute_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure validity
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * eps)
        return grad
    
    # Main optimization loop
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        
        iter_count += 1
        
        # Compute gradient
        try:
            grad = compute_gradient(h)
        except:
            break
        
        # Compute step within trust region
        step = -grad * trust_radius / np.linalg.norm(grad)
        
        # Limit step to stay in [0,1] and maintain sum constraint
        h_new = h + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            trust_radius = min(2 * trust_radius, 0.5)
        else:
            trust_radius *= 0.5
            if trust_radius < 1e-6:
                # Add small perturbation to escape local minimum
                pert = np.random.normal(0, 0.01, n_points)
                h_new = h + pert
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = evaluate_c5(h_new)
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    best_h = h.copy()
                trust_radius = 0.1
        if trust_radius < 1e-6:
            trust_radius = 0.1
    
    return (best_h.tolist(), best_c5, n_points)
