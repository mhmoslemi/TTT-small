import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Initialize with triangular shape
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure initial h satisfies the constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, n_points)
    
    # Optimization parameters
    learning_rate = 0.01
    perturb_rate = 0.05
    decay_every = 100
    perturb_every = 50
    max_iter = 10000
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Evaluate current state
        current_C5 = evaluate_C5(h, n_points)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Compute gradient (approximated via finite differences)
        gradient = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            C5_plus = evaluate_C5(h_plus, n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / np.sum(h_minus) * (n_points / 2)
            C5_minus = evaluate_C5(h_minus, n_points)
            
            gradient[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Update step
        h = h - learning_rate * gradient
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Perturb occasionally
        if iteration % perturb_every == 0:
            noise = np.random.normal(0, perturb_rate, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
        # Decay learning rate
        if iteration % decay_every == 0:
            learning_rate *= 0.95
            
    return (best_h, best_C5, n_points)

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    # Use correlate with full mode to compute overlap for all shifts
    # correlate(h, 1-h) == [sum h[i] (1-h[i+j]) for all valid shifts j]
    correlation = correlate(h, 1 - h, mode='full')
    # Take the maximum correlation value and multiply by dx
    C5 = np.max(correlation) * dx
    return C5
