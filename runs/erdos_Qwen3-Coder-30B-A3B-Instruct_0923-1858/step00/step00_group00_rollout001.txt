import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h[1] = 0.5
        h[-2] = 0.5

    # Discretization
    dx = 2.0 / n_points
    max_iter = 10000
    temperature = 0.1
    decay_rate = 0.99
    perturb_scale = 0.1
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        perturbation = np.random.normal(0, perturb_scale, n_points)
        h_candidate = h + perturbation
        
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        c5 = compute_c5(h_candidate, dx)
        
        # Accept or reject
        if c5 < best_c5:
            h = h_candidate
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if np.random.random() < np.exp(-(c5 - best_c5) / temperature):
                h = h_candidate
            if no_improvement_count > 1000:
                # Reset if no improvement for too long
                h = best_h.copy()  # Reset to best known
                no_improvement_count = 0
                perturb_scale *= 0.9
                temperature *= decay_rate
        
        # Temperature and perturbation decay
        temperature *= decay_rate
        if iteration % 100 == 0:
            perturb_scale *= 0.9

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    # Compute the overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # Using scipy's correlate with 'full' mode
    # 1-h is the flipped version of h
    conv = correlate(h, 1 - h, mode='full')
    # The full convolution includes zero-padding, but we want to extract 
    # the part that corresponds to valid overlaps
    # Since h is length n_points and we're doing full convolution of h with 1-h,
    # result is length 2*n_points - 1
    # The central part, corresponding to overlap, is from index n_points-1 to 2*n_points-2
    # But we want max over all lags, so just take max of the full convolution
    max_overlap = np.max(conv) * dx
    return max_overlap
