import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h0)
    except NameError:
        n_points = 47
        # Create initial sparse construction
        h0 = np.zeros(n_points)
        # Place 1s at positions spaced by sqrt(n_points)
        spacing = max(1, int(np.sqrt(n_points)) + 1)
        for i in range(0, n_points, spacing):
            h0[i] = 1.0
        # Scale to have integral = 1
        h0 *= n_points / (2 * np.sum(h0))
    
    dx = 2.0 / n_points
    
    # Optimization function
    def obj(h):
        # Normalize to meet constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap integral for all shifts
        h_folded = np.concatenate([h, np.zeros(n_points-1)])
        h_rev = np.concatenate([np.zeros(n_points-1), 1-h])
        corr = np.convolve(h_folded, h_rev, mode='valid')
        c5 = np.max(corr) * dx
        return c5
    
    # Gradient of objective
    def grad_obj(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Direct gradient computation via finite differences assuming smoothness
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = h_p * (n_points / 2) / np.sum(h_p)
            
            h_m = h.copy()
            h_m[i] -= eps
            h_m = h_m * (n_points / 2) / np.sum(h_m)
            
            grad[i] = (obj(h_p) - obj(h_m)) / (2 * eps)
        
        return grad
    
    # Gradient-based optimization using L-BFGS-B
    def optimize_h(h_init):
        best_h = h_init.copy()
        best_val = obj(h_init)
        
        # Track iterations
        iter_count = 0
        
        # Use scipy.optimize.minimize with L-BFGS-B for constrained optimization
        # We'll manually project onto [0, 1] and integral constraint
        def fun(x):
            # Direct computation of objective without minimizing internal sum
            h = np.clip(x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return obj(h)
        
        def jac(x):
            return grad_obj(x)
        
        # Apply constraints in a loop
        def project(x):
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            return x
        
        x0 = h_init.copy()
        while time.time() - start_time < budget_s - 1 and iter_count < 1000:
            try:
                # At each iteration, do one optimization step
                res = minimize(fun, x0, method='L-BFGS-B', jac=jac, bounds=[(0,1)]*n_points, options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    new_x = project(res.x)
                    new_val = obj(new_x)
                    if new_val < best_val:
                        best_h = new_x.copy()
                        best_val = new_val
                    x0 = new_x
                else:
                    break
                iter_count += 1
            except Exception:
                break
                
        return best_h, best_val
    
    h_opt, c5_opt = optimize_h(h0)
    
    # Final check and cleaning
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    final_c5 = obj(h_opt)
    
    return (h_opt, final_c5, n_points)
