import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize initial h to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Solver parameters
    solver = 'MOSEK' if cp.installed_solvers() else 'SCS'
    
    # Main optimization loop
    start_time = time.time()
    max_iter = 1000
    trust_radius = 0.1
    rho = 0.1
    max_backtrack = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        overlaps = signal.correlate(h, 1 - h, mode='full')
        # Take the full overlap and normalize by step size (2 / n_points)
        dx = 2.0 / n_points
        c5_val = np.max(overlaps) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        # Approximate non-convex terms using linearization
        # Compute expected values for convolution
        expected_h = np.convolve(h, np.ones(n_points), mode='same') / n_points
        
        # Formulate convex relaxation
        # Objective: minimize max_k Σ h[i] - h[i] * E[h[i+k]] 
        # Linearize the product term h[i] * E[h[i+k]]
        # We want to minimize sum(h * (1 - expected_h))
        gradient = 1 - expected_h
        
        # Solve convex subproblem
        try:
            # Create quadratic program
            d = cp.Variable(n_points)
            
            # Convex objective: ||d||^2 + gradient^T * d
            obj = cp.Minimize(cp.norm(d, 2)**2 + gradient.T @ d)
            
            # Constraints
            constraints = [
                cp.sum(d) == 0,  # Maintain sum constraint
                d <= 1 - h,      # Upper bound
                d >= -h,         # Lower bound
            ]
            
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=solver, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                d_val = d.value
            else:
                d_val = np.zeros(n_points)
        except:
            d_val = np.zeros(n_points)
        
        # Line search
        alpha = 1.0
        for _ in range(max_backtrack):
            h_new = h + alpha * d_val
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new <= c5_val - rho * alpha * np.dot(gradient, d_val):
                h = h_new
                break
            alpha *= 0.5
            
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), c5_final, n_points)
