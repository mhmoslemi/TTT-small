def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""
    # Ensure h is a numpy array
    h = np.array(h)
    # Compute the overlap integral
    # We need to compute the maximum of all correlations
    n = len(h)
    dx = 2.0 / n
    # Request full correlation (including all shifts)
    corr = correlate(h, 1 - h, mode="full")
    # The correlation values are for shifts from -n+1 to n-1
    # Since our domain is [0,2], we want shifts from 0 to 2
    # But correlation shifts can be negative
    # We only care about positive shifts (0 to 2)
    # Since h has length n, correlation takes values from -n+1 to n-1
    # The relevant shifts for 0 to 2 length are from n-1 down to 0+1
    # But say shift k means shift k positions, the actual shift in interval is k*dx
    # We want shifts k*dx where 0 <= k*dx <= 2, i.e., 0 <= k <= n
    # But since correlation is for indices, we look at values at full correlation
    shift_idx = np.arange(-n + 1, n)
    shift_vals = shift_idx * dx
    # We are interested in the shifts that correspond to k*dx in [0,2]
    # This corresponds to shift_idx * dx >= 0 and shift_idx * dx <= 2
    # So shift_idx >= 0 and shift_idx <= n
    # The valid correlation indices range from n-1 to n-1 + n-1 = 2n-2
    # But correlation indices run from 0 to 2n-2
    # shift_idx = 0..2n-2, and we need shift_idx*dx in [0,2] => idx in [0,n]
    # But we actually care about indices where idx*dx in [0,2], idx in [0,n]
    valid_idx = np.where((shift_vals >= 0) & (shift_vals <= 2))[0]
    valid_correlations = corr[valid_idx]
    if len(valid_correlations) == 0:
        return 0.0
    return np.max(valid_correlations) * dx
