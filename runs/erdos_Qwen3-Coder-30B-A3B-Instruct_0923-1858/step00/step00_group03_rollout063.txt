import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def binary_convert(h, n_points):
    """Convert h to binary with exact sum constraint"""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h
    diff = target_sum - current_sum
    if diff > 0:  # Need to add 1s
        indices = np.argsort(h)
        for i in range(diff):
            h[indices[i]] = 1
    else:  # Need to subtract 1s
        indices = np.argsort(h)[::-1]
        for i in range(-diff):
            h[indices[i]] = 0
    return h

def compute_C5(h, dx):
    """Compute C5 from binary h"""
    # Compute correlation for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # Extract only valid correlations (non-zero shifts)
    mid = len(corr) // 2
    valid_corr = corr[mid + 1:]  # Exclude shift 0; also negative shifts don't count as valid for this problem
    # Take max of valid correlations
    max_corr = np.max(valid_corr)
    return max_corr * dx

    new_h = h.copy()
    new_h[i] = 0

    return new_h
