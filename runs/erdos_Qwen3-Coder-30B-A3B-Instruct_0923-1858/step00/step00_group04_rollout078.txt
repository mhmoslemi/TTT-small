import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[1::2] = 0.0
        h *= n_points / (2 * np.sum(h))
    
    # Optimization parameters
    max_iter = 10000
    perturb_freq = 100
    epsilon = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Identify indices causing maximum overlap
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        k = max_k - (n_points - 1)
        
        # Create a perturbed h to reduce overlap at k
        h_new = h.copy()
        
        # Direct adjustment: reduce h where it contributes to overlap at k
        # This corresponds to decreasing h[i] when i+k is in range
        # For correlation h[i] * (1 - h[(i + k) % n_points])
        for idx in range(n_points):
            if (idx + k) % n_points < n_points:
                # Add a small negative adjustment to reduce overlap
                if k < 0:
                    # h[i] * (1 - h[i - k]) if i - k >= 0
                    if idx - k >= 0:
                        h_new[idx] -= epsilon
                elif k == 0:
                    # h[i] * (1 - h[i]) 
                    h_new[idx] -= epsilon
                else:
                    # h[i] * (1 - h[i + k]) if i + k < n_points
                    if idx + k < n_points:
                        h_new[idx] -= epsilon
                        
        # Random weight adjustment: reduce some locations, increase others
        # at random to reduce correlation
        indices = np.random.permutation(n_points)
        for idx in indices[:n_points//20]:
            h_new[idx] = max(0, h_new[idx] - epsilon)
            
        for idx in indices[:n_points//20]:
            if np.random.rand() > 0.5:
                h_new[idx] = min(1, h_new[idx] + epsilon)
                
        # Normalize to maintain constraint sum(h) = n_points / 2
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Clip
        h_new = np.clip(h_new, 0, 1)
        
        # Apply perturbation periodically
        if i % perturb_freq == 0:
            # Distribute mass
            h_indices = np.argsort(h_new)
            for idx in h_indices[:n_points//20]:
                h_new[idx] = min(1, h_new[idx] + epsilon/2)
            for idx in h_indices[-n_points//20:]:
                h_new[idx] = max(0, h_new[idx] - epsilon/2)
            # Re-normalize and clip
            h_new *= (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
    
    # Final evaluation
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * (2.0 / n_points)
    
    return (best_h.tolist(), final_c5, n_points)
