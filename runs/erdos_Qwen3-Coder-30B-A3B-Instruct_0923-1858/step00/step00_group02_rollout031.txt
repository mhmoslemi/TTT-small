import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate C5 for given h and dx"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, dx, target_sum):
    """Project h to satisfy sum constraint"""
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    factor = target_sum / current_sum
    h_proj = h * factor
    # Clip to [0, 1]
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parse initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct triangular waveform
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Project to sum constraint
    h = project_h(h, dx, target_sum)
    
    # Optimization parameters
    trust_radius = 0.1
    min_trust_radius = 1e-6
    max_iter = 10000
    eps = 1e-6
    perturb_freq = 100
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient estimation using finite differences
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[j] += epsilon
            if h_plus[j] > 1:
                h_plus[j] = 1
            h_plus = project_h(h_plus, dx, target_sum)
            
            h_minus[j] -= epsilon
            if h_minus[j] < 0:
                h_minus[j] = 0
            h_minus = project_h(h_minus, dx, target_sum)
            
            c5_plus = evaluate_c5(h_plus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update step
        step = -grad
        
        # Project step to trust region
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
        
        # Try update
        h_new = h + step
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_h(h_new, dx, target_sum)
        
        # Evaluate new objective
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept update
        if new_c5 < best_c5 - eps:
            h = h_new.copy()
            best_c5 = new_c5
            trust_radius = min(2 * trust_radius, 0.5)
        else:
            trust_radius = max(trust_radius / 2, min_trust_radius)
        
        # Perturb every few iterations
        if i % perturb_freq == 0 and i > 0:
            noise = np.random.normal(0, 0.05, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = project_h(h_new, dx, target_sum)
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5 - eps:
                h = h_new.copy()
                best_c5 = new_c5
                
        if abs(best_c5 - 0.3807) < 1e-6:
            break
    
    return (best_h.tolist(), best_c5, n_points)
