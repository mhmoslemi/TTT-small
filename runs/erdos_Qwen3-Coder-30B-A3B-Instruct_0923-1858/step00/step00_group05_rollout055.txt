import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for the given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part (the maximum value)
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to enforce sum(h) = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, n_points):
    """Compute gradient of overlap with respect to h using finite differences"""
    # This is a simplified approach that uses a finite difference approximation
    # The actual gradient computation is complex, so we use a simple method:
    # approximate gradient using a small perturbation
    eps = 1e-5
    grad = np.zeros_like(h)
    base_overlap = evaluate_overlap(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        overlap_plus = evaluate_overlap(h_plus)
        grad[i] = (overlap_plus - base_overlap) / eps
    return grad

def optimize_step_function(h0, n_points, max_iter=1000, learning_rate=1e-3, tolerance=1e-5):
    """Optimize step function using gradient descent with projections"""
    h = h0.copy()
    last_overlap = evaluate_overlap(h)
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        # Compute gradient (can be improved with more accurate gradient)
        grad = compute_gradient(h, n_points)
        # Update step
        h = h - learning_rate * grad
        # Project onto feasible set
        h = project_h(h, n_points)
        # Evaluate new overlap
        current_overlap = evaluate_overlap(h)
        # Check improvement
        if abs(current_overlap - last_overlap) < tolerance:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        last_overlap = current_overlap
        # Decrease learning rate if no improvement
        if no_improvement_count > 10:
            learning_rate *= 0.9
            no_improvement_count = 0
        if no_improvement_count > 50:
            break
    return h, last_overlap

def initialize_h(n_points):
    """Initialize step function h"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to correct size
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
        h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        # Add a small amount of random noise
        np.random.seed(42)
        h += np.random.normal(0, 0.01, n_points)
        h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 70
    
    # Initialize step function
    h = initialize_h(n_points)
    
    # Optimizing directly at the target resolution
    h_opt, current_c5 = optimize_step_function(h, n_points, max_iter=500, learning_rate=1e-3)
    
    # Try improving with further refinement
    if current_c5 < best_c5 and (time.time() - start_time) < budget_s - 10:
        h_refined, refined_c5 = optimize_step_function(h_opt, n_points, max_iter=500, learning_rate=1e-4)
        if refined_c5 < current_c5:
            h_opt, current_c5 = h_refined, refined_c5
    
    # Since large enough C5 seems to be hard to find, try to crank up resolution if we're early
    if (time.time() - start_time) < budget_s / 2 and current_c5 > 0.381:
        # Try 100 points
        h_upsampled = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, 70), h_opt)
        h_upsampled = project_h(h_upsampled, 100)
        h_finely, final_c5 = optimize_step_function(h_upsampled, 100, learning_rate=1e-4, max_iter=1000)
        if final_c5 < current_c5:
            h_opt, current_c5 = h_finely, final_c5
    
    # Final validation of result
    if current_c5 < best_c5:
        best_h = h_opt
        best_c5 = current_c5

    # Early exit if no good solution was found (fallback to initial)
    if best_c5 == np.inf:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_c5 = evaluate_overlap(best_h)
        else:
            best_h = np.full(n_points, 0.5)
            best_c5 = evaluate_overlap(best_h)
        
    return (best_h, best_c5, n_points)
