        # Normalize
        h = h / np.sum(h) * (n_points / 2)
        
    # Optimization
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.005
    decay_rate = 0.9
    min_lr = 1e-6
    
    # Precompute correlation indices
    full_corr_len = 2 * n_points - 1
    
    def compute_gradient(h_vec):
        # Compute integral of h(x)(1 - h(x + k)) for all k
        h_extended = np.concatenate([h_vec, np.zeros(n_points - 1)])
        ones_extended = np.concatenate([np.ones_like(h_vec), np.zeros(n_points - 1)])
        
        # Calculate correlation of h with (1-h) 
        corr = np.correlate(h_extended, ones_extended - h_vec, mode='full')
        
        # Take maximum over all shifts
        max_idx = np.argmax(corr)
        
        # Compute derivative of correlation w.r.t. h
        grad = np.zeros_like(h_vec)
        
        # For the specific shift that maximizes the integral
        shift = max_idx - (n_points - 1)
        
        # Gradient: for each position i, how much does the integral change
        for i in range(n_points):
            if 0 <= i + shift < n_points:
                grad[i] += (1 - h_vec[i + shift])  # Intuitive gradient direction
            
        return grad

    # Gradient descent with projections
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time > time_limit:
            break
            
        # Compute current gradient
        grad = compute_gradient(h)
        
        # Project onto feasible region: 0 <= h <= 1
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)

            best_c5 = current_c5
            
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h_values, dx):
    """Evaluate the overlap integral using full cross-correlation"""
    h = np.array(h_values)
    n = len(h)
    # Compute full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx
