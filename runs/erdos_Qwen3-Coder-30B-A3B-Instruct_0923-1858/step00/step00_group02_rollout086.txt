import numpy as np
import cvxpy as cp
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    n_points = 45
    dx = 2.0 / n_points
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Convex optimization parameters
    max_iter = 1000
    trust_radius = 0.1
    rho = 0.1
    beta = 0.5
    max_ls_iter = 20
    update_freq = 50
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Landmark solution: C5 = 0.38085
    target_c5 = 0.3807
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        correlations = signal.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlations) * dx
        
        # If we've reached target, stop early
        if c5_current <= target_c5:
            break
            
        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # We'll use gradient descent with line search
        # For simplicity, compute numerical gradient and take small steps
        
        # Gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2e-5)
        
        # Descent direction: negative gradient
        d = -grad
        
        # Project direction to feasible set
        # d[i] is valid if h[i] + d[i] stays in [0,1]
        # So we need: -h[i] <= d[i] <= (1-h[i])
        step_size = trust_radius
        alpha = 1.0
        current_h = h.copy()
        current_c5 = c5_current
        
        # Backtracking line search
        for ls_iter in range(max_ls_iter):
            new_h = h + alpha * d
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            new_c5 = evaluate_erdos_solution(new_h)
            
            if new_c5 <= current_c5 - rho * alpha * np.dot(d, d):
                h = new_h.copy()
                break
            else:
                alpha *= beta
                
        # Adjust trust region
        if alpha < 0.9:  # Step was small
            trust_radius = max(1e-4, trust_radius * 0.5)
        else:
            trust_radius = min(1.0, trust_radius * 2.0)
    
    # Final evaluation and return
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h)
    return (final_h, final_c5, n_points)
