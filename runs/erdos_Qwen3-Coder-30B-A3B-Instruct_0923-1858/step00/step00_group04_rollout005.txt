import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[1::2] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Precompute correlation terms for efficiency
    def compute_C5(h, dx):
        h_extended = np.concatenate([h, h])  # for circular correlation
        correlations = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            corr = np.sum(h * (1 - h_extended[k:k+n_points]))
            correlations[k] = corr
        return np.max(correlations) * dx
    
    # Optimization
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    max_iters = 10000
    iter_count = 0
    last_improvement = 0
    
    # Frank-Wolfe style optimization with line search
    while iter_count < max_iters and time.time() - start_time < budget_s - 1:
        # Compute gradient estimate based on current h
        h_extended = np.concatenate([h, h])
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = 0
            for k in range(n_points):
                idx = (i + k) % n_points
                grad[i] += (1 - h_extended[k] - h[idx]) * 1
        
        # Scaled gradient descent step (Frank-Wolfe)
        step_size = 0.01
        h_candidate = h - step_size * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Project back to feasible set (sum = n_points/2)
        h_mean = np.mean(h_candidate)
        h_candidate = h_candidate - h_mean + (n_points / 2) / n_points
        
        # Evaluate new candidate
        new_C5 = compute_C5(h_candidate, dx)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_candidate.copy()
            last_improvement = iter_count
        
        h = h_candidate.copy()
        
        iter_count += 1
        if iter_count % 100 == 0:
            # Occasionally perturb to avoid local minima
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h - np.mean(h) + (n_points / 2) / n_points
        
        if iter_count - last_improvement > 200:
            break
    
    # Final check
    final_C5 = compute_C5(best_h, dx) 
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
