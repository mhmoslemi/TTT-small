import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup problem
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h - if initial_h_values is available, use it
    try:
        h_init = initial_h_values.copy()
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    except:
        h_init = np.ones(n_points) * 0.5
        # Make it a valid solution
        h_init[10:35] = 1.0
        h_init[:10] = 0.0
        h_init[35:] = 0.0
        h_init = h_init * (n_points / 2.0) / np.sum(h_init)
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = np.inf
    
    # Objective: Maximize the worst correlation
    def objective(h):
        h = np.array(h)
        # Compute correlations
        correlations = np.correlate(h, 1 - h, mode="full")
        # Apply dx
        correlations *= dx
        return -np.max(correlations)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h - 1.0, -h])
    
    # Bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Step function to maximize overlap
    def binary_search_optimize(h_guess, time_budget):
        start = time.time()
        C_min = 0
        C_max = 0.5
        best_h = h_guess.copy()
        best_c = np.inf
        
        for i in range(15):
            if time.time() - start > time_budget * 0.95:
                break
                
            C_candidate = (C_min + C_max) / 2
            
            # Formulate and solve optimization
            h = Variable(n_points)
            
            # Constraints
            constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
            
            # Correlation constraint for each shift
            for k in range(2 * n_points - 1):
                corr = 0.0
                for i in range(n_points):
                    if 0 <= i + k < n_points:
                        corr += h[i] * (1 - h[i + k]) * dx
                constraints.append(corr <= C_candidate)
            
            # Objective: minimize a dummy variable
            obj = Minimize(0)
            
            prob = Problem(obj, constraints)
            
            try:
                prob.solve(solver=CVXOPT, verbose=False)
                
                if prob.status == 'optimal':
                    h_solution = h.value
                    if np.max(np.correlate(h_solution, 1 - h_solution, mode="full") * dx) <= C_candidate + 1e-6:
                        C_max = C_candidate
                        best_h = h_solution.copy()
                        best_c = C_candidate
                    else:
                        C_min = C_candidate
                else:
                    C_min = C_candidate
                    
            except:
                C_min = C_candidate
                
        return best_h, best_c
    
    # Main optimization loop
    elapsed = time.time() - start_time
    while elapsed < budget_s * 0.95:
        # Create a slightly perturbed version
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Optimize with this version
        h_new, c5_new = binary_search_optimize(h_perturbed, budget_s - elapsed)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        # Try to refine with line search
        def f_to_minimize(x):
            h_test = np.exp(x) / (1 + np.exp(x))
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2.0) / np.sum(h_test)
            corr = np.correlate(h_test, 1 - h_test, mode="full") * dx
            return -np.max(corr)
            
        try:
            res = minimize(f_to_minimize, np.log(h / (1 - h + 1e-10)), method='L-BFGS-B', bounds=[(-20, 20)] * n_points, options={'maxiter': 20})
            h_test = np.exp(res.x) / (1 + np.exp(res.x))
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2.0) / np.sum(h_test)
            corr = np.correlate(h_test, 1 - h_test, mode="full") * dx
            if -np.max(corr) < best_c5:
                best_h = h_test
                best_c5 = -np.max(corr)
        except:
            pass
        
        # Shift current h to randomized version
        h = h_new
        elapsed = time.time() - start_time
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(final_corr)
    
    if not (0 <= final_c5 <= 0.5) or np.sum(best_h) < n_points/2 - 1e-6 or np.sum(best_h) > n_points/2 + 1e-6:
        try:
            # Reoptimize once more
            _, best_c5 = binary_search_optimize(initial_h_values if 'initial_h_values' in globals() else np.ones(n_points) * 0.5, budget_s - elapsed)
            best_h = initial_h_values if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        except:
            pass
    
    return (best_h, best_c5, n_points)
