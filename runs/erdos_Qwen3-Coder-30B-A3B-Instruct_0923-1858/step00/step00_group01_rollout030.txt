import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Ensure sum = 1
        h = h * n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Optimization
    max_iter = 1000
    lr = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # compute C5 using correlation
        corr = np.correlate(h, 1-h, mode="full")
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Stop if no improvement for 200 iterations
        if patience > 200:
            break
            
        # Compute gradient (approximate derivative)
        grad = np.zeros_like(h)
        eps = 1e-5
            
        # finite differences for gradient approx
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * n_points / (2 * np.sum(h_plus))  # rescale
            
            c5_plus = np.max(np.correlate(h_plus, 1-h_plus, mode="full")) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # gradient descent step
        h -= lr * grad
        
        # project to feasible set
        h = np.clip(h, 0, 1)
        h = h * n_points / (2 * np.sum(h))
        
        # reduce learning rate
        if it % 100 == 0 and it > 0:
            lr *= 0.1
            
        # increase resolution if needed
        if it == 100:
            n_points = 20
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(best_h)), 
                          best_h)
            h = h * n_points / (2 * np.sum(h))
            
        # increase resolution at later iterations
        if it == 400:
            n_points = 64
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(best_h)), 
                          best_h)
            h = h * n_points / (2 * np.sum(h))
            
    # Final evaluation
    corr = np.correlate(best_h, 1-best_h, mode="full")
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, len(best_h))
