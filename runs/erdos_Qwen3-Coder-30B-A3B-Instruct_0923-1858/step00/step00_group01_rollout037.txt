import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    else:
        # Create a simple initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.45
        h[-1] = 0.55
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_C5(h_vals):
        # Compute the cross-correlation, mode='full' gives all shifts
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vals):
        # Ensure h is in [0,1] and sum(h)*dx = 1
        h_vals = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_vals)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-6:
            scale = target_sum / current_sum
            h_vals *= scale
        return h_vals
    
    def perturb_h(h_vals, max_transfer=0.05):
        # Clone the array to make changes
        new_h = h_vals.copy()
        # Get all indices with potential overlap
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        shift_idx = np.argmax(corr)
        # Get the actual shift
        shift = shift_idx - (len(h_vals) - 1)
        
        # Create indices to consider for transfer
        indices = np.arange(len(h_vals))
        # Focus on regions that contribute to the correlation at this shift
        start_idx = max(0, shift)
        end_idx = min(len(h_vals), len(h_vals) + shift)
        
        # Simple random transfer
        i = np.random.randint(len(h_vals))
        j = np.random.randint(len(h_vals))
        # Ensure we're transferring from i to j
        transfer = min(h_vals[i], 1 - h_vals[j])
        if transfer > 0:
            amount = np.random.uniform(0, min(max_transfer, transfer))
            new_h[i] -= amount
            new_h[j] += amount
        return project_h(new_h)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    last_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < 1000:
        iter_count += 1
        new_h = perturb_h(best_h)
        new_C5 = compute_C5(new_h)
        if new_C5 < best_C5:
            best_h = new_h
            best_C5 = new_C5
            last_improvement = 0
        else:
            last_improvement += 1
            # Occasionally try a different strategy
            if last_improvement > 100 and np.random.rand() < 0.1:
                # Try a more aggressive reset
                new_h = h.copy()
                for _ in range(10):
                    new_h = perturb_h(new_h, 0.1)
                new_C5 = compute_C5(new_h)
                if new_C5 < best_C5:
                    best_h = new_h
                    best_C5 = new_C5
                    last_improvement = 0
        
        # Occasionally interpolate to a higher resolution
        if iter_count % 50 == 0:
            n_points = len(best_h)
            new_n = 2 * n_points

            x_new = np.linspace(0, 2, new_n)
            x_old = np.linspace(0, 2, n_points)
            # Linear interpolation
            new_h = np.interp(x_new, x_old, best_h)
            # Project to maintain constraints
            new_h = project_h(new_h)
            # Check if this improved the value
            new_C5 = compute_C5(new_h) * dx_new
            if new_C5 < best_C5:
                best_h = new_h
                best_C5 = new_C5
                last_improvement = 0
            else:
                # Revert to previous resolution if bad
                best_h = project_h(best_h)
                best_C5 = compute_C5(best_h)
    
    # Final projection to enforce constraint
    best_h = project_h(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
