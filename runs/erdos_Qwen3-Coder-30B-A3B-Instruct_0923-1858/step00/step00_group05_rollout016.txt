import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral C5
    corr = np.correlate(h, 1 - h, mode="full")
    # Left-align the correlation result to cover all lags from 0 to 2
    # The full correlation is of length 2*n_points - 1
    # We want the maximum value over all lags from 0 to 2 (i.e., 0 to n_points-1 in correlation indexing)
    # The first element corresponds to lag 0, and last element to lag -(n_points-1)
    # So we take the middle portion needed
    mid = len(corr) // 2
    # We need the maximum from lags 0 to 2, which is indices from 0 to n_points where the lag is 0 to n_points - 1
    # Indices from mid to mid + n_points - 1 gives the appropriate range
    c5 = np.max(corr[mid:mid + len(h)]) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    def init_individual():
        h = np.random.rand(n_points)
        # Ensure sum is exactly n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Wrapper for evaluation
    def fitness(individual):
        h = np.array(individual)
        c5 = evaluate_c5(h, dx)
        return c5
    
    def constraint_sums(individual):
        h = np.array(individual)
        return np.sum(h) - n_points / 2
    
    # Constraints for optimization
    cons = {'type': 'eq', 'fun': constraint_sums}
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try using initial construction if available  
    if 'initial_h_values' in globals():
        best_individual = np.array(initial_h_values)
        best_c5 = evaluate_c5(best_individual, dx)
    else:
        # Start with a uniform distribution
        best_individual = np.ones(n_points) * 0.5
        # adjust to sum to n_points/2
        best_individual = best_individual * (n_points / 2) / np.sum(best_individual)
        best_c5 = evaluate_c5(best_individual, dx)
    
    # Use larger generation size for evolution and iteratively improve the best
    max_gens = 50
    pop_size = 100
    
    # Genetic algorithm to explore the space
    for gen in range(max_gens):
        if time.time() - start_time > budget_s - 5:
            break
            
        population = [init_individual() for _ in range(pop_size)]
        fitness_scores = [fitness(ind) for ind in population]
        
        # Sort by fitness
        idxs = np.argsort(fitness_scores)
        population = [population[i] for i in idxs]
        fitness_scores = [fitness_scores[i] for i in idxs]
        
        # Save best from this generation
        current_best = population[0]
        current_c5 = fitness_scores[0]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_individual = current_best.copy()
        
        # Local optimization on top individuals
        for _ in range(5):  # 5 local optimizations
            if time.time() - start_time > budget_s - 5:
                break
            # Take top 10
            top_indices = np.random.choice(range(10), size=3, replace=False)
            for i in top_indices:
                ind = population[i].copy()
                
                # Wrapper for scipy function
                def fun(h_vec):
                    return evaluate_c5(h_vec, dx)
                
                # Attempt local optimization
                try:
                    result = minimize(fun, ind, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
                    if result.success:
                        new_h = result.x
                        new_c5 = evaluate_c5(new_h, dx)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_individual = new_h.copy()
                except:
                    pass
    
    # Final evaluation of best
    final_c5 = evaluate_c5(best_individual, dx)
    
    return (list(best_individual), final_c5, n_points)
