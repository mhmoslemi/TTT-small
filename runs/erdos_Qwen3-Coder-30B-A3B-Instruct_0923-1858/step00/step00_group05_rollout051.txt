def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant for given step function h."""
    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integral values for all k
    overlaps = corr[len(corr)//2:] * dx
    return np.max(overlaps)
