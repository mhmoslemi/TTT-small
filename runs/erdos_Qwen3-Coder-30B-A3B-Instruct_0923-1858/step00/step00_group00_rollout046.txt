import time

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all shifts (the full correlation gives us 2*n-1 values)
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h):
    """Projects h to satisfy the constraints: 0 <= h[i] <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    n = len(h)
    target_sum = n / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def genetic_algorithm_step(population, fitnesses, elite_size=10, mutation_rate=0.05):
    """Single generation of genetic algorithm."""
    n = len(population[0])
    new_population = []
    
    # Elites (best solutions)
    elite_indices = np.argsort(fitnesses)[:elite_size]
    for i in elite_indices:
        new_population.append(population[i].copy())
        
    # Generate offspring
    while len(new_population) < len(population):
        # Tournament selection
        parent1 = population[np.random.choice(elite_indices)]
        parent2 = population[np.random.choice(elite_indices)]
        
        # Crossover (blend)
        alpha = 0.5
        child = alpha * parent1 + (1 - alpha) * parent2
        
        # Mutation
        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, 0.1, n)
            child += noise
        
        # Project to feasible region
        child = project_to_feasible(child)
        new_population.append(child)
        
    return new_population

    dx = 2.0 / n

    h_plus = h.copy()
    h_minus = h.copy()
    grad = np.zeros_like(h)

        h_plus[i] += eps
        h_minus[i] -= eps

        c5_minus = evaluate_c5(h_minus)

        # Reset for next dimension
        h_plus[i] -= eps
        h_minus[i] += eps
    
    # Update
    h_new = h - lr * grad
    
    # Project to feasible region
    h_new = project_to_feasible(h_new)
    
    return h_new
