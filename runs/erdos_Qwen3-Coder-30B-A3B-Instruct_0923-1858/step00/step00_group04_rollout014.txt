```python
import numpy as np
from scipy import signal
import time
import random

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Maximum value divided by n (to account for the dx factor)
    c5 = np.max(corr) * dx
    return c5

def is_feasible(h):
    """Check if h satisfies the constraints"""
    if not all(0 <= x <= 1 for x in h):
        return False
    n = len(h)
    return abs(sum(h) - n / 2) < 1e-10

def rescale_to_sum(h, target_sum):
    """Rescale h to have the target sum while maintaining bounds"""
    current_sum = sum(h)
    if abs(current_sum) < 1e-10:
        # If current sum is essentially 0, we create a balanced vector
        h = [0.5] * len(h)
        h[0] = 0.5 * target_sum
        h[-1] = 0.5 * target_sum
    else:
        h = np.array(h) * (target_sum / current_sum)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale again to match target sum
    current_sum = sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def create_initial_population(n_points, pop_size, initial_h_values=None):
    """Create initial population"""
    population = []
    
    if initial_h_values is not None and len(initial_h_values) == n_points:
        population.append(np.array(initial_h_values))
    else:
        # Simple initial solution
        population.append(np.ones(n_points) * 0.5)
    
    # Fill the rest randomly
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = rescale_to_sum(h, n_points / 2)
        population.append(h)
    
    return population

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    selected_idx = random.randint(0, len(population) - 1)
    for _ in range(k - 1):
        idx = random.randint(0, len(population) - 1)
        if fitness[idx] < fitness[selected_idx]:
            selected_idx = idx
    return population[selected_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover operation"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, mutation_rate=0.1, epsilon=0.1):
    """Apply mutation by perturbing randomly selected indices"""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if random.random() < mutation_rate:
            # Perturb with small amount
            delta = random.uniform(-epsilon, epsilon)
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
    
    # Rescale to maintain sum constraint
    h = rescale_to_sum(h, n / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 42  # Use the known size for robustness
    pop_size = 30
    generations = 1000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_size = int(0.1 * pop_size)
    
    start_time = time.time()
    
    # Create initial population
    population = create_initial_population(n_points, pop_size, initial_h_values)
    
    # Track best solution
    global_best = None
    global_best_c5 = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_c5(h)
        fitness.append(c5)
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best = h.copy()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for early stopping
        if gen > 200 and fitness[0] == fitness[-1]:
            break
            
        # Keep elite
        new_population = population[:elite_size]
        new_fitness = fitness[:elite_size]
        
        # Generate offspring
        for _ in range(pop_size - elite_size):
            if random.random() < crossover_prob:
                # Select two parents
                parent1 = tournament_selection(population, fitness)
                parent2 = tournament_selection(population, fitness)
                # Crossover
                child1, child2 = blend_crossover(parent1, parent2)
                # Mutation
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                # Add to population
                new_population.extend([child1, child2])
                new_fitness.extend([evaluate_c5(child1), evaluate_c5(child2)])
            else:
                # Random mutate one individual
                parent = tournament_selection(population, fitness)
                child = mutate(parent, mutation_rate)
                new_population.append(child)
                new_fitness.append(evaluate_c5(child))
        
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Update best
        for h, c5 in zip(population, fitness):
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best = h.copy()
    
    return (global_best, global_best_c5, n_points)
```
