import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[1::2] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    
    # Frank-Wolfe with DC programming
    eta = 0.05
    eps = 0.02
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute the gradient direction using a simple finite difference estimate
        # We want to minimize C5 = max_k ∫ h(x)(1-h(x+k)) dx
        # This is same as maximize sum_k h_i * (1 - h_{i+k})
        # => maximize sum_i h_i * (n_points - sum_k h_{i+k})
        # We linearize around current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Gradient step estimate (simplified)
        # Let's compute the gradient directly from the cross-correlation structure
        cross_corr = np.correlate(h, 1 - h, mode='full')
        # Remove padding at ends
        cross_corr = cross_corr[n_points - 1:2*n_points - 1]
        grad_h = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if k < n_points and (i+k) % n_points < n_points:
                    grad_h[i] += (1 - h[(i+k) % n_points]) * dx
        
        # Now use Frank-Wolfe linear subproblem
        # We want to minimize ⟨ ∇f(h), d ⟩ subject to bounds
        # But also ∫ d(x) dx = 0 (so that h remains feasible for sum)
        d = -grad_h
        # Project to [0,1]
        d = np.clip(d, 0, 1)
        d -= np.mean(d)  # Adjust to keep sum unchanged
        # Scale appropriately
        d = d * eta / (np.max(np.abs(d)) + 1e-8)
        # Advance
        h_new = h + d
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Rescale to ensure sum = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if update improved objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally perturb to escape local minima
        if iter_count % 20 == 0:
            # Add small random noise
            noise = np.random.normal(0, eps, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            # Rescale again
            h = h * (n_points / 2) / np.sum(h)
            
        if no_improvement_count > max_no_improvement:
            max_no_improvement += 20  # increase patience
            # Random restart
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
    
    return (list(best_h), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Compute the full correlation and return max
    correlation = np.correlate(h, 1-h, mode="full") * dx
    # Take the middle portion, which corresponds to actual overlaps
    mid = len(correlation) // 2
    # Valid (non-zero) correlations are from mid - n+1 to mid + n-1
    valid_corr = correlation[mid - len(h) + 1 : mid + len(h) - 1]
    return np.max(valid_corr)
