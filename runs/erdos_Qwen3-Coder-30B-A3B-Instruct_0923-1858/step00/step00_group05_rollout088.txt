import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    # This mimics the behavior described in the problem:
    # Compute the overlap integral for all lag values
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute auto-correlation of h with (1-h), mode = full
    corr = correlate(h, 1 - h, mode='full')
    # The result should be evaluated to be maximized over all k
    # The convolution results correspond to lag indices from -n+1 to n-1
    # So the actual values are for lags from 0 to 2n-2
    # The maximum value scaled by dx is our estimate
    max_val = np.max(corr) * dx
    return max_val

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral of h should be 1
    
    # use initial construction if available

    # normalize to target_sum
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # optimization loop
    iter_count = 0
    tolerance = 1e-5
    max_iter = 10000  # (600s/300ms)*20  with some margin
    step_size = 0.1
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        if iter_count > max_iter:
            break
            
        # compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        max_overlaps = np.max(corr)
        max_idx = np.argmax(corr)
        
        # Current C5 is max_overlaps * dx
        c5 = max_overlaps * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # check if converged
        if best_c5 < 0.3807:
            break
            
        # identify indices with significant overlap (close to max)
        threshold = 0.99 * max_overlaps
        idxs = np.where(corr >= threshold)[0]
        if len(idxs) == 0:
            idxs = [max_idx]
            
        # try to reduce these by solving a QP-like problem in CVXPY
        # try to adjust h values to reduce overlap at these indices
        # manuall adjustments: increase regions that should reduce overlap:
        # but be gentler to avoid numerical issues
        
        for _ in range(5):  # few iterations per step for stability
            #改善间隔优化 - 每个点的梯度估计
            # 逐个更新来避免速度快的不收敛问题
            
            # compute gradient estimate using finite differences
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                eps = 1e-4
                # add small perturbation
                h_plus = h.copy()
                h_minus = h.copy()
                
                if h[i] + eps > 1:
                    h_plus[i] = 1.0
                else:
                    h_plus[i] += eps
                    
                if h[i] - eps < 0:
                    h_minus[i] = 0.0
                else:
                    h_minus[i] -= eps
                    
                # check overlap with perturbed h
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[i] = 0.5 * (np.max(corr_plus) - np.max(corr_minus)) / eps
                grad[i] = max(-1, min(1, grad[i]))
                
            # gradient descent step
            h_new = h - step_size * grad
            
            # project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # renormalize to target sum
            h_new = h_new * target_sum / np.sum(h_new)
            
            # evaluate new value
            test_c5 = evaluate_erdos_solution(h_new, n_points)
            if test_c5 < c5:
                h = h_new.copy()
                c5 = test_c5
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                if c5 < 0.3807:
                    break
                    
    # final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
