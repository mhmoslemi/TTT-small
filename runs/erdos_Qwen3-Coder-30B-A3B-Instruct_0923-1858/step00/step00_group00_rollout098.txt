import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations using numpy's correlate with full mode
    corr = np.correlate(h, 1-h, mode='full')  
    # Only take the relevant part where shifts are in [-2, 2]
    # Full correlation is 2*n-1 long, center is at n-1
    # We want maximum of correlations from shifts -2 to 2
    max_corr = np.max(corr[n-2:n+2])  # indices from n-2 to n+1 inclusive -> 4 shifts
    return max_corr * dx

def project_to_feasible(h):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n/2"""
    n = len(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to have correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

def genetic_algorithm_step(h_values, fitnesses, pop_size, n_points, mu_rate=0.1):
    """Perform one step of genetic algorithm"""
    # Tournament selection
    selected = []
    for _ in range(pop_size):
        tournament_indices = np.random.choice(len(h_values), size=5, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(h_values[winner_idx])
    
    # Elitism: keep best 10%
    elite_count = pop_size // 10
    elite_indices = np.argsort(fitnesses)[:elite_count]
    elites = [h_values[i] for i in elite_indices]
    
    # Create offspring
    next_gen = elites[:]
    
    while len(next_gen) < pop_size:
        parent1 = random.choice(selected)
        parent2 = random.choice(selected)
        # Blend crossover
        alpha = 0.5
        child = alpha * parent1 + (1 - alpha) * parent2
        # Add mutation
        if random.random() < mu_rate:
            mutation_strength = 0.1 / (len(next_gen) + 1)  # Decrease with generation
            noise = np.random.normal(0, mutation_strength, len(child))
            child += noise
            
        child = project_to_feasible(child)
        next_gen.append(child)
    
    return next_gen
