import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
        
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    # Optimization parameters
    max_iter = 50000
    learning_rate = 0.01
    perturbation_rate = 0.1
    perturbation_freq = 100
    
    # Best solution tracking
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        correlations = np.correlate(h, 1 - h, mode="full")
        C5_candidate = np.max(np.abs(correlations)) * dx
        
        # Update best solution if improved
        if C5_candidate < best_C5:
            best_h = h.copy()
            best_C5 = C5_candidate
            
        # Compute gradient using finite differences for stability
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            C5_plus = np.max(np.abs(corr_plus)) * dx
            
            grad[i] = (C5_plus - C5_candidate) / eps
            
        # Gradient descent step
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Perturbation for escape from local minima
        if iteration % perturbation_freq == 0:
            pert = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            
        # Adaptive learning rate
        if iteration % 1000 == 0 and iteration > 0:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
    
    # Final validation and recomputation of C5
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, n_points)
