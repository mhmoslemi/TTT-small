import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from parent
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Default construction: a simple piecewise constant function
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
        # Normalize
        h_init = h_init * n_points / (2 * np.sum(h_init))
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Start with current best
    h_current = h_init.copy()
    c5_bound = max(np.correlate(h_current, 1 - h_current, mode="full")) * dx
    
    start_time = time()
    best_h = h_current.copy()
    best_c5 = c5_bound
    
    # Constraint collection
    max_shifts = n_points - 1
    
    # Parameters for cutting plane method
    max_iter = 1000
    top_p = 0.2  # Top 20% of constraints to add at each iteration
    min_improvement = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for full range
        overlaps = np.correlate(h_current, 1 - h_current, mode="full")
        if len(overlaps) != 2 * n_points - 1:
            overlaps = np.correlate(h_current, 1 - h_current, mode="full")
        
        # Sort by overlap (descending)
        sorted_indices = np.argsort(overlaps)[::-1]
        
        # Select top k constraints
        k = max(1, int(len(sorted_indices) * top_p))
        top_constraints = sorted_indices[:k]
        
        # Formulate optimization problem
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C5)
        
        constraints = [
            h[i] >= 0 for i in range(n_points)
        ] + [
            h[i] <= 1 for i in range(n_points)
        ] + [
            cp.sum(h) == n_points / 2.0
        ]
        
        # Add overlap constraints
        for idx in top_constraints:
            # Compute overlap for shift idx with offset
            shift_offset = idx - (n_points - 1)
            # Calculate indices that will be involved in convolution
            # Left edge
            left_start = max(0, -shift_offset)
            right_end = min(n_points, n_points - shift_offset)
            if shift_offset < 0:
                # h[i] * (1 - h[i-shift_offset])
                # This must be:
                # sum_{i} h[i] * (1 - h[i-shift_offset]) for i in [left_start, right_end)
                left_h = h[left_start:right_end]
                right_h = h[left_start-shift_offset:right_end-shift_offset]
                left_h_vals = h_current[left_start:right_end]
                right_h_vals = 1 - h_current[left_start-shift_offset:right_end-shift_offset]
                nan_or_inf = np.any(np.isnan(left_h_vals)) or np.any(np.isinf(left_h_vals))
                if nan_or_inf:
                    continue
                conv = cp.sum(cp.multiply(left_h, right_h))
                constraints.append(conv <= C5)
            else:
                # h[i-shift_offset] * (1 - h[i])
                left_start = max(0, shift_offset)
                right_end = min(n_points, n_points - shift_offset)
                left_h = h[left_start-shift_offset:right_end-shift_offset]
                right_h = h[left_start:right_end]
                
                left_h_vals = h_current[left_start-shift_offset:right_end-shift_offset]
                right_h_vals = 1 - h_current[left_start:right_end]
                nan_or_inf = np.any(np.isnan(left_h_vals)) or np.any(np.isinf(left_h_vals))
                if nan_or_inf:
                    continue
                conv = cp.sum(cp.multiply(left_h, right_h))
                constraints.append(conv <= C5)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6, reltol=1e-6)
        except Exception as e:
            continue
            
        if prob.status != cp.OPTIMAL:
            continue
            
        h_opt = h.value
        c5_opt = C5.value
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            
        # Compute full overlaps to identify violations
        full_overlaps = np.correlate(h_opt, 1 - h_opt, mode="full")
        max_over = max(full_overlaps) * dx
        
        if abs(max_over - c5_opt) < min_improvement:
            break
            
        if max_over > c5_opt:
            # Add new constraints
            violation_indices = np.where(full_overlaps * dx > c5_opt + min_improvement)[0]
            if len(violation_indices) == 0:
                # No violation detected, terminate
                break
            # Add more constraints
            h_current = h_opt
            continue
        else:
            h_current = h_opt
            c5_bound = c5_opt
    
    # Final validation
    h_out = best_h.copy()
    h_out = np.clip(h_out, 0, 1)
    h_out = h_out * n_points / (np.sum(h_out) * dx)
    final_c5 = max(np.correlate(h_out, 1 - h_out, mode="full")) * dx

    return (list(h_out), final_c5, n_points)
