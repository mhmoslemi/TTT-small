import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a triangular wave approximation
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    def project(h):
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to satisfy sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def gradient(h, epsilon=1e-4):
        # Compute numerical gradient using finite differences
        grad = np.zeros_like(h)
        base_val = evaluate(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project(h_plus)
            forward_val = evaluate(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project(h_minus)
            backward_val = evaluate(h_minus)
            
            grad[i] = (forward_val - backward_val) / (2 * epsilon)
        return grad
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    start_time = time.time()
    iterations = 0
    trust_radius = 0.1
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Evaluate current point
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()  

        # Compute gradient
        grad = gradient(h)
        
        # Trust region step
        step = -trust_radius * grad / (np.linalg.norm(grad) + 1e-10)
        new_h = h + step
        
        new_h = project(new_h)
        new_c5 = evaluate(new_h)
        
        if new_c5 < current_c5:
            # Accept step
            h = new_h
            trust_radius = min(2 * trust_radius, 0.5)  # Increase trust radius
        else:
            # Reject step
            trust_radius /= 2  # Decrease trust radius
            
        # Occasionally perturb to escape local minima
        if iterations % 100 == 0 and trust_radius < 0.01:
            h = project(h + 0.05 * np.random.normal(0, 1, n_points))
            trust_radius = 0.1  # Reset trust radius
            
        iterations += 1
    
    # final validation
    final_h = project(best_h)
    final_c5 = evaluate(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
