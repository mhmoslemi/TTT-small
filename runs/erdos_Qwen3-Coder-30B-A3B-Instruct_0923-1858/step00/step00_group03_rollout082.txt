import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        # Create a reasonable starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    
    # Project to ensure sum constraint
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Adjust to achieve exact sum
            adjustment = (n_points / 2 - current_sum) / n_points
            h_vec += adjustment
            h_vec = np.clip(h_vec, 0, 1)
        return h_vec
    
    # Gradient descent
    max_iter = 20000
    alpha = 0.1
    last_improvement = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlaps = []
        for k in range(2 * n_points - 1):
            if k < n_points:
                shift = k
                overlap = np.sum(h[:-shift] * (1 - h[shift:])) * dx if shift < n_points else 0
            else:
                shift = 2 * n_points - 2 - k
                overlap = np.sum(h[shift:] * (1 - h[:-shift])) * dx if shift < n_points else 0
            overlaps.append(overlap)
        
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Compute gradient
        gradient = np.zeros_like(h)
        if max_overlap_idx < n_points:
            # Find corresponding shifts for max overlap
            k = max_overlap_idx
            # Gradient w.r.t. h[i] for 0 <= i < n_points-k
            gradient[:-k] += (1 - h[k:])
            # Gradient w.r.t. h[i+k] for 0 <= i < n_points-k
            gradient[k:] -= h[:-k]
        else:
            k = 2 * n_points - 2 - max_overlap_idx
            # Gradient w.r.t. h[i] for 0 <= i < n_points-k
            gradient[k:] += (1 - h[:-k])
            # Gradient w.r.t. h[i+k] for 0 <= i < n_points-k
            gradient[:-k] -= h[k:]
        
        # Update step
        h -= alpha * gradient
        h = project_h(h)
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h, n_points, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Adaptive step size
        if last_improvement > 100:
            alpha *= 0.9
            last_improvement = 0
            
        if alpha < 1e-10:
            break
    
    # Final rounding to binary values while preserving sum
    def round_binary(h_vec, n_points):
        # Sort values in descending order to identify where to set to 1
        sorted_indices = np.argsort(-h_vec)
        target_sum = n_points / 2
        rounded = np.zeros_like(h_vec)
        current_sum = 0
        
        for idx in sorted_indices:
            if current_sum + h_vec[idx] <= target_sum:
                rounded[idx] = 1
                current_sum += 1
            else:
                break
        # Assign remaining to make sum exactly equal
        remaining = int(target_sum - current_sum)
        for idx in sorted_indices:
            if remaining <= 0: break
            if rounded[idx] == 0:
                rounded[idx] = 1
                remaining -= 1
        return rounded
    
    final_h = round_binary(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, n_points, dx)
    
    return (final_h.tolist(), final_c5, n_points)
