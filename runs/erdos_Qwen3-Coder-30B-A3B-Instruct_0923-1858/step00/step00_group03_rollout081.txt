import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: step function with 5 steps
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    alpha = 0.1
    iteration = 0
    
    while time.time() - start_time < budget_s:
        if iteration % 100 == 0 and iteration > 0:
            alpha *= 0.9
        
        # Compute all overlaps
        full_corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(full_corr)
        max_overlap = full_corr[max_k]
        
        # Gradient based on maximum overlap
        gradient = np.zeros(n_points)
        k_shift = max_k - (n_points - 1)
        if k_shift >= 0:
            for i in range(n_points - k_shift):
                gradient[i] += (1 - h[i + k_shift])
            for i in range(k_shift, n_points):
                gradient[i] -= h[i - k_shift]
        else:
            k_shift = -k_shift
            for i in range(n_points - k_shift):
                gradient[i] += (1 - h[i + k_shift])
            for i in range(k_shift, n_points):
                gradient[i] -= h[i - k_shift]
        
        # Update h
        h -= alpha * gradient
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h *= target_sum / current_sum
        
        # Evaluate new solution
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        iteration += 1
    
    # Final rounding for binary values, maintaining sum
    # This improves solution quality slightly
    mean_val = np.mean(best_h)
    binary_h = np.zeros_like(best_h)
    binary_h[best_h >= mean_val] = 1.0
    current_sum = np.sum(binary_h)
    diff = target_sum - current_sum
    
    # Adjust if needed
    if diff != 0:
        # Flip minimums to 1s to make sum correct
        if diff > 0:
            indices = np.argsort(best_h)[:int(diff)]
            binary_h[indices] = 1.0
        else:
            indices = np.argsort(best_h)[-int(-diff):]
            binary_h[indices] = 0.0
    
    final_c5 = compute_c5(binary_h, dx)
    
    return (binary_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    # Use scipy.signal.correlate for accurate computation
    corr = correlate(h, 1 - h, mode='full')
    # Take only the overlapping parts
    c5_val = np.max(corr) * dx
    return c5_val
