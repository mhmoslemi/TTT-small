import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute full correlation with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Take only the relevant part (shifts from -1 to 1, i.e. len(h)-1 to 2*len(h)-2)
    # But we need the positive shifts (from 0 to len(h)-1), so take from len(h)-1 to 2*len(h)-2
    # However, for integral, we consider shifts k=0 to 2, so we look at the middle of correlation
    # vector we care about the part at indices [len(h)-1:len(h)*2-1]
    # the relevant values for k = 0 to 2 (2*len(h)-1 includes max k = 2len(h)-2)
    # Actually just write it clearly: for shift k, indices in correlation go from -n+1 to n-1
    # So shift k = 0 corresponds to index n-1 in full correlation
    # shift k = 1 to n to 2*n-2, etc
    # We care about shifts 0 to n-1 for the valid range, but in [0,2],
    # and h(x) is discrete on 0, 2 with n points, so dx = 2/n
    # But we want correlation for shifts of size dx * i = 2/n * i in [0, 2]
    # So shift in samples (dx = 2/n) is k = i * n / 2 (i = shift in [0,2])
    # Actually, let's compute shifts of length dx in [0,2], i.e., all integer shifts that
    # will result in valid overlapping areas in our discretized domain
    # We start at shift k = 0 (i=0), and up to i=2/dx, which is 2/(2/n) = n
    # Since we have n samples covering [0,2], a shift of 2/n in the continuous case
    # maps to one sample in discrete space
    # So we look at shifts k=0, 1, ..., n-1 in discrete space
    # The corresponding shifts in continuous are: i*dx = k*2/n (from k=0 to k=n-1)
    max_corr = np.max(corr[len(h)-1:])  # Take the middle part where shifts are in [0,2]
    return max_corr * dx

def update_h(h, learning_rate, dx):
    """Gradient step for updating h."""
    # Compute the full convolution of h with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Now compute gradients of the objective w.r. to h
    # Let C5 = max_k integral h(x) (1 - h(x+k)) dx
    # This is `max_k sum_i h[i] * (1 - h[i+k]) * dx`
    # We compute partial derivative of C5 = max_k of (sum_i h[i] * (1 - h[i+k]))
    # At each shift k, the equation is: 
    # d/dh[j] [ h[i] * (1 - h[i+k]) ] for all valid i such that i+k = j
    # In the convolution, for each h[j], the influences come from h[j-k] (with j-k >=0)
    # For k being fixed, for each h[j], only h[j-k] affects it
    # So the gradient at h[j] from that fixed k is (1-h[j+k])
    # For -1 < k < 1
    # Though we derived it long-hand above we'll just use:
    # Gradient approx using correlation:
    # try:
    # We compute dC5 / dh_j for each j. Since our integral is max of correlations,
    # we only need gradient of the maximum one.
    # Another way: take the argmax k of correlation(h,1-h)
    max_corr_idx = np.argmax(corr)
    grad = np.zeros_like(h)
    # Fix the gradient w.r. to the effective h[idx] term that causes max_corr
    # If the max correlation is at position max_corr_idx, and corr = sum(h[i] * (1-h[i+k]))
    # where k = max_corr_idx - (len(h)-1)
    corr_k = max_corr_idx - (len(h) - 1)
    # Reconstruct the updated gradient if we move by the corr_k-shifted terms
    # This is complicated since the gradient of C5 = max_kcorr_c real
    # At the active shift (k is corr_k), recompute the gradient as much as possible
    # These terms are 1-h[i+k] for i s.t. i+k is valid
    # Actually, simpler approach: compute the gradient of one of the cross-correlations
    # used in max operation, and apply it.
    # Instead, we compute gradient using "lower bound"/real gradient idea:
    # d (h_i (1 - h_{i+k})) / d h_i = (1 - h_{i+k})
    # So gradient of max over shift of sum_i h_i *(1 - h_{i+k}) is:
    # ... use more explicit way: 
    # Position max_corr_idx gives the shift k = max_corr_idx - offset
    # Correlation at all positions contrbutes: 
    # But want gradient of C5 w.r to h <-- the max over k is non smooth
    # Use this: at max, use the gradient of that specific correlation...
    # Gradient update: - learning_rate * gradient of that max-combination
    # Simpler idea:
    # Recompute the effective shift k and then compute the part of gradient:
    # Let's do compute it properly, and then leave the optimization to SGD flavor
    # for i in range(len(h)):
    # grad[i] = take term 1-h[i+k] with k = max_corr_idx - (len(h)-1) if i+k is valid
    # That is bit convoluted, so we do instead:
    # Compute full correlation matrix, find which one is max, then compute that specific "
    # Inspired by: consider k = 0 to n-1, how each h_i contributes: 
    # For fixed k, there's an effect on each h[i]
    # dB_k / dh_i = (1 - h[i+k]) if i+k < n
    # But note that the shift is inclusive at both ends, the correlation definition
    # takes h[i] * (1-h[i+k]) for all i
    # So that gives us:
    # If max shift index is max_corr_idx:
    # effective shift k = max_corr_idx - (n-1)
    # What h[i] affects is when h[i] * (1-h[i+k]), then we look for influence:
    # when j = i+k (and k here is a shifted variable)  
    # dB_k / dh[j] = (1 - h[j-k])
    # which leads to: 
    # score is -dr_dh = - (1 - h[j-k])
    # Another dynamic way is to just make the guessing more accurate:
    # Better: 
    # The gradient contributions are to h[i]: from all (i+k) terms.
    # The cooling will be due to (1-h[j])
    # But directly: we simulate the gradients of those terms for the specific k shift
    # Apply a different approach:
    # For each h[i], we know that it is used in computing 1-h[i+k] for some k, 
    # that contribute to the (h(i) *(1-h(i+k))
    # Call this score:
    # The key insight: in correlation corr_idx = k + len(h) - 1, so k = corr_idx - (len(h)-1)
    # e.g., if len(h) = 5, then 1-diff corr indices: 0 to 8, middle corr_index = 4 at k=0
    # Output shape of full correlation is 2n-1
    k = max_corr_idx - (len(h) - 1)
    if k < 0:
        # Indices from 0 to len(h)+k-1
        for i in range(len(h) + k):
            if i - k >= 0:
                grad[i] += -1.0 * (1.0 - h[i - k])
    else:
        # k >= 0
        for i in range(len(h) - k):
            grad[i] += -1.0 * (1.0 - h[i + k])
    
    # Normalize gradient or not?
    # Here we only store it needed for the update step
    # Apply gradient step
    h_new = h - learning_rate * grad
    # Project to feasible region
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Rescale to sum = n/2
    h_new = h_new * (len(h) / 2.0) / np.sum(h_new)
    return h_new

    np.random.seed(seed)
    start_time = time.time()
    # Initial vector - use provided or construct triangular
    n_points = 45
    if 'initial_h_values' in globals():

        h = h[:n_points]  # ensure n_points
    else:
        h = np.linspace(0, 1, n_points)
        h = (h + 1.0-h[::-1]) * 0.5  # triangular wave
        # Ensure sum = 1
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h, best_c5 = h.copy(), evaluate_h(h, dx)
    learning_rate = 0.01
    # Counting steps for learning rate decay
    steps = 0
    perturb_freq = 50
    perturb_mag = 0.05
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:  # conservative cutoff
            break
        steps += 1
        # perform gradient update
        h_new = update_h(h, learning_rate, dx)
        # evaluate
        c5 = evaluate_h(h_new, dx)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
        h = h_new
        # adaptive learning rate
        if steps % 100 == 0:
            learning_rate *= 0.95
        # occasional random perturbation
        if steps % perturb_freq == 0:
            h_pert = h + np.random.normal(0, perturb_mag, len(h))
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2.0) / np.sum(h_pert)
            c5_pert = evaluate_h(h_pert, dx)
            if c5_pert < best_c5:
                best_h, best_c5 = h_pert.copy(), c5_pert
            h = h_pert
    
    return (best_h.tolist(), best_c5, n_points)
