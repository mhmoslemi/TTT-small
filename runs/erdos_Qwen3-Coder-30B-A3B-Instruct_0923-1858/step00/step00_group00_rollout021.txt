import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, dx):
    n = len(h)
    # Compute the full convolution
    conv = np.correlate(h, 1 - h, mode='full')
    # Take only the relevant part (shifts from -n+1 to n-1)
    # But we want maximum over all valid shifts k=1 to n-1 (in our case, 0 to n-2)
    # Actually, the shifts are from 0 to n-1 in one direction, i.e., k = 0..n-1 for our full correlation
    # We want max of int h(x)(1-h(x+k)) over all integer shifts k=0 to n-1
    return np.max(conv[n-1:2*n-1]) * dx

def project_to_feasible(h, dx, target_sum):
    # Project h to sum to target_sum while keeping in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if np.abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h = h * scale
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h
