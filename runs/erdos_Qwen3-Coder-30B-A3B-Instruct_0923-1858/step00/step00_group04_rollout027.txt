        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Perturbation parameters
    epsilon = 0.05
    perturb_freq = 100
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # The correlation is indexed from -(n-1) to (n-1)
        # We only care about the valid overlaps
        C5_val = np.max(corr) * dx
        
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h.copy()
        
        # Find indices contributing to maximum overlap
        max_idx = np.argmax(corr)
        # Convert to actual lag value (from -n+1 to n-1)
        lag = max_idx - (n_points - 1)
        
        # Create small adjustments
        # Get the current value at positions that contribute to the maximum
        h_new = h.copy()
        
        # Adjust h to reduce overlap at current lag
        k = lag
        # We want to decrease h[i] and/or increase h[(i + k) % n] to reduce this overlap
        # Move a bit of mass away from positions that contribute to the max overlap
        for i in range(n_points):
            j = (i + k) % n_points
            if h[i] > 0.01 and h[j] < 0.99:
                move_amount = min(epsilon * h[i], epsilon * (1 - h[j]))
                h_new[i] -= move_amount
                h_new[j] += move_amount
                
        # Re-normalize to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Apply perturbation periodically to escape local minima
        if iter_count % perturb_freq == 0:
            # Redistribute some mass
            indices = np.random.choice(n_points, size=min(5, n_points//4), replace=False)
            for i in indices:
                if h[i] > 0.01:

                    # Find a low-value position to move to
                    low_idx = np.argmin(h_new)

    final_C5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_C5, n_points)
