import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral of h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation h * (1 - h) in full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The correlation is of length 2*n - 1; we're interested in k=-n+1 to n-1
    # But we just need the maximum, so we take max of all values
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy sum constraint while preserving bounds [0,1]"""
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        # Distribute diff proportionally to current values, but clip at 1
        h = np.minimum(1.0, h + diff / len(h))
    else:
        # Need to decrease values
        diff = -diff
        # Try to distribute diff evenly but respect lower bound
        h = np.maximum(0.0, h - diff / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h = np.ones(n_points) * 0.5
        # Set odd positions to 0 to make it more structured
        h[1::2] = 0.0
        # Rescale to sum to target_sum
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    epsilon = 0.02
    perturb_freq = 100
    max_iter = 10000
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        # Evaluate current solution
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # If oscillating, reduce epsilon
        if iteration > 500 and iteration % 100 == 0:
            epsilon = max(0.001, epsilon * 0.95)
        
        # Compute cross-correlation at all lags
        corr = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(corr)
        max_lag = max_index - (len(h) - 1)
        max_value = corr[max_index]
        
        # Create a small perturbation to reduce overlap
        h_new = h.copy()
        weights = np.ones(len(h))
        # Adjust weights so that higher overlap areas are penalized more
        # For each x contribute to the effect of x+k in the correlation
        # We want to reduce the value at max_index
        # So we make adjustments in directions that reduce max_value
        adjustment = np.zeros(len(h))
        
        # Simple strategy: adjust elements that contribute the most to max overlap
        for i in range(len(h)):
            lag = max_lag
            j = (i + lag) % len(h)
            if h[i] > 0:
                # To reduce max_adj, we reduce h[i]
                adjustment[i] += -0.01 * epsilon
            if h[j] > 0:
                # Increase contribution of this element in the direction it contributes to max overlap
                adjustment[j] += -0.01 * epsilon        
 
        # Apply adjustment
        h_new += adjustment
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_constraint(h_new, target_sum)
        h = h_new
        
        # Perturb randomly every 100 iterations
        if iteration % perturb_freq == 0:
            # Find regions with high overlap
            corr = np.correlate(h, 1 - h, mode="full")
            # Select a random k (lag) causing significant overlap
            max_lag_indices = np.where(corr >= 0.9 * np.max(corr))[0]
            if len(max_lag_indices) > 0:
                sampled_lag = np.random.choice(max_lag_indices)
                lag = sampled_lag - (len(h) - 1)
                
                # Rand-move mass from current element to opposite region
                # Strategy: identify indices that contribute to overlap at lag
                for i in range(len(h)):
                    j = (i + lag) % len(h)
                    if h[i] > 0 and h[j] > 0:
                        # reduce i, increase j, or vice versa
                        move_amount = np.random.rand() * 0.02 * epsilon
                        h[i] = max(0, h[i] - move_amount)
                        h[j] = min(1, h[j] + move_amount)
            # Apply clipping and projection
            h = np.clip(h, 0, 1)
            h = project_to_sum_constraint(h, target_sum)
    
    # Final validation
    c5 = evaluate_erdos_solution(best_h)
    return (best_h, c5, len(best_h))
