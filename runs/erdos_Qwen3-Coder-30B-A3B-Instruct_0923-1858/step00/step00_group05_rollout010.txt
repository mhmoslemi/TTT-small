def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = signal.correlate(h, 1 - h, mode="full")
    # Maximize over all lags
    max_correlation = np.max(correlations)
    return max_correlation * dx
