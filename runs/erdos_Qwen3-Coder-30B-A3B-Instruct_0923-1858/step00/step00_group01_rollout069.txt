import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Compute C5 bound for given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, dx):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    target_sum = len(h) / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        scale = target_sum / current_sum
        h = h * scale
    return h

def crossover_h(h1, h2):
    """Create child from two parents using uniform crossover."""
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    child = np.zeros(n)
    child[:crossover_point] = h1[:crossover_point]
    child[crossover_point:] = h2[crossover_point:]
    return child

def mutate_h(h, mutation_rate=0.05):
    """Apply small random mutations to h."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Add small random noise
            h[i] += np.random.normal(0, 0.05)
    return h

def evolve_population(population, fitnesses, population_size, n_points, dx, 
                      generations, mutation_rate, elite_ratio=0.2):
    """Evolve population for given number of generations."""
    elite_size = int(elite_ratio * population_size)
    for gen in range(generations):
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        best_h = population[sorted_indices[0]]
        best_fitness = fitnesses[sorted_indices[0]]
        
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Select two parents
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutation
            child = mutate_h(child, mutation_rate)
            
            # Repair
            child = repair_h(child, dx)
            
            new_population.append(child)
            
        population = new_population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
    # Return best individual
    best_idx = np.argmin(fitnesses)
    return population[best_idx], fitnesses[best_idx]

    # Initialize population
    population_size = 75

    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)

        population.append(h)
    
    # Evolution parameters
    max_generations = 200
