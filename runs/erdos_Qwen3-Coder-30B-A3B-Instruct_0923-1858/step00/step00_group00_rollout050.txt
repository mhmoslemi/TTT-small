import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project h to satisfy sum constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vector):
        # Compute the correlation
        conv = np.correlate(h_vector, 1 - h_vector, mode='full')
        # The maximum correlation value (with appropriate scaling)
        return np.max(conv) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters for simulated annealing
    T = 0.1
    alpha = 0.99
    sigma = 0.1
    noise_scale = 0.01
    
    iterations_without_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s and iterations_without_improvement < max_no_improvement:
        # Perturb the solution
        h_candidate = h + np.random.normal(0, sigma, n_points)
        
        # Clamp to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to satisfy constraint
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate)
        
        # Accept with simulated annealing probability
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
        
        # Update temperature
        T *= alpha
        
        # Decrease noise level
        if iterations_without_improvement % 100 == 0:
            sigma *= 0.9
    
    return (best_h.tolist(), best_c5, n_points)
