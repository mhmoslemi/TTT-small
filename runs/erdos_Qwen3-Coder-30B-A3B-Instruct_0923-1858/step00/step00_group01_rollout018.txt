import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation is at the center (index n_points-1)
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h_values(h, n_points, dx):
    # Ensure h values are in [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct integral
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (1.0 / integral) * (n_points / 2)
    return np.clip(h, 0, 1)

def crossover(parent1, parent2, n_points):
    # Uniform crossover
    mask = np.random.rand(n_points) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.05):
    # Gaussian mutation
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, 0.1, len(h))
    h = h + noise * mask
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    population = []
    if initial_h_values is not None:
        h0 = np.array(initial_h_values)
        h0 = repair_h_values(h0, n_points, 2.0 / n_points)
        population.append(h0)
    
    # Fill remaining population with random valid h
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h_values(h, n_points, 2.0 / n_points)
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 47  # Start with coarse resolution
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.02
    selection_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h_values)
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness of all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection
        num_elite = int(pop_size * selection_rate)
        elite = population[:num_elite]
        
        # Generate new population
        new_population = elite[:]
        
        # Crossover
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(elite) >= 2:
                parent1, parent2 = np.random.choice(elite, size=2, replace=False)
                child = crossover(parent1, parent2, n_points)
            else:
                # Select a random elite as a parent for mutation
                parent = np.random.choice(elite)
                child = parent.copy()
            # Mutation
            child = mutate(child, mutation_rate)
            # Repair
            child = repair_h_values(child, n_points, dx)
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
