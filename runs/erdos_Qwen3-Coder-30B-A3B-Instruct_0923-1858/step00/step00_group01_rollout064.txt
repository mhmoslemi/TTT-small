import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default: simple step function
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    
    last_improvement = 0
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Target resolution (2x)
    target_n_points = 2 * n_points
    
    for iteration in range(1000):
        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Extract the relevant part
        mid = len(corr) // 2
        C5_vals = corr[mid:] * dx
        
        current_C5 = np.max(C5_vals)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iteration
            
        # If no improvement in 100 iterations, stop
        if iteration - last_improvement > 100:
            break
            
        # Identify regions that contribute most to max C5
        max_k = np.argmax(C5_vals)
        # We will avoid effects near boundaries, so focus on central region
        # Get approximate range of contribution
        max_idx = mid + max_k
        kernel_len = len(h)
        shift = max_k - (kernel_len - 1) // 2  # Shift amount to get maximum
        
        # Refine focus on where maximum occurs
        # Find critical indices
        critical_indices = range(max(0, shift), min(len(h), len(h) + shift))
        
        # Try redistributing mass in critical regions
        h_new = h.copy()
        attempts = 0
        
        while attempts < 10:  # Limit attempts
            attempts += 1
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            # Ensure no out of bounds and try to make changes
            # Transfer from i to j if beneficial
            if i == j:
                continue
                
            # Check if we are allowed to transfer
            max_transfer = min(h_new[i], 1 - h_new[j])
            if max_transfer <= 0:
                continue
            transfer = np.random.rand() * max_transfer
            
            # Apply transfer
            h_new[i] -= transfer
            h_new[j] += transfer
            
            # Clip values to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # If we maintain the constraint sum(h) = 1
            if abs(np.sum(h_new) * dx - 1.0) < 1e-5:
                break
            
        if attempts < 10:
            h = h_new.copy()
        else:
            # Alternative: modify a few points with small changes
            idxs = np.random.choice(n_points, size=min(3, n_points), replace=False)
            h_mod = h.copy()
            for i in idxs:
                new_val = h[i] + np.random.normal(0, 0.01)
                h_mod[i] = np.clip(new_val, 0, 1)
            h_mod = np.clip(h_mod, 0, 1)
            # Normalize to maintain constraint sum(h) = 1
            h_mod *= (1.0 / (np.sum(h_mod) * dx)) * dx
            h = h_mod
            
    # After optimization, interpolate to higher resolution
    if target_n_points > n_points:
        h_interp = np.interp(np.linspace(0, 2, target_n_points), 
                            np.linspace(0, 2, n_points), 
                            best_h)
        h_final = h_interp
        dx_final = 2.0 / target_n_points
    else:
        h_final = best_h
        dx_final = dx
        
    # Final check of constraint and C5 value
    expected_sum = 1.0
    actual_sum = np.sum(h_final) * dx_final
    
    if abs(actual_sum - expected_sum) > 1e-5:
        h_final = h_final * (expected_sum / actual_sum)
        
    # Recalculate final C5
    corr_final = correlate(h_final, 1 - h_final, mode='full')
    mid_final = len(corr_final) // 2
    C5_final_vals = corr_final[mid_final:] * dx_final
    final_C5 = np.max(C5_final_vals)
    
    return (h_final.tolist(), float(final_C5), len(h_final))
