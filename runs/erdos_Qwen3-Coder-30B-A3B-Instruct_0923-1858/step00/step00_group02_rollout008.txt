import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)
    else:

        n_points = 45
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            if x < 1.0:
                h[i] = x
            else:
                h[i] = 2.0 - x
        # Normalize to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    # Initial score
    c5_bound = evaluate_erdos_solution(h, n_points)
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    
    # Gradient approximation with finite differences
    def finite_diff_grad(func, x, eps=1e-4):
        n = len(x)
        grad = np.zeros(n)
        for i in range(n):
            dx = np.zeros(n)
            dx[i] = eps
            grad[i] = (func(x + dx) - func(x - dx)) / (2 * eps)
        return grad
    
    # Objective function for gradient evaluation
    def obj_func(h_vec):
        return evaluate_erdos_solution(h_vec, n_points)
    
    # Project h to feasible region
    def project_h(h_vec):
        h_proj = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h_proj)
        if sum_h > 0:
            h_proj = h_proj * (n_points / 2) / sum_h
        return h_proj
    
    # Optimize
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute current objective and gradient
        current_obj = obj_func(h)
        grad = finite_diff_grad(obj_func, h)
        
        # Try to take a gradient step
        step = -grad
        step_norm = np.linalg.norm(step)
        
        # Normalize step
        if step_norm > 0:
            step = step / step_norm
        
        # Compute proposed new point with trust region
        proposed_h = h + trust_radius * step
        
        # Project back to feasible region
        proposed_h = project_h(proposed_h)
        
        # Evaluate new objective
        new_obj = obj_func(proposed_h)

        # Accept if better
        if new_obj < current_obj:
            h = proposed_h
            c5_bound = new_obj
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
            trust_radius *= 1.2  # Increase trust radius if successful
        else:
            trust_radius *= 0.5  # Reduce trust radius if failed

        # Occasionally add noise to escape local minima
        if iter_count % 100 == 0 and trust_radius > 0.01:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = project_h(h)
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
