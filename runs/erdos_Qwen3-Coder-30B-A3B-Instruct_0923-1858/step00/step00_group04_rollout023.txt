import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    # Compute overlap integral
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_feasible(h, n_points):
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s > 0:
        h = h * (n_points / 2) / s
    return h

def crossover(parent1, parent2, alpha=0.5):
    # Blend crossover
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    # Apply mutation
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return h

def tournament_selection(population, fitnesses, k=5):
    # Select one individual using tournament selection
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 42
    dx = 2.0 / n_points
    pop_size = 30
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_size = int(0.1 * pop_size)
    replacement_size = int(0.2 * pop_size)
    
    best_h = None
    best_c5 = float('inf')
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = project_to_feasible(np.array(initial_h_values), n_points)
        population.append(h0)
    else:
        h0 = np.random.random(n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0)
    
    # Fill the rest of the population
    for _ in range(pop_size - 1):
        h = np.random.random(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    start_time = time.time()
    
    for gen in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate(h, dx) for h in population]
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitness = [fitnesses[i] for i in sorted_indices]
        
        # Keep elites
        new_population = sorted_pop[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            if np.random.random() < crossover_prob:
                parent2 = tournament_selection(population, fitnesses)
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Apply mutation
            child = mutate(child, mutation_rate)
            child = project_to_feasible(child, n_points)
            new_population.append(child)
            
        population = new_population[:pop_size]
        
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fall back to initial solution if no better found
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h_final = project_to_feasible(np.array(initial_h_values), n_points)
            final_c5 = evaluate(h_final, dx)
            return (h_final.tolist(), final_c5, n_points)
        else:
            h_final = np.ones(n_points) * 0.5
            final_c5 = evaluate(h_final, dx)
            return (h_final.tolist(), final_c5, n_points)
