import numpy as np
from scipy import signal
import time

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, dx, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h
