import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # fallback construction
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[1] = 0.0
        h0[2] = 0.0
        h0[3] = 1.0
        h0[4] = 1.0
        h0[5] = 1.0
        h0[6] = 0.0
        h0[7] = 0.0
        h0[8] = 0.0
        h0[9] = 1.0
        h0[10] = 1.0
        h0[11] = 1.0

    # ensure normalization
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    
    dx = 2.0 / n_points
    max_shifts = n_points - 1

    def compute_overlap(h, dx):
        # Use numpy correlate to compute all overlap values
        # For each k, compute integral of h(x) * (1 - h(x + k)), i.e., h * (1 - h) with mode="full"
        # But since shift is v from [0, 2], and we shift by 2 * dx units, need to align indices
        # full convolution: full length is 2 * n - 1
        h1 = 1 - h
        full_corr = np.correlate(h, h1, mode='full')
        # take only the valid overlaps (not shifted off boundary)
        start_idx = n_points - 1
        end_idx = start_idx + n_points
        overlaps = full_corr[start_idx:end_idx] * dx
        # returns the list of overlaps for shifts from - (n-1) to (n-1)
        return overlaps

    def evaluate_solution(h):
        overlaps = compute_overlap(h, dx)
        return np.max(overlaps)

    def add_constraints(prob, h_var, C_var, shifts):
        for k in shifts:
            # from indices to actual shift values: position distance from - (n-1) to (n-1)
            if k < 0:
                k_shift = -k
            else:
                k_shift = k
            # real full overlap value
            corr = np.correlate(h_var[:n_points], 1 - h_var[:n_points], mode='full')
            # get the correct indices from the full correlation
            start_idx = n_points - 1
            actual_corr = corr[start_idx + k]
            # Add constraint
            prob.add(cp.Constraint(actual_corr <= C_var / dx))

    # Cutting plane algorithm
    h_current = h0.copy()
    C_current = evaluate_solution(h_current)

    # termination criteria
    best_h = h_current.copy()
    best_C = C_current

    remaining_time = budget_s - (time() - start_time)
    max_iter = 200  # quick iterations
    iter_count = 0

    # do optimization
    while iter_count < max_iter and (time() - start_time) < budget_s - 2.0:
        iter_count += 1
        # compute overlaps
        overlaps = compute_overlap(h_current, dx)
        # find shift indices with high overlap
        # only consider indices in the central region for active constraints
        # to avoid too many constraints
        shift_indices = []
        for i in range(len(overlaps)):
            if i >= 2 and i < len(overlaps) - 2:  # skip edges
                shift_indices.append(i)
        
        # sort by overlap value descending
        sorted_shifts = sorted(shift_indices, key=lambda i: -overlaps[i])
        top_k = min(40, len(sorted_shifts) // 4)
        active_indices = sorted_shifts[:top_k]

        # build CVXPY variables and problem
        h_vars = cp.Variable(n_points)
        C_var = cp.Variable(1)

        # define constraints
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2.0)
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1.0)
        
        for idx in active_indices:
            start_idx = n_points - 1
            corr_idx = start_idx + idx - (n_points - 1)  # convert to normalized index
            # this is approximate; fix with manual computation or layers for true efficiency
            corr_val = np.correlate(h_vars[:n_points], 1 - h_vars[:n_points], mode='full')[start_idx + idx - (n_points-1)]
            constraints.append(corr_val <= C_var)
        
        prob = cp.Problem(cp.Minimize(C_var), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-8, reltol=1e-8)
            if prob.status == cp.OPTIMAL:
                h_opt = h_vars.value
                C_opt = C_var.value
                # update solution if better
                if C_opt < best_C:
                    best_C = C_opt
                    best_h = h_opt.copy()
        except Exception:
            # fallback
            pass

        # Now refine with more accurate overlaps
        h_current = best_h.copy()
        C_current = best_C

        # BREAK early if objective is very small
        if best_C <= 0.3807:
            break

    # Final validate
    final_overlap = evaluate_solution(best_h)
    return (best_h, final_overlap, n_points)
