from scipy import correlate
import time

    T = 1.0

    perturb_scale = 0.1
    max_iter = 10000
    patience = 1000
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

            perturb_scale *= 0.99
            
        # Create candidate by perturbing h
        candidate_h = h + np.random.normal(0, perturb_scale, n_points)
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Rescale to satisfy the constraint
        candidate_h *= (n_points / 2) / np.sum(candidate_h)
        
        # Calculate C5 for candidate
        candidate_C5 = max(correlate(candidate_h, 1-candidate_h, mode="full")) * dx
        
        # Accept or reject
        delta = candidate_C5 - best_C5
