import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    corr = np.correlate(h, 1-h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def clip_and_rescale(h):
    """Clip h to [0,1] and rescale to ensure integral = 1"""
    h = np.clip(h, 0, 1)
    n = len(h)
    dx = 2.0 / n
    total = np.sum(h) * dx
    if total > 0:
        h = h / total * 1.0
    return h

def gradient_descent_step(h, lr=0.01):
    """Perform a gradient descent step to minimize max correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute current correlations
    corr = np.correlate(h, 1-h, mode="full")
    max_corr_idx = np.argmax(corr)
    # Compute gradient of correlation at max point
    grad = np.zeros(n)
    for i in range(n):
        # Gradient contribution from correlation at max point
        # This is a simplified approximation
        # For exact gradient, we'd compute partial derivatives
        # of correlation w.r.t. each h[i]
        grad[i] = 0
        # Effective gradient approximation, based on finite differences
        for k in range(n):
            if i+k < n and i+k >= 0:
                grad[i] += (1-h[i+k]) * dx
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    # Update and clip
    new_h = h - lr * grad
    # Rescale to maintain integral
    new_h = clip_and_rescale(new_h)
    return new_h
