import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    except NameError:
        # Create a simple initial guess

        h[10:30] = 1.0  # Set some values to 1
        h[40:60] = 0.0  # Set some values to 0
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.1
    decay = 0.9
    tol = 1e-6
    start_time = time.time()
    
    def compute_c5(h_vals):
        # Compute overlap with all shifts
        h_pad = np.concatenate([h_vals, np.zeros(n_points)])
        overlaps = signal.correlate(h_pad, 1 - h_vals, mode='full')
        max_overlap = np.max(overlaps[n_points-1:])
        return max_overlap * dx
    
    def project_h(h_vals):
        # Clamp to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to meet sum constraint
        h_sum = np.sum(h_vals)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-10:
            # Use proportional adjustment
            diff = target_sum - h_sum
            # Distribute adjustment in proportion to current values
            if h_sum > 0:
                adjustment = diff / h_sum
                h_vals = h_vals * (1 + adjustment)
                h_vals = np.clip(h_vals, 0, 1)
                # Re-adjust to sum exactly
                h_sum = np.sum(h_vals)
                if abs(h_sum - target_sum) > 1e-10:
                    # Simple redistribution
                    excess = h_sum - target_sum
                    # Reduce values proportionally from larger ones
                    weights = h_vals / np.sum(h_vals)
                    h_vals = h_vals - excess * weights
                    h_vals = np.clip(h_vals, 0, 1)
        # Final sum check
        h_sum = np.sum(h_vals)
        if abs(h_sum - target_sum) > 1e-10:
            # Simply scale to target
            h_vals = h_vals * target_sum / h_sum
            h_vals = np.clip(h_vals, 0, 1)
        return h_vals
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all shifts
        h_pad = np.concatenate([h, np.zeros(n_points)])
        overlaps = signal.correlate(h_pad, 1 - h, mode='full')
        k_max = np.argmax(overlaps[n_points-1:]) + (n_points-1)
        max_overlap = overlaps[k_max]
        
        # Gradient update
        gradient = np.zeros_like(h)
        for j in range(n_points):
            k = k_max - n_points + 1 + j
            if k >= 0 and k < n_points:
                gradient[j] += (1 - h[k])
            k = k_max - n_points + 1 + j
            if 0 <= k < n_points:
                gradient[j] -= h[k]
        
        h_new = h - alpha * gradient
        h_new = project_h(h_new)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - tol:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Adaptive step size
        if i % 100 == 0 and i > 0:
            alpha *= decay
            
        # Early stopping if no improvement
        if i > 0 and abs(current_c5 - c5_new) < tol:
            last_improvement += 1
            if last_improvement > 500:
                break
        else:
            last_improvement = 0
        current_c5 = c5_new
    
    # Final rounding to binary
    # Use thresholding approach for binary values
    # But keep sum constraint
    avg_val = np.sum(best_h) / len(best_h)

    rounded_h = (best_h >= threshold).astype(float)
    # Normalize to meet constraint
    h_sum = np.sum(rounded_h)
    target_sum = n_points / 2
    if abs(h_sum - target_sum) > 1e-5:
        # Adjust values
        diff = target_sum - h_sum
        indices = np.argsort(rounded_h)
        if diff > 0:
            # Need to add 1 to some zeros
            for i in range(int(diff)):
                if indices[i] < len(rounded_h) - 1 and rounded_h[indices[i]] == 0:
                    rounded_h[indices[i]] = 1
        else:
            # Need to remove 1 from some ones
            for i in range(int(-diff)):
                if indices[-(i+1)] > 0 and rounded_h[indices[-(i+1)]] == 1:
                    rounded_h[indices[-(i+1)]] = 0
    
    final_c5 = compute_c5(rounded_h)
    return (rounded_h.tolist(), final_c5, n_points)
