import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    dx = 2.0 / 70
    n_points = 70
    max_iter = 10000
    learning_rate = 1e-3
    tolerance = 1e-5
    time_limit = time.time() + budget_s
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        noise = np.random.normal(0, 0.01, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent
    for iteration in range(max_iter):
        if time.time() > time_limit:
            break
            
        # Compute cross-correlation
        corr = np.correlate(h, 1-h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        # Compute gradient using finite differences for simplicity
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
            grad[i] = (corr_plus[max_corr_idx] - max_corr_val) / eps
        
        # Update step
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Check improvement
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adaptive learning rate
        if iteration % 10 == 0 and iteration > 0:
            learning_rate *= 0.9
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1-h, mode='full')
    max_overlap = np.max(corr)
    return max_overlap * dx
