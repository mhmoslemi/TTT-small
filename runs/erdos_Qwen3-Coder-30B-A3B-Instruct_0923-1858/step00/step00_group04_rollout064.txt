import numpy as np
import time
from scipy import signal
from scipy.optimize import differential_evolution
import math

def evaluate_C5(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation h * (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation (scaled appropriately)
    max_corr = np.max(corr) * dx
    return max_corr

def init_population(n_points, pop_size, use_initial=True):
    """Initialize population of step functions"""
    population = []
    if use_initial and 'initial_h_values' in globals():
        # Use initial construction if available
        h0 = np.array(initial_h_values)
        # Normalize to sum = n_points / 2
        h0 = h0 * (n_points / 2) / np.sum(h0)
        population.append(h0.copy())
        pop_size -= 1
    
    # Fill remaining population with random feasible functions
    for _ in range(pop_size):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize sum
        h = np.clip(h, 0, 1)  # Keep in [0,1]
        population.append(h.copy())
    
    return population

def mutate(h, rate=0.1, mutation_strength=0.05):
    """Apply small perturbations to h"""
    h = h.copy()
    n = len(h)
    indices = np.random.choice(n, size=int(rate*n), replace=False)
    for i in indices:
        # Add small random perturbation
        perturbation = np.random.normal(0, mutation_strength)
        h[i] += perturbation
        
    # Maintain sum constraint
    h_sum = np.sum(h)
    h = h * (n / 2) / h_sum
    h = np.clip(h, 0, 1)
    return h

def crossover(h1, h2, crossover_prob=0.7):
    """Blend crossover between two individuals"""
    h1, h2 = h1.copy(), h2.copy()
    n = len(h1)
    if np.random.random() > crossover_prob:
        return h1.copy()
        
    # Blend crossover
    alpha = np.random.random()
    h_new = alpha * h1 + (1 - alpha) * h2
    
    # Maintain sum constraint
    h_sum = np.sum(h_new)
    h_new = h_new * (n / 2) / h_sum
    h_new = np.clip(h_new, 0, 1)
    return h_new

def genetic_algorithm(n_points, budget_s=600, pop_size=30):
    """Main genetic algorithm loop"""
    start_time = time.time()
    
    # Initialize population
    population = init_population(n_points, pop_size, use_initial=True)
    fitness_history = []
    global_best = None
    global_best_fitness = float('inf')
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            fitness = evaluate_C5(h)
            fitnesses.append(fitness)
            
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = h.copy()
                no_improvement_count = 0
                
        fitness_history.append(global_best_fitness)
        
        # Selection (tournament)
        tournament_size = 5
        selected_indices = []
        for _ in range(pop_size):
            tournament = np.random.choice(len(population), size=tournament_size)
            winner = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create new population through crossover and mutation
        new_population = []
        elitism_count = pop_size // 10
        # Keep elite individuals
        elite_indices = np.argsort(fitnesses)[:elitism_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            p1_idx = selected_indices[np.random.randint(0, len(selected_indices))]
            p2_idx = selected_indices[np.random.randint(0, len(selected_indices))]
            if p1_idx == p2_idx:
                p2_idx = (p1_idx + 1) % len(selected_indices)
            p1, p2 = population[p1_idx], population[p2_idx]
            
            # Crossover
            offspring = crossover(p1, p2)
            
            # Mutate
            if np.random.random() < 0.3:  # 30% mutation rate
                offspring = mutate(offspring, rate=0.1, mutation_strength=0.1)
                
            new_population.append(offspring.copy())
            
        population = new_population.copy()
        no_improvement_count += 1
        
    if global_best is None:
        # Return best individual from final population
        fitnesses = [evaluate_C5(h) for h in population]
        best_idx = np.argmin(fitnesses)
        global_best = population[best_idx].copy()
        global_best_fitness = fitnesses[best_idx]
        
    return global_best, global_best_fitness

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    
    # Use 42 samples as in the initial solution
    n_points = 42
    h_values, c5_bound = genetic_algorithm(n_points, budget_s)
    
    return (h_values, c5_bound, n_points)
