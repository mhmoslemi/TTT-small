import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
import math

def evaluate_h(h_values, n_points):
    # Compute the overlap integral
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    # Optimize: correlation in full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap
    max_overlap = np.max(corr) * dx
    return max_overlap

def construct_initial_h(n_points):
    if 'initial_h_values' in globals():

        # Normalize to sum to n_points / 2
        h0 = h0 * (n_points / 2) / np.sum(h0)

    else:
        # Default construction
        h0 = np.full(n_points, 0.5)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        return h0

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # To avoid large memory consumption, use a smaller number of points
    n_points = 70  # same as default but can be changed if needed
    dx = 2.0 / n_points
    
    # Construct the initial solution
    h0 = construct_initial_h(n_points)
    
    # The objective function to minimize
    def objective(h):
        return evaluate_h(h, n_points)
    
    # Bounds for the variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add sum constraint using penalty
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Create a modified objective with a penalty for constraint violation
    def penalized_objective(h):
        penalty = 0.0
        sum_violation = abs(constraint_sum(h))
        if sum_violation > 1e-6:
            penalty = 1000 * sum_violation ** 2
        return objective(h) + penalty
    
    # Optimize using differential evolution first for global search
    de_result = differential_evolution(
        penalized_objective,
        bounds,
        maxiter=500,
        popsize=15,
        seed=seed,
        disp=False,
        timeout=budget_s - (time.time() - start_time) - 1
    )
    
    if not de_result.success:
        # Fall back to Nelder-Mead if DE does not converge
        try:
            # Create initial guess
            x0 = de_result.x if de_result.success else h0.copy()
            # Refine it with Nelder-Mead
            result = minimize(
                penalized_objective,
                x0,
                method='Nelder-Mead',
                options={'maxiter': int(1000), 'adaptive': True},
                tol=1e-6
            )
            h_final = result.x
        except:
            h_final = h0
    else:
        h_final = de_result.x
    
    # Final refinement and ensure constraints
    # Use the full objective again to evaluate
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final C5 computation
    c5_bound = evaluate_h(h_final, n_points)
    
    # Return the solution
    return (h_final.tolist(), c5_bound, n_points)
