import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for given h and dx."""
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h_to_constraints(h, dx):
    """Project h to satisfy constraints: h in [0,1], sum(h)*dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx)) * dx  # maintain sum = 1
    return h

def initial_h_from_parent():

    h[indices] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 47  # Standard resolution
    dx = 2.0 / n_points
    h = initial_h_from_parent()  # Will return 0s or parent if available
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Create a function to compute gradient (approximate) for optimization
    def objective_and_grad(h_vec):
        h = h_vec.copy()
        h = project_h_to_constraints(h, dx)
        c5 = evaluate_erdos_solution(h, dx)
        # Simple gradient estimation for each point
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h_to_constraints(h_plus, dx)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5) / eps
        return c5, grad

    # Optimization loop
    iter_count = 0
    learning_rate = 0.01
    max_iter = int(budget_s * 20)  # Roughly 20 iter per second

                h_plus[i] += eps
                c5_plus = fun(h_plus)
                c5 = fun(h_vec)
                grad[i] = (c5_plus - c5) / eps
            return grad

        # Employ gradient descent step
        c5, grad = objective_and_grad(best_h)
        new_h = best_h - learning_rate * grad
        new_h = project_h_to_constraints(new_h, dx)
        new_c5 = evaluate_erdos_solution(new_h, dx)
