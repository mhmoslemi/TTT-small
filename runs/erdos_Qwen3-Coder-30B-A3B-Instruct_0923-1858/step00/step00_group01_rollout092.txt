def evaluate_c5(h):
    """Evaluate the overlap integral C5 for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation using np.correlate
    corr = np.correlate(h, 1-h, mode="full")
    # Take maximum value
    c5 = np.max(corr) * dx
    return c5

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to match target sum
        scale = target_sum / current_sum
        h = h * scale
    return h

        delta = np.random.normal(0, mutation_strength, n_points) * mask
        h_mut += delta
        # Repair to maintain constraints
        h_mut = repair_h(h_mut, n_points)
    return h_mut

    child = repair_h(child, n_points)
