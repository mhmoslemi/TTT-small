import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to correct size
            h_new = np.zeros(n_points)
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x_old, h)
            h = h_new
    else:
        h = np.full(n_points, 0.5)
        h = h * target_sum / np.sum(h)
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    def compute_C5(h_vec):
        # Compute correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the valid range
        valid_corr = corr[len(h_vec)-1:]
        return np.max(valid_corr) * dx
    
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    # Define bounds (0 to 1 for all elements)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use evolutionary optimization to improve construction
    def optimize_population():
        best_h = h.copy()
        best_C5 = compute_C5(best_h)
        
        for gen in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate population
            population = []
            for _ in range(20):
                # Create random variation
                indiv = np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1)
                # Adjust to match sum constraint
                indiv = indiv * target_sum / np.sum(indiv)
                population.append(indiv.copy())
            
            # Evaluate fitness
            fitnesses = [compute_C5(indiv) for indiv in population]
            
            # Select best
            best_idx = np.argmin(fitnesses)
            if fitnesses[best_idx] < best_C5:
                best_h = population[best_idx].copy()
                best_C5 = fitnesses[best_idx]
        
        return best_h, best_C5
    
    # Do initial optimization
    h_opt, c5_opt = optimize_population()
    
    # Refine with scipy
    def obj(h_vec):
        return compute_C5(h_vec)
    
    # Define constraints
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    ]
    
    # Use bounds
    bounds_scipy = [(0, 1) for _ in range(n_points)]
    
    # Refine the solution
    try:
        result = minimize(obj, h_opt, method='SLSQP', bounds=bounds_scipy, 
                          constraints=constraints, options={'maxiter': 100})
        if result.success:
            h_refined = result.x.copy()
            c5_refined = compute_C5(h_refined)
            if c5_refined < c5_opt:
                h_opt = h_refined
                c5_opt = c5_refined
    except:
        pass
    
    # Final validation
    c5_final = compute_C5(h_opt)
    
    return (h_opt, c5_final, n_points)
