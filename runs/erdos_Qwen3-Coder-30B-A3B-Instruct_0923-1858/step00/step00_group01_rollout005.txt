import numpy as np

import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, dx):
    """Project h to ensure sum(h) * dx = 1 and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h
