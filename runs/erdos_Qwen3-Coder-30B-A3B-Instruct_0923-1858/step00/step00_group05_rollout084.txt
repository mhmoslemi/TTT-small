import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_time = lambda: time.time() - start_time
    
    # Use initial construction if available, else create a reasonable starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    
    def evaluate_C5(h_vec):
        # Compute full correlation and find maximum
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vec):
        # Ensure values are in [0,1] and sum = n_points/2
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def compute_gradient(h_vec, alpha=1e-2):
        # Simple finite difference for gradient estimation
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_C5 = evaluate_C5(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C5_plus = evaluate_C5(h_plus)
            grad[i] = (C5_plus - base_C5) / eps
        return grad
    
    def optimize_step(h_vec, lr=1e-3, max_iter=100):
        for i in range(max_iter):
            if total_time() > budget_s - 1:
                break
            grad = compute_gradient(h_vec)
            h_vec = h_vec - lr * grad
            h_vec = project_h(h_vec)
            if i % 10 == 0:
                current_C5 = evaluate_C5(h_vec)
                if i > 0 and (prev_C5 - current_C5) < 1e-6:
                    lr *= 0.9
                prev_C5 = current_C5
        return h_vec
    
    # Stage 1: Low resolution optimization (e.g., n_points=50)
    if n_points > 50:
        # Create a lower resolution version
        h_low = np.interp(np.linspace(0, 2, 50), np.linspace(0, 2, n_points), h)
        # Ensure the new h_low satisfies constraints
        h_low = project_h(h_low)
        h_low = optimize_step(h_low, lr=1e-2, max_iter=100)
        # Upsample to current resolution
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 50), h_low)
        h = project_h(h)
    
    # Stage 2: High resolution optimization
    h = optimize_step(h, lr=1e-3, max_iter=500)
    
    # Final evaluation
    final_C5 = evaluate_C5(h)
    
    return (h.tolist(), final_C5, n_points)
