import numpy as np
import time
from scipy.signal import correlate

def binaryify(h, n_points):
    """Convert h to binary with exact sum constraint."""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    
    if current_sum == target_sum:
        return h
    
    # Adjust to meet sum constraint
    diff = target_sum - current_sum
    if diff > 0:  # Need to add more 1s
        indices = np.argsort(h)
        for i in range(diff):
            h[indices[i]] = 1
    else:  # Need to remove 1s
        indices = np.argsort(1 - h)
        for i in range(-diff):
            h[indices[i]] = 0
    
    return h

def compute_C5(h, dx):
    """Compute the maximum overlap integral."""
    h1 = np.array(h, dtype=float)
    h2 = 1.0 - h1
    corr = correlate(h1, h2, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def local_search(h, n_points, dx, max_iter, temp_start=1.0, alpha=0.999, time_budget_s=600):
    """Perform local search with simulated annealing."""
    start_time = time.time()
    current_h = binaryify(h.copy(), n_points)
    current_C5 = compute_C5(current_h, dx)
    best_h = current_h.copy()
    best_C5 = current_C5
    
    temp = temp_start
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < time_budget_s:
        # Select two indices: one with 1, one with 0
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try a move: flip current_h[i] and current_h[j]
        new_h = current_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        
        new_C5 = compute_C5(new_h, dx)
        delta = new_C5 - current_C5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
                
        temp *= alpha
        iteration += 1
        
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Determine problem size
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple default construction
        h = np.zeros(n_points)
        half_points = n_points // 2
        h[:half_points] = 1
    
    # Run local search
    best_h, best_C5 = local_search(h, n_points, dx, max_iter=10000, time_budget_s=budget_s)
    
    # Final verification
    best_h = binaryify(best_h, n_points)
    final_C5 = compute_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)
