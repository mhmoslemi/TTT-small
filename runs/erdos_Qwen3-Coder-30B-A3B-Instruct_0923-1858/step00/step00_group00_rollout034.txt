import numpy as np
from scipy import signal
import time

def evaluate_overlap(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx """
    # Correlation with reflected signal
    corr = signal.correlate(h, 1-h, mode="full")
    # Find maximum correlation value and normalize
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points/2) / h_sum
    return h

def nelder_mead_simplex(h0, n_points, dx, budget_s):
    # Nelder-Mead parameters
    alpha = 1.0   # reflection
    gamma = 2.0   # expansion
    beta = 0.5    # contraction
    rho = 0.5     # shrink
    initial_size = 0.1
    max_reset_attempts = 10
    reset_interval = 50
    
    # Initial simplex
    n = len(h0)
    simplex = [h0]
    for i in range(n):
        h = h0.copy()
        h[i] += initial_size
        h = project_h(h, n_points)
        simplex.append(h)
    
    # Evaluate initial simplex
    values = []
    for h in simplex:
        v = evaluate_overlap(h, dx)
        values.append(v)
    indices = np.argsort(values)
    simplex = [simplex[i] for i in indices]
    values = [values[i] for i in indices]
    
    best_h = simplex[0]
    best_value = values[0]
    
    start_time = time.time()
    iter_count = 0
    reset_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s:
        if iter_count - last_improvement > reset_interval:
            if reset_count >= max_reset_attempts:
                break
            reset_count += 1
            last_improvement = iter_count
            # Shrink simplex
            best = simplex[0]
            for i in range(1, len(simplex)):
                simplex[i] = best + rho * (simplex[i] - best)
                simplex[i] = project_h(simplex[i], n_points)
        
        # Sort by value
        indices = np.argsort([evaluate_overlap(h, dx) for h in simplex])
        simplex = [simplex[i] for i in indices]
        values = [evaluate_overlap(h, dx) for h in simplex]
        best_h = simplex[0]
        best_value = values[0]
        
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_r = centroid + alpha * (centroid - simplex[-1])
        h_r = project_h(h_r, n_points)
        v_r = evaluate_overlap(h_r, dx)
        
        if values[0] <= v_r < values[-2]:
            simplex[-1] = h_r
        elif v_r < values[0]:
            # Expansion
            h_e = centroid + gamma * (h_r - centroid)
            h_e = project_h(h_e, n_points)
            v_e = evaluate_overlap(h_e, dx)
            if v_e < v_r:
                simplex[-1] = h_e
            else:
                simplex[-1] = h_r
        else:
            # Contraction
            h_c = centroid + beta * (simplex[-1] - centroid)
            h_c = project_h(h_c, n_points)
            v_c = evaluate_overlap(h_c, dx)
            if v_c < values[-1]:
                simplex[-1] = h_c
            else:
                # Shrink
                best = simplex[0]
                for i in range(1, len(simplex)):
                    simplex[i] = best + rho * (simplex[i] - best)
                    simplex[i] = project_h(simplex[i], n_points)
        
        iter_count += 1
        if v_r < best_value:
            last_improvement = iter_count
            best_value = v_r
            best_h = h_r
    
    return best_h, best_value
