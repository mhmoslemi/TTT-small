import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: symmetric construction centered at 1.0
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Create a slightly perturbed symmetric version
        mid = n_points // 2
        for i in range(mid):
            h[i] = 0.5 + 0.2 * (i - mid/2) / (mid/2)
            h[n_points - 1 - i] = h[i]
        if n_points % 2 == 1:
            h[mid] = 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    # Start timing
    start_time = time.time()
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h, dx)
    
    T = 0.1  # Initial temperature
    alpha = 0.99  # Annealing factor
    sigma = 0.1  # Initial perturbation strength
    patience = 0
    max_patience = 1000
    
    # Optimization iterations
    for iteration in range(100000):
        if time.time() - start_time > budget_s:
            break
            
        # Perturb
        h_candidate = h + np.random.normal(0, sigma, n_points)
        
        # Project onto feasible region [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to maintain constraint sum(h) = n_points/2
        h_candidate *= (n_points/2) / np.sum(h_candidate)
        
        # Compute C5
        C5_candidate = compute_C5(h_candidate, dx)
        
        # Accept or reject
        if C5_candidate < best_C5:
            h = h_candidate.copy()
            best_C5 = C5_candidate
            patience = 0
        else:
            delta = C5_candidate - best_C5
            if np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()
                patience = 0
            else:
                patience += 1
        
        # Update temperature
        T *= alpha
        
        # Decrease sigma over time
        if iteration % 100 == 0:
            sigma *= 0.9
            
        if patience > max_patience:
            break
            
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Compute convolution of h and (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Return maximum scaled by dx
    return np.max(conv) * dx
