import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Simple evaluation of the overlap integral
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    C5_vals = []
    for k in range(1-n, n):
        # Compute correlation at shift k
        if k >= 0:
            corr = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            corr = np.sum(h[-k:] * (1 - h[:-k]))
        C5_vals.append(corr * dx)
    return max(C5_vals)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a balanced construction
        n_points = 42
        h = np.ones(n_points) * 0.5
        # Alternative: vary based on parity
        for i in range(n_points):
            h[i] = 0.5 if i % 2 == 0 else 0.0
        # Adjust to meet constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    # Frank-Wolfe style optimization with projections and periodic perturbations
    max_iter = 10000
    learning_rate = 0.05
    perturbation_freq = 100
    perturbation_strength = 0.03
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current gradient estimate via finite difference
        current_C5 = evaluate_erdos_solution(h)
        
        # Compute a small perturbation to estimate gradient
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            # Project to feasible region
            h_eps = np.clip(h_eps, 0, 1)
            h_eps_sum = np.sum(h_eps)
            h_eps = h_eps - h_eps_sum + n_points / 2
            C5_eps = evaluate_erdos_solution(h_eps)
            grad[i] = (C5_eps - current_C5) / eps
        
        # Frank-Wolfe step: minimize linear approximation under constraints
        # Linearized objective: <grad, d>
        d = -grad  # Negative gradient direction
        
        # Project d to satisfy: sum(d) = 0, 0 ≤ d[i] ≤ 1
        # We adjust for sum constraint by shifting d to make sum zero
        d_mean = np.mean(d)
        d = d - d_mean
        
        # Perform step
        h_new = h + learning_rate * d
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        h_new = h_new - h_new_sum + n_points / 2  # Re-project to sum constraint
        
        h = h_new
        
        # Evaluate and update best solution
        C5 = evaluate_erdos_solution(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Periodic perturbations to escape local optima
        if it % perturbation_freq == 0 and it > 0:
            perturb = np.random.normal(0, perturbation_strength, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h - h_sum + n_points / 2
        
    return (best_h.tolist(), best_C5, n_points)
