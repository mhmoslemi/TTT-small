import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 50000
    lr = 0.01
    perturb_freq = 100
    perturb_mag = 0.1
    min_lr = 1e-6
    
    # Initialize h

    else:

    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.mean(h)
    
    # Store best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current C5
        correlations = np.correlate(h, 1-h, mode="full")
        C5_candidate = np.max(np.abs(correlations)) * dx
        
        # Update best if improved
        if C5_candidate < best_C5:
            best_h = h.copy()
            best_C5 = C5_candidate
        
        # Compute gradient using finite differences for simplicity
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            # Perturb h[i] and compute gradient approx
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.mean(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
            C5_plus = np.max(np.abs(corr_plus)) * dx
            
            grad[i] = (C5_plus - C5_candidate) / eps
        
        # Gradient descent step
        h = h - lr * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.mean(h)
        
        # Perturb occasionally
        if iter_count % perturb_freq == 0:
            perturb = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.mean(h)
            
        # Adaptive learning rate
        if iter_count % 1000 == 0 and iter_count > 0:
            if lr > min_lr:
                lr *= 0.95
    
    # Final check of best solution
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, n_points)
