import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a starting point
        n_points = 47
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initial rescaling to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vals):
        # Compute cross-correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The maximum value in the cross-correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_gradient(h_vals):
        # Compute cross-correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Find index of maximum
        max_idx = np.argmax(corr)
        # Compute gradient: dC5/dh = derivative of max correlation
        # Approximate using finite differences on the cross-correlation
        n = len(h_vals)
        grad = np.zeros(n)
        # For forward difference, approximate gradient at max point
        for i in range(n):
            # Compute correlation with small perturbation in h[i]
            h_pert = h_vals.copy()
            h_pert[i] += 1e-6
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode="full")
            grad[i] = (corr_pert[max_idx] - corr[max_idx]) / 1e-6
        return grad * dx
    
    def proj_grad_descent(h_vals, max_iter=100):
        nonlocal start_time
        lr = 0.01
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Compute current value and gradient
            c5_val = compute_c5(h_vals)
            try:
                grad = compute_gradient(h_vals)
            except:
                # Fallback to simple gradient
                grad = np.zeros_like(h_vals)
            # Gradient step
            h_vals -= lr * grad
            # Project to [0, 1]
            h_vals = np.clip(h_vals, 0, 1)
            # Rescale to sum to n_points / 2 (dx = 2/n_points)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            
            # Adaptive learning rate
            if i % 100 == 0 and i > 0:
                lr *= 0.5
        return h_vals

    def refine_resolution(h_old, n_new):
        # Interpolate to new size
        old_indices = np.linspace(0, 1, len(h_old))
        new_indices = np.linspace(0, 1, n_new)
        h_new = np.interp(new_indices, old_indices, h_old)
        return h_new * (n_new / 2) / np.sum(h_new)

    # Start
    c5_val = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_val

    iter_count = 0
    resolutions = [47]  # Start at 47
    # Adjust resolution schedule
    while time.time() - start_time < budget_s * 0.95:
        if len(resolutions) <= 1 and n_points >= 20:
            high_res = n_points * 2
            if high_res > 1000:
                break # Limit max points
            resolutions.append(high_res)
            h = refine_resolution(h, high_res)
            n_points = high_res
            dx = 2.0 / n_points

        # Run gradient descent with early change detection
        h = proj_grad_descent(h, 50)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        iter_count += 1
        # If we haven't improved for a while, add more points
        if iter_count % 20 == 0 and len(resolutions) < 3 and n_points >= 20:
            new_n = min(n_points * 2, 1000)
            if new_n <= 1000:
                resolutions.append(new_n)
                h = refine_resolution(h, new_n)
                n_points = new_n
                dx = 2.0 / n_points

    h_final = best_h.copy()
    final_c5 = best_c5
    
    # Final validation step
    final_c5 = compute_c5(h_final)
    
    return (h_final, final_c5, len(h_final))
