import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create triangular initialization
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Triangular initialization
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))

    # Scale to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with momentum
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    # Perturbation frequency
    perturb_freq = 50
    perturb_strength = 0.05
    
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Compute correlations for all shifts
        corr_full = np.correlate(h, 1 - h, mode="full")
        # Extract the relevant part of correlation
        c5 = max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences (simplified)
        grad = np.zeros_like(h)
        
        # Correlation for current h
        corr_current = np.correlate(h, 1 - h, mode="full")
        
        # Compute approximated gradient w.r.t. h
        # For each position i, compute contribution to gradient
        for i in range(n_points):
            # Shift the function and compute gradient contribution
            # Using finite differences approximation:
            # grad[i] ~ (C5(h+eps) - C5(h-eps)) / (2*eps)
            h_plus = h.copy()
            h_minus = h.copy()
            eps = 1e-4
            h_plus[i] += eps
            h_minus[i] -= eps
            # Clip to [0,1]
            h_plus[i] = np.clip(h_plus[i], 0, 1)
            h_minus[i] = np.clip(h_minus[i], 0, 1)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
            corr_minus = np.correlate(h_minus, 1-h_minus, mode="full")
            
            grad[i] = (max(corr_plus) - max(corr_minus)) / (2 * eps)
        
        # Update with momentum and learning rate
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project to feasible set: [0,1] and sum to n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Random perturbation to escape local minima
        if iteration % perturb_freq == 0:
            mask = np.random.rand(n_points) < 0.1
            h[mask] += np.random.normal(0, perturb_strength, np.sum(mask))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation of best solution
    corr_full = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = max(corr_full) * dx
    
    return (best_h.tolist(), final_c5, len(best_h))
