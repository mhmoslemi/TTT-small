import numpy as np
import cvxpy as cp
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create midpoint
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
        
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Trust region parameters
    trust_radius = 0.1
    rho = 0.1
    min_radius = 1e-4
    max_radius = 1.0
    
    # Iteration counter
    iteration = 0
    max_iterations = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Compute correlations for current h
        corr = signal.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        
        # Get C5 value
        current_c5 = np.max(corr)
        
        # If we are better, update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute expected values for linearization
        expected_h = np.zeros(n_points)
        for k in range(n_points):
            expected_h = expected_h + np.roll(h, k) / n_points
        
        # Formulate and solve convex subproblem
        # Linearize: h[i] * (1 - h[(i+k) % n]) ≈ h[i] - h[i] * E[h[(i+k) % n]]
        # But since E[h[(i+k)%n]] is fixed, we can compute it as:
        # Let m[i] = 1 - E[h[i]]: then linearization becomes:
        # h[i] * (1 - h[(i+k) % n]) ≈ h[i] * m[(i+k) % n]
        # Which is a trilinear term. We use a different approach:
        # We consider the dual of benefit and constraints:
        # Let's simplify the convex approximation:
        # Approximate: C5 ≈ max_k sum_i h[i] * (1 - h[i+k]) ≈ sum_i h[i] - sum_k sum_i h[i] * h[i+k]
        # But this isn't exact. We instead go with a trust region model:
        
        # Linearize the overlap function around current point
        # Let's compute the gradient of each cell wrt h[i]
        # But a simpler convex approx conceptually:
        # Because C5 = max_k sum h[i] (1-h[i+k]) , restating it:
        # We approximate each sum_i h[i](1-h[i+k]) by its tangent:
        # Gradient w.r.t h[i]: (1-h[i+k]) for i,k fixed
        
        # Compute gradient as an approximation:
        # grad_i = - sum_k (1-h[i+k]) if k = i - i (mod n) for input
        # This is not straightforward; we'll approximate by capturing local h behavior
        # So, compute gradient matrix [dJ/dh[0], ..., dJ/dh[n-1]]
        # But since C5 = max over k of these sums, it's a max of linear functions
        
        # Simpler idea:
        # At each iteration, solve for h += d, where d solves:
        # minimize  sum_i <g_i, d_i> + 0.5 * ||d||^2  subject to constraints
        # Such that, if we model approximated overlap as:
        # \hat{C5} = sum_a sum_b [h_a * (1-h_{a+b})] ~ linear in h_a -> use simple gradient
        # Better approach: 
        # Let's directly use CVX-based refinement of subproblem:
        
        # Approximate: Overlap = max_k sum_i h[i] * (1-h[i+k]) = max_k sum_i (h[i] - h[i]*h[i+k])
        # Then: dC5/dh[i] = - (sum_k (h[i+k]))  when k is determined by max
        
        m = 1 - h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points):
                grad[i] += -m[(i + k) % n_points]  # This is if we do one component manually

        # But even simpler, within one lag k per max, we do:
        # For current k at which max occurs, derivative of that term (over all i) is:
        # d/dh[i] of h[i] * (1 - h[i+k]) = 1 - h[i+k]  (for fixed lag k)
        # To get all together (that's the max over k) - we could:
        # Approximate as: sum over all k: d/dh[i] of (h[i]* (1 - h[i+k]))   (term by term)
        # For simplicity, we approximate the convergence:
        
        # Empirical model: Separate max_k case that contributes most
        max_k_ind = np.argmax(corr)
        max_k = max_k_ind - (n_points - 1)
        if max_k < 0:
            max_k += n_points  # Adjust for full mode index

        # Compute approximate subproblem gradient for general term h[i] (1 - h[i+max_k]):
        # d/dh[i] ( h[i] * (1 - h[i+max_k]) ) = (1 - h[i+max_k]) 
        # this contribution goes into gradient
        approx_grad = np.zeros(n_points)
        for i in range(n_points):
            k = (i + max_k) % n_points
            approx_grad[i] = 1 - h[k]

        # Solve the convex trust-region-like subproblem:
        # min d^T grad + 1/2 d^T d  subject to:
        #   0 <= h + d <= 1, sum(d) = 0


        # Simplify and solve as simple QCQP:
        # min d^T v + 0.5 * d^T d
        # s.t. 
        #   h + d <= 1  -> d <= (1 - h)
        #   h + d >= 0 -> d >= -h
        #   sum(d) = 0

        # Let's use convex optimization approach
        d = cp.Variable(n_points)
        
        # Objective function
        obj = cp.Minimize(cp.matmul(approx_grad.T, d) + 0.5 * cp.sum_squares(d))
        
        # Constraints
        constraints = [d <= (1 - h),
                       d >= -h,
                       cp.sum(d) == 0]
        
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                step_direction = np.array(d.value)
            else:
                step_direction = -approx_grad
        except:
            step_direction = -approx_grad
            
        # Line search for step size
        alpha = 1.0
        step_size = 1.0
        backtracked = False
        
        while alpha > 1e-8:
            new_h = h + alpha * step_direction
            new_h = np.clip(new_h, 0, 1)
            # project to correct sum
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, n_points)
            if new_c5 < current_c5 - rho * alpha * np.dot(approx_grad, step_direction):
                h = new_h
                backtracked = True
                break
            alpha *= 0.5
            
        if not backtracked:
            h = np.clip(h + step_direction, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Update trust radius
        # A simple rule: trust_radius = trust_radius * 2 if improvement, 0.5 if no improvement
        # Updated by monitoring last few steps
        iteration += 1
        
    # Final test
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the overlap integral for the given h"""
    h = np.array(h_values)
    dx = 2.0 / n_points
    result = np.correlate(h, 1-h, mode="full") * dx
    return np.max(result)
