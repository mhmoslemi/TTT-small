import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the overlap integral for h"""
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum and scale by dx
    max_val = np.max(conv) * dx
    return max_val

def project_h(h, n_points):
    """Project h onto constraints: [0,1] and sum = n_points/2"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def nelder_mead_step(vertices, f_vals, alpha=1.0, gamma=2.0, beta=0.5, rho=0.5):
    """Perform one step of Nelder-Mead simplex optimization"""
    n = len(vertices) - 1
    # Order vertices by function value
    idx = np.argsort(f_vals)
    vertices = [vertices[i] for i in idx]
    f_vals = [f_vals[i] for i in idx]
    
    # Compute centroid of best (n) vertices
    centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    h_r = centroid + alpha * (centroid - vertices[-1])
    h_r = project_h(h_r, len(h_r))
    f_r = evaluate_h(h_r, 2.0 / len(h_r))
    
    if f_vals[0] <= f_r < f_vals[-2]:
        # Accept reflection
        vertices[-1] = h_r
        f_vals[-1] = f_r
    elif f_r < f_vals[0]:
        # Expansion
        h_e = centroid + gamma * (h_r - centroid)
        h_e = project_h(h_e, len(h_e))
        f_e = evaluate_h(h_e, 2.0 / len(h_e))
        if f_e < f_r:
            vertices[-1] = h_e
            f_vals[-1] = f_e
        else:
            vertices[-1] = h_r
            f_vals[-1] = f_r
    else:
        # Contraction
        h_c = centroid + beta * (vertices[-1] - centroid)
        h_c = project_h(h_c, len(h_c))
        f_c = evaluate_h(h_c, 2.0 / len(h_c))
        if f_c < f_vals[-1]:
            vertices[-1] = h_c
            f_vals[-1] = f_c
        else:
            # Shrink
            for i in range(1, len(vertices)):
                vertices[i] = vertices[0] + rho * (vertices[i] - vertices[0])
                vertices[i] = project_h(vertices[i], len(vertices[i]))
                f_vals[i] = evaluate_h(vertices[i], 2.0 / len(vertices[i]))
    
    return vertices, f_vals
