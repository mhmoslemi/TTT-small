import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation
    correlations = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part from full correlation
    # Full length is 2*n - 1; the maximum is at center index n-1
    # The indices corresponds to shifts k = -n+1 to n-1
    # We want max over |k| in [0, n-1]
    max_corr = np.max(correlations[n-1:])  # maximum over k=0 to n-1
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Simple triangular construction
        n_points = 45
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            if x <= 1.0:
                h[i] = x
            else:
                h[i] = 2.0 - x
        h = h / np.sum(h) * (n_points / 2.0)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
        
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        h_plus = h.copy()
        h_minus = h.copy()
        
        # Gradient approximation using central differences
        for i in range(n_points):
            h_plus[i] += eps
            h_minus[i] -= eps
            
            c_plus = evaluate_erdos_solution(h_plus)
            c_minus = evaluate_erdos_solution(h_minus)
            
            grad[i] = (c_plus - c_minus) / (2.0 * eps)
            
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Compute step using trust region
        step = -grad * trust_radius / np.linalg.norm(grad + 1e-12)
        h_new = h + step
        
        # Project h_new back to feasible region
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust to maintain sum constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Check if step improved result
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < current_c5:
            h = h_new.copy()
            trust_radius = min(trust_radius * 1.2, 0.2)  # Increase trust radius
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            trust_radius = max(trust_radius * 0.5, 1e-6)  # Reduce trust radius
            
            if no_improvement_count > max_no_improvement:
                # Perturb
                noise = np.random.normal(0, 0.05, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < current_c5:
                    h = h_new.copy()
                no_improvement_count = 0
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
