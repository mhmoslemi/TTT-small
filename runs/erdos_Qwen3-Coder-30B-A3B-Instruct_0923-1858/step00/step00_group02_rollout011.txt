def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

        # Compute full convolution with (1-h)
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Take only the relevant part for maximum
        max_k = len(conv) - n_points + 1
        return np.max(conv[:max_k]) * dx

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current overlap integral
        current_C5 = compute_C5(h)
        
        # Compute gradients
        grad = np.zeros_like(h)
        h_padded = np.concatenate([h, h[:n_points-1]])
        
        # Compute gradient for each x_i
        for i in range(n_points):
            grad_i = 0
            # Contributions from all k where we compute the overlap involving h[i]
            for k in range(n_points):
                # The derivative of h[i] * (1-h[(i+k)%n]) w.r.t. h[i]
                # This contributes -h[(i+k)%n] to the gradient
                if k == 0:
                    grad_i -= 1.0  # derivative of h[i]*1
                else:
                    grad_i -= h[(i+k)%n_points]
            grad[i] = grad_i * dx

        # Second-order approximation with identity - use trust region step
        # This approximates the gradient direction, but better would be to solve QCQP
        
        # Solve trust region subproblem with quadratic model
        # min_{d} g^T d + 0.5 d^T d
        # s.t. ||d|| <= trust_radius
        d = -grad
        d_norm = np.linalg.norm(d)
        if d_norm > trust_radius:
            d *= trust_radius / d_norm
            
        # Line search to find step size
        alpha = 1.0
        max_alpha = 1.0
        step_size = 0.5
        
        best_alpha = 0.0

            h_new = h + alpha * d
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet exact sum constraint
            h_new *= (n_points / 2) / np.sum(h_new)
            new_C5 = compute_C5(h_new)
            
            if new_C5 <= best_c5_val:
                best_c5_val = new_C5

                if new_C5 < best_C5:

        iter_count += 1
        
        # Early termination for budget
        if time.time() - start_time > budget_s - 1:
            break
            
    return (best_h, best_C5, n_points)
