import numpy as np
import time
from scipy import correlate

def evaluate_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_h(h, n_points):
    """Project h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s > 0:
        h = h * (n_points / 2) / s
    else:
        h = np.full(n_points, 1.0 / 2)
    return h

def crossover(parent1, parent2, alpha=0.5):
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + noise * mask
    return project_h(h, len(h))

def tournament_selection(population, fitnesses, k=5):
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.ones(n_points) * 0.5

    pop_size = 30
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_size = int(0.1 * pop_size)
    replace_size = int(0.2 * pop_size)
    
    # Initialize population
    population = []
    population.append(initial_h)
    for _ in range(pop_size - 1):
        h = np.random.random(n_points)
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement = 0
    generation = 0

    while time.time() - start_time < budget_s - 1:
        if no_improvement > 200:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        no_improvement += 1
        if len(population) > 0 and fitnesses[0] < best_c5:
            no_improvement = 0
            
        # Create new population
        new_population = population[:elite_size]  # Keep elite
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_prob:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mutation_rate)
            
            new_population.append(child)
            
        # Replace worst
        population = new_population
        
        generation += 1
        if generation % 50 == 0:
            pass  # Intermediate checkpoints

    if best_h is None:
        # Fallback to initial
        best_h = initial_h
        best_c5 = evaluate_c5(best_h, 2.0 / n_points)

    return (best_h.tolist(), best_c5, n_points)
