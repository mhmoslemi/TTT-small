from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise use a triangular waveform
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 45
        # Triangular waveform initialization
        h = np.array([min(i, 2*n_points - i - 2) for i in range(2*n_points - 1)])
        h = h[:n_points] / np.sum(h) * n_points / 2
    
    # Ensure h is valid
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Optimization parameters
    learning_rate = 0.01
    decay_rate = 0.95
    perturbation_rate = 0.05
    perturbation_interval = 50
    gradient_update_interval = 100
    
    def evaluate_c5(h_vec):
        # Compute full correlation of h with (1-h)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Consider the maximum value over all shifts (* dx)
        return np.max(corr) * (2.0 / len(h_vec))
    
    def compute_gradient(h_vec):
        # Approximate gradient using finite differences
        # Forward difference approximation
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_c5 = evaluate_c5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    it = 0
    while time.time() - start_time < budget_s * 0.99:
        # Compute C5 and check for improvement
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Periodic learning rate decay
        if it % gradient_update_interval == 0:
            learning_rate *= decay_rate
            
        # Update step
        grad = compute_gradient(h)
        h = h - learning_rate * grad
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Random perturbation occasionally
        if it % perturbation_interval == 0 and not (it == 0):
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, size=len(h))
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        it += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_c5(h):
    # Compute full correlation of h with (1-h)
    corr = np.correlate(h, 1 - np.array(h), mode="full")
    # Consider the maximum value over all shifts (* dx)
    return np.max(corr) * (2.0 / len(h))
