        h_vals = np.array(h_vals)
        corr_full = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr_full)
        return max_corr * dx
    
    def project_h(h_vals):
        # Ensure h is in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to maintain sum(h)*dx = 1
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals = h_vals * (n_points / 2) / current_sum
        return h_vals
    
    def compute_gradient(h_vals):
        # Simplified gradient approximation using finite differences
        # This is a placeholder; more detailed derivation is complex
        h_vals = np.array(h_vals)

        # For true gradient: we would compute derivatives of the max correlation

        grad = np.zeros_like(h_vals)

        for i in range(len(h_vals)):
