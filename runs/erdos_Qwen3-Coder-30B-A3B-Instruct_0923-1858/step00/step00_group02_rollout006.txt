def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdos constant for given h."""
    # Compute the overlap integral
    correlation = np.correlate(h, 1-h, mode="full")
    # The maximum value is the maximum of all correlations (excluding the zero lag which is not relevant)
    max_correlation = np.max(correlation)
    # Normalize and return
    return max_correlation * dx

    def compute_gradient(h, eps=1e-5):

            h_plus[i] += eps
