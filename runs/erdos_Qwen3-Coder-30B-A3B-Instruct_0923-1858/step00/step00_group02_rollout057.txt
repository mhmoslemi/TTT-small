import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Create a triangular waveform as fallback
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h0 = np.maximum(0, 1 - np.abs(x - 1))
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Ensure feasibility
    h0 = np.clip(h0, 0, 1)
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    patience = 500
    
    def evaluate_c5(h, dx):
        # Compute all correlations at once
        corr = np.correlate(h, 1-h, mode="full")
        # The maximum is over the valid range, which is from -n+1 to n-1
        # but we only care about 0 to n-1 (positive shifts)
        max_k = len(corr) - n_points + 1
        return np.max(corr[n_points-1:n_points-1+max_k]) * dx
    
    def project_h(h):
        # Clamp to [0, 1] and rescale to sum to n_points/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def finite_diff_grad(h, epsilon=1e-4):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = project_h(h_plus)
            h_minus = project_h(h_minus)
            grad[i] = (evaluate_c5(h_plus, dx) - evaluate_c5(h_minus, dx)) / (2 * epsilon)
        return grad
    
    h_current = h0.copy()
    trust_radius = 0.1
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = finite_diff_grad(h_current)
        
        # Compute step using trust region
        step = -grad * trust_radius / (np.linalg.norm(grad) + 1e-8)
        
        # Ensure step respects bounds
        h_proposed = h_current + step
        h_proposed = project_h(h_proposed)
        
        # Evaluate proposed step
        c5_proposed = evaluate_c5(h_proposed, dx)
        
        if c5_proposed < best_c5:
            # Accept step
            h_current = h_proposed.copy()
            best_c5 = c5_proposed
            best_h = h_current.copy()
            trust_radius *= 1.2  # Increase trust radius
            last_improvement = i
        else:
            # Reject step
            trust_radius *= 0.5  # Decrease trust radius
            
        # Periodic perturbation
        if i % 100 == 0 and i > last_improvement + 50:
            # Add small random perturbation
            noise = np.random.normal(0, 0.05, n_points)
            h_perturbed = h_current + noise
            h_perturbed = project_h(h_perturbed)
            c5_perturbed = evaluate_c5(h_perturbed, dx)
            if c5_perturbed < best_c5:
                h_current = h_perturbed.copy()
                best_c5 = c5_perturbed
                best_h = h_current.copy()
                last_improvement = i
        
        # Early stopping if no improvement
        if i - last_improvement > patience:
            break
            
        iter_count += 1
    
    # Final computation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
