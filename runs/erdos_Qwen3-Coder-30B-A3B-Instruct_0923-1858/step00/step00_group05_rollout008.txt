import numpy as np
from scipy import signal
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h and (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # The maximum value is at the center
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n/2"""
    # Clip values
    h = np.clip(h, 0, 1)
    # Rescale to enforce sum(h) = n/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

def gradient_step(h, lr, n):
    """Perform one gradient step to reduce C5"""
    # Compute current C5
    dx = 2.0 / n
    corr = signal.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_corr = corr[max_idx]
    
    # Gradient can be approximated by finite differences of cross-correlation
    # We compute gradient of max_corr w.r.t. h using chain rule
    # Since max_corr = sum(h[i] * (1 - h[i+k])) for some k
    # This is complex; we'll do a simplified gradient estimation
    # Simple method: finite difference for objective
    epsilon = 1e-6
    
    grad = np.zeros_like(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_h(h_plus, n)  # This project here is just for numerical stability
        corr_plus = signal.correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (np.max(corr_plus) - max_corr) / epsilon
    
    # Update h
    h_new = h - lr * grad
    h_new = project_h(h_new, n)
    return h_new

def optimize_h(n_points, lr=1e-2, max_iter=1000, tol=1e-5):
    """Optimize h to minimize C5"""
    # Initialize with a good starting point or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * n_points / 2
    else:
        h = np.ones(n_points) * 0.5
    
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    no_improvement_count = 0
    
    for i in range(max_iter):
        h = gradient_step(h, lr, n_points)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if abs(prev_c5 - c5) < tol:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        prev_c5 = c5
        if no_improvement_count >= 100:
            break
        # Reduce learning rate if needed
        if i > 0 and i % 50 == 0:
            lr *= 0.9
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    n_low = 50
    max_iter_low = 200
    h_low, c5_low = optimize_h(n_low, lr=1e-2, max_iter=max_iter_low)
    print(f"Low-res C5: {c5_low}")
    
    # Now upsample to better resolution
    n_high = 70
    # Linear interpolation
    h_upsampled = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n_low), h_low)

    # High resolution optimization
    h_high, c5_high = optimize_h(n_high, lr=1e-3, max_iter=500)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_high)
