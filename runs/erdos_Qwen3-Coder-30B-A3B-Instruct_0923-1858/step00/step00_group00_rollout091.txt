import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    def compute_c5(h_vals):
        # Evaluate C5 = max_k integrate h(x)(1 - h(x+k)) dx
        # Using correlation computation
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initialize
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_count = 0
    max_no_improve = 500
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb current solution
        h_new = current_h + np.random.normal(0, perturb_scale, n_points)
        
        # Project back to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet sum constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = h_new
            current_c5 = new_c5
            
            # Update best
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cooling schedule
        T *= alpha
        
        # Adapt perturbation scale
        if i % 100 == 0 and i > 0:
            perturb_scale *= 0.99
            
        # Terminate if no improvement for a while
        if no_improve_count > max_no_improve:
            break
    
    # Final verification
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
