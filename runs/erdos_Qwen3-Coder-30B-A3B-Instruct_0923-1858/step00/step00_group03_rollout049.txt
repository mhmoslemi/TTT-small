import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to meet sum constraint
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    outer_iter = 0
    while time.time() - start_time < budget_s - 1:
        outer_iter += 1
        # Evaluate all overlaps
        conv = signal.correlate(h, 1 - h, mode='full')
        c5_vals = conv * dx
        k_max = np.argmax(c5_vals)
        max_c5 = c5_vals[k_max]
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            
        if max_c5 < 0.3807:
            break
            
        # Find which indices contribute most to maximum overlap
        # This is a heuristic, just look at the positions where overlap is significant
        if k_max < len(h):
            k = k_max
        else:
            k = k_max - len(h) + 1
            
        # Find the most influential position pair (i, i+k) for this overlap
        best_gain = -1
        best_i = None
        
        # Look at relevant pairs
        for i in range(len(h) - k):
            if h[i] > 0 and h[i + k] < 1:
                gain = h[i] * (1 - h[i + k]) - (h[i] - 0.01) * (1 - (h[i + k] + 0.01))
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    
        if best_i is not None:
            # Make small adjustment
            step_size = 0.01
            if h[best_i] > 0 and h[best_i + k] < 1:
                h[best_i] -= step_size
                h[best_i + k] += step_size
                
            # Clip values to [0,1] and maintain sum
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                # redistribute the difference
                h += diff / n_points
                
        # Small random perturbations to avoid stagnation
        if np.random.random() < 0.1:
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.005)
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                h += diff / n_points

        # Every 100 iterations, re-evaluate best
        if outer_iter % 100 == 0:
            # Evaluate current state
            conv = signal.correlate(h, 1 - h, mode='full')
            c5_vals = conv * dx
            current_max_c5 = np.max(c5_vals)
            if current_max_c5 < best_c5:
                best_c5 = current_max_c5
                best_h = h.copy()
                
    # Final evaluation
    conv = signal.correlate(best_h, 1 - best_h, mode='full')
    c5_vals = conv * dx
    final_c5 = np.max(c5_vals)
    
    return (best_h.tolist(), final_c5, len(best_h))
