import numpy as np
import time
import random
from scipy import signal

def evaluate_h(h, dx):
    # Compute the overlap integral for all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take only the valid correlations
    half_len = len(h)
    valid_corr = corr[half_len - 1:]
    return np.max(valid_corr) * dx

def binary_convert(h, n_points):
    # Convert to binary, ensuring sum equals n_points / 2
    h_bin = np.round(h).astype(int)
    current_sum = np.sum(h_bin)
    target_sum = n_points // 2
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones_indices = np.where(h_bin == 1)[0]
        to_flip = current_sum - target_sum
        flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
        h_bin[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(h_bin == 0)[0]
        to_flip = target_sum - current_sum
        flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
        h_bin[flip_indices] = 1
    return h_bin

def compute_c5(h, dx):
    # Align with the evaluation function
    return evaluate_h(h, dx)

    h_candidate = binary_convert(h_candidate, n_points)

        h_candidate = binary_convert(h_candidate, n_points)

        i = random.choice(ones)

        h_new = h_candidate.copy()

    best_h = binary_convert(best_h, n_points)
