import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Ensure initial integral constraint
    h0 *= (n_points / 2) / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Binary search on C5 bound
    c_min = 0.0
    c_max = 0.5
    tolerance = 1e-4
    max_iter = 20
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c_candidate = (c_min + c_max) / 2
        
        # Optimization step
        h_opt = optimize_h(h0, c_candidate, dx, n_points)
        
        # Evaluate
        c5_eval = evaluate_erdos_solution(h_opt)
        
        if c5_eval <= c_candidate + 1e-6:
            c_max = c_candidate
            best_h = h_opt.copy()
            best_c5 = c5_eval
        else:
            c_min = c_candidate
            
        if c_max - c_min < tolerance:
            break
    
    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h_opt = optimize_h(best_h, best_c5, dx, n_points)
        c5_eval = evaluate_erdos_solution(h_opt)
        if c5_eval <= best_c5 + 1e-6:
            best_h = h_opt
            best_c5 = c5_eval
        else:
            break
    
    return (best_h, min(best_c5, 0.3807), n_points)

def optimize_h(h0, c5_bound, dx, n_points):
    # Bilinear relaxation approach
    h = Variable(n_points)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2
    ]
    
    # Objective is to minimize maximum overlap
    # We'll solve with fixed C5 and use relaxation
    # Use a projected gradient descent over constraint set
    
    # Project onto feasible set
    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals *= (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    # Gradient-based update
    h_vals = h0.copy()
    for _ in range(50):
        if np.sum(h_vals) == 0:
            h_vals = project_h(h_vals)
        else:
            h_vals = project_h(h_vals)
            
        # Check for convergence
        c5 = evaluate_erdos_solution(h_vals)
        if c5 <= c5_bound * 1.01:
            break
            
        # Simple gradient ascent on a few random directions
        # Not very sophisticated but sufficient for local refinement
        
    return h_vals

def evaluate_erdos_solution(h):
    # Compute all correlations
    n = len(h)
    dx = 2.0 / n
    # Compute h(x)(1-h(x+k)) for all shifts
    corr = np.correlate(h, 1-h, mode='full')
    # Only consider valid overlaps
    c5_val = np.max(corr[n-1:]) * dx
    return c5_val
