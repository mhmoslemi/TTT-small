import numpy as np
from scipy import signal
import time

def evaluate_c5(h):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value of the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to make sum = n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback: construct a reasonable starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    step_size = 0.01
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Find dominant lag k* by computing overlap for all lags
        conv = signal.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(conv)
        # Adjust k_star for proper indexing (center is at n-1)
        lag = k_star - (n_points - 1)
        
        # Compute gradient at dominant k*
        # grad_i = d/dh_i [int h(x)(1-h(x+k*)) dx] = (1 - h(x+k*) - h(x)) |_{x=i}
        # But simplify: only active when i+k* is within bounds
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = lag
            if 0 <= j + k < n_points:
                # Add gradient contribution from this pair
                grad[j] += 1 - h[j + k]  # We're not using 1-h since we want to minimize
            if 0 <= j - k < n_points:
                # Note: h(x+k) becomes h(x-k) in flipped correlation sense
                grad[j] += -h[j - k]
                
        # Update using gradient descent
        h_new = h - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new.copy()
        # Occasionally reduce step size
        if i % 50 == 0:
            step_size *= 0.99
            
    return (best_h.tolist(), best_c5, n_points)
