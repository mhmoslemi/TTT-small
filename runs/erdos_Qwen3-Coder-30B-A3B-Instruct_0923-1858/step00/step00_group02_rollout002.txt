        h -= lr * grad

        # Decay learning rate
        if i % 100 == 0:
            lr *= decay_rate
            
    return (best_h, best_c5, n_points)

def evaluate_objective(h, n_points):
    dx = 2.0 / n_points
    # correlation with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # take max over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, n_points):

    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')

    max_shift_idx = np.argmax(corr)

        h_plus[i] += eps
        h_minus[i] -= eps

        val_minus = evaluate_objective(h_minus, n_points)
        
        grad[i] = (val_plus - val_minus) / (2 * eps)
