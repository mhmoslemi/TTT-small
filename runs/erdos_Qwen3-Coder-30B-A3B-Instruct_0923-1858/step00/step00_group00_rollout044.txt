import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Fallback: create symmetric h with sum = n_points/2
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = h[-1] = 0.25  # make it more symmetric and valid
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Annealing parameters
    temp = 1.0
    alpha = 0.995
    perturb_decay = 0.99
    max_iter = 10000
    patience = 200
    no_improve_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        h_new = h + np.random.normal(0, temp, n_points)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_h
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            if np.random.random() < np.exp(-(c5_new - best_c5) / temp):
                h = h_new
        
        # Cooling
        temp *= alpha
        if i % 100 == 0:
            temp *= perturb_decay
            
        # Early stopping
        if no_improve_count > patience:
            break
    
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute correlation with (1-h) and find maximum
    # The correlation will be of length 2*n_points - 1
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
