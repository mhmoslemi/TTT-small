import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Initialize with symmetric construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        # Make it reasonably smooth
        x = np.linspace(0, 2, n_points)
        h0 = np.clip(0.5 + 0.3 * np.sin(np.pi * x), 0, 1)
        
    dx = 2.0 / n_points
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Binary search for optimal C5
    C_min = 0.0
    C_max = 0.5
    best_C = C_max
    best_h = h0.copy()
    
    for iteration in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        C_candidate = (C_min + C_max) / 2.0
        
        # Formulate feasibility problem
        h_vars = Variable(n_points)
        
        # Convex relaxation
        # For each k, we want sum(h[i] * (1 - h[i+k]) * dx) <= C_candidate
        # This is equivalent to: sum(h[i] - h[i]*h[i+k]) * dx <= C_candidate
        
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2
        ]
        
        # Add the overlap constraints for each k
        # Overlap integral: sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) * dx 
        # For k=0, this becomes h[i]*(1-h[i]) = h[i] - h[i]^2, so we want sum(h[i] - h[i]^2) <= C
    
        # Relax the bilinear terms with McCormick envelope
        # For h[i] * h[i+k], introduce variables y[i,k] with constraints:
        # y[i,k] >= h[i] + h[i+k] - 1
        # y[i,k] >= 0
        # y[i,k] <= h[i]
        # y[i,k] <= h[i+k]
        
        # Since we can't easily set up a convex model directly with convolution, 
        # we will use a local heuristic approach:
        # We optimize using scipy.optimize
        from scipy.optimize import minimize
        
        def compute_overlap(h_val):
            # h_val should be length n_points
            h_val = np.array(h_val)
            h = h_val * dx
            # compute convolution h * (1-h)
            full_conv = np.correlate(h, 1 - h, mode='full')
            # Extract valid indices for overlap 
            k_start = n_points - 1
            overlap_values = full_conv[k_start:k_start + n_points]
            return np.max(overlap_values)
            
        def obj_func(h_val):
            return compute_overlap(h_val)
            
        def constraint_func(h_val):
            return np.sum(h_val) - n_points / 2
            
        # Construct bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        res = minimize(
            obj_func, 
            h0.copy(), 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint_func},
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-6},
            tol=1e-6
        )

        if res.success:
            h_opt = res.x
            C_val = compute_overlap(h_opt)
            if C_val <= C_candidate:
                C_max = C_candidate
                best_C = C_val
                best_h = h_opt.copy()
            else:
                C_min = C_candidate
        else:
            # If no improvement, say it's infeasible
            C_min = C_candidate
            
    # Ensure final solution is exactly feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h.tolist(), float(best_C), int(n_points))
