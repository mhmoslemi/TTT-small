import numpy as np
import cvxpy as cp
import time
from scipy import signal

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The valid range is from -n+1 to n-1, but we only care about the actual shift ranges
    # For k in 0 to n-1, only compute the valid ones
    # Actual shifts are from -n+1 to n-1, but our domain is [0,2] and integration is over full range
    # Looking at the full result, valid overlaps are where i >= n-1 because shift k=i-(n-1) 
    # When we have h[0:n] and 1-h[0:n], the correlation gives shifts from -n+1 to n-1
    # The valid positive shifts from 0 to 2 corresponds to indices
    # We take k from 0 to n-1 inclusive in the middle of the full conolution.
    # The i in range(len(overlaps)) such that overlap at shift k = i - (n-1) corresponds to index k = i - (n-1) 
    # For k = 0 to n-1, index i = k + (n-1)
    # All overlaps from 0 shifts to n-1 shifts are then i from n-1 to 2*(n-1)
    valid_overlaps = overlaps[n-1:2*(n-1)+1]
    return max(valid_overlaps) * dx

        h0 += np.random.normal(0, 0.01, n_points)
        h0 = np.clip(h0, 0, 1)
        # Scale to sum to n_points / 2
        h0 *= (n_points / 2) / np.sum(h0)
        
    # Set up discretization
    dx = 2.0 / n_points
    h_prev = h0.copy()
    
    # Keep track of best solution and time remaining
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Solve repeatedly using a cutting-plane approach
    full_time_limit = time.time() + budget_s
    max_iter = 100  # Soft limit to prevent indefinite loop
    iter_count = 0
    
    while iter_count < max_iter and time.time() < full_time_limit:
        iter_count += 1
        # At the start of each iteration, compute full h before solving
        h = h_prev.copy()
        # Check how close is the current solution
        c5 = evaluate_erdos_solution(h)
        
        # Early exit if solution is good enough
        if c5 <= 0.3807:
            break
            
        # Find most constraining constraints (potentially)
        # Evaluate for all shifts to get a list of overlaps
        # Use direct calculation of the full correlation for reference 
        shifts = np.arange(0, n_points)
        overlaps_all = []
        for i in range(n_points):
            k = i - (n_points - 1)
            if k >= 0:  # positive shift case 
                overlap = np.sum(h[:n_points-k] * (1 - h[k:]))
            else:  # negative shift
                k = -k
                overlap = np.sum(h[k:] * (1 - h[:n_points-k]))
            overlaps_all.append(overlap)
        overlaps_all = np.array(overlaps_all)

        # Identify most restrictive shifts that contribute to high overlaps    
        # Sort overlaps by their size, consider top 20%
        top_indices = np.argsort(overlaps_all)[::-1][:max(10, n_points // 5)]
        
        # Formulate the problem with additional constraints
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Convolution function to compute overlap for a fixed shift k:
        # overlap[k] = sum(h[i] * (1 - h[i + k])) if k >= 0 
        #             = sum(h[i + k] * (1 - h[i])) if k < 0
        
        # We'll recompute for accuracy on these specific constraints
        
        # Objective is to minimize C5  
        objective = cp.Minimize(C5)
        
        constraints = [0 <= h_vars, h_vars <= 1, cp.sum(h_vars) == n_points / 2, C5 >= 0]
        
        # Add constraint upper bounds
        for idx in top_indices:
            k = idx - (n_points - 1)  # k = shift
            if k >= 0:  # Positive shift
                # indices from 0 to n_points - k-1 for first part, k to n_points-1 for second
                # sum h[i] * (1 - h[i+s])  for i=0 to n_points-k-1, s=k
                overlap_expr = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:  # Negative shift
                k = -k
                # sum h[i+k] * (1 - h[i]) for i=0 to n_points-k-1
                overlap_expr = cp.sum(cp.multiply(h_vars[k:], 1 - h_vars[:n_points-k]))
            
            constraints.append(overlap_expr <= C5 / dx)
            
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6, reltol=1e-6)
        except Exception:
            break
        if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            h_sol = h_vars.value
            c5_sol = C5.value
            if c5_sol < best_c5:
                best_h = h_sol.copy()
                best_c5 = c5_sol
            h_prev = h_sol.copy()
        else:
            break

    return (best_h.tolist(), best_c5, n_points)
