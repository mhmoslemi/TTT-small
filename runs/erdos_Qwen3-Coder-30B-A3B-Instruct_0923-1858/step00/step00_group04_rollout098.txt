import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * target_sum / current_sum

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with given initial h"""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
    else:
        population.append(np.full(n_points, 0.5))
    
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = project_to_constraint(h, n_points / 2)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection"""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def blend_crossover(p1, p2, alpha=0.5):
    """Blend crossover between two parents"""
    child = alpha * p1 + (1 - alpha) * p2
    return np.clip(child, 0, 1)

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    """Mutate h with Gaussian noise"""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + noise * mask
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    pop_size = 30
    max_generations = 1000
    elite_rate = 0.1
    mutation_rate = 0.1
    
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_to_constraint(initial_h, n_points / 2)
        
    population = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Elitism
        elite_size = max(1, int(elite_rate * pop_size))
        sorted_indices = np.argsort(fitnesses)
        new_population = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        needed = pop_size - elite_size
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = blend_crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            child = project_to_constraint(child, n_points / 2)
            new_population.append(child)
            
        population = new_population
            
    # Final validation
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h, final_c5, n_points)
    else:
        # Fallback if nothing found
        return (np.full(n_points, 0.5), 1.0, n_points)
