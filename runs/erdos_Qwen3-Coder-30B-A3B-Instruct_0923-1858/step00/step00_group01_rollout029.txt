import numpy as np
from scipy import optimize
import time

def clip_and_rescale(h, dx):
    h = np.clip(h, 0, 1)
    scale = 1.0 / (np.sum(h) * dx)
    h *= scale
    return h

def evaluate_C5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def gradient_step(h, dx, alpha=0.01):
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr)
    
    # Compute gradient of correlation w.r.t h
    # For the max correlation at k_max, we need to backpropagate
    # This is a rough approximation using finite differences
    nh = len(h)
    grad = np.zeros_like(h)
    
    # Approximate gradient by finite differences
    eps = 1e-6
    for i in range(nh):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = np.correlate(h, 1 - h, mode="full")
        grad[i] = (corr_plus[k_max] - corr_minus[k_max]) / eps
    
    return grad

    alpha = 0.01
