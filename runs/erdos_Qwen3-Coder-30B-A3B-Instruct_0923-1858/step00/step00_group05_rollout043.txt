import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_overlap(h):
    """Compute the maximum overlap integral C5 for given h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation h ⋆ (1 - h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value in the correlation is the C5 bound
    return np.max(corr) * dx

def fitness(h):
    """Fitness function to minimize (negative of max overlap)."""
    return -evaluate_overlap(h)

def mutate_individual(h, mutation_rate=0.1):
    """Mutate a single individual with random perturbation."""
    h_new = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_new[i] = random.uniform(0, 1)
    # Ensure sum constraint after mutation
    scale = (len(h) / 2) / np.sum(h_new)
    h_new *= scale
    return h_new

def crossover(parent1, parent2):
    """Perform two-point crossover between two parents."""
    n = len(parent1)
    point1 = random.randint(1, n - 2)
    point2 = random.randint(point1 + 1, n - 1)
    child1 = np.concatenate([parent1[:point1], parent2[point1:point2], parent1[point2:]])
    child2 = np.concatenate([parent2[:point1], parent1[point1:point2], parent2[point2:]])
    # Ensure sum constraint
    scale1 = (n / 2) / np.sum(child1)
    scale2 = (n / 2) / np.sum(child2)
    child1 *= scale1
    child2 *= scale2
    return child1, child2

def local_optimize(h, max_iter=50):
    """Refine h using scipy's minimize with bounds and constraint."""
    n = len(h)
    dx = 2.0 / n
    bounds = [(0, 1) for _ in range(n)]
    def obj(h_vec):
        return evaluate_overlap(h_vec)
    
    def constraint(h_vec):
        return np.sum(h_vec) - n / 2

    cons = {"type": "eq", "fun": constraint}
    result = minimize(obj, h, method="SLSQP", bounds=bounds, constraints=cons, 
                      options={"maxiter": max_iter, "ftol": 1e-6})
    return result.x if result.success else h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)

    start_time = time.time()
    # Use initial construction if available, otherwise create a uniform one

    else:

    h = h / np.sum(h) * (len(h) / 2)
    
    # GA parameters
    pop_size = 100
    num_generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 10
    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h)
    
    # Start GA
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 5:
            break
        # Generate offspring
        population = []
        for _ in range(pop_size):
            population.append(mutate_individual(h, mutation_rate))
        
        # Evaluate fitness
        fitness_scores = [fitness(individual) for individual in population]
        # Selection with tournament
        selected = []
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness_scores)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Crossover and create new population
        new_pop = []
        for i in range(0, pop_size - 1, 2):
            if random.random() < crossover_rate:
                child1, child2 = crossover(selected[i], selected[i+1])
                new_pop.extend([child1, child2])
            else:
                new_pop.extend([selected[i], selected[i+1]])
        population = new_pop
        
        # Local optimization for top candidates
        fitness_scores = [fitness(individual) for individual in population]
        sorted_indices = np.argsort(fitness_scores)[:5]
        for idx in sorted_indices:
            optimized = local_optimize(population[idx])
            fitness_opt = fitness(optimized)
            if fitness_opt < fitness_scores[idx]:
                population[idx] = optimized
                fitness_scores[idx] = fitness_opt
        
        # Update best found
        current_best_score = min(fitness_scores)
        if current_best_score < fitness(best_h):
            best_idx = np.argmin(fitness_scores)
            best_h = population[best_idx].copy()
            best_c5 = evaluate_overlap(best_h)
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    return list(best_h), final_c5, len(best_h)
