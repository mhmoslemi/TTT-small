import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values"""
    h = np.array(h_values)
    n = len(h)
    # Compute correlation with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr / n

def initialize_h(n_points):
    """Initialize h with given number of points"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    return h

def solve_convex_subproblem(h_prev, n_points):
    """Solve convex approximated subproblem"""
    h_prev = np.array(h_prev)
    # Compute expected values
    expected_h = np.zeros(n_points)
    for i in range(n_points):
        expected_h[i] = h_prev[(i + 1) % n_points]
    # Formulate the quadratic program
    h = cp.Variable(n_points)
    # Constraint: sum = n_points / 2
    constraints = [cp.sum(h) == n_points / 2]
    # Bounds
    constraints += [h >= 0, h <= 1]
    # Approximate objective using linearization around h_prev
    # Objective: sum over all lags of h[i] * (1 - expected_h[i])
    # So we minimize - sum(h[i] * (1 - expected_h[i])) 
    # This is equivalent to minimizing -sum(h[i] - h[i] * expected_h[i])
    # = -sum(h[i]) + sum(h[i] * expected_h[i])
    # = -(n_points/2) + sum(h[i] * expected_h[i])
    # So we minimize sum(h[i] * expected_h[i]) 
    obj = cp.Minimize(cp.sum(cp.multiply(h, expected_h)))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        else:
            return h_prev  # fallback
    except:
        return h_prev  # fallback

def backtracking_line_search(h_prev, d, n_points, alpha=1.0, beta=0.5, max_iter=20):
    """Perform backtracking line search"""
    h_prev = np.array(h_prev)
    d = np.array(d)
    for _ in range(max_iter):
        h_new = h_prev + alpha * d
        h_new = np.clip(h_new, 0, 1)
        # Rescale to ensure correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        if evaluate_erdos_solution(h_new) < evaluate_erdos_solution(h_prev):
            return h_new, alpha
        alpha *= beta
    return h_prev, alpha

def run(seed=42, budget_s=600, **kwargs):
    """Optimize h to minimize C5"""
    np.random.seed(seed)
    n_points = 45
    start_time = time.time()
    
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Precompute some initial data
    if n_points < 1000:
        update_freq = 50
    else:
        update_freq = 200
    
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        if iter_count % update_freq == 0:
            # Compute the convex approximation
            h_new = solve_convex_subproblem(h, n_points)
            # Compute descent direction
            d = h_new - h
            # Perform line search to find step size
            new_h, _ = backtracking_line_search(h, d, n_points)
            h = new_h
        else:
            # Perturb with small random step
            d = np.random.random(n_points) - 0.5
            d = d * 0.01
            h = h + d
            h = np.clip(h, 0, 1)
            # Rescale to sum = n_points / 2
            h = h * (n_points / 2) / np.sum(h)
        
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early stopping if improvement is minimal
        if iter_count > 100 and abs(c5 - best_c5) < 1e-6:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
