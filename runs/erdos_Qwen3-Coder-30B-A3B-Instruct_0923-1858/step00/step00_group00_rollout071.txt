def evaluate_C5(h_values, dx):

    # Compute correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # Find maximum over all possible shifts
    max_corr = np.max(corr)
    return max_corr * dx
