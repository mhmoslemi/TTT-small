import numpy as np
import time
from scipy import correlate

def evaluate_h(h_vals, dx):
    """Evaluate C5 for given h_values."""
    n_points = len(h_vals)
    corr = correlate(h_vals, 1 - h_vals, mode='full')
    max_corr = np.max(np.abs(corr))
    return max_corr * dx

def project_to_feasible(h_vals, n_points):
    """Project h_vals to satisfy constraints."""
    # Clip to [0, 1]
    h_proj = np.clip(h_vals, 0, 1)
    # Adjust sum to be exactly n_points / 2
    current_sum = np.sum(h_proj)
    if current_sum > 0:
        h_proj = h_proj * (n_points / 2) / current_sum
    return h_proj

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parent solutions."""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h_vals, mutation_rate=0.1, noise_std=0.01):
    """Apply mutation to a solution."""
    mask = np.random.random(len(h_vals)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h_vals))
    h_mut = h_vals + noise * mask
    return h_mut

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

    best_h = None
