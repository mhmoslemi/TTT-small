import numpy as np
from scipy import signal
import time

def evaluate_overlap(h):
    """Evaluate the maximum overlap for given h."""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h
