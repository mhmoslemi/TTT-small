import numpy as np
from scipy import optimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create symmetric initialization centered at 1.0
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Add a peak in the middle
        mid = n_points // 2
        h[mid-2:mid+3] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vals):
        # Compute the overlap integral using convolution
        conv_result = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The maximum value corresponds to the maximum overlap
        max_overlap = np.max(conv_result)
        return max_overlap * dx

    def project_h(h_vals):
        # Ensure h is in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to ensure sum = target_sum
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            new_h = h_vals * (target_sum / current_sum)
        else:
            new_h = np.ones_like(h_vals) * (target_sum / n_points)
        return new_h

    def perturb_h(h_vals, temperature):
        # Add noise with current temperature
        noise = np.random.normal(0, temperature, len(h_vals))
        perturbed = h_vals + noise
        return perturbed

    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    temperature = 0.1
    alpha = 0.99
    patience = 0
    max_patience = 1000

    while time.time() - start_time < budget_s:
        # Try a perturbation
        h_candidate = perturb_h(h, temperature)
        h_candidate = project_h(h_candidate)
        new_C5 = compute_C5(h_candidate)
        
        # Accept or reject
        delta_C5 = new_C5 - current_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / temperature):
            h = h_candidate
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Adjust temperature
        temperature *= alpha
        
        # Early termination if no improvement for a long time
        if patience > max_patience:
            break
            
    return (best_h.tolist(), best_C5, n_points)
