import numpy as np
from time import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 given a step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value and scale by dx
    c5 = np.max(corr) * dx
    return c5

    # Normalize to sum to 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Simple optimization using gradient-like adjustment
    max_iter = 5000
    for i in range(max_iter):
        if time() - start_time > budget_s - 1:
            break

        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        # Adjust based on which entries contribute most to this k_max overlap
        shift = k_max - (n_points - 1)
        # Target for descent: reduce overlap at index k_max
        # Only operate on valid indices
        if shift >= 0:
            indices = np.arange(n_points - shift)
            shift_idxs = indices + shift
        else:
            indices = np.arange(-shift, n_points)
            shift_idxs = indices - shift
        if indices.size == 0:
            indices = np.array([0], dtype=int)
            shift_idxs = np.array([0], dtype=int)
        # Find which pairs cause the most overlap (left and right)
        overlaps = h[indices] * (1 - h[shift_idxs])
        max_idx = np.argmax(overlaps)
        i1 = indices[max_idx]
        i2 = shift_idxs[max_idx]
        
        # Adjust these two points to reduce overlap
        if i1 < n_points and i2 < n_points:
            # Prefer to decrease one of them, decrease the one that contributes more
            val1 = h[i1]
            val2 = h[i2]
            # Propose adjustment to decrease the contribution
            # Since overlap = val1 * (1 - val2), to reduce it:
            # If val2 > 0.5, decreasing it helps
            # If val2 <= 0.5, increasing it helps, but no more than 1
            # We pick a small step size
            step = 0.001
            if val2 > 0.5:
                # Decrease val2 unless it hits 0
                h[i2] = max(0, h[i2] - step)
            else:
                # Increase val2 unless it hits 1
                h[i2] = min(1, h[i2] + step)
            # Similarly adjust one of them to decrease the overlap term
            # ideally balance to preserve sum (only adjust one element per iteration)
            h[i1] = max(0, min(1, h[i1] - step * (1 - val2) * 0.5))
        
        # Project h back within [0,1]
        h = np.clip(h, 0, 1)
        # Re-normalize to have correct sum
        h = h * (n_points / 2) / np.sum(h)

        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
