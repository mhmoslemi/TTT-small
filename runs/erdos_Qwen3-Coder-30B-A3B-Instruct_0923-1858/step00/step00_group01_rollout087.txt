import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Start with the initial solution
    h = np.array(initial_h_values, dtype=float)
    h = h / np.sum(h) * target_sum
    
    # Track time
    start_time = time.time()
    
    # Optimization parameters
    max_iterations = 1000
    patience = 100
    best_c5 = float('inf')
    best_h = None
    
    # Main optimization loop
    last_improvement = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early termination if no progress
        if iteration - last_improvement > patience:
            break
            
        # Determine shift that gives maximum correlation
        max_k = np.argmax(corr_full)
        # The indexing in correlate is tricky; we want the actual shift
        # For a correlation of size 2*n-1, the middle index corresponds to k=0
        k_offset = len(h) - 1
        shift_k = max_k - k_offset
        
        # Determine overlapping regions that contribute to the maximum
        # For shift k, the overlap is over indices where both h[i] and (1-h[i+k]) contribute
        # Examine where the overlap is biggest
        indices_to_perturb = []
        if shift_k >= 0:
            # h[i] * (1 - h[i+shift_k]) for i in range(len(h)-shift_k)
            # So the range of i that we care about
            indices_to_perturb = list(range(len(h) - shift_k))
        else:
            # h[i-shift_k] * (1 - h[i]) for i in range(-shift_k, len(h))
            indices_to_perturb = list(range(-shift_k, len(h)))
            
        # Generate candidates by perturbing h values in relevant regions
        candidates = []
        num_candidates = 50
        for _ in range(num_candidates):
            # Copy original
            h_candidate = h.copy()
            
            # Try to modify h values in relation to the max contributing area
            i = np.random.choice(indices_to_perturb)
            j = np.random.choice(indices_to_perturb)
            
            if i == j:
                continue
            
            # Determine how much to transfer
            transfer = min(h_candidate[i], 1 - h_candidate[j])
            if transfer <= 0:
                continue
            
            amount = np.random.uniform(0, transfer)
            h_candidate[i] -= amount
            h_candidate[j] += amount
            
            # Ensure feasibility
            good = True
            if np.any(h_candidate < 0) or np.any(h_candidate > 1):
                good = False
            if abs(np.sum(h_candidate) * dx - 1.0) > 1e-6:
                good = False
                
            if good:
                # Compute new C5 for this candidate
                corr_candidate = correlate(h_candidate, 1 - h_candidate, mode='full')
                new_c5 = np.max(corr_candidate) * dx
                candidates.append((new_c5, h_candidate))
                
        if candidates:
            # Take the best candidate
            new_c5, new_h = min(candidates, key=lambda x: x[0])
            if new_c5 < c5:
                h = new_h
                continue
    
    # Interpolation to higher resolution if needed
    if len(best_h) < 100:
        # Double the resolution
        new_h = np.interp(np.linspace(0, 2, 2 * len(best_h)), np.linspace(0, 2, len(best_h)), best_h)
        final_h = new_h
        final_dx = 2.0 / len(final_h)
        final_c5 = np.max(correlate(final_h, 1 - final_h, mode='full')) * final_dx
    else:
        final_h = best_h
        final_dx = dx
        final_c5 = best_c5

    return (final_h, final_c5, len(final_h))
