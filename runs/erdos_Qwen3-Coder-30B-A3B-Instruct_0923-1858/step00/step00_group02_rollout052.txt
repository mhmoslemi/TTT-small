import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    
    # Use initial construction if available, else create triangular
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create triangular waveform
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, h[::-1]])
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure constraint: sum(h) = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Optimization parameters
    eta = 0.01
    perturbation_rate = 0.05
    perturbation_freq = 50
    decay_freq = 100
    decay_factor = 0.95
    
    # Main loop
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        if iteration % decay_freq == 0:
            eta *= decay_factor
        
        # Compute current C5 and gradient
        c5 = evaluate_c5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_c5(h_minus, n_points)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        
        # Project to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturbation
        if iteration % perturbation_freq == 0 and iteration - last_improvement > 5:
            noise = np.random.normal(0, perturbation_rate, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = signal.correlate(h, 1 - h, mode='full')
    # Take maximum (divided by dx for normalization)
    max_correlation = np.max(correlations) * dx
    return max_correlation
