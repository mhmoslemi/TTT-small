import numpy as np
from time import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: piecewise constant
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / len(h)
    target_sum = len(h) / 2.0
    
    # Normalize to correct sum
    h = h * target_sum / np.sum(h)
    
    t0 = time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 10000
    for iteration in range(max_iter):
        if time() - t0 > budget_s - 1.0:
            break
            
        # Compute all overlaps
        # Use numpy's correlate to get full convolution
        conv = np.correlate(h, 1-h, mode='full')
        # Compute overlaps for each shift
        overlaps = np.array([conv[len(h)-1+i] * dx for i in range(len(conv))])
        # The shift with maximum overlap is the one with largest value
        k_star = np.argmax(overlaps)
        max_overlap = overlaps[k_star]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Find the most impactful pair to adjust
        # We look at pairs (i, i + k_star) that contribute to that overlap
        # The contribution of pair (i, i + k_star) is h[i] * (1 - h[i + k_star])
        # To reduce it, we can decrease h[i] or increase h[i + k_star]
        # Since we want to minimize max overlap, we should reduce the largest contributions
        best_reduction = -1
        best_i = -1
        best_j = -1
        
        # Check all valid pairs
        for i in range(len(h) - k_star):
            j = i + k_star
            if j >= len(h):
                break
            contribution = h[i] * (1 - h[j])
            if contribution > best_reduction:
                best_reduction = contribution
                best_i = i
                best_j = j
                
        if best_i != -1:
            # Try reducing h[best_i] and increasing h[best_j] by small amount
            # But ensure feasibility
            delta = min(h[best_i], 1 - h[best_j], 0.01)
            if delta > 0:
                h[best_i] -= delta
                h[best_j] += delta
                
                # Project back to constraints
                h = np.clip(h, 0, 1)
                
                # Normalize to maintain sum constraint
                current_sum = np.sum(h)
                diff = target_sum - current_sum
                if abs(diff) > 1e-6:
                    # Distribute the adjustment proportionally to maintain average
                    h += diff / len(h)
