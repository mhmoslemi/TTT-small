import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with initial_h_values if available, otherwise triangular
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Triangular initialization
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based optimization with perturbations
    lr = 0.01
    perturb_rate = 0.05
    perturb_interval = 50
    decay_interval = 100
    decay_factor = 0.95
    last_improvement = 0
    
    for i in range(1000000):  # Large number to allow early stopping
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Compute current C5
        c5_vals = np.correlate(h, 1-h, mode="full") * dx
        c5 = np.max(c5_vals)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i
            
        # Perturbation
        if i - last_improvement > perturb_interval:
            perturb_mask = np.random.random(n_points) < perturb_rate
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            last_improvement = i

        # Gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            
            c5_plus = np.max(np.correlate(h_plus, 1-h_plus, mode="full") * dx)
            
            grad[j] = (c5_plus - c5) / eps

        # Update with gradient descent and learning rate decay
        h -= lr * grad
        if i % decay_interval == 0 and i > 0:
            lr *= decay_factor

        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
    # Final evaluation
    c5_vals = np.correlate(best_h, 1-best_h, mode="full") * dx
    final_c5 = np.max(c5_vals)
    
    return (best_h.tolist(), final_c5, len(best_h))
