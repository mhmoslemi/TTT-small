import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h0)
    except:

        n_points = 45

    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-6
    best_C = C_high
    best_h = h0.copy()
    
    # Constraint matrix setup
    dx = 2.0 / n_points
    # Precompute all shift indices
    shift_indices = {}
    for k in range(1 - n_points, n_points):  # k in [-n+1, n-1]
        shift_indices[k] = np.arange(n_points)
        if k > 0:
            shift_indices[k] = np.roll(shift_indices[k], -k)
        elif k < 0:
            shift_indices[k] = np.roll(shift_indices[k], -k)
    
    while time.time() - start_time < budget_s - 1 and C_high - C_low > tolerance:
        C = (C_low + C_high) / 2.0
        
        # Formulate LP
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2.0)
        
        # Overlap constraints
        for k in range(1 - n_points, n_points):
            # Compute sum_{i=0}^{n-1} h[i] * (1 - h[(i+k)%n])
            idx = shift_indices[k]
            # h[i] * (1 - h[(i+k)%n]) = h[i] - h[i] * h[(i+k)%n]
            # But we use linear constraint: h[i] - h[i] * h[(i+k)%n] <= C*dx
            # This is non-convex; approximate with cross terms or discrete.
            # A better approach is to expand the dot product:
            # sum_i h[i] * (1 - h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]
            
            # For convex relaxation, instead of enforcing equality for all k,
            # use direct constraints with overlap sums
            # This is a quadratic constraint. We relax to linear or use bisection.
            
            # Let's linearize by fixing a chosen finite subset of overlaps
            # Ideal approach: If we solve LP feasibility for an upper bound C:
            #    max sum h[i] s.t. for each k: sum h[i] (1 - h[shift(i,k)]) <= C * n
            #    and h[i] >= 0, h[i] <= 1, sum h[i] = n/2
            
            # Actually do one pass to compute overlaps.
            # For small n, the matrix will be manageable.
            
            # Use convex relaxation of overlap constraints:
            # h[i]*(1 - h[i+k]) <= h[i] - h[i]*h[i+k]  
            # Relax this: h[i]*h[i+k] >= h[i]*h[i+k] (this isn't helping)  
            # The nonlinearity is in _product_ terms.
            # A workaround: add constraints: the overlap <= C * dx
            
            # Brute force computation of overlap sums
            terms = h_vars + 0  # copy
            # i = 0, 1, ..., n-1  term: h[i] * (1 - h[(i+k)%n]) 
            delayed_constraint = cp.sum(cp.multiply(h_vars, 1 - h_vars[idx])) <= C * n_points * dx
            constraints.append(delayed_constraint)
        
        # Objective (minimize C)
        # The problem is to check feasibility, not minimize C directly.
        
        # Instead of solver, we test feasibility: solve the LP for current C
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Unreliable approach:
        # sol = prob.solve(solver=cp.ECOS, verbose=False)

        #     C_low = C

        #     best_h = h_vars.value.copy()
        #     best_C = C
            
        # Simpler: binary search on feasibility by computing max overlap
        try:
            sol = prob.solve(solver=cp.ECOS, verbose=False)
            if not np.isnan(sol) and sol != np.inf:
                # Check the actual overlap values
                h_val = h_vars.value
                overlaps = np.zeros(2 * n_points - 1)
                for k in range(1 - n_points, n_points):
                    if k < 0:
                        overlap = np.sum(h_val * np.roll(h_val, -k))
                    elif k > 0:
                        overlap = np.sum(h_val * np.roll(h_val, k))
                    else:
                        overlap = np.sum(h_val * (1 - h_val))
                    overlaps[k + n_points - 1] = overlap
                
                # compute integral: max overlap
                max_overlap = np.max(overlaps) * dx
                if max_overlap <= C:
                    # Valid
                    C_high = C
                    best_h = h_val.copy()
                    best_C = C
                else:
                    C_low = C
            else:
                C_low = C
        except:
            C_low = C

    if time.time() - start_time > budget_s * 0.98:
        # fallback to best old
        pass

    # Make sure final h is feasible
    h_final = np.clip(best_h, 0, 1)
    total = np.sum(h_final)
    if abs(total - n_points / 2.0) > 1e-6:
        h_final = h_final * (n_points / 2.0) / total
    h_final = np.clip(h_final, 0, 1)

    # Final check
    overlaps = []
    for k in range(1 - n_points, n_points):
        if k < 0:
            overlap = np.sum(h_final * np.roll(h_final, -k))
        elif k > 0:
            overlap = np.sum(h_final * np.roll(h_final, k))
        else:
            overlap = np.sum(h_final * (1 - h_final))
        overlaps.append(overlap)
    
    final_c5 = max(overlaps) * dx
    
    return (h_final, final_c5, n_points)
