import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
        h = h * (0.5 / np.mean(h))
    
    # Initial evaluation
    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Gradient descent parameters
    alpha = 0.01
    perturbation_rate = 0.1
    max_iter = 10000
    iteration = 0
    
    def compute_gradient(h):
        # Compute cross-correlation of h with (1-h)
        # This computes sum_i h[i] * (1 - h[i+k]) for each k
        h_rev = h[::-1]
        corr = signal.correlate(h, 1-h, mode='full')
        # The gradient for each h[i] is sum_k (1 - h[i+k]) * delta_k
        # where delta_k = 1 if there's overlap, otherwise 0
        # We can compute this more efficiently
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(-n_points+1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
        return grad
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Current value
        corr = np.correlate(h, 1-h, mode="full")
        C5_candidate = np.max(np.abs(corr)) * dx
        
        # Update best
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = h.copy()
        
        # Compute gradient approximately
        grad = compute_gradient(h)
        
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
        
        # Take step
        h = h - alpha * grad
        
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy sum constraint
        h = h * (0.5 / np.mean(h))
        
        # Periodically perturb
        if iteration % 100 == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.mean(h))
        
        # Update learning rate
        if iteration % 1000 == 0:
            alpha *= 0.98
            
        iteration += 1
    
    # Final evaluation of best solution
    final_eval = evaluate_erdos_solution(best_h)
    
    return (best_h, final_eval, n_points)
