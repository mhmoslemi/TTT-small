import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for given h."""
    # compute all overlaps for shifts 0 to 2*dx (in discrete steps)
    # this uses full mode correlation
    overlaps = correlate(h, 1-h, mode='full') * dx
    # return max overlap across all valid shifts (the overlapping part)
    # Minus the last half (which is from shift > 2.0)
    # The overlap for shift k is at index k + len(h) - 1 = corner of the middle part
    mid = len(h)
    return np.max(overlaps[mid:]) if len(overlaps) > mid else 0.0

                    if i < len(zero_indices):

        h = (h > np.mean(h)).astype(int)

        if actual_sum != target_sum:

                # Flip 1s to 0s

        best_change = 0.0

                if len(ones_idx) > 0:

                    h[idx] = 0
            # update the maximum overlap
            current_c5 = evaluate_h(h, dx)
    
    return (h.tolist(), current_c5, n_points)
