import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
    
    dx = 2.0 / n_points
    h0 = h0 / (np.sum(h0) * dx) * (n_points / 2)
    
    start_time = time.time()
    
    C_min, C_max = 0.0, 0.5
    best_h = h0.copy()
    best_C = float('inf')
    
    # Binary search for C5
    for iter in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        C_mid = (C_min + C_max) / 2.0
        
        # Construct and solve optimization problem
        h = Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2
        ]
        
        # Add overlap constraints
        # We'll use an approximation approach for the feasibility check
        def compute_overlap(h_vals):
            # compute the convolution manually
            h_vals = np.array(h_vals)
            h_conj = 1 - h_vals
            # valid convolution with full mode
            # replicate to allow full convolution
            h_ext = np.concatenate([h_vals, np.zeros(n_points)])
            h_conj_ext = np.concatenate([h_conj, np.zeros(n_points)])
            corr = np.correlate(h_ext, h_conj_ext, mode="full")
            # get the max over the full range
            max_overlap = np.max(corr[n_points-1:]) * dx
            return max_overlap
        
        # Use scipy.optimize with bounds
        def obj(h_vals):
            return compute_overlap(h_vals)
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(obj, h0, bounds=bounds, method='L-BFGS-B')
        
        if result.success and compute_overlap(result.x) <= C_mid:
            C_max = C_mid
            best_h = result.x.copy()
            best_C = min(best_C, C_mid)
        else:
            C_min = C_mid
    
    # Final check
    final_overlap = compute_overlap(best_h)
    
    return (best_h.tolist(), final_overlap, n_points)
