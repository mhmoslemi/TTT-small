import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Timing setup
    start_time = time.time()
    
    # Parameters
    n_points = 45  # Using the initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Simple initial construction
        h = np.ones(n_points) * 0.5
    
    def evaluate_C5(h_vals):
        # Compute overlap integral using convolution
        h = np.array(h_vals)
        h_flip = 1 - h
        # Auto-correlation (with shift) - full mode
        conv = np.correlate(h, h_flip, mode="full")
        # Take the max of the valid shifts
        # We must be careful about the indices... 
        # shift k can be from -n+1 to n-1 (inclusive)
        # but we want valid overlaps, for a valid overlap at shift k,
        # we have [0, n-1] and [k, k+n-1] overlapping (w/o boundary issues)
        # which need to be valid indices
        # valid shift k range: -n+1 to n-1 but we compute from 0 to 2*n-2
        # For full convolution, valid overlap at shift k (0 to 2*n-2):
        #   - overlap indices in h: [0, n-1] => [k, k+n-1] overlap in range:
        #   index k to min(k+n-1, n-1) = min(k+n-1, n-1) in the first range
        #   index k to min(k+n-1, n-1)  
        #   overlap length: max(0, min(n-1, k+n-1) - max(0, k) + 1) 
        #   This is complex to compute manually
        # Let's compute manually for clarity
        max_overlap = 0.0
        for k in range(1-n_points, n_points): # shift from -n+1 to n-1
            start1 = max(0, k)
            end1 = min(n_points, n_points + k)
            start2 = max(0, -k)
            end2 = min(n_points, n_points - k)
            if end1 <= start1 or end2 <= start2:
                continue
            overlap_sum = np.sum(h[start1:end1] * h_flip[start2:end2])
            overlap = overlap_sum * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    def constraint_bounds(h_vals):
        return np.concatenate([h_vals, 1-h_vals])
    
    # Optimization wrapper
    def optimize_step():
        # Define objective
        def obj(h_vals):
            return evaluate_C5(h_vals)
        
        # Define constraints
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Let's use scipy minimize with L-BFGS-B
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8})
        
        if not res.success:
            print("Optimization failed:", res.message)
            
        return res.x
    
    # Start optimization
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Check current value
    current_C5 = evaluate_C5(h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h.copy()
    
    # Main optimization loop
    while time.time() - start_time < 0.8 * budget_s:
        h_new = optimize_step()
        h_new = np.clip(h_new, 0, 1)
        
        new_C5 = evaluate_C5(h_new)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            
        h = h_new
        
        # Check if we're getting close to stopping condition
        if time.time() - start_time > 0.9 * budget_s:
            break
        
        # Early stopping if we've reached an acceptable bound
        if best_C5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_C5, n_points)
