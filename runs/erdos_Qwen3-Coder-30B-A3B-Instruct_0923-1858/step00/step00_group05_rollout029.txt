import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    # Compute all correlations
    corr = np.correlate(h, 1-h, mode='full')
    # Extract the valid range (correlations for k from -(n-1) to n-1)
    # We want the maximum value among correlations for lags k = 1,2,...,n-1 (positive lags)
    # But we also need to consider all lags from -(n-1) to n-1 and their symmetric counterparts
    # Since we care about overlap at different shifts, we consider all lags from 0 to 2*n-2
    # The formal overlap terms computed by correlation are for lags from -(n-1) to n-1
    # But actual limits are for [0,2] domain, with k stride of dx
    # The real values for C5 are max_k of the integral over 0,2
    max_val = 0
    # We consider each valid shift k from 1 to n-1 (not full range)
    # But since correlations are shifted, think carefully about how to extract overlaps

    # It's simpler to take all the full correlations
    # Since we're integrating over [0,2], dx = 2 / n points
    # We shift k from 0 to n-1, index as shifts from 0 to 2*n-2
    # For shift k=0: [0,...n-1] and [0,...n-1] gives overlap
    # For shift k=1: [0,...n-2] and [1,...n-1] gives overlap
    # etc...
    # So we want max among all valid indices: shift index from n-1 to 2*n-2  
    # The sum is dense at shift 0, so to avoid double-counting, let's restrict shifts to up to n-1 
    # Max over corr indices from n-1 to 2*n-2 => indices 0 to n-1 equivalent indices in corr
    # But let's just get the full and exact
    # max_k (h(x)(1 - h(x+k)) dx) = max_k int h(x)dx - int h(x)h(x+k)dx = max_k [ int h(x)dx - integral ]
    # But it's simpler: 
    # let c = np.correlate(h, 1-h, mode='full')
    # indices 0 to 2*n -2: correlation of h with 1-h at various lags
    # for i = 0 to 2*n-2, it's   sum(h[j] * (1 - h[j+i])) for valid j
    # We compute it one way, in terms of interpreting which lag we 'mean' in [0,2]
    # For shift index from 0 to 2*n-2 in full correlation:
    # 0 means no shift (lag = 0)
    # 1 means shift by one to the right (lag = 1)  
    # ...  
    # n-1 means shift by n-1
    # n means shift by -1  
    # ...
    # 2*n-2 means shift by -(n-1)  (see how correlate works)
    # So shifts in [0, n-1] correspond to lags 0 to n-1 (positive)
    # and [n, 2*n-2] to negative lags after the first n: i-n means shift i-n (negative)
    # We use the shifted expressions, particularly for physics, but
    # in our problem it seems more direct to look at lags and get maximum,  
    # from i = n-1 to 2*n-2 in c, since indices 0 to n-2 were negative and likely not needed or non-physical  
    # What we want to maximize is sum_{i} h[i] * (1 - h[i+k]) dx for k = 1 to n-1  
    # In the full correlation [k = -(n-1) to (n-1)] where k is the shift between indices  
    # For h[i-k] to be valid, we require 0 <= i-k <= n-1 => k <= i <= k+n-1  
    # When k is positive, it's shifting h forward, (n-1+k) to 2n-2 in full
    # So for lags 1 to n-1, we look at specifc indices
    # When we do corr(h, 1-h):
    # index k=0 in corr -> sum h[i]*(1-h[i]) = an overlap at zero
    # index k=1 -> sum h[i]*(1-h[i+1]) = delay 1, and so on
    # index that is [0, n-1] overlaps -> k from 0 to n-1, else go to negative
    # For positive lags, from indices [n-1, 2n-2], i.e., corr[n-1: 2*(n-1)+1]
    # Sizes are len(correlation) = 2*n-1.  
    # The lag k is represented as index i relative to middle of corr = n-1
    # lag 0    -> index n-1  
    # lag 1    -> index n  
    # ...
    # lag (n-1) -> index 2*(n-1) = 2*n -2
    # lag -(n-1) -> index 0  
    # lag -(n-2) -> index 1
    # ...
    # lag -1   -> index n-2
    # Max to consider: k from 1 to n-1. This corresponds to indices n to 2*(n-1) = 2n-2.
    # We should be using indices in corr that corresponds to k = 1 to n-1
    # This means corr[n:2*n-1] if 0-based indexing, which is corr[n-1:2*n-2] if redone with limits
    # Applied to the left/range: indices n, n+1, ..., 2*(n-1)
    # So we compute max corr[n-1:2*n-1] because 0 to 2*n-2 ... 

    # Correctly:
    # corr indices 0 ... 2*n-2  
    # index k=0 in corr is correlation offset 0: h[i]*1-h[i]  
    # index k=1 in corr is overlap of h after offset of 1  
    # index k=2  ->... shift by 2  
    # positive shifts, n-1 to 2*n-2
    # So max corr[i] for i = n: 2*n-2 = 2*n-2+1 -n = n-1 items  
    # corr[n]: corr[2*n-2] inclusive is 2*n-2+1 - n = n-1 items  
    # Go from index n-1 to 2*(n-1) = 2*n-2 for shifts 0 to n-1 => want [1 to n-1] = indices n to 2*n-2 
    # From n-1 to 2*n-2 inclusive => 2*n-2 - (n-1) +1 = n items
    # max_k for k=1 to n-1 = max corrs[n] to corrs[2*n-2]  
    # So index n-1 to 2*n-2, that is n to 2*n-1 inclusive.  
    # But 2*n-1 is out of range of [0, 2*n-2]  
    # corr is 0 to 2*n-2: so max_index starts at n to 2*n-2 inclusive  
    # So valid lags k=1 to n-1 corr in indices [n, 2*n-2]
    # Index at lag=1 corresponds to index n  
    # Index at lag=n-1   corresponds to index 2*n-2  

    # Set max_val = max of corr from index n to 2*n-2  
    if len(corr) != 2*n-1:
        raise ValueError("Correlation length mismatch")

    max_val = np.max(corr[n-1:2*n-1])
    return max_val * dx
    
def objective_and_constraint(h_vals, n_points):
    """Compute objective and check constraints"""
    dx = 2.0 / n_points
    C5 = evaluate_c5(h_vals, dx)
    return C5

def constraint_sum(h_vals, n_points):
    dx = 2.0 / n_points
    return np.sum(h_vals) * dx - 1.0

def get_bounds(n_points):
    return [(0, 1) for _ in range(n_points)]

            raise ValueError("Initial h does not satisfy integral constraint")
        h = initial_h_values.copy()
    else:
        # Build initial feasible solution
        h = np.ones(n_points) * 0.5  # Equal distribution
        if n_points % 2 == 1:
            h[n_points // 2] = 1.0
        h = h * n_points / 2.0 / np.sum(h)  # Scale to match constraint
        # Ensure it's within bounds
        h = np.clip(h, 0, 1)
    
    # Optimization
    bounds = get_bounds(n_points)
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_points,)}]

    best_C5 = float('inf')
    best_h = h.copy()
    dx = 2.0 / n_points

    # Use L-BFGS-B to minimize, but it handles general constraints
    # α is all h[i], subject to bounds and sum = n_points/2  
    # If time is short enough, do random restarts

    max_restarts = 5
    restarts = 0
    while restarts <= max_restarts and (time.time() - start_time) < budget_s:
        # Try to optimize from current h
        try:
            # If we are not in safe mode (we are not looping restarts too fast), 
            # we aim for a simple local optimization.
            # Goal: minimize max_k integral h(x)(1-h(x+k)) dx
            options = {'maxiter': 1000, 'disp': False, 'ftol': 1e-9}
            
            # Use SLSQP from scipy.optimize
            # Variable: h_vars = [h[0], ..., h[n-1]], 0 <= h[i] <= 1
            # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2.0 
            # We embed the result directly for simplicity here
            def obj_func(h_vars):
                # h_vars is the flattened variables
                return evaluate_c5(h_vars, dx)
                
            # Optimization round
            res = minimize(
                fun=obj_func,
                x0=h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options=options,
                tol=1e-9
            )
            
            # After optimization, update best
            if res.success:
                h_opt = res.x
                # Validate that h_opt satisfies constraints
                sum_check = np.sum(h_opt) * dx
                if abs(sum_check - 1.0) < 1e-6:
                    C5_opt = objective_and_constraint(h_opt, n_points)
                    if C5_opt < best_C5:
                        best_C5 = C5_opt
                        best_h = h_opt.copy()
                    # Break early if we've become quite good
                    if (time.time() - start_time) > 0.6 * budget_s and best_C5 < 0.38:
                        break
                        
            h = np.random.rand(n_points)  # Add randomness for next restart
            h = h * n_points / 2.0 / np.sum(h)
            h = np.clip(h, 0, 1)
            
        except Exception as e:
            # If optimization fails, try restarting or continue
            print(f"Optimization failed: {e}")
        
        restarts += 1
        
    # Final validation
    final_C5 = evaluate_c5(best_h, dx)
    if not np.allclose(np.sum(best_h) * dx, 1.0, atol=1e-6):
        raise ValueError("Final solution does not satisfy integral constraint")
    
    return (best_h, final_C5, n_points)
