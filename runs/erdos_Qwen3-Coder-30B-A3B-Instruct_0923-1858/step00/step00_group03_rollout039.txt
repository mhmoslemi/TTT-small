import numpy as np
import time
from scipy.signal import correlate

def binary_convert(h, n_points):
    """Convert h to binary with exact sum constraint."""
    h = np.array(h)
    h_rounded = np.round(h).astype(int)
    current_sum = np.sum(h_rounded)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h_rounded
    diff = target_sum - current_sum
    if diff > 0:
        # Need to flip some 0s to 1s
        zero_indices = np.where(h_rounded == 0)[0]
        flip_indices = zero_indices[:diff]
        h_rounded[flip_indices] = 1
    else:
        # Need to flip some 1s to 0s
        one_indices = np.where(h_rounded == 1)[0]
        flip_indices = one_indices[:abs(diff)]
        h_rounded[flip_indices] = 0
    return h_rounded

def compute_C5(h, dx):
    """Compute the maximum overlap integral.""" 
    h = np.array(h)
    corr = correlate(h, 1 - h, mode='full')
    # Only consider valid overlaps
    n = len(h)
    max_corr = 0
    for k in range(-n + 1, n):
        if k >= 0:
            valid_len = n - k
        else:
            valid_len = n + k
        if valid_len > 0:
            overlap = corr[k + n - 1] * dx
            max_corr = max(max_corr, overlap)
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    # Initialize from the provided initial_h_values if exists
    if 'initial_h_values' in globals():
        h = binary_convert(initial_h_values, n_points)
    else:
        # Default initialization: place 1s uniformly in first half
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    # Check and enforce sum constraint
    h = binary_convert(h, n_points)
    
    # Initial C5 value
    c5_best = compute_C5(h, dx)
    best_h = h.copy()
    
    # Temp parameters for simulated annealing
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    iter_count = 0
    start_time = time.time()
    
    # Annealing loop
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        # Create a neighbor by flipping two bits
        # Find 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try swapping
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Check the new sum
        new_sum = np.sum(h_new)
        if new_sum != n_points / 2:
            continue
        
        # Compute new C5
        c5_new = compute_C5(h_new, dx)
        
        # Accept or reject
        delta = c5_new - c5_best
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            c5_best = c5_new
            best_h = h.copy()
        
        T *= alpha
        
        if iter_count % 1000 == 0:
            # Early break if very small improvement
            if T < 1e-10:
                break
                
    # Final evaluation
    final_c5 = compute_C5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
