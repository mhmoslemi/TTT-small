import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create symmetric initial construction
        n_points = 45
        h = np.ones(n_points)
        # Ensure sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = max(correlate(h, 1-h, mode="full")) * dx
    start_time = time.time()
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    no_improve_limit = 1000
    no_improve_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        
        # Project onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        C5_new = max(correlate(h_new, 1-h_new, mode="full")) * dx
        
        # Accept or reject based on simulated annealing
        delta = C5_new - best_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cool down temperature
        T *= alpha
        
        # Decrease perturbation scale periodically
        if i % 100 == 0:
            perturb_scale *= 0.99
            
        # Terminate if no improvement for a while
        if no_improve_count > no_improve_limit:
            break
    
    return (best_h.tolist(), best_C5, n_points)
