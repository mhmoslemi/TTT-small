import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integrals for all shifts
    # h is assumed to be a 1D array
    # The computation is equivalent to np.correlate(h, 1-h, mode="full") * dx
    overlap = correlate(h, 1-h, mode="full")
    # The overlap integral at each shift
    # For the full convolution output, we only care about the relevant part
    # The indices we want are from -n+1 to n-1, but we're interested in the
    # specific valid range.
    # overlaps are computed such that the result.size = 2*n - 1
    # overlap[0] corresponds to shift = -(n-1)
    # overlap[-1] corresponds to shift = n-1
    # We want the maximum over all shifts (positive and negative)
    return np.max(overlap) * dx

def project_h(h, dx, n_points):
    # Ensure h is in [0, 1]
    h = np.clip(h, 0, 1)
    # Project to satisfy sum constraint
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Normalize differences to ensure sum = target_sum
    delta = target_sum - current_sum
    if abs(delta) < 1e-10:
        return h
    # Distribute the delta
    # We'll add a constant value to all entries (this doesn't keep bounds)
    # Or we'll adjust proportional to 1/h if h is not zero
    # Simple: just scale and clamp
    h = h + delta / n_points
    return np.clip(h, 0, 1)

    h = project_h(h, dx, n_points)
    
    start_time = time.time()
    
    # Parameters
    max_iter = 100000
    alpha = 0.1
    decay_factor = 0.9
    tol = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute the overlap integrals for all shifts
        # Too slow to do it directly, but compute for current h
        # Equivalent to using correlate
        corr = correlate(h, 1-h, mode="full")
        # The overlap for shift k is corr[k + n-1] for k in [-n+1, n-1]
        # So for shift k, the overlap is corr[k + n-1] * dx
        k_idx = np.argmax(corr)
        max_shift = k_idx - (n_points - 1)
        # Get the gradient 
        grad = np.zeros_like(h)
        # i goes from 0 to n_points-1
        # The term: h[i](1 - h[i + k]) appears in the sum for overlap k
        # When summing h[i] * (1 - h[i + k]), we get:
        # grad[i] = 1 - h[i + k]
        # When summing (1 - h[i]) * h[i + k], we get:
        # grad[i + k] = -h[i]
        # So for the max shift k:
        for i in range(n_points):
            ind = i + max_shift
            if ind >= 0 and ind < n_points:
                grad[i] += 1 - h[ind]
                grad[ind] -= h[i]
        
        # Update h
        h = h - alpha * grad
        h = project_h(h, dx, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Update alpha
        if it % 100 == 0:
            alpha *= decay_factor
            
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
