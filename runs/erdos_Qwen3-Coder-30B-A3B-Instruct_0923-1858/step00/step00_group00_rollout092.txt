import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2.0)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters for simulated annealing
    T = 1.0
    alpha = 0.995
    perturb_scale = 0.1
    max_iter = 10000
    patience = 100
    
    iter_no_improve = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2.0)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            iter_no_improve = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - best_c5) / T):
                h = h_new
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        
        # Cooling schedule
        T *= alpha
        if i % 100 == 0:
            perturb_scale *= 0.99
        
        # Early stopping
        if iter_no_improve > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute full convolution of h with (1-h)
    correlation = signal.correlate(h, 1 - h, mode='full')
    # The last n_points-1 entries correspond to valid overlaps
    overlaps = correlation[len(h)-1:]
    return np.max(overlaps) * dx
