import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise create a sensible starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a simple symmetric starting configuration
        n_points = 70
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Ensure sum equals 1 (n_points/2 * dx)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Timing variables
    start_time = time.time()
    elapsed = 0
    iteration = 0
    
    # Gradient descent parameters
    alpha = 0.1
    decay_factor = 0.9
    
    # Optimization loop
    while elapsed < budget_s - 1:
        # Compute overlaps for all shifts
        # This uses convolution; we'll compute it explicitly for gradient
        C5_vals = []
        for k in range(n_points*2 - 1):
            # This is a full convolution, so we compute it properly
            # Simpler change: just compute proper convolution or discrete equivalent
            # But for direct gradient computation, we'll do it manually
            corr = 0.0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    corr += h[i] * (1 - h[j])
            C5_vals.append(corr * dx)
        
        max_k = np.argmax(C5_vals)
        
        # Compute gradient: only for the max overlap case
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_k
            if j < n_points:
                grad[i] += (1 - h[j])
                if i + max_k < n_points:
                    grad[i + max_k] -= h[i]
        
        # Update h
        h = h - alpha * grad
        
        # Project onto feasible set
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            # Adjust proportionally
            scaling = (n_points/2) / current_sum
            h = h * scaling
        
        # Optionally round to binary for final result
        # Actually, don't round yet - keep continuous for gradient steps
        # But let's also check for improvement early
        
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        iteration += 1
        if iteration % 100 == 0:
            alpha *= decay_factor
        
        elapsed = time.time() - start_time
        
        # Debugging/checking: if we've had a large number of iterations, check early stopping
        if iteration > 1000 and abs(alpha) < 1e-8:
            break
        if elapsed > budget_s - 2:
            break
    
    # Final rounding to binary if needed (optional, but sometimes useful)
    # But since gradient suggests continuous values will help, we'll use the continuous
        
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Evaluate C5 using the built-in helper, assuming it exists in pre-imported context
    # Or compute it manually
    h = np.array(h)
    n = len(h)
    # Compute correlation: convolve h with (1-h)
    h_extended = np.pad(h, (0, n-1), mode='constant')
    ones_minus_h_extended = np.pad(1-h, (0, n-1), mode='constant')
    correlation = np.correlate(h_extended, ones_minus_h_extended, mode='full') 
    # We only care about the "full" correlation, which length 2*n - 1
    # But only the relevant parts (shifts k from 0 to n-1 are relevant)
    max_val = 0
    for k in range(n):
        val = correlation[n-1+k] * dx
        if val > max_val:
            max_val = val
    return max_val
