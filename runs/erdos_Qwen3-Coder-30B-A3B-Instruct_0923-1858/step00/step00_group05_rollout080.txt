import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Time tracking
    start_time = time.time()
    
    # Initialize from parent construction if available
    try:
        h_init = initial_h_values.copy()
        n_points = len(h_init)
    except NameError:
        # Create a basic starting construction
        n_points = 70
        h_init = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Optimization parameters
    max_iter = 10000
    min_improvement = 1e-6
    lr = 1e-2
    
    # Helper function to compute overlap
    def compute_overlap(h):
        h = np.array(h)
        corr = signal.correlate(h, 1 - h, mode='full')
        corr *= 2.0 / len(h)
        return np.max(corr)
    
    # Helper function to project onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Compute initial overlap
    current_h = h_init.copy()
    current_C5 = compute_overlap(current_h)
    
    best_C5 = current_C5
    best_h = current_h.copy()
    
    # Low-resolution optimization (initial coarse search)
    n_low = 50
    h_low = np.interp(np.linspace(0, n_points - 1, n_low), 
                      np.arange(n_points), current_h)
    h_low = project(h_low)
    
    low_lr = 1e-2
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient approximation using finite differences
        eps = 1e-4
        grad = np.zeros_like(h_low)
        base_overlap = compute_overlap(h_low)
        
        for i in range(len(h_low)):
            h_plus = h_low.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            plus_overlap = compute_overlap(h_plus)
            
            grad[i] = (plus_overlap - base_overlap) / eps
        
        # Update step
        h_low = h_low - low_lr * grad
        h_low = project(h_low)
        
        # Adaptive learning rate
        new_C5 = compute_overlap(h_low)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = np.interp(np.linspace(0, n_points - 1, n_points), 
                               np.arange(n_low), h_low)
        
        if new_C5 < current_C5:
            current_C5 = new_C5
            current_h = np.interp(np.linspace(0, n_points - 1, n_points), 
                                  np.arange(n_low), h_low)
        else:
            low_lr *= 0.95
            
        if abs(new_C5 - current_C5) < min_improvement:
            break
            
        if time.time() - start_time > budget_s - 10:
            break
    
    # High-res refinement
    h_high = project(best_h)
    lr = 1e-3
    
    # Optimize at higher resolution using projected gradient descent
    prev_C5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation
        eps = 1e-4
        grad = np.zeros_like(h_high)
        base_overlap = compute_overlap(h_high)
        
        for j in range(len(h_high)):
            h_plus = h_high.copy()
            h_plus[j] += eps
            h_plus = project(h_plus)
            plus_overlap = compute_overlap(h_plus)
            grad[j] = (plus_overlap - base_overlap) / eps
            
        # Update step
        h_high = h_high - lr * grad
        h_high = project(h_high)
        
        # Check improvement
        new_C5 = compute_overlap(h_high)
        
        if new_C5 < prev_C5:
            prev_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_high.copy()
        else:
            lr *= 0.95
            
        if abs(new_C5 - prev_C5) < min_improvement:
            break
    
    # Final verification
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
