import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum correlation value
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_constraint(h, n_points):
    # Project h to satisfy sum(h) * dx = 1, i.e. sum(h) = n_points / 2
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    diff = target_sum - current_sum
    # Distribute diff among elements to adjust sum
    delta = diff / n_points
    h_proj = h + delta
    # Ensure bounds [0,1]
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    
    # Initialize h from initial_h_values or create a balanced one
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Create a balanced construction
        h = np.ones(n_points) * 0.5
        # Slightly adjust to make sum = n_points / 2
        h[0] = 0.5
        h[1] = 0.5
        h = project_to_constraint(h, n_points)
    
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    dx = 2.0 / n_points
    
    # Optimization parameters
    perturb_freq = 100
    perturb_strength = 0.05
    step_size = 0.01
    max_iter = 100000
    
    t0 = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Calculate correlations
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max] * dx

        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()

        # Find indices that cause the worst overlap
        # k_max is the full index in [0, 2*n-2] corresponding to a shift of k=(k_max - n+1)
        k_shift = k_max - (n_points - 1)
        
        # Get valid convolution indices
        indices_1 = np.arange(n_points)
        indices_2 = (indices_1 + k_shift) % n_points
        
        # Adjust h to reduce the overlap at k_shift
        #   h[i] * (1 - h[i + k]) contributes to the overlap
        # So reduce h[i] where i + k is also a high h value
        delta = np.zeros_like(h)
        for j in range(n_points):
            idx = (j + k_shift) % n_points
            if h[j] > 0.1 and h[idx] > 0.1:
                delta[j] -= step_size * h[j] * h[idx] / (1.0 + h[j] + h[idx])
                delta[idx] -= step_size * h[j] * h[idx] / (1.0 + h[j] + h[idx])
        h = h + delta
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, n_points)

        # Apply perturbation occasionally to avoid local minima
        if i % perturb_freq == 0 and i > 0:
            # Perturb h: decrease some values and increase others
            p = np.random.permutation(n_points)
            num_to_adjust = int(0.1 * n_points)
            for j in range(num_to_adjust):
                idx1 = p[j]
                idx2 = p[(j + 1) % n_points]
                # Move mass from idx1 to idx2
                move_amount = min(perturb_strength, h[idx1])
                h[idx1] -= move_amount
                h[idx2] += move_amount
            h = np.clip(h, 0, 1)
            h = project_to_constraint(h, n_points)
    
    # Final check and return
    latest_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, latest_c5, n_points)
