import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    # Compute all overlaps using full correlation
    conv = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    best_h = h0.copy()
    best_c5 = evaluate_overlap(h0, dx)
    
    start_time = time.time()
    max_iter = 50
    iter_count = 0
    
    # Start with all shifts as constraints
    k_range = np.arange(-n_points + 1, n_points)
    k_width = len(k_range)
    
    # Use cutting plane method
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        iter_count += 1
        # Formulate problem in CVXPY
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C5)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
            C5 >= 0
        ]
        
        # Add all currents violations to constraints
        conv = correlate(best_h, 1 - best_h, mode='full')
        max_overlap = np.max(conv) * dx
        
        # The full overlap conv needs to satisfy this for all k
        for i, k in enumerate(k_range):
            conv_val = conv[i] * dx
            # We only include violated constraints if bump up C5
            # As we vary h to minimize C5, constraint bounds will change
            if conv_val > best_c5 * 0.99:
                # Add to constraints
                shift_idx = i + n_points - 1
                constraints.append(h[shift_idx:shift_idx+n_points] @ (1 - h[:n_points-shift_idx]) <= C5 * dx if k > 0 else
                                   h[:n_points-k] @ (1 - h[-k:]) <= C5 * dx if k < 0 else
                                   h @ (1 - h) <= C5 * dx)
        
        # Solve the CVXPY problem
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h.value
                new_c5 = C5.value
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except:
            # fallback on simple gradient descent
            pass

        gradient = np.zeros(n_points)

            h_plus = h_test.copy()
            h_minus = h_test.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Clamp and rescale to sum = n_points / 2
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            h_minus = h_minus / np.sum(h_minus) * (n_points / 2)

                val_plus = evaluate_overlap(h_plus, dx)
                val_minus = evaluate_overlap(h_minus, dx)
                gradient[i] = (val_plus - val_minus) / (2 * eps)
            except:
                gradient[i] = 0.0
                
        # Gradient descent step
        lr = 0.01
        new_h = best_h - lr * gradient

        new_h = np.clip(new_h, 0, 1)

            best_h = new_h
            best_c5 = test_c5
    
    # Ensure final constraints satisfied
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
