import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the full cross-correlation h ⋆ (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap value
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_constraint(h, n_points):
    """Ensure sum(h) = n_points / 2 by projecting onto the constraint."""
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust all values linearly
    scale = target_sum / current_sum
    h = h * scale
    return h
