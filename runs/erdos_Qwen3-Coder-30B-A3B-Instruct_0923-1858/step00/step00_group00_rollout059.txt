import numpy as np
from scipy import signal
import time

def evaluate_h(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_h(h):
    """Project h onto feasible region: [0,1] and sum(h) = n/2"""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum = n/2
    n = len(h)
    h = h * (n / 2) / np.sum(h)
    return h

    x_best = vertices[0]

    # Compute centroid (excluding worst vertex)
    x_centroid = np.mean(vertices[:-1], axis=0)
    
    # Reflection
    x_r = x_centroid + alpha * (x_centroid - x_worst)

    # If reflection is better than best
    if f_r < values[0]:

            vertices[-1] = x_e
            values[-1] = f_e

                    vertices[i] = project_h(vertices[i])
                # Re-evaluate
                values = [evaluate_h(v) for v in vertices]
    
    return vertices, values
