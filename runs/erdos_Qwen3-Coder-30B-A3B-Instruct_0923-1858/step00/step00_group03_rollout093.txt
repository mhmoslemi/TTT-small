import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # The overlap integral for a given h(x)
    # We want to minimize max_k int h(x)(1 - h(x+k)) dx
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    corr = correlate(h, 1 - h, mode="full")
    # Take only the relevant part and scale by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def ensure_sum_constraint(h, target_sum):
    # Ensure h sum equals target_sum by flipping bits if needed
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    diff = int(current_sum - target_sum)
    if diff > 0:
        # Need to flip diff 1s to 0s
        indices = np.argwhere(h == 1).flatten()
        if len(indices) >= diff:
            # Flip the first diff ones
            for i in range(diff):
                h[indices[i]] = 0
    elif diff < 0:
        # Need to flip abs(diff) 0s to 1s
        indices = np.argwhere(h == 0).flatten()
        if len(indices) >= abs(diff):
            # Flip the first abs(diff) zeros
            for i in range(abs(diff)):
                h[indices[i]] = 1
    return h
