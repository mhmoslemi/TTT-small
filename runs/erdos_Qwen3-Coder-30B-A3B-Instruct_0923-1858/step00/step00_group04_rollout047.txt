import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_h(h, n_points):
    """Project h to satisfy the constraint sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def init_population(n_points, pop_size, initial_h=None):
    """Initialize population with feasible candidates"""
    pop = []
    if initial_h is not None:
        pop.append(project_h(initial_h.copy(), n_points))
    
    for _ in range(pop_size - len(pop)):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        pop.append(h)
    return pop

def tournament_selection(pop, fitness, k=5):
    """Select one individual using tournament selection"""
    idxs = np.random.choice(len(pop), k)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return pop[best_idx]

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.01):
    """Mutate a candidate"""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + mask * noise
    return project_h(h, len(h))
