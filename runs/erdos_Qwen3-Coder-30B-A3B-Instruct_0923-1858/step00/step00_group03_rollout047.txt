import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start from initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * target_sum / current_sum

    def evaluate_c5(h_vals):
        # Compute all correlations for shift k in [0, n_points-1]
        # Using correlation with (1-h) as per the problem definition
        # mode="full" means we compute correlations for all possible lags
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Extract the relevant part (shifts 0 to n_points-1)
        # Because of symmetry and the way correlation is defined,
        # we take the last n_points entries which correspond to 
        # positive shifts (k = 0 to n_points-1)
        corr_vals = corr[len(corr) // 2:]
        return np.max(corr_vals) * dx

    def project_feasible(h_vals):
        # Clamp values to [0,1]
        h_new = np.clip(h_vals, 0, 1)
        # Adjust sum to target
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-10:
            # Distribute adjustment proportionally
            diff = target_sum - current_sum
            # Add diff uniformly to smallest elements
            sorted_indices = np.argsort(h_new)
            adjustment_per_element = diff / len(h_new)
            h_new[sorted_indices[:int(np.ceil(abs(diff)) / np.abs(adjustment_per_element))]] += adjustment_per_element
            h_new = np.clip(h_new, 0, 1)
        return h_new

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    last_improvement = 0
    iteration = 0

    while time.time() - start_time < budget_s - 1 and iteration < 50000:
        iteration += 1
        # Evaluate overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        corr_vals = corr[len(corr) // 2:]
        k_star = np.argmax(corr_vals)
        max_overlap = corr_vals[k_star]

        # Find which positions contribute most to max_overlap
        # This means looking at all (i, i+k_star) pairs such that 
        # they contribute to the k_star overlap
        # For shift k_star, entries h[i] and h[i + k_star] are involved for i in [0, n_points - 1 - k_star]
        max_contributions = -np.inf
        best_pair = None

        for i in range(n_points - k_star):
            # Contribution of the pair (i, i+k_star)
            contrib = h[i] * (1 - h[i + k_star])
            if contrib > max_contributions:
                max_contributions = contrib
                best_pair = (i, i + k_star)

        if best_pair is not None:
            i, j = best_pair
            # Compute step size that minimizes the overlap for shift k_star
            # derivatives w.r.t. h[i] and h[j]:
            # d/dh[i] = 1 - h[j] and d/dh[j] = -h[i]
            # So for strict optimization, we look along the direction:
            # -grad = (h[j] - 1, h[i])
            # Optimal change:
            step_size = 0.02
            h_trial = h.copy()
            h_trial[i] += step_size * (h[j] - 1)
            h_trial[j] += step_size * h[i]
            # Ensure feasibility
            h_trial = project_feasible(h_trial)

            new_c5 = evaluate_c5(h_trial)
            if new_c5 < best_c5:
                h = h_trial
                best_c5 = new_c5
                last_improvement = iteration
            else:
                # Try a smaller step to avoid overshoot
                step_size = 0.005
                h_trial = h.copy()
                h_trial[i] += step_size * (h[j] - 1)
                h_trial[j] += step_size * h[i]
                h_trial = project_feasible(h_trial)
                new_c5 = evaluate_c5(h_trial)
                if new_c5 < best_c5:
                    h = h_trial
                    best_c5 = new_c5
                    last_improvement = iteration
        else:
            # Step size adjustment fallback
            h = project_feasible(h + np.random.normal(0, 0.001, n_points))
            
        # Occasionally re-evaluate best
        if iteration % 100 == 0:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
        # Early exit if no improvement for a while
        if iteration - last_improvement > 1000:
            break

    return (best_h.tolist(), best_c5, n_points)
