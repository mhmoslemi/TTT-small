    last_improvement = 0

    def compute_gradient(h):
        # Compute A(k) = sum(h) - 0 (constant)
        # Compute B(k) = sum(h[i] * h[(i+k)%n])
        B_matrix = np.zeros((n_points, n_points))
        for k in range(n_points):
            for i in range(n_points):
                B_matrix[k, i] = h[i] * h[(i + k) % n_points]
        B_sum = np.sum(B_matrix, axis=1)
        return B_sum
    
    # Frank-Wolfe iterations
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 100:
        iteration += 1

        # Compute current correlation
        corrs = np.correlate(h, 1 - h, mode="full")
        # The C5 we care about is max over all shifts (excluding zero shift which is Identical)
        c5_val = np.max(corrs[1:]) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            last_improvement = iteration

        # Compute gradient direction more carefully using DC structure
        # linearized as: A(k) - B(k)
        # A is constant vector (sum of h)
        A = np.sum(h) * np.ones(n_points)
        B = np.zeros(n_points)
        for k in range(n_points):
            # sum over i of h[i] * h[(i+k) % n]
            B[k] = sum(h[i] * h[(i + k) % n_points] for i in range(n_points))
        
        # Differentiability structure
        g = A - B  # This is gradient of C5 = max_k (A_k - B_k)

        # Frank-Wolfe linearized problem:
        # minimize ⟨g, d⟩ subject to 0 ≤ d ≤ 1, sum(d) = 0
        # This is a projection problem
        d = -g
        d = np.clip(d, 0, 1)
        # Ensure sum(d) = 0
        d = d - np.mean(d)
        d = d * 1.0  # Avoid scalar multiplication affecting magnitude
        
        # Line search: find step size
        alpha = 1.0
        step = alpha * d
        new_h = h + step
        new_h = np.clip(new_h, 0, 1)
        # Enforce constraint sum = n_points / 2
        new_h = new_h - np.mean(new_h) + (n_points / 2) / n_points

        # Evaluate
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            h = new_h
            best_c5 = c5_new
            last_improvement = iteration

            # Small random perturbation occasionally
            if iteration % 100 == 0:
                noise = np.random.normal(0, 0.02, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h - np.mean(h) + (n_points / 2) / n_points

    return (best_h.tolist(), best_c5, n_points)
