import numpy as np
import time
from scipy.signal import correlate

def binary_convert(h, n_points):
    """Convert h to binary with exact sum constraint."""
    h = np.round(h).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    
    if current_sum == target_sum:
        return h
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = current_sum - target_sum
        indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(indices, diff, replace=False)
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        diff = target_sum - current_sum
        indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(indices, diff, replace=False)
        h[flip_indices] = 1
        
    return h

def compute_C5(h, dx):
    """Compute the maximum overlap integral C5."""
    # Convert to binary for optimization
    h_binary = np.array(h, dtype=int)
    # Compute correlation of h with (1-h)
    corr = correlate(h_binary, 1 - h_binary, mode='full')
    # Extract the valid range of correlations
    start_idx = len(h_binary) - 1
    end_idx = 2 * len(h_binary) - 2
    correlations = corr[start_idx:end_idx]
    # Maximum correlation scaled by dx
    return np.max(correlations) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    
    # Initialize from `initial_h_values` if available

    else:

        mid = n_points // 2

    h = binary_convert(h, n_points)
    
    # Initialize optimization parameters
    current_C5 = compute_C5(h, dx)
    best_C5 = current_C5
    best_h = h.copy()
    
    total_time = 0
    # Simulated annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Randomly select two indices i,j such that h[i]=1 and h[j]=0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try flipping h[i] from 1 to 0 and h[j] from 0 to 1
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        new_C5 = compute_C5(h_new, dx)
        
        delta_C5 = new_C5 - current_C5
        
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        T *= alpha
    
    return (best_h.tolist(), best_C5, n_points)
