import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant given a step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations and take maximum
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    # Clip values
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Add the difference equally to all entries
        h += diff / n_points
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Constants
    n_points = 42
    max_iter = 10000
    eta = 0.05  # Learning rate
    perturb_freq = 100
    perturb_strength = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create an initial construction
        h = np.ones(n_points) * 0.5
        # Set some values to 0 to achieve balance
        for i in range(n_points):
            if i % 2 == 1:
                h[i] = 0.0
        h = project_h(h, n_points)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    t_start = time.time()
    iter_count = 0
    
    while iter_count < max_iter:
        # Check time budget
        if time.time() - t_start > budget_s:
            break
            
        # Compute gradient estimate
        # For DC function max_k [sum(h) - sum(h_i * h_{i+k})], 
        # the gradient w.r.t. h_i is -h_{i+k} + 1 for k such that max is achieved.
        # But for practical optimization, we use the subgradient approach.
        # Here we employ a Frank-Wolfe like approach.
        
        # Compute current cross-correlations
        # We will work with a subgradient approach
        # To find the k that maximizes integral of h*(1-h_{k}), 
        # compute all cross-correlations
        
        # Use all correlations but efficiently
        h1 = h.copy()
        h2 = 1 - h.copy()
        
        corrs = correlate(h1, h2, mode='full')
        # The correlation corresponding to the maximum overlap
        max_k = np.argmax(corrs)
        
        # The correlation index starts from -(n_points-1) to n_points-1
        # So, the shift k is:
        k = max_k - (n_points - 1)
        
        # We approximate the gradient:
        # This approach is to iteratively adjust h to reduce the current overlapping value
        # By computing the gradient of the specific cross-correlation term
        
        # Perturb h to find the k pair and get an idea
        # For cross-correlation, it's complicated, so we do a direct descent.
        
        # Set up the model
        # Let's compute the one where maximum overlap arises
        # For k, we want to compute d/dh_i of h_i * h_{i+k}
        # This involves manipulating indices
        
        grad = np.zeros(n_points)
        # Compute h_i * h_{i+k} for all i
        # Indexing with wraparound for k
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] -= h[j]  # This is part of the - sum 
            grad[i] += 1     # Because of 1 - h_{i+k} term
            
        # Minimize ⟨grad, d⟩ over d subject to 0 ≤ d ≤ 1, sum d = 0
        # We do that via Frank-Wolfe approach
        
        # Solve the linearized problem
        # min η * d^T * grad s.t. 0 ≤ d ≤ 1, sum d = 0
        
        # (using sign - max does not work directly, we do a line search)
        # A simple way: let all nonnegative elements equal 1, others 0, then scale to satisfy sum=0
        
        # For unbiased update (sum d = 0), we use:
        # Take gradient, and project it such that ∑d = 0 and 0 ≤ d ≤ 1
        d = grad.copy()
        
        # Project d to sum(d) = 0
        mean_d = np.mean(d)
        d = d - mean_d
        
        # Clip to [0, 1] and then project to feasible set with sum = 0
        # This is a simple procedure:
        d = np.clip(d, 0, 1)
        d = d - np.mean(d)  # now sum should be 0
        
        # Update h
        h_new = h + eta * d
        
        # Project to satisfy constraints (0 ≤ h ≤ 1, sum h = n_points/2)
        h_new = project_h(h_new, n_points)
        
        # Evaluate 
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Perturbation
        if iter_count % perturb_freq == 0 and iter_count > 0:
            # Random perturbation with small strength
            perturb = np.random.normal(0, perturb_strength, n_points)
            h = h + perturb
            h = project_h(h, n_points)
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
