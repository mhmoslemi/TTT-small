import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Optimization parameters
    max_iter = 1000
    patience = 100
    best_C5 = float('inf')
    best_h = h.copy()
    patience_count = 0
    
    def compute_C5(h_vec):
        # Compute correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Get maximum value (not including zero shift)
        max_corr = np.max(corr)
        return max_corr * dx
    
    def perturb_h(h_vec):
        h_new = h_vec.copy()
        # Compute current correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Find max correlation index
        max_k_idx = np.argmax(corr)
        # Maximum correlation is the middle (index n_points-1) value
        # Which corresponds to shift k from -n_points+1 to n_points-1
        k_shift = max_k_idx - (n_points - 1)
        # Determine overlap regions based on shift k
        # If k > 0 then h[i] overlaps with 1-h[i+k]
        if k_shift > 0:
            start_idx = k_shift
            end_idx = n_points
            overlap_len = end_idx - start_idx
        elif k_shift < 0:
            start_idx = 0
            end_idx = n_points + k_shift
            overlap_len = end_idx - start_idx
        else:
            # k = 0, all values mutually exclusive
            start_idx = 0
            end_idx = 0
            overlap_len = 0
            
        # Select random overlap region for redistribution
        if overlap_len > 0:
            # Sample indices from overlapping regions
            idx_i = int(np.random.randint(max(0, start_idx), min(n_points, end_idx)))
            idx_j = int(np.random.randint(0, n_points))
            if idx_i == idx_j:
                return h_new
            # Compute possible transfer
            max_transfer = min(h_new[idx_i], 1 - h_new[idx_j])
            if max_transfer <= 0:
                return h_new
            # Transfer a random amount
            frac = np.random.random()
            transfer = frac * max_transfer
            h_new[idx_i] -= transfer
            h_new[idx_j] += transfer
            # Project back to feasible range
            h_new[idx_i] = np.clip(h_new[idx_i], 0, 1)
            h_new[idx_j] = np.clip(h_new[idx_j], 0, 1)
        # Ensure constraint is still satisfied
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        return h_new
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb
        h_candidate = perturb_h(h)
        # Check if better
        C5_candidate = compute_C5(h_candidate)
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = h_candidate.copy()
            patience_count = 0
        else:
            patience_count += 1
        
        # Update current solution
        h = h_candidate.copy()
        
        # Early termination if no improvement
        if patience_count >= patience:
            break
            
    # Final check
    final_C5 = compute_C5(best_h)
    
    # Interpolate to higher resolution if needed
    h_final_interp = np.interp(np.linspace(0, 2, 2*n_points), 
                               np.linspace(0, 2, n_points),
                               best_h)
    final_C5_interp = compute_C5(h_final_interp)
    
    # Use interpolator if better
    if final_C5_interp < final_C5:
        h_final = h_final_interp
        final_C5 = final_C5_interp
    else:
        h_final = best_h

    return (h_final.tolist(), final_C5, len(h_final))
