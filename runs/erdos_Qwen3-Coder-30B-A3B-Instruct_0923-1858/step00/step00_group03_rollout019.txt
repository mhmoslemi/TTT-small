import numpy as np
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for all shifts using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum of all correlations, scaled
    return np.max(corr) * dx

        # Flip 0s to 1s
        zero_indices = np.where(h == 0)[0]
        flips = target_sum - current_sum
        for i in range(flips):
            h[zero_indices[i]] = 1
    elif current_sum > target_sum:
        # Flip 1s to 0s
        one_indices = np.where(h == 1)[0]

        for i in range(flips):
            h[one_indices[i]] = 0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    max_iter = 10000
    iter_per_reset = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Every few iterations, reset to explore new areas
        if iteration % iter_per_reset == 0 and iteration > 0:
            # Reset to a small random perturbation of best solution
            h = best_h.copy()
            # Flip some bits
            indices_to_flip = np.random.choice(n_points, size=5, replace=False)
            for idx in indices_to_flip:
                h[idx] = 1 - h[idx]
            # Rebalance to meet sum constraint
            current_sum = np.sum(h)
            while current_sum < target_sum:
                zero_indices = np.where(h == 0)[0]
                if len(zero_indices) == 0:
                    break
                flip_idx = np.random.choice(zero_indices)
                h[flip_idx] = 1
                current_sum += 1
            while current_sum > target_sum:
                one_indices = np.where(h == 1)[0]
                if len(one_indices) == 0:
                    break
                flip_idx = np.random.choice(one_indices)
                h[flip_idx] = 0
                current_sum -= 1

        # Evaluate overlaps
        corr = np.correlate(h, 1 - h, mode="full")

        max_shift = max_overlap_idx - (n_points - 1)  # Convert to actual shift

            best_flip_gain = -np.inf
            best_flip = None

                h[idx] = new_val

                current_sum = np.sum(h)

                    # Find a 1 to toggle to 0
                    one_indices = np.where(h == 1)[0]
                    if len(one_indices) > 0:
                        to_flip = np.random.choice(one_indices)
                        h[to_flip] = 0
                elif current_sum < target_sum:
                    # Find a 0 to toggle to 1
                    zero_indices = np.where(h == 0)[0]
                    if len(zero_indices) > 0:
                        to_flip = np.random.choice(zero_indices)
                        h[to_flip] = 1

        # Re-evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return (best_h.tolist(), best_c5, n_points)
