import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting point: initial_h_values if available, else symmetric construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Basic construction
        n_points = 45
        h0 = np.full(n_points, 0.5)
    
    # Ensure sum condition:
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Prepare shift indices
    shifts = np.arange(-n_points+1, n_points)  # all possible shifts
    shift_indices = []  # [ (i, i+k) for each k, i ]
    
    for k in shifts:
        i_indices = np.arange(n_points)
        k_indices = (i_indices + k) % n_points
        shift_indices.append((i_indices, k_indices))
        
    # Binary search on C
    C_low = 0.0
    C_high = 0.38085  # Known upper bound
    best_C = C_high
    best_h = h0.copy()
    tolerance = 1e-6
    
    while C_high - C_low > tolerance and time.time() - start_time < budget_s - 1:
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate and solve LP
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Overlap constraints
        for i_idx, k_idx in shift_indices:
            # This is sum h[i] * (1 - h[k]), where k = (i + shift) % n
            # That's equivalent to sum h[i] - h[i]*h[k]
            # So we compute (sum h[i]) - sum h[i]*h[k]
            # We also must ensure we consider the correct modulo operations
            # We directly compute the dot products for efficiency
            
            # Get the two relevant arrays
            h_i = h[i_idx]
            h_k = h[k_idx]
            # Put back into the full vector form or work cleverly
            # Actually: compute - sum(h[i] * h[k]) + sum(h[i]) = sum(h[i]) - sum(h[i]*h[k])
            # (total sum - overlap)
            overlap_term = cp.sum(cp.multiply(h_i, h_k))
            total_sum = cp.sum(h)
            
            # Overlap integral <= C * n
            # So: total_sum - overlap_term <= C * n
            # Rearranged: overlap_term >= total_sum - C * n
            # Or +:overlap_term <= C * n - total_sum
            # That contradicts thinking. Rearranging properly:
            # Integral = sum h[i] (1 - h[i+k]) = sum h[i] - sum h[i] h[i+k]
            # This simulates:
            # sum h[i] (1 - h[i+k]) = sum h[i] - sum h[i] h[i+k] 
            # = total - sum h[i] h[i+k]
            # Each shift k: total - sum(h[i]*h[k]) <= C * n
            # Therefore: sum(h[i]*h[k]) >= total - C*n
            
            # Let's just denote:
            # sum h[i] * (1 - h[i+k]) <= C * n
            # sum h[i] - sum h[i] h[i+k] <= C * n
            # sum h[i] - (sum h[i] h[i+k]) <= C * n
            # sum h[i] - C * n <= sum h[i] h[i+k]
            # total_sum - C*n <= overlap_term
            # Collapse this: C*n >= total_sum - overlap
            # OR: overlap <= total_sum - C*n. Let's use this.
            
            # overlap must be <= total - C*n. 
            # i.e. total - overlap >= C*n. 
            # So: total - sum(h[i] h[k]) >= C * n
            # so: -sum(h[i] h[k]) >= C*n - total
            # i.e. sum(h[i] h[k]) <= total - C*n
            
            # But in the original expression we want:
            # sum h[i](1 - h[i+k]) = total - sum h[i] h[i+k]
            # If we want to be <= C * n, then:
            # total - sum h[i] h[i+k] <= C * n 
            # Rearranging: sum h[i] h[i+k] >= total - C*n
            
            # So: sum(h[i] h[i+k]) >= total - C*n
            # The formulation below enforces: sum(h[i] h[i+k]) >= total - C*n
            
            # But it's more natural for LP to (sum h[i] (1 - h[i+k])) <= C*n 
            # left = sum(h_i * (1-h_k))
            # right = C * n
            # So: sum(h[i] - h[i]*h[i+k]) <= C*n
            # => sum(h[i]) - sum(h[i]*h[i+k]) <= C*n  
            # => 1 - sum(h[i]*h[i+k]) / (n/2) <= C  
            
            # Actually: we want sum h[i] (1 - h[i+k]) <= C * n 
            # That's equivalent to: n - sum h[i]*h[i+k] <= C*n
            # That's: sum h[i]*h[i+k] >= n - C*n = n*(1-C)
            
            # So to satisfy constraints:
            # sum(h[i]*h[i+k]) >= n*(1-C)
            
            # furthest complication emphasis: simpler approach:
            # Want sum h[i](1 - h[i+n]) = sum h - sum h[i] h[i+k] <= C*n
            # So: sum h[i] h[i+k] >= sum(h) - C*n  
            # We can compute constraints directly as  
            # sum h[i] h[i+k] >= total_sum - C * n
            
            constraint = cp.sum(cp.multiply(h_i, h_k)) >= total_sum - C_mid * n_points
            constraints.append(constraint)
            
        # Objective is just to check feasibility, not to minimize anything
        # We don't have an objective to minimize
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                # Feasible
                h_opt = h.value
                if h_opt is not None:
                    # Update best solution (if we're getting better)
                    # We can also compute C5 manually for better comparison
                    best_h = h_opt.copy()
                    best_C = C_mid
                    C_high = C_mid
                else:
                    C_low = C_mid          # Infeasible
            else:
                C_low = C_mid
        except Exception as e:
            C_low = C_mid
                
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation of the best solution
    # get the actual C5 value
    overlaps = []
    for k in shifts:
        if k == 0:
            continue  # skip identity
        i_idx = np.arange(n_points)
        k_idx = (i_idx + k) % n_points
        overlap = np.sum(best_h[i_idx] * (1 - best_h[k_idx]))
        overlaps.append(overlap)
    
    final_C = max(overlaps) if overlaps else 0.0
    
    # Clamp h in case precision issues
    best_h = np.clip(best_h, 0, 1)
    # Rescale to ensure sum=1
    if np.sum(best_h) > 0:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    # Return exactly what is requested
    return (best_h.tolist(), final_C, n_points)
