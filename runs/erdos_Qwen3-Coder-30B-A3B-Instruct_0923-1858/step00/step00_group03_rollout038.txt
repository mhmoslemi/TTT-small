import numpy as np
import time
from scipy import signal

def evaluate(h, dx):
    """Evaluate C5 for given binary h and dx"""
    conv = signal.correlate(h, 1 - h, mode='full')
    max_val = np.max(conv) * dx
    return max_val

def binary_convert(h, n_points):
    """Convert h to binary with correct sum constraint"""
    # Round to nearest binary values
    h_bin = np.round(h).astype(int)
    # Adjust to have exactly n_points/2 ones
    current_sum = np.sum(h_bin)
    target_sum = n_points // 2
    diff = target_sum - current_sum
    
    if diff > 0:
        # Need to add more 1s
        indices = np.argsort(h_bin)
        for i in range(diff):
            h_bin[indices[i]] = 1
    elif diff < 0:
        # Need to remove some 1s
        indices = np.argsort(1 - h_bin)
        for i in range(-diff):
            h_bin[indices[i]] = 0
    
    return h_bin

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or a uniform construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple uniform construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h.astype(int)
    
    h = binary_convert(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Select two indices: one with 1 and one with 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Temporarily flip
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Compute new C5
        new_c5 = evaluate(h_new, dx)
        
        # Accept or reject
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        T *= alpha
    
    return (best_h.tolist(), best_c5, n_points)
