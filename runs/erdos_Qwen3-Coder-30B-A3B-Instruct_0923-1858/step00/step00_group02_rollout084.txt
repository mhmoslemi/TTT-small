def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation
    correlation = np.correlate(h, 1 - h, mode='full')
    # Take the maximum
    max_correlation = np.max(correlation)
    return max_correlation * dx

def project_h(h, n_points):
    """Project h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def compute_gradient(h, eps=1e-4):
    """Compute gradient using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h)

        h_minus[i] -= eps
        h_plus = project_h(h_plus, n)
        h_minus = project_h(h_minus, n)
        grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * eps)
    return grad

def trust_region_step(h, grad, trust_radius, n_points):
    """Compute trust region step"""
    # Simple steepest descent step
    step = -grad
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    h_new = h + step
    h_new = project_h(h_new, n_points)
    return h_new
