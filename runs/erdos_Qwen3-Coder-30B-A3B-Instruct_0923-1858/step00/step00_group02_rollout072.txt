import numpy as np
import cvxpy as cp
import time

def evaluate_h(h, n_points):
    """Evaluate the maximum overlap for given h"""
    # Compute all overlaps using correlation
    h_corr = np.correlate(h, 1 - h, mode='full')
    # Only consider the "valid" overlaps (shifts from -n+1 to n-1)
    center = len(h_corr) // 2
    overlaps = h_corr[center - n_points + 1: center + n_points - 1]
    return np.max(overlaps) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Number of points
    n_points = 45  # Use the provided initial solution size
    
    # Initialize from the provided solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, n_points)
    
    # Binary search bounds
    c_low = 0.0
    c_high = 0.38085
    tol = 1e-6
    
    # Start timing
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and c_high - c_low > tol:
        c_mid = (c_low + c_high) / 2
        
        # Define the optimization problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for all shifts
        dx = 2.0 / n_points
        for k in range(-n_points + 1, n_points):
            # Compute the overlap for shift k
            # indices: i and (i+k)%n_points
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            
            constraints.append(overlap_expr <= c_mid * n_points)
        
        # Create and solve problem
        objective = cp.Minimize(0)  # Dummy objective since we're checking feasibility
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Feasible, update lower bound
                c_low = c_mid
                best_h = h_var.value.copy()
                best_c5 = c_mid
            else:
                # Infeasible, update upper bound
                c_high = c_mid
        except Exception:
            # If solver fails, assume infeasible
            c_high = c_mid
            
    # Final validation step
    # Compute actual maximum overlap for the best solution found
    actual_max_overlap = evaluate_h(best_h, n_points)
    
    # Return best solution found within budget
    return (list(best_h), float(actual_max_overlap), n_points)
