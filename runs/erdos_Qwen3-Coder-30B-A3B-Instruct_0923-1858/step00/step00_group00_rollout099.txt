import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h_values, dx):
    """Computes C5 bound using correlation method."""
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0≤h≤1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def init_population(n_points, population_size, seed=42):
    """Initialize population with feasible solutions"""
    np.random.seed(seed)
    population = []
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = project_h(init_h, n_points)
        population.append(init_h.copy())
    else:
        # Symmetric random initialization
        init_h = np.ones(n_points) * 0.5
        init_h = project_h(init_h, n_points)
        population.append(init_h.copy())

    # Add random feasible individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        population.append(h.copy())
    
    return population

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover of two parent solutions"""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, sigma):
    """Add Gaussian noise"""
    noise = np.random.normal(0, sigma, len(h))
    new_h = h + noise
    return project_h(new_h, len(h))

def tournament_selection(population, fitnesses, k=5):
    """Select best individual from tournament"""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def genetic_algorithm(n_points, population_size=50, generations=200, 
                      elitism_rate=0.2, mutation_rate=0.2, seed=42):
    """Perform genetic algorithm optimization"""
    np.random.seed(seed)

    # Initialize population
    population = init_population(n_points, population_size, seed)
    dx = 2.0 / n_points
    fitnesses = [evaluate_c5(h, dx) for h in population]
    
    # Track best solution
    best_fitness = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    # Evolution parameters
    sigma = 0.1
    
    for gen in range(generations):
        # Evaluate population
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
        
        # Elitism
        elite_count = int(elitism_rate * population_size)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        elites = [population[i].copy() for i in elite_indices]
        
        # Generate new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Tournament selection of parents
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child = mutate(child, sigma)
            
            new_population.append(child)
        
        population = new_population[:population_size]
        sigma *= 0.95  # Decrease noise over generations
    
    return best_individual, best_fitness

def gradient_descent_refinement(h_init, dx, max_iter=200, lr=0.01):
    """Refine solution using gradient descent (finite differences)"""
    h = h_init.copy()
    epsilon = 1e-5
    current_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_h(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[j] = (c5_plus - current_c5) / epsilon
        
        # Gradient step
        h = h - lr * grad
        
        # Project to feasible region
        h = project_h(h, len(h))
        
        # Evaluate new state
        new_c5 = evaluate_c5(h, dx)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            lr *= 0.999  # Adaptive learning rate
        else:
            lr *= 0.9  # Reduce learning rate if no improvement
    
    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    n_points = 45  # Use moderate resolution for fast computation
    dx = 2.0 / n_points
    
    # Start with initial solution if available

    # GA part
    print("Starting genetic algorithm...")
    ga_h, ga_c5 = genetic_algorithm(n_points, population_size=50, generations=100, seed=seed)
    
    if time.time() - start_time > 0.9 * budget_s:
        return (ga_h.tolist(), ga_c5, n_points)
    
    # Refinement with gradient descent
    print("Refining with gradient descent...")
    ref_h, ref_c5 = gradient_descent_refinement(ga_h, dx, max_iter=150)
    
    final_c5 = min(ga_c5, ref_c5)
    final_h = ref_h if ref_c5 < ga_c5 else ga_h
    
    # Ensure feasibility and final evaluation
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Final check
    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.2f}s, Final C5: {final_c5:.6f}")
    
    if elapsed > 0.9 * budget_s:
        # Try a few more cycles with updated start points
        h_guess = initial_h_values if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        h_guess = project_h(h_guess, n_points)
        final_h, final_c5 = gradient_descent_refinement(h_guess, dx, max_iter=50)
        final_h = project_h(final_h, n_points)
        final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
