import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Default: linear construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Make sure integral constraint is satisfied
        h[0] = 1.0
        h[1] = 0.0
        h[2] = 0.0
        h[3] = 0.0
        # Distribute the remaining mass reasonably
        h[4:] = 1.0 / (n_points - 4)
        h[4] = 1.0 / (n_points - 4) * 0.5  # slightly adjust to fix integral
    
    dx = 2.0 / n_points
    max_iter = 1000
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            break
            
        # Find k where max occurs
        k_max = np.argmax(corr)
        # determine overlap region in indices 
        lag = k_max - (len(h) - 1)
        # Depending on lag, determine which part of the signal overlaps with itself
        if lag < 0:
            # h[i] overlaps with 1-h[i-lag] for i=0,...,len(h)+lag-1
            # and for i=0,..., n_points-1-lag, so
            # overlap indices for h are from 0 to n_points-1-lag
            # for 1-h, indices are from -lag to n_points-1
            # Overlap in h: [0, n_points-1-lag]
            # Overlap in (1-h): [-lag, n_points-1]
            start_h = 0
            end_h = n_points - 1 - (-lag)
        else:
            # h[i+lag] overlaps with 1-h[i] for i=0,...,n_points-lag-1
            # h indices: [lag, n_points-1]  
            # (1-h) indices: [0, n_points-1-lag]
            start_h = lag
            end_h = n_points - 1
            
        # We'll try to perturb around this region
        region_indices = list(range(start_h, end_h + 1))
        if len(region_indices) == 0:
            continue
            
        # Now make perturbations
        new_h = h.copy()
        trials = 10
        
        for _ in range(trials):
            # Select two indices from region_indices
            i, j = np.random.choice(region_indices, 2, replace=False)
            max_transfer = min(new_h[i], 1 - new_h[j])
            if max_transfer <= 0:
                continue
            amount = np.random.rand() * max_transfer
            if amount == 0:
                continue
            new_h[i] -= amount
            new_h[j] += amount
            
            # Ensure bounds and integral constraint
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h) * dx
            scale = (n_points / 2) / current_sum
            new_h *= scale
            
            # Compute new C5
            corr_new = correlate(new_h, 1 - new_h, mode='full')
            c5_new = np.max(corr_new) * dx
            if c5_new < c5:
                h = new_h
                break  # Take this change
        # End trial loop
    
    # Final validation and interpolation if needed
    corr_final = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr_final) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    # Optionally interpolate to higher resolution. Start from 2*n_points
    # interpolate h to 2*n_points, preserving integral
    n_interp = 2 * n_points
    h_interp = np.interp(np.linspace(0, 2, n_interp), np.linspace(0, 2, n_points), best_h)
    h_interp = np.clip(h_interp, 0, 1)
    # Normalize to integral = 1  
    integral = np.sum(h_interp) * (2.0 / n_interp)
    h_interp = h_interp * (1.0 / integral)
    
    # Re-compute final C5 on interpolation
    corr_final_interp = correlate(h_interp, 1 - h_interp, mode='full')
    final_c5_interp = np.max(corr_final_interp) * (2.0 / n_interp)
    
    return (h_interp, final_c5_interp, n_interp)
