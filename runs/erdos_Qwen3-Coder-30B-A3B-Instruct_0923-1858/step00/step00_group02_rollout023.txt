import numpy as np
import cvxpy as cp
import time

def evaluate_overlap(h, n_points):
    """Evaluate maximum overlap for given h"""
    dx = 2.0 / n_points
    h_full = np.concatenate([h, h])
    overlaps = []
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            overlap = np.sum(h[:-k] * (1 - h_full[k:]))
        else:
            overlap = np.sum(h[abs(k):] * (1 - h_full[:n_points + k]))
        overlaps.append(overlap * dx)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize using initial_h_values if available, else create basic vector
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(initial_h_values)
    else:
        n_points = 50
        h_current = np.full(n_points, 0.5)

    # Compute initial overlap to use as starting upper bound
    start_time = time.time()
    current_C = evaluate_overlap(h_current, n_points)
    best_C = current_C
    best_h = h_current.copy()

    # Binary search bounds
    C_low = 0.0
    C_high = min(0.38085, current_C + 0.01)
    tolerance = 1e-6
    iterations = 0

    while (C_high - C_low) > tolerance and (time.time() - start_time) < (budget_s - 1):
        iterations += 1
        C_mid = (C_low + C_high) / 2
        
        # Formulate and solve the LP feasibility problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            0 <= h_vars, 
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        dx = 2.0 / n_points
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                # This corresponds to h[i] * (1 - h[i+k])
                expr = 0
                for i in range(n_points - k):
                    expr += h_vars[i] * (1 - h_vars[i + k])
            else:
                # This corresponds to h[i] * (1 - h[i+k+n])
                expr = 0
                for i in range(-k):
                    expr += h_vars[i] * (1 - h_vars[i - k])
            constraints.append(expr <= C_mid * n_points * dx)

        # Dummy objective: just check feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVER_ERROR:
                # Check if the solution is valid (within bounds)
                h_solution = h_vars.value
                if h_solution is not None:
                    valid = True
                    for val in h_solution:
                        if val < 0 or val > 1:
                            valid = False
                            break
                    if valid and abs(np.sum(h_solution) - n_points / 2) < 1e-6:
                        C_high = C_mid
                        best_C = C_mid
                        best_h = h_solution.copy()
                        continue  # Found feasible, try to tighten
        except:
            pass  # Solve failed, move on

        # If infeasible, move C_low up
        C_low = C_mid
    
    return (best_h.tolist(), best_C, n_points)
