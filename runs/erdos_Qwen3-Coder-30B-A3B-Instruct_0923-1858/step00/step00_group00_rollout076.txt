import numpy as np
from scipy import optimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction if available, otherwise create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    # Precompute convolution
    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(conv) * dx
    
    # Perturbation and projection
    def perturb_and_project(h_vec, sigma):
        h_new = h_vec + np.random.normal(0, sigma, len(h_vec))
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        return h_new
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.99
    sigma = 0.1
    T = T0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improvement_count = 0
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % 100 == 0:
            sigma *= 0.9
        
        h_candidate = perturb_and_project(h, sigma)
        c5_candidate = compute_c5(h_candidate)
        
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Accept with probability
        delta_c5 = c5_candidate - best_c5
        if delta_c5 > 0 and np.random.random() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
            
        T *= alpha
        
        # Early termination
        if no_improvement_count > 1000:
            break
    
    # Final computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
