import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: 
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h /= np.sum(h) * 2.0 / n_points  # Correct to sum to 1
    
    dx = 2.0 / n_points
    max_iter = 1000
    start_time = time.time()
    time_limit = budget_s - 1.0  # leave margin
    
    # Initial overlap computation
    h_flipped = 1.0 - h
    full_overlap = correlate(h, h_flipped, mode='full')
    full_overlap *= dx
    C5 = np.max(full_overlap)
    best_C5 = C5
    best_h = h.copy()
    
    # Constraint inclusion: select highest overlaps
    max_shift = n_points - 1
    shifts = np.arange(-max_shift, max_shift + 1)
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Re-evaluate overlap for current h
        full_overlap = correlate(h, h_flipped, mode='full') * dx
        
        # Find violations (largest ones)
        violation_indices = np.where(full_overlap > C5 + 1e-8)[0]
        if len(violation_indices) == 0:
            # No new violations - check if improved
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
            break
        
        # Add more constraints
        sorted_indices = np.argsort(full_overlap)[::-1]
        top_V = min(20, len(sorted_indices))
        added_constraints = sorted_indices[:top_V]
        
        # Construct CVX problem
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == 1.0,
            C5_var >= 0
        ]
        
        # Add overlap constraints
        for idx in added_constraints:
            shift_val = shifts[idx]
            shift_idx = idx - max_shift  # shift index
            # Compute correct partial convolution for this shift
            # We use a simplified approach, use finite differences via direct scalar
            # For now, we directly compute all overlaps from current h as sufficient for this small case
            if 0 <= shift_idx < len(h):
                # overlap[k] = sum h[i] * (1 - h[i+k]) over valid range
                # This may be incorrect or we compute full overlap accurately via the convolution
                pass # we'll ignore for now and rely on the error checking to work
            
            # Simpler approach: just include the direct overlap values from computations
            constraints.append(
                cp.sum(cp.multiply(
                    h_vars[shift_idx:] if shift_idx >= 0 else h_vars[:shift_idx],
                    h_flipped[shift_idx:] if shift_idx <= 0 else h_flipped[:-shift_idx]
                )) <= C5_var
            )
            
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        
        # Try multiple solvers
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, 
                       mosek_params={'MSK_DPAR_INTPNT_CO_TOL_PFEAS': 1e-8})
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                # Fallback
                try:
                    prob.solve(solver=cp.SCS, verbose=False)
                except:
                    pass  # Skip if fails
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_vars.value
            if h_new is not None:
                h = np.maximum(0, np.minimum(1, h_new))
                h /= np.sum(h) * 2.0 / n_points  # Re-scale?
                h_flipped = 1.0 - h
                full_overlap = correlate(h, h_flipped, mode='full') * dx
                C5 = np.max(full_overlap)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()
        else:
            # Fallback
            pass
    
    # Final check with evaluation:
    h_final = best_h
    h_flip = 1.0 - h_final
    full_overlap = correlate(h_final, h_flip, mode='full') * dx
    final_C5 = np.max(full_overlap)
    
    # Return final result
    return (h_final.tolist(), float(final_C5), n_points)
