import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h):
    """Evaluate the C5 constant for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Maximum value of correlation
    max_corr = np.max(corr)
    # Scale to get C5
    c5 = max_corr * dx
    return c5

def normalize_h(h):
    """Normalize h to satisfy sum constraint."""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    else:
        # Rescale to match target sum
        h = h * target_sum / current_sum
        return h

def mutate_h(h, mutation_rate=0.1, epsilon=0.1):
    """Mutate a step function h."""
    h = h.copy()
    n = len(h)
    # Variables to adjust
    indices = np.random.choice(n, size=int(mutation_rate * n), replace=False)
    for i in indices:
        # Add small random perturbation
        perturb = random.uniform(-epsilon, epsilon)
        h[i] += perturb
        # Clip to [0,1]
        h[i] = max(0, min(1, h[i]))
    # Rescale to maintain sum constraint
    h = normalize_h(h)
    return h

def crossover_h(h1, h2, crossover_rate=0.7):
    """Blend crossover between two step functions."""
    h1 = h1.copy()
    h2 = h2.copy()
    n = len(h1)
    if random.random() > crossover_rate:
        return h1.copy()
    # Choose crossover point
    point = random.randint(1, n - 1)
    # Blend parts
    offspring = np.concatenate([h1[:point], h2[point:]])
    # Rescale to maintain sum constraint
    offspring = normalize_h(offspring)
    return offspring

def genetic_algorithm(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    population_size = 30
    generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_size = int(0.1 * population_size)
    tournament_size = 5
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        # Use starting solution if available
        h_start = np.array(initial_h_values)
        h_start = normalize_h(h_start)
        population.append(h_start)
    else:
        # Generate initial population
        h_start = np.ones(n_points) * 0.5
        h_start = normalize_h(h_start)
        population.append(h_start)
    
    # Fill rest of population with random functions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = normalize_h(h)
        population.append(h)
        
    current_best_h = None
    current_best_c5 = float('inf')
    
    # Start timing
    start_time = time.time()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(c5)
            
        # Track best
        for i, c5 in enumerate(fitnesses):
            if c5 < current_best_c5:
                current_best_c5 = c5
                current_best_h = population[i].copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Tournament selection with replacement
        new_population = []
        
        # Preserve elite
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection for parent 1
            tournament_indices = np.random.choice(len(sorted_population), tournament_size)
            tournament_fitnesses = [sorted_fitnesses[i] for i in tournament_indices]
            parent1_index = tournament_indices[np.argmin(tournament_fitnesses)]
            
            # Tournament selection for parent 2
            tournament_indices = np.random.choice(len(sorted_population), tournament_size)
            tournament_fitnesses = [sorted_fitnesses[i] for i in tournament_indices]
            parent2_index = tournament_indices[np.argmin(tournament_fitnesses)]
            
            parent1 = sorted_population[parent1_index]
            parent2 = sorted_population[parent2_index]
            
            # Crossover
            child = crossover_h(parent1, parent2, crossover_rate)
            # Mutation
            child = mutate_h(child, mutation_rate)
            
            new_population.append(child)
            
        population = new_population
    
    # Final evaluation of best
    final_c5 = evaluate_h(current_best_h)
    
    return (current_best_h, final_c5, len(current_best_h))

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization to find improved C5 bound."""
    h_values, c5_bound, n_points = genetic_algorithm(seed, budget_s)
    return (h_values, c5_bound, n_points)
