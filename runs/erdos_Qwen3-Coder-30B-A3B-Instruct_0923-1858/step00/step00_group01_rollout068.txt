import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for a given h."""
    dx = 2.0 / len(h)
    # Compute cross-correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    # The overlap integral is max of all correlations
    c5 = np.max(corr) * dx
    return c5

def repair_h(h):
    """Ensure h is within bounds and sums to 1."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s != 0:
        h = h * (2.0 / s)  # Scale to make integral = 1
    return h

def crossover_h(h1, h2):
    """Create offspring via crossover of two parent vectors."""
    n = len(h1)
    # Randomly select crossover point
    cp = np.random.randint(1, n)
    child = np.concatenate([h1[:cp], h2[cp:]])
    return child

def mutate_h(h, rate=0.05):
    """Apply small random mutations to h."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < rate:
            # Add small normally distributed noise
            noise = np.random.normal(0, 0.05)
            h[i] += noise
    return repair_h(h)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population of h vectors."""
    population = []
    if initial_h is not None:
        population.append(initial_h)
        pop_size -= 1
    for _ in range(pop_size):
        # Generate random vector with same sum constraint
        h = np.random.rand(n_points)
        h = repair_h(h)
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with fine resolution
    n_points = 47
    dx = 2.0 / n_points
    pop_size = 50
    
    # Initialize population
    if 'initial_h_values' in globals():

        initial_h = repair_h(initial_h)
    else:
        initial_h = None
    
    population = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 50:
        # Evaluate fitness of all in population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Early termination if we have a good solution
        if best_c5 < 0.3807:
            break
            
        # Keep top 20% as elites
        elite_size = max(2, pop_size // 5)
        elites = population[:elite_size]
        
        # Generate new population
        new_population = elites[:]
        
        # Fill rest of population
        while len(new_population) < pop_size:
            if len(elites) < 2:
                # If too few elites, create random individuals
                h = np.random.rand(n_points)
                h = repair_h(h)
            else:
                # Select two random elites
                i1, i2 = np.random.choice(len(elites), 2, replace=False)
                h1, h2 = elites[i1], elites[i2]
                # Crossover
                child = crossover_h(h1, h2)
                # Mutate
                child = mutate_h(child, rate=0.03)
                h = child
            new_population.append(h)
            
        population = new_population
        generation += 1
        
        # Check for stagnation
        if generation > 1 and fitnesses[0] >= best_c5:
            no_improvement += 1
        else:
            no_improvement = 0
    
    if best_h is None:
        # Fallback to the single best from last evaluation
        # Use the best from the final population if available
        if len(population) > 0:
            best_h = population[0]
            best_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
