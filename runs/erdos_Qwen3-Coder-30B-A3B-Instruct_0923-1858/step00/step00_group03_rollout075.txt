import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the pre-imported initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback to a simple construction if no initial_h_values
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Function to evaluate C5
    def evaluate_c5(h_values):
        # Calculate the full cross-correlation
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    # Optimization parameters
    max_iter = 5000
    step_size = 0.01
    tol = 1e-6
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient descent loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient of the objective w.r.t. h
        # This is a simplified approach: 
        # we approximate gradient by finite difference
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] = min(1.0, h_plus[j] + eps)
            h_minus = h.copy()
            h_minus[j] = max(0.0, h_minus[j] - eps)
            
            # clip to ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Adjust for sum constraint
            sum_diff = (np.sum(h_plus) - np.sum(h_minus)) / 2.0
            h_plus -= sum_diff / n_points
            h_minus -= sum_diff / n_points
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update h with gradient
        h_new = h - step_size * grad
        
        # Ensure h stays in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        sum_factor = (n_points / 2.0) / np.sum(h_new)
        h_new *= sum_factor
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        current_c5 = evaluate_c5(h_new)
        
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
    
    return (list(best_h), float(best_c5), n_points)
