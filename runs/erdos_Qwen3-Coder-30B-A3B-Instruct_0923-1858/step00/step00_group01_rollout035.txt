import time

    h = h * (n_points / 2.0) / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    last_improvement = 0
    iteration = 0
    
    def compute_c5(h_array):
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def apply_perturbation(h_array, iterations):
        # Find the max correlation index
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Determine overlap region for maximum
        shift = max_idx - (len(h_array) - 1)
        overlap_start = max(0, -shift)
        overlap_end = min(len(h_array), len(h_array) - shift)
        
        # Create candidate by redistributing values in overlap
        h_new = h_array.copy()
        trials = min(50, iterations)
        candidate_c5 = float('inf')
        best_candidate = h_new.copy()
        
        for _ in range(trials):
            # Pick random indices within overlap
            i = np.random.randint(overlap_start, overlap_end)
            j = np.random.randint(0, len(h_array))
            
            # Compute max transfer
            max_transfer = min(h_new[i], 1 - h_new[j])
            if max_transfer <= 1e-8:
                continue
            transfer = np.random.uniform(0, max_transfer)
            
            # Apply transfer
            h_new[i] -= transfer
            h_new[j] += transfer
            
            # Enforce bounds
            h_new[i] = np.clip(h_new[i], 0, 1)
            h_new[j] = np.clip(h_new[j], 0, 1)
            
            # Adjust for sum constraint
            diff = np.sum(h_array) - np.sum(h_new)
            if abs(diff) > 1e-10:
                # Distribute adjustment among indices
                step = diff / len(h_new)
                h_new += step
            
            c5_new = compute_c5(h_new)
            if c5_new < candidate_c5:
                candidate_c5 = c5_new
                best_candidate = h_new.copy()
        
        return best_candidate, candidate_c5
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000:

            last_improvement = iteration

        h_new, c5_new = apply_perturbation(best_h, iteration)
        
        if c5_new < best_c5:

            best_c5 = c5_new
            
        iteration += 1
        
        # Occasionally interpolate to higher resolution
        if iteration % 100 == 0 and len(best_h) < 200:

            h_interp = np.interp(

    # Ensure constraints

    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6
    
    return (best_h.tolist(), final_c5, len(best_h))
