import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        n_points = len(h)
    else:
        n_points = 42

        # Balance the sum to be exactly n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    # Scale h to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to compute overlap
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        dx = 2.0 / len(h_vec)
        # Compute all cross-correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum
        max_corr = np.max(np.abs(corr))
        return max_corr * dx

    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    # Initial constraint: integral = 1
    def constraint_eq(x):
        return np.sum(x) * (2.0 / len(x)) - 1.0
    
    # Frank-Wolfe inspired approach with line search
    def objective(h_vec):
        return compute_c5(h_vec)
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute gradient estimate via finite difference
        epsilon = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_minus = h.copy()
            h_minus[i] -= epsilon
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        
        # Solve linear subproblem: minimize <grad, d> subject to d feasible
        # This forces max gradient direction
        d = -grad.copy()
        # Project d to satisfy sum constraint
        d_sum = np.sum(d)
        d = d - d_sum / n_points  # Zero sum constraint
        
        # Line search along d
        alpha = 1.0
        h_trial = h + alpha * d
        h_trial = np.clip(h_trial, 0, 1)
        # Rescale to sum constraint
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        new_c5 = compute_c5(h_trial)
        if new_c5 < best_c5:
            best_h = h_trial.copy()
            best_c5 = new_c5
            h = h_trial.copy()
        else:
            # Try smaller step
            alpha = 0.5
            h_trial = h + alpha * d
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            new_c5 = compute_c5(h_trial)
            if new_c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = new_c5
                h = h_trial.copy()
            else:
                h = h * 0.99  # Small relaxation
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Periodic random perturbation
        if iteration % 50 == 0:
            pert = np.random.normal(0, 0.01, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h, best_c5, n_points)
