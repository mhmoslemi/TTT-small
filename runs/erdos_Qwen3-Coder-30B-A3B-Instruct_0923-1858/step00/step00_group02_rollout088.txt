        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute overlap correlations
        corrs = correlate(h, 1-h, mode="full")
        corrs = corrs * dx
        max_correlation = np.max(corrs)
        
        # Formulate convex approximation
        # Gradient of objective w.r.t. h:
        # For each i, sum over k of (1 - (h[i] + h[i-k]) * (2 - h[i-k] - h[i]))*dx
        # Let's simplify by making linear approximation
        # h[i] * (1 - h[i]) = linear approx around previous iterate
        
        # Compute approximate gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            grad[i] = 1 - 2 * h[i]
            
        # Compute descent direction using trust region approach
        # Minimize: grad^T * d + 0.5 * ||d||^2 subject to:
        # 0 <= h + d <= 1, sum(d) = 0 (since sum(h) is fixed)
        # Trust region constraint: ||d|| <= trust_radius
        
        # Solve QP with trust region constraint: 
        # min 0.5 * d^T * I * d + grad^T * d
        # s.t.     0 <= h + d <= 1
        #          sum(h + d) = n_points / 2
        
        # simpler approach: project gradient descent step onto trust region
        d = -grad
        d_norm = np.linalg.norm(d)
        if d_norm > trust_radius and d_norm > 1e-10:
            d = d * trust_radius / d_norm
            
        # Apply step
        new_h = h + d
        
        # Project to feasible region 
        new_h = np.clip(new_h, 0, 1)
        # Rescale to maintain sum = n_points / 2
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Line search
        alpha = 1.0
        step_size = 0.5
        for _ in range(max_line_search):
            candidate_h = h + alpha * d
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            if candidate_c5 < current_c5:
                h = candidate_h
                break
            alpha *= step_size
            
        # Update trust region
        if alpha < 0.2:
            trust_radius *= 0.5
        else:
            trust_radius = min(trust_radius * 2.0, 1.0)
        trust_radius = max(trust_radius, 1e-4)
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
