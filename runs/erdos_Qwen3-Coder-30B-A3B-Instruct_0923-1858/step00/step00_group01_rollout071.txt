import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a basic starting point (dense rough approximation)
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h *= target_sum / np.sum(h)  # Normalize to correct sum
    
    def compute_c5(h_vec):
        # Compute the overlap integral
        # Since h is supposed to be normalized to have sum(h)*dx = 1,
        # we compute cross correlation with (1 - h)
        corr_full = np.correlate(h_vec, 1 - h_vec, mode='full')
        # The full cross-correlation includes all lags, but we only care
        # about lags from 0 to 2*dx (i.e., 0 to n_points)
        # In practice, the maximum will occur at a lag corresponding to one cycle
        # So we check the center values (corresponding to lags 0 to n_points)
        max_val = np.max(corr_full)
        return max_val * dx

    def objective(h_vec):
        return compute_c5(h_vec)

    # Constraint function
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0

    # Bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}

    # Optimizer settings
    options = {'maxiter': 10000, 'ftol': 1e-9, 'gtol': 1e-9}
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Main loop of optimization
    while time.time() - start_time < budget_s - 1:
        # Try random perturbations
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        
        # Apply perturbation
        new_h = h + perturb
        # Project to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Rescale to satisfy constraint
        new_h *= target_sum / np.sum(new_h)
        
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

        # Small step of gradient descent (using finite differences)
        grad_h = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus *= target_sum / np.sum(h_plus)
            val_plus = compute_c5(h_plus)
            grad_h[i] = (val_plus - best_c5) / eps
        
        step_size = 0.01
        h_new = h - step_size * grad_h
        h_new = np.clip(h_new, 0, 1)
        h_new *= target_sum / np.sum(h_new)
        h = h_new.copy()
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new

        if time.time() - start_time > budget_s - 1:
            break

    return (best_h.tolist(), best_c5, len(best_h))
