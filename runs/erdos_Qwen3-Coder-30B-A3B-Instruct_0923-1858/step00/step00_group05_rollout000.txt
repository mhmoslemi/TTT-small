import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Computes the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value of the correlation
    # over all shifts, scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def initial_h_values_fn():
    """Create a feasible initial h using the provided initial_h_values if available."""
    try:
        return initial_h_values
    except NameError:
        # Fallback: create a simple symmetric construction with sum=1
        n = 70
        h = np.ones(n) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    h0 = initial_h_values_fn()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Adjust to make sum equal to 1 exactly
    current_sum = np.sum(h0)
    target_sum = 1.0
    if abs(current_sum - target_sum) > 1e-10:
        h0 = h0 * target_sum / current_sum

    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]

    # Objective function that we want to minimize
    def objective(h):
        return evaluate_erdos_solution(h)

    # Optimizer
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Indices for sampling
    indices = np.arange(n_points)

    # Gradient approximation and improves
    def gradient_approx(h):
        epsilon = 1e-5
        grad = np.zeros_like(h)
        f0 = objective(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            grad[i] = (objective(h_plus) - f0) / epsilon
        return grad

    # Chunked optimization with random restarts
    max_restarts = 10
    restart = 0
    while restart < max_restarts and (time.time() - start_time) < budget_s * 0.95:
        if (time.time() - start_time) > budget_s * 0.95:
            break

        # SLSQP with constraints
        try:
            # Get random initialization nearby
            h_start = h0 + np.random.normal(0, 0.1, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * target_sum / np.sum(h_start)

            res = minimize(
                objective, 
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'gtol': 1e-8},
                callback=lambda x: None  # No callback needed
            )

            if res.success and evaluate_erdos_solution(res.x) < best_c5:
                best_h = res.x.copy()
                best_c5 = evaluate_erdos_solution(res.x)
        except Exception as e:
            pass  # Skip if optimization fails

        restart += 1

    # Final validation check
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        # Adjust if necessary
        best_h = best_h * (1.0 / final_sum)

    # Verify final solution
    actual_c5 = evaluate_erdos_solution(best_h)
    
    if actual_c5 > best_c5:
        best_c5 = actual_c5

    return (best_h, best_c5, n_points)
