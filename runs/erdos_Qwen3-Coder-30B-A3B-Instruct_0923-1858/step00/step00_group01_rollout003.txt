import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # The full correlation has length 2*n - 1
    # The maximum corresponds to all valid lags k from -(n-1) to (n-1)
    max_corr = np.max(corr)
    return max_corr * dx

def clip_and_rescale(h, dx):
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale so that sum(h)*dx = 1
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def gradient_step(h, dx, alpha=0.01):
    # Compute current correlation and corresponding max
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    # Get the gradient w.r.t. to each h[i]
    # We'll estimate the gradient numerically for now,
    # but we could also derive an exact form
    n = len(h)
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = clip_and_rescale(h_plus, dx)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
    # Compute step: we want to minimize the maximum correlation
    # So we go against the gradient
    h_new = h - alpha * grad
    # Then reclip and rescale
    h_new = clip_and_rescale(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        dx = 2.0 / n
        h = clip_and_rescale(h, dx)
        
    dx = 2.0 / n
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    # Gradient descent with decreasing learning rate
    alpha = 0.01
    max_iter = 1000
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:  # Leave a bit of margin
            break
            
        h_new = gradient_step(h, dx, alpha)
        C5 = evaluate_erdos_solution(h_new)
        if C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = C5
            
        h = h_new
        
        # Every 100 iterations, reduce learning rate
        if (i + 1) % 100 == 0:
            alpha *= 0.1
            
    # Double resolution if possible and improve
    h2 = np.interp(np.linspace(0, 2, 2*n), np.linspace(0, 2, n), best_h)
    dx2 = 2.0 / (2*n)
    h2 = clip_and_rescale(h2, dx2)
    best_h = h2
    best_C5 = evaluate_erdos_solution(best_h)
    
    # Another round of optimization at higher resolution
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
            break

        C5 = evaluate_erdos_solution(h_new)
        if C5 < best_C5:

            best_C5 = C5
            
        h2 = h_new

            alpha *= 0.1
            
    # Ensure final solution is feasible
    final_h = clip_and_rescale(best_h, 2.0 / len(best_h))
    final_C5 = evaluate_erdos_solution(final_h)
