import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for given h values"""
    n = len(h)
    # Compute all correlations
    conv = np.correlate(h, 1 - h, mode='full')
    # Only consider valid shifts (k from -n+1 to n-1)
    max_val = 0
    for k in range(-n+1, n):
        idx = k + n - 1
        if 0 <= idx < len(conv):
            val = conv[idx] * dx
            max_val = max(max_val, val)
    return max_val

def project_feasible(h, n_points):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate_h(h, sigma):
    """Add Gaussian noise"""
    return h + np.random.normal(0, sigma, len(h))

def crossover_h(h1, h2, alpha=0.5):
    """Blend two h values"""
    return alpha * h1 + (1 - alpha) * h2

def genetic_algorithm(initial_h, n_points, budget_s):
    """Genetic algorithm for optimizing h"""
    n_pop = 50
    n_gen = 200
    mutation_sigma = 0.1
    mutation_decay = 0.95
    tournament_size = 5
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(n_pop - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    start_time = time.time()
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        current_best = fitness[0][0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = fitness[0][1].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                break
                
        # Select elites (top 20%)
        elites = [h for _, h in fitness[:n_pop//5]]
        
        # Generate new population
        new_pop = elites.copy()
        while len(new_pop) < n_pop:
            # Tournament selection
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament = [population[i] for i in tournament_indices]
            tournament.sort(key=lambda x: evaluate_h(x, dx))
            parent1 = tournament[0]
            parent2 = tournament[1] if len(tournament) > 1 else parent1
            
            # Crossover and mutation
            child = crossover_h(parent1, parent2, 0.5)
            child = mutate_h(child, mutation_sigma)
            child = project_feasible(child, n_points)
            new_pop.append(child)
            
        population = new_pop
        mutation_sigma *= mutation_decay
    
    return best_h, best_c5

def gradient_descent_step(h, dx, learning_rate=0.01):
    """Perform one step of gradient descent using finite differences"""
    n = len(h)
    epsilon = 1e-5
    c5_current = evaluate_h(h, dx)
    
    # Compute gradient numerically
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_feasible(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_current) / epsilon
    
    # Apply gradient descent
    h_new = h - learning_rate * grad
    h_new = project_feasible(h_new, n)
    
    return h_new, evaluate_h(h_new, dx)

def refined_optimization(initial_h, n_points, budget_s):
    """Refine the best solution found by genetic algorithm"""
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
    learning_rate = 0.01
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.99
            
        h_new, c5_new = gradient_descent_step(h, dx, learning_rate)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    else:
        # Fallback symmetric initialization
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        h0 = project_feasible(h0, n_points)
    
    # Genetic algorithm
    h_ga, c5_ga = genetic_algorithm(h0, n_points, budget_s)
    
    # Refined optimization with gradient descent
    h_refined, c5_refined = refined_optimization(h_ga, n_points, budget_s)
    
    # Return best result
    if c5_refined < c5_ga:
        return (h_refined, c5_refined, n_points)
    else:
        return (h_ga, c5_ga, n_points)
