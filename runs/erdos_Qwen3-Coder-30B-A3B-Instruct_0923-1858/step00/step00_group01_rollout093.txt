import numpy as np
import time
from scipy import signal

def evaluate_h(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the correlation of h with (1-h) for all lags
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # The maximum value over all lags is the bound C5
    c5 = np.max(corr) * dx
    return c5

def repair_h(h_values, n_points):
    # Ensure h values are in [0,1]
    h_values = np.clip(h_values, 0, 1)
    # Adjust to satisfy sum(h) * dx = 1
    dx = 2.0 / n_points
    current_sum = np.sum(h_values)
    target_sum = n_points / 2.0
    scale_factor = target_sum / current_sum if current_sum > 0 else 1.0
    h_values = h_values * scale_factor
    h_values = np.clip(h_values, 0, 1)
    return h_values

def crossover_h(h1, h2):
    # Simple point crossover
    n_points = len(h1)
    crossover_point = np.random.randint(1, n_points)
    child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return child

def mutate_h(h_values, mutation_rate=0.05):
    h_new = h_values.copy()
    n_points = len(h_values)
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            # Add small random noise
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
    return h_new

def evolve_h(initial_h, n_points, population_size=50, generations=100, 
            mutation_rate=0.05, elite_ratio=0.2):
    population = []
    # Initialize population
    if initial_h is not None:
        population.append(initial_h.copy())
        # Add some random h's
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    else:
        # Create random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(generations):
        # Evaluate all individuals
        fitness_scores = []
        for h in population:
            c5 = evaluate_h(h)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Track best
        current_best = population[0]
        current_c5 = fitness_scores[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        
        # Keep elite
        n_elite = max(1, int(elite_ratio * population_size))
        elite = population[:n_elite]
        
        # Generate offspring by crossover and mutation
        offspring = []
        while len(offspring) < population_size - n_elite:
            # Select two parents
            p1, p2 = np.random.choice(elite, size=2, replace=False)
            child = crossover_h(p1, p2)
            child = mutate_h(child, mutation_rate)
            child = repair_h(child, n_points)
            offspring.append(child)
        
        # Fill population
        population = elite + offspring
    
    return best_h, best_c5
