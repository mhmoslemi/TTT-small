import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.9
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.1
    tol = 1e-6
    
    start_time = time.time()
    
    # Normalize to ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlap integrals
        corrs = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corrs) * dx
        
        # Find which shift gives max overlap
        shift_idx = np.argmax(corrs)
        k = shift_idx - (n_points - 1)
        
        # Compute gradient for the current maximum overlap
        grad = np.zeros_like(h)
        if k >= 0:
            grad[k:] += (1 - h[:-k])
            grad[:-k] -= h[k:]
        else:
            grad[:k] += (1 - h[-k:])
            grad[-k:] -= h[:k]
            
        # Update h
        h = h - alpha * grad
        
        # Project h onto feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Check for improvement
        c5_new = evaluate_erdos_solution(h, dx)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        # Adapt learning rate
        if it % 100 == 0 and it > 0:
            alpha *= 0.9
            
        if abs(c5_new - c5) < tol:
            break
            
    # Final refinement with binary rounding if beneficial
    h_final = best_h.copy()
    h_rounded = np.round(h_final).astype(float)
    sum_diff = np.sum(h_rounded) - n_points / 2
    if abs(sum_diff) > 1e-6:
        # Adjust the rounded solution to meet sum constraint
        indices = np.argsort(np.abs(h_final - h_rounded))[:int(abs(sum_diff))]
        for i in indices:
            if sum_diff > 0:
                h_rounded[i] = max(0, h_rounded[i] - 1)
            else:
                h_rounded[i] = min(1, h_rounded[i] + 1)
    
    final_c5 = evaluate_erdos_solution(h_rounded, dx)
    
    return (h_rounded, final_c5, n_points)

def evaluate_erdos_solution(h_vals, dx):
    n = len(h_vals)
    corr_full = np.correlate(h_vals, 1 - h_vals, mode='full')
    return np.max(corr_full) * dx
