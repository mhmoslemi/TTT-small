import numpy as np
import time
import random
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluates C5 for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx

def project_h(h, target_sum):
    """Projects h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.full_like(h, target_sum / len(h))
    else:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population."""
    population = []
    if initial_h is not None:
        population.append(initial_h)
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = project_h(h, n_points / 2)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    selected_idx = np.random.choice(len(population), k)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def crossover(parent1, parent2, alpha=0.5):
    """Blend crossover."""
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(individual, mutation_rate=0.1, noise_std=0.01):
    """Gaussian mutation."""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(individual))
    individual = individual + mask * noise
    return project_h(individual, len(individual) / 2)
