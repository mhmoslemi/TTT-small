import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Computes C5 = max_k int h(x)(1 - h(x+k)) dx
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the convolved values
    max_overlap = np.max(conv)
    c5 = max_overlap * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Ensure h is feasible: sum(h) = n_points / 2, h[i] in [0,1]
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Optimization setup
    dx = 2.0 / n_points
    
    def objective(h_vec):
        # Evaluate max overlap
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv) * dx
    
    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial C5 and best
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Try multiple local optimizations with different random restarts
    attempts = 0
    while time.time() - start_time < budget_s - 1 and attempts < 50:
        attempts += 1
        # Random perturbation of the solution
        h_test = h.copy() + np.random.normal(0, 0.02, n_points)
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Optimize
        try:
            res = minimize(
                objective,
                h_test,

                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100}
            )

                    best_h = res.x.copy()
            # Early exit if we are close
            if best_c5 < 0.3807:
                break
        except Exception:
            pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
