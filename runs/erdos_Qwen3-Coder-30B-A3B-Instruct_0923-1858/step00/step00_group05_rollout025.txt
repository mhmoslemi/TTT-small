import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # This is a placeholder - in practice this would be provided by the harness
    # For now we will compute the C5 value directly
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap for each shift k is corr[n_points-1+k] * dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial solution from harness if available

        n_points = len(h0)

    except NameError:
        # Construct a feasible initial solution
        n_points = 100
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.5
        h0[n_points-1] = 0.5
        # Normalize to sum to n_points/2
        sum_h0 = np.sum(h0)
        h0 = h0 * (n_points / 2) / sum_h0
        feasible = True
    
    # Ensure constraints
    if not feasible:
        # Handle constraint violation more carefully
        h0 = np.clip(h0, 0, 1)
        sum_h0 = np.sum(h0)
        h0 = h0 * (n_points / 2) / sum_h0
    
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute all correlations and return max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        # The maximum is at index n_points - 1
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Setup constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Run optimization with time tracking
    start_time = time.time()
    
    # For better stability with large number of variables,
    # we may want a few restarts
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try 10 random restarts
    for i in range(10): 
        # Create a small random perturbation
        pert = np.random.normal(0, 0.05, n_points)
        h_start = h0 + pert
        h_start = np.clip(h_start, 0, 1)
        # Re-normalize
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Optimize from this restart
        try:
            res = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 5000, 'ftol': 1e-8},
                tol=1e-8
            )
            
            if res.success:
                new_c5 = objective(res.x)
                if new_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = new_c5
        except Exception as e:
            continue
            
        # Check time
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Final check
    c5_final = evaluate_erdos_solution(best_h)
    
    # Ensure the solution has correct sum and bounds
    assert np.isclose(np.sum(best_h), n_points / 2), "Sum constraint violated"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Domain constraint violated"
    
    return (best_h, c5_final, n_points)
