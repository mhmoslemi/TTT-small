import numpy as np
from scipy.optimize import minimize
from scipy import signal
import time

def evaluate_erdos_solution(h_vals, dx=2.0 / 47):
    """
    Evaluate the C5 constant for a given h function.
    """
    h = np.array(h_vals)
    # Compute full cross-correlation of h with (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    # Scale by dx
    c5 = max_corr * dx
    return c5

        h[block_start:block_end] = 1.0
        h[:block_start] = 0.0
        h[block_end:] = 0.0
        # Normalize to integral = 1
        integral = np.sum(h) * 2.0 / n_points
        h *= (1.0 / integral)

        actual_integral = np.sum(h_vec) * dx
