import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h_vals, dx):
    # Evaluate the maximum overlap
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = start_time + budget_s
    
    # Initialize h
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 2000
    iter_count = 0
    
    # Helper to project onto feasible set
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Finite difference gradient approximation
    def gradient_estimate(h, eps=1e-4):
        grad = np.zeros_like(h)
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(len(h)):
            h_plus[i] += eps
            h_minus[i] -= eps
            # Project both
            h_plus = project_h(h_plus)
            h_minus = project_h(h_minus)
            # Evaluate
            val_plus = evaluate_erdos_solution(h_plus, dx)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (val_plus - val_minus) / (2 * eps)
            # Restore
            h_plus[i] -= eps
            h_minus[i] += eps
        return grad
    
    # Optimization loop
    for iter_count in range(max_iter):
        if time.time() > max_time:
            break
            
        current_C5 = evaluate_erdos_solution(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Compute gradient
        try:
            grad = gradient_estimate(h)
        except Exception as e:
            # Fallback to simple random search if gradient fails
            h = project_h(h + np.random.normal(0, 0.01, len(h)))
            continue
            
        # Trust region step
        step = -grad * trust_radius
        
        # Update h
        h_new = h + step
        
        # Apply projection
        h_new = project_h(h_new)
        
        # Evaluate new point
        new_C5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if better
        if new_C5 <= current_C5:
            h = h_new
            trust_radius = min(2.0 * trust_radius, 1.0)  # Expand if improves
        else:
            trust_radius = max(trust_radius / 2.0, 1e-6)  # Shrink
            
        # Perturbation if necessary
        if iter_count % 100 == 0 and iter_count > 0:
            # Add noise to escape local minima
            noise = np.random.normal(0, 0.02, len(h))
            h = project_h(h + noise)
        
    return (best_h, best_C5, n_points)
