import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: a simple step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Make it a valid construction
        h[0] = 0.4
        h[1] = 0.6
        h[-1] = 0.4
        h[-2] = 0.6
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Constraint adjustment
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h *= (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient descent with projection
    max_iter = 10000
    alpha = 0.1
    alpha_decay = 0.9
    last_improvement = 0
    t0 = time.time()
    
    for i in range(max_iter):
        if time.time() - t0 > budget_s - 1:
            break
            
        # Compute all overlaps
        # This computes a full convolution, where we extract maximum
        # indices k such that max_k over [k] in overlap = sum h[i] * (1 - h[i-k])
        # fft-based for speed:
        # Correlate h and (1-h) 
        h1 = 1 - h
        # Use full mode for all shifts; handle wraparound properly
        # For overlap integral we compute:
        # \int h(x)(1 - h(x+k)) dx = \sum h[i] * (1 - h[i + k]) dx over [0,2]
        # Check both: h[i] * (1 - h[i+k]) for all valid i, k
        # This corresponds to computing cross-correlation of h and (1-h), 
        # but shifted by k, so k runs from -(n-1) to (n-1)
        # The maximum of these values is the C5 bound
        # We compute all correlations using fft: 2*n - 1 results
        corr_full = signal.correlate(h, h1, mode='full') 
        # Get the 'k' values that correspond to the different shifts (from -n+1 to n-1)
        # For efficiency compute maximum directly without overwriting full overlap
        # Actually, the true overlap integral is just the corresponding term max_k of \sum_k sum_i h[i](1-h[i+k]) dx
        # But there are n-k+1 terms for each k (where k is shift in range 0,...n-1)
        # e.g., this is how to compute overlap integral:
        # For each shift k=0,...,n-1:
        #    sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) = sum h[i] * (1-h[i+k])
        # We need to compute this for all k
        # Compute full correlations (cross-correlation h*(1-h)) but we only care about overlapping parts
        # We can reformulate overlap: 
        # Need to compute for all k = 0..n-1: sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) 
        # But more correctly: 
        # We compute all coordinates of x[k] = sum_{i=0}^{n-1} h[i] * (1 - h[i+k])  
        # For each k (positive or negative), but only where valid indices exist.
        # Since h is of size n, (1-h) is of size n,
        # x is of size 2*n - 1 due to full correlation mode.
        # Here x[0] represents shift k = -n+1 ... x[n-1] shift k=0 ... x[2*n-2] = shift k=n-1
        # We only need those for k > 0
        # Actually, x[k] for k=0 is shift=0. Overlap computation for that shift is
        # sum_{i=0}^{n-1} h[i] * (1 - h[i])  = sum_{i=0}^{n-1} (h[i] - h[i]^2) 
        # The symmetric k leads to same value.
        # Let's reframe: Each k should represent the shift of 1-h with h, so smaller k means
        # shifting the 1-h one to the left.
        # But the solution is to extract all the relevant overlapping integrals.
        # We can also compute directly:
        c5_values = []
        for k in range(n_points):
            # For shift k, overlap = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k]) 
            # Let's compute all such sums:
            overlap = 0.0
            for i in range(n_points - k):
                overlap += h[i] * (1 - h[i + k])
            c5_values.append(overlap * dx)
        c5_val = max(c5_values)
        
        # Just optimize C5 in an approximation way:
        # Compute gradient: for i update by derivative of  max_k of overlap_k, 
        # where overlap_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k]) 
        # The contribution to overlap_k to gradient of h[i] is:
        # For h[i], if i+k < n, then term (1-h[i+k]) appears.
        # For h[i+k], if i+k < n, then term -h[i] appears.
        # Since we're looking at max_k, instead of computing symbolic derivatives
        # we'll work with the maximum mismatch.
        # Let's do one long cycle of update using the gradient for the current maximum k.
        
        # Get the index k where max overlap occurs:
        max_idx = np.argmax(c5_values)
        k_max = max_idx
        
        # Compute the gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Contribute to gradient of h[i]
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            # Contribute to gradient of h[i+k_max]
            if i < n_points - k_max:
                grad[i + k_max] -= h[i]
        
        # Update 
        h = h - alpha * grad
        
        # Project back (clamp and rescale)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
        
        # Check if improved
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Decrease step size
        if i % 100 == 0:
            alpha *= alpha_decay
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_c5, n_points)

def evaluate_erdos_solution(h, dx):
    n = len(h)
    # Compute all overlapped integrals
    # overlap_k = sum_{i=0}^{n-1-k} h[i] (1 - h[i+k]) dx for k=0,...,n-1
    overlaps = []
    for k in range(n):
        overlap = 0.0
        for i in range(n - k):
            overlap += h[i] * (1 - h[i + k])
        overlaps.append(overlap * dx)
    return max(overlaps)
