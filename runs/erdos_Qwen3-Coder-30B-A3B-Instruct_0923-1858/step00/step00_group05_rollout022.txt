    # Optimization settings
    alpha = 0.01
    perturbation_rate = 0.1
    max_iter = 10000
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        # Compute gradient via finite differences for numerical stability
        gradient = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.mean(h_plus))
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.mean(h_minus))
            
            c_plus = evaluate_erdos_solution(h_plus)
            c_minus = evaluate_erdos_solution(h_minus)
            gradient[i] = (c_plus - c_minus) / (2 * eps)
            
        # Gradient descent step
        h = h - alpha * gradient
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.mean(h))
        
        # Perturbation occasionally
        if iteration % 100 == 0:
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.mean(h))
            
        # Adapt learning rate
        if iteration % 500 == 0 and iteration > 0:
            alpha *= 0.9
            if alpha < 1e-6:
                alpha = 1e-6
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, n_points)
