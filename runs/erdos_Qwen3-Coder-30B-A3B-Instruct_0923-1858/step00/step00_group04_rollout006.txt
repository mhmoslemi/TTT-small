import time

from scipy.linalg import eigh

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h)

        h[::2] = 0.5

        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()

    last_improvement = 0

        # Compute current gradient-like approximations via finite differences
        # Estimate gradient using ADMM-like approach
        
        # Create low-rank approximation to speed up cross-correlation
        # Use top 10 components of autocorrelation
        h_centered = h - np.mean(h)

        for i in range(n_points):
            corr_matrix[i] = np.correlate(h_centered, np.roll(h_centered, i), mode="full")[n_points-1:]
        
        # Get top 10 eigenvectors of covariance matrix
        try:
            _, V = eigh(corr_matrix, eigvals=(n_points-10, n_points-1))
            V = V[:, ::-1]  # Reverse to get largest eigenvalues first
        except:
            V = np.eye(n_points)
        
        # Project h onto the low-rank subspace
        if V.shape[1] < V.shape[0]:
            h_proj = V @ (V.T @ h)
        else:
            h_proj = h
        
        # Frank-Wolfe step with gradient estimation  
        # Simplified: compute directional derivative at current h
        h_grad = np.zeros(n_points)
        for i in range(n_points):
            h_grad[i] = np.sum(h_proj * np.roll(h_proj, i)) - np.sum(h_proj)
        
        # Frank-Wolfe direction (simplify to gradient descent for robustness)
        direction = h_grad
        direction = direction - np.mean(direction)  # Zero mean to maintain constraint
        
        # Line search for optimal step size (simplified)
        step_sizes = [0.01, 0.05, 0.1, 0.2]

    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    """Compute the maximum overlap integral using correlation."""
    n = len(h)
    # Compute full cross-correlation
    cross_correlation = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags (excluding trivial zeros)
    max_overlap = np.max(cross_correlation)
    return max_overlap * dx
