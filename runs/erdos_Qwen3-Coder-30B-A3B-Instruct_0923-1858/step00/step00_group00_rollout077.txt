import numpy as np
from scipy import optimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a symmetric initialization
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Initialize with a simple step function
        mid = n_points // 2
        h[:mid] = 0.0
        h[mid:] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    temp = 0.1
    temp_decay = 0.99
    sigma = 0.1
    sigma_decay = 0.9
    max_no_improvement = 1000
    no_improvement_count = 0
    start_time = time.time()
    
    # Annealing schedule parameters
    t = 0
    while time.time() - start_time < budget_s:
        # Perturb current solution
        delta = np.random.normal(0, sigma, n_points)
        h_candidate = h + delta
        
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate *= (n_points / 2.0) / np.sum(h_candidate)
        
        # Compute C5
        c5_candidate = compute_c5(h_candidate, dx)
        
        # Accept or reject
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if np.random.rand() < np.exp(-(c5_candidate - best_c5) / temp):
                h = h_candidate.copy()
        
        # Anneal temperature and sigma
        temp *= temp_decay
        if t % 100 == 0:
            sigma *= sigma_decay
        
        t += 1
        if no_improvement_count > max_no_improvement:
            break
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_c5), int(n_points))

def compute_c5(h, dx):
    # Compute convolution
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Find maximum
    max_val = np.max(conv)
    return max_val * dx
