import numpy as np
import scipy.signal as signal
import time
from math import sqrt

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value of the correlation scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create balanced starting point
        h = np.ones(n_points) * 0.5
        # Distribute mass to satisfy sum constraint
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * (target_sum / np.sum(h))
    
    # Ensure initial h satisfies the sum constraint and bounds
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization loop
    iteration = 0
    max_iter = 10000  # Roughly 1e4 for 600s budget
    pert_freq = 100
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        iteration += 1
        
        # Compute full cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Reshape to make indexing easier (it's symmetric around center)
        corr_full = corr.copy()
        
        # Compute overlap for full correlation
        overlap = corr_full * dx
        
        # Find maximum index in overlap
        max_idx = np.argmax(overlap)
        max_k = max_idx - (len(h) - 1)
        
        # Compute difference from the correlation
        h_new = h.copy()
        h_old = h.copy()
        
        # Smaller perturbation
        eps = 0.01
        
        # Create effective gradient approximation
        # Adjust h[i] based on how much contribution i has to max_k overlap
        for i in range(n_points):
            # Add a small negative perturbation to region i to reduce its contribution to max_k
            prod = 0.0
            for j in range(n_points):
                k = (j - i) % n_points
                prod += h_old[j] * (1 - h_old[k])
            # Apply adjustment to reduce that contribution
            # Currently step-size: perturb by small epsilon
            h_new[i] = min(1.0, h_new[i] - eps * (1.0 if i == 0 else 0.0))
            h_new[i] = max(0.0, h_new[i])
        
        # Project to sum constraint
        current_sum = np.sum(h_new)
        h_new = h_new * (target_sum / current_sum)
        
        # Verify bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Check for improvement
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Periodically apply more aggressive perturbation to escape local minima
        if iteration % pert_freq == 0:
            # Perturb globally: redistribute part of high-weight regions to low-weight regions
            sorted_indices = np.argsort(h)
            # Pick top 5% indices to redistribute from
            cutoff_idx = int(0.05 * n_points)
            to_reduce = sorted_indices[-cutoff_idx:]
            to_add = sorted_indices[:cutoff_idx]
            
            total_reduced = sum(h[i] for i in to_reduce)
            amount_per_target = total_reduced / len(to_add)
            
            h_new = h.copy()
            for i in to_reduce:
                h_new[i] -= amount_per_target
            for i in to_add:
                h_new[i] += amount_per_target
            
            # Clip and reproject
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            # Check new value
            candidate_c5 = evaluate_erdos_solution(h_new)
            if candidate_c5 < current_c5:
                h = h_new
                current_c5 = candidate_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
