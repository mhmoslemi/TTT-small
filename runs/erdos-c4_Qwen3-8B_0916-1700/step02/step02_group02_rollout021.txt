import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, roots, Eq, solve, N
from sympy.polys.polytools import resultant
from time import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    def hermite_poly(n, z):
        return hermite(n, z)
    
    def exact_eval(coeffs):
        m = len(coeffs)
        z = symbols('z')
        H = [hermite(4*k, z) for k in range(m)]
        Q = sum(c*h for c, h in zip(coeffs, H))
        Q = Q.subs(z, 0)
        Q = Q - Q.as_poly().coeffs()[0]
        Q = Q.subs(z, z)
        Q = Q.as_poly()
        eq = Eq(Q, 0)
        sol = solve(eq, z)
        real_roots = [float(N(r)) for r in sol if r.is_real and r > 0]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def num_eval(coeffs):
        m = len(coeffs)
        z_vals = np.linspace(-10, 10, 1000)
        H = [hermite(4*k, z_vals) for k in range(m)]
        Q = np.dot(coeffs, H)
        Q -= Q[0]
        Q = Q / z_vals**2
        real_roots = np.roots(np.poly1d(Q))
        real_roots = real_roots[np.isreal(real_roots)]
        real_roots = real_roots[real_roots > 0]
        if not real_roots.size:
            return float('inf')
        return (real_roots[-1])**2 / (2 * np.pi)
    
    def score(coeffs):
        try:
            return exact_eval(coeffs)
        except:
            return num_eval(coeffs)
    
    def refine(coeffs):
        m = len(coeffs)
        return np.concatenate([coeffs, np.random.rand(m)])
    
    def perturb(coeffs):
        return coeffs * (1 + np.random.normal(0, 0.1, len(coeffs)))
    
    def optimize():
        initial = np.random.rand(3)
        best = initial.copy()
        best_score = score(best)
        
        for m in [3, 5, 7]:
            if m > len(best):
                best = refine(best)
            for _ in range(20):
                start = time()
                result = minimize(lambda x: score(x), x0=best, method='L-BFGS-B',
                                  bounds=[(0, 10) for _ in range(m)], tol=1e-10)
                if time() - start > budget_s * 0.6:
                    break
                if score(result.x) < best_score:
                    best = result.x
                    best_score = score(best)
                if m < 7:
                    best = np.concatenate([best[:m], refine(best[m:])])
                if best_score < 0.32:
                    break
        
        while time() < budget_s - 10:
            pert = perturb(best)
            if score(pert) < best_score:
                best = pert
                best_score = score(best)
        
        return best[:7]
    
    return optimize()
