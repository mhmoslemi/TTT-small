import numpy as np
from scipy.optimize import minimize
from sympy import symbols, diff, hermite, Poly, roots

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    m = 8
    # Geometric progression initialization
    initial = np.geomspace(1e-4, 1, m)
    
    def evaluate(coeffs):
        z = symbols('z')
        H = [hermite(4*k, z) for k in range(m)]
        Q = sum(c * h for c, h in zip(coeffs, H))
        # Enforce Q(0) = 0
        Q = Q - Q.as_poly().coeffs()[0]
        # Find r_max with exact root isolation
        P = Poly(Q, z)
        roots = roots(P)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    # Multi-start optimization
    best = np.inf
    best_coeff = None
    for _ in range(10):
        start = np.geomspace(1e-4, 1, m) * np.random.rand(m)
        res = minimize(evaluate, x0=start, method='L-BFGS-B', 
                      options={'ftol': 1e-10, 'maxiter': 100, 'disp': False}, 
                      tol=1e-10)
        if res.fun < best:
            best = res.fun
            best_coeff = res.x
    
    return best_coeff
