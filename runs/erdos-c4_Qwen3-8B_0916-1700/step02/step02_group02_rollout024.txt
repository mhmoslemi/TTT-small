import numpy as np
from scipy.optimize import minimize
from scipy.stats import qmc
from sympy import symbols, hermite, simplify, Eq, solve
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    m = 8
    initial = np.random.rand(m)
    bounds = [(0, 1) for _ in range(m)]
    
    def evaluate_coeffs(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q.as_poly().as_expr()
        Q = Q.subs(z, z)
        Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        r_max = max(real_roots) if real_roots else 0
        return r_max**2 / (2 * np.pi)
    
    def surrogate(coeffs):
        return evaluate_coeffs(coeffs)
    
    def objective(coeffs):
        return surrogate(coeffs)
    
    def search():
        points = qmc.sample(m, 100, seed=seed)
        best = np.inf
        best_coeffs = None
        for p in points:
            coeffs = np.array(p)
            score = objective(coeffs)
            if score < best:
                best = score
                best_coeffs = coeffs
        return best_coeffs
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        coeffs = search()
        result = minimize(objective, x0=coeffs, method='L-BFGS-B', 
                          bounds=bounds, options={'ftol': 1e-10, 'maxiter': 50})
        if result.fun < best:
            best = result.fun
            best_coeffs = result.x
    
    return best_coeffs[:4]
