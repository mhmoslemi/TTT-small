import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time
from sympy import symbols, hermite, simplify
from scipy.optimize import Bounds

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    # Define Hermite polynomials up to H_32 for coefficients up to 8
    z = symbols('z')
    H = [simplify(hermite(k, z)) for k in range(0, 32, 4)]  # H_0, H_4, H_8, ..., H_32
    
    def evaluate_c4_coefficients(coeffs):
        # Build the polynomial Q(z) = sum c_k H_{4k}(z)
        Q = sum(c * h for c, h in zip(coeffs, H[:len(coeffs)]))
        # Enforce Q(0) = 0 and flip sign at infinity
        if Q.subs(z, 0) != 0:
            Q = -Q
        # Remove z² factor and find r_max
        try:
            # Find roots of Q(z)/z²
            roots = np.roots([1, 0, 0] + list(Q.as_poly().all_coeffs()[::-1]))
            real_roots = roots[np.isreal(roots)]
            positive_roots = real_roots[real_roots > 0]
            # Remove roots of even multiplicity (approximate check)
            positive_roots = np.unique(positive_roots)
            if len(positive_roots) == 0:
                return float('inf')
            r_max = positive_roots[-1]
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Hybrid initialization: mix random, Chebyshev, and Hermite-based guesses
    initial_guesses = []
    # Random guess
    initial_guesses.append(np.random.uniform(-1, 1, 8))
    # Chebyshev guess
    theta = np.linspace(0, np.pi, 9)
    initial_guesses.append(np.cos(theta))
    # Hermite-based guess
    initial_guesses.append(np.array([0.5**k for k in range(8)]))
    
    best_result = np.inf
    best_coeffs = np.zeros(8)
    
    for guess in initial_guesses:
        result = minimize(
            fun=evaluate_c4_coefficients,
            x0=guess,
            bounds=[(-1, 1) for _ in range(8)],
            method='L-BFGS-B',
            options={'maxiter': int(budget_s / 0.1)}
        )
        if result.fun < best_result:
            best_result = result.fun
            best_coeffs = result.x
    
    return best_coeffs
