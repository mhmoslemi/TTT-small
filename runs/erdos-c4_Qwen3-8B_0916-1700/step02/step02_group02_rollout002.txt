import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, roots, diff, simplify

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def exact_hermite(n, z):
        return hermite(n, z)
    
    def evaluate(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(c * exact_hermite(4*k, z) for k, c in enumerate(coeffs))
        Q = Q - Q.subs(z, 0)
        Q = Q.as_poly()
        roots = roots(Q)
        real_roots = [r for r in roots if r.is_real and r > 0]
        r_max = max(real_roots) if real_roots else 1
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate(coeffs)
    
    # Initial grid search with structured coefficients
    base = np.linspace(0.1, 1.5, 10)
    initial = []
    for i in range(2):
        for j in range(2):
            for k in range(2):
                initial.append(base[i] * base[j] * base[k])
    initial = np.array(initial)
    
    # Optimization
    result = minimize(objective, x0=initial, method='L-BFGS-B', 
                      options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False})
    
    return result.x[:8]
