import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Generate diverse initial points with symmetry and exploration
    initial_points = []
    for i in range(8):
        if i % 2 == 0:
            coeffs = np.random.uniform(-1, 1, 8)
            coeffs[0] = 0  # Ensure Q(0)=0
            initial_points.append(coeffs)
        else:
            coeffs = np.zeros(8)
            coeffs[0] = 0
            coeffs[1] = 1.0
            coeffs[2] = -1.0
            coeffs[3] = 0.5
            initial_points.append(coeffs)
    
    best_coeffs = initial_points[0]
    best_value = objective(best_coeffs)
    
    # Evaluate all initial points
    for i in range(len(initial_points)):
        value = objective(initial_points[i])
        if value < best_value:
            best_value = value
            best_coeffs = initial_points[i]
    
    # Refine with L-BFGS-B from best point
    result = minimize(
        fun=objective,
        x0=best_coeffs,
        bounds=[(-1, 1) for _ in range(8)],
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
