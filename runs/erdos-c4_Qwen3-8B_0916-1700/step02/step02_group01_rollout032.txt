from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)

        [0.0, 0.5, 1.0, -0.5, 0.0, 0.5, 1.0, -0.5]
    ])
    
    best_value = float('inf')
    best_coeffs = grid[0, :]
    
    # Evaluate all grid points
    for i in range(grid.shape[0]):
        value = objective(grid[i, :])
        if value < best_value:
            best_value = value
            best_coeffs = grid[i, :]
    
    # Refine with L-BFGS-B starting from best_coeffs
    initial_guess = best_coeffs
    bounds = [(-1, 1) for _ in range(8)]
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
