from scipy.optimize import minimize, Bounds
from scipy.linalg import lstsq

def run(seed=42, budget_s=200, **kwargs):

        if n == 0:
            return np.array([1.0])
        coeffs = np.zeros(n+1)
        coeffs[0] = 1.0
        for k in range(1, n):
            coeffs[k] = 2*k*coeffs[k-1]
        coeffs[n] = 1.0
        return coeffs
    
    def evaluate_c4_coefficients(coeffs):
        import sympy
        z = sympy.symbols('z')
        Q = sum(c * sympy.hermite(n, z) for n, c in enumerate(coeffs))
        Q = Q.as_poly()
        # Ensure Q(0) = 0
        if Q.subs(z, 0) != 0:
            Q = Q - Q.subs(z, 0)
        # Flip sign if necessary
        if Q.lead_term().is_positive:
            Q = -Q
        # Remove z^2 factor
        Q = Q.as_poly().div(z**2)[0]
        roots = Q.all_roots()
        real_roots = [r for r in roots if r.is_real]
        real_roots.sort()
        # Find r_max
        r_max = None
        for r in real_roots:
            if r > 0:
                r_max = r
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    def score(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def optimize(coeffs):
