from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    # Define Hermite polynomials up to degree 32 (for 8 coefficients)
    H = [lambda z, k=k: hermite(k)(z) for k in range(32)]
    
    def evaluate_c4_coefficients(coeffs):
        # Construct Q(z) and perform exact calculation
        def Q(z):
            val = 0
            for k in range(len(coeffs)):
                val += coeffs[k] * H[k](z)
            return val
        
        # Find r_max numerically
        def find_r_max():
            # Simple root-finding with sign change
            def f(z):
                return Q(z) / (z**2 + 1e-12)
            
            # Brute-force search for sign change
            for r in np.logspace(-2, 3, 500):
                if f(r) > 0:
                    continue
                for dr in np.logspace(-3, -1, 100):
                    if f(r + dr) < 0:
                        return (r + dr)**2
            return 1.0
        
        r_max_squared = find_r_max()
        return r_max_squared / (2 * np.pi)
    
    # Define bounds for coefficients
    bounds = [(-1, 1) for _ in range(8)]
    
    # Use differential evolution for global search
    result = differential_evolution(
        fun=evaluate_c4_coefficients,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=int(budget_s / 0.1),
        tol=1e-6,
        seed=seed
    )
    
    # Refine with local optimization
    refined_result = minimize(
        fun=evaluate_c4_coefficients,
        x0=result.x,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return refined_result["x"]
