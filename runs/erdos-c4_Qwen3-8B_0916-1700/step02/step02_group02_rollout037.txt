import numpy as np
from scipy.optimize import minimize
import time
from scipy.optimize import Bounds

def hermite(n, x):
    if n == 0:
        return 1
    elif n == 1:
        return 2*x
    else:
        return 2*x*hermite(n-1, x) - 2*(n-1)*hermite(n-2, x)

def evaluate_c4_coefficients(coeffs):
    m = len(coeffs)
    H = []
    for k in range(m):
        h = hermite(4*k, np.linspace(-10, 10, 1000))
        H.append(h)
    Q = np.dot(coeffs, H)
    Q -= Q[0]
    roots = np.roots(np.poly1d(Q))
    real_roots = roots[np.isreal(roots)]
    r_max = np.max(real_roots[real_roots > 0])
    return r_max**2 / (2 * np.pi)

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def surrogate(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        x = np.linspace(-10, 10, 1000)
        H = [hermite(4 * k, x) for k in range(m)]
        Q = np.dot(coeffs, H)
        Q -= Q[0]
        roots = np.roots(np.poly1d(Q))
        real_roots = roots[np.isreal(roots)]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def objective_wrapper(coeffs):
        return surrogate(coeffs)
    
    def init_coeffs(n):
        return np.random.rand(n)
    
    best_coeffs = init_coeffs(4)
    best_value = objective(best_coeffs)
    
    for n in range(4, 8):
        initial = init_coeffs(n)
        result = minimize(objective_wrapper, x0=initial, method='L-BFGS-B', 
                          bounds=Bounds(0, 1), options={'ftol': 1e-10, 'maxiter': 100})
        if objective(result.x) < best_value:
            best_coeffs = result.x
            best_value = objective(best_coeffs)
