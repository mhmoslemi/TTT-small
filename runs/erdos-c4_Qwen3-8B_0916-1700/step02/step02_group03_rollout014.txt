import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, Poly, solve

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    z = symbols('z')
    hermites = [Hermite(k) for k in range(8)]
    
    # Symbolic initialization with Hermite polynomials at z=0
    initial_coeff = [1.0 if k % 2 == 0 else 0.0 for k in range(8)]
    
    def evaluate_c4_coefficients(coeff):
        Q = sum(c * hermites[i](z) for i, c in enumerate(coeff))
        Q = Q.as_poly().remove term(z**2)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        r_max = max(real_roots) if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    result = minimize(
        fun=objective,
        x0=initial_coeff,
        bounds=[(-1, 1) for _ in range(8)],
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
