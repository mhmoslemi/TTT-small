import numpy as np
from scipy.optimize import differential_evolution
from sympy import symbols, hermite, Poly, solve, Eq, Rational
from fractions import Fraction

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def hermite_coeffs(n):
        z = symbols('z')
        return [Fraction(hermite(i, z).as_expr().coeff(z, j) for j in range(i+1)) for i in range(n+1)]
    
    def eval_Q(coeffs, z_vals):
        Q = np.zeros(len(z_vals))
        for i, c in enumerate(coeffs):
            H = hermite(4*i, z_vals)
            Q += c * H
        return Q
    
    def find_r_max(coeffs):
        z_vals = np.linspace(-10, 10, 10000)
        Q = eval_Q(coeffs, z_vals)
        Q -= Q[0]  # Ensure Q(0) = 0
        sign_changes = np.sign(np.roll(Q, 1) - Q)
        roots = np.where(np.abs(sign_changes) > 1e-10)[0]
        real_roots = z_vals[roots]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max
    
    def objective(coeffs):
        z_vals = np.linspace(-10, 10, 10000)
        Q = eval_Q(coeffs, z_vals)
        Q -= Q[0]
        r_max = find_r_max(coeffs)
        return r_max**2 / (2 * np.pi)
    
    result = differential_evolution(objective, bounds=[[-1, 1] for _ in range(8)], 
                                    strategy='best1bin', popsize=15, 
                                    maxiter=50, tol=1e-10, mutation=(0.5, 1), 
                                    recombination=0.8, seed=seed)
    return result.x
