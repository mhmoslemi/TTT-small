import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.cmaes import fit
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def hermite(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
    
    def evaluate_c4_coefficients(coeffs):
        m = len(coeffs)
        z_vals = np.linspace(-10, 10, 1000)
        H = np.array([hermite(4*k, z_vals) for k in range(m)])
        Q = np.dot(coeffs, H)
        Q -= Q[0]  # Enforce Q(0) = 0
        
        # Find r_max with exact root isolation
        roots = np.roots(np.poly1d(Q))
        real_roots = roots[np.isreal(roots)]
        real_roots = real_roots[real_roots > 0]
        r_max = np.max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    # Initial structured coefficients with alternating signs
    initial = np.array([1e-3, -1e-3, 1e-3, -1e-3, 1e-3, -1e-3, 1e-3, -1e-3])
    
    # Phase 1: CMA-ES for global search
    result = fit(
        func=evaluate_c4_coefficients,
        x0=initial[:8],
        bounds=Bounds(-1e-2, 1e-2),
        sigma=0.1,
        maxiter=50,
        seed=seed,
        ftol=1e-10,
        disp=False
    )
    
    # Phase 2: L-BFGS-B refinement
    refined = result['x']
    result = minimize(
        fun=evaluate_c4_coefficients,
        x0=refined,
        method='L-BFGS-B',
        bounds=Bounds(-1e-2, 1e-2),
        options={'ftol': 1e-10, 'maxiter': 100, 'disp': False}
    )
    
    # Return the best coefficients
    return result.x[:8]
