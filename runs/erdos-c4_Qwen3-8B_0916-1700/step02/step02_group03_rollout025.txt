import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.special import hermite
from sympy import symbols, Eq, solve

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(c * hermite(4*k)(z) for k, c in enumerate(coeffs))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots = sorted(real_roots)
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    # Initialize with Chebyshev polynomials
    theta = np.linspace(0, np.pi, 6)
    initial_guess = np.cos(theta)
    
    # Differential evolution with adaptive bounds
    def bounds_func(x):
        return [(-1, 1) for _ in range(len(x))]
    
    result_de = differential_evolution(
        func=objective,
        bounds=bounds_func,
        strategy='best1bin',
        popsize=15,
        maxiter=int(budget_s / 10),
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Local refinement with L-BFGS-B
    result_l_bfgs = minimize(
        fun=objective,
        x0=result_de.x,
        bounds=bounds_func,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 5)}
    )
    
    return result_l_bfgs["x"]
