import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.special import hermite
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coeffs(c):
        def Q(z):
            total = 0
            for k in range(len(c)):
                total += c[k] * hermite(4*k, z)
            return total
        
        def find_r_max(c):
            def f(z):
                return Q(z) / (z**2)
            
            try:
                result = root_scalar(f, bracket=[1, 100], method='brentq')
                return result.root**2 / (2*np.pi)
            except:
                return float('inf')
        
        result = minimize(lambda c: find_r_max(c), x0=np.random.rand(8), method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)])
        return result.x
    
    start = time.time()
    best = evaluate_coeffs(np.random.rand(8))
    for _ in range(10):
        if time.time() - start > budget_s - 5:
            break
        next_step = np.random.rand(8) * 2 - 1
        next_step = np.clip(next_step, 0, 10)
        current = evaluate_coeffs(next_step)
        if current < best:
            best = current
    return np.random.rand(8)
