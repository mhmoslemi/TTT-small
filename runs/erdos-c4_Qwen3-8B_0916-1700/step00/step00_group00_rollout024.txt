import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    x = np.random.rand(C4_MAX_COEFF_COUNT) * 2 - 1  # Initial guess
    
    def objective(c):
        # Numerical estimation of U(c)
        z = sp.symbols('z')
        H = [sp.hermite(h) for h in range(4*C4_MAX_COEFF_COUNT)]
        Q = sum(c[k] * H[4*k] for k in range(len(c)))
        Q = Q.subs(z, 0).as_numerical()  # Ensure Q(0) = 0
        Q = Q / z**2
        roots = sp.roots(Q).real
        r_max = max(roots[roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def evaluate_c4_coefficients(c):
        z = sp.symbols('z')
        H = [sp.hermite(h) for h in range(4*len(c))]
        Q = sum(c[k] * H[4*k] for k in range(len(c)))
        Q = Q.subs(z, 0).as_numerical()
        Q = Q / z**2
        roots = sp.roots(Q).real
        r_max = max(roots[roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def optimize():
        result = minimize(objective, x, method='L-BFGS-B', tol=1e-10, options={'maxiter': 100})
        return result.x
    
    start = time()
    while time() - start < budget_s:
        try:
            best_c = optimize()
            best_bound = evaluate_c4_coefficients(best_c)
            print(f"Current best bound: {best_bound}")
        except:
            pass
    return best_c.tolist()
