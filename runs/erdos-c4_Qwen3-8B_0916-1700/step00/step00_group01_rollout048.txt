def evaluate_surrogate(c):
    def Q(z):
        return sum(c[k] * np.polynomial.hermite.hermegauss(4*k, z) for k in range(len(c)))
    try:
        r = fsolve(lambda z: Q(z), 10)

def run(seed=42, budget_s=200):
