from scipy.optimize import minimize_scalar
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=200, **kwargs):

    def eval_obj(c):

            result = evaluate_c4_coefficients(c)
            return result
        except:
            return np.inf
    
    def fast_surrogate(c):
        coeffs = np.array(c)
        z = np.linspace(0, 10, 1000)
        Q = 0
        for k in range(len(coeffs)):
            Q += coeffs[k] * np.hermite.hermval(z, 4*k)
        Q /= z**2
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = 10
        for root in z[sign_changes]:
            if np.abs(Q[root]) < 1e-8:
                r_max = root
        return r_max**2 / (2 * np.pi)
    
    def optimize():
        result = differential_evolution(
            fun=eval_obj,
            bounds=[(-1, 1) for _ in range(8)],
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        return result.x
    
    start_time = time.time()
    best_coeffs = optimize()

        step = 0.01
        new_coeffs = best_coeffs + np.random.normal(0, step, 8)
        new_coeffs = np.clip(new_coeffs, -1, 1)
