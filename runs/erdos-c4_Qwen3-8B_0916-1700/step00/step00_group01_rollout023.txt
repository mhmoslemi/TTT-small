import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, expand, solve, Eq
import time

def run(seed=42, budget_s=200):
    np.random.seed(seed)
    # Initial random coefficients
    coeffs = np.random.rand(8) - 0.5
    best_coeffs = coeffs
    best_score = float('inf')
    start_time = time.time()

    def objective(x):
        nonlocal best_score
        if time.time() - start_time > budget_s:
            return best_score
        coeffs = x
        # Construct Q(z)
        z = symbols('z')
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(len(coeffs)))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Flip if negative at infinity
        if Q.as_coefficients_dict().get(z**np.inf, 0) < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q / z**2
        # Find r_max: largest positive real root where Q(z) changes sign
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        real_roots = [r for r in real_roots if np.isfinite(r)]
        # Filter by sign change
        sign_changes = []
        for r in real_roots:
            if len(sign_changes) == 0:
                sign_changes.append(r)
            else:
                prev = sign_changes[-1]
                if prev < r:
                    sign_changes.append(r)
        if not sign_changes:
            return float('inf')
        r_max = sign_changes[-1]
        score = r_max**2 / (2 * np.pi)
        return score

    result = minimize(objective, x0=best_coeffs, method='L-BFGS-B', bounds=[[-10, 10] for _ in range(8)], options={'maxiter': 1000})
    return result.x
