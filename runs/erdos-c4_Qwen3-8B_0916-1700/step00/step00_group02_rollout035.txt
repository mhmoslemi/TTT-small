import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, sqrt, Rational, simplify

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    best_coeff = None
    best_value = float('inf')

    def objective(coeffs):
        c = coeffs
        z = symbols('z')
        expr = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        expr = expr.subs(z, 0)
        expr = expr.as_numerical()
        # For exact check, we'll use symbolic solution later
        return np.mean((np.abs(c) - 1)**2)

    def surrogate_bound(coeffs):
        c = coeffs
        z = symbols('z')
        expr = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        # Enforce Q(0) = 0, adjust signs
        expr = expr - expr.subs(z, 0)
        expr = expr / z**2
        roots = solve(expr, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        # Remove roots of even multiplicity
        unique_roots = []
        for r in real_roots:
            if not unique_roots or abs(r - unique_roots[-1]) > 1e-8:
                unique_roots.append(r)
        if not unique_roots:
            return float('inf')
        r_max = max(unique_roots)
        return r_max**2 / (2 * np.pi)

    def exact_check(coeffs):
        c = coeffs
        z = symbols('z')
        expr = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        expr = expr - expr.subs(z, 0)
        expr = expr / z**2
        roots = solve(expr, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        unique_roots = []
        for r in real_roots:
            if not unique_roots or abs(r - unique_roots[-1]) > 1e-8:
                unique_roots.append(r)
        if not unique_roots:
            return float('inf')
        r_max = max(unique_roots)
        return r_max**2 / (2 * np.pi)

    def generate_initial_guesses(num_guesses=20):
        guesses = []
        for _ in range(num_guesses):
            coeffs = np.random.randn(C4_MAX_COEFF_COUNT)
            coeffs = np.where(np.abs(coeffs) > 1e-6, coeffs, np.sign(coeffs))
            guesses.append(coeffs)
        return guesses

    initial_guesses = generate_initial_guesses()

    for guess in initial_guesses:
        result = minimize(objective, guess, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(C4_MAX_COEFF_COUNT)], options={'maxiter': 100})
        coeffs = result.x
        # Use surrogate to get approximate bound
        bound = surrogate_bound(coeffs)
        if bound < best_value:
            best_value = bound
            best_coeff = coeffs
        # Check exact value if surrogate bound is low
        if bound < 0.35:
            try:
                exact = exact_check(coeffs)
                if exact < best_value:
                    best_value = exact
                    best_coeff = coeffs
            except:
                pass

    return best_coeff
