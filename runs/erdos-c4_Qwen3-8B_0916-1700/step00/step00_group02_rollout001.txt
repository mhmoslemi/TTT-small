from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=200, **kwargs):

        if leading.as_poly().coeffs()[-1] < 0:
            poly = -poly

        poly = poly / z**2

        roots = solve(poly, z, domain=S.Reals)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([float(r) for r in real_roots if r > 0])
        # Find r_max: largest root where sign changes
        for r in real_roots[::-1]:
            if real_roots.index(r) > 0:
                r_prev = real_roots[real_roots.index(r) - 1]
                if np.sign(poly.subs(z, r_prev)) != np.sign(poly.subs(z, r)):
                    return r**2 / (2 * np.pi)
        return float('inf')
    
    def surrogate(c):

        from scipy.optimize import fsolve
