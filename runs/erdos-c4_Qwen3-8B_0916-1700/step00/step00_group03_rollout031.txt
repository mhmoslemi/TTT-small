import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_value = float('inf')
    start_time = time.time()
    
    def evaluate_coefficients(coeffs):
        from sympy import symbols, hermite, roots, Rational, simplify
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Flip sign to make positive at infinity
        if Q.as_leading_term(z) < 0:
            Q = -Q
        # Remove z² factor
        Q = Q / z**2
        # Find largest positive real root
        roots_list = roots(Q, domain='Real')
        real_roots = [r for r in roots_list if r > 0]
        real_roots = [r for r in real_roots if r != 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 1.0
        U = r_max**2 / (2 * np.pi)
        return U
    
    def objective(coeffs):
        return evaluate_coefficients(coeffs)
    
    def random_search():
        nonlocal best_coeff, best_value
        for _ in range(100):
            coeffs = np.random.uniform(-1, 1, 8)
            val = objective(coeffs)
            if val < best_value:
                best_value = val
                best_coeff = coeffs.copy()
    
    def gradient_search():
        nonlocal best_coeff, best_value
        initial_guess = np.zeros(8)
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(-1,1)]*8)
        if result.fun < best_value:
            best_value = result.fun
            best_coeff = result.x
    
    def fine_tune():
        nonlocal best_coeff, best_value
        for i in range(8):
            def f(x):
                coeffs = best_coeff.copy()
                coeffs[i] = x
                return evaluate_coefficients(coeffs)
            result = minimize_scalar(f, bounds=(-1,1), method='bounded')
            if result.fun < best_value:
                best_coeff[i] = result.x
                best_value = result.fun
    
    random_search()
    gradient_search()
    fine_tune()
    
    if time.time() - start_time < budget_s - 1:
        # Additional refinement
        for _ in range(5):
            def f(x):
                coeffs = best_coeff.copy()
                coeffs[0] = x
                return evaluate_coefficients(coeffs)
            result = minimize_scalar(f, bounds=(-1,1), method='bounded')
            if result.fun < best_value:
                best_coeff[0] = result.x
                best_value = result.fun
    
    return best_coeff
