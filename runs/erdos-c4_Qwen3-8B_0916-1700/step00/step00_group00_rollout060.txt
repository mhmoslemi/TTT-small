from scipy.optimize import minimize

from sympy import symbols, hermite, simplify, roots
from sympy.polys import Poly
import time

def run(seed=42, budget_s=200, **kwargs):

    c0 = np.random.rand(8)

    start_time = time.time()

            Q += c[k] * hermite(4*k, z)

        # Filter out even multiplicity roots
        unique_roots = []

            if abs(r) > 1e-8:
                if not unique_roots or abs(r - unique_roots[-1]) > 1e-8:
                    unique_roots.append(r)
        if not unique_roots:
            return float('inf')
        r_max = max(unique_roots)
        return r_max**2 / (2 * np.pi)

        return evaluate(coeffs)
