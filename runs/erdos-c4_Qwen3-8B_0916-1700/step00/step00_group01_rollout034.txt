import numpy as np
import scipy.optimize as opt
from sympy import symbols, Hermite, solve, real_roots, pi

def run(seed=42, budget_s=200):
    np.random.seed(seed)
    c = np.random.rand(8)
    for _ in range(100):
        try:
            def objective(c):
                coeffs = c
                z = symbols('z')
                Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(8))
                Q = Q - Q.as_leading_coefficient() * z**2
                roots = real_roots(Q)
                r_max = max(roots) if roots else 1
                return r_max**2 / (2 * pi)
            result = opt.minimize(objective, c, method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)])
            c = result.x
        except:
            break
    return c
