import numpy as np
import scipy.optimize as opt
import scipy.special as sp

def run(seed=42, budget_s=200, **kwargs):

            return sp.hermt(n, z)

        return r_max**2 / (2 * np.pi)
    
    # Initial grid search
    best_coeffs = None
