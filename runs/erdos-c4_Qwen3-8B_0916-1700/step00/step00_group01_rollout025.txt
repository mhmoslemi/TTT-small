import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, fsolve
from sympy import symbols, hermite, solve, Rational, pi
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(c):
        c = np.array(c, dtype=np.float64)
        # Build Hermite polynomial
        z = symbols('z')
        H = [hermite(k, z) for k in range(8)]
        Q = sum(c[k] * H[k] for k in range(8))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positive at infinity
        if Q.as_leading_coefficient() < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q.as_poly().remove_factor(z**2)
        Q = Q.as_expr()
        # Find r_max
        try:
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            # Find largest root with sign change
            real_roots.sort()
            r_max = None
            for i in range(len(real_roots)-1):
                if Q.subs(z, real_roots[i]) * Q.subs(z, real_roots[i+1]) < 0:
                    r_max = real_roots[i]
                    break
            if r_max is not None:
                return r_max**2 / (2 * pi)
            return float('inf')
        except:
            return float('inf')
    
    # Initial guess
    best_coeff = np.random.rand(8)
    best_value = objective(best_coeff)
    
    # Gradient descent with restarts
    for _ in range(10):
        start_time = time.time()
        result = minimize(lambda c: objective(c), best_coeff, method='L-BFGS-B', bounds=[(0, 100) for _ in range(8)], options={'maxiter': 100})
        if time.time() - start_time > budget_s:
            break
        if objective(result.x) < best_value:
            best_value = objective(result.x)
            best_coeff = result.x
    
    return best_coeff
