import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, simplify, roots, Poly, S

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    c_count = 8
    best_coeff = None
    best_value = float('inf')
    
    def objective(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(c_count))
        Q = Q.as_expr()
        Q = Q.subs(z, 0)
        c_m = -Q / (HermiteH(0, 0) if c_count > 0 else 1)
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(c_count)) + c_m * HermiteH(0, z)
        Q = Q - Q.as_poly().coeffs()[0] * z**2
        Q = Q.as_poly()
        roots_list = roots(Q, z)
        real_roots = [r for r in roots_list.keys() if r.is_real]
        real_roots = sorted(r for r in real_roots if r > 0)
        r_max = real_roots[-1] if len(real_roots) > 0 else 1.0
        value = r_max**2 / (2 * np.pi)
        return value
    
    def surrogate_objective(coeffs):
        z = np.linspace(-100, 100, 10000)
        c = coeffs
        Q = np.zeros_like(z)
        for k in range(c_count):
            H = np.vectorize(lambda x: HermiteH(4*k, x))
            Q += c[k] * H(z)
        Q = Q - Q[0] * z**2
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = np.max(z[sign_changes])
        return r_max**2 / (2 * np.pi)
    
    for _ in range(20):
        start_time = time.time()
        initial = np.random.uniform(-1, 1, c_count)
        result = minimize(surrogate_objective, initial, method='L-BFGS-B', bounds=[(-1, 1) for _ in range(c_count)])
        coeffs = result.x
        val = objective(coeffs)
        if val < best_value:
            best_value = val
            best_coeff = coeffs
        if time.time() - start_time > budget_s / 2:
            break
    
    return best_coeff
