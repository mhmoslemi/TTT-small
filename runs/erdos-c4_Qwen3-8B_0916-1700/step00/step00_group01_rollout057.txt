import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, roots, simplify, Rational
from time import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    coeffs = np.random.randn(8)
    best_coeffs = coeffs.copy()
    best_bound = float('inf')
    
    def evaluate_coeffs(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q.as_independent(z)[1].subs(z, 0)
        Q = Q / z**2
        roots_list = roots(Q, z, multiple=False)
        real_roots = [r.evalf() for r in roots_list if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return r_max**2 / (2*np.pi)
    
    def surrogate(c):
        return evaluate_coeffs(c)
    
    def optimize():
        nonlocal best_coeffs, best_bound
        res = minimize(lambda c: surrogate(c), x0=best_coeffs, method='L-BFGS-B', bounds=[(-2, 2) for _ in range(8)])
        if res.success:
            current_bound = res.fun
            if current_bound < best_bound:
                best_bound = current_bound
                best_coeffs = res.x
    
    start_time = time()
    while time() - start_time < budget_s:
        optimize()
    
    return best_coeffs
