import numpy as np
import time
from scipy.optimize import root_scalar
from scipy.optimize._numdiff import approx_derivative
from sympy import symbols, hermite, Rational, Poly, roots

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_upper = float('inf')
    time_start = time.time()
    
    def evaluate(coeffs):
        nonlocal best_upper
        if len(coeffs) != 8:
            return float('inf')
        
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(8))
        
        # Enforce Q(0) = 0
        Q0 = Q.subs(z, 0)
        if Q0 != 0:
            Q = Q - Q0
        
        # Ensure positivity at infinity
        leading_term = Q.as_leading_term(z)
        if leading_term.as_poly().coeffs()[0] < 0:
            Q = -Q
        
        # Remove z^2 factor
        Q = Q / z**2
        Q = Q.simplify()
        
        # Find real roots of Q(z)
        try:
            roots_list = roots(Q, real=True)
        except:
            return float('inf')
        
        real_roots = [r for r in roots_list if r.is_real]
        real_roots = np.sort(real_roots)
        
        # Find sign change across positive real roots
        r_max = None
        for i in range(len(real_roots) - 1):
            r1, r2 = real_roots[i], real_roots[i+1]
            if np.isclose(r1, r2):
                continue
            if r1 < 0:
                continue
            if r2 < 0:
                continue
            # Check sign change between r1 and r2
            def f(x):
                return Q.subs(z, x).evalf()
            try:
                val1 = f(r1)
                val2 = f(r2)
                if np.sign(val1) != np.sign(val2):
                    r_max = r2
                    break
            except:
                pass
        
        if r_max is None:
            return float('inf')
        
        upper = r_max**2 / (2 * np.pi)
        return upper
    
    def objective(coeffs):
        return evaluate(coeffs)
    
    for _ in range(20):
        if time.time() - time_start > budget_s - 1:
            break

        best_coeff = coeffs
        best_upper = evaluate(coeffs)
        
        for _ in range(100):
            if time.time() - time_start > budget_s - 1:
                break

            best_coeff -= 0.1 * grad
            current_upper = evaluate(best_coeff)
            if current_upper < best_upper:
                best_upper = current_upper
                best_coeff = best_coeff.copy()
    
    return best_coeff.tolist()
