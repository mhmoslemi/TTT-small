import time
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from numpy.polynomial.hermite_e import hermeval, hermeder

def hermite_eval(z, coeffs):

def c4_bound(coeffs):

        Q0 = Q(0)

        except:
            return float('inf')
    except:
        return float('inf')

def run(seed=42, budget_s=200, **kwargs):
