import numpy as np
import sympy as sp
from scipy.optimize import minimize
from itertools import product
from time import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_bound = float('inf')
    start_time = time()
    
    def objective(coeffs):
        nonlocal best_bound
        if time() - start_time > budget_s - 1:
            return best_bound
        
        # Convert coeffs to symbolic variables
        c = coeffs
        z = sp.symbols('z')
        
        # Compute Hermite polynomials H_{4k}(z)
        H = [sp.hermite(n)(z) for n in range(len(c))]
        
        # Construct Q(z)
        Q = sum(c[k] * H[k] for k in range(len(c)))
        
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        
        # Make Q positive at infinity by flipping sign if necessary
        if Q.as_infinity().is_positive:
            pass
        else:
            Q = -Q
        
        # Remove z^2 factor
        Q = Q / z**2
        
        # Find r_max
        expr = Q
        roots = sp.solvers.roots(expr)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted(real_roots)
        
        r_max = None
        for r in real_roots:
            if r > 0 and expr.subs(z, r).evalf() * expr.subs(z, r + 1e-6).evalf() < 0:
                r_max = r
                break
        
        if r_max is None:
            return float('inf')
        
        bound = r_max**2 / (2 * np.pi)
        if bound < best_bound:
            best_bound = bound
            best_coeffs = coeffs
        return bound
    
    # Initial random search
    for _ in range(10):
        coeffs = np.random.uniform(-1, 1, size=8)

            break

            break

        res = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-1, 1) for _ in range(8)])
    
    return best_coeffs
