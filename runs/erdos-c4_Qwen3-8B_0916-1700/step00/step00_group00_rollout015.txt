from sympy import symbols, Hermite, Poly, Rational, diff, solve, roots, simplify
from time import time

def run(seed=42, budget_s=200, **kwargs):

    best_bound = float('inf')
    
    def evaluate_coefficients(coeffs):

            # Find sign change
            sign_changes = []
            for r in real_roots:
                if r > 0:
                    sign_changes.append(r)
            if not sign_changes:

            return float('inf')
