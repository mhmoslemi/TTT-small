import numpy as np
from scipy.optimize import root_scalar
import sympy as sp
from itertools import product
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_bound = float('inf')
    time_start = time.time()
    
    def eval_U(coeffs):
        z = sp.symbols('z')
        H = [sp.hermite(h) for h in range(4*len(coeffs)+1)]

        roots = sp.roots(Q.as_poly())
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted(r for r in real_roots if r > 0)
        for r in real_roots:
            if r > 0:
                if sp.sign(Q.subs(z, r - 1e-3)) * sp.sign(Q.subs(z, r + 1e-3)) < 0:
                    return r**2 / (2*np.pi)
        return float('inf')
    
    def fast_surrogate(coeffs):
        z = sp.symbols('z')
        H = [sp.hermite(h) for h in range(4*len(coeffs)+1)]
        Q = sum(c * H[4*i] for i, c in enumerate(coeffs))
        Q = Q - Q.as_poly().coeffs()[0] * z**2
        try:
            r = root_scalar(lambda x: Q.subs(z, x), bracket=[1e-3, 100], method='brentq')
