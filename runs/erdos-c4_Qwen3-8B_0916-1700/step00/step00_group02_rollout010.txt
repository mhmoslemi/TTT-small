Use gradient descent with a fast numerical surrogate to find promising coefficient vectors, then validate with exact rational arithmetic via `evaluate_c4_coefficients`. Focus on low C₄ bounds by balancing Hermite coefficients to minimize r_max²/(2π).

import numpy as np
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_surrogate(c):
        m = len(c)
        z = np.linspace(-10, 10, 10000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += c[k] * hermite(4*k, z)
        Q /= z**2
        sign_changes = np.count_nonzero(np.sign(Q[1:] - Q[:-1]) != 0)
        r_max = 0.0
        for i in range(len(z) - 1):
            if np.sign(Q[i]) != np.sign(Q[i+1]):
                r_max = np.max(z[np.abs(Q) > 1e-6])
                break
        return r_max**2 / (2 * np.pi)
    
    def hermite(n, z):
        if n == 0:
            return np.ones_like(z)
        elif n == 1:
            return 2*z
        else:
            return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
    
    best_c = np.random.rand(8)
    best_score = evaluate_surrogate(best_c)
    
    for _ in range(50):
        start_time = time.time()
        grad = np.zeros(8)
        for i in range(8):
            c_plus = best_c.copy()
            c_plus[i] += 1e-5
            c_minus = best_c.copy()
            c_minus[i] -= 1e-5
            grad[i] = (evaluate_surrogate(c_plus) - evaluate_surrogate(c_minus)) / (2e-5)
        
        step = 0.1 * np.sign(grad)
        new_c = best_c - step
        
        new_score = evaluate_surrogate(new_c)
        if new_score < best_score:
            best_c = new_c
            best_score = new_score
        
        if time.time() - start_time > budget_s / 2:
            break
    
    return best_c
