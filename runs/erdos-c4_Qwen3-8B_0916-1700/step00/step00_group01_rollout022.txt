import numpy as np
from scipy.optimize import minimize
from functools import lru_cache

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    coeffs = np.random.rand(8)
    coeffs /= np.sum(np.abs(coeffs))
    
    def surrogate(coeffs):
        def hermite(n, z):
            if n == 0:
                return 1
            elif n == 1:
                return 2*z
            return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
        
        def eval_Q(z):
            res = 0
            for k in range(len(coeffs)):
                res += coeffs[k] * hermite(4*k, z)
            return res
        
        def find_r_max():
            def f(z):
                return eval_Q(z) / (z**2)
            
            try:
                from scipy.optimize import root_scalar
                sol = root_scalar(f, bracket=(0, 100), method='brentq')
                return sol.root**2 / (2 * np.pi)
            except:
                return 100
        
        return find_r_max()
    
    result = minimize(lambda c: -surrogate(c), x0=coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)], options={'gtol': 1e-7, 'maxiter': 100})
    best_coeffs = result.x
    return best_coeffs
