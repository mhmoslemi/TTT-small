import numpy as np
import random
from scipy.optimize import minimize
from sympy import symbols, HermiteH, Poly, solve, Rational, N, pi

def run(seed=42, budget_s=200):
    random.seed(seed)
    z = symbols('z')
    max_coeff_count = 8
    
    def evaluate_coefficients(coeffs):
        m = len(coeffs)
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_poly()
        Q = Q.subs(z, 0)
        if Q.coeffs()[0] != 0:
            Q = -Q
        Q = Q.as_poly()
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [N(r) for r in real_roots]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if r > 0:
                possible_r_max = r
                break
        else:
            possible_r_max = 1e-6
        return possible_r_max
    
    def objective(coeffs):
        r_max = evaluate_coefficients(coeffs)
        return r_max ** 2 / (2 * pi)

    best_coeffs = None
    best_bound = float('inf')
    
    for _ in range(20):
        initial_guess = [random.uniform(-1, 1) for _ in range(max_coeff_count)]
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(-1, 1) for _ in range(max_coeff_count)], options={'maxiter': 100})
        if result.success:
            current_bound = result.fun
            if current_bound < best_bound:
                best_bound = current_bound
                best_coeffs = result.x
    
    return best_coeffs
