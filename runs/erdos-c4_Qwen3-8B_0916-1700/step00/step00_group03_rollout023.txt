import numpy as np
from scipy.optimize import minimize, root_scalar
from scipy.special import hermiti

def run(seed=42, budget_s=200, **kwargs):

    def objective(c):
        coeffs = np.array(c)
        # Compute Q(z) = sum c_k H_{4k}(z)
        def Q(z):
            val = 0
            for k in range(len(coeffs)):
                val += coeffs[k] * hermiti(4*k, z)
            return val
        # Ensure Q(0) = 0 and positive at infinity

        sign = 1
        try:
            # Check sign at large z
            sign = 1 if Q(10) > 0 else -1
        except:
            sign = 1

            return sign * Q(z)
        # Remove z^2 factor by finding r_max

            return adjusted_Q(z) / (z**2)
        # Find largest real root where f(z) changes sign
        left = 0.1
        right = 100
        try:
            # Use root_scalar to find sign change
            sol = root_scalar(f, bracket=[left, right], method='brentq')
            r_max = sol.root
            # Check if r_max is positive and f crosses zero
            if r_max > 0 and f(r_max) * f(0.1) < 0:
                return r_max**2 / (2 * np.pi)
        except:
            pass
        return 1.0  # Fallback

    # Use gradient descent with random restarts
    best_coeffs = None
    best_score = float('inf')

    for _ in range(20):
