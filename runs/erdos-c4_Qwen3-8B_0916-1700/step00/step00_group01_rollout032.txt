import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import Bounds
from scipy.optimize.linesearch import line_search_wolfe
from scipy.optimize.minimize import MINIMIZE_ALL
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    start_time = time.time()

    def numerical_surrogate(coeffs):
        coeffs = np.array(coeffs)
        m = len(coeffs)
        z = np.linspace(-10, 10, 1000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * np.polynomial.hermite.hermite_e(k, z)
        Q /= z**2
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = z[sign_changes[-1]] if sign_changes.size else 10.0
        return r_max**2 / (2 * np.pi)

    def objective(coeffs):
        return numerical_surrogate(coeffs)

    def refine_coefficients(coeffs):
        coeffs = np.array(coeffs)
        m = len(coeffs)
        z = np.linspace(-10, 10, 1000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * np.polynomial.hermite.hermite_e(k, z)
        Q /= z**2
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = z[sign_changes[-1]] if sign_changes.size else 10.0
        return r_max**2 / (2 * np.pi)

    for _ in range(20):
        initial = np.random.uniform(-1, 1, 8)
        result = minimize(objective, initial, method='L-BFGS-B', bounds=[(-1, 1)]*8, options={'ftol': 1e-8, 'maxiter': 100})
        final_coeffs = result.x
        final_bound = refine_coefficients(final_coeffs)
        if final_bound < best_bound:
            best_bound = final_bound
            best_coeff = final_coeffs
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_coeff
