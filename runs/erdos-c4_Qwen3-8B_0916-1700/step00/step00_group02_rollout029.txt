from scipy.optimize import minimize

        """Approximate Hermite polynomial using recurrence."""
        H = np.zeros_like(z)
        if n == 0:
            H[:] = 1
        elif n == 1:
            H[:] = 2*z
        else:
            H_prev = np.copy(H)
            H_prev2 = np.copy(H)

            H = H_prev
        return H
