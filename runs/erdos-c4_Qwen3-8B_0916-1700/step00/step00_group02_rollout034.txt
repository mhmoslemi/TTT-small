import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    def surrogate(coeffs):
        return np.sum(coeffs ** 2)
    
    def exact_eval(coeffs):
        from sympy import symbols, HermiteH, solve, Poly, roots
        z = symbols('z')
        Q = sum(c * HermiteH(4*k, z) for k, c in enumerate(coeffs))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        if Q.as_poly().all_coeffs()[-1] < 0:
            Q *= -1
        r_roots = roots(Q.as_poly(), z)
        real_roots = [r for r in r_roots.as_real_imag() if r[0].is_real]
        real_roots = sorted([r for r in real_roots if r != 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1][0]
        if r_max <= 0:
            return float('inf')
        return r_max ** 2 / (2 * np.pi)
    
    def objective(coeffs):
        return exact_eval(coeffs)
    
    def render(coeffs):
        return coeffs.tolist()
    
    def num_eval(coeffs):
        return exact_eval(coeffs)
    
    def evaluate_c4_coefficients(coeffs):
        return num_eval(coeffs)
    
    def inner_search():
        result = minimize(surrogate, np.random.rand(8), method='L-BFGS-B', bounds=[(-10, 10)]*8, options={'maxiter': 50})
        return result.x
    
    start_time = time.time()
    best_coeffs = inner_search()
    best_score = evaluate_c4_coefficients(best_coeffs)
    elapsed = time.time() - start_time
    
    while elapsed < budget_s - 10:
        new_coeffs = best_coeffs + np.random.normal(0, 0.1, 8)
        new_coeffs = np.clip(new_coeffs, -10, 10)
        score = evaluate_c4_coefficients(new_coeffs)
        if score < best_score:
            best_coeffs = new_coeffs
            best_score = score
        elapsed = time.time() - start_time
    
    return render(best_coeffs)
