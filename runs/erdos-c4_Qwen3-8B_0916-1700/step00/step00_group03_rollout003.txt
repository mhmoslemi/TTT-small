import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def numerical_surrogate(coeffs):
        c = np.array(coeffs)
        c = np.concatenate((c, [0]))
        def hermite_4k(k):
            return np.polyspace.her_moments(4*k, 100)
        H = np.array([hermite_4k(k) for k in range(len(c))])
        Q = np.dot(c, H)
        try:
            roots = np.roots(Q)
            real_roots = np.real(roots[np.isfinite(roots)])
            positive_roots = real_roots[real_roots > 0]
            positive_roots = np.sort(positive_roots)
            idx = np.argmax(np.sign(np.polyval(Q, positive_roots[1:])))
            r_max = positive_roots[idx]
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    def objective(x):
        return numerical_surrogate(x)
    
    # Initial guess
    x0 = np.random.rand(8)
    result = minimize(objective, x0, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)], tol=1e-10, options={'maxiter': 200})
    return result.x.tolist()
