import numpy as np
import scipy.optimize as opt
import sympy as sp
from itertools import product

def run(seed=42, budget_s=200, **kwargs):

    best_coeff = None

        c = np.array(c, dtype=np.float64)
        if len(c) != 8:
            return float('inf')
        
        z = sp.symbols('z')

            Q = -Q
        
        # Remove z^2 factor
        Q = Q / z**2
        
        # Find r_max
        try:
            roots = sp.solvers.solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            
            # Sort and find sign change
            real_roots.sort()
            sign_changes = []
            for i in range(len(real_roots)-1):
                if np.sign(Q.subs(z, real_roots[i])) != np.sign(Q.subs(z, real_roots[i+1])):

                bound = r_max**2 / (2 * np.pi)
                return bound
            else:
                return float('inf')

    # Gradient descent with random restarts
