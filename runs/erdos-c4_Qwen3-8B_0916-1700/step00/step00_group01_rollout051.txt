import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Eq, solve, Hermite, Poly, roots, simplify
import time

def hermite_eval(z, coeffs):
    z = np.array(z, dtype=float)
    res = np.zeros_like(z)
    for k, c in enumerate(coeffs):
        h = Hermite(4*k)(z)
        res += c * h
    return res

def calculate_r_max(coeffs):
    z = symbols('z')
    poly = sum(c * Hermite(4*k)(z) for k, c in enumerate(coeffs))
    poly = poly - poly.as_coefficients_dict()[z**2] * z**2
    roots = solve(poly, z)
    real_roots = [r.evalf() for r in roots if r.is_real]
    real_roots = sorted([r for r in real_roots if r > 0])
    for r in real_roots:
        if np.isclose(poly.subs(z, r), 0, atol=1e-10):
            return r**2
    return np.inf

def run(seed=42, budget_s=200):
    np.random.seed(seed)
    initial_coeffs = np.random.rand(8)
    best_coeffs = initial_coeffs
    best_bound = np.inf

    def objective(coeffs):
        nonlocal best_coeffs, best_bound
        start_time = time.time()
        if time.time() - start_time > budget_s:
            return best_bound
        coeffs = np.clip(coeffs, 0, 1)
        try:
            r_max_sq = calculate_r_max(coeffs)
            bound = r_max_sq / (2 * np.pi)

                best_bound = bound
                best_coeffs = coeffs
            return bound
