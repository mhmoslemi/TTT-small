import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermitePolynomial, simplify, Poly, solve, Eq, Rational
from sympy.abc import z

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_value = float('inf')
    start_time = time.time()

    def objective(coeffs):
        nonlocal best_value
        if time.time() - start_time > budget_s:
            return best_value
        c = coeffs
        # Build Q(z)
        z_sym = symbols('z')
        H = [HermitePolynomial(4*k, z_sym) for k in range(len(c))]
        Q = sum(c[k] * H[k] for k in range(len(c)))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z_sym, 0)
        # Check positivity at infinity
        leading_term = Q.as_coefficients_dict().get(z_sym**len(c), 0)
        if leading_term < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q / z_sym**2
        # Find r_max: largest positive real root of Q(z) changing sign
        poly = Poly(Q, z_sym)
        roots = solve(poly, z_sym)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            # Check sign change
            val0 = Q.subs(z_sym, r - 1e-6)
            val1 = Q.subs(z_sym, r + 1e-6)
            if val0 * val1 < 0:
                r_max = r
                break
        else:
            r_max = 1
        U = r_max**2 / (2 * np.pi)
        return U

    def evaluate(coeffs):
        return objective(coeffs)

    # Initial guess
    initial_guess = np.random.rand(8)
    result = minimize(
        lambda x: evaluate(x),
        initial_guess,
        method='L-BFGS-B',
        bounds=[(None, None) for _ in range(8)],
        options={'maxiter': 1000}
    )

    best_coeff = result.x
    best_value = result.fun

    # Check a few random restarts
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        guess = np.random.rand(8)
        res = minimize(
            lambda x: evaluate(x),
            guess,
            method='L-BFGS-B',
            bounds=[(None, None) for _ in range(8)],
            options={'maxiter': 100}
        )
        if res.fun < best_value:
            best_value = res.fun
            best_coeff = res.x

    return best_coeff
