import time
from scipy.optimize import minimize
from sympy import symbols, HermitePolynomial, simplify, solve, roots, S, pi

def run(seed=42, budget_s=200):
    np.random.seed(seed)
    z = symbols('z')
    H = [HermitePolynomial(4*k, z) for k in range(8)]
    
    def hermite_eval(coeffs):
        expr = sum(c * H[k] for k, c in enumerate(coeffs))
        expr = expr.subs(z, 0)
        if expr != 0:
            expr = -expr
        expr = expr / (z**2)
        expr = expr.expand()
        roots = solve(expr, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r != 0]
        real_roots = sorted([r for r in real_roots if r > 0])
        max_root = max(real_roots)
        return max_root**2 / (2 * pi)
    
    def evaluate_c4_coefficients(coeffs):

    def surrogate_eval(coeffs):
        return hermite_eval(coeffs)
    
    initial_guess = np.random.rand(8)
    best_coeffs = initial_guess.copy()
    best_value = surrogate_eval(best_coeffs)
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break

            return surrogate_eval(x)
