import numpy as np
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    def fitness(coeffs):
        def surrogate(coeffs):
            try:
                Q = np.poly1d([0]*2 + [1])  # Placeholder
                for k in range(len(coeffs)):
                    H = np.polynomial.hermite.hermfromroots([-1, 1]) * np.poly1d([1, 0])
                    if k == 0:
                        Q = coeffs[k] * H
                    else:
                        Q += coeffs[k] * H**k
                # This is a placeholder; actual computation is done in evaluate_c4_coefficients
                return 1.0 / (np.sqrt(np.sum(coeffs**2)) + 1e-6)
            except:
                return np.inf
        return surrogate(coeffs)
    
    def generate_initial_population(size=50):
        return [np.random.uniform(-1, 1, 8) for _ in range(size)]
    
    def crossover(a, b):
        return [random.choice([a[i], b[i]]) for i in range(8)]
    
    def mutate(individual, rate=0.1):
        for i in range(len(individual)):
            if random.random() < rate:
                individual[i] += np.random.normal(0, 0.1)
        return individual
    
    population = generate_initial_population()
    best = population[0]
    best_fitness = fitness(best)
    
    for _ in range(100):
        if budget_s <= 0:
            break
        candidates = []
        for ind in population:
            candidates.append((fitness(ind), ind))
        candidates.sort()
        population = [c[1] for c in candidates[:20]]
        next_gen = []
        for _ in range(10):
            a, b = random.sample(population, 2)
            child = crossover(a, b)
            child = mutate(child)
            next_gen.append(child)
        population = next_gen
        current_best = min(population, key=fitness)
        if fitness(current_best) < best_fitness:
            best = current_best
            best_fitness = fitness(best)
        budget_s -= 10
    
    return best
