import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

# Define Hermite polynomials
def hermite(n, z):
    return sp.hermite(n, z)

def evaluate_Q(coeffs, z):
    m = len(coeffs)
    Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
    return Q

def find_r_max(Q, z):
    # Remove z^2 factor and find largest positive real root where Q/z^2 changes sign
    Q_z2 = Q / z**2
    roots = sp.solvers.roots(Q_z2)
    real_roots = [r for r in roots if r.is_real and r > 0]
    real_roots = sorted(real_roots)
    for r in real_roots:
        if Q_z2.subs(z, r).evalf() != 0:
            return r
    return None

def objective(coeffs):
    z = sp.symbols('z')
    Q = evaluate_Q(coeffs, z)
    # Enforce Q(0) = 0
    Q = Q - Q.subs(z, 0)
    # Find r_max
    r_max = find_r_max(Q, z)
    if r_max is None:
        return float('inf')
    return r_max**2 / (2*np.pi)

def run(seed=42, budget_s=200):
    np.random.seed(seed)
    # Initial guess
    coeffs = np.random.rand(8)
    # Optimization
    result = minimize(objective, coeffs, method='L-BFGS-B', options={'maxiter': 100})
    # Return best coefficients
    return result.x
