import random
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, Poly, roots

def evaluate_c4_coefficients(coefficients):
    z = symbols('z')
    Q = sum(c * HermiteH(4*k, z) for k, c in enumerate(coefficients))
    Q = Q.as_poly().as_expr()

    try:
        r = root_scalar(lambda x: Q.subs(z, x), bracket=[0, 100], method='brentq')
        return r.root**2 / (2 * np.pi)
    except:
        return float('inf')
