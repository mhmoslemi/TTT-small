import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Poly, diff
from time import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    
    def evaluate_coefficients(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.as_expr()
        Q = Q - Q.subs(z, 0)  # Ensure Q(0) = 0
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def score_coefficients(c):
        try:
            return evaluate_coefficients(c)
        except:
            return float('inf')
    
    def objective(c):
        return score_coefficients(c)
    
    initial_guess = np.random.rand(C4_MAX_COEFF_COUNT)
    
    # Gradient descent with numerical optimization
    def minimize_with_restart():
        best_coeff = initial_guess.copy()
        best_score = objective(best_coeff)
        
        for _ in range(10):
            start_time = time()
            res = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(0, 10) for _ in range(C4_MAX_COEFF_COUNT)])
            elapsed = time() - start_time
            if elapsed > budget_s / 2:
                break
            
            if res.fun < best_score:
                best_score = res.fun
                best_coeff = res.x
        
        return best_coeff

    best_coeff = minimize_with_restart()
    return best_coeff.tolist()
