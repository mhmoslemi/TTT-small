from scipy.optimize import minimize, root_scalar
from scipy.optimize.optimize import OptimizeResult
from scipy.special import hermite
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        m = len(coeffs)
        def Q(z):
            res = 0
            for k in range(m):
                res += coeffs[k] * hermite(4*k, z)
            return res
        
        # Ensure Q(0) = 0 and positive at infinity
        Q0 = Q(0)
        if Q0 != 0:
            coeffs = coeffs * (Q(0) ** -1)
        
        # Find r_max
        def f(z):
            return Q(z) / (z**2)
        
        try:
            r_max = root_scalar(f, bracket=[1, 100], method='brentq').root
            return r_max**2 / (2 * np.pi)
        except:
            return np.inf
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Initial grid search
    grid = np.linspace(-1, 1, 100)
    best_score = np.inf
    best_coeffs = np.zeros(8)
    
    for i in range(8):
        for j in range(8):
            coeffs = np.zeros(8)
            coeffs[i] = grid[i]
            coeffs[j] = grid[j]
            score = objective(coeffs)
            if score < best_score:
                best_score = score
                best_coeffs = coeffs
    
    # Refine with L-BFGS-B
    bounds = [(-1, 1) for _ in range(8)]
    result = minimize(
        fun=objective,
        x0=best_coeffs,
        bounds=bounds,
        method='L-BFGS-B',

    )
    
    return result["x"]
