import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(x):
        return evaluate_c4_coefficients(x)
    
    def fitness(x):
        try:
            return evaluate_c4_coefficients(x)
        except:
            return np.inf
    
    population_size = 20
    genetic_params = {
        'mutation_rate': 0.1,
        'mutation_strength': 0.5,
        'tournament_size': 5,
        'num_elites': 2,
        'max_iterations': int(budget_s / 10)
    }
    
    def crossover(parents):
        offspring = []
        for _ in range(population_size - genetic_params['num_elites']):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = (p1 + np.random.rand() * (p2 - p1)) * (1 - np.random.rand())
            offspring.append(child)
        return offspring
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if np.random.rand() < genetic_params['mutation_rate']:
                mutated[i] += np.random.normal(0, genetic_params['mutation_strength'])
                mutated[i] = np.clip(mutated[i], -1, 1)
        return mutated
    
    parents = [np.random.rand(8) * 2 - 1 for _ in range(population_size)]
    
    for _ in range(genetic_params['max_iterations']):
        fitnesses = np.array([fitness(ind) for ind in parents])
        sorted_indices = np.argsort(fitnesses)
        parents = np.array(parents)[sorted_indices]
        
        elites = parents[:genetic_params['num_elites']]
        offspring = crossover(parents)
        offspring = [mutate(ind) for ind in offspring]
        parents = np.concatenate((elites, offspring))
    
    best = parents[np.argmin(fitnesses)]
    return best
