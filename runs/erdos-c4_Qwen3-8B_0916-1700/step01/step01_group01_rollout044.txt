import numpy as np
from scipy.optimize import Bounds
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    def evaluate_coeffs(coeffs):
        H = np.polyspace(4, len(coeffs))
        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q /= np.polyint(Q, len(coeffs)-2)
        roots = np.roots(np.polyder(Q, len(coeffs)-2))
        real_roots = roots[np.isreal(roots)]
        real_roots = real_roots[real_roots > 0]
        r_max = np.max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def exact_evaluate(coeffs):

        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q /= np.polyint(Q, len(coeffs)-2)
        roots = np.roots(np.polyder(Q, len(coeffs)-2))
        real_roots = roots[np.isreal(roots)]
        real_roots = real_roots[real_roots > 0]
        r_max = np.max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def numerical_surrogate(coeffs):
        return evaluate_coeffs(coeffs)
    
    def grid_search():
        grid = np.logspace(-3, 3, 100)
        grid = np.concatenate((grid, np.logspace(-2, 2, 100)))
        grid = np.unique(grid)
        best = np.full(8, 1.0)
        best_val = float('inf')
        for i in range(8):
            best[i] = grid[np.random.randint(len(grid))]
        for _ in range(100):
            for i in range(8):
                idx = np.random.randint(len(grid))
                best[i] = grid[idx]
            val = numerical_surrogate(best)
            if val < best_val:
                best_val = val
                return best
        return best
    
    def refine_candidate(coeffs):
        for _ in range(10):
            H = np.polyspace(4, len(coeffs))
            Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
            Q /= np.polyint(Q, len(coeffs)-2)
            roots = np.roots(np.polyder(Q, len(coeffs)-2))
            real_roots = roots[np.isreal(roots)]
            real_roots = real_roots[real_roots > 0]
            r_max = np.max(real_roots)
            if r_max < 0.1:
                continue
            H_ = np.polyspace(4, len(coeffs))
            Q_ = np.sum(coeffs[:, np.newaxis] * H_, axis=0)
            Q_ /= np.polyint(Q_, len(coeffs)-2)
            Q_ = Q_ / Q_[np.isreal(Q_)]
            coeff_diff = np.gradient(Q_)
            coeff_diff = coeff_diff / np.max(np.abs(coeff_diff))

            coeffs = np.clip(coeffs, 0.001, 100)
        return coeffs

    initial = np.random.choice(initial, size=8, replace=True)

                      tol=1e-8)
    final = result.x
    final = np.clip(final, 0.001, 100)
    final = refine_candidate(final)
    final = np.clip(final, 0.001, 100)
    return final
