import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    initial_guess = np.random.rand(8)
    
    def numerical_surrogate(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        H = np.polyspace(4, m)
        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q /= np.polyint(Q, m-2)
        poly = np.polynomial.Polynomial.fromroots(np.roots(np.polyder(Q, m-2)))
        real_roots = np.real(poly.roots())
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def exact_evaluator(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        H = np.polyspace(4, m)
        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q /= np.polyint(Q, m-2)
        poly = np.polynomial.Polynomial.fromroots(np.roots(np.polyder(Q, m-2)))
        real_roots = np.real(poly.roots())
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return exact_evaluator(coeffs)
    
    def evaluate_c4_coefficients(coeffs):
        return exact_evaluator(coeffs)
    
    def multi_start_search():
        best = np.inf
        best_coeffs = None
        for _ in range(10):
            guess = np.random.rand(8)
            result = minimize(objective, x0=guess, method='L-BFGS-B',

    start_time = time.time()
