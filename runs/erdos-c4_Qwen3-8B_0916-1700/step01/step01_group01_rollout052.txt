import numpy as np
import sympy as sp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    z = sp.symbols('z')
    H = [sp.herite(n, z) for n in range(4*8 + 1)]
    
    def evaluate_coeffs(coeffs):
        Q = sum(c * H[4*i] for i, c in enumerate(coeffs))
        Q = Q / sp.Poly(Q).integrated()
        roots = sp.roots(Q)
        real_roots = [r for r in roots if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    def numerical_surrogate(coeffs):

        H_vals = [sp.lambdify(z, H[i], 'numpy')(0) for i in range(len(H))]
        Q = sum(c * H_vals[i] for i, c in enumerate(coeffs))
        Q /= np.polyint(Q, len(coeffs)-2)
        root = np.roots(np.polyder(Q, len(coeffs)-2))
        real_roots = root[np.isreal(root)]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_coeffs(coeffs)
    
    def bounds_generator():
        return [(0.01, 10.0) for _ in range(8)]
    
    initial_guess = np.array([1.0 for _ in range(8)])
    result = minimize(objective, x0=initial_guess, method='L-BFGS-B',
