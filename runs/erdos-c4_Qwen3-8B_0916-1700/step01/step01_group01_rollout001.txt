import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, sqrt, pi

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    m = 4
    c = np.random.rand(m)
    
    def objective(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        z = symbols('z')
        H = [Hermite(k, z) for k in range(4*m)]
        Q = sum(coeffs[k] * H[4*k] for k in range(m))
        Q = Q / z**2
        roots = np.roots(np.array([Q.coeff(z**i) for i in range(2*m-1)], dtype=float))
        real_roots = roots[np.isreal(roots)]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    result = minimize(objective, x0=c, method='L-BFGS-B', 
                      options={'ftol': 1e-10, 'maxiter': 100, 'disp': False}, 
                      tol=1e-10)
    return result.x
