import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate(x):
        return evaluate_c4_coefficients(x)
    
    # Initial grid search on log scale
    log_grid = np.logspace(-3, 3, 50)
    best = np.inf
    best_coeffs = np.zeros(8)
    
    for i in range(8):
        for val in log_grid:
            coeff = np.zeros(8)
            coeff[i] = val
            # Try to find good sign via a small perturbation
            for j in range(8):
                if j != i:
                    coeff[j] = np.random.uniform(-1, 1)
            bound = evaluate(coeff)
            if bound < best:
                best = bound
                best_coeffs = coeff.copy()
    
    # Local optimization
    result = minimize(
        evaluate,
        x0=best_coeffs,
        method='L-BFGS-B',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    return result.x
