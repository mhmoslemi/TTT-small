import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, simplify, Eq, diff

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def calc_upper_bound(coeffs):
        z = symbols('z')
        Q = sum(c * HermiteH(4*k, z) for k, c in enumerate(coeffs))
        Q = Q - Q.as_coeff_add()[0][0] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r).evalf() != 0:
                r_max = r
                break
        else:
            r_max = 1
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return calc_upper_bound(x)
    
    # Initial grid search
    grid = np.linspace(-1, 1, 50)
    best = np.array([0.5]*8)
    best_val = float('inf')
    
    for i in range(8):
        for j in range(8):
            if i == j:
                continue
            coeffs = np.zeros(8)
            coeffs[i] = grid[np.random.randint(len(grid))]
            coeffs[j] = grid[np.random.randint(len(grid))]
            val = objective(coeffs)
            if val < best_val:
                best = coeffs
                best_val = val
    
    # Fine-tune with gradient descent
    result = minimize(
        objective,
        x0=best,
        method='L-BFGS-B',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    return result.x
