import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, HermiteH, solve, N, Eq, Poly, Roots

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    x = symbols('x')
    H = [HermiteH(k, x) for k in range(8)]
    
    def exact_evaluate(coeffs):
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q.as_poly()
        Q = Q.as_expr()
        Q = Q / (x**2)
        roots = solve(Q, x)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def numerical_surrogate(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        Q = sum(c * HermiteH(k, x) for k, c in enumerate(coeffs))
        Q = Q / (x**2)
        roots = np.roots([float(Q.as_poly()[i]) for i in range(len(Q.as_poly())-1, -1, -1)])
        real_roots = roots[np.isreal(roots)]
        real_roots = real_roots[real_roots > 0]
        if not real_roots.size:
            return float('inf')
        r_max = np.max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    result = minimize(numerical_surrogate, x0=np.random.rand(8), 
                      method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 100}, 
                      tol=1e-10)
    best = result.x
    for _ in range(5):
        perturb = np.random.normal(0, 0.1, 8)
        candidate = best + perturb
        candidate = np.clip(candidate, -1, 1)
        score = numerical_surrogate(candidate)
        if score < numerical_surrogate(best):
            best = candidate
    return best
