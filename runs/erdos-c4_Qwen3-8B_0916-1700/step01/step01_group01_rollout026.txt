from scipy.optimize import minimize
import time
from sympy import symbols, HermiteH, roots, Rational, expand, simplify

def run(seed=42, budget_s=200, **kwargs):

    m = 8

    def evaluate(coeffs):
        var = symbols('z')
        H = [HermiteH(k, var) for k in range(m)]
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q.subs(var, 0).as_numerical_enumerator()
        Q = Q / (var**2)
        Q = Q.expand()
        roots = roots(Q, var, domain='Real')
        real_roots = [r for r in roots if r > 0]
        r_max = max(real_roots) if real_roots else 1
        return r_max**2 / (2 * np.pi)
