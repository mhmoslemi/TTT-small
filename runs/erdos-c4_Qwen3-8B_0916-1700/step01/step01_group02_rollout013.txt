import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time
from sympy import symbols, Hermite, solve, Eq, Rational

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(c * Hermite(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.as_poly()
        # Enforce Q(0) = 0
        if Q.coeffs()[0] != 0:
            Q = Q - Q.coeffs()[0]
        # Flip sign if necessary to make Q positive at infinity
        if Q.LC() < 0:
            Q = -Q
        # Remove z^2 factor
        if Q.as_dict().get(2, 0) != 0:
            Q = Q / z**2
        # Find real roots
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        # Find r_max
        r_max = 0
        for r in real_roots:
            if r > r_max:
                r_max = r
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Initial heuristic guess
    base_guess = np.array([Rational(1, 2), Rational(1, 2), Rational(1, 2), Rational(1, 2), 0, 0, 0, 0])
    
    # Grid search
    grid_points = 10
    best_coeffs = base_guess
    best_value = objective(best_coeffs)
    
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    for m in range(grid_points):
                        for n in range(grid_points):
                            for p in range(grid_points):
                                for q in range(grid_points):
                                    coeffs = np.array([Rational(i, grid_points), Rational(j, grid_points),

                                                      Rational(m, grid_points), Rational(n, grid_points),

                                    if value < best_value:

                                        best_coeffs = coeffs
    
    # Local optimization
    result = minimize(

        bounds=[(-1, 1) for _ in range(8)],
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
