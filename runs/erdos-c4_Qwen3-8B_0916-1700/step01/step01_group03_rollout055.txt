import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.optimize import root_scalar
from scipy.special import hermgauss

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_upper_bound(coeffs):
        c = coeffs.copy()
        c[0] = 1.0  # Fix c₀ to 1 for normalization
        coeffs = c[:8]
        
        def Q(z):
            res = 0
            for k in range(8):
                res += coeffs[k] * hermgauss(4*k + 1)[0][0] * np.exp(-z**2 / 2)
            return res / z**2
        
        try:
            r_max = root_scalar(lambda z: Q(z), bracket=[1, 100], method='brentq').root
            return r_max**2 / (2 * np.pi)
        except:
            return np.inf
    
    bounds = [(-1, 1) for _ in range(8)]
    
    # Initial global search with differential evolution
    result = differential_evolution(evaluate_upper_bound, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-8)
    
    # Refine with L-BFGS-B
    refined_result = minimize(
        evaluate_upper_bound,
        x0=result.x,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8}
    )
    
    return refined_result.x
