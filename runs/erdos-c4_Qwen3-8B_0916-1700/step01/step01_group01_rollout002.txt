import numpy as np
from scipy.optimize import differential_evolution
from symengine import symbols, hermite, simplify, expand

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    z = symbols('z')
    H = [hermite(k, z) for k in range(8)]
    
    def objective(coeffs):
        c = np.array(coeffs, dtype=float)
        Q = sum(c[k] * H[k] for k in range(len(c)))
        Q = Q - Q.as_expr().coeff(z**2) * z**2
        Q = simplify(Q)
        roots = np.real(np.roots(np.array([coeff for coeff in Q.as_coeffs() if coeff != 0])))
        real_roots = roots[roots > 0]
        r_max = np.max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def numerical_surrogate(coeffs):
        c = np.array(coeffs, dtype=float)
        Q = sum(c[k] * H[k] for k in range(len(c)))
        Q = Q - Q.as_expr().coeff(z**2) * z**2
        Q = simplify(Q)
        coeffs_list = [float(coeff) for coeff in Q.as_coeffs()]
        roots = np.roots(coeffs_list)
        real_roots = roots[np.isreal(roots)]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    result = differential_evolution(numerical_surrogate, bounds=[[-10, 10] for _ in range(8)], 
                                   strategy='best1bin', popsize=20, tol=1e-5, 
                                   maxiter=int(budget_s/10), seed=seed)
    return result.x
