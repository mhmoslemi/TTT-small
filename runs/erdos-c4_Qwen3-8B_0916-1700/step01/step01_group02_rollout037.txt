import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coefficients):
        from sympy import symbols, Hermite, roots, diff, simplify
        z = symbols('z')
        Q = sum(c * Hermite(4*k, z) for k, c in enumerate(coefficients))
        Q = Q - Q.as_leading_coefficient()*z**2
        roots_list = roots(Q, z, domain='R')
        r_max = 0
        for r in roots_list:
            if r.is_real and r > r_max:
                if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                    r_max = r
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Initial guess with low-degree contributions
    initial_guess = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    
    # Optimization settings
    bounds = [(-1, 1) for _ in range(8)]
    
    # Hybrid optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
