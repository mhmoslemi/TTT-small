import numpy as np
from scipy.optimize import minimize, root_scalar
from scipy.interpolate import interp1d
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    # Known good initial basis
    initial_basis = np.array([0.5, -0.2, 0.3, -0.1, 0.4, -0.2, 0.3, -0.1])
    
    def objective(x):
        # Fast numerical surrogate
        try:
            r_max = root_scalar(lambda z: (np.polyval(x, z) / (z**2)), bracket=[0, 100], method='brentq')
            return r_max.root**2 / (2 * np.pi)
        except:
            return np.inf
    
    # Hybrid optimization
    result = minimize(
        objective,
        x0=initial_basis,
        method='L-BFGS-B',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    return result.x
