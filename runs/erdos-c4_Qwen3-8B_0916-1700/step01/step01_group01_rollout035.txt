import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, roots, S

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    m = 4
    x = symbols('x')
    H = [HermiteH(k, x) for k in range(m)]
    
    def exact_objective(coeffs):
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q / (Q.as_poly().leader)
        roots = roots(Q, x, domain=S.Reals)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    def numerical_surrogate(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        H_vals = [HermiteH(k, 0).evalf() for k in range(m)]
        Q = sum(c * h for c, h in zip(coeffs, H_vals))
        Q /= Q
        def f(z):
            return sum(c * HermiteH(k, z).evalf() for k, c in enumerate(coeffs))
        z = np.linspace(0, 10, 1000)
        sign_changes = np.where(np.sign(f(z)) != np.sign(f(z[0])))[0]
        if not sign_changes.size:
            return float('inf')
        r_max = np.max(z[sign_changes[-1]])
        return (r_max**2) / (2 * np.pi)
    
    result = minimize(numerical_surrogate, x0=np.random.rand(m), method='L-BFGS-B',

    return result.x
