import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite
from sympy import symbols, Eq, solve, Rational, S, expand, latex
from time import time

def compute_Q(coeffs, z):
    m = len(coeffs)
    Q = 0
    for k in range(m):
        H = hermite(4*k, z)
        Q += coeffs[k] * H
    return Q

def find_r_max(Q, z):
    z_val = symbols('z')
    Q_z = Q.subs(z_val, z)
    roots = solve(Q_z, z)
    real_roots = [r.evalf() for r in roots if r.is_real]
    real_roots = sorted([r for r in real_roots if r > 0])
    for r in real_roots:
        if r > 0:
            return r
    return None

def evaluate_c4_coefficients(coeffs):
    m = len(coeffs)
    z = symbols('z')
    Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
    Q = Q.subs(z, z).simplify()
    Q = Q - Q.subs(z, 0)
    Q = Q / z**2
    r_max = find_r_max(Q, z)
    if r_max is None:
        return float('inf')
    return r_max**2 / (2 * np.pi)

def run(seed=42, budget_s=200, **kwargs):

        return evaluate_c4_coefficients(x)
