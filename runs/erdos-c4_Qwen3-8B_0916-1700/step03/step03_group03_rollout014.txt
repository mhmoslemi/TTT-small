import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Grid search with 6 coefficients, structured near zero
    grid = np.array([
        [0.0, 0.25, 0.5, 0.75, 1.0, -0.5],
        [0.0, 0.25, 0.5, 0.75, 1.0, -0.5],
        [0.0, 0.25, 0.5, 0.75, 1.0, -0.5],
        [0.0, 0.25, 0.5, 0.75, 1.0, -0.5],
        [0.0, 0.25, 0.5, 0.75, 1.0, -0.5],
        [0.0, 0.25, 0.5, 0.75, 1.0, -0.5]
    ])
    
    best_coeffs = grid[0, :]
    best_value = objective(best_coeffs)
    
    # Evaluate all grid points
    for i in range(grid.shape[0]):
        value = objective(grid[i, :])
        if value < best_value:
            best_value = value
            best_coeffs = grid[i, :]
    
    # Refine with L-BFGS-B starting from best_coeffs
    initial_guess = best_coeffs
    bounds = [(-1, 1) for _ in range(6)]
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
