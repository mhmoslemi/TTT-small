import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Initial structured grid search with 6 coefficients
    grid = np.array([
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5],
        [0.0, 0.5, 1.0, 1.5, 2.0, 2.5]
    ])
    
    best_coeffs = grid[0, :]
    best_value = objective(best_coeffs)
    
    # Evaluate all grid points
    for i in range(grid.shape[0]):
        value = objective(grid[i, :])
        if value < best_value:
            best_value = value
            best_coeffs = grid[i, :]
    
    # Refine with coordinate descent
    initial_guess = best_coeffs
    bounds = [(-2, 2) for _ in range(8)]
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1), 'disp': False}
    )
    
    return result["x"]
