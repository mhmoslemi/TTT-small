from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
from sympy import symbols, hermite, solve, Interval, N, Rational

def hermite_poly(n):
    z = symbols('z')
    return hermite(n, z)

def compute_r_max(coeffs):
    m = len(coeffs)
    z = symbols('z')
    Q = sum(coeffs[k] * hermite_poly(4*k) for k in range(m))
    Q = Q - Q.subs(z, 0)  # Ensure Q(0) = 0

    r_max = None
    for r in real_roots:
        if r_max is None or r > r_max:
            r_max = r
    return r_max

def compute_upper_bound(coeffs):
    r_max = compute_r_max(coeffs)
    if r_max is None:
        return float('inf')
    return r_max**2 / (2 * np.pi)

def run(seed=42, budget_s=200, **kwargs):
