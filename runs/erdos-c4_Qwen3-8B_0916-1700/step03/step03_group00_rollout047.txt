import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate(x):
        coeffs = np.array(x)
        m = len(coeffs)
        z = np.linspace(-10, 10, 10000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * hermite(4*k, z)
        Q /= z**2
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = 0
        for idx in sign_changes:
            root = z[idx]
            if root > r_max:
                r_max = root
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate(x)
    
    # Try different coefficient counts
    for n in [4, 5, 6, 7]:
        if n > 8:
            continue
        
        # Crafted initial guess with alternating signs
        initial_guess = np.array([1.0 if i % 2 == 0 else -1.0 for i in range(n)])
        
        # Global optimization with differential evolution
        result = differential_evolution(
            fun=objective,
            bounds=[(-2, 2) for _ in range(n)],
            strategy='best1bin',
            popsize=15,
            maxiter=int(budget_s / 50),
            disp=False
        )
        
        # Local refinement with L-BFGS-B
        refined = minimize(
            fun=objective,
            x0=result.x,
            bounds=[(-2, 2) for _ in range(n)],
            method='L-BFGS-B',
            options={'maxiter': int(budget_s / 30)}
        )
        
        # Return best result
        return refined["x"]
