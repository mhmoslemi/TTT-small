import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, roots, S, pi

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    initial = np.array([1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01])
    
    def evaluate_coeffs(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        Q = Q.as_numerical()
        Q = Q / z**2
        r = roots(Q, domain=S.Reals)
        real_roots = [x.evalf() for x in r if x.is_real and x > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return r_max**2 / (2 * pi)
    
    def surrogate(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        z = np.linspace(-10, 10, 1000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * np.polyspecial.hermitenorm(4*k, z)
        Q -= Q[0]
        Q /= z**2
        r = np.roots(np.polynomial.Polynomial(Q))
        real_roots = r[np.isreal(r)]
        real_roots = real_roots[real_roots > 0]
        if not real_roots.size:
            return float('inf')
        return (np.max(real_roots))**2 / (2 * np.pi)
    
    result = minimize(lambda x: surrogate(x), x0=initial, method='L-BFGS-B', 
                      options={'ftol': 1e-12, 'maxiter': 150}, tol=1e-12)
    return result.x
