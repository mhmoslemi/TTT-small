import numpy as np
from scipy.optimize import Bounds, minimize
from scipy.special import hermite
from scipy.optimize import root_scalar

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)

    initial = np.array([1.0, 0.5, 0.25, 0.125])
    
    def objective(coeffs):

        m = len(coeffs)
        # Compute Hermite polynomials up to H_{4*(m-1)}
        z = np.linspace(-10, 10, 10000)
        H = np.array([hermite(4 * k, z) for k in range(m)])
        Q = np.dot(coeffs, H)
        # Enforce Q(0) = 0 by subtracting the constant term
        Q -= Q[0]
        # Find r_max numerically with exact root isolation
        def f(r):
            return Q[np.abs(z - r) < 1e-3].min()
        try:
            r_max = root_scalar(lambda r: f(r), bracket=[0, 10], method='brentq').root
        except:
            r_max = 10
        return r_max**2 / (2 * np.pi)
    
    # Bounds for coefficients
    bounds = Bounds(lb=-5, ub=5)
    
    # Multi-start optimization with adaptive precision
    results = []
    for _ in range(10):
        x0 = np.random.uniform(-5, 5, size=4)
        res = minimize(objective, x0=x0, bounds=bounds, method='L-BFGS-B',

        results.append(res)
    
    best = min(results, key=lambda r: r.fun)
    return best.x
