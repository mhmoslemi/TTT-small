import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
from sympy import symbols, diff, simplify, solve

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        # Exact symbolic evaluation of r_max^2 / (2π)
        z = symbols('z')
        Q = sum(c * hermite(4 * i, z) for i, c in enumerate(coeffs))
        Q = Q - Q.as_poly().coefs[-1] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots = sorted(real_roots)
        r_max = real_roots[-1] if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def hermite(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        else:
            return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
    
    # Initial dense grid search with 8 coefficients
    grid = np.array([
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5),
        np.linspace(-1, 1, 5)
    ]).T
    
    best_coeffs = grid[0, :]
    best_value = evaluate_c4_coefficients(best_coeffs)
    
    # Evaluate all grid points
    for i in range(grid.shape[0]):
        value = evaluate_c4_coefficients(grid[i, :])
        if value < best_value:
            best_value = value
            best_coeffs = grid[i, :]
    
    # Refine with coordinate descent
    initial_guess = best_coeffs
    bounds = [(-1, 1) for _ in range(8)]
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
