from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def hermite(n, z):
        if n == 0:
            return np.ones_like(z)
        elif n == 1:
            return 2*z
        else:
            return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
    
    def evaluate(coeffs):
        m = len(coeffs)
        z = np.linspace(-10, 10, 10000)
        H = np.array([hermite(4*k, z) for k in range(m)])
        Q = np.dot(coeffs, H)
        Q -= Q[0]  # Ensure Q(0) = 0
        sign_change = np.diff(np.sign(Q))
        positive_roots = np.where(sign_change > 0)[0]
        r_max = np.max(z[positive_roots])
        return r_max**2 / (2 * np.pi)
    
    def surrogate(coeffs):
        return evaluate(coeffs)
    
    initial = np.array([1.0, 1.0, 1.0, 1.0, 0.5, 0.5, 0.5, 0.5])
    
    result = minimize(
        surrogate,
        x0=initial,
        method='L-BFGS-B',
        bounds=[(-10, 10) for _ in range(8)],
        options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 200, 'disp': False}
    )
    
    return result.x
