import numpy as np
from scipy.optimize import minimize, fmin_powell
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    m = 8
    initial = np.random.rand(m) * 2 - 1
    
    def objective(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        z = np.linspace(-10, 10, 1000)
        H = np.array([hermite(4*k, z) for k in range(m)])
        Q = np.dot(coeffs, H)
        Q -= Q[0]  # Enforce Q(0) = 0
        roots = np.roots(np.poly1d(Q))
        real_roots = roots[np.isreal(roots)]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    # Initial Powell's search
    result = fmin_powell(objective, x0=initial, ftol=1e-10, maxiter=100)
    # Refine with Newton-like steps
    for _ in range(5):
        result = minimize(objective, x0=result, method='L-BFGS-B', tol=1e-12, options={'maxiter': 50})
    return result.x
