import numpy as np
from scipy.optimize import Bounds

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Initial grid search with 8 coefficients
    grid = np.random.uniform(-1, 1, (10, 8))
    best_value = np.inf
    best_coeffs = grid[0, :]
    
    for i in range(grid.shape[0]):
        value = objective(grid[i, :])
        if value < best_value:
            best_value = value
            best_coeffs = grid[i, :]
    
    # Bayesian optimization with L-BFGS-B
    from scipy.optimize import Bounds, minimize
    
    bounds = Bounds([-1, 1], [-1, 1], shape=(8,))
    
    result = minimize(
        fun=objective,
        x0=best_coeffs,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    return result["x"]
