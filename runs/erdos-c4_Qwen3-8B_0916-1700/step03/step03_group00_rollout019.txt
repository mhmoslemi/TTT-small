from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite
from sympy import symbols, simplify, Eq, solve

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def num_surrogate(coeffs):
        return np.linalg.norm(coeffs)
    
    def exact_eval(coeffs):
        z = symbols('z')
        poly = sum(c * hermite(4 * k, z) for k, c in enumerate(coeffs))
        poly = poly.as_poly()
        poly = poly - poly.as_poly().coeffs()[0] * z**2
        roots = solve(poly, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return np.inf
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def evaluate(coeffs):
        return exact_eval(coeffs)
    
    def surrogate(coeffs):
        return num_surrogate(coeffs)
    
    def objective(coeffs):
        return evaluate(coeffs)
    
    def initial_guess():
        return np.random.uniform(-1, 1, 8)
    
    def refine_guess(guess):
        res = minimize(
            fun=surrogate,
            x0=guess,
            bounds=[(-1, 1) for _ in range(8)],
            method='L-BFGS-B',
            options={'maxiter': 50}
        )
        return res['x']

    )
    
    refined = refine_guess(result.x)
