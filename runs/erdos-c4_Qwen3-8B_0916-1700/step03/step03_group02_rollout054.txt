import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import Bounds
from scipy.integrate import quad
import time

def hermite(n, z):
    if n == 0:
        return 1.0
    elif n == 1:
        return 2 * z
    else:
        return 2 * z * hermite(n-1, z) - 2 * (n-1) * hermite(n-2, z)

def exact_evaluate(coeffs):
    m = len(coeffs)
    H = []
    for k in range(m):
        h = hermite(4*k, np.linspace(-10, 10, 10000))
        H.append(h)
    Q = np.dot(coeffs, H)
    Q -= Q[0]  # Enforce Q(0) = 0
    # Find r_max
    roots = np.roots(np.poly1d(Q))
    real_roots = roots[np.isreal(roots)]
    positive_roots = real_roots[real_roots > 0]
    r_max = np.max(positive_roots)
    return r_max**2 / (2 * np.pi)

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    # Structured initialization with known bounds
    initial = np.array([0.5, 0.3, 0.2, 0.1, 0.1, 0.05, 0.02, 0.01])
    
    def objective(coeffs):
        return exact_evaluate(coeffs)
    
    # Optimization with hybrid strategy
    result = minimize(objective, x0=initial, method='L-BFGS-B', 
                      bounds=Bounds(0.001, 1.0), 
                      options={'ftol': 1e-12, 'maxiter': 200, 'disp': False}, 
                      tol=1e-12)
    return result.x
