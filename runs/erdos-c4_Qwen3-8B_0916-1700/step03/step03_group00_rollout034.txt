import numpy as np
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import OptimizeResult
from scipy.integrate import quad
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8

    def construct_Q(coeffs):
        m = len(coeffs)
        Q = np.zeros(1000, dtype=np.complex128)
        for k in range(m):
            H = hermite(4*k, 1.0)
            Q += coeffs[k] * H
        return Q

    def evaluate_c4_coefficients(coeffs):
        try:
            m = len(coeffs)
            Q = np.zeros(1000, dtype=np.complex128)
            for k in range(m):
                H = hermite(4*k, 1.0)
                Q += coeffs[k] * H
            Q = Q / (1 + np.abs(Q))
            r_max = 0.0
            for z in np.linspace(0, 100, 1000):
                if np.isclose(Q[np.isclose(z, 0)], 0):
                    continue
                sign = np.sign(np.real(Q[np.isclose(z, 0)]))
                if np.abs(z) > r_max and np.sign(np.real(Q[np.isclose(z, 0)])) != sign:
                    r_max = z
            return r_max ** 2 / (2 * np.pi)
        except:
            return float('inf')

    def objective(x):
        return evaluate_c4_coefficients(x)

    best_coeffs = None
    best_value = float('inf')

    for n in [4, 5, 6, 7, 8]:
        if n > 8:
            continue
        initial_guess = np.random.rand(n) * 2 - 1
        bounds = [(-1, 1) for _ in range(n)]

        result = minimize(
            fun=objective,
            x0=initial_guess,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': int(budget_s / 0.3)}
        )
        refined_result = minimize(
            fun=objective,
            x0=result['x'],
            bounds=bounds,
            method='Nelder-Mead',
            options={'maxiter': int(budget_s / 0.5)}
        )

        if refined_result['fun'] < best_value:
            best_value = refined_result['fun']
            best_coeffs = refined_result['x']

    return best_coeffs
