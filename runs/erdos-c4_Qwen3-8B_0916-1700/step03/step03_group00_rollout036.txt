import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, HermiteH, simplify, solve, roots, Eq
from sympy.polys.polytools import Poly

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        expr = sum(c * HermiteH(4*k, z) for k, c in enumerate(coeffs))
        expr = expr.subs(z, 0)
        expr = expr.as_numerical()
        expr = expr / z**2
        expr = expr.as_numerical()
        roots = solve(expr, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 1
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Start with 4 coefficients
    n = 4
    initial_guess = np.random.rand(n) * 2 - 1
    
    # First global optimization with differential evolution
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=[(-1, 1) for _ in range(n)],
        method='Nelder-Mead',
        options={'maxiter': int(budget_s / 0.1)}
    )
    
    # Refine with a local optimizer
    refined_result = minimize(
        fun=objective,
        x0=result['x'],
        bounds=[(-1, 1) for _ in range(n)],
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.2)}
    )
    
    return refined_result["x"]
