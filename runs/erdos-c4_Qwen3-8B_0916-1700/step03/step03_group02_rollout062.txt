import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def hermite(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        else:
            return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
    
    def evaluate_coeffs(coeffs):
        m = len(coeffs)
        z = np.linspace(-10, 10, 10000)
        H = np.array([hermite(4*k, z) for k in range(m)])
        Q = np.dot(coeffs, H)
        Q -= Q[0]  # Enforce Q(0) = 0
        roots = np.roots(np.poly1d(Q))
        real_roots = roots[np.isreal(roots)]
        r_max = np.max(real_roots[real_roots > 0])
        return r_max**2 / (2 * np.pi)
    
    # Initial structured grid search
    initial_guesses = []
    for a in np.linspace(0.1, 1.0, 5):
        for b in np.linspace(0.1, 1.0, 5):
            for c in np.linspace(0.1, 1.0, 5):
                for d in np.linspace(0.1, 1.0, 5):
                    initial_guesses.append([a, b, c, d])
    
    best_score = float('inf')
    best_coeff = None
    
    for guess in initial_guesses:
        try:
            result = minimize(
                lambda x: evaluate_coeffs(x),
                x0=guess,
                method='L-BFGS-B',
                bounds=Bounds(0.1, 1.0),
                options={'ftol': 1e-12, 'maxiter': 100, 'disp': False}
            )
            if result.fun < best_score:
                best_score = result.fun
                best_coeff = result.x
        except:
            pass
    
    return best_coeff
