import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Poly, solve, roots
import time

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    # Use exact polynomial arithmetic with known bounds
    initial = np.array([1.0, 0.5, 0.3, 0.2, 0.1])
    
    def evaluate_coeffs(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Find r_max with exact root isolation
        Q_poly = Poly(Q, z)
        real_roots = roots(Q_poly, domain='Real')
        real_roots = [r.evalf() for r in real_roots if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    # Optimization with adaptive strategy
    def objective(coeffs):
        return evaluate_coeffs(coeffs)
    
    # Initial search with random perturbations
    results = []
    for _ in range(5):
        perturbed = initial + np.random.uniform(-0.5, 0.5, len(initial))
        result = minimize(objective, x0=perturbed, method='L-BFGS-B',
                          options={'ftol': 1e-12, 'maxiter': 100}, tol=1e-12)
        results.append(result.x)
    
    # Select the best result
    best = min(results, key=objective)
    return best
