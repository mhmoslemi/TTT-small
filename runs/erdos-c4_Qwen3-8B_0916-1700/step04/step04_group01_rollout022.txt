import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, Poly, hermite, simplify

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q.as_independent(z)[1].subs(z, 0)
        Q = Q / z**2
        roots = [r.evalf() for r in Poly(Q, z).all_roots() if r.as_real_imag()[1] != 0]
        real_roots = [r for r in roots if r.is_real]
        if not real_roots:
            return np.inf
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Try different coefficient counts
    best_coeffs = None
    best_value = np.inf
    
    # Try with 6 coefficients
    n = 6
    initial_guess = np.random.rand(n) * 2 - 1
    
    # Set bounds
    bounds = [(-2, 2) for _ in range(n)]
    
    # Use a hybrid optimization strategy: global search with differential evolution
    # followed by local refinement with Nelder-Mead and L-BFGS-B
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': int(budget_s / 0.3)}
    )
    
    # Refine with a multi-start local optimizer
    for _ in range(5):
        start = np.random.rand(n) * 4 - 2
        refined_result = minimize(
            fun=objective,
            x0=start,
            bounds=bounds,
            method='L-BFGS-B',

        )
        if refined_result['fun'] < best_value:
            best_value = refined_result['fun']
            best_coeffs = refined_result['x']
    
    return best_coeffs
