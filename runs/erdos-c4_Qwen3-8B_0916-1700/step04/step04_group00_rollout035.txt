import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_score = float('inf')
    
    # Initial random guess
    initial_guess = np.random.rand(8) * 2 - 1
    
    # Optimization parameters
    learning_rate = 0.1
    decay_rate = 0.995
    current_rate = learning_rate
    steps = 0
    max_steps = int(budget_s * 0.8)
    
    # Surrogate function for fast numerical evaluation
    def surrogate(coeffs):
        return np.linalg.norm(coeffs)**2 / (2 * np.pi)
    
    # Main optimization loop
    for _ in range(max_steps):
        # Gradient descent step
        grad = np.random.randn(8) * 0.1  # Simple gradient approximation
        new_coeffs = initial_guess - current_rate * grad
        score = surrogate(new_coeffs)
        
        # Update best coefficients
        if score < best_score:
            best_score = score
            best_coeffs = new_coeffs
        
        # Decay learning rate
        current_rate *= decay_rate
        steps += 1
    
    return best_coeffs
