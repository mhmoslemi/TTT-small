def run(seed=42, budget_s=200, **kwargs):

    best_score = float('inf')

        Q = 0
        for i in range(len(coeffs)):
            Q += coeffs[i] * HermiteH(4*i, z)
        return Q

        coeffs_dict = {coeffs[i]: coeffs_array[i] for i in range(len(coeffs))}

        z_squared = z**2
        Q_normalized = Q / z_squared
        # Find roots of Q_normalized
        roots = solve(Q_normalized, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        # Find r_max
        r_max = 0
        for r in real_roots:
            if r > r_max:
                r_max = r
        if r_max == 0:
            return float('inf')
        return r_max**2 / (2 * np.pi)
