import numpy as np
from scipy.optimize import differential_evolution
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def objective(coeffs):
        m = len(coeffs)
        z = np.linspace(-10, 10, 1000)
        H = np.array([hermite(4 * k, z) for k in range(m)])
        Q = np.dot(coeffs, H)
        Q -= Q[0]
        
        # Find r_max
        real_roots = np.roots(np.poly1d(Q))
        real_roots = real_roots[np.isreal(real_roots)]
        r_max = np.max(real_roots[real_roots > 0])
        
        return r_max**2 / (2 * np.pi)
    
    # Use differential evolution with random initializations
    result = differential_evolution(
        fun=objective,
        bounds=[[-10.0, 10.0] for _ in range(8)],
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-12,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )
    
    return result.x[:8]
