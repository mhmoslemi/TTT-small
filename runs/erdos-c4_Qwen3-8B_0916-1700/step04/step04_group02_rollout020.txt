from scipy.optimize import differential_evolution, minimize
from sympy import symbols, Eq, solve, hermite, diff, simplify

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    
    def hermite_poly(n):
        z = symbols('z')
        return hermite(n, z)

        Q = sum(coeffs[k] * hermite_poly(4*k) for k in range(len(coeffs)))

        r_max = compute_r_max(Q)
