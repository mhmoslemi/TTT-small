from scipy.optimize import minimize, Bounds
from sympy import symbols, Poly, hermite

def run(seed=42, budget_s=200, **kwargs):

    n = 6  # Use 6 coefficients for better resolution
    
    # Define symbolic variables
    z = symbols('z')
    H = [hermite(i, z) for i in range(4*n)]
    
    def eval_Q(c):
        Q = sum(c[i] * H[i] for i in range(len(c)))
        return Q
    
    def compute_bound(c):

        Q = eval_Q(c)

                if val > 0 and r > 0:
                    roots.append(r)
            except:
                pass
        if not roots:
            return float('inf')
        r_max = max(roots)
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return compute_bound(x)
    
    # Initial guess with symmetry and scaling
    initial_guess = np.random.rand(n) * 2 - 1
    initial_guess[0] = 1.0  # Ensure Q(0) = 0
    
    # Global optimization with differential evolution
    result = minimize(

        x0=initial_guess,
        bounds=[(-1, 1) for _ in range(n)],
