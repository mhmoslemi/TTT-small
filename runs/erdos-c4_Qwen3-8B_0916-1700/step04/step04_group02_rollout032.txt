import numpy as np
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    initial_guess = np.random.rand(8) * 2 - 1
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    # Initial optimization with L-BFGS-B
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': int(budget_s / 0.3)}
    )
    
    # Refinement with Nelder-Mead
    refined_result = minimize(
        fun=objective,
        x0=result['x'],
        method='Neldorf',
        bounds=Bounds([-1, 1], np.full(8, 1)),
        options={'maxiter': int(budget_s / 0.5)}
    )
    
    return refined_result["x"]
