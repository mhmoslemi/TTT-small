import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    initial_guess = np.random.rand(8) * 2 - 1
    
    def evaluate_and_refine(x):
        def helper(x):
            coeffs = x
            m = len(coeffs)
            Q = np.zeros_like(x)
            for k in range(m):
                H = hermite(4*k, 1.0)
                Q += coeffs[k] * H
            Q[0] = 0  # Enforce Q(0) = 0
            try:
                r_max = root_scalar(lambda z: Q[z]/z**2, bracket=[0, 100], method='brentq').root
                return r_max**2 / (2*np.pi)
            except:
                return np.inf
        return helper(x)
    
    # First optimization with L-BFGS-B
    result = minimize(
        fun=evaluate_and_refine,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': int(budget_s / 0.3)}
    )
    
    # Refinement with Nelder-Mead
    refined_result = minimize(
        fun=evaluate_and_refine,
        x0=result['x'],
        method='Nelder-Mead',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': int(budget_s / 0.5)}
    )
    
    return refined_result["x"]
