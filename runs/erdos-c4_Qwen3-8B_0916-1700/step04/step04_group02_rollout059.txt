from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):

        Q = np.zeros_like(z)

            H = hermite(4*k, z)
            Q += coefficients[k] * H
        # Enforce Q(0) = 0 and positive at infinity
        if np.isnan(Q[0]):
            return np.inf

        sign_changes = np.sign(np.gradient(Q))
        r_max = 0
        for i in range(len(z)-1):
            if sign_changes[i] != sign_changes[i+1] and z[i+1] > 0:
                r_max = z[i+1]
        if r_max == 0:
            return np.inf
        return r_max**2 / (2 * np.pi)
    
    # Initial differential evolution
    result = differential_evolution(
        fun=evaluate_Q,
