import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite as H

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    def evaluate(coeffs):
        m = len(coeffs)
        def Q(z):
            res = 0
            for k in range(m):
                res += coeffs[k] * H(4*k, z)
            return res
        # Find r_max
        try:
            from scipy.optimize import root_scalar
            def f(z):
                return Q(z) / (z**2)
            sol = root_scalar(f, bracket=(-100, 100), method='brentq')
            r_max = sol.root
            return r_max**2 / (2 * np.pi)
        except:
            return np.inf
    
    # Use differential evolution with adaptive bounds
    result = differential_evolution(
        func=evaluate,
        bounds=[(-1, 1) for _ in range(8)],
        strategy='best1bin',
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        maxiter=int(budget_s / 50)
    )
    
    # Refine with L-BFGS-B
    refined_result = minimize(
        fun=evaluate,
        x0=result.x,
        method='L-BFGS-B',
        bounds=[(-1, 1) for _ in range(8)],
        options={'maxiter': int(budget_s / 50)}
    )
    
    return refined_result["x"]
