from scipy.optimize import minimize, OptimizeResult
from scipy.special import hermite
from collections import deque

def run(seed=42, budget_s=200, **kwargs):

    def evaluate_c4_coefficients(coeffs):
        m = len(coeffs)
        z = np.linspace(-10, 10, 10000)
        Q = np.zeros_like(z)
        for k in range(m):
            H = hermite(4*k, z)
            Q += coeffs[k] * H
        Q /= z**2
        sign_changes = 0
        for i in range(len(z)-1):

                sign_changes += 1

            return float('inf')
        return z[-1]**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    def optimize_with_restart():
        results = []
        for _ in range(3):
            init = np.random.rand(len(coeffs)) * 2 - 1

                x0=init,
                method='L-BFGS-B',
                bounds=[(-1, 1) for _ in range(len(coeffs))],

            )
            results.append(result)
        return min(results, key=lambda r: r.fun)
    
    best_result = optimize_with_restart()
    
    return best_result.x
