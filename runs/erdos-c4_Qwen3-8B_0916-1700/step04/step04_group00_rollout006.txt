import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_score = float('inf')
    
    def evaluate_c4_coefficients(coeffs):
        m = len(coeffs)
        z = np.linspace(-5, 5, 1000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * hermite(4*k, z)
        Q /= z**2
        roots = np.roots(np.roots([1, 0, 0, 0], z)[::-1])
        r_max = np.max(roots[roots > 0])
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    bounds = [(-1, 1) for _ in range(7)]
    
    # Initial global search with DE
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=int(budget_s / 2),
        tol=1e-8
    )
    
    # Local refinement with Nelder-Mead
    refined_result = minimize(
        fun=objective,
        x0=result.x,
        bounds=bounds,
        method='Nelder-Mead',
        options={'xatol': 1e-8, 'maxiter': int(budget_s / 2)}
    )
    
    if refined_result.fun < best_score:
        best_score = refined_result.fun
        best_coeffs = refined_result.x
    
    return best_coeffs
