import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.optimize import OptimizeResult
from sympy import symbols, hermite, simplify, Poly

def run(seed=42, budget_s=200, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_score = float('inf')
    
    # Define Hermite polynomials up to H_32
    z = symbols('z')
    H = [simplify(hermite(k, z)) for k in range(33)]
    
    # Function to compute Q(z) and find r_max
    def evaluate_c4_coefficients(coeffs):
        m = len(coeffs)
        Q = 0
        for k in range(m):
            Q += coeffs[k] * H[4*k]
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positivity at infinity: flip sign if necessary
        if Q.as_infinite_limit().evalf() < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q / z**2
        # Find largest positive real root
        try:
            roots = Poly(Q, z).all_roots()
            real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
            r_max = max(real_roots) if real_roots else 0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Try different coefficient counts
    for count in range(1, 9):
        if count > 8:
            continue
        
        # Initialize with random values
        initial_guess = np.random.rand(count) * 2 - 1
        
        # Set bounds
        bounds = [(-1, 1) for _ in range(count)]
        
        # First global optimization with differential evolution
        result = minimize(
            fun=evaluate_c4_coefficients,
            x0=initial_guess,
            bounds=bounds,
            method='L-B96',
            options={'maxiter': int(budget_s / 0.4)}
        )
        
        # Refine with a local optimizer
        refined_result = minimize(
            fun=evaluate_c4_coefficients,
            x0=result['x'],
            bounds=bounds,
            method='Nelder-Mead',
            options={'maxiter': int(budget_s / 0.3)}
        )
        
        # Use multi-start with 5 random initial guesses
        for _ in range(5):
            guess = np.random.rand(count) * 2 - 1
            res = minimize(
                fun=evaluate_c4_coefficients,
