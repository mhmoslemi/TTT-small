def project_feasible(h, n):

        else:

def c5_subgrad(h):
    """Compute subgradient of c5"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = corr >= mx - 1e-14
    nact = float(np.sum(active))
    if nact == 0:
        return mx * dx, np.zeros(n)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if not np.any(active[k]):
            continue
        shift = k - (n - 1)
        for j in range(n):
            g = 0.0
            jm = j - shift
            jp = j + shift
            if 0 <= jm < n:
                g += 1.0 - h[jm]
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += g
    grad *= dx / nact
    return mx * dx, grad
