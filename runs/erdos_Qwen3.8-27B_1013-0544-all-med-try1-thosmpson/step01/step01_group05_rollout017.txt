from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_val(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def grad_lag(h, L, n):
    grad = np.zeros(n)
    if L == 0:
        grad[:] = (1.0 - h) - h
    elif L > 0:
        grad[L:] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += (1.0 - h[L2:])
        grad[L2:] -= h[:n - L2]
    return grad

def subgrad_step(h, n, dx, step, active_count=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.01, c_max * 0.001)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 3:
        active = np.argsort(corr)[-3:]
    elif len(active) > active_count:
        active = np.argsort(corr)[-active_count:]
    n_half = n - 1
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += grad_lag(h, L, n)
    grad /= float(len(active))
    return h, grad, c_max

def slsqp_eq_fun(x, target_sum):
