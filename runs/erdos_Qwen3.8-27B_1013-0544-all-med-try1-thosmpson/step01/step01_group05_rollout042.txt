import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def coord_descent(h, dx, target_sum, max_sweeps, check_time=None, start_time=None, time_limit=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for sweep in range(max_sweeps):
        if check_time and time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):

            slopes = np.zeros(2 * n - 1)

                s = 0.0
                n1 = j - k

                    if 0 <= n1 <= n - 1 - k:
                        s += 1.0 - h[n1]
                    if 0 <= j <= n - 1 - k:
                        s -= h[j + k]
                else:
                    if -k <= n1 <= n - 1:
                        s += 1.0 - h[n1]
                    if -k <= j <= n - 1:
                        s -= h[j + k]
                slopes[ki] = s

            lo, hi = 0.0, 1.0

                else:

def gradient_polish(h, dx, target_sum, time_limit, start_time):
    n = len(h)
    n_half = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step_size = 0.001
    momentum = np.zeros(n)
    mu = 0.5
    
    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(0.01, c_max * 0.001)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-3:]
        elif len(active) > 20:
            active = np.argsort(corr)[-20:]
        
        grad = np.zeros(n)
        for idx in active:
            L = int(idx) - n_half
            grad += compute_grad_lag(h, L)
        grad /= float(len(active))

        h_new = project_feasible(h + momentum, target_sum)
        c5_new = compute_c5(h_new, dx)

            best_h = h_new.copy()
            h = h_new

            h = best_h + np.random.normal(0, 0.01, n)
            h = project_feasible(h, target_sum)
            step_size = 0.001
            momentum = np.zeros(n)

def linearized_step(h, active_indices, target_sum):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Build LP: minimize t
    # subject to: for each active k: grad_k · (h - h0) + corr[k] <= t
    #             0 <= h <= 1
    #             sum(h) = target_sum
    
    # Variables: h[0], ..., h[n-1], t (total n+1 variables)
    # Objective: minimize c · x where c = [0, ..., 0, 1]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: grad_k[0]*h[0] + ... + grad_k[n-1]*h[n-1] - t <= corr[k] - grad_k·h0
    # i.e., [grad_k, -1] · x <= corr[k] - grad_k·h0
    
    A_ub_rows = []
    b_ub_rows = []
    for ki in active_indices:
        L = ki - n_half
        g = compute_grad_lag(h, L)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(float(corr[ki] - np.dot(g, h)))
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    # Equality constraint: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: 0 <= h[i] <= 1, t free (or t >= 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def coord_descent(h, dx, target_sum, max_sweeps, start_time, deadline):
    n = len(h)
    n_half = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            slopes = np.zeros(2 * n - 1)
            for ki in range(2 * n - 1):
                k = ki - n_half
                s = 0.0
                n1 = j - k
                if k >= 0:
                    if 0 <= n1 <= n - 1 - k:
                        s += 1.0 - h[n1]
                    if 0 <= j <= n - 1 - k:
                        s -= h[j + k]
                else:
                    if -k <= n1 <= n - 1:
                        s += 1.0 - h[n1]
                    if -k <= j <= n - 1:
                        s -= h[j + k]
                slopes[ki] = s
            a = corr - slopes * h[j]
            lo, hi2 = 0.0, 1.0
            for _ in range(60):
                m1 = lo + (hi2 - lo) / 3.0
                m2 = hi2 - (hi2 - lo) / 3.0
                if np.max(a + slopes * m1) <= np.max(a + slopes * m2):
                    hi2 = m2
                else:
                    lo = m1
            h[j] = 0.5 * (lo + hi2)
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5

def linearized_solve(h, target_sum):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.005, c_max * 0.002)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) > 25:
        active = np.argsort(corr)[-25:]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    rows = []
    bvals = []
    for ki in active:
        L = int(ki) - n_half
        g = compute_grad_lag(h, L)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        bvals.append(float(corr[ki] - np.dot(g, h)))
    A_ub = np.array(rows)
    b_ub = np.array(bvals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def gradient_polish(h, dx, target_sum, start_time, deadline):

        h_new = project_feasible(h + momentum, target_sum)
