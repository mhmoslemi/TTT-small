import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, target_sum):
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)


def greedy_binary(n, seed=0):
    m = n // 2
    h = np.zeros(n)
    rng = np.random.RandomState(seed)
    order = list(range(n))
    rng.shuffle(order)
    for _ in range(m):
        best_pos = -1
        best_max = 1e9
        for pos in order:
            if h[pos] == 0:
                h[pos] = 1
                corr = np.correlate(h, 1 - h, mode="full")
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[n - 1] = False
                cm = float(np.max(corr[mask]))
                h[pos] = 0
                if cm < best_max - 1e-9 or (abs(cm - best_max) < 1e-9 and best_pos < 0):
                    best_max = cm
                    best_pos = pos
        h[best_pos] = 1
        order.remove(best_pos)
    return h


def binary_sa(h_bin, n_iter, t0, t1, seed):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    m = n // 2
    h = h_bin.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    h_f = h.astype(float)
    corr = np.correlate(h_f, 1.0 - h_f, mode="full")
    cur_val = float(np.max(corr))
    best_val = cur_val
    best_h = h.copy()
    for it in range(n_iter):
        t = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        i = ones[rng.randint(m)]
        j = zeros[rng.randint(m)]
        h[i] = 0
        h[j] = 1
        h_f[i] = 0.0
        h_f[j] = 1.0
        corr = np.correlate(h_f, 1.0 - h_f, mode="full")
        new_val = float(np.max(corr))
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(t, 1e-10)):
            cur_val = new_val
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
            h_f[i] = 1.0
            h_f[j] = 0.0
    return best_h, best_val * (2.0 / n)


def slsqp_refine(h_init, beta, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx

    def con_fun(x):
        return x.sum() - target_sum

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    h_new = project_h(res.x, target_sum)
    return h_new
