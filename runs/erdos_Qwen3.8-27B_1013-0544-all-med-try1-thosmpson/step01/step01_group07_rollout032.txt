```python
def flat_spectrum_search(n, max_iter=1000, seed=42):
    """Find binary sequence with flat power spectrum via iterative projection."""
    rng = np.random.RandomState(seed)
    # Start with random ±1 sequence with zero mean
    a = rng.choice([-1, 1], n)
    # Fix to have equal +1 and -1
    pos = (a == 1).sum()
    target = n // 2
    if pos > target:
        idx = np.where(a == 1)[0]
        a[idx[rng.choice(pos, pos-target, replace=False)]] = -1
    elif pos < target:
        idx = np.where(a == -1)[0]
        a[idx[rng.choice(n-pos, target-pos, replace=False)]] = 1
    
    for it in range(max_iter):
        H = np.fft.fft(a)
        # Set DC to 0
        H[0] = 0
        # Equalize magnitudes
        avg_mag = np.mean(np.abs(H[1:]))
        phases = np.angle(H[1:])
        H[1:] = avg_mag * np.exp(1j * phases)
        # IFFT and threshold
        a_new = np.real(np.fft.ifft(H))
        a_thresh = np.where(a_new >= 0, 1, -1)
        # Fix count
        pos = (a_thresh == 1).sum()
        if pos != target:
            # Adjust
            sorted_idx = np.argsort(a_new)
            if pos > target:
                for i in sorted_idx[:pos-target]:
                    a_thresh[i] = -1
            else:
                for i in sorted_idx[-(target-pos):]:
                    a_thresh[i] = 1
        a = a_thresh
    
    return (a + 1) / 2  # Convert to {0, 1}
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def to_binary(h, n):
    target = n // 2
    b = (h >= 0.5).astype(float)
    current = int(b.sum())
    if current > target:
        one_idx = np.where(b == 1)[0]
        order = np.argsort(h[one_idx])
        b[one_idx[order[:current - target]]] = 0
    elif current < target:
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        b[zero_idx[order[:target - current]]] = 1
    return b

def flat_spectrum(n, max_iter=200, seed=42):
    rng = np.random.RandomState(seed)
    a = rng.choice([-1.0, 1.0], n)
    target = n // 2
    pos = int((a > 0).sum())
    if pos > target:
        idx = np.where(a > 0)[0]
        sel = rng.choice(idx, pos - target, replace=False)
        a[sel] = -1
    elif pos < target:
        idx = np.where(a < 0)[0]
        sel = rng.choice(idx, target - pos, replace=False)
        a[sel] = 1
    
    for _ in range(max_iter):
        H = np.fft.fft(a)
        H[0] = 0
        mags = np.abs(H[1:])
        avg = np.mean(mags)
        phases = np.angle(H[1:])
        H[1:] = avg * np.exp(1j * phases)
        a_new = np.real(np.fft.ifft(H))

        pos = int((a_thresh > 0).sum())

            if pos > target:
                a_thresh[order[:pos - target]] = -1
            else:
                a_thresh[order[-(target - pos):]] = 1
        a = a_thresh

def local_search(h, dx, max_time, seed=42, strategy="random_best"):
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n // 2
    current_c5 = compute_c5(h, dx)

    best_c5 = current_c5
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        # Generate candidate swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        K = min(200, len(ones) * len(zeros))
        i_sel = rng.choice(ones, K, replace=False) if len(ones) >= K else rng.choice(ones, K, replace=True)
        j_sel = rng.choice(zeros, K, replace=False) if len(zeros) >= K else rng.choice(zeros, K, replace=True)
        
        best_delta = current_c5
        best_pair = None
        for i, j in zip(i_sel, j_sel):
            h[i], h[j] = h[j], h[i]
            c5_try = compute_c5(h, dx)
            h[i], h[j] = h[j], h[i]
            if c5_try < best_delta:
                best_delta = c5_try
                best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            current_c5 = best_delta
            if current_c5 < best_c5:

                best_h = h.copy()

            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h, dx)
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def to_binary(h, n):
    target = n // 2
    b = (h >= 0.5).astype(float)
    current = int(b.sum())
    if current > target:
        one_idx = np.where(b == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:current - target]:
            b[one_idx[idx]] = 0
    elif current < target:
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - current]:
            b[zero_idx[idx]] = 1
    return b

def flat_spectrum(n, max_iter, seed_val):
    rng = np.random.RandomState(seed_val)
    a = rng.choice([-1.0, 1.0], n)
    target = n // 2
    pos = int((a > 0).sum())
    if pos > target:
        idx = np.where(a > 0)[0]
        sel = rng.choice(idx, pos - target, replace=False)
        a[sel] = -1.0
    elif pos < target:
        idx = np.where(a < 0)[0]
        sel = rng.choice(idx, target - pos, replace=False)
        a[sel] = 1.0
    for _ in range(max_iter):
        H = np.fft.fft(a)
        H[0] = 0
        mags = np.abs(H[1:])
        avg = np.mean(mags)
        if avg < 1e-10:
            break
        phases = np.angle(H[1:])
        H[1:] = avg * np.exp(1j * phases)
        a_new = np.real(np.fft.ifft(H))
        a_thresh = np.where(a_new >= 0, 1.0, -1.0)
        pos = int((a_thresh > 0).sum())
        if pos != target:
            order = np.argsort(a_new)
            if pos > target:
                for i in range(pos - target):
                    a_thresh[order[i]] = -1.0
            else:
                for i in range(target - pos):
                    a_thresh[order[n - 1 - i]] = 1.0
        a = a_thresh
    return (a + 1.0) / 2.0

def make_block_pattern(n, period):
    h = np.zeros(n)
    duty = period // 2
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return h

def make_mod_pattern(n, p):
    h = np.zeros(n)
    for i in range(n):
        if (i * p) % (p + 1) < (p + 1) // 2 + 1:
            h[i] = 1.0
    return h

def local_search(h_init, dx, max_time, seed_val):
    rng = np.random.RandomState(seed_val)
    n = len(h_init)
    h = h_init.copy()
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    t0 = time.time()
    batch_size = 150
    while time.time() - t0 < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        K = min(batch_size, len(ones) * len(zeros))
        i_sel = rng.choice(ones, size=K, replace=(K > len(ones)))
        j_sel = rng.choice(zeros, size=K, replace=(K > len(zeros)))
        best_try_c5 = current_c5
        best_i = -1
        best_j = -1
        for idx in range(K):
            i = i_sel[idx]
            j = j_sel[idx]
            h[i], h[j] = h[j], h[i]
            c5_try = compute_c5(h, dx)
            h[i], h[j] = h[j], h[i]
            if c5_try < best_try_c5:
                best_try_c5 = c5_try
                best_i = i
                best_j = j
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            current_c5 = best_try_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h, dx)
    return best_h, best_c5

def directed_local_search(h_init, dx, max_time, seed_val):

    h = h_init.copy()

    best_h = h.copy()
    best_c5 = current_c5
    t0 = time.time()

        h_one = h.copy()
        corr = np.correlate(h_one, 1.0 - h_one, mode="full")
        max_val = corr.max()
        max_indices = np.where(corr == max_val)[0]
        
        improved = False
        for k_idx in max_indices:
            if time.time() - t0 > max_time:
                break
            shift = k_idx - (n - 1)
            if shift >= 0:
                valid_start = shift
                valid_end = n
            else:
                valid_start = 0
                valid_end = n + shift
            contributing = []
            for i in range(valid_start, valid_end):
                j_idx = i - shift
                if 0 <= j_idx < n:
                    if h[i] > 0.5 and h[j_idx] < 0.5:
                        contributing.append((i, j_idx))
            if len(contributing) > 0:
                for (i, j_idx) in contributing[:5]:
                    zero_positions = np.where(h == 0)[0]
                    one_positions = np.where(h == 1)[0]
                    if j_idx in one_positions and i in zero_positions:
                        h[i], h[j_idx] = h[j_idx], h[i]
                        c5_try = compute_c5(h, dx)

                            improved = True
                            if c5_try < best_c5:
                                best_c5 = c5_try
                                best_h = h.copy()
                        else:

                    break
        if not improved:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
```
