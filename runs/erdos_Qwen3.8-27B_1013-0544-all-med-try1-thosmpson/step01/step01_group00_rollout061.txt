def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(v, target):
    v = np.asarray(v, dtype=float)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def get_soft_grad(h, temp):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = corr.max()
    w = np.exp((corr - cm) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        jlo1 = max(0, s)
        jhi1 = min(n, s + n)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += wk * (1.0 - h[jlo1 - s:jhi1 - s])
        jlo2 = max(0, -s)
        jhi2 = min(n, n - s)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= wk * h[jlo2 + s:jhi2 + s]
    return grad
