import numpy as np

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

_G = {}

def obj_smax(x):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    t = _G['t']
    sh = t * (corr - mc)
    return (mc + (1.0/t) * np.log(np.sum(np.exp(sh)))) * _G['dx']

def obj_group(x):
    hf = _G['h_full'].copy()
    hf[_G['idx']] = x
    corr = np.correlate(hf, 1 - hf, mode="full")
    mc = np.max(corr)
    t = _G['t']
    sh = t * (corr - mc)
    return (mc + (1.0/t) * np.log(np.sum(np.exp(sh)))) * _G['dx']

def con_sum(x):
    return np.sum(x) - _G['target']

def obj_lbbf(x):
    return obj_smax(x) + _G['pen_w'] * (np.sum(x) - _G['target'])**2

def slsqp_opt(h, target, t, dx, maxiter=500):
    _G['target'] = target
    _G['t'] = t
    _G['dx'] = dx
    try:
        res = minimize(obj_smax, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * len(h),
                       constraints=[{'type': 'eq', 'fun': con_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_sum(res.x, target)
    except Exception:
        return h

def lbbf_opt(h, target, t, dx, maxiter=2000):

                           bounds=[(0.0, 1.0)] * gs,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - fsum}],
                           options={'maxiter': 80, 'ftol': 1e-10})
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                _G['h_full'] = h
        except Exception:
            pass
    return h, best_c5

def obj_penalty(x):

    corr = np.correlate(xc, 1 - xc, mode="full")

    thr = _G['thr']
