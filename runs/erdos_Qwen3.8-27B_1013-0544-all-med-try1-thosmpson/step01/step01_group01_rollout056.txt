import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_grad(h, beta=100.0):
    """Analytical gradient of smooth_max(corr * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-12:
            continue
        L = p - (n - 1)
        if L >= 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            La = -L
            grad[:n - La] += w * (1.0 - h[La:])
            grad[La:] -= w * h[:n - La]
    grad *= dx
    return grad


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def pgd_optimize(h, target_sum, betas, lrs, max_iters=200):
    """Projected gradient descent with multiple beta/lr combos"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in betas:
        for lr in lrs:
            h_curr = best_h.copy()
            for it in range(max_iters):
                grad = compute_grad(h_curr, beta)
                grad -= np.mean(grad)  # project onto sum-constant subspace
                c5_curr = compute_c5(h_curr)
                # Backtracking
                step = lr
                improved = False
                for _ in range(15):
                    h_new = project_feasible(h_curr - step * grad, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_curr - 1e-14:
                        h_curr = h_new
                        improved = True
                        break
                    step *= 0.5
                if not improved:
                    break
            c5 = compute_c5(h_curr)
            if c5 < best_c5:
                best_h, best_c5 = h_curr.copy(), c5
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradual refinement from initial
    betas_seq = [30.0, 50.0, 100.0, 200.0, 500.0]
    for beta in betas_seq:
        if time_left() < 30:
            break
        h_new, c5 = pgd_optimize(best_h, target_sum, [beta],
                                  [0.05, 0.02, 0.01, 0.005, 0.001], 100)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 2: Diverse restarts
    attempt = 0
    while time_left() > 40:
        attempt += 1
        strategy = attempt % 8
        
        if strategy == 0:
            scale = 0.03 + 0.07 * np.exp(-attempt / 30.0)
            h0 = best_h + np.random.randn(n_points) * scale
        elif strategy == 1:
            # Block pattern
            bsz = np.random.randint(4, 12)
            pattern = np.random.random(bsz)
            h0 = np.tile(pattern, n_points // bsz + 1)[:n_points]
        elif strategy == 2:
            # Binary random
            h0 = (np.random.random(n_points) < 0.5).astype(float)
        elif strategy == 3:
            # Sinusoidal + noise
            freq = np.random.uniform(1, 8)
            phase = np.random.uniform(0, 2 * np.pi)
            h0 = 0.5 + 0.25 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
        elif strategy == 4:
            # Swap from best
            h0 = best_h.copy()
            for _ in range(np.random.randint(3, 12)):
                i, j = np.random.choice(n_points, 2, replace=False)
                h0[i], h0[j] = h0[j], h0[i]
        elif strategy == 5:
            # Block randomization
            blen = np.random.randint(5, max(6, n_points // 5))
            bs = np.random.randint(0, max(1, n_points - blen))
            h0 = best_h.copy()
            h0[bs:bs+blen] = 0.3 + 0.4 * np.random.random(blen)
        elif strategy == 6:
            # Mix best with binary
            binary = (np.random.random(n_points) < 0.5).astype(float)
            h0 = 0.6 * best_h + 0.4 * binary
        else:
            # Anti-symmetric perturbation
            h0 = best_h.copy()
            half = n_points // 2
            h0[:half] += 0.05 * np.random.randn(half)
            h0[half:] -= 0.05 * np.random.randn(half)

        h0 = project_feasible(h0, target_sum)
        
        # Optimize
        h_new, c5 = pgd_optimize(h0, target_sum,
                                  [50.0, 100.0, 200.0],
                                  [0.03, 0.01, 0.003], 80)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 3: SLSQP polish
    if time_left() > 30:
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        
        def cons_fun(x):
            return x.sum() - target_sum
        
        def cons_jac(x):
            return np.ones(n_points)
        
        constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        def obj_func(h):
            dx = 2.0 / len(h)
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            m = np.max(corr)
            return m + np.log(np.sum(np.exp(150.0 * (corr - m)))) / 150.0
        
        def obj_grad(h):
            return compute_grad(h, 150.0)
        
        try:
            res = minimize(obj_func, best_h, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Final
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
