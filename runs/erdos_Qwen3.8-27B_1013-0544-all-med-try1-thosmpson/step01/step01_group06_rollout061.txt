def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_lag(h, k):
    """Correct gradient of corr_k(h) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

    best_c5 = compute_c5(h_init)

    # Phase 1: Main gradient optimization
    d1 = t0 + budget_s * 0.38
    best_h, best_c5 = optimize(h_init, target, d1, rng, alpha_init=50, alpha_max=400)

    # Phase 2: Multi-resolution exploration
    rem = budget_s - (time.time() - t0) - 25
    for nt in [140, 160, 200, 240, 260, 300]:
        if rem < 60:
            break
        hi = interpolate(best_h, nt)
        hi = project(hi, nt / 2.0)
        pd = time.time() + min(rem * 0.28, 150.0)
        ho, c5o = optimize(hi, nt / 2.0, pd, rng, alpha_init=40, alpha_max=350)
        if c5o < best_c5:
            best_h, best_c5 = ho, c5o
        rem = budget_s - (time.time() - t0) - 20

    # Phase 3: High-alpha polish (focus on true max)
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        best_h, best_c5 = optimize(best_h, len(best_h) / 2.0,
                                   t0 + budget_s - 10, rng,
                                   alpha_init=150, alpha_max=1200, lr_init=0.008, momentum=0.2)

    # Phase 4: Local random refinement
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h, best_c5 = try_local_refine(best_h, len(best_h) / 2.0,
                                           t0 + budget_s - 2, rng)

    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
