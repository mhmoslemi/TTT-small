            if c5_pol < best_c5 - 1e-14:
                best_c5 = c5_pol
                best_h = h_pol
                pop[best_idx] = h_pol
                pop_c5[best_idx] = c5_pol
        gen += 1

    # Phase 2: Upsample and DE at 2n
    if remaining() > 180:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        T2_end = T1 + budget_s * 0.33

        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

        N_pop2 = 60
        pop2 = np.zeros((N_pop2, n2))
        pop2_c5 = np.zeros(N_pop2)
        pop2[0] = h_up
        pop2_c5[0] = c5_up
        for i in range(1, N_pop2):
            if i % 4 == 1:
                h_t = h_up + rng.randn(n2) * 0.04
            elif i % 4 == 2:
                h_t = np.tile(best_h + rng.randn(n) * 0.03, 2)
            elif i % 4 == 3:
                h_t = np.repeat(best_h, 2) + rng.randn(n2) * 0.02
            else:
                h_t = h_up + rng.randn(n2) * 0.06
            pop2[i] = project_feasible(h_t, n2)
            pop2_c5[i] = compute_c5(pop2[i], dx2)

        best_idx2 = np.argmin(pop2_c5)
        if pop2_c5[best_idx2] < best_c5:
            best_c5 = pop2_c5[best_idx2]
            best_h = pop2[best_idx2].copy()

        gen2 = 0
        while elapsed() < T2_end:
            prog2 = min(1.0, (elapsed() - T1) / max(1, T2_end - T1))
            F2 = 0.8 - 0.5 * prog2
            Cr2 = 0.85 - 0.35 * prog2
            pop2, pop2_c5, best_idx2 = de_generation(pop2, pop2_c5, n2, dx2, F2, Cr2, rng, best_idx2)

            if pop2_c5[best_idx2] < best_c5 - 1e-14:
                best_c5 = pop2_c5[best_idx2]
                best_h = pop2[best_idx2].copy()

            if gen2 % 10 == 0 and remaining() > 60:
                h_pol = slsqp_optimize(pop2[best_idx2].copy(), n2, dx2, target2, max_iter=120)
                c5_pol = compute_c5(h_pol, dx2)
                if c5_pol < best_c5 - 1e-14:
                    best_c5 = c5_pol
                    best_h = h_pol
                    pop2[best_idx2] = h_pol
                    pop2_c5[best_idx2] = c5_pol
            gen2 += 1

    # Phase 3: Final local polishing
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    tgt_f = n_f / 2.0
    polish_count = 0
    while elapsed() < budget_s - 8:
        h_pol = slsqp_optimize(best_h.copy(), n_f, dx_f, tgt_f, max_iter=400)
        c5_pol = compute_c5(h_pol, dx_f)
        if c5_pol < best_c5 - 1e-14:
            best_c5 = c5_pol
            best_h = h_pol
            polish_count = 0
        else:
            polish_count += 1
            if polish_count > 5:
                h_try = best_h + rng.randn(n_f) * 0.008
                h_try = project_feasible(h_try, n_f)
                h_pol = slsqp_optimize(h_try, n_f, dx_f, tgt_f, max_iter=300)
                c5_pol = compute_c5(h_pol, dx_f)
                if c5_pol < best_c5 - 1e-14:
                    best_c5 = c5_pol
                    best_h = h_pol
                    polish_count = 0
                elif polish_count > 15:
                    break
            else:
                h_try = best_h + rng.randn(n_f) * 0.015
                h_try = project_feasible(h_try, n_f)
                h_pol = slsqp_optimize(h_try, n_f, dx_f, tgt_f, max_iter=250)
                c5_pol = compute_c5(h_pol, dx_f)
                if c5_pol < best_c5 - 1e-14:
                    best_c5 = c5_pol
                    best_h = h_pol
                    polish_count = 0

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
