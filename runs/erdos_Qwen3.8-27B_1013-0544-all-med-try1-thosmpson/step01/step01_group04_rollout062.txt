def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.count_nonzero(free) > 0:
            h[free] -= diff / np.count_nonzero(free)
        else:
            h -= diff / n
        h = np.clip(h, 0.0, 1.0)

def de_generation(pop, fit, rng, F, Cr, dim, dx):
    N = len(pop)
    nfit = fit.copy()
    npop = pop.copy()
    idx_all = np.arange(N)

        cmask[j0] = True

        trial = project_feasible(trial, dim)
        f = compute_c5(trial, dx)
        if f < nfit[i]:
            npop[i] = trial
            nfit[i] = f
    return npop, nfit

        if np.isfinite(res.fun):
            return project_feasible(res.x, n), res.fun

        pass
