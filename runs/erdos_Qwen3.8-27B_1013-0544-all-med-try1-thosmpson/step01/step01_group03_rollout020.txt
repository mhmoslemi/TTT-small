def overlap_grad_single(h, m):
    """Gradient of overlap at lag m w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        # (1-h[j+m]) for j in [0, n-1-m]

For m = n/2 - 1: overlap = ∑_{i=0}^{1} h[i](1-h[i+n/2-1]) = h[0](1-h[n/2-1]) + h[1](1-h[n/2]) = 1*0 + 1*1 = 1.

Hmm wait, let me be more careful. For m = n/2:
overlap(n/2) = ∑_{i=0}^{n/2-1} h[i] * (1 - h[i+n/2])
h[i] = 1 for i < n/2
h[i+n/2] = 0 for i < n/2
So overlap = n/2 * 1 * 1 = n/2.

C₅ = (n/2) * dx = (n/2) * (2/n) = 1.

So the step function gives C₅ = 1. Much worse!

The key insight is that for "full" correlation, the zero-lag term is the largest (it has n terms vs. n-|m| terms for lag m). So the zero-lag overlap is:
overlap(0) = ∑ h[i](1-h[i]) = ∑ h[i] - ∑ h[i]² = n/2 * dx - ∑ h[i]²

Wait no, in the discrete formula:
overlap(0) = ∑_{i=0}^{n-1} h[i] * (1-h[i])

And C₅ = max_k overlap(k) * dx.

For the constant h = 0.5: overlap(0) = n * 0.25 = n/4. C₅ = n/4 * 2/n = 1/2.

For a "two-level" h with values a and b (fraction f at level b, 1-f at level a):
overlap(0) = ∑ h[i](1-h[i]) = (number at level b)*b*(1-b) + (number at level a)*a*(1-a)
= (nf/2) * b(1-b) + (n(1-f)/2) * a(1-a) ... wait, the constraint is ∫h = 1, so the weighted average is 1/2 (since ∫h dx = ∑h[i]*dx = (∑h[i])*2/n = 1, so ∑h[i] = n/2).

For two-level h with values a and b, fraction of points at level b is p:
p*b + (1-p)*a = 1/2 (from the sum constraint)
overlap(0) = n*(p*b*(1-b) + (1-p)*a*(1-a))
C₅(0) = overlap(0) * dx = 2*(p*b*(1-b) + (1-p)*a*(1-a))

To minimize C₅(0), we want to minimize p*b*(1-b) + (1-p)*a*(1-a).

If a=0, b=1: p = 1/2, C₅(0) = 2*(0 + 0) = 0. 
But then C₅ at other lags will be large.

If a=0.5, b=0.5: C₅(0) = 2*(0.25) = 0.5.

So there's a trade-off: making h more "binary" reduces the zero-lag overlap but increases other lags.

The optimal balance is what we're trying to find. The current best (0.38085) is better than both extremes.

OK so now I understand the problem better. The key challenge is the trade-off between:
- Zero lag: want h to be "binary" (close to 0 or 1) to minimize h(1-h)
- Large lags: want h to be "balanced" at all shifts

This is a fundamentally non-trivial optimization problem.

Now, let me think about my optimization strategy more carefully.

The bottleneck gradient approach should work well:
1. Compute all overlaps
2. Find the worst ones
3. Compute the gradient to reduce those specific overlaps
4. Take a projected step

The gradient of overlap(m) w.r.t. h[j] is:
(1-h[j+m]) * I[0≤j+m<n] - h[j-m] * I[0≤j-m<n]

To DECREASE overlap(m), we want to move h in the direction that makes (1-h[j+m]) smaller where h[j+m] is large, and makes h[j-m] larger where it's small. Intuitively:
- Where h[j] is high and h[j+m] is low: increase h[j+m] (or decrease h[j])
- Where h[j] is low and h[j+m] is high: decrease h[j+m] (or increase h[j])

Wait, that's reducing the "mismatch" which is what we want!

Actually, to decrease overlap(m) = ∑ h[i](1-h[i+m]):
- Where h[i] is high and h[i+m] is low: either decrease h[i] or increase h[i+m]
- The gradient tells us: ∂/∂h[i] = (1-h[i+m]) (positive, so increasing h[i] increases overlap)
  and ∂/∂h[i+m] = -h[i] (negative, so increasing h[i+m] decreases overlap)

So to decrease overlap(m): decrease h where h[i+m] is low (i.e., where the "other" is low), and increase h where h[i-m] is high (i.e., where the "partner" is high).

This makes sense: we want h[i] and h[i+m] to be either both high or both low, to minimize h[i](1-h[i+m]).

The bottleneck approach should work. Let me refine my implementation.

One concern: the bottleneck gradient only looks at the worst few lags. Once those are improved, the next-worst become the bottleneck. This is like a "water-filling" approach and should work well for maximin problems.

Let me also add momentum to the gradient method for faster convergence.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, tau):
    tx = tau * x
    tm = np.max(tx)
    return float((tm + np.log(np.sum(np.exp(tx - tm)))) / tau)

def smooth_max_weights(x, tau):
    tx = tau * x
    tm = np.max(tx)
    w = np.exp(tx - tm)
    return w / w.sum()

def overlap_grad(h, m):
    """Gradient of overlap at lag m w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        if n > m:
            grad[0:n-m] += (1.0 - h[m:n])
            grad[m:n] -= h[0:n-m]
    else:
        mp = -m
        if n > mp:
            grad[mp:n] += (1.0 - h[0:n-mp])
            grad[0:n-mp] -= h[mp:n]
    return grad

def weighted_overlap_grad(h, weights):
    """Weighted sum of overlap gradients"""
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-10:
            continue
        m = k - (n - 1)
        grad += weights[k] * overlap_grad(h, m)
    return grad

def bottleneck_optimize(h, target, max_iters=1000, lr_init=0.5, top_k=10, momentum=0.7):
    """Optimize using bottleneck gradient with momentum"""
    n = len(h)
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    lr = lr_init
    
    for it in range(max_iters):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        # Select worst lags
        idx = np.argsort(corr)[:top_k]
        w = np.zeros(len(corr))
        w[idx] = 1.0 / top_k
        
        grad = weighted_overlap_grad(h_curr, w)
        v = momentum * v - lr * grad  # gradient descent
        h_new = project_simplex(h_curr + v, target)
        c5 = evaluate_c5(h_new)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        else:
            # Revert and reduce lr
            v *= 0.5
            lr *= 0.9
            h_new = h_curr  # don't move
        
        h_curr = h_new
        if it % 100 == 0 and it > 0:
            lr *= 0.95
    
    return best_h, best_c5

def slsqp_refine(h, target, tau, maxiter=300):
    """SLSQP with smooth max objective"""
    n = len(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        val = smooth_max_val(corr, tau)
        w = smooth_max_weights(corr, tau)
        grad = weighted_overlap_grad(x, w)
        return val, grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,

        return x, evaluate_c5(x)
    except Exception:
