        g[:n - p] += (1.0 - h[p:])
        if p > 0:

    return g

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def grad_smooth(h, alpha, top_k=60):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha)
    order = np.argsort(w)[::-1]
    grad = np.zeros(n)
    for idx in order[:top_k]:
        if w[idx] < 1e-12:
            break
        k = idx - (n - 1)
        grad += w[idx] * grad_corr_single(h, k, n)
    return grad * (2.0 / n)

def grad_variance(h):
    n = len(h)
    corr = compute_corr(h)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        k = idx - (n - 1)
        grad += 2.0 * corr[idx] * grad_corr_single(h, k, n)
    return grad * (2.0 / n) / len(corr)

def grad_focused(h, top_shifts=3):
    n = len(h)
    corr = compute_corr(h)
    order = np.argsort(corr)[::-1][:top_shifts]
    grad = np.zeros(n)
    for idx in order:
        k = idx - (n - 1)
        grad += grad_corr_single(h, k, n)
    return grad * (2.0 / n) / top_shifts

def project_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def optimize(h0, target_sum, deadline, rng, alpha_init=200, use_var=False):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    momentum = 0.85
    lr = 0.004
    alpha = alpha_init
    patience = 0
    stall = 0

    while time.time() < deadline:
        corr = compute_corr(h)
        c5 = np.max(corr) * (2.0 / n)

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            stall = 0
        else:
            patience += 1
            if patience % 100 == 0:
                stall += 1

        # Choose gradient type
        if use_var and stall > 2:
            g1 = grad_smooth(h, alpha, top_k=40)
            g2 = grad_variance(h)
            grad = 0.6 * g1 + 0.4 * g2
        elif patience > 40 and patience % 30 < 15:
            grad = grad_focused(h, top_shifts=3)
        else:
            grad = grad_smooth(h, alpha, top_k=60)

        # Adaptive alpha
        if patience > 60:
            alpha = min(alpha * 1.08, 600.0)

        # Nesterov step
        v = momentum * v - lr * grad
        h_new = h - v
        h_new = project_box(h_new, target_sum)
        new_c5 = compute_c5(h_new)

        if new_c5 <= c5 + 1e-14:
            h = h_new
            if patience < 15 and lr < 0.04:
                lr = min(lr * 1.04, 0.04)
        else:
            lr *= 0.92
            v *= 0.4
            if patience > 120:
                # Structured perturbation targeting the worst shift
                max_idx = np.argmax(corr)
                max_k = max_idx - (n - 1)
                g_worst = grad_corr_single(h, max_k, n)
                g_worst -= np.mean(g_worst)
                scale = 0.015 + 0.005 * min(stall, 5)
                h = best_h - scale * g_worst / (np.max(np.abs(g_worst)) + 1e-10)
                h = project_box(h, target_sum)
                # Also try random perturbation
                h_rand = best_h + rng.randn(n) * scale
                h_rand = project_box(h_rand, target_sum)
                if compute_c5(h_rand) < compute_c5(h):
                    h = h_rand
                v = np.zeros(n)
                lr = 0.003
                alpha = min(alpha + 40, 600)
                patience = 0
                if stall > 10:
                    break

    return best_h, best_c5

def try_binary_refine(h, target_sum, deadline, rng):
    """Try pushing toward binary and refining."""
    n = len(h)
    # Create binary version
    h_bin = np.where(h > 0.5, 1.0, 0.0).copy()
    # Fix to meet sum constraint
    n_ones = int(np.sum(h_bin))
    need = target_sum - n_ones
    if need > 0:
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:int(need)]
        h_bin[zeros_idx[order]] = 1.0
    elif need < 0:
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:int(-need)]
        h_bin[ones_idx[order]] = 0.0
    # Handle fractional
    frac = need - int(need) if need > 0 else int(-need) - (-need)
    if abs(frac) > 1e-10:
        remaining = int(np.sum(h_bin)) - target_sum
        if remaining > 0:
            ones_idx = np.where(h_bin == 1.0)[0]
            h_bin[ones_idx[0]] = 1.0 - remaining
        elif remaining < 0:
            zeros_idx = np.where(h_bin == 0.0)[0]
            h_bin[zeros_idx[0]] = -remaining

    bin_c5 = compute_c5(h_bin)
    if time.time() > deadline:
        return h_bin, bin_c5

    # Refine from binary
    h_ref, c5_ref = optimize(h_bin, target_sum, min(deadline, time.time() + 60), rng, alpha_init=300)
    return h_ref, c5_ref

def interpolate(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        h0 = np.full(n, 0.5) + rng.randn(n) * 0.3
        h0 = project_box(h0, n / 2.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_box(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    # Phase 1: Main optimization at original resolution
    deadline1 = t0 + budget_s * 0.35
    best_h, best_c5 = optimize(best_h, target_sum, deadline1, rng, alpha_init=200, use_var=False)

    # Phase 2: Binary refinement attempt
    elapsed = time.time() - t0
    if budget_s - elapsed > 100:
        deadline2 = t0 + budget_s * 0.55
        h_bin, c5_bin = try_binary_refine(best_h, target_sum, deadline2, rng)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin

    # Phase 3: Variance-aware optimization
    elapsed = time.time() - t0
    if budget_s - elapsed > 80:
        deadline3 = t0 + budget_s * 0.75
        best_h, best_c5 = optimize(best_h, target_sum, deadline3, rng, alpha_init=250, use_var=True)

    # Phase 4: Try higher resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        n_try = 300
        h_interp = interpolate(best_h, len(best_h), n_try)
        h_interp = project_box(h_interp, n_try / 2.0)
        deadline4 = t0 + budget_s - 10
        h_opt, c5_opt = optimize(h_interp, n_try / 2.0, deadline4, rng, alpha_init=150, use_var=False)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try

    # Final feasibility check
    best_h = project_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
