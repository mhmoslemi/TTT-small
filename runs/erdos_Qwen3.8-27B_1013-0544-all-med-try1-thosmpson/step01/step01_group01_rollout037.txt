    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.1, 0.9)  # More aggressive clipping to avoid boundary issues
        # Re-clip and adjust
        s2 = h.sum()
        if abs(s2 - target_sum) > 1e-12:
            h = np.clip(h + (target_sum - s2) / len(h), 0.0, 1.0)
    return h

def subgradient_step(h, target_sum, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_star = np.argmax(corr)
    s = m_star - (n - 1)  # the shift
    # Gradient: grad[j] = dx * [(1-h[j-s])_valid - h[j+s]_valid]
    grad = np.zeros(n)
    for j in range(n):

        idx1 = j - s
        if 0 <= idx1 < n:

    h_new = h - alpha * grad

    lo, hi = -2.0, 2.0  # range for lambda
    
    def sum_at_lambda(lam):
        return np.clip(h_raw - lam, 0.0, 1.0).sum()

        if sum_at_lambda(mid) > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def project_h(h_raw, target_sum):
    """Euclidean projection onto {0<=h<=1, sum=target_sum} via bisection."""
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h_raw - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h_raw - 0.5*(lo+hi), 0.0, 1.0)

def compute_grad(h, beta):
    """Compute gradient of smooth_max(correlate(h,1-h))*dx using softmax weights."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_shifted = corr - corr.max()
    weights = np.exp(beta * corr_shifted)
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= ws
    
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-8:
            continue
        s = m - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            sa = -s
            grad[:n-sa] += w * (1.0 - h[sa:])
            grad[sa:] -= w * h[:n-sa]
    return grad * dx

def gradient_descent(h, target_sum, beta_init, beta_final, n_steps, alpha_init, alpha_final):
    """Run gradient descent with annealed beta and step size."""
    n = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for it in range(n_steps):
        t = it / max(n_steps - 1, 1)
        beta = beta_init * (beta_final / beta_init) ** t
        alpha = alpha_init * (alpha_final / alpha_init) ** t
        
        grad = compute_grad(h, beta)
        h_new = h - alpha * grad
        h_new = project_h(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    
    return best_h, best_c5

        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        
        def con_fun(x):
            return np.sum(x) - target_sum
        
        def con_jac(x):
            return np.ones(n)
        
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        def obj_func(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max_val(corr, 100.0) * dx
        
        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,

            h_new = project_h(res
