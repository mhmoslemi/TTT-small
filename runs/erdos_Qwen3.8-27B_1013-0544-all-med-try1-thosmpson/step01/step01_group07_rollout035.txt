def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    # p_k = softmax(t * dx * corr)
    shifted = t * dx * (corr - np.max(corr))  # numerical stability
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient
    # ∂corr[k]/∂h[j] = (1 - h[j+s_k]) - h[j] * I(0 <= j+s_k < n)
    # where s_k = k - (n-1)
    # So: grad[j] = dx * sum_k p[k] * (1 - h[j+k-(n-1)]) - h[j] * sum_k p[k] * I(0 <= j+k-(n-1) < n)
    # Here h[idx] = 0 if idx not in [0, n-1]
    
    grad = np.zeros(n)
    for j in range(n):
        # s_k = k - (n-1), so j + s_k = j + k - (n-1)
        idx = j + np.arange(2*n-1) - (n-1)  # = j - (n-1) + k for k=0..2n-2
        # h[idx] where idx in [0, n-1], else 0
        valid = (idx >= 0) & (idx < n)
        h_vals = np.zeros(2*n-1)
        h_vals[valid] = h[idx[valid]]
        # I(0 <= j+k-(n-1) < n) = valid
        grad[j] = dx * (np.sum(p * (1 - h_vals)) - h[j] * np.sum(p * valid.astype(float)))
    
    return grad

    p = exp_s / np.sum(exp_s)
    
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2*n-1)[None, :]
    M = j_idx + k_idx - (n-1)
    valid = (M >= 0) & (M < n)
    h_vals = np.where(valid, h[np.clip(M, 0, n-1)], 0.0)
    
    term1 = (1.0 - h_vals) @ p
    term2 = h * (valid.astype(np.float64) @ p)
    return dx * (term1 - term2)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def project_onto_constraint(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    # h = clip(v - lambda, 0, 1), find lambda so sum(h) = target_sum
    lo = v.min() - target_sum / len(v)
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Adjust the elements closest to 0.5
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_optimize(h, dx, target_sum, t, maxiter=500):
    """Optimize using L-BFGS-B with penalty for sum constraint."""
    n = len(h)
    mu = 10.0  # penalty for sum constraint
    
    def obj(x):
        h_x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_x, 1 - h_x, mode="full")
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        smooth = (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx
        penalty = mu * (np.sum(h_x) - target_sum) ** 2
        return smooth + penalty
    
    def grad(x):
        h_x = np.clip(x, 0.0, 1.0)
        # Gradient of smooth max
        g = smooth_max_grad(h_x, t, dx)
        # Gradient of penalty
        g += 2 * mu * (np.sum(h_x) - target_sum)
        # Mask by bounds
        g = np.where((h_x > 0) & (h_x < 1), g, 0)
        return g
    
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'gtol': 1e-6})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

def subgradient_step(h, dx, target_sum, alpha):
    """One subgradient step with projection."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    k_max = np.argmax(corr)
    s = k_max - (n - 1)
    
    # Subgradient of corr[k_max] w.r.t. h
    # ∂corr[k]/∂h[j] = (1 - h[j+s]) - h[j] * I(0 <= j+s < n)
    idx = np.arange(n) + s
    valid = (idx >= 0) & (idx < n)
    h_shifted = np.where(valid, h[np.clip(idx, 0, n-1)], 0.0)
    sg = (1.0 - h_shifted) - h * valid.astype(float)
    sg *= dx
    
    # Subgradient step
    v = h - alpha * sg
    # Project onto constraint
    return project_onto_constraint(v, target_sum)

def make_fourier_init(n, seed_val, n_harmonics=6):
    """Generate a Fourier-based initial candidate."""
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(k * np.pi * x + phase)
    h = np.clip(h, 0.0, 1.0)
    # Rescale to have correct integral
    if np.sum(h) > 0:
        h = h * (n / 2.0) / np.sum(h)
        h = np.clip(h, 0.0, 1.0)
    return project_onto_constraint(h, n / 2.0)

def anneal_optimize(h, dx, target_sum, t_start=5, t_end=200, n_steps=10, maxiter_per=200):
    """Run L-BFGS-B with increasing temperature."""
    for i in range(n_steps):
        t = t_start * (t_end / t_start) ** (i / max(n_steps - 1, 1))
        h_new = lbfgsb_optimize(h, dx, target_sum, t=t, maxiter=maxiter_per)
        c5_new = compute_c5(h_new, dx)
        c5_old = compute_c5(h, dx)
        if c5_new < c5_old:
            h = h_new
    return h

def slsqp_polish(h, dx, target_sum, t=100, maxiter=500):
    """Final polish with SLSQP."""
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        h_new = np.clip(res.x, 0.0, 1.0)
        # Fix sum
        diff = target_sum - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    except Exception:
        return h

        g = smooth_max_grad(hx, t, dx)
        g += 2.0 * mu * (np.sum(hx) - target_sum)
        return np.where((hx > 1e-6) & (hx < 1.0 - 1e-6), g, 0.0)
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-5, 'maxls': 50})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

def subgrad_step(h, dx, target_sum, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr)
    sg = np.zeros(n)
    for j in range(n):
        jm = j - k_max
        jp = j + k_max
        if 0 <= jm < n:
            sg[j] += (1.0 - h[jm])
        if 0 <= jp < n:
            sg[j] -= h[jp]
    sg *= dx
    v = h - alpha * sg
    return project_constraint(v, target_sum)

def subgrad_phase(h, dx, target_sum, n_steps=300, alpha0=0.3, decay=0.97):
    out = h.copy()
    alpha = alpha0
    best = compute_c5(out, dx)
    for i in range(n_steps):
        out = subgrad_step(out, dx, target_sum, alpha)
        c = compute_c5(out, dx)
        if c < best:
            best = c
        else:
            alpha *= 0.8
        alpha *= decay
        alpha = max(alpha, 0.01)
    return out, best

def fourier_init(n, seed_val, nharm=8):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, nharm + 1):
        amp = rng.uniform(0.1, 0.5) / (k ** 0.5)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(k * np.pi * x + ph)
    h = np.clip(h, 0.0, 1.0)
    return project_constraint(h, n / 2.0)

def block_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2.0, n, endpoint=False)
    nblocks = rng.randint(2, 6)
    h = np.zeros(n)
    for _ in range(nblocks):
        start = rng.randint(0, n)
        width = rng.randint(n // 20, n // 4)
        end = min(start + width, n)
        h[start:end] = 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_constraint(h, n / 2.0)

def slsqp_polish(h, dx, target_sum, t=300, maxiter=2000):
    n = len(h)
    try:
        res = minimize(lambda x: smooth_max_obj(x, t, dx), h.copy(),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        hn = np.clip(res.x, 0.0, 1.0)
        hn = project_constraint(hn, target_sum)
        return hn
    except Exception:
        return h

def anneal_lbfgsb(h, dx, target, t_seq, maxiter_per=250):
    for t in t_seq:
        h = lbfgsb_step(h, dx, target, t=t, maxiter=maxiter_per)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    ts0 = n0 / 2.0
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0

    # Phase 1: L-BFGS-B annealing from initial
    if elapsed() < budget_s - 300:
        t_seq = [3, 5, 10, 20, 40, 80, 150, 300]
        h1 = anneal_lbfgsb(best_h, dx0, ts0, t_seq, maxiter_per=300)
        c1 = compute_c5(h1, dx0)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Subgradient refinement
    if elapsed() < budget_s - 200:
        h2, c2 = subgrad_phase(best_h, dx0, ts0, n_steps=400, alpha0=0.4, decay=0.98)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: More L-BFGS-B with higher t
    if elapsed() < budget_s - 150:
        h3 = anneal_lbfgsb(best_h, dx0, ts0, [50, 100, 200, 400], maxiter_per=400)
        c3 = compute_c5(h3, dx0)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Try different resolutions
    for n_try in [n0 // 2, n0 * 3 // 2, n0 * 2]:
        if elapsed() > budget_s - 80:
            break
        if n_try < 20 or n_try > 400:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0, 2, best_n + 1)[:-1]
        x_new = np.linspace(0, 2, n_try + 1)[:-1]
        h_t = np.interp(x_new, x_old, best_h)
        h_t = project_constraint(h_t, ts_t)
        h_t = anneal_lbfgsb(h_t, dx_t, ts_t, [5, 15, 50, 150], maxiter_per=200)
        ct = compute_c5(h_t, dx_t)
        if ct < best_c5:
            best_h, best_c5, best_n = h_t, ct, n_try

    # Phase 5: Fourier restarts
    if elapsed() < budget_s - 60:
        for fs in range(12):
            if elapsed() > budget_s - 40:
                break
            h_f = fourier_init(best_n, seed + fs * 13 + 200, nharm=6 + fs % 4)
            h_f = anneal_lbfgsb(h_f, 2.0 / best_n, best_n / 2.0, [5, 20, 80], maxiter_per=200)
            cf = compute_c5(h_f, 2.0 / best_n)
            if cf < best_c5:
                best_h, best_c5 = h_f, cf

    # Phase 6: Block restarts
    if elapsed() < budget_s - 30:
        for bs in range(6):
            if elapsed() > budget_s - 20:
                break
            h_b = block_init(best_n, seed + bs * 17 + 500)
            h_b = anneal_lbfgsb(h_b, 2.0 / best_n, best_n / 2.0, [3, 10, 40, 100], maxiter_per=200)
            cb = compute_c5(h_b, 2.0 / best_n)
            if cb < best_c5:
                best_h, best_c5 = h_b, cb

    # Phase 7: SLSQP polish
    if elapsed() < budget_s - 5:
        dx_f = 2.0 / best_n
        h_p = slsqp_polish(best_h, dx_f, best_n / 2.0, t=300, maxiter=3000)
        cp = compute_c5(h_p, dx_f)
        if cp < best_c5:
            best_h, best_c5 = h_p, cp

    # Final validation
    dx_final = 2.0 / best_n
    best_h = project_constraint(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, best_n
