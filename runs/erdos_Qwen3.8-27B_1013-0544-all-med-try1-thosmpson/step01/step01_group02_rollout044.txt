    if n_active == 0:
        return h.copy()
    grad = np.zeros(n)
    for idx in range(n_total):
        if not active_mask[idx]:
            continue
        shift = idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + shift < n:
                g += 1.0 - h[j + shift]
            if 0 <= j - shift < n:
                g -= h[j - shift]
            grad[j] += g
    grad *= dx / n_active
    h_new = h - eta * grad
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    def elapsed():
        return time.time() - t0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: SLSQP continuation
    for alpha in [10, 30, 100, 300, 1000, 3000, 10000]:
        if budget_s - elapsed() < 25:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-16})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 2: Subgradient descent
    if budget_s - elapsed() > 100:
        eta = 0.01
        active_eps = 0.01
        for it in range(2000):
            if budget_s - elapsed() < 80:
                break
            h_try = subgradient_step(best_h, n, eta, active_eps)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
            eta *= 0.998
            active_eps = max(0.001, active_eps * 0.995)

    # Phase 3: SLSQP polish after subgradient
    for alpha in [1000, 5000, 10000, 20000]:
        if budget_s - elapsed() < 15:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-17})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 4: Simulated annealing with 2-point moves
    if budget_s - elapsed() > 150:
        T = 0.003
        T_min = 1e-9
        cooling = 0.997
        no_improve = 0
        sa_patience = 3000
        while budget_s - elapsed() > 100:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d_mag = rng.uniform(0.005, 0.15)
            d = d_mag if rng.random() < 0.5 else -d_mag
            hi = best_h[i] + d
            hj = best_h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h_try = best_h.copy()
            h_try[i] = hi
            h_try[j] = hj
            c = compute_c5(h_try, n)
            delta = c - best_c5
            if delta < -1e-15:
                best_h, best_c5 = h_try, c
                no_improve = 0
                T = max(T * 0.95, T_min)
            else:
                no_improve += 1
                if no_improve > sa_patience:
                    T = 0.005
                    no_improve = 0
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    best_h, best_c5 = h_try, c
            T = max(T * cooling, T_min)

    # Phase 5: SLSQP polish after SA
    for alpha in [2000, 10000, 30000]:
        if budget_s - elapsed() < 15:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-17})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 6: Fine systematic 2-point scan
    if budget_s - elapsed() > 40:
        deltas = [0.0005, 0.001, 0.002, 0.005, 0.01]
        count = 0
        while budget_s - elapsed() > 20 and count < 50000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            found = False
            for d in deltas:
                for s in [1, -1]:
                    hi = best_h[i] + s * d
                    hj = best_h[j] - s * d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-15:
                            best_h, best_c5 = h_try, c
                            found = True
                            break
                if found:
                    break

    # Phase 7: Another round of SA with lower temperature if time permits
    if budget_s - elapsed() > 60:
        T = 0.001
        T_min = 1e-9
        cooling = 0.998
        no_improve = 0
        while budget_s - elapsed() > 25:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d_mag = rng.uniform(0.002, 0.05)
            d = d_mag if rng.random() < 0.5 else -d_mag
            hi = best_h[i] + d
            hj = best_h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h_try = best_h.copy()
            h_try[i] = hi
            h_try[j] = hj
            c = compute_c5(h_try, n)
            delta = c - best_c5
            if delta < -1e-15:
                best_h, best_c5 = h_try, c
                no_improve = 0
                T = max(T * 0.95, T_min)
            else:
                no_improve += 1
                if no_improve > 2000:
                    T = 0.002
                    no_improve = 0
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    best_h, best_c5 = h_try, c
            T = max(T * cooling, T_min)

    # Phase 8: Final SLSQP with very high alpha
    if budget_s - elapsed() > 10:
        try:
            res = minimize(obj_smooth, best_h, args=(n, 50000), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 10000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 9: Very fine final scan
    if budget_s - elapsed() > 5:
        deltas = [0.0002, 0.0005, 0.001]
        for _ in range(10000):
            if budget_s - elapsed() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                for s in [1, -1]:
                    hi = best_h[i] + s * d
                    hj = best_h[j] - s * d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-15:
                            best_h, best_c5 = h_try, c
                            break
                else:
                    continue
                break

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
