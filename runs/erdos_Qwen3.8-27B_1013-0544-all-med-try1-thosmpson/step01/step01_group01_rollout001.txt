            for _ in range(rng.randint(1, 4)):
                i, j = rng.choice(n_points, 2, replace=False)
                h0[i], h0[j] = h0[j] + 0.02 * rng.randn(), h0[i] + 0.02 * rng.randn()

        h0 = project_h(h0, target_sum)
        beta = 300 if attempt % 2 == 0 else 100
        obj = make_obj(beta, dx)
        try:
            res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Final high-precision polish
    if time_left() > 15:
        for beta in [2000, 5000]:
            if time_left() < 10:
                break
            obj = make_obj(beta, dx)
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                               constraints=constraints, options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
