    dx = 2.0 / n

    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    log_sum = np.log(np.sum(np.exp(shifted)))

        if h_try.sum() > target_sum:

- Plus contributions from all other terms where j appears

        h0 = project_h(h0, target_sum)

        if att % 4 == 3 and tl() > 60:
            hn, cd = trust_constr(h0, 800)
        else:
            beta = 300 if att < 25 else 500
            mi = 400 if tl() > 200 else 250
            hn, cd = slsqp(h0, beta, mi, 1e-13)
        if cd < best_c5:
            best_h, best_c5 = hn, cd

    # Final polish
    if tl() > 20:
        for b in [400, 700, 1000]:
            if tl() < 12:
                break
            hn, cd = slsqp(best_h, b, 5000, 1e-15)
            if cd < best_c5:
                best_h, best_c5 = hn, cd
    return best_h, best_c5, n


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_parent = len(initial_h_values)
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_parent = 70
        h_parent = np.full(n_parent, 0.5)

    best_overall_h = project_h(h_parent, n_parent / 2.0)
    best_overall_c5 = compute_c5(best_overall_h)
    best_overall_n = n_parent

    # Optimize with parent n
    time_per_n = min(500, budget_s * 0.5)
    h_opt, c5_opt, n_opt = optimize_n(n_parent, h_parent, start_time,
                                       start_time + time_per_n, seed)
    if c5_opt < best_overall_c5:
        best_overall_h, best_overall_c5, best_overall_n = h_opt, c5_opt, n_opt

    # Try a few other n values if time allows
    remaining = budget_s - (time.time() - start_time)
    n_candidates = [100, 140]
    if n_parent != 70:
        n_candidates = [70, 100]

    for nc in n_candidates:
        if remaining < 120:
            break
        # Resample parent to new n
        x_old = np.linspace(0, 2, n_parent, endpoint=False) + 1.0 / n_parent
        x_new = np.linspace(0, 2, nc, endpoint=False) + 1.0 / nc
        h_resampled = np.interp(x_new, x_old, np.concatenate([h_parent, [h_parent[0]]]),
                                period=2)
        h_resampled = project_h(h_resampled, nc / 2.0)

        t_per = min(remaining * 0.45, 300)
        h_opt2, c5_opt2, n_opt2 = optimize_n(nc, h_resampled, time.time(),
                                              time.time() + t_per, seed + nc)
        if c5_opt2 < best_overall_c5:
            best_overall_h, best_overall_c5, best_overall_n = h_opt2, c5_opt2, n_opt2
        remaining = budget_s - (time.time() - start_time)

    # Final global polish on best
    if remaining > 40:
        t_left = remaining - 5
        h_final, c5_final, _ = optimize_n(best_overall_n, best_overall_h,
                                          time.time(), time.time() + t_left, seed + 1000)
        if c5_final < best_overall_c5:
            best_overall_h, best_overall_c5 = h_final, c5_final

    best_overall_h = project_h(best_overall_h, best_overall_n / 2.0)
    best_overall_c5 = compute_c5(best_overall_h)
    return best_overall_h, best_overall_c5, best_overall_n
