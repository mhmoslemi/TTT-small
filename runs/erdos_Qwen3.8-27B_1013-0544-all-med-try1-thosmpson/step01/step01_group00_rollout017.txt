The function g(x) = max_k(a_k*x + b_k) is convex and piecewise linear. Its minimum is at:
- x = 0, or
- x = 1, or
- A point where two lines a_i*x + b_i = a_j*x + b_j both achieve the maximum

The breakpoints are where a_i*x + b_i = a_j*x + b_j, i.e., x = (b_j - b_i)/(a_i - a_j).

1. Compute g(0) and g(1)
2. Find all breakpoints x_kl = (b_l - b_k)/(a_k - a_l) in [0,1]
3. The minimum must be at x=0, x=1, or at a breakpoint
4. Evaluate g at all breakpoints and take the minimum

1. Start at x = 0.5 (or current value)

4. If s > 0, move left; if s < 0, move right; if s = 0, we're at the minimum
5. Bisect to find where the subgradient changes sign

- If right_subgrad < 0: minimum is to the right
- If 0 is between them: we're at the minimum

    al[m1] += 1.0 - h[l - k[m1]]
    m2 = (l + k >= 0) & (l + k < n)
    al[m2] -= h[l + k[m2]]
    fc = np.correlate(h, 1.0 - h, mode='full')
    c = fc - aj * h[j] - al * h[l]
    c_obj = np.array([0.0, 0.0, 1.0])
    A_ub = np.column_stack([aj, al, -np.ones(2 * n - 1)])
    b_ub = -c
    A_eq = np.array([[1.0, 1.0, 0.0]])
    b_eq = np.array([h[j] + h[l]])
    bounds = [(0.0, 1.0), (0.0, 1.0), (None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[0], res.x[1]
    return h[j], h[l]


def t_obj_v(v, n):
    return v[-1]


def t_ineq_v(v, n):
    hh = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
    return t - corr


def t_eq_v(v, n):
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac_v(v, n):
    jj = np.zeros(len(v))
    jj[:n] = 1.0
    return jj


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: 1D coordinate descent (many sweeps)
    stall = 0
    for sweep in range(3000):
        if tleft() < budget_s * 0.38:
            break
        perm = rng.permutation(n)
        for j in perm:
            h[j] = coord_1d(h, j)
        h = project_h(h, n * 0.5)
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-13:
            c5_best = c5
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 150:
            break

    # Phase 2: Pairwise LP refinement
    for it in range(2000):
        if tleft() < budget_s * 0.65:
            break
        frac = np.where((h_best > 0.005) & (h_best < 0.995))[0]
        if len(frac) < 2:
            break
        n_pairs = min(30, len(frac) * (len(frac) - 1) // 2)
        for _ in range(n_pairs):
            idx = rng.choice(len(frac), 2, replace=False)
            j, l = frac[idx[0]], frac[idx[1]]
            nj, nl = pairwise_lp_solve(h_best, j, l)
            h_best[j] = nj
            h_best[l] = nl
        c5 = compute_c5(h_best)
        if c5 < c5_best:
            c5_best = c5

    # Phase 3: More 1D CD after LP
    h = h_best.copy()
    stall = 0
    for sweep in range(2000):
        if tleft() < budget_s * 0.82:
            break
        perm = rng.permutation(n)
        for j in perm:
            h[j] = coord_1d(h, j)
        h = project_h(h, n * 0.5)
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-13:
            c5_best = c5
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 100:
            break

    # Phase 4: SLSQP polish with t-variable
    if tleft() > 80:
        c5_cur = compute_c5(h_best)
        v0 = np.append(h_best.copy(), c5_cur + 0.003)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': lambda v: t_ineq_v(v, n)},
            {'type': 'eq', 'fun': lambda v: t_eq_v(v, n), 'jac': lambda v: t_eq_jac_v(v, n)}
        ]
        res = minimize(lambda v: t_obj_v(v, n), v0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        if res.success:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_h(h_new, n * 0.5)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new

    # Phase 5: Final 1D CD polish
    if tleft() > 30:
        h = h_best.copy()
        for sweep in range(500):
            if tleft() < 10:
                break
            perm = rng.permutation(n)
            for j in perm:
                h[j] = coord_1d(h, j)
            h = project_h(h, n * 0.5)
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
