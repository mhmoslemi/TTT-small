import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(x, dx):
    return compute_c5(x, dx)

def con_func(x, target):
    return np.sum(x) - target

def con_jac(x, target):
    return np.ones(len(x))

def slsqp_optimize(h_start, n, dx, target_sum, max_iter=300, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': con_func, 'jac': con_jac, 'args': (target_sum,), 'jac_args': (target_sum,)}
    res = minimize(obj_func, h_start, args=(dx,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    return project_feasible(res.x, n)

def run_de(h_init, n, dx, target_sum, rng, pop_size, max_gens,
           F_start, F_end, Cr_start, Cr_end, time_limit, t0):

    for i in range(4, pop_size):

    fit = np.array([compute_c5(pop[i], dx) for i in range(pop_size)])
    best_i = int(np.argmin(fit))
    best_h = pop[best_i].copy()
    best_f = fit[best_i]
    for gen in range(max_gens):
        if time.time() - t0 > time_limit:
            break
        progress = gen / max_gens
        F = F_start + (F_end - F_start) * progress
        Cr = Cr_start + (Cr_end - Cr_start) * progress
        for i in range(pop_size):
            if rng.random() < 0.4:
                others = [j for j in range(pop_size) if j != i]
                a, b, c = rng.choice(others, 3, replace=False)
                v = pop[a] + F * (pop[b] - pop[c])
            else:
                others = [j for j in range(pop_size) if j != best_i and j != i]
                p1, p2 = rng.choice(others, 2, replace=False)
                v = pop[best_i] + F * (pop[p1] - pop[p2])
            mask = rng.random(n) < Cr
            mask[rng.randint(n)] = True

            u = project_feasible(u, n)
            fu = compute_c5(u, dx)

                pop[i] = u
                fit[i] = fu
            if fu < best_f:
                best_f = fu
                best_h = u.copy()
                best_i = i
    return best_h, best_f

def guided_perturb(h, n, rng, strength):

    mx = np.max(corr)
    active = np.where(corr > 0.93 * mx)[0]
    h_try = h.copy()

            idx2 = np.arange(lag_abs, n)
            h_try[idx2] += strength * h[idx2 - lag_abs] / n_active
    h_try += rng.randn(n) * strength * 0.3
    return project_feasible(h_try, n)
