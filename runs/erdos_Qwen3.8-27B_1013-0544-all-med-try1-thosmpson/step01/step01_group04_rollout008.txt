import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Overflow protection
    if np.max(shifted) > 500:
        return (m + np.log(2*n-1) / beta) * dx  # approx
    log_sum = m + np.log(np.sum(np.exp(shifted)))
    return log_sum * dx

def smooth_objective_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    if np.max(shifted) > 500:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = np.exp(shifted) / np.sum(np.exp(shifted))
    # grad[i] = dx * sum_k p_k * (1 - h[i+l_k] - h[i-l_k])
    # where l_k = k - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        for k in range(2*n-1):
            if p[k] < 1e-15:
                continue
            l = k - (n-1)
            hv = 0.0
            if 0 <= i+l < n:
                hv += h[i+l]
            if 0 <= i-l < n:
                hv += h[i-l]
            val += p[k] * (1.0 - hv)
        grad[i] = val * dx
    return grad

def compute_c5(h, dx):

    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:

        if np.any(free):

            h = np.clip(h, 0.0, 1.0)

        p = exps / np.sum(exps)

            val += p[k] * (1.0 - hv)

    return obj, grad

    best_c5 = compute_c5(h, dx)
    for sweep in range(max_sweeps):
        improved = 0.0
        for i in range(n):
            old = h[i]
            current_sum = np.sum(h)
            # 1D objective: minimize max_k corr[k](t) + penalty*(current_sum - target + t - old)^2

            h[i] = 0.0

            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = old
            A = corr0
            B = corr1 - corr0

                    obj_fn, h_try, jac=grad_fn,
                    method='SLSQP', bounds=bounds, constraints=constraint,
                    options={'maxiter': 200, 'ftol': 1e-15}
                )
                h2 = project_feasible(res2.x, n)
                c5_2 = compute_c5(h2, dx)
                if c5_2 < best_c5 - 1e-15:
                    best_h = h2
                    best_c5 = c5_2
                    improved = True
            if not improved:
                break

    return (best_h, best_c5, n)
