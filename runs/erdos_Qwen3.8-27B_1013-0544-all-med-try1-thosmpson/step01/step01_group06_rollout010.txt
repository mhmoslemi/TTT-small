```python
def project_onto_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_at_shift(h, k):
    """Gradient of corr[k] with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def topk_gradient(h, top_k=5, sharpness=40):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    w = np.exp(sharpness * (top_vals - top_vals.max()))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(top_idx)); w[-1] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        k = idx - (n - 1)
        grad += w[i] * grad_at_shift(h, k)
    return grad * (2.0 / n)

def smooth_grad_full(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr); w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_at_shift(h, k)
    return grad * (2.0 / n)

def optimize_hybrid(h0, target_sum, deadline, rng, mode='topk'):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.015
    momentum = 0.85
    alpha = 50
    top_k = 5
    patience = 0
    stall = 0
    best_lr = lr

    while time.time() < deadline:
        if mode == 'topk':
            grad = topk_gradient(h, top_k, alpha)
        else:
            grad = smooth_grad_full(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            best_lr = lr
            if patience == 0 and lr < 0.04:
                lr = min(lr * 1.04, 0.04)
            if alpha < 300:
                alpha = min(alpha * 1.03, 300)
        else:
            h = best_h.copy()
            patience += 1
            lr *= 0.93
            if patience > 40:
                pert_scale = 0.015 + 0.01 * stall
                h_pert = best_h + rng.randn(n) * pert_scale
                h_pert = project_box_sum(h_pert, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
                    h = best_h.copy()
                v = np.zeros(n)
                lr = best_lr * 0.5
                alpha = max(20, alpha * 0.5)
                patience = 0
                stall += 1
                if stall > 12:
                    break
    return best_h, best_c5

def local_search(h0, target_sum, deadline, rng):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    scale = 0.03
    while time.time() < deadline:
        gs = rng.randint(1, min(8, n // 15 + 1))
        idx = rng.choice(n, size=gs, replace=False)
        delta = rng.randn(gs) * scale
        h_new = h.copy()
        h_new[idx] += delta
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            scale = min(scale * 1.1, 0.08)
        else:
            scale = max(scale * 0.95, 0.005)
    return h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = 0.5 * np.ones(100)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n

    # Phase 1: Top-k subgradient at native resolution
    d1 = t0 + budget_s * 0.35
    best_h, best_c5 = optimize_hybrid(h0, target_sum, d1, rng, mode='topk')
    best_n = len(best_h)

    # Phase 2: Local search polish
    elapsed = time.time() - t0
    rem = budget_s - elapsed - 20
    if rem > 25:
        d2 = t0 + budget_s * 0.5
        best_h, best_c5 = local_search(best_h, len(best_h) / 2.0, d2, rng)

    # Phase 3: Try smooth-grad mode at same resolution
    elapsed = time.time() - t0
    rem = budget_s - elapsed - 20
    if rem > 50:
        d3 = t0 + budget_s * 0.65
        h_smooth, c5_smooth = optimize_hybrid(best_h.copy(), len(best_h) / 2.0, d3, rng, mode='smooth')
        if c5_smooth < best_c5:
            best_h = h_smooth
            best_c5 = c5_smooth

    # Phase 4: Try lower resolutions
    elapsed = time.time() - t0
    rem = budget_s - elapsed - 30
    for n_try in [120, 80, 60]:
        if rem < 50:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        d4 = time.time() + min(rem * 0.4, 120)
        h_opt, c5_opt = optimize_hybrid(h_interp, n_try / 2.0, d4, rng, mode='topk')
        # Quick local search
        if compute_c5(h_opt) < best_c5:
            d5 = time.time() + min(rem * 0.1, 30)
            h_opt, c5_opt = local_search(h_opt, n_try / 2.0, d5, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        rem = budget_s - (time.time() - t0) - 20

    # Phase 5: Final local search
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h, best_c5 = local_search(best_h, len(best_h) / 2.0, t0 + budget_s - 2, rng)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
