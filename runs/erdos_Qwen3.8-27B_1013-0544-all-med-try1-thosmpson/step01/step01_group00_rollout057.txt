    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}"""
    h = h.copy()
    # Bisection on tau
    lo, hi = -1.0, 1.0
    for _ in range(100):
        tau = (lo + hi) / 2
        projected = np.clip(h - tau, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_objective(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = (corr - np.max(corr)) / temp
    s = np.exp(shifted)
    s /= s.sum()

    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]

    ip = i_idx + k
    iq = i_idx - k
    valid_p = (ip >= 0) & (ip < n)
    valid_q = (iq >= 0) & (iq < n)
    ip_c = np.clip(ip, 0, n - 1)
    iq_c = np.clip(iq, 0, n - 1)
    
    h_ip = np.where(valid_p, h[ip_c], 0.0)
    h_iq = np.where(valid_q, h[iq_c], 0.0)
    
    s_r = s[None, :]
    grad = dx * ((s_r * valid_p).sum(1) - (s_r * h_ip).sum(1) - (s_r * h_iq).sum(1))
    return grad

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.5

    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    def obj_and_grad(h):
        val = smooth_objective(h, temp)
        g = smooth_gradient(h, temp)
        return val, g

                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def smooth_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = (corr - np.max(corr)) / temp
    s = np.exp(shifted)
    s /= s.sum()
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    k = m_idx - (n - 1)
    ip = i_idx + k
    iq = i_idx - k
    valid_p = (ip >= 0) & (ip < n)
    valid_q = (iq >= 0) & (iq < n)
    ip_c = np.clip(ip, 0, n - 1)
    iq_c = np.clip(iq, 0, n - 1)
    h_ip = np.where(valid_p, h[ip_c], 0.0)
    h_iq = np.where(valid_q, h[iq_c], 0.0)
    s_r = s[None, :]
    return dx * ((s_r * valid_p).sum(1) - (s_r * h_ip).sum(1) - (s_r * h_iq).sum(1))

            hb, cb = binary_sa(h_bin, n_iter=150000, T_init=0.004, T_final=0.00003, seed=sa_seed)
            if cb < c5_best:
                h_best = hb
                c5_best = cb
                h_bin = hb.copy()

    # Phase 4: Projected gradient polishing on current best
    for temp, lr in [(0.001, 0.03), (0.0005, 0.015), (0.0002, 0.008)]:

        if cn < c5_best:

    n_cur = len(h_best)

    for i in range(len(pert_scales)):

            break
        sc = pert_scales[i]
        if i % 3 == 2:
            # Fourier perturbation
            H = np.fft.rfft(h_best)
            freqs = np.arange(len(H))
            mask = (freqs > n_cur * 0.1) & (freqs < n_cur * 0.4)
            H[mask] += sc * rng.randn(*H[mask].shape)
            h_pert = np.real(np.fft.irfft(H, n=n_cur))
        else:

        h_pert = project_feasible(h_pert, n_cur * 0.5)
        hn, cn = t_var_slsqp(h_pert, max_iter=400)
        if cn < c5_best:
            h_best, c5_best = hn, cn

    for n_try in [100, 120, 150, 200, 240, 300]:
        if rem() < 200:
            break
        if n_try == len(h_best):
            continue

        h_cur = h_int
        c5_cur = compute_c5(h_cur)
        for temp in [0.005, 0.002, 0.001]:
            if rem() < 100:
                break
            hn, cn = slsqp_smooth(h_cur, temp, max_iter=300)
            if cn < c5_cur:
                h_cur, c5_cur = hn, cn
        if rem() > 120:
            hn, cn = t_var_slsqp(h_cur, max_iter=500)
            if cn < c5_cur:
                h_cur, c5_cur = hn, cn
        if c5_cur < c5_best:
            h_best, c5_best = h_cur, c5_cur
