import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_f(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()

def slsqp_step(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_gradient(h, n, alpha, steps=5000, init_lr=0.01):
    """Momentum-based projected gradient descent."""
    best = h.copy()
    best_c5 = compute_c5(h, n)
    v = np.zeros(n)
    mu = 0.9
    lr = init_lr
    
    for step in range(steps):
        g = grad_smooth(h, n, alpha)
        v = mu * v - lr * g
        h_new = h + v  # Note: v already has the negative sign from gradient
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best = h_new.copy()
            best_c5 = c5
        h = h_new
        # Decay learning rate
        if step % 500 == 0:
            lr *= 0.8
    
    return best, best_c5

    target = n / 2.0
    # Find lambda such that sum(clip(h_i - lambda, 0, 1)) = target
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0, 1)
    # Adjust for floating point
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # Distribute the remaining difference
        if diff > 0:
            room = 1.0 - x
            total = room.sum()
            if total > 0:
                x += (diff / total) * room
        else:
            total = x.sum()
            if total > 0:
                x += (diff / total) * x
    return np.clip(x, 0, 1)

def compute_f(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def obj_smooth(h, n, alpha):
    f = compute_f(h, n)
    mx = alpha * f.max()
    return float((mx + np.log(np.sum(np.exp(alpha * f - mx)))) / alpha)

def grad_smooth(h, n, alpha):

    mx = alpha * f.max()
    p = np.exp(alpha * f - mx)
    p /= p.sum()
    grad = np.zeros(n)

        w = p[(n - 1) + lag]
        if abs(w) < 1e-16:

        j0 = max(0, lag)
        j1 = min(n - 1, n - 1 + lag)
        if j0 <= j1:
            grad[j0:j1 + 1] += w * (1.0 - h[j0 - lag:j1 - lag + 1])
        j0b = max(0, -lag)
        j1b = min(n - 1, n - 1 - lag)
        if j0b <= j1b:
            grad[j0b:j1b + 1] -= w * h[j0b + lag:j1b + lag + 1]

                h_try = best_h.copy()
                h_try[i] = hi
                h_try[j] = hj
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h = h_try
                    best_c5 = c
                    improved = True
                    break
        if not improved and count % 500 == 0:
            g = grad_smooth(best_h, n, 5000)
            for _g in range(5):
                ii = int(np.argmax(g))
                jj = int(np.argmin(g))
                if ii == jj:
                    break
                dd = 0.005
                hi = best_h[ii] - dd
                hj = best_h[jj] + dd
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_try = best_h.copy()
                    h_try[ii] = hi
                    h_try[jj] = hj
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_h = h_try
                        best_c5 = c

    if budget_s - (time.time() - t0) > 15:
        h_try = slsqp_opt(best_h, n, 10000, maxiter=5000)
        c_try = compute_c5(h_try, n)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
