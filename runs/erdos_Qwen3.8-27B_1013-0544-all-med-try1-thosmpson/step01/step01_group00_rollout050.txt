def grad_lse_step(h, temp, step_size, n_steps=100):
    """Take gradient steps on the LSE objective"""
    n = len(h)
    dx = 2.0 / n

        m = np.max(corr)
        weights = np.exp((corr - m) / temp)
        weights /= np.sum(weights)

        # corr_k = sum_i h[i] * (1-h[i+k_a]) * dx where k_a is the appropriate shift
        # For full correlation indexed from -(n-1) to (n-1):
        # corr_full[m] = sum_i h[i] * (1-h[i - (m-n+1)]) * dx
        # d corr_full[m] / d h[j] = (1-h[j-(m-n+1)])*dx - h[j+(m-n+1)]*dx (if indices valid)

        # grad = sum_k weights[k] * d(corr_k)/d(h)
        # Approximate with finite differences for a few key coordinates
        
        # Actually, let me just use the fact that:
        # corr_full[m] = sum_{i: valid} h[i] * (1 - h[i - (m - n + 1)])
        # Let lag = m - n + 1
        # = sum_i h[i] - sum_i h[i] * h[i - lag]
        # For valid i: 0 <= i < n and 0 <= i-lag < n
        
        # d/dh[j] of sum_i h[i]*(1-h[i-lag]):
        # = sum over i where i=j: (1-h[j-lag]) + sum over i where i-lag=j: -h[i]
        # = (1-h[j-lag])*[0<=j-lag<n] - h[j+lag]*[0<=j+lag<n]
        
        grad = np.zeros(n)
        for m_idx in range(len(corr)):
            w = weights[m_idx]
            if w < 1e-10:
                continue
            lag = m_idx - (n - 1)
            if lag >= 0:
                # i ranges from lag to n-1
                for j in range(lag, n):
                    grad[j] += w * (1 - h_cur[j - lag]) * dx
                    if j + lag < n:
                        grad[j] += w * (-h_cur[j + lag]) * dx

        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lse_refine(h0, temp, lr, n_steps=200):
    n = len(h0)
    dx = 2.0 / n
    h_cur = h0.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for _ in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        m = np.max(corr)
        w = np.exp((corr - m) / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for m_idx in range(len(corr)):
            wi = w[m_idx]
            if wi < 1e-12:
                continue
            lag = m_idx - (n - 1)
            if lag >= 0:
                idx = np.arange(lag, n)
                grad[idx] += wi * (1.0 - h_cur[idx - lag]) * dx
                valid = idx + lag < n
                grad[idx[valid]] -= wi * h_cur[idx[valid] + lag] * dx
            else:
                idx = np.arange(0, n + lag)
                grad[idx] += wi * (1.0 - h_cur[idx - lag]) * dx
                valid = idx + lag < n
                grad[idx[valid]] -= wi * h_cur[idx[valid] + lag] * dx
        h_trial = h_cur - lr * grad
        h_trial = project_sum_box(h_trial, n * 0.5)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h_cur = h_trial
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5_best = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    c5_cur = c5_best

    # Phase 1: t-var optimization at original n (180)
    for ftol in [1e-10, 1e-11, 1e-12]:
        if tleft() < 50:
            break
        h_new, c5_new = opt_t(h_cur, max_iter=400, ftol=ftol)
        if c5_new < c5_cur:
            h_cur, c5_cur = h_new, c5_new
            if c5_new < c5_best:
                h_best, c5_best = h_new.copy(), c5_new

    # LSE smoothing at original n
    for temp in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if tleft() < 40:
            break
        h_new, c5_new = opt_lse(h_cur, temp, max_iter=300, ftol=1e-12)
        if c5_new < c5_cur:
            h_cur, c5_cur = h_new, c5_new
            if c5_new < c5_best:
                h_best, c5_best = h_new.copy(), c5_new

    # Gradient-based refinement at original n
    if tleft() > 60:
        h_new, c5_new = grad_lse_refine(h_cur, 0.002, 0.005, n_steps=150)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new
            h_cur, c5_cur = h_new, c5_new

    # Phase 2: Upgrade to n=360
    n2 = 360
    h360 = interp_h(h_best, n2)
    c5_360 = compute_c5(h360)
    h_best_360 = h360.copy()
    c5_best_360 = c5_360

    if tleft() > 120:
        h_new, c5_new = opt_t(h360, max_iter=500, ftol=1e-11)
        if c5_new < c5_best_360:
            h360, c5_360 = h_new, c5_new
            h_best_360, c5_best_360 = h360.copy(), c5_new
            if c5_new < c5_best:
                h_best, c5_best = h360.copy(), c5_new

    # LSE at n=360
    for temp in [0.01, 0.005, 0.002, 0.001]:
        if tleft() < 70:
            break
        h_new, c5_new = opt_lse(h360, temp, max_iter=300, ftol=1e-12)
        if c5_new < c5_best_360:
            h360, c5_360 = h_new, c5_new
            h_best_360, c5_best_360 = h360.copy(), c5_new
            if c5_new < c5_best:
                h_best, c5_best = h360.copy(), c5_new

    # Gradient refinement at n=360
    if tleft() > 80:
        h_new, c5_new = grad_lse_refine(h360, 0.002, 0.003, n_steps=200)
        if c5_new < c5_best_360:
            h360, c5_360 = h_new, c5_new
            h_best_360, c5_best_360 = h360.copy(), c5_new
            if c5_new < c5_best:
                h_best, c5_best = h360.copy(), c5_new

    # Another t-var pass at n=360
    if tleft() > 120:
        h_new, c5_new = opt_t(h360, max_iter=500, ftol=1e-12)
        if c5_new < c5_best_360:
            h360, c5_360 = h_new, c5_new
            h_best_360, c5_best_360 = h360.copy(), c5_new
            if c5_new < c5_best:
                h_best, c5_best = h360.copy(), c5_new

    # Phase 3: Diversified restarts at n=360
    for i in range(12):
        if tleft() < 50:
            break
        if i % 4 == 0:
            h_p = h_best_360 + rng.randn(n2) * (0.02 + 0.008 * i)
        elif i % 4 == 1:
            h_p = h_best_360.copy()
            s = rng.randint(0, n2 - 40)
            h_p[s:s+40] += rng.randn(40) * 0.08
        elif i % 4 == 2:
            sh = rng.randint(1, 8)
            h_p = np.roll(h_best_360, sh) + rng.randn(n2) * 0.015
        else:
            h_p = h_best_360.copy()
            for _ in range(3):
                s = rng.randint(0, n2 - 20)
                h_p[s:s+20] += rng.randn(20) * 0.06
        h_p = project_h(h_p, n2 * 0.5)
        h_new, c5_new = opt_t(h_p, max_iter=300, ftol=1e-11)
        if c5_new < c5_best_360:
            h360 = h_new
            h_best_360, c5_best_360 = h_new.copy(), c5_new
            if c5_new < c5_best:
                h_best, c5_best = h_new.copy(), c5_new

    # Phase 4: Final refinement
    if tleft() > 50:
        h_new, c5_new = opt_t(h_best, max_iter=400, ftol=1e-13)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    if tleft() > 30:
        h_new, c5_new = opt_lse(h_best, 0.0005, max_iter=200, ftol=1e-13)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    if tleft() > 20:
        h_new, c5_new = grad_lse_refine(h_best, 0.001, 0.002, n_steps=100)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
