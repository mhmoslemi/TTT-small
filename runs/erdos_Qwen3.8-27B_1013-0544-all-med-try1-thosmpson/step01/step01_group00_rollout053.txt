    m = np.max(corr)
    shifted = (corr - m) / temp
    # Numerically stable log-sum-exp
    max_s = np.max(shifted)
    return m + temp * (max_s + np.log(np.sum(np.exp(shifted - max_s))))

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def slsqp_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def slsqp_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def slsqp_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def slsqp_obj(v):
    return v[-1]


def softmax_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    max_s = np.max(shifted)
    return m + temp * (max_s + np.log(np.sum(np.exp(shifted - max_s)))) - temp * np.log(len(corr))


def sum_constraint(h):
    return np.sum(h) - len(h) * 0.5


def sum_constraint_jac(h):
    return np.ones(len(h))


def optimize_tvar(h0, max_iter=500, ftol=1e-12):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.005)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    res = minimize(slsqp_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_softmax(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    res = minimize(softmax_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def subgradient_step(h, lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] += (1.0 - h[lag:n]) * dx
        grad[lag:] -= h[:n - lag] * dx
    elif lag < 0:
        la = -lag
        grad[la:] += (1.0 - h[:n - la]) * dx
        grad[:n - la] -= h[la:n] * dx
    else:
        grad = (1.0 - 2.0 * h) * dx
    return grad


def subgradient_descent(h0, n_iters=300, lr0=0.04):
    n = len(h0)
    target = n * 0.5
    h_cur = h0.copy()
    best = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * (2.0 / n)
        k_max = np.argmax(corr)
        lag = k_max - n + 1
        grad = subgradient_step(h_cur, lag)
        lr = lr0 / (1.0 + 0.03 * it)
        h_new = h_cur - lr * grad
        h_new = project_h(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best:
            best = c_new
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best


def accel_subgradient(h0, n_iters=200, lr0=0.02):
    n = len(h0)
    target = n * 0.5
    h_cur = h0.copy()
    h_prev = h0.copy()

    best_h = h_cur.copy()

        h_y = (1.0 - t_k) * h_cur + t_k * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode='full') * (2.0 / n)

        lag = k_max - n + 1

        lr = lr0 / (it + 1.0)
        h_new = h_y - lr * grad
        h_new = project_h(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best:
            best = c_new
            best_h = h_new.copy()
        h_prev = h_cur
        h_cur = h_new
    return best_h, best

    x_old = np.linspace(0, 1, n_old, endpoint=False) + 0.5 / n_old

    h_new = np.interp(x_new, x_old, h_old)
