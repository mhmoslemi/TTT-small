from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def softmax_weighted_grad(h, T):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    shifted = T * (corr - corr.max())
    exp_v = np.exp(shifted)
    p = exp_v / exp_v.sum()
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-10:
            continue
        L = k - n_half
        if L == 0:
            grad += w * (omh - h)
        elif L > 0:
            grad[L:] += w * omh[:n - L]
            grad[:n - L] -= w * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += w * omh[L2:]
            grad[L2:] -= w * h[:n - L2]
    return grad

def active_set_grad(h, n_top=10):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    n_half = n - 1
    c_max = float(np.max(corr))
    tol = max(0.05, c_max * 0.001)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 3:
        active = np.argsort(corr)[-max(3, len(active)):]
    elif len(active) > n_top:
        active = np.argsort(corr)[-n_top:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        if L == 0:
            grad[:] += omh - h
        elif L > 0:
            grad[L:] += omh[:n - L]
            grad[:n - L] -= h[L:]
        else:
            L2 = -L
            grad[:n - L2] += omh[L2:]
            grad[L2:] -= h[:n - L2]
    grad /= float(len(active))
    return grad

def optimize_smoothed(h, n, target_sum, dx, T_start, T_end, lr_init, max_iter, tleft_fn):
    """Projected gradient with soft-max smoothing and backtracking."""
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    momentum = np.zeros(n)
    mu = 0.65
    lr = lr_init
    stag = 0

    for it in range(max_iter):
        if tleft_fn() < 30:
            break
        frac = it / max(max_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        grad = softmax_weighted_grad(h, T)
        momentum = mu * momentum - lr * grad
        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(10):
            h_new = project_feasible(h + alpha * momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 + 1e-12:
                h = h_new
                improved = True
                break
            alpha *= 0.5
        if improved:
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stag = 0
            else:
                stag += 1
        else:
            stag += 1
            lr *= 0.7
            momentum = np.zeros(n)

                lr = lr_init
                stag = 0
    return best_h, best_c5

def optimize_active_set(h, n, target_sum, dx, tleft_fn):

def slsqq_polish(h, n, target_sum, dx):
    """SLSQP refinement."""
    def obj(x):
        return compute_c5(x, dx)
    def eq(x):
        return np.sum(x) - target_sum
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': 3000, 'ftol': 1e-15})
        if res.x is not None:
            h_ref = project_feasible(res.x, target_sum)
            c5_ref = compute_c5(h_ref, dx)
            return h_ref, c5_ref
    except Exception:
        pass
    return h, compute_c5(h, dx)

def make_initial(n, seed, style=0):
    """Generate algorithmic initializations."""
    rng = np.random.RandomState(seed)
    if style == 0:
        return np.full(n, 0.5)
    elif style == 1:
        # Binary-ish: random with 50% ones
        h = (rng.rand(n) > 0.5).astype(float)
        return h
    elif style == 2:
        # Periodic-like
        period = max(3, n // 7)
        h = np.tile([0.0, 0.5, 1.0][:n % 3 or 3], n // 3 + 1)[:n]
        h = np.clip(h + rng.normal(0, 0.1, n), 0, 1)
        return h
    elif style == 3:
        # Fourier-based
        k = rng.randint(2, n // 2)
        h = 0.5 + 0.3 * np.cos(2 * np.pi * k * np.arange(n) / n)
        return np.clip(h, 0, 1)
    else:
        return np.clip(rng.uniform(0.1, 0.9, n), 0, 1)
