```python
def _subgrad_step(h, target, top_k=5, step=0.01):
    """Compute subgradient of the max and take a step."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    
    # Gradient of corr[lag] with respect to h[j]
    # corr[m] = sum_j h[j+m] * (1-h[j]) for valid j (in full correlation)
    # Actually, let me just use the known formula
    # For np.correlate(a, v, 'full'):
    # out[k] = sum_i a[i+k] * v[i] for i where both indices are valid
    # Here a=h, v=1-h
    # out[k] = sum_i h[i+k]*(1-h[i])
    # d(out[k])/d(h[j]) = (1-h[j-k]) [when j-k is valid, from the h[i+k] term with i=j-k]
    #                    + sum_i h[i+k]*(-delta(j-i)) [from the (1-h[i]) term]
    #                    = (1-h[j-k]) - h[j+k] [when j+k is valid]
    
    # This is getting complicated. Let me just use numerical gradient for the top terms.
    # Actually, for the purpose of subgradient descent, I can approximate:
    # The subgradient of max_k f_k(h) is the gradient of one of the active f_k.
    # For f_k(h) = sum_j h[j+k](1-h[j]), the gradient w.r.t. h[j] is:
    #   (1 - h[j-k]) - h[j+k] (with boundary handling)
    
    grad = np.zeros(n)
    for li in top_idx:
        lag = li - (n - 1)  # convert to actual lag (can be negative)
        for j in range(n):
            # Contribution from h[j] as the "first" factor: (1-h[j-lag]) if j-lag in [0,n)
            j1 = j - lag
            if 0 <= j1 < n:
                grad[j] += (1.0 - h[j1])
            # Contribution from h[j] as the "second" factor: -h[j+lag] if j+lag in [0,n)
            j2 = j + lag
            if 0 <= j2 < n:
                grad[j] -= h[j2]
    
    # Normalize
    gn = np.linalg.norm(grad)
    if gn < 1e-12:
        return h
    # Move in negative gradient direction
    h_new = h - step * grad / gn
    return project_h(h_new, target)
```

```python
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _pen_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, _g_beta) * _g_dx + _g_mu * (x.sum() - _g_target) ** 2

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, _g_beta) * _g_dx

def _con(x):
    return x.sum() - _g_target

def _jac(x):
    return np.ones(x.shape)

def _time_left(t0, budget):
    return budget - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_beta, _g_dx, _g_target, _g_mu
    np.random.seed(seed)
    t0 = time.time()
    budget = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n = 560
    _g_target = n / 2.0
    _g_dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    x_old = np.linspace(0, 2, len(h_parent))
    x_new = np.linspace(0, 2, n)
    h0 = project_h(np.interp(x_new, x_old, h_parent), _g_target)
    c5_0 = compute_c5(h0)
    if c5_0 < best_c5:
        best_h, best_c5 = h0.copy(), c5_0

    # Phase 1: L-BFGS-B with penalty (fast exploration)
    _g_mu = 500.0
    for beta in [25, 50, 100, 200, 400]:
        if _time_left(t0, budget) < 300:
            break
        _g_beta = float(beta)
        for attempt in range(2):
            if _time_left(t0, budget) < 150:
                break
            try:
                res = minimize(_pen_obj, h0, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'disp': False})
                h1 = project_h(res.x, _g_target)
                c5_1 = compute_c5(h1)
                if c5_1 < c5_0:
                    h0, c5_0 = h1, c5_1
                if c5_1 < best_c5:
                    best_h, best_c5 = h1.copy(), c5_1
            except Exception:
                pass
            if attempt == 0:
                h0 = project_h(h0 + np.random.randn(n) * 0.01, _g_target)

    # Phase 2: SLSQP with increasing sharpness
    for beta in [100, 250, 500, 1000, 2000, 4000]:
        if _time_left(t0, budget) < 80:
            break
        _g_beta = float(beta)
        h_cur = h0.copy()
        for restart in range(2):
            if _time_left(t0, budget) < 40:
                break
            try:
                res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': _con, 'jac': _jac}],
                               options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
                h_new = project_h(res.x, _g_target)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_cur):
                    h_cur = h_new
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass
            if restart == 0 and _time_left(t0, budget) > 40:
                mask = np.random.random(n) < 0.3
                h_cur = project_h(np.clip(h_cur + np.random.randn(n) * 0.025 * mask, 0, 1),
                                  _g_target)
        h0 = h_cur

    # Phase 3: Simulated annealing for local escape
    if _time_left(t0, budget) > 60:
        cur_h = best_h.copy()
        cur_c5 = best_c5
        T = 0.002
        for it in range(8000):
            if _time_left(t0, budget) < 20:
                break
            bs = np.random.randint(0, max(1, n - 10))
            sz = np.random.randint(3, 12)
            old = cur_h[bs:bs + sz].copy()
            cur_h[bs:bs + sz] = np.clip(old + np.random.randn(sz) * 0.05, 0, 1)
            cur_h = project_h(cur_h, _g_target)
            nc = compute_c5(cur_h)
            if nc < cur_c5 or np.random.random() < np.exp(-(nc - cur_c5) / T):
                cur_c5 = nc
                if nc < best_c5:
                    best_h, best_c5 = cur_h.copy(), nc
            else:
                cur_h[bs:bs + sz] = old
                cur_h = project_h(cur_h, _g_target)
            T = max(0.0001, T * 0.999)

    # Phase 4: Final high-precision SLSQP polish
    if _time_left(t0, budget) > 10:
        _g_beta = 5000.0
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _con, 'jac': _jac}],
                           options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False})
            h_f = project_h(res.x, _g_target)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f
        except Exception:
            pass

    return best_h, best_c5, int(len(best_h))
```
