def project_h(h, target_sum):

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_phase(h, target, dx, betas, max_iter_per_beta, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 15:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter_per_beta, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def subgrad_phase(h, target, time_fn, max_iters, m_factor, step_list, perturb_every, perturb_scale, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(m_factor, max(3, n // 8))
        top = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]
        g = compute_subgrad(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in step_list:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)

                    best = ht
                    best_c5 = c5t
                    vel *= 0.5
                    no_improve = 0
            if no_improve > perturb_every * 10:
                break
    return best, best_c5
