    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx

        grad[l:] += -h[:n-l] * dx
    return grad

        n_try = n_parent + delta
        if 80 <= n_try <= 500:
            n_list.append(n_try)
    n_list = sorted(set(n_list))

    for n in n_list:
        if tl() < 60:
            break
        target = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        x = np.linspace(0, 2, n)

        # Build diverse inits
        inits = [project_h(resample(h_parent, n), target)]
        if best_state[2] != n:
            inits.append(project_h(resample(best_state[1], n), target))
        # Cosine modes
        for mode in [1, 2, 3, 4]:
            for amp in [0.25, 0.35, 0.45]:
                hc = 0.5 + amp * np.cos(mode * np.pi * x)
                inits.append(project_h(hc, target))
        # Asymmetric
        ha = 0.5 + 0.2 * np.cos(np.pi * x) + 0.15 * np.sin(2 * np.pi * x)
        inits.append(project_h(ha, target))
        # Bump
        hb = 0.5 + 0.3 * np.exp(-((x - 1.0) / 0.3) ** 2)
        inits.append(project_h(hb, target))

        # Limit number of inits to keep time manageable
        max_inits = 8 if tl() > 500 else 5
        inits = inits[:max_inits]

        for h_init in inits:
            if tl() < 30:
                break
            cur = h_init.copy()
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n

            # SLSQP phase with increasing beta
            for beta in [40, 80, 160, 320, 640]:
                if tl() < 12:
                    break
                try:
                    res = minimize(lambda hh: obj_sm(hh, dx, beta), cur,
                                   method='SLSQP', bounds=bounds,
                                   args=(),
                                       constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target,)}],
                                   options={'maxiter': 800, 'ftol': 1e-15})
                    hn = project_h(res.x, target)
                    c5n = compute_c5(hn)
                    if c5n < cur_c5:
                        cur = hn
                        cur_c5 = c5n
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
                except Exception:
                    pass

            # Subgradient Adam phase
            cur, cur_c5 = subgrad_opt(cur, n, target, dx,
                                      max_iters=8000, time_fn=tl, min_time=8,
                                      best_state=best_state)

    # Final aggressive polish
    if tl() > 10:
        bh = best_state[1].copy()
        bc5 = best_state[0]
        bn = len(bh)
        tf = bn / 2.0
        df = 2.0 / bn
        bf = Bounds(np.zeros(bn), np.ones(bn))

        for beta in [800, 1600, 3200, 6400, 12800]:
            if tl() < 5:
                break
            try:
                res = minimize(lambda hh: obj_sm(hh, df, beta), bh,
                               method='SLSQP', bounds=bf,
                               constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (tf,)}],
                               options={'maxiter': 5000, 'ftol': 1e-18})
                hn = project_h(res.x, tf)
                c5n = compute_c5(hn)
                if c5n < bc5:
                    bh = hn
                    bc5 = c5n
            except Exception:
                pass

        # Exact subgradient fine polish
        for it in range(30000):
            if tl() < 3:
                break
            corr = np.correlate(bh, 1.0 - bh, mode="full")
            cv = corr * df
            center = bn - 1
            # Use all lags within 5% of max for better subgradient
            thresh = cv.max() - 0.05 * (cv.max() - cv.min())
            active = np.where(cv >= thresh)[0]
            if len(active) == 0:
                active = np.argsort(cv)[::-1][:3]
            # Softmax weight over active
            av = cv[active]
            wts = np.exp((av - av[0]) * 400.0)
            wts /= wts.sum()
            g = np.zeros(bn)
            for ai, w in zip(active, wts):
                g += w * grad_lag(bh, int(ai) - center, bn, df)
            gn = np.linalg.norm(g)
            if gn < 1e-16:
                break
            g /= gn
            step = 0.0015 / (1 + 0.00005 * it)
            ok = False
            for _ in range(30):
                ht = project_h(bh - step * g, tf)
                c5t = compute_c5(ht)
                if c5t < bc5 - 1e-18:
                    bh = ht
                    bc5 = c5t
                    ok = True
                    break
                step *= 0.5
            if not ok and it > 5000:
                # Try tiny random kick
                bh2 = project_h(bh + np.random.normal(0, 1e-5, bn), tf)
                c52 = compute_c5(bh2)
                if c52 < bc5:
                    bh = bh2
                    bc5 = c52
                else:
                    if it > 15000:
                        break

    best_state[0] = min(best_state[0], bc5)
    if bc5 < best_state[0]:
        best_state[1] = bh
        best_state[2] = bn

    return best_state[1], best_state[0], best_state[2]
