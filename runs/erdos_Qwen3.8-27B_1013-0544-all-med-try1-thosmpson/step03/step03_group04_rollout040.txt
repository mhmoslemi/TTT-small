            try:
                res = minimize(lambda x: _slsqp_obj(x, beta, dx_val), h_sls,
                               method='SLSQP', bounds=bounds, constraints=constr,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_res = project_h(res.x, target)
                c5_res = compute_c5(h_res)
                if c5_res < c5_sls - 1e-15:
                    h_sls = h_res
                    c5_sls = c5_res
                if c5_res < best_c5:
                    best_h = h_sls.copy()
                    best_c5 = c5_res
            except Exception:
                pass

    # Phase 3: Local search with random perturbations (20% time)
    phase3_end = t0 + budget_s * 0.95
    h_ls = best_h.copy()
    c5_ls = best_c5
    pert_scale = 0.03

    while elapsed() < phase3_end:
        # Random perturbation
        mask = np.random.random(n) < 0.1
        pert = np.random.randn(n) * pert_scale
        pert[~mask] = 0
        h_trial = project_h(h_ls + pert, target)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_ls - 1e-14:
            h_ls = h_trial
            c5_ls = c5_trial
            if c5_ls < best_c5:
                best_h = h_ls.copy()
                best_c5 = c5_ls
            pert_scale = min(0.08, pert_scale * 1.02)
        else:
            pert_scale = max(0.005, pert_scale * 0.98)

    # Phase 4: Final SLSQP polish
    if elapsed() < t0 + budget_s - 5:
        h_final = best_h.copy()
        c5_final = best_c5
        for beta in [5000, 10000, 20000]:
            if elapsed() > t0 + budget_s - 5:
                break
            try:
                res = minimize(lambda x: _slsqp_obj(x, beta, dx_val), h_final,
                               method='SLSQP', bounds=bounds, constraints=constr,
                               options={'maxiter': 2000, 'ftol': 1e-15})
                h_res = project_h(res.x, target)
                c5_res = compute_c5(h_res)
                if c5_res < c5_final - 1e-15:
                    h_final = h_res
                    c5_final = c5_res
                if c5_res < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_res
            except Exception:
                pass

    # Final validation
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(len(best_h))
