def project_feasible(h, n):

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_grad(c, alpha):
    """Returns (soft_max_value, softmax_weights)"""
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    exp_s = np.exp(s - mx)
    log_sum = mx + np.log(exp_s.sum())
    val = m + log_sum / alpha
    w = exp_s / exp_s.sum()
    return val, w

def c5_obj_smooth(x, n, alpha, penalty=1000.0):
    c = compute_c_vec(x, n)
    val, _ = soft_max_grad(c, alpha)
    sum_pen = penalty * (x.sum() - n/2.0)**2
    return val + sum_pen

def c5_grad_smooth(x, n, alpha, penalty=1000.0):

    c = np.correlate(x, 1 - x, mode='full') * dx
    val, w = soft_max_grad(c, alpha)
    
    grad = np.zeros(n)
    n_half = n - 1

        lag = k - n_half

def random_start(n, rng, method='uniform'):
    x = np.linspace(0, 2, n)
    if method == 'uniform':
        h = rng.uniform(0, 1, n)
    elif method == 'sine':
        k = rng.randint(1, 8)
        amp = rng.uniform(0.2, 0.45)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + amp * np.sin(k * np.pi * x + phase)
    elif method == 'square':
        period = rng.randint(2, 10)
        duty = rng.uniform(0.3, 0.5)
        h = np.array([1.0 if (i % period) < period * duty else 0.0 for i in range(n)])
    elif method == 'bimodal':
        c1 = rng.uniform(0.2, 0.8)
        c2 = rng.uniform(0.2, 0.8)
        w1 = rng.uniform(0.1, 0.3)
        w2 = rng.uniform(0.1, 0.3)
        h = 0.3 + 0.7 * np.exp(-((x-c1)/w1)**2) + 0.7 * np.exp(-((x-c2)/w2)**2)
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (all top-level)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def obj_pen(x, n, alpha, pen):
    c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
    return smooth_max(c, alpha) + pen * (x.sum() - n/2.0)**2

def obj_pen_grad(x, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    es = np.exp(s - mx)
    w = es / es.sum()
    val = m + (mx + np.log(es.sum())) / alpha
    
    grad = np.zeros(n)
    nh = n - 1
    for k in range(len(c)):
        if w[k] < 1e-10:
            continue
        lag = k - nh
        wk = w[k]
        # Term 1: (1-h[i-lag]) for valid i-lag
        i1 = max(0, lag)
        i2 = min(n, n + lag)
        if i1 < i2:
            grad[i1:i2] += wk * (1.0 - x[i1-lag:i2-lag])
        # Term 2: -h[i+lag] for valid i+lag
        i3 = max(0, -lag)
        i4 = min(n, n - lag)
        if i3 < i4:
            grad[i3:i4] -= wk * x[i3+lag:i4+lag]
    grad *= dx
    grad += 2.0 * pen * (x.sum() - n/2.0)
    
    return val + pen * (x.sum() - n/2.0)**2, grad

def lbfgsb_opt(h0, n, alpha, pen, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h0, method='L-BFGS-B',
                      jac=obj_pen_grad, args=(n, alpha, pen),
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except:
        return h0

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for d in deltas:
            if time.time() > deadline:
                break
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i+1, n):
                    if time.time() > deadline:
                        break
                    for s in (1.0, -1.0):
                        hi, hj = h[i]+s*d, h[j]-s*d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi-s*d, hj+s*d
    return h, best_c

def make_starts(n, rng, h_init):

def interpolate_h(h, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)
