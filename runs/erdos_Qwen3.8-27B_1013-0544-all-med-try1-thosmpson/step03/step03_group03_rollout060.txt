import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(corr, beta, h):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    n = len(h)
    grad = np.zeros(n)
    thresh = 1e-10
    for k in range(len(corr)):
        wk = w[k] / w_sum
        if wk < thresh:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m2 = n - lag
            idx = np.arange(m2)
            grad[idx] += wk * (1.0 - h[idx + lag])
            grad[idx + lag] -= wk * h[idx]
        else:
            la = -lag
            m2 = n - la
            idx = np.arange(m2)
            grad[idx + la] += wk * (1.0 - h[idx])
            grad[idx] -= wk * h[idx + la]
    return grad

def obj_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def grad_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = smooth_max_grad(corr, beta, h)
    return g * dx

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    best_c5 = compute_c5(best_h)

    if tl() > 400:
