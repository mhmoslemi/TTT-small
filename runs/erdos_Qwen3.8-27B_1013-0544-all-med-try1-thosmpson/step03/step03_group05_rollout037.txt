```python
def benders_step(h_cur, n, target, n_active=10):
    """One Benders iteration: solve LP with linearized constraints."""
    dx = 2.0 / n
    corr = all_correlations(h_cur)
    # Get top active lags
    active_idx = np.argsort(corr)[::-1][:n_active]
    
    # Linearize: f_k(h) ≈ f_k(h_cur) + grad_k(h_cur)^T (h - h_cur)
    # We want: f_k(h) ≤ t for all active k
    # => f_k(h_cur) + grad_k^T (h - h_cur) ≤ t
    # => grad_k^T h - t ≤ grad_k^T h_cur - f_k(h_cur)
    
    # LP: min t
    # s.t. grad_k^T h - t ≤ rhs_k for active k
    #      sum(h) = target
    #      0 ≤ h ≤ 1
    
    from scipy.optimize import linprog
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    A_rows = []
    b_rows = []
    center = n - 1
    for idx in active_idx:
        lag = idx - center
        g = grad_for_single_lag(h_cur, lag, n, dx)
        f_val = corr[idx]
        # g^T h - t ≤ g^T h_cur - f_val
        # g^T h - t ≤ g^T h_cur * ... 
        # Actually: g^T (h - h_cur) + f_val ≤ t
        # g^T h - t ≤ g^T h_cur - f_val... wait
        # f_k(h) ≈ f_k(h_cur) + g^T (h - h_cur) ≤ t
        # g^T h - t ≤ g^T h_cur - f_val... no
        # f_k(h_cur) + g^T(h - h_cur) ≤ t
        # g^T h - t ≤ t - f_val + g^T h_cur - ... 
        # Let me redo: f_val + g^T h - g^T h_cur ≤ t
        # g^T h - t ≤ g^T h_cur - f_val
        a_row = np.zeros(n + 1)
        a_row[:n] = g
        a_row[-1] = -1
        b_val = g.dot(h_cur) - f_val
        A_rows.append(a_row)
        b_rows.append(b_val)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds_lp = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds_lp, method='highs')
    if res.success:
```

```python
def pairwise_exchange(h, i, j, target, dx, n):
    """Try to improve h by transferring mass from i to j (or vice versa)."""
    # h[i] -> h[i] + delta, h[j] -> h[j] - delta
    # Constraint: 0 <= h[i]+delta <= 1, 0 <= h[j]-delta <= 1
    # => -h[i] <= delta <= 1-h[i] and h[j]-1 <= delta <= h[j]
    lo = max(-h[i], h[j]-1)
    hi = min(1-h[i], h[j])
    if lo >= hi:
        return h, compute_c5(h)
    
    # Try to minimize max correlation by binary search on delta
    # This is tricky because the objective is max of linear functions in delta
    # (each correlation is linear in delta when only two variables change)
    
    # Actually each c_k is linear in delta:
    # c_k(delta) = c_k + delta * (∂c_k/∂h[i] - ∂c_k/∂h[j])
    # We want to minimize max_k c_k(delta)
    # This is a 1D piecewise-linear minimization
    
    # Compute the slope for each lag
    slopes = np.zeros(2*n-1)
    center = n - 1
    for k_idx in range(2*n-1):
        lag = k_idx - center
        gi = grad_lag(np.zeros(n), lag, n, dx)  # unit gradient for h
        # Actually, let's just compute the partial derivatives directly
        # ∂c_k/∂h[i] and ∂c_k/∂h[j]
        corr = all_corr(h)
        h_trial_i = h.copy()
        h_trial_i[i] += 1.0  # Perturb to get gradient... no this is wasteful
    
    # Better: compute analytically
    # For lag k: ∂c_k/∂h[i] = dx * (1-h[i+k]) if i < n-k (i is the "first" index)
    #                      + dx * (-h[i-k]) if i >= k (i is the "second" index)
    # This is getting complicated. Let me just do a line search.
    
    # Golden section search on [lo, hi] to minimize max correlation
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hc = h.copy(); hc[i] += c; hc[j] -= c
    hd = h.copy(); hd[i] += d; hd[j] -= d
    fc = compute_c5(np.clip(hc, 0, 1))  # Need to project but maintain sum
    fd = compute_c5(np.clip(hd, 0, 1))
    
    for _ in range(30):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            hc = h.copy(); hc[i] += c; hc[j] -= c
            hc = np.clip(hc, 0, 1)
            # Need to maintain sum... this is getting complicated
            fc = compute_c5(hc)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            hd = h.copy(); hd[i] += d; hd[j] -= d
            hd = np.clip(hd, 0, 1)
            fd = compute_c5(hd)
    
    opt_delta = (a + b) / 2
    h_new = h.copy()
    h_new[i] += opt_delta
    h_new[j] -= opt_delta
    h_new = np.clip(h_new, 0, 1)
    h_new = project_h(h_new, target)
    return h_new, compute_c5(h_new)
```

Hmm, the pairwise exchange is getting complicated because of the projection. Let me simplify: just do a line search without projection, since the sum is preserved by the exchange (h[i]+delta + h[j]-delta = h[i]+h[j]).

Actually that's a great point! If I transfer δ from j to i, the sum is preserved automatically (as long as both stay in [0,1]). So I don't need the projection!

This makes the pairwise exchange much cleaner:
- Pick (i, j)
- Find optimal δ ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]
- The objective max_k c_k(δ) is piecewise linear in δ (since each c_k is linear in δ)
- The minimum of a piecewise linear function is at a breakpoint or endpoint

But computing all breakpoints is expensive. A line search (golden section) would work for the smoothed version.

Actually, for the exact max (not smoothed), the minimum of a piecewise-linear convex function... wait, it's not necessarily convex. The max of linear functions IS convex. So max_k c_k(δ) is convex in δ! (Each c_k is linear in δ, and max of linear functions is convex.)

Wait, is each c_k linear in δ? Let's check:
c_k = Σ_m h[m] * (1-h[m+k]) * dx

If only h[i] and h[j] change (by +δ and -δ respectively):
c_k(δ) = c_k(0) + δ * [∂c_k/∂h[i] - ∂c_k/∂h[j]]

Yes! Each c_k is linear in δ. So max_k c_k(δ) is convex in δ. The minimum can be found by:
- Binary search on the subgradient
- Or ternary search (since it's convex and piecewise linear)

This is a very clean 1D optimization! Let me implement this properly.

The subgradient of max_k c_k(δ) at a point δ is:
{c_k'(δ) for all k that achieve the maximum at δ}

where c_k'(δ) = ∂c_k/∂h[i] - ∂c_k/∂h[j] (the slope of c_k with respect to δ).

We want to find δ* where 0 is in the subgradient, i.e., there exist maximizing k's with slopes on both sides of 0.

Algorithm:
1. Compute all slopes s_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]
2. At δ=0, find which k's achieve the max
3. If all max-achieving slopes are positive, decrease δ
4. If all max-achieving slopes are negative, increase δ
5. Otherwise, δ* = 0 is optimal (or we need to find the exact breakpoint)

Actually, simpler approach: since it's 1D convex, just do a ternary search or binary search on the derivative.

Let me implement this:

```python
def pairwise_opt(h, i, j, n, dx, max_iters=50):
    """Optimal 1D search for transferring mass between i and j."""
    target = n / 2.0  # Not needed since sum is preserved
    center = n - 1
    
    # Bounds on delta
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h.copy()
    
    # Compute corr at delta=0
    corr0 = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # For each lag, compute the slope: how c_k changes per unit delta
    # c_k = sum_m h[m] * (1-h[m+k]) * dx
    # If h[i] -> h[i]+d and h[j] -> h[j]-d:
    # Contribution of h[i] to c_k: 
    #   if i < n-|lag|: appears as h[i]*(1-h[i+lag]) -> derivative wrt h[i] is (1-h[i+lag])*dx
    #   if i >= |lag|: appears as -h[i-lag]*h[i] -> derivative wrt h[i] is -h[i-lag]*dx
    # Total slope for c_k: (∂c_k/∂h[i]) - (∂c_k/∂h[j])
    
    # This is expensive to compute for all k. Let me just do a line search.
    # Since the function is convex in delta, ternary search works.
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    m1 = a + (1 - gr) * (b - a)
    m2 = a + gr * (b - a)
    
    def eval_delta(d):
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        # No need to clip since d is in [lo, hi]
        corr = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
        return np.max(corr)
    
    f1 = eval_delta(m1)
    f2 = eval_delta(m2)
    
    for _ in range(max_iters):
        if b - a < 1e-9:
            break
        if f1 < f2:
            b = m2
            m2 = m1
            f2 = f1
            m1 = a + (1 - gr) * (b - a)
            f1 = eval_delta(m1)
        else:
            a = m1
            m1 = m2
            f1 = f2
            m2 = a + gr * (b - a)
            f2 = eval_delta(m2)
    
    opt_d = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] += opt_d
    h_new[j] -= opt_d
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize, linprog, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:]) * dx
        grad[lag:] += -h[:n - lag] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad


def smooth_max_val(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta


def resample_h(h_old, n_new):
    if len(h_old) == n_new:
        return h_old.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h_old)), h_old)


def benders_lp(h_cur, n, target, n_active=25):
    dx = 2.0 / n
    corr = all_corr(h_cur)
    active = np.argsort(corr)[::-1][:n_active]
    center = n - 1
    A_rows = []
    b_rows = []
    for idx in active:
        lag = idx - center
        g = grad_lag(h_cur, lag, n, dx)
        f_val = corr[idx]
        A_rows.append(np.concatenate([g, [-1.0]]))
        b_rows.append(g.dot(h_cur) - f_val)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds_lp = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
        if res.success:
            return project_h(res.x[:n], target)
    except Exception:
        pass
    return h_cur.copy()


def smooth_obj_fn(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(c, beta) * dx


def sum_con_fn(x, target):
    return np.sum(x) - target


def sum_jac_fn(x):
    return np.ones(x.shape)


def sg_descent(h, n, target, max_iters, time_left_fn, min_time, best_state):

    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1

        if time_left_fn() < min_time:
            break

        top_k = min(5, len(corr))

        vals = corr[top_idx]
        temp = 80.0 + 8.0 * it
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):

            grad += w * grad_lag(cur, lag, n, dx)

def pairwise_exchange(h, i, j, n, dx):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    m1 = a + (1.0 - gr) * (b - a)
    m2 = a + gr * (b - a)

    def eval_d(d):
        hn = h.copy()
        hn[i] += d
        hn[j] -= d
        c = np.dot(hn, 1.0 - hn)
        corr = np.correlate(hn, 1.0 - hn, mode="full")
        return float(np.max(corr) * dx)

    f1 = eval_d(m1)
    f2 = eval_d(m2)
    for _ in range(35):
        if b - a < 1e-10:
            break
        if f1 < f2:
            b = m2
            m2 = m1
            f2 = f1
            m1 = a + (1.0 - gr) * (b - a)
            f1 = eval_d(m1)
        else:
            a = m1
            m1 = m2
            f1 = f2
            m2 = a + gr * (b - a)
            f2 = eval_d(m2)
    opt_d = (a + b) / 2.0
    hn = h.copy()
    hn[i] += opt_d
    hn[j] -= opt_d
    return hn

            if time_left() < 3:
                break
            i, j = np.random.randint(0, n_f, 2)
            if i == j:
                continue
            h_new = pairwise_exchange(best_state[1], i, j, n_f, dx_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_state[0] - 1e-13:
                best_state[0] = c5_new
                best_state[1] = h_new

    return best_state[1], best_state[0], len(best_state[1])
```
