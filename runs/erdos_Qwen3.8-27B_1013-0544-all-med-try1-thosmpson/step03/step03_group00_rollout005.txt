def lbfgs_solve(h, beta, maxiter, top_k=None, ftol=1e-15):

    target = n / 2.0
    
    def obj(x):
        return smoothmax_obj_grad(x, beta, top_k)[0]
    
    def grad(x):
        return smoothmax_obj_grad(x, beta, top_k)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

    n = len(h)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c = corr[top_idx]
    m = c.max()
    shifted = beta * (c - m)
    exps = np.exp(shifted - shifted.max())
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        wk = w[idx]
        if wk < 1e-15:
            continue
        k = top_idx[idx]
        s = k - n + 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_hi > i_lo:
            j1_lo = i_lo + s
            j1_hi = i_hi + s
            grad[j1_lo:j1_hi] += wk * (1.0 - h[i_lo:i_hi])
            grad[i_lo:i_hi] += wk * (-h[i_lo + s:i_hi + s])
    grad *= dx
    return val, grad


def lbfgs_solve(h, beta, maxiter, top_k):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def grad_fn(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    try:
        res = minimize(obj, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter,
                                              'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def subgrad_refine(h, n_steps, step_size, top_k, seed):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(-corr)[:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - n + 1
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            if i_hi > i_lo:
                j1_lo = i_lo + s
                j1_hi = i_hi + s
                grad[j1_lo:j1_hi] += (1.0 - h[i_lo:i_hi])
                grad[i_lo:i_hi] += (-h[i_lo + s:i_hi + s])
        h -= step_size * grad
        h = project_feasible(h, target)
    return h


def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.randint(0, 2, size=n).astype(float)
    return project_feasible(h, n / 2.0)


def block_init(n, n_blocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            h[b * block_size:(b + 1) * block_size] = 1.0
    if n % n_blocks != 0:
        h[-(n % n_blocks):] = 1.0
    return project_feasible(h, n / 2.0)


def sinusoid_init(n, freq, amp, seed):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x)
    return project_feasible(h, n / 2.0)


def coord_descent(h, n_iters, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    deltas = [-0.1, -0.05, 0.05, 0.1, -0.2, 0.2]
    for _ in range(n_iters):
        j = rng.randint(n)
        for delta in deltas:
            h_trial = h.copy()
            h_trial[j] = np.clip(h[j] + delta, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            c = compute_c5(h_trial)
            if c < best_c5 - 1e-10:
                h = h_trial
                best_c5 = c
                break
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_c5 = compute_c5(h_cur)

    n = len(h_cur)
    for beta in [10, 30, 80, 200, 500, 1000]:
