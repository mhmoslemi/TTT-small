def project_feasible(h, target_sum):

        else:

            h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        (30, 12, 300),
        (60, 25, 300),
        (120, 50, 300),
        (250, 100, 250),
        (400, 200, 200),
        (600, None, 150),
    ]
    for beta, tk, mi in schedule:
        if rem() < 50:
            break
        h_cur = optimize_lfbgbsb(h_cur, beta, tk, mi)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            h_best = h_cur.copy()

    # Phase 2: Basin hopping with structured + random noise
    rng = np.random.RandomState(seed + 7)
    for hop in range(20):
        if rem() < 70:
            break
        amp = 0.04 * (0.82 ** hop)
        x = np.linspace(0, 2, n)
        f1 = rng.randint(1, 6)
        p1 = rng.uniform(0, 2 * np.pi)
        f2 = rng.randint(1, 10)
        p2 = rng.uniform(0, 2 * np.pi)
        noise = amp * (0.7 * np.sin(2 * np.pi * f1 * x + p1) + 0.3 * np.cos(2 * np.pi * f2 * x + p2))
        noise += 0.4 * amp * rng.randn(n)
        h_pert = project_feasible(h_best + noise, n / 2.0)
        h_local = optimize_lfbgbsb(h_pert, 300, 50, 200)
        h_local = optimize_lfbgbsb(h_local, 600, 100, 150)
        c = compute_c5(h_local)
        if c < best_c5:
            best_c5 = c
            h_best = h_local.copy()

    # Phase 3: Multi-resolution attempts
    for n_try in [150, 250, 350, 500]:
        if rem() < 140:
            break
        x_old = (np.arange(n) + 0.5) * (2.0 / n)
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        f = interp1d(x_old, h_best, kind='cubic', fill_value='extrapolate')
        h_try = np.clip(f(x_new), 0.0, 1.0)
        h_try = project_feasible(h_try, n_try / 2.0)
        for beta, tk, mi in [(40, 15, 200), (150, 50, 250), (400, 100, 200), (700, 200, 150)]:
            if rem() < 50:
                break
            h_try = optimize_lfbgbsb(h_try, beta, tk, mi)
        c = compute_c5(h_try)
        if c < best_c5:
            best_c5 = c
            h_best = h_try.copy()
            n = n_try

    # Phase 4: Fine basin hopping at best resolution
    rng2 = np.random.RandomState(seed + 777)
    for hop in range(12):
        if rem() < 45:
            break
        amp = 0.015 * (0.75 ** hop)
        x = np.linspace(0, 2, n)
        f = rng2.randint(2, 12)
        p = rng2.uniform(0, 2 * np.pi)
        noise = amp * np.sin(2 * np.pi * f * x + p) + 0.5 * amp * rng2.randn(n)
        h_pert = project_feasible(h_best + noise, n / 2.0)
        h_local = optimize_lfbgbsb(h_pert, 500, 80, 200)
        h_local = optimize_lfbgbsb(h_local, 1000, None, 150)
        c = compute_c5(h_local)
        if c < best_c5:
            best_c5 = c
            h_best = h_local.copy()

    # Phase 5: Final high-beta polish
    if rem() > 30:
        h_fin = optimize_lfbgbsb(h_best, 1200, None, 300)
        c = compute_c5(h_fin)
        if c < best_c5:
            best_c5 = c
            h_best = h_fin

    # Phase 6: Micro hops for final squeeze
    rng3 = np.random.RandomState(seed + 31337)
    for hop in range(6):
        if rem() < 30:
            break
        amp = 0.005 * (0.6 ** hop)
        x = np.linspace(0, 2, n)
        f = rng3.randint(1, 20)
        p = rng3.uniform(0, 2 * np.pi)
        noise = amp * np.sin(2 * np.pi * f * x / 2.0 + p)
        h_pert = project_feasible(h_best + noise, n / 2.0)
        h_local = optimize_lfbgbsb(h_pert, 1500, None, 200)
        c = compute_c5(h_local)
        if c < best_c5:
            best_c5 = c
            h_best = h_local.copy()

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
