import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def lbfgs_opt(h, beta, maxiter, top_k=None, lam=50.0):
    n = len(h)
    target = n / 2.0
    ones = np.ones(n)

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return v + lam * (x.sum() - target) ** 2

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        return g + 2.0 * lam * (x.sum() - target) * ones

    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def thue_morse_init(n, seed):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = t.astype(float)
    return project_feasible(h, n / 2.0)


def golden_partition(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    pos = rng.rand()
    for i in range(n):
        pos = (pos + 1.0 / phi) % 1.0
        h[i] = 1.0 if pos < 0.5 else 0.0
    return project_feasible(h, n / 2.0)


def block_pattern(n, n_blocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    bs = n // n_blocks
    for b in range(n_blocks):
        s = b * bs
        e = min(s + bs, n)
        if rng.rand() < 0.5:
            h[s:e] = 1.0
    return project_feasible(h, n / 2.0)


def sine_pattern(n, freq, seed):
    x = np.linspace(0, 2, n, endpoint=False)
    rng = np.random.RandomState(seed)
    ph = rng.rand() * 2 * np.pi
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + ph)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)


def sawtooth_pattern(n, period, seed):
    x = np.arange(n) / n * 2.0
    rng = np.random.RandomState(seed)
    ph = rng.rand() * period
    t = ((x + ph) % period) / period
    h = np.where(t < 0.5, 1.0, 0.0)
    return project_feasible(h, n / 2.0)


def multi_beta_opt(h, betas, maxiter, top_k=None, lam=50.0):
    for beta in betas:
        h = lbfgs_opt(h, beta, maxiter, top_k, lam)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n / 2.0)
    best_c5 = compute_c5(h_best)

    # Phase 1: Try diverse structural initial conditions at coarse resolution
    coarse_inits = []
    for nc in [50, 80]:
        coarse_inits.append((thue_morse_init(nc, seed), nc))
        coarse_inits.append((golden_partition(nc, seed + 1), nc))
        coarse_inits.append((block_pattern(nc, 4, seed + 2), nc))
        coarse_inits.append((sine_pattern(nc, 3, seed + 3), nc))
        coarse_inits.append((sawtooth_pattern(nc, 4.0 / 5.0, seed + 4), nc))

    for h_c, nc in coarse_inits:
        if remaining() < 60:
            break
        h_opt = multi_beta_opt(h_c, [5, 20, 60, 150, 400], 150, top_k=8, lam=30)
        h_interp = interp_to_n(h_opt, n)
        c = compute_c5(h_interp)
        if c < best_c5:
            best_c5 = c
            h_best = h_interp.copy()

    # Phase 2: L-BFGS continuation on current best
    for beta in [30, 80, 200, 400, 800]:
        if remaining() < 50:
            break
        lam = 30.0 if beta < 200 else 80.0
        h_new = lbfgs_opt(h_best, beta, 300, top_k=None, lam=lam)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()

    # Phase 3: Basin hopping with diverse perturbations
    rng = np.random.RandomState(seed + 500)
    for hop in range(20):
        if remaining() < 50:
            break
        noise = 0.03 * (0.82 ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, n / 2.0)
        h_loc = multi_beta_opt(h_pert, [80, 250, 600], 200, top_k=12, lam=80)
        c_loc = compute_c5(h_loc)
        if c_loc < best_c5:
            best_c5 = c_loc
            h_best = h_loc.copy()

    # Phase 4: Try finer resolution
    for n_fine in [250, 320, 400]:
        if remaining() < 100:
            break
        h_fine = interp_to_n(h_best, n_fine)
        for beta in [30, 100, 300, 700]:
            if remaining() < 50:
                break
            h_fine = lbfgs_opt(h_fine, beta, 250, top_k=15, lam=80)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c5:
            best_c5 = c_fine
            h_best = h_fine.copy()
            n = n_fine

    # Phase 5: Fine basin hopping on best resolution
    rng2 = np.random.RandomState(seed + 900)
    for hop in range(12):
        if remaining() < 45:
            break
        noise = 0.01 * (0.85 ** hop)
        h_pert = h_best + rng2.randn(n) * noise
        h_pert = project_feasible(h_pert, n / 2.0)
        h_loc = multi_beta_opt(h_pert, [200, 500, 1000], 200, top_k=10, lam=150)
        c_loc = compute_c5(h_loc)
        if c_loc < best_c5:
            best_c5 = c_loc
            h_best = h_loc.copy()

    # Phase 6: Final high-beta polish with full gradient
    if remaining() > 35:
        h_fin = lbfgs_opt(h_best, 1500, 600, top_k=None, lam=500)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 7: One more basin hop with very small noise
    if remaining() > 40:
        rng3 = np.random.RandomState(seed + 2000)
        for _ in range(5):
            if remaining() < 30:
                break
            h_pert = h_best + rng3.randn(n) * 0.004
            h_pert = project_feasible(h_pert, n / 2.0)
            h_loc = lbfgs_opt(h_pert, 800, 300, top_k=None, lam=500)
            c_loc = compute_c5(h_loc)
            if c_loc < best_c5:
                best_c5 = c_loc
                h_best = h_loc

    return h_best, best_c5, len(h_best)
