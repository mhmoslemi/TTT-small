def project_box_sum(h, target):
    """Project onto {x: 0<=x<=1, sum(x)=target} via bisection on watermark."""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

            result = np.clip(result, 0, 1)

        corr = np.correlate(x, 1.0 - x, mode="full")

        m = np.max(active_corr)

def lbfgs_solve(h, beta, top_k, maxiter, pen_lambda):

        v += pen_lambda * s * s
        g = g + 2 * pen_lambda * s

                                'maxls': 50})

    except:

def project_box_sum(h, target):
    """Project onto {x: 0<=x<=1, sum(x)=target} using water-filling."""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - r.sum()
    if abs(diff) > 1e-12:
        free = (r > 1e-10) & (r < 1 - 1e-10)
        if free.sum() > 0:
            r[free] += diff / free.sum()
            r = np.clip(r, 0, 1)
    return r

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

    """Compute smooth max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
