def compute_slopes(h, n, i):
    """Compute slope_r for all r when changing h[i]."""
    # correlate[r] = sum_j h[j]*(1-h[j+r-(n-1)])
    # Change in correlate[r] when h[i] changes by 1:
    # slope_r = (1-h[i+shift])*I(0<=i+shift<n) - h[i-shift]*I(0<=i-shift<n)
    # where shift = r - (n-1)
    # So for shift = s: r = s + n - 1
    # slope = (1-h[i+s])*I(0<=i+s<n) - h[i-s]*I(0<=i-s<n)
    
    shifts = np.arange(-(n-1), n)
    # For each shift s:
    # term1: (1-h[i+s]) if 0 <= i+s < n else 0
    # term2: -h[i-s] if 0 <= i-s < n else 0
    idx1 = i + shifts
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[idx1[valid1]], 0.0)
    
    idx2 = i - shifts
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, -h[idx2[valid2]], 0.0)
    
    return term1 + term2

    base = current_corr - h[i] * slopes

    # Minimize max_r (base[r] + new_h_i * slopes[r]) over new_h_i in [0,1]

        f2 = np.max(base + m2 * slopes)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

Let me also think about the "pair optimization" approach. Instead of optimizing one coordinate at a time, optimize a pair (i, j) with the constraint h_i + h_j = const (to maintain the sum). This is a 2D problem but with one constraint, so it's effectively 1D.

For a pair (i, j) with h_i + h_j = s (fixed), we parameterize h_i = t, h_j = s - t, where t ∈ [max(0, s-1), min(1, s)].

The change in correlate[r] when we change (h_i, h_j) from (old_i, old_j) to (t, s-t):
delta_r = (t - old_i) * slope_r(i) + (s-t - old_j) * slope_r(j)
= (t - old_i) * slope_r(i) + (old_i - t) * slope_r(j)  [since s = old_i + old_j, so s-t-old_j = old_i-t]
= (t - old_i) * (slope_r(i) - slope_r(j))

So the change is still linear in t! Great.

new_corr[r] = current_corr[r] + (t - old_i) * (slopes_i[r] - slopes_j[r])

And we minimize max_r new_corr[r] over t. Again a 1D convex piecewise-linear problem, solvable by ternary search.

This is the same cost as single-coordinate optimization but maintains the sum constraint exactly (no projection needed).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room < 1e-15:
                break
            h += (diff / total_room) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def compute_slopes(h, n, i):
    shifts = np.arange(-(n-1), n)
    idx1 = i + shifts
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.zeros(2*n - 1)
    term1[valid1] = 1.0 - h[idx1[valid1]]
    idx2 = i - shifts
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.zeros(2*n - 1)
    term2[valid2] = -h[idx2[valid2]]
    return term1 + term2

def ternary_max(obj_func, lo, hi, niter=60):
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_func(m1)
        f2 = obj_func(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_step(h, n, i, corr):
    slopes = compute_slopes(h, n, i)
    base = corr - h[i] * slopes
    lo, hi = 0.0, 1.0
    def f(t):
        return np.max(base + t * slopes)
    opt = ternary_max(f, lo, hi, 50)
    return opt

def pair_step(h, n, i, j, corr):
    slopes_i = compute_slopes(h, n, i)
    slopes_j = compute_slopes(h, n, j)
    diff_slope = slopes_i - slopes_j
    old_i = h[i]
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        return old_i, old_i
    base_corr = corr - old_i * slopes_i - (s - old_i) * slopes_j
    # new_corr = base_corr + t * slopes_i + (s-t) * slopes_j = base_corr + s*slopes_j + t*(slopes_i - slopes_j)
    const = base_corr + s * slopes_j
    def f(t):
        return np.max(const + t * diff_slope)
    opt = ternary_max(f, lo, hi, 50)
    return opt, s - opt

def coord_descent(h, n, n_sweeps, deadline, rng):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = np.max(corr) * (2.0/n)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        indices = list(range(n))
        rng.shuffle(indices)
        for i in indices:
            opt = coord_step(h, n, i, corr)
            if abs(opt - h[i]) > 1e-12:
                slopes = compute_slopes(h, n, i)
                corr += (opt - h[i]) * slopes
                h[i] = opt
        h = project_feasible(h, n)
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        c = np.max(new_corr) * (2.0/n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            corr = new_corr
        else:
            corr = new_corr
    return best_h, best_c

def pair_descent(h, n, n_sweeps, deadline, rng, max_pairs_per_sweep=None):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = np.max(corr) * (2.0/n)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        if max_pairs_per_sweep is None:
            max_pairs = n * (n-1) // 2
        else:
            max_pairs = min(max_pairs_per_sweep, n*(n-1)//2)
        
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(pairs)
        pairs = pairs[:max_pairs]
        
        for i, j in pairs:
            if time.time() > deadline:
                break
            new_i, new_j = pair_step(h, n, i, j, corr)
            if abs(new_i - h[i]) > 1e-12 or abs(new_j - h[j]) > 1e-12:
                slopes_i = compute_slopes(h, n, i)
                slopes_j = compute_slopes(h, n, j)
                corr += (new_i - h[i]) * slopes_i + (new_j - h[j]) * slopes_j
                h[i] = new_i
                h[j] = new_j
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = np.max(corr) * (2.0/n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

    """Minimize max(base + t*slope) over t in [lo, hi]."""

        m2 = hi - (hi - lo) / 3.0
        if np.max(base + m1 * slope) < np.max(base + m2 * slope):

        else:
            lo = m1

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h = h + (diff / tr) * room
        else:
            tr = h.sum()
            if tr < 1e-15:
                break
            h = h * (1.0 + diff / tr)
    return np.clip(h, 0.0, 1.0)

def get_slopes(h, n, i):
    m = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    s = np.zeros(m)
    idx1 = i + shifts
    v1 = (idx1 >= 0) & (idx1 < n)
    kv1 = np.where(v1)[0]
    s[kv1] += 1.0 - h[idx1[kv1]]
    idx2 = i - shifts
    v2 = (idx2 >= 0) & (idx2 < n)
    kv2 = np.where(v2)[0]
    s[kv2] -= h[idx2[kv2]]
    return s

def all_slopes_arr(h, n):
    return np.array([get_slopes(h, n, i) for i in range(n)])

def ternary_min(base, slope, lo, hi, niter=55):
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(base + m1 * slope) < np.max(base + m2 * slope):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, n_sweeps, deadline, rng):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    stale = 0
    for sw in range(n_sweeps):
        if time.time() > deadline:
            break
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for i in idxs:
            sl = get_slopes(h, n, i)
            base = corr - h[i] * sl
            opt = ternary_min(base, sl, 0.0, 1.0, 50)
            if abs(opt - h[i]) > 1e-11:
                corr += (opt - h[i]) * sl
                h[i] = opt
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = float(np.max(corr) * (2.0 / n))
        if c < best_c - 1e-14:
            best_c, best_h = c, h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 80:
                break
    return best_h, best_c

def pair_descent(h, n, n_iters, deadline, rng):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sLi = get_slopes(h, n, i)
        sLj = get_slopes(h, n, j)
        ds = sLi - sLj
        sv = h[i] + h[j]
        lo = max(0.0, sv - 1.0)
        hi = min(1.0, sv)
        if hi - lo < 1e-14:
            continue
        base = corr + h[i] * ds
        opt = ternary_min(base, ds, lo, hi, 45)
        nj = sv - opt
        if abs(opt - h[i]) > 1e-12 or abs(nj - h[j]) > 1e-12:
            corr += (opt - h[i]) * sLi + (nj - h[j]) * sLj
            h[i] = opt
            h[j] = nj
        if it % 1000 == 0:
            h = project_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c = float(np.max(corr) * (2.0 / n))
            if c < best_c - 1e-14:
                best_c, best_h = c, h.copy()
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = (c - m) * alpha
    if sh.max() > 500:
        sh = sh - sh.max()
    return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = (c - m) * alpha
    if sh.max() > 500:
        sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    sl = all_slopes_arr(x, n)
    return (dx / alpha) * (sl.T @ w)

def sum_fun(x, n):
    return float(x.sum() - n / 2.0)

def sum_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h
    try:
        res = minimize(lse_obj, h, args=(n, alpha), method='SLSQP',
                       jac=lambda x, n, a: lse_grad(x, n, a),
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def make_init(n, rng, h_parent):
    x = np.linspace(0, 2, n)
    starts = []
    if h_parent is not None and len(h_parent) == n:
        starts.append(h_parent.copy())
    starts.append(np.full(n, 0.5))
    for k in [1, 2, 3, 5, 7, 11]:
        starts.append(project_feasible(0.5 + 0.35 * np.sin(k * np.pi * x) + 0.1 * np.cos(2 * k * np.pi * x), n))
    for period in [3, 4, 5, 6, 7, 8, 9, 10]:
        pat = np.array([1.0 if (i % period) < period // 2 else 0.0 for i in range(n)])
        starts.append(project_feasible(pat, n))
    for _ in range(8):
        starts.append(project_feasible(rng.uniform(0, 1, n), n))
    for amp in [0.15, 0.25, 0.4]:
        starts.append(project_feasible(0.5 + amp * np.sin(np.pi * x) + 0.08 * np.sin(5 * np.pi * x), n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    h_parent = None
    n = 100
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n = len(h_parent)
    except Exception:
        pass

    best_c = float('inf')
    best_h = None

    starts = make_init(n, rng, h_parent)
    phase_deadline = t0 + budget_s * 0.55

    for si, s in enumerate(starts):
        if time.time() > phase_deadline:
            break
        alloc = (phase_deadline - time.time()) / max(1, len(starts) - si)
        sub_dl = time.time() + min(alloc + 10, phase_deadline - time.time())

        h = project_feasible(s.copy(), n)
        c0 = compute_c5(h, n)
        if c0 < best_c:
            best_c, best_h = c0, h.copy()

        # Coord descent
        h1, c1 = coord_descent(h, n, 3000, sub_dl, rng)
        if c1 < best_c:
            best_c, best_h = c1, h1.copy()
        h = best_h.copy()

        # SLSQP with a few alphas
        for alpha in [500, 5000, 50000]:
            if time.time() > sub_dl:
                break
            h2 = slsqp_step(h, n, alpha, 2000, sub_dl)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()
            h = best_h.copy()

        # Pair descent
        h3, c3 = pair_descent(h, n, 20000, sub_dl, rng)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 2: Deep refinement of best solution
    h = best_h.copy()
    # High-alpha SLSQP
    for alpha in [100000, 500000, 2000000]:
        if time.time() > deadline - 120:
            break
        h2 = slsqp_step(h, n, alpha, 4000, min(time.time() + 80, deadline - 120))
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()
        h = best_h.copy()

    # Long coordinate descent
    h, c = coord_descent(h, n, 50000, deadline - 120, rng)
    if c < best_c:
        best_c, best_h = c, h.copy()
    h = best_h.copy()

    # Long pair descent
    h, c = pair_descent(h, n, 500000, deadline - 30, rng)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Final coord descent
    h, c = coord_descent(best_h, n, 5000, deadline, rng)
    if c < best_c:
        best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
