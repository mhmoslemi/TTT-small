    h_best, best_c5 = basin_hop(h_best, best_c5,
                                 n_hops=12, beta_list=[100, 300],
                                 maxiter=250, noise_base=0.03,
                                 noise_decay=0.72, seed=seed + 1,
                                 t0=t0, budget_s=budget_s, T=0.004)

    # Phase 4: Block perturbation hopping
    if remaining() > 200:
        h_best, best_c5 = basin_hop(h_best, best_c5,
                                     n_hops=8, beta_list=[200, 500],
                                     maxiter=250, noise_base=0.01,
                                     noise_decay=0.8, seed=seed + 50,
                                     t0=t0, budget_s=budget_s,
                                     use_block=True, T=0.003)

    # Phase 5: Multi-resolution
    for n_try in [300, 400, 550]:
        if remaining() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_c = optimize_at_resolution(h_interp, [50, 200, 400], 350, t0, budget_s)
        c_try = compute_c5(h_c)
        if c_try < best_c5:
            best_c5 = c_try
            h_best = h_c.copy()

    # Phase 6: Fine basin hopping at best resolution
    h_best, best_c5 = basin_hop(h_best, best_c5,
                                 n_hops=8, beta_list=[300, 600],
                                 maxiter=300, noise_base=0.012,
                                 noise_decay=0.75, seed=seed + 200,
                                 t0=t0, budget_s=budget_s, T=0.002)

    # Phase 7: Block perturbation at fine resolution
    if remaining() > 100:
        h_best, best_c5 = basin_hop(h_best, best_c5,
                                     n_hops=6, beta_list=[400, 700],
                                     maxiter=250, noise_base=0.005,
                                     noise_decay=0.8, seed=seed + 300,
                                     t0=t0, budget_s=budget_s,
                                     use_block=True, T=0.0015)

    # Phase 8: High-beta SLSQP polish
    if remaining() > 40:
        h_fin = slsqp_solve(h_best, 800, 600, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 9: Final momentum squeeze
    if remaining() > 30:
        h_m2, c_m2 = momentum_opt(h_best, 1000, 400, 0.02, 0.92, 15, t0, budget_s)
        if c_m2 < best_c5:
            best_c5 = c_m2
            h_best = h_m2

    # Final SLSQP
    if remaining() > 20:
        h_fin2 = slsqp_solve(h_best, 1200, 400, top_k=None)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
