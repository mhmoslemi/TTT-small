After optimization, we have h* with C₅ = max_k g_k(h*). Let K* be the set of lags where g_k is close to the maximum (say within 1% of the max). We want to find a direction d such that g_k(h* + εd) decreases for all k in K*.

The gradient of g_k at h* is:
∇g_k[h*][i] = dx * [(1 - h*[i+k]) - h*[i-k]] (with boundary handling)

We want to find d that minimizes the maximum of {∇g_k[h*] · d : k ∈ K*}.

This is a linear programming problem! We can solve it with scipy.

Actually, it's simpler: we want d such that ∇g_k[h*] · d < 0 for all k ∈ K*. This is a feasibility problem. If it's infeasible, we're at a (KKT) local minimum.

If it's feasible, we take a small step in direction d.

This is essentially a cutting plane / active set method.

```python
def active_set_step(h, n, tol=0.01):
    """Find a direction that decreases all near-maximal lags"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    
    # Active set: lags where c is within tol of max
    active = np.where(c > c_max - tol * c_max)[0]
    
    # Compute gradients for active lags
    # grad_k[i] = dx * [(1-h[i+k]) - h[i-k]] for valid indices
    n_lags = len(active)
    n_pts = n
    G = np.zeros((n_lags, n_pts))
    
    for idx, m in enumerate(active):
        lag = m - (n - 1)
        for i in range(n_pts):
            g = 0.0
            if 0 <= i + lag < n_pts:
                g += (1 - h[i + lag])
            if 0 <= i - lag < n_pts:
                g -= h[i - lag]
            G[idx, i] = g * dx
    
    # We want to find d such that G @ d < 0 (all components negative)
    # and ||d|| ≤ 1 (for a bounded step)
    # Also need to respect box constraints: for i where h[i]=0, d[i] ≥ 0
    #                              for i where h[i]=1, d[i] ≤ 0
    # And sum(d) = 0 (to maintain the sum constraint)
    
    # This is a LP: minimize t subject to G@t_dir ≤ t, sum(t_dir)=0, box constraints
    # If t < 0, we have a descent direction
    
    from scipy.optimize import linprog
    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize: t
    # Subject to: G @ d + t >= 0  =>  -G @ d - t <= 0
    #             sum(d) = 0
    #             -1 <= d_i <= 1
    #             t is free (but bounded by -1 to 1 for numerical stability)
    
    n_var = n_pts + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: -G @ d - t <= 0  =>  G @ d + t >= 0  =>  -(G @ d + t) <= 0
    A_ub = np.hstack([-G, -np.ones((n_lags, 1))])
    b_ub = np.zeros(n_lags)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n_pts] = 1.0
    b_eq = [0.0]
    
    # Bounds: -1 <= d_i <= 1, -1 <= t <= 1
    bounds = [(-1, 1)] * n_pts + [(-1, 1)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x[-1] < -1e-6:
        # We found a descent direction
        d = res.x[:n_pts]
        # Take a step
        step = 0.1  # small step
        h_new = h + step * d
        h_new = project(h_new, n)
        return h_new
    else:
        return None  # No descent direction found (at local minimum)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        d = target - h.sum()
        if abs(d) < 1e-13 * n:
            break
        if d > 0:
            room = 1.0 - h
            if room.sum() < 1e-15: break
            h = h + d * room / room.sum()
        else:
            if h.sum() < 1e-15: break
            h = h + d * h / h.sum()
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, alpha):
    """Objective and gradient for LSE-smoothed C5"""
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -100, 0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    val = m + np.log(exp_s.sum()) / alpha
    
    # Gradient: sum_m w[m] * (dc[dm]/dx) * dx
    # dc[m]/dx[j] = (1-x[j+lag]) * I(0<=j+lag<n) - x[j-lag] * I(0<=j-lag<n)
    # where lag = m - (n-1)
    grad = np.zeros(n)
    for m in range(len(c_raw)):
        if w[m] < 1e-12:
            continue
        lag = m - (n - 1)
        # First term: (1-x[j+lag]) for j where 0<=j+lag<n
        # j ranges from max(0,-lag) to min(n, n-lag)-1
        j_start = max(0, -lag)
        j_end = min(n, n - lag)
        if j_end > j_start:
            grad[j_start:j_end] += w[m] * dx * (1 - x[j_start + lag:j_end + lag])
        # Second term: -x[j-lag] for j where 0<=j-lag<n
        # j ranges from max(0,lag) to min(n+lag, n)-1 ... wait
        # 0<=j-lag<n means lag<=j<n+lag, intersected with 0<=j<n
        j_start2 = max(0, lag)
        j_end2 = min(n, n + lag)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w[m] * dx * x[j_start2 - lag:j_end2 - lag]
    
    return val, grad
```

Now let me write the gradient computation efficiently:

```python
def obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -100, 0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    val = m + np.log(exp_s.sum()) / alpha
    
    grad = np.zeros(n)
    for m_idx in range(len(c_raw)):
        wm = w[m_idx]
        if wm < 1e-12:
            continue
        lag = m_idx - (n - 1)
        # Term 1: (1-x[j-lag]) for j where 0<=j-lag<n, i.e., lag<=j<n+lag
        # Intersect with [0,n): max(0,lag) <= j < min(n, n+lag)
        j1s = max(0, lag)
        j1e = min(n, n + lag)
        if j1e > j1s:
            grad[j1s:j1e] += wm * dx * (1 - x[j1s - lag:j1e - lag])
        # Term 2: -x[j+lag] for j where 0<=j+lag<n, i.e., -lag<=j<n-lag
        # Intersect with [0,n): max(0,-lag) <= j < min(n, n-lag)
        j2s = max(0, -lag)
        j2e = min(n, n - lag)
        if j2e > j2s:
            grad[j2s:j2e] -= wm * dx * x[j2s + lag:j2e + lag]
    
    return val, grad
```

This is O(n) per lag (vectorized), and O(n) lags, so O(n²) total. For n=100, that's 10000 operations per gradient evaluation. Very fast.

F(h) = LSE_objective(h) + μ * (sum(h) - n/2)²

where μ is a large number. As μ → ∞, the constraint is enforced exactly.

The gradient of the penalty term is: 2μ * (sum(h) - n/2) * ones(n)
