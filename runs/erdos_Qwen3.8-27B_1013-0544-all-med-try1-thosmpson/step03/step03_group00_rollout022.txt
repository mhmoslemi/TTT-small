    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)

        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def overlap_and_grad(h):
    """Compute overlaps at all lags and the gradient structure."""
    n = len(h)
    dx = 2.0 / n
    o = np.correlate(h, 1.0 - h, mode="full")
    return o * dx  # returns overlap values (already scaled by dx)

def smoothmax_obj_grad(h, beta, top_k=None):
    """Compute smoothmax of overlaps and its gradient."""
    n = len(h)
    dx = 2.0 / n
    o = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(o):
        top_idx = np.argsort(-o)[:top_k]
    else:
        top_idx = np.arange(len(o))
    o_sel = o[top_idx]
    m = o_sel.max()
    # Avoid overflow
    exps = np.exp(beta * (o_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    # Gradient: vectorized
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-15:
            continue
        ell = k - (n - 1)  # actual lag
        if ell >= 0:
            # grad[ell:] += wi * (1 - h[:n-ell])
            if ell < n:
                grad[ell:] += wi * (1.0 - h[:n-ell])
                grad[:n-ell] += -wi * h[ell:]
        else:
            m2 = -ell
            # grad[:n-m2] += wi * (1 - h[m2:])
            if m2 < n:
                grad[:n-m2] += wi * (1.0 - h[m2:])
                grad[m2:] += -wi * h[:n-m2]
    grad *= dx
    return val, grad

def slsqp(h, beta, maxiter, top_k=None):

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return smoothmax_obj_grad(x, beta, top_k)[0]

        return smoothmax_obj_grad(x, beta, top_k)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h, target)

def lp_step(h, max_lags=None):
    """One LP linearization step."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    o = np.correlate(h, 1.0 - h, mode="full")
    
    # Select top lags by overlap value
    if max_lags is not None and max_lags < len(o):
        top_idx = np.argsort(-o)[:max_lags]
    else:
        top_idx = np.arange(len(o))
    
    # Build LP: min t s.t. grad_k·h - t ≤ rhs_k, 0≤h≤1, sum(h)=target
    # Using linprog
    # Variables: [h_0,...,h_{n-1}, t]
    # Objective: minimize t → c = [0,...,0,1]
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []
    
    for k in top_idx:
        ell = k - (n - 1)
        o_k = o[k]
        # Compute grad_k (gradient of overlap at lag k)
        g = np.zeros(n)
        if ell >= 0:
            if ell < n:
                g[ell:] = (1.0 - h[:n-ell])
                g[:n-ell] += -h[ell:]
        else:
            m2 = -ell
            if m2 < n:
                g[:n-m2] = (1.0 - h[m2:])
                g[m2:] += -h[:n-m2]
        # Constraint: g·h - t ≤ g·h_old - o_k
        # (since overlap_k(h) ≈ o_k + g·(h-h_old) = o_k - g·h_old + g·h)
        # We want: o_k - g·h_old + g·h ≤ t
        # → g·h - t ≤ g·h_old - o_k
        rhs = g.dot(h) - o_k
        row = np.zeros(n_vars)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return project_feasible(h_new, target)
    else:
        return h.copy()

            h_pert = h_best.copy()
            start = rng.randint(0, n - n // 5)
            length = n // 5 + rng.randint(0, n // 5)
            h_pert[start:start + length] = 1.0 - h_pert[start:start + length]
        h_pert = project_feasible(h_pert, target)
        h_local = slsqp_solve(h_pert, beta, maxiter, top_k)
        h_local, _ = blend_lp(h_local, alpha=0.8, max_lags=80)
        h_local = slsqp_solve(h_local, beta, maxiter // 2, top_k)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5

def make_block_init(n, nblocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    boundaries = sorted(rng.choice(range(1, n), size=nblocks - 1, replace=False))
    segments = np.split(np.arange(n), boundaries)
    for seg in segments:
        if rng.rand() > 0.5:
            h[seg] = 1.0
    return project_feasible(h, n / 2.0)

def make_periodic_init(n, period, duty, seed):
    rng = np.random.RandomState(seed)
    phase = rng.rand() * period
    h = np.zeros(n)
    for i in range(n):
        pos = (i + phase) % period
        if pos < period * duty:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)
    n = len(h_cur)
    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: SLSQP continuation
    for beta in [15, 40, 100, 200, 400]:
        if remaining() < 50:
            break
        h_new = slsqp_solve(h_cur, beta, 350, top_k=30)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new

    # Phase 2: LP-based refinement with line search
    for _ in range(8):
        if remaining() < 60:
            break
        h_cur, c_cur = blend_lp(h_cur, alpha=0.6, max_lags=100)
        if c_cur < best_c5:
            best_c5 = c_cur
            h_best = h_cur.copy()
        h_cur = slsqp_solve(h_cur, 200, 200, top_k=20)
        c_cur = compute_c5(h_cur)
        if c_cur < best_c5:
            best_c5 = c_cur
            h_best = h_cur.copy()

    # Phase 3: Diversified basin hopping
    h_best, best_c5 = diversified_hops(
        h_best, best_c5, n_hops=12, beta=200, maxiter=300,
        noise=0.03, seed=seed + 1, t0=t0, budget_s=budget_s, top_k=20)

    # Phase 4: More aggressive hops
    h_best, best_c5 = diversified_hops(
        h_best, best_c5, n_hops=10, beta=300, maxiter=250,
        noise=0.05, seed=seed + 100, t0=t0, budget_s=budget_s, top_k=15)

    # Phase 5: Try diverse initializations
    inits = [
        make_periodic_init(n, 7, 0.5, seed),
        make_periodic_init(n, 11, 0.5, seed + 1),
        make_block_init(n, 6, seed + 2),
        make_block_init(n, 10, seed + 3),
    ]
    for h_init in inits:
        if remaining() < 80:
            break
        h_try = slsqp_solve(h_init, 100, 300, top_k=30)
        h_try, c_try = blend_lp(h_try, alpha=0.5, max_lags=80)
        h_try = slsqp_solve(h_try, 300, 200, top_k=20)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            h_best = h_try.copy()

    # Phase 6: Fine polish with high beta
    if remaining() > 50:
        h_fin = slsqp_solve(h_best, 500, 500, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 7: Final LP squeeze
    if remaining() > 35:
        h_fin, _ = blend_lp(h_best, alpha=0.4, max_lags=None)
        h_fin = slsqp_solve(h_fin, 400, 400, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 8: Final micro-hops
    if remaining() > 30:
        h_best, best_c5 = diversified_hops(
            h_best, best_c5, n_hops=5, beta=500, maxiter=200,
            noise=0.01, seed=seed + 500, t0=t0, budget_s=budget_s, top_k=10)

    return h_best, best_c5, len(h_best)
