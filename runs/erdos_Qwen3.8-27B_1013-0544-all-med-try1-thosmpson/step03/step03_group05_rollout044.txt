                    if c5_new < cur_c5:
                        cur = h_new
                        cur_c5 = c5_new
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
                except Exception:
                    pass

            # Subgradient with momentum
            cur, cur_c5, _ = subgradient_with_momentum(
                cur, n, target, dx, max_iters=2500,
                time_fn=time_left, min_time=5, best_state=best_state
            )

            # Pair swap local search
            cur, cur_c5, _ = pair_swap_search(
                cur, n, target, dx, max_iters=5000,
                time_fn=time_left, min_time=5, best_state=best_state
            )

            # Another round of subgradient
            if time_left() > 30:
                cur, cur_c5, _ = subgradient_with_momentum(
                    cur, n, target, dx, max_iters=1500,
                    time_fn=time_left, min_time=5, best_state=best_state
                )

    # Final polish on best solution
    if time_left() > 10:
        best_h = best_state[1].copy()
        best_n = best_state[2]
        target_f = best_n / 2.0
        dx_f = 2.0 / best_n
        best_c5 = compute_c5(best_h)

        # Fine subgradient
        for it in range(8000):
            if time_left() < 3:
                break
            corr_val = compute_all_overlaps(best_h)
            top_idx = np.argsort(corr_val)[::-1][:3]
            vals = corr_val[top_idx]
            spread = max(vals[0] - vals[-1], 1e-10)
            temp = max(100.0, 5.0 / spread)
            weights = np.exp((vals - vals[0]) * temp)
            weights /= weights.sum()
            grad = np.zeros(best_n)
            center = best_n - 1
            for t_idx, w in zip(top_idx, weights):
                lag = int(t_idx) - center
                grad += w * grad_for_lag(best_h, lag, best_n, dx_f)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad /= gnorm
            step = 0.004 / (1.0 + 0.0005 * it)
            improved = False
            for s_mult in [1.0, 0.5, 0.25, 0.125]:
                h_trial = project_h(best_h - step * s_mult * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
                step *= 0.5
            if not improved:
                T = 0.001 * np.exp(-it / 2000.0)
                noise = np.random.normal(0, T, best_n)
                h_trial = project_h(best_h + noise, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 + T * 2:
                    best_h = h_trial
                    best_c5 = c5_trial

        # Final pair swaps
        for it in range(10000):
            if time_left() < 2:
                break
            i, j = np.random.choice(best_n, 2, replace=False)
            a, b = best_h[i], best_h[j]
            if abs(a - b) < 1e-8:
                continue
            best_h[i], best_h[j] = b, a
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
            else:
                best_h[i], best_h[j] = a, b

    final_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
