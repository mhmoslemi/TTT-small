    m = np.max(corr)

    B = fftconvolve(h, w_full)[n-1:n-1+n]
    grad = dx * (A - B)
    return val, grad

    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_func(x):
        v, _ = smoothmax_val_grad_fft(x, beta)
        return v
    
    def obj_grad(x):
        _, g = smoothmax_val_grad_fft(x, beta)
        return g
    
    def con_func(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

    target = n / 2.0
    x = h.copy().astype(float)

        if v_true < best_v:

            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
