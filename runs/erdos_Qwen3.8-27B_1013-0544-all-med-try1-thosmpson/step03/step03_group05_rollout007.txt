```python
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:

        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def obj_smooth_max(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx


def con_sum(x, target):
    return np.sum(x) - target


def con_sum_jac(x, target):
    return np.ones(x.shape)


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)


def slsqp_refine(h, n, target, dx, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    obj = partial(obj_smooth_max, dx=dx, beta=beta)
    con = partial(con_sum, target=target)
    con_j = partial(con_sum_jac, target=target)
    cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(res.x, target)
    except Exception:
        return h


def momentum_descent(h0, n, target, dx, max_iters, start_time, budget_s, min_time, best_state):
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    v = np.zeros(n)
    beta_mom = 0.88
    base_step = 0.04
    stall_count = 0

    for it in range(max_iters):
        elapsed = time.time() - start_time
        if budget_s - elapsed < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(5, max(3, n // 50))
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 180.0 + 250.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        grad -= grad.mean()
        v = beta_mom * v - base_step * grad
        v -= v.mean()
        step = 1.0
        improved = False
        for _ in range(35):
            h_trial = project_h(cur + step * v, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-16:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                stall_count = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved:
            stall_count += 1
            T = 0.004 * np.exp(-stall_count / 100.0)
            if T > 1e-7:
                noise = np.random.normal(0, T, n)
                noise -= noise.mean()
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 4:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
                    stall_count = 0
            elif stall_count > 50:
                break
    return cur, cur_c5, n


def fine_polish(h, n, target, dx, start_time, budget_s, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    v = np.zeros(n)
    beta_mom = 0.92
    for it in range(30000):
        if budget_s - (time.time() - start_time) < 3:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = 3
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 350.0 + 150.0 * it / 30000.0
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        grad -= grad.mean()
        v = beta_mom * v - 0.0015 * grad
        v -= v.mean()
        step = 1.0
        for _ in range(25):
            h_trial = project_h(cur + step * v, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-17:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        else:
            if it < 200:
                T = 0.0015 * np.exp(-it / 80.0)
                noise = np.random.normal(0, T, n)
                noise -= noise.mean()
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T:
                    cur = h_trial
                    cur_c5 = c5_trial
            elif it > 500:
                break
    return cur, cur_c5, n


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    n_list = sorted(set([n_parent, 300, 350, 400]))
    n_list = [n for n in n_list if 100 <= n <= 500]

    for n in n_list:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 80:
            break
        target = n / 2.0
        dx = 2.0 / n
        x = np.linspace(0, 2, n)

        inits = [project_h(resample_h(best_state[1], n), target)]
        for mode in range(1, 5):
            h_f = 0.5 + 0.28 * np.cos(mode * np.pi * x) + 0.08 * np.sin(2 * mode * np.pi * x)
            inits.append(project_h(h_f, target))
        h_g = 0.5 + 0.35 * np.exp(-((x - 1.0) / 0.35) ** 2)
        inits.append(project_h(h_g, target))
        h_bb = 0.5 + 0.25 * (np.exp(-((x - 0.35) / 0.22) ** 2) + np.exp(-((x - 1.65) / 0.22) ** 2))
        inits.append(project_h(h_bb, target))
        h_anti = 0.5 + 0.3 * np.sin(np.pi * x)
        inits.append(project_h(h_anti, target))

        for h_init in inits:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 50:
                break
            cur = project_h(h_init.copy(), target)
            cur_c5 = compute_c5(cur)

            if cur_c5 >= best_state[0] + 0.01:
                continue

            for beta in [100, 300, 800]:
                remaining = budget_s - (time.time() - start_time)
                if remaining < 25:
                    break
                cur_new = slsqp_refine(cur, n, target, dx, beta, 400)
                c5_new = compute_c5(cur_new)
                if c5_new < cur_c5:
                    cur = cur_new
                    cur_c5 = c5_new

            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n

            cur, cur_c5, _ = momentum_descent(
                cur, n, target, dx,
                max_iters=4000, start_time=start_time,
                budget_s=budget_s, min_time=15, best_state=best_state
            )

    best_h = best_state[1]
    best_c5 = best_state[0]
    best_n = best_state[2]

    if budget_s - (time.time() - start_time) > 20:
        best_h, best_c5, best_n = fine_polish(
            best_h, best_n, best_n / 2.0, 2.0 / best_n,
            start_time=start_time, budget_s=budget_s, best_state=best_state
        )

    return best_state[1], best_state[0], len(best_state[1])
```
