    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad_fast(x, beta, top_k)[0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_smoothmax(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    shifted = beta * (c_sel - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    return corr, top_idx, w, val

def grad_smoothmax(h, top_idx, w):

    dx = 2.0 / n

        if i_hi2 > i_lo2:

    grad *= dx

    target = n / 2.0
    h = h_init.copy()

    stall_count = 0

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_parts(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    shifted = np.clip(beta * (c_sel - m), -500, 500)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    return top_idx, w, val

def grad_smoothmax(h, top_idx, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w[i]
        if wk < 1e-15:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return grad

def fista_optimize(h_init, beta_start, beta_end, n_iter, step_init, t0, budget_s, top_k=40):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    y = h.copy()
    t_param = 1.0
    step = step_init
    best_val = float('inf')
    best_h = h.copy()
    stall = 0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 5:
            break
        frac = it / max(n_iter - 1, 1)
        beta = max(1.0, beta_start * (beta_end / beta_start) ** frac)
        y = project_feasible(y, target)
        top_idx, w, val = smoothmax_parts(y, beta, top_k)
        if val < best_val:
            best_val = val
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        if stall > 600:
            rng = np.random.RandomState(1000 + it)
            y = project_feasible(best_h + rng.randn(n) * 0.015, target)
            h = y.copy()
            t_param = 1.0
            stall = 0
        grad = grad_smoothmax(y, top_idx, w)
        x_new = project_feasible(y - step * grad, target)
        val_new = smoothmax_parts(x_new, beta, top_k)[2]
        if val_new > val + 1e-14:
            step *= 0.6
        elif step < step_init * 0.9:
            step = min(step * 1.1, step_init)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x_new + (t_param - 1.0) / t_new * (x_new - h)
        h = x_new
        t_param = t_new
    return best_h, best_val

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_parts(x, beta, top_k)[2]
    def objg(x):
        ti, w, _ = smoothmax_parts(x, beta, top_k)
        return grad_smoothmax(x, ti, w)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.array(f(x_new))
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def make_starts(h_base, seed, n_starts=7):
    rng = np.random.RandomState(seed)
    n = len(h_base)
    target = n / 2.0
    starts = [h_base.copy()]
    starts.append(project_feasible(1.0 - h_base, target))
    starts.append(project_feasible(np.roll(h_base, n // 4), target))
    starts.append(project_feasible(np.roll(h_base, n // 3), target))
    starts.append(project_feasible(h_base + rng.randn(n) * 0.06, target))
    starts.append(project_feasible(h_base + rng.randn(n) * 0.12, target))
    x = np.linspace(0, 6 * np.pi, n)
    starts.append(project_feasible(h_base + 0.05 * np.sin(2 * x), target))
    return starts[:n_starts]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)
    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: FISTA warmup
    h1, _ = fista_optimize(h_best, 8, 200, 4000, 0.025, t0, budget_s, top_k=40)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5, h_best = c1, h1

    # Phase 2: SLSQP continuation
    for beta in [100, 300, 600]:
        if remaining() < 50:
            break
        h2 = slsqp_solve(h_best, beta, 500, top_k=30)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, h_best = c2, h2

    # Phase 3: Multi-start exploration
    starts = make_starts(h_best, seed, 7)
    for i, hs in enumerate(starts):
        if remaining() < 45:
            break
        ht, _ = fista_optimize(hs, 20, 400, 2500, 0.02, t0, budget_s, top_k=40)
        if remaining() > 25:
            ht = slsqp_solve(ht, 400, 300, top_k=20)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, h_best = ct, ht

    # Phase 4: Resolution upgrade
    for n_try in [300, 400]:
        if remaining() < 120:
            break
        h_try = interp_to_n(h_best, n_try)
        ht, _ = fista_optimize(h_try, 8, 250, 2500, 0.025, t0, budget_s, top_k=40)
        if remaining() > 40:
            ht = slsqp_solve(ht, 400, 350, top_k=30)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, h_best = ct, ht

    # Phase 5: Multi-start at best resolution
    n_best = len(h_best)
    starts2 = make_starts(h_best, seed + 50, 5)
    for i, hs in enumerate(starts2):
        if remaining() < 40:
            break
        ht, _ = fista_optimize(hs, 50, 600, 2000, 0.015, t0, budget_s, top_k=30)
        if remaining() > 20:
            ht = slsqp_solve(ht, 600, 300, top_k=20)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, h_best = ct, ht

    # Phase 6: Final high-beta SLSQP squeeze
    if remaining() > 30:
        hs = slsqp_solve(h_best, 800, 500, top_k=15)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, h_best = cs, hs

    # Phase 7: One more FISTA at very high beta
    if remaining() > 30:
        hf, _ = fista_optimize(h_best, 200, 1000, 2000, 0.008, t0, budget_s, top_k=15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, h_best = cf, hf

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
