def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = corr - m
    exps = np.exp(np.minimum(beta * shifted, 0.0))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n - t])
            grad[:n - t] += wk * (-h[t:])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def slsqp(h0, beta, maxiter=300):

        v, _ = smoothmax_obj_grad(x, beta)
        return v

        return g

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
