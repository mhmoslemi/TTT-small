def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha, mu):
    """Returns objective value and gradient for smooth log-sum-exp + penalty."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    s = alpha * (c - cmax)
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    W = exp_s.sum()
    w = exp_s / W  # softmax weights
    
    # Objective
    f = (cmax + (s_max + np.log(W)) / alpha) + mu * (x.sum() - n/2.0)**2
    
    # Gradient of log-sum-exp part: weighted average of dc[k]/dh[i]
    # dc[k]/dh[i] with m = k - (n-1):
    # = (1 - x[i+m]) * [i+m in range] - x[i-m] * [i-m in range]
    grad = np.zeros(n)
    for k in range(2*n - 1):
        m = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if m >= 0:
            if m == 0:
                grad += wk * (1.0 - 2.0 * x)
            else:
                grad[:n-m] += wk * (1.0 - x[m:])
                grad[m:] -= wk * x[:n-m]
        else:

                grad += wk * (1.0 - 2.0 * x)
            else:
                grad[am:] += wk * (1.0 - x[:n-am])
                grad[:n-am] -= wk * x[am:]
    
    # Add penalty gradient
    grad += 2.0 * mu * (x.sum() - n/2.0)
    
    return f, grad

        return smooth_obj_and_grad(x, n, alpha, mu)[0]
    def jac(x):
        return smooth_obj_and_grad(x, n, alpha, mu)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= s * d
                            h[j] += s * d
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    s = alpha * (c - cmax)
    sm = s.max()
    exp_s = np.exp(s - sm)
    W = exp_s.sum()
    w = exp_s / W
    f = (cmax + (sm + np.log(W)) / alpha) + mu * (x.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif m > 0:
            grad[:n - m] += wk * (1.0 - x[m:])
            grad[m:] -= wk * x[:n - m]
        else:
            a = -m
            grad[a:] += wk * (1.0 - x[:n - a])
            grad[:n - a] -= wk * x[a:]
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return f, grad

def lbfgs_opt(h, n, alpha, mu, maxiter=5000, ftol=1e-18, gtol=1e-14):
    def fun(x):
        return smooth_obj_grad(x, n, alpha, mu)[0]
    def jac(x):
        return smooth_obj_grad(x, n, alpha, mu)[1]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                                'iprint': -1, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i], h[j] = h[i] - s * d, h[j] + s * d
    return h, best_c

def coord_descent(h, n, deadline, n_sweep=200):
    """Exact coordinate descent: for each i, find best h[i] minimizing max c[k]."""
    best_c = compute_c5(h, n)
    grid = np.linspace(0, 1, n_sweep)
    improved = True
    while improved:
        improved = False
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            best_val = best_c
            best_hi = old
            for g in grid:
                h[i] = g
                rest_sum = h.sum() - g
                # adjust another coordinate to maintain sum
                # Actually just evaluate without constraint for speed, project at end
                c = compute_c5(h, n)
                if c < best_val:
                    best_val = c
                    best_hi = g
            if best_hi != old:
                h[i] = best_hi
                if best_val < best_c:
                    best_c = best_val
                    improved = True
                elif improved:
                    improved = True
            else:
                h[i] = old
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Progressive alpha L-BFGS-B
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        mu = max(100.0, alpha * 0.01)
        ht = lbfgs_opt(best_h, n, alpha, mu, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    # Phase 2: Multiple restarts
    n_restarts = 20
    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        scale = 0.03 + 0.12 * (r / n_restarts)
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 25:
                break
            ht = lbfgs_opt(hp, n, alpha, max(500.0, alpha * 0.01), maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = ht
                break

    # Phase 3: High-alpha refinement
    for alpha in [1000000, 5000000, 20000000]:
        if time.time() > deadline - 15:
            break
        ht = lbfgs_opt(best_h, n, alpha, 10000.0, maxiter=4000, gtol=1e-12)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht

    # Phase 4: Pair local search
    if time.time() < deadline - 15:
        best_h, best_c = pair_local(best_h, n, deadline,
                                    [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    # Phase 5: Final L-BFGS polish
    if time.time() < deadline - 10:
        ht = lbfgs_opt(best_h, n, 5000000, 10000.0, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 6: Final pair search
    if time.time() < deadline - 5:
        best_h, best_c = pair_local(best_h, n, deadline,
                                    [0.002, 0.001, 0.0005, 0.0002])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def make_starts(h_base, n, rng):

    x = np.linspace(0, 2, n)

        starts.append(project_feasible(0.5 * h_base + 0.5 * pat, n))
