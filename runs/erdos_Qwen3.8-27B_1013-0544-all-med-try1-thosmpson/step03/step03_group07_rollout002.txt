    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

        if abs(diff2) > 1e-12:

            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)

    for beta in [50, 100, 200, 500, 1000, 2000]:
        if tl() < 40:
            break
        h_new = lbgfb_optimize(h_best, beta, maxiter=500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Symmetrize and re-optimize
    if tl() > 60:
        h_sym = symmetrize(h_best)
        c_sym = compute_c5(h_sym)
        if c_sym < best_c5:
            h_best, best_c5 = h_sym, c_sym
            h_new = lbgfb_optimize(h_best, 1000, maxiter=500)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 3: Aggressive multi-start (diverse perturbations)
    for trial in range(40):
        if tl() < 25:
            break
        if trial % 3 == 0:
            scale = 0.01 + 0.005 * (trial % 8)
            h_pert = project_feasible(h_best + np.random.randn(n) * scale, n/2.0)
        elif trial % 3 == 1:
            m_blocks = int(rng.integers(3, 12))
            h_pert = random_block_start(n, m_blocks, rng)
            h_pert = 0.3 * h_best + 0.7 * h_pert
            h_pert = project_feasible(h_pert, n/2.0)
        else:
            h_pert = symmetrize(project_feasible(h_best + np.random.randn(n) * 0.02, n/2.0))
        h_new = lbgfb_optimize(h_pert, 1500, maxiter=500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: Higher resolution
    for n_try in [600, 800, 1200]:
        if tl() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        n = n_try
        for beta in [500, 1000, 2000, 4000]:
            if tl() < 50:
                break
            h_new = lbgfb_optimize(h_interp, beta, maxiter=400)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()

    # Phase 5: Symmetrize at higher resolution
    if tl() > 40:
        h_sym = symmetrize(h_best)
        c_sym = compute_c5(h_sym)
        if c_sym < best_c5:
            h_best, best_c5 = h_sym, c_sym
            h_new = lbgfb_optimize(h_best, 3000, maxiter=500)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 6: Final polish with very high beta
    for beta in [3000, 5000, 10000]:
        if tl() < 20:
            break
        h_new = lbgfb_optimize(h_best, beta, maxiter=600)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 7: More multi-start at final resolution
    for trial in range(20):
        if tl() < 15:
            break
        scale = 0.003 + 0.002 * (trial % 5)
        h_pert = project_feasible(h_best + np.random.randn(len(h_best)) * scale, len(h_best)/2.0)
        h_new = lbgfb_optimize(h_pert, 3000, maxiter=500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    h_best = project_feasible(h_best, len(h_best)/2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
