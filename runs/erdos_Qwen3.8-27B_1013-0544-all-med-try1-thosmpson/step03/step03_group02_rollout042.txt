Let me also add a "restart" mechanism: if the coordinate descent stalls, try a small random perturbation and continue.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(xs)) / len(x)) / beta)

def sml_obj(h, params):
    beta, dx = params
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def sml_grad(h, params):
    beta, dx = params
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(xs)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        if lag == 0:
            js = np.arange(i_min, i_max + 1)
            grad[js] += wk * (1.0 - 2.0 * h[js])
        else:
            js1 = np.arange(i_min, i_max + 1)
            grad[js1] += wk * (1.0 - h[js1 + lag])
            js2 = np.arange(i_min + lag, i_max + lag + 1)
            js2 = js2[(js2 >= 0) & (js2 < n)]
            if len(js2) > 0:
                grad[js2] -= wk * h[js2 - lag]
    return dx * beta * grad

def exact_coord_sweeps(h, target, n_sweeps, t0, budget_s, n_samples=49):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    lags = np.arange(-(n - 1), n)
    stall_count = 0

    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < 8:
            break
        order = np.random.permutation(n)
        improved = False
        for j in order:
            if budget_s - (time.time() - t0) < 4:
                break
            h_save = h[j]
            h[j] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode="full")
            h[j] = h_save

            jp = j + lags
            jm = j - lags
            valid_p = (jp >= 0) & (jp < n)
            valid_m = (jm >= 0) & (jm < n) & (lags != 0)
            c_arr = np.zeros(2 * n - 1)
            c_arr[valid_p] += (1.0 - h[jp[valid_p]])
            c_arr[valid_m] -= h[jm[valid_m]]
            c_arr[n - 1] = 1.0

            vs = np.linspace(0.0, 1.0, n_samples)
            vals_all = corr_base[None, :] + c_arr[None, :] * vs[:, None]
            vals_all[:, n - 1] -= vs ** 2
            f_vals = np.max(vals_all, axis=1) * dx
            best_idx = np.argmin(f_vals)
            best_v = vs[best_idx]

            if 0 < best_idx < n_samples - 1:
                lo = vs[max(0, best_idx - 2)]
                hi = vs[min(n_samples - 1, best_idx + 2)]
                vs2 = np.linspace(lo, hi, 65)
                vals2 = corr_base[None, :] + c_arr[None, :] * vs2[:, None]
                vals2[:, n - 1] -= vs2 ** 2
                f2 = np.max(vals2, axis=1) * dx
                best_v = vs2[np.argmin(f2)]

            h[j] = best_v
            h = project_h(h, target)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True

        if not improved:
            stall_count += 1
            if stall_count >= 3:
                break
        else:
            stall_count = 0

    return h, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(256, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)

    # Phase 1: Coordinate descent at parent resolution
    n0 = len(h_parent)
    tgt0 = n0 / 2.0
    h1 = project_h(best_h, tgt0)
    h1, c5_1 = exact_coord_sweeps(h1, tgt0, 12, t0, budget_s, n_samples=49)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c5_1, n0

    # Phase 2: L-BFGS-B with increasing beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h2 = best_h.copy()
        n2 = len(h2)
        tgt2 = n2 / 2.0
        bnds = Bounds(np.zeros(n2), np.ones(n2))
        for beta in [80, 200, 500, 1200]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            try:
                res = minimize(sml_obj, h2, args=(beta, 2.0 / n2),
                               method='L-BFGS-B', jac=sml_grad, bounds=bnds,
                               options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-14})
                h2 = project_h(res.x, tgt2)
                c5_2 = compute_c5(h2)
                if c5_2 < best_c5:
                    best_h, best_c5, best_n = h2.copy(), c5_2, n2
            except Exception:
                pass

    # Phase 3: More coordinate descent (finer sampling)
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        h3 = best_h.copy()
        h3, c5_3 = exact_coord_sweeps(h3, best_n / 2.0, 15, t0, budget_s, n_samples=81)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, best_n

    # Phase 4: Try slightly different resolutions
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        for n_try in [256, 384]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 35:
                break
            if n_try == best_n:
                continue
            h_try = resample_h(best_h, n_try)
            h_try = project_h(h_try, n_try / 2.0)
            h_try, c5_try = exact_coord_sweeps(h_try, n_try / 2.0, 8, t0, budget_s, n_samples=49)
            if c5_try < best_c5:
                best_h, best_c5, best_n = h_try, c5_try, n_try

    # Phase 5: Perturbation restarts at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        nf = best_n
        tf = nf / 2.0
        for att in range(5):
            remaining = budget_s - (time.time() - t0)
            if remaining < 20:
                break
            scale = 0.015 / (att + 1)
            hp = project_h(best_h + np.random.randn(nf) * scale, tf)
            hp, cp = exact_coord_sweeps(hp, tf, 8, t0, budget_s, n_samples=49)
            if cp < best_c5:
                best_h, best_c5 = hp, cp

    # Phase 6: Final L-BFGS polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        nf = best_n
        tf = nf / 2.0
        bnds = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [1500, 3000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            try:
                res = minimize(sml_obj, best_h, args=(beta, 2.0 / nf),
                               method='L-BFGS-B', jac=sml_grad, bounds=bnds,
                               options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-14})
                best_h = project_h(res.x, tf)
                best_c5 = compute_c5(best_h)
            except Exception:
                pass

    # Phase 7: Final coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = exact_coord_sweeps(best_h, best_n / 2.0, 8, t0, budget_s, n_samples=129)

    return best_h, best_c5, best_n
