    return np.ones(x_in.shape)

    h = np.clip(h.copy(), 0.0, 1.0)

    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, target):
    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter):
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except Exception:
        return h0

def _eval_pair_t(t, h, i, j):
    h[i] += t
    h[j] -= t
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(corr) * dx)
    h[i] -= t
    h[j] += t
    return val

def pair_search(h, n, target, n_sweeps, t0, budget, rng, pps=2000):
    cur = h.copy()
    best = compute_c5(cur)
    for sw in range(n_sweeps):
        if budget - (time.time() - t0) < 5:
            break
        imp = 0
        for _ in range(pps):
            if budget - (time.time() - t0) < 2:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            t_lo = -cur[i]
            t_hi = 1.0 - cur[j]
            if t_hi - t_lo < 1e-10:
                continue
            best_t = 0.0
            best_f = best
            span = t_hi - t_lo
            for frac in (0.15, 0.4, 0.65, 0.85):
                t = t_lo + frac * span
                f = _eval_pair_t(t, cur, i, j)
                if f < best_f:
                    best_f = f
                    best_t = t
            if best_f < best - 1e-13:
                lo = max(t_lo, best_t - 0.02)
                hi = min(t_hi, best_t + 0.02)
                if hi - lo > 1e-6:
                    try:
                        res = minimize_scalar(_eval_pair_t, args=(cur, i, j),
                                             bounds=(lo, hi), method='bounded',
                                             options={'xatol': 1e-7})
                        if res.fun < best_f:
                            best_f = res.fun
                            best_t = res.x
                    except Exception:
                        pass
                cur[i] = np.clip(cur[i] + best_t, 0.0, 1.0)
                cur[j] = np.clip(cur[j] - best_t, 0.0, 1.0)
                best = compute_c5(cur)
                imp += 1
        if imp == 0 and sw >= 2:
            break
    return cur, best

Let me review for potential issues:
1. The `_eval_pair_t` function modifies `h` in place and reverts. This is safe as long as no exception occurs between the modification and reversion. But if an exception occurs (e.g., in np.correlate), the values won't be reverted. However, np.correlate on a valid array shouldn't raise an exception.

2. The `minimize_scalar` call with `args=(cur, i, j)` - the function signature is `_eval_pair_t(t, h, i, j)` so `t` is the variable and `h, i, j` are the args. This should work.

3. In `pair_search`, after accepting a move, I compute `best = compute_c5(cur)` which recomputes the full C5. This is a safety check to make sure the actual C5 matches what we expected. It adds a bit of overhead but ensures correctness.

4. The multi-start phase tries different perturbation strategies. The "block roll" and "block swap" are structural perturbations that might help escape local minima.

5. The resolution change (Phase 4) might help if the optimal at n=320 is slightly worse than at n=240 or n=280 due to the discretization.

But that's 320*319/2 ≈ 51K pairs, and each requires ~10 C5 evaluations, so ~510K evaluations per sweep. At ~20μs per evaluation, that's ~10s per sweep. I could afford 1-2 such sweeps.
