    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:

        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        top_idx = np.argsort(-corr)[:top_k]
    else:

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -50, 50)))

def momentum_sigmoid(x, beta, n_iter, lr, mom, top_k, t0, deadline):
    n = len(x)
    tgt = n / 2.0
    xc = x.copy()
    v = np.zeros(n)

            break
        h = sigmoid(xc)
        _, grad_h = smoothmax_val_grad(h, beta, top_k)
        # Chain rule: d/dx = d/dh * dh/dx = d/dh * h * (1-h)
        grad_x = grad_h * h * (1.0 - h)
        # Sum constraint: project onto sum(sigmoid(x)) = n/2
        # Use a simple correction
        diff = tgt - h.sum()
        # Adjust x to fix the sum
        # dh/dx = h*(1-h), so dx ≈ diff / sum(h*(1-h))

        h_new = sigmoid(xc)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nk = min(top_k, len(corr))
    top_idx = np.argsort(-corr)[:nk]
    c_sel = corr[top_idx]
    m = c_sel[0]
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i in range(nk):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = int(top_idx[i]) - (n - 1)
        j1lo, j1hi = max(0, -s), min(n, n - s)
        if j1hi > j1lo:
            grad[j1lo:j1hi] += wi * (1.0 - h[j1lo+s:j1hi+s])
        j2lo, j2hi = max(0, s), min(n, n + s)
        if j2hi > j2lo:
            grad[j2lo:j2hi] -= wi * h[j2lo-s:j2hi-s]
    grad *= dx
    return val, grad

                best, bh = c, h_new
    if time.time() < dl - 20:
        h_new = slsqp_solve(bh, 600, 300, 8)
        c = compute_c5(h_new)
        if c < best:
            best, bh = c, h_new

    return bh, best, n
