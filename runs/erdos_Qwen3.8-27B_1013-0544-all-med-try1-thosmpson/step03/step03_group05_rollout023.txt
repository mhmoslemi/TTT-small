def binary_pattern(n, period, duty=0.5):
    """Create a binary pattern with given period and duty cycle."""
    h = np.zeros(n)
    # period is in grid units, duty is fraction of period that's 1
    ones_per_period = int(round(period * duty))
    for start in range(0, n, period):
        end = min(start + ones_per_period, n)
        h[start:end] = 1.0
    # Project to sum = n/2
    target = n / 2.0
    # Adjust by making it slightly non-binary
    # Find how many we need to change
    s = h.sum()
    if abs(s - target) > 0.01:
        # Need to adjust - make a partial cell
        deficit = target - s
        if deficit > 0:
            # Need more 1s: change a 0 to partial
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = min(1.0, deficit)
        else:
            # Need fewer 1s: change a 1 to partial
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = max(0.0, 1.0 + deficit)
    return h

            # Simpler: use ternary search or just evaluate at a few points.
            # Since f is convex, minimum is where the slope changes sign.
            
            # Find the slope at x=0: it's the max a[s] (among those where b[s] is max)
            # Actually, f'(0) = max of a[s] for s in argmax(b[s])
            # And f'(1) = max of a[s] for s in argmax(a[s]+b[s])
            
            # Let me just use a numerical approach:
            # Evaluate f at x = 0, 0.25, 0.5, 0.75, 1 and the old value
            # Then narrow down.
            
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0, old_val]
            # Also add intersection points of top lines
            sorted_idx = np.argsort(-b)[:top_k]  # top k by intercept
            for ii in range(min(top_k, len(sorted_idx))):
                for jj in range(ii+1, min(top_k, len(sorted_idx))):
                    si, sj = sorted_idx[ii], sorted_idx[jj]
                    if abs(a[si] - a[sj]) > 1e-15:
                        x_int = (b[sj] - b[si]) / (a[si] - a[sj])
                        if 0.0 <= x_int <= 1.0:
                            candidates.append(x_int)
            
            best_x = old_val
            best_f = np.inf
            for x in candidates:
                x = np.clip(x, 0.0, 1.0)
                f_val = np.max(a * x + b)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x
                    
            # Also try a fine search around the best candidate
            if 0 < best_x < 1:
                for dx_search in [0.01, 0.005, 0.001]:
                    for delta in [-dx_search, dx_search]:
                        x_try = np.clip(best_x + delta, 0, 1)
                        f_val = np.max(a * x_try + b)
                        if f_val < best_f:
                            best_f = f_val
                            best_x = x_try
            
            h[i] = best_x
            # Update overlaps
            delta = best_x - old_val
            if abs(delta) > 1e-15:
                overlaps += a * delta
    
    # Project to satisfy sum constraint
    h = project_sum(h, n / 2.0)
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[n - 1:] * dx

def coord_descent_sweep(h, top_k=12, project_every=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    overlaps = compute_overlaps(h)
    for i in range(n):
        old_val = h[i]
        a = np.zeros(n)
        a[0] = 1.0 - 2.0 * old_val
        for s in range(1, n):
            coef = 0.0
            if s <= i:
                coef += 1.0 - h[i - s]
            if s <= n - 1 - i:
                coef -= h[i + s]
            a[s] = coef
        b = overlaps - a * old_val
        sorted_idx = np.argsort(-b)[:top_k]
        a_top = a[sorted_idx]
        b_top = b[sorted_idx]
        candidates = np.array([0.0, 0.25, 0.5, 0.75, 1.0, old_val])
        pts = []
        for ii in range(top_k):
            for jj in range(ii + 1, top_k):
                if abs(a_top[ii] - a_top[jj]) > 1e-14:
                    xi = (b_top[jj] - b_top[ii]) / (a_top[ii] - a_top[jj])
                    if -0.01 <= xi <= 1.01:
                        pts.append(xi)
        if pts:
            candidates = np.concatenate([candidates, np.array(pts)])
        candidates = np.clip(candidates, 0.0, 1.0)
        f_vals = np.max(a[:, None] * candidates[None, :] + b[:, None], axis=0)
        best_idx = np.argmin(f_vals)
        best_x = candidates[best_idx]
        if 0.01 < best_x < 0.99:
            for eps in [0.02, 0.005, 0.001]:
                for d in [-eps, eps]:
                    xt = np.clip(best_x + d, 0, 1)
                    ft = np.max(a * xt + b)
                    if ft < f_vals[best_idx] - 1e-15:
                        best_x = xt
                        f_vals[best_idx] = ft
        h[i] = best_x
        delta = best_x - old_val
        if abs(delta) > 1e-15:
            overlaps += a * delta
        if (i + 1) % project_every == 0:
            h = project_sum(h, target)
            overlaps = compute_overlaps(h)
    h = project_sum(h, target)
    return h, compute_c5(h)

def binary_init(n, period):
    ones = max(1, int(round(period * 0.5)))
    h = np.zeros(n)
    for start in range(0, n, period):
        h[start:min(start + ones, n)] = 1.0
    return project_sum(h, n / 2.0)

def fourier_init(n, freqs, amps, phases):
    x = np.arange(n, dtype=float)
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.cos(2.0 * np.pi * f * x / n + p)
    return project_sum(h, n / 2.0)

def smooth_max_obj(x, dx, beta):
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    return m + np.log(np.sum(np.exp((c - m) * beta)) / len(c)) / beta

def sum_con_fun(x, target):
    return x.sum() - target

def sum_con_jac(x, target):
    return np.ones(x.shape)

def slsqp_polish(h, n, target, time_left_fn, max_iter=800):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in [100, 300, 800, 2000]:
        if time_left_fn() < 10:
            break
        h = project_sum(h, target)
        try:
            res = minimize(
                lambda x: smooth_max_obj(x, dx, beta),
                h, method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: sum_con_fun(x, target),
                              'jac': lambda x: sum_con_jac(x, target)}],
                options={'maxiter': max_iter, 'ftol': 1e-15})
            h_new = project_sum(res.x, target)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        except Exception:
            pass
    return h

def resample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    return project_sum(np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h), n_new / 2.0)

    np.random.seed(seed)

    def tl():

        inits.append(fourier_init(n, [2, 3], [0.22, 0.18], [0, np.pi / 4]))

                break

            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    return best_h, best_c5, len(best_h)
