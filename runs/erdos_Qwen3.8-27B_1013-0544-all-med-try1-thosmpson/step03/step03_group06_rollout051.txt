        if tl() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        wi = int(np.argmax(corr))
        wl = wi - (n - 1)
        best_h, best_c5, imp = do_subgrad_step(best_h, target, [wl], fine_steps)
        if not imp and it % 200 == 199:
            # Try paired moves on the worst lag
            al = abs(wl)
            if 0 < al < n:
                nv = n - al
                if nv > 0:
                    for _ in range(5):
                        i = np.random.randint(nv)
                        j = i + al if wl > 0 else i - wl
                        for d in [0.005, -0.005, 0.002, -0.002, 0.001]:
                            h_t = best_h.copy()
                            h_t[i] += d
                            h_t[j] -= d
                            if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                                h_t = project_h(h_t, target)
                                if compute_c5(h_t) < best_c5:
                                    best_h = h_t
                                    best_c5 = compute_c5(h_t)
                                    break

    return best_h, best_c5, len(best_h)
