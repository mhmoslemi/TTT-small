def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):

    n = len(h)

    deficit = target - len(ones)

        order = zeros[np.argsort(h[zeros])]
        h[order[:deficit]] = 1.0
    elif deficit < 0:
        order = ones[np.argsort(h[ones], ascending=False)]
        h[order[:(-deficit)]] = 0.0

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target):
    h = h.copy()
    # Sort by value, set top 'target' to 1, rest to 0
    idx = np.argsort(h)[::-1]
    h[:] = 0.0
    h[idx[:target]] = 1.0
    return h

def sa_binary(h, budget=50.0, seed=42):
    """Simulated annealing on binary sequence."""
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    h = to_binary(h, target)
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_val = np.max(corr)
    best_val = cur_val
    
    rng = np.random.RandomState(seed)
    t0 = time.time()
    T = cur_val * 0.5  # in units of correlation, not C5
    T_min = 0.5
    
    while time.time() - t0 < budget:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = np.max(new_corr)
        
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            corr = new_corr
            if cur_val < best_val:
                best_val = cur_val
        else:
            h[i], h[j] = 1.0, 0.0
        
        T = max(T * 0.9995, T_min)

    """SLSQP refinement with smooth max."""
    target = n_pts / 2.0

    x_new = np.linspace(0, 2, n_pts)

        c_shift = np.clip(500.0 * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(c_shift)) / len(c)) / 500.0) * dx
    
    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    def con_fun(x):
        return np.sum(x) - target

        return np.ones(n_pts)

        h_new = project_h(res.x, target)

    except Exception:
        return h_cur
    finally:
        pass

def compute_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
