For a "good" f, we want R_f(k) to be large and uniform for all k. The maximum possible uniform value would be if f is an eigenfunction of the translation operator with eigenvalue 1 (i.e., constant), but ∫f=0 rules that out.

The next best would be for f to be "almost" an eigenfunction for all translations, which is impossible. So we need a trade-off.

In the Fourier domain: if f has Fourier series f(x) = sum_k a_k cos(kπx) + b_k sin(kπx) (periodic on [0,2]), then R_f(k) = sum_j (a_j² + b_j²)/2 * cos(jπk) (for the periodic case).

To maximize the minimum of R_f, we want the Fourier coefficients to be concentrated at frequencies where cos(jπk) is large for all k simultaneously. But cos(jπk) oscillates, so for any single frequency, there are shifts where it's negative.

The optimal strategy is to put energy at multiple frequencies to "average out" the oscillations. This is related to the theory of "low autocorrelation sequences" or "Golay sequences."

This spectral perspective suggests that a good starting point might be:

    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_conv_weighted(h, weights):
    """Compute sum_k weights[k] * grad(conv[k]) efficiently."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:n] += w * (1.0 - h[:n-s])
                grad[:n-s] -= w * h[s:n]
        else:
            sp = -s
            if sp < n:
                grad[:n-sp] += w * (1.0 - h[sp:n])
                grad[sp:n] -= w * h[:n-sp]
    return grad

    conv = compute_conv(h)
    m = conv.max()
    exps = np.exp((conv - m) / t)
    sm = t * (m / t + np.log(exps.sum()))
    penalty = mu * 0.5 * (h.sum() - n / 2.0) ** 2
    return sm * (2.0 / n) + penalty

    m = conv.max()
    exps = np.exp((conv - m) / t)
    weights = exps / exps.sum()

    y = h0.copy()

        obj_old = soft_max_obj(h, t, mu, n)

            lr /= 2
            h_new = np.clip(y - lr * grad, 0.0, 1.0)

                break

            y = np.clip(y, 0.0, 1.0)  # Keep y in bounds (optional)
            h = h_new
            k_ista += 1
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
    return best_h, best_obj

def coord_descent(h, n_passes=10, n_active=30):
    """Coordinate descent: for each i, solve 1D subproblem."""

    h = h.copy()

            a_vals = []
            C_vals = []

                s = k_idx - (n - 1)
                # a = coefficient of h[i] in conv[k_idx]

                if s >= 0:

                if 0 <= i - s < n:
                    a += 1.0 - h[i - s]
                if 0 <= i + s < n:
                    a -= h[i + s]
                C = conv[k_idx] - a * h[i]
                a_vals.append(a)
                C_vals.append(C)
            a_vals = np.array(a_vals)
            C_vals = np.array(C_vals)
            # Solve: min_{t in [0,1]} max_k (C_k + a_k * t)
            # Try endpoints and pair intersections
            candidates = [0.0, 1.0]
            for p in range(len(a_vals)):
                for q in range(p + 1, len(a_vals)):
                    da = a_vals[p] - a_vals[q]
                    if abs(da) < 1e-15:
                        continue
                    t_int = (C_vals[q] - C_vals[p]) / da
                    if 0.0 <= t_int <= 1.0:
                        candidates.append(t_int)
            best_t = h[i]
            best_val = np.max(C_vals + a_vals * h[i])
            for t_c in candidates:
                val = np.max(C_vals + a_vals * t_c)
                if val < best_val:
                    best_val = val
                    best_t = t_c
            h[i] = best_t
    # Project to feasible
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def compute_conv(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_conv_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:n] += w * (1.0 - h[:n-s])
                grad[:n-s] -= w * h[s:n]
        else:
            sp = -s
            if sp < n:
                grad[:n-sp] += w * (1.0 - h[sp:n])
                grad[sp:n] -= w * h[:n-sp]
    return grad

def fista_smooth(h0, t, mu, n, max_iter=2000):
    """FISTA for soft_max + penalty, projecting onto [0,1]."""
    h = h0.copy()
    y = h0.copy()
    k_ac = 1
    # Lipschitz estimate
    L = 8.0 / t + mu
    lr = 1.0 / L
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        conv = compute_conv(y)
        m = conv.max()
        exps = np.exp((conv - m) / t)
        weights = exps / exps.sum()
        grad_sm = grad_conv_weighted(y, weights) * (2.0 / n)
        grad_pen = mu * (y.sum() - n / 2.0)
        grad = grad_sm + grad_pen
        
        h_new = np.clip(y - lr * grad, 0.0, 1.0)

            if lr < 1e-12:

            h_new = np.clip(y - lr * grad, 0.0, 1.0)

        y = h_new + ((k_ac - 1.0) / beta) * (h_new - h)

        best_c5 = c5_final
    return best_h, best_c5

def coord_descent(h, n_passes=50, n_active=40):
    """Coordinate descent with exact 1D subproblem."""
    n = len(h)
    h = h.copy()
    conv = compute_conv(h)
    
    for p in range(n_passes):
        # Project periodically
        if p > 0 and p % 10 == 0:
            h = project_feasible(h, n)
            conv = compute_conv(h)
        
        for i in range(n):
            # Get active shifts
            if n_active < len(conv):
                idx = np.argpartition(conv, -n_active)[-n_active:]
            else:
                idx = np.arange(len(conv))
            na = len(idx)
            
            # Compute a_k for each active k
            a = np.zeros(na)
            for j, k_idx in enumerate(idx):
                s = k_idx - (n - 1)
                coef = 0.0
                if 0 <= i - s < n:
                    coef += 1.0 - h[i - s]
                if 0 <= i + s < n:
                    coef -= h[i + s]
                a[j] = coef
            C = conv[idx] - a * h[i]
            
            # 1D minimax: min_t max_j (C_j + a_j * t), t in [0,1]
            # Candidates: 0, 1, and pair intersections
            best_t = h[i]
            best_val = np.max(C + a * h[i])
            
            # Check endpoints
            val0 = np.max(C)
            if val0 < best_val:
                best_val, best_t = val0, 0.0
            val1 = np.max(C + a)
            if val1 < best_val:
                best_val, best_t = val1, 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def compute_conv(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_conv_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:n] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:n]
        else:
            sp = -s
            if sp < n:
                grad[:n - sp] += w * (1.0 - h[sp:n])
                grad[sp:n] -= w * h[:n - sp]
    return grad
