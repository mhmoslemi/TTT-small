import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_val(h):
    return compute_corr(h).max()

def grad_lag(h, n_lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, n_lag)
    e1 = min(N, N + n_lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - n_lag:e1 - n_lag]
    s2 = max(0, -n_lag)
    e2 = min(N, N - n_lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + n_lag:e2 + n_lag]
    return g

def topk_gradient(h, corr, k, alpha):
    N = len(h)
    if k >= len(corr):
        k = len(corr)
    top_k = np.argsort(corr)[-k:]
    c = corr[top_k] - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        return grad_lag(h, top_k[-1] - (N - 1))
    w /= wsum
    g = np.zeros(N)
    for idx, wi in zip(top_k, w):
        if wi < 1e-6:
            continue
        g += wi * grad_lag(h, idx - (N - 1))
    return g

def soft_gradient_full(h, corr, alpha):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        return grad_lag(h, np.argmax(corr) - (N - 1))
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-8:
            continue
        g += w[j] * grad_lag(h, j - (N - 1))
    return g

def optimize(h, N, rng, t_end, alpha_init, alpha_final, topk, max_iter=100000):

    corr = compute_corr(h)
    best_val = val
    best_h = h.copy()
    lr = 0.025
    stagnant = 0
    momentum = np.zeros(N)
    beta = 0.55

    for it in range(max_iter):

            break

        alpha = alpha_init + (alpha_final - alpha_init) * frac
        k = topk if topk < 2 * N else 2 * N

        if k <= 10:

        else:

def local_search(h, N, rng, t_end, max_iters=50000):
    val = corr_val(h)
    corr = compute_corr(h)
    best_val = val
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() > t_end:
            break
        interior = np.where((h > 5e-5) & (h < 1.0 - 5e-5))[0]
        n_int = len(interior)
        if n_int < 2:
            break
        i1 = interior[rng.randint(n_int)]
        i2 = interior[rng.randint(n_int)]
        if i1 == i2:
            continue
        accepted = False
        for eps in (0.02, 0.01, 0.005):
            if h[i1] < eps or h[i2] > 1.0 - eps:
                continue
            h[i1] -= eps
            h[i2] += eps
            val_new = corr_val(h)
            if val_new < val - 1e-15:
                val = val_new
                accepted = True
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
            else:
                h[i1] += eps
                h[i2] -= eps
                break
        if not accepted:
            h[i1] += eps if h[i1] < eps else 0.0
            h[i2] -= eps if h[i2] > 1.0 - eps else 0.0
    return best_h, best_val

    x0 = np.linspace(0, 1, len(h_src), endpoint=False)
    x1 = np.linspace(0, 1, n_dst, endpoint=False)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

        # Adjust to get sum right
    h = project_feasible(h, N)
    return h
