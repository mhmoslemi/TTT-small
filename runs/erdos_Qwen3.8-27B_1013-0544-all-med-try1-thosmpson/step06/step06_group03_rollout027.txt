def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def soft_max_val_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    # log-sum-exp for numerical stability
    shifted = (conv - m) / t
    logsumexp = m / t + np.log(np.sum(np.exp(shifted)))
    val = t * logsumexp * dx
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
    return val, grad * dx

        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    v = np.zeros(n)

    lr = lr_init

        h_new = project_sum(h - lr * v, n)

            lr *= 0.5
            v = beta * v
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.05, 0.1)
    return best_h, best_c5

                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_sum(res.x, n)
    except:
        return h_init

    shifted = (conv - m) / t
    sumexp = np.sum(np.exp(shifted))
    val = t * (m / t + np.log(sumexp)) * dx
    weights = np.exp(shifted) / sumexp
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-11:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
    return val, grad * dx

def project_sum(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def pgd_nesterov(h_init, max_iter, lr_init, t_init, t_final, beta_init):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    beta = beta_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        frac = it / max(max_iter - 1, 1)
        t = t_init * (t_final / t_init) ** frac
        beff = beta_init * (1.0 - 0.3 * frac)
        c5s, grad = soft_max_val_grad(h, t)
        v = beff * v + (1.0 - beff) * grad
        h_new = project_sum(h - lr * v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        diff = c5_new - c5s
        if diff > 0.5 * lr * np.sum(grad * v) + 1e-18:
            lr *= 0.5
            v = beff * v
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.03, 0.08)
    return best_h, best_c5

def lbfgs_smooth(h_init, t, max_iter=3000):
    n = len(h_init)
    target = n / 2.0
    def obj(h):
        v, _ = soft_max_val_grad(h, t)
        return v + 50.0 * (np.sum(h) - target) ** 2
    def grad(h):
        _, g = soft_max_val_grad(h, t)
        return g + 100.0 * (np.sum(h) - target)
    try:
        res = minimize(obj, h_init.copy(), jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_sum(res.x, n)
    except Exception:
        return h_init

def fourier_warm_start(n, seed, budget_s=30):
    t_start = time.time()
    x = np.linspace(0, 2.0, n, endpoint=False)
    rng = np.random.RandomState(seed)
    best_h, best_c5 = None, np.inf
    for m in range(2, 7):
        if time.time() - t_start > budget_s:
            break
        for _ in range(15):
            if time.time() - t_start > budget_s:
                break
            coeffs = rng.randn(2 * m) * 0.15
            for _ in range(300):
                h = np.full(n, 0.5)
                for j in range(m):
                    h += coeffs[2*j] * np.cos((j+1)*np.pi*x)
                    h += coeffs[2*j+1] * np.sin((j+1)*np.pi*x)
                h = np.clip(h, 0.0, 1.0)
                h = project_sum(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                step = 0.008 * (1.0 - _ / 400.0)
                for j in range(2 * m):
                    coeffs[j] += 1e-5
                    hp = np.full(n, 0.5)
                    for k2 in range(m):
                        hp += coeffs[2*k2]*np.cos((k2+1)*np.pi*x)
                        hp += coeffs[2*k2+1]*np.sin((k2+1)*np.pi*x)
                    hp = np.clip(hp, 0.0, 1.0)
                    hp = project_sum(hp, n)
                    g = (compute_c5(hp) - c5) / 1e-5
                    coeffs[j] -= 1e-5
                    coeffs[j] -= step * g
    return best_h, best_c5

def hard_pgd_polish(h, max_iter, lr_init):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        conv = np.correlate(best_h, 1.0 - best_h, mode="full")
        c5 = np.max(conv) * dx
        # Use top-3 shifts for better gradient
        top_idx = np.argsort(conv)[-3:]
        grad = np.zeros(n)
        for idx in top_idx:
            shift = idx - (n - 1)
            if shift >= 0:
                if shift < n:
                    grad[shift:] += (1.0 - best_h[:n - shift])
                    grad[:n - shift] -= best_h[shift:]
            else:
                s = -shift
                if s < n:
                    grad[:n - s] += (1.0 - best_h[s:])
                    grad[s:] -= best_h[:n - s]
        grad *= dx / len(top_idx)
        h_new = project_sum(best_h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-14:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(200, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Fourier warm start for diverse initializations
    if tl() > 100:
        fw_h, fw_c5 = fourier_warm_start(n0, seed, budget_s=25)
        if fw_h is not None and fw_c5 < best_c5:
            best_h, best_c5 = fw_h, fw_c5

    # Phase 2: L-BFGS-B with annealing
    for t_val in [0.08, 0.04, 0.02, 0.01, 0.005]:
        if tl() < 40:
            break
        ht = lbfgs_smooth(best_h, t_val, max_iter=4000)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Momentum PGD with geometric annealing
    if tl() > 80:
        ht, c5t = pgd_nesterov(best_h, max_iter=40000, lr_init=0.008,
                                t_init=0.03, t_final=0.001, beta_init=0.9)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = sorted(set([max(80, n_base//2), 150, 200, 250, 320, 400, 500, 640]))
    n_candidates = [nt for nt in n_candidates if nt != n_base or nt <= 500]
    for n_try in n_candidates:
        if tl() < 120:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_sum(hi, n_try)

        for t_val in [0.03, 0.01]:
            if tl() < 60:
                break
            ht = lbfgs_smooth(hi, t_val, max_iter=2000)
        if tl() > 40:
            ht, c5t = pgd_nesterov(ht, max_iter=25000, lr_init=0.005,
                                    t_init=0.02, t_final=0.0005, beta_init=0.88)
        else:
            c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Diverse restarts (Fourier + random)
    n_r = len(best_h)
    x = np.linspace(0, 2.0, n_r, endpoint=False)
    for r in range(50):
        if tl() < 30:
            break
        # Mix of random and structured perturbations
        if r % 3 == 0:
            hr = np.full(n_r, 0.5)
            freq = rng.randint(1, 6)
            hr += 0.25 * np.sin(freq * np.pi * x + rng.uniform(0, 2*np.pi))
            hr += 0.15 * rng.randn(n_r)
        elif r % 3 == 1:
            hr = best_h + 0.1 * rng.randn(n_r)
        else:
            hr = rng.uniform(0.0, 1.0, n_r)
        hr = np.clip(hr, 0.0, 1.0)
        hr = project_sum(hr, n_r)
        ht, c5t = pgd_nesterov(hr, max_iter=4000, lr_init=0.015,
                                t_init=0.05, t_final=0.005, beta_init=0.85)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Final hard PGD polish
    if tl() > 20:
        best_h, best_c5 = hard_pgd_polish(best_h, max_iter=50000, lr_init=0.002)

    # Phase 7: One more smooth pass on best
    if tl() > 15:
        best_h = lbfgs_smooth(best_h, 0.003, max_iter=3000)
        ht, c5t = hard_pgd_polish(best_h, max_iter=30000, lr_init=0.001)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
