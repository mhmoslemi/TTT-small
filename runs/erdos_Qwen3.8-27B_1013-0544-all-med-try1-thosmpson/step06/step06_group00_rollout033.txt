    #             + delta * dx * I(j in valid range for second) * (-h[j-m])

    # result[k] = sum_i a[i] * b[i + k - (len(b)-1)]

    # result[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # valid i: 0<=i<n and 0<=i+k-(n-1)<n, i.e., (n-1)-k <= i <= n-1+k-(n-1) = k
    # Wait: 0 <= i + k - (n-1) < n means (n-1)-k <= i < (n-1)-k+n = 2n-1-k
    # And 0 <= i < n.
    # So i ranges from max(0, (n-1)-k) to min(n-1, 2n-2-k)
    
    # Let lo = max(0, n-1-k) and hi = min(n-1, 2n-2-k)
    # result[k] = sum_{i=lo}^{hi} h[i] * (1-h[i+k-(n-1)])
    
    # For h[j] to appear:
    # As first factor: j is in [lo, hi], and the second factor is (1-h[j+k-(n-1)])
    # As second factor: j+k-(n-1) is in [lo, hi], i.e., j is in [lo-(k-(n-1)), hi-(k-(n-1))] = [lo-n+1+k, hi-n+1+k]
    #   The first factor is h[j-(k-(n-1))] = h[j-n+1+k]
    
    # This is getting really messy. Let me just do it numerically.
    
    new_overlaps = overlaps.copy()
    # For each k, compute the change
    for k in range(2*n-1):
        m = k - (n-1)  # shift
        lo = max(0, n-1-k)
        hi = min(n-1, 2*n-2-k)
        change = 0.0
        # h[j] as first factor: j in [lo, hi]
        if lo <= j <= hi:
            # second factor index: j + m = j + k - (n-1)
            second_idx = j + m
            if 0 <= second_idx < n:
                change += delta * (1.0 - h[second_idx])
        # h[j] as second factor: j = i + m for some i in [lo, hi]
        # i = j - m
        i_second = j - m
        if lo <= i_second <= hi:
            first_idx = i_second  # = j - m
            if 0 <= first_idx < n:
                change -= delta * h[first_idx]
        new_overlaps[k] += change * dx
    
    return new_overlaps

But I can vectorize the update! Instead of looping over k, I can compute the changes for all k simultaneously using numpy.

Let me think about this more carefully. For a fixed j and new value, the change in overlap at shift m (index k = m + n - 1) is:
Δoverlap(m) = delta * dx * [(1-h[j+m]) * I(valid_first) - h[j-m] * I(valid_second)]

where:
- valid_first: 0 <= j < n and 0 <= j+m < n, i.e., -j <= m <= n-1-j
- valid_second: 0 <= j-m < n and 0 <= j < n, i.e., j-n+1 <= m <= j

So:
Δoverlap(m) = delta * dx * [(1-h[j+m]) * I(max(-j, -(n-1)) <= m <= min(n-1-j, n-1)) - h[j-m] * I(max(j-n+1, -(n-1)) <= m <= min(j, n-1))]

Since m ranges from -(n-1) to (n-1):
- valid_first: max(-j, -(n-1)) <= m <= n-1-j (since j >= 0, -j >= -(n-1) is always true; and n-1-j <= n-1 is always true)
  So: -j <= m <= n-1-j
- valid_second: j-n+1 <= m <= j (since j-n+1 >= -(n-1) means j >= 2, and j <= n-1)
  Actually: max(j-n+1, -(n-1)) <= m <= min(j, n-1)
  Since j <= n-1, min(j, n-1) = j
  And j-n+1 >= -(n-1) iff j >= 2 (for n >= 3)
  So for j >= 2: j-n+1 <= m <= j
  For j = 0: max(1-n, -(n-1)) = 1-n, so 1-n <= m <= 0
  For j = 1: max(2-n, 1-n) = 2-n (for n>1), so 2-n <= m <= 1
  
  Actually, j-n+1 is always >= -(n-1) when j >= 2. For j=0: 0-n+1 = 1-n = -(n-1). For j=1: 1-n+1 = 2-n = -(n-2) > -(n-1).
  
  So valid_second: max(j-n+1, -(n-1)) <= m <= j
  = (j-n+1 if j>=1 else -(n-1)) ... hmm this is getting complicated.

Wait, there's a problem. After I project to feasibility, the overlaps I computed are no longer valid because the projection changed other coordinates. I need to recompute the overlaps from scratch after projection.

This is a fundamental issue with coordinate descent under a sum constraint. When I change h[j] and then project, the projection might change many coordinates.

One approach: instead of projecting after each coordinate update, I can:
1. Change h[j] to a new value
2. Adjust all other coordinates proportionally to maintain the sum (like a "flat" projection)
3. This is still O(n) per update

Or I can use a different constraint handling:
- Allow the sum to drift slightly during the sweep
- Project at the end of each sweep

This is a common technique in coordinate descent. Let me use this:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(np.clip(temp * (corr - m), -50, 50))
    w = exp_vals / exp_vals.sum()
    obj = (m + np.log(exp_vals.sum()) / temp) * dx
    
    # Gradient
    grad = np.zeros(n)
    m_vals = np.arange(-(n-1), n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        m = k - (n-1)
        wk = w[k] * dx
        lo1 = max(0, n-1-k)
        hi1 = min(n-1, 2*n-2-k)
        if lo1 <= hi1:
            # h[j] as first factor, j in [lo1, hi1]
            second_idx = np.arange(lo1, hi1+1) + m
            valid = (second_idx >= 0) & (second_idx < n)
            grad[lo1:hi1+1][valid] += wk * (1.0 - h[second_idx[valid]])
            # h[j] as second factor: j = i + m for i in [lo1, hi1]
            # j in [lo1+m, hi1+m]
            first_idx = np.arange(lo1, hi1+1)  # these are i values
            j_vals = first_idx + m
            valid2 = (j_vals >= 0) & (j_vals < n)
            grad[j_vals[valid2]] -= wk * h[first_idx[valid2]]
    return obj, grad

def slsqp_optimize(h_start, n, temp, maxiter=300, tol=1e-12):
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj_and_grad(x, temp)[0]
    
    def grad(x):
        return smooth_obj_and_grad(x, temp)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_start, jac=grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': tol})
    return project_feasible(res.x, ts)

def coord_descent_efficient(h, n_levels=8, max_sweeps=50, time_budget=120, t_start=None):
    """Efficient coordinate descent with vectorized overlap updates."""
    n = len(h)
    dx = 2.0 / n
    ts = n / 2.0
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(overlaps)
    
    m_vals = np.arange(-(n-1), n)
    levels = np.linspace(0, 1, n_levels)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.time() - t_start > time_budget:
            break
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_v = orig
            best_ov = None
            
            for v in levels:
                if abs(v - orig) < 1e-15:
                    continue
                delta = v - orig
                
                # Compute change in all overlaps
                # For shift m (index k = m + n - 1):
                # Δov = delta * dx * [(1-h[j+m])*I(valid_first) - h[j-m]*I(valid_second)]
                change = np.zeros(2*n-1)
                
                # First term: h[j] as first factor
                # Valid when: 0 <= j < n and 0 <= j+m < n → -j <= m <= n-1-j
                lo1 = max(-j, -(n-1))
                hi1 = min(n-1-j, n-1)
                if lo1 <= hi1:
                    idx = m_vals[lo1 + (n-1): hi1 + (n-1) + 1] + j
                    change[lo1 + (n-1): hi1 + (n-1) + 1] += 1.0 - h[idx]
                
                # Second term: h[j] as second factor (j = i + m, so i = j - m)
                # Valid when: 0 <= j-m < n and 0 <= j < n → j-n+1 <= m <= j
                lo2 = max(j-n+1, -(n-1))
                hi2 = min(j, n-1)
                if lo2 <= hi2:
                    idx2 = j - m_vals[lo2 + (n-1): hi2 + (n-1) + 1]
                    change[lo2 + (n-1): hi2 + (n-1) + 1] -= h[idx2]
                
                new_ov = overlaps + delta * dx * change
                new_c5 = np.max(new_ov)
                
                if new_c5 < best_c - 1e-14:
                    best_c = new_c5
                    best_v = v
                    best_ov = new_ov
            
            if best_v != orig:
                h[j] = best_v
                if best_ov is not None:
                    overlaps = best_ov
                c5 = best_c
                improved = True
        
        # Project at end of sweep
        h = project_feasible(h, ts)
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        c5 = np.max(overlaps)
        
        if not improved:
            break
    
    return h, c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ev = np.exp(np.clip(temp * (corr - m), -50, 50))
    w = ev / ev.sum()
    obj = (m + np.log(ev.sum()) / temp) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m_val = k - (n - 1)
        wk = w[k] * dx
        lo = max(0, n - 1 - k)
        hi = min(n - 1, 2 * n - 2 - k)
        if lo > hi:
            continue
        # h[j] as first factor: j in [lo, hi], paired with h[j+m_val]
        sec = np.arange(lo, hi + 1) + m_val
        v1 = (sec >= 0) & (sec < n)
        if v1.any():
            jj = np.arange(lo, hi + 1)[v1]
            grad[jj] += wk * (1.0 - h[sec[v1]])
        # h[j] as second factor: j = i + m_val, i in [lo, hi]
        jj = np.arange(lo, hi + 1) + m_val
        v2 = (jj >= 0) & (jj < n)
        if v2.any():
            grad[jj[v2]] -= wk * h[np.arange(lo, hi + 1)[v2]]
    return obj, grad

def slsqp_step(h_start, n, temp, maxiter=200):
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj_grad(x, temp)[0]
    def grad(x):
        return smooth_obj_grad(x, temp)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_start, jac=grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, ts)

def cd_update_overlap(overlaps, h, j, v, n, dx):
    """Fast update of overlaps when h[j] changes to v."""
    delta = v - h[j]
    if abs(delta) < 1e-15:
        return overlaps
    m_vals = np.arange(-(n - 1), n)
    change = np.zeros(2 * n - 1)
    lo1 = max(-j, -(n - 1))
    hi1 = min(n - 1 - j, n - 1)
    if lo1 <= hi1:
        sl = slice(lo1 + n - 1, hi1 + n)
        idx = m_vals[sl] + j
        change[sl] += 1.0 - h[idx]
    lo2 = max(j - n + 1, -(n - 1))
    hi2 = min(j, n - 1)
    if lo2 <= hi2:
        sl2 = slice(lo2 + n - 1, hi2 + n)
        idx2 = j - m_vals[sl2]
        change[sl2] -= h[idx2]
    return overlaps + delta * dx * change

            orig = h[j]

                new_ov = cd_update_overlap(overlaps, h, j, v, n, dx)

                c5 = best_c

                           time_budget=min(80, remaining() - 15), t_start=time.time())
        if c < best_c5:
            best_c5 = c
            best_h = h_fin

    # Phase 6: Final SLSQP polish
    if remaining() > 30:
        h_pol = slsqp_step(best_h, len(best_h), 5.0, maxiter=100)
        c = compute_c5(h_pol)
        if c < best_c5:
            best_c5 = c
            best_h = h_pol

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
