    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, shift, N):
    """Gradient contribution for a specific shift."""
    g = np.zeros(N)
    # For the correlation term at this shift:
    # sum_i h[i] * (1 - h[i+shift]) for valid i
    if shift >= 0:
        # i goes from 0 to N-1-shift
        idx1 = np.arange(0, N - shift)
        g[idx1] += (1.0 - h[idx1 + shift])
        # h[i+shift] term: j = i+shift, j goes from shift to N-1
        if shift > 0:
            idx2 = np.arange(shift, N)
            g[idx2] -= h[idx2 - shift]
    else:
        s = -shift
        # i goes from s to N-1
        idx1 = np.arange(s, N)
        g[idx1] += (1.0 - h[idx1 - s])
        if s > 0:
            idx2 = np.arange(0, N - s)
            g[idx2] -= h[idx2 + s]
    return g

    c = corr - corr.max()
    # Select top k
    if top_k < len(corr):
        top_idx = np.argsort(c)[-top_k:]
        w_full = np.zeros(len(corr))
        w_vals = c[top_idx]
        w_vals = w_vals - w_vals.max()  # for numerical stability
        w_exp = np.exp(alpha * w_vals)

    val, corr, dx = compute_obj_corr(h)
    try_update(h, val)
    
    target = N / 2.0
    
    for round in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(N - 1):
            # Try optimizing h[i] and h[i+1] jointly
            # Constraint: h[i] + h[i+1] = s (fixed)
            s = h[i] + h[i+1]
            # h[i] = t, h[i+1] = s - t, where t in [max(0, s-1), min(1, s)]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi <= t_lo:
                continue
            
            # Golden section search over t
            best_t = h[i]
            best_val = val
            gratio = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = t_lo, t_hi
            c = b - gratio * (b - a)
            dpt = a + gratio * (b - a)
            
            # Evaluate at c and d
            h[i], h[i+1] = c, s - c
            vc, _, _ = compute_obj_corr(h)
            h[i], h[i+1] = dpt, s - dpt
            vd, _, _ = compute_obj_corr(h)
            
            for _ in range(30):
                if vc < vd:
                    b = dpt; dpt = c; vd = vc
                    c = b - gratio * (b - a)
                    h[i], h[i+1] = c, s - c
                    vc, _, _ = compute_obj_corr(h)
                else:
                    a = c; c = dpt; vc = vd
                    dpt = a + gratio * (b - a)
                    h[i], h[i+1] = dpt, s - dpt
                    vd, _, _ = compute_obj_corr(h)
                if abs(b - a) < 1e-10:
                    break
            
            # Take the better of c and dpt
            if vc < vd:
                h[i], h[i+1] = c, s - c
                new_val = vc
            else:
                h[i], h[i+1] = dpt, s - dpt
                new_val = vd
            
            if new_val < val - 1e-15:
                val = new_val
                improved = True
                try_update(h, val)
            else:
                # Revert
                h[i], h[i+1] = best_t, s - best_t
        
        if not improved:
            break
    
    return h, val

    best_val = obj_and_corr(h_init)[0]

    # Phase 1: Momentum GD at parent resolution
    N1 = len(h_init)
    t_p1 = min(t0 + 300, t_end - 350)
    h, v = momentum_gd(h_init.copy(), N1, rng, t_p1,
                       lr=0.04, mu=0.85, top_k=10, alpha=5.0)
    if v < best_val:
        best_val = v
        best_h = h

    # Phase 2: Multi-resolution exploration
    n_options = [64, 80, 96, 144, 160]
    for n_t in n_options:
        if time.time() > t_end - 180:
            break
        if len(best_h) == n_t:
            h_start = best_h.copy()
        else:
            h_start = interpolate_to_n(best_h, n_t)
        t_phase = min(time.time() + 80, t_end - 120)
        h, v = momentum_gd(h_start, n_t, rng, t_phase,
                           lr=0.03, mu=0.8, top_k=8, alpha=4.0)
        if v < best_val:
            best_val = v
            best_h = h

    # Phase 2b: Try Thue-Morse based starts
    if time.time() < t_end - 150:
        N_tm = len(best_h)
        h_tm = thue_morse_init(N_tm)
        t_phase = min(time.time() + 60, t_end - 100)
        h, v = momentum_gd(h_tm, N_tm, rng, t_phase,
                           lr=0.05, mu=0.8, top_k=10, alpha=5.0)
        if v < best_val:
            best_val = v
            best_h = h

    # Phase 3: Refine at best resolution
    N_best = len(best_h)
    t_p3 = min(time.time() + 180, t_end - 100)
    h, v = momentum_gd(best_h.copy(), N_best, rng, t_p3,
                       lr=0.02, mu=0.9, top_k=15, alpha=8.0)
    if v < best_val:
        best_val = v
        best_h = h

    # Phase 4: Pairwise CD
    N_best = len(best_h)
    t_p4 = min(time.time() + 150, t_end - 40)
    h, v = pairwise_cd(best_h.copy(), N_best, t_p4)
    if v < best_val:
        best_val = v
        best_h = h

    # Phase 5: Final momentum GD fine-tuning
    N_best = len(best_h)
    if time.time() < t_end:
        h, v = momentum_gd(best_h.copy(), N_best, rng, t_end,
                           lr=0.01, mu=0.95, top_k=20, alpha=12.0)
        if v < best_val:
            best_val = v
            best_h = h

    # Phase 6: One more pairwise pass if time allows
    N_best = len(best_h)
    if time.time() < t_end - 10:
        h, v = pairwise_cd(best_h.copy(), N_best, t_end)
        if v < best_val:
            best_val = v
            best_h = h

    return best_h, best_val, len(best_h)
