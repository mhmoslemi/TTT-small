import cvxpy as cp

def qp_step(h, n, target, max_lags=50):
    """Solve linearized QP at h, return new h."""
    dx = 2.0 / n
    corr = get_correlations(h)
    # Get top lags
    n_lags = 2 * n - 1
    top_k = min(max_lags, n_lags)
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    
    # Linearize c_k at h
    # c_k(h) = sum_i h[i+shift]*(1-h[i]) * dx
    # c_k(h + ε) ≈ c_k(h) + ε^T ∇c_k(h) * dx
    # (where ∇c_k is the gradient w.r.t. h before multiplying by dx)
    
    # For each top lag, compute c_k(h) and ∇c_k(h)
    # ∇c_k(h)[j] = [(1-h[j-shift]) - h[j+shift]] (for valid indices) * dx
    
    # Build the QP:
    # minimize t
    # subject to: c_k(h) + ∇c_k(h)^T (h_new - h) <= t for k in top_idx
    #             sum(h_new) = target
    #             0 <= h_new <= 1
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    base = h.copy()
    constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
    
    n_center = n - 1  # center of full correlation
    for L in top_idx:
        shift = L - n_center
        c_k_val = corr[L]
        # Gradient of c_k w.r.t. h
        grad_k = np.zeros(n)
        # h[j] in first factor: need j-shift valid
        lo1, hi1 = max(0, shift), min(n, n + shift)
        if lo1 < hi1:
            grad_k[lo1:hi1] += (1.0 - h[np.arange(lo1, hi1) - shift])
        # h[j] in second factor: need j+shift valid  
        lo2, hi2 = max(0, -shift), min(n, n - shift)
        if lo2 < hi2:

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_opt(h, n, temp, target, maxiter=300):
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    
    def obj(x):

        m = np.max(corr)
        s = np.exp(temp * (corr - m))
        return m + np.log(s.sum()) / temp
    
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, target)

def grad_c_L(h, L, n, dx):
