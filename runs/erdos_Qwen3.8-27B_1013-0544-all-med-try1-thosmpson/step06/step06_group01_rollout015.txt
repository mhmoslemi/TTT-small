```python
    # h[i] appears as 1-h[j-k+n-1] when j-k+n-1=i, i.e., k=j-i+n-1: contributes h[j]*(1-h[i]) if valid
    
    # Let's compute coef_k: the coefficient of h[i] in C(k)
    # C(k) = sum_j h[j]*(1-h[j-k+n-1])
    # Terms with h[i]:
    #   j=i: h[i]*(1-h[i-k+n-1]) -> coef is (1-h[i-k+n-1]) if 0<=i-k+n-1<n
    #   j-k+n-1=i -> k=j-i+n-1: h[j]*(1-h[i]) -> coef is -h[j] where j=k-i+n-1, valid if 0<=j<n
    #     So for a given k: j = k-i+n-1, valid if 0 <= k-i+n-1 < n, i.e., i-n+1 <= k <= i-1
    #     In that case, coef_k gets -h[k-i+n-1]
    
    # So: coef_k = (1-h[i-k+n-1]) * [0<=i-k+n-1<n] - h[k-i+n-1] * [0<=k-i+n-1<n]
    #         = (1-h[i-k+n-1]) * [k in [i-(n-1), i+(n-1)]] - h[k-i+n-1] * [k in [i-(n-1), i-1]]
    
    k_range = np.arange(2*n-1)
    # First term: valid when 0 <= i-k+n-1 < n, i.e., i-n+1 <= k <= i+n-1
    idx1 = i - k_range + n - 1  # this is i-k+n-1
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[idx1[valid1]], 0.0)
    
    # Second term: valid when 0 <= k-i+n-1 < n, i.e., i-n+1 <= k <= i-1
    idx2 = k_range - i + n - 1  # this is k-i+n-1
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, h[idx2[valid2]], 0.0)
    
    coef = term1 - term2
    B = C_current - h[i] * coef  # C(k) without h[i]'s contribution
    
    # Now minimize max_k (B[k] + t*coef[k]) over t in [0,1]
    # Find breakpoints
    # For each pair (k1, k2) where coef[k1] != coef[k2]:
    #   t = (B[k2] - B[k1]) / (coef[k1] - coef[k2])
    
    # Collect all candidate t values
    candidates = [0.0, 1.0, h[i]]  # include current value
    unique_coefs = np.unique(coef)
    if len(unique_coefs) > 1:
        # For each pair of distinct coefficients
        for a in unique_coefs:
            mask_a = (coef == a)
            for b in unique_coefs:
                if b == a:
                    continue
                mask_b = (coef == b)
                # t = (B_b - B_a) / (a - b) for lines a*t+B_a and b*t+B_b
                # But we need to consider all pairs of lines, not just groups
                # This is O(K^2) which is fine for K=189
                break
        # Actually let's just do it O(K^2) directly
        K = 2*n-1
        for k1 in range(K):
            for k2 in range(k1+1, K):
                if abs(coef[k1] - coef[k2]) < 1e-12:
                    continue
                t = (B[k2] - B[k1]) / (coef[k1] - coef[k2])
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    
    # Evaluate at all candidates
    best_t = h[i]
    best_val = np.max(B + h[i] * coef)
    for t in candidates:
        val = np.max(B + t * coef)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t
```

```python
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))

    w = es / se
    j_arr = np.arange(nc)

        a = i - j_arr + n - 1
        b = i + j_arr - n + 1

        vb = (b >= 0) & (b < n)

        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)

        if v < best_val:
            best_val = v
            best_t = t
    return best_t
```
