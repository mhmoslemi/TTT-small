```python
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def _repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-13
            if not mask.any():
                x[:] = target / n
                break
            x[mask] += diff * slack[mask] / max(slack[mask].sum(), 1e-30)
        else:
            mask = x > 1e-13
            if not mask.any():
                break
            x[mask] += diff * x[mask] / max(x[mask].sum(), 1e-30)
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _c_vals(h):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / h.size)

def _exact_c(h):
    return float(_c_vals(h).max())

def _smooth_c(h, tau):
    c = _c_vals(h)
    cm = float(c.max())
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return cm
    return cm + tau * np.log(s)

def _smooth_grad(h, tau):
    c = _c_vals(h)
    n = h.size
    cm = float(c.max())
    if tau <= 0:
        k = int(np.argmax(c))
        w = {k: 1.0}
    else:
        z = (c - cm) / tau
        z -= z.max()
        e = np.exp(z)
        s = e.sum()
        if s <= 0 or not np.isfinite(s):
            w = {int(np.argmax(c)): 1.0}
        else:
            w = {}
            for k in range(len(c)):
                if e[k] > 1e-10 * s:
                    w[k] = e[k] / s
    grad = np.zeros(n)
    for k, wi in w.items():
        s = k - (n - 1)
        # For lag s, c = sum_{i valid} h[i]*(1-h[i+s])
        # d/dc/dh[j] = sum over contributions
        # j appears as "first" factor when 0<=j<n and 0<=j+s<n
        # j appears as "second" factor when 0<=j<n and 0<=j-s<n
        if s >= 0:
            # first: j in [0, n-1-s], contributes (1-h[j+s])
            # second: j in [s, n-1], contributes -h[j-s]
            grad[:n-s] += wi * (1.0 - h[s:n])
            grad[s:] -= wi * h[:n-s]
        else:
            sp = -s  # positive
            # first: j in [sp, n-1], contributes (1-h[j-sp])
            # second: j in [0, n-1-sp], contributes -h[j+sp]
            grad[sp:] += wi * (1.0 - h[:n-sp])
            grad[:n-sp] -= wi * h[sp:n]
    return grad * (2.0 / n)

def _slsqp_smooth(h0, tau, maxiter):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5*n, 'jac': lambda x: np.ones(n)}]
    try:
        res = so.minimize(
            lambda x: _smooth_c(x, tau),
            h0, method='SLSQP',
            jac=lambda x: _smooth_grad(x, tau),
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return res.x if res.success or _exact_c(res.x) < _exact_c(h0) else None
    except:
        return None
```

```python
import numpy as np

def _repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(80):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            m = slack > 1e-13
            if not m.any():
                x[:] = target / n
                break
            x[m] += diff * slack[m] / max(slack[m].sum(), 1e-30)
        else:
            m = x > 1e-13
            if not m.any():
                break
            x[m] += diff * x[m] / max(x[m].sum(), 1e-30)
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _cv(h):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / h.size)

def _ec(h):
    return float(_cv(h).max())

def _sg(h, tau):
    """Smooth objective value."""
    c = _cv(h)
    cm = c.max()
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    return cm + tau * np.log(s) if s > 0 else cm

def _sgg(h, tau):
    """Smooth objective gradient."""
    c = _cv(h)
    n = h.size
    cm = c.max()
    if tau <= 0:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        z = (c - cm) / tau
        z -= z.max()
        e = np.exp(z)
        s = e.sum()
        w = e / s if s > 0 else (lambda x: (x.__setitem__(int(np.argmax(c)), 1.0), x)[1])(np.zeros(len(c)))
    grad = np.zeros(n)
    for k in range(len(c)):
        wi = w[k]
        if wi < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:

            grad[:n-s] -= wi * h[s:]
        else:
            sp = -s
            grad[:n-sp] += wi * (1.0 - h[sp:])
            grad[sp:] -= wi * h[:n-sp]
    return grad * (2.0 / n)

        else:
            sp = -s
            grad[:n-sp] += (1.0 - h[sp:])
            grad[sp:] -= h[:n-sp]

def _slsqp(fun, jac, h0, mi=200):

                       bounds=b, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-13})
```

```python
def _project(x, target):
    """Project x onto {h: 0<=h<=1, sum(h)=target}."""
    # Bisection on lambda
    # h_i = clamp(x_i - lambda, 0, 1)
    # sum(h_i) = target
    lo, hi = -2.0, 2.0

        hx = np.clip(x - mid, 0, 1)
        s = hx.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
```

For the coordinate-level search:

Actually, with the sum constraint, if we change h[i] from v_old to v_new, we need to compensate by changing other coordinates. The simplest compensation is to distribute the change (v_old - v_new) / (n-1) to all other coordinates. This gives a 1D problem that we can solve.

```python
def _coord_search(x, i, n_steps=20):
    """Try to improve coordinate i by varying it and compensating in others."""
    n = x.size
    target = 0.5 * n
    old_v = x[i]
    best_v = old_v
    best_c = _ec(x)
    
    # Try several values for h[i]
    for v in np.linspace(0, 1, n_steps):
        if abs(v - old_v) < 1e-6:
            continue
        x_new = x.copy()
        x_new[i] = v
        # Compensate: distribute (old_v - v) to other coordinates
        comp = (old_v - v)
        # Add comp to all other coordinates proportionally to their room
        others = np.ones(n)
        others[i] = 0
        # Need to distribute comp among others while keeping [0,1]
        # Simple: add comp/(n-1) to each other
        add = comp / (n - 1)
        x_new[others.astype(bool)] += add
        x_new = _repair(x_new)
        c = _ec(x_new)
        if c < best_c:
            best_c = c
            best_v = v
            best_x = x_new.copy()
    
    if best_v != old_v:
        return best_x, best_c
    return x, best_c
```
