def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corrs) * dx)

def _obj_grad(p, n_pts, t, lam):
    h = np.empty(n_pts)
    h[:-1] = p
    h[-1] = n_pts / 2.0 - np.sum(p)
    
    pen = 0.0
    pen_g = np.zeros(n_pts - 1)
    if h[-1] < 0.0:
        pen = lam * h[-1] * h[-1]
        pen_g[:] = -2.0 * lam * h[-1]
    elif h[-1] > 1.0:
        pen = lam * (h[-1] - 1.0) * (h[-1] - 1.0)
        pen_g[:] = 2.0 * lam * (h[-1] - 1.0)
    
    corrs = np.correlate(h, 1.0 - h, mode='full')
    sm = t * logsumexp(corrs / t)
    
    ex = np.exp(corrs / t)
    w = ex / ex.sum()
    
    gh = np.zeros(n_pts)
    for j in range(n_pts):
        g = 0.0
        for tau in range(-(n_pts - 1), n_pts):

            jt = j + tau
            if 0 <= jt <= n_pts - 1:
                g += (1.0 - h[jt]) * w[k]
            js = j - tau
            if 0 <= js <= n_pts - 1:
                g += (-h[js]) * w[k]
        gh[j] = g / t
    
    gp = gh[:-1] - gh[-1] + pen_g
    return sm + pen, gp

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corrs) * dx)

def _fix_sum(h, n):
    target = n / 2.0
    return np.clip(h + (target - np.sum(h)) / n, 0.0, 1.0)

def _grad_h(h, corrs, t):
    n = len(h)
    ex = np.exp(corrs / t)
    w = ex / ex.sum()
    first = np.zeros(n)
    second = np.zeros(n)
    for tau in range(-(n - 1), n):
        k = tau + n - 1
        wv = w[k]
        if wv < 1e-18:
            continue
        j_lo = max(0, -tau)
        j_hi = min(n - 1, n - 1 - tau)
        if j_lo <= j_hi:
            first[j_lo:j_hi + 1] += (1.0 - h[j_lo + tau:j_hi + 1 + tau]) * wv
        j2_lo = max(0, tau)
        j2_hi = min(n - 1, n - 1 + tau)
        if j2_lo <= j2_hi:
            second[j2_lo:j2_hi + 1] += (-h[j2_lo - tau:j2_hi + 1 - tau]) * wv
    return (first + second) / t

def _obj_grad(p, n_pts, t, lam):
    h = np.empty(n_pts)
    h[:-1] = p
    h[-1] = n_pts / 2.0 - np.sum(p)
    pen = 0.0
    pen_g = np.zeros(n_pts - 1)
    if h[-1] < 0.0:
        pen = lam * h[-1] * h[-1]
        pen_g[:] = -2.0 * lam * h[-1]
    elif h[-1] > 1.0:
        pen = lam * (h[-1] - 1.0) * (h[-1] - 1.0)
        pen_g[:] = 2.0 * lam * (h[-1] - 1.0)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    sm = t * logsumexp(corrs / t)
    gh = _grad_h(h, corrs, t)
    gp = gh[:-1] - gh[-1] + pen_g
    return sm + pen, gp

def _optimize(h_init, n_pts, temps, max_iter, lam, t0, budget):
    best = h_init.copy()
    best_val = compute_c5(best)
    for t in temps:
        if time.time() - t0 > budget - 15:
            break
        params = best[:-1].copy()
        try:
            res = minimize(_obj_grad, params, args=(n_pts, t, lam),
                           jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * (n_pts - 1),
                           options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-7})
        except Exception:
            continue
        h_new = np.empty(n_pts)
        h_new[:-1] = np.clip(res.x, 0.0, 1.0)
        h_new[-1] = n_pts / 2.0 - np.sum(h_new[:-1])
        h_new = _fix_sum(h_new, n_pts)
        c5 = compute_c5(h_new)
        if c5 < best - 1e-10:
            best = h_new.copy()
            best_val = c5
    return best, best_val

    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()

    best_h = h0.copy()
