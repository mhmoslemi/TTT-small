    """Compute soft-max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    exps = np.exp(shifted)
    logsum = m / t + np.log(np.sum(exps))
    obj = t * logsum * dx
    weights = exps / np.sum(exps)
    # Gradient
    grad = np.zeros(n)
    n_conv = len(conv)
    for idx in range(n_conv):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        # d/dh[i] of sum_j h[j](1-h[j+k])
        if k >= 0:

    dx = 2.0 / n

def project_to_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on shift."""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, t):
    """Compute soft-max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    vals = conv * dx
    m = np.max(vals)
    exps = np.exp((vals - m) / t)
    sum_exps = np.sum(exps)
    obj = (m + t * np.log(sum_exps))
    weights = exps / sum_exps
    
    # Gradient computation
    grad = np.zeros(n)
    n_conv = len(conv)
    for idx in range(n_conv):
        w = weights[idx]
        if w < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad += w * (1 - 2 * h)
            else:
                tmp = np.zeros(n)
                tmp[:n-k] += (1 - h[k:])
                tmp[k:] -= h[:n-k]
                grad += w * tmp
        else:
            s = -k
            tmp = np.zeros(n)
            tmp[s:] += (1 - h[:n-s])
            tmp[:n-s] -= h[s:]
            grad += w * tmp
    return obj, grad * dx

        if s > target:
            hi = mid
        else:

    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")

def coordinate_descent(h, t, n_sweeps=5, time_fn=None, min_time=30):
    n = len(h)
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)

                return o + 100.0 * (np.sum(h) - nc / 2.0) ** 2

            def grad_lbfgs(h):
                _, g = soft_max_obj_grad(h, t)
                return g + 200.0 * (np.sum(h) - nc / 2.0)

            try:
                res = minimize(obj_lbfgs, best_h, jac=grad_lbfgs, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * nc,
                               options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14})
                ht = project_feasible(res.x, nc)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 4: Random restarts with structured perturbation
    n_restarts = 30
    for r in range(n_restarts):
        if remaining() < 50:
            break
        # Structured perturbation: add smooth noise
        sigma = 0.1 * (1.0 - r / n_restarts) + 0.02
        noise = rng.normal(0, sigma, n)
        # Smooth it a bit
        for _ in range(3):
            noise = np.concatenate([noise[-1:], noise[:-1]]) * 0.5 + noise * 0.5
        h_r = project_feasible(best_h + noise, n)
        c5r = compute_c5(h_r)
        # Quick optimization from this start
        ht, c5t = pgd_soft(h_r, 0.01, 5000, 0.01, momentum=0.3, time_fn=remaining, min_time=40)
        ht, c5t2 = pgd_exact(ht, 5000, 0.005, time_fn=remaining, min_time=30)
        if c5t2 < c5t:
            c5t = c5t2
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Try different n by interpolation
    for n_try in [160, 256, 400]:
        if remaining() < 180:
            break
        xo = np.linspace(0, 2.0, n, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        ht, c5t = pgd_soft(hi, 0.02, 5000, 0.02, momentum=0.3, time_fn=remaining, min_time=60)
        ht, c5t = pgd_soft(ht, 0.005, 10000, 0.01, momentum=0.5, time_fn=remaining, min_time=40)
        ht, c5t = pgd_exact(ht, 10000, 0.003, time_fn=remaining, min_time=30)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Final aggressive polish on best solution
    if remaining() > 30:
        ht, c5t = pgd_soft(best_h, 0.0005, 30000, 0.005, momentum=0.5, time_fn=remaining, min_time=20)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    if remaining() > 15:
        ht, c5t = pgd_exact(best_h, 50000, 0.002, time_fn=remaining, min_time=10)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Final exact evaluation
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
