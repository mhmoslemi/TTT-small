```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def make_bivalent(h, n):
    """Convert h to bivalent with exactly n/2 ones."""
    h = (h >= 0.5).astype(float)
    count = h.sum()
    target = n // 2
    if count > target:
        idx = np.where(h == 1)[0]
        # Remove the ones with lowest "importance" (just random for now)
        perm = np.random.permutation(len(idx))
        h[idx[perm[:int(count - target)]]] = 0
    elif count < target:
        idx = np.where(h == 0)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:int(target - count)]]] = 1
    return h

def local_search_bivalent(h, max_rounds=500, seed=42):
    """Greedy local search with random restarts."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    
    for round_num in range(max_rounds):
        improved = False
        # Get current positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_gain = 0
        best_swap = None
        
        # Try all swaps (for n <= 400 this is feasible)
        for z in zeros:
            for o in ones:
                h[z] = 1
                h[o] = 0
                c5_new = compute_c5(h)
                h[z] = 0
                h[o] = 1
                if c5_new < best_c5 - best_gain - 1e-15:
                    best_gain = best_c5 - c5_new
                    best_swap = (z, o)
        
        if best_swap is not None:
            h[best_swap[0]] = 1
            h[best_swap[1]] = 0
            best_c5 = compute_c5(h)
            improved = True
        else:
            # Stuck - do random perturbation
            if len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(3):
                    z1 = rng.choice(zeros)
                    o1 = rng.choice(ones)
                    h[z1] = 1
                    h[o1] = 0
                # Re-evaluate
                best_c5 = compute_c5(h)
            break
    
    return h, best_c5
```

```python
def evaluate_swap_effect(h, z, o, C):
    """Compute new max after swapping h[z]:0->1, h[o]:1->0.
    C is the current correlation (positive lags only, length n)."""
    n = len(h)
    # Compute ΔC for all lags
    dC = np.zeros(n)
    # z as first element
    if z < n:
        k_max = n - 1 - z
        dC[:k_max+1] += (1 - h[z:z+k_max+1])
    # z as second element  
    if z >= 1:
        dC[1:z+1] -= h[z-z:0:-1]  # h[z-k] for k=1..z
    # Wait, this indexing is getting confused. Let me be more explicit.
    
    # For k = 1, 2, ..., min(z, n-1): z is second element in pair (z-k, z)
    # dC[k] -= h[z-k]
    # h[z-1], h[z-2], ..., h[z-z] = h[0]
    # So dC[1:k_z+1] -= h[z-1], h[z-2], ..., h[z-k_z]
    k_z = min(z, n-1)
    dC[1:k_z+1] -= h[z-1:z-1-k_z:-1]

    k_o_second = min(o, n-1)
    dC[1:k_o_second+1] += h[o-1:o-1-k_o_second:-1]
    
    new_C = C + dC
```

**Key insight for improvement**: The previous code's main weakness is that it uses a simple subgradient method which converges slowly. A better approach is:

1. **Frank-Wolfe (Conditional Gradient)**: At each iteration, solve a linear minimization over the constraint set, then take a step. The linear minimization over {h: 0 ≤ h ≤ 1, sum(h) = n/2} is easy (just pick the n/2 smallest coordinates and set them to 1, rest to 0).

2. **This naturally produces bivalent solutions!** So Frank-Wolfe might be ideal for this problem.

Let me implement Frank-Wolfe:

```
Algorithm: Frank-Wolfe for min f(h) s.t. h ∈ P = {h: 0≤h≤1, sum(h)=n/2}
1. h_0 = initial point in P
2. For t = 0, 1, 2, ...:
   a. Compute g = ∇f(h_t) (or subgradient)
   b. Solve linear problem: s_t = argmin_{s ∈ P} g^T s
      This gives a bivalent vertex of P
   c. α_t = 2/(t+2) (or line search)
   d. h_{t+1} = (1-α_t) h_t + α_t s_t
```

The linear minimization in step (b): minimize g^T s over s ∈ P.
Since P is the permutahedron (intersection of [0,1]^n with sum=n/2), the minimum is achieved at a vertex, which is a 0-1 vector with n/2 ones. Specifically, set s[i]=1 for the n/2 indices with the smallest g[i], and s[i]=0 otherwise.

This is O(n log n) per iteration (sorting). Each iteration also requires computing the subgradient, which is O(n²) (computing the full correlation).

So each iteration is O(n² + n log n) ≈ O(n²). For n=320, that's about 100K operations per iteration. We can do maybe 10K-100K iterations in 1100s.

But wait, for the max function, the subgradient is not unique. At a point where multiple lags achieve the maximum, the subgradient is the average of the corresponding individual gradients (weighted by the "active" set).

For Frank-Wolfe with a max function, we can use:

- The "smooth" approximation with soft-max (gives a true gradient)
- The "average over active set" subgradient

```python
def soft_max_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    # Compute all lags
    conv = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # Softmax weights
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / T)
    w /= w.sum()
    
    # Gradient: for each i, sum over all lags
    grad = np.zeros(n)
    for i in range(n):
        for j in range(2*n - 1):
            lag = j - (n - 1)  # lag from -(n-1) to n-1
            wi = w[j]
            # First term: (1-h[i+lag]) if 0 <= i+lag < n
            if 0 <= i + lag < n:
                grad[i] += wi * (1.0 - h[i + lag])
            # Second term: -h[i-lag] if 0 <= i-lag < n
            if 0 <= i - lag < n:
                grad[i] -= wi * h[i - lag]
    return grad * dx
```

```python
def frank_wolfe(h_init, T_schedule, max_iter_per_T=1000):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for T in T_schedule:
        for it in range(max_iter_per_T):
            # Compute gradient
            conv = np.correlate(h, 1.0 - h, mode="full")
            conv_shifted = conv - conv.max()
            w = np.exp(conv_shifted / T)
            w /= w.sum()
            
            # Build W matrix
            indices = np.arange(n)
            diffs = indices[None, :] - indices[:, None] + (n-1)
            W = np.where((diffs >= 0) & (diffs <= 2*n-2), w[diffs.clip(0, 2*n-2)], 0.0)
            
            dx = 2.0 / n
            g = (W @ (1.0 - h) - W.T @ h) * dx
            
            # Frank-Wolfe step: find vertex s
            # s[i]=1 for n/2 indices with smallest g[i]
            threshold_idx = np.argpartition(g, n//2)[:n//2]
            s = np.zeros(n)
            s[threshold_idx] = 1.0
            
            # Step size
            alpha = 2.0 / (it + 2)
            h_new = (1.0 - alpha) * h + alpha * s
            
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h = h_new
    
    return best_h, best_c5
```
