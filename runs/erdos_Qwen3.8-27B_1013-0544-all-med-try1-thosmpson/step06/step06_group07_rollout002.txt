    y = h.copy()
    beta = 0.5  # momentum parameter
    
    while time.time() - start < budget_s:
        t += 1
        # Evaluate at y
        c = np.correlate(y, 1.0 - y, mode="full") * dx
        C = c.max()
        
        if C < best_c:
            best_c = C
            best_h = h.copy()

        active_lags = idx - (n-1)
        
        # Subgradient
        g = active_subgrad(y, n, dx, active_lags)
        
        # Step size (Polak-Ribiere style)
        # Use 1/sqrt(t) decay
        alpha = 1.0 / (np.sqrt(t) * 0.1)
        # Clamp to reasonable range
        alpha = min(alpha, 5.0)
        
        # Nesterov step
        beta_new = (1.0 + np.sqrt(1.0 + 4.0*beta*beta)) / 2.0
        new_h = y - alpha * g
        new_h = project_box_sum(new_h, target)
        
        y = new_h + (beta_new/beta - 1.0) * (new_h - h)
        h = new_h
        beta = beta_new
    
    # Final check
    c = np.correlate(h, 1.0-h, mode="full") * dx
    if c.max() < best_c:
        best_c = c.max()
        best_h = h.copy()
    
    return best_h, best_c

import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def project_box_sum(y, target):
    n = len(y)
    lo = y.min() - target
    hi = y.max() + target
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def lag_grad(h, k, n, dx):
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:])
            g[k:] -= h[:m]
    else:
        kp = -k
        m = n - kp
        if m > 0:
            g[kp:] += (1.0 - h[:m])
            g[:m] -= h[kp:]
    return g * dx

def compute_c(h, n, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def subgrad_loop(h0, n, budget_s, topk=4, alpha0=0.5, seed=0):
    h = h0.copy().astype(float)
    dx = 2.0 / n
    target = 0.5 * n
    best_h = h.copy()
    best_c = float(compute_c(h, n, dx).max())
    start = time.time()
    t = 0
    alpha = alpha0
    rng = np.random.RandomState(seed)
    stuck = 0

    while time.time() - start < budget_s:
        t += 1
        c = compute_c(h, n, dx)
        C = c.max()
        if C < best_c - 1e-13:
            best_c = C
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 50:
            h = project_box_sum(h + rng.normal(0, 0.01, n), target)
            alpha = alpha0 * 0.5
            stuck = 0

        idx = np.argpartition(c, -topk)[-topk:]
        active = idx - (n - 1)
        g = np.zeros(n)
        for k in active:
            g += lag_grad(h, int(k), n, dx)
        g /= topk

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        # Backtracking line search
        accepted = False
        a = alpha
        for _ in range(20):
            y = project_box_sum(h - a * g, target)
            cy = compute_c(y, n, dx).max()
            if cy < C - 1e-14:
                h = y
                alpha = min(a * 1.05, 5.0)
                accepted = True
                break
            a *= 0.5
            if a < 1e-10:
                break
        if not accepted:
            alpha *= 0.7
            if alpha < 1e-8:
                break
    return best_h, best_c

def smooth_obj(h, n, tau):
    dx = 2.0 / n
    c = compute_c(h, n, dx)
    C = c.max()
    z = (c - C) / tau
    z -= z.max()
    return C + tau * np.log(np.exp(z).sum())

def smooth_grad(h, n, tau, topk=8):
    dx = 2.0 / n
    c = compute_c(h, n, dx)
    C = c.max()
    z = (c - C) / tau
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    if topk and topk < len(w):
        idx = np.argsort(w)[::-1][:topk]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w2 /= w2.sum()
        w = w2
    g = np.zeros(n)
    for i in np.nonzero(w > 1e-10)[0]:
        k = int(i) - (n - 1)
        g += w[i] * lag_grad(h, k, n, dx)
    return g

def slsqp_smooth(x0, n, tau, maxiter, topk=8):
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n,
             "jac": lambda x: np.ones(n)}]
    try:
        res = so.minimize(lambda x: smooth_obj(x, n, tau), x0,
                          jac=lambda x: smooth_grad(x, n, tau, topk),
                          method="SLSQP", bounds=bounds,
                          constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
        return res.x if res is not None else x0
    except Exception:
        return x0

def slsqp_exact(x0, n, maxiter, topk=3):
    dx = 2.0 / n

    def fun(x):
        return compute_c(x, n, dx).max()

    def jac(x):
        c = compute_c(x, n, dx)
        C = c.max()
        tol = max(1e-9, 1e-5 * C)
        act = np.where(c >= C - tol)[0]
        if len(act) > topk:
            act = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for i in act:
            g += lag_grad(x, int(i) - (n - 1), n, dx)
        g /= len(act)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n,
             "jac": lambda x: np.ones(n)}]
    try:
        res = so.minimize(fun, x0, jac=jac, method="SLSQP",
                          bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
        return res.x if res is not None else x0
    except Exception:
        return x0

def resample(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return h0.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x1, x0, h0)
    return project_box_sum(y, 0.5 * n_new)

def repair(h):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    return project_box_sum(x, 0.5 * len(x))
