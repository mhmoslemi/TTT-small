        if abs(d) < 1e-15:

def smooth_max_objective(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    # Numerically stable log-sum-exp
    shifted = temp * (corr - m)
    max_shifted = np.max(shifted)
    s = np.exp(shifted - max_shifted).sum()
    return (m + (max_shifted + np.log(s)) / temp) * dx

    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = temp * (corr - m)

    grad = np.zeros(n)

            dC[:n-Lp] = 1 - h[Lp:]
            dC[Lp:] -= h[:n-Lp]
            grad += w * dC * dx
    return grad

        h_curr = h_new

            h = project_feasible(h, target)
            if best_c < c5 - tol:
                c5 = best_c
                improved = True
        if not improved:

def coord_descent(h, n_levels=25, max_sweeps=20):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_v = h[j]
            for v in np.linspace(0.0, 1.0, n_levels):
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h = orig.copy()
            h[j] = best_v
            h = project_feasible(h, ts)
            if best_c < c5 - 1e-14:
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5


def pair_exchange(h, n_trials=50000):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(n_trials):
        i, j = np.random.randint(0, n, 2)

            continue

        h[i], h[j] = oj, oi
        h = project_feasible(h, ts)
        c = compute_c5(h)
        if c < c5 - 1e-14:
            c5 = c
        else:
            h[i], h[j] = oi, oj
            h = project_feasible(h, ts)
    return h, c5
