def compute_all_gains(h, overlap):
    """Compute the new max overlap for all possible single swaps."""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_o = len(ones)
    n_z = len(zeros)
    
    # Precompute remove effects: for each one position i, effect on overlap vector
    # remove_i[l] = contribution change at lag l when h[i] goes from 1 to 0
    # = h[i-l] - 1 (if 0 <= i-l < n) + h[i+l] (if 0 <= i+l < n)
    
    # lags range from -(n-1) to (n-1), stored as indices 0 to 2n-2
    # lag = idx - (n-1)
    
    remove_effects = np.zeros((n_o, 2*n-1))
    add_effects = np.zeros((n_z, 2*n-1))
    
    lags = np.arange(2*n-1) - (n-1)  # ranges from -(n-1) to (n-1)
    
    for idx, i in enumerate(ones):
        # h[i] goes from 1 to 0
        # Effect 1: terms where i is the "h[j+lag]" part, i.e., j+lag = i, j = i-lag
        # Before: h[i]*(1-h[i-lag]) = 1-h[i-lag], After: 0. Change: -(1-h[i-lag]) = h[i-lag]-1
        # Valid when 0 <= i-lag < n
        valid1 = (i - lags >= 0) & (i - lags < n)
        remove_effects[idx, valid1] = h[i - lags[valid1]] - 1
        
        # Effect 2: terms where i is the "(1-h[j])" part, i.e., j = i
        # Before: h[i+lag]*(1-h[i]) = 0, After: h[i+lag]*1 = h[i+lag]. Change: +h[i+lag]
        # Valid when 0 <= i+lag < n
        valid2 = (i + lags >= 0) & (i + lags < n)
        remove_effects[idx, valid2] += h[i + lags[valid2]]
    
    for idx, j in enumerate(zeros):
        # h[j] goes from 0 to 1
        # Effect 1: terms where j is the "h[j2+lag]" part, j2+lag = j, j2 = j-lag
        # Before: h[j]*(1-h[j-lag]) = 0, After: 1*(1-h[j-lag]) = 1-h[j-lag]. Change: +(1-h[j-lag])
        # Valid when 0 <= j-lag < n
        valid1 = (j - lags >= 0) & (j - lags < n)
        add_effects[idx, valid1] = 1 - h[j - lags[valid1]]
        
        # Effect 2: terms where j is the "(1-h[j2])" part, j2 = j
        # Before: h[j+lag]*(1-h[j]) = h[j+lag]*1 = h[j+lag], After: 0. Change: -h[j+lag]
        # Valid when 0 <= j+lag < n
        valid2 = (j + lags >= 0) & (j + lags < n)
        add_effects[idx, valid2] -= h[j + lags[valid2]]
    
    return ones, zeros, remove_effects, add_effects

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n_ones = int(np.sum(h))

            if remaining() < 40:
                break
            h_c, c5_c = vec_greedy(init, max_time=min(30.0, max(5.0, time_per_n)))
            if c5_c < best_c5:
                best_c5, best_h, best_n = c5_c, h_c.copy(), n

    # Phase 3: SA refinement on best
    if remaining() > 40:
        sa_time = min(remaining() * 0.4, 200.0)
        h_sa, c5_sa = sa_refine(best_h, max_time=sa_time, seed=seed + 500)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa

    # Phase 4: Final greedy polish
    if remaining() > 15:
        h_f, c5_f = vec_greedy(best_h, max_time=min(remaining() * 0.7, 120.0))
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Final verification
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
