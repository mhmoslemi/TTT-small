    h = h0.copy()
    for f in range(1, n_harmonics + 1):
        a = amp / f * rng.uniform(0.5, 1.5)
        h += a * np.sin(2 * np.pi * f * t / 2.0 + rng.uniform(0, 2 * np.pi))
    h += 0.01 * rng.normal(0, 1, n)
    return repair(h)


def jitter(h0, scale, seed):
    rng = np.random.RandomState(seed)
    return repair(h0 + rng.normal(0.0, scale, h0.size))


def optimize_single(x0, t_limit, seed):
    x = repair(x0.copy())
    best_x = x.copy()
    best_c = exact_c(x)
    start = time.time()
    n = x.size
    scale = 1.0 if n <= 80 else (0.7 if n <= 140 else 0.5)
    stages = [(0.05, 200), (0.02, 300), (0.008, 350), (0.003, 400), (0.001, 400), (0.0004, 350)]
    sstages = []
    for tau, mi in stages:
        el = time.time() - start
        rem = t_limit - el
        if rem < 5.0:
            break
        sstages.append((tau, max(30, int(mi * scale))))
    for tau, mi in sstages:
        if time.time() - start > t_limit * 0.75:
            break

        def f(x):
            v, _, _ = smooth_eval(x, tau)
            return v

        def j(x):
            _, _, w = smooth_eval(x, tau)
            return compute_gradient(x, w)

        res = _slsqp(f, j, x, mi)
        if res is not None:
            xr = repair(res.x)
            c = exact_c(xr)
            if c < best_c:
                best_c = c
                best_x = xr.copy()
            x = xr if c <= best_c + 5e-4 else best_x
    el = time.time() - start
    tl = t_limit - el
    if tl > 3.0:
        xn, cn = nesterov_projected(best_x, 0.001, 500, time.time() + min(tl * 0.6, 20.0))
        if cn < best_c:
            best_c = cn
            best_x = xn
    tl2 = t_limit - (time.time() - start)
    if tl2 > 2.0:
        for sg in [0.01, 0.005, 0.002, 0.001]:
            if time.time() - start > t_limit - 0.5:
                break
            cand = jitter(best_x, sg, seed + int(sg * 1e5))
            cc = exact_c(cand)
            if cc < best_c:
                best_c = cc
                best_x = cand
                xn, cn = nesterov_projected(best_x, 0.0005, 200, time.time() + 5.0)
                if cn < best_c:
                    best_c = cn
                    best_x = xn
    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)

    raw = globals().get("initial_h_values", None)
    parent = None if raw is None else np.asarray(raw, dtype=float)
    if parent is not None and parent.size < 10:
        parent = None

    best_x = None
    best_c = 1e9

    ns = []
    if parent is not None:
        ns.append(parent.size)
    for n in [60, 70, 80, 100, 120, 140, 160]:
        if n not in ns and 40 <= n <= 200:
            ns.append(n)
    ns = ns[:5]

    for ni, n in enumerate(ns):
        if time_left(deadline) < 60.0:
            break
        if parent is not None:
            base = resample(parent, n)
        else:
            base = repair(np.full(n, 0.5))
        cb = exact_c(base)
        if cb < best_c:
            best_c = cb
            best_x = base.copy()

        starts = [base, jitter(base, 0.03, sv + n), fourier_perturb(base, sv + n * 7)]
        if n <= 100:
            starts.append(fourier_perturb(base, sv + n * 13, 3, 0.12))

        per_start = min(45.0, max(10.0, time_left(deadline) * 0.18))
        for si, s in enumerate(starts[:3]):
            if time_left(deadline) < 15.0:
                break
            x, c = optimize_single(s, per_start, sv + 1000 * n + si * 31)
            if c < best_c:
                best_c = c
                best_x = x.copy()

    if best_x is None:
        best_x = repair(np.full(70, 0.5))
        best_c = exact_c(best_x)

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
