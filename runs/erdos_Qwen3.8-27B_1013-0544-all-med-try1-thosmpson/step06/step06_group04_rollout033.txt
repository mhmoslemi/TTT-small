import numpy as np
from scipy.optimize import minimize

def compute_overlaps(h):
    """Compute all overlap values for given h."""
    n = len(h)
    overlaps = np.zeros(2*n - 1)
    # For lag k = -(n-1) to n-1
    # overlap at position m in the array corresponds to lag k = m - (n-1)
    for k in range(-(n-1), n):
        if k >= 0:
            # sum h[i]*(1-h[i+k]) for i=0 to n-1-k
            idx = np.arange(0, n-k)
            overlaps[n-1+k] = np.sum(h[idx] * (1 - h[idx+k]))
        else:
            m = -k
            # sum h[i]*(1-h[i-m]) for i=m to n-1
            idx = np.arange(m, n)
            overlaps[n-1+k] = np.sum(h[idx] * (1 - h[idx-m]))
    return overlaps

def smooth_max(overlaps, t):
    """Smooth max approximation."""
    scaled = overlaps / t
    max_val = np.max(scaled)
    return t * (max_val + np.log(np.sum(np.exp(scaled - max_val))))

def smooth_max_grad(overlaps, t):
    """Gradient of smooth max with respect to overlaps."""
    scaled = overlaps / t
    max_val = np.max(scaled)
    exps = np.exp(scaled - max_val)
    weights = exps / np.sum(exps)
    return weights  # d(smooth_max)/d(overlaps) = weights

def full_gradient(h, t):
    """Full gradient of smooth_max(overlaps(h), t) with respect to h."""
    n = len(h)
    overlaps = compute_overlaps(h)
    weights = smooth_max_grad(overlaps, t)
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        m = n - 1 + k  # index in overlaps array
        w = weights[m]
        if w == 0:
            continue
        if k == 0:
            grad += w * (1 - 2*h)
        elif k > 0:
            # d(overlap(k))/dh[i] = (1-h[i+k]) for i in [0, n-1-k] - h[i-k] for i in [k, n-1]
            front = np.zeros(n)
            front[:n-k] = 1 - h[k:]  # (1-h[i+k]) for i=0 to n-1-k
            back = np.zeros(n)
            back[k:] = -h[:n-k]  # -h[i-k] for i=k to n-1
            grad += w * (front + back)
        else:
            m_val = -k
            # d(overlap(-m))/dh[i] = (1-h[i-m]) for i in [m, n-1] - h[i+m] for i in [0, n-1-m]
            front = np.zeros(n)
            front[:n-m_val] = -h[m_val:]  # -h[i+m] for i=0 to n-1-m
            back = np.zeros(n)
            back[m_val:] = 1 - h[:n-m_val]  # (1-h[i-m]) for i=m to n-1
            grad += w * (front + back)

            continue
        k = p - n + 1
        lo = 0 if k >= 0 else -k
        hi = n - 1 - k if k >= 0 else n - 1
        g[lo:hi + 1] += w * (1.0 - x[lo + k:hi + k + 1])
        g[lo + k:hi + k + 1] -= w * x[lo:hi + 1]
    g += 2.0 * penalty * (np.sum(x) - n * 0.5)
    return g

def _fun(x, n, t, penalty):
    return _obj_core(x, n, t, penalty)[0]

def _jac(x, n, t, penalty):
    return _grad_core(x, n, t, penalty)

def optimize_level(h, time_limit, max_iter=400):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n * 0.5)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    temps = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002]
    for t in temps:
        if time.time() - t0 > time_limit:
            break
        pen = 1e5 if t > 0.01 else 1e4
        try:
            res = minimize(_fun, h, jac=_jac, args=(n, t, pen),
                           method='L-BFGS-B',
                           bounds=[(1e-12, 1.0 - 1e-12)] * n,
                           options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                    'ftol': 1e-12, 'gtol': 1e-8, 'iprint': -1})
        except Exception:
            break
        h = np.clip(res.x, 0.0, 1.0)
        h = project_to_sum(h, n * 0.5)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def make_random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    h = project_to_sum(h, n * 0.5)
    return h

def make_block_h(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, min(int(rng.randint(1, max(2, n // 8))), n - pos))
        h[pos:pos + blen] = 1.0
        pos += 2 * blen + int(rng.randint(0, 3))
    h = project_to_sum(h, n * 0.5)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = project_to_sum(h_init, n0 * 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n0
    print("Initial: n=%d, C5=%.6f" % (n0, best_c5), flush=True)

    levels = []
    for f in [1, 2, 4, 8]:
        lv = n0 * f
        if lv <= 1120:
            levels.append(lv)

    time_fracs = [0.08, 0.18, 0.35, 0.35]
    idx = 0
    for target_n in levels:
        elapsed = time.time() - t_start
        remaining = budget_s * 0.92 - elapsed
        if remaining < 20:
            break
        tf = time_fracs[idx] if idx < len(time_fracs) else 0.1
        idx += 1
        time_alloc = min(remaining * tf, 400.0)
        if time_alloc < 10:
            continue

        if target_n == best_n:
            h = best_h.copy()
        else:
            factor = target_n // best_n
            if factor < 1:
                continue
            h = np.repeat(best_h, factor)
            h = project_to_sum(h, target_n * 0.5)

        mi = 300 if target_n <= 140 else 200
        h_opt, c5 = optimize_level(h, time_alloc, max_iter=mi)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = target_n
            print("Improved (main): n=%d, C5=%.6f" % (target_n, best_c5), flush=True)

        remaining = budget_s * 0.92 - (time.time() - t_start)
        n_restarts = min(4, max(1, int(remaining // 80)))
        for r in range(n_restarts):
            if time.time() - t_start > budget_s * 0.92:
                break
            if r % 2 == 0:
                h_rand = make_random_h(target_n, seed + r * 137 + 3)
            else:
                h_rand = make_block_h(target_n, seed + r * 71 + 11)
            ta = min(remaining * 0.25, 80.0)
            if ta < 5:
                break
            h_opt_r, c5_r = optimize_level(h_rand, ta, max_iter=150)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_opt_r.copy()
                best_n = target_n
                print("Improved (restart %d): n=%d, C5=%.6f" % (r, target_n, best_c5), flush=True)
            remaining = budget_s * 0.92 - (time.time() - t_start)
            if remaining < 20:
                break

    # Final local refinement at best resolution
    remaining = budget_s * 0.92 - (time.time() - t_start)
    if remaining > 20:
        h_final, c5_final = optimize_level(best_h, min(remaining, 60.0), max_iter=200)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            print("Final refinement: C5=%.6f" % best_c5, flush=True)

    best_c5 = compute_c5(best_h)
    print("Result: n=%d, C5=%.6f" % (len(best_h), best_c5), flush=True)
    return best_h, best_c5, len(best_h)
