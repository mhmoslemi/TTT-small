```python
def compute_c5(h):

def make_binary(h, n):

def normalize_continuous(h, n):
```

```python
    dx = 2.0 / n

    start = time.time()

    # corr[k] for k = -(n-1) ... (n-1)
    # After np.correlate(h, 1-h, 'full'), index i corresponds to lag i-(n-1)

        h_vec = np.clip(h_vec, 0, 1)
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full') * dx
        # Soft max for smoothness
        alpha = 50.0

    def objective_smooth(h_vec):

        alpha = 30.0
        return float(np.log(np.sum(np.exp(alpha * (corr - np.max(corr)))) / len(corr)) / alpha + np.max(corr))

        else:
            break
```

```python
def subgradient_c5(h):
    """Compute subgradient of C5 at h."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_val = np.max(corr)
    # Find all lags achieving max (within tolerance)
    mask = corr >= max_val - 1e-10
    n_max = np.sum(mask)
    
    grad = np.zeros(n)
    for m in np.where(mask)[0]:
        lag = m - (n - 1)
        for j in range(n):
            g = 0.0
            j_minus = j - lag
            if 0 <= j_minus < n:
                g += (1.0 - h[j_minus])
            j_plus = j + lag
            if 0 <= j_plus < n:
                g -= h[j_plus]
            grad[j] += g / n_max
    grad *= dx
    return grad, max_val
```

```python
    return grad
```

Actually, I can vectorize the gradient computation:

```python
def soft_max_gradient_v(h, alpha=50):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    weights /= np.sum(weights)
    
    # Vectorized: for each lag k, the gradient w.r.t. h[j] is:
    # (1-h[j-k]) for valid j-k - h[j+k] for valid j+k
    # This is: convolve(1-h, something) - convolve(h, something)
    # Actually, let's think of it as:
    # grad[j] = sum_k weights[k] * [(1-h[j-k]) * 1[j-k valid] - h[j+k] * 1[j+k valid]]
    # = sum_k weights[k] * (1-h[j-k]) - sum_k weights[k] * h[j+k]
    # = (1 - conv(h, w_rev))[j] - conv(h, w)[j]  (approximately)
    
    # More precisely:
    # First term: sum_k weights[k] * (1-h[j-k]) for valid j-k
    #   = sum_{j-k valid} weights[k] - sum_{j-k valid} weights[k]*h[j-k]
    #   = (sum of weights where k is in range[-j, n-1-j]) - (cross-corr of h and weights at offset j)
    
    # This is getting complex. Let me just use the loop version and hope it's fast enough.
    
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        lag = k_idx - (n - 1)
        # (1-h[j-lag]) for valid j-lag
        j_start = max(0, lag)
        j_end = min(n, n + lag)
        if j_start < j_end:
            grad[j_start:j_end] += w * (1.0 - h[j_start - lag:j_end - lag])
        # -h[j+lag] for valid j+lag
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w * h[j_start2 + lag:j_end2 + lag]
    grad *= dx
    return grad
```

This is O(n²) but vectorized within each lag, so it should be fast (about 2n-1 slicing operations). For n=112, that's 223 operations, each touching at most n elements. Total: ~223 * 112 ≈ 25000 operations, which is very fast in numpy.

    return float(np.max(corr)), corr

def make_binary(h, n, seed=0):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        idxs = rng.choice(ones, size=n_ones - target, replace=False)
        h[idxs] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        idxs = rng.choice(zeros, size=target - n_ones, replace=False)
        h[idxs] = 1.0
    return h

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr > 1e-10:
                h += room * (diff / tr)
        else:
            t = np.sum(h)
            if t > 1e-10:
                h *= (1.0 + diff / t)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / float(n_old)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    return np.interp(x_new, x_old, h_old)

def local_search_binary(h_init, max_time, seed=42):
    n = len(h_init)
    h = h_init.copy().astype(float)
    current_c5, corr = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    while time.time() - start < max_time:
        worst_idx = int(np.argmax(corr))
        k_w = worst_idx - (n - 1)
        cands = []
        if k_w >= 0:
            for i in range(n - k_w):
                cands.append((i, i + k_w))
        else:
            for i in range(-k_w, n):
                cands.append((i + k_w, i))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nr = min(25, len(ones) * len(zeros))
        for _ in range(nr):
            cands.append((int(rng.choice(ones)), int(rng.choice(zeros))))
        best_gain = 0.0
        best_a = best_b = -1
        for a, b in cands:
            if a == b or h[a] == h[b]:
                continue
            h[a], h[b] = h[b], h[a]
            new_c5, _ = compute_c5(h)
            h[a], h[b] = h[b], h[a]
            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain
                best_a, best_b = a, b
        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]
            current_c5, corr = compute_c5(h)
            stall = 0
        else:
            stall += 1
            if stall >= 40:
                np_p = int(rng.randint(3, 7))
                for _ in range(np_p):
                    a = int(rng.choice(ones))
                    b = int(rng.choice(zeros))
                    h[a], h[b] = h[b], h[a]
                current_c5, corr = compute_c5(h)
                stall = 0
    return h, current_c5

def subgradient_step(h, dx):
    """Compute one subgradient step for C5(h)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    mask = corr >= max_val - 1e-10
    idxs = np.where(mask)[0]
    n_max = len(idxs)
    grad = np.zeros(n)
    for ki in idxs:
        lag = ki - (n - 1)
        js, je = max(0, lag), min(n, n + lag)
        if js < je:
            grad[js:je] += (1.0 - h[js - lag:je - lag])
        js2, je2 = max(0, -lag), min(n, n - lag)
        if js2 < je2:
            grad[js2:je2] -= h[js2 + lag:je2 + lag]
    grad /= n_max
    return grad * dx

def project_feasible(y, n):
    """Project y onto {0 <= x <= 1, sum(x) = n/2}."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > n / 2.0:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_optimize(h_init, max_time, seed=42):
    n = len(h_init)
    dx = 2.0 / n
    h = normalize_continuous(h_init, n).astype(float)
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    alpha = 0.05
    decay = 0.9995
    while time.time() - start < max_time:
        grad = subgradient_step(h, dx)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)[0]
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        alpha *= decay
    return best_h, best_c5
