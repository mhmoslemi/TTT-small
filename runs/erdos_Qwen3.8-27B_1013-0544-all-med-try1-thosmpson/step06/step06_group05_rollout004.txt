def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (target / total)
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def local_search(h0, max_time, seed, use_double=False):
    n = len(h0)
    h = make_binary(h0, n)
    cur = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    while time.time() - start < max_time and stall < 5:
        best_gain = 1e-14
        best_move = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                nc = compute_c5(h)
                g = cur - nc
                if g > best_gain:
                    best_gain = g; best_move = (i, j)
                h[i] = 1.0; h[j] = 0.0
        if use_double and best_move is None and time.time() - start < max_time * 0.7:
            n_try = min(5000, max(100, len(ones) * len(zeros) // 2))
            for _ in range(n_try):
                i1 = int(rng.choice(ones)); i2 = int(rng.choice(ones))
                j1 = int(rng.choice(zeros)); j2 = int(rng.choice(zeros))
                if i1 == i2 or j1 == j2:
                    continue
                h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                nc = compute_c5(h)
                g = cur - nc
                if g > best_gain:
                    best_gain = g; best_move = (i1, i2, j1, j2)
                h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0

                h[a] = 0.0; h[b] = 0.0; h[c] = 1.0; h[d] = 1.0
            cur = compute_c5(h)
            stall = 0
        else:
            stall += 1
    return h, cur

    start = time.time()
    best = make_binary(h0, n).copy()
    best_c5 = compute_c5(best)
    t = 0
    while time.time() - start < max_time:
        t += 1
        h_p = best.copy()
        n_flip = max(2, n // 10 + rng.randint(-2, 6))
        n_flip = min(n_flip, n - 2)
        pos = rng.choice(n, n_flip, replace=False)

def ga_search(h_init, n, max_time, seed, pop=32):
    rng = np.random.RandomState(seed)
    start = time.time()

    pop_l = [make_binary(h_init, n)]
