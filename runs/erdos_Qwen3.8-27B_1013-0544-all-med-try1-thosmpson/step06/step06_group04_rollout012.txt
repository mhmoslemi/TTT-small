import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective_factory(n, beta):
    def obj(x):
        x = project_feasible(x, n)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        corr_shifted = corr - m
        return m + np.log(np.sum(np.exp(beta * corr_shifted))) / beta
    return obj

            obj = smooth_objective_factory(n, beta)

                h_cand = project_feasible(res.x, n)

                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
                    h_init = h_cand.copy()
            except Exception:
                pass
