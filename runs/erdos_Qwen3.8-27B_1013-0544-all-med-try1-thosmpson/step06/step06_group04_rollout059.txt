import numpy as np
from scipy.optimize import minimize
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def upsample(h, factor=2):
    return np.repeat(h, factor)

def downsample(h, factor=2):
    n = len(h)
    m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1)

def project(h, target_sum):
    # Project h onto {0 <= h <= 1, sum(h) = target_sum}
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Adjust: if diff > 0, increase elements with most room
    # If diff < 0, decrease elements with most margin
    # Simple approach: add/subtract uniformly, then re-clip
    for _ in range(100):
        if diff > 0:
            room = 1 - h
            mask = room > 1e-10
            if not np.any(mask): break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            h[h > 0] += diff / np.sum(h > 0)
            h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12: break
    return h

def compute_gradient(h, n_active=5):
    """Compute subgradient of C5 = max_k corr[k]"""
    n = len(h)
    dx = 2.0 / n
    comp = 1 - h
    corr = np.correlate(h, comp, mode='full') * dx
    
    # Find top n_active shifts
    top_k = np.argsort(corr)[-n_active:]
    
    # For each active shift, compute gradient contribution
    # corr[k] = dx * sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
    # Let's verify: for k in 0..2n-2, the lag is k-(n-1)
    # corr[k] = dx * sum_i h[i] * comp[i + k - (n-1)]
    # = dx * sum_i h[i] * (1 - h[i + k - (n-1)])
    
    grad = np.zeros(n)
    for k in top_k:
        lag = k - (n - 1)  # this is the shift
        # corr[k] = dx * sum_i h[i] * (1 - h[i + lag]) for valid i
        # Valid i: 0 <= i < n and 0 <= i + lag < n
        # So i ranges from max(0, -lag) to min(n-1, n-1-lag)
        i_min = max(0, -lag)
        i_max = min(n-1, n-1-lag)
        if i_min > i_max:

        i_vals = np.arange(i_min, i_max+1)
