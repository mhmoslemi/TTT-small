    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()

                mask = h > 0.0
            if mask.sum() == 0:
                break

def smooth_max_obj(x, temp, penalty):

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

            else:

            if mask.sum() == 0:
                break

def smooth_obj(x, temp, penalty):
    n = len(x)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = temp * (corr - m)
    smax = m + np.log(np.exp(shifted).sum()) / temp
    return smax * dx + penalty * (x.sum() - target) ** 2

    return project_feasible(res.x, n / 2.0)

        improved = False

            best_c = c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_binary(bits):
    n = len(bits)
    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        if abs(d) < 1e-13:
            break

def smooth_grad(x, temp, penalty):

    a = 1.0 - x

    grad = np.zeros(n)

        t1 = np.dot(p[n-1-j:2*n-1-j], a[:n-j])
