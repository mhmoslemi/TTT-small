def normalize_continuous(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            total = np.sum(h)
            if total > 1e-12:

            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)

            adj = min(abs(diff), min(1.0 - h[idx], h[idx]) if diff < 0 else 1.0 - h[idx])
            h[idx] += np.sign(diff) * adj
