def project_sum(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Bisection for water-filling

        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def upsample_binary(h0, n_new):
    n0 = len(h0)
    if n0 == n_new:
        return h0.copy()
    x0 = np.arange(n0) / n0 * 2.0
    x1 = np.arange(n_new) / n_new * 2.0
    h_cont = np.interp(x1, x0, h0)
    h = (h_cont >= 0.5).astype(float)
    target = n_new // 2
    current = int(h.sum())
    if current > target:
        idx = np.where(h == 1)[0]
        order = np.argsort(h_cont[idx])
        for k in range(current - target):
            h[idx[order[k]]] = 0
    elif current < target:
        idx = np.where(h == 0)[0]
        order = np.argsort(-h_cont[idx])
        for k in range(target - current):
            h[idx[order[k]]] = 1
    return h

def local_search_vectorized(h, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    current_c5 = evaluate_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        # Precompute Delta
        Delta = np.zeros((n, 2*n - 1))
        d_vals = np.arange(2*n - 1) - (n - 1)
        for i in range(n):
            valid_a = (i - d_vals >= 0) & (i - d_vals < n)
            Delta[i, valid_a] -= (1.0 - h[i - d_vals[valid_a]])
            valid_b = (i + d_vals >= 0) & (i + d_vals < n)
            Delta[i, valid_b] += h[i + d_vals[valid_b]]
        
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Generate all (or subset of) swaps
        n_swaps = len(ones) * len(zeros)
        if n_swaps <= 100000:
            i_list, j_list = np.meshgrid(ones, zeros, indexing='ij')
            i_flat = i_list.ravel().astype(int)
            j_flat = j_list.ravel().astype(int)
        else:
            n_try = min(100000, n_swaps)
            i_flat = rng.choice(ones, n_try, replace=False) if n_try <= n_swaps else rng.choice(ones, n_try, replace=True)
            j_flat = rng.choice(zeros, n_try, replace=True)
        
        # Batch evaluation
        best_c5 = current_c5
        best_i, best_j = -1, -1
        batch_size = 5000
        n_total = len(i_flat)
        
        for start_b in range(0, n_total, batch_size):
            end_b = min(start_b + batch_size, n_total)
            ib = i_flat[start_b:end_b]
            jb = j_flat[start_b:end_b]
            # new_corr = corr_full + Delta[ib] - Delta[jb]
            c5_batch = np.max(corr_full[None, :] + Delta[ib] - Delta[jb], axis=1) * dx
            best_in_batch = np.argmin(c5_batch)
            c5_val = c5_batch[best_in_batch]
            if c5_val < best_c5 - 1e-12:
                best_c5 = c5_val
                best_i = ib[best_in_batch]
                best_j = jb[best_in_batch]
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            current_c5 = best_c5
        else:
            # Local optimum reached, try random perturbation
            improved = False
            for _ in range(200):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0
                h[j] = 1.0
                c5 = evaluate_c5(h)
                if c5 < current_c5 - 1e-12:
                    current_c5 = c5
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if not improved:
                break
    
    return h, current_c5

    dx = 2.0 / n

    return float(np.max(c))

def upsample_binary(h0, n_new):
    n0 = len(h0)
    if n0 == n_new:
        return h0.copy()
    x0 = np.arange(n0, dtype=float) / n0 * 2.0
    x1 = np.arange(n_new, dtype=float) / n_new * 2.0
    h_cont = np.interp(x1, x0, h0)
    h = (h_cont >= 0.5).astype(float)
    target = n_new // 2
    current = int(h.sum())
    if current > target:
        idx = np.where(h == 1)[0]
        order = np.argsort(h_cont[idx])
        for k in range(current - target):
            h[idx[order[k]]] = 0
    elif current < target:
        idx = np.where(h == 0)[0]
        order = np.argsort(-h_cont[idx])
        for k in range(target - current):
            h[idx[order[k]]] = 1
    return h

        n_swaps = len(ones) * len(zeros)
        if n_swaps <= 100000:
            i_list, j_list = np.meshgrid(ones, zeros, indexing='ij')
            i_flat = i_list.ravel()
            j_flat = j_list.ravel()
        else:
            n_try = 100000
            i_flat = rng.choice(ones, n_try, replace=True)
            j_flat = rng.choice(zeros, n_try, replace=True)

        best_c5 = current_c5
        best_i, best_j = -1, -1
        batch_size = 5000
        n_total = len(i_flat)

        for bs in range(0, n_total, batch_size):
            be = min(bs + batch_size, n_total)
            ib = i_flat[bs:be]
            jb = j_flat[bs:be]
            c5_batch = np.max(corr_full[None, :] + Delta[ib] - Delta[jb], axis=1) * dx
            bi = int(np.argmin(c5_batch))
            if c5_batch[bi] < best_c5 - 1e-12:
                best_c5 = float(c5_batch[bi])
                best_i = int(ib[bi])
                best_j = int(jb[bi])

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            current_c5 = best_c5
            stall = 0
        else:
            stall += 1
            found = False
            n_attempts = 200 if stall < 3 else 500
            for _ in range(n_attempts):
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                h[i] = 0.0
                h[j] = 1.0
                c5 = evaluate_c5(h)
                if c5 < current_c5 - 1e-12:
                    current_c5 = c5
                    stall = 0
                    found = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if not found and stall >= max_stall:
                break
            elif not found:
                break

    return h, current_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.asarray(initial_h_values, dtype=float)
    # Ensure valid
    h0 = np.clip(h0, 0, 1)
    n0 = len(h0)
    target_sum = n0 // 2
    if int(h0.sum()) != target_sum:
        h0 = (h0 >= 0.5).astype(float)
        cur = int(h0.sum())
        if cur > target_sum:
            idx = np.where(h0 == 1)[0]
            h0[idx[cur - target_sum:]] = 0
        elif cur < target_sum:
            idx = np.where(h0 == 0)[0]
            h0[idx[:target_sum - cur]] = 1

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    resolutions = sorted(set([n0, 140, 200, 280, 350, 490, 700]))

    for n in resolutions:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 25:
            break

        time_per = min(100, remaining * 0.4)
        if time_per < 8:
            time_per = 8

        # Start 1: upsampled from current best
        h_up = upsample_binary(best_h, n)
        h1, c5_1 = local_search(h_up, time_limit=time_per * 0.55, seed=seed + n)

        # Start 2: random
        remaining2 = budget_s - (time.time() - start_time)
        if remaining2 > 25:
            h_r = random_binary(n, seed=seed + 5000 + n)
            h2, c5_2 = local_search(h_r, time_limit=time_per * 0.45, seed=seed + 6000 + n)
            if c5_2 < c5_1:
                h1, c5_1 = h2, c5_2

        # Start 3: complement of upsampled
        remaining3 = budget_s - (time.time() - start_time)
        if remaining3 > 25:
            h_comp = 1.0 - upsample_binary(best_h, n)
            h3, c5_3 = local_search(h_comp, time_limit=min(30, time_per * 0.35), seed=seed + 7000 + n)
            if c5_3 < c5_1:
                h1, c5_1 = h3, c5_3

        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
