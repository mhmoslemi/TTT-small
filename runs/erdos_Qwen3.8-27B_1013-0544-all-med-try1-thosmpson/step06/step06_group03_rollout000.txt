    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    total = np.sum(exps)
    if total < 1e-300:
        idx = np.argmax(conv)
        return _conv_grad_single(h, idx - (n - 1)) * dx
    top_idx = np.argsort(exps)[-K:]
    grad = np.zeros(n)
    for idx in top_idx:
        w = exps[idx] / total
        if w < 1e-12:
            continue
        grad += w * _conv_grad_single(h, idx - (n - 1))
    return grad * dx

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h_init, max_iter=20000, lr=0.008, t_start=0.05, t_end=0.001,
                   t_decay=0.98, K=50, time_limit=None, t0=None):
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    t_mom = 1.0
    temp = t_start
    best_c5 = compute_c5(x)
    best_x = x.copy()
    for it in range(max_iter):
        if t0 is not None and time.time() - t0 > time_limit:
            break
        grad = soft_max_grad_topk(y, temp, K)
        y_new = project_feasible(y - lr * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        if it > 0:
            y = project_feasible(y_new + (t_mom - 1.0) / t_new * (y_new - x), n)
        else:
            y = y_new
        x = y_new
        t_mom = t_new
        temp = max(temp * t_decay, t_end)
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x.copy()
    return best_x, best_c5

def pgd_optimize(h_init, max_iter=50000, lr_init=0.003, use_smooth=False,
                 smooth_t=0.01, K=50, time_limit=None, t0=None):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if t0 is not None and time.time() - t0 > time_limit:
            break
        if use_smooth:
            grad = soft_max_grad_topk(h, smooth_t, K)
            c5 = compute_c5(h)
        else:
            conv = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(conv) * dx
            idx = np.argmax(conv)
            grad = _conv_grad_single(h, idx - (n - 1)) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-15:
            lr *= 0.5
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.03, 0.05)
    return best_h, best_c5

def lbfgs_refine(h, n, t_val, time_limit=None, t0=None):
    def obj(h2):
        return soft_max_obj(h2, t_val) + 500.0 * (np.sum(h2) - n / 2.0) ** 2
    def grad(h2):
        return soft_max_grad_topk(h2, t_val, K=80) + 1000.0 * (np.sum(h2) - n / 2.0)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def structured_perturb(h, rng, n, mode):
    h2 = h.copy()
    if mode == 0:
        h2 += rng.normal(0, 0.04, n)
    elif mode == 1:
        bs = rng.randint(max(2, n // 10), max(3, n // 5))
        i = rng.randint(0, n - 2 * bs)
        j = rng.randint(0, n - 2 * bs)
        if j < i:
            i, j = j, i
        tmp = h2[i:i+bs].copy()
        h2[i:i+bs] = h2[j:j+bs]
        h2[j:j+bs] = tmp
    elif mode == 2:
        freq = rng.randint(2, 10)
        phase = rng.uniform(0, 2 * np.pi)
        h2 += 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    else:
        i = rng.randint(0, n // 3)
        j = rng.randint(2 * n // 3, n)
        h2[i:j] = 1.0 - h2[i:j]
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(320, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: FISTA with decreasing temperature
    if rem() > 120:
        ht, c5t = fista_optimize(best_h, max_iter=20000, lr=0.006,
                                  t_start=0.06, t_end=0.001, t_decay=0.985,
                                  K=60, time_limit=rem() - 100, t0=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: L-BFGS refinements
    if rem() > 60:
        for tv in [0.02, 0.005, 0.002]:
            if rem() < 30:
                break
            ht = lbfgs_refine(best_h, n0, tv, time_limit=25, t0=t0)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 3: Fine PGD with exact subgradient
    if rem() > 50:
        ht, c5t = pgd_optimize(best_h, max_iter=40000, lr_init=0.001,
                               use_smooth=False, time_limit=rem() - 40, t0=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Structured perturbations + local optimization
    for rnd in range(30):
        if rem() < 40:
            break
        hp = structured_perturb(best_h, rng, n0, rnd % 4)
        ht, c5t = fista_optimize(hp, max_iter=6000, lr=0.004,
                                  t_start=0.03, t_end=0.001, t_decay=0.975,
                                  K=40, time_limit=min(18, rem() - 25), t0=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
            if rem() > 30:
                ht, c5t2 = pgd_optimize(best_h, max_iter=10000, lr_init=0.0005,
                                        use_smooth=False,
                                        time_limit=min(12, rem() - 20), t0=t0)
                if c5t2 < best_c5:
                    best_h, best_c5 = ht, c5t2

    # Phase 5: Try different resolutions
    for n_try in [256, 384, 400, 500]:
        if rem() < 100:
            break
        nc = len(best_h)
        if n_try == nc:
            continue
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        ht, c5t = fista_optimize(hi, max_iter=8000, lr=0.005,
                                  t_start=0.03, t_end=0.001, t_decay=0.98,
                                  K=50, time_limit=min(35, rem() - 50), t0=t0)
        if c5t < best_c5:
            ht, c5t2 = pgd_optimize(ht, max_iter=10000, lr_init=0.001,
                                     use_smooth=False,
                                     time_limit=min(15, rem() - 40), t0=t0)
            if c5t2 < c5t:
                c5t = c5t2
            best_h, best_c5 = ht, c5t

    # Final polish
    if rem() > 25:
        ht, c5t = pgd_optimize(best_h, max_iter=60000, lr_init=0.0003,
                               use_smooth=False, time_limit=rem() - 15, t0=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    if rem() > 15:
        ht = lbfgs_refine(best_h, len(best_h), 0.001, time_limit=rem() - 10, t0=t0)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
