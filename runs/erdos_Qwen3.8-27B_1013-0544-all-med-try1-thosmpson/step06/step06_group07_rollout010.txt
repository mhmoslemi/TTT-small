import warnings

warnings.simplefilter("ignore")

def project_onto_simplex(y, target):
    """Project y onto {h: 0<=h<=1, sum(h)=target} using bisection."""
    n = y.size
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(y - lam, 0.0, 1.0)

def compute_c(h):
    """Compute all correlation values c_k."""
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(compute_c(h)))

def grad_c_k(h, k):
    """Gradient of c_k with respect to h."""
    n = h.size
    dx = 2.0 / n
    # c_k = dx * sum_i h[i] * (1 - h[(i+k) mod n])
    # dc_k/dh_j = dx * [(1 - h[(j+k) mod n]) - h[(j-k) mod n]]
    idx_fwd = (np.arange(n) + k) % n
    idx_bwd = (np.arange(n) - k) % n
    return dx * ((1.0 - h[idx_fwd]) - h[idx_bwd])

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    x = project_onto_simplex(x, target)
    # Final fix for numerical issues
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)[::-1]
        else:
            order = np.argsort(x)[::-1]
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
            else:
                add = max(diff, -x[idx])
            x[idx] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    return x

def resample(h0, n_new):

        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x1, x0, h0)
    return repair(y)

def subgradient_step(x, active_weights, step_size):
    """Take a projected subgradient step."""
    n = x.size
    dx = 2.0 / n
    grad = np.zeros(n)
    for k, w in zip(*active_weights):
        grad += w * grad_c_k(x, k)
    y = x - step_size * grad
    return project_onto_simplex(y, 0.5 * n)

            if s < 1e-300:

import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

def _project(y, target):
    n = y.size
    if n == 0:
        return y.copy()
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    n = x.size
    return _project(x, 0.5 * n)

def _compute_c(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _exact_c(h):
    return float(np.max(_compute_c(h)))

def _grad_c_k(h, k):
    n = h.size
    dx = 2.0 / n
    k = int(k) % n
    f = (np.arange(n) + k) % n
    b = (np.arange(n) - k) % n
    return dx * ((1.0 - h[f]) - h[b])

def _resample(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return _repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return _repair(np.interp(x1, x0, h0))

def _con_fun(x):
    return np.array([x.sum() - 0.5 * x.size])

def _con_jac(x):
    return np.ones((1, x.size))

def _proj_subgrad(x0, tlimit, seed, n_act=4):
    rng = np.random.RandomState(seed)
    x = _repair(x0)
    best_x, best_c = x.copy(), _exact_c(x)
    deadline = time.time() + tlimit
    step, no_imp = 0.4, 0

    while time.time() < deadline:
        c = _compute_c(x)
        C = c.max()
        if C < best_c - 1e-14:
            best_c, best_x, no_imp = C, x.copy(), 0
        else:
            no_imp += 1

        tol = max(1e-9, 0.012 * C)
        act = np.where(c >= C - tol)[0]
        if len(act) > n_act:
            act = np.argsort(c)[::-1][:n_act]

        if len(act) == 1:
            w = 1.0
        else:
            z = (c[act] - C) / (0.004 * C + 1e-10)
            w_arr = np.exp(z)
            w_sum = w_arr.sum()
            if w_sum < 1e-300:
                g = _grad_c_k(x, act[0])
            else:
                g = sum((w_arr[j] / w_sum) * _grad_c_k(x, act[j]) for j in range(len(act)))

        if np.linalg.norm(g) < 1e-12:
            break

        acc = False
        for fct in [1.0, 0.6, 0.36, 0.216, 0.13, 0.08]:
            y = _project(x - step * fct * g, 0.5 * x.size)
            if _exact_c(y) < C - 1e-13:
                x = y
                acc = True
                step = min(step * 1.06, 2.5)
                break
        if not acc:
            step *= 0.6
            if step < 1e-8:
                found = False
                for s_ in [0.01, 0.03, 0.06, 0.1]:
                    y = _project(x + rng.normal(0.0, s_, x.size), 0.5 * x.size)
                    if _exact_c(y) < best_c:

    return best_x, best_c

def _slsqp_smooth(x0, tlimit, seed):
    x = _repair(x0)
    n = x.size
    best_x, best_c = x.copy(), _exact_c(x)
    t0 = time.time()
    stages = [(0.04, 120), (0.01, 180), (0.003, 180), (0.0008, 140)]

    for tau, mi in stages:
        if time.time() - t0 > tlimit * 0.88:
            break

        def _sm_fun(xx):
            cc = _compute_c(xx)
            Cv = np.max(cc)
            z = (cc - Cv) / tau
            z -= np.max(z)
            e = np.exp(z)
            s = e.sum()
            if s > 1e-300:
                return Cv + tau * np.log(s)
            return Cv

        def _sm_jac(xx):
            cc = _compute_c(xx)
            Cv = np.max(cc)
            z = (cc - Cv) / tau
            z -= np.max(z)
            e = np.exp(z)
            s = e.sum()
            if s > 1e-300:
                w = e / s
            else:
                w = np.zeros(cc.size)
                w[int(np.argmax(cc))] = 1.0
            g = np.zeros(n)
            for k in range(cc.size):
                if w[k] > 1e-10:
                    idx_f = (np.arange(n) + int(k)) % n
                    idx_b = (np.arange(n) - int(k)) % n
                    g += w[k] * (2.0 / n) * ((1.0 - xx[idx_f]) - xx[idx_b])
            return g

        try:
            res = so.minimize(_sm_fun, x, method='SLSQP', jac=_sm_jac,
                              bounds=[(0.0, 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                              options={'maxiter': mi, 'ftol': 1e-12})
        except Exception:
            break
        if res is not None:
            xr = _repair(res.x)
            cr = _exact_c(xr)
            if cr < best_c:
                best_c, best_x = cr, xr.copy()
            x = xr if cr <= best_c + 2e-4 else best_x
    return best_x, best_c

def _sim_anneal(x0, tlimit, seed):

    x = _repair(x0)

    cool = 0.996
    n = x.size

        delta = cy - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            x = y
            if cy < best_c:
                best_c, best_x = cy, y.copy()
        T *= cool
    return best_x, best_c

def _fine_polish(x0, tlimit, seed):
    x = _repair(x0)
    best_x, best_c = x.copy(), _exact_c(x)
    deadline = time.time() + tlimit
    step = 0.15
    rng = np.random.RandomState(seed)

        g = sum(_grad_c_k(x, int(k)) for k in act) / max(1, len(act))
        if np.linalg.norm(g) < 1e-12:

        acc = False
        for fct in [1.0, 0.5, 0.25, 0.125, 0.0625]:

                else:
                    break
