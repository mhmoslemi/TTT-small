    e1 = min(N, N + n_lag)

    c_shifted = corr - corr.max()

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def get_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def grad_lag(h, n_lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, n_lag)
    e1 = min(N, N + n_lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - n_lag:e1 - n_lag]
    s2 = max(0, -n_lag)
    e2 = min(N, N - n_lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + n_lag:e2 + n_lag]
    return g

def soft_gradient(h, corr, N, alpha):
    c_shifted = corr - corr.max()
    w = np.exp(alpha * c_shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_lag(h, k_star - (N - 1))
    w = w / wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-5:
            continue
        g += w[j] * grad_lag(h, j - (N - 1))
    return g

def backtracking_line_search(h, g, N, val, corr, max_trials=20):
    """Find step size that decreases the objective."""
    lr = 0.1
    for _ in range(max_trials):
        h_new = project_feasible(h - lr * g, N)
        corr_new = get_corr(h_new)
        val_new = corr_new.max()
        if val_new < val - 1e-14:
            return h_new, corr_new, val_new, lr
        lr *= 0.5
    return h, corr, val, lr

def optimize_phase1(h, N, rng, t_end, try_update, max_iter=50000):
    """Main gradient optimization with adaptive alpha and backtracking."""
    corr = get_corr(h)
    val = corr.max()
    try_update(h, val)
    
    best_h = h.copy()
    best_val = val
    alpha = 10.0
    lr = 0.05
    stagnation = 0
    it = 0
    
    while it < max_iter and time.time() < t_end:
        it += 1
        g = soft_gradient(h, corr, N, alpha)
        h_new, corr_new, val_new, lr = backtracking_line_search(h, g, N, val, corr)
        
        if val_new < best_val - 1e-13:
            best_val = val_new
            best_h = h_new.copy()
            try_update(h_new, val_new)
            h = h_new
            corr = corr_new
            val = val_new
            stagnation = 0
            # Slowly increase alpha
            alpha = min(alpha * 1.005, 200.0)
            lr = min(lr * 1.1, 0.2)
        else:
            stagnation += 1
            if stagnation % 20 == 0:
                # Perturb and reset
                scale = 0.02 + 0.03 * min(stagnation / 100.0, 2.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                corr = get_corr(h)
                val = corr.max()
                lr = 0.05
                alpha = max(alpha * 0.5, 5.0)
                stagnation = 0
            elif stagnation > 100:

                h = project_feasible(h + rng.randn(N) * 0.1, N)
                corr = get_corr(h)
                val = corr.max()
                lr = 0.05
                alpha = 10.0
                stagnation = 0
    
    return best_h, best_val

def coordinate_descent(h, N, rng, t_end, try_update, max_sweeps=100):
    """Coordinate descent: optimize each coordinate one at a time."""
    corr = get_corr(h)
    val = corr.max()
    try_update(h, val)
    
    best_h = h.copy()
    best_val = val
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(N):
            if time.time() > t_end:
                break
            # Try a few values for h[i]
            old_val = h[i]
            candidates = [0.0, 1.0, old_val - 0.1, old_val + 0.1, old_val - 0.05, old_val + 0.05]
            # Also try the projected optimal
            candidates = [c for c in candidates if 0 <= c <= 1]
            best_c = old_val
            best_v = val
            h_saved = h.copy()
            for c in candidates:
                h[i] = c
                # Adjust sum constraint: redistribute the difference
                diff = (c - old_val)
                if abs(diff) < 1e-10:
                    continue
                # Project to maintain sum constraint
                h[i] = c
                h_proj = project_feasible(h, N)
                corr_c = get_corr(h_proj)
                v_c = corr_c.max()
                if v_c < best_v:
                    best_v = v_c
                    best_c = c
                    best_h_proj = h_proj.copy()
                h = h_saved.copy()
            if best_c != old_val and best_v < val - 1e-12:
                h = best_h_proj
                val = best_v
                corr = get_corr(h)
                improved = True
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                    try_update(h, val)
        if not improved:
            break
    
    return best_h, best_val

def interpolate_to_n(h_src, n_dst):
    x0 = np.linspace(0, 1, len(h_src), endpoint=False)
    x1 = np.linspace(0, 1, n_dst, endpoint=False)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def binary_local_search(h, N, rng, t_end, try_update, max_iters=5000):
    """Local search on binary sequences: swap a 1 with a 0."""
    # First round to binary
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly N/2 ones
    num_ones = int(h_bin.sum())
    target = N // 2
    if num_ones > target:
        # Remove excess ones
        ones_idx = np.where(h_bin == 1)[0]
        remove = rng.choice(ones_idx, num_ones - target, replace=False)
        h_bin[remove] = 0
    elif num_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zeros_idx, target - num_ones, replace=False)
        h_bin[add] = 1
    
    corr = get_corr(h_bin)
    val = corr.max()
    try_update(h_bin, val)
    
    best_h = h_bin.copy()
    best_val = val
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Pick a random 1 and a random 0, swap them
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_bin[i] = 0
        h_bin[j] = 1
        corr_new = get_corr(h_bin)
        val_new = corr_new.max()
        if val_new < val - 1e-15:
            val = val_new
            if val < best_val:
                best_val = val
                best_h = h_bin.copy()
                try_update(h_bin, val)
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    
    return best_h, best_val

        remove = rng.choice(ones_idx, num_ones - target, replace=False)
        h_bin[remove] = 0
    elif num_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zeros_idx, target - num_ones, replace=False)
        h_bin[add] = 1
    corr = get_corr(h_bin)
    val = corr.max()
    try_update(h_bin, val)
    best_h = h_bin.copy()
    best_val = val
    for p in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(N)
        for i in order:
            if time.time() > t_end:
                break
            if h_bin[i] != 1.0:
                continue
            zeros_idx = np.where(h_bin == 0)[0]
            if len(zeros_idx) == 0:
                break
            h_bin[i] = 0
            best_j = -1
            best_v = val
            for j in zeros_idx:
                h_bin[j] = 1
                v = get_corr(h_bin).max()
                h_bin[j] = 0
                if v < best_v:
                    best_v = v
                    best_j = j
            if best_j >= 0:
                h_bin[best_j] = 1
                val = best_v
                improved = True
                if val < best_val:
                    best_val = val
                    best_h = h_bin.copy()
                    try_update(h_bin, val)
            else:
                h_bin[i] = 1
        if not improved:
            break
    return best_h, best_val

def interpolate_to_n(h_src, n_dst):
    x0 = np.linspace(0, 1, len(h_src), endpoint=False)
    x1 = np.linspace(0, 1, n_dst, endpoint=False)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    best_h = None
    best_val = float('inf')
    def try_update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()
    if h_init is not None:
        v = get_corr(h_init).max()
        try_update(h_init, v)
    N_init = len(h_init) if h_init is not None else 120
    # Phase 1: gradient optimize at init res
    if h_init is not None and time.time() < t_end - 200:
        t_phase = min(time.time() + 150, t_end - 200)
        gradient_optimize(h_init.copy(), N_init, rng, t_phase, try_update)
    # Phase 2: binary SA from best
    if best_h is not None and time.time() < t_end - 180:
        t_phase = min(time.time() + 180, t_end - 180)
        binary_sa(best_h, len(best_h), rng, t_phase, try_update)
    # Phase 3: multi-res gradient
    n_opts = sorted(set([60, 80, 100, N_init, 140, 160]))
    for n_t in n_opts:
        if time.time() > t_end - 100:
            break
        if best_h is not None:
            h_s = interpolate_to_n(best_h, n_t)
        elif h_init is not None:
            h_s = interpolate_to_n(h_init, n_t)
        else:
            h_s = 0.5 * np.ones(n_t)
        t_phase = min(time.time() + 60, t_end - 50)
        gradient_optimize(h_s, n_t, rng, t_phase, try_update)
    # Phase 4: binary greedy pairs on best
    if best_h is not None and time.time() < t_end - 80:
        t_phase = min(time.time() + 100, t_end - 80)
        binary_greedy_pairs(best_h, len(best_h), rng, t_phase, try_update)
    # Phase 5: more binary SA
    if best_h is not None and time.time() < t_end - 50:
        t_phase = min(time.time() + 80, t_end - 50)
        binary_sa(best_h, len(best_h), rng, t_phase, try_update)
    # Phase 6: final gradient polish
    if best_h is not None and time.time() < t_end:
        gradient_optimize(best_h.copy(), len(best_h), rng, t_end, try_update, max_iter=15000)
    if best_h is None:
        best_h = 0.5 * np.ones(47)
        best_val = get_corr(best_h).max()
    return best_h, best_val, len(best_h)
