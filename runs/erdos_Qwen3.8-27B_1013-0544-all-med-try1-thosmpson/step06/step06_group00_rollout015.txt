import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, temp, lam):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) * temp, -500, 0)
    logsum = m + np.log(np.sum(np.exp(shifted))) / temp
    return logsum * dx + lam * (h.sum() - target) ** 2

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfqb_opt(h, n, temp, lam, maxiter=200):
    res = minimize(smooth_obj, h, args=(temp, lam),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, n / 2.0)

def coord_descent(h, max_sweeps=40):
    n = len(h)
    target = n / 2.0
    c5_val = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5_val
            best_v = orig
            for v in (0.0, 1.0, max(0.0, orig - 0.25), min(1.0, orig + 0.25)):
                if abs(v - orig) < 1e-10:
                    continue
                h[j] = v
                h = project_feasible(h, target)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            if best_v != orig:
                h[j] = best_v
                h = project_feasible(h, target)
                c5_val = best_c
                improved = True
            else:
                h[j] = orig
                h = project_feasible(h, target)
        if not improved:
            break
    return h, c5_val

def binary_sa(h_init, n, n_iters, t_start, t_end, deadline):
    h = h_init.astype(int).copy()
    target = n // 2
    n_ones = h.sum()
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            np.random.shuffle(ones)
            h[ones[target:]] = 0
        else:
            zeros = np.where(h == 0)[0]
            np.random.shuffle(zeros)
            h[zeros[:target - n_ones]] = 1
    c5_val = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = c5_val
    for it in range(n_iters):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0, 1
        c_new = compute_c5(h.astype(float))
        if c_new <= c5_val or np.random.rand() < np.exp(-max(c_new - c5_val, 0) / max(t, 1e-12)):
            c5_val = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_c5
