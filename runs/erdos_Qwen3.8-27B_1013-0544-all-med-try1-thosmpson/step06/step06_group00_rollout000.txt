    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        m = idx - (n - 1)
        t1 = np.ones(n)
        t2 = np.zeros(n)
        if m >= 0:
            t1[:n - m] = 1.0 - h[m:]
            t2[m:] = h[:n - m]
        else:
            mm = -m
            t1[mm:] = 1.0 - h[:n - mm]
            t2[:n - mm] = h[mm:]
        grad += t1 - t2
    return grad / top_k * dx

def subgrad_softmax(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    sh = np.clip(corr - mx, -60.0, 0.0)
    w = np.exp(temp * sh)
    ws = w.sum()
    active = np.where(w / ws > 1e-5)[0]
    grad = np.zeros(n)
    for idx in active:
        m = idx - (n - 1)
        wi = w[idx] / ws
        t1 = np.ones(n)
        t2 = np.zeros(n)
        if m >= 0:
            t1[:n - m] = 1.0 - h[m:]
            t2[m:] = h[:n - m]
        else:
            mm = -m
            t1[mm:] = 1.0 - h[:n - mm]
            t2[:n - mm] = h[mm:]
        grad += wi * (t1 - t2)
    val = (mx + np.log(ws) / temp) * dx
    return val, grad * dx

def proj_subgrad_momentum(h, n_iters=2000, top_k=5, init_step=0.08, beta=0.85):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = init_step
    patience = 0
    for it in range(n_iters):
        sg = subgrad_topk(h, top_k=top_k)
        vel = beta * vel - step * sg
        h_new = h + vel
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        if it > 0 and it % 100 == 0:
            if patience > 50:
                step *= 0.75
                vel *= 0.5
            elif step < 0.3:
                step *= 1.05
        step = min(step, 0.5)
    return best_h, best_c5

def proj_subgrad_softmax(h, n_iters=1500, temp=20.0, init_step=0.05, beta=0.8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = init_step
    for it in range(n_iters):
        _, sg = subgrad_softmax(h, temp)
        vel = beta * vel - step * sg
        h_new = h + vel
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 199:
            step *= 0.9
        step = max(step, 0.002)
    return best_h, best_c5

def coord_descent(h, max_sweeps=12, n_levels=13):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    levels = np.linspace(0.0, 1.0, n_levels)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_hj = orig.copy()
            for v in levels:
                trial = orig.copy()
                trial[j] = v
                trial = project_feasible(trial, target)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_hj = trial.copy()
            if best_c < c5 - 1e-13:
                h = best_hj
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def interpolate_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Subgradient + momentum at n0
    for top_k in [3, 5, 8]:
        if remaining() < 120:
            break
        h1, c1 = proj_subgrad_momentum(best_h, n_iters=1500, top_k=top_k,
                                         init_step=0.06, beta=0.85)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Softmax subgradient
    if remaining() > 200:
        for temp in [30.0, 10.0]:
            if remaining() < 100:
                break
            h2, c2 = proj_subgrad_softmax(best_h, n_iters=1200, temp=temp,
                                           init_step=0.04, beta=0.8)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

    # Phase 3: Coordinate descent polish
    if remaining() > 80:
        h3, c3 = coord_descent(best_h, max_sweeps=15, n_levels=15)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: Resolution exploration
    for n_try in [256, 512, 768]:
        if remaining() < 180:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_cur = h_interp.copy()
        # Subgradient at new resolution
        for top_k in [5, 8]:
            if remaining() < 100:
                break
            h_cur, c_tmp = proj_subgrad_momentum(h_cur, n_iters=1000, top_k=top_k,
                                                   init_step=0.05, beta=0.85)
        if remaining() > 60:
            h_cur, c_tmp = coord_descent(h_cur, max_sweeps=8, n_levels=11)
        c_final = compute_c5(h_cur)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_cur.copy()

    # Phase 5: Randomized perturbations at best resolution
    n_best = len(best_h)
    n_restarts = 0
    while remaining() > 100 and n_restarts < 20:
        n_restarts += 1
        scale = 0.15 * (1.0 + 0.5 * np.random.randn())
        scale = max(0.03, min(0.3, scale))
        h_r = best_h + np.random.randn(n_best) * scale
        h_r = project_feasible(h_r, n_best / 2.0)
        h_r, c_r = proj_subgrad_momentum(h_r, n_iters=800, top_k=5,
                                          init_step=0.05, beta=0.85)
        if remaining() > 50:
            h_r, c_r = coord_descent(h_r, max_sweeps=6, n_levels=10)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 6: Final subgradient with varying top_k
    if remaining() > 60:
        for tk in [2, 4, 7, 12]:
            if remaining() < 30:
                break
            h_f, c_f = proj_subgrad_momentum(best_h, n_iters=800, top_k=tk,
                                              init_step=0.03, beta=0.9)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f.copy()

    # Phase 7: Final coordinate descent
    if remaining() > 20:
        h_f, c_f = coord_descent(best_h, max_sweeps=10, n_levels=17)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
