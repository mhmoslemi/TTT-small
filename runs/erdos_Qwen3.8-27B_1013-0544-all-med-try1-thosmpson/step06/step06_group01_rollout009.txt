    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(1.0 - h)[:int(np.ceil(d2))]
                for k in idx:
                    amt = min(1.0 - h[k], d2)
                    h[k] += amt
                    d2 -= amt
                    if d2 < 1e-14:
                        break
            else:
                d2 = -d2
                idx = np.argsort(h)[-int(np.ceil(d2)):]
                for k in idx:
                    amt = min(h[k], d2)
                    h[k] -= amt
                    d2 -= amt
                    if d2 < 1e-14:
                        break

    """Precompute gradient coefficients for all i and all lags.
    Returns array G of shape (n, 2n-1) where G[i,l] = dC_l/dh_i * (1/dx)
    (we factor out dx to keep numbers manageable)
    """

    offsets = np.arange(nc) - (n - 1)
    G = np.zeros((n, nc))
    for i in range(n):
        idx1 = i + offsets
        idx2 = i - offsets
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        G[i, m1] = 1.0 - h[np.clip(idx1[m1], 0, n-1)]
        G[i, m2] -= h[np.clip(idx2[m2], 0, n-1)]
    return G

import numpy as np

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                for k in np.argsort(1.0 - h)[:max(1, int(np.ceil(d2)))]:
                    amt = min(1.0 - h[k], d2)
                    h[k] += amt
                    d2 -= amt
                    if d2 < 1e-14:
                        break
            else:
                d2 = -d2
                for k in np.argsort(h)[-max(1, int(np.ceil(d2))):]:
                    amt = min(h[k], d2)
                    h[k] -= amt
                    d2 -= amt
                    if d2 < 1e-14:
                        break
    return h

    nc = 2 * n - 1
    a = np.zeros(nc)
    idx1 = i + offsets
    idx2 = i - offsets
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    a[m1] = (1.0 - h[np.clip(idx1[m1], 0, n - 1)]) * dx
    a[m2] -= h[np.clip(idx2[m2], 0, n - 1)] * dx
    return a

def _coord_descent(h, n, C, max_sweeps=400, tol=1e-11, n_eval=120):
    dx = 2.0 / n
    nc = 2 * n - 1
    offsets = np.arange(nc) - (n - 1)
    best_c5 = np.max(C)
    prev_best = best_c5 + 1.0

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            a = _grad_row(h, i, n, offsets, dx)
            deltas = np.linspace(lo, hi, n_eval)
            vals = np.max(C + np.outer(deltas, a), axis=1)
            best_idx = np.argmin(vals)
            best_delta = deltas[best_idx]
            if abs(best_delta) < 1e-13:
                continue
            C_new = C + best_delta * a
            h[i] += best_delta
            c5_new = np.max(C_new)
            if c5_new < best_c5 - tol:
                C = C_new
                best_c5 = c5_new
                improved = True
        h = _fix(h, n)
        C = _c5_array(h)
        if np.max(C) < best_c5:
            best_c5 = np.max(C)
        if not improved or abs(prev_best - best_c5) < tol * 0.1:
            break
        prev_best = best_c5
    return h, C, best_c5

        ht, Ct, ct = _coord_descent(ht, nb, Ct, max_sweeps=250, tol=1e-11, n_eval=100)
        ht = _fix(ht, nb)
        Ct = _c5_array(ht)
        ct = np.max(Ct)
        if ct < cb - 1e-9:
            cb, hb, Cb = ct, ht, Ct

    # Phase 5: Random restarts
    for r in range(15):
        if _el() > budget_s * 0.93:
            break
        scale = 0.01 + 0.05 * rng.rand()
        ht = _fix(hb + rng.randn(nb) * scale, nb)
        Ct = _c5_array(ht)
        ht, Ct, ct = _coord_descent(ht, nb, Ct, max_sweeps=200, tol=1e-11, n_eval=80)
        ht = _fix(ht, nb)
        Ct = _c5_array(ht)
        ct = np.max(Ct)
        if ct < cb - 1e-9:
            cb, hb, Cb = ct, ht, Ct

    # Phase 6: Final high-precision polish
    if _el() < budget_s * 0.97:
        hb, Cb, cb = _coord_descent(hb, nb, Cb, max_sweeps=800, tol=1e-12, n_eval=250)
        hb = _fix(hb, nb)
        Cb = _c5_array(hb)
        cb = np.max(Cb)

    return (hb, cb, nb)
