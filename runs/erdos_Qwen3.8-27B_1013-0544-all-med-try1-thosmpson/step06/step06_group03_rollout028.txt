```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-12:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift) * dx
    return c5, grad, conv

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    x = h_init.copy()
    z = h_init.copy()
    tk = 1.0

    t0 = time.time()

def as_direction(h, s_active, n):
    """Compute the optimal direction to reduce c_{s_active} subject to feasibility."""
    g = _conv_grad_single(h, s_active)
    dx = 2.0 / n
    g = g * dx
    
    # Bisection for threshold
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift) * dx
    return c5, conv, grad

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    return t * (m / t + np.log(np.sum(np.exp((conv - m) / t)))) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    active = np.where(weights > 1e-9)[0]
    for idx in active:
        shift = idx - (n - 1)
        grad += weights[idx] * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def fista_smooth(h_init, t, max_iter, max_time, n):
    x = h_init.copy()
    z = h_init.copy()
    tk = 1.0
    best_c5 = compute_c5(x)
    best_x = x.copy()
    t0 = time.time()
    L = 8.0 / (t * n) + 4.0 / n
    step = 1.0 / (2.5 * L)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - t0 > max_time:
            break
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        momentum = (tk - 1.0) / tk_new
        y = x + momentum * (x - z)
        y_clip = np.clip(y, 0.0, 1.0)
        grad = soft_max_grad(y_clip, t)
        x_new = project_feasible(y - step * grad, n)
        if it % 200 == 0:
            c5_new = compute_c5(x_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_x = x_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 3:
                    z = x_new.copy()
                    tk = 1.0
                    no_improve = 0
        z = x
        x = x_new
        tk = tk_new
    c5_f = compute_c5(x)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_x = x.copy()
    return best_x, best_c5

def lbfgs_smooth(h_init, t, max_time, n):
    target = n / 2.0
    t0 = time.time()
    def obj(hv):
        if time.time() - t0 > max_time:
            return soft_max_obj(hv, t) + 1e6
        return soft_max_obj(hv, t) + 200.0 * (np.sum(hv) - target) ** 2
    def grad(hv):
        return soft_max_grad(hv, t) + 400.0 * (np.sum(hv) - target)
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h_init.copy()

def subgrad_pgd(h_init, max_iter, max_time, n):
    x = h_init.copy()
    best_c5 = compute_c5(x)
    best_x = x.copy()
    t0 = time.time()
    lr = 0.004
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > max_time:
            break
        c5, conv, grad = c5_eval(x)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        x_new = project_feasible(x - lr * grad, n)
        c5_new = compute_c5(x_new)
        if c5_new > c5 + 1e-12:
            lr *= 0.7
            if lr < 1e-13:
                break
        else:
            x = x_new
            lr = min(lr * 1.01, 0.015)
        if stall > 3000:
            lr *= 0.5
            stall = 0
    return best_x, best_c5

def compute_as_dir(h, s_active, n):
    g = _conv_grad_single(h, s_active) * (2.0 / n)
    lo, hi = g.min() - 0.01, g.max() + 0.01
    if hi - lo < 1e-14:
        return np.zeros(n)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_val = -h[g > mid].sum() + (1.0 - h[g < mid]).sum()
        if s_val > 0:
            hi = mid
        else:
            lo = mid
    threshold = (lo + hi) / 2.0
    d = np.zeros(n)
    d[g > threshold] = -h[g > threshold]
    d[g < threshold] = 1.0 - h[g < threshold]
    d -= d.sum() / n
    return d

def active_set_polish(h_init, n, max_time=60):
    x = h_init.copy()
    best_c5 = compute_c5(x)
    best_x = x.copy()
    t0 = time.time()
    alpha = 0.5
    stuck = 0
    while time.time() - t0 < max_time:
        c5, conv, _ = c5_eval(x)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x.copy()
        s_active = np.argmax(conv) - (n - 1)
        d = compute_as_dir(x, s_active, n)
        if np.max(np.abs(d)) < 1e-14:
            break
        improved = False
        a = alpha
        for _ in range(12):
            x_new = project_feasible(x + a * d, n)
            c5_new = compute_c5(x_new)
            if c5_new < best_c5 - 1e-14:
                x = x_new
                best_c5 = c5_new
                best_x = x.copy()
                improved = True
                alpha = min(a * 1.05, 1.0)
                break
            a *= 0.5
        if not improved:
            alpha *= 0.6
            stuck += 1
            if stuck > 50 or alpha < 1e-10:
                break
        else:
            stuck = 0
    return best_x, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def tl():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(256, 0.5)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    # Phase 1: FISTA with soft-max at multiple temperatures
    for t_val in [0.08, 0.04, 0.02, 0.01, 0.005, 0.003]:
        if tl() < 100:

        mt = min(90, tl() * 0.22)

        if c5t < best_c5:

    # Phase 2: L-BFGS-B refinement
    if tl() > 60:
        ht = lbfgs_smooth(best_h, 0.01, max_time=min(80, tl() * 0.3), n=n)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    # Phase 3: Subgradient PGD
    if tl() > 40:
        ht, c5t = subgrad_pgd(best_h, max_iter=60000, max_time=min(100, tl() * 0.3), n=n)
        if c5t < best_c5:

    # Phase 4: Active set polish
    if tl() > 30:
        ht, c5t = active_set_polish(best_h, n, max_time=min(80, tl() * 0.25))
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    # Phase 5: Perturbed restarts from best
    for _ in range(20):
        if tl() < 50:
            break
        noise = rng.normal(0, 0.05, n)
        hp = project_feasible(best_h + noise, n)
        ht, c5t = fista_smooth(hp, 0.02, max_iter=3000, max_time=min(15, tl() * 0.1), n=n)
        ht, c5t = subgrad_pgd(ht, max_iter=10000, max_time=min(15, tl() * 0.1), n=n)
        if c5t < best_c5:

    # Phase 6: Final polish
    if tl() > 25:

        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        if tl() > 15:
            ht, c5t = active_set_polish(best_h, n, max_time=min(30, tl() * 0.5))
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
    return best_h, best_c5, n
```
