        if d > 0:

            m = s > 1e-14

import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

_TAU = 0.01

def time_left(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    t = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(50):
        d = t - float(x.sum())
        if abs(d) <= 1e-14:
            break
        if d > 0:
            s = 1.0 - x
            m = s > 1e-14
            if not np.any(m):
                break
            x[m] += d * s[m] / float(s[m].sum())
        else:
            m = x > 1e-14
            if not np.any(m):
                break
            x[m] += d * x[m] / float(x[m].sum())
        np.clip(x, 0.0, 1.0, out=x)
    d = t - float(x.sum())
    if abs(d) > 1e-10:
        k = int(np.floor(t))
        x[:] = 0.0
        x[:k] = 1.0
        r = t - k
        if r > 0 and k < n:
            x[k] = r
    return x

def cval(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def ecx(h):
    return float(np.max(cval(h)))

def subgrad(x, tol=1e-6):
    c = cval(x)
    C = float(np.max(c))
    act = np.where(c >= C - max(1e-8, tol * C))[0]
    n = x.size
    g = np.zeros(n)
    for k in act:
        m = int(k) - (n - 1)
        if m >= 0:
            g[:n - m] += 1.0 - x[m:]
            g[m:] -= x[:n - m]
        else:
            mm = -m
            g[:n - mm] -= x[mm:]
            g[mm:] += 1.0 - x[:n - mm]
    return g * (2.0 / n) / len(act)

def sfun(x):
    c = cval(x)
    cm = float(np.max(c))
    tau = _TAU
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return cm
    return cm + tau * np.log(s)

def sjac(x):
    c = cval(x)
    cm = float(np.max(c))
    tau = _TAU
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    n = x.size
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(c.size):
        if w[k] < 1e-12:
            continue
        m = k - (n - 1)
        wt = w[k] * dx
        if m >= 0:
            g[:n - m] += wt * (1.0 - x[m:])
            g[m:] -= wt * x[:n - m]
        else:
            mm = -m
            g[:n - mm] -= wt * x[mm:]
            g[mm:] += wt * (1.0 - x[:n - mm])
    return g

def cfun(x):
    return np.array([x.sum() - 0.5 * x.size])

def cjac(x):
    return np.ones((1, x.size))

def sa_opt(x0, tl, seed):
    rng = np.random.RandomState(seed)
    x = x0.copy()
    bx = x.copy()
    bc = ecx(x)
    dl = time.time() + tl
    T = 0.018
    n = x.size
    stall = 0
    while time.time() < dl:
        r = rng.random()
        if r < 0.55:
            idx = rng.choice(n, 2, replace=False)
            d = rng.normal(0, T)
            xn = x.copy()
            xn[idx[0]] += d
            xn[idx[1]] -= d
            xn = repair(xn)
        elif r < 0.82:
            bl = int(rng.randint(3, max(4, n // 4)))
            st = int(rng.randint(0, n - bl))
            xn = x.copy()
            xn[st:st + bl] += rng.normal(0, T * 0.6, bl)
            xn = repair(xn)
        else:
            xn = repair(x + rng.normal(0, T * 0.35, n))
        cn = ecx(xn)
        if cn < bc - 1e-14:
            bc, bx, x, stall = cn, xn.copy(), xn, 0
        elif cn < bc or rng.random() < np.exp(-(cn - bc) / T):
            x = xn
        else:
            stall += 1
        T = max(0.00008, T * 0.9986)
        if stall > 250:
            T = max(T * 2.5, 0.007)
            x = repair(bx + rng.normal(0, 0.012, n))
            stall = 0
    return bx, bc

def psg_opt(x0, tl, seed):

    x = x0.copy()
    bx, bc = x.copy(), ecx(x)
    dl = time.time() + tl
    step = 0.035
    px, pg = None, None
    stall = 0
    while time.time() < dl:
        c = ecx(x)
        if c < bc - 1e-14:
            bc, bx, stall = c, x.copy(), 0
        else:
            stall += 1

        if np.linalg.norm(g) < 1e-12:
            break

            y = repair(x - step * g)
            cy = ecx(y)
        if cy < c - 1e-14:
            x = y
        px, pg = x.copy(), g.copy()
    return bx, bc

def slsqp_pol(x0, tl, seed):
    x = x0.copy()
    bx, bc = x.copy(), ecx(x)
    st = time.time()
    bnds = [(0.0, 1.0)] * x.size
    cons = [{"type": "eq", "fun": cfun, "jac": cjac}]
    for tau in [0.012, 0.004, 0.0015, 0.0006]:
        if time.time() - st > tl * 0.88:
            break
        _TAU = tau
        mi = 280 if tau > 0.005 else 200
        try:
            res = so.minimize(sfun, x, method="SLSQP", jac=sjac,
                              bounds=bnds, constraints=cons,
                              options={"maxiter": mi, "ftol": 1e-12})
            if res is not None:
                xr = repair(res.x)
                c = ecx(xr)
                if c < bc:
                    bc, bx = c, xr.copy()
                x = xr if c <= bc + 1e-4 else bx
        except Exception:
            pass
    return bx, bc

def coord_desc(x0, tl, seed):
    rng = np.random.RandomState(seed)
    x = x0.copy()
    bx, bc = x.copy(), ecx(x)
    dl = time.time() + tl
    n = x.size
    step = 0.012
    while time.time() < dl:
        imp = False
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            if time.time() > dl:
                break
            for d in (step, -step, 0.5 * step, -0.5 * step):
                xt = x.copy()
                xt[i] += d
                xt = repair(xt)
                ct = ecx(xt)
                if ct < bc - 1e-13:
                    bx, bc, x, imp = xt.copy(), ct, xt, True
                    break
        if not imp:
            step *= 0.55
            if step < 0.0004:
                break
    return bx, bc

def resample(h0, nn):
    h0 = np.asarray(h0, dtype=float)
    n0, nn = h0.size, int(nn)
    if n0 == 0:
        return repair(np.full(nn, 0.5))
    if n0 == nn:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(nn) + 0.5) * (2.0 / nn)
    return repair(np.interp(x1, x0, h0))
