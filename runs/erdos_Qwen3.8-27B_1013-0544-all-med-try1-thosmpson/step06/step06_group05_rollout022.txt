def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_subgradient(h):
    """Compute subgradient of the max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = np.argmax(corr)
    
    # Compute gradient of corr[k] with respect to h
    grad = np.zeros(n)
    if k < n:
        # corr[k] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        for j in range(n):
            # h[j] as first factor (h[i+k] where i=j-k): valid if 0<=j-k<=n-1-k
            if k <= j < n:
                i = j - k
                if 0 <= i <= n-1-k:
                    grad[j] += (1.0 - h[i])
            # h[j] as second factor ((1-h[i]) where i=j): valid if 0<=j<=n-1-k
            if j <= n-1-k:
                grad[j] += -h[j+k]
    else:
        k2 = k - n + 1
        # corr[k] = sum_{i=k2}^{n-1} h[i] * (1-h[i-k2])
        for j in range(n):
            # h[j] as first factor (h[i] where i=j): valid if k2<=j<=n-1
            if k2 <= j < n:
                grad[j] += (1.0 - h[j-k2])
            # h[j] as second factor ((1-h[i-k2]) where i-k2=j, i=j+k2): valid if j+k2<=n-1
            j2 = j + k2
            if j2 < n:
                grad[j] += -h[j2]
    
    return grad * dx, k

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    h = h.copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            if not np.any(mask):
                break

        else:
            mask = h > 0.0
            if not np.any(mask):

def subgradient_descent(h0, max_time=100.0, seed=42):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project(h, n)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    
    lr = 0.1
    decay = 0.999
    
    while (time.time() - start) < max_time:
        grad, k = get_subgradient(h)
        # Take step
        h_new = h - lr * grad
        h_new = project(h_new, n)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= decay
        
        # Add small noise to escape local minima
        if rng.random() < 0.1:
            noise = rng.normal(0, 0.01, n)
            noise -= np.mean(noise)
            h = project(h + noise, n)
    
    return best_h, best_c5

    return project(np.interp(x_new, x_src, h_src), n_new)

    n = len(h)
    g = np.zeros(n)

        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        k2 = k - n + 1
        g[k2:] += 1.0 - h[:n-k2]
        g[:n-k2] -= h[k2:]
    return g

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(cr)
    w = np.exp(beta * (cr - cm))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w); w[np.argmax(cr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        wk = w[k]
        if k < n:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            k2 = k - n + 1
            grad[k2:] += wk * (1.0 - h[:n-k2])
            grad[:n-k2] -= wk * h[k2:]
    return grad * dx, cm

        mom = mu * mom + (1 - mu) * grad
        h_new = project(h - lr * mom, n)
        nc = compute_c5(h_new)
        if nc < cur:
            h, cur = h_new, nc
            lr = min(lr * 1.01, 0.5)
            if nc < best_c:

        else:
            lr *= 0.92
        if rng.random() < 0.02:
            noise = rng.normal(0, 0.03, n)
            noise -= noise.mean()
            h = project(h + noise, n)
    return best_h, best_c

            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Binary optimization
    if rem() > 120:
        h_bin = make_binary(best_h, n_init)
        t3 = min(180.0, rem() * 0.15)
        if t3 > 20:
            h3, c5_3 = binary_search(h_bin, t3, seed=seed + 1000)
            if c5_3 < best_c5:
                best_c5, best_h = c5_3, h3.copy()
            # Also try continuous from binary
            if rem() > 60:
                h3c, c5_3c = gd_optimize(h3, min(60.0, rem() * 0.1), seed=seed + 1500)
                if c5_3c < best_c5:
                    best_c5, best_h = c5_3c, h3c.copy()

    # Phase 4: Multi-resolution exploration
    for n_try in [128, 160, 192, 224, 256, 320]:
        if rem() < 100:
            break
        h_interp = interp_to(best_h, n_try)
        t_g = min(120.0, rem() * 0.12)
        if t_g < 15:
            break
        h4, c5_4 = gd_optimize(h_interp, t_g, seed=seed + n_try)
        if rem() > 50:
            h5, c5_5 = local_search(h4, min(50.0, rem() * 0.08), seed=seed + n_try + 50)
            if c5_5 < c5_4:
                h4, c5_4 = h5, c5_5
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Final refinement
    if rem() > 30:
        h6, c5_6 = local_search(best_h, rem() * 0.5, seed=seed + 5000)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()
    if rem() > 20:
        h7, c5_7 = gd_optimize(best_h, rem() * 0.7, seed=seed + 6000)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    best_h = project(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
