        return h

def conv_single_shift(h, shift):
    """Compute correlation value for a single shift and its gradient."""
    n = len(h)
    if shift >= 0:
        if shift >= n:
            return 0.0, np.zeros(n)

        grad[:n-shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n-shift]
        return val, grad

            return 0.0, np.zeros(n)
        overlap = h[s:] * (1.0 - h[:n-s])
        val = np.sum(overlap)
        grad = np.zeros(n)

        grad[:n-s] -= h[s:]

def topk_grad(h, k=5):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def conv_single_shift(h, shift):
    n = len(h)
    if shift >= 0:
        if shift >= n:
            return 0.0, np.zeros(n)
        val = np.sum(h[:n-shift] * (1.0 - h[shift:]))
        grad = np.zeros(n)
        grad[:n-shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n-shift]
        return val, grad
    else:
        s = -shift
        if s >= n:
            return 0.0, np.zeros(n)
        val = np.sum(h[s:] * (1.0 - h[:n-s]))
        grad = np.zeros(n)
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
        return val, grad

def topk_subgrad(h, k=5):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    # Get top k shifts
    if k >= len(conv):
        top_idx = np.arange(len(conv))
        weights = np.ones(k) / k
    else:
        top_idx = np.argpartition(conv, -k)[-k:]
        top_conv = conv[top_idx]
        # Use softmax-style weights
        m = top_conv.max()
        exps = np.exp((top_conv - m) / max(top_conv.max() - top_conv.min(), 1e-10))
        weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for i in range(min(k, len(top_idx))):
        shift = top_idx[i] - (n - 1)
        _, g = conv_single_shift(h, shift)
        grad += weights[i] * g
    return grad * dx, c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def conv_single_shift(h, shift):
    n = len(h)
    if shift >= 0:
        if shift >= n:
            return np.zeros(n)
        grad = np.zeros(n)
        grad[:n - shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n - shift]
        return grad
    else:
        s = -shift
        if s >= n:
            return np.zeros(n)
        grad = np.zeros(n)
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
        return grad

def topk_subgrad(h, k):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    kk = min(k, len(conv))
    top_idx = np.argpartition(conv, -kk)[-kk:]
    grad = np.zeros(n)
    for i in range(kk):
        shift = top_idx[i] - (n - 1)
        grad += conv_single_shift(h, shift)
    return grad * dx / kk, c5

def pgd_optimize(h_init, max_iter=30000, lr_init=0.003, topk=5, lr_min=1e-12, mom=0.0):
    n = len(h_init)
    h = h_init.copy()

    best_h = h.copy()
    lr = lr_init
    v = np.zeros(n)
    for it in range(max_iter):
        grad, c5 = topk_subgrad(h, topk)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if mom > 0:
            v = mom * v + (1.0 - mom) * grad
            step = v
        else:
            step = grad
        h_new = project_feasible(h - lr * step, n)
        c5_new = compute_c5(h_new)
        if c5_new <= c5 + 1e-15:
            h = h_new
            lr = min(lr * 1.03, 0.015)
        else:
            lr *= 0.6
            v *= 0.5
            if lr < lr_min:
                break
    return best_h, best_c5

    n_old = len(h)

    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
