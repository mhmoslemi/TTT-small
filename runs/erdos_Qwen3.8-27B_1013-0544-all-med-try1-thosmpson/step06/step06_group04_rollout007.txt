```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def grad_qk(h, k, dx):
    n = len(h)
    if abs(k) >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if k >= 0:
        j_min, j_max = 0, n - 1 - k
    else:
        j_min, j_max = -k, n - 1
    if j_min > j_max:
        return grad
    # Part 1: h[i] from h[j+k] factor, i = j+k
    i_lo = max(j_min + k, 0)
    i_hi = min(j_max + k, n - 1)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += (1.0 - h[idx - k])
    # Part 2: h[i] from (1-h[j]) factor, i = j
    i_lo2 = max(j_min, 0)
    i_hi2 = min(j_max, n - 1)
    if i_lo2 <= i_hi2:
        idx = np.arange(i_lo2, i_hi2 + 1)
        jk = idx + k
        valid = (jk >= 0) & (jk < n)
        grad[idx[valid]] += -h[jk[valid]]
    return grad * dx

def project_feasible(p, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def optimize_smooth(h, n, time_limit=100, beta=20.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    t0 = time.time()
    alpha = 1.0
    while time.time() - t0 < time_limit:
        c5, corr = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        c_max = np.max(corr)
        weights = np.exp(beta * (corr - c_max))
        w_sum = np.sum(weights)
        if w_sum < 1e-150:
            break
        g = np.zeros(n)
        active = weights / w_sum > 0.005
        for idx in range(len(corr)):
            if active[idx]:
                k = idx - (n - 1)
                g += (weights[idx] / w_sum) * grad_qk(h, k, dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        improved = False
        a = alpha
        for _ in range(40):
            h_try = project_feasible(h - a * g, n)
            c5_try, _ = compute_c5(h_try)
            if c5_try < c5 - 1e-12:
                h = h_try
                alpha = min(a * 1.5, 5.0)
                improved = True
                break
            a *= 0.5
            if a < 1e-10:
                break
        if not improved:
            alpha = a
            if alpha < 1e-8:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    global initial_h_values
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(70) * 0.5
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)

    # Phase 1: optimize at original resolution
    remaining = budget_s - (time.time() - t_start) - 400
    t_lim = max(30, min(100, remaining))
    if t_lim > 20:
        h, c5 = optimize_smooth(h, n, time_limit=t_lim, beta=15.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2-5: progressive upsampling
    for target_n in [140, 280, 560, 1120]:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 120
        if remaining < 80 or target_n <= n:
            continue
        if target_n % n != 0:
            ratio = max(1, target_n // n)
            target_n = n * ratio
        ratio = target_n // n
        h_new = np.repeat(h, ratio)
        h_new = project_feasible(h_new, target_n)
        t_lim = min(remaining * 0.6, 200)
        h_new, c5_new = optimize_smooth(h_new, target_n, time_limit=t_lim, beta=20.0)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        n = target_n
        h = h_new

    # Phase 6: final polish with higher beta
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 10
    if remaining > 30:
        h_polish = optimize_smooth(h, n, time_limit=min(remaining, 60), beta=50.0)
        if h_polish[1] < best_c5:
            best_h, best_c5 = h_polish

    final_c5, _ = compute_c5(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
```
