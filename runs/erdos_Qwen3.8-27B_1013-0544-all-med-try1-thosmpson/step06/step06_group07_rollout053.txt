            lo, hi = dlo, dhi
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(c + m1 * d) < np.max(c + m2 * d):
                    hi = m2
                else:
                    lo = m1
            delta = (lo + hi) / 2.0
            x[i] += delta
            x[j] -= delta

    return best_x, best_c

def slsqp_smooth(h, tau, maxiter):
    n = h.size
    dx = 2.0 / n

    def obj(v):
        c = np.correlate(v, 1.0 - v, mode="full") * dx
        cm = float(np.max(c))
        z = (c - cm) / tau
        z -= z.max()
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            return cm
        return cm + tau * np.log(s)

    def obj_grad(v):
        c = np.correlate(v, 1.0 - v, mode="full") * dx
        cm = float(np.max(c))
        z = (c - cm) / tau
        z -= z.max()
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad = np.zeros(n)
        for k in range(c.size):
            if w[k] < 1e-10:
                continue
            s2 = k - (n - 1)
            if s2 >= 0:
                grad[s2:] += w[k] * (1.0 - v[:n - s2]) * dx
                if n - s2 > 0:
                    grad[:n - s2] += w[k] * (-v[s2:]) * dx
            else:
                sa = -s2
                if n - sa > 0:
                    grad[:n - sa] += w[k] * (1.0 - v[sa:]) * dx
                grad[sa:] += w[k] * (-v[:n - sa]) * dx
        return grad

    def cons(v):
        return np.array([v.sum() - 0.5 * n])

    def cons_jac(v):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(obj, h, method="SLSQP", jac=obj_grad,
                          bounds=bounds,
                          constraints=[{"type": "eq", "fun": cons, "jac": cons_jac}],
                          options={"maxiter": maxiter, "ftol": 1e-12})
        return res.x
    except Exception:
        return None

def subgrad_optimize(h, max_time, seed):
    n = h.size
    x = h.copy()
    best_x = x.copy()
    best_c = c5_exact(x)
    rng = np.random.RandomState(seed)
    start = time.time()
    step = 0.05
    mom = np.zeros(n)
    mu = 0.7
    no_imp = 0

    while time.time() - start < max_time:
        c = c5_vals(x)
        C = float(np.max(c))
        if C < best_c - 1e-14:
            best_c = C; best_x = x.copy(); no_imp = 0
        else:
            no_imp += 1
        if no_imp > 300:
            break

        tol = max(1e-8, 0.005 * C)
        active = np.where(c >= C - tol)[0]
        g = subgrad_of_max(x, active)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        accepted = False
        for _ in range(20):
            y = x - step * (g / gn)
            y = repair(y)
            cy = c5_exact(y)
            if cy < C - 1e-13:
                mom = mu * mom + (1 - mu) * (x - y)
                x = y
                accepted = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        if not accepted:
            step = min(0.1, step * 1.5)

        if no_imp > 50 and rng.rand() < 0.15:
            x = repair(x + rng.normal(0, 0.008, n))
            step = 0.03

    return best_x, best_c

def nelder_mead_optimize(h, max_time, seed):
    n = h.size
    x = h.copy()
    best_x = x.copy()
    best_c = c5_exact(x)
    start = time.time()
    rng = np.random.RandomState(seed)

    def fun(v):
        return c5_exact(v)

    options = {"maxiter": 500, "xatol": 1e-10, "fatol": 1e-12}
    try:
        res = so.minimize(fun, x, method="Nelder-Mead", options=options)
        if res.x is not None:
            xr = repair(res.x)
            cr = c5_exact(xr)
            if cr < best_c:
                best_c = cr; best_x = xr
    except Exception:
        pass

    if time.time() - start < max_time * 0.5:
        for trial in range(5):
            if time.time() - start > max_time:
                break
            pert = 0.02 + 0.02 * trial
            x0 = repair(best_x + rng.normal(0, pert, n))
            try:
                res = so.minimize(fun, x0, method="Nelder-Mead", options=options)
                if res.x is not None:
                    xr = repair(res.x)
                    cr = c5_exact(xr)
                    if cr < best_c:
                        best_c = cr; best_x = xr
            except Exception:
                pass

    return best_x, best_c

def resample(h, n_new):
    n = h.size
    if n == n_new:
        return repair(h)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        base = repair(np.asarray(raw, dtype=float))
    else:
        base = repair(np.full(70, 0.5))

    best_x = base.copy()
    best_c = c5_exact(best_x)

    def try_optimize(h, method, t, sd):
        nonlocal best_x, best_c
        if t < 5:
            return
        if method == "pcd":
            x, c = pair_cd_optimize(h, t, sd)
        elif method == "slsqp":
            n = h.size
            for tau in [0.02, 0.005, 0.001]:
                if time.time() > deadline - 5:
                    break
                rem = min(t, deadline - time.time() - 5)
                if rem < 5:
                    break
                mi = max(30, int(rem * 8))
                r = slsqp_smooth(h, tau, mi)
                if r is not None:
                    xr = repair(r)
                    cr = c5_exact(xr)
                    if cr < best_c:
                        best_c = cr; best_x = xr.copy()
                    h = xr
                h = best_x.copy()
        elif method == "subgrad":
            x, c = subgrad_optimize(h, t, sd)
        elif method == "nm":
            x, c = nelder_mead_optimize(h, t, sd)
        else:
            return
        if c < best_c:
            best_c = c; best_x = x.copy()

    # Phase 1: Pair CD at base resolution
    t1 = min(200, (deadline - time.time()) * 0.25)
    h1 = best_x.copy()
    try_optimize(h1, "pcd", t1, sv)

    # Phase 2: SLSQP smooth refinement
    t2 = min(250, (deadline - time.time()) * 0.25)
    h2 = best_x.copy()
    try_optimize(h2, "slsqp", t2, sv + 100)

    # Phase 3: Subgradient polish
    t3 = min(150, (deadline - time.time()) * 0.2)
    h3 = best_x.copy()
    try_optimize(h3, "subgrad", t3, sv + 200)

    # Phase 4: Nelder-Mead
    t4 = min(200, (deadline - time.time()) * 0.25)
    h4 = best_x.copy()
    try_optimize(h4, "nm", t4, sv + 300)

    # Phase 5: Pair CD again on improved solution
    t5 = min(150, (deadline - time.time()) * 0.3)
    h5 = best_x.copy()
    try_optimize(h5, "pcd", t5, sv + 400)

    # Phase 6: Try different resolution - upsample
    if deadline - time.time() > 60:
        for n_new in [100, 140]:
            if deadline - time.time() < 40:
                break
            h_up = resample(best_x, n_new)
            c_up = c5_exact(h_up)
            # Quick pair CD at higher resolution
            t6 = min(40, (deadline - time.time()) * 0.4)
            x6, c6 = pair_cd_optimize(h_up, t6, sv + 500 + n_new)
            if c6 < best_c:
                best_c = c6; best_x = x6.copy()
            if c_up < best_c:
                best_c = c_up; best_x = h_up.copy()

    # Final SLSQP polish
    if deadline - time.time() > 30:
        h7 = best_x.copy()
        n = h7.size
        r = slsqp_smooth(h7, 0.001, max(30, int((deadline - time.time()) * 5)))
        if r is not None:
            xr = repair(r)
            cr = c5_exact(xr)
            if cr < best_c:
                best_c = cr; best_x = xr.copy()

    best_x = repair(best_x)
    best_c = c5_exact(best_x)
    return best_x, best_c, int(best_x.size)
