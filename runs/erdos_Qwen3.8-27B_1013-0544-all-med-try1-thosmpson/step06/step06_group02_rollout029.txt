import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    if 0 <= lag < N:
        g[:N-lag] += (1.0 - h[lag:N])
        g[lag:] -= h[:N-lag]
    elif -N < lag < 0:
        j = -lag
        g[j:] += (1.0 - h[:N-j])
        g[:N-j] -= h[j:]
    return g

def objective_gradient(h, corr, dx, n_top=3, soft=1.0):
    N = len(h)
    max_val = corr.max()
    # Soft weights: exp(-soft * (max_val - corr))
    w = np.exp(-soft * (max_val - corr))
    w /= w.sum()
    # But also cap to top n_top for efficiency
    top_k = np.argsort(corr)[-n_top:]
    mask = np.zeros(len(corr))
    mask[top_k] = 1.0
    w = w * mask
    wsum = w.sum()
    if wsum < 1e-10:
        k = np.argmax(corr)
        return grad_lag(h, k - (N-1)) * dx
    w /= wsum
    g = np.zeros(N)
    for k in range(len(w)):
        if w[k] > 1e-8:
            g += w[k] * grad_lag(h, k - (N-1))
    return g * dx

def line_search(h, g, val, corr, dx, lr, N, max_backtrack=10):
    """Backtracking line search"""
    for i in range(max_backtrack):
        h_try = project_feasible(h - lr * g, N)
        corr_try = compute_corr(h_try)
        val_try = corr_try.max() * dx

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    if 0 <= lag < N:
        g[:N-lag] += (1.0 - h[lag:N])
        g[lag:] -= h[:N-lag]
    elif -N < lag < 0:
        j = -lag
        g[j:] += (1.0 - h[:N-j])
        g[:N-j] -= h[j:]
    return g

def obj_grad(h, corr, dx, n_top=5, soft=3.0):
    N = len(h)
    max_val = corr.max()
    w = np.exp(-soft * (max_val - corr))
    w /= w.sum()
    top_idx = np.argsort(corr)[-n_top:]
    m = np.zeros(len(corr))
    m[top_idx] = 1.0
    w *= m
    ws = w.sum()
    if ws < 1e-10:
        return grad_lag(h, np.argmax(corr) - (N-1)) * dx
    w /= ws
    g = np.zeros(N)
    for k in range(len(w)):
        if w[k] > 1e-7:
            g += w[k] * grad_lag(h, k - (N-1))
    return g * dx
