Alternative approach: Successive Linearization (outer approximation)

1. Start with h^(0)
2. At each iteration:
   a. Compute the current correlate values: c_k = correlate_k(h^(t)) * dx for all k
   b. Find the active constraints (those close to the max)
   c. Linearize the problem around h^(t):
      - The bilinear term h[i]*(1-h[j]) can be linearized as:
        h[i]*(1-h[j]) ≈ h0[i]*(1-h0[j]) + (h[i]-h0[i])*(1-h0[j]) - h0[i]*(h[j]-h0[j])
      - This gives a linear program!
   d. Solve the LP using CVXPY
   e. Update h^(t+1) = solution of LP
3. Repeat until convergence

This is essentially a cutting-plane / outer-approximation method for bilinear programs. Each LP is a standard linear program that CVXPY can solve efficiently.

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def _eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def _fix_sum(h, n):
    """Fix the sum constraint while maintaining 0<=h<=1"""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:
        mask = h > 1e-9
    if np.sum(mask) == 0:
        mask = np.ones(n, dtype=bool)
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def _lp_linearize_one_step(h, n):
    """One step of successive linearization. Returns LP solution."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    con = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    # Compute linearized constraints
    for m in range(2 * n - 1):
        s = m - (n - 1)
        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)
        if n_lo > n_hi:
            continue
        
        # f_m(h) = sum_{nn=n_lo}^{n_hi} h[nn] * (1-h[nn+s])
        # grad_j = sum of contributions
        # For j in [n_lo, n_hi]: grad_j += (1 - h[j+s])
        # For j in [n_lo+s, n_hi+s]: grad_j += -h[j-s]  (where j-s in [n_lo, n_hi])
        
        grad = np.zeros(n)
        f_val = 0.0
        for nn in range(n_lo, n_hi + 1):
            f_val += h[nn] * (1 - h[nn + s])
            grad[nn] += (1 - h[nn + s])
            grad[nn + s] -= h[nn]
        
        # Linearized: f(h0) + grad @ (h_new - h0) <= t/dx
        # => dx * (grad @ h_new - grad @ h0 + f_val) <= t
        # => dx * grad @ h_new <= t - dx * f_val + dx * grad @ h0
        
        rhs = -dx * f_val + dx * np.dot(grad, h)
        con.append(dx * cp.sum(grad * h_var) - t <= rhs)
    
    prob = cp.Problem(cp.Minimize(t), con)
    
    try:
        prob.solve(solver=cp.ECOS)
    except:
        try:
            prob.solve(solver=cp.CLARABEL)
        except:
            return h
    
    if h_var.value is None:
        return h
    return np.clip(h_var.value, 0, 1)

def _successive_lin(h, n, maxiter=50, tol=1e-7, damp=1.0):
    """Successive linearization with damping"""
    h = h.copy()
    best = _eval_c5(h, n)
    
    for it in range(maxiter):
        h_lp = _lp_linearize_one_step(h, n)
        h_new = (1 - damp) * h + damp * h_lp
        h_new = _fix_sum(h_new, n)
        
        val = _eval_c5(h_new, n)
        if val < best:
            best = val
            h = h_new
        elif abs(val - best) < tol:
            h = h_new
            break
        else:
            # Damp more
            damp = max(0.1, damp * 0.8)
            h_new = (1 - damp) * h + damp * h_lp
            h_new = _fix_sum(h_new, n)
            val = _eval_c5(h_new, n)
            h = h_new
            if val < best:
                best = val
    
    return h, best

    dx = 2.0 / n
    h = h.copy()
    best = _eval_c5(h, n)

        n_hi = min(n - 1, n - 1 - s)

        for nn in range(n_lo, n_hi + 1):

    step = 0.01
    for it in range(steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        corr = np.correlate(h, 1 - h, mode='full')
        k = int(np.argmax(corr))
        s = k - (n - 1)
        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)
        grad = np.zeros(n)
        for nn in range(n_lo, n_hi + 1):
            grad[nn] += (1.0 - h[nn + s])
            grad[nn + s] -= h[nn]
        h -= step * grad * dx
        h = _fix_sum(h, n)
        val = _eval_c5(h, n)
        if val < best:
            best = val
        step = max(0.002, step * 0.997)
    return h, best

def _multi_subgrad(h, n, total_steps=1000, time_limit=None, start_time=None):
    """Subgradient with alternating multiple active constraints."""
    dx = 2.0 / n
    h = h.copy()
    best = _eval_c5(h, n)
    step = 0.008
    for it in range(total_steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        avg = np.mean(corr)
        # Use gradient of all above-average constraints weighted
        grad = np.zeros(n)
        count = 0
        for m in range(2 * n - 1):
            if corr[m] > avg:
                s = m - (n - 1)
                n_lo = max(0, -s)
                n_hi = min(n - 1, n - 1 - s)
                for nn in range(n_lo, n_hi + 1):
                    grad[nn] += (1.0 - h[nn + s])
                    grad[nn + s] -= h[nn]
                count += 1
        if count > 0:
            h -= step * grad / count * dx
            h = _fix_sum(h, n)
            val = _eval_c5(h, n)
            if val < best:
                best = val
        step = max(0.001, step * 0.998)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()

    def _tl(frac=1.0):
        return (time.time() - start) < frac * budget_s

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _fix_sum(h, n)
    best_h = h.copy()
    best_c5 = _eval_c5(h, n)

    # Phase 1: SL from initial
    h1, c51 = _successive_lin(h, n, maxiter=40, tol=1e-8,
                               time_limit=0.3 * budget_s, start_time=start)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51

    # Phase 2: Multi-subgradient
    h2, c52 = _multi_subgrad(best_h, n, total_steps=800,
                             time_limit=0.45 * budget_s, start_time=start)
    if c52 < best_c5:
        best_h, best_c5 = h2, c52

    # Phase 3: Another SL round
    h3, c53 = _successive_lin(best_h, n, maxiter=30, tol=1e-9,
                               time_limit=0.6 * budget_s, start_time=start)
    if c53 < best_c5:
        best_h, best_c5 = h3, c53

    # Phase 4: Subgradient refinement
    h4, c54 = _subgrad(best_h, n, steps=500,
                       time_limit=0.7 * budget_s, start_time=start)
    if c54 < best_c5:
        best_h, best_c5 = h4, c54

    # Phase 5: Upsample to 2n
    if _tl(0.75):
        n2 = 2 * n
        h2 = np.repeat(best_h, 2).astype(float)
        h2 = _fix_sum(h2, n2)
        h2, c55 = _successive_lin(h2, n2, maxiter=20, tol=1e-7,
                                   time_limit=0.88 * budget_s, start_time=start)
        if c55 < best_c5:
            best_h, best_c5 = h2, c55

    # Phase 6: Final subgradient at best resolution
    if _tl(0.9):
        nb = len(best_h)
        h6, c56 = _subgrad(best_h, nb, steps=400,
                           time_limit=0.97 * budget_s, start_time=start)
        if c56 < best_c5:
            best_h, best_c5 = h6, c56

    # Final verification
    best_h = _fix_sum(best_h, len(best_h))
    final_c5 = _eval_c5(best_h, len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
