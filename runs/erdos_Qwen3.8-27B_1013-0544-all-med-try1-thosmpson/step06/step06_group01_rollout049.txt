Wait, that's the circular correlation, not the linear one. The full (linear) correlation can be computed by zero-padding:

If we zero-pad h to length 2n-1 (or next power of 2), then:
C_full = ifft(fft(h_padded) * conj(fft(1-h_padded)))

This gives O(n log n) instead of O(n²).

                if add > 1e-15:
                    h[i] += add
                    d -= add
                if abs(d) < 1e-13:
                    break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                sub = min(-d, h[i])
                if sub > 1e-15:
                    h[i] -= sub
                    d += sub
                if abs(d) < 1e-13:
                    break
    return h

def _solve_1d(c, a, lo, hi):
    if hi - lo < 1e-15:
        return 0.5 * (lo + hi)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = c + mid * a
        m = np.max(vals)
        active = vals >= m - 1e-12
        sg = np.mean(a[active])
        if sg > 1e-11:
            hi = mid
        elif sg < -1e-11:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def _cd(h, n, sweeps=500, tol=1e-11):
    h = h.copy()
    k_arr = np.arange(2 * n - 1)
    for _ in range(sweeps):
        C = np.correlate(h, 1.0 - h, mode='full')
        mx = 0.0
        for i in range(n):
            idx1 = i + k_arr - (n - 1)
            idx2 = i - k_arr + (n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2 * n - 1)
            a[v1] = 1.0 - h[idx1[v1]]
            a[v2] -= h[idx2[v2]]
            if np.all(np.abs(a) < 1e-14):
                continue
            delta = _solve_1d(C, a, -h[i], 1.0 - h[i])
            if abs(delta) > 1e-14:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                C += delta * a
                mx = max(mx, abs(delta))
        if mx < tol:
            break
    return h

def _interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def _sm_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    return (m + np.log(np.sum(es))) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2

def _sm_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    k_arr = np.arange(2 * n - 1)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = i + k_arr - (n - 1)
        idx2 = i - k_arr + (n - 1)
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        ta = np.zeros(2 * n - 1)
        tb = np.zeros(2 * n - 1)
        ta[v1] = 1.0 - h[idx1[v1]]
        tb[v2] = h[idx2[v2]]
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _local_pair(h, n, max_pairs=2000):
    """Try swapping small amounts between pairs of coordinates."""
    h = h.copy()
    dx = 2.0 / n
    best = _c5_val(h)
    rng = np.random.RandomState(123)
    improved = True
    iters = 0
    while improved and iters < 50:
        improved = False
        iters += 1
        for _ in range(max_pairs):
            i, j = rng.randint(0, n), rng.randint(0, n)
            if i == j:
                continue
            for delta in [0.005, 0.01, 0.02, 0.002]:
                for sign in [1, -1]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h_try[i] + sign * delta, 0, 1)
                    h_try[j] = np.clip(h_try[j] - sign * delta, 0, 1)
                    c5 = _c5_val(h_try)
                    if c5 < best - 1e-12:
                        h = h_try
                        best = c5
                        improved = True
                        break
                if improved:
                    break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _fix_sum(initial_h_values.astype(float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_val(h)
    best_n = n

    def _try_update(h_try, n_try):
        nonlocal best_h, best_c5, best_n
        h_try = _fix_sum(h_try, n_try)
        c5 = _c5_val(h_try)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5, best_n = h_try.copy(), c5, n_try

    # Phase 1: Coordinate descent
    h1 = _cd(h, n, sweeps=600)
    _try_update(h1, n)

    # Phase 2: Multiple resolutions
    n_candidates = []
    for n_try in [70, 75, 80, 85, 90, 95, 100, 105, 110, 115, 120, 130, 140]:
        if n_try <= 0:
            continue
        h_try = _interp_h(best_h, best_n, n_try)
        h_try = _fix_sum(h_try, n_try)
        h_try = _cd(h_try, n_try, sweeps=400)
        _try_update(h_try, n_try)
        if _el() > budget_s * 0.35:
            break

    # Phase 3: Random restarts
    for r in range(80):
        if _el() > budget_s * 0.65:
            break
        scale = 0.01 + 0.05 * rng.rand()
        h_try = best_h + rng.randn(best_n) * scale
        h_try = _fix_sum(h_try, best_n)
        h_try = _cd(h_try, best_n, sweeps=300)
        _try_update(h_try, best_n)

    # Phase 4: Structured initializations
    for r in range(20):
        if _el() > budget_s * 0.78:
            break
        n_try = best_n
        typ = r % 4
        if typ == 0:
            h_try = np.where(np.arange(n_try) % 4 < 2, 1.0, 0.0)
        elif typ == 1:
            x = np.linspace(0, 2, n_try)
            h_try = np.clip(0.5 + 0.5 * np.sin(2 * np.pi * rng.rand(3, 1).T @ x / 2 + rng.rand(3)), 0, 1)
            h_try = h_try.mean(axis=0)
        elif typ == 2:
            h_try = np.clip(0.5 + rng.randn(n_try) * 0.3, 0, 1)
        else:
            h_try = np.clip(1.0 - np.abs(np.linspace(-1, 1, n_try) - rng.uniform(-0.5, 0.5)), 0, 1)
        h_try = _fix_sum(h_try, n_try)
        h_try = _cd(h_try, n_try, sweeps=300)
        _try_update(h_try, n_try)

    # Phase 5: Soft-max L-BFGS polish
    if _el() < budget_s * 0.90:
        for alpha in [500, 2000, 8000]:
            if _el() > budget_s * 0.92:
                break
            res = minimize(_sm_grad, best_h.copy(), args=(alpha, 2.0),
                          jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * best_n,
                          options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-14})
            if res.x is not None:
                _try_update(res.x, best_n)

    # Phase 6: Final coordinate descent
    if _el() < budget_s * 0.97:
        h_fin = _cd(best_h, best_n, sweeps=800)
        _try_update(h_fin, best_n)

    # Phase 7: Pair local search
    if _el() < budget_s * 0.99:
        h_ps, c5_ps = _local_pair(best_h, best_n, max_pairs=3000)
        if c5_ps < best_c5:
            best_h, best_c5 = h_ps, c5_ps

    return (best_h, best_c5, best_n)
