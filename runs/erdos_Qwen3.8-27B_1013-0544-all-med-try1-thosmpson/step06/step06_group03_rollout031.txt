def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

        idxs = np.where(h == 1.0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:int(cur_sum - target)]:
            h[i] = 0.0
    elif cur_sum < target:
        idxs = np.where(h == 0.0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:int(target - cur_sum)]:
            h[i] = 1.0
    C, S = compute_sensitivity(h)
    best_c5 = np.max(C) * dx
    best_h = h.copy()
    for step in range(max_steps):
        # Find best flip that reduces max C
        best_val = np.max(C)
        best_j = -1
        best_dir = 0
        # Vectorized: for each j, compute max(C + S[j]) if h[j]==0, max(C - S[j]) if h[j]==1
        max_plus = np.max(C[None, :] + S, axis=1)  # if h[j]==0, flip to 1: C + S[j]
        max_minus = np.max(C[None, :] - S, axis=1)  # if h[j]==1, flip to 0: C - S[j]
        # For h[j]==0, candidate is max_plus[j]; for h[j]==1, candidate is max_minus[j]
        candidate = np.where(h == 0, max_plus, max_minus)
        # Only consider flips that maintain sum (allow drift of 1)
        cur_sum = h.sum()
        if cur_sum >= target:
            # Can only flip 1->0 or allow 0->1 if sum < target+1
            candidate = np.where(h == 0, np.inf, max_minus)
        else:
            candidate = np.where(h == 1, np.inf, max_plus)
        j = np.argmin(candidate)
        if candidate[j] >= best_val:
            break
        # Execute flip
        if h[j] == 0:
            h[j] = 1.0
            C += S[j]
        else:
            h[j] = 0.0
            C -= S[j]
        # Recompute S
        _, S = compute_sensitivity(h)
        new_max = np.max(C) * dx
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
    return best_h, best_c5

def thue_morse_init(n):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return tm

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = thue_morse_init(128)
        h0 = project_feasible(h0, len(h0))

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuous PGD with cooling on base n
    for lr_i in [0.006, 0.003, 0.001]:
        if tl() < 40:
            break
        ht, c5t = pgd_cool(best_h, max_iter=20000, lr_init=lr_i, t_init=0.04, t_min=0.001, t_decay=0.9997)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Binary local search on base n
    if tl() > 60:
        hb = np.where(best_h >= 0.5, 1.0, 0.0)
        hb, c5b = binary_local_search(hb, max_steps=3000)
        if c5b < best_c5:
            best_h, best_c5 = hb, c5b

    # Phase 3: Try different n values
    n_candidates = [64, 96, 128, 160, 192, 256, 320, 400, 480]
    if n0 not in n_candidates:
        n_candidates.append(n0)
    n_candidates.sort()

    for n_try in n_candidates:
        if tl() < 120:
            break
        if n_try == n0 and n_try == len(best_h):
            continue
        nc = len(best_h)
        hi = interpolate_h(best_h, n_try)
        hi = project_feasible(hi, n_try)

        # Continuous optimization
        for lr_i in [0.005, 0.002]:
            if tl() < 60:
                break
            ht, c5t = pgd_cool(hi, max_iter=15000, lr_init=lr_i, t_init=0.03, t_min=0.001, t_decay=0.9998)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

        # Binary local search
        if tl() > 60:
            hb = np.where(best_h >= 0.5, 1.0, 0.0)
            hb, c5b = binary_local_search(hb, max_steps=2000)
            if c5b < best_c5:
                best_h, best_c5 = hb, c5b

    # Phase 4: Random restarts at various n
    for n_try in [64, 128, 192, 256]:
        for _ in range(6):
            if tl() < 40:
                break
            hr = rng.uniform(0.0, 1.0, n_try)
            hr = project_feasible(hr, n_try)
            ht, c5t = pgd_cool(hr, max_iter=8000, lr_init=0.008, t_init=0.06, t_min=0.005, t_decay=0.999)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
            if tl() < 30:
                break

    # Phase 5: Thue-Morse based starts
    for n_try in [96, 128, 192, 256]:
        if tl() < 50:
            break
        ht = thue_morse_init(n_try)
        ht = project_feasible(ht, n_try)
        ht, c5t = pgd_cool(ht, max_iter=10000, lr_init=0.004, t_init=0.03, t_min=0.002, t_decay=0.9995)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        if tl() > 30:
            hb = np.where(best_h >= 0.5, 1.0, 0.0)
            hb, c5b = binary_local_search(hb, max_steps=2000)
            if c5b < best_c5:
                best_h, best_c5 = hb, c5b

    # Final polish
    if tl() > 30:
        ht, c5t = pgd_cool(best_h, max_iter=30000, lr_init=0.0005, t_init=0.01, t_min=0.001, t_decay=0.9999)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Final binary polish
    if tl() > 20:
        hb = np.where(best_h >= 0.5, 1.0, 0.0)
        hb, c5b = binary_local_search(hb, max_steps=3000)
        if c5b < best_c5:
            best_h, best_c5 = hb, c5b

    return best_h, best_c5, len(best_h)
