import numpy as np
import time

def _c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_vals(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        idx = np.argmax(1.0 - h) if d2 > 0 else np.argmax(h)
        h[idx] += d2
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_step(h, i, c5):
    """Find optimal value of h[i] to minimize max C[k]."""
    n = len(h)
    dx = 2.0 / n
    k_min = max(0, n - 1 - i)
    k_max = min(2 * n - 2, n - 1 + i)
    k_vals = np.arange(k_min, k_max + 1)
    
    # Coefficient of h[i] in C[k] for k != n-1 (linear part)
    # For k = n-1: term is h[i]*(1-h[i]) = h[i] - h[i]^2
    # For k != n-1: term is h[i]*(1-h[i+k-n+1])
    
    idx_partner = i + k_vals - n + 1
    valid = (idx_partner >= 0) & (idx_partner < n)
    idx_clipped = np.clip(idx_partner, 0, n-1)
    
    # For each k, C[k] as function of x = h[i]:
    # C[k] = x * (1-h[partner]) * dx + rest (for k where partner exists and partner != i)
    # C[n-1] = x*(1-x)*dx + rest (the self-term)
    
    # Let's compute: for each k in k_vals, what is C[k] = f_k(x)?
    # C[k] = x*(1-h[partner_k])*dx + C[k] - x*(1-h[partner_k])*dx (if partner valid)
    #      = x*(1-h[partner_k])*dx + b_k  where b_k = C[k] - x*(1-h[partner_k])*dx
    
    # But for k = n-1, partner = i, so the term is x*(1-x)*dx not x*(1-h[i])*dx
    # We handle this by subtracting the current contribution and adding the new one
    
    # Current contribution of h[i] to C[k]:
    # For valid partner: h[i] * (1-h[partner]) * dx
    # For k=n-1 (partner=i): h[i] * (1-h[i]) * dx
    
    current_x = h[i]
    contrib = np.where(valid, current_x * (1.0 - h[idx_clipped]) * dx, 0.0)
    b = c5[k_vals] - contrib  # constant part
    
    # New C[k] when h[i] = x:
    # For valid partner (partner != i): x * (1-h[partner]) * dx + b_k
    # For k=n-1 (partner=i): x*(1-x)*dx + b_k
    
    # Coefficient a_k for linear part (1-h[partner])*dx
    a = np.where(valid, (1.0 - h[idx_clipped]) * dx, 0.0)
    
    # self_mask: True where partner == i (i.e., k == n-1)
    self_mask = (idx_partner == i)
    
    # f_k(x) = a_k * x + b_k for non-self k
    # f_self(x) = x*(1-x)*dx + b_self for self k
    # f(x) = max(f_1(x), ..., f_m(x))
    
    def f_of_x(x):
        vals = a * x + b
        if np.any(self_mask):
            vals[self_mask] = x * (1.0 - x) * dx + b[self_mask]
        return np.max(vals)
    
    # Evaluate at grid points
    x_grid = np.linspace(0, 1, 101)
    f_vals = np.array([f_of_x(x) for x in x_grid])
    x_coarse = x_grid[np.argmin(f_vals)]
    
    # Refine
    lo = max(0.0, x_coarse - 2.0 * (1.0/100.0))
    hi = min(1.0, x_coarse + 2.0 * (1.0/100.0))
    x_fine = np.linspace(lo, hi, 201)
    f_fine = np.array([f_of_x(x) for x in x_fine])
    x_best = x_fine[np.argmin(f_fine)]
    
    return x_best

def _coord_descent(h, n, n_sweeps=100, random_order=False, rng=None):
    h = h.copy()
    best_c5 = _c5_max(h)
    
    for sweep in range(n_sweeps):
        c5 = _c5_vals(h)
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        any_change = False
        for i in order:
            new_val = _coord_step(h, i, c5)
            if abs(new_val - h[i]) > 1e-9:
                h[i] = new_val
                any_change = True
                c5 = _c5_vals(h)  # update after each change
        
        new_c5 = np.max(c5)
        if new_c5 < best_c5:
            best_c5 = new_c5
        if not any_change:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    n0 = len(initial_h_values)

import numpy as np
import time

def _c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_vals(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        idx = np.argmax(1.0 - h) if d2 > 0 else np.argmax(h)
        h[idx] += d2
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_step_vec(h, i, c5):
    n = len(h)
    dx = 2.0 / n
    k_min = max(0, n - 1 - i)
    k_max = min(2 * n - 2, n - 1 + i)
    k_vals = np.arange(k_min, k_max + 1)
    idx_partner = i + k_vals - n + 1
    valid = (idx_partner >= 0) & (idx_partner < n)
    idx_clipped = np.clip(idx_partner, 0, n - 1)
    current_x = h[i]
    contrib = np.where(valid, current_x * (1.0 - h[idx_clipped]) * dx, 0.0)
    b = c5[k_vals] - contrib
    a = np.where(valid, (1.0 - h[idx_clipped]) * dx, 0.0)
    self_mask = (idx_partner == i)
    x_grid = np.linspace(0.0, 1.0, 401)
    vals = a[None, :] * x_grid[:, None] + b[None, :]
    if np.any(self_mask):
        vals[:, self_mask] = x_grid[:, None] * (1.0 - x_grid[:, None]) * dx + b[None, self_mask]
    f_vals = np.max(vals, axis=1)
    x_coarse = x_grid[np.argmin(f_vals)]
    step = 1.0 / 400.0
    lo = max(0.0, x_coarse - 2.0 * step)
    hi = min(1.0, x_coarse + 2.0 * step)
    x_fine = np.linspace(lo, hi, 401)
    vals = a[None, :] * x_fine[:, None] + b[None, :]
    if np.any(self_mask):
        vals[:, self_mask] = x_fine[:, None] * (1.0 - x_fine[:, None]) * dx + b[None, self_mask]
    f_fine = np.max(vals, axis=1)
    x_best = x_fine[np.argmin(f_fine)]
    return x_best

def _coord_descent(h, n, n_sweeps, random_order, rng):
    h = h.copy()
    for sweep in range(n_sweeps):
        c5 = _c5_vals(h)
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        any_change = False
        for i in order:
            new_val = _coord_step_vec(h, i, c5)
            if abs(new_val - h[i]) > 1e-10:
                h[i] = new_val
                any_change = True
                c5 = _c5_vals(h)
        if not any_change:
            break
    return h

def _pair_local(h, n, n_pairs, rng):
    h = h.copy()
    best = _c5_max(h)
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    for _ in range(n_pairs):
        improved = False
        for _ in range(200):
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            for d in deltas:
                for sign in (1, -1):
                    hi = h[i] + sign * d
                    hj = h[j] - sign * d
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i], h[j] = hi, hj
                    c5 = _c5_max(h)
                    if c5 < best - 1e-12:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved:
            break
    return h, best
