        grad[s:] += (1.0 - h[:n-s])
        # j+s valid: j+s < n, so j from 0 to n-1-s

    for _ in range(50):

        else:

def compute_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] += (1.0 - h)
            grad[:] -= h
        else:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    else:
        sa = -s
        grad[:n-sa] += (1.0 - h[sa:])
        grad[sa:] -= h[:n-sa]
    return grad * dx

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(z, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(z - lam, 0.0, 1.0)

def compute_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    else:
        sa = -s
        grad[:n - sa] += (1.0 - h[sa:])
        grad[sa:] -= h[:n - sa]
    return grad * dx

def multishift_gradient(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_use = min(top_k, 2 * n - 1)
    top_idx = np.argsort(-corr)[:k_use]
    vals = corr[top_idx]
    max_val = vals[0]
    min_val = vals[-1]
    if max_val > min_val + 1e-10:
        weights = (vals - min_val) / (max_val - min_val)
    else:
        weights = np.ones(k_use) / k_use
    weights = weights ** 2
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        s = int(idx) - (n - 1)
        grad += w * compute_gradient(h, s)
    return grad

def gradient_descent(h0, max_time, seed):
    n = len(h0)
    target = n / 2.0
    h = project_onto_constraint(np.asarray(h0, dtype=float).copy(), target)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    lr = 4.0
    no_improve = 0
    top_k = 7

    while (time.time() - start) < max_time:
        grad = multishift_gradient(h, top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        best_local_c5 = current_c5
        best_h_try = h
        for factor in [1.0, 0.5, 0.25, 2.0, 4.0]:
            step = lr * factor
            h_new = h - step * grad
            h_new = project_onto_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_h_try = h_new
        if best_local_c5 < current_c5 - 1e-14:
            h = best_h_try
            current_c5 = best_local_c5
            no_improve = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 20.0)
        else:
            no_improve += 1
            lr *= 0.85
            if no_improve > 40:
                noise_scale = min(0.05 * (no_improve - 40) / 40.0, 0.25)
                h = best_h + rng.normal(0, noise_scale, n)
                h = project_onto_constraint(h, target)
                current_c5 = compute_c5(h)
                lr = 4.0
                no_improve = 0
                top_k = max(3, top_k - 2)
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_onto_constraint(h_new, n_new / 2.0)

def binary_swaps(h0, max_time, seed):
    n = len(h0)
    sorted_idx = np.argsort(-h0)
    h = np.zeros(n)
    h[sorted_idx[:n // 2]] = 1.0
    current_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 0.0
        best_move = None
        n_tries = min(3000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain
                best_move = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            current_c5 -= best_gain
        else:
            break
    return h, compute_c5(h)

def itersa_refine(h0, max_time, seed):
    """Iterated local search: perturb then gradient descent"""
    n = len(h0)
    target = n / 2.0
    h = project_onto_constraint(np.asarray(h0, dtype=float).copy(), target)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    n_restarts = 5
    time_per = max_time / n_restarts
    for r in range(n_restarts):
        if r > 0:
            scale = 0.15 * (1 + r * 0.3)
            h = best_h + rng.normal(0, scale, n)
            h = project_onto_constraint(h, target)
        remain = max(time.time() - start, 0)
        t_sub = max(time_per - (remain - r * time_per), 2.0)
        h_sub, c5_sub = gradient_descent(h, t_sub, seed + r * 1000)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
    return best_h, best_c5
