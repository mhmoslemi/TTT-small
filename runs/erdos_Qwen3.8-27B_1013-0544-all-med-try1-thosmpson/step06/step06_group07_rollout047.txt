def compute_all_gradients(x):
    """Compute gradient of c_k w.r.t. x for all k. Returns array of shape (2n-1, n)."""
    n = x.size
    dx = 2.0 / n
    m = 2 * n - 1
    grads = np.zeros((m, n), dtype=float)
    
    for k in range(m):
        d = k - (n - 1)
        # S_k = {i : max(0,-d) <= i <= min(n-1, n-1-d)}
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        
        # Term 1: (1 - h[i+d]) for i in [lo, hi]
        # h[i+d] for i in [lo, hi] → h[lo+d : hi+d+1]
        term1 = np.zeros(n)
        term1[lo:hi+1] = 1.0 - x[lo+d:hi+d+1]
        
        # Term 2: -h[i] for i in S_k, contributes to index i+d
        # So for j = i+d, i = j-d, valid when i in [lo,hi], so j in [lo+d, hi+d]
        term2 = np.zeros(n)
        jlo = lo + d
        jhi = hi + d
        term2[jlo:jhi+1] = -x[lo:hi+1]
        
        grads[k] = dx * (term1 + term2)
    
    return grads

def max_subgradient(x):
    """Compute a subgradient of f(x) = max_k c_k(x) at x."""
    n = x.size
    dx = 2.0 / n
    c = c_values(x)
    C = np.max(c)
    tol = max(1e-8, 0.01 * C)
    active = np.where(c >= C - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        for i in range(lo, hi + 1):
            j = i - d
            # term: h[i] * (1-h[j]) * dx
            # d/dh[i] = (1-h[j]) * dx
            grad[i] += (1.0 - x[j]) * dx
            # d/dh[j] = -h[i] * dx  (only if j != i, otherwise counted once)
            if j != i:
                grad[j] -= x[i] * dx
            else:
                grad[j] -= x[i] * dx  # still valid: d/dh[i] of h[i](1-h[i]) = 1-2h[i]
                # total for i=j: (1-h[i])*dx - h[i]*dx = (1-2h[i])*dx ✓
    
    grad /= len(active)
    return grad

def smooth_max_val(c, tau):
    cm = float(np.max(c))
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    return cm + tau * float(np.log(s)) if s > 0 else cm


def smooth_w(c, tau):
    cm = float(np.max(c))
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s
