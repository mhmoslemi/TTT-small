def binary_local_search(h, N, rng, t_end, max_iters=1000000):
    h = h.astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, 'full')
    val = corr.max() * dx
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Pick random i from ones, j from zeros
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[i] = 0
        h[j] = 1
        
        # Evaluate
        corr_new = np.correlate(h, 1-h, 'full')
        val_new = corr_new.max() * dx
        
        if val_new < val:
            val = val_new
            corr = corr_new
            # Update ones and zeros
            ones = ones[ones != i]
            zeros = zeros(zeros != j)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return h, val

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, k):
    N = len(h)
    g = np.zeros(N)
    # corr at lag k: sum_i h[i] * (1-h[i+k])
    # d/dh[m] = (1-h[m+k]) if m+k < N, minus h[m-k] if m-k >= 0
    if k >= 0:
        g[:N-k] += (1.0 - h[k:])
        g[k:] -= h[:N-k]
    else:
        k = -k
        g[k:] += (1.0 - h[:N-k])
        g[:N-k] -= h[k:]
    return g

def soft_grad(h, corr, dx, alpha=20.0, top_k=8):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr) - (N - 1)
        return grad_lag(h, k_star) * dx
    # Use only top top_k lags
    top_idx = np.argsort(w)[-top_k:]
    g = np.zeros(N)
    w_sum = 0.0
    for j in top_idx:
        if w[j] < 1e-10:
            continue
        lag = j - (N - 1)
        if lag == 0:
            continue  # Skip lag 0 (always zero contribution to change)
        g += w[j] * grad_lag(h, lag)
        w_sum += w[j]
    if w_sum > 0:
        g /= w_sum
    return g * dx

def alt_linear_refine(h, k, N, n_sweeps=50):
    """Alternating linear method to maximize S_k = sum_i h[i]*h[i+k]."""
    h = h.copy()
    for _ in range(n_sweeps):
        c = np.zeros(N)
        if k > 0:
            c[:N-k] += h[k:]
            c[k:] += h[:N-k]
        elif k < 0:
            kk = -k
            c[kk:] += h[:N-kk]
            c[:N-kk] += h[kk:]
        else:
            c = 2.0 * h
        
        # Project: set top N/2 values to 1, rest to 0 (for binary case)
        # Or for continuous: use the projection
        # For now, do a soft version
        sorted_idx = np.argsort(c)[::-1]
        h_new = np.zeros(N)
        h_new[sorted_idx[:N//2]] = 1.0
        h = project_feasible(h_new, N)
    return h

def binary_sa(h, N, rng, t_end, t0_temp=0.5, t_end_temp=0.01, max_iters=500000):
    """Simulated annealing on binary sequences."""
    h = np.round(h)
    h = project_feasible(h, N)  # Ensure constraint
    # Make it properly binary
    target = N // 2
    ones = np.sum(h == 1.0)
    if ones > target:
        # Randomly set some 1s to 0s
        one_idx = np.where(h == 1.0)[0]
        to_zero = rng.choice(one_idx, ones - target, replace=False)
        h[to_zero] = 0.0
    elif ones < target:
        zero_idx = np.where(h == 0.0)[0]
        to_one = rng.choice(zero_idx, target - ones, replace=False)
        h[to_one] = 1.0
    
    dx = 2.0 / N
    val, corr, _ = evaluate(h)
    best_h = h.copy()
    best_val = val
    it = 0
    total = max_iters
    
    while time.time() < t_end and it < total:
        it += 1
        progress = it / total
        T = t0_temp * (t_end_temp / t0_temp) ** progress
        
        # Pick random swap
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        val_new = np.correlate(h, 1.0 - h, mode='full').max() * dx
        
        delta = val_new - val
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            val = val_new
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_val

def gradient_refine(h, N, rng, t_end, max_iter=100000):

    val, corr, dx = evaluate(h)
    best_val = val
    best_h = h.copy()

    stagnant = 0
    
    for it in range(max_iter):

        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new

                val, corr, _ = evaluate(h)
                lr = 0.02
                stagnant = 0

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, k):
    N = len(h)
    g = np.zeros(N)
    if k >= 0:
        g[:N-k] += (1.0 - h[k:])
        g[k:] -= h[:N-k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:N-kk])
        g[:N-kk] -= h[kk:]
    return g

def soft_grad(h, corr, dx, alpha=20.0, top_k=6):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr) - (N - 1)
        return grad_lag(h, k_star) * dx
    top_idx = np.argsort(w)[-top_k:]
    g = np.zeros(N)
    w_sum = 0.0
    for j in top_idx:
        if w[j] < 1e-10:
            continue
        lag = j - (N - 1)
        if lag == 0:
            continue
        g += w[j] * grad_lag(h, lag)
        w_sum += w[j]
    if w_sum > 0:
        g /= w_sum
    return g * dx

def binary_sa(h_start, N, rng, t_end, t0_temp=0.5, t_end_temp=0.01, max_iters=5000000):
    h = np.round(h_start * 10) / 10  # Start from a rounded version
    # Make properly binary
    target = N // 2
    # Sort by value, set top N/2 to 1
    idx = np.argsort(h)[::-1]
    h = np.zeros(N)
    h[idx[:target]] = 1.0
    
    dx = 2.0 / N
    val, corr, _ = evaluate(h)
    best_h = h.copy()
    best_val = val
    it = 0
    
    while time.time() < t_end and it < max_iters:
        it += 1
        progress = min(it / max_iters, 1.0)
        T = t0_temp * (t_end_temp / t0_temp) ** progress
        
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        val_new = np.correlate(h, 1.0 - h, mode='full').max() * dx
        
        delta = val_new - val
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            val = val_new
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_val

def gradient_refine(h_start, N, rng, t_end, max_iter=200000):
    h = h_start.copy()
    val, corr, dx = evaluate(h)
    best_val = val
    best_h = h.copy()
    lr = 0.03
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = soft_grad(h, corr, dx, alpha=15.0, top_k=6)
        h_new = project_feasible(h - lr * g, N)
        val_new, corr_new, _ = evaluate(h_new)
        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            lr = min(lr * 1.02, 0.1)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            stagnant += 1
            lr *= 0.8
            if stagnant > 30:
                h = project_feasible(best_h + rng.randn(N) * 0.05, N)
                val, corr, _ = evaluate(h)
                lr = 0.03
                stagnant = 0
    
    return best_h, best_val

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) * 2.0 / len(h_src)
    x1 = np.arange(n_dst) * 2.0 / n_dst
    h = np.interp(x1, x0, h_src)
