def compute_grad_k(x, k):
    n = x.size
    g = np.zeros(n)
    if k >= 0:
        hi = n - 1 - k
        if hi >= 0:
            g[0:hi+1] += (1.0 - x[k:hi+1+k])
            g[k:hi+1+k] -= x[0:hi+1]
    else:
        kk = -k
        g[kk:n] += (1.0 - x[0:n-kk])
        g[0:n-kk] -= x[kk:n]
    return g

def smooth_obj_w(x, tau):
    c = c_values(x)
    cm = float(np.max(c))
    if tau <= 1e-10:
        w = np.zeros(c.size); w[int(np.argmax(c))] = 1.0
        return cm, w
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(c.size); w[int(np.argmax(c))] = 1.0
        return cm, w
    return cm + tau * np.log(s), e / s

def grad_smooth(x, tau):
    n = x.size
    dx = 2.0 / float(n)
    _, w = smooth_obj_w(x, tau)
    m = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        grad += wi * compute_grad_k(x, k)
    return grad * dx

def grad_exact(x, tol=1e-7):
    n = x.size
    dx = 2.0 / float(n)
    c = c_values(x)
    C = float(np.max(c))
    active = np.where(c >= C - max(1e-9, tol * C))[0]
    if active.size > 10:
        active = np.argsort(c)[::-1][:10]
    grad = np.zeros(n)
    for idx in active:
        grad += compute_grad_k(x, int(idx) - (n - 1))
    return grad * dx / float(active.size)

def run_slsqp_smooth(x0, tau, maxiter):
    def fun(x):
        v, _ = smooth_obj_w(x, tau)
        return v
    def jac(x):
        return grad_smooth(x, tau)
    bounds = [(0.0, 1.0)] * x0.size
    cons = [{'type': 'eq', 'fun': lambda x: np.array([x.sum() - 0.5 * x.size]),
             'jac': lambda x: np.ones((1, x.size))}]
    try:
        res = so.minimize(fun, x0, method='SLSQP', jac=jac, bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return res
    except Exception:
        return None

def exact_polish(x0, time_limit, seed=42):
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    rng = np.random.RandomState(seed)
    deadline = time.time() + max(0.1, float(time_limit))
    step = 0.5
    no_imp = 0
    while time.time() < deadline:
        cvals = c_values(x)
        C = float(np.max(cvals))
        if C < best_c - 1e-14:
            best_c = C; best_x = x.copy(); no_imp = 0
        else:
            no_imp += 1
        if no_imp > 20:
            improved = False
            for sg in (0.01, 0.005, 0.002, 0.001, 0.0005):
                y = repair(x + rng.normal(0, sg, x.size))
                cy = exact_c(y)
                if cy < best_c:
                    best_c = cy; best_x = y.copy(); x = y; no_imp = 0; improved = True; break
            if not improved:
                break
        if no_imp > 8:
            active = np.argsort(cvals)[::-1][:4]
        else:
            active = np.where(cvals >= C - max(1e-8, 1e-6 * C))[0]
        grad = np.zeros(x.size)
        dx = 2.0 / float(x.size)
        for idx in active:
            grad += compute_grad_k(x, int(idx) - (x.size - 1))
        grad = grad * dx / float(active.size)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        accepted = False
        for _ in range(20):
            y = repair(x - step * grad)
            cy = exact_c(y)
            if cy < C - 1e-13:
                x = y; accepted = True; break
            step *= 0.5
            if step < 1e-12:
                break
        if not accepted:
            no_imp += 1
        if rng.rand() < 0.1:
            y = repair(x + rng.normal(0, 0.003, x.size))
            if exact_c(y) < best_c:
                best_c = exact_c(y); best_x = y.copy(); x = y; no_imp = 0
    return best_x, best_c

def binary_optimize(h0, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    x = repair(h0)
    n = x.size
    target = int(round(0.5 * n))
    order = np.argsort(x)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    best_c = exact_c(b)
    best_b = b.copy()
    deadline = time.time() + time_limit
    stall = 0
    while time.time() < deadline:
        C = exact_c(b)
        if C < best_c - 1e-14:
            best_c = C; best_b = b.copy(); stall = 0
        else:
            stall += 1
        if stall > 500:
            if best_c < 0.381:
                break
            nf = rng.randint(2, 6)
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
            fi = rng.choice(ones, min(nf, len(ones)), replace=False)
            fj = rng.choice(zeros, min(nf, len(zeros)), replace=False)
            b[fi] = 0.0; b[fj] = 1.0
            if exact_c(b) > best_c + 0.01:
                b[fi] = 1.0; b[fj] = 0.0
            stall = 0
            continue
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        improved = False
        for _ in range(300):
            i = int(rng.randint(len(ones)))
            j = int(rng.randint(len(zeros)))
            pi, pj = ones[i], zeros[j]
            b[pi] = 0.0; b[pj] = 1.0
            cn = exact_c(b)
            if cn < C - 1e-14:
                ones[i], zeros[j] = zeros[j], ones[i]
                improved = True
                break
            b[pi] = 1.0; b[pj] = 0.0
        if not improved:
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
    return best_b, best_c

def resample(h0, n_new):
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def structured_start(n, seed, freqs=None, amp=0.4):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    if freqs is None:
        freqs = [1, 2, 3]
    for i, f in enumerate(freqs):
        h += (amp / (i + 1)) * np.sin(2 * np.pi * f * t / 2.0 + rng.uniform(0, 6.2832))
    h += 0.01 * rng.normal(0, 1, n)
    return repair(h)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    best_x = None
    best_c = 1e9

    if raw is not None and len(raw) >= 10:
        n0 = int(len(raw))
        x0 = repair(np.array(raw, dtype=float))
        if exact_c(x0) < best_c:
            best_c = exact_c(x0); best_x = x0.copy()
    else:
        n0 = 70
        x0 = repair(np.full(n0, 0.5))

    # Phase 1: Smooth SLSQP optimization at base resolution
    x = x0.copy()
    for tau, mi in [(0.05, 300), (0.02, 400), (0.01, 500), (0.005, 500), (0.002, 400), (0.001, 300)]:
        if time_left(deadline) < 40:
            break
        res = run_slsqp_smooth(x, tau, mi)
        if res is not None and res.success:
            xr = repair(res.x)
            c = exact_c(xr)
            if c < best_c:
                best_c = c; best_x = xr.copy()
            if c <= best_c + 5e-4:
                x = xr
            else:
                x = best_x

    # Phase 2: Exact polish
    if time_left(deadline) > 30:
        tl = min(60.0, time_left(deadline) * 0.5)
        xp, cp = exact_polish(best_x, tl, sv)
        if cp < best_c:
            best_c = cp; best_x = xp

    # Phase 3: Binary optimization
    if time_left(deadline) > 20:
        tl = min(40.0, time_left(deadline) * 0.4)
        bb, bc = binary_optimize(best_x, tl, sv + 999)
        if bc < best_c:
            best_c = bc; best_x = bb.copy()

    # Phase 4: More smooth + exact from binary solution or different starts
    if time_left(deadline) > 50:
        starts = [best_x]
        for i in range(3):
            starts.append(structured_start(n0, sv + 1000 + i * 77, [1, 2, 3, 5]))
        for i, h0 in enumerate(starts):
            if time_left(deadline) < 30:
                break
            x2 = h0.copy()
            for tau, mi in [(0.02, 300), (0.005, 400), (0.002, 300)]:
                if time_left(deadline) < 15:
                    break
                res = run_slsqp_smooth(x2, tau, mi)
                if res is not None and res.success:
                    xr = repair(res.x)
                    c = exact_c(xr)
                    if c < best_c:
                        best_c = c; best_x = xr.copy()
                    if c <= best_c + 3e-4:
                        x2 = xr
                    else:
                        x2 = best_x
            if time_left(deadline) > 10:
                xp, cp = exact_polish(best_x, min(15.0, time_left(deadline) * 0.4), sv + 5000 + i)
                if cp < best_c:
                    best_c = cp; best_x = xp

    # Phase 5: Try a finer resolution
    if time_left(deadline) > 40:
        n_fine = n0 + (n0 // 2)
        if n_fine % 2 == 1:
            n_fine += 1
        xf = resample(best_x, n_fine)
        cf = exact_c(xf)
        if cf < best_c:
            best_c = cf; best_x = xf.copy()
        for tau, mi in [(0.01, 300), (0.003, 400)]:
            if time_left(deadline) < 15:
                break
            res = run_slsqp_smooth(xf, tau, mi)
            if res is not None and res.success:
                xr = repair(res.x)
                c = exact_c(xr)
                if c < best_c:
                    best_c = c; best_x = xr.copy()
        if time_left(deadline) > 10:
            xp, cp = exact_polish(best_x, min(20.0, time_left(deadline) * 0.5), sv + 9000)
            if cp < best_c:
                best_c = cp; best_x = xp

    # Final polish
    if time_left(deadline) > 5:
        xp, cp = exact_polish(best_x, min(15.0, time_left(deadline) * 0.6), sv + 77777)
        if cp < best_c:
            best_c = cp; best_x = xp

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
