def pairwise_optstep(h, n, i, j, c):
    """One pairwise optimization step. Returns new h and new c."""
    dx = 2.0 / n
    # Compute delta by unit perturbation
    h_p = h.copy()
    h_p[i] += 1.0
    h_p[j] -= 1.0
    c_p = np.correlate(h_p, 1 - h_p, mode='full') * dx
    delta = c_p - c  # change per unit d
    
    # Feasible range for d
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    
    if d_lo >= d_hi - 1e-12:
        return h, c
    
    # Find optimal d (linear approximation)
    m = len(c)
    # Vectorized intersection finding
    da = delta[:, None]
    db = delta[None, :]
    diff = da - db
    cc = c[None, :] - c[:, None]  # cc[a,b] = c[b] - c[a]
    with np.errstate(divide='ignore', invalid='ignore'):
        d_cross = cc / diff
    
    mask = (np.abs(diff) > 1e-14) & (d_cross >= d_lo) & (d_cross <= d_hi)
    np.fill_diagonal(mask, False)
    
    valid_d = d_cross[mask]
    candidates = np.unique(np.concatenate([[d_lo, d_hi], valid_d]))
    
    # Evaluate max at each candidate
    # vals[k] = max_p (c[p] + candidates[k] * delta[p])
    # = (candidates[:,None] * delta[None,:]).T + c  -> max along axis 1
    if len(candidates) > 0:
        vals = np.max(candidates[:, None] * delta[None, :] + c[None, :], axis=1)
        best_idx = np.argmin(vals)
        best_d = candidates[best_idx]
    else:
        best_d = 0.0
    
    if abs(best_d) < 1e-14:
        return h, c
    
    h_new = h.copy()
    h_new[i] += best_d
    h_new[j] -= best_d
    c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
    
    return h_new, c_new

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def find_optimal_d(c, delta, d_lo, d_hi):
    m = len(c)
    if d_lo >= d_hi:
        return 0.0
    da = delta[:, None]
    db = delta[None, :]
    diff = da - db
    cc = c[None, :] - c[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        d_cross = cc / diff
    mask = (np.abs(diff) > 1e-14) & (d_cross >= d_lo) & (d_cross <= d_hi)
    np.fill_diagonal(mask, False)
    valid_d = d_cross[mask]
    candidates = np.unique(np.concatenate([[d_lo, d_hi], valid_d]))
    if len(candidates) == 0:
        return 0.0
    vals = np.max(candidates[:, None] * delta[None, :] + c[None, :], axis=1)
    return candidates[np.argmin(vals)]

    h_p[j] -= 1.0

    delta = c_p - c
    best_d = find_optimal_d(c, delta, d_lo, d_hi)
    if abs(best_d) < 1e-14:
        return h, c
    h_new = h.copy()
    h_new[i] += best_d
    h_new[j] -= best_d
    c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
    return h_new, c_new

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def find_optimal_d(c, delta, d_lo, d_hi):
    if d_lo >= d_hi:
        return 0.0
    m = len(c)
    da = delta[:, None]
    db = delta[None, :]
    diff = da - db
    cc = c[None, :] - c[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        d_cross = cc / diff
    mask = (np.abs(diff) > 1e-14) & (d_cross >= d_lo) & (d_cross <= d_hi)
    np.fill_diagonal(mask, False)
    valid_d = d_cross[mask]
    candidates = np.unique(np.concatenate([[d_lo, d_hi], valid_d]))
    if len(candidates) == 0:
        return 0.0
    vals = np.max(candidates[:, None] * delta[None, :] + c[None, :], axis=1)
    return float(candidates[np.argmin(vals)])

def pairwise_step(h, n, i, j, c):
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-12:
        return h, c
    h_p = h.copy()
    h_p[i] += 1.0
    h_p[j] -= 1.0
    c_p = np.correlate(h_p, 1 - h_p, mode='full')
    dx = 2.0 / n
    delta = (c_p - c) * dx
    best_d = find_optimal_d(c, delta, d_lo, d_hi)
    if abs(best_d) < 1e-14:
        return h, c
    h_new = h.copy()
    h_new[i] += best_d
    h_new[j] -= best_d
    c_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
    return h_new, c_new
