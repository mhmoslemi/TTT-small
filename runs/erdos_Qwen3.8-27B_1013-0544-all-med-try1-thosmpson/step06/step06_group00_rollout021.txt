```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_delta_coeffs(h, i):
    """Get the coefficient vector d_i for changing h[i]."""
    n = len(h)
    m = 2 * n - 1
    d = np.zeros(m)
    # First factor: k in [n-1-i, n-1+i]
    k1_start = n - 1 - i
    k1_end = n - 1 + i
    for k in range(k1_start, k1_end + 1):
        j2 = i + k - (n - 1)
        d[k] += (1.0 - h[j2])
    # Second factor: k in [i, i+n-1]
    k2_start = i
    k2_end = i + n - 1
    for k in range(k2_start, min(k2_end + 1, m)):
        j1 = i - k + (n - 1)
        d[k] -= h[j1]
    return d

def coord_descent_fast(h, max_sweeps=30, n_trial=30):
    """Fast coordinate descent using analytical delta computation."""
    n = len(h)
    ts = n / 2.0
    dx = 2.0 / n
    m = 2 * n - 1
    
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            a = h[i]
            if a <= 0 or a >= 1:
                # Still try to move towards interior
                pass
            
            d = get_delta_coeffs(h, i)
            # Find best t in [-a, 1-a] to minimize max(corr + t*d)
            # Use grid search
            t_vals = np.linspace(-a, 1 - a, n_trial)
            best_t = 0
            best_val = np.max(corr)
            for t in t_vals:
                new_max = np.max(corr + t * d)
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_t = t
            
            if abs(best_t) > 1e-12:
                h[i] = a + best_t
                # Project to maintain feasibility
                h = project_feasible(h, ts)
                corr = compute_corr(h)
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
                else:
                    # Revert (projection may have hurt)
                    h[i] = a
                    corr = compute_corr(h)
        if not improved:
            break
```

```python
    t_hi = 1.0 - a

    f_hi = np.max(corr + t_hi * d)
```
