    n = len(h)

    s = np.sum(np.exp(shifted - max_shifted))
    return (m + max_shifted / temp + np.log(s) / temp) * dx

Actually, there's a smarter way. Note that:
sum_k p_k * corr_k is a weighted sum of correlations.
The gradient of a weighted sum of correlations can be computed as a convolution!

sum_k p_k * corr_k = sum_k p_k * sum_i h[i+k-n+1] * (1-h[i])

Let me define the "reversed" weights: q_s = p_{s+n-1} where s is the true lag (ranges from -(n-1) to n-1).

Then: sum_s q_s * (sum_i h[i+s]*(1-h[i]))
= sum_s q_s * (sum_i h[i+s] - sum_i h[i+s]*h[i])
= sum_i h[i] * (sum_s q_s * 1[valid]) - sum_i sum_s q_s * h[i+s]*h[i] * 1[valid]

The first term is linear: (A^T 1) * h where A is the "validity matrix"
The second term is quadratic: h^T B h where B[i,j] = q_{j-i} * 1[valid pair]

So the gradient is: linear part - 2 * B * h (approximately, need to be careful about the bilinear form)

Actually, for the bilinear form h^T B h where B is not symmetric:
∂/∂h = (B + B^T) * h

And B[i,j] = q_{j-i} * 1[valid_i,j]
B^T[i,j] = B[j,i] = q_{i-j} * 1[valid_j,i]

By symmetry of the validity condition (pair (i,j) is valid iff pair (j,i) is valid for the correlation sum), we have B^T[i,j] = q_{i-j} * 1[valid_i,j] = B[j,i].

        a1s, a1e = max(0, lo + s), min(n - 1, hi + s)
        if a1s <= a1e:
            arr = np.arange(a1s, a1e + 1)
            g[arr] += (1.0 - h[arr - s])
        a2s, a2e = max(0, lo), min(n - 1, hi)
        if a2s <= a2e:
            arr = np.arange(a2s, a2e + 1)
            g[arr] -= h[arr + s]
    return (g / k) * dx

def grad_lse(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_s = np.exp(temp * (corr - m))
    p = exp_s / exp_s.sum()
    g = np.zeros(n)
    for kk in range(len(corr)):
        w = p[kk]
        if w < 1e-12:
            continue
        s = kk - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        a1s, a1e = max(0, lo + s), min(n - 1, hi + s)
        if a1s <= a1e:
            arr = np.arange(a1s, a1e + 1)
            g[arr] += w * (1.0 - h[arr - s])
        a2s, a2e = max(0, lo), min(n - 1, hi)
        if a2s <= a2e:
            arr = np.arange(a2s, a2e + 1)
            g[arr] += -w * h[arr + s]
    return g * dx

def projected_grad_descent(h, target, mode='topk', k=10, temp=50.0,
                            max_iter=500, lr_init=0.01, lr_decay=0.999,
                            patience=50, tol=1e-14):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    lr = lr_init
    best_local = c5
    best_local_h = h.copy()
    no_improve = 0
    
    for it in range(max_iter):
        if mode == 'topk':
            g = grad_topk(h, k)
        else:
            g = grad_lse(h, temp)
        
        # Gradient descent step (we want to minimize, so subtract gradient)
        h_new = h - lr * g
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5 - tol:
            h = h_new
            c5 = c5_new
            no_improve = 0
            if c5 < best_local - tol:
                best_local = c5
                best_local_h = h.copy()
                lr = lr_init * 0.5  # reduce step on improvement? No, keep same
        else:
            no_improve += 1
            lr *= lr_decay
        
        if no_improve >= patience:
            break
    
    return h, c5

def coord_descent(h, target, n_levels=20, max_sweeps=10):
    n = len(h)
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_arr = h.copy()
            for v in np.linspace(0.0, 1.0, n_levels):
                h[j] = v
                d = target - h.sum()
                if abs(d) > 1e-12:
                    mask = (h > 0) & (h < 1)
                    if mask.sum() > 0:
                        h[mask] += d / mask.sum()
                        np.clip(h, 0.0, 1.0, out=h)
                    else:
                        h[j] = orig
                        break
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_arr = h.copy()
            if best_c < c5 - 1e-13:
                h = best_arr
                c5 = best_c
                improved = True
            else:
                h[j] = orig
        if not improved:
            break
    return h, c5

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
