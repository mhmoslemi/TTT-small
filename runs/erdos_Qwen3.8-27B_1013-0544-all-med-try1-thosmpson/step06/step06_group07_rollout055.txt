    """Gradient of c[k_idx] w.r.t. x. k_idx is index in full correlation."""
    n = len(x)

    grad = np.zeros(n)

        if kk < n:

            grad[:n - kk] -= x[kk:]
    else:
        kk = -lag
        if kk < n:
            grad[:n - kk] += 1.0 - x[kk:]
            grad[kk:] -= x[:n - kk]
    return grad * dx

import numpy as np
import time
import warnings

warnings.simplefilter("ignore")


def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(40):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-15
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-15
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / x[mask].sum()
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - x)[::-1] if diff > 0 else np.argsort(x)[::-1]
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
            else:
                add = max(diff, -x[idx])
            x[idx] += add
            diff -= add
    np.clip(x, 0.0, 1.0, out=x)
    return x


def c_values(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c(h):
    return float(np.max(c_values(h)))


def grad_lag(x, k_idx):
    n = len(x)
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        kk = lag
        if kk < n:
            grad[kk:] += 1.0 - x[:n - kk]
            grad[:n - kk] -= x[kk:]
    else:
        kk = -lag
        if kk < n:
            grad[:n - kk] += 1.0 - x[kk:]
            grad[kk:] -= x[:n - kk]
    return grad * dx


def smooth_obj_and_w(x, tau, topk=8):
    cv = c_values(x)
    C = float(np.max(cv))
    z = (cv - C) / tau
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        w = np.zeros_like(cv)
        w[int(np.argmax(cv))] = 1.0
        return C, cv, w
    w = e / s
    if topk and topk < len(cv):
        idx = np.argsort(w)[::-1][:topk]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        ss = w2.sum()
        if ss > 0:
            w = w2 / ss
    return C, cv, w


def compute_smooth_grad(x, tau, topk=8):
    C, cv, w = smooth_obj_and_w(x, tau, topk)
    g = np.zeros_like(x)
    for i in np.where(w > 1e-12)[0]:
        g += w[i] * grad_lag(x, int(i))
    return g


def sa_phase(x, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(x)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    n = x.size
    T = 0.035
    cooling = 0.9998
    step_base = 0.04

    while time.time() < deadline:
        frac = 1.0 - (deadline - time.time()) / max(time_limit, 1e-6)
        noise = step_base * (1.0 - 0.5 * frac)
        y = x + rng.normal(0, noise, n)
        y = repair(y)
        cy = exact_c(y)
        cx = exact_c(x)
        d = cy - cx
        if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-9)):
            x = y
            if cy < best_c:
                best_c = cy
                best_x = x.copy()
        T *= cooling
        if T < 0.0004 and frac < 0.95:
            T = 0.006
    return best_x, best_c


def grad_phase(x, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(x)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    n = x.size
    step = 0.06
    no_imp = 0

    while time.time() < deadline:
        rem = (deadline - time.time()) / max(time_limit, 1e-6)
        tau = 0.025 * max(rem, 0.05)
        g = compute_smooth_grad(x, tau, topk=8)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        accepted = False
        for _ in range(18):
            y = repair(x - step * g / gn)
            cy = exact_c(y)
            if cy < exact_c(x) - 1e-12:
                x = y
                step = min(step * 1.2, 0.3)
                accepted = True
                break
            step *= 0.45
        if accepted:
            cx = exact_c(x)
            if cx < best_c:
                best_c = cx
                best_x = x.copy()
            no_imp = 0
        else:
            step *= 0.6
            no_imp += 1
            if no_imp > 35:
                y = repair(x + rng.normal(0, 0.012, n))
                if exact_c(y) < best_c:
                    best_c = exact_c(y)
                    best_x = y.copy()
                    x = y
                    no_imp = 0
                    step = 0.05
                else:
                    break
    return best_x, best_c


def polish_phase(x, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(x)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    n = x.size
    step = 0.02
    no_imp = 0

    while time.time() < deadline:
        cv = c_values(x)
        C = float(np.max(cv))
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 50:
            y = repair(x + rng.normal(0, 0.004, n))
            if exact_c(y) < best_c:
                best_c = exact_c(y)
                best_x = y.copy()
                x = y
                no_imp = 0
            else:
                break
        tol = max(1e-8, 0.004 * C)
        active = np.where(cv >= C - tol)[0]
        if len(active) > 4:
            active = np.argsort(cv)[::-1][:4]
        g = np.zeros(n)
        for k in active:
            g += grad_lag(x, int(k))
        g /= max(len(active), 1)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        for _ in range(12):
            y = repair(x - step * g / gn)
            cy = exact_c(y)
            if cy < C - 1e-11:
                x = y
                step = min(step * 1.15, 0.08)
                break
            step *= 0.5
    return best_x, best_c


def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x1, x0, h0)
    return repair(y)


def full_optimize(x0, time_limit, seed, sa_frac=0.35, grad_frac=0.40, polish_frac=0.25):
    x = x0
    best_x = x0.copy()
    best_c = exact_c(x0)
    unchanged = time.time() - (time.time() - time_limit)
    if deadline is None:
        deadline = time.time() + time_limit
    if time.time() < deadline - time_limit * polish_frac:
        sa_t = time_limit * sa_frac
        x1, c1 = sa_phase(x, sa_t, seed)
        if c1 < best_c:
            best_c = c1
            best_x = x1
    if time.time() < deadline - time_limit * polish_frac * 0.5:
        gr_t = min(time_limit * grad_frac, deadline - time.time() - 10)
        x2, c2 = grad_phase(best_x, gr_t, seed + 500)
        if c2 < best_c:
            best_c = c2
            best_x = x2
    if time.time() < deadline - 2:
        po_t = min(time_limit * polish_frac, deadline - time.time() - 2)
        x3, c3 = polish_phase(best_x, po_t, seed + 900)
        if c3 < best_c:
            best_c = c3
            best_x = x3
    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and len(raw) >= 10:
        x0 = repair(np.asarray(raw, dtype=float))
    else:
        x0 = repair(np.full(70, 0.5))

    best_x = x0.copy()
    best_c = exact_c(x0)
    n0 = x0.size

    # Primary optimization at native resolution
    if deadline - time.time() > 120:
        t1 = min(380.0, (deadline - time.time()) * 0.42)
        x1, c1 = full_optimize(x0, t1, seed_val)
        if c1 < best_c:
            best_c = c1
            best_x = x1

    # Second pass with perturbation for diversity
    if deadline - time.time() > 100:
        rng = np.random.RandomState(seed_val + 313)
        x_p = repair(best_x + rng.normal(0, 0.025, best_x.size))
        t2 = min(220.0, (deadline - time.time()) * 0.30)
        x2, c2 = full_optimize(x_p, t2, seed_val + 4000)
        if c2 < best_c:
            best_c = c2
            best_x = x2

    # Higher resolution attempts
    for mult in (2, 3):
        n_new = n0 * mult
        if deadline - time.time() > 70:
            x_up = resample_to(best_x, n_new)
            t3 = min(160.0, (deadline - time.time()) * 0.30)
            t3 = max(t3, 30.0)
            x3, c3 = full_optimize(x_up, t3, seed_val + 7000 * mult)
            if c3 < best_c:
                best_c = c3
                best_x = x3

    # Final multi-start refinement at best resolution
    if deadline - time.time() > 40:
        nb = best_x.size
        for trial in range(4):
            if deadline - time.time() < 25:
                break
            rng = np.random.RandomState(seed_val + 11000 + trial * 191)
            x_r = repair(best_x + rng.normal(0, 0.015, nb))
            t4 = min(70.0, (deadline - time.time()) * 0.35)
            t4 = max(t4, 10.0)
            x4, c4 = full_optimize(x_r, t4, seed_val + 15000 + trial * 7)
            if c4 < best_c:
                best_c = c4
                best_x = x4

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
