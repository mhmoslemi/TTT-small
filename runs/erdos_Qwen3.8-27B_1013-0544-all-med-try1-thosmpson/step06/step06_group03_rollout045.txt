Flipped: $\int (1-h(x)) \cdot h(x+k) dx$

These are actually the same integral! Because:
$\int_0^2 h(x)(1-h(x+k)) dx$ and $\int_0^2 (1-h(x))h(x+k) dx$

By substitution x' = x+k:
$\int_0^2 (1-h(x))h(x+k) dx = \int_k^{k+2} (1-h(x'-k))h(x') dx'$

This isn't exactly the same as the original unless we have periodic boundary conditions. So the flip might give a different C5 value. Let me just try it as a perturbation.

- C5(g) = max_k ∫ g(x)(1-g(x+k)) dx = max_k ∫ (1-h(x))h(x+k) dx

These are different in general (they correspond to different "directions" of the overlap). So the flip gives a different feasible point and potentially different C5.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    return t * (m / t + np.log(np.sum(np.exp(shifted)))) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += (1.0 - h[shift:])
            g[shift:] += (-h[:n - shift])
    else:
        s = -shift
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] += (-h[s:])
    return g

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

def subgrad_at_argmax(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    return _conv_grad_single(h, shift) * dx

def pgd_optimize(h_init, max_iter=50000, lr_init=0.005, smooth_t=None, momentum=0.9):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    use_smooth = smooth_t is not None
    for it in range(max_iter):
        if use_smooth:
            grad = soft_max_grad(h, smooth_t)
        else:
            grad = subgrad_at_argmax(h)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)
        if use_smooth:
            obj_cur = soft_max_obj(h, smooth_t)
            obj_new = soft_max_obj(h_new, smooth_t)
            if obj_new > obj_cur + 1e-12:
                lr = max(lr * 0.5, 1e-12)
            else:
                h = h_new
                lr = min(lr * 1.05, 0.1)
        else:
            if c5_new > best_c5 + 0.003:
                lr = max(lr * 0.5, 1e-12)
            else:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                lr = min(lr * 1.05, 0.1)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5

def coordinate_polish(h, max_sweeps=5):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            candidates = [0.0, 1.0, old + 0.15, old - 0.15, old + 0.05, old - 0.05]
            best_val = old
            best_c = best_c5
            for c in candidates:
                if c < 0.0 or c > 1.0:
                    continue
                h[j] = c
                h_bal = project_feasible(h, n)
                c_try = compute_c5(h_bal)
                if c_try < best_c:
                    best_c = c_try
                    best_val = h_bal
                    h = h_bal
                    improved = True
            if best_val != old and h[j] != best_val:
                h[j] = best_val
                h = project_feasible(h, n)
                best_c5 = compute_c5(h)
            else:
                h[j] = old
                h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(320, 0.5)

    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Annealed smooth optimization
    if tl() > 80:
        for t_val in [0.1, 0.05, 0.03, 0.01, 0.005]:
            if tl() < 30:
                break
            ht, c5t = pgd_optimize(best_h, max_iter=25000, lr_init=0.004,
                                   smooth_t=t_val, momentum=0.85)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 2: Hard subgradient with momentum
    if tl() > 60:
        ht, c5t = pgd_optimize(best_h, max_iter=60000, lr_init=0.002,
                               smooth_t=None, momentum=0.9)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Basin hopping with structured perturbations
    if tl() > 120:
        for hop in range(25):
            if tl() < 60:
                break
            strength = 0.04 + 0.04 * np.exp(-hop * 0.15)
            noise = rng.normal(0, strength, n)
            h_pert = project_feasible(best_h + noise, n)
            ht, c5t = pgd_optimize(h_pert, max_iter=15000, lr_init=0.003,
                                   smooth_t=0.02, momentum=0.85)
            ht, c5t2 = pgd_optimize(ht, max_iter=15000, lr_init=0.001,
                                    smooth_t=None, momentum=0.9)
            if c5t2 < best_c5:
                best_h, best_c5 = ht, c5t2
            # Try complement
            h_comp = project_feasible(1.0 - best_h, n)
            c5_comp = compute_c5(h_comp)
            if c5_comp < best_c5:
                best_h, best_c5 = h_comp, c5_comp
                ht, c5t = pgd_optimize(best_h, max_iter=20000, lr_init=0.002,
                                       smooth_t=None, momentum=0.9)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t

    # Phase 4: Coordinate polish
    if tl() > 40:
        hp, c5p = coordinate_polish(best_h, max_sweeps=8)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    # Phase 5: Fine-tuning with very small steps
    if tl() > 40:
        ht, c5t = pgd_optimize(best_h, max_iter=100000, lr_init=0.0003,
                               smooth_t=None, momentum=0.95)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Another coordinate polish pass
    if tl() > 20:
        hp, c5p = coordinate_polish(best_h, max_sweeps=10)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    # Phase 7: Try L-BFGS-B on smooth objective from current best
    if tl() > 60:
        for t_val in [0.005, 0.002]:
            if tl() < 30:
                break
            def obj_fn(hv, tv=t_val):
                return soft_max_obj(hv, tv) + 1000.0 * (np.sum(hv) - n / 2.0) ** 2
            def grad_fn(hv, tv=t_val):
                return soft_max_grad(hv, tv) + 2000.0 * (np.sum(hv) - n / 2.0)
            try:
                res = minimize(obj_fn, best_h, jac=grad_fn, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12})
                ht = project_feasible(res.x, n)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 8: Final polish
    if tl() > 20:
        ht, c5t = pgd_optimize(best_h, max_iter=50000, lr_init=0.0002,
                               smooth_t=None, momentum=0.95)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        hp, c5p = coordinate_polish(best_h, max_sweeps=5)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    return best_h, best_c5, len(best_h)
