```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(h, target):
    """Clip and adjust sum to target"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            room = 1.0 - h
            rsum = room.sum()
            if rsum < 1e-15:
                break
            h += d * room / rsum
        else:
            hsum = h.sum()
            if hsum < 1e-15:
                break
            h += d * h / hsum
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    f = (m + np.log(es.sum()) / alpha) + beta * (np.sum(h) - 0.5 * n) ** 2
    return f

def _smooth_max_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    w = es / es.sum()
    # Gradient of smooth max
    grad = np.zeros(n)
    nc = len(C)
    for k in range(nc):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        j1 = max(0, -s)
        j2 = min(n - 1, n - 1 - s)
        if j1 <= j2:
            grad[j1:j2+1] += w[k] * dx * (1.0 - h[j1+s:j2+s+1])
        j3 = max(0, s)
        j4 = min(n - 1, n - 1 + s)
        if j3 <= j4:
            grad[j3:j4+1] -= w[k] * dx * h[j3-s:j4-s+1]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return _smooth_max_obj(h, alpha, beta), grad

def _lbfgs(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_smooth_max_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    if res.x is not None:
        return _project(res.x, 0.5 * n)
    return h

def _subgrad(h, n, n_iters=200, step_scale=0.5):
    """Subgradient method with decreasing step"""
    h = h.copy().astype(float)
    target = 0.5 * n
    best_h = h.copy()
    best_c5 = _c5_max(h)
    dx = 2.0 / n
    
    for it in range(1, n_iters + 1):
        C = _c5_array(h)
        max_val = np.max(C)
        
        # Active set
        tol = max(1e-3, 0.01 * max_val)
        active = np.where(C >= max_val - tol)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            j1 = max(0, -s)
            j2 = min(n - 1, n - 1 - s)
            if j1 <= j2:
                grad[j1:j2+1] += (1.0 - h[j1+s:j2+s+1])
            j3 = max(0, s)
            j4 = min(n - 1, n - 1 + s)
            if j3 <= j4:
                grad[j3:j4+1] -= h[j3-s:j4-s+1]
        grad *= dx / max(len(active), 1)
        
        # Step
        step = step_scale / np.sqrt(it + 1.0)
        h_try = h - step * grad
        h_try = _project(h_try, target)
        c5_try = _c5_max(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h = h_try
    return best_h, best_c5

def _coord_descent(h, n, n_passes=10, grid_size=11):
    """Coordinate descent: for each j, try a few values of h[j]"""
    h = h.copy().astype(float)
    target = 0.5 * n
    best_c5 = _c5_max(h)
    grid = np.linspace(0, 1, grid_size)
    
    for p in range(n_passes):
        improved = False
        for j in range(n):
            orig = h[j]
            best_t = orig
            best_local = best_c5
            for t in grid:
                if abs(t - orig) < 1e-6:
                    continue
                h[j] = t
                h2 = _project(h, target)
                c5 = _c5_max(h2)
                if c5 < best_local:
                    best_local = c5
                    best_t = t
            h[j] = best_t
        h = _project(h, target)
        c5 = _c5_max(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            improved = True
        elif not improved and p > 2:
            break
    return h, best_c5

def _pairwise_search(h, n, n_iters=5000, rng=None):
    """Random pairwise swap search"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy().astype(float)
    target = 0.5 * n
    best_c5 = _c5_max(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = h[i] - h[j]
        if abs(d) < 1e-6:
            continue
        # Move some mass from higher to lower (or vice versa)
        delta = min(abs(d), 0.1) * rng.choice([-1, 1])
        h_try = h.copy()
        h_try[i] = np.clip(h[i] + delta, 0, 1)
        # Compensate j to maintain sum
        diff = h[i] - h_try[i]
```

```python
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            room = 1.0 - h
            r = room.sum()
            if r < 1e-15:
                break
            h += d * room / r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += d * h / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _om(h, alpha, beta):
    """Smooth max objective"""
    n = len(h)
    C = _c5_array(h)
    m = np.max(C)
    f = m + np.log(np.sum(np.exp(alpha * (C - m)))) / alpha
    f += beta * (np.sum(h) - 0.5 * n) ** 2
    return f

def _omg(h, alpha, beta):
    """Smooth max objective + gradient"""
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    w = es / es.sum()
    f = m + np.log(es.sum()) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    grad = np.zeros(n)
    nc = len(C)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j1 = max(0, -s)
        j2 = min(n - 1, n - 1 - s)
        if j1 <= j2:
            grad[j1:j2+1] += wk * dx * (1.0 - h[j1+s:j2+s+1])
        j3 = max(0, s)
        j4 = min(n - 1, n - 1 + s)
        if j3 <= j4:
            grad[j3:j4+1] -= wk * dx * h[j3-s:j4-s+1]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

        pass

    h = h.copy()
    target = 0.5 * n

            zeros = np.setdiff1d(zeros, j)
            zeros = np.append(zeros, i)
        else:
            h[i], h[j] = 1.0, 0.0
    return h, c5
```

```python
    """Update C = correlate(h, 1-h, 'full') when h[i] -> h[i]-1 and h[j] -> h[j]+1
    (i.e., flip i from 1 to 0, j from 0 to 1)
    Returns new C and new max."""
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            room = 1.0 - h
            r = room.sum()
            if r < 1e-15:
                break
            h += d * room / r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += d * h / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _omg(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = es.sum()
    w = es / se
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    grad = np.zeros(n)
    nc = len(C)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j1 = max(0, -s)
        j2 = min(n - 1, n - 1 - s)
        if j1 <= j2:
            grad[j1:j2+1] += wk * dx * (1.0 - h[j1+s:j2+s+1])
        j3 = max(0, s)
        j4 = min(n - 1, n - 1 + s)
        if j3 <= j4:
            grad[j3:j4+1] -= wk * dx * h[j3-s:j4-s+1]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, alpha, beta, mi=400):
    n = len(h)
    try:
        res = minimize(_omg, h, args=(alpha, beta), jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': mi, 'ftol': 1e-14, 'gtol': 1e-7})
        if res.x is not None:
            return _project(res.x, 0.5 * n)
    except:
        pass
    return h

    stall = 0
    for it in range(n_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc5 = _c5_max(h)
        if nc5 < c5 - 1e-12:
            c5 = nc5
            stall = 0
            ones = np.setdiff1d(ones, i)
            ones = np.append(ones, j)
            zeros = np.setdiff1d(zeros, j)
            zeros = np.append(zeros, i)
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
            if stall > 800:
                # Perturb: flip 2 random pairs
                if len(ones) >= 2 and len(zeros) >= 2:
                    i1 = ones[rng.randint(len(ones))]
                    i2 = ones[rng.randint(len(ones))]
                    j1 = zeros[rng.randint(len(zeros))]
                    j2 = zeros[rng.randint(len(zeros))]

                        ones = np.append(ones, [j1, j2])
                        zeros = np.setdiff1d(zeros, [j1, j2])
                        zeros = np.append(zeros, [i1, i2])
                        stall = 0
    return h, c5

def _subgrad_optimize(h, n, n_iters, step_scale):
    h = h.copy().astype(float)
    target = 0.5 * n
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5_max(h)
    for it in range(1, n_iters + 1):
        C = _c5_array(h)
        mv = np.max(C)
        tol = max(0.005, 0.005 * mv)
        active = np.where(C >= mv - tol)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            j1 = max(0, -s)
            j2 = min(n - 1, n - 1 - s)

            j3 = max(0, s)
            j4 = min(n - 1, n - 1 + s)
            if j3 <= j4:
                grad[j3:j4+1] -= h[j3-s:j4-s+1]
        grad *= dx / len(active)

            best_c5 = c5_new
            best_h = h_new.copy()
```
