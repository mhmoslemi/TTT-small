        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return w, grad

def _lbfgs(h, alpha, beta, maxiter=500):
    n = len(h)
    def fun(x):
        f = _softmax_val(x, alpha, beta)
        _, g = _softmax_grad(x, alpha, beta)
        return f, g
    res = minimize(fun, h, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 20,
                            'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
    if res.x is None:
        return h
    return _fix(res.x, n)

def _proj_grad(h, alpha, beta, eta=0.01, n_steps=200):
    n = len(h)
    best_f = _softmax_val(h, alpha, beta)
    for _ in range(n_steps):
        _, grad = _softmax_grad(h, alpha, beta)
        moved = False
        for te in [eta, eta * 0.5, eta * 0.25, eta * 0.1, eta * 0.05]:
            h_new = _fix(h - te * grad, n)
            f_new = _softmax_val(h_new, alpha, beta)
            if f_new < best_f - 1e-15:
                h, best_f, moved = h_new, f_new, True
                break
        if not moved:
            break
    return h

def _structured_inits(n, seed=0):
    rng = np.random.RandomState(seed)
    results = []
    h = np.zeros(n); h[:n // 2] = 1.0; results.append(h.copy())
    h = np.zeros(n); h[:n // 3] = 1.0; h[2 * n // 3:] = 1.0; results.append(h.copy())
    h = np.zeros(n); h[1:n // 2] = 1.0; results.append(h.copy())
    for period in [5, 7, 11, 13, 17, 23]:
        h = np.zeros(n); duty = period // 2
        for s in range(0, n, period):
            h[s:min(s + duty, n)] = 1.0
        results.append(h.copy())
    for _ in range(4):
        h = rng.binomial(1, 0.5, n)
        k = np.ones(3) / 3
        h = np.convolve(h, k, mode='same')
        results.append((h > 0.5).astype(float))
    return results

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    # Phase 1: Progressive optimization at initial n
    alpha_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    for alpha in alpha_schedule:
        if _el() > budget_s * 0.22:
            break
        beta = max(1.0, 0.02 * alpha)
        hb = _lbfgs(hb, alpha, beta, maxiter=1000)
        cn = _c5_max(hb)
        if cn < cb:
            cb = cn
        if _el() < budget_s * 0.22:
            hb = _proj_grad(hb, alpha, beta, eta=0.005, n_steps=100)
            cn = _c5_max(hb)
            if cn < cb:
                cb = cn

    # Phase 2: Try multiple n values
    n_cands = [40, 50, 55, 60, 65, 70, 75, 80, 85, 90, 100, 110, 120, 132, 150]
    n_cands = [n for n in n_cands if n != nb and 30 <= n <= 200]

    for n in n_cands:
        if _el() > budget_s * 0.52:
            break
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        ht = _fix(np.interp(x_new, x_old, hb), n)
        improved = False
        for alpha in [20, 50, 100, 300, 800]:
            if _el() > budget_s * 0.55:
                break
            beta = max(1.0, 0.02 * alpha)
            ht = _lbfgs(ht, alpha, beta, maxiter=700)
            ht = _proj_grad(ht, alpha, beta, eta=0.003, n_steps=50)
            cn = _c5_max(ht)
            if cn < cb - 1e-10:
                cb, hb, nb = cn, ht, n
                improved = True
                x_old = np.linspace(0, 2, nb, endpoint=False)
        if not improved and _el() < budget_s * 0.55:
            # Try structured init at this n
            for hi in _structured_inits(n, seed=seed + n)[:3]:
                if _el() > budget_s * 0.56:
                    break
                hi = _fix(hi.astype(float), n)
                for alpha in [50, 200]:
                    if _el() > budget_s * 0.57:
                        break
                    beta = max(1.0, 0.02 * alpha)
                    ht2 = _lbfgs(hi, alpha, beta, maxiter=400)
                    cn2 = _c5_max(ht2)
                    if cn2 < cb:
                        cb, hb, nb = cn2, ht2, n
                        hi = hb.copy()
                        x_old = np.linspace(0, 2, nb, endpoint=False)
                        break

    # Phase 3: Structured initializations at best n
    if _el() < budget_s * 0.70:
        for hi in _structured_inits(nb, seed=seed + 100):
            if _el() > budget_s * 0.73:
                break
            hi = _fix(hi.astype(float), nb)
            for alpha in [30, 100, 300, 800]:
                if _el() > budget_s * 0.75:
                    break
                beta = max(1.0, 0.02 * alpha)
                ht = _lbfgs(hi, alpha, beta, maxiter=500)
                ht = _proj_grad(ht, alpha, beta, eta=0.003, n_steps=30)
                cn = _c5_max(ht)
                if cn < cb:
                    cb, hb = cn, ht
                    hi = hb.copy()

    # Phase 4: Random restarts
    if _el() < budget_s * 0.85:
        n_restarts = 25
        for r in range(n_restarts):
            if _el() > budget_s * 0.87:
                break
            scale = 0.015 + 0.05 * rng.rand()
            ht = _fix(hb + rng.randn(nb) * scale, nb)
            for alpha in [50, 200, 600]:
                if _el() > budget_s * 0.89:
                    break
                beta = max(1.0, 0.02 * alpha)
                ht = _lbfgs(ht, alpha, beta, maxiter=400)
                cn = _c5_max(ht)
                if cn < cb:
                    cb, hb = cn, ht

    # Phase 5: Final intensive polish
    if _el() < budget_s * 0.96:
        for alpha in [2000, 5000, 10000, 50000]:
            if _el() > budget_s * 0.97:
                break
            beta = max(5.0, 0.01 * alpha)
            hn = _lbfgs(hb, alpha, beta, maxiter=2000)
            hn = _proj_grad(hn, alpha, beta, eta=0.001, n_steps=200)
            hn = _fix(hn, nb)
            cn = _c5_max(hn)
            if cn < cb:
                cb, hb = cn, hn

    # Phase 6: Coordinate descent on worst lags
    if _el() < budget_s * 0.99:
        C = _c5_array(hb)
        worst = np.argsort(C)[-10:]
        for k_idx in worst:
            if _el() > budget_s * 0.99:
                break
            # Try moving mass at positions that affect this lag
            n = nb
            dx = 2.0 / n
            k = k_idx - (n - 1)
            for _ in range(20):
                if _el() > budget_s * 0.995:
                    break
                best_delta = 0
                best_c5 = _c5_max(hb)
                for trial in range(40):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    for delta in [0.005, 0.01, 0.02, -0.005, -0.01]:
                        ht = hb.copy()
                        ht[i] = np.clip(ht[i] + delta, 0, 1)
                        ht[j] = np.clip(ht[j] - delta, 0, 1)
                        ht = _fix(ht, n)
                        cn = _c5_max(ht)
                        if cn < best_c5 - 1e-12:
                            hb, best_c5, best_delta = ht, cn, delta
                            if _c5_max(hb) < cb:
                                cb = best_c5
                            break
                    if best_delta != 0:
                        break
        hb = _fix(hb, nb)
        cb = _c5_max(hb)

    return (hb, cb, nb)
