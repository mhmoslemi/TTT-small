def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        if m.sum() > 0:
            h[m] = np.clip(h[m] + diff / m.sum(), 0.0, 1.0)
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    N = len(h)
    return compute_corr(h).max() * 2.0 / N


def grad_single_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    if lag >= 0:
        e = min(N, N - lag)
        if e > 0:
            g[:e] += 1.0 - h[lag:lag + e]
    else:
        s = -lag
        if s < N:
            g[s:] -= h[:N - s]
    return g


def soft_gradient(h, corr, dx, beta):
    N = len(h)
    s = corr - corr.max()
    mx = np.max(s)
    w = np.exp(beta * (s - mx))
    ws = w.sum()

        k = np.argmax(corr)
        return grad_single_lag(h, k - (N - 1)) * dx
    w /= ws
    g = np.zeros(N)

        if w[j] < 1e-6:
            continue
        lag = j - (N - 1)
        g += w[j] * grad_single_lag(h, lag)
    return g * dx
