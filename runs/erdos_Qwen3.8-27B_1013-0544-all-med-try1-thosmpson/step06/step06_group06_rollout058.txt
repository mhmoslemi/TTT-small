- Indirect effect (h[i] appears as the second factor, i.e., n+k = i, so n = i-k): for each k where 0 <= i-k < n, c[k+n-1] gets -h[i-k]*t

If h[j] changes by -t:
- Direct effect: for each k where 0 <= j+k < n, c[k+n-1] gets -t*(1-h[j+k])
- Indirect effect: for each k where 0 <= j-k < n, c[k+n-1] gets +h[j-k]*t

So d[k+n-1] = (1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]

def compute_direction(h, n, i, j):
    """Compute d[m] = dc[m]/dt for the move h[i]+=t, h[j]-=t."""
    d = np.zeros(2*n - 1)
    offset = n - 1
    
    # h[i] += t effects:
    # Direct: for k where 0 <= i+k < n (i.e., k from -i to n-1-i)
    for k in range(-i, n - i):
        m = k + offset
        d[m] += (1.0 - h[i+k])
    # Indirect: for k where 0 <= i-k < n (i.e., k from i-n+1 to i)
    for k in range(i - n + 1, i + 1):
        m = k + offset
        d[m] -= h[i-k]
    
    # h[j] -= t effects:
    # Direct: for k where 0 <= j+k < n
    for k in range(-j, n - j):
        m = k + offset
        d[m] -= (1.0 - h[j+k])
    # Indirect: for k where 0 <= j-k < n
    for k in range(j - n + 1, j + 1):
        m = k + offset
        d[m] += h[j-k]
    
    return d
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(0.0, 1.0 - h)
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_fun(h):
    return h.sum() - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def compute_direction(h, n, i, j):
    d = np.zeros(2 * n - 1)
    offset = n - 1
    for k in range(-i, n - i):
        d[k + offset] += (1.0 - h[i + k])
    for k in range(max(0, i - n + 1), i + 1):
        d[k + offset] -= h[i - k] if i - k >= 0 else 0.0
    for k in range(-j, n - j):
        d[k + offset] -= (1.0 - h[j + k])
    for k in range(max(0, j - n + 1), j + 1):
        d[k + offset] += h[j - k] if j - k >= 0 else 0.0
    return d

def find_optimal_t(c, d, t_min, t_max):
    if t_max - t_min < 1e-12:
        return 0.0
    a, b = t_min, t_max
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(c + m1 * d) < np.max(c + m2 * d):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def compute_direction(h, n, i, j):
    d = np.zeros(2 * n - 1)
    offset = n - 1
    # For h[i] += t:
    # Term 1: (1-h[i+L]) for L where 0 <= i+L < n → L from -i to n-1-i → m from offset-i to offset+n-1-i
    lo1 = max(0, offset - i)
    hi1 = min(2*n-2, offset + n - 1 - i)
    for m in range(lo1, hi1 + 1):
        L = m - offset
        d[m] += (1.0 - h[i + L])
    
    # Term 2: -h[i-L] for L where 0 <= i-L < n → L from i-n+1 to i → m from offset+i-n+1 to offset+i
    lo2 = max(0, offset + i - n + 1)
    hi2 = min(2*n-2, offset + i)
    for m in range(lo2, hi2 + 1):
        L = m - offset
        d[m] -= h[i - L]
    
    # For h[j] -= t:
    # Term 3: -(1-h[j+L]) for L where 0 <= j+L < n
    lo3 = max(0, offset - j)
    hi3 = min(2*n-2, offset + n - 1 - j)
    for m in range(lo3, hi3 + 1):
        L = m - offset
        d[m] -= (1.0 - h[j + L])
    
    # Term 4: +h[j-L] for L where 0 <= j-L < n
    lo4 = max(0, offset + j - n + 1)
    hi4 = min(2*n-2, offset + j)
    for m in range(lo4, hi4 + 1):
        L = m - offset
        d[m] += h[j - L]
    
    return d
```
