            if elapsed() > budget_s * 0.7:
                break
            # Different perturbation strategies
            strategy = r % 3
            if strategy == 0:
                scale = 0.03 + 0.02 * (r % 5)
                h_pert = best_h + rng.normal(0, scale, size=n)
            elif strategy == 1:
                # Targeted: shift mass from high to low correlation regions
                h_pert = best_h.copy()
                for _ in range(5):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i != j:
                        d = rng.uniform(0.01, 0.08)
                        if h_pert[i] >= d and h_pert[j] + d <= 1:
                            h_pert[i] -= d
                            h_pert[j] += d
            else:
                # Binary perturbation
                h_pert = best_h.copy()
                nb = rng.randint(n // 4 + 1)
                for _ in range(nb):
                    idx = rng.randint(n)
                    h_pert[idx] = rng.choice([0.0, 1.0, 0.5])
            h_pert = project_feasible(h_pert, n)
            try:
                res = minimize(obj_active, h_pert, args=(n, 2000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-16})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-15:
                    best_h, best_c5 = h_try, c
                    # If improved, do quick 2-point polish
                    if elapsed() < budget_s * 0.7:
                        best_h, best_c5 = two_point_search(best_h, n, rng, n_trials=500)
            except Exception:
                pass

    # Phase 5: Fine 2-point search with smaller deltas
    if elapsed() < budget_s * 0.85:
        max_iters = int((budget_s * 0.85 - elapsed()) / 0.003)
        max_iters = max(max_iters, 1000)
        best_h, best_c5 = two_point_search(best_h, n, rng, n_trials=max_iters)

    # Phase 6: Final SLSQP polish with very high alpha
    if elapsed() < budget_s - 10:
        for alpha in [5000, 20000]:
            if elapsed() > budget_s - 5:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(n, alpha, 0), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 10000, 'ftol': 1e-18})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-15:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Final 2-point polish
    if elapsed() < budget_s - 3:
        best_h, best_c5 = two_point_search(best_h, n, rng, n_trials=3000)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
