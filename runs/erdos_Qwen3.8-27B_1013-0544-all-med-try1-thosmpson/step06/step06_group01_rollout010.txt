def _lbfgs_optimize(h, n, alpha, beta, t0, deadline, maxiter=500):
    def _obj(h_in):
        C = _c5_array(h_in)
        m = np.max(C)
        sh = alpha * C
        es = np.exp(sh - m)
        se = np.sum(es)
        f = (m + np.log(se)) / alpha + beta * (np.sum(h_in) - 0.5 * n) ** 2
        return f
    
    def _obj_grad(h_in):
        C = _c5_array(h_in)
        m = np.max(C)
        sh = alpha * C
        es = np.exp(sh - m)
        se = np.sum(es)
        f = (m + np.log(se)) / alpha + beta * (np.sum(h_in) - 0.5 * n) ** 2
        w = es / se
        g = np.zeros(n)

def _coord_descent(h, n, t0, deadline, max_sweeps=200):
    h = h.copy()
    best_c5 = _c5_max(h)
    best_h = h.copy()
    nc = 2 * n - 1
    dx = 2.0 / n
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        C = _c5_array(h)
        Cmax = float(np.max(C))
        if Cmax < best_c5 - 1e-12:
            best_c5 = Cmax
            best_h = h.copy()
        # Compute Jacobian
        J = np.zeros((nc, n))
        for k in range(nc):
            lag = k - (n - 1)
            if lag == 0:
                J[k, :] = (1.0 - h) * dx
            elif lag > 0:
                J[k, lag:] = (1.0 - h[:n - lag]) * dx
                J[k, :n - lag] -= h[lag:] * dx
            else:
                lag = -lag
                J[k, :n - lag] = (1.0 - h[lag:]) * dx
                J[k, lag:] -= h[:n - lag] * dx
        improved = False
        for d_val in deltas:
            if time.time() > deadline:
                break
            for j in range(n):
                if h[j] < d_val:
                    continue
                h[j] -= d_val
                best_k = -1
                best_c5_try = Cmax
                for k_idx in range(n):
                    if k_idx == j or h[k_idx] + d_val > 1.0:
                        continue
                    h[k_idx] += d_val
                    c5_try = _c5_max(h)
                    if c5_try < best_c5_try:
                        best_c5_try = c5_try
                        best_k = k_idx
                    h[k_idx] -= d_val
                if best_k >= 0:
                    h[best_k] += d_val
                    improved = True
                    break
                else:
                    h[j] += d_val
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5
