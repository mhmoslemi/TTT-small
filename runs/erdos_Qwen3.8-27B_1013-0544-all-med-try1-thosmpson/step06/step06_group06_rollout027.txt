    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth_max(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    # numerically stable logsumexp
    smax = s.max()
    if smax > 350:
        lse = smax + np.log(np.sum(np.exp(s - smax)))
    else:
        lse = np.log(np.sum(np.exp(s)))
    return float(m + lse / alpha)

def obj_topk_max(x, n, k, alpha):
    """Smooth approximation of max of top-k values."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    # Sort in descending order
    c_sorted = np.sort(c)[::-1]
    topk = c_sorted[:k]
    m = topk.max()
    s = alpha * (topk - m)
    smax = s.max()
    lse = smax + np.log(np.sum(np.exp(s - smax)))

    # Actually, use: mean of top-k smoothed
    # A better smooth: (sum of top-k) / k, smoothed
    # For now, just use smooth max of top-k
    return float(m + lse / alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def obj_smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    exp_s = np.exp(s - smax)
    w = exp_s / exp_s.sum()
    lse = smax + np.log(exp_s.sum())
    f = m + lse / alpha
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-16:
            continue
        sj = j - (n - 1)
        if sj >= 0:
            term1[:n - sj] += w[j] * x[sj:]
            term2[sj:] += w[j] * x[:n - sj]
        else:
            term1[-sj:] += w[j] * x[:n + sj]
            term2[:n + sj] += w[j] * x[-sj:]
    grad = (1.0 / alpha) * (1.0 - term1 - term2)
    return f, grad

def obj_smooth_only(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    exp_s = np.exp(s - smax)
    lse = smax + np.log(exp_s.sum())
    return float(m + lse / alpha)

def subgradient_step(h, n, eta, n_worst=10):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-n_worst:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / n_worst
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for j in top_idx:
        sj = j - (n - 1)
        if sj >= 0:
            term1[:n - sj] += w[j] * h[sj:]
            term2[sj:] += w[j] * h[:n - sj]
        else:
            term1[-sj:] += w[j] * h[:n + sj]
            term2[:n + sj] += w[j] * h[-sj:]
    grad = (1.0 - term1 - term2)
    h_new = h - eta * grad
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

    def try_slsqp(h_init, alpha, maxiter=2000):
        try:
            res = minimize(obj_smooth_grad, h_init, args=(n, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            return h_try, c
        except Exception:
            return h_init, compute_c5(h_init, n)

    # Phase 1: Progressive SLSQP with analytical gradient

            j = rng.randint(n)
            if i == j:
                continue

            for s in [1, -1]:
                hi = h_ls[i] + s * d
                hj = h_ls[j] - s * d
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h_ls[i] = hi
                    h_ls[j] = hj
                    c = compute_c5(h_ls, n)
                    if c < c5_ls - 1e-15:
                        c5_ls = c
                        if c5_ls < best_c5:
                            best_h = h_ls.copy()
                            best_c5 = c5_ls
                    else:
                        h_ls[i] -= s * d
                        h_ls[j] += s * d
                        break

    if remaining() > 10:
        for alpha in [50000, 100000, 200000]:
            if remaining() < 8:

            h_try, c = try_slsqp(best_h, alpha, maxiter=5000)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c

    # Final feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
