```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)

        w = weights[idx]
        if w < 1e-12:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

    target = n / 2.0
    lo, hi = -2.0, 2.0

        if s > target:

        else:

    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def nesterov_smooth(h_init, n, t, max_iter, lr_init):
    """Nesterov-accelerated gradient descent on soft-max objective."""
    h = h_init.copy()
    y = h.copy()  # Nesterov momentum variable
    beta = 0.9  # momentum
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iter):
        # Evaluate gradient at y (Nesterov lookahead)
        y_proj = project_feasible(y, n)
        obj_val, grad = soft_max_obj(y_proj, t), soft_max_grad(y_proj, t)
        
        # Update

        h_new = project_feasible(h_new, n)
        
        # Nesterov update for y
        beta_new = (1 + np.sqrt(1 + 4 * beta * beta)) / 2
        y = h_new + (beta_new / beta - 1) * (h_new - h)
        
        beta = beta_new
        
        # Check improvement
        c5 = compute_c5(h_new)
        if c5 < best_c5:

            best_h = h_new.copy()

            lr *= 0.8
            if lr < 1e-15:
                break
        else:
            lr = min(lr * 1.02, 0.05)
        
        h = h_new

def coordinate_descent(h, n, max_sweeps=50, tol=1e-10):
    """Coordinate descent: for each coordinate, try 0, 1, and current value."""
    h = h.copy()
    current_c5 = compute_c5(h)
```

Now, for the Nesterov method, I need to be careful about the feasible set. Standard Nesterov works on unconstrained problems. For constrained problems, I can use the "projected Nesterov" variant:

y_k = x_k + beta_k (x_k - x_{k-1})  [momentum step]
x_{k+1} = proj(y_k - lr * grad(proj(y_k)))  [gradient step with projection]

But this can have issues. A simpler approach is to apply Nesterov momentum on the projected space:

x_{k+1} = proj(x_k + momentum * (x_k - x_{k-1}) - lr * grad(x_k))

This is the "heavy ball" method projected onto the feasible set.

Let me refine the Nesterov implementation:

```python
def soft_max_grad_fast(h, t, threshold=1e-8):

    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)

    significant = weights > threshold
    for idx in np.where(significant)[0]:
        w = weights[idx]
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t, threshold=1e-8):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    sig = np.where(weights > threshold)[0]
    for idx in sig:
        shift = idx - (n - 1)
        grad += weights[idx] * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def nesterov_smooth(h_init, n, t, max_iter, lr_init, momentum_init=0.9):
    h = h_init.copy()
    h_prev = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    momentum = momentum_init
    for it in range(max_iter):
        grad = soft_max_grad(h, t)
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_feasible(h_new, n)
        h_prev = h.copy()
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Adaptive: compare objective values
        obj_new = soft_max_obj(h, t)
        obj_old = soft_max_obj(h_prev, t)
        if obj_new > obj_old + 1e-13:
            lr *= 0.6
            if lr < 1e-16:
                break
        else:
            lr = min(lr * 1.015, 0.04)
        # Decay momentum slightly
        momentum = min(0.99, momentum * 1.001)
    return best_h, best_c5

def subgrad_pgd(h_init, n, max_iter, lr_init):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        c5, grad = c5_subgrad(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-14:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.04, 0.02)
    return best_h, best_c5

def topk_subgrad_pgd(h_init, n, max_iter, lr_init, k=5):
    """Use top-k subgradients averaged."""
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        dx = 2.0 / n
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Top-k shifts with softmax weighting
        top_idx = np.argsort(conv)[-k:]
        m = conv[top_idx].max()
        w = np.exp((conv[top_idx] - m) / 0.01)
        w = w / w.sum()
        grad = np.zeros(n)
        for i, idx in enumerate(top_idx):
            shift = idx - (n - 1)
            grad += w[i] * _conv_grad_single(h, shift)
        h_new = project_feasible(h - lr * grad * dx, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-14:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.03, 0.015)
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def diverse_init(n, rng, idx):
    if idx % 5 == 0:
        return rng.uniform(0, 1, n)
    elif idx % 5 == 1:
        freq = max(2, rng.randint(2, n // 4 + 1))
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        return np.clip(h, 0, 1)
    elif idx % 5 == 2:
        bs = max(2, rng.randint(2, n // 6 + 1))
        h = np.zeros(n)
        h[0::2 * bs:bs] = 1.0
        return h
    elif idx % 5 == 3:
        f1 = max(1, rng.randint(1, n // 10 + 1))
        f2 = max(1, rng.randint(n // 10, n // 3 + 1))
        p1, p2 = rng.uniform(0, 2 * np.pi, 2)
        h = 0.5 + 0.2 * np.sin(2 * np.pi * f1 * np.arange(n) / n + p1)
        h += 0.2 * np.sin(2 * np.pi * f2 * np.arange(n) / n + p2)
        return np.clip(h, 0, 1)
    else:
        # Sawtooth-like
        period = max(3, rng.randint(3, n // 5 + 1))
        duty = rng.uniform(0.3, 0.7)
        h = np.zeros(n)
        phase = rng.uniform(0, period)
        for i in range(n):
            pos = (i + phase) % period
            h[i] = 1.0 if pos < duty * period else 0.0
        return h

def lbfgs_optimize(h_init, n, t, max_iter=500):
    def obj(h):
        return soft_max_obj(h, t) + 100.0 * (np.sum(h) - n / 2.0) ** 2
    def grad(h):
        return soft_max_grad(h, t) + 200.0 * (np.sum(h) - n / 2.0)
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h_init
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def fast_correlate(a, b):
    return fftconvolve(a, b[::-1], mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = fast_correlate(h, 1.0 - h)
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t, threshold=1e-10):
    n = len(h)
    dx = 2.0 / n
    conv = fast_correlate(h, 1.0 - h)
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    sig = np.where(weights > threshold)[0]
    for idx in sig:
        shift = int(idx) - (n - 1)
        grad += weights[idx] * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def nesterov_smooth(h_init, n, t, max_iter, lr_init, momentum_init=0.85):
    h = h_init.copy()
    h_prev = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    momentum = momentum_init
    for it in range(max_iter):
        grad = soft_max_grad(h, t)
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_feasible(h_new, n)
        h_prev = h.copy()
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        obj_new = soft_max_obj(h, t)
        obj_old = soft_max_obj(h_prev, t)
        if obj_new > obj_old + 1e-14:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            lr = min(lr * 1.02, 0.03)
        momentum = min(0.98, momentum + 0.0005)
    return best_h, best_c5

def subgrad_pgd(h_init, n, max_iter, lr_init):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        c5, grad = c5_subgrad(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-14:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.03, 0.015)
    return best_h, best_c5

def topk_subgrad_pgd(h_init, n, max_iter, lr_init, k=5):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        dx = 2.0 / n
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        top_idx = np.argsort(conv)[-k:]
        m = conv[top_idx].max()
        w = np.exp((conv[top_idx] - m) / 0.005)
        w = w / w.sum()
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            shift = int(top_idx[i]) - (n - 1)
            grad += w[i] * _conv_grad_single(h, shift)
        h_new = project_feasible(h - lr * grad * dx, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-14:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.02, 0.01)
    return best_h, best_c5

def lbfgs_optimize(h_init, n, t, max_iter=300):
    def obj_fc(h):
        return soft_max_obj(h, t) + 200.0 * (np.sum(h) - n / 2.0) ** 2
    def grad_fc(h):
        return soft_max_grad(h, t) + 400.0 * (np.sum(h) - n / 2.0)
    try:
        res = minimize(obj_fc, h_init, jac=grad_fc, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h_init.copy()

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def diverse_init(n, rng, idx):
    if idx % 5 == 0:
        return rng.uniform(0, 1, n)
    elif idx % 5 == 1:
        freq = max(2, rng.randint(2, n // 4 + 1))
        phase = rng.uniform(0, 2 * np.pi)
        arr = np.arange(n)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * arr / n + phase)
        return np.clip(h, 0, 1)
    elif idx % 5 == 2:
        bs = max(2, rng.randint(2, n // 6 + 1))
        h = np.zeros(n)
        for start in range(0, n, 2 * bs):
            h[start:start + bs] = 1.0
        return h
    elif idx % 5 == 3:
        f1 = max(1, rng.randint(1, n // 10 + 1))
        f2 = max(2, n // 10 + rng.randint(0, n // 6))
        p1, p2 = rng.uniform(0, 2 * np.pi, 2)
        arr = np.arange(n)
        h = 0.5 + 0.2 * np.sin(2 * np.pi * f1 * arr / n + p1)
        h += 0.2 * np.sin(2 * np.pi * f2 * arr / n + p2)
        return np.clip(h, 0, 1)
    else:
        period = max(3, rng.randint(3, n // 5 + 1))
        duty = rng.uniform(0.3, 0.7)
        phase = rng.uniform(0, period)
        h = np.zeros(n)
        for i in range(n):
            pos = (i + phase) % period
            if pos < duty * period:
                h[i] = 1.0
        return h

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
```
