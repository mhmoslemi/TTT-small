import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        excess = n_ones - target
        # Turn off the ones that are most "redundant" (e.g., in a cluster)
        # Simple approach: turn off random ones
        to_flip = ones[:excess]
        h[to_flip] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        deficit = target - n_ones
        to_flip = zeros[:deficit]
        h[to_flip] = 1
    return h

def full_local_search(h, max_time=30.0, seed=42):
    """Exhaustive single-swap local search."""
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 1e-15
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            current_c5 = compute_c5(h)
        else:
            break
    
    return h, current_c5

def guided_perturbation(h, n_max_lags=5, n_swaps=3):
    """Perturb h by targeting the worst lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find the lags with highest correlation
    # corr has length 2n-1, index n-1 is lag 0
    top_lags = np.argsort(corr)[-n_max_lags:]
    
    rng = np.random.RandomState(int(sum(h) * 1000))  # pseudo-random seed
    
    h_new = h.copy()
    for lag_idx in top_lags:
        lag = lag_idx - (n - 1)
        # At this lag, find positions where h[i]=1 and h[i+lag]=0
        # (or h[i]=0 and h[i+lag]=1, depending on the sign of lag)
        if lag >= 0:
            # result[lag_idx] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # We want to reduce this: find i where h[i+lag]=1 and h[i]=0
            # and swap h[i+lag] with h[i]
            candidates = []
            for i in range(n - lag):
                if h_new[i + lag] == 1 and h_new[i] == 0:
                    candidates.append(i)
        else:
            # result[lag_idx] = sum_{i=-lag}^{n-1} h[i+lag]*(1-h[i])
            # i+lag >= 0, so i >= -lag
            m = -lag
            candidates = []
            for i in range(m, n):
                if h_new[i - m] == 1 and h_new[i] == 0:
                    candidates.append(i)
        
        if candidates:
            i = rng.choice(candidates)
            if lag >= 0:
                h_new[i + lag] = 0
                h_new[i] = 1
            else:
                h_new[i - abs(lag)] = 0
                h_new[i] = 1
    
    return h_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[ones[target:]] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - n_ones]] = 1.0
    return h

def single_swap_ls(h, max_time=5.0):
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 1e-15
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain; best_i, best_j = i, j
                h[i] = 1.0; h[j] = 0.0
        
        if best_i >= 0:
            h[best_i] = 0.0; h[best_j] = 1.0
            current_c5 = compute_c5(h)
        else:
            break
    
    return h, current_c5

def double_swap_ls(h, max_time=5.0, seed=42):
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)

        zeros = np.where(h == 0)[0]

            j1, j2 = zeros[j1], zeros[j2]

            new_c5 = compute_c5(h)

                best_move = (i1, i2, j1, j2)

        for _ in range(n_samples):
            i1 = rng.randint(n_o); i2 = rng.randint(n_o)
            j1 = rng.randint(n_z); j2 = rng.randint(n_z)
            if i1 == i2 or j1 == j2:
                continue
            a, b = ones[i1], ones[i2]
            c, d = zeros[j1], zeros[j2]
            h[a] = 0; h[b] = 0; h[c] = 1; h[d] = 1
            new_c5 = compute_c5(h)
            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain; best_move = (a, b, c, d)
            h[a] = 1; h[b] = 1; h[c] = 0; h[d] = 0
        if best_move is not None:
            a, b, c, d = best_move
            h[a] = 0; h[b] = 0; h[c] = 1; h[d] = 1
            current_c5 = compute_c5(h)
        else:
            break
    return h, current_c5

def guided_perturb(h, rng, strength=3):
    n = len(h)
    h_new = h.copy()
    corr = np.correlate(h_new, 1.0 - h_new, mode='full')
    top_k = min(strength + 2, len(corr))
    top_indices = np.argsort(corr)[-top_k:]
    for idx in top_indices:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            candidates = [i for i in range(n - lag) if h_new[i + lag] == 1 and h_new[i] == 0]
        else:
            m = -lag
            candidates = [i for i in range(m, n) if h_new[i] == 1 and h_new[i - m] == 0]
        if candidates and rng.random() < 0.7:
            i = int(rng.choice(candidates))
            if lag >= 0:
                h_new[i + lag] = 0.0; h_new[i] = 1.0
            else:
                h_new[i] = 0.0; h_new[i + lag] = 1.0
    remaining_swaps = strength - top_k + strength
    for _ in range(max(0, strength - len(top_indices))):
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h_new[int(rng.choice(ones))] = 0.0
        h_new[int(rng.choice(zeros))] = 1.0
    return h_new

def iterated_ls(h0, max_time=100.0, seed=42):
    n = len(h0)
    best_h = h0.astype(float).copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    start = time.time()
    rng = np.random.RandomState(seed)
    stall_count = 0
    while (time.time() - start) < max_time:
        current_h, current_c5 = single_swap_ls(current_h, max_time=1.5)
        if current_c5 < best_c5 - 1e-12:
            best_c5 = current_c5
            best_h = current_h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if (time.time() - start) < max_time - 5.0:
                current_h, current_c5 = double_swap_ls(current_h, max_time=2.0, seed=seed + stall_count)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    stall_count = 0
            if stall_count >= 2:
                pert_strength = min(2 + stall_count // 2, 8)
                current_h = guided_perturb(best_h, rng, pert_strength)
                current_h, current_c5 = single_swap_ls(current_h, max_time=1.5)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    stall_count = 0
    return best_h, best_c5

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return make_binary(h_new, n_new)

def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    positions = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def remaining():
        return budget_s - (time.time() - t0)
    
    h_init = np.asarray(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_bin = make_binary(h_init, n_init)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: ILS on initial construction
    if remaining() > 100:
        t1 = min(250.0, remaining() * 0.35)
        h1, c5_1 = iterated_ls(h_bin, max_time=t1, seed=seed)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: Try different resolutions
    resolutions = [56, 70, 84, 98, 126, 168, 224]
    for n in resolutions:
        if remaining() < 100:
            break
        if n % 2 != 0:
            continue
        h_interp = interpolate_to_n(best_h, n)
        t_i = min(150.0, remaining() * 0.25)
        if t_i < 15:
            break
        h_i, c5_i = iterated_ls(h_interp, max_time=t_i, seed=seed + 1000 + n)
        if c5_i < best_c5:
            best_c5 = c5_i
            best_h = h_i.copy()
    
    # Phase 3: Try random starts at the best resolution
    best_n = len(best_h)
    n_random_starts = 3
    for rs in range(n_random_starts):
        if remaining() < 80:
            break
        h_rand = random_binary(best_n, seed=seed + 5000 + rs * 7)
        t_r = min(80.0, remaining() * 0.2)
        if t_r < 15:
            break
        h_r, c5_r = iterated_ls(h_rand, max_time=t_r, seed=seed + 6000 + rs)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 4: Final polish on best
    if remaining() > 40:
        t_f = remaining() * 0.85
        h_f, c5_f = iterated_ls(best_h, max_time=t_f, seed=seed + 9999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
