def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    log_sum = np.log(np.sum(np.exp(shifted - mx))) + mx

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    log_sum = np.log(np.sum(np.exp(shifted - mx))) + mx
    return m + log_sum / alpha

def subgradient_step(h, n, top_m):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:]) * dx
            g[lag:] -= h[:n - lag] * dx
        else:
            L = -lag
            g[L:] += (1.0 - h[:n - L]) * dx
            g[:n - L] -= h[L:] * dx
    g /= float(len(top_idx))
    return g, c.max()

def sum_con_fun(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 80
        h0 = np.full(n0, 0.5)

    n = 200
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h_interp = np.interp(x_new, x0, h0)
    h_start = project_feasible(h_interp, n)

    best_h = h_start.copy()
    best_c5 = compute_c5_val(best_h, n)

    # Try alternative starting points
    alts = [
        np.full(n, 0.5),
        np.where(np.arange(n) < n // 2, 1.0, 0.0),
        0.5 + 0.4 * np.sin(np.pi * np.arange(n) / n),
        0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) / n),
    ]
    for alt in alts:
        a = project_feasible(alt.copy(), n)
        c = compute_c5_val(a, n)
        if c < best_c5:
            best_h, best_c5 = a.copy(), c

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: Subgradient descent with momentum (broad exploration)
    h_sg = best_h.copy()
    v = np.zeros(n)
    mom = 0.85
    eta0 = 0.012
    max_iters = 6000
    stall = 0
    for t in range(max_iters):
        if budget_s - (time.time() - t0) < 180:
            break
        g, c_val = subgradient_step(h_sg, n, 5)
        eta = eta0 / (1.0 + t * 2.0 / max_iters)
        v = mom * v + (1.0 - mom) * g
        h_sg = project_feasible(h_sg - eta * v, n)
        if t % 50 == 0:
            c_check = compute_c5_val(h_sg, n)
            if c_check < best_c5 - 1e-10:
                best_c5 = c_check
                best_h = h_sg.copy()
                stall = 0
            else:
                stall += 1
        if stall > 40 and t > 1000:
            # Restart with perturbation
            h_sg = project_feasible(h_sg + rng.normal(0, 0.1, n), n)
            v = np.zeros(n)
            stall = 0

    c_sg = compute_c5_val(h_sg, n)
    if c_sg < best_c5:
        best_c5, best_h = c_sg, h_sg.copy()

    # Phase 2: SLSQP polishing with increasing alpha
    alphas = [30, 100, 500, 2000, 10000]
    for alpha in alphas:
        if budget_s - (time.time() - t0) < 50:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_try = project_feasible(res.x, n)
            c = compute_c5_val(h_try, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 3: Multi-start with subgradient + SLSQP
    for r in range(5):
        if budget_s - (time.time() - t0) < 120:
            break
        scale = 0.08 + 0.06 * r
        h_pert = project_feasible(best_h + rng.normal(0, scale, n), n)

        v2 = np.zeros(n)
        for t in range(400):
            if budget_s - (time.time() - t0) < 50:
                break
            g, _ = subgradient_step(h_pert, n, 3)
            eta = 0.006 / (1.0 + t / 150.0)
            v2 = 0.8 * v2 + 0.2 * g
            h_pert = project_feasible(h_pert - eta * v2, n)

        c_pert = compute_c5_val(h_pert, n)
        if c_pert < best_c5:
            best_h, best_c5 = h_pert.copy(), c_pert

        if budget_s - (time.time() - t0) < 25:
            break
        try:
            res = minimize(smooth_obj, h_pert, args=(n, 5000), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-15})
            h_try = project_feasible(res.x, n)
            c = compute_c5_val(h_try, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 4: Fine 2-point optimization
    if budget_s - (time.time() - t0) > 25:
        deltas = [0.001, 0.002, 0.004, 0.008, 0.015, 0.03]
        count = 0
        while budget_s - (time.time() - t0) > 10 and count < 60000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                improved = False
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5_val(h_try, n)
                        if c < best_c5 - 1e-15:
                            best_h = h_try
                            best_c5 = c
                            improved = True
                            break
                if improved:
                    break

    # Final high-precision SLSQP
    if budget_s - (time.time() - t0) > 15:
        try:
            res = minimize(smooth_obj, best_h, args=(n, 100000), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5_val(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5_val(best_h, n)
    return best_h, best_c5, n
