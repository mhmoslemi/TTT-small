    n = len(h)

    # At lambda = +inf: sum = 0

        s = projected.sum()

            lo = mid
        else:
            hi = mid

from scipy.special import logsumexp
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlations(h)) * dx)

def project_box_sum(h, S):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(c, mu):
    obj = mu * logsumexp(c / mu)
    weights = np.exp(c / mu) / np.sum(np.exp(c / mu))
    return obj, weights

def grad_correlations(h, weights):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    grad = np.zeros(n)
    m = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        shift = k - m

        elif shift > 0:

            grad[shift:] += w * (-h[:n - shift])
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] += w * (-h[s:])
    return grad

    n = len(h)

        c = compute_correlations(h_local)
        sm, _ = smooth_max_and_grad(c, mu)
        return sm + rho * (h_local.sum() - S) ** 2

        h_local = np.clip(x, 0.0, 1.0)
        c = compute_correlations(h_local)
        _, weights = smooth_max_and_grad(c, mu)
        g = grad_correlations(h_local, weights)
        g += 2.0 * rho * (h_local.sum() - S) * np.ones(n)
        return g

                      bounds=[(0, 1)] * n,
                      options={'maxiter': max_iter, 'maxfun': 3 * max_iter, 'gtol': 1e-5})

Let me also think about whether the gradient computation is correct.

For shift k > 0:
c(k) = sum_{i=0}^{n-1-k} x[i](1 - x[i+k])

d/dx[j] c(k):
- If j is in [0, n-1-k]: x[j] appears in term i=j as x[j](1-x[j+k]), derivative is (1-x[j+k])
- If j is in [k, n-1]: x[j] appears in term i=j-k as x[j-k](1-x[j]), derivative is -x[j-k]

So:
d/dx[j] c(k) = (1 - x[j+k]) * 1[j ≤ n-1-k] + (-x[j-k]) * 1[j ≥ k]

In vectorized form:

grad_shift_k[:n-k] += 1.0 - x[k:]  # (1-x[j+k]) for j in [0, n-1-k]
grad_shift_k[k:] += -x[:n-k]       # (-x[j-k]) for j in [k, n-1]

For shift -k (k > 0):
c(-k) = sum_{i=k}^{n-1} x[i](1 - x[i-k])

d/dx[j] c(-k):
- If j is in [k, n-1]: x[j] appears in term i=j as x[j](1-x[j-k]), derivative is (1-x[j-k])
- If j is in [0, n-1-k]: x[j] appears in term i=j+k as x[j+k](1-x[j]), derivative is -x[j+k]

So:
d/dx[j] c(-k) = (1 - x[j-k]) * 1[j ≥ k] + (-x[j+k]) * 1[j ≤ n-1-k]

In vectorized form:
grad_shift_neg_k = np.zeros(n)
grad_shift_neg_k[k:] += 1.0 - x[:n-k]    # (1-x[j-k]) for j in [k, n-1]
grad_shift_neg_k[:n-k] += -x[k:]          # (-x[j+k]) for j in [0, n-1-k]
