def repair(h):
    """Project h onto {x : 0 <= x <= 1, sum(x) = n/2}"""
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    target = 0.5 * n
    np.clip(x, 0, 1, out=x)
    # Water-filling to satisfy sum constraint
    for _ in range(100):
        diff = target - x.sum()
        if abs(diff) < 1e-12:

        if diff > 0:

            if not mask.any():
                break
            x[mask] += diff / mask.sum()

def perturb(h, scale, seed):
    rng = np.random.RandomState(seed)
    return repair(h + rng.normal(0, scale, h.size))


def structured_perturb(h, seed, freq):
    rng = np.random.RandomState(seed)
    n = h.size
    t = (np.arange(n) + 0.5) * (2.0 / n)
    h2 = h + 0.02 * np.sin(2 * np.pi * freq * t / 2.0 + rng.uniform(0, 6.28))
    return repair(h2)


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        init = np.asarray(raw, dtype=float)
    else:
        init = np.full(70, 0.5)

    best_x = init.copy()
    best_c = exact_c(best_x)

    n0 = int(init.size)

    # Phase 1: Coordinate descent at base resolution with various params
    configs = [
        (5, 0.0, 100, 0),
        (7, 0.0, 150, 1),
        (5, 0.01, 100, 2),
        (5, -0.01, 100, 3),
        (10, 0.0, 120, 4),
        (3, 0.0, 150, 5),
    ]
    for topk, lam, sweeps, ps in configs:
        if time.time() > deadline - 120:
            break
        h_start = repair(init + np.random.RandomState(seed_val * 7 + ps).normal(0, 0.005, n0))
        hx, cx = coord_descent(h_start, sweeps, topk=topk, lam=lam, seed=seed_val + ps)
        if cx < best_c:
            best_c = cx
            best_x = hx

    # Phase 2: Multi-start at base resolution
    for trial in range(5):
        if time.time() > deadline - 100:
            break
        h_start = perturb(best_x, 0.03, seed_val + 100 + trial)
        hx, cx = coord_descent(h_start, 80, topk=5, lam=0.0, seed=seed_val + 200 + trial)
        if cx < best_c:
            best_c = cx
            best_x = hx

    # Phase 3: Try higher resolutions
    res_list = sorted(set([n0, n0 + 10, n0 + 20, n0 + 40, 100, 140, 200]))
    res_list = [n for n in res_list if 20 <= n <= 200 and n != n0]
    for n_new in res_list[:4]:
        if time.time() > deadline - 80:
            break
        h_new = resample(best_x, n_new)
        hx, cx = coord_descent(h_new, 100, topk=5, lam=0.0, seed=seed_val + 300 + n_new)
        if cx < best_c:
            best_c = cx
            best_x = hx
        # Second pass with different topk
        if time.time() > deadline - 40:
            break
        h_new2 = resample(best_x, n_new)
        hx2, cx2 = coord_descent(h_new2, 80, topk=7, lam=0.0, seed=seed_val + 400 + n_new)
        if cx2 < best_c:
            best_c = cx2
            best_x = hx2

    # Phase 4: Intensive polish at best resolution
    n_best = best_x.size
    if time.time() < deadline - 20:
        for trial in range(4):
            if time.time() > deadline - 10:
                break
            scale = [0.002, 0.005, 0.01, 0.02][trial]
            h_start = perturb(best_x, scale, seed_val + 500 + trial)
            hx, cx = coord_descent(h_start, 60, topk=7, lam=0.0, seed=seed_val + 600 + trial)
            if cx < best_c:
                best_c = cx
                best_x = hx

    # Phase 5: SLSQP refinement on the best
    if time.time() < deadline - 15:
        n_b = best_x.size
        bounds = [(0.0, 1.0)] * n_b
        cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n_b,
                 "jac": lambda x: np.ones(n_b)}]

        def obj(x):
            c = c_values(x)
            tau = 0.005
            z = (c - c.max()) / tau
            z = z - z.max()
            s = np.exp(z).sum()
            return c.max() + tau * np.log(s)

        def obj_jac(x):
            c = c_values(x)
            tau = 0.005
            z = (c - c.max()) / tau
            z = z - z.max()
            e = np.exp(z)
            w = e / e.sum()
            n = x.size
            dx = 2.0 / n
            grad = np.zeros(n)
            for ki in np.where(w > 1e-10)[0]:
                lag = int(ki) - (n - 1)
                for j in range(n):
                    li = j - lag
                    ri = j + lag
                    s_v = 0.0
                    if 0 <= li < n:
                        s_v += x[li]
                    if 0 <= ri < n:
                        s_v += x[ri]
                    grad[j] += w[ki] * dx * (1.0 - s_v)
            return grad

        try:
            res = so.minimize(obj, best_x, method="SLSQP", jac=obj_jac,
                              bounds=bounds, constraints=cons,
                              options={"maxiter": 60, "ftol": 1e-14})
            if res is not None:
                xr = repair(res.x)
                cr = exact_c(xr)
                if cr < best_c:
                    best_c = cr
                    best_x = xr
        except Exception:
            pass

    # Phase 6: Final coordinate descent polish
    if time.time() < deadline - 5:
        hx, cx = coord_descent(best_x, 200, topk=7, lam=0.0, seed=seed_val + 999)
        if cx < best_c:
            best_c = cx
            best_x = hx

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
