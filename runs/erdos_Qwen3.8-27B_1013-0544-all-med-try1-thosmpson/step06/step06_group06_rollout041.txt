            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_func(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    s = np.max(shifted)
    return c_max + (s + np.log(np.sum(np.exp(shifted - s)))) / alpha

def smooth_grad_func(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    s = np.max(shifted)
    weights = np.exp(shifted - s)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    grad_h = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-14:
            continue
        sk = k - (n - 1)
        js = idx + sk
        m1 = (js >= 0) & (js < n)
        if m1.any():
            grad_h[m1] += w * (1.0 - x[js[m1]]) * dx
        jm = idx - sk
        m2 = (jm >= 0) & (jm < n)
        if m2.any():
            grad_h[m2] -= w * x[jm[m2]] * dx
    return grad_h

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    res = minimize(smooth_obj_func, h, args=(n, alpha),
                   jac=smooth_grad_func, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def fine_tune(h, n, rng, max_iters, steps):

    it = 0

            if improved:

    return h, c5

def simulated_anneal(h, n, rng, t_start, t_end, n_iter):
    c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    T = t_start
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / n_iter)
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        d = rng.uniform(-0.1, 0.1)
        hi, hj = h[i] + d, h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i], h[j] = hi, hj
        c_new = compute_c5(h, n)
        delta = c_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = c_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = hi - d, hj + d
    return best_h, best_c5

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
