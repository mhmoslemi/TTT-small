def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):

def grad_smooth(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3000):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            if d > 0:
                i = int(np.argmin(h))
                h[i] = min(1.0, h[i] + d)
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] + d)
        else:
            h[mask] += d / cnt
            np.clip(h, 0, 1, out=h)
    return h

def grad_smooth(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(np.clip(temp * (corr - m), -60, 60))
    sw = w.sum()
    if sw > 1e-300:
        w /= sw
    else:
        w = np.zeros_like(corr)
        w[corr.argmax()] = 1.0
    b = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sft = k - (n - 1)
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        if sft >= 0:
            if sft < n:
                grad[sft:] += wk * b[:n - sft]
                grad[:n - sft] -= wk * h[sft:]
        else:
            a = -sft
            if a < n:
                grad[:n - a] += wk * b[a:]
                grad[a:] -= wk * h[:n - a]
    return grad * dx

def grad_hard(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    thresh = m - 1.0 / n
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([corr.argmax()])
    active = active[np.argsort(corr[active])[::-1]][:top_k]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    b = 1.0 - h
    grad = np.zeros(n)
    for k in active:
        sft = k - (n - 1)
        wk = 1.0 / len(active)
        if sft >= 0:
            if sft < n:
                grad[sft:] += wk * b[:n - sft]
                grad[:n - sft] -= wk * h[sft:]
        else:
            a = -sft
            if a < n:
                grad[:n - a] += wk * b[a:]
                grad[a:] -= wk * h[:n - a]
    return grad * dx

def anneal_grad(h, temp, alpha, n_steps, ts, momentum=0.85, hard_every=200):
    n = len(h)
    cur = h.copy()
    v = np.zeros(n)
    for it in range(n_steps):
        if it % hard_every == hard_every - 1:
            g = grad_hard(cur, top_k=3)
        else:
            g = grad_smooth(cur, temp)
        v = momentum * v - alpha * g
        cur = project_feasible(cur + v, ts)
        if it % 500 == 499:
            v *= 0.3
    return cur

def coord_descent(h, ts, n_levels=7, max_sweeps=5):
    n = len(h)
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            ov = h[j]
            bv, bc = ov, c5
            for v in np.linspace(0, 1, n_levels):
                if abs(v - ov) < 1e-10:
                    continue
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < bc - 1e-13:
                    bc, bv = c, v
            if bv != ov:
                h[j] = bv
                h = project_feasible(h, ts)
                c5 = bc
                improved = True
            else:
                h[j] = ov
        if not improved:
            break
    return h, c5

def make_init(n, mode, rng):
    if mode == 0:
        return np.full(n, 0.5)
    elif mode == 1:
        x = np.arange(n)
        return 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x / n + rng.uniform(0, 6.28))
    elif mode == 2:
        x = np.arange(n)
        return 0.5 + 0.2 * np.cos(2 * np.pi * 3 * x / n) + 0.1 * np.cos(2 * np.pi * 7 * x / n)
    elif mode == 3:
        center = n // 2
        x = np.abs(np.arange(n) - center) / (n / 2)
        return np.clip(0.5 + 0.4 * np.exp(-2 * x * x), 0, 1)
    elif mode == 4:
        x = np.arange(n)
        freq = rng.integers(2, 8)
        return 0.5 + 0.25 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 6.28))
    else:
        f = np.fft.rfft(rng.randn(n))
        f[0] = 0
        f[1:min(4, len(f))] = 0
        sig = np.fft.irfft(f * np.exp(-np.linspace(0, 3, len(f))), n)
        sig = sig - sig.min()
        if sig.max() > 0:
            sig = sig / sig.max()
        return 0.5 + 0.3 * (sig - 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    n = 384
    ts = n / 2.0
    rng = np.random.RandomState(seed)

    # Try parent first
    if 'initial_h_values' in dir() or True:
        try:
            h_p = initial_h_values.astype(float).copy()
            if len(h_p) != n:
                x_old = np.arange(len(h_p)) / len(h_p)
                x_new = np.arange(n) / n
                h_p = np.interp(x_new, x_old, h_p)
            h_p = project_feasible(h_p, ts)
        except:
            h_p = np.full(n, 0.5)

    # Multi-start: try several initializations
    candidates = [h_p]
    for mode in [1, 2, 3, 4, 5]:
        h_init = make_init(n, mode, rng)
        h_init = project_feasible(h_init, ts)
        candidates.append(h_init)

    # Quick evaluation of each candidate with short optimization
    best_h = None
    best_c5 = 10.0
    for h_c in candidates:
        if rem() < 80:
            break
        h_opt = anneal_grad(h_c, 50.0, 0.015, 800, ts, momentum=0.85)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 1: Full annealed gradient from best
    cur = best_h.copy()
    for temp in [400.0, 150.0, 60.0, 20.0, 7.0, 2.5, 1.0]:
        if rem() < 50:
            break
        alpha = 0.018
        steps = min(2500, max(500, int(rem() * 15)))
        cur = anneal_grad(cur, temp, alpha, steps, ts, momentum=0.85, hard_every=300)
        c = compute_c5(cur)
        if c < best_c5:
            best_c5 = c
            best_h = cur.copy()

    # Phase 2: Coordinate descent polish
    if rem() > 40:
        h_cd, c_cd = coord_descent(best_h, ts, n_levels=7, max_sweeps=8)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 3: Random restarts
    n_rest = 0
    while rem() > 100 and n_rest < 30:
        n_rest += 1
        sc = rng.uniform(0.04, 0.2)
        h_r = best_h + rng.randn(n) * sc
        h_r = project_feasible(h_r, ts)
        h_r = anneal_grad(h_r, 30.0, 0.012, 1200, ts, momentum=0.82, hard_every=200)
        c = compute_c5(h_r)
        if c < best_c5:
            best_c5 = c
            best_h = h_r.copy()
            if rem() > 50:
                h_r2, c2 = coord_descent(best_h, ts, n_levels=7, max_sweeps=4)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_r2.copy()

    # Phase 4: Additional diverse starts
    if rem() > 120:
        for mode in [1, 2, 4, 5]:
            if rem() < 60:
                break
            h_i = make_init(n, mode, rng)
            h_i = project_feasible(h_i, ts)
            h_i = anneal_grad(h_i, 20.0, 0.01, 1000, ts, momentum=0.8)
            c = compute_c5(h_i)
            if c < best_c5:
                best_c5 = c
                best_h = h_i.copy()

    # Phase 5: Final fine coordinate descent
    if rem() > 25:
        h_cd, c_cd = coord_descent(best_h, ts, n_levels=11, max_sweeps=12)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
