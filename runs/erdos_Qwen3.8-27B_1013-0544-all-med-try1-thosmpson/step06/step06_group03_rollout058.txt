import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = np.sum(exps)
    obj = (m + t * np.log(s)) * dx
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[:n - shift] += w * (1.0 - h[:n - shift])
                grad[shift:] -= w * h[shift:]
        else:
            s2 = -shift
            if s2 < n:
                grad[s2:] += w * (1.0 - h[s2:])
                grad[:n - s2] -= w * h[:n - s2]
    return obj, grad * dx

def fista_run(h_init, t_schedule, lr, max_iter_per_t):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for t in t_schedule:
        y = h.copy()
        t_k = 1.0
        for it in range(max_iter_per_t):
            _, grad_y = softmax_obj_grad(y, t)
            h_new = project_feasible(y - lr * grad_y, n)
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = h_new + ((t_k - 1.0) / t_next) * (h_new - h)
            t_k = t_next
            h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def fista_adaptive(h_init, t, lr_init, max_iter):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    y = h.copy()
    t_k = 1.0
    lr = lr_init
    for it in range(max_iter):
        obj_y, grad_y = softmax_obj_grad(y, t)
        h_new = project_feasible(y - lr * grad_y, n)
        obj_new = compute_c5(h_new)
        if obj_new > obj_y + 1e-8:
            lr *= 0.5
            h_new = project_feasible(y - lr * grad_y, n)
        else:
            lr = min(lr * 1.05, lr_init)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_next) * (h_new - h)
        t_k = t_next
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def smooth_perturb(h, rng, sigma, width):
    n = len(h)
    noise = rng.normal(0, 1, n)
    kernel = np.ones(width) / width
    noise = np.convolve(noise, kernel, mode='same')
    return h + sigma * noise
