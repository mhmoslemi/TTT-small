```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            # All at bounds, need to adjust
            if d > 0:
                # Need to increase sum, increase the ones closest to 1
                idx = np.argsort(h)[-int(abs(d))+1:]
                h[idx] = np.minimum(h[idx] + d / len(idx), 1.0)
            else:
                idx = np.argsort(h)[:int(abs(d))+1:]
                h[idx] = np.maximum(h[idx] + d / len(idx), 0.0)
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final projection
    h = np.clip(h, 0.0, 1.0)
    # If sum still off, do a final adjustment
    d = target_sum - h.sum()
    if abs(d) > 1e-12:
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective_and_grad(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_arg = np.clip(temp * (corr - m), -500, 500)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w = w / w_sum
    smooth_max = m + np.log(w_sum) / temp
    obj = smooth_max * dx
    sum_dev = h.sum() - n / 2.0
    obj += penalty * sum_dev ** 2
    
    # Gradient
    j_idx = np.arange(n)
    t_idx = np.arange(-(n-1), n)
    # h_padded: h is at positions n-1 to 2n-2 in an array of length 3n-1
    h_pad = np.zeros(3 * n - 1)
    h_pad[n-1:2*n-1] = h
    
    # S1[j] = sum_t w[t] * (1-h[j+t]) * 1{0<=j+t<n}
    valid1 = (j_idx[:, None] + t_idx[None, :] >= 0) & (j_idx[:, None] + t_idx[None, :] < n)
    idx1 = j_idx[:, None] + t_idx[None, :] + (n - 1)
    S1 = (w[None, :] * (1.0 - h_pad[idx1]) * valid1).sum(axis=1)
    
    # S2[j] = sum_t w[t] * h[j-t] * 1{0<=j-t<n}
    valid2 = (j_idx[:, None] - t_idx[None, :] >= 0) & (j_idx[:, None] - t_idx[None, :] < n)
    idx2 = j_idx[:, None] - t_idx[None, :] + (n - 1)
    S2 = (w[None, :] * h_pad[idx2] * valid2).sum(axis=1)
    
    grad = dx / temp * (S1 - S2) + 2.0 * penalty * sum_dev
    return obj, grad

def lbfgs_optimize(h_start, temp, penalty, maxiter=500):
    n = len(h_start)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_objective_and_grad(x, temp, penalty)[0]
    
    def grad(x):
        return smooth_objective_and_grad(x, temp, penalty)[1]
    
    res = minimize(obj, h_start, jac=grad, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    h_out = project_feasible(res.x, n / 2.0)
    return h_out

def coord_descent_fast(h, max_sweeps=10, ternary_iters=40):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Save current value
            orig = h[j]
            
            # Compute the effect of changing h[j] on each correlation
            # For each lag t, corr[t] changes linearly with h[j]
            # slope_t = (1-h[j+t])*1{valid} - h[j-t]*1{valid}
            # intercept_t = corr[t] - slope_t * h[j]
            
            # Compute slopes and intercepts
            t_vals = np.arange(-(n-1), n)
            # h_padded
            h_pad = np.zeros(3 * n - 1)
            h_pad[n-1:2*n-1] = h
            
            valid1 = (j + t_vals >= 0) & (j + t_vals < n)
            valid2 = (j - t_vals >= 0) & (j - t_vals < n)
            
            slopes = np.zeros(len(t_vals))
            slopes[valid1] += (1.0 - h_pad[j + t_vals[valid1] + n - 1])
            slopes[valid2] -= h_pad[j - t_vals[valid2] + n - 1]
            
            # Current corr values
            corr_current = np.correlate(h, 1.0 - h, mode="full")
            intercepts = corr_current - slopes * orig
            
            # Now find h[j] in [0,1] that minimizes max_t (intercept_t + slope_t * v) * dx
            # plus penalty for sum constraint
            # After changing h[j] from orig to v, sum changes by (v - orig)
            # New sum = target + (v - orig) [approximately, since we'll project]
            # Penalty: penalty * (v - orig)^2 (approximately)
            
            # For simplicity, just minimize max_t (intercept_t + slope_t * v) over v in [0,1]
            # using ternary search (it's convex in v)
            
            def f_1d(v):
                return np.max(intercepts + slopes * v)
            
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(ternary_iters):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if f_1d(m1) < f_1d(m2):
                    b = m2
                else:
                    a = m1
            
            v_best = (a + b) / 2.0
            
            # Check if improvement
            new_corr_max = f_1d(v_best)
            new_c5 = new_corr_max * (2.0 / n)
            
            if new_c5 < c5 - 1e-14:
                h[j] = v_best
                c5 = new_c5
                improved = True
            # else: keep original (h[j] is already orig)
        
        if not improved:
            break
    
    # Project to feasible
    h = project_feasible(h, target)
    c5 = compute_c5(h)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return budget_s - elapsed()
    
    # Initialize
```

```python
def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        step = d / mask.sum()
        h[mask] += step

    h = np.clip(h, 0.0, 1.0)

def grad_smooth(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_arg = np.clip(temp * (corr - m), -500, 500)
    w = np.exp(exp_arg)
    w /= w.sum()
    
    # Compute gradient using O(n^2) matrix approach
    j_idx = np.arange(n)[:, None]

    h_pad[n-1:2*n-1] = h

    idx1 = j_idx + t_idx + (n - 1)

    idx2 = j_idx - t_idx + (n - 1)

    grad = dx / temp * (S1 - S2) + 2.0 * penalty * sum_dev

    res = minimize(obj, h_start, jac=grad, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
    return project_feasible(res.x, n / 2.0)

def coord_descent(h, max_sweeps=15, ternary_iters=40):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    t_vals = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            valid1 = (j + t_vals >= 0) & (j + t_vals < n)
            valid2 = (j - t_vals >= 0) & (j - t_vals < n)
            
            slopes = np.zeros(len(t_vals))
            slopes[valid1] += (1.0 - h[j + t_vals[valid1]])
            slopes[valid2] -= h[j - t_vals[valid2]]
            
            u_lo = -h[j]
            u_hi = 1.0 - h[j]
            if u_hi - u_lo < 1e-10:
                continue
            
            a, b = u_lo, u_hi
            for _ in range(ternary_iters):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(corr + slopes * m1) < np.max(corr + slopes * m2):
                    b = m2
                else:
                    a = m1
            
            v_best = (a + b) / 2.0 + h[j]
            v_best = np.clip(v_best, 0.0, 1.0)
            delta = v_best - h[j]
            
            if abs(delta) < 1e-14:
                continue
            
            new_c5 = float(np.max(corr + slopes * delta)) * dx
            if new_c5 < c5 - 1e-14:
                h[j] = v_best
                corr += slopes * delta
                c5 = new_c5
                improved = True
        
        if not improved:
            break
    
    h = project_feasible(h, target)
    c5 = compute_c5(h)
    return h, c5
```

```python
def grad_smooth(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_arg = np.clip(temp * (corr - m), -500, 500)
    w = np.exp(exp_arg)
    w /= w.sum()
    
    omh = 1.0 - h
    w_rev = w[::-1]
    
    S1 = np.correlate(omh, w, mode="full")[3*n-3 : 2*n-3 : -1]
    S2 = np.correlate(h, w_rev, mode="full")[3*n-3 : 2*n-3 : -1]
    
    sum_dev = h.sum() - n / 2.0
    grad = dx / temp * (S1 - S2) + 2.0 * penalty * sum_dev
    return grad
```
