    target = 0.5 * n
    for _ in range(80):
        d = target - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            s = 1.0 - x; m = s > 1e-14
            if not m.any(): break
            x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-14
            if not m.any(): break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    d = target - x.sum()
    if abs(d) > 1e-11:
        o = np.argsort(1-x)[::-1] if d > 0 else np.argsort(x)[::-1]
        for i in o:
            if abs(d) < 1e-14: break
            if d > 0 and x[i] < 1:
                a = min(d, 1-x[i]); x[i] += a; d -= a
            elif d < 0 and x[i] > 0:
                a = max(d, -x[i]); x[i] += a; d -= a
    return x

def cvals(h):
    n = h.size
    return np.correlate(h, 1-h, mode="full") * (2.0/n)

def c5(h):
    return float(cvals(h).max())

def _opt_t(x, i, f_base, n):
    m = 2*n - 1
    ks = np.arange(-(n-1), n)
    ik = i + ks; imk = i - ks
    vk = (ik >= 0) & (ik < n)
    vmk = (imk >= 0) & (imk < n)
    sk = np.clip(ik, 0, n-1); smk = np.clip(imk, 0, n-1)
    hk = np.where(vk, x[sk], 0.0)
    hmk = np.where(vmk, x[smk], 0.0)
    a = vk.astype(float); b = vmk.astype(float)
    C = b - a*hk - b*hmk
    B = f_base - x[i]*C
    R0 = f_base[n-1] - x[i]*(1.0-x[i])
    ts = np.linspace(0, 1, 601)
    gl = np.maximum.reduce(B[None,:] + ts[:,None]*C[None,:])
    g_no0 = np.delete(gl, n-1, axis=1).max(axis=1)
    g = np.maximum(g_no0, R0 + ts - ts**2)
    idx = g.argmin()
    lo, hi = max(0, idx-2), min(600, idx+2)
    ts2 = np.linspace(float(ts[lo]), float(ts[hi]), 51)
    gl2 = np.maximum.reduce(B[None,:] + ts2[:,None]*C[None,:])
    g2_no0 = np.delete(gl2, n-1, axis=1).max(axis=1)
    g2 = np.maximum(g2_no0, R0 + ts2 - ts2**2)
    return float(ts2[g2.argmin()])

def cd_pass(x, rng):
    n = x.size
    x = x.copy()
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    for i in order:
        f = cvals(x)
        x[i] = _opt_t(x, i, f, n)
    return x

def cd_optimize(x0, time_limit, rng, max_passes=200):
    x = x0.copy()
    best_x, best_c = x.copy(), c5(x)
    t0 = time.time()
    improved = 0
    for p in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        xn = cd_pass(x, rng)
        cn = c5(xn)
        if cn < best_c - 1e-12:
            best_c = cn; best_x = xn.copy(); x = xn; improved = 0
        elif cn < best_c + 1e-10:
            x = xn; improved += 1
        else:
            improved += 1
        if improved > 3:
            break
    return best_x, best_c

_SMO = {}
def _smooth_fun(x):
    n = x.size; dx = 2.0/n
    c = np.correlate(x, 1-x, mode="full")*dx
    tau = _SMO["tau"]
    z = (c - c.max())/tau; z -= z.max()
    e = np.exp(z); s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return float(c.max())
    return float(c.max() + tau*np.log(s))

def _smooth_jac(x):
    n = x.size; dx = 2.0/n
    c = np.correlate(x, 1-x, mode="full")*dx
    tau = _SMO["tau"]
    z = (c - c.max())/tau; z -= z.max()
    e = np.exp(z); w = e/e.sum()
    if not np.isfinite(w).all() or w.sum() <= 0:
        w = np.zeros(2*n-1); w[n-1] = 1.0
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12: continue
        lag = k - (n-1)
        for i in range(n):
            if 0 <= i+lag < n:
                grad[i] += w[k] * (-x[i+lag]) if lag != 0 else w[k]*(1-2*x[i])
            if 0 <= i-lag < n:
                if lag != 0:
                    grad[i] += w[k] * (1-x[i-lag])
    return grad * dx

def _cons_fun(x):
    return np.array([x.sum() - 0.5*x.size])
def _cons_jac(x):
    return np.ones((1, x.size))

def slsqp_run(x0, tau, maxiter, time_limit):
    n = x0.size
    _SMO["tau"] = tau
    bounds = [(0.0, 1.0)]*n
    cons = [{"type": "eq", "fun": _cons_fun, "jac": _cons_jac}]
    tl = time.time() + time_limit
    if time.time() > tl:
        return x0, c5(x0)
    try:
        res = so.minimize(_smooth_fun, x0, method="SLSQP", jac=_smooth_jac,
                          bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-12})
        if res is not None:
            xr = repair(res.x)
            return xr, c5(xr)
    except Exception:
        pass
    return x0, c5(x0)

def full_optimize(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(h0)
    best_x, best_c = x.copy(), c5(x)
    # Phase A: CD
    ta = min(time_limit * 0.4, max(1.0, time_limit / 3))
    x, c = cd_optimize(x, ta, rng)
    if c < best_c: best_c, best_x = c, x
    # Phase B: SLSQP with decreasing tau
    tb = time_limit - (time.time() - (time.time() - time_limit))  # remaining
    remaining = time_limit - (time.time() - (time.time() - time_limit))
    elapsed = time.time() - (time.time() - time_limit)
    for tau in (0.05, 0.02, 0.008, 0.003):
        if time.time() - (time.time() - time_limit) > time_limit * 0.85:
            break
        tl_s = min(3.0, (time_limit - (time.time() - (time.time()-time_limit))) * 0.15)
        if tl_s < 0.5: break
        x2, c2 = slsqp_run(best_x, tau, 100, tl_s)
        if c2 < best_c: best_c, best_x = c2, x2
    # Phase C: final CD polish
    remaining = time_limit - (time.time() - (time.time() - time_limit))
    if remaining > 0.5:
        x3, c3 = cd_optimize(best_x, min(remaining * 0.6, 5.0), rng)
        if c3 < best_c: best_c, best_x = c3, x3
    return best_x, best_c

def resample(h0, n_new):
    n0 = h0.size
    if n0 == n_new: return repair(h0)
    x0 = (np.arange(n0)+0.5)*(2.0/n0)
    x1 = (np.arange(n_new)+0.5)*(2.0/n_new)
    return repair(np.interp(x1, x0, h0))

def random_start(n, seed):
    rng = np.random.RandomState(seed)
    t = (np.arange(n)+0.5)*(2.0/n)
    f = rng.randint(1, max(2, n//6+1))
    amp = 0.3 + 0.2*rng.rand()
    h = 0.5 + amp*np.sin(2*np.pi*f*t/2 + rng.uniform(0, 6.28))
    h += 0.04*rng.normal(0, 1, n)
    return repair(h)

def structured_start(n, seed, freq):
    rng = np.random.RandomState(seed)
    t = (np.arange(n)+0.5)*(2.0/n)
    h = np.full(n, 0.5)
    for m in (1, 2, 3):
        h += (0.4/m)*0.5*np.sin(2*np.pi*freq*m*t/2 + rng.uniform(0, 6.28))
    h += 0.01*rng.normal(0, 1, n)
    return repair(h)

def jitter(h0, scale, seed):
    rng = np.random.RandomState(seed)
    return repair(h0 + rng.normal(0, scale, h0.size))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)
    if init is not None and init.size >= 10:
        n0 = int(init.size)
        if n0 > 300: n0 = 200
    else:
        n0 = 70
        init = np.full(n0, 0.5)

    best_x, best_c = init.copy(), c5(init)

    # Strategy: try multiple n values and starting points
    n_list = sorted(set([n0, min(n0+10, 200), max(n0-10, 40), min(n0+20, 200)]), key=lambda x: -abs(x-n0))
    n_list = [n for n in n_list if 40 <= n <= 200]

    starts_per_n = 3
    per_time = budget_s * 0.08

    total_starts = len(n_list) * starts_per_n
    alloc = (budget_s * 0.85) / max(1, total_starts)

    attempt = 0
    for n in n_list:
        if time.time() > deadline - 30:
            break
        base = resample(init, n) if n != n0 else init.copy()
        starts = [base]
        starts.append(jitter(base, 0.03, seed_val+attempt*31+n))
        starts.append(random_start(n, seed_val+attempt*53+n))
        if n != n0:
            starts.append(structured_start(n, seed_val+attempt*71+n, 2))
        for si, h0 in enumerate(starts[:starts_per_n+1]):
            attempt += 1
            if time.time() > deadline - 20:
                break
            tl = min(alloc * 1.2, max(3.0, deadline - time.time() - 10))
            x, c = full_optimize(h0, tl, seed_val + attempt*1000 + si)
            if c < best_c:
                best_c = c; best_x = x.copy()
            # Also try CD-only from this result (fast extra polish)
            if time.time() < deadline - 15:
                rng2 = np.random.RandomState(seed_val + attempt*999)
                x2, c2 = cd_optimize(x, min(2.0, (deadline-time.time())*0.1), rng2)
                if c2 < best_c:
                    best_c = c2; best_x = x2.copy()

    # Final polish on best
    if time.time() < deadline - 5:
        rng_f = np.random.RandomState(seed_val + 77777)
        xf, cf = cd_optimize(best_x, min(10.0, (deadline-time.time())-3), rng_f)
        if cf < best_c:
            best_c, best_x = cf, xf
    if time.time() < deadline - 3:
        _SMO["tau"] = 0.001
        xb, cb = slsqp_run(best_x, 0.001, 80, max(1.0, deadline - time.time() - 2))
        if cb < best_c:
            best_c, best_x = cb, xb

    best_x = repair(best_x)
    best_c = c5(best_x)
    return best_x, best_c, int(best_x.size)
