    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_value(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = temp * (corr - m)
    return (m + np.log(np.exp(shifted).sum()) / temp) * dx

def soft_max_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = temp * (corr - m)
    exp_s = np.exp(shifted)
    S = exp_s.sum()
    p = exp_s / S
    A = np.correlate(p, 1.0 - h, mode='full')[:n]
    B = np.convolve(h, p, mode='full')[n-1:n-1+n]
    g = (A - B) * dx
    g = g - g.mean()
    return g

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(x, temp):
    return soft_max_value(x, temp)

def grad_func(x, temp):
    return soft_max_grad(x, temp)

def lbfgs_step(h, n, temp, maxiter):
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h, args=(temp,), jac=grad_func, method='L-BFGS-B',
                   bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxls': 100, 'maxfun': 10000})
    h_out = project_feasible(res.x, ts)
    return h_out

def coord_descent_fast(h, max_sweeps=40, n_levels=16):
    n = len(h)
    ts = n / 2.0
    f = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(f) * (2.0 / n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            v_old = h[j]
            best_v = v_old
            best_c5 = c5
            levels = np.linspace(0.0, 1.0, n_levels)
            for v_new in levels:
                if abs(v_new - v_old) < 1e-10:
                    continue
                delta = v_new - v_old
                s = shifts
                idx1 = j + s
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = j - s
                valid2 = (idx2 >= 0) & (idx2 < n)
                change = np.zeros(2 * n - 1)
                if np.any(valid1):
                    change[valid1] += delta * (1.0 - h[idx1[valid1]])
                if np.any(valid2):
                    change[valid2] -= delta * h[idx2[valid2]]
                f_new = f + change
                c5_new = np.max(f_new) * (2.0 / n)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_v = v_new
            if best_v != v_old:
                h[j] = best_v
                h = project_feasible(h, ts)
                f = np.correlate(h, 1.0 - h, mode='full')
                c5 = np.max(f) * (2.0 / n)
                improved = True
        if not improved:
            break
    return h, c5
