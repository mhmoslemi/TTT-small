import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

    g = np.zeros(N)
    if L >= 0:
        g[L:] += (1.0 - h[:N-L])
        g[:N-L] -= h[L:]
    else:
        M = -L
        g[:N-M] += (1.0 - h[M:])
        g[M:] -= h[:N-M]
    return g

def soft_grad(h, corr, dx, alpha, k_top=None):
    N = len(h)
    if k_top is not None:
        # Top-k gradient
        idx = np.argpartition(corr, -k_top)[-k_top:]
        w = np.zeros(len(corr))
        w[idx] = 1.0 / k_top
    else:
        c = corr - corr.max()
        w = np.exp(alpha * c)
        wsum = w.sum()
        if wsum < 1e-300:
            k_star = np.argmax(corr)
            return grad_lag(h, k_star - (N - 1)) * dx
        w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-6:
            continue
        g += w[j] * grad_lag(h, j - (N - 1))
    return g * dx

def nesterov_optimize(h, N, rng, t_end, try_update, max_iter=30000):

    try_update(h, val)
    best_h = h.copy()
    best_val = val
    v = np.zeros(N)  # momentum velocity
    beta = 0.9
    lr = 0.02

The objective is max_L ∑_n h[n+L](1-h[n]). If we think of h as a "signal" and we want all its autocorrelation-like quantities with (1-h) to be small, this is related to "low autocorrelation" sequences.

Actually wait, for binary h: ∑ h[n](1-h[n]) = 0 for all n, so c_0 = 0. The max is over L ≠ 0.

For non-binary h, c_0 = ∑ h[n](1-h[n]) > 0. So there's a trade-off: making h more binary reduces c_0 but might increase other c_L.

This suggests the optimal h is close to binary but with some smoothing at the transitions.

**Key insight**: If h is binary (indicator of a set A with |A| = N/2), then:

This is related to "difference sets" in combinatorics! A (v,k,λ) difference set is a subset D of a group of order v with |D| = k such that every non-identity element is represented exactly λ times as a difference of two elements of D.
