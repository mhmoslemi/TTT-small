import cvxpy as cp
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(1.0, h[idx] + d2)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _slp_step(h_old, n, solver=cp.GLPK):
    """One step of successive linearization."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == 0.5 * n]
    
    # For each lag m from -(n-1) to (n-1)
    for m in range(-(n-1), n):
        # C(m) ≈ sum_{i valid} h[i] * (1 - h_old[i+m])
        # = sum_{i valid} (1 - h_old[i+m]) * h[i]
        # Constraint: this * dx <= t
        # i.e., sum (1 - h_old[i+m]) * h[i] <= t / dx
        
        coeff = np.zeros(n)
        for i in range(max(0, -m), min(n, n - m)):
            coeff[i] = (1.0 - h_old[i + m])
        
        # Skip if all coefficients are zero (shouldn't happen for m=0 where coeff = 1-h_old)
        if np.all(coeff == 0):

        if c5_new < best_c5 - 1e-10:

            best_c5 = c5_new
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
            h_cur = _fix(0.5 * h_cur + 0.5 * h_new, n)
    return best_h, best_c5

def _grad_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    nc = len(C)
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _grad_optimize(h, n, deadline, alpha=500, maxiter=300):
    beta = 0.05 * alpha
    res = minimize(_grad_obj, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 5,
                            'ftol': 1e-14, 'gtol': 1e-12})
    if res.x is not None and time.time() < deadline:
        return _fix(res.x, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    def _el():
        return time.time() - t0

    # Phase 1: SLP at original n
    deadline1 = t0 + budget_s * 0.25
    hb, cb = _slp_optimize(hb, nb, deadline1, max_iters=400)

    # Phase 2: Try different n values
    n_cands = sorted(set([nb + d for d in [-10, -5, -3, 3, 5, 10, 15, 20, 30]
                          if 60 <= nb + d <= 250]))
    for n_abs in [80, 85, 90, 95, 100, 105, 110, 120, 130, 150, 180, 200]:
        if 60 <= n_abs <= 250 and n_abs != nb:
            n_cands.append(n_abs)
    n_cands = sorted(set(n_cands))

    deadline2 = t0 + budget_s * 0.55
    for n in n_cands:
        if time.time() > deadline2:
            break
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        ht = _fix(np.interp(x_new, x_old, hb), n)
        ht, ct = _slp_optimize(ht, n, deadline2, max_iters=150)
        if ct < cb - 1e-9:
            cb, hb, nb = ct, ht, n

    # Phase 3: Random restarts with SLP
    deadline3 = t0 + budget_s * 0.85
    attempts = 0
    while time.time() < deadline3 and attempts < 200:
        attempts += 1
        scale = 0.04 + 0.08 * rng.rand()
        ht = _fix(hb + rng.randn(nb) * scale, nb)
        ht, ct = _slp_optimize(ht, nb, deadline3, max_iters=80)
        if ct < cb - 1e-9:
            cb, hb = ct, ht
        # Also try a different n occasionally
        if attempts % 10 == 0:
            n_try = nb + int(rng.randint(-15, 16))
            if 60 <= n_try <= 250 and n_try != nb:
                x_old = np.linspace(0, 2, nb, endpoint=False)
                x_new = np.linspace(0, 2, n_try, endpoint=False)
                ht2 = _fix(np.interp(x_new, x_old, hb + rng.randn(nb) * 0.05), n_try)
                ht2, ct2 = _slp_optimize(ht2, n_try, deadline3, max_iters=80)
                if ct2 < cb - 1e-9:
                    cb, hb, nb = ct2, ht2, n_try

    # Phase 4: Gradient polish
    deadline4 = t0 + budget_s * 0.97
    if time.time() < deadline4:
        for alpha in [300, 800, 1500]:
            if time.time() > deadline4:
                break
            hn = _grad_optimize(hb, nb, deadline4, alpha=alpha, maxiter=200)
            cn = _c5_max(hn)
            if cn < cb - 1e-10:
                cb, hb = cn, hn

    # Phase 5: Final SLP polish
    deadline5 = t0 + budget_s * 0.995
    if time.time() < deadline5:
        hb, cb = _slp_optimize(hb, nb, deadline5, max_iters=200)

    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
