def compute_corr(h):

def to_binary(h, N):

    h = np.array(h, dtype=float)
    # Sort by value, set top N/2 to 1
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(N)
    b[sorted_idx[:N//2]] = 1.0
    return b

    ones = list(np.nonzero(h)[0])
    zeros = list(np.nonzero(1-h)[0])
    
    corr = compute_corr(h)

    cur_score = cur_val * 2.0 / N
    try_update(h, cur_score)
    
    T = 0.5  # initial temperature (in units of corr values)
    T_min = 0.01
    cool = 0.9995
    
    best_val = cur_val
    best_h = h.copy()

        ib = rng.randint(len(zeros))

            h[b] = 0

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    N = len(h)
    return compute_corr(h).max() * (2.0 / N)

def to_binary_exact(h, rng):
    """Ensure h is binary with exactly len(h)//2 ones."""
    N = len(h)
    target = N // 2
    b = np.round(h).astype(float)
    n_ones = int(b.sum())
    if n_ones > target:
        idx = np.nonzero(b)[0]
        rng.shuffle(idx)
        b[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.nonzero(1 - b)[0]
        rng.shuffle(idx)
        b[idx[:target - n_ones]] = 1
    return b

def local_search_binary(h, rng, t0, t_end, try_update, max_iter=1000000):
    """Swap-based local search with restarts."""
    N = len(h)
    h = h.copy()
    
    corr = compute_corr(h)
    cur_max = corr.max()
    cur_c5 = cur_max * (2.0 / N)
    try_update(h, cur_c5)
    
    best_max = cur_max
    best_h = h.copy()
    
    ones = np.nonzero(h)[0]
    zeros = np.nonzero(1 - h)[0]
    
    T = 1.0
    T_min = 0.1
    cool_rate = 0.999
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        
        # Pick random swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        # Apply swap
        h[a] = 0
        h[b] = 1
        
        new_corr = compute_corr(h)
        new_max = new_corr.max()
        delta = new_max - cur_max
        
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            # Accept
            cur_max = new_max
            corr = new_corr
            ones[np.where(ones == a)[0][0]] = b
            zeros[np.where(zeros == b)[0][0]] = a
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0

            h[b] = 0

            cur_max = corr.max()
            no_improve = 0

def sa_binary(h, rng, t_end, try_update, max_iter=10000000):
    N = len(h)
    h = to_binary(h, rng)
    cur_max = int(compute_corr(h).max())
    try_update(h, cur_max * 2.0 / N)
    best_max = cur_max
    best_h = h.copy()
    
    ones = list(np.nonzero(h)[0])
    zeros = list(np.nonzero(1 - h)[0])
    
    T = 1.0
    T_min = 0.01
    cool = 0.9998
    stag = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        
        M = 20
        best_nm = 9999
        best_a = best_b = -1
        for _ in range(M):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = 0, 1
            nm = int(compute_corr(h).max())
            h[a], h[b] = 1, 0
            if nm < best_nm:
                best_nm = nm
                best_a, best_b = a, b
        
        if best_a < 0:
            stag += 1
        else:
            delta = best_nm - cur_max
            if delta <= 0 or (T > T_min and rng.random() < np.exp(-max(-delta, 0.0) / T)):
                h[best_a], h[best_b] = 0, 1
                ia = ones.index(best_a)
                ib = zeros.index(best_b)
                ones[ia], zeros[ib] = best_b, best_a
                cur_max = best_nm
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                    stag = 0
                    try_update(best_h, best_max * 2.0 / N)
                else:
                    stag += 1
            else:
                stag += 1
        
        if T > T_min:
            T *= cool
        elif stag > 300:
            stag = 0
            for _ in range(rng.randint(3, 10)):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a], h[b] = 0, 1
                ia = ones.index(a)
                ib = zeros.index(b)
                ones[ia], zeros[ib] = b, a
            cur_max = int(compute_corr(h).max())
            T = 0.3
    
    try_update(best_h, best_max * 2.0 / N)
    return best_h, best_max * 2.0 / N

def continuous_refine(h, t_end, try_update):
    N = len(h)
    dx = 2.0 / N
    h = h.astype(float).copy()
    h = np.clip(h, 0, 1)
    diff = N / 2.0 - h.sum()
    if abs(diff) > 1e-8:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    h = np.clip(h, 0, 1)
    
    val = c5_of(h)
    try_update(h, val)
    
    lr = 0.02
    alpha = 15.0
    
    for it in range(30000):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        cm = corr.max()
        c = corr - cm
        w = np.exp(alpha * c)
        wsum = w.sum()
        if wsum < 1e-300:
            break
        w /= wsum
        
        g = np.zeros(N)
        for j in range(len(w)):
            if w[j] < 1e-5:
                continue
            k = j - (N - 1)
            g += w[j] * grad_Sk(h, k)
        g *= dx
        
        h_new = np.clip(h - lr * g, 0, 1)
        diff = N / 2.0 - h_new.sum()
        if abs(diff) > 1e-8:
            mask = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
            if mask.sum() > 0:
                h_new[mask] += diff / mask.sum()
        h_new = np.clip(h_new, 0, 1)
        
        val_new = c5_of(h_new)
        if val_new < val - 1e-13:
            h = h_new
            val = val_new
            try_update(h, val)
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.9
    
    return h, val
