from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - n / 2.0

    return np.ones(n)

        if time.time() - t_start > time_limit:
            break

        j = rng.randint(n)
        if i == j:
            continue
        # Try transfer from i to j (and vice versa)
        # h[i] + t, h[j] - t, t in [max(-h[i], -(1-h[j])), min(1-h[i], h[j])]
        t_min = max(-h[i], -(1.0 - h[j]))
        t_max = min(1.0 - h[i], h[j])
        if t_max - t_min < 1e-6:

        # Try several values of t

        candidates = []
        for frac in [0.25, 0.5, 0.75]:

        for s in [1, -1]:

                t = s * d
                if t_min <= t <= t_max:
                    candidates.append(t)
        # Also try t near current best
        candidates.append(0.0)

        best_c = best_c5

            h_try[i] += t
            h_try[j] -= t
            if h_try[i] < -1e-10 or h_try[i] > 1 + 1e-10 or h_try[j] < -1e-10 or h_try[j] > 1 + 1e-10:

            h_try = np.clip(h_try, 0, 1)
            c = compute_c5(h_try, n)
            if c < best_c:

                best_t = t
        if best_t != 0.0:
            h[i] += best_t
            h[j] -= best_t
            h = np.clip(h, 0, 1)
            best_c5 = best_c
    return h, best_c5

        lo1 = max(0, shift)
        hi1 = min(n, n + shift)  # i such that 0 <= i - shift < n
        # Second factor: -h[i + shift] for valid i + shift
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)  # i such that 0 <= i + shift < n

        # grad[i] -= h[i + shift] for i in [lo2, hi2)

            grad[idx1] += (1 - h[idx1 - shift])
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= h[idx2 + shift]

    grad /= len(active)
    
    # Projected step

    h_new = project_box_sum(h_new, n)
    return h_new

- Polyak step size: eta_t = (C_t - C*) / ||grad||² where C_t is current value and C* is optimal (unknown). Approximate with eta_t = (C_t - C_lower) / ||grad||² where C_lower is a known lower bound.
- Or use a simple diminishing schedule with backtracking.

Actually, for this problem, I think the most effective approach is to combine:
1. Subgradient for direction (tells us which coordinates to adjust and in which direction)
2. Line search for step size (find the best step along the subgradient direction)

def subgradient_line_search(h, n, max_active=8):

    c = np.correlate(h, 1 - h, mode='full') * dx
    max_val = c.max()

    if len(active) > max_active:
        active = active[:max_active]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += (1.0 - h[idx + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx - shift]
    grad *= dx / len(active)
    
    # Line search: try several step sizes along -grad direction
    best_c5 = compute_c5(h, n)
    best_h = h

        c_try = compute_c5(h_try, n)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con_func(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def compute_subgradient(h, n, max_active=8):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_val = c.max()
    active = np.where(c >= max_val - 1e-10)[0]
    if len(active) > max_active:
        active = active[:max_active]
    elif len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += (1.0 - h[idx + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx - shift]
    grad *= dx / len(active)
    return grad

def subgradient_step(h, n, rng, max_active=8):
    grad = compute_subgradient(h, n, max_active)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    for eta in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        h_try = project_box_sum(h - eta * grad, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c5 - 1e-14:
            best_c5 = c_try
            best_h = h_try
            break
    return best_h, best_c5

def local_pair_search(h, n, rng, n_trials, time_limit):
    best_c5 = compute_c5(h, n)
    t_start = time.time()
    for _ in range(n_trials):
        if time.time() - t_start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        t_min = max(-h[i], -(1.0 - h[j]))
        t_max = min(1.0 - h[i], h[j])
        if t_max - t_min < 1e-9:
            continue
        best_t = 0.0
        best_c = best_c5
        range_dt = t_max - t_min
        for frac in [0.2, 0.4, 0.5, 0.6, 0.8]:
            t = t_min + frac * range_dt
            h_try = h.copy()
            h_try[i] += t
            h_try[j] -= t
            c = compute_c5(h_try, n)
            if c < best_c:
                best_c = c
                best_t = t
        for s in [1, -1]:
            for d in [0.005, 0.02, 0.05, 0.1]:
                t = s * d
                if t_min + 1e-10 <= t <= t_max - 1e-10:
                    h_try = h.copy()
                    h_try[i] += t
                    h_try[j] -= t
                    c = compute_c5(h_try, n)
                    if c < best_c:
                        best_c = c
                        best_t = t
        if best_t != 0.0:
            h = h.copy()
            h[i] += best_t
            h[j] -= best_t
            h = np.clip(h, 0, 1)
            h = project_box_sum(h, n)
            best_c5 = best_c
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    # Phase 4: Intensive local search on best (remaining time - 20s)
