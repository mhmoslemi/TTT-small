    dx = 2.0 / n
    one_minus_h = 1.0 - h
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8

        corr = np.correlate(h, one_minus_h, mode="full")
        # Smooth max
        shift = np.max(corr)
        weights = np.exp((corr - shift) / temp)
        weights /= weights.sum()

        if mask.sum() > 0:
            h_new[mask] += d / mask.sum()
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    return h

    for t in range(1, n_iters + 1):
        progress = (t - 1) / max(1, n_iters - 1)
        temp = temp_start * (temp_end / temp_start) ** progress
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(corr)
        shifted = np.clip((corr - max_c) / temp, -60, 0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        top_idx = np.argsort(corr)[-top_k:]
        w_top = weights[top_idx].copy()
        w_sum = w_top.sum()
        if w_sum < 1e-30:
            w_top = np.ones(top_k) / top_k
        else:
            w_top /= w_sum
        
        grad = compute_grad_topk(h, top_idx, w_top) * dx
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = np.clip(h_new, 0.0, 1.0)
        d = n / 2.0 - h_new.sum()
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_new[mask] += d / mask.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
        
        if t % 100 == 0 or t == n_iters:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                last_improve = t
            elif t - last_improve > patience:
                m = np.zeros(n)
                v = np.zeros(n)
                h = best_h.copy()
                last_improve = t
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n_levels=15, max_sweeps=8):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    h = h.copy()
    grid = np.linspace(0.0, 1.0, n_levels)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_h = h.copy()
            for v in grid:
                if abs(v - orig) < 1e-10:
                    continue
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
            if best_c < c5 - 1e-14:
                h = best_h
                c5 = best_c
                improved = True
            else:
                h[j] = orig
        if not improved:
            break
    return h, c5

def interp_h(h_from, n_to):
    x_old = np.linspace(0, 1, len(h_from), endpoint=False)
    x_new = np.linspace(0, 1, n_to, endpoint=False)
    return np.interp(x_new, x_old, h_from)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Adam at original resolution - explore
    if remaining() > 120:
        h_a, c = adam_anneal(h0, 10000, lr=0.08, temp_start=200.0, temp_end=1.0, top_k=20, seed=seed)
        if c < best_c5:
            best_c5 = c
            best_h = h_a
    
    # Phase 2: Adam - exploit
    if remaining() > 100:
        h_a, c = adam_anneal(best_h, 10000, lr=0.03, temp_start=30.0, temp_end=0.1, top_k=25, seed=seed + 100)
        if c < best_c5:
            best_c5 = c
            best_h = h_a
    
    # Phase 3: Fine Adam with very low temp
    if remaining() > 80:
        h_a, c = adam_anneal(best_h, 8000, lr=0.01, temp_start=5.0, temp_end=0.01, top_k=30, seed=seed + 200)
        if c < best_c5:
            best_c5 = c
            best_h = h_a
    
    # Phase 4: Coordinate descent polish
    if remaining() > 40:
        h_cd, c = coord_descent(best_h, n_levels=15, max_sweeps=6)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd
    
    # Phase 5: Multi-resolution search
    res_list = [64, 96, 128, 192, 256, 512, 768]
    for n_try in res_list:
        if remaining() < 100:
            break
        n_cur = len(best_h)
        h_new = interp_h(best_h, n_try)
        h_new = project_feasible(h_new, n_try / 2.0)
        
        h_a, c = adam_anneal(h_new, 6000, lr=0.05, temp_start=80.0, temp_end=0.5, top_k=15, seed=seed + n_try)
        if c < best_c5:
            best_c5 = c
            best_h = h_a
    
    # Phase 6: Random restarts from best
    n_restarts = 0
    while remaining() > 80 and n_restarts < 25:
        n_restarts += 1
        if remaining() < 60:
            break
        scale = 0.15 * np.sqrt(1.0 / len(best_h))
        h_r = best_h + np.random.randn(len(best_h)) * scale
        h_r = project_feasible(h_r, len(best_h) / 2.0)
        h_r, c = adam_anneal(h_r, 4000, lr=0.04, temp_start=40.0, temp_end=0.2, top_k=15, seed=seed + 1000 + n_restarts)
        if c < best_c5 - 1e-9:
            best_c5 = c
            best_h = h_r
    
    # Phase 7: Try a few fresh random starts at low res then upsample
    if remaining() > 150:
        for n_low in [32, 48, 64]:
            if remaining() < 100:
                break
            for trial in range(3):
                if remaining() < 60:
                    break
                h_rand = np.random.rand(n_low) * 2.0
                h_rand = project_feasible(h_rand, n_low / 2.0)
                h_up = interp_h(h_rand, max(n_low * 2, 128))
                h_up = project_feasible(h_up, len(h_up) / 2.0)
                h_a, c = adam_anneal(h_up, 5000, lr=0.05, temp_start=60.0, temp_end=0.3, top_k=15, seed=seed + 5000 + n_low * 10 + trial)
                if c < best_c5 - 1e-9:
                    best_c5 = c
                    best_h = h_a
    
    # Phase 8: Final coordinate descent
    if remaining() > 25:
        n_levels = 25 if len(best_h) <= 256 else 15
        h_cd, c = coord_descent(best_h, n_levels=n_levels, max_sweeps=5)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd
    
    # Final feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
