            lo = m1
    return (lo + hi) * 0.5

def lagr_cd(x0, time_limit, seed):
    x = repair(x0).copy()
    n = x.size
    dx = 2.0 / float(n)
    s0 = n - 1
    s_vals = np.arange(2 * n - 1) - (n - 1)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + max(1.0, float(time_limit))
    rng = np.random.RandomState(int(seed))
    mu = 0.0
    mu_lr = 0.04
    no_imp = 0
    k_top = 3

    while time.time() < deadline:
        c_cur = c_vals(x)
        k_top = min(2 * n - 1, 3 + (no_imp // 15) * 2)
        order = rng.permutation(n)
        for ii in order:
            i = int(ii)
            idx_p = i + s_vals
            idx_m = i - s_vals
            vp = (idx_p >= 0) & (idx_p < n)
            vm = (idx_m >= 0) & (idx_m < n)
            a = np.zeros(2 * n - 1)
            a[vp] += (1.0 - x[idx_p[vp]])
            a[vm] += (-x[idx_m[vm]])
            a *= dx
            a[s0] = dx
            b = c_cur - a * x[i]
            b[s0] = c_cur[s0] - dx * x[i] + dx * x[i] * x[i]
            t_new = min_1d(a, b, s0, dx, mu)
            x[i] = t_new

        diff = float(x.sum()) - 0.5 * n
        mu += mu_lr * diff
        mu = float(np.clip(mu, -5.0, 5.0))
        x = repair(x)
        c = exact_c(x)
        if c < best_c - 1e-15:
            best_c = c
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 40:
            scale = 0.003 + 0.002 * min(3, (no_imp - 40) // 20)
            x = repair(x + rng.normal(0.0, scale, n))
            mu = 0.0
            no_imp = 0
        if no_imp > 150:
            break
    return best_x, best_c

def pgd_refine(x0, time_limit, seed):
    x = repair(x0).copy()
    n = x.size
    dx = 2.0 / float(n)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + max(1.0, float(time_limit))
    rng = np.random.RandomState(int(seed))
    v = np.zeros(n)
    mom = 0.85
    alpha = 0.5
    no_imp = 0

    while time.time() < deadline:
        c = c_vals(x)
        C = float(np.max(c))
        if C < best_c - 1e-15:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 60:
            x = repair(x + rng.normal(0.0, 0.008, n))
            v[:] = 0.0
            no_imp = 0
            continue
        idx = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        s_vals = np.arange(2 * n - 1) - (n - 1)
        for k in idx:
            s = int(k) - (n - 1)
            ip = np.arange(n) + s
            im = np.arange(n) - s
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            g[vp] += (1.0 - x[ip[vp]])
            g[vm] += (-x[im[vm]])
        g *= dx / float(len(idx))
        v = mom * v + (1.0 - mom) * g
        x_new = repair(x - alpha * v)
        c_new = exact_c(x_new)
        if c_new < C:
            x = x_new
            alpha = min(2.0, alpha * 1.05)
        else:
            alpha *= 0.7
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        x0 = repair(np.asarray(raw, dtype=float))
    else:
        n = 70
        t = (np.arange(n) + 0.5) * 2.0 / n
        x0 = repair(0.5 + 0.35 * np.sin(2.0 * np.pi * t) + 0.15 * np.sin(4.0 * np.pi * t))

    best_x = x0.copy()
    best_c = exact_c(x0)

    # Phase 1: Lagrangian CD at native resolution (45% budget)
    t1 = (deadline - time.time()) * 0.45
    x1, c1 = lagr_cd(x0, t1, sv)
    if c1 < best_c:
        best_c, best_x = c1, x1

    # Phase 2: PGradient refine (15% budget)
    t2 = (deadline - time.time()) * 0.25
    x2, c2 = pgd_refine(best_x, t2, sv + 500)
    if c2 < best_c:
        best_c, best_x = c2, x2

    # Phase 3: Multi-resolution (25% budget)
    rem = deadline - time.time()
    for nn in (50, 80, 100, 140):
        if time.time() > deadline - 80:
            break
        xn = resample(best_x, nn)
        tn = min(rem * 0.22, 80.0)
        xn, cn = lagr_cd(xn, tn, sv + 1000 + nn)
        if cn < best_c:
            best_c, best_x = cn, xn
        rem = deadline - time.time()

    # Phase 4: Final polish (remaining)
    tl = deadline - time.time() - 3.0
    if tl > 10.0:
        n_best = best_x.size
        xp = resample(best_x, n_best)
        xp, cp = lagr_cd(xp, tl * 0.6, sv + 9999)
        if cp < best_c:
            best_c, best_x = cp, xp
        tl2 = deadline - time.time() - 2.0
        if tl2 > 5.0:
            xr, cr = pgd_refine(best_x, tl2, sv + 7777)
            if cr < best_c:
                best_c, best_x = cr, xr

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
