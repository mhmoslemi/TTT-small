def softmax_grad(h, corr, dx, alpha, top_k=None):
    """Compute weighted gradient using softmax over lags"""
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        idx = np.argmax(corr)
        return grad_k(h, idx - (N-1)) * dx
    w /= wsum
    if top_k is not None and top_k < len(w):
        # Only use top-k weights
        top_idx = np.argsort(w)[-top_k:]
        w_masked = np.zeros_like(w)
        w_masked[top_idx] = w[top_idx]
        w_masked /= w_masked.sum()
        w = w_masked
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-7:
            continue
        g += w[j] * grad_k(h, j - (N-1))
    return g * dx

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_obj(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_k(h, k):
    N = len(h)
    g = np.zeros(N)
    if k >= 0:
        nv = N - k
        if nv > 0:
            g[0:nv] += (1.0 - h[k:N])
            g[k:N] -= h[0:nv]
    else:
        kp = -k
        nv = N - kp
        if nv > 0:
            g[kp:N] += (1.0 - h[0:nv])
            g[0:nv] -= h[kp:N]
    return g

def softmax_grad(h, corr, dx, alpha, top_k=25):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        idx = int(np.argmax(corr))
        return grad_k(h, idx - (N - 1)) * dx
    w /= wsum
    g = np.zeros(N)
    if top_k < len(w):
        top_idx = np.argsort(w)[-top_k:]
        ws = w[top_idx]
        ws /= ws.sum()
        for wi, ji in zip(ws, top_idx):
            if wi > 1e-8:
                g += wi * grad_k(h, int(ji) - (N - 1))
    else:
        for j in range(len(w)):
            if w[j] > 1e-8:
                g += w[j] * grad_k(h, j - (N - 1))
    return g * dx

def nesterov_opt(h, N, rng, t_end, alpha_start=25.0, lr0=0.04, max_iter=200000):
    val, corr, dx = compute_obj(h)
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    lr = lr0
    best_val = val
    best_h = h.copy()
    stagnant = 0
    alpha = alpha_start
    for it in range(max_iter):
        if time.time() > t_end:
            break
        beta = (t - 1.0) / (t + 1.0)
        y = h_curr + beta * (h_curr - h_prev)
        y = project_feasible(y, N)
        val_y, corr_y, _ = compute_obj(y)
        g = softmax_grad(y, corr_y, dx, alpha, top_k=25)
        h_new = project_feasible(y - lr * g, N)
        val_new, _, _ = compute_obj(h_new)
        if val_new < val - 1e-16:
            h_prev = h_curr
            h_curr = h_new
            val = val_new
            t += 1.0
            lr = min(lr * 1.005, 0.12)
            stagnant = 0
            if val < best_val:
                best_val = val
                best_h = h_curr.copy()
        else:
            stagnant += 1
            lr *= 0.85
            if stagnant > 35:
                if stagnant % 35 == 0:
                    alpha = max(12.0, alpha * 0.75)
                    lr = lr0
                scale = 0.02 + 0.04 * min(stagnant / 200.0, 1.0)
                h_curr = project_feasible(best_h + rng.randn(N) * scale, N)
                h_prev = h_curr.copy()
                val, _, _ = compute_obj(h_curr)
                t = 1.0
                stagnant = 0
                if val < best_val:
                    best_val = val
                    best_h = h_curr.copy()
    return best_h, best_val

def interpolate_to_n(h_src, n_dst):
    x0 = np.linspace(0, 1, len(h_src), endpoint=False)
    x1 = np.linspace(0, 1, n_dst, endpoint=False)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def structured_perturb(h, rng, N):
    mode = rng.randint(1, max(2, N // 8))
    amp = 0.02 + 0.04 * rng.random()
    phase = rng.uniform(0, 2.0 * np.pi)
    x = np.arange(N) / N * 2.0 * np.pi
    pert = amp * np.sin(mode * x + phase) + 0.015 * rng.randn(N)
    return project_feasible(h + pert, N)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_h = None
    best_val = float('inf')

    if h_init is not None:
        v_init = compute_obj(h_init)[0]
        best_h = h_init.copy()
        best_val = v_init

    # Phase 1: Nesterov refinement at original resolution
    if h_init is not None and time.time() < t_end - 150:
        N0 = len(h_init)
        t_phase = min(t0 + 350, t_end - 150)
        h, v = nesterov_opt(h_init.copy(), N0, rng, t_phase,
                           alpha_start=25.0, lr0=0.04, max_iter=300000)
        if v < best_val:
            best_val = v
            best_h = h

    # Phase 2: Multi-resolution exploration
    if time.time() < t_end - 60:
        n_candidates = [50, 60, 70, 80, 90, 100, 110, 130, 140, 160]
        n_candidates = [n for n in n_candidates if n != len(h_init) if h_init is not None]
        rng.shuffle(n_candidates)
        n_candidates = n_candidates[:6]
        for n_t in n_candidates:
            if time.time() > t_end - 40:
                break
            if best_h is not None:
                h_start = interpolate_to_n(best_h, n_t)
            else:
                h_start = project_feasible(0.5 * np.ones(n_t) + 0.05 * rng.randn(n_t), n_t)
            t_phase = min(time.time() + 100, t_end - 30)
            h, v = nesterov_opt(h_start, n_t, rng, t_phase,
                               alpha_start=25.0, lr0=0.04, max_iter=150000)
            if v < best_val:
                best_val = v
                best_h = h

    # Phase 3: Structured restarts on best solution
    if best_h is not None and time.time() < t_end - 40:
        N_best = len(best_h)
        n_restarts = 12
        for r in range(n_restarts):
            if time.time() > t_end - 15:
                break
            h_start = structured_perturb(best_h, rng, N_best)
            t_phase = min(time.time() + 55, t_end - 10)
            h, v = nesterov_opt(h_start, N_best, rng, t_phase,
                               alpha_start=30.0, lr0=0.03, max_iter=100000)
            if v < best_val:
                best_val = v
                best_h = h

    # Phase 4: High-alpha exploitation (sharpen the max)
    if best_h is not None and time.time() < t_end - 15:
        N_best = len(best_h)
        t_phase = t_end - 3
        h, v = nesterov_opt(best_h.copy(), N_best, rng, t_phase,
                           alpha_start=60.0, lr0=0.015, max_iter=200000)
        if v < best_val:
            best_val = v
            best_h = h

    # Phase 5: Very fine exploitation
    if best_h is not None and time.time() < t_end - 3:
        N_best = len(best_h)
        h, v = nesterov_opt(best_h.copy(), N_best, rng, t_end - 1,
                           alpha_start=80.0, lr0=0.008, max_iter=100000)
        if v < best_val:
            best_val = v
            best_h = h

    if best_h is None:
        best_h = project_feasible(0.5 * np.ones(120), 120)
        best_val = compute_obj(best_h)[0]

    return best_h, best_val, len(best_h)
