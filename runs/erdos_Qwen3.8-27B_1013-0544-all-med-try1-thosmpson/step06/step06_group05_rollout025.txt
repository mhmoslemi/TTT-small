import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.round(np.asarray(h, dtype=float))
    h = np.clip(h, 0, 1)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        h[ones[target:]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1
    return h

                h[j] = 1.0

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fast(h, corr):
    """Update c5 efficiently after a known change."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def make_binary(h, n, rng=None):
    h = np.round(np.asarray(h, dtype=float)).clip(0, 1)
    n_ones = int(np.sum(h))
    target = n // 2
    if rng is None:
        rng = np.random.RandomState(0)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(len(ones), n_ones - target, replace=False)
        h[ones[idx]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(len(zeros), target - n_ones, replace=False)
        h[zeros[idx]] = 1
    return h

def local_search(h, max_time, seed=42, use_double=True):
    """Best-improvement local search for binary h."""
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    start = time.time()
    no_improve_count = 0
    
    while (time.time() - start) < max_time:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_gain = 1e-10
        best_move = None
        best_type = 'single'
        
        # Single swaps
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                h[i] = 1.0; h[j] = 0.0
        
        # Double swaps (sampled) if time allows
        if use_double and (time.time() - start) < max_time * 0.6:
            n_pairs = min(len(ones) * (len(ones)-1) // 2 * 2, 200)
            rng = np.random.RandomState(int(time.time() * 1000) % 100000)
            for _ in range(n_pairs):
                i1_idx = rng.randint(len(ones))
                i2_idx = rng.randint(len(ones))
                j1_idx = rng.randint(len(zeros))
                j2_idx = rng.randint(len(zeros))
                if i1_idx == i2_idx or j1_idx == j2_idx:
                    continue
                i1, i2 = ones[i1_idx], ones[i2_idx]
                j1, j2 = zeros[j1_idx], zeros[j2_idx]
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i1, i2, j1, j2)
                    best_type = 'double'
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
        
        if best_move is not None:
            if best_type == 'single':
                i, j = best_move
                h[i] = 0.0; h[j] = 1.0
            else:
                i1, i2, j1, j2 = best_move
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            current_c5 = compute_c5(h)

        else:

            if no_improve_count >= 2:

def simulated_annealing(h, max_time, seed=42):
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    
    T0 = 0.1
    Tend = 0.001
    
    while (time.time() - start) < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T0 * (Tend / T0) ** frac
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if rng.random() < 0.8:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0; h[j] = 1
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1; h[j] = 0
        else:
            # Double swap
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
    
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old)

def golden_ratio_init(n, seed=0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    indices = set()
    for i in range(n // 2):
        idx = int((i + 1) * n * phi) % n
        if idx not in indices:
            h[idx] = 1.0
            indices.add(idx)
            if len(indices) == n // 2:
                break
    # Fill remaining if needed
    if len(indices) < n // 2:
        remaining = np.setdiff1d(np.arange(n), list(indices))
        for idx in remaining[:n//2 - len(indices)]:
            h[idx] = 1.0
    return h

2. Perturb x* (change k random positions) → get x'.
3. Do local search from x' → get new local optimum.
4. Keep the best of all local optima found.
5. Repeat.

This is a standard approach for combinatorial optimization and might be more effective than SA for this problem.

Also, I realize that for the local search, I should be smarter about which swaps to try. Instead of trying ALL O(n²) swaps every iteration, I can:
1. First, identify the "worst" lags (those with c(k) close to the maximum).
2. For those lags, identify which positions are contributing most to the overlap.
3. Try swaps that reduce the contribution at those specific lags.

This is a form of "guided local search" or "large neighborhood search".

Let me implement this:

```python
def guided_local_search(h, max_time, seed=42):
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        # Find the worst lag
        max_idx = np.argmax(corr)  # index in the full correlation
        max_lag = max_idx - (n - 1)  # actual lag k
        
        # Find positions contributing to this lag
        # c(k) = sum_i h[i] * (1-h[i+k]) for valid i
        # The positions i where h[i]=1 and h[i+k]=0 are the "bad" ones
        if max_lag >= 0:
            bad_ones = []
            for i in range(n - max_lag):
                if h[i] > 0.5 and h[i + max_lag] < 0.5:
                    bad_ones.append(i)
        else:
            bad_ones = []
            for i in range(-max_lag, n):
                if h[i] > 0.5 and h[i + max_lag] < 0.5:
                    bad_ones.append(i)
        
        if not bad_ones:
            break
        
        # Try to move one of the bad ones to a position that helps
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        best_gain = 1e-10
        best_move = None
        
        # Try moving bad ones to zeros (prioritize bad ones)
        for i in bad_ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                h[i] = 1; h[j] = 0
        
        # If no improvement from bad ones, try all swaps
        if best_move is None:
            for i in ones:
                for j in zeros:
                    h[i] = 0; h[j] = 1
                    new_c5 = compute_c5(h)
                    gain = current_c5 - new_c5
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i, j)
                    h[i] = 1; h[j] = 0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0; h[j] = 1
            current_c5 = compute_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            break
    
    return h, compute_c5(h)

Hmm, but this is still O(n²) per iteration and might not be much better than the vanilla local search.

Let me think about this from a completely different angle. What if I use a **gradient-based approach on the continuous relaxation**?

The continuous problem: minimize max_k ∫ h(x)(1-h(x+k)) dx subject to 0 ≤ h ≤ 1 and ∫h = 1.

This is a min-max problem. We can write it as:
min_{h, t} t
subject to: ∫ h(x)(1-h(x+k)) dx ≤ t for all k
           0 ≤ h(x) ≤ 1
           ∫ h = 1

This is a convex optimization problem! (The constraints are bilinear in h, so... wait, they're not convex because h*(1-h) is not convex.)

Actually, h(x)*(1-h(x+k)) is bilinear in h (linear in h(x) and linear in h(x+k)), so the constraint ∫ h(x)(1-h(x+k)) dx ≤ t is NOT convex in h. So this is a non-convex problem.

However, if we fix all variables except one, say h(x₀), then the constraint becomes:
h(x₀) * ∫_{x: x+k=x₀} (1-h(x)) dx + (terms not involving h(x₀)) ≤ t
which is linear in h(x₀). So we can do coordinate descent!

For each coordinate i, the objective (max over k of c(k)) as a function of h[i] (with all other h fixed) is:
max_k [a_k * h[i] + b_k]
where a_k and b_k depend on the other h values and the lag k.

Specifically, c(k) = sum_j h[j](1-h[j+k]). The terms involving h[i] are:
- When j = i: h[i](1-h[i+k])
- When j+k = i, i.e., j = i-k: h[i-k](1-h[i])

So c(k) = h[i](1-h[i+k]) + h[i-k](1-h[i]) + (terms not involving h[i])
        = h[i](1-h[i+k]) - h[i]*h[i-k] + h[i-k] + c₀(k)
        = h[i](1 - h[i+k] - h[i-k]) + h[i-k] + c₀(k)

Wait, that's not right either. Let me be more careful.

c(k) = sum_j h[j] * (1 - h[j+k])

Terms involving h[i]:
1. j = i: h[i] * (1 - h[i+k])
2. j+k = i, so j = i-k: h[i-k] * (1 - h[i])

So c(k) = h[i](1-h[i+k]) + h[i-k](1-h[i]) + rest(k)
        = h[i] - h[i]*h[i+k] + h[i-k] - h[i]*h[i-k] + rest(k)
        = h[i](1 - h[i+k] - h[i-k]) + h[i-k] + rest(k)

This is LINEAR in h[i]! (The coefficients depend on other h values but not on h[i].)

So for each k, c(k) is linear in h[i]. Therefore, max_k c(k) is the max of linear functions of h[i], which is convex in h[i].

This means we can do coordinate descent: for each i, minimize max_k c(k) over h[i] ∈ [0,1], holding all other h fixed. This is a 1D convex optimization problem!

The minimum of max_k (a_k * x + b_k) over x ∈ [0,1] can be found by:
- If all a_k ≤ 0: minimum at x = 1 (or the constraint boundary)
- If all a_k ≥ 0: minimum at x = 0
- Otherwise: the minimum is at the intersection of the two "active" lines

Actually, for a piecewise-linear convex function f(x) = max_k (a_k * x + b_k), the minimum over [0,1] is at:
- x = 0, if f'(0⁺) ≥ 0 (i.e., the max slope at x=0 is ≥ 0)
- x = 1, if f'(1⁻) ≤ 0 (i.e., the min slope at x=1 is ≤ 0)
- x = x* where two lines cross, if f'(0⁺) < 0 < f'(1⁻)

The crossing point of lines (a_p, b_p) and (a_q, b_q) is at x = (b_q - b_p) / (a_p - a_q).

This is implementable! And it gives exact coordinate minima.

But wait, we also have the constraint sum(h) = n/2. In coordinate descent, we need to handle this. Options:
1. Use a Lagrange multiplier: minimize max_k c(k) + λ * (sum(h) - n/2). Then adjust λ to satisfy the constraint.
2. Use a "swap" version: instead of changing h[i] alone, change h[i] and h[j] simultaneously (one up, one down) to preserve the sum.

Option 2 is cleaner. For a pair (i, j), we change h[i] by δ and h[j] by -δ, maintaining the sum. The new c(k) becomes:
c(k) → c(k) + δ * (a_k^(i) - a_k^(j))
where a_k^(i) = 1 - h[i+k] - h[i-k] (coefficient of h[i] in c(k)) and similarly for j.

So we're looking for δ that minimizes max_k [c(k) + δ * d_k] where d_k = a_k^(i) - a_k^(j).

This is again a 1D convex problem! The minimum of max_k (c_k + δ * d_k) over δ such that h[i]+δ ∈ [0,1] and h[j]-δ ∈ [0,1].

The optimal δ can be found by:
- If all d_k ≤ 0: increase h[i] as much as possible (δ = 1 - h[i])
- If all d_k ≥ 0: decrease h[i] as much as possible (δ = -h[i])
- Otherwise: find the crossing point

This is very efficient! Each "step" is O(n) to compute the d_k values and O(n) to find the optimal δ. And we can cycle through all pairs (i, j) in a random order.

This is essentially a **subgradient method** or **coordinate descent** for the min-max problem, and it should converge to a local minimum.

Let me implement this:

```python
def coord_descent(h, max_time, seed=42):
    n = len(h)
    h = h.astype(float).copy()
    h = np.clip(h, 0, 1)
    # Enforce sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Redistribute
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-10:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-10:
                h *= (1.0 + diff / total)
        h = np.clip(h, 0, 1)
    # Fix any remaining discrepancy
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        improved = False
        # Random order of pairs
        indices = rng.permutation(n)
        
        for i in indices:
            # Try to adjust h[i] (and compensate with a random j)
            # Actually, for simplicity, let's do: pick j, find optimal δ
            j = rng.randint(n)
            if i == j:
                continue
            
            # Compute d_k = a_k^(i) - a_k^(j) for all k
            # a_k^(i) = coefficient of h[i] in c(k)
            # c(k) = sum_m h[m](1-h[m+k])
            # Terms with h[i]: m=i gives h[i](1-h[i+k]), m=i-k gives h[i-k](1-h[i])
            # So a_k^(i) = (1-h[i+k]) - h[i-k] (with bounds checking)
            
            # Similarly for j
            # d_k = a_k^(i) - a_k^(j)
            
            # For the full correlation (2n-1 values):
            # Index in full corr = n-1+k where k is the lag
            # For lag k:
            #   a_k^(i) = (1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n]
            
            # Let's compute this for all k at once
            k_vals = np.arange(-(n-1), n)  # all possible lags
            
            # a_k^(i): (1-h[i+k]) for valid i+k, minus h[i-k] for valid i-k
            a_i = np.zeros(2*n-1)
            valid_ip = (k_vals >= -i) & (k_vals <= n-1-i)  # i+k in [0,n-1]
            a_i[valid_ip] += (1 - h[i + k_vals[valid_ip]])
            valid_ik = (k_vals >= i-n+1) & (k_vals <= i)  # i-k in [0,n-1]
            a_i[valid_ik] -= h[i - k_vals[valid_ik]]
            
            a_j = np.zeros(2*n-1)
            valid_jp = (k_vals >= -j) & (k_vals <= n-1-j)
            a_j[valid_jp] += (1 - h[j + k_vals[valid_jp]])
            valid_jk = (k_vals >= j-n+1) & (k_vals <= j)
            a_j[valid_jk] -= h[j - k_vals[valid_jk]]
            
            d = a_i - a_j
            
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # Find δ that minimizes max_k (corr[k] + δ * d[k])
            # subject to: 0 ≤ h[i] + δ ≤ 1 and 0 ≤ h[j] - δ ≤ 1
            # i.e., δ ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]
            
            delta_min = max(-h[i], h[j] - 1)
            delta_max = min(1 - h[i], h[j])
            
            if delta_min >= delta_max:
                continue
            
            # f(δ) = max_k (corr[k] + δ * d[k])
            # This is a piecewise linear convex function.
            # Find its minimum on [delta_min, delta_max].
            
            # The minimum is at a point where two lines cross, or at a boundary.
            # Lines: f_p(δ) = corr[p] + δ * d[p]
            # f_p(0) = corr[p]
            # f_p(delta) = corr[p] + delta * d[p]
            
            # At δ = 0, the max is corr_max = max(corr)
            # We want to find if we can reduce the max by changing δ.
            
            # The subgradient of f(δ) at the maximum point is the set of d[p] for p that achieve the max.
            # If 0 is in the subgradient (i.e., there exist p,q with d[p] < 0 < d[q] both achieving the max),
            # then δ=0 is optimal for this pair.
            # Otherwise, we should move in the direction that reduces the max.
            
            # Simple approach: try a few candidate δ values
            # 1. δ such that the max decreases (if possible)
            # 2. The boundaries
            
            best_delta = 0
            best_val = np.max(corr)
            
            # Try boundary values
            for delta_test in [delta_min, delta_max, 0]:
                val = np.max(corr + delta_test * d)
                if val < best_val:
                    best_val = val
                    best_delta = delta_test
            
            # Try the "crossing" points
            # Sort by slope d[k]
            # The upper envelope changes at crossing points
            # For efficiency, just try a few:
            max_idx = np.argmax(corr)
            if d[max_idx] != 0:
                # At the current max, the slope is d[max_idx]
                # If d[max_idx] < 0, increasing δ helps (reduce the max)
                # If d[max_idx] > 0, decreasing δ helps
                # The new max will be at a different k
                # Try δ = -(corr[max_idx] - corr[second_max]) / (d[max_idx] - d[second_max])
                sorted_corr = np.sort(corr)[::-1]
                if len(sorted_corr) > 1 and sorted_corr[1] > 1e-10:
                    # Find second max
                    corr_copy = corr.copy()
                    corr_copy[max_idx] = -np.inf
                    second_idx = np.argmax(corr_copy)
                    if d[max_idx] != d[second_idx]:
                        delta_cross = -(corr[max_idx] - corr[second_idx]) / (d[max_idx] - d[second_idx])
                        if delta_min < delta_cross < delta_max:
                            val = np.max(corr + delta_cross * d)
                            if val
