For the 2-point optimization, here's a key insight: if the active lag is m*, then the gradient of f_{m*} tells us which pairs (i,j) would most help. Specifically, if we transfer from j to i, the change in f_{m*} is:
Δf_{m*} = dx * [(1-h[i+m*] - h[i-m*]) - (1-h[j+m*] - h[j-m*])] * δ (approximately)

So we want to transfer from j where (1-h[j+m*] - h[j-m*]) is large (i.e., where f_{m*} increases most) to i where (1-h[i+m*] - h[i-m*]) is small (or negative).

This gives us a **greedy coordinate descent** that's much more directed than random swaps.

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target_sum = n / 2.0
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h = h + (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def run_slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_try = project_feasible(res.x, n)
        return h_try, compute_c5(h_try, n)
    except Exception:
        return h, compute_c5(h, n)

def active_gradient(h, n, top_k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    m = c.max()
    weights = np.exp(100 * (c[idx] - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, mi in zip(weights, idx):
        lag = mi - (n - 1)
        for k in range(n):
            if 0 <= k - lag < n:
                grad[k] += w * (1 - h[k - lag])
            if 0 <= k + lag < n:
                grad[k] -= w * h[k + lag]
    grad *= dx
    return grad, c

def directed_cd(h, n, rng, t0, time_limit):
    c5 = compute_c5(h, n)
    step = 0.05
    no_improve = 0
    while time.time() - t0 < time_limit and no_improve < 200:
        grad, c = active_gradient(h, n, top_k=5)
        i = np.argmin(grad)
        j = np.argmax(grad)
        if i == j:
            no_improve += 1
            continue
        # Also try random pairs for diversity
        if rng.random() < 0.3:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        max_d = min(1.0 - h[i], h[j])
        if max_d < 1e-6:
            # Try reverse
            i, j = j, i
            max_d = min(1.0 - h[i], h[j])
            if max_d < 1e-6:
                no_improve += 1
                continue
        best_d = 0.0
        best_val = c5
        fracs = [0.1, 0.25, 0.5, 0.75, 1.0]
        for frac in fracs:
            d = max_d * frac
            h_try = h.copy()
            h_try[i] += d
            h_try[j] -= d
            val = compute_c5(h_try, n)
            if val < best_val:
                best_val = val
                best_d = d
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5 = best_val
            no_improve = 0
            step = max(0.02, step * 0.95)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                step = max(0.01, step * 0.7)
    return h, c5

def make_sinusoidal_init(n, freq=1.0, amp=0.4):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x)
