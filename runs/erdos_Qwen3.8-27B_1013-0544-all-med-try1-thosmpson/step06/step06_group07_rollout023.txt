        C = float(np.max(c))
        if C < best_c:
            best_c = C
            best_h = h.copy()

        thresh = C - active_tol
        active = np.where(c >= thresh)[0]
        if len(active) > max_active:
            idx = np.argsort(c)[::-1][:max_active]
            active = idx

        c_vec[0] = 1.0  # minimize t

            b_ub[i] = rhs

        A_eq[0, 1:] = 1.0
        b_eq = np.array([n/2.0])

        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res is None or not res.success:
            break
        
        new_h = res.x[1:]
        new_h = repair(new_h)
        new_C = exact_c(new_h)

            else:

import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")


def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-15
            if not mask.any():
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-15
            if not mask.any():
                break
            x[mask] += diff * x[mask] / x[mask].sum()
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        x = np.clip(x, 0, 1)
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                room = 1.0 - x[idx]
                if room > 0:
                    a = min(diff, room)
                    x[idx] += a
                    diff -= a
            else:
                room = x[idx]
                if room > 0:
                    a = max(diff, -room)
                    x[idx] += a
                    diff -= a
    return x


def compute_c(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c(h):
    return float(np.max(compute_c(h)))


def grad_c(h, k):
    """Gradient of the k-th component of compute_c with respect to h."""
    n = h.size
    dx = 2.0 / n
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = dx * (1.0 - 2.0 * h)
        else:
            g[:n - lag] += dx * (1.0 - h[lag:])
            g[lag:] += -dx * h[:n - lag]
    else:
        m = -lag
        g[m:] += dx * (1.0 - h[:n - m])
        g[:n - m] += -dx * h[m:]
    return g


def lp_step(h, active_set):
    """One LP cutting plane step. Returns new h or None."""
    n = h.size
    c = compute_c(h)
    m = len(active_set)
    # LP variables: [t, h_0, ..., h_{n-1}], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, k in enumerate(active_set):
        g = grad_c(h, k)
        rhs = float(g @ h) - float(c[k])
        A_ub[i, 0] = -1.0
        A_ub[i, 1:] = g
        b_ub[i] = rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(None, None)] + [(0.0, 1.0)] * n
    try:
        res = so.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None
    if res is None or not res.success:
        return None
    return res.x[1:]


def lp_polish(h0, time_limit, max_active=8, tol=1e-5):
    """Iterative LP cutting plane."""
    n = h0.size
    h = repair(h0)
    best_h = h.copy()
    best_c = exact_c(h)
    t0 = time.time()
    for it in range(2000):
        if time.time() - t0 > time_limit:
            break
        c = compute_c(h)
        C = float(np.max(c))
        if C < best_c - 1e-15:
            best_c = C
            best_h = h.copy()
        # Active set: top max_active lags
        idx = np.argsort(c)[::-1][:max_active]
        new_h = lp_step(h, idx)
        if new_h is None:
            break
        new_h = repair(new_h)
        new_C = exact_c(new_h)
        if new_C < C - 1e-12:
            h = new_h
        else:
            # Fallback: try smaller step
            improved = False
            for alpha in (0.5, 0.25, 0.1):
                frac = repair(h + alpha * (new_h - h))
                fc = exact_c(frac)
                if fc < C:
                    h = frac
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c


def diversified_starts(best_h, n, seed, count=6):
    """Generate diverse starting points around best_h."""
    rng = np.random.RandomState(seed)
    starts = []
    for i in range(count):

        s = rng.randn(n)
        s -= s.mean()  # keep sum roughly the same
        h_try = repair(best_h + scale * s)
        starts.append(h_try)
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)
    np.random.seed(seed_val)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size > 10:
        h0 = repair(np.asarray(raw, dtype=float))
    else:
        h0 = repair(np.full(70, 0.5))

    n = h0.size

        if time.time() > t_phase2_end:
