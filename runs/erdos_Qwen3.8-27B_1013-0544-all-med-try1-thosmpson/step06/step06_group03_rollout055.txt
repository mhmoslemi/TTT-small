- Divide h into blocks of size b
- For each block, optimize those b coordinates while keeping others fixed

For a fixed block, the objective is max_k conv[k] * dx, where conv[k] depends linearly on the block variables (since conv[k] = Σ h[i](1-h[i+k]), and if we fix all h except the block, then conv[k] is linear in the block variables). So the subproblem is:
minimize max_k (a_k · x + b_k)
subject to 0 ≤ x_j ≤ 1, sum(x_j) = target_block

This is a linear program (LP)! We can solve it efficiently.

Actually wait, conv[k] = Σ_i h[i](1-h[i+k]). If we fix all h except the block B, then:
conv[k] = (terms not involving B) + (terms involving B)

The terms involving B are of the form: for i in B, h[i] appears as h[i] * (1-h[i+k]) and (1-h[i]) * h[i+k] (when the other index is in B or when i+k is in B).

Actually it's more complex. Let me denote the block as h_B and the rest as h_R.
conv[k] = Σ_{i ∉ B} h[i](1-h[i+k]) + cross terms + Σ_{i ∈ B} h[i](1-h[i+k])

The cross terms are: for i ∉ B, i+k ∈ B: h[i](1-h[i+k]) - these are linear in h_B
And for i ∈ B, i+k ∉ B: h[i](1-h[i+k]) - these are also linear in h_B.
And for i ∈ B, i+k ∈ B: h[i](1-h[i+k]) - these are quadratic in h_B.

So if the block is small and doesn't overlap with its own shifts, the problem is approximately linear. But in general, it's a quadratic program.

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr_init
    for it in range(max_iter):
        progress = it / max(max_iter - 1, 1)
        cur_t = t_sm * (t_an / t_sm) ** progress if t_sm > t_an else t_sm
        c5, sg = worst_shift_grad(h)
        if cur_t > 0.004:
            _, smg = smooth_obj_grad(h, cur_t)
            grad = 0.4 * sg + 0.6 * smg
        else:
            grad = sg
        v = 0.92 * v - lr * grad
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > c5 + 0.008:
            v *= 0.3
            lr = max(lr * 0.93, 1e-7)
            h_new = project_feasible(h + v, n)
            if compute_c5(h_new) < best_c5:
                best_c5 = compute_c5(h_new)
                best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def sa_refine(h0, max_iter, t_start=0.03, t_end=0.0003, seed=42):
    n = len(h0)
    h = h0.copy()
    h_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = h_c5
    rng = np.random.RandomState(seed)
    n_pert = max(4, n // 35)
    for it in range(max_iter):
        progress = it / max(max_iter - 1, 1)
        T = t_start * (t_end / t_start) ** progress
        step = 0.12 * (1.0 - 0.85 * progress)
        idxs = rng.randint(0, n, size=n_pert)
        h_new = h.copy()
        h_new[idxs] += rng.uniform(-step, step, size=n_pert)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < h_c5 or rng.rand() < np.exp(-(c5_new - h_c5) / max(T, 1e-12)):
            h = h_new
            h_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        if it % 800 == 400:
            nl = n // 10
            idxs2 = rng.randint(0, n, size=nl)
            ht = h.copy()
            ht[idxs2] += rng.uniform(-0.2, 0.2, size=nl)
            ht = project_feasible(ht, n)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                h = ht.copy()
                h_c5 = ct
    return best_h, best_c5

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    xo = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    xn = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    hi = np.zeros(n_dst)
    for i in range(n_dst):
        pos = xn[i]
        if pos >= 2.0:
            pos -= 2.0
        j = int(pos * n_src / 2.0)
        j0 = j % n_src
        j1 = (j + 1) % n_src
        frac = (pos * n_src / 2.0) - j
        hi[i] = h_src[j0] * (1.0 - frac) + h_src[j1] * frac
    return hi

def lbfgs_smooth(h0, n, t_val, max_iter=2000):
    def obj(h):
        return smooth_obj_grad(h, t_val)[0] + 1000.0 * (np.sum(h) - n / 2.0) ** 2
    def grad(h):
        return smooth_obj_grad(h, t_val)[1] + 2000.0 * (np.sum(h) - n / 2.0)
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(320, 0.5)
    n_best = len(h_best)
    best_c5 = compute_c5(h_best)

    # Phase 1: Momentum GD at initial n
    if tl() > 200:
        h, c5 = momentum_optimize(h_best, 60000, 0.004, t_sm=0.04, t_an=0.003)
        if c5 < best_c5:
            best_c5, h_best = c5, h
        n_best = len(h_best)

    # Phase 2: SA at initial n
    if tl() > 300:
        h, c5 = sa_refine(h_best, 250000, t_start=0.025, t_end=0.0002, seed=seed)
        if c5 < best_c5:
            best_c5, h_best = c5, h
        n_best = len(h_best)

    # Phase 3: L-BFGS-B polish
    if tl() > 150:
        for tv in [0.01, 0.005]:
            if tl() < 30:
                break
            ht = lbfgs_smooth(h_best, n_best, tv, 1500)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_c5, h_best = c5t, ht
                n_best = len(h_best)

    # Phase 4: Multi-resolution
    for n_try in [240, 200, 400, 480, 600, 800]:
        if tl() < 150:
            break
        if n_try == n_best:
            continue
        hi = project_feasible(interp_h(h_best, n_try), n_try)
        if tl() > 80:
            h, c5 = momentum_optimize(hi, 25000, 0.003, t_sm=0.03, t_an=0.002)
        else:
            h, c5 = hi, compute_c5(hi)
        if tl() > 40 and c5 < best_c5 + 0.004:
            h2, c52 = sa_refine(h, 60000, t_start=0.015, t_end=0.0002, seed=seed + n_try)
            if c52 < c5:
                h, c5 = h2, c52
        if c5 < best_c5:
            best_c5, h_best = c5, h
            n_best = n_try

    # Phase 5: Final momentum polish
    if tl() > 60:
        h, c5 = momentum_optimize(h_best, 40000, 0.0015, t_sm=0.01, t_an=0.001)
        if c5 < best_c5:
            best_c5, h_best = c5, h
        n_best = len(h_best)

    # Phase 6: Final SA
    if tl() > 60:
        h, c5 = sa_refine(h_best, 150000, t_start=0.008, t_end=0.0001, seed=seed + 9999)
        if c5 < best_c5:
            best_c5, h_best = c5, h
        n_best = len(h_best)

    return h_best, best_c5, n_best
