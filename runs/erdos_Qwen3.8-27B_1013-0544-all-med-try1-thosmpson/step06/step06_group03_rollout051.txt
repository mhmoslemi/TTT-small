def conv_at_shift(h, shift):

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def conv_grad_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += (1.0 - h[shift:])
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    return g

def smooth_obj_grad(h, t, k=None):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    if k is not None and k < len(conv):
        top_idx = np.argpartition(-conv, k - 1)[:k]
        vals = conv[top_idx]
        centered = (vals - m) / t
        mc = np.max(centered)
        exps = np.exp(centered - mc)
        s = np.sum(exps)
        obj = t * (m / t + mc + np.log(s)) * dx
        weights = exps / s
        grad = np.zeros(n)
        for j in range(k):
            w = weights[j]
            if w < 1e-10:
                continue
            shift = top_idx[j] - (n - 1)
            grad += w * conv_grad_shift(h, shift)
    else:
        centered = (conv - m) / t
        mc = np.max(centered)
        exps = np.exp(centered - mc)
        s = np.sum(exps)
        obj = t * (m / t + mc + np.log(s)) * dx

            hi = mid
        else:
