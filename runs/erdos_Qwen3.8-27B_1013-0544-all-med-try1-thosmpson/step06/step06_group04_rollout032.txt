    """Project h onto {0 ≤ h[i] ≤ 1, sum(h) = target_sum}."""
    # This is a projection onto the intersection of the box [0,1]^n and the hyperplane sum = target_sum

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def project_feasible(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def smooth_gradient(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_vals = np.exp((corr - max_c) * beta)
    S = np.sum(exp_vals)
    weights = exp_vals / S
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                grad[d:n] += w * dx * (1.0 - h[0:n-d])
                grad[0:n-d] -= w * dx * h[d:n]
        else:
            m = -d
            if m < n:
                grad[0:n-m] += w * dx * (1.0 - h[m:n])
                grad[m:n] -= w * dx * h[0:n-m]
    return grad

def gradient_descend(h, dx, target_sum, beta, lr, deadline, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = smooth_gradient(h, dx, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        alpha = lr
        found = False
        for _ in range(12):
            h_try = project_feasible(h - alpha * grad, target_sum)
            c5_try, _ = compute_c5(h_try, dx)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                found = True
                break
            alpha *= 0.5
        if not found:
            h_try = project_feasible(h - alpha * grad, target_sum)
            if np.linalg.norm(h_try - h) > 1e-14:
                h = h_try
        if it % 150 == 149:
            beta = min(beta * 1.4, 150.0)
            lr = max(lr * 0.7, 0.001)
    return best_h, best_c5

def coord_descent(h, dx, n, target_sum, deadline, rounds=5):
    for rnd in range(rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            cur = np.max(corr)
            cands = [0.0, 1.0, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9,
                     max(0.0, old - 0.1), min(1.0, old + 0.1),
                     max(0.0, old - 0.05), min(1.0, old + 0.05)]
            best_v = old
            best_c = cur
            for cv in cands:
                h[j] = cv
                ct = np.max(np.correlate(h, 1.0 - h, mode='full') * dx)
                if ct < best_c - 1e-12:
                    best_c = ct
                    best_v = cv
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return project_feasible(h, target_sum)

def interpolate_h(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    return np.interp(x_new, x_old, h)

def make_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_feasible(h, float(n) / 2.0)

def make_block_init(n, seed):
    rng = np.random.RandomState(seed)
    n_blocks = rng.randint(3, 8)
    h = np.zeros(n)
    sizes = rng.multinomial(n, np.ones(n_blocks) / n_blocks)
    val = 0
    for i, sz in enumerate(sizes):
        h[val:val+sz] = 0.0 if i % 2 else 1.0
        val += sz
    return project_feasible(h, float(n) / 2.0)
