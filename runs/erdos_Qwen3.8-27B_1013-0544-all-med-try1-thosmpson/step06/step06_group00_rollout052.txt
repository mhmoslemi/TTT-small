import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(corr, temp):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(temp * (corr - m)))) / temp

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        step = d / mask.sum()
        if step > 0:
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            h[mask] = np.maximum(0.0, h[mask] + step)
    return h

def obj_fn(z, temp, n):
    ts = n / 2.0
    h = np.clip(z, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, temp)
    penalty = 200.0 * (h.sum() - ts) ** 2
    return sm * (2.0 / n) + penalty

def optimize_lbfgsb(h_start, n, temp, maxiter=500):
    ts = n / 2.0
    res = minimize(obj_fn, h_start, args=(temp, n), method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, ts)

def binary_local_search(h_init, n, n_flips=3000, seed=0):

        idx = np.where(h == 1)[0]
        to_remove = rng.choice(idx, int(n_ones - ts), replace=False)
        h[to_remove] = 0
    elif n_ones < ts:
        idx = np.where(h == 0)[0]
        to_add = rng.choice(idx, int(ts - n_ones), replace=False)

            zeros = np.where(h == 0)[0]

            k = zeros[rng.randint(len(zeros))]
            h[j] = 0.0
            h[k] = 1.0
            c_new = compute_c5(h)
            if c_new < c5:
                c5 = c_new
            else:
                h[j] = 1.0
                h[k] = 0.0
        else:
            ones = np.where(h == 1)[0]
            if len(ones) == 0:
                break
            k = ones[rng.randint(len(ones))]
            h[j] = 1.0
            h[k] = 0.0
            c_new = compute_c5(h)
            if c_new < c5:
                c5 = c_new
            else:
                h[j] = 0.0
                h[k] = 1.0
    return h, c5

def binary_full_sweep(h, n, max_rounds=3):
    """Try all single 1->0 and 0->1 swaps exhaustively"""
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        for i in range(len(ones_idx)):
            j = ones_idx[i]
            for l in range(len(zeros_idx)):
                k = zeros_idx[l]
                h[j] = 0.0
                h[k] = 1.0
                c_new = compute_c5(h)
                if c_new < c5 - 1e-14:
                    c5 = c_new
                    ones_idx = np.where(h == 1)[0]
                    zeros_idx = np.where(h == 0)[0]
                    improved = True
                    break
                else:
                    h[j] = 1.0
                    h[k] = 0.0
            if improved:

        if not improved:
            break
    return h, c5

def single_coord_descent(h, n_levels=20, max_sweeps=15):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    vals = np.linspace(0, 1, n_levels)

        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_v = orig
            for v in vals:
                if abs(v - orig) < 1e-7:

                h[j] = v

                c = compute_c5(h)
                if c < best_c - 1e-14:

            if best_c < c5 - 1e-14:
                h[j] = best_v
                h = project_feasible(h, ts)
                c5 = best_c
                improved = True
