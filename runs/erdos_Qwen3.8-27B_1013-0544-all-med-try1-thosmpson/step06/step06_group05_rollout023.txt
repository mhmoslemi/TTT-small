For a binary sequence, flipping h[i] from 1 to 0 and h[j] from 0 to 1:

corr_k = sum_i h[i] * (1-h[i+k])

If we flip h[a] from 1 to 0:
- For each k, corr_k changes by:

  - Term where i+k=a (i.e., i=a-k): we change h[a-k]*(1-h[a]) = h[a-k]*(1-1) = 0 to h[a-k]*(1-0) = h[a-k]

= h[a-k] + h[a+k] - h[b-k] - h[b+k]

(where we define h[x] = 0 for x outside [0, n-1])

            else:
                break
    return best_h, best_c

def ls_targeted(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = make_binary(h0, n, seed)
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    while (time.time() - start) < max_time:
        corr = np.correlate(h, 1 - h, mode='full')
        bottleneck_idx = int(np.argmax(corr))
        k_star = bottleneck_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        no, nz = len(ones), len(zeros)
        pa = np.zeros(n)
        for i in range(n):
            v = 0.0
            if 0 <= i + k_star < n:
                v += h[i + k_star]
            if 0 <= i - k_star < n:
                v += h[i - k_star]
            pa[i] = v
        bg = 0.0
        bm = None
        candidates = []
        for i in ones:
            for j in zeros:
                delta = pa[i] - pa[j]
                if delta < 0:
                    candidates.append((-delta, i, j))
        candidates.sort(reverse=True)
        for _, i, j in candidates[:50]:
            h[i] = 0; h[j] = 1
            nc = compute_c5(h)
            g = cur - nc
            if g > bg:
                bg = g; bm = (i, j, 1)
            h[i] = 1; h[j] = 0
        if bm is None:
            for _ in range(3000):
                i1 = ones[rng.randint(no)]
                i2 = ones[rng.randint(no)]
                j1 = zeros[rng.randint(nz)]
                j2 = zeros[rng.randint(nz)]
                if i1 == i2 or j1 == j2:
                    continue
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                nc = compute_c5(h)
                g = cur - nc
                if g > bg:
                    bg = g; bm = (i1, i2, j1, j2, 2)
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
        if bm is not None:
            if bm[-1] == 1:
                h[bm[0]] = 0; h[bm[1]] = 1
            else:
                h[bm[0]] = 0; h[bm[1]] = 0; h[bm[2]] = 1; h[bm[3]] = 1
            cur = compute_c5(h)
            stall = 0
            if cur < best_c:
                best_c = cur; best_h = h.copy()
        else:
            stall += 1
            if stall >= 4:
                k = min(4, no, nz)
                ko = rng.choice(ones, k, replace=False)
                kz = rng.choice(zeros, k, replace=False)
                for x in ko:
                    h[x] = 0
                for x in kz:
                    h[x] = 1
                cur = compute_c5(h)
                stall = 0
            else:
                break
    return best_h, best_c

def continuous_refine(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = np.asarray(h0, dtype=float).copy()
    h = np.clip(h, 0, 1)
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    start = time.time()
    rng = np.random.RandomState(seed)
    def obj(x):
        xx = np.clip(x, 0, 1)
        return compute_c5(xx)
    for attempt in range(20):
        if (time.time() - start) > max_time * 0.85:
            break
        res = minimize(obj, h, method='Nelder-Mead',
                       options={'maxiter': 300, 'xatol': 1e-9, 'fatol': 1e-12,
                                'maxfev': 2000})
        h = np.clip(res.x, 0, 1)
        nc = compute_c5(h)
        if nc < best_c:
            best_c = nc; best_h = h.copy()
        if (time.time() - start) > max_time * 0.85:
            break
        h = np.clip(h + rng.normal(0, 0.05, n), 0, 1)
    diff = n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - best_h
            tr = np.sum(room)
            if tr > 1e-12:
                best_h += room * (diff / tr)
        else:
            t = np.sum(best_h)
            if t > 1e-12:
                best_h *= (1.0 + diff / t)
        best_h = np.clip(best_h, 0, 1)
    return best_h, compute_c5(best_h)

def msequence_start(n, seed=0):
    m = 1
    while (1 << m) - 1 < n:
        m += 1
    L = (1 << m) - 1
    poly = {6: 0b10011, 7: 0b1000111, 5: 0b10111, 4: 0b1011, 8: 0b10010001}
    p = poly.get(m, 0b1000111)
    reg = 1
    seq = []
    for _ in range(L + n):
        seq.append(reg & 1)
        bit = 0
        for k in range(m):
            if p & (1 << k):
                bit ^= (reg >> k) & 1
        reg = (reg >> 1) | (bit << (m - 1))
    seq = np.array(seq[:L])
    if n == L:
        return seq.astype(float)
    if n < L:
        return seq[:n].astype(float)
    padded = np.zeros(n)
    padded[:L] = seq
    extra = n - L
    padded[L:] = rng_local = np.random.RandomState(seed).random(extra)
    return padded

def build_random_h(n, seed=0):
    rng = np.random.RandomState(seed)
    return rng.random(n)
