    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(c.size)
        w[int(c.argmax())] = 1.0
        val = C
    else:
        w = e / s
        val = C + tau * np.log(s)
    grad = np.zeros(n)
    m = c.size
    for ki in range(m):
        wi = w[ki]
        if wi < 1e-12:
            continue
        k = ki - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * x)
        elif k > 0:
            grad[:n - k] += wi * (1.0 - x[k:])
            grad[k:] -= wi * x[:n - k]
        else:
            ak = -k
            grad[ak:] += wi * (1.0 - x[:ak])
            grad[:ak] -= wi * x[ak:]
    grad *= dx
    return val, grad

def slsqp_polish(x0, tau, max_iter):
    x0 = np.asarray(x0, dtype=float)
    n = x0.size
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        v, _ = smooth_fun_and_jac(x, tau)
        return v

    def jac(x):
        _, g = smooth_fun_and_jac(x, tau)
        return g

    def cfun(x):
        return np.array([x.sum() - 0.5 * n])

    def cjac(x):
        return np.ones((1, n))

    try:
        res = so.minimize(fun, x0, method="SLSQP", jac=jac, bounds=bounds,
                          constraints={"type": "eq", "fun": cfun, "jac": cjac},
                          options={"maxiter": max_iter, "ftol": 1e-14})
        if res is not None and res.success is not False:
            return repair(res.x)
    except Exception:
        pass
    return None

def resample(h0, n_new):
    n0 = h0.size
    if n0 == n_new:
        return h0.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    idx = rng.permutation(n)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1
    return b

def structured_binary(n, seed, freq):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    phase = rng.uniform(0, 2 * np.pi)
    v = 0.5 + 0.35 * np.sin(2 * np.pi * freq * t / 2.0 + phase)
    v += 0.05 * rng.normal(0, 1, n)
    return to_binary(v, seed)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    best_x = None
    best_c = 1e9

    def update(x, c):
        nonlocal best_x, best_c
        if c < best_c:
            best_c = c
            best_x = x.copy()

    # Phase 1: Binary local search at n=70 from diverse starts
    if init is not None and init.size >= 10:
        n_base = int(init.size)
        if n_base > 100:
            n_base = 100
    else:
        n_base = 70

    n_list = [n_base]
    for dn in [10, 20, 30, -10, 40]:
        nn = n_base + dn
        if 40 <= nn <= 160:
            n_list.append(nn)

    # Focus on n_base first
    n = n_base
    if init is not None:
        base_b = to_binary(resample(init, n), seed_val)
    else:
        base_b = random_binary(n, seed_val)

    b, c = bin_ls(base_b, seed_val, max_time=8.0)
    update(b, c)

    # Try many random binary starts at n
    n_starts = 25
    for si in range(n_starts):
        if time.time() > deadline - 120:
            break
        if si == 0:
            b0 = base_b
        elif si % 3 == 1:
            b0 = random_binary(n, seed_val + si * 131)
        elif si % 3 == 2:
            b0 = structured_binary(n, seed_val + si * 97, 1 + (si % 4))
        else:
            b0 = to_binary(base_b + 0.05 * np.random.RandomState(seed_val + si).normal(0, 1, n), seed_val + si)
        tl = min(4.0, max(1.0, (deadline - time.time()) * 0.08))
        b, c = bin_ls(b0, seed_val + 5000 + si, max_time=tl)
        update(b, c)

    # Phase 2: SLSQP polish from best binary solution
    if best_x is not None and time.time() < deadline - 60:
        n_pol = best_x.size
        for tau in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline - 30:
                break
            x0 = best_x.astype(float)
            x0 += 0.001 * np.random.RandomState(seed_val + int(tau * 10000)).normal(0, 1, n_pol)
            x0 = repair(x0)
            xr = slsqp_polish(x0, tau, max_iter=300)
            if xr is not None:
                cx = exact_c(xr)
                update(xr, cx)
            # Also try from the binary directly
            xr = slsqp_polish(best_x.astype(float), tau, max_iter=200)
            if xr is not None:
                cx = exact_c(xr)
                update(xr, cx)

    # Phase 3: Try other n values with resampling
    for n2 in n_list:
        if n2 == n_base or time.time() > deadline - 80:
            continue
        if best_x is not None:
            h2 = resample(best_x, n2)
            b2 = to_binary(h2, seed_val + n2)
        else:
            b2 = random_binary(n2, seed_val + n2)
        tl = min(5.0, max(1.0, (deadline - time.time()) * 0.06))
        b, c = bin_ls(b2, seed_val + 9000 + n2, max_time=tl)
        update(b, c)
        # SLSQP on this
        if time.time() < deadline - 40:
            for tau in [0.002, 0.001]:
                if time.time() > deadline - 20:
                    break
                xr = slsqp_polish(b.astype(float), tau, max_iter=200)
                if xr is not None:
                    cx = exact_c(xr)
                    update(xr, cx)

    # Phase 4: Final aggressive polish at best n
    if best_x is not None and time.time() < deadline - 20:
        n_fin = best_x.size
        b_fin = to_binary(best_x, seed_val + 999)
        tl = min(15.0, max(2.0, (deadline - time.time()) * 0.3))
        b2, c2 = bin_ls(b_fin, seed_val + 7777, max_time=tl)
        update(b2, c2)
        if time.time() < deadline - 10:
            for tau in [0.001, 0.0005, 0.0002]:
                if time.time() > deadline - 5:
                    break
                x0 = repair(b2.astype(float) + 0.0005 * np.random.RandomState(seed_val + tau * 1e6).normal(0, 1, n_fin))
                xr = slsqp_polish(x0, tau, max_iter=400)
                if xr is not None:
                    cx = exact_c(xr)
                    update(xr, cx)
                xr = slsqp_polish(best_x.astype(float), tau, max_iter=300)
                if xr is not None:
                    cx = exact_c(xr)
                    update(xr, cx)

    if best_x is None:
        best_x = repair(np.full(70, 0.5))
        best_c = exact_c(best_x)

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
