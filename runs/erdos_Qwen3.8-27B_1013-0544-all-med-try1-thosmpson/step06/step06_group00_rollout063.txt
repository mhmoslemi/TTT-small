def compute_delta(h, j, j2, n):

def slsqp_step(h, n, temp, maxiter=300):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        M = np.max(corr)
        scores = np.exp((corr - M) / temp)
        return (M + temp * (np.log(scores.sum()) - np.log(len(corr)))) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        M = np.max(corr)
        scores = np.exp((corr - M) / temp)
        p = scores / scores.sum()
        grad = np.zeros(n)
        for j in range(n):
            lo1 = max(0, n - 1 - j)
            hi1 = min(2 * n - 2, n - 1 + j)
            k1 = j + np.arange(lo1, hi1 + 1) - n + 1
            grad[j] += np.dot(p[lo1:hi1+1], (1.0 - x[k1]))
            lo2 = max(0, j + 1)
            hi2 = min(2 * n - 2, j + n - 1)
            k2 = j - np.arange(lo2, hi2 + 1) + n - 1
            grad[j] -= np.dot(p[lo2:hi2+1], x[k2])
        return grad * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, target)
