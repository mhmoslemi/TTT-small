def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
