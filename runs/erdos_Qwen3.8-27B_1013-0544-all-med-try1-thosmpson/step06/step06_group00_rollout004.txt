```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            # Need to adjust differently
            if d > 0:
                # Need to increase sum: find largest values < 1
                idx = np.argsort(h)[-1:]
                h[idx] = np.clip(h[idx] + d, 0, 1)
            else:
                idx = np.argsort(h)[:1]
                h[idx] = np.clip(h[idx] + d, 0, 1)
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_topk(h, corr, k_top=5):
    """Compute averaged subgradient of top-k correlations."""
    n = len(h)
    topk_idx = np.argsort(corr)[-k_top:]
    grad = np.zeros(n)
    for k in topk_idx:
        offset = k - (n - 1)
        # term1: (1 - h[j+offset]) where 0<=j+offset<n
        j_start = max(0, -offset)
        j_end = min(n, n - offset)
        if j_start < j_end:
            grad[j_start:j_end] += (1.0 - h[j_start + offset:j_end + offset])
        # term2: -h[j-offset] where 0<=j-offset<n
        j_start2 = max(0, offset)
        j_end2 = min(n, n + offset)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= h[j_start2 - offset:j_end2 - offset]
    return grad / len(topk_idx)

def subgrad_descent(h, n, max_iter=2000, k_top_start=10, k_top_end=3, 
                    step_init=0.3, tol=1e-10):
    """Subgradient descent with adaptive k_top and step size."""
    h = h.copy()
    target_sum = n / 2.0
    dx = 2.0 / n
    best = h.copy()
    best_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    
    step = step_init
    best_step = step
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        
        if c5 < best_val:
            best_val = c5
            best = h.copy()
            best_step = step
        
        # Progressive reduction of k_top
        progress = it / max_iter
        k_top = max(k_top_end, int(k_top_start * (1 - progress) + k_top_end * progress))
        
        grad = compute_grad_topk(h, corr, k_top)
        
        # Try step
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx
        
        if c5_new < c5:
            h = h_new
            step = min(step * 1.1, 2.0)
        else:
            step = step * 0.8
            if step < 0.01:
                # Try smaller step
                step = 0.05
            if step < 0.001:
                break
        
    return best, best_val

def coord_descent_binary(h, target_sum, max_sweeps=10):
    """Coordinate descent that only tries binary values (0 or 1) plus current."""
    n = len(h)
    h = h.copy()
    c5, _ = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            # Try 0 and 1 (and maybe midpoints)
            for v in [0.0, 1.0]:
                if v == orig:
                    continue
                h[j] = v
                h = project_feasible(h, target_sum)
                c, _ = compute_c5(h)
                if c < c5 - 1e-12:
                    c5 = c
                    improved = True
                    break
            else:
                h[j] = orig
                h = project_feasible(h, target_sum)
                # Re-project
                if h.sum() != target_sum:
                    h = project_feasible(h, target_sum)
        if not improved:
            break
    return h, c5

def make_structured_start(n, mode=0, rng=None):
    """Create diverse structured starting points."""
    x = np.linspace(0, 2, n)
    if mode == 0:
        # Bistable: 0.5 + 0.5*cos
        h = 0.5 + 0.4 * np.cos(2 * np.pi * x)
    elif mode == 1:

        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    elif mode == 2:
        # Two bumps
        h = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
    elif mode == 3:
        # Asymmetric
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif mode == 4:
        # Random smooth
        if rng is None:
            rng = np.random
        f = rng.randn(n)
        k = np.exp(-0.5 * (np.arange(n) - n//2)**2 / (n//8)**2)
        h = np.convolve(f, k, mode='same')
        h = h - h.min()
        if h.max() > 0:
            h = h / h.max()
    else:
        h = np.full(n, 0.5)
    h = project_feasible(h, n / 2.0)
    return h

def slsqp_optimize(h, n, temp=50, maxiter=300):
    """SLSQP with smooth max."""
    dx = 2.0 / n
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        s = np.exp(temp * (corr - m))
        return (m + np.log(s.sum()) / temp) * dx
    
    def obj_grad(hv):
        # Numerical gradient (expensive but reliable)
        eps = 1e-7
        g = np.zeros(n)
        f0 = obj(hv)
        for i in range(n):
            hv[i] += eps
            f1 = obj(hv)
            hv[i] -= 2*eps
            f2 = obj(hv)
            hv[i] += eps
            g[i] = (f1 - f2) / (2*eps)
        return g
    
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, ts)
```

```python
    significant = w > 1e-8
    for k in np.where(significant)[0]:
        offset = k - (n - 1)
        j_start = max(0, -offset)
        j_end = min(n, n - offset)
        if j_start < j_end:
            grad[j_start:j_end] += w[k] * (1.0 - h[j_start + offset:j_end + offset])
        j_start2 = max(0, offset)
        j_end2 = min(n, n + offset)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w[k] * h[j_start2 - offset:j_end2 - offset]
    
    obj_val = (m + np.log(exp_vals.sum()) / temp) * dx
    grad *= dx
    return obj_val, grad
```

```python
    corr = np.correlate(h, 1.0 - h, mode="full")
```

```python
def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    js = max(0, -offset)
    je = min(n, n - offset)
    if js < je:
        grad[js:je] += (1.0 - h[js + offset:je + offset])
    js2 = max(0, offset)
    je2 = min(n, n + offset)
    if js2 < je2:
        grad[js2:je2] -= h[js2 - offset:je2 - offset]
    return grad

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(temp * (corr - m))
    w = exp_vals / exp_vals.sum()
    obj_val = (m + np.log(exp_vals.sum()) / temp) * dx
    grad = np.zeros(n)
    thresh = 1e-8
    sig = np.where(w > thresh)[0]
    for k in sig:
        grad += w[k] * grad_corr_k(h, k, n)
    grad *= dx
    return obj_val, grad
```
