from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(a, b, lo=0.0, hi=1.0):
    for _ in range(200):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        max_val = vals.max()
        tol = 1e-13 * max(1.0, abs(max_val))
        active = vals >= (max_val - tol)
        if not active.any():
            idx = np.argmax(vals)
            active = np.zeros(len(a), dtype=bool)
            active[idx] = True
        slopes = b[active]
        if slopes.min() > 1e-14:
            hi = mid
        elif slopes.max() < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def compute_s(h, j, n):
    n_lags = 2 * n - 1
    k = np.arange(-(n - 1), n)
    s = np.ones(n_lags)
    jk = j + k
    m1 = (jk >= 0) & (jk < n)
    if m1.any():
        s[m1] -= h[jk[m1]]
    jmk = j - k
    m2 = (jmk >= 0) & (jmk < n)
    if m2.any():
        s[m2] -= h[jmk[m2]]
    return s

def lagrangian_cd(h, n, lam, n_sweeps, rng):
    h = h.copy()
    c_full = np.correlate(h, 1 - h, mode='full')
    best_local = np.max(c_full)
    for sw in range(n_sweeps):
        order = rng.permutation(n)
        max_change = 0.0
        for j in order:
            s = compute_s(h, j, n)
            d = c_full - s * h[j]
            new_j = solve_1d_minimax(d, s + lam)
            delta = new_j - h[j]
            if abs(delta) > 1e-15:
                h[j] = new_j
                c_full += delta * s
                max_change = max(max_change, abs(delta))
        cmax = np.max(c_full)
        if cmax < best_local:
            best_local = cmax
        if max_change < 1e-10:
            break
    return h, c_full, best_local

def objective_smooth(h_flat, n, alpha):
    c = np.correlate(h_flat, 1 - h_flat, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h, n, budget_remaining):
    if budget_remaining < 5:
        return h, compute_c5(h, n)
    dx = 2.0 / n
    def obj(x):
        return objective_smooth(x, n, 5000)
    def sfun(x):
        return x.sum() - n / 2.0
    def sjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    try:
        maxiter = 5000 if budget_remaining > 60 else 2000
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_try = project_feasible(res.x, n)
        c = compute_c5(h_try, n)
        if c < compute_c5(h, n):
            return h_try, c
    except Exception:
        pass
    return h, compute_c5(h, n)
