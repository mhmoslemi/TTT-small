2. (1-h[i]) appears in the second factor: contributes -δ * h[some_k]

Both are linear in δ. So c[k] is linear in h[i]. ✓

            # (using a linear penalty for simplicity, or quadratic)

            # The max part is piecewise linear. The quadratic part makes it strongly convex.
            # Use a fine grid + local refinement
            grid_coarse = np.linspace(0, 1, 200)
            t_dev = grid_coarse - v

            pen = lam * (sum_curr + t_dev - target) ** 2
            total = max_corr + pen
            coarse_best = grid_coarse[np.argmin(total)]
            
            # Local refinement around coarse_best
            lo = max(0, coarse_best - 1.0/200)
            hi = min(1, coarse_best + 1.0/200)
            grid_fine = np.linspace(lo, hi, 100)
            t_dev_f = grid_fine - v

            pen_f = lam * (sum_curr + t_dev_f - target) ** 2
            total_f = max_corr_f + pen_f
            best_t = grid_fine[np.argmin(total_f)]
            
            if abs(best_t - v) > 1e-10:

                improved = True
        
        h = project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def coord_descent(h, n, max_sweeps=100, num_grid=64, lam=10.0):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    grid = np.linspace(0.0, 1.0, num_grid)
    target = n / 2.0
    for sweep in range(max_sweeps):
        improved = False
        sum_curr = h.sum()
        for i in range(n):
            v = h[i]
            c_curr = np.correlate(h, 1 - h, mode='full')
            h[i] = v + 1.0
            c_plus = np.correlate(h, 1 - h, mode='full')
            h[i] = v
            a = c_plus - c_curr
            t_dev = grid - v
            corr_vals = c_curr[None, :] + t_dev[:, None] * a[None, :]
            max_corr = corr_vals.max(axis=1)
            sum_pen = lam * (sum_curr + t_dev - target) ** 2
            total = max_corr + sum_pen
            best_t = grid[np.argmin(total)]
            if abs(best_t - v) > 1e-9:
                h[i] = best_t
                sum_curr += (best_t - v)
                improved = True
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5

def coord_descent_fine(h, n, max_sweeps=100, lam=50.0):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    target = n / 2.0
    for sweep in range(max_sweeps):
        improved = False
        sum_curr = h.sum()
        for i in range(n):
            v = h[i]
            c_curr = np.correlate(h, 1 - h, mode='full')
            h[i] = v + 1.0
            c_plus = np.correlate(h, 1 - h, mode='full')
            h[i] = v
            a = c_plus - c_curr
            coarse = np.linspace(0.0, 1.0, 200)
            td = coarse - v
            mc = (c_curr[None, :] + td[:, None] * a[None, :]).max(axis=1)
            pen = lam * (sum_curr + td - target) ** 2
            cb = coarse[np.argmin(mc + pen)]
            lo = max(0.0, cb - 0.005)
            hi = min(1.0, cb + 0.005)
            fine = np.linspace(lo, hi, 200)
            td2 = fine - v
            mc2 = (c_curr[None, :] + td2[:, None] * a[None, :]).max(axis=1)
            pen2 = lam * (sum_curr + td2 - target) ** 2
            best_t = fine[np.argmin(mc2 + pen2)]
            if abs(best_t - v) > 1e-10:
                h[i] = best_t
                sum_curr += (best_t - v)
                improved = True
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5

    rng = np.random.RandomState(seed)
    try:

    for alpha in [10, 50, 200, 1000, 5000]:

        try:

            h_try = project_feasible(res.x, n)

            h_try = project_feasible(res.x, n)
