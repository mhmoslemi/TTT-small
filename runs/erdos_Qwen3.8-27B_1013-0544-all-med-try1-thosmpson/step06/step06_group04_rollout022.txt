def compute_subgradient(h):
    """Compute subgradient of C5 at h"""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    k_max = np.argmax(corr)
    shift = k_max - (n - 1)
    
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        js = j - shift
        if 0 <= js < n:
            g += (1.0 - h[js])
        jp = j + shift
        if 0 <= jp < n:
            g -= h[jp]
        grad[j] = g
    return grad * dx

def smooth_c5_and_grad(h, beta=10.0):
    """Compute smooth approximation of C5 and its gradient"""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    # Smooth max
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    smooth_val = (max_corr + np.log(np.sum(weights)) / beta) * dx
    
    # Gradient: weighted sum of individual lag gradients
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        shift = k_idx - (n - 1)
        for j in range(n):
            g = 0.0
            js = j - shift
            if 0 <= js < n:
                g += (1.0 - h[js])
            jp = j + shift
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += weights[k_idx] * g
    return smooth_val, grad * dx
