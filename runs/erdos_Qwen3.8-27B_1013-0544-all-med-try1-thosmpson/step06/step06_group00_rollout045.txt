from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            # Force: distribute to all
            h += d / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, top_k=3, step_size=0.01):
    """One subgradient step targeting the top-k worst lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Get top-k lags
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for m in top_idx:
        s = m - n + 1
        a = abs(s)
        if a == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[:n-a] += 1.0 - h[a:]
            grad[a:] -= h[:n-a]
    
    grad *= step_size * dx
    h_new = h - grad
    h_new = project_feasible(h_new, n / 2.0)
    return h_new, corr[-1] * dx  # return c5 of original

def projected_subgradient(h, max_iters=5000, init_step=0.05, decay=0.999):
    """Adaptive step-size projected subgradient method."""
    n = len(h)
    step = init_step
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    
    for it in range(max_iters):
        h_new, _ = subgradient_step(h, top_k=3, step_size=step)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:

            h = h_new

            h = h_new

            patience += 1

def smooth_max_obj(h, temp=50.0):
    """Smooth approximation of max using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    s = np.sum(np.exp(temp * (corr - m)))
    return (m + np.log(s) / temp) * dx

def slsqp_refine(h, temp=20.0, maxiter=300):
    """SLSQP with smooth max objective."""
    n = len(h)
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_max_obj(x, temp)
    
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    h_out = project_feasible(res.x, ts)
    return h_out, compute_c5(h_out)

        x_old = np.arange(len(best_h), dtype=float)
        x_new = np.arange(n_try, dtype=float)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try / 2.0)
        for temp in [40.0, 15.0]:
            if remaining() < 50:
                break
            h_new = slsqp_minimize(h_new, temp, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new
            if remaining() > 40:
                h_cd, c2 = coord_descent(best_h, n_levels=15, max_sweeps=8)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cd
    
    # Phase 7: Final polish
    if remaining() > 30:
        h_cd, c = coord_descent(best_h, n_levels=31, max_sweeps=12)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd
    
    if remaining() > 20:
        h_ref = active_lag_slsqp(best_h, n_active=6, maxiter=200)
        c = compute_c5(h_ref)
        if c < best_c5:
            best_c5 = c
            best_h = h_ref
    
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
