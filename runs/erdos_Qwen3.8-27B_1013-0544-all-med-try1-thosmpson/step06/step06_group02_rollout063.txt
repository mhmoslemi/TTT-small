def grad_lag(h, s):
    """Gradient of the correlation at shift s (positive or negative) w.r.t. h."""
    N = len(h)
    g = np.zeros(N)
    if s >= 0:
        # corr = sum_{i=0}^{N-1-s} h[i]*(1-h[i+s])
        # d/dh[i]: (1-h[i+s]) for i < N-s (as first factor)
        #          -h[i-s] for i >= s (as second factor in 1-h[i])
        ns = N - s

        g[s:] -= h[:ns]  # h[i-s] for i in [s, N)

        ns = N - ss
        g[:ns] -= h[ss:ss+ns]  # h[i+ss] for i in [0, ns)  
        g[ss:] += (1.0 - h[:ns])  # 1-h[i-ss] for i in [ss, N)
    return g

    v = beta2 * v + (1 - beta2) * g**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps), m, v
