import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def compute_c5(h):
    return np.max(compute_correlations(h))

def softmax_weighted_corr(h, temp):
    """Compute softmax-weighted maximum of correlations."""
    corr = compute_correlations(h)
    max_c = np.max(corr)
    # Softmax approximation of max
    w = np.exp((corr - max_c) / temp)
    w /= w.sum()
    return max_c + temp * np.log(w.sum())  # This is the log-sum-exp

def gradient_of_corr_k(h, k_offset):
    """Gradient of the k-th correlation element w.r.t. h."""
    n = len(h)
    k = k_offset - (n - 1)  # actual lag
    grad = np.zeros(n)
    for j in range(n):
        # h[j] appears in the sum as h[i+k] when i = j-k
        if 0 <= j - k <= n - 1:
            grad[j] += (1.0 - h[j - k])
        # h[j] appears as (1-h[i]) when i = j
        if 0 <= j + k <= n - 1:
            grad[j] += (-h[j + k])
    return grad

def subgradient_descent(h, n_iter, lr_schedule, top_m=10, temp=0.01):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(n_iter):
        corr = compute_correlations(h)
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Get top-m lags
        top_idx = np.argsort(corr)[-top_m:]
        # Weight by softmax
        top_corr = corr[top_idx]
        w = np.exp((top_corr - np.max(top_corr)) / temp)
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for m, k_idx in enumerate(top_idx):
            if w[m] > 1e-8:
                grad += w[m] * gradient_of_corr_k(h, k_idx)
        
        # Adaptive step size
        lr = lr_schedule(t)
        
        # Gradient step
        h_new = h - lr * grad
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        diff = target_sum - h_new.sum()
        mask = (h_new > 0) & (h_new < 1)
        if mask.sum() > 0:
            h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    return h, best_c5

def subgradient_step(h, top_m=20, temp=0.05, lr=0.01):
    """One subgradient descent step."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(corr)
    
    # Get top-m lags
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
    else:
        top_idx = np.arange(len(corr))
    
    # Softmax weights
    top_val = corr[top_idx]
    w = np.exp((top_val - np.max(top_val)) / temp)
    w /= w.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    n_half = n - 1  # center of full correlation
    for m, k_idx in enumerate(top_idx):
        if w[m] < 1e-10:
            continue
        k = k_idx - n_half  # actual lag
        # grad[j] = dx * [(1-h[j+k]) * 1_{0<=j+k<n} - h[j-k] * 1_{0<=j-k<n}]
        # Vectorized:
        # Term 1: (1-h[j+k]) for j where 0<=j+k<n
        lo1 = max(0, -k)
        hi1 = min(n, n - k)  # j ranges from lo1 to hi1-1
        if lo1 < hi1:
            grad[lo1:hi1] += w[m] * dx * (1.0 - h[lo1+k:hi1+k])
        # Term 2: -h[j-k] for j where 0<=j-k<n
        lo2 = max(0, k)
        hi2 = min(n, n + k)  # j ranges from lo2 to hi2-1
        if lo2 < hi2:
            grad[lo2:hi2] += w[m] * dx * (-h[lo2-k:hi2-k])
    
    return h - lr * grad, c5

def compute_weighted_gradient(h, weights_by_lag):
    """
    Compute weighted sum of gradients of correlations.
    weights_by_lag: dict mapping lag k to weight w_k
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k, w in weights_by_lag.items():
        if abs(w) < 1e-12:
            continue
        if k >= 0:
            # First factor: j from k to n-1
            if k < n:
                grad[k:n] += w * dx * (1.0 - h[:n-k])
            # Second factor: j from 0 to n-k-1
            if k < n:
                grad[:n-k] += w * dx * (-h[k:])
        else:
            kp = -k
            # First factor: j from 0 to n-kp-1
            if kp < n:
                grad[:n-kp] += w * dx * (1.0 - h[kp:])
            # Second factor: j from kp to n-1
            if kp < n:
                grad[kp:] += w * dx * (-h[:n-kp])
    return grad

        temp = max(0.01, 0.1 * cur_temp)
        h_opt, c5 = subgrad_optimize(h_pert, target0, 300, top_m, temp, 0.03, 0.003)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        else:

            delta = c5 - best_c5
            if np.random.rand() < np.exp(-delta / max(cur_temp, 1e-6)):
                best_h = h_opt.copy()
        
        # Decay temperature
        cur_temp *= 0.995
    
    # Phase 3: Try different resolutions
    for n_try in [256, 320, 448, 512, 640, 768]:
        if remaining() < 120:
            break
        h_new = interpolate_h(best_h, n_try)
        h_new = project_feasible(h_new, n_try / 2.0)

            (2, 0.01, 0.003, 0.0003, 1000),
        ]:
            if remaining() < 60:
                break
            n_iter_adj = min(n_iter, max(100, int(remaining() * 2)))
            h_new, c5 = subgrad_optimize(h_new, n_try / 2.0, n_iter_adj, top_m, temp, lr0, lr1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    if remaining() > 30:
        n_fin = len(best_h)
        h_cd, c5 = coord_descent_fast(best_h, n_fin / 2.0, n_levels=7, max_sweeps=5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd
    
    # Final check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
