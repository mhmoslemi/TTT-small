    corr = np.correlate(h, one_minus_h, mode='full')

    return corr.max() * dx

def to_binary(h, N, rng):
    """Convert to binary maintaining sum constraint."""
    target = N // 2
    # Round and fix
    h_bin = (h >= 0.5).astype(float)
    cur_sum = h_bin.sum()
    if cur_sum > target:
        # Need to flip some 1s to 0s
        ones = np.where(h_bin == 1.0)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[ones])
        for idx in order[:int(cur_sum - target)]:
            h_bin[ones[idx]] = 0.0
    elif cur_sum < target:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:int(target - cur_sum)]:
            h_bin[zeros[idx]] = 1.0
    return h_bin

            val = val_new
            corr = corr_new
            if val < best_val:
                best_val = val
                best_h = h_bin.copy()

            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):

        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid

def grad_lag(h, k):
    N = len(h)

        g[k:] -= h[:N-k]

        g[kk:] += (1.0 - h[:N-kk])
    return g
