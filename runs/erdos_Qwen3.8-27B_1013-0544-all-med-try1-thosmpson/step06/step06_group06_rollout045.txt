def get_subgradient_direction(h, n):
    """Get a direction that reduces the max overlap."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    # Use soft weights
    s = c - m
    w = np.exp(s - s.max())
    w /= w.sum()
    
    # For each lag k, the overlap is sum_j h[j]*(1-h[j+shift_k])
    # The "pressure" on h[i] from lag k is:
    #   from h[i] as first factor: (1 - h[i+shift_k]) if valid
    #   from h[i-shift_k] as second factor: -h[i-shift_k] if valid
    # Combined: for shift >= 0:
    #   pressure[i] += w[k] * [(1-h[i+shift]) for i < n-shift]  (as first factor)
    #   pressure[i] += w[k] * [-h[i-shift] + (1-h[i+shift... no)]
    
    # Actually let me think more carefully.
    # c[k] = dx * sum_j h[j] * (1-h[j+shift]) where shift depends on k
    # dc[k]/dh[i] = dx * [(1-h[i+shift]) if i+shift < n, from the h[i] term]
    #             + dx * [-h[i-shift]] if i-shift >= 0, from the (1-h[i]) term in some j
    
    # This is getting complex. Let me just use a simpler approach:
    # The subgradient of max is just the gradient of the active constraint.
    # For the max, take the k with the highest value and compute its gradient.
    
    k_max = np.argmax(c)
    shift = k_max - (n - 1)
    
    grad = np.zeros(n)
    if shift >= 0:
        # c[k_max] = dx * sum_{j=0}^{n-shift-1} h[j] * (1-h[j+shift])
        for j in range(n - shift):
            grad[j] += dx * (1 - h[j + shift])
            grad[j + shift] += dx * (-h[j])
    else:
        s2 = -shift
        # c[k_max] = dx * sum_{j=0}^{n-1-s2} h[j+s2] * (1-h[j])
        for j in range(n - s2):
            grad[j + s2] += dx * (1 - h[j])
            grad[j] += dx * (-h[j + s2])
    
    # Normalize
    norm = np.linalg.norm(grad)
    if norm > 1e-15:
        grad /= norm
    return -grad  # Negative gradient = descent direction

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):

        if abs(diff) < 1e-15:
