import numpy as np
from scipy.optimize import minimize
import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def _solve_1d_max_line(a, b, lo=0.0, hi=1.0):
    """Find t in [lo,hi] minimizing max_m (a[m]*t + b[m])."""
    n = len(a)
    # Candidate points: lo, hi, and all intersection points in (lo, hi)
    candidates = [lo, hi]
    for i in range(n):
        for j in range(i+1, n):
            da = a[i] - a[j]
            if abs(da) > 1e-12:
                t = (b[j] - b[i]) / da
                if lo < t < hi:
                    candidates.append(t)
    # Evaluate at all candidates
    best_t = lo
    best_val = np.inf
    for t in candidates:
        vals = a * t + b
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t, best_val

def _coord_sweep(h, n, max_sweeps=50, tol=1e-10):
    """Coordinate descent with exact 1D optimization."""
    dx = 2.0 / n
    best = _c5_max(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        # Precompute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for i in range(n):
            # Compute a_m = d(corr[m])/d(h[i])
            n_c = 2 * n - 1
            a = np.zeros(n_c)
            for m in range(n_c):
                p = i + m - (n - 1)
                q = i - m + (n - 1)
                coeff = 0.0
                if 0 <= p < n:
                    coeff += 1.0 - h[p]
                if 0 <= q < n:
                    coeff += -h[q]
                a[m] = coeff * dx
            b = corr - a * h[i]
            t_star, _ = _solve_1d_max_line(a, b)
            if abs(t_star - h[i]) > 1e-12:
                h[i] = t_star
                improved = True
        # Recompute
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(new_corr * dx)
        if new_c5 < best - 1e-12:
            best = new_c5
        if not improved:
            break
    return h, best

import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _fix_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h += d / n
        h = np.clip(h, 0.0, 1.0)
        # Fix residual
        for _ in range(10):
            d2 = target - np.sum(h)
            if abs(d2) < 1e-12:
                break
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(h[idx] + d2, 1.0)
            else:
                idx = np.argmax(h)
                h[idx] = max(h[idx] + d2, 0.0)
    return h

def _solve_1d(a, b):
    """Find t in [0,1] minimizing max_m (a[m]*t + b[m])."""
    n = len(a)
    # Sort by slope for potential optimization, but brute force is fine for n~200
    best_t = 0.0
    best_val = np.max(b)  # at t=0
    val_at_1 = np.max(a + b)
    if val_at_1 < best_val:
        best_val = val_at_1
        best_t = 1.0
    # Check all pairwise intersections
    for i in range(n):
        ai, bi = a[i], b[i]
        for j in range(i + 1, n):
            aj, bj = a[j], b[j]
            da = ai - aj
            if abs(da) < 1e-14:
                continue
            t = (bj - bi) / da
            if t > 0 and t < 1:
                v = np.max(a * t + b)
                if v < best_val - 1e-14:
                    best_val = v
                    best_t = t
    return best_t

def _coord_sweep(h, n, max_sweeps=100):
    """Coordinate descent with exact 1D subproblem."""
    dx = 2.0 / n
    n_c = 2 * n - 1
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best = np.max(corr * dx)
    
    for sweep in range(max_sweeps):
        any_changed = False
        for i in range(n):
            # Compute derivative a[m] = d(corr[m])/d(h[i])
            p = np.arange(i - (n - 1), i + n)
            q = 2 * i - p
            valid_p = (p >= 0) & (p < n)
            valid_q = (q >= 0) & (q < n)
            a = np.zeros(n_c)
            a[valid_p] += (1.0 - h[p[valid_p]])
            a[valid_q] += -h[q[valid_q]]
            a *= dx
            
            # b[m] = corr[m] - a[m] * h[i]
            b = corr - a * h[i]
            
            # Solve 1D problem
            t_star = _solve_1d(a, b)
            
            if abs(t_star - h[i]) > 1e-10:
                delta = t_star - h[i]
                h[i] = t_star
                corr += delta * a
                any_changed = True
        
        # Enforce constraint
        h = _fix_constraint(h, n)
        # Recompute corr after constraint fix
        corr = np.correlate(h, 1.0 - h, mode='full')
        current = np.max(corr * dx)
        
        if current < best:
            best = current
        if not any_changed:
            break
    return h, best

    # da[i,j] = a[i] - a[j] for i < j
    # t[i,j] = (b[j] - b[i]) / da[i,j]

    i_idx, j_idx = np.triu_indices(n, k=1)

        v1 = np.max(a + b)

    vals = a[None, :] * candidates[:, None] + b[None, :]

    best_idx = np.argmin(max_vals)

from scipy.optimize import minimize
import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _fix_constraint(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(20):
            d2 = target - np.sum(h)
            if abs(d2) < 1e-12:
                break
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                add = min(d2, 1.0 - h[idx])
                h[idx] += add
            else:
                idx = np.argmax(h)
                sub = min(-d2, h[idx])
                h[idx] -= sub
    return h

def _solve_1d(a, b):
    n = len(a)
    i_idx, j_idx = np.triu_indices(n, k=1)
    da = a[i_idx] - a[j_idx]
    db = b[j_idx] - b[i_idx]
    valid = np.abs(da) > 1e-14
    t = np.empty(len(da))
    t[~valid] = -1.0
    t[valid] = db[valid] / da[valid]
    t_valid = t[(t > 0) & (t < 1) & valid]
    
    if len(t_valid) == 0:
        v0 = np.max(b)
        v1 = np.max(a + b)
        return 0.0 if v0 <= v1 else 1.0
    
    candidates = np.concatenate([[0.0, 1.0], t_valid])
    vals = a[None, :] * candidates[:, None] + b[None, :]
    max_vals = np.max(vals, axis=1)
    best_idx = np.argmin(max_vals)
    return float(candidates[best_idx])

    n_c = 2 * n - 1

    corr = np.correlate(h, 1.0 - h, mode='full').copy()

            q = 2 * i - p
            vp = (p >= 0) & (p < n)
            vq = (q >= 0) & (q < n)
            a = np.zeros(n_c)
            a[vp] += (1.0 - h[p[vp]])
            a[vq] += -h[q[vq]]
            a *= dx
            b = corr - a * h[i]
            t_star = _solve_1d(a, b)
            if abs(t_star - h[i]) > 1e-10:
                delta = t_star - h[i]
                h[i] = t_star
                corr += delta * a
                any_changed = True
        
        h = _fix_constraint(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        current = np.max(corr) * dx
        if current < best:
            best = current
        if not any_changed:
            break
    return h, best

def _solve_1d_full(a, b, n):
    n_c = 2 * n - 1
    cqr = n - 1  # quadratic index
    cands = {0.0, 1.0, 0.5}
    
    # Linear-linear intersections (excluding quadratic)
    lin_idx = [i for i in range(n_c) if i != cqr]
    for i in range(len(lin_idx)):
        for j in range(i+1, len(lin_idx)):
            mi, mj = lin_idx[i], lin_idx[j]
            da = a[mi] - a[mj]
            if abs(da) > 1e-14:
                t = (b[mj] - b[mi]) / da
                if 0 < t < 1:
                    cands.add(t)
    
    # Linear-quadratic intersections
    # Quadratic: t - t² + b[cqr]
    # Line: a[mi]*t + b[mi]
    # t - t² + b[cqr] = a[mi]*t + b[mi]
    # t² + (a[mi]-1)*t + (b[mi]-b[cqr]) = 0
    for mi in lin_idx:
        A = 1.0
        B = a[mi] - 1.0
        C = b[mi] - b[cqr]
        disc = B*B - 4*A*C
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            t1 = (-B + sqrt_disc) / (2*A)
            t2 = (-B - sqrt_disc) / (2*A)
            if 0 < t1 < 1:
                cands.add(t1)
            if 0 < t2 < 1:
                cands.add(t2)
    
    # Evaluate objective at all candidates
    best_t = 0.0
    best_val = np.inf
    for t in cands:
        vals = a * t + b
        vals[cqr] = t - t*t + b[cqr]  # override with quadratic
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t
