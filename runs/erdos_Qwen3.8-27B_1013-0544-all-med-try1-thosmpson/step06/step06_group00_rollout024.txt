import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            # Need to add/remove from boundary values
            if d > 0:
                # Need to increase sum: set some 0s toward 1
                idx = np.where(h < 0.5)[0]
                if len(idx) == 0:
                    break
                for i in idx:
                    need = min(d, 1.0 - h[i])
                    h[i] += need
                    d -= need
                    if abs(d) < 1e-12:
                        break
            else:
                idx = np.where(h > 0.5)[0]
                if len(idx) == 0:
                    break
                for i in idx:
                    need = min(-d, h[i])
                    h[i] -= need
                    d += need
                    if abs(d) < 1e-12:
                        break
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = temp * (corr - m)
    # clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    s = np.exp(shifted)
    val = m + np.log(s.sum()) / temp
    return val * dx

def run_slsqp(h_start, n, temp, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    def obj(x):
        return smooth_obj(x, temp)
    def obj_grad(x):
        # Use numerical gradient for reliability
        return obj(x)  # Let SLSQP compute gradient
    res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                                'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, ts)

        b[i_zero] = 1

        zeros_set.discard(i_zero)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        adj = d / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0, 1)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(temp * (corr - m), -400, 400)
    s = np.exp(shifted)
    return (m + np.log(s.sum()) / temp) * dx

def slsqp_optimize(h, n, temp, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    res = minimize(lambda x: smooth_obj(x, temp), h, method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                                'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, ts)

def coord_descent(h, n_levels=20, max_sweeps=15):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_v = orig
            for v in np.linspace(0, 1, n_levels):
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_c < c5 - 1e-14:
                h[j] = best_v
                h = project_feasible(h, ts)
                c5 = best_c
                improved = True
            else:
                h[j] = orig
        if not improved:
            break
    return h, c5

def binary_search_1d(h, j, lo, hi, ts, n, n_evals=12):
    """Find best value for h[j] in [lo, hi] using golden section or bisection."""
    best_v = None
    best_c = float('inf')
    for v in np.linspace(lo, hi, n_evals + 1):
        h[j] = v
        h = project_feasible(h, ts)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_v = v
    h[j] = best_v
    h = project_feasible(h, ts)
    return best_c

def binary_sa(h_start, n, max_iter=20000, T_start=0.5, T_end=0.01):
    """SA on binary vectors with exactly n//2 ones."""
    b = (h_start > 0.5).astype(int).copy()
    ones_idx = np.where(b == 1)[0]
    if len(ones_idx) > n // 2:
        remove = np.random.choice(ones_idx, len(ones_idx) - n//2, replace=False)
        b[remove] = 0
    elif len(ones_idx) < n // 2:
        zeros_idx = np.where(b == 0)[0]
        add = np.random.choice(zeros_idx, n//2 - len(ones_idx), replace=False)
        b[add] = 1
    
    current_e = compute_c5(b.astype(float))
    best_b = b.copy()
    best_e = current_e
    
    for it in range(max_iter):
        T = T_start * (T_end / T_start) ** (it / max_iter)
        ones_idx = np.where(b == 1)[0]
        zeros_idx = np.where(b == 0)[0]
        i1 = np.random.choice(ones_idx)
        i0 = np.random.choice(zeros_idx)
        b[i1] = 0
        b[i0] = 1
        new_e = compute_c5(b.astype(float))
        d = new_e - current_e
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            current_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_b = b.copy()
        else:
            b[i1] = 1
            b[i0] = 0
    
    return best_b.astype(float), best_e

def interpolate(h_from, n_to):
    n_from = len(h_from)
    x_old = np.arange(n_from) / n_from
    x_new = np.arange(n_to) / n_to
    h_new = np.interp(x_new, x_old, h_from)
    return project_feasible(h_new, n_to / 2.0)

    t0 = time.time()

        return budget_s - (time.time() - t0)
    
    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)

    best_c5 = compute_c5(h0)

    for temp in [2000, 500, 200, 100, 50, 20, 10, 5]:

            best_c5 = c
            best_h = h_new.copy()

        if c < best_c5:

        c = compute_c5(h_pol)
        if c < best_c5:
            best_c5 = c
            best_h = h_pol.copy()
            if time_left() > 60:
                h_cd, c2 = coord_descent(best_h, n_levels=15, max_sweeps=10)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(temp * (corr - m), -400, 400)
    s = np.exp(shifted)
    return (m + np.log(s.sum()) / temp) * dx

                   constraints=[{'type': 'eq', 'fun': constr, 'jac': constr_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, ts)
