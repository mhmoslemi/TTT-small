_g_alpha = 500.0

def _slsqp_obj(x):
    n = _g_n
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = g.max()
    shifted = _g_alpha * (g - m)
    exp_s = np.exp(shifted - shifted.max())
    return m + np.log(exp_s.sum()) / _g_alpha

def _slsqp_grad(x):
    n = _g_n
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    shifted = _g_alpha * (g - m)
    exp_s = np.exp(shifted - shifted.max())
    w = exp_s / exp_s.sum()
    F = _compute_F(x, n)
    return (F.T @ w) * dx

def _cons_fun(x):
    return x.sum() - _g_n / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def compute_c5(h, n):

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        if i_lo < i_hi:

        j_lo = max(0, d)
        j_hi = min(n, n + d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo - d:j_hi - d]
    return F

def binary_local_search(h_init, n, deadline, max_passes=500):

    idx = np.argsort(h_init)[::-1]
