            best_h = h.copy()

            continue
        h_bin[z] = 1.0
        h_bin[o] = 0.0
        c_new = compute_c5(h_bin, n)
        delta = c_new - cur_c
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
        else:
            h_bin[z] = 0.0
            h_bin[o] = 1.0
        T = max(T * cooling, T_min)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

        if c_small < best_c * (n / n_small) * (n_small / n):  # just compare directly

            h_up = upsample(h_small, n_small, n)
            c_up = compute_c5(h_up, n)
            if c_up < best_c:

                best_h = h_up

    p4 = t0 + 0.80 * (budget_s - 5)

        # Hill climb after SA
        hc_dl = min(p5 + 30, dl - 20)
        if time.time() < hc_dl:

            if c_hc2 < best_c:

    best_c = compute_c5(best_h, n)
