    """Smoothed max optimization with projected gradient + Nesterov momentum"""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    m = h.copy()  # momentum position
    v = np.zeros(n)  # velocity
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        # Compute g = correlation * dx
        g = np.correlate(h, 1 - h, mode='full') * dx
        # Softmax weights
        g_shifted = alpha * (g - g.max())
        w = np.exp(g_shifted)
        w /= w.sum()
        # Gradient: dx * F^T @ w
        # F[i, k] = effect of h[i] on g[k] (before dx)
        # So ∂(g[k])/∂h[i] = F[i,k] * dx
        # ∇_i obj = sum_k w[k] * F[i,k] * dx
        F = compute_F(h, n)  # shape (n, 2n-1)
        grad = (F * w[None, :]).sum(axis=1) * dx
        
        # Nesterov step
        beta = 0.9  # momentum
        v = beta * v - grad  # negative gradient direction with momentum
        m = m + v
        
        # Project to feasible set
        h_new = project_feasible(m, n)
        
        # Backtracking line search
        f_new = compute_c5(h_new, n)
        if f_new < best_c:
            best_c = f_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    i_idx = np.arange(n)[:, None]
    ip = i_idx + dm[None, :]
    im = i_idx - dm[None, :]
    vp = (ip >= 0) & (ip <= n - 1)
    vm = (im >= 0) & (im <= n - 1)
    ipc = np.clip(ip, 0, n - 1)
    imc = np.clip(im, 0, n - 1)
    return (1.0 - h[ipc]) * vp - h[imc] * vm

def smooth_grad(h, n, deadline, lr=0.004, maxiter=20000, alpha0=500.0, alpha1=5000.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    v = np.zeros(n)
    beta = 0.88
    for it in range(maxiter):
        if time.time() > deadline:
            break
        alpha = alpha0 + (alpha1 - alpha0) * (it / maxiter)
        g = compute_g(h_cur, n)
        gs = alpha * (g - g.max())
        w = np.exp(gs)
        w /= w.sum()
        F = compute_F(h_cur, n)
        grad = (F * w[None, :]).sum(axis=1) * dx
        v = beta * v - lr * grad
        h_new = project_feasible(h_cur + v, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c

def binary_sa(h, n, deadline, rng, directed=True):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    g = compute_g(h_bin, n)
    best_c = float(g.max())
    cur_c = best_c
    best_h = h_bin.copy()
    dm = np.arange(2 * n - 1) - (n - 1)
    pos = np.arange(n)
    T = 0.005
    no_improve = 0
    step_count = 0
    recompute_every = 10000
    while time.time() < deadline:
        T = max(T * 0.99996, 0.00002)
        if directed:
            m_star = int(np.argmax(g))
            m_val = dm[m_star]
            nm = pos + m_val
            valid = (nm >= 0) & (nm < n)
            contrib = valid & (h_bin == 1) & (h_bin[np.clip(nm, 0, n - 1)] == 0)
            ci = np.where(contrib)[0]
            if len(ci) == 0:
                ci = np.where(h_bin == 1)[0]
            i = ci[rng.randint(len(ci))]
        else:
            ones_idx = np.where(h_bin == 1)[0]
            i = ones_idx[rng.randint(len(ones_idx))]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(zeros_idx) == 0:
            break
        j = zeros_idx[rng.randint(len(zeros_idx))]
        ip = i + dm
        im = i - dm
        dp = np.zeros(2 * n - 1)
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        dp[vp] = -(1.0 - h_bin[ip[vp]])
        dp[vm] += h_bin[im[vm]]
        jp = j + dm
        jm = j - dm
        dj = np.zeros(2 * n - 1)
        vp2 = (jp >= 0) & (jp < n)
        vm2 = (jm >= 0) & (jm < n)
        dj[vp2] = 1.0 - h_bin[jp[vp2]]
        dj[vm2] -= h_bin[jm[vm2]]
        new_g = g + dp + dj
        new_c = float(new_g.max())
        if new_c <= cur_c + 1e-14:
            accept = True
        else:
            delta = new_c - cur_c
            if delta > 0.05:
                accept = False
            else:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            g = new_g
            cur_c = new_c
            step_count += 1
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
            if step_count % recompute_every == 0:
                g = compute_g(h_bin, n)
                cur_c = float(g.max())
        else:
            no_improve += 1
        if no_improve > 80000:
            no_improve = 0
            T = min(T * 4.0, 0.005)
    return best_h, best_c

def continuous_polish(h, n, deadline, max_rounds=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        target = n // 2
        idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
        blend = 0.3 + 0.1 * r
        h_mixed = blend * h_bin + (1 - blend) * h
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (max_rounds - r))
        h2, c2 = smooth_grad(h_mixed, n, sub_dl, lr=0.002, maxiter=5000, alpha0=1000.0, alpha1=10000.0)
        if c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: smooth gradient (12% budget)
    p1 = t0 + 0.12 * (budget_s - 5)
    h, c = smooth_grad(best_h.copy(), n, p1, lr=0.004, maxiter=8000, alpha0=500.0, alpha1=3000.0)
    if c < best_c:
        best_c, best_h = c, h.copy()
    # Phase 2: directed binary SA (40% budget)
    p2 = t0 + 0.52 * (budget_s - 5)
    h, c = binary_sa(best_h.copy(), n, p2, rng, directed=True)
    if c < best_c:
        best_c, best_h = c, h.copy()
    # Phase 3: smooth polish from binary (13% budget)
    p3 = t0 + 0.65 * (budget_s - 5)
    h, c = continuous_polish(best_h.copy(), n, p3, max_rounds=3)
    if c < best_c:
        best_c, best_h = c, h.copy()
    # Phase 4: undirected binary SA (20% budget)
    p4 = t0 + 0.85 * (budget_s - 5)
    rng2 = np.random.RandomState(seed + 123)
    h, c = binary_sa(best_h.copy(), n, p4, rng2, directed=False)
    if c < best_c:
        best_c, best_h = c, h.copy()
    # Phase 5: final smooth polish (remaining)
    if time.time() < dl - 10:
        h, c = smooth_grad(best_h.copy(), n, dl - 3, lr=0.001, maxiter=10000, alpha0=2000.0, alpha1=20000.0)
        if c < best_c:
            best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
