def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):

        if abs(diff) < 1e-12:

            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
