import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += d / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_gradient(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = g - m
    lse = sh.max() + np.log(np.exp(sh - sh.max()).sum())
    val = m + lse / alpha
    w = np.exp(sh - sh.max())
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wt = w[k] * dx
        s = k - (n - 1)
        if s >= 0:
            v = n - s
            if v > 0:
                grad[:v] += wt * (1.0 - h[s:])
                grad[s:] -= wt * h[:v]
        else:
            s2 = -s
            v = n - s2
            if v > 0:
                grad[s2:] += wt * (1.0 - h[:v])
                grad[:v] -= wt * h[s2:]
    return grad, val

def _slsqp_obj(x, n, alpha):
    return smooth_gradient(x, n, alpha)[1]

def _slsqp_jac(x, n, alpha):
    return smooth_gradient(x, n, alpha)[0]

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h0, args=(n, alpha), jac=_slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() < deadline:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def nesterov_opt(h0, n, deadline, max_iter=200000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    x = h0.copy()
    x_prev = x.copy()
    beta = 0.9
    for it in range(max_iter):
        if time.time() > deadline:
            break
        alpha = min(1000.0 * (it + 10) ** 1.4, 1e8)
        if it == 0:
            y = x
        else:
            y = x + beta * (x - x_prev)
            y = np.clip(y, 0.0, 1.0)
        grad_y, _ = smooth_gradient(y, n, alpha)
        lr = 0.15 / (1.0 + it * 0.0003)
        x_new = project_feasible(y - lr * grad_y, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = x_new.copy()
        x_prev = x
        x = x_new
        if it % 50000 == 0 and it > 0:
            beta = max(0.5, beta * 0.95)
    return best_h, best_c

def coord_descent(h0, n, deadline, max_sweeps=500):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    h = h0.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            hi = h[i]
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            h[i] = c
            fc = compute_c5(h, n)
            h[i] = d
            fd = compute_c5(h, n)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    h[i] = c
                    fc = compute_c5(h, n)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    h[i] = d
                    fd = compute_c5(h, n)
            best_i = (a + b) / 2.0
            h[i] = best_i
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                improved = True
            else:
                h[i] = hi
        if not improved:
            break
        h = project_feasible(h, n)
    return best_h, best_c

def binary_local(h0, n, deadline, max_sweeps=30, seed=123):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h, best_c

    p6 = t0 + 0.97 * (budget_s - 3)
    h, c = coord_descent(best_h.copy(), n, p6, max_sweeps=200)
    if c < best_c - 1e-14:
        best_c = c
        best_h = h.copy()

    # Final: quick polish
    if time.time() < dl - 1:
        h = slsqp_opt(best_h.copy(), n, 1e8, 500, dl - 1)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
