The objective is max_k ∑_i h[i](1-h[i+k]) * dx.

For the best solutions, we expect the maximum to be achieved at several different k values (it's a max, so the tightest constraints are the binding ones).

    """L-BFGS-B with softmax approximation of max."""
    dx = 2.0 / n
    
    def objective(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        logsum = sm + np.log(np.exp(sh - sm).sum())
        val = m + logsum / alpha
        val += 1e6 * (x.sum() - n / 2.0) ** 2
        return val
    
    def gradient(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = int(np.argmax(c))
        k = m - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e6 * (x.sum() - n / 2.0)
        return g
    
    try:
