def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.max() - target, h.min() + (n - target)  # rough bounds
    # Actually, let's use tighter bounds
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix any remaining numerical issues
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h_proj
            s = room.sum()
            if s > 1e-14:
                h_proj += diff / s * room
        else:
            s = h_proj.sum()
            if s > 1e-14:
                h_proj += diff / s * h_proj
    return np.clip(h_proj, 0.0, 1.0)

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    # Use bisection on lambda for exact projection
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h_p
            s = room.sum()
            if s > 1e-14:
                h_p += diff / s * room
        else:
            s = h_p.sum()
            if s > 1e-14:
                h_p += diff / s * h_p
    return np.clip(h_p, 0.0, 1.0)

def grad_smooth(h, alpha, n):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    # Compute gradient
    c1 = np.correlate(p, h, mode='full')[0:n]
    c2 = np.convolve(p, h, mode='full')[n-1:2*n-1]
    grad = dx * (1.0 - c1 - c2)
    return grad, g, p

    best_c = compute_c5(h, n)
    cur = h.copy()
    vel = np.zeros(n)

    alpha_start = 50.0
    alpha_end = 500000.0

        cur = project_feasible(cur, n)
        
        c = compute_c5(cur, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = cur.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h_p
            s = room.sum()
            if s > 1e-14:
                h_p += diff / s * room
        else:
            s = h_p.sum()
            if s > 1e-14:
                h_p += diff / s * h_p
    return np.clip(h_p, 0.0, 1.0)

def grad_smooth(h, alpha, n):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    shifted = alpha * (g - g_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    c1 = np.correlate(p, h, mode='full')[0:n]
    c2 = np.convolve(p, h, mode='full')[n - 1:2 * n - 1]
    grad = dx * (1.0 - c1 - c2)
    return grad, g

def smooth_gd(h, n, deadline, max_iters=100000, alpha_start=30.0, alpha_end=2000000.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    vel = np.zeros(n)
    momentum = 0.85
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        step = min(0.08, 3.0 / alpha)
        grad, g = grad_smooth(cur, alpha, n)
        vel = momentum * vel - step * grad
        cur = cur + vel
        cur = project_feasible(cur, n)
        if it % 50 == 0:
            c = compute_c5(cur, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = cur.copy()
    c = compute_c5(cur, n)
    if c < best_c:
        best_c = c
        best_h = cur.copy()
    return best_h, best_c

def pair_descent_fast(h, n, deadline, max_sweeps=100):
    best_h = h.copy()

    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        improved = 0
        n_changed = 0
        for i in range(0, n, 2):
            if time.time() > deadline:
                break
            j = i + 1
            if j >= n:
                j = i
                if i == n - 1:
                    break
            jp = i + dm
            jm = i - dm
            j2p = j + dm
            j2m = j - dm
            s = np.zeros(2 * n - 1)
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            s[vp] = dx * (1.0 - h[jp[vp]])
            s[vm] = s[vm] - dx * h[jm[vm]] if vm.any() else s[vm]
            vp2 = (j2p >= 0) & (j2p < n)
            vm2 = (j2m >= 0) & (j2m < n)
            s[vp2] = s[vp2] - dx * (1.0 - h[j2p[vp2]])
            s[vm2] = s[vm2] + dx * h[j2m[vm2]]
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-14:
                continue
            candidates = [0.0, d_lo, d_hi]
            top5 = np.argsort(g)[::-1][:5]
            for a in range(5):
                for b in range(a + 1, 5):
                    ds = s[top5[b]] - s[top5[a]]
                    if abs(ds) > 1e-14:
                        dc = (g[top5[a]] - g[top5[b]]) / ds
                        if d_lo < dc < d_hi:
                            candidates.append(dc)
            best_delta = 0.0
            best_val = g.max()
            for d in candidates:
