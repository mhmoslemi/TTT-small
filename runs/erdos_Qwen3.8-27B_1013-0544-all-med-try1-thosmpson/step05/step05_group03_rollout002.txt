import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_local_search(hb, n, deadline, max_improvements=50000):
    """Fast greedy 1-swap local search."""
    best_c = compute_c5(hb, n)
    ones = set(np.where(hb == 1)[0].tolist())
    zeros = set(np.where(hb == 0)[0].tolist())
    improvements = 0
    while improvements < max_improvements:
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    improvements += 1
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_local_search(hb, n, deadline, seed=0, max_iters=200000):
    rng = np.random.RandomState(seed)
    current_c = compute_c5(hb, n)
    best_c = current_c
    best_hb = hb.copy()
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    no_improve = 0
    limit = max(1000, n * 8)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb[i], hb[j] = 1.0, 0.0
        c = compute_c5(hb, n)
        if c < current_c - 1e-15:
            ones[j] = ones[-1]; ones.pop()
            ones.append(i)
            zeros[i] = zeros[-1]; zeros.pop()
            zeros.append(j)
            current_c = c
            no_improve = 0
            if c < best_c:
                best_c = c
                best_hb = hb.copy()
        else:
            hb[i], hb[j] = 0.0, 1.0
            no_improve += 1
        if no_improve > limit:
            break
    return best_hb, best_c

def sa_refine(hb, n, deadline, seed=0, t_start=0.02, t_end=0.0003):
    rng = np.random.RandomState(seed)
    current_c = compute_c5(hb, n)
    best_c = current_c
    best_hb = hb.copy()
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    elapsed_est = 0.000025
    est_iter = max(5000, int((deadline - time.time()) / elapsed_est))
    rate = (t_end / t_start) ** (1.0 / est_iter)
    temp = t_start
    for it in range(est_iter):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb[i], hb[j] = 1.0, 0.0
        c = compute_c5(hb, n)
        delta = c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            ones[j] = ones[-1]; ones.pop()
            ones.append(i)
            zeros[i] = zeros[-1]; zeros.pop()
            zeros.append(j)
            current_c = c
            if c < best_c:
                best_c = c
                best_hb = hb.copy()
        else:
            hb[i], hb[j] = 0.0, 1.0
        temp *= rate
    return best_hb, best_c

def slsqp_smooth(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sh_m = sh.max()
    return m + (sh_m + np.log(np.exp(sh - sh_m).sum())) / alpha

def slsqp_smooth_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sh_m = sh.max()
    vals = np.exp(sh - sh_m)
    w = vals / vals.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += wk * (1.0 - x[s:n])
            if s < n:
                grad[s:n] -= wk * x[0:n-s]
        else:
            ss = -s
            grad[ss:n] += wk * (1.0 - x[0:n-ss])
            if n - ss > 0:
                grad[0:n-ss] -= wk * x[ss:n]
    return grad * dx

def project_grad_step(hb, n, deadline, alpha=100000, max_iter=20000):
    h = hb.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = slsqp_smooth_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = lr / gn
        found = False
        for _ in range(20):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                h = h_new
                found = True
                break
            step *= 0.5
        if not found:
            lr *= 0.8
            stall += 1
            if lr < 1e-12 or stall > 100:
                break
        else:
            stall = 0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.02, 10.0)
    return best_h, best_c
