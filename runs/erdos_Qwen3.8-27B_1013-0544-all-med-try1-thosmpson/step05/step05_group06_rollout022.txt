    """grad[j, m] = dg(d_m)/dh[j] where d_m = m - (n-1)"""
    dm = np.arange(2*n-1) - (n-1)

                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                
                # Find optimal delta: min max_d (g[d] + delta*s[d])
                # Candidates: boundaries + intersections of top-K lines
                candidates = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(m1 + 1, K):
                        d1, d2 = top_idx[m1], top_idx[m2]
                        ds = s[d1] - s[d2]
                        if abs(ds) > 1e-12:
                            dc = (g[d2] - g[d1]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d
                
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    sweep_improved += 1
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if sweep_improved == 0:
            break
    
    return best_h, best_c

                m2lo = max(0, o)

This is better because:
1. We compute g once per sweep (or maintain it incrementally)
2. We find the best swap in one pass (not first-improvement)
3. Applying the best swap is O(n) to update g

But finding the best swap is O(n² * n) = O(n³) per sweep = 15.6M for n=250. In numpy, each dg computation involves a few array operations on ~250-element arrays, so maybe 10-20μs. 62500 * 15μs ≈ 1 second per sweep. That's reasonable.

Actually, I can make it even faster by not using np.zeros(2*n-1) and fancy indexing for each pair. Instead, I can precompute some structures.

For a binary h, the delta_g for swapping o (1→0) and z (0→1) is:
dg(d) = A_o(d) + B_z(d)

where:
A_o(d) = -(1-h[o+d])*1{o+d valid} + h[o-d]*1{o-d valid} (removing 1 at o)
B_z(d) = (1-h[z+d])*1{z+d valid} - h[z-d]*1{z-d valid} (adding 1 at z)

For binary h:
- (1-h[o+d]) = 1 if h[o+d]=0, else 0
- h[o-d] = 1 if h[o-d]=1, else 0

So A_o(d) is non-zero only for d in [-o, n-1-o] ∪ [o-n+1, o], and its values are in {-1, 0, 1}.

Similarly for B_z.

The key insight: A_o depends only on o and the current h. If I precompute A_o for all o, then for each pair (o,z), I just add A_o + B_z.

But A_o changes when h changes (after a swap). So I'd need to recompute after each swap.

For now, let me just do the straightforward O(n³) per sweep approach and see if it's fast enough.

The LP linearization is different from SLSQP. SLSQP uses a smooth approximation of the max and optimizes it with gradient-based methods. LP linearization directly solves the linearized problem at each step.

The LP approach has the advantage that it handles the non-smooth max exactly (at the linearization point), while SLSQP only approximates it.

- This allows the search to explore more of the solution space

This is a key difference from the previous code's binary polish, which only accepts improvements.

```python
import numpy as np
from scipy.optimize import linprog
import time

# Helper functions (top-level, no closures)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_g(h, n):
    dm = np.arange(2*n-1) - (n-1)
    j_idx = np.arange(n)[:, None]
    d_idx = dm[None, :]
    jp = j_idx + d_idx
    mask1 = (jp >= 0) & (jp < n)
    h_jp = np.where(mask1, h[np.clip(jp, 0, n-1)], 0.0)
    term1 = np.where(mask1, 1.0 - h_jp, 0.0)
    jm = j_idx - d_idx
    mask2 = (jm >= 0) & (jm < n)
    h_jm = np.where(mask2, h[np.clip(jm, 0, n-1)], 0.0)
    term2 = np.where(mask2, -h_jm, 0.0)
    return (term1 + term2).astype(float)

def lp_optimize(h, n, deadline):
    """LP-based successive linearization."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ns = 2 * n - 1
    
    while time.time() < deadline:
        g = compute_g(h, n)
        grad = compute_grad_g(h, n)
        
        # Build LP with all shifts
        A_ub = np.zeros((ns, n + 1))
        b_ub = np.zeros(ns)
        for d in range(ns):
            A_ub[d, :n] = grad[:, d]
            A_ub[d, n] = -1.0
            b_ub[d] = float(grad[:, d] @ h) - g[d]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        
        # Backtracking
        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = (1.0 - alpha) * h + alpha * h_lp
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                accepted = True
                break
        
        if not accepted:
            break
    
    return best_h, best_c

def pair_descent(h, n, deadline, K=15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        g = compute_g(h, n)
        grad = compute_grad_g(h, n)
        top_idx = np.argsort(g)[::-1][:K]
        current_max = g.max()
        improved = 0
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = grad[i] - grad[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(a + 1, K):
                        da, db = top_idx[a], top_idx[b]
                        ds = s[da] - s[db]
                        if abs(ds) > 1e-12:
                            dc = (g[db] - g[da]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                
                best_d = 0.0
                best_v = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    v = float(np.max(g + d * s))
                    if v < best_v - 1e-14:
                        best_v = v
                        best_d = d
                
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    g += best_d * s
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved += 1
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            break
    
    return best_h, best_c

def binary_search_anneal(h, n, deadline):
    """Binary local search with simulated annealing."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    g = compute_g(h_bin, n)
    
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    T = 0.001  # initial temperature
    T_min = 1e-6
    T_factor = 0.95
    it = 0
    
    while time.time() < deadline and T > T_min:
        it += 1
        # Pick random pair
        o = int(list(ones).__iter__().__next__()) if ones else None
        if o is None:
            break
        # Use random selection
        o_list = list(ones)
        z_list = list(zeros)
        o = o_list[np.random.randint(len(o_list))]
        z = z_list[np.random.randint(len(z_list))]
        
        # Compute delta_g
        dg = np.zeros(2*n-1)
        # Remove 1 at o
        m1lo = max(0, n-1-o)
        m1hi = min(2*n-2, 2*n-2-o)
        if m1lo <= m1hi:
            dv = np.arange(m1lo, m1hi+1) - (n-1)
            dg[m1lo:m1hi+1] -= (1.0 - h_bin[o + dv])
        m2lo = max(0, o)
        m2hi = min(2*n-2, o+n-1)
        if m2lo <= m2hi:
            dv = np.arange(m2lo, m2hi+1) - (n-1)
            dg[m2lo:m2hi+1] += h_bin[o - dv]
        # Add 1 at z
        m3lo = max(0, n-1-z)
        m3hi = min(2*n-2, 2*n-2-z)
        if m3lo <= m3hi:
            dv = np.arange(m3lo, m3hi+1) - (n-1)
            dg[m3lo:m3hi+1] += (1.0 - h_bin[z + dv])
        m4lo = max(0, z)
        m4hi = min(2*n-2, z+n-1)
        if m4lo <= m4hi:
            dv = np.arange(m4lo, m4hi+1) - (n-1)
            dg[m4lo:m4hi+1] -= h_bin[z - dv]
        
        c_new = float(np.max(g + dg))
        delta = c_new - float(np.max(g))
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h_bin[o] = 0.0
            h_bin[z] = 1.0
            g += dg
            ones.discard(o)
            ones.add(z)
            zeros.discard(z)
            zeros.add(o)
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
        
        T *= T_factor
    
    return best_h, best_c
