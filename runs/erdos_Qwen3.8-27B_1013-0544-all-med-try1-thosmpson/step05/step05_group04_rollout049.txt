    for a in alphas:
        if time.time() > deadline - 30:
            break
        h_new = slsqp_opt(h, n, a, maxiter=1200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            h = h_new
    if time.time() < deadline - 20:
        h, best_c = bottleneck_reduction(h, n, deadline - 20, top_m=5)
    rem = deadline - time.time() - 5
    if rem > 10:
        if n <= 130:
            h, best_c = fine_pair_search(h, n, time.time() + rem)
        else:
            h, best_c = near_pair_search(h, n, time.time() + rem, radius=50)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Optimize initial construction
    alloc1 = min(250, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_single(best_h, n_init, t0 + alloc1)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt

    # Phase 2: Try other n values
    x_init = np.linspace(0, 2, n_init)
    for n_try in [150, 200, 300]:
        if time.time() > deadline - 100:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_single(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with perturbation
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended fine pair search on best
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 130:
            best_h, best_c = fine_pair_search(best_h, n, deadline - 3)
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3, radius=60)

    # Phase 5: Final SLSQP polish + tiny pair search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_opt(best_h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            if n <= 130:
                best_h, best_c = fine_pair_search(best_h, n, deadline - 0.5)
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 0.5, radius=30)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
