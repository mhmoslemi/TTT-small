def compute_g(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

    best_h = h.copy()

    zeros_set = set(zeros_list)

            break

            b_cand = zeros_list[np.argmin(vb)]
            delta = vec[b_cand] - vec[a_cand]
            new_max = float(np.max(g + delta))
            impr = c - new_max
            if impr > best_impr:
                best_impr = impr

        # Also try a few random pairs

                best_a, best_b = a, b

            delta = vec[b] - vec[a]

            g += delta

            stall = 0

                best_c = c
                best_h = h.copy()
        else:
            stall += 1

def subgradient_descent(h, n, deadline, seed=42):

    nm1 = n - 1
    rng = np.random.RandomState(seed)
    best_h = h.copy()

    eta = 0.05  # initial step size

        if time.time() > deadline:
            break
        
        g = compute_g(h, n)
        c = float(g.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Subgradient: average of vec columns at argmax shifts
        max_val = g.max()
        argmax_mask = g >= max_val - 1e-10
        m_indices = np.where(argmax_mask)[0]
        
        vec = compute_vec(h, n)
        s = vec[:, m_indices].mean(axis=1)  # subgradient

        for _ in range(5):

            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < c:
                h = h_new
                break
            eta *= 0.5
        else:
            eta *= 0.95
        
        if eta < 1e-6:
            break
    
    return best_h, best_c

def compute_subgradient(h, n, m_indices):
    """Compute subgradient of max g at h, averaged over m_indices."""
    nm1 = n - 1
    s = np.zeros(n)
    for m_idx in m_indices:
        d = m_idx - nm1
        for i in range(n):
            contrib = 0.0
            j = i + d
            if 0 <= j < n:
                contrib += (1.0 - h[j])
            j2 = i - d
            if 0 <= j2 < n:
                contrib -= h[j2]
            s[i] += contrib
    s /= len(m_indices)
    return s

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_vec(h, n):
    nm1 = n - 1
    vec = np.zeros((n, 2 * n - 1))
    for idx_d in range(2 * n - 1):
        d = idx_d - nm1
        hp = np.zeros(n)
        hm = np.zeros(n)
        if d >= 0:
            if d < n:
                hp[:n - d] = h[d:]
                hm[d:] = h[:n - d]
        else:
            da = -d
            if da < n:
                hp[da:] = h[:n - da]
                hm[:da] = h[da:]
        vec[:, idx_d] = 1.0 - hp - hm
    return vec

def update_vec_swap(vec, a, b, n):
    nm1 = n - 1
    lo = max(0, a - nm1)
    hi = min(n, a + nm1 + 1)
    i = np.arange(lo, hi)
    vec[i, a - i + nm1] += 1.0
    vec[i, i - a + nm1] += 1.0
    lo = max(0, b - nm1)
    hi = min(n, b + nm1 + 1)
    j = np.arange(lo, hi)
    vec[j, b - j + nm1] -= 1.0
    vec[j, j - b + nm1] -= 1.0

    g = compute_g(h, n)
    c = float(g.max())
    best_h = h.copy()
    best_c = c

    zeros_arr = np.where(h == 0)[0].copy()

                zeros_arr[ib] = a
            g = compute_g(h, n)
            c = float(g.max())
            stall = 0
            continue

        best_impr = 0.0
        best_ia = -1
        best_ib = -1
        for m in top_idx:

            vb = vec[zeros_arr, m]
            iac = int(np.argmax(va))
            ibc = int(np.argmin(vb))

            b = int(zeros_arr[ibc])

def slsqp_optimize(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sm = alpha * (g - m)
        sm = sm - sm.max()
        return m + np.log(np.exp(sm).sum()) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_step(h, n, eta):
    nm1 = n - 1
    g = compute_g(h, n)
    c = float(g.max())
    max_v = g.max()
    m_idx = np.where(g >= max_v - 1e-10)[0]
    s = np.zeros(n)
    for mi in m_idx:
        d = mi - nm1
        for i in range(n):
            v = 0.0
            j = i + d
            if 0 <= j < n:
                v += (1.0 - h[j])
            j2 = i - d
            if 0 <= j2 < n:
                v -= h[j2]
            s[i] += v
    s /= len(m_idx)
    return h - eta * s, c

def subgradient_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.03
    stale = 0
    while time.time() < deadline:
        h_new, c = subgrad_step(h, n, eta)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < c:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        else:
            eta *= 0.9
            stale += 1
        if stale > 200:
            break
    return best_h, best_c
