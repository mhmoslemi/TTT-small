def compute_gradient(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    g_shifted = g - g.max()
    exp_g = np.exp(alpha * g_shifted)
    w = exp_g / exp_g.sum()  # shape: (2n-1,)

                grad[i] += w[idx] * (1 - h[i+d])

        return sm + pen_w * (x.sum() - target) ** 2

    def grad(x):
        gsm = compute_grad_sm(x, n, alpha)
        return gsm + 2.0 * pen_w * (x.sum() - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def build_F(h, n, dm):
    F = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        d = int(dm[m])
        i_lo, i_hi = max(0, d), min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo, j_hi = max(0, -d), min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m] -= h[j_lo + d:j_hi + d]
    return F

def pair_descent(h, n, deadline, max_sweeps=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = build_F(h, n, dm)
        g = compute_g(h, n)
        cur_max = g.max()
        K = min(12, 2 * n - 1)
        top_idx = np.argsort(g)[::-1][:K]
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            hi = h[i]
            if hi < 0.01 or hi > 0.99:
                continue
            for j in range(i + 1, n):
                hj = h[j]
                if abs(hi - hj) < 0.01:
                    continue
                s = F[i] - F[j]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-14:
                    continue
                cands = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                cands.append(dc)
                best_d, best_v = 0.0, cur_max
                for d in cands:
                    d = float(np.clip(d, d_lo, d_hi))
                    v = float(np.max(g + d * s))
                    if v < best_v - 1e-14:
                        best_v, best_d = v, d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    g += best_d * s
                    v1 = (i + dm >= 0) & (i + dm < n)
                    F[i + dm[v1], m_idx[v1]] -= best_d
                    v2 = (i - dm >= 0) & (i - dm < n)
                    F[i - dm[v2], m_idx[v2]] -= best_d
                    v3 = (j + dm >= 0) & (j + dm < n)
                    F[j + dm[v3], m_idx[v3]] += best_d
                    v4 = (j - dm >= 0) & (j - dm < n)
                    F[j - dm[v4], m_idx[v4]] += best_d
                    cur_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if improved == 0:
            stall += 1
            if stall > 4:
                break
        else:
            stall = 0
    return best_h, best_c

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    for _ in range(30):
        if time.time() > deadline:
            break
        improved = False
        for z in sorted(zeros):
            for o in sorted(ones):
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.discard(o); ones.add(z)
                    zeros.discard(z); zeros.add(o)
                    improved = True
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing sharpness (30% budget)
    p1 = t0 + 0.30 * (budget_s - 5)
    h = best_h.copy()
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 8000000]
    for alpha in alphas:
        if time.time() > p1:
            break
        h = lbfgsb_phase(h, n, alpha, p1, maxiter=250)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Pair descent (55% budget)
    p2 = t0 + 0.85 * (budget_s - 5)
    h, c = pair_descent(best_h.copy(), n, p2, max_sweeps=2000)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 3: L-BFGS-B polish + pair descent (remaining)
    if time.time() < dl - 30:
        h = lbfgsb_phase(best_h.copy(), n, 10000000, dl - 30, maxiter=150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        h, c = pair_descent(best_h.copy(), n, dl - 3, max_sweeps=500)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Check binary solution
    if time.time() < dl - 20:
        hb, cb = binary_polish(best_h.copy(), n, dl - 5)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
