def sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def smooth_subgradient(h, n, alpha_sg=200.0):
    """Compute the smoothed subgradient of max(c) w.r.t. h."""
    dx = 2.0 / n
    c = get_corr(h, n)
    curr = c.max()
    # Softmax weights over lags
    w = np.exp(alpha_sg * (c - curr))
    w /= w.sum()
    # For each lag, compute gradient
    # c[k] for k = 0..2n-2, shift s = k - (n-1)
    g = np.zeros(n)
    # We'll iterate over active lags (those with significant weight)
    # For efficiency, only process lags with w > threshold
    threshold = w.max() * 1e-6
    active = np.where(w > threshold)[0]
    for k in active:
        s = int(k) - (n - 1)
        wk = w[int(k)] * dx
        if s == 0:
            g += wk * (1 - 2 * h)
        elif s > 0:
            # ∂/∂h[j] of ∑_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # = (1-h[j+s]) for j=0..n-1-s (first factor)
            # + (-h[j-s]) for j=s..n-1 (second factor)
            g[:n-s] += wk * (1 - h[s:])
            g[s:] += wk * (-h[:n-s])
        else:
            sp = -s
            g[sp:] += wk * (1 - h[:n-sp])
            g[:n-sp] += wk * (-h[sp:])
    # Remove mean to respect sum constraint
    g -= g.mean()
    return g, c, curr

def momentum_subgradient(h, n, deadline, alpha_sg=200.0, gamma=0.9):
    """Nesterov-style momentum subgradient method."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    lr = 0.01
    t = 1.0
    stagnation = 0
    
    while time.time() < deadline:
        t += 1
        beta = (1 + np.sqrt(1 + 4 * t * t)) / 2
        # Nesterov extrapolation
        momentum_factor = (beta - 1) / beta
        h_extrap = h_curr + momentum_factor * (h_curr - h_prev)
        h_extrap = project_feasible(h_extrap, n)
        
        g, c, curr = smooth_subgradient(h_extrap, n, alpha_sg)
        
        # Backtracking line search
        accepted = False
        for trial in range(20):
            h_new = h_extrap - lr * g
            h_new = project_feasible(h_new, n)
            new_c = compute_c5(h_new, n)
            if new_c < curr - 1e-15:
                h_prev = h_curr
                h_curr = h_new
                accepted = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = h_new.copy()
                    stagnation = 0
                break
            lr *= 0.5
        if not accepted:
            lr *= 0.5
            h_prev = h_curr
            stagnation += 1
            if stagnation > 200:
                # Reset momentum and increase alpha_sg
                alpha_sg *= 2
                lr = 0.01
                stagnation = 0
                if alpha_sg > 1e6:
                    break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_eq(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def smooth_sgrad(h, n, alpha_sg=300.0):
    dx = 2.0 / n
    c = get_corr(h, n)
    curr = c.max()
    w = np.exp(alpha_sg * (c - curr))
    w /= w.sum()
    n_corr = len(c)
    g = np.zeros(n)
    thresh = w.max() * 1e-5
    active = np.where(w > thresh)[0]
    for kk in active:
        k = int(kk)
        s = k - (n - 1)
        wk = w[kk] * dx
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] += wk * (-h[:n - s])
        else:
            p = -s
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] += wk * (-h[p:])
    g -= g.mean()
    return g, c, curr

def momentum_subgrad(h, n, deadline, alpha_sg=300.0):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    lr = 0.005
    t_counter = 1.0
    stagnation = 0
    while time.time() < deadline:
        t_counter += 1
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t_counter * t_counter)) / 2.0
        mom = (beta - 1.0) / beta
        h_ex = h_curr + mom * (h_curr - h_prev)
        h_ex = project_feasible(h_ex, n)
        g, c, curr = smooth_sgrad(h_ex, n, alpha_sg)
        accepted = False
        for _ in range(25):
            h_new = project_feasible(h_ex - lr * g, n)
            new_c = compute_c5(h_new, n)
            if new_c < curr - 1e-15:
                h_prev = h_curr
                h_curr = h_new
                accepted = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = h_new.copy()
                    stagnation = 0
                break
            lr *= 0.5
        if not accepted:
            lr *= 0.5
            h_prev = h_curr
            stagnation += 1
            if stagnation > 300:
                alpha_sg = min(alpha_sg * 1.5, 1e6)
                lr = 0.005
                stagnation = 0
                if alpha_sg > 1e6:
                    break
    return best_h, best_c

def guided_polish(h, n, deadline, num_worst=5):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
        top_k = np.argsort(c)[-num_worst:]
        g = np.zeros(n)
        for kk in top_k:
            s = int(kk) - (n - 1)
            if s == 0:
                g += (1.0 - 2.0 * h)
            elif s > 0:
                g[:n - s] += (1.0 - h[s:])
                g[s:] += (-h[:n - s])
            else:
                p = -s
                g[p:] += (1.0 - h[:n - p])
                g[:n - p] += (-h[p:])
        g -= g.mean()
        g *= dx
        if g.max() < 1e-14:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-13:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > 300:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < curr - 1e-15:
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 200:
                step_idx = min(step_idx + 1, len(steps) - 1)
                stall = 0
                if step_idx >= len(steps) - 1:
                    break
    return best_h, best_c
