            if tot < 1e-14:

def build_gradient_matrix(h, n):
    """Build G where G[k,j] = dc(k)/dh[j] for all k, j."""
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    j = np.arange(n)
    for k_idx in range(2*n-1):
        s = k_idx - (n-1)
        G[k_idx, :] = dx
        vp = (j + s >= 0) & (j + s < n)
        vm = (j - s >= 0) & (j - s < n)
        G[k_idx, vp] -= dx * h[j[vp] + s]
        G[k_idx, vm] -= dx * h[j[vm] - s]

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k_idx in range(2 * n - 1):
        s = k_idx - (n - 1)
        vp = (j + s >= 0) & (j + s < n)
        vm = (j - s >= 0) & (j - s < n)
        G[k_idx, vp] += dx * (1.0 - h[j[vp] + s])
        G[k_idx, vm] += -dx * h[j[vm] - s]
    return G

def lp_linearize(h, n, deadline, max_iters=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline - 1:
            break
        c = get_corr(h, n)
        G = build_gradient_matrix(h, n)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        cons.append(G @ h_var - t_var <= -c + G @ h)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            try:
                kwargs = {'max_iters': 20000, 'eps': 1e-9, 'verbose': False}
                if solver == cp.CLARABEL:
                    kwargs = {'tol_abs': 1e-9, 'tol_gap': 1e-9, 'verbose': False}
                elif solver == cp.ECOS:
                    kwargs = {'max_iters': 10000, 'abstol': 1e-8, 'reltol': 1e-8, 'verbose': False}
                else:
                    kwargs = {'max_iters': 20000, 'eps': 1e-9, 'verbose': False}
                prob.solve(solver=solver, **kwargs)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:

        h_new = np.clip(h_var.value, 0.0, 1.0)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def slsqp_round(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                best_c = c
                best_h = ht
    # Phase 3: Multi-lag polish
    if time.time() < deadline - 80:
        h_m, c_m = multi_lag_polish(best_h, n, min(deadline - 60, time.time() + 180), num_lags=5)
        if c_m < best_c:
            best_c = c_m
            best_h = h_m
    # Phase 4: Directed polish
    if time.time() < deadline - 40:
        h_d, c_d = directed_polish(best_h, n, min(deadline - 30, time.time() + 150))
        if c_d < best_c:
            best_c = c_d
            best_h = h_d
    # Phase 5: Final LP pass
    if time.time() < deadline - 30:
        h_f, c_f = lp_linearize(best_h, n, min(deadline - 20, time.time() + 80), max_iters=50)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    # Phase 6: Final polish
    if time.time() < deadline - 5:
        h_p, c_p = directed_polish(best_h, n, deadline - 2)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
