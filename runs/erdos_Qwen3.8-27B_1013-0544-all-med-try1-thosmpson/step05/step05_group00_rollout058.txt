            best_h = h_new.copy()
        
        h = h_new
        t = t_new
        
    return best_h

        vmj = (j_minus >= 0) & (j_minus < n)
        hpj = h[np.clip(j_plus, 0, n-1)] * vpj

                mj1 = np.clip(mj1, 0, m-1)

                idx = final.argmin()

                    best_i = zeros[ii]
                    best_j = ones[idx]
        else:

            mx = combined.max(axis=2)  # (nz, no)

def nesterov_descent(h, n, alpha, iterations, lr_init, deadline):
    """Projected Nesterov gradient on smoothed objective."""
    dx = 2.0 / n
    m = 2 * n - 1
    offsets = np.arange(-(n-1), n)
    
    h = h.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_f = np.inf
    t = 1.0
    lr = lr_init
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        
        corr_y = np.correlate(y, 1 - y, mode='full') * dx
        max_c = corr_y.max()
        sh = alpha * (corr_y - max_c)
        w = np.exp(sh - sh.max())
        w /= w.sum()
        
        # Gradient: grad[i] = dx * (1 - A[i] - B[i])
        # A[i] = sum_k w[k] * h[i + offsets[k]]
        # B[i] = sum_k w[k] * h[i - offsets[k]]
        A = np.zeros(n)
        B = np.zeros(n)
        for k in range(m):
            wk = w[k]
            if wk < 1e-10:
                continue
            off = offsets[k]
            sp = max(0, -off)
            ep = min(n, n - off)
            if sp < ep:
                A[sp:ep] += wk * h[sp+off:ep+off]
            sq = max(0, off)
            eq = min(n, n + off)
            if sq < eq:
                B[sq:eq] += wk * h[sq-off:eq-off]
        
        grad = dx * (1.0 - A - B)
        
        # Nesterov
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        if it > 0:
            y = h + beta * (h - y)
        else:
            y = h.copy()
        
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
        f_new = corr_new.max()
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        
        h = h_new
        t = t_new
        lr = lr_init / np.sqrt(it + 1)
    
    return best_h, best_f

    offsets = np.arange(-(n-1), n)
    h = h.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_f = compute_c5(h, n)
    t_val = 1.0
    for it in range(iterations):
        if time.time() > deadline:
            break
        corr_y = np.correlate(y, 1 - y, mode='full') * dx
        max_c = corr_y.max()
        sh = alpha * (corr_y - max_c)
        w = np.exp(sh - sh.max())
        w /= w.sum()
        A = np.zeros(n)
        B = np.zeros(n)
        for k in range(m):
            wk = w[k]
            if wk < 1e-10:
                continue
            off = offsets[k]
            sp = max(0, -off)
            ep = min(n, n - off)
            if sp < ep:
                A[sp:ep] += wk * h[sp+off:ep+off]
            sq = max(0, off)
            eq = min(n, n + off)
            if sq < eq:
                B[sq:eq] += wk * h[sq-off:eq-off]
        grad = dx * (1.0 - A - B)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        if it > 0:
            y = h + beta * (h - y)
        else:
            y = h.copy()
        lr = lr_init / np.sqrt(it + 1)
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        f_new = compute_c5(h_new, n)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h = h_new
        t_val = t_new
    return best_h, best_f

def vectorized_binary_search(h, n, deadline, max_sweeps=5000, seed=42):
    dx = 2.0 / n
    m = 2 * n - 1
    offsets = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    h = np.round(h).clip(0, 1).astype(float)
    target = n // 2
    ones_idx = np.where(h == 1)[0]
    if len(ones_idx) > target:
        h[ones_idx[target:]] = 0
    elif len(ones_idx) < target:
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[:target - len(ones_idx)]] = 1
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 80:
            if time.time() > deadline - 5:
                break
            for _ in range(3):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = corr.max()
            stall = 0
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz = len(zeros)
        no = len(ones)
        if nz == 0 or no == 0:
            break
        i_plus = zeros[:, None] + offsets[None, :]
        i_minus = zeros[:, None] - offsets[None, :]
        vp = (i_plus >= 0) & (i_plus < n)
        vm = (i_minus >= 0) & (i_minus < n)
        part1 = (1.0 - h[np.clip(i_plus, 0, n-1)]) * vp
        part2 = h[np.clip(i_minus, 0, n-1)] * vm
        flip_up = part1 - part2
        j_plus = ones[:, None] + offsets[None, :]
        j_minus = ones[:, None] - offsets[None, :]
        vpj = (j_plus >= 0) & (j_plus < n)
        vmj = (j_minus >= 0) & (j_minus < n)
        part1j = -(1.0 - h[np.clip(j_plus, 0, n-1)]) * vpj
        part2j = h[np.clip(j_minus, 0, n-1)] * vmj
        flip_down = part1j + part2j
        base = corr[None, :] + flip_up
        mem = nz * no * m * 8
        jmi = ones[None, :] - zeros[:, None]
        best_impr = cur_max
        best_i = -1
        best_j = -1
        if mem < 200e6:
            combined = base[:, None, :] + flip_down[None, :, :]
            mx = combined.max(axis=2)
            mc0 = base[:, n-1][:, None] + flip_down[None, :, n-1] - 2.0
            mj1 = np.clip(n - 1 + jmi, 0, m-1)
            idx1 = zeros[:, None] * m + mj1
            b_at1 = base.ravel()[idx1]
            idx1f = ones[None, :] * m + mj1
            fd_at1 = flip_down.ravel()[idx1f]
            mc1 = b_at1 + fd_at1 + 1.0
            mj2 = np.clip(n - 1 - jmi, 0, m-1)
            idx2 = zeros[:, None] * m + mj2
            b_at2 = base.ravel()[idx2]
            idx2f = ones[None, :] * m + mj2
            fd_at2 = flip_down.ravel()[idx2f]
            mc2 = b_at2 + fd_at2 + 1.0
            final = np.maximum(mx, np.maximum(np.maximum(mc0, mc1), mc2))
            flat_idx = final.argmin()
            ii, jj = np.unravel_index(flat_idx, final.shape)
            best_impr = final[flat_idx]
            best_i = zeros[ii]
            best_j = ones[jj]
        else:
            arange_no = np.arange(no)
            for ii in range(nz):
                b = base[ii]
                cm_mx = (b[None, :] + flip_down).max(axis=1)
                mc0_i = b[n-1] + flip_down[:, n-1] - 2.0
                mj1_i = np.clip(n - 1 + (ones - zeros[ii]), 0, m-1)
                mc1_i = b[mj1_i] + flip_down[arange_no, mj1_i] + 1.0
                mj2_i = np.clip(n - 1 - (ones - zeros[ii]), 0, m-1)
                mc2_i = b[mj2_i] + flip_down[arange_no, mj2_i] + 1.0
                fin = np.maximum(cm_mx, np.maximum(np.maximum(mc0_i, mc1_i), mc2_i))
                idx = fin.argmin()
                if fin[idx] < best_impr:
                    best_impr = fin[idx]
                    best_i = zeros[ii]
                    best_j = ones[idx]
        if best_impr >= cur_max - 1e-12:
            stall += 1
            continue
        stall = 0
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_max = corr.max()
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP polish
    p1 = t0 + 0.08 * budget_s
    h = project_feasible(best_h, n)
    for alpha in [5000, 50000, 500000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 2000, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Nesterov
    p2 = t0 + 0.15 * budget_s
    if time.time() < p2:
        for alpha in [2000, 20000, 200000]:
            if time.time() > p2:
                break
            hn, fn = nesterov_descent(best_h, n, alpha, 3000, 0.05, p2)
            c = compute_c5(hn, n)
            if c < best_c:
                best_c = c
                best_h = hn.copy()
    # Phase 3: Main vectorized binary search
    p3 = t0 + 0.72 * budget_s
    hb, cb = vectorized_binary_search(best_h, n, p3, max_sweeps=10000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Phase 4: SLSQP polish from binary
    p4 = t0 + 0.82 * budget_s
    if time.time() < p4:
        for alpha in [100000, 1000000, 10000000]:
            if time.time() > p4:
                break
            hp = slsqp_smooth(best_h + 0.02 * rng.randn(n), n, alpha, 3000, p4)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()
    # Phase 5: More binary search
    p5 = dl - 8
    if time.time() < p5:
        hb, cb = vectorized_binary_search(best_h, n, p5, max_sweeps=10000, seed=seed+100)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 6: Final SLSQP
    if time.time() < dl - 3:
        hp = slsqp_smooth(best_h.copy(), n, 50000000, 2000, dl - 2)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
