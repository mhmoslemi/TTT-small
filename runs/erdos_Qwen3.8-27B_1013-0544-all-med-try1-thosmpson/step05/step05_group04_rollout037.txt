```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    """Return all correlation values (not just the max)"""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def optimal_pair_step(corr_values, h, i, j, n):
    """Find the optimal d for moving h[i]+=d, h[j]-=d that minimizes max correlation.
    
    corr_values: the full correlation array (length 2n-1)
    Returns: (best_d, best_max_val)
    """
    # Slopes: s[L] = h[j+L-n+1] - h[i+L-n+1] with out-of-bounds = 0
    # For L in [0, 2n-2], the effective index for h is L - (n-1)
    # corr[L] = sum_m h[m] * (1-h)[m + L - n + 1]
    # After change: corr_new[L] = corr[L] + d*((1-h0)[i+L-n+1] - (1-h0)[j+L-n+1])
    # = corr[L] + d * (h0[j+L-n+1] - h0[i+L-n+1])  (with h0[out_of_bounds]=0)
    
    lag = np.arange(2*n-1) - (n-1)  # lag ranges from -(n-1) to n-1
    idx_i = i + lag  # index into h for the i-term
    idx_j = j + lag  # index into h for the j-term
    
    # h[idx] is valid only if 0 <= idx < n
    hi_valid = (idx_i >= 0) & (idx_i < n)
    hj_valid = (idx_j >= 0) & (idx_j < n)
    
    hi_vals = np.zeros(2*n-1)
    hj_vals = np.zeros(2*n-1)
    hi_vals[hi_valid] = h[idx_i[hi_valid]]
    hj_vals[hj_valid] = h[idx_j[hj_valid]]
    
    slopes = hj_vals - hi_vals  # s[L]
    
    # We want to minimize max_L (corr_values[L] + d * slopes[L])
    # subject to: max(0, -h[i]/d_sign, ...) bounds on d
    # h[i] + d >= 0 => d >= -h[i]
    # h[i] + d <= 1 => d <= 1 - h[i]
    # h[j] - d >= 0 => d <= h[j]
    # h[j] - d <= 1 => d >= h[j] - 1
    d_min = max(-h[i], h[j] - 1)
    d_max = min(1 - h[i], h[j])
    
    if d_min >= d_max:
        return 0.0, corr_values.max()
    
    # The function f(d) = max_L (corr_values[L] + d * slopes[L]) is convex in d
    # Find its minimum over [d_min, d_max]
    # At the optimum, either d is at a bound, or two slopes are "active"
    
    # Efficient approach: find the minimum of the upper envelope
    # For a set of lines a_L * d + b_L, the minimum of the max is where
    # two lines (or more) intersect at the minimum point of the upper envelope.
    
    # Simple approach: ternary search or bisection on the subgradient
    # The subgradient at d is: {slope[L] : L is active at d}
    # At the minimum, 0 is in the convex hull of active slopes.
    
    # Even simpler: since f is convex and piecewise linear, use bisection
    # on the "direction" of the subgradient.
    
    # Actually, the most efficient approach for piecewise linear convex function:
    # The minimum is at a breakpoint or at the boundary.
    # Breakpoints occur where corr[L1] + d*s[L1] = corr[L2] + d*s[L2]
    # i.e., d = (corr[L2] - corr[L1]) / (s[L1] - s[L2])
    
    # But there are O(n²) breakpoints. Instead, use the fact that at the optimum,
    # there are at most 2 active constraints (in 1D).
    
    # For 1D, the optimal d* is where the slope changes sign.
    # Slope of f at d (from the right) = max{s[L] : L active at d}
    # Slope of f at d (from the left) = min{s[L] : L active at d}
    
    # Bisection approach:
    lo, hi = d_min, d_max
    for _ in range(60):  # enough for double precision
        mid = (lo + hi) / 2
        vals = corr_values + mid * slopes
        active_max = vals.max()
        active = vals >= active_max - 1e-15
        min_slope = slopes[active].min()
        max_slope = slopes[active].max()
        if min_slope > 1e-12:
            # Function is increasing at mid, go left
            hi = mid
        elif max_slope < -1e-12:
            # Function is decreasing at mid, go right
            lo = mid
        else:
            # 0 is in the subgradient, we're at a minimum
            lo = hi = mid
            break
    
    d_opt = (lo + hi) / 2
    optimal_val = (corr_values + d_opt * slopes).max()
    return d_opt, optimal_val

def pair_search_line(h, n, deadline, max_sweeps=100):
    """Pair search with optimal line search steps."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    corr = compute_c5_all(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all pairs
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i+1, n):
                d_opt, opt_val = optimal_pair_step(corr, h, i, j, n)
                if abs(d_opt) < 1e-15:
                    continue
                # Check improvement
                new_max = opt_val * dx
                if new_max < best_c - 1e-14:
                    h[i] += d_opt
                    h[j] -= d_opt
                    best_c = new_max
                    corr = compute_c5_all(h, n)
                    improved = True
        if not improved:
            break
    return h, best_c
```

```python
def vectorized_pair_step(corr, h, i, n, active_lags):
    """For a fixed i, compute optimal d for all j simultaneously."""
    lag = active_lags - (n-1)  # shape (m,)
    
    # For each j in [0, n), compute slopes[j, L] = h[j+lag[L]] - h[i+lag[L]]
    # h[j+lag[L]]: for j in [0,n) and lag[L] in [-(n-1), n-1]
    # j+lag[L] ranges over [-(n-1), 2n-2)
    
    # Build a (n, m) matrix for h[j+lag[L]]
    # Index: rows are j, columns are L
    idx = np.arange(n)[:, None] + lag[None, :]  # shape (n, m)
    valid = (idx >= 0) & (idx < n)
    h_at_idx = np.where(valid, h[np.clip(idx, 0, n-1)], 0.0)
    
    # Similarly for h[i+lag[L]]
    idx_i = i + lag  # shape (m,)
    valid_i = (idx_i >= 0) & (idx_i < n)
    h_i_at_idx = np.where(valid_i, h[np.clip(idx_i, 0, n-1)], 0.0)  # shape (m,)
    
    slopes = h_at_idx - h_i_at_idx[None, :]  # shape (n, m)
    
    # For each j, find optimal d:
    # bounds: d_min_j = max(-h[i], h[j]-1), d_max_j = min(1-h[i], h[j])
    d_min = np.maximum(-h[i], h - 1)  # shape (n,)
    d_max = np.minimum(1 - h[i], h)  # shape (n,)
    
    # Bisection for all j simultaneously
    lo = d_min.copy()
    hi = d_max.copy()
    
    corr_active = corr[active_lags]  # shape (m,)
    
    for _ in range(50):

        vals = corr_active[None, :] + mid[:, None] * slopes  # shape (n, m)
        m_vals = vals.max(axis=1)  # shape (n,)
        act = vals >= m_vals[:, None] - 1e-15  # shape (n, m)

        hi = np.where(max_sl < -1e-14, mid, hi)
        # If neither condition, leave as is (converged)
    
    d_opt = (lo + hi) / 2  # shape (n,)
    # Compute the resulting max for each j
    vals_final = corr_active[None, :] + d_opt[:, None] * slopes
    new_max = vals_final.max(axis=1)  # shape (n,)
    
    return d_opt, new_max
```

One more idea: **projected subgradient with momentum**. The objective C₅(h) = max_k f_k(h) is convex in h (as the max of linear functions... wait, are the f_k linear in h?).

f_k(h) = dx * sum_i h[i] * (1-h[i+k]) = dx * (sum_i h[i] - sum_i h[i]*h[i+k])

This is NOT linear in h because of the h[i]*h[i+k] term. It's quadratic. So C₅ is the max of quadratic (concave) functions, which is not necessarily convex.

Wait, actually: f_k(h) = dx * (sum(h) - R_h(k)) where R_h(k) = sum_i h[i]*h[i+k].
The term R_h(k) is a quadratic form in h (specifically, h^T P_k h / 2 where P_k is a permutation matrix). So f_k is the difference of a linear term and a quadratic term, making it concave (since the quadratic term is being subtracted and P_k is PSD).

Wait: R_h(k) = sum_i h[i]*h[i+k] = h^T Q_k h where Q_k has Q_k[i,i+k] = 1 (and by symmetry Q_k[i+k,i] = 1... no, it's not symmetric in the usual sense for a general k).

Actually, R_h(k) = sum_i h[i]*h[i+k]. If I write this as h^T A_k h where A_k[i,j] = 1 if j = i+k (mod n), then:
h^T A_k h = sum_{i,j} h[i] * A_k[i,j] * h[j] = sum_i h[i] * h[i+k]

But A_k is not symmetric (A_k[i,j] = 1 if j=i+k, but A_k[j,i] = 1 if i=j+k, which is different). However, h^T A_k h = h^T A_k^T h, so we can use (A_k + A_k^T)/2.

(A_k + A_k^T)/2 has [i,j] = (1+1)/2 = 1 if j=i+k or i=j+k, i.e., |i-j|=k (mod n). So it's a circulant matrix with 1's at positions ±k.

This matrix is PSD (it's the adjacency matrix of a cycle graph, whose eigenvalues are 2cos(2πjk/n) which are bounded but not all positive... wait, they can be negative).

So f_k(h) = dx * (sum(h) - h^T B_k h) where B_k = (A_k + A_k^T)/2.
Since B_k is not necessarily PSD, f_k is not necessarily concave.

This means C₅ = max_k f_k is not necessarily convex, and the problem is non-convex. This explains why local methods can get stuck.

2. Very effective local search (to minimize within that basin)

For the local search, since the problem is non-convex, the pair search with line search is still valid (it's a coordinate-wise local search that doesn't require convexity).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _obj_args(x, n, alpha):
    return obj_smooth(x, n, alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(lambda x: obj_smooth(x, n, alpha), h,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def find_active(corr, n, tol=1e-6):
    mx = corr.max()
    active = np.where(corr >= mx - tol)[0]
    if len(active) < 2:
        top = np.argsort(corr)[-min(4, len(corr)):]
        active = top
    return active

def vectorized_pair_search(h, n, deadline, tol=1e-8):
    """Vectorized pair search: for each i, find best j and optimal step."""
    dx = 2.0 / n
    corr = compute_c5_all(h, n)
    best_c = float(corr.max())
    total_sweeps = 0

    while time.time() < deadline:
        active = find_active(corr, n, tol=1e-5)
        m = min(len(active), 20)
        active = active[:m] if m < len(active) else active

        best_improvement = 1e-14
        best_i, best_j, best_d = -1, -1, 0.0

        for i in range(n):
            if time.time() > deadline:
                break
            lag = active - (n - 1)
            idx_j = np.arange(n)[:, None] + lag[None, :]
            valid_j = (idx_j >= 0) & (idx_j < n)
            h_j = np.where(valid_j, h[np.clip(idx_j, 0, n-1)], 0.0)

            idx_i = i + lag
            valid_i = (idx_i >= 0) & (idx_i < n)
            h_i = np.where(valid_i, h[np.clip(idx_i, 0, n-1)], 0.0)

            slopes = h_j - h_i[None, :]

            d_min = np.maximum(-h[i], h - 1.0)
            d_max = np.minimum(1.0 - h[i], h)

            lo = d_min.copy()
            hi = d_max.copy()

            corr_active = corr[active]

            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = corr_active[None, :] + mid[:, None] * slopes
                mvals = vals.max(axis=1)
                act = vals >= mvals[:, None] - 1e-15
                min_sl = np.where(act, slopes, np.inf).min(axis=1)
                max_sl = where_act = np.where(act, slopes, -np.inf).max(axis=1)
                lo = np.where(min_sl > 1e-13, mid, lo)
                hi = np.where(max_sl < -1e-13, mid, hi)

            d_opt = (lo + hi) / 2.0
            vals_final = corr_active[None, :] + d_opt[:, None] * slopes
            new_max = vals_final.max(axis=1)

            valid_mask = (np.arange(n) != i) & (d_opt > 1e-12) | (np.arange(n) != i) & (d_opt < -1e-12)
            valid_mask = valid_mask & (new_max < best_c - 1e-14)
            if not valid_mask.any():
                continue

            candidates = np.where(valid_mask)[0]
            best_idx = candidates[np.argmin(new_max[candidates])]
            if new_max[best_idx] < best_c - best_improvement:
                best_improvement = best_c - new_max[best_idx]
                best_i, best_j, best_d = i, best_idx, d_opt[best_idx]

        if best_i >= 0:
            h[best_i] += best_d
            h[best_j] -= best_d
            h = np.clip(h, 0.0, 1.0)
            corr = compute_c5_all(h, n)
            best_c = float(corr.max())
            total_sweeps += 1
        else:
            break

    return h, best_c, total_sweeps

def fine_pair_search(h, n, deadline, tol=1e-10):
    """Finer pair search with tighter tolerance."""
    dx = 2.0 / n
    corr = compute_c5_all(h, n)
    best_c = float(corr.max())

    while time.time() < deadline:
        active = find_active(corr, n, tol=1e-7)
        m = min(len(active), 10)
        active = active[:m] if m < len(active) else active

        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lag = active - (n - 1)
            idx_j = np.arange(n)[:, None] + lag[None, :]
            valid_j = (idx_j >= 0) & (idx_j < n)
            h_j = np.where(valid_j, h[np.clip(idx_j, 0, n-1)], 0.0)
            idx_i = i + lag
            valid_i = (idx_i >= 0) & (idx_i < n)
            h_i = np.where(valid_i, h[np.clip(idx_i, 0, n-1)], 0.0)
            slopes = h_j - h_i[None, :]

            d_min = np.maximum(-h[i], h - 1.0)
            d_max = np.minimum(1.0 - h[i], h)
            lo = d_min.copy()
            hi = d_max.copy()
            corr_active = corr[active]

            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = corr_active[None, :] + mid[:, None] * slopes
                mvals = vals.max(axis=1)
                act = vals >= mvals[:, None] - 1e-15
                min_sl = np.where(act, slopes, np.inf).min(axis=1)
                max_sl = np.where(act, slopes, -np.inf).max(axis=1)
                lo = np.where(min_sl > 1e-14, mid, lo)
                hi = np.where(max_sl < -1e-14, mid, hi)

            d_opt = (lo + hi) / 2.0
            vals_final = corr_active[None, :] + d_opt[:, None] * slopes
            new_max = vals_final.max(axis=1)

            valid_mask = (np.arange(n) != i) & ((np.abs(d_opt) > 1e-10)) & (new_max < best_c - 1e-15)
            if not valid_mask.any():

            best_idx = candidates[np.argmin(new_max[candidates])]
            h[i] += d_opt[best_idx]
            h[best_idx] -= d_opt[best_idx]
            h = np.clip(h, 0.0, 1.0)
            best_c = float(new_max[best_idx])
            corr = compute_c5_all(h, n)
            improved = True

        if not improved:
            break

    return h, best_c

def fixed_delta_pair(h, n, deadline, deltas):
    """Fallback pair search with fixed deltas."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n - 1):
                if time.time() > deadline:
                    return h, best_c
                j = (i + 1) % n
                for sign in [1, -1]:
                    di = sign * d
                    if 0 <= h[i] + di <= 1 and 0 <= h[j] - di <= 1:
                        h[i] += di
                        h[j] -= di
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= di
                            h[j] += di
            if not improved:
                break
    return h, best_c
```
