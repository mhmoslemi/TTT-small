def compute_S(h, n):

    dm = np.arange(2*n-1) - (n-1)

from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, n, alpha, penalty):
    g = compute_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    obj = m + (max_s + np.log(sum_exp)) / alpha
    
    grad = np.zeros(n)
    dm = np.arange(2*n-1) - (n-1)
    for idx in range(2*n-1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        d = dm[idx]
        lo1 = max(0, d)
        hi1 = min(n, d + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1-d:hi1-d])
        lo2 = max(0, -d)
        hi2 = min(n, n-d)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2+d:hi2+d]
    
    sv = h.sum() - n/2.0
    grad += 2.0 * penalty * sv
    obj += penalty * sv * sv
    return obj, grad

def run_lbfgs(h, n, alpha, penalty, deadline, maxiter=300):
    def fun(x):
        return smooth_max_obj_grad(x, n, alpha, penalty)[0]
    def fprime(x):
        return smooth_max_obj_grad(x, n, alpha, penalty)[1]

                                     bounds=[(0.0,1.0)]*n,
                                     maxiter=maxiter, pgtol=1e-14, iprint=0)
        return res
    except Exception:
        return h

def coord_descent(h, n, deadline, max_sweeps=200, K=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2*n-1) - (n-1)
    g = compute_g(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        idx_order = np.random.permutation(n)
        for i in idx_order:
            if time.time() > deadline:
                break
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            s = np.zeros(2*n-1)
            v1 = (i - dm >= 0) & (i - dm < n)
            s[v1] += (1.0 - h[i - dm[v1]])
            v2 = (i + dm >= 0) & (i + dm < n)
            s[v2] -= h[i + dm[v2]]
            K_ = min(K, 2*n-1)
            top = np.argsort(g)[::-1][:K_]
            cur_max = g.max()
            best_d = 0.0
            best_v = cur_max
            cands = [lo, hi, 0.0]
            for a in range(K_):
                for b in range(a+1, K_):
                    k1, k2 = top[a], top[b]
                    ds = s[k1] - s[k2]
                    if abs(ds) > 1e-14:
                        dv = (g[k2] - g[k1]) / ds
                        if lo < dv < hi:
                            cands.append(dv)
            for dv in cands:
                dv = float(np.clip(dv, lo, hi))
                v = (g + dv * s).max()
                if v < best_v - 1e-15:
                    best_v = v
                    best_d = dv
            if best_d != 0.0:
                h[i] += best_d
                g += best_d * s
                improved = True
        h = project_feasible(h, n)
        g = compute_g(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def subgrad_momentum(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    vel = np.zeros(n)
    mu = 0.9
    dm = np.arange(2*n-1) - (n-1)
    step = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        cur_max = g.max()
        K = min(15, 2*n-1)
        top = np.argsort(g)[::-1][:K]
        gaps = g[top] - g[top].min() + 1e-10
        w = gaps ** 6
        w /= w.sum()
        grad = np.zeros(n)
        for a in range(K):
            idx = top[a]
            wi = w[a]
            d = dm[idx]
            lo1 = max(0, d)
            hi1 = min(n, d + n)
            if lo1 < hi1:
                grad[lo1:hi1] += wi * (1.0 - h[lo1-d:hi1-d])
            lo2 = max(0, -d)
            hi2 = min(n, n-d)
            if lo2 < hi2:

        vel = mu * vel - step * grad
        h_new = project_feasible(h + vel, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.01, 1.0)
        else:
            step = max(step * 0.95, 0.001)
        h = h_new
    return best_h, best_c

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)

                else:
                    hb[z] = 0.0
                    hb[o] = 1.0
                if improved:
