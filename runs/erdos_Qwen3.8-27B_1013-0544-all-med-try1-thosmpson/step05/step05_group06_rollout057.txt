import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_descent(h, n, deadline, max_iter=20000):
    """Accelerated subgradient descent on active shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    K = 15
    step = 0.02
    momentum = np.zeros(n)
    beta = 0.9
    t_best = time.time()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        # Active shifts
        active = np.argsort(g)[::-1][:K]
        # Weight by proximity to max
        gaps = current_max - g[active]
        weights = np.exp(-gaps / 0.01)
        weights /= weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, s_val in zip(weights, active):
            d = int(s_val - (n - 1))
            if d == 0:
                grad += w * (1 - 2 * h)
            elif d > 0:
                grad[:n-d] += w * (1 - h[d:])
                grad[d:] -= w * h[:n-d]
            else:
                da = -d
                grad[da:] += w * (1 - h[:n-da])
                grad[:da] -= w * h[da:]
        
        # Momentum step
        momentum = beta * momentum - step * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new

            if it - (t_best and 0) > 100:  # simplified: just decay step

            if step < 0.001:
                step = 0.02
                momentum = np.zeros(n)
