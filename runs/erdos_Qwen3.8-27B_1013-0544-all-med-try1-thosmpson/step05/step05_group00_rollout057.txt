    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    best_c = cur_c

    t_end = 0.01

            break
        # Progress-based temperature
        progress = it / n_iters_budget
        T = t_start * (t_end / t_start) ** progress

            corr = new_corr
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)

def compute_c5_val(h, n):

    rng = np.random.RandomState(seed)

    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]

    for phase in range(n_phases):
        phase_deadline = time.time() + phase_time
        if phase > 0:
            # Perturb current best
            h = best_h.copy()
            n_perturb = max(5, n // 20)

        cur_max = corr.max()
        
        while it < max_it and time.time() < phase_deadline:
            it += 1
            # Temperature
            progress = min(it / max_it, 1.0)
            T = T0 * (Tend / T0) ** progress
            
            # Random swap
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]

                corr = new_corr
                # Update index sets
                zeros_idx = np.delete(zeros_idx, np.where(zeros_idx == i)[0][0])

                    best_h = h.copy()

                h[j] = 1.0

    dx = 2.0 / n
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    corr = np.correlate(h_bin, 1 - h_bin, mode='full').astype(np.float64)
    cur_max = corr.max()
    
    improved = True
    while improved and time.time() < deadline:

        best_delta = 1e-10
        best_i, best_j = -1, -1

            if time.time() > deadline:
                return h_bin, cur_max * dx
            for j in ones:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                new_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
                new_max = new_corr.max()
                delta = new_max - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h_bin[i] = 0.0
                h_bin[j] = 1.0
        
        if best_i >= 0:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
            corr = np.correlate(h_bin, 1 - h_bin, mode='full').astype(np.float64)
            cur_max = corr.max()
            ones.discard(best_j)
            ones.add(best_i)
            zeros.discard(best_i)
            zeros.add(best_j)
            improved = True
    
    return h_bin, cur_max * dx

import numpy as np
import time

def compute_c5_scalar(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def sa_single_swap(h, n, deadline, seed=42, T0=5.0, T1=0.01, max_iter=10**8):
    rng = np.random.RandomState(seed)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_c = cur_max
    best_h = h.copy()
    
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        progress = min(it / max_iter, 1.0)
        T = T0 * (T1 / T0) ** progress
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h[i] = 1.0
        h[j] = 0.0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = float(new_corr.max())
        
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-10)):
            cur_max = new_max
            corr = new_corr
            # Update sets
            mask_z = (zeros == i)
            zeros = np.delete(zeros, np.where(mask_z)[0][0])
            ones = np.append(ones, i)
            mask_o = (ones == j)
            # Careful: j might have been added if i==j (impossible since i was zero, j was one)
            idx_j = np.where(ones == j)[0]
            # j should still be in ones (we just added i)
            if len(idx_j) > 0:
                ones = np.delete(ones, idx_j[0])
            zeros = np.append(zeros, j)
            
            if cur_max < best_c:
                best_c = cur_max
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    
    return best_h, best_c * (2.0/n)

def greedy_local_search(h, n, deadline):
    dx = 2.0 / n
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    
    while time.time() < deadline:
        best_delta = 0.0
        best_i = -1
        best_j = -1
        
        for i in sorted(zeros):
            for j in sorted(ones):
                h[i] = 1.0
                h[j] = 0.0
                nc = float(np.correlate(h, 1 - h, mode='full').max())
                h[i] = 0.0
                h[j] = 1.0
                if nc < cur_max - 1e-10:
                    best_delta = nc - cur_max
                    best_i = i
                    best_j = j
                    break
            if best_i >= 0:
                break
        
        if best_i < 0:
            break
        
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
        cur_max = float(corr.max())
        ones.discard(best_j)
        ones.add(best_i)
        zeros.discard(best_i)
        zeros.add(best_j)
    
    return h, cur_max * dx

**Alternative approach: Frequency domain**

This is the cross-correlation of h and d. In the frequency domain:
C(f) = H(f) * conjugate(D(f))

where H and D are the DFTs of h and d. The max of c[k] is related to the L∞ norm of C.
