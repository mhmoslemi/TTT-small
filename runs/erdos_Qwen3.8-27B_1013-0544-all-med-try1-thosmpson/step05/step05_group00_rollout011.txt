import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def bcd_coordinate(h, j, corr, n, x_lo=0.0, x_hi=1.0, n_iter=50):
    """Solve the 1D subproblem for coordinate j using ternary search."""
    # alpha[ell] = 1 - h[j+ell] - h[j-ell] (with OOB handling)
    # beta[ell] = corr[ell] - h[j] * alpha[ell]
    
    # Compute alpha and beta
    ell = np.arange(-(n-1), n)  # lags
    # j+ell valid when 0 <= j+ell < n
    jp = j + ell
    jm = j - ell
    h_jp = np.where((jp >= 0) & (jp < n), h[jp[(jp >= 0) & (jp < n)]], 0.0)
    h_jm = np.where((jm >= 0) & (jm < n), h[jm[(jm >= 0) & (jm < n)]], 0.0)
    alpha = 1.0 - h_jp - h_jm
    beta = corr - h[j] * alpha
    
    # Minimize max_ell (beta[ell] + x * alpha[ell]) over x in [x_lo, x_hi]
    # Ternary search
    best_x = h[j]
    best_val = np.inf
    a, b = x_lo, x_hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        v1 = np.max(beta + m1 * alpha)
        v2 = np.max(beta + m2 * alpha)
        if v1 < v2:
            b = m2
        else:
            a = m1
        if v1 < best_val:
            best_val = v1
            best_x = m1
        if v2 < best_val:
            best_val = v2
            best_x = m2
    
    # Also check endpoints
    v0 = np.max(beta + x_lo * alpha)
    v1 = np.max(beta + x_hi * alpha)
    if v0 < best_val:
        best_val = v0
        best_x = x_lo
    if v1 < best_val:
        best_val = v1
        best_x = x_hi
    
    return best_x, best_val

    dx = 2.0 / n
    corr = compute_corr(h, n)
    cur_c = np.max(corr) * dx
    improved_total = 0.0
    
    for j in order:
        if deadline and time.time() > deadline:
            break
        new_x, new_val = bcd_coordinate(h, j, corr, n, n_iter=n_iter)
        if new_val * dx < cur_c - 1e-15:
            # Update correlation
            ell = np.arange(-(n-1), n)
            jp = j + ell
            jm = j - ell
            h_jp = np.where((jp >= 0) & (jp < n), h[jp[(jp >= 0) & (jp < n)]], 0.0)

import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def bcd_one_coord(h, j, corr, n, n_sub=60):
    """Solve 1D subproblem for coordinate j.
    Returns new value of h[j] that minimizes max_ell C[ell]."""
    ell = np.arange(-(n-1), n)
    jp = j + ell
    jm = j - ell
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    h_jp = np.zeros(2*n-1)
    h_jm = np.zeros(2*n-1)
    h_jp[mask_p] = h[jp[mask_p]]
    h_jm[mask_m] = h[jm[mask_m]]
    alpha = 1.0 - h_jp - h_jm  # slope coefficient for each ell
    beta = corr - h[j] * alpha  # intercept for each ell
    
    # Minimize max_ell (beta[ell] + x * alpha[ell]) over x in [0,1]
    # Using ternary search
    a, b = 0.0, 1.0
    best_x = h[j]
    best_v = np.max(beta + best_x * alpha)
    
    for _ in range(n_sub):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        v1 = np.max(beta + m1 * alpha)
        v2 = np.max(beta + m2 * alpha)
        if v1 < v2:
            b = m2
        else:
            a = m1
        if v1 < best_v:
            best_v = v1
            best_x = m1
        if v2 < best_v:
            best_v = v2
            best_x = m2
    
    # Check endpoints
    v0 = np.max(beta)
    v1 = np.max(beta + alpha)
    if v0 < best_v:
        best_v = v0
        best_x = 0.0
    if v1 < best_v:
        best_v = v1
        best_x = 1.0
    
    return best_x, best_v

    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        h_work = h.copy()  # Work on a copy, project at end
        corr = compute_corr(h_work, n)
        cur_max = np.max(corr)

            if new_max < cur_max - 1e-15:

                jm = idx - ell
                mask_p = (jp >= 0) & (jp < n)

                h_jm = np.zeros(2*n-1)

                alpha = 1.0 - h_jp - h_jm

                h_work[idx] = new_x
                cur_max = new_max

        h_projected = project_feasible(h_work, n)
        c = compute_c5(h_projected, n)
        if c < best_c:
            best_c = c
            best_h = h_projected.copy()
        h = h_projected.copy()
    
    return best_h, best_c

Let me also think about a different approach: **direct 1D optimization using the structure**.

The function f(x) = max_ell (beta[ell] + x * alpha[ell]) is convex and piecewise linear. Its minimum can be found exactly by:
1. Finding the upper envelope of the lines
2. The minimum is at the lowest point of the envelope

To find the upper envelope of L lines in 1D:
- Sort lines by slope: alpha[ell]
- Use a stack: for each line (in order of slope), add it to the stack if it's above the previous stack top at their intersection point
- O(L log L)

Then the minimum of the envelope is at one of the vertices (intersection points of consecutive lines on the envelope) or at x=0 or x=1.

Let me implement this:

```python
def find_min_piecewise_linear(alpha, beta, x_lo=0.0, x_hi=1.0):
    """Find x that minimizes max_ell (beta[ell] + x*alpha[ell]) for x in [x_lo, x_hi]."""
    L = len(alpha)
    # Sort by slope
    order = np.argsort(alpha)
    a = alpha[order]
    b = beta[order]
    
    # Build upper envelope using stack
    # Each element on stack: (x_start, line_index) where line_index refers to position in sorted array
    # The line is active from x_start to the next line's x_start
    stack_x = []  # x-coordinates where new lines become active
    stack_idx = []  # indices of lines (in sorted order)
    
    for i in range(L):
        # Find intersection with current top of stack
        while stack_idx:
            j = stack_idx[-1]
            if a[i] == a[j]:
                # Parallel lines, keep the one with higher intercept
                if b[i] > b[j]:
                    stack_idx.pop()
                    if stack_idx:
                        # Recheck with new top
                        continue
                    else:
                        break
                else:
                    break
            # Intersection x = (b[j] - b[i]) / (a[i] - a[j])
            x_int = (b[j] - b[i]) / (a[i] - a[j])
            if x_int >= stack_x[-1]:
                # Line i becomes active after line j's start, so remove j
                stack_idx.pop()
                stack_x.pop()
            else:
                break
        if not stack_idx:
            stack_x.append(x_lo)  # Active from x_lo (or -inf)
        else:
            j = stack_idx[-1]
            if a[i] != a[j]:
                x_int = (b[j] - b[i]) / (a[i] - a[j])
            else:
                x_int = x_lo  # shouldn't happen due to parallel handling above
            stack_x.append(max(x_int, x_lo))
        stack_idx.append(i)
    
    # Now find the minimum on the envelope
    # The envelope consists of segments, each from a line
    # Minimum is at: x_lo, x_hi, or at a breakpoint (stack_x[k] for k=1,...,len-1)
    candidates = [x_lo, x_hi] + [x for x in stack_x[1:] if x_lo < x < x_hi]
    
    best_x = x_lo
    best_v = np.inf
    for x in candidates:
        v = np.max(beta + x * alpha)
        if v < best_v:
            best_v = v
            best_x = x
    return best_x, best_v
```
