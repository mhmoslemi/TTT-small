def optimize_binary_swap(h, n, corr, active_set):
    """Find best swap (i: 0->1, j: 1->0) that reduces max corr."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, np.inf
    
    # Precompute g_j[k] for all j and k in active_set
    # g_j[k] = (1-2*h[k-j]) * [0<=k-j<n] * [k!=2j]
    
    # For efficiency, precompute g for all positions
    # g[p, k] = (1-2*h[k-p]) * [0<=k-p<n] * [k!=2p]
    # This is a (n x |active|) matrix
    
    g = np.zeros((n, len(active_set)))
    for idx, k in enumerate(active_set):
        # g[p, idx] = (1-2*h[k-p]) * [0<=k-p<n] * [k!=2p]
        # k-p is valid when 0 <= k-p < n, i.e., k-n+1 <= p <= k
        p_min = max(0, k - n + 1)
        p_max = min(n - 1, k)
        for p in range(p_min, p_max + 1):
            if k != 2 * p:
                g[p, idx] = 1 - 2 * h[k - p]
    
    # For each candidate swap (i in zeros, j in ones):
    # new_max = max over k in active of (corr[k] + g[i,idx] - g[j,idx])
    # We want to minimize this.
    
    # First, compute the "base" for each i:
    # M_i[idx] = corr[active[idx]] + g[i, idx]
    
    active_corr = corr[active_set]
    
    best_val = np.inf
    best_swap = None
    
    for i in zeros:
        M_i = active_corr + g[i]  # shape (|active|,)
        # For each j in ones: new_max = max(M_i - g[j])
        # We want min over j of max(M_i - g[j])
        
        # Vectorized: for all j at once
        D = M_i[None, :] - g[ones[:, None] if False else ones][:, None, :]  # This is getting messy
        # Actually: for each j, value = max(M_i - g[j])
        # = max over idx of (M_i[idx] - g[j, idx])
        
        # Let's do it differently: for each j, compute max(M_i - g[j])
        # This is a (|ones|, |active|) matrix of (M_i[None,:] - g[ones][None,:,:])
        # then take max over last axis, then min over first axis
        
        diff = M_i[None, :] - g[ones]  # shape (|ones|, |active|)
        new_maxes = diff.max(axis=1)  # shape (|ones|,)
        j_idx = new_maxes.argmin()
        j_val = new_maxes[j_idx]
        
        if j_val < best_val:
            best_val = j_val
            best_swap = (i, ones[j_idx])
    
    return best_swap, best_val

Let me also think about a completely different approach: **simulated annealing with smart moves**.

The idea:
1. Start from a good initial configuration
2. At each step, generate a random "smart" move (not just a single swap, but a pattern of swaps)
3. Accept/reject based on simulated annealing criterion
4. The "smart" moves focus on fixing the worst shifts

Types of smart moves:
- Single swap (as before)
- Block swap: swap a block of h values with another block
- Shift move: shift a segment of h
- Complementary move: for the worst shift k*, try to fix the misalignments

Actually, I think the most promising approach is:

1. **Subgradient phase** (fast, O(N log N) per iteration using the formula derived)
2. **Binary local search with smart move selection** (O(N²|A|) per iteration but finds best move)
3. **Periodic restarts at different resolutions**

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, top_k=5, lr=0.01):
    """One subgradient step for the minimax problem."""
    corr = np.correlate(h, 1 - h, mode='full')
    # Find top-k shifts
    top_idx = np.argsort(corr)[-top_k:]
    active = top_idx
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        # g[j] += (1 - 2*h[k-j]) * [0 <= k-j < n] * [k != 2*j]
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            if k != 2 * j:
                g[j] += (1 - 2 * h[k - j])
    g /= len(active)
    
    # Take step
    h_new = h - lr * g
    return project_feasible(h_new, n), corr

def fast_binary_local_search(h, n, deadline, max_iters=50000, top_k=5, seed=42):
    """Fast binary local search using smart move selection."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    
    # Ensure binary
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Find active set (top shifts)
        top_idx = np.argsort(corr)[-top_k:]
        active = top_idx
        active_corr = corr[active]
        
        # Compute g matrix: g[p, a] for p in [0,n), a in active
        g = np.zeros((n, len(active)))
        for a_idx, k in enumerate(active):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            valid_js = js[js != k / 2]  # exclude self-term
            if len(valid_js) > 0:
                g[valid_js, a_idx] = 1 - 2 * h[k - valid_js]
        
        # For each zero position i, compute M_i = active_corr + g[i]
        # For each one position j, new_max(i,j) = max(M_i - g[j])
        # Find pair (i,j) that minimizes new_max
        
        # Vectorized computation
        # M for all zeros: shape (|zeros|, |active|)
        M = active_corr[None, :] + g[zeros]  # (|zeros|, |active|)
        # D for all (i,j) pairs: M_i - g[j] for all i in zeros, j in ones
        # Shape: (|zeros|, |ones|, |active|)
        # This might be too large: 125 * 125 * 5 = 78125, which is fine
        
        D = M[:, None, :] - g[ones][None, :, :]  # (|zeros|, |ones|, |active|)
        new_maxes = D.max(axis=2)  # (|zeros|, |ones|)
        
        # Find best swap
        flat_idx = new_maxes.argmin()
        i_idx = flat_idx // len(ones)
        j_idx = flat_idx % len(ones)
        best_new_max = new_maxes[i_idx, j_idx]
        
        i = zeros[i_idx]
        j = ones[j_idx]
        
        if best_new_max < cur_c - 1e-15:
            # Accept the swap
            h[i], h[j] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            stall_count = 0
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            
            # Update ones and zeros
            zeros = np.delete(zeros, i_idx)
            ones = np.delete(ones, j_idx)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
        else:
            # No improvement found - try random swap or break
            stall_count += 1
            if stall_count > 50:
                # Try a random perturbation
                if len(ones) > 0 and len(zeros) > 0:
                    i = rng.choice(zeros)
                    j = rng.choice(ones)
                    h[i], h[j] = 1.0, 0.0
                    corr = np.correlate(h, 1 - h, mode='full').astype(float)
                    cur_c = corr.max() * dx
                    zeros = np.delete(zeros, np.where(zeros == i)[0])
                    ones = np.delete(ones, np.where(ones == j)[0])
                    zeros = np.append(zeros, j)
                    ones = np.append(ones, i)
                    stall_count = 0
                else:
                    break
            elif stall_count > 10:
                break
    
    return best_h, best_c

def fast_binary_search(h_start, n, deadline, top_k=5, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    g = build_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    stall = 0
    temp = 0.01  # annealing temperature
    
    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        active = top_idx
        active_corr = corr[active]
        g_act = g[:, active]
        
        M = active_corr[None, :] + g_act[zeros]
        D = M[:, None, :] - g_act[ones][None, :, :]
        new_maxes = D.max(axis=2)
        
        flat = new_maxes.argmin()
        ii = flat // len(ones)
        jj = flat % len(ones)
        best_val = new_maxes[ii, jj]
        i = zeros[ii]
        j = ones[jj]
        new_c = best_val * dx
        
        if new_c < cur_c - 1e-12:
            # Accept improving move
            g_i = g[i].copy()
            g_j = g[j].copy()
            corr = corr + g_i - g_j
            cur_c = corr.max() * dx
            
            p_idx = np.arange(n)
            mi = (p_idx != i) & (p_idx + i < 2*n - 1)
            g[p_idx[mi], p_idx[mi] + i] -= 2.0
            mj = (p_idx != j) & (p_idx + j < 2*n - 1)
            g[p_idx[mj], p_idx[mj] + j] += 2.0
            h[i], h[j] = 1.0, 0.0
            
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            stall = 0
            temp = max(0.001, temp * 0.999)
        else:
            # Try to accept a non-improving move (annealing)
            stall += 1
            if stall > 20:
                # Pick a random swap
                ri = rng.randint(len(zeros))
                rj = rng.randint(len(ones))
                ri_val = zeros[ri]
                rj_val = ones[rj]
                
                g_ri = g[ri_val].copy()
                g_rj = g[rj_val].copy()
                
                # Compute what the new max would be
                new_corr = corr + g_ri - g_rj
                new_c_rand = new_corr.max() * dx
                
                # Accept with probability exp(-(new_c - cur_c)/temp)
                if np.random.rand() < np.exp(-(new_c_rand - cur_c) / temp):
                    corr = new_corr
                    cur_c = new_c_rand
                    p_idx = np.arange(n)
                    mi = (p_idx != ri_val) & (p_idx + ri_val < 2*n - 1)
                    g[p_idx[mi], p_idx[mi] + ri_val] -= 2.0
                    mj = (p_idx != rj_val) & (p_idx + rj_val < 2*n - 1)
                    g[p_idx[mj], p_idx[mj] + rj_val] += 2.0
                    h[ri_val], h[rj_val] = 1.0, 0.0
                    
                    zeros = np.delete(zeros, ri)
                    ones = np.delete(ones, rj)
                    zeros = np.append(zeros, rj_val)
                    ones = np.append(ones, ri_val)
                
                temp = max(0.001, temp * 0.95)
                stall = 0
    
    return best_h, best_c

            if s < 1e-14:

    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Main binary search from parent (45% time)
    p1 = t0 + 0.45 * (budget_s - 5)
    hb, cb = fast_binary_search(h0, n, p1, top_k=10, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Phase 2: Alternative initializations (20% time)
    p2 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p2:
        h_tm = thue_morse_init(n, rng)
        rem2 = p2 - time.time() - 10
        if rem2 > 30:
            t_split = rem2 / 2.0
            hb, cb = fast_binary_search(h_tm, n, time.time() + t_split, top_k=10, seed=seed + 100)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
            if time.time() < p2 - 10:
                h_rand = np.zeros(n)
                h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
                hb, cb = fast_binary_search(h_rand, n, p2 - 5, top_k=10, seed=seed + 200)
                if cb < best_c:
                    best_c = cb
                    best_h = hb.copy()
    
    # Phase 3: SLSQP continuous polish (10% time)
    p3 = t0 + 0.75 * (budget_s - 5)
    if time.time() < p3:
        h_cont = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [100000, 1000000, 5000000]:
            if time.time() > p3 - 5:
                break
            h_cont = slsqp_smooth(h_cont, n, alpha, 1500, p3 - 3)
            c = compute_c5(h_cont, n)
            if c < best_c:
                best_c = c
                best_h = h_cont.copy()
    
    # Phase 4: Extended binary polish on best (remaining time)
    p4 = dl - 10
    if time.time() < p4 - 5:
        hb, cb = fast_binary_search(best_h, n, p4, top_k=10, seed=seed + 300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 5: Final SLSQP micro-polish
    if time.time() < dl - 3:
        hf = project_feasible(best_h + 0.01 * rng.randn(n), n)
        hf = slsqp_smooth(hf, n, 5000000, 1000, dl - 2)
        c = compute_c5(hf, n)
        if c < best_c:
            best_c = c
            best_h = hf.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
