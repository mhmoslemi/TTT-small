                        t1, t2 = top_idx[m1], top_idx[m2]
                        ds = s[t2] - s[t1]
                        if abs(ds) > 1e-15:
                            dc = (g[t1] - g[t2]) / ds
                            dc = float(np.clip(dc, d_lo, d_hi))
                            candidates.append(dc)
                for d in candidates:
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    improved = True
        h = project_feasible(h, n)
        c = float(np.max(compute_g(h, n)))
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    np.random.seed(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = float(np.max(compute_g(best_h, n)))
    # Phase 1: ALO
    p1 = t0 + 0.45 * (budget_s - 5)
    h, c = alternating_lp(best_h.copy(), n, p1, max_iter=2000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 2: Softmax GD
    p2 = t0 + 0.65 * (budget_s - 5)
    h, c = softmax_gd(best_h.copy(), n, p2, max_iter=10000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 3: Pair descent
    p3 = t0 + 0.85 * (budget_s - 5)
    h, c = pair_descent(best_h.copy(), n, p3, max_sweeps=500)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 4: ALO + pair polish
    if time.time() < dl - 15:
        h, c = alternating_lp(best_h.copy(), n, dl - 10, max_iter=200)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h, c = pair_descent(best_h.copy(), n, dl - 2, max_sweeps=100)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = float(np.max(compute_g(best_h, n)))
    return best_h, best_c, n
