import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    w = exp_sh / exp_sh.sum()
    val = m + (sm + np.log(exp_sh.sum())) / alpha
    # Gradient
    g = np.zeros(n)
    for midx in range(len(c)):
        wm = w[midx]
        if wm < 1e-12:
            continue
        k = midx - (n - 1)
        if k >= 0:
            idx1 = np.arange(0, n - k)
            g[idx1] += wm * (1.0 - x[idx1 + k])
            if k > 0:
                idx2 = np.arange(k, n)
                g[idx2] -= wm * x[idx2 - k]
        else:

            g[idx1] += wm * (1.0 - x[idx1 - ks])
            idx2 = np.arange(0, n - ks)
            g[idx2] -= wm * x[idx2 + ks]
    # Add constraint gradient
    val += 500.0 * (x.sum() - n / 2.0) ** 2
    g += 1000.0 * (x.sum() - n / 2.0)
    return val, g

    """L-BFGS with alpha annealing from alpha_start to alpha_end."""
    h = project_feasible(h.copy(), n)

    per_alpha = total_iter // n_alphas

        if time.time() > deadline:
            break
        def obj(x):
            v, _ = smooth_obj_and_grad(x, n, alpha)
            return v
        def grad(x):
            _, g = smooth_obj_and_grad(x, n, alpha)
            return g
        try:
            res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': per_alpha, 'ftol': 1e-16, 'gtol': 1e-10})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

    """Coordinate descent on the smooth objective."""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)

            if s < 1e-14:

    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)

    val += penalty * (x.sum() - n / 2.0) ** 2

    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    w = exp_sh / exp_sh.sum()
    g = np.zeros(n)
    for midx in range(len(c)):
        wm = w[midx]
        if wm < 1e-12:

            if k > 0:

            g[ks:] += wm * (1.0 - x[:n-ks])

    """One Frank-Wolfe step with the smooth objective."""
    _, g = smooth_grad(h, n, alpha, 0.0)
    # Oracle: binary vector minimizing <g, x> with sum(x) = n/2
    idx = np.argsort(g)
    x = np.zeros(n)
    x[idx[:n//2]] = 1.0
    return x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, penalty):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    val = m + (sm + np.log(exp_sh.sum())) / alpha
    val += penalty * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha, penalty):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    w = exp_sh / exp_sh.sum()
    g = np.zeros(n)
    for midx in range(len(c)):
        wm = w[midx]
        if wm < 1e-10:
            continue
        k = midx - (n - 1)
        if k >= 0:
            g[:n - k] += wm * (1.0 - x[k:n])
            if k > 0:
                g[k:] -= wm * x[:n - k]
        else:
            ks = -k
            g[ks:] += wm * (1.0 - x[:n - ks])
            g[:n - ks] -= wm * x[ks:]
    g += 2.0 * penalty * (x.sum() - n / 2.0)
    return g

def lbfgs_round(h, n, alpha, penalty, maxiter, deadline):
    def obj(x):
        return smooth_obj(x, n, alpha, penalty)
    def grad(x):
        return smooth_grad(x, n, alpha, penalty)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def frank_wolfe(h, n, alpha, deadline, max_iter=5000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(max_iter):
        if time.time() > deadline:
            break
        gamma = 2.0 / (t + 2)
        if gamma < 0.005:
            break
        _, g = smooth_grad(h, n, alpha, 0.0)
        idx = np.argsort(g)
        x = np.zeros(n)
        x[idx[:n // 2]] = 1.0
        h_new = project_feasible((1 - gamma) * h + gamma * x, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def greedy_cd(h, n, deadline, alpha=50000, max_rounds=200):
