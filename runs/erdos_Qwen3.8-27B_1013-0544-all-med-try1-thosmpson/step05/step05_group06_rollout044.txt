def optimal_ci(a, b, K_top=50):
    # a[m] = coefficient of h[i] in g(m)
    # b[m] = constant term
    # Find t in [0,1] minimizing max_m (a[m]*t + b[m])
    K = len(a)
    candidates = [0.0, 1.0]
    
    # Find line active at t=0 and t=1
    m0 = int(np.argmax(b))
    m1 = int(np.argmax(a + b))
    
    # Add intersection of m0 and m1
    if a[m0] != a[m1]:
        t = (b[m1] - b[m0]) / (a[m0] - a[m1])
        if 0 <= t <= 1:
            candidates.append(t)
    
    # Add intersections with top lines
    idx0 = np.argsort(b)[::-1][:K_top]
    idx1 = np.argsort(a + b)[::-1][:K_top]
    
    for i0 in idx0:
        for i1 in idx1:
            if a[i0] == a[i1]:
                continue
            t = (b[i1] - b[i0]) / (a[i0] - a[i1])
            if 0 <= t <= 1:
                candidates.append(t)
    
    best_t = candidates[0]
    best_val = float('inf')
    for t in candidates:
        val = float(np.max(a * t + b))
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, max_sweeps=10000):
    """Exact coordinate descent for the min-max problem."""
    K = 2 * n - 1
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    P = 30  # number of top lines to consider for intersection candidates
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            # Compute a_m and b_m for all shifts m (corresponding to correlate positions)
            # correlate position p corresponds to shift m = p - (n-1)
            a = np.zeros(K)
            b = np.zeros(K)
            
            hi = h[i]
            
            for p in range(K):
                m = p - (n - 1)  # shift
                if m == 0:
                    # g(0) = sum h[j](1-h[j]) = n/2 - sum h[j]^2
                    # As function of h[i]: n/2 - h[i]^2 - sum_{j!=i} h[j]^2
                    # This is handled separately
                    continue
                elif m > 0:
                    # g(m) = sum_{j=0}^{n-m-1} h[j] * (1-h[j+m]) * dx
                    # h[i] as first factor: j=i, need i < n-m → coeff (1-h[i+m])
                    if i < n - m:
                        a[p] += (1.0 - h[i + m])
                    # h[i] as second factor: j+m=i, j=i-m, need 0<=i-m and i-m < n-m → m<=i<n
                    if 0 <= i - m and i - m < n - m:
                        a[p] += (-h[i - m])
                        b[p] += h[i - m]
                else:
                    m2 = -m  # m2 > 0
                    # g(-m2) = sum_{j=0}^{n-m2-1} h[j+m2] * (1-h[j]) * dx
                    # h[i] as h[j+m2]: j=i-m2, need 0<=i-m2 and i-m2 < n-m2 → m2<=i<n
                    if 0 <= i - m2 and i - m2 < n - m2:
                        a[p] += (1.0 - h[i - m2])
                    # h[i] as h[j]: j=i, need 0<=i<n-m2
                    if 0 <= i and i < n - m2:
                        a[p] += (-h[i + m2])
                        b[p] += h[i + m2]
            
            # Now find optimal t in [0,1]
            # Objective: max(max_{m!=0} (a_m * t + b_m) * dx, (n/2 - t^2 - S) * dx)
            # where S = sum_{j!=i} h[j]^2
            S = (h * h).sum() - hi * hi
            
            # Candidates
            candidates = [0.0, 1.0]
            
            # Top lines by value at t=0.5
            vals_mid = a * 0.5 + b
            top_idx = np.argsort(vals_mid)[::-1][:P]
            
            for x in range(len(top_idx)):
                for y in range(x + 1, len(top_idx)):
                    i1, i2 = top_idx[x], top_idx[y]
                    da = a[i1] - a[i2]
                    if abs(da) > 1e-14:
                        t = (b[i2] - b[i1]) / da
                        if 0.0 <= t <= 1.0:
                            candidates.append(t)
            
            # Intersections with g(0) curve: n/2 - t^2 - S = a*t + b
            # t^2 + a*t + (b + S - n/2) = 0
            for x in range(len(top_idx)):
                i1 = top_idx[x]
                # t^2 + a[i1]*t + (b[i1] + S - n/2) = 0
                disc = a[i1]**2 - 4.0 * (b[i1] + S - n/2.0)
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-a[i1] + sqrt_disc) / 2.0
                    t2 = (-a[i1] - sqrt_disc) / 2.0
                    if 0.0 <= t1 <= 1.0:
                        candidates.append(t1)
                    if 0.0 <= t2 <= 1.0:
                        candidates.append(t2)
            
            # Also add t where derivative of g(0) equals slope of top line
            # g(0) derivative: -2*t; set equal to a[i1]: t = -a[i1]/2
            for x in range(min(5, len(top_idx))):
                i1 = top_idx[x]
                t = -a[i1] / 2.0
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
            
            # Evaluate
            def eval_obj(t):
                # g(m) for m != 0
                vals = a * t + b
                # g(0)
                g0 = n / 2.0 - t * t - S
                return max(float(np.max(vals)), float(g0))
            
            best_t = hi  # default: keep current value
            best_val = eval_obj(hi)
            
            for t in candidates:
                t = float(np.clip(t, 0.0, 1.0))
                val = eval_obj(t)
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t
            
            if best_t != hi:
                h[i] = best_t
                improved = True
        
        # Project to feasible
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if not improved and sweep > 5:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_ab(i, h, n):
    K = 2 * n - 1
    a = np.zeros(K)
    b = np.zeros(K)
    m_pos = np.arange(1, n)
    p_pos = n - 1 + m_pos
    v1 = m_pos < (n - i)
    if v1.any():
        a[p_pos[v1]] += (1.0 - h[i + m_pos[v1]])
    v2 = m_pos <= i
    if v2.any():
        a[p_pos[v2]] += (-h[i - m_pos[v2]])
        b[p_pos[v2]] += h[i - m_pos[v2]]
    m_neg = np.arange(1, n)
    p_neg = n - 1 - m_neg
    v3 = m_neg <= i
    if v3.any():
        a[p_neg[v3]] += (1.0 - h[i - m_neg[v3]])
    v4 = m_neg < (n - i)
    if v4.any():
        a[p_neg[v4]] += (-h[i + m_neg[v4]])
        b[p_neg[v4]] += h[i + m_neg[v4]]
    return a, b

def find_optimal_t(a, b, hi, n, S, P=40):
    K = len(a)
    candidates = [0.0, 1.0, hi]
    vals_mid = a * 0.5 + b
    top_idx = np.argsort(vals_mid)[::-1][:P]
    for x in range(len(top_idx)):
        for y in range(x + 1, len(top_idx)):
            i1, i2 = top_idx[x], top_idx[y]
            da = a[i1] - a[i2]
            if abs(da) > 1e-13:
                t = (b[i2] - b[i1]) / da
                if 0.0 < t < 1.0:
                    candidates.append(t)
    for x in range(min(15, len(top_idx))):
        i1 = top_idx[x]
        disc = a[i1]**2 - 4.0 * (b[i1] + S - n / 2.0)
        if disc >= 0:
            sd = np.sqrt(disc)
            t1 = (-a[i1] + sd) / 2.0
            t2 = (-a[i1] - sd) / 2.0
            if 0.0 < t1 < 1.0:
                candidates.append(t1)
            if 0.0 < t2 < 1.0:
                candidates.append(t2)
        t = -a[i1] / 2.0
        if 0.0 < t < 1.0:
            candidates.append(t)
    def eval_obj(t):
        vals = a * t + b
        g0 = n / 2.0 - t * t - S
        return max(float(np.max(vals)), float(g0))
    best_t = hi
    best_val = eval_obj(hi)

def coord_descent(h, n, deadline, P=40, noise_scale=0.0, max_sweeps=50000):
    K = 2 * n - 1
    best_h = h.copy()
    best_c = compute_c5(h, n)
    order = list(range(n))
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % 20 == 0:
            np.random.shuffle(order)
        sweep_improved = False
        for i in order:
            a, b = compute_ab(i, h, n)
            S = (h * h).sum() - h[i] * h[i]
            best_t = find_optimal_t(a, b, h[i], n, S, P)
            if best_t != h[i]:
                h[i] = best_t
                sweep_improved = True
        if noise_scale > 0:
            mask = np.random.rand(n) < 0.05
            h[mask] += np.random.randn(mask.sum()) * noise_scale
            h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if not sweep_improved and stall > 30:
            break
    return best_h, best_c

def pair_descent_fast(h, n, deadline, max_sweeps=200):
    K = 2 * n - 1
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        top_K = min(6, K)
        top_idx = np.argsort(g)[::-1][:top_K]
        for i in range(n):
            for j in range(i + 1, n):
                s1 = np.zeros(K)
                s2 = np.zeros(K)
                a1, b1 = compute_ab(i, h, n)
                a2, b2 = compute_ab(j, h, n)
                s = a1 - a2
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi, 0.0]
                for m1 in range(top_K):
                    for m2 in range(top_K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-13:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = g.max()
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-12:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    improved = True
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

def binary_polish(h, n, deadline, max_rounds=30):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1.0)[0].tolist())
    zeros = list(np.where(h_bin == 0.0)[0].tolist())
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for z in sorted(zeros):
            for o in sorted(ones):
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.discard(o)
                    ones.add(z)
                    zeros.remove(z)
                    zeros.append(o)
                    improved = True
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def random_binary_init(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = random_binary_init(n, rng)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Coordinate descent from initial (50% time)
    p1 = t0 + 0.45 * (budget_s - 10)
    h = best_h.copy()
    h, c = coord_descent(h, n, p1, P=50, noise_scale=0.0)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 2: Coordinate descent with noise from perturbed start (20% time)
    p2 = t0 + 0.65 * (budget_s - 10)
    if time.time() < p2:
        h_pert = best_h + rng.randn(n) * 0.1
        h_pert = project_feasible(h_pert, n)
        h, c = coord_descent(h_pert, n, p2, P=50, noise_scale=0.02)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 3: Try random binary init + coord descent (15% time)
    p3 = t0 + 0.80 * (budget_s - 10)
    if time.time() < p3:
        for attempt in range(3):
            if time.time() > p3:
                break
            h_rand = random_binary_init(n, rng)
            h, c = coord_descent(h_rand, n, p3, P=40, noise_scale=0.0)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    # Phase 4: Pair descent polish (10% time)
    p4 = t0 + 0.90 * (budget_s - 10)
    if time.time() < p4:
        h, c = pair_descent_fast(best_h.copy(), n, p4, max_sweeps=100)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 5: Binary polish (5% time)
    p5 = t0 + 0.95 * (budget_s - 10)
    if time.time() < p5:
        hb, cb = binary_polish(best_h.copy(), n, p5, max_rounds=20)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Final: one more coord descent pass from best
    if time.time() < dl:
        h, c = coord_descent(best_h.copy(), n, dl, P=60, noise_scale=0.0, max_sweeps=2000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
