    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def weighted_grad(h, w, n):
    dx = 2.0 / n
    W = w.sum()
    grad = np.full(n, W)
    for m in range(-(n - 1), n):
        idx = m + n - 1
        if 0 <= idx < 2 * n - 1:
            wm = w[idx]
            if wm == 0.0:
                continue
            j_lo1 = max(0, -m)
            j_hi1 = min(n - 1, n - 1 - m)
            if j_lo1 <= j_hi1:
                grad[j_lo1:j_hi1 + 1] -= wm * h[j_lo1 + m:j_hi1 + m + 1]
            j_lo2 = max(0, m)
            j_hi2 = min(n - 1, n - 1 + m)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2 + 1] -= wm * h[j_lo2 - m:j_hi2 - m + 1]
    grad *= dx
    return grad

def weighted_obj(h, w, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.dot(w, c) * dx)

def softmax_weights(f, beta):
    f_shift = f - f.max()
    w = np.exp(beta * f_shift)
    s = w.sum()
    if s < 1e-30:
        return np.ones_like(f) / len(f)
    return w / s

def grad_descent_reweight(h, n, deadline, beta_start=3.0, beta_end=80.0,
                          lr0=0.3, momentum=0.7, max_iter=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    vel = np.zeros(n)
    n_iter = max_iter
    stall_count = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (it / max_iter)
        f = compute_all_f(h, n)
        w = softmax_weights(f, beta)
        grad = weighted_grad(h, w, n)
        vel = momentum * vel - lr0 * grad
        h_new = h + vel
        h_new = project_feasible(h_new, n)
        wobj_new = weighted_obj(h_new, w, n)
        wobj_old = weighted_obj(h, w, n)
        if wobj_new > wobj_old:
            vel *= 0.5
            h_new = h - 0.3 * vel
            h_new = project_feasible(h_new, n)
            wobj_new = weighted_obj(h_new, w, n)
            if wobj_new > wobj_old:
                lr0 *= 0.9
                vel *= 0.3
            else:
                h = h_new
        else:
            h = h_new
            lr0 = min(lr0 * 1.02, 1.0)
        c = compute_c5(h, n)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                lr0 *= 0.95
        if it % 500 == 0 and it > 0:
            lr0 = max(lr0, 0.1)
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def build_cosine_init(n, mode, amplitude=0.5, phase=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amplitude * np.cos(np.pi * mode * x + phase)
    return project_feasible(h, n)

def multi_start_grad(h_init, n, deadline, rng):
    best_c = compute_c5(h_init, n)
    best_h = h_init.copy()
    alloc_each = 80
    n_starts = max(3, int((deadline - time.time()) / alloc_each))
    scales = [0.01, 0.03, 0.06, 0.1, 0.15]
    for s in range(n_starts):
        sub_deadline = min(time.time() + alloc_each, deadline)
        scale = scales[s % len(scales)]
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        h_opt, c_opt = grad_descent_reweight(hp, n, sub_deadline,
                                             beta_start=3.0, beta_end=60.0,
                                             lr0=0.3, momentum=0.7)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = build_cosine_init(n0, 3, 0.45)

    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)

    # Phase 1: Gradient descent with reweighting on initial n
    t_phase1_end = min(t0 + budget_s * 0.45, deadline - 60)
    best_h, best_c = grad_descent_reweight(best_h, n0, t_phase1_end,
                                           beta_start=5.0, beta_end=100.0,
                                           lr0=0.4, momentum=0.75)

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 50:
        for alpha in [2000, 10000, 100000, 1000000]:
            if time.time() > deadline - 40:
                break
            ht = slsqp_optimize(best_h, n0, alpha, maxiter=2000)
            c = compute_c5(ht, n0)
            if c < best_c - 1e-12:
                best_c = c
                best_h = ht

    # Phase 3: Try different n (interpolate and optimize)
    for n_try in [150, 120]:
        if n_try >= n0 or time.time() > deadline - 40:
            continue
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(60, deadline - time.time() - 30)
        if alloc < 20:
            break
        h_opt, c_opt = grad_descent_reweight(h_interp, n_try, time.time() + alloc,
                                             beta_start=5.0, beta_end=80.0,
                                             lr0=0.35, momentum=0.7)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
            n0 = n_try

    # Phase 4: Multi-start with gradient descent
    if time.time() < deadline - 50:
        n_cur = len(best_h)
        best_h, best_c = multi_start_grad(best_h, n_cur, deadline - 30, rng)

    # Phase 5: Try structured initializations at different n
    for n_struct in [200, 150]:
        if time.time() > deadline - 40:
            break
        for mode in [2, 3, 4, 5]:
            if time.time() > deadline - 25:
                break
            hs = build_cosine_init(n_struct, mode, 0.45)
            alloc = min(25, deadline - time.time() - 15)
            if alloc < 10:
                break
            h_opt, c_opt = grad_descent_reweight(hs, n_struct, time.time() + alloc,
                                                 beta_start=5.0, beta_end=80.0,
                                                 lr0=0.3, momentum=0.7)
            if c_opt < best_c:
                best_h = h_opt
                best_c = c_opt

    # Phase 6: Final SLSQP + gradient polish
    n_final = len(best_h)
    if time.time() < deadline - 20:
        ht = slsqp_optimize(best_h, n_final, 500000, maxiter=3000)
        c = compute_c5(ht, n_final)
        if c < best_c:
            best_c = c
            best_h = ht
        best_h, best_c = grad_descent_reweight(best_h, n_final, deadline - 10,
                                               beta_start=50.0, beta_end=200.0,
                                               lr0=0.2, momentum=0.5)

    # Phase 7: Local search
    if time.time() < deadline - 5:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        best_h, best_c = pair_search(best_h, n_final, deadline - 2, deltas, max_passes=100)

    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
