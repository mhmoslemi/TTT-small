            grad[lag:] -= w * dx * h[:i]
        else:
            to = -lag
            i = n - to
            grad[to:] += w * dx * (1.0 - h[:i])
            grad[:i] -= w * dx * h[to:]
    return grad

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_of_max(h, n, k):
    dx = 2.0 / n
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[:m] += dx * (1.0 - h[lag:])
        grad[lag:] -= dx * h[:m]
    else:
        to = -lag
        m = n - to
        grad[to:] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[to:]
    return grad

def nesterov_subgrad(h0, n, deadline, gamma=0.3, max_iter=3000000):
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        g = compute_all_g(y, n)
        k = int(np.argmax(g))
        grad = subgrad_of_max(y, n, k)
        lr = gamma / np.sqrt(t_new)
        h_new = project_feasible(y - lr * grad, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        h_prev = h
        h = h_new
        t = t_new
    return best_h, best_c

def mom_subgrad(h0, n, deadline, mu=0.85, gamma=0.5, max_iter=2000000):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t = it + 1
        g = compute_all_g(h, n)
        k = int(np.argmax(g))
        grad = subgrad_of_max(h, n, k)
        lr = gamma / np.sqrt(t)
        v = mu * v - lr * grad
        h = project_feasible(h + v, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def smooth_grad_topk(h, n, alpha, topk=8):
    g = compute_all_g(h, n)
    g_max = g.max()
    exps = np.exp(alpha * (g - g_max))
    nm1 = 2 * n - 1
    if topk >= nm1:
        idx = np.arange(nm1)
        w = exps / exps.sum()
    else:
        idx = np.argpartition(exps, -topk)[-topk:]
        w_top = exps[idx]
        s = w_top.sum()
        w = np.zeros(nm1)
        if s > 1e-15:
            w[idx] = w_top / s
        else:
            w[int(np.argmax(g))] = 1.0
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in idx:
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += wk * dx * (1.0 - h[lag:])
            grad[lag:] -= wk * dx * h[:m]
        else:
            to = -lag
            m = n - to
            grad[to:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[to:]
    return g_max, grad

def smooth_grad_descend(h0, n, deadline, alpha, max_iter=200000):
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, grad = smooth_grad_topk(y, n, alpha, topk=8)
        lr = 1.0 / (t_new + 1.0)
        h_new = project_feasible(y - lr * grad, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        t = t_new
    return best_h, best_c

def binary_local_search(h0, n, deadline, max_sweeps=100000, seed=123):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_iters = min(len(zones) * 3, len(zones) * len(ones))
        for _ in range(n_iters):
            if time.time() > deadline:
                break
            i = zeros[int(rng.randint(len(zones)))]
            j = ones[int(rng.randint(len(ones)))]
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            c = compute_c5(h_bin, n)
            if c < best_c - 1e-15:
                best_c = c
                idx2 = np.where(h_bin == 1)[0]
                ones = idx2
                zeros = np.where(h_bin == 0)[0]
                improved = True
            else:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
        if not improved:
            break
    return h_bin, best_c

def two_flip_search(h0, n, deadline, seed=77):
    target = n // 2
    h_bin = h0.copy()
    if not np.allclose(h_bin, np.round(h_bin)):
        idx = np.argsort(h_bin)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    rng = np.random.RandomState(seed)
    for it in range(500000):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i1 = zeros[int(rng.randint(len(zeros)))]
        i2 = zeros[int(rng.randint(len(zeros)))]
        j1 = ones[int(rng.randint(len(ones)))]
        j2 = ones[int(rng.randint(len(ones)))]
        if i1 == i2 or j1 == j2:
            continue
        h_bin[i1] = 1.0; h_bin[i2] = 1.0
        h_bin[j1] = 0.0; h_bin[j2] = 0.0
        c = compute_c5(h_bin, n)
        if c < best_c - 1e-15:
            best_c = c
        else:
            h_bin[i1] = 0.0; h_bin[i2] = 0.0
            h_bin[j1] = 1.0; h_bin[j2] = 1.0
    return h_bin, best_c

def simulated_annealing(h0, n, deadline, t0=0.1, t1=0.0001, seed=42):
    target = n // 2
    h_bin = h0.copy()
    if not np.allclose(h_bin, np.round(h_bin)):
        idx = np.argsort(h_bin)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
    cur_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    best_c = cur_c
    rng = np.random.RandomState(seed)
    max_iters = 1000000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t0 * (t1 / t0) ** (it / max_iters)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        a = zeros[int(rng.randint(len(zeros)))]
        b = ones[int(rng.randint(len(ones)))]
        h_bin[a] = 1.0
        h_bin[b] = 0.0
        new_c = compute_c5(h_bin, n)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[a] = 0.0
            h_bin[b] = 1.0
    return best_h, best_c

def interpolate_to_n(h, n_src, n_dst):
    if n_src == n_dst:
        return h.copy()
    xo = np.linspace(0, 2.0, n_src)
    xn = np.linspace(0, 2.0, n_dst)
    return project_feasible(np.interp(xn, xo, h, left=0, right=0), n_dst)
