        (3000000, 0.02, 0.0005, 0.3, 0.005),
        (4000000, 0.008, 0.0002, 0.15, 0.003),
        (3000000, 0.003, 0.0001, 0.08, 0.001),
    ]
    for n_iter, e_s, e_m, d_s, d_m in sa_rounds:
        if remaining() < 120:
            break
        best_h, best_c5 = fast_sa(best_h, n, t0, budget_s, rng, n_iter, e_s, e_m, d_s, d_m)
        if remaining() < 60:
            break
        ht = slsqp_opt(best_h, n, 100000, 3000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 3: Batch local search (greedy)
    if remaining() > 60:
        best_h, best_c5 = batch_local(best_h, n, t0, budget_s, rng, n_rounds=500)

    # Phase 4: More SA from refined point
    if remaining() > 120:
        best_h, best_c5 = fast_sa(best_h, n, t0, budget_s, rng, 2000000, 0.005, 0.0001, 0.1, 0.002)

    # Phase 5: SLSQP polish
    for alpha in [100000, 500000, 2000000]:
        if remaining() < 20:
            break
        ht = slsqp_opt(best_h, n, alpha, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 6: Try n=96
    if remaining() > 200:
        n2 = 96
        h2 = make_initial(n2, seed + 10)
        h2 = slsqp_opt(h2, n2, 5000, 3000, 1e-15)
        h2, c2 = fast_sa(h2, n2, t0, budget_s, rng, 1500000, 0.03, 0.001, 0.3, 0.005)
        h2 = slsqp_opt(h2, n2, 100000, 3000, 1e-16)
        c2 = compute_c5(h2, n2)
        if c2 < best_c5:
            best_h, best_c5, n = h2, c2, n2

    # Phase 7: Final batch local + SLSQP
    if remaining() > 30:
        best_h, best_c5 = batch_local(best_h, n, t0, budget_s, rng, n_rounds=200)
    if remaining() > 15:
        ht = slsqp_opt(best_h, n, 2000000, 8000, 1e-20)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
