    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_max(h, top_k, dx):
    """Subgradient of max correlation using top-k lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    j = np.arange(n)
    for k in top_idx:
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    return grad / top_k * dx

def projected_subgrad(h, max_iters, lr0, top_k, threshold_frac):
    """Projected subgradient with adaptive top_k and backtracking."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = project_sum(h.copy(), target)
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        corr = np.correlate(x, 1.0 - x, mode="full")
        max_corr = np.max(corr)
        # Adaptive top_k: include lags within threshold_frac of max
        active_mask = corr >= max_corr * (1.0 - threshold_frac)
        k_eff = max(3, int(active_mask.sum()))
        k_eff = min(k_eff, max(top_k, 3))
        
        g = subgrad_max(x, k_eff, dx)
        
        # Backtracking line search
        lr = lr0 * (0.95 ** (t // 100))
        for attempt in range(20):
            x_new = project_sum(x - lr * g, target)
            v_new = compute_c5(x_new)
            if v_new <= best_v + 1e-10:  # Accept if not much worse than best
                break
            lr *= 0.5
        else:
            lr = 0.01
            x_new = project_sum(x - lr * g, target)
            v_new = compute_c5(x_new)
        
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_v

def slsqp_smooth(h, beta, maxiter, top_k=None):
    """SLSQP with smooth max objective."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        if top_k is not None and top_k < len(corr):
            idx = np.argsort(corr)[-top_k:]
            c = corr[idx]
        else:
            c = corr
        m = np.max(c)
        e = np.exp(beta * (c - m))
        return (m + np.log(np.sum(e)) / beta) * dx
    
    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        if top_k is not None and top_k < len(corr):
            idx = np.argsort(corr)[-top_k:]
            c = corr[idx]
        else:
            c = corr
        m = np.max(c)
        e = np.exp(beta * (c - m))
        w = e / np.sum(e)
        grad = np.zeros(n)
        j = np.arange(n)
        if top_k is not None and top_k < len(corr):
            for i, k in enumerate(idx):
                wi = w[i]
                if wi < 1e-12: continue
                s = k - (n - 1)
                js = j - s; jp = j + s
                ms = (js >= 0) & (js < n)
                mp = (jp >= 0) & (jp < n)
                if ms.any(): grad[ms] += wi * (1.0 - x[js[ms]])
                if mp.any(): grad[mp] += wi * (-x[jp[mp]])
        else:
            for k in range(len(corr)):
                wk = w[k]
                if wk < 1e-12: continue
                s = k - (n - 1)
                js = j - s; jp = j + s
                ms = (js >= 0) & (js < n)
                mp = (jp >= 0) & (jp < n)
                if ms.any(): grad[ms] += wk * (1.0 - x[js[ms]])
                if mp.any(): grad[mp] += wk * (-x[jp[mp]])
        return grad * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_sum(res.x, target)
    except Exception:
        return project_sum(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_sum(h_new, n_new / 2.0)
