def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_x = np.exp(x_shift)
    weights = exp_x / np.sum(exp_x)
    return weights  # These are the softmax weights

def project_polytope(h, target_sum):
    """Project onto {0 <= h <= 1, sum(h) = target_sum}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Bisection on the Lagrange multiplier
    lo, hi = -target_sum, 1.0 - target_sum / n  # rough bounds
    # Actually: we need sum(clamp(h - t, 0, 1)) = target_sum
    # f(t) = sum(clamp(h - t, 0, 1)) is decreasing in t
    # We want f(t) = target_sum
    
    def f(t):
        return np.sum(np.clip(h - t, 0, 1))
    
    # Find range of t
    lo = h.min() - 1  # f(lo) >= n (all 1s)
    hi = h.max() + 1  # f(hi) <= 0 (all 0s)
    # Actually simpler:
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if f(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0, 1)

def compute_corr_and_grad(h):
    """Compute correlation and gradient of corr[k] w.r.t. h for all k."""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    # corr has length 2n-1, indexed from lag -(n-1) to lag (n-1)
    # corr[k + n - 1] corresponds to lag k
    
    # Gradient: for each lag, grad[lag, m] is the partial derivative
    # But we want sum over lags of weight[lag] * grad_lag[m]
    # For the smooth max, we compute:
    # grad_m = sum_k weight[k] * d(corr[k])/d(h[m])
    
    return corr

def corr_gradient(h, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h."""
    n = len(h)
    num_lags = 2 * n - 1
    grad = np.zeros(n)
    
    for k_idx in range(num_lags):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        lag = k_idx - (n - 1)
        # d(corr[lag])/d(h[m]) for each m
        # corr[lag] = sum_i h[i+lag]*(1-h[i]) for valid i
        # d/dh[m]: 
        #   if m-lag valid (0 <= m-lag < n): contributes (1-h[m-lag])
        #   if m+lag valid (0 <= m+lag < n): contributes -h[m+lag]
        if lag == 0:
            # d/dh[m] of h[m]*(1-h[m]) = 1 - 2*h[m]
            for m in range(n):
                grad[m] += w * (1.0 - 2.0 * h[m])
        else:
            for m in range(n):
                i1 = m - lag
                i2 = m + lag
                g = 0.0
                if 0 <= i1 < n:
                    g += (1.0 - h[i1])
                if 0 <= i2 < n:
                    g -= h[i2]
                grad[m] += w * g
    
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) / len(x)) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -500, 500)
    e = np.exp(s)
    return e / np.sum(e)

def project_polytope(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def smooth_grad(h, beta, dx):
    """Compute gradient of smooth_max(corr(h,1-h), beta) * dx w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, beta)
    # grad = convolve(1-h, w) - convolve(h, w_reversed), both sliced [n-1:2n-1]
    part_a = np.convolve(1.0 - h, w, mode='full')[n-1:2*n-1]
    part_b = np.convolve(h, w[::-1], mode='full')[n-1:2*n-1]
    return (part_a - part_b) * dx

def projected_gradient_descent(h_init, n_iter, betas, lr_base=0.01, dx=None):
    """Run projected gradient descent with increasing beta."""
    h = h_init.copy()
    n = len(h)
    target_sum = n / 2.0
    if dx is None:
        dx = 2.0 / n
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    it = 0
    for beta in betas:
        # Inner iterations for this beta
        n_inner = max(100, n_iter // len(betas))
        for i in range(n_inner):
            it += 1
            # Compute gradient
            g = smooth_grad(h, beta, dx)
            
            # Step size with decay
            lr = lr_base / (1.0 + 0.01 * it)
            
            # Backtracking line search
            for shrink in range(10):
                h_trial = project_polytope(h - lr * g, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < compute_c5(h):
                    break
                lr *= 0.5
            
            h = project_polytope(h - lr * g, target_sum)
            c5_curr = compute_c5(h)
            if c5_curr < best_c5:
                best_h = h.copy()
                best_c5 = c5_curr
    
    return best_h, best_c5
