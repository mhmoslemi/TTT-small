def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:

            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

Actually with top_k=1, the "smooth max" is just the max value itself, and the gradient is the gradient of the active term. This is actually what we want at the optimum!

So the strategy should be:
- Start with larger top_k and smaller beta (smoother landscape)
- Progressively reduce top_k and increase beta (sharper, more accurate)
- End with top_k=1 or 2 for the final polish

3. Optimize the single term corr[k*] using SLSQP (this is a smooth quadratic objective!)
4. Check if other terms exceeded corr[k*]
5. If yes, add them to the active set
6. Repeat

The advantage is that each SLSQP call has a smooth objective (a single quadratic form), which SLSQP handles very well.

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

# Helper functions (no closures)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        d2 = target_sum - h.sum()
        if abs(d2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] += d2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_vals_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    if top_k is not None and top_k < nc:
        idx = np.argsort(corr)[-top_k:]
        ct = corr[idx]
    else:
        idx = np.arange(nc)
        ct = corr
    m = ct.max()
    e = np.exp(beta * (ct - m))
    sw = e.sum()
    w = e / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for i in range(len(idx)):
        wk = w[i]
        if wk < 1e-12:
            continue
        s = idx[i] - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_solve(h0, beta, top_k, maxiter):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        v, _ = smooth_vals_grad(x, beta, top_k)
        return v
    
    def objg(x):
        _, g = smooth_vals_grad(x, beta, top_k)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def cfun(x):
        return np.array([x.sum() - target])
    
    def cjac(x):
        return np.ones((1, n))
    
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def make_periodic(n, periods, duty=0.5):
    h = np.zeros(n)
    points_per_period = n // periods
    on = int(points_per_period * duty)
    for p in range(periods):
        start = p * points_per_period
        h[start:start+on] = 1.0
    return project_feasible(h, n / 2.0)

def make_wave(n, f1, f2, a1, a2):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + a1 * np.cos(np.pi * f1 * x) + a2 * np.sin(np.pi * f2 * x)
    return project_feasible(h, n / 2.0)
