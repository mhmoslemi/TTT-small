        g = np.zeros(n)
        if d >= 0:
            g[d:] += (1.0 - h[:n-d])
            g[:n-d] -= h[d:]
        else:
            ad = -d
            g[:n-ad] += (1.0 - h[ad:])
            g[ad:] -= h[:n-ad]

    h_new = h - alpha * grad
    return project_h(h_new, target_sum)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def subgrad_one(h, idx, n):
    d = idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n-d])
        g[:n-d] -= h[d:]
    else:
        ad = -d
        g[:n-ad] += (1.0 - h[ad:])
        g[ad:] -= h[:n-ad]
    return g

def subgradient_step(h, target_sum, alpha, top_k=5):
    n = len(h)
    dx = 2.0 / n
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(full_corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_indices:
        grad += subgrad_one(h, idx, n) / len(top_indices)
    grad *= dx
    h_new = h - alpha * grad
    return project_h(h_new, target_sum)

def con_fun(x):
    return np.sum(x) - len(x) / 2.0

def con_jac(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    def obj_func(h_arr, beta):
        corr = np.correlate(h_arr, 1.0 - h_arr, mode="full")
        return smooth_max(corr, beta) * dx

    # Phase 1: Subgradient descent
    if tleft() > 150:
        h_cur = best_h.copy()
        alpha = 0.005
        for it in range(8000):
            if tleft() < 80:
                break
            h_new = subgradient_step(h_cur, target_sum, alpha, top_k=5)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                h_cur = h_new
            else:
                h_cur = 0.5 * h_cur + 0.5 * h_new
                h_cur = project_h(h_cur, target_sum)
            alpha = max(0.0005, alpha * 0.9995)

    # Phase 2: SLSQP with increasing beta
    for beta in [80, 200, 500, 1200, 3000]:
        if tleft() < 50:
            break
        try:
            res = minimize(obj_func, best_h, args=(beta,), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 8000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Phase 3: Free-coordinate SLSQP (reduced dimensionality)
    if tleft() > 100:
        h_work = best_h.copy()
        for round_num in range(10):
            if tleft() < 40:
                break
            free_mask = (h_work > 0.01) & (h_work < 0.99)
            free_idx = np.where(free_mask)[0]
            n_free = len(free_idx)
            if n_free < 4:
                break
            fixed_sum = h_work[~free_mask].sum()
            free_target = target_sum - fixed_sum

            def obj_reduced(x, fi=free_idx, fw=h_work, fdx=dx):
                hf = fw.copy()
                hf[fi] = x
                corr = np.correlate(hf, 1.0 - hf, mode="full")
                return smooth_max(corr, 2000) * fdx

            def con_reduced(x, ft=free_target):
                return np.sum(x) - ft

            x0 = h_work[free_idx].copy()
            try:
                res = minimize(obj_reduced, x0, method='SLSQP',
                               bounds=Bounds(np.zeros(n_free), np.ones(n_free)),
                               constraints=[{'type': 'eq', 'fun': con_reduced}],
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_work[free_idx] = np.clip(res.x, 0, 1)
                h_work = project_h(h_work, target_sum)

                if c5_w < best_c5:

            except Exception:
                pass

    # Phase 4: Targeted SA focusing on worst lags

        c5_cur = best_c5
        T = 0.006
        cool = 0.9992
        no_imp = 0
        for it in range(100000):

                n_adj = np.random.randint(2, 5)

                        else:
                            h_new[i + d] -= 0.02 + 0.04 * np.random.random()
                elif d < 0:
                    ad = -d
                    idx = np.random.choice(n - ad, n_adj, replace=False)
                    for i in idx:
                        if h_cur[i + ad] < 0.5:
                            h_new[i + ad] += 0.02 + 0.04 * np.random.random()
                        else:
                            h_new[i] -= 0.02 + 0.04 * np.random.random()
                else:
                    idx = np.random.choice(n, n_adj, replace=False)
                    for i in idx:
                        h_new[i] += 0.06 if h_cur[i] > 0.5 else -0.06
            elif r < 0.60:
                # Gaussian on subset
                n_pts = np.random.randint(3, 12)
                idx = np.random.choice(n, n_pts, replace=False)
                h_new[idx] += np.random.randn(n_pts) * 0.04
            elif r < 0.80:
                # Block swap
                blen = np.random.randint(2, min(8, n // 8))
                i = np.random.randint(0, n - blen)
                j = np.random.randint(0, n - blen)
                if abs(j - i) >= blen:
                    tmp = h_new[i:i+blen].copy()
                    h_new[i:i+blen] = h_new[j:j+blen]
                    h_new[j:j+blen] = tmp
            else:
                # Pair exchange
                i, j = np.random.choice(n, 2, replace=False)
                h_new[i], h_new[j] = h_new[j], h_new[i]

            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(150.0, delta / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:

                else:
                    no_imp += 1
            else:
                no_imp += 1
            T = max(0.0001, T * cool)
            if no_imp > 400:
                T = max(T, 0.004)
                no_imp = 0

    # Phase 5: Multi-restart SLSQP from perturbed best
    if tleft() > 40:
        for attempt in range(8):
            if tleft() < 20:
                break
            h0 = best_h.copy()
            n_p = np.random.randint(3, 10)
            idx = np.random.choice(n, n_p, replace=False)
            h0[idx] += np.random.randn(n_p) * 0.03
            h0 = project_h(h0, target_sum)
            beta = np.random.choice([1000, 2000, 5000])
            try:
                res = minimize(obj_func, h0, args=(beta,), method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
