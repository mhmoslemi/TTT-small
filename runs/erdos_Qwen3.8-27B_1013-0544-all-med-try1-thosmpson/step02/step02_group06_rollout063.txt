from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def _obj_flat(h, n, target_sum, beta, lam, gamma):
    """Objective with smooth max + variance penalty."""
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
    sm = smooth_max(c, beta)
    var_pen = gamma * np.var(c)
    sum_pen = lam * (h.sum() - target_sum) ** 2
    return sm + var_pen + sum_pen

def _obj_pure(h, n, target_sum, beta, lam):
    """Pure smooth max objective."""
    c = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(c, beta) * (2.0/n)
    sum_pen = lam * (h.sum() - target_sum) ** 2
    return sm + sum_pen

    else:
        func = lambda h: _obj_pure(h, n, target_sum, beta, lam)
    try:
        r = minimize(func, h0, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_box_sum(r.x, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def _obj(h, n, ts, beta, lam, gamma):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    sm = smooth_max(c, beta) * dx
    if gamma > 0:
        var_pen = gamma * np.var(c) * dx
    else:
        var_pen = 0.0
    sum_pen = lam * (h.sum() - ts) ** 2
    return sm + var_pen + sum_pen

def _optimize_stage(h0, n, ts, beta, lam, gamma, maxiter):
    bnds = [(0.0, 1.0)] * n
    def func(h):
        return _obj(h, n, ts, beta, lam, gamma)
    try:
        r = minimize(func, h0, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_box_sum(r.x, ts)
    except Exception:
        return h0

def _gen_block(n, n_blocks, seed):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def _softmax_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(shifted)
    w /= w.sum()
    return w


def _objective(h, n, ts, beta, lam, gamma):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    sm = _smooth_max_val(corr, beta) * dx
    gp = gamma * np.var(corr) * dx if gamma > 0 else 0.0
    sp = lam * (h.sum() - ts) ** 2
    return sm + gp + sp


def _gradient(h, n, ts, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = _softmax_weights(corr, beta)
    dx = 2.0 / n
    A = np.zeros(n)
    B = np.zeros(n)
    idx = np.arange(n)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        if ws < 1e-15:
            continue
        va = (idx + s) >= 0
        va &= (idx + s) < n
        A[va] += ws * (1.0 - h[idx[va] + s])
        vb = (idx - s) >= 0
        vb &= (idx - s) < n
        B[vb] += ws * h[idx[vb] - s]
    grad = dx * (A - B) + 2.0 * lam * (h.sum() - ts)
    return grad


def _opt_stage(h0, n, ts, beta, lam, gamma, maxiter):
    bnds = [(0.0, 1.0)] * n

    def fun(h):
        return _objective(h, n, ts, beta, lam, gamma)

    def jac(h):
        return _gradient(h, n, ts, beta, lam)

    try:
        r = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_box_sum(r.x, ts)
    except Exception:
        return h0

    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_box_sum(h, n / 2.0)


def _gen_two_tone(n, f1, a1, p1, f2, a2, p2):

    h = 0.5 + a1 * np.sin(np.pi * f1 * x + p1) + a2 * np.sin(np.pi * f2 * x + p2)
    return project_box_sum(h, n / 2.0)


def _gen_block(n, nblocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    sizes = np.zeros(nblocks, dtype=int)
    rem = n
    for i in range(nblocks - 1):
        sizes[i] = max(1, rem // (nblocks - i))
        rem -= sizes[i]
    sizes[-1] = rem
    pos = 0
    for b in range(nblocks):
        h[pos:pos + sizes[b]] = rng.uniform(0.0, 1.0)
        pos += sizes[b]
    return project_box_sum(h, n / 2.0)


def _try_rounding(h, ts):
    n = len(h)
    best = h.copy()
    bc = compute_c5(h)
    dist = np.minimum(h, 1.0 - h)
    order = np.argsort(dist)
    for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        k = int(n * frac)
        h_new = h.copy()
        for i in range(k):
            h_new[order[i]] = 0.0 if h[order[i]] < 0.5 else 1.0
        h_new = project_box_sum(h_new, ts)
        c = compute_c5(h_new)
        if c < bc:
            best, bc = h_new.copy(), c
    return best, bc

        ns = rng.randint(4, max(5, n // 5))
        for _ in range(ns):
            i, j = rng.choice(n, 2, replace=False)
            h[i], h[j] = h[j], h[i]
        h += rng.randn(n) * 0.005
    return project_box_sum(h, ts)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_p = project_box_sum(h_init, n_init / 2.0)
        c = compute_c5(h_p)
        if c < best_c5:
            best_h, best_c5, best_n = h_p, c, n_init

    n_values = [50, 60, 70, 80, 100]
    if best_n > 0 and best_n not in n_values:
        n_values.insert(0, best_n)

    rng = np.random.RandomState(seed)

    # Phase 1: Quick scan over n values
    for n in n_values:
        if tl() < 150:
            break
        ts = n / 2.0
        starts = []
        if best_n == n and best_h is not None:
            starts.append(best_h.copy())
        for f in [1, 2, 3]:
            for a in [0.35, 0.45]:
                starts.append(_gen_sin(n, f, a, 0.0))
                starts.append(_gen_sin(n, f, a, np.pi / 2))
        starts.append(_gen_two_tone(n, 1.5, 0.35, 0.3, 4.0, 0.1, 1.1))
        starts.append(_gen_block(n, 5, seed + n))
        h_r = rng.uniform(0, 1, n)
        starts.append(project_box_sum(h_r, ts))

        for h0 in starts[:8]:
            if tl() < 20:
                break
            h_cur = h0.copy()
            for beta, lam, gamma, mi in [(100, 100, 0.01, 200), (500, 300, 0, 300), (1500, 800, 0, 300)]:
                if tl() < 8:
                    break
                h_new = _opt_stage(h_cur, n, ts, beta, lam, gamma, mi)
                if compute_c5(h_new) < compute_c5(h_cur):
                    h_cur = h_new
            if compute_c5(h_cur) < best_c5:
                best_c5 = compute_c5(h_cur)
                best_h = h_cur.copy()
                best_n = n

    # Phase 2: Focused optimization on best few n
    n_focus = sorted(set([best_n] + [v for v in n_values if abs(v - best_n) <= 20]))
    for n in n_focus:
        if tl() < 200:
            break
        ts = n / 2.0
        starts = []
        if best_n == n and best_h is not None:
            starts.append(best_h.copy())
        for f in [1, 2, 3, 4]:
            for a in [0.3, 0.38, 0.42, 0.46, 0.49]:
                for p in [0.0, np.pi / 4, np.pi / 2, 3 * np.pi / 4]:
                    starts.append(_gen_sin(n, f, a, p))
        for _ in range(4):
            starts.append(_gen_two_tone(n, rng.uniform(0.8, 2.5), rng.uniform(0.25, 0.45),
                                        rng.uniform(0, 6.28), rng.uniform(2, 6), rng.uniform(0.05, 0.15),
                                        rng.uniform(0, 6.28)))
        for nb in [4, 6, 8, 10]:
            starts.append(_gen_block(n, nb, seed + n * 7 + nb))

        for si, h0 in enumerate(starts[:20]):
            if tl() < 25:
                break
            if tl() < 12 and si > 5:
                break
            h_cur = h0.copy()
            c_cur = compute_c5(h_cur)
            for beta, lam, gamma, mi in [(80, 80, 0.02, 200), (200, 150, 0.005, 250),
                                         (500, 300, 0, 300), (1000, 500, 0, 300), (2000, 1000, 0, 250)]:
                if tl() < 8:
                    break
                h_new = _opt_stage(h_cur, n, ts, beta, lam, gamma, mi)
                c_new = compute_c5(h_new)
                if c_new < c_cur:
                    h_cur = h_new
                    c_cur = c_new
            if c_cur < best_c5:
                best_c5 = c_cur
                best_h = h_cur.copy()
                best_n = n

    # Phase 3: Diverse restarts from best
    n = best_n
    ts = n / 2.0
    it = 0
    while tl() > 80:
        it += 1
        h = _perturb(best_h, n, ts, it % 5, rng)
        h_cur = h.copy()
        c_cur = compute_c5(h_cur)
        for beta, lam, mi in [(500, 300, 250), (1000, 500, 250), (2000, 800, 200)]:
            if tl() < 10:
                break
            h_new = _opt_stage(h_cur, n, ts, beta, lam, 0, mi)
            c_new = compute_c5(h_new)
            if c_new < c_cur:
                h_cur = h_new
                c_cur = c_new
        if c_cur < best_c5:
            best_c5 = c_cur
            best_h = h_cur.copy()

    # Phase 4: Rounding + re-optimize
    if tl() > 40:
        h_r, c_r = _try_rounding(best_h, ts)
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r
            for beta, lam, mi in [(500, 300, 300), (1000, 500, 300), (2000, 1000, 200)]:
                if tl() < 12:
                    break
                h_new = _opt_stage(best_h, n, ts, beta, lam, 0, mi)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h = c, h_new.copy()

    # Phase 5: Final high-precision polish
    if tl() > 25:
        for beta, lam, mi in [(2000, 1000, 500), (4000, 2000, 400), (8000, 4000, 300)]:
            if tl() < 10:
                break
            h_new = _opt_stage(best_h, n, ts, beta, lam, 0, mi)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_n = 70
    else:
        best_h = project_box_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
