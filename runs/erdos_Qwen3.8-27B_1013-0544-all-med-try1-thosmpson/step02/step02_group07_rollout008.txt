    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break

        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

            break

        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)
