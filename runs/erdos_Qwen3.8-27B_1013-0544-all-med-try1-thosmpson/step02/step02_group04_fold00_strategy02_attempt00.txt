    corr = correlate(h, np.ones_like(h), mode="full")
    
    # Temporal smoothing
    window_size = min(10, len(corr)//2)
    if window_size > 0:
        from scipy.signal import savgol_filter
        corr = savgol_filter(corr, window_size, 2)
    
    # Adjust for autocorrelation energy vs overlap
    shift_energy = []
    for lag in range(corr.shape[0]//2+1):
        if lag == 0:
            energy = 0.1  # Avoid self-overlap penalty
        else:
            energy = corr[lag] / (1 + norm_val**1.5 / 1000)
        shift_energy.append(energy)
    
    # Return the mean overlap (analytical-like)
    return np.min(shift_energy)  # SMOOTHING the max operation

def objective_slepian(h, target_sum):
    """Core objective function using spectral formulation"""
    n = len(h)
    dx_val = 2.0 / n
    
    # Scale for integral constraint
    scaled = h * (target_sum / np.sum(h) + 1e-6) if np.sum(h) != 0 else h
    
    # Compute autocorrelation via spectral filtering
    if n < 5:  # Small case default handling
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
    else:
        # Use frequency domain for large arrays
        autocorr = np.correlate(scaled, np.ones_like(scaled), mode="full")
        
        # Apply spectral window to mimic Slepian properties
        kernel = np.exp(-0.1 * (np.arange(len(autocorr)) - len(autocorr)//2)**2)
        autocorr = np.min(autocorr / kernel if np.max(kernel)>0 else autocorr)
        max_corr = np.max(autocorr) * dx_val
    
    # Smooth max objective: constructs a smooth envelope
    if len(max_corr.shape) > 0:  # Deal with vector output from correlate()
        max_corr = max_corr[-1]
    
    # Finely-tuned cost function
    return smooth_max(max_corr, beta=5.0)
