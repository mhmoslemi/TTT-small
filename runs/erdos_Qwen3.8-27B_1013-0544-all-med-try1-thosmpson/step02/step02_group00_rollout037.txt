def compute_subgradient(h, top_k_indices):
    n = len(h)
    grad = np.zeros(n)
    for lag_idx in top_k_indices:
        lag = int(lag_idx) - (n - 1)
        if lag >= 0:
            # V = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # dV/dh[j]:
            #   if 0 <= j < n-lag: (1-h[j+lag])
            #   if lag <= j < n: -h[j-lag]
            #   Note: if lag <= j < n-lag, both terms apply
            if lag == 0:
                # V = sum h[i]^2 ... wait no
                # V = sum_{i=0}^{n-1} h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
                # dV/dh[j] = 1 - 2*h[j]

            else:

                grad[:n-lag] += 1.0 - h[lag:]
                # j in [lag, n): grad[j] -= h[j-lag]
                grad[lag:] -= h[:n-lag]
        else:
            d = -lag
            # V = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])

            grad[d:] += 1.0 - h[:n-d]
            grad[:n-d] -= h[d:]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, top_k_indices):
    n = len(h)
    grad = np.zeros(n)
    for lag_idx in top_k_indices:
        lag = int(lag_idx) - (n - 1)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n - lag] += 1.0 - h[lag:]
            grad[lag:] -= h[:n - lag]
        else:
            d = -lag
            grad[d:] += 1.0 - h[:n - d]
            grad[:n - d] -= h[d:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 128
        h_parent = np.full(n_base, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base
    
    # Resolutions to try
    n_list = sorted(set([n_base, 128, 160, 200]))
    n_list = [n for n in n_list if 64 <= n <= 320]
    
    def resample(h_src, n_new):
        n_src = len(h_src)
        x_old = np.linspace(0, 2, n_src)
        x_new = np.linspace(0, 2, n_new)
        return project_h(np.interp(x_new, x_old, h_src), n_new / 2.0)
    
    def subgradient_search(h, n_pts, max_iters=5000, base_step=0.05):
        """Vectorized subgradient with momentum."""
        target = n_pts / 2.0
        cur = h.copy()
        cur_c5 = compute_c5(cur)
        vel = np.zeros(n_pts)
        momentum = 0.8
        
        for it in range(max_iters):
            if time_left() < 3:
                break
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            # Adaptive number of top lags
            top_vals = np.sort(corr)[::-1]
            # Use top 5 if they're close, else top 3
            if top_vals[2] > 0.95 * top_vals[0]:
                k = 5
            else:
                k = 3
            top_k = np.argsort(corr)[::-1][:k]
            grad = compute_subgradient(cur, top_k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad /= gnorm
            
            # Heavy ball
            vel = momentum * vel - base_step / (1.0 + 0.005 * it) * grad
            h_trial = project_h(cur + vel, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5:
                cur = h_trial
                cur_c5 = c5_trial
            else:
                vel *= 0.5
                # Try smaller step
                h_trial = project_h(cur - 0.01 * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5:
                    cur = h_trial
                    cur_c5 = c5_trial
        
        return cur, cur_c5
    
    def slsqp_refine(h, n_pts, betas=[50, 100, 200, 400, 800], maxiter=300):
        """SLSQP with smooth max."""
        target = n_pts / 2.0
        dx = 2.0 / n_pts
        cur = h.copy()
        cur_c5 = compute_c5(cur)
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

                break
            
            def obj(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, beta) * dx
            
            def con(x_in):
                return np.sum(x_in) - target
            
            def con_jac(x_in):
                return np.ones(x_in.shape)
            
            constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
            
            try:
                res = minimize(obj, cur, method='SLSQP', bounds=bounds,
                              constraints=constraints,
                              options={'maxiter': maxiter, 'ftol': 1e-12})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    cur = h_new
                    cur_c5 = c5_new
            except Exception:
                pass
        
        return cur, cur_c5
    
    # Main loop over resolutions
    for n_pts in n_list:
        if time_left() < 30:
            break
        
        target = n_pts / 2.0
        
        # Generate initializations
        inits = []
        
        # From parent
        inits.append(resample(h_parent, n_pts))
        
        # From current best
        if len(best_h) != n_pts:
            inits.append(resample(best_h, n_pts))
        else:
            inits.append(best_h.copy())
        
        # Structured: alternating blocks
        block_size = max(2, n_pts // 16)
        for offset in [0, 1]:
            h_block = np.zeros(n_pts)
            for b in range(n_pts // block_size + 1):
                start = b * block_size
                end = min(start + block_size, n_pts)
                h_block[start:end] = 0.7 if (b + offset) % 2 == 0 else 0.3
            inits.append(project_h(h_block, target))
        
        # Sinusoidal
        for freq in [1, 2, 3]:
            for amp in [0.2, 0.3, 0.4]:
                s = 0.5 + amp * np.sin(2 * np.pi * freq * np.arange(n_pts) / n_pts)
                inits.append(project_h(s, target))
        
        # Sawtooth
        ramp = np.linspace(0, 1, n_pts)
        inits.append(project_h(ramp, target))
        inits.append(project_h(1.0 - ramp, target))
        
        # Random smooth
        for _ in range(3):
            rough = np.random.uniform(0, 1, n_pts)
            # Smooth with moving average
            kernel = np.ones(5) / 5
            smooth = np.convolve(rough, kernel, mode='same')
            inits.append(project_h(smooth, target))
        
        # Deduplicate (keep unique ones)
        unique_inits = [inits[0]]
        for init in inits[1:]:
            if all(np.linalg.norm(init - u) > 0.1 for u in unique_inits):
                unique_inits.append(init)
        
        for h_init in unique_inits:
            if time_left() < 15:
                break
            
            cur_c5 = compute_c5(h_init)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = h_init.copy(), cur_c5, n_pts
            
            # SLSQP refinement

            if cur_c5 < best_c5:

            # Subgradient search
            cur, cur_c5 = subgradient_search(cur, n_pts, max_iters=3000, base_step=0.04)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
            
            # One more SLSQP with higher beta

                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
    
    # Final polish
    if time_left() > 5:
        n_final = len(best_h)
        cur, cur_c5 = slsqp_refine(best_h, n_final, betas=[800, 1200, 2000], maxiter=2000)
        if cur_c5 < best_c5:

        if cur_c5 < best_c5:
            best_h, best_c5 = cur, cur_c5
    
    return best_h, best_c5, len(best_h)

Also, I want to add **coordinate descent**: pick a single index j and try to optimize h[j] while keeping everything else fixed. Given all other values, the objective as a function of h[j] is:

Actually, for a fixed lag k, the contribution of h[j] to V_k is:
- If j is the "first" index: h[j]*(1-h[j+k]) → linear in h[j] with coefficient (1-h[j+k])
- If j is the "second" index: h[j-k]*(1-h[j]) → linear in h[j] with coefficient -h[j-k] (well, (1-h[j]) is linear in h[j])

Wait: V_k = sum_i h[i](1-h[i+k])
The term involving h[j]:
- When i=j: h[j](1-h[j+k])
- When i+k=j, i.e., i=j-k: h[j-k](1-h[j])

So V_k as a function of h[j] is:
h[j](1-h[j+k]) + h[j-k](1-h[j]) = h[j] - h[j]h[j+k] + h[j-k] - h[j-k]h[j]
= h[j](1 - h[j+k] - h[j-k]) + h[j-k]

This is linear in h[j]! So the maximum over k of these linear functions is a convex piecewise-linear function of h[j]. The minimum of this over [0,1] can be found by checking the breakpoints.

This gives us an exact coordinate descent step! For each coordinate j, we can compute the optimal value exactly.

```python
def coordinate_descent_step(h, j, top_k_indices):
    """Find optimal h[j] given top_k lags."""
    n = len(h)
    # For each active lag, compute the coefficient of h[j] in V_k
    # and the constant term
    coeffs = []
    consts = []
    for lag_idx in top_k_indices:
        lag = int(lag_idx) - (n - 1)
        # V_lag = sum_i h[i](1-h[i+lag])
        # Coefficient of h[j]:
        #   If j is first index (i=j, need j+lag in [0,n-1]): coeff += (1-h[j+lag])
        #   If j is second index (i+lag=j, i=j-lag, need j-lag in [0,n-1]): coeff += -h[j-lag]
        #   (because (1-h[j]) contributes -h[j] * h[j-lag])
        # Constant term (independent of h[j]):
        #   If j is second index: + h[j-lag]
        
        coeff = 0.0
        const = 0.0
        if 0 <= j + lag < n:
            coeff += (1.0 - h[j + lag])
        if 0 <= j - lag < n and lag != 0:
            coeff += (-h[j - lag])
            const += h[j - lag]
        elif lag == 0:
            # V_0 = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
            # The h[j] term is h[j] - h[j]^2, which is NOT linear!
            # So for lag=0, we can't use this trick directly
            pass
        coeffs.append(coeff)
        consts.append(const)
    
    # If lag 0 is active, we have a quadratic term. Handle separately.
    # For simplicity, if lag 0 is in top_k, skip coordinate descent for that lag.
    # f(h[j]) = max_k (coeff_k * h[j] + const_k) for non-zero lags
    # Plus for lag 0: h[j] - h[j]^2 (if present)
    
    # For now, handle only non-zero lags (linear case)
    non_zero = [(c, v) for c, v in zip(coeffs, consts) if c != 0 or v != 0]
    if not non_zero:
        return h[j]  # h[j] doesn't appear in any active lag
    
    # f(t) = max_k (c_k * t + d_k) for t in [0,1]
    # This is a convex piecewise-linear function
    # Minimum is at a breakpoint or at 0 or 1
    # Breakpoints: c_i * t + d_i = c_j * t + d_j → t = (d_j - d_i)/(c_i - c_j)
    
    candidates = [0.0, 1.0]
    for i in range(len(non_zero)):
        for j in range(i+1, len(non_zero)):
            c1, d1 = non_zero[i]
            c2, d2 = non_zero[j]
            if abs(c1 - c2) > 1e-15:
                t = (d2 - d1) / (c1 - c2)
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    
    best_t = h[j]
    best_val = float('inf')
    for t in candidates:
        val = max(c * t + d for c, d in non_zero)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t
```

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, top_k_indices):
    n = len(h)
    grad = np.zeros(n)
    for lag_idx in top_k_indices:
        lag = int(lag_idx) - (n - 1)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n - lag] += 1.0 - h[lag:]
            grad[lag:] -= h[:n - lag]
        else:
            d = -lag
            grad[d:] += 1.0 - h[:n - d]
            grad[:n - d] -= h[d:]
    return grad

def resample(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    x_old = np.linspace(0, 2, n_src)
    x_new = np.linspace(0, 2, n_new)

    def subgradient_search(h_init, n_pts, max_iters=5000, base_step=0.04, momentum=0.7):
        target = n_pts / 2.0
        cur = h_init.copy()
        cur_c5 = compute_c5(cur)
        vel = np.zeros(n_pts)

            corr = np.correlate(cur, 1.0 - cur, mode="full")
            top_vals = np.sort(corr)[::-1]
            k = 5 if (len(top_vals) > 5 and top_vals[4] > 0.92 * top_vals[0]) else 3
            top_k = np.argsort(corr)[::-1][:k]
            grad = compute_subgradient(cur, top_k)
            gnorm = np.linalg.norm(grad)
