        # Alternating
        alt = np.where(np.arange(n_points) % 2 == 0, 0.7, 0.3)
        inits.append(project_h(alt, target))
        # From best (resampled)
        if len(best_h) != n_points:
            x_b = np.linspace(0, 2, len(best_h))
            x_n = np.linspace(0, 2, n_points)
            inits.append(project_h(np.interp(x_n, x_b, best_h), target))

        for h_init in inits:
            if time_left() < phase1_budget * 0.3 + 20:
                break
            cur = h_init.copy()
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_points

            # CD rounds
            cur, cur_c5 = paired_cd(cur, 200, dx, n_ternary=30, rng=rng)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_points

    # Phase 2: Focused optimization on best resolution (45% of time)
    phase2_budget = budget_s * 0.45
    phase2_start = time.time()
    n_opt = best_n
    target = n_opt / 2.0
    dx = 2.0 / n_opt
    cur = best_h.copy()

    # If best is at a different resolution, work at that resolution
    if n_opt == best_n:
        cur = best_h.copy()
    else:
        x_b = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, n_opt)
        cur = project_h(np.interp(x_n, x_b, best_h), n_opt / 2.0)

    # Extended CD with basin hopping
    hop_count = 0
    while time.time() - phase2_start < phase2_budget and time_left() > 60:
        # CD pass
        cur, cur_c5 = paired_cd(cur, 500, dx, n_ternary=30, rng=rng)
        if cur_c5 < best_c5 - 1e-15:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_opt

        # Occasional basin hop
        hop_count += 1
        if hop_count % 3 == 0:
            cur_hop, c5_hop = basin_hop(cur, 5, dx, rng, noise_scale=0.08 + 0.05 * rng.random())
            if c5_hop < best_c5 - 1e-15:
                best_h, best_c5, best_n = cur_hop.copy(), c5_hop, n_opt
                cur = cur_hop.copy()

    # Phase 3: SLSQP polish (remaining time)
    if time_left() > 30:
        polish_budget = min(time_left() - 10, budget_s * 0.15)
        best_h, best_c5 = slsqp_polish(best_h, polish_budget, maxiter=5000)

    # Final CD polish
    if time_left() > 10:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        best_h, best_c5 = paired_cd(best_h, 1000, dx_f, n_ternary=35, rng=rng)

    return best_h, best_c5, len(best_h)
