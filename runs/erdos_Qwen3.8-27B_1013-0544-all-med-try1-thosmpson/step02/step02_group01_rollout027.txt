import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    shifts = np.arange(len(corr)) - (n - 1)
    for idx, s in enumerate(shifts):
        wk = w[idx]
        if wk < 1e-10:
            continue
        # dc/dh[j] = (1-h[j-s]) * [valid] - h[j+s] * [valid]
        js = np.arange(n) - s
        jp = np.arange(n) + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):

        return v
    
    def objg(x):

        return g

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    shifts = np.arange(len(corr)) - (n - 1)
    for idx, s in enumerate(shifts):
        wk = w[idx]
        if wk < 1e-10:
            continue
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] -= wk * h[jp[mp]]
    grad *= dx
    return val, grad
