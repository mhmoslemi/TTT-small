def _compute_AB(h, ones, zeros, n):
    """Precompute A (effect of 1->0) and B (effect of 0->1) for all positions."""
    k = np.arange(-(n-1), n)
    size = 2*n - 1
    
    A = np.zeros((len(ones), size))  # for each one position
    for idx, i in enumerate(ones):
        m1 = (k >= -i) & (k <= n-1-i)
        A[idx, m1] += h[i + k[m1]] - 1.0
        m2 = (k >= i-n+1) & (k <= i)
        A[idx, m2] += h[i - k[m2]]
    
    B = np.zeros((len(zeros), size))  # for each zero position
    for idx, j in enumerate(zeros):
        m3 = (k >= -j) & (k <= n-1-j)
        B[idx, m3] += 1.0 - h[j + k[m3]]
        m4 = (k >= j-n+1) & (k <= j)
        B[idx, m4] -= h[j - k[m4]]
    
    return A, B

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def _objective(h, n, target_sum, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2

def _make_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _compute_AB(h, ones, zeros, n):
    k = np.arange(-(n - 1), n)
    size = 2 * n - 1
    A = np.zeros((len(ones), size))
    for idx, i in enumerate(ones):
        m1 = (k >= -i) & (k <= n - 1 - i)
        A[idx, m1] += h[i + k[m1]] - 1.0
        m2 = (k >= i - n + 1) & (k <= i)
        A[idx, m2] += h[i - k[m2]]
    B = np.zeros((len(zeros), size))
    for idx, j in enumerate(zeros):
        m3 = (k >= -j) & (k <= n - 1 - j)
        B[idx, m3] += 1.0 - h[j + k[m3]]
        m4 = (k >= j - n + 1) & (k <= j)
        B[idx, m4] -= h[j - k[m4]]
    return A, B

def _single_swap_delta(h, i, j, n):
    k = np.arange(-(n - 1), n)
    dc = np.zeros(2 * n - 1)
    m1 = (k >= -i) & (k <= n - 1 - i)
    dc[m1] += h[i + k[m1]] - 1.0

    dc[m3] += 1.0 - h[j + k[m3]]
    m4 = (k >= j - n + 1) & (k <= j)
    dc[m4] -= h[j - k[m4]]
    return dc

        A, B = _compute_AB(h, ones, zeros, n)
        term = corr[None, None, :] + A[:, None, :] + B[None, :, :]
        new_maxes = np.max(term, axis=2)

def sa_binary(h0, n_iter=30000, time_limit=8.0, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        T = max(0.005, 0.8 * (0.9995 ** it))
        ones = np.where(h == 1.0)[0]

        j = zeros[rng.randint(len(zeros))]
        dc = _single_swap_delta(h, i, j, n)

        delta = new_max - cur_max

        ts = n_f / 2.0
        bnds = [(0.0, 1.0)] * n_f
        for beta, lam in [(2000, 3000), (5000, 5000)]:
            if time.time() - t0 > budget_s - 10:
                break
            try:
                r = minimize(_objective, overall_best_h, args=(n_f, ts, beta, lam),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 5000, 'ftol': 1e-18})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < overall_best_c5:
                    overall_best_c5 = c
                    overall_best_h = h_new
            except Exception:
                pass

    overall_best_h = project_box_sum(overall_best_h, len(overall_best_h) / 2.0)
    overall_best_c5 = compute_c5(overall_best_h)
    return overall_best_h, overall_best_c5, len(overall_best_h)
