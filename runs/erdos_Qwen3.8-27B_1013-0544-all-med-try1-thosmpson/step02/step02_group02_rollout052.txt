    """Optimize coordinate i using precomputed base/slope + ternary search."""
    target = n / 2.0
    # Compute current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    old_v = h[i]
    
    # Compute slope for each lag
    # corr[lag_index] where lag = lag_index - (n-1)
    lags = np.arange(-(n-1), n)
    slope = np.zeros(2*n - 1)
    for lag_idx, lag in enumerate(lags):
        s = 0.0
        j = i + lag
        if 0 <= j < n:
            s += (1.0 - h[j])
        j = i - lag
        if 0 <= j < n:
            s -= h[j]
        slope[lag_idx] = s
    
    base = corr - old_v * slope
    sum_others = h.sum() - old_v
    
    def obj_at(v):
        vals = base + v * slope
        c = np.max(vals)
        s = sum_others + v
        return c + mu * (s - target) ** 2
    
    # Ternary search
    a, b = 0.0, 1.0
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if obj_at(m1) < obj_at(m2):
            b = m2
        else:
            a = m1
    opt_v = (a + b) / 2.0
    h[i] = opt_v

        else:
            total = h.sum()
            if total < 1e-14:
                break
            h = np.clip(h * (target_sum / total), 0.0, 1.0)

def pair_step(h, n, i, j, n_iter):
    """Optimize h[i] and h[j] jointly (maintaining their sum)."""
    old_i, old_j = h[i], h[j]
    s = old_i + old_j
    if s < 1e-12 or s > 2.0 - 1e-12:
        return
    v_min = max(0.0, s - 1.0)
    v_max = min(1.0, s)
    if v_max - v_min < 1e-12:
        return
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(-(n-1), n)
    nlags = 2 * n - 1
    
    j1_i = i + lags
    j2_i = i - lags
    j1_j = j + lags
    j2_j = j - lags
    
    v1i = (j1_i >= 0) & (j1_i < n)
    v2i = (j2_i >= 0) & (j2_i < n)
    v1j = (j1_j >= 0) & (j1_j < n)
    v2j = (j2_j >= 0) & (j2_j < n)
    
    si = np.zeros(nlags)
    sj = np.zeros(nlags)
    si[v1i] += 1.0 - h[np.clip(j1_i[v1i], 0, n-1)]
    si[v2i] -= h[np.clip(j2_i[v2i], 0, n-1)]
    sj[v1j] += 1.0 - h[np.clip(j1_j[v1j], 0, n-1)]
    sj[v2j] -= h[np.clip(j2_j[v2j], 0, n-1)]
    
    base = corr - old_i * si - old_j * sj
    sl = si - sj
    
    a, b = v_min, v_max
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(base + m1 * sl) < np.max(base + m2 * sl):
            b = m2
        else:
            a = m1
    v = (a + b) / 2.0
    h[i] = v
    h[j] = s - v
