import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(h, beta):
    """Value and gradient of smooth max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = m + np.log(sw) / beta
    val *= dx
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        # d(corr_s)/d(h[j]) = (1-h[j+s])*I[j+s valid] - h[j-s]*I[j-s valid]
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:n-s]
        else:
            ns = -s
            if n - ns > 0:
                grad[ns:] += wk * (1.0 - h[:n-ns])
                grad[:n-ns] -= wk * h[ns:]
    grad *= dx
    return val, grad

def slsqp_adaptive(h, betas, maxiter_per):
    """SLSQP with adaptive beta continuation"""
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    x = h.copy()
    for beta in betas:
        def obj(xx):
            v, _ = smooth_max(xx, beta)
            return v
        def objg(xx):
            _, g = smooth_max(xx, beta)
            return g
        cons = [{'type': 'eq', 'fun': lambda xx: xx.sum() - target,
                 'jac': lambda xx: np.ones(n)}]
        try:
            res = minimize(obj, x, jac=objg, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per, 'ftol': 1e-16})
            x = res.x
        except:
            pass
        x = project_h(x, target)
    return x

def nesterov(h, iters, lr, top_k):
    """Nesterov accelerated subgradient"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    v = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(iters):
        if t > 0:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - v)
        else:
            y = x.copy()
        y = project_h(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                if n - s > 0:
                    grad[:n-s] += (1.0 - y[s:])
                    grad[s:] -= y[:n-s]
            else:
                ns = -s
                if n - ns > 0:
                    grad[ns:] += (1.0 - y[:n-ns])
                    grad[:n-ns] -= y[ns:]
        grad = grad / top_k * dx
        v = x.copy()
        x = project_h(y - lr * grad, target)
        cv = compute_c5(x)
        if cv < best_v:
            best_v = cv
            best_x = x.copy()
    return best_x, best_v

def interp_resample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_h(h_new, n_new / 2.0)

def coordinate_descent(h, max_rounds=5):
    """One-coordinate-at-a-time minimization"""
    n = len(h)
    target = n / 2.0
    x = h.copy()
    best_v = compute_c5(x)
    for rnd in range(max_rounds):
        for i in range(n):
            # Try a few values around x[i]
            candidates = np.linspace(0, 1, 21)
            best_cand = x[i]
            best_local = best_v
            for c in candidates:
                if abs(c - x[i]) < 1e-10:
                    continue
                x[i] = c
                x = project_h(x, target)
                v = compute_c5(x)
                if v < best_local:
                    best_local = v
                    best_cand = c
            x[i] = best_cand
            x = project_h(x, target)
            if best_local < best_v:
                best_v = best_local
                break  # restart
        if best_v >= compute_c5(x):
            break
    return x, best_v

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:ns]
        else:
            ns = -s
            rem = n - ns
            if rem > 0:
                grad[ns:] += wk * (1.0 - h[:rem])
                grad[:rem] -= wk * h[ns:]
    grad *= dx
    return val, grad

def slsqp_continue(h, betas, maxiter_per):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    x = h.copy()
    for beta in betas:
        def obj(xx):
            return smooth_max(xx, beta)[0]
        def objg(xx):
            return smooth_max(xx, beta)[1]
        cons = [{'type': 'eq', 'fun': lambda xx: xx.sum() - target,
                 'jac': lambda xx: np.ones(n)}]
        try:
            res = minimize(obj, x, jac=objg, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per, 'ftol': 1e-18})
            x = res.x
        except:
            pass
        x = project_h(x, target)
    return x

def nesterov(h, iters, lr, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    v = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(iters):
        if t > 0:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - v)
        else:
            y = x.copy()
        y = project_h(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                ns = n - s
                if ns > 0:
                    grad[:ns] += (1.0 - y[s:])
                    grad[s:] -= y[:ns]
            else:
                ns = -s
                rem = n - ns
                if rem > 0:
                    grad[ns:] += (1.0 - y[:rem])
                    grad[:rem] -= y[ns:]
        grad = grad / top_k * dx
        v = x.copy()
        x = project_h(y - lr * grad, target)
        cv = compute_c5(x)
        if cv < best_v:
            best_v = cv
            best_x = x.copy()
    return best_x, best_v

def interp_resample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_h(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_g_target = 0.0
_g_n = 0

def _con_fun(xx):
    return xx.sum() - _g_target

def _con_jac(xx):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[s:] += wk * (1.0 - h[:ns])
                grad[:ns] -= wk * h[s:]
        else:
            t = -s
            rem = n - t
            if rem > 0:
                grad[t:] += wk * (1.0 - h[:rem])
                grad[:rem] -= wk * h[t:]
    grad *= dx
    return val, grad

def nesterov(h, iters, lr, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    v = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(iters):
        if t > 0:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - v)
        else:
            y = x.copy()
        y = project_h(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                ns = n - s
                if ns > 0:
                    grad[s:] += (1.0 - y[:ns])
                    grad[:ns] -= y[s:]
            else:
                t2 = -s
                rem2 = n - t2
                if rem2 > 0:
                    grad[t2:] += (1.0 - y[:rem2])
                    grad[:rem2] -= y[t2:]
        grad = grad / top_k * dx
        v = x.copy()
        x = project_h(y - lr * grad, target)
        cv = compute_c5(x)
        if cv < best_v:
            best_v = cv
            best_x = x.copy()
    return best_x, best_v

def slsqp_continue(h, betas, maxiter_per):
    global _g_target, _g_n
    n = len(h)
    _g_n = n
    _g_target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    x = h.copy()
    for beta in betas:
        def obj(xx):
            return smooth_max(xx, beta)[0]
        def objg(xx):
            return smooth_max(xx, beta)[1]
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(obj, x, jac=objg, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per, 'ftol': 1e-18})
            x = res.x
        except Exception:
            pass
        x = project_h(x, _g_target)
    return x

def interp_resample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_h(h_new, n_new / 2.0)
