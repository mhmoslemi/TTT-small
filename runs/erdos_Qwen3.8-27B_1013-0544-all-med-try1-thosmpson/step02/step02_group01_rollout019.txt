import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.interpolate import interp1d
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] = h[free] + diff2 / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    # For each lag k in corr (index 0 to 2n-2), lag = k - (n-1)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        # corr[k] = sum_i h[i] * (1 - h[i+s]) over valid i
        # d(corr[k])/dh[i] = (1 - h[i+s]) for valid (i, i+s)
        #                    + (-h[i-s]) for valid (i-s, i) i.e. when i-s is valid and (i-s)+s=i
        # First term: i valid and i+s valid => 0<=i<n, 0<=i+s<n => max(0,-s) <= i <= min(n-1, n-1-s)
        # Second term: i-s valid and i valid => 0<=i-s<n, 0<=i<n => max(0,s) <= i <= min(n-1, n-1+s)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo1 < hi1:
            idx1 = np.arange(lo1, hi1)
            grad[idx1] += wk * (1.0 - h[idx1 + s])
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] += wk * (-h[idx2 - s])
    grad *= dx
    return val, grad

def lbfgs_smoothmax(h, maxiter, beta, target):
    n = len(h)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [LinearConstraint(np.ones((1, n)), target, target)]
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def projected_adagrad(h, max_iters, lr0, beta, target, momentum=0.9):
    n = len(h)
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    G = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for t in range(max_iters):
        lr = lr0 / np.sqrt(1 + t * 0.01)
        val, grad = smoothmax_val_grad(x, beta)
        G += grad ** 2
        v = momentum * v - lr * grad / np.sqrt(G + 1e-8)
        x_new = project_feasible(x + v, target)
        cv = compute_c5(x_new)
        if cv < best_v:
            best_v = cv
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    return np.correlate(h, 1.0 - h, mode="full")

    n = len(h)
    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += diff2 / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    idx = np.argsort(corr)[-k:]
    c_top = corr[idx]

        s = idx[ki] - (n - 1)

        hi1 = min(n, n - s)
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo1 < hi1:

            grad[i1] += wk * (1.0 - h[i1 + s])
        if lo2 < hi2:
            i2 = np.arange(lo2, hi2)
            grad[i2] += wk * (-h[i2 - s])
    grad *= dx
    return val, grad

    n = len(h)

        return topk_smoothmax_val_grad(x, topk, beta)[0]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

    best_x = x.copy()

    v = np.zeros(n)

        cv = compute_c5(x_new)

            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.interpolate import interp1d
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += diff2 / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        lo1, hi1 = max(0, -s), min(n, n - s)
        lo2, hi2 = max(0, s), min(n, n + s)
        if lo1 < hi1:
            i1 = np.arange(lo1, hi1)
            grad[i1] += wk * (1.0 - h[i1 + s])
        if lo2 < hi2:
            i2 = np.arange(lo2, hi2)
            grad[i2] += wk * (-h[i2 - s])
    grad *= dx
    return val, grad


def topk_smoothmax_val_grad(h, k, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    w_top = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for ki in range(k):
        wk = w_top[ki]
        s = idx[ki] - (n - 1)
        lo1, hi1 = max(0, -s), min(n, n - s)
        lo2, hi2 = max(0, s), min(n, n + s)
        if lo1 < hi1:
            i1 = np.arange(lo1, hi1)
            grad[i1] += wk * (1.0 - h[i1 + s])
        if lo2 < hi2:
            i2 = np.arange(lo2, hi2)
            grad[i2] += wk * (-h[i2 - s])
    grad *= dx
    return val, grad


def lbfgs_optimize(h, target, maxiter, beta, topk=None):
    n = len(h)

    def obj(x):
        if topk is not None:
            return topk_smoothmax_val_grad(x, topk, beta)[0]
        return smoothmax_val_grad(x, beta)[0]

    def objg(x):
        if topk is not None:
            return topk_smoothmax_val_grad(x, topk, beta)[1]
        return smoothmax_val_grad(x, beta)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    A = np.ones((1, n))
    cons = [LinearConstraint(A, target, target)]
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-9, 'maxcor': 50, 'maxls': 50})
        out = project_feasible(res.x, target)
        if not np.all(np.isfinite(out)):
            return project_feasible(h, target)
        return out
    except Exception:
        return project_feasible(h, target)


def momentum_gd(h, target, max_iters, beta, topk, lr0=0.03, decay=0.9998, mu=0.9):
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    v = np.zeros(x.shape)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        val, grad = topk_smoothmax_val_grad(x, topk, beta)
        v = mu * v - lr * grad
        x = project_feasible(x + v, target)
        cv = compute_c5(x)
        if cv < best_v:
            best_v = cv
            best_x = x.copy()
    return best_x, best_v
