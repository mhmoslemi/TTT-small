from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() == 0:
            # Need to adjust boundary elements
            if d > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
        else:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def compute_weighted_gradient(h, n, weights):
    """
    weights: array of length 2n-1, indexed by lag offset from 0
    lag k = index - (n-1)
    Returns gradient of sum_k weights[k] * corr[k] w.r.t. h
    """
    grad = np.zeros(n)
    for idx in range(2*n-1):
        k = idx - (n-1)
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        # ∂corr/∂h[i] = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
        # For i in [0, n-1-k] when k>0: (1-h[i+k])
        # For i in [k, n-1] when k>0: -h[i-k]
        if k >= 0:
            valid_i = range(n - k)
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            k_abs = -k
            valid_i = range(k_abs, n)
            grad[k_abs:] += w * (1.0 - h[:n-k_abs])
            grad[:n-k_abs] -= w * h[k_abs:]
    return grad

    ts = n / 2.0
    h = h.copy()
    step_size = step_size_init
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Exponential weights
        shifted = corr - np.max(corr)
        weights = np.exp(beta * shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:

        weights /= w_sum
        
        # Compute gradient
        grad = compute_weighted_gradient(h, n, weights)
        
        # Projected gradient step with backtracking

        current_weighted = np.dot(weights, corr)

            new_weighted = np.dot(weights, corr_new)
            if new_weighted < current_weighted:

                break
        else:

    return h

            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            ka = -k
            grad[ka:] += w * (1.0 - h[:n - ka])
            grad[:n - ka] -= w * h[ka:]
    return grad


def exp_weights_opt(h, n, beta, n_iters, step_init):
    ts = n / 2.0
    h = h.copy()
    step = step_init
    momentum = np.zeros(n)
    mu = 0.85
    corr = np.correlate(h, 1.0 - h, mode="full")

    for it in range(n_iters):
        shifted = corr - np.max(corr)
        weights = np.exp(beta * shifted)
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= ws

        grad = compute_weighted_grad(h, n, weights)
        momentum = mu * momentum + grad

        current = np.dot(weights, corr)
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_try = h - step * factor * momentum
            h_try = project_box_sum(h_try, ts)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            new_val = np.dot(weights, corr_try)
            if new_val < current - 1e-15:
                h = h_try
                corr = corr_try
                improved = True
                break
        if improved:
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step_init * 0.1
                momentum = np.zeros(n)

    return h


def local_swap_search(h, n, max_iters):
    ts = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for _ in range(max_iters):
        if no_improve > 500:
            break
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        h = project_box_sum(h, ts)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
            h = project_box_sum(h, ts)
            no_improve += 1
    return h, best_c5


def binarize_and_reopt(h, n, alpha, ts):
    h_b = np.clip((h - 0.5) * alpha + 0.5, 0.0, 1.0)
    h_b = project_box_sum(h_b, ts)
    return h_b


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Exponential weights optimization
    for beta, iters, step in [(3, 400, 0.012), (5, 400, 0.009), (10, 350, 0.006), (18, 300, 0.004), (30, 250, 0.003)]:
        if tl() < 180:
            break
        h_new = exp_weights_opt(best_h, n, beta, iters, step)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 2: L-BFGS-B with smooth max (increasing sharpness)
    for beta, lam in [(100, 50), (300, 150), (800, 400), (2000, 1000)]:
        if tl() < 100:
            break
        try:
            r = minimize(obj_smoothmax, best_h, args=(n, ts, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 1500, 'ftol': 1e-16})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 3: Binarization attempts
    for alpha in [2.0, 3.0, 4.0, 6.0, 8.0]:
        if tl() < 80:
            break
        h_b = binarize_and_reopt(best_h, n, alpha, ts)
        for beta, lam in [(500, 200), (1500, 600)]:
            if tl() < 40:
                break
            try:
                r = minimize(obj_smoothmax, h_b, args=(n, ts, beta, lam),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 1000, 'ftol': 1e-16})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 4: Local swap search
    if tl() > 120:
        h_new, c = local_swap_search(best_h, n, max_iters=15000)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 5: More exp-weights from current best
    for beta, iters, step in [(15, 300, 0.005), (40, 200, 0.002)]:
        if tl() < 80:
            break
        h_new = exp_weights_opt(best_h, n, beta, iters, step)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 6: High-precision L-BFGS-B polish
    for beta, lam in [(4000, 2000), (8000, 5000)]:
        if tl() < 40:
            break
        try:
            r = minimize(obj_smoothmax, best_h, args=(n, ts, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 7: Final local search
    if tl() > 40:
        h_new, c = local_swap_search(best_h, n, max_iters=10000)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 8: Try different n values
    for n_try in [80, 90, 100, 120]:
        if tl() < 150:
            break
        x = np.arange(n_try) * 2.0 / n_try + 1.0 / n_try
        h_try = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x + 0.5)
        h_try = np.clip(h_try, 0, 1)
        h_try = project_box_sum(h_try, n_try / 2.0)
        bnds_try = [(0.0, 1.0)] * n_try
        ts_try = n_try / 2.0

        for beta, lam, maxit in [(200, 80, 500), (800, 300, 1000)]:
            if tl() < 70:
                break
            try:
                r = minimize(obj_smoothmax, h_try, args=(n_try, ts_try, beta, lam),
                             method='L-BFGS-B', bounds=bnds_try,
                             options={'maxiter': maxit, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts_try)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5, n = h_new, c, n_try
            except Exception:
                pass

    # Final SLSQP attempt on best
    if tl() > 30:
        try:
            def slsqp_obj(x):
                c = np.correlate(x, 1.0 - x, mode="full")
                return smooth_max(c, 5000.0) * (2.0 / n)
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
            r = minimize(slsqp_obj, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_box_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
