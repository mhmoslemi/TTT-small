    return h.sum() - target_sum

    return np.ones(n)

def lag_gradient(h, k_worst, n):
    grad = np.zeros(n)
    lag = k_worst - (n - 1)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad
