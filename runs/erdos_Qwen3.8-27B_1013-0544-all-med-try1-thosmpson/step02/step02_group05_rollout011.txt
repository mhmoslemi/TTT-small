import numpy as np
from scipy.optimize import linprog, minimize, Bounds
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_simplex(h, target_sum):
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def solve_linlp(h0, target_sum, dx):
    n = len(h0)
    n_full = 2 * n - 1
    coeff = np.zeros((n_full, n))
    const = np.zeros(n_full)
    for k_idx in range(n_full):
        s = k_idx - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        js = idx + s
        coeff[k_idx, idx] += (1.0 - h0[js])
        coeff[k_idx, js] += -h0[idx]
        const[k_idx] = h0[idx] @ (1.0 - h0[js])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A = np.zeros((n_full, n + 1))
    A[:, :n] = coeff
    A[:, -1] = -1.0
    b = -dx * const
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    beq = np.array([target_sum])
    bnds = [(0.0, 1.0)] * n + [(-1e10, 1e10)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                  bounds=bnds, method='highs')
    return res.x[:n] if res.success else h0.copy()

def _sm_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, _sm_beta) * _sm_dx

def _sum_con(x):
    return np.sum(x) - _sum_target

def _sum_jac(x):
    return np.ones(len(x))

_sm_beta = 100.0
_sm_dx = 0.028
_sum_target = 35.0

def run(seed=42, budget_s=1100, **kwargs):
    global _sm_beta, _sm_dx, _sum_target
    np.random.seed(seed)
    t_start = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h_par = None
    n_par = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_par = np.array(initial_h_values, dtype=float).copy()
        n_par = len(h_par)

    def remaining():
        return budget_s - (time.time() - t_start)

    n_list = [70, 80, 90, 100]
    if n_par in n_list:
        n_list.remove(n_par)
        n_list.insert(0, n_par)

    for n in n_list:
        if remaining() < 50:
            break

        ts = n / 2.0
        dx = 2.0 / n
        n_alloc = remaining() / max(1, len(n_list) - n_list.index(n))

        if h_par is not None and n == n_par:
            h0 = h_par.copy()
        elif h_par is not None:
            xo = np.linspace(0, 2, n_par, endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(xn, xo, h_par, period=2.0)
            h0 = project_simplex(h0, ts)
        else:
            h0 = np.full(n, 0.5)

        h_cur = h0.copy()
        c5_cur = compute_c5(h_cur)

        # Phase 1: Successive LP linearization
        t1 = time.time()
        p1_time = n_alloc * 0.35
        for it in range(20000):
            if time.time() - t1 > p1_time:
                break
            h_lp = solve_linlp(h_cur, ts, dx)
            h_lp = project_simplex(h_lp, ts)
            if np.max(np.abs(h_lp - h_cur)) < 1e-11:
                h_cur = h_lp
                break
            h_cur = h_lp

        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5_cur, n

        # Phase 2: Multi-start with LP refinement
        t2 = time.time()
        p2_time = n_alloc * 0.30
        for trial in range(30):
            if time.time() - t2 > p2_time:
                break
            scale = 0.02 + 0.05 * (trial % 6) / 5.0
            h_p = h_cur + np.random.randn(n) * scale
            h_p = project_simplex(h_p, ts)
            tp = time.time()
            for it in range(5000):
                if time.time() - tp > 3.0 or time.time() - t2 > p2_time:

                h_p = solve_linlp(h_p, ts, dx)
                h_p = project_simplex(h_p, ts)

                c5_cur = c5_p

                                                  'jac': _sum_jac}],
                                   options={'maxiter': 400, 'ftol': 1e-14})
                    h_new = project_simplex(res.x, ts)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h_cur = h_new
                        c5_cur = c5_new
                        if c5_cur < best_c5:
                            best_h, best_c5, best_n = h_cur.copy(), c5_cur, n
                except Exception:
                    pass

        # Phase 4: Final LP polish
        if remaining() > 20:
            h_fin = h_cur.copy()
            tf = time.time()
            for it in range(10000):
                if time.time() - tf > n_alloc * 0.25 or remaining() < 10:
                    break
                h_fin = solve_linlp(h_fin, ts, dx)
                h_fin = project_simplex(h_fin, ts)
                if compute_c5(h_fin) < c5_cur:
                    c5_cur = compute_c5(h_fin)
                    if c5_cur < best_c5:
                        best_h, best_c5, best_n = h_fin.copy(), c5_cur, n

    # Ultimate polish on overall best
    if best_h is not None and remaining() > 20:
        nf = best_n
        tsf = nf / 2.0
        dxf = 2.0 / nf
        h_u = best_h.copy()
        tu = time.time()
        for it in range(20000):
            if time.time() - tu > remaining() * 0.5 or remaining() < 5:
                break
            h_u = solve_linlp(h_u, tsf, dxf)
            h_u = project_simplex(h_u, tsf)
        c5_u = compute_c5(h_u)
        if c5_u < best_c5:
            best_h, best_c5 = h_u.copy(), c5_u

        # SLSQP final
        if remaining() > 15:
            _sm_beta = 500.0
            _sm_dx = dxf
            _sum_target = tsf
            bounds = Bounds(np.zeros(nf), np.ones(nf))
            try:
                res = minimize(_sm_obj, best_h, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': _sum_con,
                                              'jac': _sum_jac}],
                               options={'maxiter': 500, 'ftol': 1e-15})
                h_s = project_simplex(res.x, tsf)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_h, best_c5 = h_s.copy(), c5_s
            except Exception:
                pass

    return best_h, best_c5, best_n
