    val = smooth_max_val(corr, beta) * dx
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -700.0, 700.0)
    p = np.exp(s) / np.sum(np.exp(s))
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = p[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n-la])
            grad[:n-la] -= w * h[la:]
    grad *= dx
    return val, grad

def fista(h0, ts, t0, budget, beta):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    h_prev = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(best)
    step = 0.05
    t = 1
    stall = 0
    max_iter = 500000
    for it in range(max_iter):
        if time.time() - t0 > t0 + budget:
            break
        mu = (t - 1.0) / (t + 2.0)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        val, grad = obj_grad(y, dx, beta)
        s = step
        for _ in range(20):
            cand = project_feasible(y - s * grad, ts)
            vc, _ = obj_grad(cand, dx, beta)
            if vc <= val + np.dot(grad, cand - y) + 0.5 / s * np.sum((cand - y)**2):
                break
            s *= 0.5
        step = min(s * 1.8, 0.5)
        h_prev = h
        h = cand
        t += 1
        if it % 3000 == 2999:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-8:
                best_c5 = c5
                best = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 8:
                h = project_feasible(h + np.random.randn(n) * 0.03, ts)
                h_prev = h.copy()
                t = 1
                step = 0.05
                stall = 0
    c5 = compute_c5(h)
    if c5 < best_c5:
        best, best_c5 = h.copy(), c5
    return best, best_c5

def slsqp_obj(h, dx, beta):
    return smooth_max_val(np.correlate(h, 1.0 - h, mode="full"), beta) * dx

def slsqp_grad(h, dx, beta):
    return obj_grad(h, dx, beta)[1]

def cons_fun(h, ts):
    return h.sum() - ts

def cons_jac(h, ts):
    return np.ones_like(h)

def resample(h, n_new):
    x = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, x, h, period=2.0)

def diverse_starts(n, ts, seed):
    rng = np.random.RandomState(seed)
    candidates = []
    candidates.append(np.full(n, 0.5))
    x = np.linspace(0, 2, n, endpoint=False)
    for f in range(1, 8):
        for amp in [0.1, 0.25, 0.4]:
            candidates.append(0.5 + amp * np.sin(2 * np.pi * f * x / 2.0))
    for f in range(1, 8):
        candidates.append(0.5 + 0.3 * np.cos(2 * np.pi * f * x / 2.0 + 0.5))
    for _ in range(20):
        candidates.append(rng.rand(n))
    for _ in range(10):
        block = rng.randint(3, 15)
        pattern = np.tile(rng.rand(block), n // block + 1)[:n]
        candidates.append(pattern)
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)

    n = len(h)
    ts = n / 2.0
    h = project_feasible(h, ts)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 0: Try diverse starts to find a good initial point
    remaining = budget_s - (time.time() - t0)
    if remaining > 100:
        cands = diverse_starts(n, ts, seed)
        if 'initial_h_values' in globals() and initial_h_values is not None:
            cands.insert(0, np.array(initial_h_values, dtype=float))
        for c in cands:
            c = project_feasible(np.asarray(c, dtype=float)[:n] if len(c) >= n else np.resize(c, n), ts)
            c5 = compute_c5(c)
            if c5 < best_c5:
                best_h, best_c5 = c.copy(), c5

    # Phase 1: FISTA with multiple betas
    for beta in [80, 200, 500, 1200]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        h_f, c5_f = fista(best_h, ts, t0, remaining - 40, beta)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Phase 2: SLSQP with gradient
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        dx = 2.0 / n
        bnds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (ts,)}]
        for beta in [300, 1500, 8000]:
            rem2 = budget_s - (time.time() - t0)
            if rem2 < 12:
                break
            for trial in range(2):
                rem3 = budget_s - (time.time() - t0)
                if rem3 < 8:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * 0.008
                    x0 = project_feasible(x0, ts)
                try:
                    res = minimize(slsqp_obj, x0, args=(dx, beta), jac=slsqp_grad,
                                   method='SLSQP', bounds=bnds, constraints=cons,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new, c5
                except Exception:
                    pass

    # Phase 3: Multi-restart FISTA
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        for r in range(8):
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            scale = 0.02 + 0.02 * r
            h_r = project_feasible(best_h + np.random.randn(n) * scale, ts)
            h_r, c5_r = fista(h_r, ts, t0, rem - 15, 400 + 200 * r)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 4: Resolution exploration
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        n_cur = len(best_h)
        for n_try in sorted(set([n_cur - 15, n_cur - 10, n_cur - 5, n_cur + 5, n_cur + 10, n_cur + 15, n_cur + 20, n_cur + 25])):
            if n_try < 20 or n_try % 2 == 0:
                continue
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                continue
            ts_t = n_try / 2.0
            h_t = project_feasible(resample(best_h, n_try), ts_t)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t
            h_t, c5_t = fista(h_t, ts_t, t0, rem - 10, 300)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 5: Final SLSQP polish on best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bnds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        cons_f = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (ts_f,)}]
        for beta in [20000, 80000]:
            rem = budget_s - (time.time() - t0)
            if rem < 8:
                break
            try:
                res = minimize(slsqp_obj, best_h, args=(dx_f, beta), jac=slsqp_grad,
                               method='SLSQP', bounds=bnds_f, constraints=cons_f,
                               options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
