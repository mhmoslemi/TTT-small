def frank_wolfe(h, max_iters=10000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy()
    best_x = x.copy()
    best_f = compute_c5(x)
    
    for t in range(max_iters):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        active = np.where(corr >= m - 1e-10)[0]
        
        # Subgradient: average of gradients of active correlations
        g = np.zeros(n)
        j = np.arange(n)
        for k in active:
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                g[ms] += (1.0 - x[js[ms]])
            if mp.any():
                g[mp] += (-x[jp[mp]])
        g /= len(active)
        
        # Linear minimization: min <g, s> s.t. 0<=s<=1, sum(s)=target
        # Solution: set s[i]=1 for smallest g[i], 0 for largest
        order = np.argsort(g)
        s = np.zeros(n)
        s[order[:int(target)]] = 1.0
        # Handle fractional part
        frac = target - int(target)
        if frac > 1e-10:
            s[order[int(target)]] = frac
        
        # Step: x_new = (1-alpha)*x + alpha*s
        # Use standard FW step size: alpha = 2/(t+2)
        alpha = 2.0 / (t + 2)
        x_new = (1 - alpha) * x + alpha * s
        x_new = np.clip(x_new, 0.0, 1.0)
        # Fix sum
        x_new = project_feasible(x_new, target)
        
        f_new = compute_c5(x_new)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_f

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        m = np.max(corr[idx])
        exps = np.exp(beta * (corr[idx] - m))
        sw = np.sum(exps)
        w = exps / sw
        val = (m + np.log(sw) / beta) * dx
        grad = np.zeros(n)
        j = np.arange(n)
        for k in idx:
            wk = w[k]
            if wk < 1e-8:
                continue
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += wk * (1.0 - h[js[ms]])
            if mp.any():
                grad[mp] += wk * (-h[jp[mp]])
        grad *= dx
    else:
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        sw = np.sum(exps)
        w = exps / sw
        val = (m + np.log(sw) / beta) * dx
        grad = np.zeros(n)
        j = np.arange(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-10:
                continue
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += wk * (1.0 - h[js[ms]])
            if mp.any():
                grad[mp] += wk * (-h[jp[mp]])
        grad *= dx
    return val, grad

def frank_wolfe_smooth(h, max_iters, beta, top_k=20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_f = compute_c5(x)
    
    for t in range(max_iters):
        val, g = smoothmax_val_grad(x, beta, top_k)
        
        # FW step: find s in {0<=s<=1, sum=target} minimizing <g, s>
        order = np.argsort(g)
        s = np.zeros(n)
        int_part = int(target)
        frac_part = target - int_part
        s[order[:int_part]] = 1.0
        if frac_part > 1e-10:
            s[order[int_part]] = frac_part
        
        alpha = 2.0 / (t + 2)
        x_new = (1 - alpha) * x + alpha * s
        x_new = np.clip(x_new, 0.0, 1.0)
        x_new = project_feasible(x_new, target)
        
        f_new = compute_c5(x_new)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        x = x_new
        
        # Convergence check
        if t % 1000 == 0 and t > 0:
            gap = val - compute_c5(x)  # rough gap
            if gap < 1e-6:
                break
    
    return best_x, best_f

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):

    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)

def slsqp_refine(h, maxiter, beta, top_k=None):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return v
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def _con_fun(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

_beta_g = 100.0

def _sm_obj(x):

    exps = np.exp(_beta_g * (corr - m))
    sw = np.sum(exps)
    return (m + np.log(sw) / _beta_g) * dx
