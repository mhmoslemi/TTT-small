Let me try a different approach: **coordinate descent on the Lagrangian dual**.

The minimax problem can be written as:
min_h max_k corr_k(h) * dx

By minimax duality (Sion's theorem, since the feasible set is compact and convex, and for each h, max_k is well-defined):
= max_{p ∈ Δ_{2n-1}} min_h sum_k p_k * corr_k(h) * dx

where Δ is the simplex. The inner problem is:
min_h sum_k p_k * corr_k(h) * dx
s.t. 0 <= h <= 1, sum(h) = n/2

This is a quadratic minimization (since sum_k p_k * corr_k is still quadratic in h). We can solve it with projected gradient or L-BFGS-B.

Then we iterate:
1. Given p, solve inner problem to get h*
2. Update p: p_k = 1 if k = argmax corr_k(h*), else 0 (or smooth version)
3. Repeat

This is the Rackauckas-Burby / multiplicative weights approach for minimax.

Actually, this is essentially the **mirror descent / multiplicative weights** for the outer problem:
p_k^{(t+1)} = p_k^{(t)} * exp(η * corr_k(h^{(t)})) / Z

This is guaranteed to converge (for the maximization over p) and each step requires solving a quadratic program (or using gradient descent on the inner problem).

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h *= (1.0 + diff / total)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 700:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def cutting_plane(h, n, t0, budget_s, max_iters=200):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 15:
            break
        hs = best_h
        corr = np.correlate(hs, 1.0 - hs, mode='full')
        A_rows = np.zeros((n_lags, n + 1))
        b_vals = np.zeros(n_lags)
        omh = 1.0 - hs
        for k in range(n_lags):
            off = k - (n - 1)
            grad = np.zeros(n)
            for i in range(n):
                j1 = i + off
                j2 = i - off
                if 0 <= j1 < n:
                    grad[i] = omh[j1]
                if 0 <= j2 < n:
                    grad[i] -= hs[j2]
            A_rows[k, :n] = grad
            A_rows[k, n] = -1.0
            b_vals[k] = -corr[k] + np.dot(grad, hs)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, 1)] * n + [(0, None)]
        try:
            res = linprog(c_obj, A_ub=A_rows, b_ub=b_vals, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_new = res.x[:n]
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_h = project_feasible(h_new, n)
            best_c5 = compute_c5(best_h, n)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                ht = (1.0 - frac) * best_h + frac * h_new
                ht = project_feasible(ht, n)
                ct = compute_c5(ht, n)
                if ct < best_c5 - 1e-15:
                    best_h, best_c5 = ht, ct
                    improved = True
                    break
            if not improved and np.max(np.abs(h_new - best_h)) < 1e-12:
                break
    return best_h, best_c5
