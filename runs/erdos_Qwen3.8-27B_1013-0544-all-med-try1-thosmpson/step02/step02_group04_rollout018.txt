import numpy as np
from scipy.optimize import minimize, Bounds
import time

_beta = 100.0
_dx = 2.0 / 70.0
_npts = 70
_target = 35.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_grad(corr, h, beta, dx):
    n = len(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(shifted)
    w = e / np.sum(e)
    val = (m + np.log(np.sum(e)) / beta) * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            L = lag
            if L < n:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
        else:
            M = -lag
            if M < n:
                g[:n - M] += wk * (1.0 - h[M:])
                g[M:] -= wk * h[:n - M]
    return val, g


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    return np.clip(h, 0.0, 1.0)


def _obj(h):
    corr = corr_arr(h)
    return smooth_max_val(corr, _beta) * _dx


def _grad(h):
    corr = corr_arr(h)
    _, g = smooth_max_grad(corr, h, _beta, _dx)
    return g


def _cons_fun(x):
    return np.sum(x) - _target


def _cons_jac(x):
    return np.ones(_npts)


def _do_slsqp(h0, beta, dx, n, target_sum, maxiter, ftol):
    global _beta, _dx, _npts, _target
    _beta = beta
    _dx = dx
    _npts = n
    _target = target_sum
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x


def _perturb_top_lag(h, lag, dx, n, magnitude):
    h = h.copy()
    if lag >= 0:
        L = lag
        if L < n:
            h[L:] -= magnitude * (1.0 - h[:n - L])
            h[:n - L] += magnitude * h[L:]
    else:
        M = -lag
        if M < n:
            h[:n - M] -= magnitude * (1.0 - h[M:])
            h[M:] += magnitude * h[:n - M]
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global _beta, _dx, _npts, _target
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

        return budget_s - (time.time() - start_time)

        n_points = 70

    _target = target_sum
    _dx = dx

    best_h = project_h(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing beta
    for beta in [10, 20, 50, 100, 200, 500, 1000, 2000, 5000]:

            h_new = project_h(h_new, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:

                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: Multi-start local search with diverse strategies
    for trial in range(120):
        if time_left() < 15:
            break
        h0 = best_h.copy()
        s = trial % 8
        scale = 0.002 + 0.015 * np.exp(-trial / 40.0)

        elif s == 1:

            _, g = smooth_max_grad(corr, h0, 500, dx)
            norm = np.max(np.abs(g)) + 1e-10
            h0 -= 0.03 * g / norm
        elif s == 2:
            i, j = rng.choice(n_points, 2, replace=False)
            d = 0.06 * rng.randn()
            h0[i] += d
            h0[j] -= d
        elif s == 3:

            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] += rng.randn(blen) * scale * 3
        elif s == 4:
            corr = corr_arr(h0)
            top_k = np.argmax(corr)
            lag = top_k - (n_points - 1)
            h0 = _perturb_top_lag(h0, lag, dx, n_points, 0.01)
        elif s == 5:

            best_c5 = c5_try
            best_h = h_try.copy()
        # SLSQP polish

        except Exception:
            pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
