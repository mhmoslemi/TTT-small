def apply_corr_grad(h, weights, n):
    """
    Compute sum_k weights[k] * grad(corr[k])/grad(h)
    where grad(corr[k]) is the gradient of corr[k] with respect to h.
    """
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        lag = k - n_half
        if lag >= 0:
            # corr[k] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # d/dh[i] for i < n-lag: (1 - h[i+lag])
            # d/dh[i] for i >= lag: -h[i-lag]
            upper = n - lag
            grad[:upper] += w * (1.0 - h[lag:lag+upper])
            grad[lag:lag+upper] -= w * h[:upper]
        else:
            la = -lag
            upper = n - la
            # corr[k] = sum_{i=la}^{n-1} h[i-la] * (1-h[i])
            # d/dh[i] for i >= la: (1 - h[i])
            # d/dh[i] for i < n-la: -h[i+la]
            grad[la:la+upper] += w * (1.0 - h[la:la+upper])
            grad[:upper] -= w * h[la:la+upper]
    return grad

    n_corr = len(corr)
    # Get top-k lags
    idx_sorted = np.argsort(corr)[::-1][:top_k]

        wsum = 1.0
    weights /= wsum

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
    else:
        w /= s
    return w


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def apply_corr_grad(h, weights, n):
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        lag = k - n_half
        if lag >= 0:
            upper = n - lag
            grad[:upper] += w * (1.0 - h[lag:lag + upper])
            grad[lag:lag + upper] -= w * h[:upper]
        else:
            la = -lag
            upper = n - la
            grad[la:la + upper] += w * (1.0 - h[la:la + upper])
            grad[:upper] -= w * h[la:la + upper]
    return grad


def obj_func(hv, dx, beta):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    return smooth_max(corr, beta) * dx

    return dx * apply_corr_grad(hv, w, len(hv))

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
