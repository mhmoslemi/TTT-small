- new_j = old_j - t ∈ [0, 1]
- max_l corr_new[l] is minimized

- 0 ≤ old_i + t ≤ 1 → -old_i ≤ t ≤ 1 - old_i
- 0 ≤ old_j - t ≤ 1 → old_j - 1 ≤ t ≤ old_j

And the objective is max_l (corr_old[l] + t * s_l) where s_l = s_i(l) - s_j(l).

This is the maximum of linear functions of t, which is convex. We can minimize it using bisection on the subgradient.

This is much more efficient! We only need to compute s_l = s_i(l) - s_j(l) for each pair, which is O(n) per pair (since s_i(l) is nonzero only for O(n) lags).

```python
def pair_step(h, i, j, target_sum, n):
    """One pairwise coordinate descent step."""
    old_i = h[i]
    old_j = h[j]
    
    # Compute s_l = s_i(l) - s_j(l) for all l
    # s_i(l) = (1-h[i+shift]) if valid + (-h[i-shift]) if valid
    # where shift = l - (n-1)
    
    offset = n - 1
    s = np.zeros(2*n - 1)
    
    for l in range(2*n - 1):
        shift = l - offset
        si = 0.0
        if 0 <= i + shift < n:
            si += (1.0 - h[i + shift])
        if 0 <= i - shift < n:
            si -= h[i - shift]
        sj = 0.0
        if 0 <= j + shift < n:
            sj += (1.0 - h[j + shift])
        if 0 <= j - shift < n:
            sj -= h[j - shift]
        s[l] = si - sj
    
    # Valid range for t
    t_lo = max(-old_i, old_j - 1.0)
    t_hi = min(1.0 - old_i, old_j)
    
    if t_lo >= t_hi:
        return  # Can't move
    
    # Get current correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Minimize max_l (corr[l] + t * s[l]) over t ∈ [t_lo, t_hi]
    # Using bisection on subgradient
    a, b = t_lo, t_hi
    for _ in range(40):
        mid = (a + b) / 2.0
        vals = corr + mid * s
        i_max = np.argmax(vals)
        if s[i_max] > 1e-15:
            b = mid
        elif s[i_max] < -1e-15:
            a = mid
        else:
            break
    
    t_opt = (a + b) / 2.0
    
    # Check if this actually improves
    new_corr = corr + t_opt * s
    new_max = np.max(new_corr)
    old_max = np.max(corr)
    
    if new_max < old_max - 1e-12:
        h[i] = old_i + t_opt
        h[j] = old_j - t_opt
    
    return h
```

        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    offset = n - 1
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP (60s)
    for beta in [20, 200, 1000]:
        if tl() < 30:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        try:
            res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Subgradient with momentum (120s)
    if tl() > 60:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode="full")
        momentum = np.zeros(n)
        t_ph = time.time()
        p_budget = min(tl() * 0.15, 120)
        lr = 0.01
        for it in range(500000):
            if time.time() - t_ph > p_budget:
                break
            k_worst = np.argmax(corr)
            shift = k_worst - offset
            g = np.zeros(n)
            if shift >= 0:
                idx = np.arange(n - shift)
                g[idx] += (1.0 - h[idx + shift])
                g[idx + shift] -= h[idx]
            else:
                la = -shift
                idx = np.arange(n - la)
                g[idx + la] += (1.0 - h[idx])
                g[idx] -= h[idx + la]
            momentum = 0.9 * momentum + 0.1 * g
            h = h - lr * momentum
            h = project_feasible(h, target_sum)
            old_corr = corr
            corr = np.correlate(h, 1.0 - h, mode="full")
            if it % 5000 == 4999:
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 3: Pairwise CD (250s)
    if tl() > 60:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode="full")
        t_ph = time.time()
        p_budget = min(tl() * 0.3, 250)
        last_best = np.inf
        stall = 0
        for sweep in range(100000):
            if time.time() - t_ph > p_budget:
                break
            h, corr = cd_sweep(h, corr, n, offset)
            cur_max = np.max(corr)
            if cur_max < last_best - 1e-14:
                stall = 0
                last_best = cur_max
            else:
                stall += 1
                if stall > 200:
                    # Random restart within CD
                    for _ in range(20):
                        i = np.random.randint(n)
                        j = np.random.randint(n)
                        if i != j:
                            h, corr = pair_step(h, corr, i, j, n, offset)
                    stall = 0
            if sweep % 200 == 199:
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 4: Random perturbation + CD (250s)
    if tl() > 60:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode="full")
        t_ph = time.time()
        p_budget = min(tl() * 0.25, 250)
        for it in range(3000):
            if time.time() - t_ph > p_budget:
                break
            n_pert = np.random.randint(1, 4)
            idx = np.random.choice(n, n_pert, replace=False)
            h_try = h.copy()
            for ix in idx:
                h_try[ix] += np.random.uniform(-0.08, 0.08)
            h_try = project_feasible(h_try, target_sum)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            nsweeps = np.random.randint(3, 15)
            for _ in range(nsweeps):
                h_try, corr_try = cd_sweep(h_try, corr_try, n, offset)
            c5_try = np.max(corr_try) * dx
            if c5_try < best_c5 - 1e-12:
                best_h, best_c5 = h_try.copy(), c5_try
                h, corr = h_try.copy(), corr_try.copy()

    # Phase 5: Multi-resolution (150s)
    if tl() > 50:
        t_ph = time.time()
        p_budget = min(tl() * 0.15, 150)
        for n_try in [68, 72, 74, 76, 80, 84, 88]:
            if time.time() - t_ph > p_budget:
                break
            if n_try < 30:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            off_t = n_try - 1
            h_t = project_feasible(resample_h(best_h, n_try), ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
            try:
                res = minimize(obj_smooth, h_t, args=(dx_t, 500), method='SLSQP',
                               bounds=bnds_t, constraints=cons_t,
                               options={'maxiter': 3000, 'ftol': 1e-13})
                h_t = project_feasible(res.x, ts_t)
            except Exception:
                pass
            corr_t = np.correlate(h_t, 1.0 - h_t, mode="full")
            for _ in range(30):
                h_t, corr_t = cd_sweep(h_t, corr_t, n_try, off_t)
            c5_t = np.max(corr_t) * dx_t
            if c5_t < best_c5:
                best_h, best_c5 = h_t.copy(), c5_t
                n, dx, target_sum, offset = n_try, dx_t, ts_t, off_t
                bounds = bnds_t

    # Phase 6: Final polish
    if tl() > 15:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode="full")
        for _ in range(50):
            if tl() < 8:
                break
            h, corr = cd_sweep(h, corr, n, offset)
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        if tl() > 12:
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
            for beta in [5000, 20000]:
                if tl() < 8:
                    break
                try:
                    res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass
        # Final CD after SLSQP
        if tl() > 5:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode="full")
            for _ in range(20):
                if tl() < 2:
                    break
                h, corr = cd_sweep(h, corr, n, offset)
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    return best_h, best_c5, len(best_h)
