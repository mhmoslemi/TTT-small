def compute_dc(h, n, a, b):
    """Compute the change in correlation vector when h[a]+=d, h[b]-=d, per unit d."""
    c = np.zeros(2*n-1)
    off = n - 1
    # For index a (h[a] increases by d):
    # First factor: (1-h[a+k-off]) for valid k
    idx1 = a + np.arange(2*n-1) - off  # = a + k - (n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    c[valid1] += (1 - h[idx1[valid1]])
    # Second factor: -h[a+k-off] where a = i + k - off, so i = a - k + off
    idx2 = a - np.arange(2*n-1) + off  # = a - k + (n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    c[valid2] -= h[idx2[valid2]]
    # For index b (h[b] decreases by d):
    idx3 = b + np.arange(2*n-1) - off
    valid3 = (idx3 >= 0) & (idx3 < n)
    c[valid3] -= (1 - h[idx3[valid3]])
    idx4 = b - np.arange(2*n-1) + off
    valid4 = (idx4 >= 0) & (idx4 < n)
    c[valid4] += h[idx4[valid4]]
    return c

def coordinate_descent_step(h, corr, n, a, b, d_range):
    """Find optimal d for pair (a,b) given current correlation."""
    dc = compute_dc(h, n, a, b)
    # Objective: max(corr + d * dc)
    # Find d in d_range that minimizes this
    # Use ternary search (convex function)
    lo, hi = d_range[0], d_range[1]
    if hi - lo < 1e-12:
        return 0.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(corr + m1 * dc)
        f2 = np.max(corr + m2 * dc)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d = (lo + hi) / 2
    return d

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

def compute_dc(h, n, a, b):
    k = np.arange(2 * n - 1)
    off = n - 1
    dc = np.zeros(2 * n - 1)
    idx1 = a + k - off
    v1 = (idx1 >= 0) & (idx1 < n)
    dc[v1] += (1.0 - h[idx1[v1]])
    idx2 = a - k + off
    v2 = (idx2 >= 0) & (idx2 < n)
    dc[v2] -= h[idx2[v2]]
    idx3 = b + k - off
    v3 = (idx3 >= 0) & (idx3 < n)
    dc[v3] -= (1.0 - h[idx3[v3]])
    idx4 = b - k + off
    v4 = (idx4 >= 0) & (idx4 < n)
    dc[v4] += h[idx4[v4]]
    return dc

def cd_sweep(h, corr, n, t0, budget_s, max_passes=100):
    best_c5 = float(np.max(corr))
    for p in range(max_passes):
        if budget_s - (time.time() - t0) < 30:
            break
        improved = False
        for a in range(n):
            for b in range(a + 1, n):
                d_lo = max(-h[a], h[b] - 1.0)
                d_hi = min(1.0 - h[a], h[b])
                if d_hi - d_lo < 1e-10:
                    continue
                dc = compute_dc(h, n, a, b)
                # Quick check: can we improve?

                # If dc at argmax is 0, no improvement possible
                lo, hi = d_lo, d_hi
                # Ternary search (convex minimization)
                for _ in range(30):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = np.max(corr + m1 * dc)
                    f2 = np.max(corr + m2 * dc)
                    if f1 < f2:

                    else:

                    h[b] -= d
                    corr += d * dc
                    best_c5 = new_max
                    improved = True
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        out = project_feasible(res.x, n)
        if np.min(out) > -1e-6 and np.max(out) < 1 + 1e-6:
            return out
    except Exception:
        pass
    return h

def compute_dc(h, n, a, b):
    k = np.arange(2 * n - 1)
    off = n - 1
    dc = np.zeros(2 * n - 1)
    idx1 = a + k - off
    v1 = (idx1 >= 0) & (idx1 < n)
    dc[v1] += (1.0 - h[idx1[v1]])
    idx2 = a - k + off
    v2 = (idx2 >= 0) & (idx2 < n)
    dc[v2] -= h[idx2[v2]]
    idx3 = b + k - off
    v3 = (idx3 >= 0) & (idx3 < n)
    dc[v3] -= (1.0 - h[idx3[v3]])
    idx4 = b - k + off
    v4 = (idx4 >= 0) & (idx4 < n)
    dc[v4] += h[idx4[v4]]
    return dc
