def project(h, target_sum):
    """Project h onto {x ∈ [0,1]^n : sum(x) = target_sum}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    # Water filling
    if abs(diff) > 1e-12:
        # Binary search for threshold
        if diff > 0:
            # Need to increase sum: add to elements not at 1
            free = h < 1.0
        else:
            # Need to decrease sum: reduce elements not at 0
            free = h > 0.0
        # Simple approach: distribute equally among free elements
        n_free = free.sum()
        if n_free > 0:
            h[free] += diff / n_free
            h = np.clip(h, 0, 1)
            # If still not right, do a second pass
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-12:
                free2 = (h > 0) & (h < 1)
                if free2.sum() > 0:
                    h[free2] += diff2 / free2.sum()
                    h = np.clip(h, 0, 1)
    return h
```

def project_box_sum(h, target_sum, n):
    """Exact projection onto {x: 0≤x_i≤1, sum(x)=target_sum}"""
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.clip(h - theta, 0, 1)
```

This is the exact projection! The projection of y onto {x: 0≤x≤1, sum(x)=t} is x_i = clip(y_i - θ, 0, 1) where θ is chosen so that sum(clip(y_i - θ, 0, 1)) = t.

def projected_subgradient(h, n, target_sum, max_iter=10000, tol=1e-10):
    dx = 2.0 / n
    h = h.copy()
    prev_f = compute_c5(h)
    
    # Barzilai-Borwein step size
    step = 0.1
    
    for it in range(max_iter):
        # Compute overlap vector
        corr = np.correlate(h, 1-h, mode="full")
        max_val = corr.max()
        f_curr = max_val * dx
        
        if f_curr < prev_f:
            prev_f = f_curr
            step *= 1.05  # increase step slightly
        else:
            step *= 0.8  # decrease step
        
        # Find active lags
        active_mask = corr >= max_val - 1e-8
        n_active = active_mask.sum()
        
        # Compute subgradient
        # ∂O(k)/∂h[i] = (1-h[i+k]) - h[i-k] with boundary
        grad = np.zeros(n)
        active_indices = np.where(active_mask)[0]
        for k_idx in active_indices:
            k = k_idx - (n - 1)  # actual lag
            for i in range(n):
                ip = i + k
                im = i - k
                g = 0.0
                if 0 <= ip < n:
                    g += (1 - h[ip])
                if 0 <= im < n:
                    g -= h[im]
                grad[i] += g
        grad /= n_active
        grad *= dx
        
        # Take step
        h_new = h - step * grad
        h_new = project_box_sum(h_new, target_sum, n)
        
        # Check convergence
        if np.max(np.abs(h_new - h)) < tol:
            break
        h = h_new
    
    return h

    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # i+k ranges from k to n-1, so i ranges from 0 to n-1-k
        g[:n-k] += (1 - h[k:])  # (1-h[i+k]) for i=0..n-1-k
        if k > 0:
            g[k:] -= h[:n-k]  # -h[i-k] for i=k..n-1
    else:

        kp = -k
        # i+k = i-k' < n always for i < n, and ≥ 0 for i ≥ k'
        g[kp:] += (1 - h[:n-kp])  # (1-h[i+k]) for i=k'..n-1
        g[:n-kp] -= h[kp:]  # -h[i-k] = -h[i+k'] for i=0..n-1-k'

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project onto {x: 0≤x_i≤1, sum(x)=target_sum}"""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.clip(h - theta, 0, 1)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def compute_subgrad(h, corr, tol=1e-6):
    """Compute subgradient of max(corr) w.r.t. h"""
    n = len(h)
    max_val = corr.max()
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        if k >= 0:
            t1 = np.zeros(n); t1[:n-k] = 1.0 - h[k:]
            t2 = np.zeros(n); t2[k:] = h[:n-k]
        else:
            kp = -k
            t1 = np.zeros(n); t1[kp:] = 1.0 - h[:n-kp]
            t2 = np.zeros(n); t2[:n-kp] = h[kp:]
        grad += t1 - t2
    return grad / len(active)

def projected_subgrad(h, n, target_sum, max_iter=10000, dx=0.02857):
    """Projected subgradient with BB step size"""
    h = h.copy()
    prev_h = h.copy()
    prev_g = np.zeros(n)
    step = 0.5
    best_f = compute_c5(h)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        f_curr = float(corr.max() * dx)
        
        if f_curr < best_f:
            best_f = f_curr
        
        g = compute_subgrad(h, corr) * dx
        
        # BB step size
        s = h - prev_h
        y = g - prev_g
        sy = s @ y
        yy = y @ y
        if yy > 1e-16 and sy > 1e-16:
            step = np.clip(s @ s / sy, 1e-6, 5.0)
        
        prev_h = h.copy()
        prev_g = g.copy()
        
        h = project_box_sum(h - step * g, target_sum)
    
    return h, best_f

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def compute_subgrad(corr, h, tol=1e-6):
    n = len(h)
    max_val = corr.max()
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        if k >= 0:
            t1 = np.zeros(n)
            if n - k > 0:

                t2[:n-k] = h[k:]
        else:

            t1 = np.zeros(n)

            t2 = np.zeros(n)
            if n - kp > 0:
                t2[kp:] = h[:n-kp]
        grad += t1 - t2
    return grad / float(len(active))

    avg_h = h.copy()
    prev_h = h.copy()
    prev_g = np.zeros(n)
    step = 0.5
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        f_curr = float(np.max(corr) * dx)
        if f_curr < best_f:
            best_f = f_curr
            best_h = h.copy()
        g = compute_subgrad(corr, h) * dx
        s = h - prev_h
        y = g - prev_g
        sy = float(s @ y)
        yy = float(y @ y)
        if yy > 1e-14 and sy > 1e-14:
            step = float(np.clip(float(s @ s) / sy, 1e-8, 10.0))
        prev_h = h.copy()
        prev_g = g.copy()
        h = project_box_sum(h - step * g, target_sum)
        avg_h = (it * avg_h + h) / (it + 1)
    
    # Return both current and averaged
    f_avg = compute_c5(avg_h)
    if f_avg < best_f:
        return avg_h, f_avg
    return best_h, best_f

def binary_local_search(h, n, max_rounds=200):
    """Best-improvement swap search on binary h."""
    h = h.copy()
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    
    for _ in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_c5 = curr_c5
        best_swap = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_max = float(np.max(np.correlate(h, 1.0-h, mode="full")) * dx)
                h[i] = 1
                h[j] = 0
                if new_max < best_c5 - 1e-15:
                    best_c5 = new_max
                    best_swap = (i, j)
        
        if best_swap is None:

        h[best_swap[0]] = 0
        h[best_swap[1]] = 1
        curr_c5 = best_c5
    
    return h, curr_c5

def slsqp_polish(h0, n, target_sum, dx, betas, maxiters):
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_best = h0.copy()
    c5_best = compute_c5(h_best)

        def obj(h, _b=float(beta)):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, _b) * dx

            return np.sum(x) - target_sum

            return np.ones(n)

            if c5_new < c5_best:

        except Exception:
            pass
    return h_best, c5_best

    step = 0.5
    best_h = h.copy()
    best_f = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    for it in range(max_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        f_curr = float(np.max(corr) * dx)
        if f_curr < best_f:
            best_f = f_curr
            best_h = h.copy()
        g = compute_subgrad(corr, h) * dx
        s = h - prev_h
        y = g - prev_g
        sy = float(s @ y)
        yy = float(y @ y)
        if yy > 1e-14 and sy > 1e-14:
            step = float(np.clip(float(s @ s) / sy, 1e-8, 10.0))
        prev_h = h.copy()
        prev_g = g.copy()
        h = project_box_sum(h - step * g, target_sum)
    return best_h, best_f

def binary_local_search(h, n, max_rounds=200):
    h = h.copy()
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_c5 = curr_c5
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] -= 1.0
                h[j] += 1.0
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                h[i] += 1.0
                h[j] -= 1.0
                if new_max < best_c5 - 1e-15:
                    best_c5 = new_max
                    best_swap = (i, j)
        if best_swap is None:
            break
        h[best_swap[0]] -= 1.0
        h[best_swap[1]] += 1.0
        curr_c5 = best_c5
    return h, curr_c5

def slsqp_polish(h0, n, target_sum, dx, betas, maxiters):
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_best = h0.copy()
    c5_best = compute_c5(h_best)
    for beta, maxiter in zip(betas, maxiters):
        def obj(h, _b=float(beta)):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, _b) * dx
        def con_f(x):
            return np.sum(x) - target_sum
        def jac_f(x):
            return np.ones(n)
        try:
            res = minimize(obj, h_best, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_f, 'jac': jac_f}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_box_sum(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best, c5_best = h_new.copy(), c5_new
        except Exception:
            pass
    return h_best, c5_best

def to_binary(h, target_sum):
    h_b = (h > 0.5).astype(float)
    deficit = int(round(target_sum - h_b.sum()))
    if deficit > 0:
        zi = np.where(h_b == 0)[0]
        h_b[zi[:deficit]] = 1.0
    elif deficit < 0:
        oi = np.where(h_b == 1)[0]
        h_b[oi[:deficit]] = 0.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(70, 0.5)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    n_choices = [70, 64, 72, 80]
    for n_points in n_choices:
        if time.time() - start_time > budget_s - 60:
            break
        target_sum = n_points / 2.0
        dx = 2.0 / n_points
        t_n = time.time()
        time_alloc = min(250.0, budget_s - 60 - (time.time() - start_time))
        if len(parent) == n_points:
            h_starts = [parent.copy()]
        else:
            xp = np.linspace(0, 2, len(parent), endpoint=False)
            xn = np.linspace(0, 2, n_points, endpoint=False)
            h_i = np.interp(xn, xp, parent, period=2.0)
            h_starts = [project_box_sum(h_i, target_sum)]
        for _ in range(2):
            h_noisy = h_starts[0] + np.random.randn(n_points) * 0.08
            h_starts.append(project_box_sum(h_noisy, target_sum))
        h_starts.append(to_binary(h_starts[0], target_sum))
        for h0 in h_starts:
            if time.time() - t_n > time_alloc * 0.85:
                break
            h1, _ = projected_subgrad(h0, n_points, target_sum,
                                       max_iter=12000, dx=dx,
                                       time_limit=30, t_start=time.time())
            h2, c5_2 = slsqp_polish(h1, n_points, target_sum, dx,
                                     [80, 200, 500], [300, 500, 400])
            if c5_2 < best_c5:
                best_h, best_c5, best_n = h2.copy(), c5_2, n_points
            h_b = to_binary(h2, target_sum)
            h_b3, c5_b3 = binary_local_search(h_b, n_points, max_rounds=150)
            if c5_b3 < best_c5:
                best_h, best_c5, best_n = h_b3.copy(), c5_b3, n_points
    if best_h is not None and time.time() - start_time < budget_s - 20:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        h_pol, _ = projected_subgrad(best_h, n_f, ts, max_iter=25000, dx=dx_f,
                                      time_limit=60, t_start=time.time())
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol.copy(), c5_pol
        h_fin, c5_fin = slsqp_polish(best_h, n_f, ts, dx_f,
                                      [400, 800, 1500], [600, 800, 600])
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.copy(), c5_fin
        h_b = to_binary(best_h, ts)
        h_b3, c5_b3 = binary_local_search(h_b, n_f, max_rounds=100)
        if c5_b3 < best_c5:
            best_h, best_c5 = h_b3.copy(), c5_b3
    if best_h is None:
        best_h = parent
        best_c5 = compute_c5(parent)
        best_n = len(parent)
    return best_h, best_c5, best_n
