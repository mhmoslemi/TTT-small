import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # array of overlaps for each lag

def active_gradient(h, dx, n):
    """Compute subgradient of C5 at h"""
    overlaps = compute_overlaps(h)
    k_star = np.argmax(overlaps)
    lag = k_star - (n - 1)
    grad = np.ones(n)  # from the Σ h[i] part
    # subtract h[j+lag] for valid j
    for j in range(n):
        jp = j + lag
        if 0 <= jp < n:
            grad[j] -= h[jp]
        jn = j - lag
        if 0 <= jn < n:
            grad[j] -= h[jn]
    return grad * dx, k_star, overlaps

    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta_mom = 0.9

        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break

        h = project_h(h, target_sum)

            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def subgradient(h, n):
    """Compute the subgradient of C5 at h (using the active lag)"""
    overlaps = compute_overlaps(h)
    dx = 2.0 / n
    k_star = np.argmax(overlaps)
    lag = k_star - (n - 1)
    
    # Find valid range
    lo = max(0, -lag)
    hi = min(n, n - lag)
    
    grad = np.zeros(n)
    for i in range(lo, hi):
        grad[i] += (1.0 - h[i + lag])
        grad[i + lag] -= h[i]
    
    return grad * dx, k_star, overlaps

def proj_subgrad_descent(h, target_sum, n, t_start, t_end):
    """Projected subgradient descent with momentum and adaptive step"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta_mom = 0.7
    step = 0.5
    n = len(h)
    
    for it in range(50000):
        if time.time() > t_end:
            break
        
        grad, k_star, overlaps = subgradient(h, n)
        momentum = beta_mom * momentum + grad
        h_new = h - step * momentum
        h_new = project_h(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Occasionally increase step
            if it % 100 == 0 and step < 1.0:
                step *= 1.05
        else:
            step *= 0.95
            if step < 0.01:
                step = 0.01
                # Reset momentum
                momentum = 0.5 * momentum
        
        # Periodically increase momentum
        if it % 1000 == 0:
            beta_mom = min(0.95, 0.7 + it * 0.00003)
    
    return best_h, best_c5

def slsqp_refine(h, target_sum, n, beta, maxiter=5000):
    """SLSQP refinement with smooth max"""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr * dx, beta)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        weights = np.exp(beta * (corr - m))
        weights /= weights.sum()
        # grad of corr*m w.r.t. x
        # corr[m_idx] = sum_i x[i]*(1-x[i+lag]) * dx
        # This is complex, let's just use numerical gradient or skip
        return None  # Let SLSQP handle it numerically
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)
    
    target_sum = n / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    # Phase 1: Subgradient descent with momentum (300s)
    t_phase1_end = time.time() + min(300, budget_s * 0.3)
    h1, c5_1 = proj_subgrad_descent(best_h, target_sum, n, t_start, t_phase1_end)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    
    # Phase 2: SLSQP refinement (150s)
    if time_left() > 100:
        for beta in [100, 300, 1000, 3000]:
            if time_left() < 30:
                break
            t_slsqp_end = time.time() + min(30, time_left() - 5)
            h2, c5_2 = slsqp_refine(best_h, target_sum, n, beta, maxiter=3000)
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2
            # Subgradient again to escape
            if time_left() > 60:
                t_sg_end = time.time() + min(20, time_left() - 10)
                h3, c5_3 = proj_subgrad_descent(best_h, target_sum, n, t_start, t_sg_end)
                if c5_3 < best_c5:
                    best_h, best_c5 = h3, c5_3
    
    # Phase 3: Multi-start (300s)
    if time_left() > 100:
        n_starts = 20
        t_per_start = (time_left() - 50) / n_starts
        for s in range(n_starts):
            if time_left() < 60:
                break
            strat = s % 5
            if strat == 0:
                h0 = best_h + rng.randn(n) * 0.03
            elif strat == 1:
                h0 = best_h.copy()
                idx = rng.choice(n, n//5, replace=False)
                h0[idx] = 1.0 - h0[idx]  # Flip some values
            elif strat == 2:
                # Periodic perturbation
                freq = rng.uniform(1, 5)
                t = np.arange(n) * 2.0 / n
                h0 = best_h + 0.05 * np.sin(2*np.pi*freq*t)
            elif strat == 3:
                # Block perturbation
                nb = rng.randint(2, 5)
                bs = n // nb
                for b in range(nb):
                    h0[b*bs:(b+1)*bs] = best_h[b*bs:(b+1)*bs] + rng.randn(bs)*0.05
            else:
                h0 = best_h + rng.randn(n) * (0.01 + 0.02*rng.random())
            
            h0 = project_h(h0, target_sum)
            
            # Quick subgradient
            t_sg_end = time.time() + t_per_start * 0.6
            h_s, c5_s = proj_subgrad_descent(h0, target_sum, n, t_start, t_sg_end)
            
            # Quick SLSQP
            if time_left() > 40:
                t_sl_end = time.time() + t_per_start * 0.35
                h_s, c5_s = slsqp_refine(h_s, target_sum, n, 500, maxiter=2000)
            
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s
    
    # Phase 4: Final polish (remaining time)
    if time_left() > 20:
        for beta in [5000, 10000]:
            if time_left() < 15:
                break
            h_f, c5_f = slsqp_refine(best_h, target_sum, n, beta, maxiter=10000)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f
    
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

        if 0 <= j - lag < n:
            grad[j] += (1.0 - h[j - lag])
        # Second term: -h[j+lag] if j+lag is in [0, n)
        if 0 <= j + lag < n:

    dx = 2.0 / n
    k_star = np.argmax(overlaps)
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    if lag >= 0:
        # j+lag valid: j < n-lag, so j in [0, n-lag)
        grad[:n-lag] -= h[lag:]
        # j-lag valid: j >= lag, so j in [lag, n)
        grad[lag:] += (1.0 - h[:n-lag])
    else:
        # lag < 0, let l = -lag > 0
        l = -lag
        # j+lag = j-l valid: j >= l, so j in [l, n)
        grad[l:] -= h[:n-l]
        # j-lag = j+l valid: j < n-l, so j in [0, n-l)
        grad[:n-l] += (1.0 - h[l:])
    
    return grad * dx, k_star, overlaps

from scipy.optimize import minimize, Bounds
import time

_g_n = 0
_g_target = 0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        grad[l:] -= h[:n-l]
        grad[:n-l] += (1.0 - h[l:])

    return np.ones(x.shape[0])

    _g_target = target_sum

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_n = 0
_g_beta = 100.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smax_obj(h):
    dx = 2.0 / _g_n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return smooth_max_val(corr, _g_beta)

def smax_grad(h):
    n = _g_n
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    overlaps = raw * dx
    m = np.max(overlaps)
    weights = np.exp(_g_beta * (overlaps - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] -= w * h[lag:]
            grad[lag:] += w * (1.0 - h[:n - lag])
        else:
            l = -lag
            grad[l:] -= w * h[:n - l]
            grad[:n - l] += w * (1.0 - h[l:])
    return grad

def con_fun(x):
    return np.sum(x) - _g_target

def con_jac(x):
    return np.ones(x.shape[0])

def slsqp_refine(h, target_sum, n, beta, maxiter=5000, ftol=1e-14):
    global _g_n, _g_target, _g_beta
    _g_n = n
    _g_target = target_sum
    _g_beta = beta
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(smax_obj, h, jac=smax_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

def subgrad_active(h, n):
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    k_star = int(np.argmax(overlaps))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    else:
        l = -lag
        grad[l:] -= h[:n - l]
        grad[:n - l] += (1.0 - h[l:])
    return grad * dx, k_star

def proj_subgrad(h, target_sum, n, t_limit, step_init=0.3, mom_init=0.7):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta_mom = mom_init
    step = step_init
    it = 0
    while time.time() < t_limit:
        grad, _ = subgrad_active(h, n)
        momentum = beta_mom * momentum + grad
        h_new = h - step * momentum
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            if it % 200 == 0 and step < 1.0:
                step *= 1.04
        else:
            step *= 0.92
            if step < 0.01:
                step = 0.01
                momentum *= 0.5
        it += 1
        if it % 2000 == 0:
            beta_mom = min(0.94, mom_init + it * 0.00001)
    return best_h, best_c5

def gen_perturbation(best_h, n, rng, strat):
