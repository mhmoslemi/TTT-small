import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(s))) / beta

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def time_remaining(t0, budget_s):
    return budget_s - (time.time() - t0)

def obj_smooth(h, n, target, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target) ** 2

def obj_active(h, n, target, k_indices, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    sub = np.array([c[k] for k in k_indices])
    return smooth_max(sub, beta) * (2.0 / n) + lam * (h.sum() - target) ** 2

    best = h.copy()

            r = minimize(obj_smooth, best, args=(n, target, b, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_box_sum(r.x, target)
            c = compute_c5(h_new)
            if c < best_c:

        except Exception:
            pass
    return best, best_c
