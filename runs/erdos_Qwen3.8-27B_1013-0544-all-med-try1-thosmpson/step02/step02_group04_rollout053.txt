    no_improve = 0
    for it in range(max_iter):
        grad = get_gradient(best, beta)
        grad = grad - np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        improved = False
        for trial in range(30):
            candidate = best - step * grad
            candidate = project_h(candidate, target_sum)
            val = compute_c5(candidate)
            if val < best_val - 1e-15:
                best = candidate
                best_val = val
                improved = True
                no_improve = 0
                break
            step *= 0.55
        if not improved:
            no_improve += 1
            step = max(0.0003, step * 0.4)
            if no_improve > 80:
                break
        else:
            step = min(0.08, step * 1.08)
    return best, best_val

def coord_descent(h, target_sum, n_sweeps, base_delta):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    deltas = [base_delta, -base_delta, base_delta * 2.0, -base_delta * 2.0,
              base_delta * 0.5, -base_delta * 0.5, base_delta * 4.0, -base_delta * 4.0]
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            orig = best[i]
            for d in deltas:
                candidate = best.copy()
                candidate[i] = orig + d
                candidate = project_h(candidate, target_sum)
                val = compute_c5(candidate)
                if val < best_val - 1e-14:
                    best = candidate
                    best_val = val
                    improved = True
                    break
        if not improved:
            deltas = [d * 0.5 for d in deltas]
            if abs(deltas[0]) < 1e-9:
                break
    return best, best_val

def try_lbgfsb(h, target_sum, beta, n_points):
    from scipy.optimize import minimize
    dx = 2.0 / n_points
    class Obj:
        def __init__(self):
            pass
        def __call__(self, x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            sm = m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -100, 100)))) / beta
            penalty = 1e6 * (np.sum(x) - target_sum) ** 2
            return sm * dx + penalty
    obj = Obj()
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_points,
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14})
        h_new = project_h(res.x, target_sum)
        val = compute_c5(h_new)
        return h_new, val
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - start_time)

    # Phase 1: Gradient descent with increasing beta
    for beta in [10, 30, 100, 300, 1000]:
        if remaining() < 200:
            break
        h_new, val = gradient_descent(best_h, target_sum, beta, 4000)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Phase 2: L-BFGS-B with penalty
    if remaining() > 150:
        h_new, val = try_lbgfsb(best_h, target_sum, 200, n_points)
        if val < best_c5:
            best_h, best_c5 = h_new, val
        h_new, val = try_lbgfsb(best_h, target_sum, 1000, n_points)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Phase 3: Coordinate descent
    if remaining() > 120:
        h_new, val = coord_descent(best_h, target_sum, 15, 0.01)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Phase 4: More gradient descent at high beta
    for beta in [1000, 3000, 10000]:
        if remaining() < 120:
            break
        h_new, val = gradient_descent(best_h, target_sum, beta, 3000)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Phase 5: Fine coordinate descent
    if remaining() > 100:
        h_new, val = coord_descent(best_h, target_sum, 20, 0.003)
        if val < best_c5:
            best_h, best_c5 = h_new, val
        h_new, val = coord_descent(best_h, target_sum, 20, 0.001)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Phase 6: Perturb + re-optimize loop
    while remaining() > 60:
        scale = 0.001 + 0.004 * (remaining() / budget_s)
        h0 = best_h + rng.randn(n_points) * scale
        h0 = project_h(h0, target_sum)
        h_new, val = gradient_descent(h0, target_sum, 500, 800)
        if val < best_c5:
            best_h, best_c5 = h_new, val
        h_new2, val2 = coord_descent(h_new, target_sum, 5, 0.004)
        if val2 < best_c5:
            best_h, best_c5 = h_new2, val2

    # Phase 7: L-BFGS-B final push
    if remaining() > 40:
        h_new, val = try_lbgfsb(best_h, target_sum, 5000, n_points)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Phase 8: Ultra-fine coordinate descent
    if remaining() > 20:
        h_new, val = coord_descent(best_h, target_sum, 40, 0.0005)
        if val < best_c5:
            best_h, best_c5 = h_new, val
        h_new, val = coord_descent(best_h, target_sum, 40, 0.0002)
        if val < best_c5:
            best_h, best_c5 = h_new, val

    # Final verification
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
