```python
        bases[k] = base

    # Use a simple approach: the optimum is at a breakpoint or boundary
    # Breakpoints occur where two lines intersect: bases[k1]+alphas[k1]*x = bases[k2]+alphas[k2]*x
    # x = (bases[k2]-bases[k1]) / (alphas[k1]-alphas[k2])
    
    # Simpler: just evaluate at many points and also check boundaries
    # For 139 lines, the breakpoints are O(139^2) = ~19000, which is fine
    # But even simpler: ternary search on [0,1] since the function is convex
    
    def f(x):
        vals = bases + alphas * x
        return np.max(vals)
    
    # The function is convex (max of linear functions), so ternary search works
    a, b = 0.0, 1.0
    for _ in range(60):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        if f(m1) <= f(m2):
            b = m2
        else:
            a = m1
    
    x_opt = (a + b) / 2
    return x_opt
```

```python
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    if m < 1e-30:
        return 0.0
    return m * (np.sum((corr / m) ** p)) ** (1.0 / p) * dx

    return None, None  # Let SLSQP use finite differences
```

corr[k] = base[k] + alpha[k] * x + (quad term if lag=0)

Where:
- For lag = k-(n-1):
  - If 0 <= j+lag < n: alpha gets (1-h[j+lag]), base gets full_corr[k] - h[j]*(1-h[j+lag]) (removing the old contribution)
  - If 0 <= j-lag < n: alpha gets -h[j-lag], base gets full_corr[k] - h[j-lag]*(1-h[j])
  - Special case lag=0: the term is h[j](1-h[j]) = h[j] - h[j]^2

Let me vectorize this:

```python
def coord_optimize(h, j):
    n = len(h)
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    h_j = h[j]
    
    alphas = np.zeros(2*n - 1)
    bases = full_corr.copy()
    quad = np.zeros(2*n - 1)
    
    for lag in range(-(n-1), n):
        k = lag + (n - 1)
        # Remove old contributions of h[j] from bases[k]
        # And add the coefficient to alphas[k]
        
        jp = j + lag
        jn = j - lag
        
        if 0 <= jp < n:
            # term: h[j] * (1 - h[jp])
            if jp == j:  # lag = 0
                # h[j]*(1-h[j]) = h[j] - h[j]^2
                bases[k] -= h_j * (1.0 - h_j)
                alphas[k] += 1.0
                quad[k] += -1.0
            else:
                bases[k] -= h_j * (1.0 - h[jp])
                alphas[k] += (1.0 - h[jp])
        
        if 0 <= jn < n and jn != j:  # jn != j means lag != 0
            # term: h[jn] * (1 - h[j])
            bases[k] -= h[jn] * (1.0 - h_j)
            alphas[k] += (-h[jn])
    
    # Now f(x) = max_k (bases[k] + alphas[k]*x + quad[k]*x^2)
    # Sample to find minimum
    xs = np.linspace(0.0, 1.0, 300)
    vals = bases + alphas * xs  # shape (2n-1, 300)
    vals += quad * xs**2
    f_vals = np.max(vals, axis=0)
    
    return xs[np.argmin(f_vals)]
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def lp_obj(x, p, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    if m < 1e-30:
        return 0.0
    return m * (np.sum((corr / m) ** p)) ** (1.0 / p) * dx

def sum_eq(x, ts):
    return np.sum(x) - ts

def sum_eq_jac(x, ts):
    return np.ones(len(x))

def coord_optimize(h, j, n_samples):
    n = len(h)
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    h_j = h[j]
    alphas = np.zeros(2 * n - 1)
    bases = full_corr.copy()
    quad = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        k = lag + (n - 1)
        jp = j + lag
        jn = j - lag
        if 0 <= jp < n:
            if jp == j:
                bases[k] -= h_j * (1.0 - h_j)
                alphas[k] += 1.0
                quad[k] += -1.0
            else:
                bases[k] -= h_j * (1.0 - h[jp])
                alphas[k] += (1.0 - h[jp])
        if 0 <= jn < n and jn != j:
            bases[k] -= h[jn] * (1.0 - h_j)
            alphas[k] += (-h[jn])
    xs = np.linspace(0.0, 1.0, n_samples)
    vals = bases[None, :] + alphas[None, :] * xs[:, None]
    vals += quad[None, :] * (xs[:, None] ** 2)
    f_vals = np.max(vals, axis=0)
    return xs[np.argmin(f_vals)]

def coord_descent_sweep(h, ts, n_samples, n_sweeps):
    n = len(h)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            x_opt = coord_optimize(h, j, n_samples)
            if abs(x_opt - h[j]) > 1e-8:
                h[j] = x_opt
                improved = True
        h = project_h(h, ts)
        if not improved:
            break
    return h

def generate_diverse(n, ts, rng, best_h):
    candidates = []
    candidates.append(best_h.copy())
    for i in range(12):
        if i % 4 == 0:
            c = best_h + rng.randn(n) * 0.04
        elif i % 4 == 1:
            period = rng.randint(3, n // 2 + 1)
            t = np.arange(n)
            c = 0.5 + rng.uniform(0.2, 0.4) * np.sin(2 * np.pi * (t + rng.uniform(0, period)) / period)
        elif i % 4 == 2:
            nb = rng.randint(2, 8)
            bs = n // nb
            c = np.zeros(n)
            for b in range(nb):
                st = b * bs
                en = st + bs if b < nb - 1 else n
                c[st:en] = rng.uniform(0.1, 0.9)
        else:
            c = best_h[::-1] + rng.randn(n) * 0.03
        candidates.append(project_h(c, ts))
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (ts,)}]

    def tl():
        return budget_s - (time.time() - t0)

    best_h = project_h(h_init, ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from multiple starts
    if tl() > 100:
        starts = generate_diverse(n, ts, rng, best_h)
        for start in starts:
            if tl() < 100:
                break
            c = coord_descent_sweep(start.copy(), ts, 400, 20)
            c5 = compute_c5(c)
            if c5 < best_c5:
                best_h, best_c5 = c.copy(), c5

    # Phase 2: L_p SLSQP with increasing p
    for p in [8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192]:
        if tl() < 60:
            break
        try:
            res = minimize(lp_obj, best_h, args=(p, dx), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, ts)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except:
            pass

    # Phase 3: More coordinate descent
    if tl() > 60:
        best_h = coord_descent_sweep(best_h.copy(), ts, 800, 30)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 4: Iterated local search
    attempt = 0
    stall = 0
    while tl() > 50:
        attempt += 1
        h0 = best_h.copy()
        s = attempt % 9

        if s == 0:
            sc = 0.012 + 0.004 * np.exp(-stall / 25.0)
            h0 += rng.randn(n) * sc
        elif s == 1:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            mx = np.max(corr)
            active = np.where(corr > 0.9 * mx)[0]
            for ai in active[:10]:
                lag = ai - (n - 1)
                for j in range(n):
                    jp = j + lag
                    jn = j - lag
                    if 0 <= jp < n:
                        h0[j] -= 0.002 * (1.0 - h0[jp])
                    if 0 <= jn < n:
                        h0[j] += 0.002 * h0[jn]
        elif s == 2:
            i1, i2 = rng.choice(n, 2, replace=False)
            d = 0.05 * rng.randn()
            h0[i1] += d
            h0[i2] -= d
        elif s == 3:
            h0 = 0.93 * h0 + 0.07 * np.mean(h0)
            h0 += rng.randn(n) * 0.005
        elif s == 4:
            h0 = 0.65 * h0 + 0.35 * h0[::-1]
            h0 += rng.randn(n) * 0.007
        elif s == 5:
            spec = np.fft.rfft(h0 - 0.5)
            spec[1:] *= (1.0 + 0.1 * rng.randn(spec[1:].size))
            h0 = np.fft.irfft(spec, n=n) + 0.5
        elif s == 6:
            b1, b2 = sorted(rng.choice(n, 2, replace=False))
            h0[b1:b2 + 1] += rng.randn(b2 - b1 + 1) * 0.02
        elif s == 7:
            for _ in range(rng.randint(1, 4)):
                i1, i2 = rng.choice(n, 2, replace=False)
                tmp = h0[i1]
                h0[i1] = h0[i2] + 0.01 * rng.randn()
                h0[i2] = tmp + 0.01 * rng.randn()
        else:
            h0 += rng.randn(n) * 0.025

        h0 = project_h(h0, ts)

        # Coordinate descent from perturbed start
        h_cand = coord_descent_sweep(h0.copy(), ts, 300, 5)
        c5_cand = compute_c5(h_cand)

        # SLSQP refinement
        p = min(2048, 16 * (1 + attempt // 8))
        try:
            res = minimize(lp_obj, h_cand, args=(p, dx), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_h(res.x, ts)
            c5 = compute_c5(h_new)
        except:
            h_new, c5 = h_cand, c5_cand

        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
            stall = 0
        else:
            stall += 1

    # Phase 5: Final high-precision polish
    if tl() > 15:
        for p in [16384, 32768, 65536]:
            if tl() < 8:
                break
            try:
                res = minimize(lp_obj, best_h, args=(p, dx), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 20000, 'ftol': 1e-16})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except:
                pass

    # Final coordinate descent polish
    if tl() > 5:
        best_h = coord_descent_sweep(best_h.copy(), ts, 1000, 10)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    best_h = project_h(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
