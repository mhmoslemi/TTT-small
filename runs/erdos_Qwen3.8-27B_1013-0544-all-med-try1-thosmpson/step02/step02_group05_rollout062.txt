```python
def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = exp_vals.sum()
    weights = exp_vals / sum_exp
    obj = (m + (max_shifted + np.log(sum_exp)) / beta) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        # term1: (1-h[j+lag]) for valid j
        j_start1 = max(0, -lag)
        j_end1 = min(n, n - lag)  # exclusive
        if j_end1 > j_start1:
            grad[j_start1:j_end1] += w * (1.0 - h[j_start1 + lag : j_end1 + lag])
        # term2: -h[j-lag] for valid j
        j_start2 = max(0, lag)
        j_end2 = min(n, n + lag)  # exclusive
        if j_end2 > j_start2:
            grad[j_start2:j_end2] += w * (-h[j_start2 - lag : j_end2 - lag])
    
    grad *= dx
    return obj, grad
```

```python
def smooth_max_og(h, beta):
    """Return (objective, gradient) for smooth max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    sh = beta * (corr - mv)
    ms = np.max(sh)
    ev = np.exp(sh - ms)
    se = ev.sum()
    wts = ev / se
    obj = (mv + (ms + np.log(se)) / beta) * dx
    grad = np.zeros(n)

        if w < 1e-12:
```

```python
def coordinate_descent_step(h, j, corr):
    """Find optimal h[j] given all other h fixed.
    corr is the current correlation vector."""
    n = len(h)
    # For each m, corr[m] = A_m * h[j] + B_m
    # where A_m = ∂corr[m]/∂h[j] and B_m = corr[m] - A_m * h[j]
    # We want to minimize max_m (A_m * h[j] + B_m) over h[j] ∈ [0,1]
    
    A = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        # ∂corr[m]/∂h[j] = (1-h[j-k])*1_{0<=j-k<n} + (-h[j+k])*1_{0<=j+k<n}
        a = 0.0
        if 0 <= j-k < n:
            a += (1.0 - h[j-k])
        if 0 <= j+k < n:
            a += (-h[j+k])
        A[m] = a
    
    B = corr - A * h[j]
    
    # Now minimize max_m (A[m]*t + B[m]) for t in [0,1]
    # The max of linear functions is convex piecewise linear
    # The minimum is at a point where two lines cross, or at a boundary
    
    # Find the optimal t by checking all breakpoints
    # Breakpoints occur where A[i]*t + B[i] = A[j]*t + B[j]
    # t = (B[j] - B[i]) / (A[i] - A[j])
    
    best_val = float('inf')
    best_t = h[j]
    
    # Check boundaries
    for t_cand in [0.0, 1.0]:
        vals = A * t_cand + B
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t_cand
    
    # Check crossing points
    for i in range(len(A)):
        for j2 in range(i+1, len(A)):
            if abs(A[i] - A[j2]) < 1e-15:
                continue
            t_cross = (B[j2] - B[i]) / (A[i] - A[j2])
            if 0.0 < t_cross < 1.0:
                vals = A * t_cross + B
                v = np.max(vals)
                if v < best_val:
                    best_val = v
                    best_t = t_cross
    
    return best_t
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

_ctarget = [0.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _corr_grad_contrib(h, j, k, n):
    """Contribution to dcord[k]/dh[j] for given j, k."""
    a = 0.0
    jk = j - k
    jpk = j + k
    if 0 <= jk < n:
        a += (1.0 - h[jk])
    if 0 <= jpk < n:
        a += (-h[jpk])
    return a

def smooth_max_og(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    sh = beta * (corr - mv)
    ms = np.max(sh)
    ev = np.exp(sh - ms)
    se = ev.sum()
    wts = ev / se
    obj = (mv + (ms + np.log(se)) / beta) * dx
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = wts[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += -w * h[k:]
        else:
            ak = -k
            grad[:n-ak] += w * (1.0 - h[ak:])
            grad[ak:] += -w * h[:n-ak]
    grad *= dx
    return obj, grad

def sm_obj(h, beta):
    return smooth_max_og(h, beta)[0]

def sm_grad(h, beta):
    return smooth_max_og(h, beta)[1]

def topk_obj(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    return np.mean(corr[idx]) * dx

def topk_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    w = 1.0 / top_k
    for i in idx:
        k = i - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += -w * h[k:]
        else:
            ak = -k
            grad[:n-ak] += w * (1.0 - h[ak:])
            grad[ak:] += -w * h[:n-ak]
    grad *= dx
    return grad

def con_fun(x):
    return np.sum(x) - _ctarget[0]

def con_jac(x):
    return np.ones(len(x))

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

    xn = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xn, xo, h_src)

def coord_descent_round(h, ts, max_rounds=3):
    """Cyclic coordinate descent: optimize each h[j] to minimize max corr, then project."""
    n = len(h)
    h = h.copy()
    for _ in range(max_rounds):
        for j in range(n):
            corr = np.correlate(h, 1.0 - h, mode="full")
            A = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                k = m - (n - 1)
                A[m] = _corr_grad_contrib(h, j, k, n)
            B = corr - A * h[j]
            top_idx = np.argsort(A * h[j] + B)[::-1][:12]
            best_val = np.max(A * h[j] + B)
            best_t = h[j]
            cands = [0.0, 1.0, h[j]]
            for ii in range(len(top_idx)):
                for jj in range(ii + 1, len(top_idx)):
                    da = A[top_idx[ii]] - A[top_idx[jj]]
                    if abs(da) < 1e-12:
                        continue
                    tc = (B[top_idx[jj]] - B[top_idx[ii]]) / da
                    if -1e-9 <= tc <= 1.0 + 1e-9:
                        cands.append(np.clip(tc, 0.0, 1.0))
            for tc in cands:
                tc = np.clip(tc, 0.0, 1.0)
                v = np.max(A * tc + B)
                if v < best_val:
                    best_val = v
                    best_t = tc
            h[j] = best_t
        h = project_h(h, ts)
    return h
```
