import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_active(h, active_lags):
    """
    Compute gradient of sum of corr[m] for m in active_lags.
    active_lags are indices into the full correlation (0 to 2n-2).
    """
    n = len(h)
    grad = np.zeros(n)
    for m in active_lags:
        l = m - (n - 1)  # actual lag
        # ∂corr[m]/∂h[j] = (1-h[j+l])*[0≤j+l<n] - h[j-l]*[0≤j-l<n]
        for j in range(n):
            val = 0.0
            j_plus = j + l
            if 0 <= j_plus < n:
                val += (1.0 - h[j_plus])
            j_minus = j - l
            if 0 <= j_minus < n:
                val -= h[j_minus]
            grad[j] += val
    return grad
