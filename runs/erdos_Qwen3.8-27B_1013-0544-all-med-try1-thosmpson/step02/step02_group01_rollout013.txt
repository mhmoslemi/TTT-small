import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute uniformly first
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, top_k=3):
    """Compute subgradient at the top-k worst shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    j = np.arange(n)
    grad = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    return grad / top_k * dx

def binary_push(h, strength, target_sum):
    """Push h toward binary values."""
    h_new = h.copy()
    # Push values away from 0.5 toward 0 or 1
    for i in range(len(h)):
        if h_new[i] > 0.5:
            h_new[i] = min(1.0, h_new[i] + strength * (1.0 - h_new[i]))
        else:
            h_new[i] = max(0.0, h_new[i] - strength * h_new[i])
    return project_feasible(h_new, target_sum)

def nesterov_subgrad(h, max_iters, lr0, decay, top_k, binary_strength=0.0):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        if binary_strength > 0:
            y = binary_push(y, binary_strength, target)
        grad = subgradient_step(y, top_k)
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    j = np.arange(n)
    grad = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    return grad / top_k * dx

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    corr_shifted = corr - m
    # Numerical stability
    max_shift = np.max(corr_shifted)
    if max_shift > -50:
        corr_shifted = corr_shifted - max_shift
        m_adjusted = m + max_shift
    else:
        m_adjusted = m
    exps = np.exp(beta * corr_shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m_adjusted + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    active = w > 1e-12
    for k in np.where(active)[0]:
        wk = w[k]
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_sg(h, max_iters, lr0, decay, top_k):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        grad = subgrad_topk(y, top_k)
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def local_search_pairs(h, n_swaps=1000):
    """Try random pair swaps to improve the solution."""
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_v = compute_c5(best)
    corr = compute_corr(best)
    max_corr = np.max(corr)
    
    ones = np.where(best > 0.5)[0]
    zeros = np.where(best <= 0.5)[0]
    
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Try swap
        trial = best.copy()
        trial[i] = best[j]
        trial[j] = best[i]
        trial = project_feasible(trial, target)
        v = compute_c5(trial)
        if v < best_v - 1e-10:
            best = trial
            best_v = v
            # Update ones/zeros
            ones = np.where(best > 0.5)[0]
            zeros = np.where(best <= 0.5)[0]
    
    return best, best_v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def tl():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 200
        h_best = np.full(n0, 0.5)
    
    best_c5 = compute_c5(h_best)
    
    # Phase 1: Nesterov subgradient at current resolution
    if tl() > 50:
        h1, c1 = nesterov_sg(h_best, 5000, 0.04, 0.9995, top_k=3)
        if c1 < best_c5:
            h_best, best_c5 = h1, c1
    
    # Phase 2: SLSQP polish
    if tl() > 30:
        h2 = slsqp_refine(h_best, maxiter=800, beta=200)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            h_best, best_c5 = h2, c2
    
    # Phase 3: Higher resolution
    for n_try in [300, 400, 600]:
        if tl() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        if tl() > 80:
            h_n, c_n = nesterov_sg(h_interp, 4000, 0.015, 0.9997, top_k=3)
            if c_n < best_c5:
                h_best, best_c5 = h_n, c_n
                h_interp = h_best.copy()
        if tl() > 40:
            h_s = slsqp_refine(h_interp, maxiter=600, beta=300)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                h_best, best_c5 = h_s, c_s
    
    # Phase 4: Pair swap local search at final resolution
    if tl() > 60:
        # Push toward binary first
        n = len(h_best)
        h_bin = np.where(h_best > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n / 2.0)
        if compute_c5(h_bin) < best_c5:
            h_best, best_c5 = h_bin, best_c5
        
        # Local search
        if tl() > 40:
            h_ls, c_ls = local_search_pairs(h_best, n_swaps=5000)
            if c_ls < best_c5:
                h_best, best_c5 = h_ls, c_ls
    
    # Phase 5: More SLSQP at higher beta
    if tl() > 30:
        h_f = slsqp_refine(h_best, maxiter=1000, beta=500)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f
    
    # Phase 6: Final Nesterov + SLSQP cycle
    if tl() > 40:
        h_p, c_p = nesterov_sg(h_best, 3000, 0.01, 0.9998, top_k=1)
        if c_p < best_c5:

        if tl() > 20:

            c_g = compute_c5(h_g)
            if c_g < best_c5:

**Idea: Use the dual problem**

The primal problem is:
min_h max_k c_k(h) subject to 0 ≤ h ≤ 1, 1^T h = n/2

By introducing a variable t:
min_{h,t} t
subject to c_k(h) ≤ t for all k
         0 ≤ h ≤ 1
         1^T h = n/2

The Lagrangian is:
L(h, t, λ, μ, ν) = t + λ^T (c(h) - t) + μ^T h + ν (1^T h - n/2) - μ'^T (h - 1)

where λ_k ≥ 0 are the dual variables for the constraint c_k(h) ≤ t, and μ, μ' are for the bound constraints.

At optimality, the dual variable λ is a probability distribution over the shifts, and the optimal h minimizes:
∑_k λ_k * c_k(h) - (1 - ∑λ_k) * t

For the dual to be tight, ∑λ_k = 1 (the constraint t ≥ max c_k is active).

So the optimal h minimizes:
∑_k λ_k * c_k(h) = E_λ[c(h)]

This is the expected overlap under the distribution λ. The optimal λ puts weight on the shifts that are "most active" at the optimal h.

This suggests an iterative approach:
1. Given λ, minimize E_λ[c(h)] over h (this is a convex problem if we relax the integer constraint!)
2. Update λ to be the empirical distribution of the worst shifts at the new h
3. Repeat

Wait, is E_λ[c(h)] convex in h?
c_k(h) = ∑_i h_i (1 - h_{i+k}) * dx = dx * (∑_i h_i - ∑_i h_i * h_{i+k})

This is NOT convex in h (because of the -h_i * h_{i+k} term). So the inner problem is still non-convex.

However, if we use the variable a_i = 2h_i - 1 ∈ [-1, 1], then:
c_k(a) = dx * (n-k)/4 - dx/4 * ∑ a_i * a_{i+k} (for balanced a)

The term ∑ a_i * a_{i+k} is a quadratic form, so c_k(a) is a quadratic function of a. Minimizing a weighted sum of quadratics is still non-convex.

OK, so the dual approach doesn't directly help. But it gives insight: the optimal solution is one where the "expected" overlap (under the worst-shift distribution) is minimized.
