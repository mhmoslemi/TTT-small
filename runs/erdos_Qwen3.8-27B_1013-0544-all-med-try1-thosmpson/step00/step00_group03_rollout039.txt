            continue

        # dc[k]/dh[j]: j appears in h[i+k] when i+k=j, i.e., i=j-k (need 0<=j-k<n and 0<=i<n)

        if 0 <= j - k < n:

def compute_smooth_gradient(h, dx, T=0.01):
    n = len(h)
    c = np.correlate(h, 1-h, mode='full')
    # Softmax weights
    c_scaled = c / T
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2*n - 1):
            if w[m] < 1e-10:
                continue
            k = m - (n - 1)
            # Term 1: (1-h[j-k]) if 0 <= j-k < n
            idx1 = j - k
            if 0 <= idx1 < n:
                s += w[m] * (1 - h[idx1])
            # Term 2: -h[j+k] if 0 <= j+k < n
            idx2 = j + k
            if 0 <= idx2 < n:
                s -= w[m] * h[idx2]
        grad[j] = dx * s
    return grad

def project_feasible(h, target_sum):
    n = len(h)
    # Find lambda such that sum(clip(h[i] - lambda, 0, 1)) = target_sum
    lo = h.min() - target_sum / n - 1
    hi = h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        g = np.clip(h - mid, 0, 1)
        s = g.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    g = np.clip(h - mid, 0, 1)
    # Adjust residual
    residual = target_sum - g.sum()
    if abs(residual) > 1e-10:
        # Distribute residual among non-bound variables
        free = (g > 1e-8) & (g < 1 - 1e-8)
        if free.sum() > 0:
            g[free] += residual / free.sum()
            g = np.clip(g, 0, 1)
    return g

import time

def project_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - target_sum / n - 1
    hi = h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        g = np.clip(h - mid, 0, 1)
        s = g.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    g = np.clip(h - (lo + hi) / 2, 0, 1)
    residual = target_sum - g.sum()
    if abs(residual) > 1e-10:
        free = (g > 1e-8) & (g < 1 - 1e-8)
        if free.sum() > 0:
            g[free] += residual / free.sum()
            g = np.clip(g, 0, 1)
    return g

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def compute_smooth_gradient(h, dx, T=0.01):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    c_scaled = c / T
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            if w[m] < 1e-10:
                continue
            k = m - (n - 1)
            idx1 = j - k
            if 0 <= idx1 < n:
                s += w[m] * (1.0 - h[idx1])
            idx2 = j + k
            if 0 <= idx2 < n:
                s -= w[m] * h[idx2]
        grad[j] = dx * s
    return grad

def gradient_descent(h, dx, target_sum, max_time, n_points):
    start_time = time.time()
    lr = 0.05
    T = 0.01
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    patience = 0
    
    n_iter = 0
    while time.time() - start_time < max_time:
        n_iter += 1
        # Adaptive temperature
        T = max(0.001, 0.01 * (0.999 ** n_iter))
        
        grad = compute_smooth_gradient(h, dx, T)
        
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_new = h - step * grad
            h_new = project_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                break
            step *= 0.5
        
        c5_curr = compute_c5(h, dx)
        if c5_curr < best_c5 - 1e-8:
            best_c5 = c5_curr
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 100:
            break
        
        # Decay learning rate
        lr = max(0.001, lr * 0.999)
    
    return best_h, best_c5

def coordinate_descent(h, dx, target_sum, max_time, n_points):
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    n = n_points
    target = target_sum
    
    while time.time() - start_time < max_time:
        improved = False
        for j in range(n):

            # Simpler: use the projection approach

import numpy as np

def project_feasible(h, target_sum):
    """Project h onto {g: 0<=g<=1, sum(g)=target_sum}"""
    n = len(h)
    lo = h.min() - target_sum / n - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        g = np.clip(h - mid, 0.0, 1.0)
        s = g.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    g = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - g.sum()
    if abs(residual) > 1e-12:
        free = (g > 1e-10) & (g < 1.0 - 1e-10)
        if free.sum() > 0:
            g[free] += residual / free.sum()
            g = np.clip(g, 0.0, 1.0)
    return g

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_gradient(h, dx, T):
    """Compute gradient of smoothed max using softmax approximation."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c / T
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / np.sum(exp_c)

            if kk < n:
                grad[:n - kk] += wm * one_minus_h[kk:]
                grad[kk:] += wm * neg_h[:n - kk]

    iter_list = [200, 200, 200, 200, 200, 200, 200, 300]
    for idx, T in enumerate(T_list):
        if remaining < 30.0:
            break
        t_sub = time.time()
        max_it = iter_list[idx]
        h_new = slsqp_optimize(best_h, dx, target_sum, n_points, T, max_it)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
        remaining = budget_s - (time.time() - t_start) - 10.0

    # Phase 2: Gradient descent refinement
    if remaining > 30.0:
        t_sub_start = time.time()
        max_time_phase2 = min(120.0, remaining * 0.5)
        best_h, best_c5 = grad_descent_refine(
            best_h, dx, target_sum, n_points,
            T=0.001, max_iter=20000, init_lr=0.02
        )
        remaining = budget_s - (time.time() - t_start) - 10.0

    # Phase 3: Local search
    if remaining > 15.0:
        time_limit = min(180.0, remaining * 0.6)
        best_h, best_c5 = local_search(
            best_h, dx, target_sum, n_points,
            time_limit + t_start, t_start
        )
        remaining = budget_s - (time.time() - t_start) - 10.0

    # Phase 4: More SLSQP from slightly perturbed starts
    if remaining > 30.0:
        for attempt in range(4):
            if remaining < 25.0:
                break
            pert = 0.05 * np.random.randn(n_points)
            h_try = project_feasible(best_h + pert, target_sum)
            c5_try = compute_c5(h_try, dx)
            start_h = h_try if c5_try < best_c5 else best_h
            c5_start = min(c5_try, best_c5)
            h_opt = slsqp_optimize(start_h, dx, target_sum, n_points, 0.005, 200)
            c5_opt = compute_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            # Continue with gradient descent from best
            h_gd, c5_gd = grad_descent_refine(
                best_h, dx, target_sum, n_points,
                T=0.001, max_iter=3000, init_lr=0.01
            )
            if c5_gd < best_c5:
                best_h = h_gd
                best_c5 = c5_gd
            remaining = budget_s - (time.time() - t_start) - 10.0

    # Final: quick local search to polish
    if remaining > 5.0:
        best_h, best_c5 = local_search(
            best_h, dx, target_sum, n_points,
            budget_s - 3.0, t_start
        )

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n_points
