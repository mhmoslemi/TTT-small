def compute_max_corr(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def fix_sum_binary(h, target):
    n = len(h)
    num_ones = int(np.sum(h))
    if num_ones > target:
        one_pos = np.where(h == 1)[0]
        for idx in np.random.choice(one_pos, num_ones - target, replace=False):
            h[int(idx)] = 0
    elif num_ones < target:
        zero_pos = np.where(h == 0)[0]
        for idx in np.random.choice(zero_pos, target - num_ones, replace=False):
            h[int(idx)] = 1
    return h
