    target_sum = n0 / 2.0

    h_best = project_onto_feasible(h0, target_sum)
    c5_best = c5_value(h_best, dx0)

    # Main optimization on n0 points with increasing alpha
    alphas_main = [15, 30, 50, 80, 120, 200, 300, 500, 800]
    h_best, c5_best = optimize_from_start(h_best, dx0, n0, target_sum,
                                          alphas_main, t_start, t_start + budget_s * 0.4)

    # Random restarts
    if time.time() < t_start + budget_s * 0.6:
        for r in range(4):
            if time.time() > t_start + budget_s * 0.6:
                break
            h_rand = project_onto_feasible(rng.uniform(0, 1, n0), target_sum)
            alphas_r = [30, 60, 100, 200]
            h_r, c5_r = optimize_from_start(h_rand, dx0, n0, target_sum,
                                            alphas_r, t_start, t_start + budget_s * 0.6)
            if c5_r < c5_best:
                h_best = h_r
                c5_best = c5_r

    # Structured starts: block patterns
    if time.time() < t_start + budget_s * 0.7:
        for pattern_seed in [1, 2, 3]:
            if time.time() > t_start + budget_s * 0.7:
                break
            rng_p = np.random.RandomState(pattern_seed + 100)
            # Create a "noisy" structured start
            h_struct = np.zeros(n0)
            n_blocks = rng_p.randint(3, 7)
            block_sizes = rng_p.dirichlet(np.ones(n_blocks) * 0.5) * n0
            pos = 0
            for bs in block_sizes:
                bs_int = max(1, int(round(bs)))
                fill = rng_p.uniform(0.7, 1.0)
                h_struct[int(pos):int(pos) + bs_int] = fill
                pos += bs_int
            h_struct = project_onto_feasible(h_struct, target_sum)
            alphas_s = [40, 80, 150, 300]
            h_s, c5_s = optimize_from_start(h_struct, dx0, n0, target_sum,
                                            alphas_s, t_start, t_start + budget_s * 0.7)
            if c5_s < c5_best:
                h_best = h_s
                c5_best = c5_s

    # Polish with subgradient
    t_polish_end = t_start + budget_s * 0.88
    h_best, c5_best = polish_subgradient(h_best, dx0, n0, target_sum, t_polish_end)

    # Try higher resolution
    n1 = n0 * 2
    dx1 = 2.0 / n1
    target_sum1 = n1 / 2.0
    if time.time() < t_start + budget_s * 0.93:
        x_grid_old = np.arange(n0) * dx0
        x_grid_new = np.arange(n1) * dx1
        # Extend h0 to handle boundary: tile-like extension
        h_ext = np.concatenate([h_best, h_best[:1]])
        h_interp = np.interp(x_grid_new, x_grid_old, h_ext)
        h1 = project_onto_feasible(h_interp, target_sum1)
        alphas_h = [20, 50, 100, 200]
        h1_opt, c5_1 = optimize_from_start(h1, dx1, n1, target_sum1,
                                           alphas_h, t_start, t_start + budget_s * 0.97)
        if c5_1 < c5_best:
            h_best = h1_opt
            c5_best = c5_1
            n0 = n1

    # Final verification
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_onto_feasible(h_best, len(h_best) / 2.0)
    c5_final = c5_value(h_best, 2.0 / len(h_best))

    return h_best, c5_final, len(h_best)
