import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-10:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff >= -1e-10:
                    break
    return h

def grad_corr(h, lag):
    """Gradient of corr at given lag (lag = index - (n-1))."""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        for i in range(n - lag):
            grad[i + lag] += (1.0 - h[i])
            grad[i] -= h[i + lag]
    else:
        t = -lag
        for i in range(n - t):
            grad[i] += (1.0 - h[i + t])
            grad[i + t] -= h[i]
    return grad

def soft_max_gradient(h, beta):
    """Compute soft-max weighted gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Soft max weights
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    # Weighted gradient
    grad = np.zeros(n)
    n_half = n - 1
    for k in range(2 * n - 1):
        lag = k - n_half
        if weights[k] > 1e-6:
            grad += weights[k] * grad_corr(h, lag)
    return grad * dx

            best_h = h2.copy()

    # Phase 3: Binary local search
    if budget_s - (time.time() - t0) > 60:
        hb, c5b = binary_local_search(best_h, rng, 8000, 30)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()
        # Also refine binary with coord opt (allows non-binary)
        if budget_s - (time.time() - t0) > 40:
            h2, c5_2 = coord_opt(hb.copy(), 3, [0.02, 0.05, 0.1])
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 4: Try different n values
    for n_try in [50, 60, 80, 100, 120, 150]:
        if budget_s - (time.time() - t0) < 100:
            break
        n = n_try
        x_old = np.linspace(0, 2, n_p, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, h_p)
        h = project_feasible(h)
        c5_init = compute_c5(h)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h.copy()

        # Gradient descent
        h, c5_h = gradient_descent(h, 30.0, 0.015, 2000, t0, budget_s, 100)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()

        # Coord opt
        if budget_s - (time.time() - t0) > 50:
            h2, c5_2 = coord_opt(h, 2, [0.02, 0.05, 0.1])
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 5: Random restarts on best
    if budget_s - (time.time() - t0) > 80:
        n = len(best_h)
        for restart in range(3):
            if budget_s - (time.time() - t0) < 40:
                break
            h_r = best_h + rng.randn(n) * 0.08
            h_r = project_feasible(h_r)
            h_r, c5_r = gradient_descent(h_r, 50.0, 0.01, 500, t0, budget_s, 40)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final refinement
    if budget_s - (time.time() - t0) > 20:
        h2, c5_2 = coord_opt(best_h.copy(), 4, [0.01, 0.02, 0.05, 0.1])
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
