        contrib = h[shift:shift + max_i + 1] * (1.0 - h[:max_i + 1])
        n_best = int(np.argmax(contrib))
        i = n_best
        j = n_best + shift
        delta = 0.03 * (1.0 + rng.random() * 0.5)
        return make_pair_move(h, i, j, delta, rng)
    else:
        shift_abs = -shift
        max_i = n - 1 - shift_abs
        if max_i < 0:
            return None
        # For negative shift: c_{-k} = sum h[n-k]*(1-h[n])
        # Transfer from n-k to n... or equivalently from n to n+k
        contrib = h[:max_i + 1] * (1.0 - h[shift_abs:shift_abs + max_i + 1])
        n_best = int(np.argmax(contrib))
        i = n_best
        j = n_best + shift_abs
        delta = -0.03 * (1.0 + rng.random() * 0.5)
        return make_pair_move(h, i, j, delta, rng)

def large_perturb(h, target_sum, rng):
    """Larger perturbation for escaping local optima."""
    n = len(h)
    h_new = h.copy()
    n_perturb = max(5, n // 5)
    indices = rng.choice(n, n_perturb, replace=False)
    for idx in indices:
        h_new[idx] = rng.uniform(0, 1)
    return fix_sum(h_new, target_sum, rng)

def upsample(h, new_n, target_sum_new, rng):
    """Upsample h to new_n points using linear interpolation."""
    old_n = len(h)
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    # Create new grid points and interpolate
    old_x = np.arange(old_n) * dx_old
    new_x = (np.arange(new_n) + 0.5) * dx_new
    old_x_mid = (np.arange(old_n) + 0.5) * dx_old
    h_interp = np.interp(new_x, old_x_mid, h)
    h_interp = np.clip(h_interp, 0, 1)
    return fix_sum(h_interp, target_sum_new, rng)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(47) * 0.5
    
    n = len(h)
    target_sum = n / 2.0
    h = fix_sum(h, target_sum, rng)
    
    best_h = h.copy()
    best_score = evaluate_h(h)
    
    # Phase 1: SA at initial resolution
    T = 0.005
    T_min = 1e-8
    cooling_rate = 0.9995
    no_improve = 0
    stall_limit = 3000
    phase1_time = budget_s * 0.5
    
    while time.time() - t0 < t0 + phase1_time:
        r = rng.random()
        h_new = None
        
        if r < 0.55:
            # Random pair swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                delta = rng.uniform(-0.12, 0.12)
                h_new = make_pair_move(h, i, j, delta, rng)
        elif r < 0.80:
            # Targeted move
            h_new = targeted_move(h, rng)
        elif r < 0.90:
            # Multi-pair move
            h_new = h.copy()
            for _ in range(rng.randint(2, 5)):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    delta = rng.uniform(-0.05, 0.05)
                    res = make_pair_move(h_new, i, j, delta, rng)
                    if res is not None:
                        h_new = res
        else:
            # Large perturbation
            h_new = large_perturb(h, target_sum, rng)
        
        if h_new is None:
            continue
        
        score = evaluate_h(h_new)
        delta_s = score - best_score
        
        if delta_s < 0:
            h = h_new
            if score < best_score:
                best_h = h_new.copy()
                best_score = score
                no_improve = 0
            else:
                no_improve += 1
        else:
            if T > T_min and rng.random() < np.exp(-delta_s / T):
                h = h_new
            no_improve += 1
        
        T = max(T * cooling_rate, T_min)
        
        if no_improve > stall_limit:
            # Restart temperature and perturb
            T = 0.005
            no_improve = 0
            h = best_h.copy()
            # Small kick
            for _ in range(3):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    delta = rng.uniform(-0.1, 0.1)
                    res = make_pair_move(h, i, j, delta, rng)
                    if res is not None:
                        h = res
    
    # Phase 2: Try higher resolutions
    remaining = budget_s - (time.time() - t0)
    
    for new_n in [n * 2, n * 3, n * 4]:
        if remaining < 60:
            break
        phase_budget = remaining * 0.35
        new_target = new_n / 2.0
        
        # Upsample best solution
        h_up = upsample(best_h, new_n, new_target, rng)
        # Also try a fresh upsample with perturbation
        h_up2 = h_up.copy()
        for _ in range(new_n // 4):
            i = rng.randint(new_n)
            h_up2[i] = np.clip(h_up2[i] + rng.normal(0, 0.1), 0, 1)
        h_up2 = fix_sum(h_up2, new_target, rng)
        
        candidates = [h_up, h_up2]
        cand_scores = [evaluate_h(candidates[0]), evaluate_h(candidates[1])]
        
        for start_idx in range(2):
            if remaining < 30:
                break
            h_cur = candidates[start_idx].copy()
            cur_score = cand_scores[start_idx]
            local_best = h_cur.copy()
            local_best_score = cur_score
            
            T2 = 0.003
            n2 = new_n
            stall = 0
            phase_end = time.time() + phase_budget
            
            while time.time() < phase_end:
                r = rng.random()
                h_new = None
                if r < 0.6:
                    i = rng.randint(n2)
                    j = rng.randint(n2)
                    if i != j:
                        delta = rng.uniform(-0.08, 0.08)
                        h_new = make_pair_move(h_cur, i, j, delta, rng)
                elif r < 0.85:
                    h_new = targeted_move(h_cur, rng)
                else:
                    i = rng.randint(n2)
                    j = rng.randint(n2)
                    if i != j:
                        delta = rng.uniform(-0.03, 0.03)
                        h_new = make_pair_move(h_cur, i, j, delta, rng)
                
                if h_new is None:
                    continue
                
                score = evaluate_h(h_new)
                delta_s = score - local_best_score
                
                if delta_s < 0:
                    h_cur = h_new
                    if score < local_best_score:
                        local_best = h_new.copy()
                        local_best_score = score
                        stall = 0
                    else:
                        stall += 1
                else:
                    if T2 > 1e-9 and rng.random() < np.exp(-delta_s / T2):
                        h_cur = h_new
                    stall += 1
                
                T2 = max(T2 * 0.9997, 1e-9)
                
                if stall > 2000:
                    T2 = 0.003
                    h_cur = local_best.copy()
                    stall = 0
            
            # Compare with global best (need to compare on same or evaluate properly)
            # The score is computed at different resolutions but represents the same quantity
            if local_best_score < best_score:
                best_h = local_best
                best_score = local_best_score
                n = new_n
        
        remaining = budget_s - (time.time() - t0)
    
    # Final verification
    final_score = evaluate_h(best_h)
    if final_score < best_score:
        best_score = final_score
    
    return best_h, best_score, len(best_h)
