        g[k:] += 1.0 - h[:n-k]

    h = h.astype(float).copy()

    best_h = h.copy()

            best_h = h.copy()
        step *= decay

        else:

        h1, c51 = subgrad_opt(bh, 40000, tgt, si, dc, be)
        if c51 < bc5:
            bc5, bh = c51, h1

    # Phase 2: Smooth L-BFGS-B continuation
    if el() < budget_s * 0.72:
        for tau in [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            if el() > budget_s * 0.72:
                break
            res = minimize(sobj, bh, jac=sgrad, args=(tgt, tau, 500.0),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h2 = project_feasible(res.x, tgt)
            c52 = compute_c5(h2)[0]
            if c52 < bc5:
                bc5, bh = c52, h2

    # Phase 3: Fine subgradient polish
    if el() < budget_s * 0.88:
        h3, c53 = subgrad_opt(bh, 150000, tgt, 0.001, 0.99999, 0.2)
        if c53 < bc5:
            bc5, bh = c53, h3

    # Phase 4: Random and structured restarts
    nr = 0
    while el() < budget_s * 0.97 and nr < 50:
        nr += 1
        r = np.random.rand()
        if r < 0.4:
            h_p = bh + np.random.randn(n) * (0.01 + 0.04 * np.random.rand())
        elif r < 0.6:
            x = np.arange(n) * 2.0 / n
            period = np.random.choice([0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
            amp = 0.1 + 0.1 * np.random.rand()
            phase = np.random.rand() * 2.0 * np.pi
            h_p = 0.5 + amp * np.cos(2.0 * np.pi * x / period + phase)
        elif r < 0.8:
            h_p = 0.5 + 0.15 * np.random.choice([-1.0, 1.0], size=n)
        else:
            h_p = bh * np.random.uniform(0.7, 1.3, size=n)
        h_p = project_feasible(h_p, tgt)
        h4, c54 = subgrad_opt(h_p, 20000, tgt, 0.01, 0.999, 0.5)
        if c54 < bc5:
            bc5, bh = c54, h4
        # Try L-BFGS-B polish on improved solution
        if c54 < bc5 + 0.001 and el() < budget_s * 0.97:
            res = minimize(sobj, h4, jac=sgrad, args=(tgt, 0.01, 500.0),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 200, 'ftol': 1e-14})
            h5 = project_feasible(res.x, tgt)
            c55 = compute_c5(h5)[0]
            if c55 < bc5:
                bc5, bh = c55, h5

    # Final projection and evaluation
    bh = project_feasible(bh, tgt)
    bc5 = compute_c5(bh)[0]
    return bh, bc5, n
