```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def sumexp_objective(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    # log-sum-exp for numerical stability
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def project_onto_constraint(y, S):
    """Project y onto {h : 0<=h[i]<=1, sum(h)=S}"""
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Correct for any tiny numerical error in the sum
    err = S - h.sum()
    if abs(err) > 1e-12:
        # Distribute error among non-bound elements
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if mask.any():
            h[mask] += err / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def projected_gradient(h0, beta=50.0, max_iter=5000, tol=1e-9):
    n = len(h0)
    S = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h)
    lr = 0.1
    
    for it in range(max_iter):
        # Compute objective and gradient
        dx = 2.0 / n
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full') * dx
        m = np.max(corr)
        # Softmax weights
        exp_bc = np.exp(beta * (corr - m))
        p = exp_bc / exp_bc.sum()
        
        # Gradient of log-sum-exp objective
        # grad[i] = sum_k p[k] * dc_k/dh[i]
        # dc_k/dh[i] = dx * ((1-h)[i-k] - h[i+k]) with bounds
        grad = np.zeros(n)
        for i in range(n):
            acc = 0.0
            for k in range(len(p)):
                ik = i - k  # index into (1-h)
                ik2 = i + k  # index into h
                term = 0.0
                if 0 <= ik < n:
                    term += (1 - h[ik])
                if 0 <= ik2 < n:
                    term -= h[ik2]
                acc += p[k] * term
            grad[i] = acc * dx
        
        # Line search
        for _ in range(30):
            h_new = project_onto_constraint(h - lr * grad, S)
            f_new = sumexp_objective(h_new, beta)
            f_old = sumexp_objective(h, beta)
            if f_new < f_old:
                break
            lr *= 0.5
        else:
            break
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        
        if lr < 1e-10:
            break
    
    return h, best_c5
```

- At shift k: ∫ h(x)(1-h(x+k)) dx = measure of {x: h(x)=1, h(x+k)=0, x and x+k both in [0,2]}

For a random 0/1 function with density 1/2:
E[∫ h(x)(1-h(x+k)) dx] ≈ (1/4) * (2-|k|) for |k| ≤ 2

The maximum of (1/4)(2-|k|) over k is at k=0, giving 1/2. But at k=0, it's actually 0 for 0/1 functions!

So the max is achieved at k near 0 but not 0, or at k=1, etc.

For a step function with n=80 samples, the "shift" k takes discrete values. The shift in the correlation is in units of dx = 2/80 = 0.025.

```python
import numpy as np
import time

def get_c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(get_c5_vector(h)))

def smooth_max(c, beta):
    m = np.max(c)
    return m + float(np.log(np.sum(np.exp(beta * (c - m)))) / beta)

def project_onto_constraint(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_c_at_k(h, k):
    """Gradient of c[k] with respect to h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    g = 1.0 - h
    for i in range(n):
        val = 0.0
        ik = i - k
        if 0 <= ik < n:
            val += g[ik]
        ik2 = i + k
        if 0 <= ik2 < n:
            val -= h[ik2]
        grad[i] = val * dx
    return grad

def smooth_gradient(h, beta=50.0):
    """Gradient of smooth_max objective"""
    n = len(h)
    c = get_c5_vector(h)
    m = np.max(c)
    exp_bc = np.exp(beta * (c - m))
    p = exp_bc / exp_bc.sum()
    
    # Compute sum_k p[k] * grad_k
    # grad_k[i] = dx * ((1-h)[i-k] * 1{valid} - h[i+k] * 1{valid})
    dx = 2.0 / n
    g = 1.0 - h
    grad = np.zeros(n)
    
    for i in range(n):
        acc = 0.0
        for k in range(len(p)):
            ik = i - k
            if 0 <= ik < n:
                acc += p[k] * g[ik]
            ik2 = i + k
            if 0 <= ik2 < n:
                acc -= p[k] * h[ik2]
        grad[i] = acc * dx
    return grad

def projected_gradient_optimize(h0, max_iter=5000, beta=50.0, lr_init=0.1):

    S = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iter):

        # Backtracking line search

        for _ in range(30):

            f_new = smooth_max(get_c5_vector(h_new), beta)
            if f_new <= f_old:

                accepted = True

            lr = lr_init
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def get_c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(get_c5_vector(h)))

def smooth_max(c, beta):
    m = np.max(c)
    return m + float(np.log(np.sum(np.exp(beta * (c - m)))) / beta)

def project_onto_constraint(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = get_c5_vector(h)
    m = np.max(c)
    p = np.exp(beta * (c - m))
    p /= p.sum()
    g = 1.0 - h
    i_idx = np.arange(n)
    k_idx = np.arange(len(p))
    ik = i_idx[:, None] - k_idx[None, :]
    ik2 = i_idx[:, None] + k_idx[None, :]
    mask1 = (ik >= 0) & (ik < n)
    mask2 = (ik2 >= 0) & (ik2 < n)
    ik_safe = np.clip(ik, 0, n - 1)
    ik2_safe = np.clip(ik2, 0, n - 1)
    term1 = (p[None, :] * g[ik_safe] * mask1).sum(axis=1)
    term2 = (p[None, :] * h[ik2_safe] * mask2).sum(axis=1)
    return dx * (term1 - term2)

def projected_gradient_optimize(h0, max_iter, beta, lr_init=0.05):
    n = len(h0)
    S = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    last_best = 0
    for it in range(max_iter):
        grad = smooth_gradient(h, beta)
        f_old = smooth_max(get_c5_vector(h), beta)
        accepted = False
        for _ in range(40):
            h_new = project_onto_constraint(h - lr * grad, S)
            f_new = smooth_max(get_c5_vector(h_new), beta)
            if f_new <= f_old + 1e-14:

                accepted = True

        if not accepted:

        if lr < lr_init * 0.05:
            lr = lr_init
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            last_best = it
        elif it - last_best > max_iter // 2:
            break
    return best_h, best_c5
```
