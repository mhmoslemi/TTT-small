import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5_val(h):
    dx = 2.0 / len(h)
    return float(np.max(_corrs(h)) * dx)

def _smooth_max(vals, s):
    m = np.max(vals)
    shifted = np.clip(s * (vals - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / s

def _obj_pen(x, n, s, pw):
    corr = _corrs(x)
    sm = _smooth_max(corr, s)
    d = np.sum(x) - n * 0.5
    return sm + pw * d * d

def _obj_smooth(x, n, s):
    return _smooth_max(_corrs(x), s)

def _feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(1000):
        c = np.sum(h)
        if abs(c - tgt) < 1e-14:
            return h
        if c < tgt:
            m = h < 1.0
            k = int(np.sum(m))
            if k == 0:
                h[:] = tgt / n
                return h
            h[m] = np.minimum(h[m] + (tgt - c) / k, 1.0)
        else:
            m = h > 0.0
            k = int(np.sum(m))
            if k == 0:
                h[:] = tgt / n
                return h
            h[m] = np.maximum(h[m] - (c - tgt) / k, 0.0)
    return h

def _interp(h, nn):
    no = len(h)
    if no == nn:
        return h.copy()
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    xe = np.concatenate([xo - 2.0, xo, xo + 2.0])
    he = np.concatenate([h, h, h])
    return np.interp(xn, xe, he)

def _opt_lbfgs(h_in, n, tlim):
    h = _feasible(h_in.copy(), n)
    bds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    bv = _c5_val(h)
    for s in [20.0, 50.0, 120.0, 300.0, 800.0]:
        if time.time() > tlim:
            break
        r = minimize(_obj_pen, h, method='L-BFGS-B', bounds=bds,
                     args=(n, s, 1e4),
                     options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-14})
        h2 = _feasible(r.x, n)
        v2 = _c5_val(h2)
        if v2 < bv:
            bv = v2
            best = h2.copy()
        h = h2
    return best, bv

def _opt_slsqp(h_in, n, tlim):
    h = _feasible(h_in.copy(), n)
    bds = Bounds(np.zeros(n) + 0.0, np.ones(n) - 0.0)
    ones = np.ones(n)
    lc = LinearConstraint(ones, n * 0.5, n * 0.5)
    best = h.copy()
    bv = _c5_val(h)
    for s in [50.0, 150.0, 400.0]:
        if time.time() > tlim:
            break
        r = minimize(_obj_smooth, h, method='SLSQP', bounds=bds,
                     args=(n, s), constraints=[lc],
                     options={'maxiter': 400, 'ftol': 1e-12})
        h2 = _feasible(r.x, n)
        v2 = _c5_val(h2)
        if v2 < bv:
            bv = v2
            best = h2.copy()
        h = h2
    return best, bv

def _binary_ls(h_init, n, tlim):
    h = (h_init > 0.5).astype(float)
    tgt = n // 2
    n1 = int(np.sum(h))
    if n1 > tgt:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - tgt]] = 0
    elif n1 < tgt:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:tgt - n1]] = 1
    best = h.copy()
    bv = _c5_val(h)
    while time.time() < tlim:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for j in range(i + 1, n):
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
                    v = _c5_val(h)
                    if v < bv - 1e-13:
                        bv = v
                        best = h.copy()
                        improved = True
                    else:
                        h[i], h[j] = h[j], h[i]
                    if time.time() > tlim:
                        return best, bv
        if not improved:
            break
    return best, bv

def _coord_polish(h_in, n, tlim):
    h = _feasible(h_in.copy(), n)
    bv = _c5_val(h)
    candidates = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    for it in range(50):
        if time.time() > tlim:
            break
        improved = False
        for i in range(n):
            orig = h[i]
            best_c = orig
            best_v = bv
            for c in candidates:
                h[i] = c
                v = _c5_val(h)
                if v < best_v:
                    best_v = v
                    best_c = c
            if best_c != orig:
                h[i] = best_c
                improved = True
                bv = best_v
            else:
                h[i] = orig
        if not improved:
            break
    h = _feasible(h, n)
    elif time.time() > tlim:
        break
    bv = _c5_val(h)
    return h, bv

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    dl = t0 + budget_s - 25

    if initial_h_values is not None:

        hs = bh.copy() if nt == len(bh) else _interp(bh, nt)
        nrestart = 10 if nt <= 140 else 5

            if r == 0:

            elif r % 5 == 1:

            elif r % 5 == 2:

                idx = np.random.choice(nt, min(12, nt), replace=False)
                hcur[idx] = 1.0 - hcur[idx]
            elif r % 5 == 3:

                idx = np.random.choice(nt, min(25, nt), replace=False)

            elif r % 5 == 4:

                hcur = hs.copy()
                mask = np.random.random(nt) < 0.3
                hcur[mask] = np.random.random(int(np.sum(mask)))
            hcur = _feasible(hcur, nt)

            ho, vo = _opt_lbfgs(hcur, nt, dl)
            if vo < bc:
                bc = vo
                bh = ho

            if vb < bc:

                bc = vp2
                bh = hp2

    return bh, bc, len(bh)
