def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * 2.0 / n

def hill_climb(h, rng, max_rounds=1000):
    """Best-improvement hill climbing: try all swaps, pick the best."""
    n = len(h)
    h = h.astype(float).copy()
    score = compute_c5(h)
    
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = None
        best_new_score = score
        
        for i in ones:
            for j in zeros:

                h[j] = 1

                h[i] = 1
                h[j] = 0

        h[i] = 0
        h[j] = 1
        score = best_new_score

score(i,j) = max_k [corr(k) + ΔC(i,j,k)]

where ΔC(i,j,k) = -(1-h[i+k]) + h[i-k] + (1-h[j+k]) - h[j-k]

Actually, let me think about it differently. For a fixed i:
- The contribution of removing h[i]=1 to shift k is: -(1-h[i+k]) for the m=i term, +h[i-k] for the m=i-k term
- Let's call this r_i[k] = h[i-k] - (1-h[i+k]) (with out-of-bounds = 0)

For a fixed j:
- The contribution of adding h[j]=1 to shift k is: +(1-h[j+k]) for the m=j term, -h[j-k] for the m=j-k term
- Let's call this a_j[k] = (1-h[j+k]) - h[j-k] (with out-of-bounds = 0)

Then ΔC(i,j,k) = r_i[k] + a_j[k]

And new_corr(i,j,k) = corr[k] + r_i[k] + a_j[k]
new_score(i,j) = max_k new_corr(i,j,k) * 2/n

So I can precompute r_i[k] for all i in ones and a_j[k] for all j in zeros, and then compute all scores.

    corr = np.correlate(h, 1.0 - h, mode='full')
    score = np.max(corr) * dx
    
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Precompute r_i[k] for all i in ones
        # r_i[k] = h[i-k] - (1-h[i+k]) with out-of-bounds = 0
        # We represent shift k as index k+offset in the corr array
        # k ranges from -(n-1) to n-1
        
        shifts = np.arange(-(n-1), n)  # shape (2n-1,)
        
        # For each i in ones, compute r_i
        # r_i[k] = h[i-k] if 0<=i-k<n else 0, minus (1-h[i+k]) if 0<=i+k<n else 0
        r = np.zeros((len(ones), 2*n-1))
        for idx, i in enumerate(ones):
            # h[i-k] for k where 0<=i-k<n, i.e., i-n+1 <= k <= i
            k_valid_1 = (shifts >= (i-n+1)) & (shifts <= i)
            r[idx, k_valid_1] = h[i - shifts[k_valid_1]]
            # minus (1-h[i+k]) for k where 0<=i+k<n, i.e., -i <= k <= n-1-i
            k_valid_2 = (shifts >= (-i)) & (shifts <= (n-1-i))
            r[idx, k_valid_2] -= (1.0 - h[i + shifts[k_valid_2]])
        
        # For each j in zeros, compute a_j
        # a_j[k] = (1-h[j+k]) - h[j-k] with out-of-bounds = 0
        a = np.zeros((len(zeros), 2*n-1))
        for idx, j in enumerate(zeros):
            # (1-h[j+k]) for k where 0<=j+k<n, i.e., -j <= k <= n-1-j
            k_valid_2 = (shifts >= (-j)) & (shifts <= (n-1-j))
            a[idx, k_valid_2] = (1.0 - h[j + shifts[k_valid_2]])
            # minus h[j-k] for k where 0<=j-k<n, i.e., j-n+1 <= k <= j
            k_valid_1 = (shifts >= (j-n+1)) & (shifts <= j)
            a[idx, k_valid_1] -= h[j - shifts[k_valid_1]]
        
        # Compute all scores: score(i,j) = max_k(corr[k] + r[i,k] + a[j,k]) * dx
        # This is a (|ones|, |zeros|, 2n-1) computation
        # scores[np.ix_(ones_idx, zeros_idx)] = max over k
        
        # Use broadcasting: r has shape (n1, 2n-1), a has shape (n0, 2n-1)
        # r[:, None, :] + a[None, :, :] + corr[None, None, :] has shape (n1, n0, 2n-1)
        # Then max over last axis gives (n1, n0)
        
        # For memory efficiency with larger n, do it in chunks
        n1 = len(ones)
        n0 = len(zeros)
        
        # For n=42: 21*21*83 = 36K floats = 288KB. Fine.
        # For n=252: 126*126*503 = 8M floats = 63MB. Might be too much.
        
        if n1 * n0 * (2*n-1) < 50_000_000:
            all_scores = np.max(corr[None, None, :] + r[:, None, :] + a[None, :, :], axis=2) * dx
        else:
            # Fallback to sequential
            all_scores = np.zeros((n1, n0))
            for ii in range(n1):
                for jj in range(n0):
                    all_scores[ii, jj] = np.max(corr + r[ii] + a[jj]) * dx
        
        best_ii, best_jj = np.unravel_index(np.argmin(all_scores), all_scores.shape)
        best_new_score = all_scores[best_ii, best_jj]
        
        if best_new_score >= score - 1e-14:
            break
        
        # Apply the best swap
        i = ones[best_ii]
        j = zeros[best_jj]
        h[i] = 0.0
        h[j] = 1.0
        score = best_new_score
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    return h, score

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * 2.0 / n

def hill_climb_best_fast(h, max_rounds=500):
    n = len(h)
    h = h.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = np.max(corr) * dx
    shifts = np.arange(-(n - 1), n)

    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        r = np.zeros((n1, 2 * n - 1))
        for idx in range(n1):
            i = ones[idx]
            m1 = (shifts >= i - n + 1) & (shifts <= i)
            r[idx, m1] = h[i - shifts[m1]]
            m2 = (shifts >= -i) & (shifts <= n - 1 - i)
            r[idx, m2] -= (1.0 - h[i + shifts[m2]])
        a = np.zeros((n0, 2 * n - 1))
        for idx in range(n0):
            j = zeros[idx]
            m2 = (shifts >= -j) & (shifts <= n - 1 - j)
            a[idx, m2] = (1.0 - h[j + shifts[m2]])
            m1 = (shifts >= j - n + 1) & (shifts <= j)
            a[idx, m1] -= h[j - shifts[m1]]
        all_scores = np.max(corr[None, None, :] + r[:, None, :] + a[None, :, :], axis=2) * dx
        flat_idx = np.argmin(all_scores)
        best_new = all_scores.flat[flat_idx]
        if best_new >= score - 1e-14:
            break
        bi, bj = np.unravel_index(flat_idx, (n1, n0))
        h[ones[bi]] = 0.0
        h[zeros[bj]] = 1.0
        score = best_new
        corr = np.correlate(h, 1.0 - h, mode='full')
    return h, score

def hill_climb_first(h, rng, max_steps=10000):
    n = len(h)
    h = h.astype(float).copy()
    score = compute_c5(h)
    stuck = 0
    for _ in range(max_steps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
        ns = compute_c5(h)
        if ns < score - 1e-14:
            score = ns
            stuck = 0
        else:
            h[i], h[j] = h[j], h[i]
            stuck += 1
            if stuck > 3000:
                break
    return h, score

def upsample(h, target_n, rng):
    n = len(h)
    ratio = max(1, target_n // n)
    h_up = np.repeat(h, ratio)[:target_n]
    target_sum = target_n // 2
    cur_sum = int(h_up.sum())
    diff = target_sum - cur_sum
    if diff > 0:
        idx = np.where(h_up == 0)[0]
        perm = rng.permutation(len(idx))
        h_up[idx[perm[:diff]]] = 1.0
    elif diff < 0:
        idx = np.where(h_up == 1)[0]
        perm = rng.permutation(len(idx))
        h_up[idx[perm[:abs(diff)]]] = 0.0
    return h_up

def do_kick(h, rng, strength):
    h = h.copy()
    for _ in range(strength):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
    return h

def time_remaining(t0, budget_s, fraction):
    return (time.time() - t0) < budget_s * fraction
