import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx

def project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() < target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.any():
            i = np.where(mask)[0][0]
            hp[i] = np.clip(hp[i] + diff, 0.0, 1.0)
    return hp

def smooth_grad(h, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = c.max()
    ec = np.exp(beta * (c - cm))
    w = ec / ec.sum()
    grad = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-9:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] -= wi * h[s:]
        else:
            sp = -s
            grad[:n - sp] += wi * (1.0 - h[sp:])
            grad[sp:] -= wi * h[:n - sp]
    return grad, cm * (2.0 / n)
