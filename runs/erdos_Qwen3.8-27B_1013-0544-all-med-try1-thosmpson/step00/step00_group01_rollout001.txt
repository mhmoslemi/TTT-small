    i_start = max(0, n)

        # grad[i_start:i_end] += 1 - h[j:j + (i_end - i_start)]
        grad[i_start:i_end] += 1.0 - h[j:j + (i_end - i_start)]

    # i ranges from max(0, -n) to min(N-1, N-1-n)
    i_start2 = max(0, -n)
    i_end2 = min(N, N - n)
    if i_start2 < i_end2:
        j2 = i_start2 + n  # = max(0, -n) + n
        grad[i_start2:i_end2] -= h[j2:j2 + (i_end2 - i_start2)]
    
    return grad

    e1 = min(N, N + n_lag)
    if s1 < e1:
