        if s > target: lo = t

    c = np.correlate(h, 1.0-h, mode='full') * dx

from scipy.optimize import minimize

import time

def _eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(x, target):
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - t, 0.0, 1.0))
        if s > target:
            lo = t
        else:
            hi = t
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _smooth_max_c(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def _smoobj(x, dx, beta, lam):
    c = _smooth_max_c(x, dx)
    cm = float(c.max())
    cs = beta * (c - cm)
    ec = np.exp(cs)
    smax = cm + np.log(float(ec.sum())) / beta
    pen = lam * (float(np.sum(x)) * dx - 1.0) ** 2
    return smax + pen

def _smograd(x, dx, beta, lam):
    n = len(x)
    c = _smooth_max_c(x, dx)
    cm = float(c.max())
    cs = beta * (c - cm)
    ec = np.exp(cs)
    w = ec / float(ec.sum())
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[:n - m] += w[k] * dx * (1.0 - x[m:])
            grad[m:] += w[k] * dx * (-x[:n - m])
        else:
            p = -m
            grad[p:] += w[k] * dx * (1.0 - x[:n - p])
            grad[:n - p] += w[k] * dx * (-x[p:])
    pv = 2.0 * lam * (float(np.sum(x)) * dx - 1.0) * dx
    grad += pv
    return grad

            grad[p:] += w[k] * dx * (1.0 - h[:n - p])
            grad[:n - p] += w[k] * dx * (-h[p:])

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(x, target):
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - t, 0.0, 1.0))
        if s > target:
            lo = t
        else:
            hi = t
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _corr_c(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_from_weights(x, w, dx):
    n = len(x)
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[:n - m] += w[k] * dx * (1.0 - x[m:])
            grad[m:] += w[k] * dx * (-x[:n - m])
        else:
            p = -m
            grad[p:] += w[k] * dx * (1.0 - x[:n - p])
            grad[:n - p] += w[k] * dx * (-x[p:])
    return grad

def _smoobj(x, dx, beta, lam):
    c = _corr_c(x, dx)
    cm = float(c.max())
    cs = beta * (c - cm)
    ec = np.exp(cs)
    smax = cm + np.log(float(ec.sum())) / beta
    pen = lam * (float(np.sum(x)) * dx - 1.0) ** 2
    return smax + pen

def _smograd(x, dx, beta, lam):
    n = len(x)
    c = _corr_c(x, dx)
    cm = float(c.max())
    cs = beta * (c - cm)
    ec = np.exp(cs)
    w = ec / float(ec.sum())
    grad = _grad_from_weights(x, w, dx)
    pv = 2.0 * lam * (float(np.sum(x)) * dx - 1.0) * dx
    grad += pv
    return grad

def _pgd(h0, dx, target, t0, tlim, seed=0):

    mom = np.zeros(n)
    ni = 0

        c_raw = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c_raw.max())
        wexp = np.exp(25.0 * (c_raw - cm))
        w = wexp / float(wexp.sum())
        grad = _grad_from_weights(h, w, dx)
        mom = beta_m * mom + (1.0 - beta_m) * grad
        h = _proj(h - alpha * mom, target)
        c5 = _eval(h, dx)
        if c5 < bc:
            bc = c5
            bh = h.copy()
            ni = 0
        else:
            ni += 1
            if ni % 400 == 0:
                alpha *= 0.75
                if alpha < 0.005:
                    alpha = 0.4
                    h = _proj(h + 0.08 * rng.randn(n), target)
                    mom = np.zeros(n)
                    ni = 0
        if it % 20000 == 0 and it > 0:
            alpha = max(alpha * 0.93, 0.01)
    return bh, bc

def _lbgfs_opt(h0, dx, target, beta, lam, t0, tlim):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    f = partial(_smoobj, dx=dx, beta=beta, lam=lam)
    g = partial(_smograd, dx=dx, beta=beta, lam=lam)
    try:
        res = minimize(f, h0.copy(), jac=g, method='L-BFGS-B',
                       bounds=bounds,

        return _proj(res.x, target), _eval(_proj(res.x, target), dx)
    except Exception:
        return h0.copy(), _eval(h0, dx)
