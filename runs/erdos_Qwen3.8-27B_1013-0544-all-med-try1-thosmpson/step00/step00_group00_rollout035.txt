    return np.sum(h) - len(h) / 2.0

def eq_jacobian(h):
    return np.ones(len(h))

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for k in max_idx:
        s = k - (n - 1)
        if s >= 0:
            for j in range(s, n):
                grad[j] += (1.0 - h[j - s]) / len(max_idx)
            for j in range(0, n - s):
                grad[j] -= h[j + s] / len(max_idx)
        else:
            t = -s
            for j in range(0, n - t):
                grad[j] += (1.0 - h[j + t]) / len(max_idx)
            for j in range(t, n):
                grad[j] -= h[j - t] / len(max_idx)
