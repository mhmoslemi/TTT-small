import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_feasible(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_subgradient(h, m_star, n):
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += 1.0 - h[:n - k]
            grad[:n - k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            grad[:n - kk] += 1.0 - h[kk:]
            grad[kk:] -= h[:n - kk]
    return grad

def subgrad_opt(h0, tlim, seed):
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    h = project_onto_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    ss, ni = 0.01, 0
    t0 = time.time()
    while time.time() - t0 < tlim:
        corr = np.correlate(h, 1.0 - h, mode='full')
        ms = int(np.argmax(corr))
        cc = float(corr[ms] * dx)
        if cc < bc:
            bh, bc, ni = h.copy(), cc, 0
        else:
            ni += 1
        g = compute_subgradient(h, ms, n)
        hn = project_onto_feasible(h - ss * g, ts)
        cn = compute_c5(hn, dx)
        if cn < cc:
            h, ss = hn, min(ss * 1.1, 0.03)
        else:
            ss = max(ss * 0.7, 0.0005)
        if ni > 300:
            h = project_onto_feasible(h + 0.03 * rng.randn(n), ts)
            ss, ni = 0.01, 0
    return bh, bc

def smooth_lbfgs(h0, tlim, seed):
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    T_val = 40.0
    mu = 1e4
    x0 = project_onto_feasible(h0.copy(), ts)
    t0 = time.time()

    def obj(x):
        if time.time() - t0 > tlim:
            return 1e10
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sm = m + np.log(np.mean(np.exp(T_val * (corr - m))))
        return sm * dx + mu * (np.sum(x) - ts) ** 2

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        w = np.exp(T_val * (corr - m))
        w /= w.sum()
        g = np.zeros(n)
        active = np.where(w > 1e-7)[0]
        for ki in active:
            g += w[ki] * compute_subgradient(x, int(ki), n)
        return dx * g + 2.0 * mu * (np.sum(x) - ts)

    res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 2000, 'maxfun': 8000, 'iprint': 0})
    h_f = project_onto_feasible(np.clip(res.x, 0, 1), ts)
    return h_f, compute_c5(h_f, dx)

def perturb_search(h0, tlim, seed):
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    h = project_onto_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    scale = 0.05
    t0 = time.time()
    while time.time() - t0 < tlim:
        hn = project_onto_feasible(h + scale * rng.randn(n), ts)
        cn = compute_c5(hn, dx)
        if cn < bc:
            bh, bc, h = hn.copy(), cn, hn
            scale = min(scale * 1.03, 0.1)
        else:
            scale = max(scale * 0.97, 0.003)
            if rng.random() < 0.1:
                h = bh.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    best_n = n0

    def elapsed():
        return time.time() - t_start

    # Phase 1: Subgradient optimization at base resolution

            xo = np.linspace(0.0, 2.0, n0, endpoint=False)

            hi = np.interp(xn, xo, best_h)

    return best_h, c5_final, nf
