import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()

def project_constraint(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) < 1e-10:
        return h
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, t):
    """Compute smooth max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Compute softmax weights
    c_max = c.max()
    shifted = t * (c - c_max)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    weights = exp_shifted / exp_shifted.sum()
    
    val = c_max + (shifted_max + np.log(exp_shifted.sum())) / t
    
    # Compute gradient
    grad = np.zeros(n)
    s = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        s_k = s[k]
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # First term: add w*(1-h[i+s_k]) for i where 0<=i+s_k<n
        i_lo = max(0, -s_k)
        i_hi = min(n, n - s_k)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo:i_hi + s_k])
        # Second term: subtract w*h[i-s_k] for i where 0<=i-s_k<n
        j_lo = max(0, s_k)
        j_hi = min(n, n + s_k)
        if j_lo < j_hi:
            grad[j_lo:j_hi] -= w * h[j_lo - s_k:j_hi - s_k]
    
    return val, grad

def projected_grad_descent(h, target_sum, t_values, max_iter=500, tol=1e-8):
    """Run projected gradient descent with increasing t."""
    best_h = h.copy()
    best_obj = evaluate_c5(h)
    
    for t in t_values:
        h = best_h.copy()
        step = 0.01
        for it in range(max_iter):
            val, grad = smooth_max_and_grad(h, t)
            # Projected gradient step
            h_new = project_constraint(h - step * grad, target_sum)
            new_val, _ = smooth_max_and_grad(h_new, t)
            
            if new_val < val:
                h = h_new
                # True objective
                true_val = evaluate_c5(h)
                if true_val < best_obj:
                    best_obj = true_val
                    best_h = h.copy()
            else:
                step *= 0.5
                if step < 1e-10:

        h = project_constraint(h, target_sum)
        if evaluate_c5(h) < best_obj:
            best_obj = evaluate_c5(h)
            best_h = h.copy()
    
    return best_h, best_obj

                step *= 0.5
                if step < 1e-13:
                    break
        h = project_constraint(h, target_sum)
        tv = evaluate_c5(h)
        if tv < best_obj:
            best_obj = tv
            best_h = h.copy()
    return best_h, best_obj

def threshold_refine(h, target_sum, rng):
    n = len(h)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    for _ in range(200):
        ms = np.sort(h)[::-1]
        if 0 < int(target_sum) < n:
            thr = ms[int(target_sum) - 1]
        else:
            break
        h_bin = (h >= thr).astype(float)
        if h_bin.sum() < target_sum:
            h_bin = np.zeros(n)
            idx = np.argsort(h)[::-1][:int(target_sum)]
            h_bin[idx] = 1.0
        elif h_bin.sum() > target_sum:
            h_bin = np.zeros(n)
            idx = np.argsort(h)[::-1][:int(target_sum)]
            h_bin[idx] = 1.0
        else:
            h_bin[(h > thr).astype(float)] = 1.0
            if h_bin.sum() != target_sum:
                h_bin = np.zeros(n)
                idx = np.argsort(h)[::-1][:int(target_sum)]
                h_bin[idx] = 1.0
        v = evaluate_c5(h_bin)
        if v < best_val:
            best_val = v
            best_h = h_bin.copy()
        if best_val < 0.38:
            break
        h = best_h + rng.normal(0, 0.02, n)
        h = project_constraint(h, target_sum)
    return best_h, best_val

def local_search_binary(h_bin, target_sum, rng, max_iters=5000, deadline=None):
    n = len(h)
    best_val = evaluate_c5(h_bin)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    temp = 0.05
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        v = evaluate_c5(h_new)
        if v < best_val or rng.random() < np.exp((best_val - v) / temp):
            best_h = h_new
            best_val = v
            ones = np.where(best_h == 1.0)[0]
            zeros = np.where(best_h == 0.0)[0]
        temp *= 0.999
        if temp < 0.001:
            temp = 0.001
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    deadline = start + budget_s * 0.92

    n = 80
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        h0 = np.full(n, 0.5)

    ts = n / 2.0
    h0 = project_constraint(h0, ts)

    best_h = h0.copy()
    best_val = evaluate_c5(h0)

    # Phase 1: continuous optimization from initial
    h1, v1 = optimize_h(h0, ts, [8, 15, 30, 60, 120, 250, 500], max_iter=400, deadline=start + budget_s * 0.30)
    if v1 < best_val:
        best_val, best_h = v1, h1

    # Phase 2: threshold + local search
    h2, v2 = threshold_refine(best_h, ts, rng)
    if v2 < best_val:
        best_val, best_h = v2, h2
    if best_h is not None and np.all((best_h == 0) | (best_h == 1)):
        h3, v3 = local_search_binary(best_h, ts, rng, max_iters=20000, deadline=start + budget_s * 0.55)
        if v3 < best_val:
            best_val, best_h = v3, h3

    # Phase 3: perturbations from best
    for trial in range(80):
        if time.time() > deadline:
            break
        scale = 0.08 if trial < 20 else 0.03
        h_try = best_h + rng.normal(0, scale, n)
        h_try = project_constraint(h_try, ts)
        h4, v4 = optimize_h(h_try, ts, [30, 80, 160, 320], max_iter=250, deadline=min(start + budget_s * 0.02 * (trial + 1), deadline))
        if v4 < best_val:
            best_val, best_h = v4, h4

    # Phase 4: binary local search from rounded best
    h_round = project_constraint(best_h, ts)
    thr_v = np.sort(h_round)[::-1]
    if 0 < int(ts) < n:
        thr = thr_v[int(ts) - 1]
    else:
        thr = 0.5
    h_bin = np.zeros(n)
    idx_bin = np.argsort(h_round)[::-1][:int(ts)]
    h_bin[idx_bin] = 1.0
    h5, v5 = local_search_binary(h_bin, ts, rng, max_iters=30000, deadline=deadline)
    if v5 < best_val:
        best_val, best_h = v5, h5

    # Phase 5: try other n
    for n_try in [64, 96, 128, 160]:
        if time.time() > deadline:
            break
        ts_try = n_try / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_constraint(h_interp, ts_try)
        h6, v6 = optimize_h(h_interp, ts_try, [20, 50, 100, 200], max_iter=300, deadline=min(time.time() + budget_s * 0.08, deadline))
        if v6 < best_val:
            best_val, best_h = v6, h6
        # binary search on this n
        hr = np.zeros(n_try)
        ix = np.argsort(h6)[::-1][:int(ts_try)]
        hr[ix] = 1.0
        h7, v7 = local_search_binary(hr, ts_try, rng, max_iters=15000, deadline=deadline)
        if v7 < best_val:
            best_val, best_h = v7, h7

    # Final polish
    for _ in range(5):
        if time.time() > deadline:
            break
        h_try = best_h + rng.normal(0, 0.01, len(best_h))
        h_try = project_constraint(h_try, len(best_h) / 2.0)
        h8, v8 = optimize_h(h_try, len(best_h) / 2.0, [100, 200, 400], max_iter=200, deadline=deadline)
        if v8 < best_val:
            best_val, best_h = v8, h8

    return best_h, best_val, len(best_h)
