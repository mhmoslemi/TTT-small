                h[i] -= remaining
                remaining = 0
                break
            h[i] = 0.0
            remaining -= avail
    if abs(np.sum(h) - target_sum) > 1e-8:
        return None
    return np.clip(h, 0, 1)

def objective_fn(h, dx, temperature, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = smooth_max(corr, temperature)
    c_v = np.sum(h) * dx - 1.0
    return sm + penalty * c_v * c_v

def run_optimize(h_cur, n, dx, bounds, configs, t0, budget_s, margin):
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h, dx)
    for (penalty, temp, maxit) in configs:
        if time.time() - t0 > budget_s - margin:
            break
        try:
            res = minimize(
                objective_fn, h_cur, args=(dx, temp, penalty),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            h_cand = np.clip(res.x, 0, 1)
            h_fix = fix_constraint(h_cand, dx)
            if h_fix is not None:
                c5 = compute_c5(h_fix, dx)
                if c5 < best_c5 - 1e-12:
                    best_h = h_fix.copy()
                    best_c5 = c5
                    h_cur = h_fix.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.asarray(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    h0 = fix_constraint(h0, dx)
    if h0 is None:
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Main optimization configs: (penalty, temperature, maxiter)
    configs = [
        (50, 0.1, 300),
        (100, 0.05, 300),
        (200, 0.03, 400),
        (500, 0.02, 400),
        (500, 0.01, 400),
        (1000, 0.007, 500),
        (2000, 0.004, 500),
        (2000, 0.002, 500),
        (5000, 0.001, 600),
        (5000, 0.0005, 600),
        (10000, 0.0002, 700),
        (10000, 0.0001, 800),
        (20000, 0.00005, 800),
    ]

    h_cur = best_h.copy()
    best_h, best_c5 = run_optimize(h_cur, n, dx, bounds, configs, t0, budget_s, 120)

    # Restarts with perturbation
    for restart in range(4):
        if time.time() - t0 > budget_s - 90:
            break
        scale = 0.02 * (restart + 1)
        h_cur = best_h + np.random.randn(n) * scale
        h_cur = np.clip(h_cur, 0, 1)
        h_fix = fix_constraint(h_cur, dx)
        if h_fix is None:
            continue
        h_cur = h_fix
        result = run_optimize(h_cur, n, dx, bounds, configs[3:], t0, budget_s, 40)
        if result[1] < best_c5 - 1e-12:
            best_h = result[0].copy()
            best_c5 = result[1]

    # Try higher resolution
    for n_try in [90, 180, 360]:
        if time.time() - t0 > budget_s - 60:
            break
        dx_try = 2.0 / n_try
        h_interp = np.zeros(n_try)
        scale = n_try / n
        for i in range(n_try):
            h_interp[i] = best_h[int(i / scale)]
        h_interp = fix_constraint(h_interp, dx_try)
        if h_interp is None:
            continue
        c5_try = compute_c5(h_interp, dx_try)
        if c5_try <= best_c5 + 0.01:
            bounds_try = [(0.0, 1.0)] * n_try
            configs_hi = [(500, 0.01, 300), (1000, 0.005, 400),
                          (2000, 0.002, 400), (5000, 0.001, 500),
                          (5000, 0.0005, 500), (10000, 0.0002, 600)]
            result = run_optimize(h_interp, n_try, dx_try, bounds_try,
                                  configs_hi, t0, budget_s, 30)
            if result[1] < best_c5 - 1e-12:
                best_h = result[0].copy()
                best_c5 = result[1]
                n = n_try
                dx = dx_try
                bounds = bounds_try

    # Coordinate descent polish
    h_work = best_h.copy()
    for it in range(500):
        if time.time() - t0 > budget_s - 5:
            break
        improved = False
        for i in range(n):
            for d in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                v = h_work[i] + d
                if v < -1e-10 or v > 1.0 + 1e-10:
                    continue
                h_test = h_work.copy()
                h_test[i] = np.clip(v, 0, 1)
                h_fix = fix_constraint(h_test, dx)
                if h_fix is None:
                    continue
                c5_test = compute_c5(h_fix, dx)
                if c5_test < best_c5 - 1e-13:
                    best_h = h_fix.copy()
                    best_c5 = c5_test
                    h_work = h_fix.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_h, best_c5, n
