import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max(x, T):
    x_max = np.max(x)
    if T < 1e-8:
        return x_max
    return float(x_max + np.log(np.sum(np.exp((x - x_max) * T))) / T)


def project_constraint(h, target_sum):
    h = h + (target_sum - np.sum(h)) / len(h)
    return np.clip(h, 0.0, 1.0)


def pen_objective(h, T, target_sum, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = smooth_max(corr, T)
    cv = np.sum(h) - target_sum
    return sm + penalty * cv * cv


def coordinate_descent(current, n, target_sum, max_sweeps, t_start, budget_s):
    best_c5 = compute_c5(current)
    for sweep in range(max_sweeps):
        if time.time() - t_start > budget_s - 60:
            break
        improved = False
        for i in range(n):
            best_val = current[i]
            best_obj = best_c5
            for val in np.linspace(0.0, 1.0, 11):
                if val == current[i]:
                    continue
                trial = current.copy()
                trial[i] = val
                trial = project_constraint(trial, target_sum)
                obj_val = compute_c5(trial)
                if obj_val < best_obj - 1e-12:
                    best_obj = obj_val
                    best_val = val
            if best_val != current[i]:
                current[i] = best_val
                current = project_constraint(current, target_sum)
                best_c5 = compute_c5(current)
                improved = True
        if not improved:
            break
    return current, best_c5


def binary_rounding(h, n, target_sum):
    """Try rounding to near-binary solutions."""
    idx = np.argsort(h)
    k_floor = int(np.floor(target_sum))
    k_ceil = int(np.ceil(target_sum))
    candidates = []
    # Round down the smallest k_floor, up the largest n-k_ceil, middle to fractional
    h1 = h.copy()
    h1[idx[:k_floor]] = 0.0
    h1[idx[-(n - k_ceil):]] = 1.0
    h1[idx[k_floor:n - (n - k_ceil)]] = target_sum - k_ceil
    h1 = project_constraint(h1, target_sum)
    candidates.append(h1)
    # Alternative: round up k_ceil smallest to 1, rest to 0, adjust
    h2 = h.copy()
    h2[idx[:k_ceil]] = 1.0
    h2[idx[k_ceil:]] = 0.0
    # Need to reduce sum by (k_ceil - target_sum)
    excess = k_ceil - target_sum
    if excess > 0:
        h2[idx[k_ceil - 1]] = 1.0 - excess
    h2 = project_constraint(h2, target_sum)
    candidates.append(h2)
    best = h.copy()
    best_c5 = compute_c5(h)
    for c in candidates:
        c5 = compute_c5(c)
        if c5 < best_c5:
            best_c5 = c5
            best = c.copy()
    return best, best_c5
