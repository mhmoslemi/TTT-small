The projection onto {h : 0 ≤ h ≤ 1, sum(h) = n/2} is a well-studied problem. It's equivalent to finding the closest point in the hypercube to a point that also lies on the hyperplane sum(h) = n/2.

This can be done by:
1. Find λ such that sum(clip(x - λ, 0, 1)) = n/2
2. Return clip(x - λ, 0, 1)

This is a 1D root-finding problem (bisection on λ).

Let me implement this:

```python
def project_feasible(x, target_sum):
    """Project x onto {h : 0 ≤ h ≤ 1, sum(h) = target_sum}."""
    # Binary search for lambda
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(x - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)
```

This is O(n) per iteration, 60 iterations, so O(60n) total. Very fast.

Now for the soft-max gradient:
f_τ(h) = (1/τ) log(∑_k exp(τ * d_k(h))) where d_k = dx * c_k

∂f_τ/∂h[j] = (1/τ) * (∑_k exp(τ*d_k) * τ * ∂d_k/∂h[j]) / (∑_k exp(τ*d_k))
            = (∑_k p_k * ∂d_k/∂h[j]) where p_k = exp(τ*d_k) / ∑_k exp(τ*d_k)

So the gradient is a weighted average of the individual constraint gradients, with weights proportional to the softmax values. This is efficient to compute.

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_all_d(h):
    """Compute all d_k = dx * correlate[k]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(x, target_sum):
    """Project x onto {h : 0 ≤ h ≤ 1, sum(h) = target_sum}."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(x - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0, 1)

def softmax_grad(h, tau):
    """Compute gradient of soft-max of d_k values."""
    n = len(h)
    dx = 2.0 / n
    d = compute_all_d(h)  # shape (2n-1,)
    
    # Softmax weights
    d_max = np.max(d)
    exp_d = np.exp(tau * (d - d_max))
    p = exp_d / np.sum(exp_d)
    
    # Compute gradient: sum_k p_k * grad(d_k)
    # d_k = dx * c_l where l = k - (n-1)
    # grad(d_l)[j] = dx * [(1-h[j-l])*I(valid) - h[j+l]*I(valid)]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        l = k - (n - 1)
        j = np.arange(n)
        # term 1: (1 - h[j-l]) where 0 <= j-l < n
        jl = j - l
        valid1 = (jl >= 0) & (jl < n)
        t1 = np.zeros(n)
        t1[valid1] = 1.0 - h[jl[valid1]]
        # term 2: h[j+l] where 0 <= j+l < n
        jp = j + l
        valid2 = (jp >= 0) & (jp < n)
        t2 = np.zeros(n)
        t2[valid2] = h[jp[valid2]]
        grad += p[k] * dx * (t1 - t2)
    
    return grad, np.max(d)

def gradient_descent(h_init, t0, budget_s, tau=50.0, lr=0.01):
    """Gradient descent with projection for the soft-max objective."""
    h = h_init.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() - t0 < budget_s - 30:
        grad, c5 = softmax_grad(h, tau)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient step
        h_new = h - lr * grad
        # Project
        h_new = project_feasible(h_new, target_sum)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            # Can decrease learning rate
            lr *= 0.999
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

def slsqp_optimize(h_init, t0, budget_s):
    """SLSQP optimization with auxiliary variable."""
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def objective(x):
        return x[-1]  # minimize t
    
    def objective_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    cons = []
    # Inequality constraints: d_k(h) - t <= 0
    for k in range(2 * n - 1):
        l = k - (n - 1)
        def cons_fun(x, l=l, k=k):
            h = x[:n]
            # c_l = sum_{n in S_l} h[n+l] * (1-h[n])
            # S_l = {n : max(0,-l) <= n < min(n, n-l)}
            n_start = max(0, -l)
            n_end = min(len(h), len(h) - l)
            if n_start >= n_end:
                return -x[-1]  # constraint is -t <= 0, always satisfied
            total = 0.0
            for i in range(n_start, n_end):
                total += h[i + l] * (1.0 - h[i])
            return dx * total - x[-1]
        cons.append({'type': 'ineq', 'fun': cons_fun})
    
    # Equality constraint: sum(h) - target_sum = 0
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    cons.append({'type': 'eq', 'fun': eq_fun})
    
    bounds = [(0, 1)] * n + [(0, 1)]
    x0 = np.append(h_init, compute_c5(h_init))
    
    try:
        res = opt.minimize(objective, x0, jac=objective_grad,
                         method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': 200, 'ftol': 1e-9})
        if res.x[-1] < compute_c5(h_init):
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    
    return h_init, compute_c5(h_init)

def upsample(h, new_n):
    """Upsample h from len(h) to new_n using linear interpolation."""
    n = len(h)
    old_x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    # Actually, the samples are at positions (i+0.5)*dx for i=0..n-1
    # Or at positions i*dx for i=0..n-1? Let's use the midpoint convention.
    # The problem says "n_points samples over [0,2]" with dx = 2/n
    # So samples are at x_i = i * dx for i = 0, ..., n-1
    # But for integration, we use sum(h) * dx = 1
    
    # For upsampling, use linear interpolation
    old_x = np.arange(n) * (2.0 / n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    # Fix the sum
    h_new = project_feasible(h_new, new_n / 2.0)
    return h_new

def simulate_anneal(h_init, t0, budget_s, n_iter=100000):
    """Simulated annealing with swap moves."""
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T = 0.01
    i = 0
    while i < n_iter and time.time() - t0 < budget_s - 30:
        # Try a small perturbation
        idx = np.random.randint(0, n)
        delta = np.random.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        # Fix sum
        diff = target_sum - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Distribute to another index
            idx2 = np.random.randint(0, n)
            while idx2 == idx:
                idx2 = np.random.randint(0, n)
            adj = np.clip(-diff, -(1 - h_new[idx2]), h_new[idx2])
            h_new[idx2] += adj
            # Still might not be exact, use projection
            h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        delta_c = c5_new - compute_c5(h)
        
        if delta_c < 0 or np.random.random() < np.exp(-delta_c / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= 0.9999
        i += 1
    
    return best_h, best_c5

∂f/∂h[j] = dx * ∑_k p_k * [(1-h[j-l_k])*I(valid) - h[j+l_k]*I(valid)]

I can compute this more efficiently by noting that for each j:
∂f/∂h[j] = dx * ∑_k p_k * (1-h[j-l_k]) * I(0≤j-l_k<n) - dx * ∑_k p_k * h[j+l_k] * I(0≤j+l_k<n)

The first sum: for fixed j, we sum over k such that 0 ≤ j-l_k < n, i.e., j-n < l_k ≤ j, i.e., j-n < k-(n-1) ≤ j, i.e., 1 < k ≤ j+n-1.
So k ranges from max(0, j-n+2) to min(2n-2, j+n-1).

The second sum: for fixed j, we sum over k such that 0 ≤ j+l_k < n, i.e., -j < l_k < n-j, i.e., -(j+1) < k-(n-1) < n-j-1, i.e., n-1-j < k < 2n-2-j.
So k ranges from max(0, n-j) to min(2n-2, 2n-2-j).

This is O(n) per j, giving O(n^2) total. Same as before but with clearer structure.

For n=47, this is about 2200 operations. Very fast.

import scipy.optimize as opt

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:

        else:
            hi = mid

def softmax_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    d = corr * dx  # shape (2n-1,)
    
    # Softmax
    d_max = d.max()
    exp_d = np.exp(tau * (d - d_max))
    Z = exp_d.sum()
    p = exp_d / Z  # shape (2n-1,)
    
    # Gradient: df/dh[j] = dx * sum_k p[k] * [(1-h[j-l])*I(0<=j-l<n) - h[j+l]*I(0<=j+l<n)]
    # where l = k - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: sum_k p[k] * (1-h[j-l_k]) where 0 <= j-l_k < n
        # l_k = k - (n-1), so j-l_k = j-k+(n-1)
        # 0 <= j-k+(n-1) < n => k-(n-1) <= j < k-(n-1)+n => k <= j+n-1 and k > j-1
        # k in [max(0, j), min(2n-2, j+n-1)]
        k_lo = max(0, j)
        k_hi = min(2*n-2, j+n-1)
        if k_lo <= k_hi:
            lks = np.arange(k_lo, k_hi+1) - (n-1)
            idx = j - lks  # = j - k + (n-1), should be in [0, n)
            t1 = (p[k_lo:k_hi+1] * (1.0 - h[idx])).sum()
        else:
            t1 = 0.0

def downsample(h, new_n):
    """Downsample by averaging blocks."""
    n = len(h)
    if new_n >= n:
        return h.copy()
    blocks = n // new_n
    h_ds = np.zeros(new_n)
    for i in range(new_n):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    d = corr * dx
    d_max = d.max()
    exp_d = np.exp(tau * (d - d_max))
    Z = exp_d.sum()
    p = exp_d / Z
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        t1 = 0.0
        if k_lo <= k_hi:
            lks = np.arange(k_lo, k_hi + 1) - (n - 1)
            idx = j - lks
            t1 = (p[k_lo:k_hi + 1] * (1.0 - h[idx])).sum()
        k_lo2 = max(0, n - 1 - j)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - j)
        t2 = 0.0
        if k_lo2 <= k_hi2:
            lks2 = np.arange(k_lo2, k_hi2 + 1) - (n - 1)
            idx2 = j + lks2
            t2 = (p[k_lo2:k_hi2 + 1] * h[idx2]).sum()
        grad[j] = dx * (t1 - t2)
    return d.max(), grad

            best_h = h_li.copy()

    n = len(best_h)
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
