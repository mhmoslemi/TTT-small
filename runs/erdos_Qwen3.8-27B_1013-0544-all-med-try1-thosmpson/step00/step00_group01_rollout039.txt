def downscale_h(h, factor, rng):
    """Downscale h by averaging."""
    n = len(h)
    new_n = n // factor
    if new_n < 2:
        return h
    # Average blocks
    padded = np.pad(h, (0, n - new_n * factor), mode='edge')
    new_h = padded[:new_n * factor].reshape(new_n, factor).mean(axis=1)
    new_h = np.clip(new_h, 0, 1)
    # Adjust sum
    target = new_n / 2.0
    current = np.sum(new_h)
    if current > 0:
        new_h *= target / current
        new_h = np.clip(new_h, 0, 1)
    return new_h

    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    stall = 0
    tc = time.time()
    for _ in range(20):
        compute_c5(h)
    spd = 20.0 / max(time.time() - tc, 1e-6)
    nit = int(spd * tlim * 0.7)
    for _ in range(nit):
        if time.time() - t0 > tlim:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        km = np.argmax(c)
        lag = km - (n - 1)
        if lag >= 0:
            contrib = h[lag:] * (1.0 - h[:n - lag])
        else:
            contrib = h[:n + lag] * (1.0 - h[-lag:])
        ii = int(np.argmax(contrib))
        jj = ii + lag
        if jj < 0 or jj >= n or ii == jj:
            i, j = int(rng.integers(n)), int(rng.integers(n))
        else:
            i, j = ii, jj
        dm = min(h[i], 1.0 - h[j])
        dn = -min(h[j], 1.0 - h[i])
        if dm - dn < 1e-12:
            continue
        d = (rng.random() * 2.0 - 1.0) * (dm - dn) * 0.25
        oi, oj = h[i], h[j]
        h[i] = oi - d
        h[j] = oj + d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i], h[j] = oi, oj
            continue
        nc = compute_c5(h)
        if nc < cur - 1e-14:
            cur = nc
            if nc < best:
                best = nc
                best_h = h.copy()
            stall = 0
        else:
            h[i], h[j] = oi, oj
            stall += 1
        if stall > 3000:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    tg = time.time()
    tl = budget_s * 0.93
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf

    starts = []
    try:
        init = np.array(initial_h_values, dtype=float)
        starts.append(normalize_h(init))
    except Exception:
        pass

    for n in [47, 94]:
        starts.append(gen_bin(n, rng))
        for f in [1, 2, 3, 5]:
            starts.append(gen_sine(n, f, 0.4, rng))
        starts.append(gen_bumps(n, 3, rng))
        starts.append(gen_bumps(n, 5, rng))
        starts.append(gen_bumps(n, 2, rng))

    scored = []
    for h in starts:
        c5 = compute_c5(h)
        scored.append((c5, h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    scored.sort()

    nproc = min(10, len(scored))
    for idx in range(nproc):
        if time.time() - tg > tl - 15:
            break
        c5, h = scored[idx]
        n = len(h)
        rem = tl - (time.time() - tg)
        if rem < 10:
            break
        frac = 1.0 / (idx + 2.0)
        ta = max(5.0, min(rem * 0.35, rem * frac))

        h2, c2 = sa_loop(h, ta * 0.55, time.time(), rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

        rem = tl - (time.time() - tg)
        t2 = min(ta * 0.2, rem * 0.25)
        if t2 > 3:
            h3, c3 = ls_loop(h2, t2, time.time(), rng)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

        rem = tl - (time.time() - tg)
        t3 = min(ta * 0.2, rem * 0.2, 25.0)
        if t3 > 3:
            h4, c4 = targeted_ls(h2, t3, time.time(), rng)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

        rem = tl - (time.time() - tg)
        if rem > 15 and best_h is not None and len(best_h) == n:
            hr = best_h + rng.normal(0, 0.015, n)
            hr = normalize_h(hr)
            h5, c5r = sa_loop(hr, min(rem * 0.15, 20.0), time.time(), rng)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h5.copy()

    if best_h is not None:
        rem = tl - (time.time() - tg)
        if rem > 5:
            hf, cf = ls_loop(best_h, rem * 0.5, time.time(), rng)
            if cf < best_c5:
                best_c5 = cf
                best_h = hf
        rem = tl - (time.time() - tg)
        if rem > 5:
            hf2, cf2 = targeted_ls(best_h, rem * 0.4, time.time(), rng)
            if cf2 < best_c5:
                best_c5 = cf2
                best_h = hf2

    if best_h is None:
        best_h = gen_bin(47, rng)
        best_c5 = compute_c5(best_h)

    best_h = normalize_h(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
