import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(h, eps):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = (vals - m) / eps
    return m + eps * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad(h, eps):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = (vals - m) / eps
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    grad = np.zeros(N)
    for mi in range(len(corr)):
        s = mi - (N - 1)
        p = w[mi] * dx / eps
        if p < 1e-14:
            continue
        if s >= 0:
            grad[s:] += p * (1.0 - h[:N - s])
            grad[:N - s] -= p * h[s:]
        else:
            grad[:N + s] += p * (1.0 - h[-s:])
            grad[-s:] -= p * h[:N + s]
    return grad

def projected_gradient(h, N, eps, max_iter, deadline):
    step = 2.0
    best_f = smooth_max_val(h, eps)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, eps)
        h_new = project_feasible(h - step * g, N)
        f_new = smooth_max_val(h_new, eps)
        if f_new < best_f - 1e-12:
            h = h_new
            best_f = f_new
            step = min(step * 1.08, 20.0)
        else:
            step *= 0.5
            if step < 1e-9:
                break
    return h

def coord_descent(h, N, max_rounds, deadline):
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        base_c = compute_c5(h)
        for j in range(N):
            old_val = h[j]
            best_t = old_val
            best_c = base_c
            cands = np.concatenate([[0.0, 1.0], np.linspace(0.05, 0.95, 15)])
            for t in cands:
                h[j] = t
                h = project_feasible(h, N)
                c = compute_c5(h)
                if c < best_c - 1e-11:
                    best_c = c
                    best_t = t
            h[j] = best_t
            h = project_feasible(h, N)
            if best_t != old_val:
                improved = True
    return h

def upsample_h(h_old, N_new):
    N_old = len(h_old)
    x_old = np.arange(N_old) / N_old * 2.0
    x_new = np.arange(N_new) / N_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, N_new)

def make_binary_feasible(N, rng):
    h = np.zeros(N)
    idx = rng.choice(N, size=N // 2, replace=False)
    h[idx] = 1.0
    return h
