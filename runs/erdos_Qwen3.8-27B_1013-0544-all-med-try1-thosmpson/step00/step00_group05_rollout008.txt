def project_to_constraint(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    # Binary search on additive offset
    lo, hi = -np.abs(s - target_sum) - 1.0, np.abs(s - target_sum) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_temp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_temp) < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
