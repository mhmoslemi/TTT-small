    n = len(h)

    return np.max(c) * dx

def make_binary(f, n):
    """Convert a continuous function to binary, maintaining sum constraint."""
    h = (f > 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    target = n // 2
    current = int(h.sum())
    if current > target:
        # Need to flip some 1s to 0s

        np.random.shuffle(indices_1)
        h[indices_1[:current-target]] = 0
    elif current < target:
        indices_0 = np.where(h == 0)[0]
        np.random.shuffle(indices_0)
        h[indices_0[:target-current]] = 1
    return h

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def to_binary(f, n, rng):
    h = (f >= 0.5).astype(float)
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        order = rng.permutation(len(idx))
        h[idx[order[:cur - target]]] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        order = rng.permutation(len(idx))
        h[idx[order[:target - cur]]] = 1
    return h

def interp_to(h_old, n_new, rng):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = np.interp(x_new, x_old, h_old)
    return to_binary(f, n_new, rng)
