        best_c5, best_h = compute_c5(h0), h0.copy()
        oh, oc = optimize_from(h0, n0, sched, 200, t0, budget_s)
        if oc < best_c5:
            best_c5, best_h = oc, oh
    except Exception:
        pass

    for n in [44, 46, 48, 50, 42, 40, 52, 54, 38, 56, 60]:
        if budget_s - (time.time() - t0) < 12:
            break
        for h0 in [gen_random(n, rng), gen_block(n, rng), gen_sine(n, 2, rng),
                   gen_sine(n, 3, rng), gen_sine(n, 5, rng), gen_gauss(n, rng), gen_rect(n, rng)]:
            if budget_s - (time.time() - t0) < 7:
                break
            oh, oc = optimize_from(h0, n, sched, 120, t0, budget_s)
            if oc < best_c5:
                best_c5, best_h = oc, oh

    if budget_s - (time.time() - t0) > 20 and best_h is not None:
        nc = len(best_h)
        if nc % 2 == 0:
            oh, oc = search_flip(best_h, nc, 5000, rng, t0, budget_s)
            if oc < best_c5:
                best_c5, best_h = oc, oh
        oh, oc = search_cont(best_h, nc, 10000, rng, t0, budget_s)
        if oc < best_c5:
            best_c5, best_h = oc, oh

    if budget_s - (time.time() - t0) > 10 and best_h is not None:
        oh, oc = optimize_from(best_h, len(best_h), [500, 2000, 5000], 100, t0, budget_s)
        if oc < best_c5:
            best_c5, best_h = oc, oh

    if best_h is None:
        best_h = gen_block(44, rng)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
