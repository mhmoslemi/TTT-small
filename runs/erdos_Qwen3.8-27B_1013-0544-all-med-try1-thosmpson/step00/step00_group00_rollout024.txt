import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def grad_corr_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        m = -k
        grad[:n - m] += (1 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_obj(x, beta, n, dx):
    corr = correlate(x, 1 - x, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(corr - mc))) / beta) * dx

def smooth_grad(x, beta, n, dx):
    corr = correlate(x, 1 - x, mode='full')
    mc = np.max(corr)
    ec = np.exp(corr - mc)
    p = ec / np.sum(ec)
    grad = np.zeros(n)
    for ki in range(len(corr)):
        if p[ki] < 1e-12:
            continue
        k = ki - (n - 1)
        grad += p[ki] * grad_corr_lag(x, k)
    return grad * dx

def pg_step(h, ts, lr):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    ki = np.argmax(corr)
    c5c = corr[ki] * dx
    k = ki - (n - 1)
    g = grad_corr_lag(h, k)
    hn = project_feasible(h - lr * g, ts)
    c5n = compute_c5(hn)
    return hn, c5n, c5c
