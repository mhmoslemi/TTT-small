import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def subgrad_optimize(h, target, dx, rem_fn, max_iter, m_act, step0, momentum, perturb_every):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        if rem_fn() < 5:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        na = min(m_act, max(5, n // 20))
        top_idx = np.argsort(corr)[::-1][:na]
        lags = [int(i) - (n - 1) for i in top_idx]
        w = np.exp(np.clip(100.0 * (corr[top_idx] - corr[top_idx[0]]), -400, 0))
        w /= w.sum()
        g = np.zeros(n)
        for wi, lag in zip(w, lags):
            g += wi * lag_gradient(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        vel = momentum * vel + (1.0 - momentum) * g
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break
        d = -vel / vn
        step = step0 * np.exp(-no_improve / (max_iter * 0.15))
        found = False
        for _ in range(7):
            h_new = project_h(best + step * d, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best = h_new
                best_c5 = c5_new
                no_improve = 0
                found = True
                break
            step *= 0.5
        if not found:
            no_improve += 1
            if no_improve % perturb_every == perturb_every - 1:
                scale = 0.012 * np.exp(-no_improve / 6000.0)
                p = np.random.randn(n) * scale
                h_p = project_h(best + p, target)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best = h_p
                    best_c5 = c5_p
                    vel *= 0.4
                    no_improve = 0
            if no_improve > perturb_every * 15:
                break
    return best, best_c5

def lbfgs_phase(h, target, dx, rem_fn, betas, penalty, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if rem_fn() < 20:
            break
        def obj_lp(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx + penalty * (h_in.sum() - target) ** 2
        try:
            res = minimize(obj_lp, best, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def slsqp_phase(h, target, dx, rem_fn, betas, maxiter):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if rem_fn() < 25:
            break
        def obj_sp(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con_sp(h_in):
            return h_in.sum() - target
        def jac_sp(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_sp, best, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_sp, 'jac': jac_sp}],
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    n = len(best_h)
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B (fast, find good basin)
    best_h, best_c5 = lbfgs_phase(best_h, target, dx, rem,
                                   [50, 200, 800], 50.0, 300)

    # Phase 2: Subgradient refinement
    best_h, best_c5 = subgrad_optimize(best_h, target, dx, rem,
                                        max_iter=25000, m_act=25,
                                        step0=0.018, momentum=0.85, perturb_every=350)

    # Phase 3: SLSQP polish
    best_h, best_c5 = slsqp_phase(best_h, target, dx, rem,
                                   [500, 2000, 5000], 2000)

    # Phase 4: Subgradient again with smaller steps
    best_h, best_c5 = subgrad_optimize(best_h, target, dx, rem,
                                        max_iter=20000, m_act=30,
                                        step0=0.01, momentum=0.88, perturb_every=500)

    # Phase 5: Try perturbed restart
    if rem() > 80:
        p = np.random.randn(n) * 0.006
        h_pert = project_h(best_h + p, target)
        c5_pert = compute_c5(h_pert)
        h_start = h_pert if c5_pert < best_c5 else best_h.copy()
        h2, c5_2 = subgrad_optimize(h_start, target, dx, rem,
                                     max_iter=15000, m_act=20,
                                     step0=0.012, momentum=0.87, perturb_every=400)
        h2, c5_2 = lbfgs_phase(h2, target, dx, rem, [1000, 3000], 100.0, 200)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 6: Another perturbed restart with different scale
    if rem() > 60:
        p = np.random.randn(n) * 0.015
        h_pert = project_h(best_h + p, target)
        c5_pert = compute_c5(h_pert)
        h_start = h_pert if c5_pert < best_c5 else best_h.copy()
        h3, c5_3 = subgrad_optimize(h_start, target, dx, rem,
                                     max_iter=12000, m_act=25,
                                     step0=0.015, momentum=0.85, perturb_every=300)
        h3, c5_3 = slsqp_phase(h3, target, dx, rem, [2000, 5000], 1500)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3

    # Phase 7: Final fine-tuning subgradient
    if rem() > 15:
        best_h, best_c5 = subgrad_optimize(best_h, target, dx, rem,
                                            max_iter=10000, m_act=35,
                                            step0=0.004, momentum=0.92, perturb_every=800)

    # Phase 8: Final L-BFGS-B + SLSQP
    if rem() > 10:
        best_h, best_c5 = lbfgs_phase(best_h, target, dx, rem, [5000, 10000], 200.0, 200)
    if rem() > 5:
        best_h, best_c5 = slsqp_phase(best_h, target, dx, rem, [8000], 1000)

    return best_h, best_c5, n
