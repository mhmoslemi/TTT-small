import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

where valid_i means 0<=i-lag_k<n (for the first term) etc.

This is O(n) per pair (to compute for all k), and there are O(n²) pairs. So total is O(n³) per sweep. For n=250, that's about 15.6M operations, which is fast (a few ms in numpy if vectorized).

    h, c = subgrad_polish(best_h.copy(), n, p2, max_iter=200000)
    if c < best_c - 1e-14:
        best_c = c
        best_h = h.copy()
    # Phase 3: Try n=200 (10% budget)
    p3 = t0 + 0.60 * (budget_s - 3)
    if time.time() < p3 - 30:
        nh = 200
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h, left=0, right=0), nh)
        for alpha in [1000, 10000, 100000, 1000000, 10000000]:
            if time.time() > p3 - 20:
                break
            hh = slsqp_opt(hh, nh, alpha, 2000, p3 - 20)
            c = compute_c5(hh, nh)
            if c < best_c - 1e-14:
                best_c = c
                best_h = hh.copy()
                n = nh
        if time.time() < p3 - 10:
            hh, c = subgrad_polish(best_h.copy(), n, p3 - 10, max_iter=50000)
            if c < best_c - 1e-14:
                best_c = c
                best_h = hh.copy()
    # Phase 4: Binary local search + simulated annealing (25% budget)
    p4 = t0 + 0.85 * (budget_s - 3)
    p4a = t0 + 0.75 * (budget_s - 3)
    hb, cb = binary_local_search(best_h.copy(), n, p4a, max_sweeps=50000, seed=123)
    if cb < best_c - 1e-14:
        best_c = cb
        best_h = hb.copy()
    # Simulated annealing from current best binary
    if time.time() < p4 - 10:
        ha, ca = simulated_annealing(best_h.copy(), n, p4 - 10,
                                     t_start=0.3, t_end=0.001, max_iters=500000, seed=999)
        if ca < best_c - 1e-14:
            best_c = ca
            best_h = ha.copy()
    # Binary local search after SA
    if time.time() < p4 - 3:
        hb2, cb2 = binary_local_search(best_h.copy(), n, p4 - 3, max_sweeps=20000, seed=456)
        if cb2 < best_c - 1e-14:
            best_c = cb2
            best_h = hb2.copy()
    # Phase 5: Final polish (15% budget)
    if time.time() < dl - 15:
        h = slsqp_opt(best_h.copy(), n, 500000000, 3000, dl - 15)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 5:
        h, c = subgrad_polish(best_h.copy(), n, dl - 5, max_iter=100000)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 2:
        best_h, best_c = pair_refine(best_h, n, dl - 1, [0.01, 0.005, 0.001, 0.0005])
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
