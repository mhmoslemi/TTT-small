    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # Need to adjust a variable at the boundary
            if diff > 0:
                # Need to increase sum, find variable at 0 and increase it
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        else:
            h[free] += diff / nfree
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_grad_focused(h, top_k):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    sum_exp = np.sum(exp_vals)

import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            if diff > 0:
                h[np.argmin(h)] = min(1.0, h[np.argmin(h)] + diff)
            else:
                h[np.argmax(h)] = max(0.0, h[np.argmax(h)] + diff)
        else:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    si = np.argsort(corr)[::-1][:top_k]
    m = np.max(corr[si])
    xs = np.clip(beta * (corr[si] - m), -50, 50)
    ev = np.exp(xs)
    se = np.sum(ev)
    wts = ev / se
    val = (m + np.log(se) / beta) * dx
    grad = np.zeros(n)
    for ii in range(len(si)):
        w = wts[ii] * dx
        k = int(si[ii])
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def _obj(x):
    return smooth_max_val_grad(x, _state['beta'], _state['top_k'])[0]

def _grad(x):
    return smooth_max_val_grad(x, _state['beta'], _state['top_k'])[1]

def _con_eq(x):
    return x.sum() - _state['target']

def _con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, target, betas, maxiter_each, top_k, time_fn, min_time=5):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    _state['target'] = target
    constraints = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < min_time:
            break
        _state['beta'] = beta
        _state['top_k'] = top_k
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                          jac=_grad, constraints=constraints,
                          options={'maxiter': maxiter_each, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def do_lbfgs_auglag(h0, target, time_fn, min_time=10):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    mu = 1.0
    lam = 0.0
    beta = 5000
    top_k = 10

        if time_fn() < min_time:
            break
        def aug_obj(x):
            v = smooth_max_val_grad(x, beta, top_k)[0]
            s = x.sum() - target
            return v + 0.5 * mu * s * s + lam * s
        def aug_grad(x):
            g = smooth_max_val_grad(x, beta, top_k)[1]
            s = x.sum() - target
            g = g + (mu * s + lam) * np.ones(n)
            return g

            res = minimize(aug_obj, h, method='L-BFGS-B', jac=aug_grad,
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-18,
                                                  'gtol': 1e-12, 'maxfun': 2000})
            h = project_h(res.x, target)
        except Exception:
            pass
        s = h.sum() - target
        if abs(s) < 1e-8:
            break
        lam = lam + mu * s
        mu = min(mu * 1.5, 1e5)
    return h, compute_c5(h)

                                3000, 8, tl)

    # Phase 2: Iterated local search with perturbations
    no_improve_count = 0
