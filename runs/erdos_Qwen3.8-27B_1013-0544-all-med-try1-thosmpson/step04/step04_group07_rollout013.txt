    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0, 2, len(h))
    new_x = np.linspace(0, 2, n_new)
    return np.interp(new_x, old_x, h)

def gradient_single_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        la = -lag
        if la < n:
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g

def smooth_max_val(corr, beta):
    m = np.max(corr)
    e = np.exp(np.clip(beta * (corr - m), -500, 500))
    return m + np.log(e.sum() / len(corr)) / beta

def sml_objective(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

    return np.ones(len(x))

        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / (it + 1) ** 2)) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 200 == 0:
                step *= 0.7
    return best_h, best_c5

def frank_wolfe(h0, n, target, n_iter, t0, budget):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = min(40, len(corr))
        top_idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_idx:
            g += grad_single_lag(h, idx - (n - 1))
        g /= top_k
        if np.linalg.norm(g) < 1e-12:
            break
        order = np.argsort(g)
        z = np.zeros(n)
        n_full = int(target)
        z[order[:n_full]] = 1.0
        rem = target - z.sum()
        if rem > 1e-10 and n_full < n:
            z[order[n_full]] = rem
        elif rem < -1e-10 and n_full > 0:
            z[order[n_full - 1]] = 1.0 + rem
        alpha = 2.0 / (it + 2)
        h_new = project_h((1 - alpha) * h + alpha * z, target)
        c = compute_c5(h_new)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h_new.copy(), c, 0
        else:
            stall += 1
        h = h_new
        if stall > 300:
            break
    return best_h, best_c5

def coord_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            for d in (0.04, -0.04, 0.02, -0.02, 0.01, -0.01):
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def pair_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(123)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for _ in range(n // 2):
            if time.time() - t0 > budget - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.04, -0.04, 0.02, -0.02, 0.01, -0.01):
                trial = cur.copy()
                trial[j] += d
                trial[k] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        if len(zeros) >= int(diff):
            z[zeros[np.argsort(h[zeros])[::-1]][:int(diff)]] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        if len(ones) >= int(-diff):
            z[ones[np.argsort(h[ones])][:int(-diff)]] = 0.0
    return z

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    base_n = len(h_parent)

    # Resolutions: parent + higher
    candidates = [base_n, 420, 448, 480, 512, 560, 600, 640, 720, 800]
    resolutions = sorted(set([r for r in candidates if 300 <= r <= 900]))

    for n_pts in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 100:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        # Initialize: resample best
        cur = project_h(resample_h(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)

        # Binary snap + anneal
        hb = snap_binary(cur, target)
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            hb, cb = binary_anneal(hb, target, t0, budget_s, max_iter=2500)
        cb = compute_c5(hb)
        if cb < cur_c5:
            cur, cur_c5 = hb, cb

        # SLSQP with multiple betas
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            for beta in [300, 800, 2000]:
                rem = budget_s - (time.time() - t0)
                if rem < 15:
                    break
                cur = do_slsqp(cur, n_pts, target, beta, bounds, 2500, 1e-13)
                cc = compute_c5(cur)
                if cc < cur_c5:
                    cur_c5 = cc

        # Frank-Wolfe
        rem = budget_s - (time.time() - t0)
        if rem > 35:
            cur_fw, cfw = frank_wolfe(cur, n_pts, target, 4000, t0, budget_s)
            if cfw < cur_c5:
                cur, cur_c5 = cur_fw, cfw

        # Nesterov subgradient
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            cur_n, cn = proj_subgrad(cur, n_pts, target, 6000, t0, budget_s)
            if cn < cur_c5:
                cur, cur_c5 = cur_n, cn

        # SLSQP polish
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur = do_slsqp(cur, n_pts, target, 3000, bounds, 4000, 1e-14)
            cur_c5 = compute_c5(cur)

        # Pair descent
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = pair_descent(cur, target, n_pts, 25, t0, budget_s)

        # Coordinate descent
        rem = budget_s - (time.time() - t0)
        if rem > 15:
            cur, cur_c5 = coord_descent(cur, target, n_pts, 20, t0, budget_s)

        # Update best
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5

    # Final intensive polish on best solution
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [3000, 5000]:
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            best_h = do_slsqp(best_h, nf, tf, beta, bf, 6000, 1e-15)
            best_c5 = compute_c5(best_h)
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            best_h, best_c5 = proj_subgrad(best_h, nf, tf, 4000, t0, budget_s)
        rem = budget_s - (time.time() - t0)
        if rem > 15:
            best_h, best_c5 = pair_descent(best_h, tf, nf, 30, t0, budget_s)
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            best_h, best_c5 = coord_descent(best_h, tf, nf, 40, t0, budget_s)
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            best_h = do_slsqp(best_h, nf, tf, 5000, bf, 5000, 1e-15)
            best_c5 = compute_c5(best_h)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
