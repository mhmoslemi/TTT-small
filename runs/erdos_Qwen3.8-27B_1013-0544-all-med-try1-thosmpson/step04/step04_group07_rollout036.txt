import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        z[zeros[order[:int(diff)]]] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:int(-diff)]]] = 0.0
    return z

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def greedy_swap_search(h, c, n, max_sweeps=50, t0=0, budget=1100):
    """Greedy swap-based local search for binary h."""
    best_c5 = np.max(c)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget - 5:
            break
        improved = False
        for j in ones:
            if time.time() - t0 > budget - 2:
                break
            for k in zeros:
                if j == k:
                    continue
                # Compute delta
                len_j = min(n, 2*n-1-j)
                len_k = min(n, 2*n-1-k)
                new_c = c.copy()
                new_c[j:j+len_j] += 2*h[:len_j] - 1
                new_c[k:k+len_k] -= 2*h[:len_k] - 1
                new_max = np.max(new_c)
                if new_max < best_c5 - 1e-15:
                    c = new_c
                    h[j] = 0
                    h[k] = 1
                    best_c5 = new_max
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
        if not improved:
            break
    
    return h, best_c5 * (2.0/n)
