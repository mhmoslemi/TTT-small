from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, beta):
    """Compute gradient of smooth_max(corr, beta) * dx where corr = correlate(h, 1-h)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    w = np.exp(shifted)
    w /= w.sum()
    
    # V_j = sum_m w[j+m] * h[m] for m=0..n-1
    V = np.correlate(w, h, mode='full')[:n]
    # W_j = sum_m w[j+m] for m=0..n-1
    W = np.correlate(w, np.ones(n), mode='full')[:n]
    
    grad = (1.0 - h) * W - V
    return grad * dx, w

def projected_grad_descent(h, n, deadline, beta=10000, max_iter=200000):
    ts = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.9
    lr = 0.05
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, w = compute_smooth_grad(h, beta)
        
        # Nesterov momentum
        v = mu * v - lr * grad
        h_new = project_feasible(h + v, ts)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    w = np.exp(shifted)
    w /= w.sum()
    V = np.correlate(w, h, mode='full')[:n]
    W = np.correlate(w, np.ones(n), mode='full')[:n]
    grad = (1.0 - h) * W - V
    return grad * dx

def projected_grad_descent(h, n, deadline, beta=10000, max_iter=500000):

    best_h = h.copy()

            break
        grad = compute_smooth_grad(h, beta)
        grad_norm = np.linalg.norm(grad)

            break

            v *= 0.5

    return best_h, best_c5

        pass
