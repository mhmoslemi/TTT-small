import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        g[:m] += (1.0 - h[k:])
        g[k:] -= h[:m]
    else:
        ka = -k
        m = n - ka
        g[ka:] += (1.0 - h[:m])
        g[:m] -= h[ka:]
    return g

def smooth_max_c5(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(np.clip(beta*(corr-m), -500, 500)))/len(corr))/beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def slsqp_pass(h0, n, target, beta, maxiter, t0, budget):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_c5, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except:
        return h0

    h = h0.copy()
    h_prev = h0.copy()

    best_h, best_c5 = h.copy(), compute_c5(h)[0]
    step = 0.04
    patience = 0
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        c5, corr = compute_c5(y)
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g += lag_grad(y, k, n)
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        
        # Backtracking
        best_trial_c5 = best_c5
        best_trial = h.copy()
        for s in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            trial = project_h(y - s * g, target)
            tc = compute_c5(trial)[0]
            if tc < best_trial_c5:
                best_trial, best_trial_c5 = trial.copy(), tc
                break
        
        mu = (1.0 - np.sqrt(1.0 + 4.0/((it+1)**2))) / 2.0
        y = best_trial + mu * (best_trial - h_prev)

        h_prev = h.copy()
        h = best_trial
        
        if best_trial_c5 < best_c5 - 1e-15:
            best_h, best_c5 = h.copy(), best_trial_c5

def pair_descent(h, n, target, n_iters, n_active, t0, budget):
    """Paired coordinate descent with line search on active lags."""
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    rng = np.random.RandomState(42)
    stall = 0
    
    for it in range(n_iters):
        if time.time() - t0 > budget - 2:
            break
        c5, corr = compute_c5(h)
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        j, k = rng.choice(n, 2, replace=False)
        
        # Compute effect on active lags for h[j]+=d, h[k]-=d
        coeffs = np.zeros(n_active)
        for m, idx in enumerate(top_idx):
            kg = int(idx) - (n - 1)
            c = 0.0
            # h[j] contribution
            if 0 <= j + kg < n:
                c += (1.0 - h[j + kg])
            if 0 <= j - kg < n:
                c -= h[j - kg]
            # h[k] contribution (with -d)
            if 0 <= k + kg < n:
                c -= (1.0 - h[k + kg])
            if 0 <= k - kg < n:
                c += h[k - kg]
            coeffs[m] = c
        
        vals = np.array([corr[idx] for idx in top_idx])
        
        # Find best d: minimize max(vals + d*coeffs)
        # Subject to 0 <= h[j]+d <= 1 and 0 <= h[k]-d <= 1
        d_min = max(-h[j], h[k] - 1.0)
        d_max = min(1.0 - h[j], h[k])
        
        if d_max - d_min < 1e-12:
            continue
        
        # Grid search
        best_d = 0
        best_v = best_c5
        grid = np.linspace(d_min, d_max, 15)
        for d in grid:
            v = np.max(vals + d * coeffs)
            if v / (2.0/n) < best_v - 1e-14:
                best_v = v / (2.0/n)
                best_d = d
        
        if best_d != 0:
            h[j] += best_d
            h[k] -= best_d
            h[j] = np.clip(h[j], 0, 1)
            h[k] = np.clip(h[k], 0, 1)

                best_h, best_c5 = h.copy(), c
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 3000:
            break
    
    return best_h, best_c5

        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

    return (m + np.log(np.sum(np.exp(np.clip(beta*(corr-m), -500, 500)))/len(corr))/beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def slsqp_pass(h0, n, target, beta, maxiter, t0, budget):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_c5, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except:
        return h0

def nesterov_opt(h0, n, target, n_iter, n_active, t0, budget):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)[0]
    step = 0.04
    patience = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        c5, corr = compute_c5(y)
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g += lag_grad(y, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        best_trial_c5 = best_c5
        best_trial = h.copy()
        for s in [step, step*0.5, step*0.25, step*0.1]:
            trial = project_h(y - s * g, target)
            tc = compute_c5(trial)[0]
            if tc < best_trial_c5:
                best_trial, best_trial_c5 = trial.copy(), tc
                break
        mu = (1.0 - np.sqrt(1.0 + 4.0/((it+1)**2))) / 2.0
        y = np.clip(best_trial + mu * (best_trial - h_prev), 0, 1)
        h_prev = h.copy()
        h = best_trial
        if best_trial_c5 < best_c5 - 1e-15:
            best_h, best_c5 = h.copy(), best_trial_c5
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                step *= 0.7
            if patience > 2000:
                break
    return best_h, best_c5

def pair_line_search(h, n, target, n_active, n_iters, t0, budget):
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    rng = np.random.RandomState(7)
    stall = 0
    for it in range(n_iters):
        if time.time() - t0 > budget - 2:
            break
        c5, corr = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        top_idx = np.argsort(corr)[::-1][:n_active]
        j, k = rng.choice(n, 2, replace=False)
        vals = np.array([corr[idx] for idx in top_idx])
        coeffs = np.zeros(n_active)
        for m, idx in enumerate(top_idx):
            kg = int(idx) - (n - 1)
            c = 0.0
            if 0 <= j + kg < n: c += (1.0 - h[j + kg])
            if 0 <= j - kg < n: c -= h[j - kg]
            if 0 <= k + kg < n: c -= (1.0 - h[k + kg])
            if 0 <= k - kg < n: c += h[k - kg]
            coeffs[m] = c
        d_min = max(-h[j], h[k] - 1.0)
        d_max = min(1.0 - h[j], h[k])
        if d_max - d_min < 1e-12:
            stall += 1
            continue
        dx = 2.0 / n
        best_d, best_v = 0.0, best_c5
        for d in np.linspace(d_min, d_max, 12):
            v = float(np.max(vals + d * coeffs) * dx)
            if v < best_v - 1e-15:
                best_v, best_d = v, d
        if best_d != 0:
            h[j] = np.clip(h[j] + best_d, 0, 1)
            h[k] = np.clip(h[k] - best_d, 0, 1)
            c = compute_c5(h)[0]
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h.copy(), c
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 5000:
            break
    return best_h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)
