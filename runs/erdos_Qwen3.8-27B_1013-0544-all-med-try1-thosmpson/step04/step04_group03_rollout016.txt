At k = n-1 (center): c[n-1] = sum_i h[i] * (1-h[i]) = 0 for binary h.
At k = n-1+m: c[n-1+m] = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
At k = n-1-m: c[n-1-m] = sum_{i=m}^{n-1} h[i] * (1-h[i-m])

So for binary h, c[n-1+m] counts positions where h[i]=1 and h[i+m]=0 for i=0..n-1-m.

The maximum of this over all m is what we want to minimize.

For a "perfect" binary arrangement where ones and zeros alternate perfectly (101010...), at shift 1: we'd have n-1 positions where h[i]=1 and h[i+1]=0, giving c ≈ n/2 (about 192). That's terrible.

For a "block" arrangement (111...1000...0), at shift n/2: we'd have about n/2 overlaps. Also bad.

The best arrangement is something like a balanced sequence. For example, if we use a Thue-Morse-like arrangement or a "modular" arrangement.

For the modular arrangement: h[i] = 1 if i mod n < n/2. Then at shift m, the overlap is approximately n/2 - m (for m < n/2) and 0 for m ≥ n/2. So max is about n/2 = 192. Bad.

For a "spread out" arrangement: place ones at positions 0, 2, 4, ..., 382 (every other position). Then at shift 1: h[i]=1 and h[i+1]=0 for all i where h[i]=1, giving 192 overlaps. Still bad.

The key insight is that we want the ones to be "as far apart as possible" in a circular sense, but since this is linear (not circular), the boundary effects matter.

Actually, for the linear correlation, the maximum overlap is achieved at small shifts. At shift m, we have n-m pairs to consider. If the density of ones is 1/2, then the expected overlap at shift m is (n-m)/4 (if ones and zeros are independent). So for m=1, expected overlap is (384-1)/4 ≈ 95.75. But we can do better with correlated arrangements.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(400):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_for_lag(h, lag_idx):
    n = len(h)
    grad = np.zeros(n)
    lag = lag_idx - (n - 1)
    if lag >= 0 and lag < n:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n - lag]
    elif lag < 0:
        la = -lag
        if la < n:
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    return grad

def softmax_subgrad(h, target, top_frac, n_iter, step_init, t0, budget, momentum=0.82):

def binary_anneal(h_bin, t0, budget, n_iter=300000, rng=None, multi_move=2):
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h_bin)
    h = h_bin.copy()
    cur_max = float(np.max(compute_lags(h)))
    best_h = h.copy()
    best_max = cur_max
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        k = min(multi_move, len(ones), len(zeros))
        a_idx = rng.choice(ones, size=k, replace=False)
        b_idx = rng.choice(zeros, size=k, replace=False)
        for a in a_idx:
            h[a] = 0.0
        for b in b_idx:
            h[b] = 1.0
        new_max = float(np.max(compute_lags(h)))
        if new_max <= cur_max + 1e-15:
            cur_max = new_max
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_h = h.copy()

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

    n_ones = n // 2

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=10000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def targeted_refine(h, target, t0, budget, n_rounds=300):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for rnd in range(n_rounds):
        if budget - (time.time() - t0) < 4:
            break
        lags = compute_lags(cur)
        k_max = np.argmax(lags)
        grad_k = gradient_for_lag(cur, k_max)
        gn = np.linalg.norm(grad_k)
        if gn < 1e-14:
            break
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            trial = project_h(cur - step * grad_k / gn, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                break
    return cur, best_c5

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.007, -0.007, 0.003, -0.003]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5
