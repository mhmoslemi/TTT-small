This is a pure combinatorial optimization. For n=200, that's choosing 100 out of 200 positions. The search space is C(200,100) which is astronomical, but local search (flipping pairs) should work well.

**Binary local search:**
- State: binary vector with exactly n/2 ones
- Move: flip h[i] from 1 to 0 and h[j] from 0 to 1 (or vice versa)
- This maintains the sum constraint automatically!
- Evaluate: C₅ = max_k (count of i where h[i]=1, h[i+k]=0) * dx

This is much cleaner! And the evaluation is fast.

Let me combine:
1. Start from initial_h_values, round to binary
2. Do binary local search (flip pairs)
3. Interpolate back to continuous (or use the binary solution directly)
4. Do SLSQP polish on the continuous version
5. Repeat

Actually, the binary solution IS a valid step function (taking values 0 and 1), so it's a valid answer. The question is whether continuous values can do better.

For the Erdős problem, I believe the optimal solution IS binary (or very close to it), because the objective is bilinear and the feasible region is a polytope - the optimum of a bilinear function over a polytope is at a vertex, which for the constraints 0≤h[i]≤1 and sum(h)=n/2 means h[i] ∈ {0,1} for all i (well, not exactly a vertex of the polytope, but close).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hp = project_feasible(res.x, n)
            if compute_c5(hp, n) <= compute_c5(h, n) + 1e-12:
                return hp
    except Exception:
        pass
    return h.copy()

def bin_local_search(h_bin, n, deadline, rng, max_rounds=200):
    """Binary local search: flip pair (1->0, 0->1)."""
    h = h_bin.copy().astype(float)
    best_c = compute_c5(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        # Try random pairs for speed
        n_tries = min(5000, len(ones) * len(zeros))
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c = compute_c5(h, n)
            if c < best_c - 1e-14:
                best_c = c
                improved = True
                ones = np.delete(ones, np.where(ones == i)[0][0])
                zeros = np.delete(zeros, np.where(zeros == j)[0][0])
                ones = np.append(ones, j)
                zeros = np.append(zeros, i)
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c

def sa_binary_search(h_bin, n, deadline, rng):
    """Simulated annealing on binary representation."""
    h = h_bin.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = 0.05
    cool = 0.9995
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_steps = 0
    while time.time() < deadline:
        n_steps += 1
        if n_steps % 1000 == 0:
            T *= cool
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h, n)
        delta = c - best_c
        if delta < 1e-14:
            best_c = c
            best_h = h.copy()
            ones = np.append(np.delete(ones, np.where(ones == i)[0][0]), j)
            zeros = np.append(np.delete(zeros, np.where(zeros == j)[0][0]), i)
        elif T > 0.001 and rng.random() < np.exp(-delta / T):
            ones = np.append(np.delete(ones, np.where(ones == i)[0][0]), j)
            zeros = np.append(np.delete(zeros, np.where(zeros == j)[0][0]), i)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def paired_continuous_search(h, n, deadline, rng, steps):
    """Continuous paired local search with adaptive steps."""
    best_c = compute_c5(h, n)
    h = h.copy()
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for _ in range(5000):
                if time.time() > deadline:
                    return h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for d in [step, -step]:
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i], h[j] = hi - d, hj + d
                if improved:
                    break
    return h, best_c

def make_binary_from_continuous(h, n, rng):
    """Convert continuous to binary with exactly n/2 ones."""
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[n // 2 - 1] if n % 2 == 0 else h_sorted[n // 2]
    h_bin = (h >= threshold).astype(float)
    # Handle ties
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        excess = n_ones - n // 2
        tie_idxs = np.where((h_bin == 1) & (h <= threshold + 1e-10))[0]
        if len(tie_idxs) >= excess:
            to_flip = rng.choice(tie_idxs, excess, replace=False)
            h_bin[to_flip] = 0
        else:
            one_idxs = np.where(h_bin == 1)[0]
            to_flip = rng.choice(one_idxs, excess - len(tie_idxs) if len(tie_idxs) < excess else 0, replace=False) if excess > len(tie_idxs) else []
            h_bin[to_flip] = 0
    elif n_ones < n // 2:
        deficit = n // 2 - n_ones
        tie_idxs = np.where((h_bin == 0) & (h >= threshold - 1e-10))[0]
        if len(tie_idxs) >= deficit:
            to_flip = rng.choice(tie_idxs, deficit, replace=False)
            h_bin[to_flip] = 1
        else:
            zero_idxs = np.where(h_bin == 0)[0]
            h_bin[rng.choice(zero_idxs, deficit, replace=False)] = 1
    return h_bin

def optimize_at_n(h_start, n, deadline, rng):
    """Full optimization pipeline for given n."""
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP continuous
    if time.time() < deadline - 10:
        h1 = h.copy()
        for alpha in [2000, 20000, 200000]:
            if time.time() > deadline - 10:

        if c1 < best_c:

            best_h = h1.copy()

    # Phase 2: Binary local search
    if time.time() < deadline - 15:

        h_b, c_b = bin_local_search(h_bin, n, time.time() + alloc_bin, rng)
        if c_b < best_c:

        h_s, c_s = sa_binary_search(h_bin2, n, time.time() + alloc_sa, rng)
