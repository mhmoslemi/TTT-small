import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def compute_lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n_ones]] = 1.0
    return hb

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def obj_and_grad(h, beta, n, pen_fac=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(x)
    e_sum = e.sum()
    alpha = e / e_sum
    val = (m + np.log(e_sum) / beta) * dx
    s = h.sum()
    val += pen_fac * (s - n / 2.0) ** 2
    grad = np.zeros(n)
    active = alpha > 1e-14
    idxs = np.where(active)[0]
    for idx in idxs:
        lag = idx - (n - 1)
        w = alpha[idx]
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    grad *= dx
    grad += 2.0 * pen_fac * (s - n / 2.0)
    return val, grad

def lgbfgs_step(h0, n, beta, pen_fac=50.0, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj_and_grad, h0, args=(beta, n, pen_fac), method='L-BFGS-B',
                     jac=True, bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': -1})
        return project_h(r.x, n / 2.0)
    except Exception:
        return h0

def binary_local_search(h_bin, n, t0, budget, max_iter=80000, rng_seed=777):
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_lags = np.correlate(cur, 1.0 - cur, mode="full")
    cur_max = cur_lags.max() * dx
    best_h = cur.copy()
    best_max = cur_max
    rng = np.random.default_rng(rng_seed)
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > budget - 5:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        cur[a] = 0.0
        cur[b] = 1.0
        new_lags = np.correlate(cur, 1.0 - cur, mode="full")
        new_max = new_lags.max() * dx
        if new_max < cur_max:
            cur_max = new_max
            cur_lags = new_lags
            if new_max < best_max:
                best_max = new_max
                best_h = cur.copy()
            stall = 0
        else:
            cur[b] = 0.0
            cur[a] = 1.0
            stall += 1
            if stall > 5000:
                break
    return best_h, best_max

def subgrad_topk(h, target, n, n_iter, step_init, t0, budget, top_k=12, momentum=0.9):

    mom = np.zeros(n)
    best_local = compute_c5(cur)
    step = step_init
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[lag:] += (1.0 - cur[:n - lag])
                    grad[:n - lag] -= cur[lag:]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = momentum * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.05, 0.08)
        else:
            stall += 1
            step *= 0.7

                best_local = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:

    return cur, best_local

def coord_polish(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004]
    for sw in range(max_sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-13:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5
