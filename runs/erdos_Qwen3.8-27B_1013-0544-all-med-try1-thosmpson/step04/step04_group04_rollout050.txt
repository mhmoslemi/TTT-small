import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # shape (2n-1,)

def compute_grad_overlap(h, k_index):
    """Gradient of overlap at lag index k_index (0 to 2n-2)."""
    n = len(h)
    dx = 2.0 / n
    lag = int(k_index) - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif 0 < lag < n:
        grad[lag:] += (1.0 - h[:-lag]) * dx
        grad[:n-lag] -= h[lag:] * dx
    elif lag < 0 and abs(lag) < n:
        p = -lag
        grad[p:] += (1.0 - h[:-p]) * dx
        grad[:n-p] -= h[p:] * dx
    return grad

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_sum_obj_and_grad(h, lam):
    """Compute weighted sum of overlaps and its gradient."""
    n = len(h)

    overlaps = compute_overlaps(h)
    val = np.dot(lam, overlaps)

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif 0 < lag < n:
        g[lag:] += (1.0 - h[:-lag]) * dx
        g[:n-lag] -= h[lag:] * dx
    elif -n < lag < 0:
        p = -lag
        g[p:] += (1.0 - h[:-p]) * dx
        g[:n-p] -= h[p:] * dx
    return g

def topk_grad(h, k):
    n = len(h)
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:k]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if abs(lag) < n:
            g += grad_for_lag(h, lag)
    return g / k  # average

        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

             'jac': lambda x: np.ones(n)}]

            break

            return m + np.log(np.exp(beta*(ov2-m)).sum())/beta

    for it in range(n_iter):

        g = topk_grad(h, k)

                found = True
                if c5_try < c5:

                break
            step *= 0.5
        if not found:
            momentum = 0.5 * momentum  # dampen if no progress

import numpy as np
from scipy.optimize import minimize
import time

_beta_box = [100.0]
_n_box = [200]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif 0 < lag < n:
        g[lag:] += (1.0 - h[:-lag]) * dx
        g[:n - lag] -= h[lag:] * dx
    elif -n < lag < 0:
        p = -lag
        g[p:] += (1.0 - h[:-p]) * dx
        g[:n - p] -= h[p:] * dx
    return g

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    n = _n_box[0]
    beta = _beta_box[0]
    ov = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = np.max(ov)
    xs = np.clip(beta * (ov - m), -500, 500)
    e = np.exp(xs)
    return float(m + np.log(e.sum()) / beta)

def _smooth_grad(x):
    n = _n_box[0]
    beta = _beta_box[0]
    ov = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = np.max(ov)
    xs = np.clip(beta * (ov - m), -500, 500)
    e = np.exp(xs)
    w = e / e.sum()
    active = np.where(w > 1e-10)[0]
    g = np.zeros(n)
    for idx in active:
        lag = int(idx) - (n - 1)
        if abs(lag) < n:
            g += w[idx] * grad_for_lag(x, lag)
    return g

def _sum_con(x):
    return x.sum() - _n_box[0] / 2.0

def _sum_jac(x):
    return np.ones(_n_box[0])

def slsqp_optimize(h0, target, betas, maxiter, time_fn):
    n = len(h0)
    _n_box[0] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    h = h0.copy()
    c5 = compute_c5(h)
    for beta in betas:
        if time_fn() < 15:
            break
        _beta_box[0] = beta
        try:
            res = minimize(_smooth_obj, h, method='SLSQP', jac=_smooth_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            hn = project_h(res.x, target)
            c5n = compute_c5(hn)
            if c5n < c5 - 1e-15:
                h, c5 = hn, c5n
        except Exception:
            pass
    return h, c5

def proj_grad(h0, target, n_iter, time_fn, k=5):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    mom = np.zeros(n)
    mu = 0.7
    step_base = 0.08
    stall = 0
    for it in range(n_iter):
        if time_fn() < 2:
            break
        ov = compute_overlaps(h)
        top_idx = np.argsort(ov)[::-1][:k]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if abs(lag) < n:
                g += grad_for_lag(h, int(lag))
        g /= k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = mu * mom + g
        base = compute_c5(h)
        step = step_base
        found = False
        for _ in range(15):
            ht = project_h(h - step * mom, target)
            ct = compute_c5(ht)
            if ct < base - 1e-15:
                h = ht
                if ct < c5:
                    c5 = ct
                stall = 0
                found = True
                break
            step *= 0.5
        if not found:
            stall += 1
            mom *= 0.5
            if stall > 200:
                pert = np.random.randn(n) * 0.02
                h = project_h(h + pert, target)
                c5 = compute_c5(h)
                mom = np.zeros(n)
                stall = 0
    return h, c5

def binary_local_search(n, n_passes, time_fn, seed_val=0):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_corr = int(np.max(np.correlate(h, 1 - h, mode="full")))
    for p in range(n_passes):
        if time_fn and time_fn() < 3:
            break
        for _ in range(n // 2):
            zeros = np.where(h == 0)[0]
            ones_pos = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones_pos) == 0:
                break
            i = rng.choice(zeros)
            j = rng.choice(ones_pos)
            h[i], h[j] = 1.0, 0.0
            mc = int(np.max(np.correlate(h, 1 - h, mode="full")))
            if mc < best_corr:
                best_corr = mc
            else:
                h[i], h[j] = 0.0, 1.0
    return h, best_corr

def basin_hop(h0, target, n_restarts, time_fn, base_sigma=0.08):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for r in range(n_restarts):
        if time_fn() < 30:
            break
        sigma = base_sigma * (1.0 + 0.3 * (r % 5))
        hp = project_h(h + np.random.randn(n) * sigma, target)
        c5p = compute_c5(hp)
        # Quick SLSQP from perturbed point
        hn, c5n = slsqp_optimize(hp, target, [200, 1000, 5000], 2000, time_fn)
        if c5n < c5 - 1e-15:
            h, c5 = hn, c5n
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()

    # Phase 1: SLSQP refinement from initial point
    if time_left() > 60:

        if time_left() < 80:

        # Convert to continuous: use the binary as a template

        x_old = np.linspace(0, 2, n_b)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_old, hb)
        h_interp = project_h(h_interp, target)
        # Optimize from this starting point

        if cn < best_c5:

    # Phase 3: Basin hopping
    if time_left() > 60:
        best_h, best_c5 = basin_hop(best_h, target, 12, time_left, base_sigma=0.06)

    # Phase 4: Projected gradient polish
    if time_left() > 20:
