        g = subgrad_dir(best_h, corr, 3)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        for step in [0.008, 0.003, 0.001, 0.0003, 0.0001]:
            ht = project_sum(best_h - step * g, target_best)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                break

    return best_h, best_c5, len(best_h)
