import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = (h > 0) & (h < 1)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    clipped = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(clipped)) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def get_active_lags(h, num_lags=20):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.001 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > num_lags:
        idx = idx[np.argsort(corr[idx])[::-1][:num_lags]]
    return idx, corr, mc

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        la = -lag
        if la < n:
            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
    return g

def targeted_descent(h, n, target, t0, budget, max_iters=10000):
    """Coordinate descent targeting active lags specifically."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - t0 > budget - 3:
            break
        idx, corr, mc = get_active_lags(best_h, num_lags=15)
        # Aggregate gradient from active lags
        g = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            g += lag_gradient(best_h, lag)
        g /= len(idx)
        
        # Try various step sizes and directions
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002]:
            trial = project_h(best_h - step * g, target)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = trial, c
                improved = True
                break
        if not improved:
            # Try random coordinate moves on active lags
            for _ in range(5):
                j = np.random.randint(n)
                for d in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                    trial = best_h.copy()
                    trial[j] += d
                    trial = project_h(trial, target)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        best_h, best_c5 = trial, c
                        improved = True
                        break
                if improved:
                    break
        if not improved and it > 200:
            # Try binary snap
            hb = (best_h > 0.5).astype(float)
            diff = target - hb.sum()
            if diff > 0:
                zeros = np.where(hb == 0)[0]
                hb[zeros[np.argsort(best_h[zeros])[::-1]][:int(diff)]] = 1.0
            elif diff < 0:

            hb = project_h(hb, target)
            cb = compute_c5(hb)
            if cb < best_c5:

            if not improved:

    return best_h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    clipped = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(clipped)) / len(x)) / beta

def obj_fn(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.success or r.fun < obj_fn(h0, beta, 2.0/n):
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

def get_top_lags(h, m=20):

    n = len(h)
    idx = np.argsort(corr)[::-1][:m]

def lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if 0 <= lag < n:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    elif -n < lag <= 0:

        g[la:] += (1.0 - h[:n-la])
        g[:n-la] -= h[la:]
    return g

            zeros.append(i); zeros.remove(j)
        else:

    n = len(h)

    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta
