        d_tmp[block_idx] = x
        return smooth_max_grad_d(d_tmp, beta)[0]

    def block_grad(x):
        d_tmp = d.copy()
        d_tmp[block_idx] = x
        _, g = smooth_max_grad_d(d_tmp, beta)
        return g[block_idx]

    def block_eq(x):
        return x.sum() + (target - d.sum() + x0.sum()) - target

    def block_eq_jac(x):
        return np.ones(b)

    # Actually the constraint is sum(d) = 0, so sum(block) = -sum(others)
    others_sum = d.sum() - x0.sum()
    target_block = -others_sum

    def block_eq2(x):
        return x.sum() - target_block

    res = minimize(block_obj, x0, method='SLSQP', jac=block_grad,
                   bounds=bounds_b,
                   constraints=[{'type': 'eq', 'fun': block_eq2, 'jac': block_eq_jac}],
                   options={'maxiter': 200, 'ftol': 1e-15})
    d_new = d.copy()
    d_new[block_idx] = res.x
    d_new = project_d(d_new)
    return d_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    # Initialize from parent or default
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(500, 0.5)
    n = len(h_init)
    d = h_init - 0.5
    d = project_d(d)
    best_d = d.copy()
    best_c5 = compute_c5(0.5 + d)

    # Phase 1: SLSQP with beta annealing on the d parameterization
    d, c5 = do_slsqp_round(d, [50, 200, 1000, 5000, 20000, 100000],
                           800, tleft, 0.0)
    if c5 < best_c5:
        best_d, best_c5 = d.copy(), c5

    # Phase 2: Multi-resolution exploration
    for n_r in [200, 300, 400, 500, 600]:
        if tleft() < 80:
            break
        if n_r == n:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_r)
        h_interp = np.interp(x_new, x_old, 0.5 + best_d)
        d_interp = project_d(h_interp - 0.5)
        d_r, c5_r = do_slsqp_round(d_interp, [100, 1000, 10000, 100000],
                                    600, tleft, 0.0)
        if c5_r < best_c5:
            best_d, best_c5 = d_r.copy(), c5_r
            n = n_r  # update n for subsequent phases

    # Phase 3: Block coordinate descent
    n = len(best_d)
    d_cur = best_d.copy()
    c5_cur = best_c5
    for iteration in range(100):
        if tleft() < 30:
            break
        h = 0.5 + d_cur
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_lags = np.argsort(corr)[::-1][:4]
        # Select block: coordinates with values in (0,1) that affect top lags
        free_idx = np.where((h > 0.05) & (h < 0.95))[0]
        if len(free_idx) < 3:
            break
        # Prefer coordinates near the "center" of the domain
        np.random.shuffle(free_idx)
        block = free_idx[:min(25, len(free_idx))]
        d_new = block_optimize(d_cur, block, 5000, tleft, 0.0)
        c5_new = compute_c5(0.5 + d_new)
        if c5_new < c5_cur - 1e-16:
            d_cur, c5_cur = d_new, c5_new
            if c5_cur < best_c5:
                best_d, best_c5 = d_cur.copy(), c5_cur
        else:
            # Try smaller block
            block_s = free_idx[:min(8, len(free_idx))]
            d_new = block_optimize(d_cur, block_s, 5000, tleft, 0.0)
            c5_new = compute_c5(0.5 + d_new)
            if c5_new < c5_cur - 1e-16:
                d_cur, c5_cur = d_new, c5_new
                if c5_cur < best_c5:
                    best_d, best_c5 = d_cur.copy(), c5_cur

    # Phase 4: Subgradient polish
    if tleft() > 20:
        steps = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
        d_cur, c5_cur = subgrad_descent(best_d, tleft, 15000,
                                        min(7, len(best_d)//20 + 3), steps, 0.0)
        if c5_cur < best_c5:
            best_d, best_c5 = d_cur.copy(), c5_cur

    # Phase 5: Final SLSQP with very high beta
    if tleft() > 40:
        d, c5 = do_slsqp_round(best_d, [50000, 200000, 1000000],
                               2000, tleft, 0.0)
        if c5 < best_c5:
            best_d, best_c5 = d, c5

    # Phase 6: One more subgradient + SLSQP cycle
    if tleft() > 20:
        steps_fine = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        d_cur, c5_cur = subgrad_descent(best_d, tleft, 5000, 5, steps_fine, 0.0)
        if c5_cur < best_c5:
            best_d, best_c5 = d_cur.copy(), c5_cur
        if tleft() > 15:
            d, c5 = do_slsqp_round(best_d, [200000, 1000000],
                                   3000, tleft, 0.0)
            if c5 < best_c5:
                best_d, best_c5 = d, c5

    best_h = 0.5 + best_d
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
