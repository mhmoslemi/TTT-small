import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -60, 0)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[m2:] += w * (1.0 - h[:n - m2])
            grad[:n - m2] -= w * h[m2:]
    return val, grad

def lbfgsb_opt(h, beta, maxiter):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val_grad(x, beta)[0]

    def grad_fn(x):
        return smooth_max_val_grad(x, beta)[1]

    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad_fn, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 80})
        return project_h(res.x, target)
    except Exception:
        return h.copy()

def subgrad_refine(h, n_iter, time_fn):
    n = len(h)
    target = n / 2.0
    momentum = np.zeros(n)
    lr = 0.004
    no_improve = 0
    c5_cur = compute_c5(h)

    for it in range(n_iter):
        if time_fn() < 5:
            break
        _, grad = smooth_max_val_grad(h, 500)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        momentum = 0.85 * momentum - lr * grad / (gnorm + 1e-10)
        if it % 200 == 199:
            momentum += (np.random.randn(n) * 0.01 - np.mean(np.random.randn(n) * 0.01))
        h_new = project_h(h + momentum, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h = h_new
            c5_cur = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                lr *= 0.9
                momentum *= 0.5
                if lr < 1e-6:
                    break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: L-BFGS-B with smooth max, increasing beta
    for beta in [50, 150, 500, 1500, 5000, 20000, 80000]:
        if time_left() < 80:
            break
        maxit = min(300, max(50, int((time_left() - 80) / (10.0) / 0.25)))
        h_new = lbfgsb_opt(best_h, beta, maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Diverse multi-start
    for trial in range(8):
        if time_left() < 150:
            break
        n_t = n_best
        if trial == 0:
            h_try = project_h(best_h + np.random.randn(n_t) * 0.015, n_t / 2.0)
        elif trial == 1:
            x = np.linspace(0, 2, n_t)
            freq = float(np.random.choice([3, 5, 7, 11, 13]))
            ph = np.random.uniform(0, 2 * np.pi)
            h_try = project_h(best_h + 0.04 * np.sin(freq * np.pi * x + ph), n_t / 2.0)
        elif trial == 2:
            x = np.linspace(0, 2, n_t)
            h_try = project_h(best_h + 0.03 * np.cos(2 * np.pi * 4 * x), n_t / 2.0)
        elif trial == 3:
            h_try = best_h.copy()
            bs = max(1, n_t // 12)
            for b in range(0, n_t - bs, bs):
                h_try[b:b + bs] = best_h[(b + bs * 3) % (n_t - bs):((b + bs * 3) % (n_t - bs)) + bs]
            h_try = project_h(h_try, n_t / 2.0)
        elif trial == 4:
            h_try = project_h(best_h + np.random.randn(n_t) * 0.04, n_t / 2.0)
        elif trial == 5:
            # Anti-correlation bias: push toward making h more binary
            h_try = best_h.copy()
            h_try += 0.06 * np.sign(h_try - 0.5) * np.random.uniform(0.3, 1.0, n_t)

        elif trial == 6:

            x = np.linspace(0, 2, n_t)
            h_try = project_h(best_h + 0.05 * np.sin(np.pi * x) + 0.03 * np.sin(3 * np.pi * x + 1.0), n_t / 2.0)
        else:
            h_try = project_h(best_h + np.random.randn(n_t) * 0.025, n_t / 2.0)

        for beta in [200, 1000, 8000]:
            if time_left() < 60:
                break
            h_try = lbfgsb_opt(h_try, beta, 120)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_h, best_c5, n_best = h_try, c5_try, len(h_try)

    # Phase 3: Subgradient refinement on best
    best_h, best_c5 = subgrad_refine(best_h, 40000, time_left)
    n_best = len(best_h)

    # Phase 4: Final L-BFGS-B polish with very high beta
    if time_left() > 40:
        for beta in [50000, 200000, 500000]:
            if time_left() < 25:
                break
            h_new = lbfgsb_opt(best_h, beta, 250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Try resolution changes
    for n_res in [400, 600, 800]:
        if time_left() < 100:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        for beta in [500, 5000, 50000]:
            if time_left() < 50:
                break
            h_res = lbfgsb_opt(h_res, beta, 150)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5 - 1e-15:
            best_h, best_c5 = h_res, c5_res

    # Phase 6: Final subgradient + L-BFGS-B cycle
    if time_left() > 30:
        best_h, best_c5 = subgrad_refine(best_h, 10000, time_left)
        if time_left() > 20:
            h_new = lbfgsb_opt(best_h, 100000, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    return best_h, best_c5, len(best_h)
