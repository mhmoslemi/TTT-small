            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fast_subgrad(h, n, deadline, max_iter=5000000, every=20):
    best_h, best_c = h.copy(), compute_c5(h, n)
    eta, beta, v = 0.03, 0.85, np.zeros(n)
    m = int(np.argmax(compute_all_g(h, n))) - (n - 1)
    for t in range(max_iter):
        if time.time() > deadline:
            break
        v = beta * v + (1.0 - beta) * subgrad_lag(h, n, m)
        h = project_fast(h - eta * v, n)
        if (t + 1) % every == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
            m = int(np.argmax(compute_all_g(h, n))) - (n - 1)
            if c < best_c:
                eta = min(eta * 1.02, 0.3)
            else:
                eta *= 0.92
            if eta < 1e-9:
                break
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

_slsqp_n = 250
_slsqp_alpha = 1000.0

def _slsqp_obj(x):
    n, a = _slsqp_n, _slsqp_alpha
    g = np.correlate(x, 1 - x, 'full') * (2.0 / n)
    m = g.max()
    s = a * (g - m)
    sm = s.max()
    return m + (sm + np.log(np.exp(s - sm).sum())) / a

def _slsqp_cons(x):
    return x.sum() - _slsqp_n / 2.0

def slsqp_step(h, n, alpha, maxiter, deadline):
    global _slsqp_n, _slsqp_alpha
    _slsqp_n, _slsqp_alpha = n, alpha
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': _slsqp_cons}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_exact(r.x, n)
    except Exception:
        return h

def binary_search(h, n, deadline, sweeps=200):
    t = n // 2
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:t]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    for _ in range(sweeps):
        if time.time() > deadline:
            break
        imp = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    ones.remove(j); ones.append(i)
                    zeros.remove(i); zeros.append(j)
                    imp = True
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
            if imp:
                break
        if not imp:
            break
    return h_bin, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n, h0 = 250, np.full(250, 0.5)
    h0 = project_exact(h0, n)
    best_h, best_c = h0.copy(), compute_c5(h0, n)

    # Phase 1: Fast subgradient with heavy ball momentum (45% budget)
    p1 = t0 + 0.45 * (budget_s - 5)
    h, c = fast_subgrad(best_h.copy(), n, p1, 5000000, 20)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: SLSQP continuation method (20% budget)
    p2 = t0 + 0.65 * (budget_s - 5)
    for a in [100, 500, 2000, 10000, 50000, 200000]:
        if time.time() > p2 - 10:
            break
        h = slsqp_step(best_h.copy(), n, a, 800, p2 - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Warm restart + perturbed subgradient (10% budget)
    p3 = t0 + 0.75 * (budget_s - 5)
    p3a = t0 + 0.68 * (budget_s - 5)
    h_pert = project_exact(best_h + 0.04 * rng.randn(n), n)
    h, c = fast_subgrad(h_pert, n, p3a, 500000, 15)
    if c < best_c:
        best_c, best_h = c, h.copy()
    h, c = fast_subgrad(best_h.copy(), n, p3, 500000, 15)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Resolution cycling (15% budget)
    p4 = t0 + 0.90 * (budget_s - 5)
    if time.time() < p4 - 30:
        for nh in [200, 300]:
            if time.time() > p4 - 20:
                break
            xo = np.linspace(0, 2, n)
            hh = project_exact(np.interp(np.linspace(0, 2, nh), xo, best_h), nh)
            hh, c = fast_subgrad(hh, nh, p4 - 25, 500000, 15)
            hh = slsqp_step(hh, nh, 5000, 500, p4 - 10)
            c = compute_c5(hh, nh)
            if c < best_c:
                best_c, best_h, n = c, hh.copy(), nh

    # Phase 5: Binary local search (7% budget)
    if time.time() < dl - 12:
        hb, cb = binary_search(best_h.copy(), n, dl - 8, 200)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Final SLSQP polish (3% budget)
    if time.time() < dl - 3:
        h = slsqp_step(best_h.copy(), n, 500000, 1000, dl - 2)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_exact(best_h, n)
    return best_h, compute_c5(best_h, n), n
