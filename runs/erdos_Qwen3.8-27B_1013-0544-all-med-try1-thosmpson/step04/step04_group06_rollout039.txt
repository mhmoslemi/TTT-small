def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corrs(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def thue_morse(n):

    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def swap_delta_array(h, i, j, n):
    """Compute the change in the correlation array (length 2n-1) when
    swapping binary h[i] (1->0) and h[j] (0->1)."""

    # For the first part: h[i+l] - h[j+l] for l = -(n-1), ..., n-1
    # i+l ranges from i-(n-1) to i+(n-1)
    # j+l ranges from j-(n-1) to j+(n-1)

    part1_i = np.where((idx1 >= 0) & (idx1 < n), h[idx1[(idx1 >= 0) & (idx1 < n)]], 0.0)
    
    idx2 = j + l
    part1_j = np.where((idx2 >= 0) & (idx2 < n), h[idx2[(idx2 >= 0) & (idx2 < n)]], 0.0)
    
    # h[i-l] for valid indices
    idx3 = i - l
    part2_i = np.where((idx3 >= 0) & (idx3 < n), h[idx3[(idx3 >= 0) & (idx3 < n)]], 0.0)
    
    idx4 = j - l
    part2_j = np.where((idx4 >= 0) & (idx4 < n), h[idx4[(idx4 >= 0) & (idx4 < n)]], 0.0)
    
    return (part1_i - part1_j) + (part2_i - part2_j)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def thue_morse_bin(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def binarize_fixed(h, n, seed=0):

        rng = np.random.RandomState(seed)
        flip = rng.choice(zeros, int(n / 2 - cur), replace=False)

    return h

def soft_max_obj(h, target, beta):
    """Soft-max smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta)
    w = e / se
    # Gradient
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:

        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

        # Increase beta over time
        beta *= 1.01
    
    return best_h, best_c5

def subgrad_polish(h, target, max_iters, t0, time_limit):

    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
