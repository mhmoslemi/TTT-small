def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)

            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_swap_anneal(h0, n, t0, budget, max_iter=200000, rng_seed=42):
    """Swap-based simulated annealing for binary sequences."""
    h = np.zeros(n)
    ones = np.where(h0 > 0.5)[0]
    h[ones] = 1.0
    # Ensure exactly n//2 ones
    if len(ones) != n // 2:
        h = np.zeros(n)
        perm = np.random.default_rng(rng_seed).permutation(n)
        h[perm[:n//2]] = 1.0

    cur_max = float(np.max(cur_lags))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(rng_seed + 1)
    
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    
    temp_start = 0.5
    temp_end = 0.001
    n_dense = int(max_iter * 0.7)
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 5:
            break
        
        # Temperature schedule
        if it < n_dense:
            temp = temp_start
        else:
            frac = (it - n_dense) / (max_iter - n_dense)
            temp = temp_start * (temp_end / temp_start) ** frac
        
        # Choose move: swap a 1 and a 0
        a = list(ones_set)[rng.integers(len(ones_set))]
        b = list(zeros_set)[rng.integers(len(zeros_set))]
        
        # Apply swap
        h[a] = 0.0
        h[b] = 1.0
        ones_set.discard(a)
        ones_set.add(b)
        zeros_set.discard(b)
        zeros_set.add(a)
        
        new_lags = compute_lags(h)
        new_max = float(np.max(new_lags))
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
            cur_max = new_max
            cur_lags = new_lags
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:

            h[a] = 1.0
            h[b] = 0.0
            ones_set.discard(b)
            ones_set.add(a)
            zeros_set.discard(a)
            zeros_set.add(b)

If we linearize around the current solution:
- Fix all h values except a small set S
- For the variables in S, solve: min t s.t. overlap_k ≤ t for all k in active set, sum constraint, 0≤h≤1
- This is an LP!

The LP is:
Variables: h[i] for i in S, and t
Minimize t
Subject to:
  For each active lag k: ∑_{i: i,i+k valid} h[i]*(1-h[i+k]) ≤ t
  ∑ h[i] = n/2
  0 ≤ h[i] ≤ 1

- If h[i+k] ≈ 0: term becomes h[i]
- If h[i+k] ≈ 1: term becomes 0
- If h[i+k] is the variable: term becomes h[i]*(1-h[i+k]) which is still bilinear

So this only works if we fix one side to be binary.

from scipy.optimize import minimize, Bounds
from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * 2.0/n)

def lags_raw(h):
    return np.correlate(h, 1.0-h, mode="full")

def project_h(h, target):
    h = np.clip(h.copy(), 0., 1.)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15: break
        free = (h > 1e-13) & (h < 1. - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + d/len(h), 0., 1.)
        else:
            h[free] += d/free.sum()
            h = np.clip(h, 0., 1.)

def resample(h, nn):
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(nx, ox, h)

def smax(x, beta):
    m = np.max(x)
    return m + (logsumexp(beta*(x-m)) - np.log(len(x))) / beta

def obj_slsqp(v, beta, n):
    return smax(np.correlate(v, 1.-v, mode="full"), beta) * 2.0/n

def sum_fun(v, tgt): return v.sum() - tgt
def sum_jac(v, tgt): return np.ones(v.shape)

def slsqp(h0, n, tgt, beta, maxit=15000):
    try:
        r = minimize(obj_slsqp, h0, args=(beta, n), method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type':'eq','fun':sum_fun,'jac':sum_jac,'args':(tgt,)}],
                     options={'maxiter':maxit,'ftol':1e-18})

        b = zi[rng.integers(len(zi))]

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def lags_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target):
    h = np.clip(h.copy(), 0., 1.)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1. - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + d / len(h), 0., 1.)
        else:
            h[free] += d / free.sum()
            h = np.clip(h, 0., 1.)
    return h

def resample(h, nn):
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(nx, ox, h)

def smax(x, beta):
    m = np.max(x)
    return m + (logsumexp(beta * (x - m)) - np.log(len(x))) / beta

def obj_slsqp(v, beta, n):
    return smax(lags_raw(v), beta) * 2.0 / n

def sum_fun(v, tgt):
    return v.sum() - tgt

def sum_jac(v, tgt):
    return np.ones(v.shape)

def slsqp(h0, n, tgt, beta, maxit=15000):
    try:
        r = minimize(obj_slsqp, h0, args=(beta, n), method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (tgt,)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return project_h(r.x, tgt)
    except Exception:
        return h0

def binary_swap_sa(h0, n, t0, budget, niter=200000, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = np.where(h0 > 0.5)[0]
    if len(ones) != n // 2:
        perm = rng.permutation(n)
        h[perm[:n // 2]] = 1.0
    else:
        h[ones] = 1.0
    cur_max = float(np.max(lags_raw(h)))
    best_h, best_max = h.copy(), cur_max
    n_half = n // 2
    for it in range(niter):
        if budget - (time.time() - t0) < 5:
            break
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        a = oi[rng.integers(len(oi))]
        b = zi[rng.integers(len(zi))]
        h[a], h[b] = 0.0, 1.0
        nm = float(np.max(lags_raw(h)))
        T = 0.4 * (1.0 - it / niter) ** 1.5 + 0.005
        if nm < cur_max or rng.random() < np.exp(-(nm - cur_max) / T):
            cur_max = nm
            if nm < best_max:
                best_max = nm
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
    return best_h, best_max

def refine_from_binary(h_bin, n, tgt, t0, budget, seed=0):
    rng = np.random.default_rng(seed)
    best = compute_c5(h_bin)
    best_h = h_bin.copy()
    for scale in [0.05, 0.03, 0.015, 0.01]:
        if budget - (time.time() - t0) < 12:
            break
        hp = project_h(h_bin + rng.standard_normal(n) * scale, tgt)
        for beta in [500, 2000, 8000]:
            if budget - (time.time() - t0) < 10:
                break
            hp = slsqp(hp, n, tgt, beta, maxit=12000)
            c = compute_c5(hp)
            if c < best:
                best = c
                best_h = hp.copy()
    return best_h, best

def coord_polish(h, n, tgt, t0, budget, max_sweeps=50):
    cur = h.copy()
    best_c = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for _ in range(max_sweeps):
        if budget - (time.time() - t0) < 3:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 2:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, tgt)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    cur, best_c, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c

def subgrad_step(h, n, tgt, t0, budget, n_iter=5000, step0=0.03, top_k=8):
    cur = h.copy()
    best = compute_c5(cur)
    mom = np.zeros(n)
    step = step0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        corr = lags_raw(cur)
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, tgt)
        c = compute_c5(trial)
        if c < best - 1e-15:
            cur, best, step = trial, c, min(step * 1.05, 0.1)
            mom *= 0.5
        else:
            step *= 0.7
            mom *= 0.3
    return cur, best

            if c < best_c5:
