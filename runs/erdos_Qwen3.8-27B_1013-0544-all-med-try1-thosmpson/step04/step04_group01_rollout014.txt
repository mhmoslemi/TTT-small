import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h *= (1 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def subgradient_optimize(h, n, budget_s, rng):
    """Momentum subgradient descent on the max correlation."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    lr = 0.01
    momentum = 0.9
    
    t0 = time.time()
    while time.time() - t0 < budget_s:
        c = np.correlate(h, 1 - h, mode='full') * dx
        m_val = c.max()
        if m_val < best_c:
            best_c = m_val
            best_h = h.copy()
        
        m_idx = c.argmax()
        offset = m_idx - (n - 1)
        
        # Compute gradient of c[m_idx] w.r.t. h
        grad = np.zeros(n)
        # Forward: h[i] contributes (1-h[i+offset]) to lag m_idx
        idx_fwd = np.arange(n) + offset
        valid_fwd = (idx_fwd >= 0) & (idx_fwd < n)
        grad[valid_fwd] += (1 - h[idx_fwd[valid_fwd]])
        # Backward: h[i] contributes -h[i-offset] to lag m_idx  
        idx_bwd = np.arange(n) - offset
        valid_bwd = (idx_bwd >= 0) & (idx_bwd < n)
        grad[valid_bwd] -= h[idx_bwd[valid_bwd]]
        grad *= dx

        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, n)
        
        # Decay learning rate
        lr *= 0.9999
    
    return best_h, best_c
