def projected_gradient(h, n, deadline, alpha_start=100, alpha_end=100000):
    """Projected gradient descent with Barziland-Borwein step size."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Log-spaced alpha sequence
    n_alphas = 10
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    prev_x = None
    prev_grad = None
    
    for alpha in alphas:
        step_scale = 1.0 / alpha  # approximate step size
        x_old = h.copy()
        g_old = smooth_max_and_grad(h, n, alpha)[1]
        
        max_iters = 500
        for it in range(max_iters):
            if time.time() > deadline - 5:
                return best_h, best_c
            
            # Projected gradient step
            # Project gradient: remove mean (for sum constraint), then clip
            g_proj = g_old - np.mean(g_old)
            # Line search
            step = step_scale
            for ls in range(20):
                h_new = h - step * g_proj
                h_new = np.clip(h_new, 0, 1)
                h_new = project_feasible(h_new, n)
                val_new, _ = smooth_max_and_grad(h_new, n, alpha)
                if val_new < smooth_max_and_grad(h, n, alpha)[0] - 1e-10 * abs(step * np.linalg.norm(g_proj)):
                    break
                step *= 0.5
            
            # BB step size update

                s = h - prev_x
                y = g_old - prev_grad
                ss = s @ s
                sy = s @ y
                if sy > 1e-10:
                    step_scale = ss / sy
                    step_scale = np.clip(step_scale, 0.01/alpha, 10/alpha)

        c = compute_c5(h, n)
        if c < best_c:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() == 0: break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.sum() == 0: break
            h[mask] += diff / mask.sum()
        np.clip(h, 0, 1, out=h)
    return h

def c5_gradient(h, n, alpha):
    """Compute value and gradient of smooth max of correlate."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    exp_sh = np.exp(sh)
    w = exp_sh / exp_sh.sum()
    val = m + np.log(exp_sh.sum()) / alpha
    
    # Compute gradient using the formula
    # grad[j] = dx * sum_k w[k] * ((1-h[j+k-n+1])*[valid1] - h[j-k+n-1]*[valid2])
    grad = np.zeros(n)
    # Vectorized approach:
    # For each k, the contribution to grad is w[k] * (term1 - term2)
    # term1[j] = 1-h[j+k-n+1] for j in [max(0,n-1-k), min(n-1, 2n-2-k)]
    # term2[j] = h[j-k+n-1] for j in [max(0,k-n+1), min(n-1, 2n-2-k)]
    
    # I'll use a different vectorization:
    # Note that sum_k w[k]*(1-h[j+k-n+1]) = 1 - sum_k w[k]*h[j+k-n+1] (for valid k)
    # = 1 - (weighted sum of h at shifted positions)
    # And sum_k w[k]*h[j-k+n-1] = weighted sum of h at other shifted positions
    
    # Let me just loop over k (399 iterations for n=200, each O(n) → total O(n²) = 40k ops)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * (1.0 - h[lo1 + k - n + 1 : hi1 + k - n + 2])
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * h[lo2 - k + n - 1 : hi2 - k + n + 1]
    
    return val, grad * dx

def projected_grad_descent(h, n, deadline, alpha_list=None):
    if alpha_list is None:
        alpha_list = [50, 200, 1000, 5000, 20000, 100000, 500000]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    
    for alpha in alpha_list:
        if time.time() > deadline - 3:
            break
        step = 1.0 / (alpha * 0.1)
        prev_y = None
        prev_x = None
        max_iter = 200
        
        for it in range(max_iter):
            if time.time() > deadline - 3:
                break
            val, g = c5_gradient(x, n, alpha)
            # Project gradient: remove mean for sum constraint
            g = g - np.mean(g)
            
            # BB step
            if prev_x is not None:
                s = x - prev_x
                y = g - prev_y
                sy = np.dot(s, y)
                ss = np.dot(s, s)
                if sy > 1e-12:
                    step = np.clip(ss / sy, 0.001/alpha, 5.0/alpha)
            
            # Line search (simple backtracking)
            for ls in range(15):
                x_new = np.clip(x - step * g, 0, 1)
                x_new = project_feasible(x_new, n)
                val_new, _ = c5_gradient(x_new, n, alpha)
                if val_new < val - 1e-4 * step * np.dot(g, g) or ls == 14:
                    break
                step *= 0.5
            
            prev_x = x.copy()
            prev_y = g.copy()
            x = x_new
            
            if it % 20 == 19:
                c = compute_c5(x, n)
                if c < best_c:
                    best_c = c
                    best_h = x.copy()
        
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
            best_h = x.copy()
    
    return best_h, best_c

def pair_refine(h, n, deadline, step_sizes=None, radius=None):
    """Pair search with given step sizes and optional radius limit."""
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005]
    if radius is None:
        radius = n  # full search
    
    best_c = compute_c5(h, n)
    for d in step_sizes:
        if time.time() > deadline - 2:
            break
        for _pass in range(30):
            if time.time() > deadline - 2:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return h, best_c
                j_start = max(0, i - radius)
                j_end = min(n, i + radius + 1)
                for j in range(j_start, j_end):
                    if i == j:
                        continue
                    # Try h[i]+=d, h[j]-=d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # Try h[i]-=d, h[j]+=d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
