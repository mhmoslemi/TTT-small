def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_full_corr(h):
    """Compute full correlation using FFT (faster for large n)."""
    n = len(h)
    size = 2 * n
    f_h = np.fft.rfft(h, size)
    f_1mh = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(f_h * f_1mh, size)[:2*n-1]
    # np.correlate(a, v, 'full') gives result of length 2n-1
    # corr_full[k] = sum_i a[i] * v[i+k-(n-1)]
    # We need to match the indexing of np.correlate
    # Actually let me just use np.correlate for correctness and use FFT only for the max
    return corr

def compute_c5_fft(h):
    """Fast C5 computation using FFT."""
    n = len(h)
    dx = 2.0 / n
    size = 2 * n
    # Linear cross-correlation via FFT
    f1 = np.fft.rfft(1.0 - h, size)
    f2 = np.fft.rfft(h, size)
    corr = np.fft.irfft(f1 * f2, size)
    # The linear correlation of a with b: corr[k] = sum_i a[i+k]*b[i]
    # Using FFT: ifft(fft(a) * fft(conj(b)))...

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_bin, n, max_iters, t0, budget, rng):
    """Swap-based local search on binary sequences."""
    cur = h_bin.copy()
    cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
    best_h = cur.copy()
    best_max = cur_max
    ones = np.where(cur == 1)[0].tolist()
    zeros = np.where(cur == 0)[0].tolist()
    stuck = 0
    
    for it in range(max_iters):
        if budget - (time.time() - t0) < 5:
            break
        # Try random swap
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        
        # Compute new correlation
        cur[a], cur[b] = 0.0, 1.0
        new_corr = np.correlate(cur, 1.0 - cur, mode="full")
        new_max = float(np.max(new_corr))
        
        if new_max < cur_max:
            cur_max = new_max
            # Update lists
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            if new_max < best_max:
                best_max = new_max
                best_h = cur.copy()
            stuck = 0
        else:
            cur[a], cur[b] = 1.0, 0.0
            stuck += 1
            if stuck > 500:
                # Perturb: do a random swap regardless
                a = ones[rng.integers(len(ones))]
                b = zeros[rng.integers(len(zeros))]
                cur[a], cur[b] = 0.0, 1.0
                cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
                ones.remove(a)
                ones.append(b)
                zeros.remove(b)
                zeros.append(a)
                stuck = 0
    
    return best_h, best_max

def smooth_max_grad(corr, beta):
    """Compute smooth max and its softmax weights."""
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    w = exp_x / np.sum(exp_x)
    return m + np.log(np.sum(exp_x)) / beta, w

def gradient_c5(h, beta):
    """Compute gradient of smooth_max C5 with respect to h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, w = smooth_max_grad(corr, beta)
    
    # corr has length 2n-1, indexed from -(n-1) to n-1
    # corr[k] = sum_i h[i+k] * (1-h[i]) for k = -(n-1) to n-1
    # Actually np.correlate(h, 1-h, 'full')[k] for k=0,...,2n-2
    # corresponds to lag = k - (n-1)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            # C(lag) = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # dC/dh[j] = (1-h[j+lag]) for j <= n-1-lag (j appears as h[i])
            #           - h[j-lag] for j >= lag (j appears as h[i+lag], i=j-lag)
            for j in range(n):
                d = 0.0
                if j <= n - 1 - lag:
                    d += (1.0 - h[j + lag])
                if j >= lag and j - lag <= n - 1 - lag:
                    d -= h[j - lag]
                grad[j] += wi * d
        else:
            la = -lag
            # C(-la) = sum_{i=la}^{n-1} h[i] * (1-h[i-la])
            # = sum_{j=0}^{n-1-la} h[j+la] * (1-h[j])
            # dC/dh[j]: j appears as h[j] in (1-h[j]) → -h[j+la] for j <= n-1-la
            #           j appears as h[j+la] → (1-h[j-la]) for j >= la
            for j in range(n):
                d = 0.0
                if j <= n - 1 - la:
                    d -= h[j + la]
                if j >= la and j - la <= n - 1 - la:
                    d += (1.0 - h[j - la])
                grad[j] += wi * d
    
    return sm * (2.0 / n), grad * (2.0 / n)

def projected_grad_desc(h, target, n, beta, n_iter, t0, budget, lr_init=0.01):
    """Projected gradient descent with backtracking."""
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    lr = lr_init
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        _, grad = gradient_c5(cur, beta)
        # Projected gradient step
        trial = cur - lr * grad
        trial = project_h(trial, target)
        c_try = compute_c5(trial)
        
        if c_try < best_c5:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

The key insight: for each lag `lag` (corresponding to correlation index idx = lag + n - 1), the gradient contribution is:
- For lag >= 0: dC/dh[j] += (1-h[j+lag]) for j <= n-1-lag, minus h[j-lag] for j >= lag

For lag >= 0:

For lag < 0 (la = -lag):
  grad[:n-la] -= h[la:]         (the first part)
  grad[la:] += (1 - h[:n-la])   (the second part)

def gradient_c5_fast(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, w = smooth_max_grad(corr, beta)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            # A: (1-h[lag:]) for positions 0 to n-1-lag
            grad[:n-lag] += wi * (1.0 - h[lag:])
            # B: -h[:n-lag] for positions lag to n-1
            grad[lag:] += wi * (-h[:n-lag])
        else:
            la = -lag
            # For C(-la) = sum_{i=0}^{n-1-la} h[i+la]*(1-h[i])
            # dC/dh[j]: j appears as h[i+la] → (1-h[i]) = (1-h[j-la]) for j >= la, j-la <= n-1-la
            #           j appears in (1-h[i]) → -h[i+la] = -h[j+la]... wait
            # C(-la) = sum_{i=0}^{n-1-la} h[i+la] * (1-h[i])
            # dC/dh[j] = sum over terms where h[j] appears
            #   h[j] appears as h[i+la] when i = j-la: contributes (1-h[j-la]) if 0 <= j-la <= n-1-la
            #   h[j] appears in (1-h[i]) when i = j: contributes -h[j+la] if 0 <= j+la... wait
            #   (1-h[i]) where i = j: -h[j+la]... no, d(1-h[i])/dh[j] = -delta(i,j)
            #   So the term h[i+la]*(1-h[i]) differentiated w.r.t. h[j]:
            #     if j = i+la (i = j-la): contributes (1-h[j-la])
            #     if j = i (i = j): contributes -h[j+la]
            # So dC/dh[j] = (1-h[j-la])*[0<=j-la<=n-1-la] + (-h[j+la])*[0<=j<=n-1-la and j+la<n]
            #             = (1-h[j-la])*[la<=j<=n-1] + (-h[j+la])*[0<=j<=n-1-la]
            
            grad[la:] += wi * (1.0 - h[:n-la])
            grad[:n-la] += wi * (-h[la:])
    
    return sm * (2.0/n), grad * (2.0/n)

The difference is non-zero only when i ∈ {a, b} or i+l ∈ {a, b}.

  But if a+l = b: h'[a+l] = h'[b] = 1, so 0*(1-1) = 0, change = 0 - (1-0) = -1
  If a+l ≠ b: h'[a+l] = h[a+l], change = 0 - (1-h[a+l]) = -(1-h[a+l])
  
  Valid when a+l in [0, n-1], i.e., -a ≤ l ≤ n-1-a

- i = b: h'[b]=1, h[b]=0. Change = 1*(1-h'[b+l]) - 0 = 1-h'[b+l]
  If b+l = a: h'[b+l] = h'[a] = 0, change = 1-0 = 1
  If b+l ≠ a: h'[b+l] = h[b+l], change = 1-h[b+l]
  
  Valid when b+l in [0, n-1], i.e., -b ≤ l ≤ n-1-b

- i+l = a, i = a-l: h[i+l] changes from h[a]=1 to h'[a]=0
  Change = h'[a-l]*(1-0) - h[a-l]*(1-1) = h[a-l] - 0 = h[a-l]
  (assuming a-l ≠ b; if a-l = b, h'[a-l] = h'[b] = 1, same result)
  Actually: h'[a-l]*(1-h'[a]) - h[a-l]*(1-h[a]) = h[a-l]*(1-0) - h[a-l]*(1-1) = h[a-l]
  
  Valid when a-l in [0, n-1] and a in [0, n-1] (always true), i.e., 0 ≤ a-l ≤ n-1, i.e., a-(n-1) ≤ l ≤ a

- i+l = b, i = b-l: h[i+l] changes from h[b]=0 to h'[b]=1
  Change = h'[b-l]*(1-1) - h[b-l]*(1-0) = 0 - h[b-l] = -h[b-l]
  
  Valid when b-l in [0, n-1], i.e., b-(n-1) ≤ l ≤ b

So the total change in C(l) for a given lag l is the sum of contributions from the valid terms above.

For a specific lag l, the change is computed in O(1) (just checking a few conditions). But to find the new max, we need to check all lags where the change is non-zero.

The lag l has a non-zero change if at least one of the four conditions is met:
- -a ≤ l ≤ n-1-a (i=a valid)
- -b ≤ l ≤ n-1-b (i=b valid)
- a-(n-1) ≤ l ≤ a (i+l=a valid)
- b-(n-1) ≤ l ≤ b (i+l=b valid)

The union of these intervals covers l from min(-a, -b, a-n+1, b-n+1) to max(n-1-a, n-1-b, a, b).

For a, b in [0, n-1], this is roughly l from -(n-1) to n-1, i.e., all lags. So we can't really reduce the number of lags to check.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def ensure_binary(h, n, rng):
    h = h.copy()
    h_ones = int(np.sum(h >= 0.5))
    h_bin = (h >= 0.5).astype(float)
    target_ones = n // 2
    if h_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        rng.shuffle(ones_idx)
        h_bin[ones_idx[target_ones:]] = 0.0
    elif h_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        rng.shuffle(zeros_idx)
        h_bin[zeros_idx[:target_ones - h_ones]] = 1.0
    return h_bin

def binary_sa(h_bin, n, t0, budget, rng, n_iter=300000, restarts=3):
    cur = h_bin.copy()
    cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
    best_h = cur.copy()
    best_max = cur_max
    ones = list(np.where(cur == 1)[0])
    zeros = list(np.where(cur == 0)[0])
    
    for restart in range(restarts):
        if budget - (time.time() - t0) < 10:
            break
        iters_per_restart = n_iter // restarts
        stuck = 0
        for it in range(iters_per_restart):
            if budget - (time.time() - t0) < 5:
                break
            a = ones[int(rng.integers(len(ones)))]
            b = zeros[int(rng.integers(len(zeros)))]
            cur[a], cur[b] = 0.0, 1.0
            new_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
            delta = new_max - cur_max
            if delta < 0:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_h = cur.copy()
                ones.remove(a)
                ones.append(b)
                zeros.remove(b)
                zeros.append(a)
                stuck = 0
            else:
                frac = (it + restart * iters_per_restart) / n_iter
                temp = 3.0 * (0.005) ** frac
                if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                    cur_max = new_max
                    ones.remove(a)
                    ones.append(b)
                    zeros.remove(b)
                    zeros.append(a)
                    stuck = 0
                else:
                    cur[a], cur[b] = 1.0, 0.0
                    stuck += 1
            if stuck > 2000 and it < iters_per_restart - 100:
                # Perturb: do 3 random swaps
                for _ in range(3):
                    a2 = ones[int(rng.integers(len(ones)))]
                    b2 = zeros[int(rng.integers(len(zeros)))]
                    cur[a2], cur[b2] = 0.0, 1.0
                    ones.remove(a2)
                    ones.append(b2)
                    zeros.remove(b2)
                    zeros.append(a2)
                cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
                stuck = 0
    return best_h, best_max

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -400, 400)
    exp_x = np.exp(x)
    w = exp_x / np.sum(exp_x)
    return m + np.log(np.sum(exp_x)) / beta, w

def grad_smooth_c5(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, w = smooth_max_and_weights(corr, beta)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] += wi * (-h[:n - lag])
        else:
            la = -lag
            grad[la:] += wi * (1.0 - h[:n - la])
            grad[:n - la] += wi * (-h[la:])
    return sm * (2.0 / n), grad * (2.0 / n)

def projected_grad_desc(h, target, n, beta, n_iter, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    lr = 0.005
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        _, grad = grad_smooth_c5(cur, beta)
        trial = cur - lr * grad
        trial = project_h(trial, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            lr = min(lr * 1.05, 0.05)
            stall = 0
        else:
            lr *= 0.6
            stall += 1
            if stall > 200 or lr < 1e-9:
                break
    return best_h, best_c5

def make_structured_inits(n, rng):
    inits = []
    # Block patterns
    for bs in [3, 5, 7, 11, 15, 21, 29, 39, 49, 65]:
        if 2 * bs > n:
            continue
        h = np.zeros(n)
        for start in range(0, n, 2 * bs):
            h[start:start + bs] = 1.0
        inits.append(ensure_binary(h, n, rng))
    # Quasi-periodic with golden ratio
    golden = (1.0 + 5.0 ** 0.5) / 2.0
    h_g = np.zeros(n)
    for i in range(n):
        if (i * golden) % 1.0 < 0.5:
            h_g[i] = 1.0
    inits.append(ensure_binary(h_g, n, rng))
    # Fibonacci-like
    fib = 1.0 / (golden - 1)
    h_f = np.zeros(n)
    for i in range(n):
        if (i * fib) % 1.0 < 0.5:
            h_f[i] = 1.0
    inits.append(ensure_binary(h_f, n, rng))
    # Random
    for _ in range(8):
        h_r = np.zeros(n)
        idx = np.arange(n)
        rng.shuffle(idx)
        h_r[idx[:n // 2]] = 1.0
        inits.append(h_r)
    return inits

    remaining = budget_s - (time.time() - t0)

        c_sa = sa_max * (2.0 / n)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa.copy()

            h_cont, c_cont = projected_grad_desc(h_cont, target, n, beta,
                                                 int(remaining * 2000), t0, budget_s)

                best_h = h_cont.copy()

        h2_bin = ensure_binary(h2, n2, rng)

        if c2 < best_c5:

            n = n2
