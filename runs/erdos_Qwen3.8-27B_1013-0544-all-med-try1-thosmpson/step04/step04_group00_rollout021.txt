    def compute_c5(h, n):

        return float(np.max(c) * dx)

        return c * dx
    
    def project_binary(h, n):

        target = n // 2
        idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
        return h_bin
    
    def project_feasible(h, n):
        """Project to [0,1]^n with sum = n/2."""
        target = n / 2.0
        h = np.clip(h, 0.0, 1.0)
        for _ in range(1000):
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
            if diff > 0:

                s = room.sum()
                if s < 1e-12:
                    break
                h = h + diff / s * room
            else:
                s = h.sum()
                if s < 1e-12:
                    break
                h = h + diff / s * h
        return np.clip(h, 0.0, 1.0)
