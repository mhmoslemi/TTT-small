def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = free.sum()
        if cnt == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def obj_func(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def c_sum(x, t):
    return x.sum() - t

def jac_sum(x, t):
    return np.ones(x.shape)

def slsqp(h0, n, target, beta, maxiter=10000):
    b = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': c_sum, 'jac': jac_sum, 'args': (target,)}]
    try:
        r = minimize(obj_func, h0, args=(beta, n), method='SLSQP',
                     bounds=b, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except:
        return h0

    best_h = h.copy()
    best_c = cur

        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_c = compute_c5(h)
        if new_c < cur:
            cur = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:

            if rng.random() < np.exp(-(new_c - cur) / max(T, 1e-15)):
                cur = new_c
            else:
                h[a] = 1.0
                h[b] = 0.0
    return best_h, best_c

def subgrad(h, target, n, n_iter, t0, budget, top_k=12, lr_init=0.03):
    cur = h.copy()
    best_c = compute_c5(cur)
    best_h = cur.copy()
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            if 0 <= lag < n:
                grad[:n-lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n-lag]
            elif -n < lag < 0:
                la = -lag
                grad[la:] += (1.0 - cur[:n-la])
                grad[:n-la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-15:

        grad /= gn
        trial = project_h(cur - lr * grad, target)

            best_c = c_try
            best_h = cur.copy()
            stall = 0
            lr = min(lr * 1.03, 0.15)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = free.sum()

            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

    f = smooth_max(corr, beta) * (2.0/n)
    penalty = mu * (h.sum() - target)**2
    return f + penalty

    f = smooth_max(corr, beta) * (2.0/n)

            grad[:n-la] -= w * h[la:]

    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def corr_grad(h):
    """Gradient of max correlation (smooth) w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ws = np.exp(np.clip(500.0 * (corr - m), -500, 500))
    ws /= ws.sum()
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = ws[idx]
        if w < 1e-9:
            continue
        lag = int(idx) - (n - 1)
        if 0 <= lag < n:
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n - lag]
        elif -n < lag < 0:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    return grad, corr

def obj_lbfgs(h, beta, target, mu, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = smooth_max(corr, beta) * (2.0 / n)
    p = mu * (h.sum() - target) ** 2
    return f + p

def obj_lbfgs_full(h, beta, target, mu, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = smooth_max(corr, beta) * (2.0 / n)
    m = np.max(corr)
    ws = np.exp(np.clip(beta * (corr - m), -500, 500))
    ws /= ws.sum()
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = ws[idx]
        if w < 1e-9:
            continue
        lag = int(idx) - (n - 1)
        if 0 <= lag < n:
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n - lag]
        elif -n < lag < 0:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    grad *= (2.0 / n)
    grad += 2.0 * mu * (h.sum() - target)
    p = mu * (h.sum() - target) ** 2
    return f + p, grad

def lbfgs_opt(h0, n, target, beta, mu, maxiter=1000):
    b = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(obj_lbfgs_full, h0, args=(beta, target, mu, n),
                     method='L-BFGS-B', jac=True, bounds=b,
                     options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def slsqp_fn(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def slsqp_fn_grad(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = smooth_max(corr, beta) * (2.0 / n)
    m = np.max(corr)
    ws = np.exp(np.clip(beta * (corr - m), -500, 500))
    ws /= ws.sum()
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = ws[idx]
        if w < 1e-9:
            continue
        lag = int(idx) - (n - 1)
        if 0 <= lag < n:
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n - lag]
        elif -n < lag < 0:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] -= w * h[la:]
    grad *= (2.0 / n)
    return f, grad

def csum(x, t):
    return x.sum() - t

def csum_jac(x, t):
    return np.ones(x.shape)

def slsqp(h0, n, target, beta, maxiter=5000):
    b = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': csum, 'jac': csum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_fn_grad, h0, args=(beta, n), method='SLSQP',
                     jac=True, bounds=b, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad(h, target, n, n_iter, t0, budget, top_k=12, lr_init=0.04):
    cur = h.copy()
    best_c = compute_c5(cur)
    best_h = cur.copy()
    lr = lr_init
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            if 0 <= lag < n:
                if lag == 0:
                    grad += (1.0 - 2.0 * cur)
                else:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            elif -n < lag < 0:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        trial = project_h(cur - lr * grad, target)
        c_try = compute_c5(trial)
        if c_try < best_c - 1e-16:
            cur = trial
            best_c = c_try
            best_h = cur.copy()
            stall = 0
            lr = min(lr * 1.04, 0.2)
        else:
            stall += 1
            lr *= 0.6
            if stall > 400:
                break
    return best_h, best_c

def coord_polish(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c - 1e-15:
                    cur = trial
                    best_c = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c

def anneal_binary(h_bin, n, t0, budget, n_iter=30000):
    h = h_bin.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(4242)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_c = compute_c5(h)
        if new_c < cur:
            cur = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            T = 0.3 * (0.005) ** (it / max(n_iter, 1))
            if rng.random() < np.exp(-(new_c - cur) / max(T, 1e-15)):
                cur = new_c
            else:
                h[a] = 1.0
                h[b] = 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 384
        best_h = np.full(n, 0.5)
        best_h[:n // 2] = 0.75
        best_h[n // 2:] = 0.25
        best_h = project_h(best_h, n / 2.0)

    best_c5 = compute_c5(best_h)
    n_cur = len(best_h)
    target = n_cur / 2.0

    # Phase 1: L-BFGS-B with increasing beta and penalty
    for beta in [50, 200, 1000, 5000, 20000]:
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        for mu in [50.0, 500.0]:
            rem = budget_s - (time.time() - t0)
            if rem < 8:
                break
            cur = lbfgs_opt(best_h, n_cur, target, beta, mu, maxiter=800)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h = cur
                best_c5 = c_try

    # Phase 2: SLSQP with exact constraint
    for beta in [200, 1000, 5000, 20000, 80000]:
        rem = budget_s - (time.time() - t0)
        if rem < 12:
            break
        best_h = slsqp(best_h, n_cur, target, beta, maxiter=8000)
        best_c5 = compute_c5(best_h)

    # Phase 3: Subgradient descent
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h, best_c5 = subgrad(best_h, target, n_cur, 10000, t0, budget_s, top_k=16, lr_init=0.03)

    # Phase 4: L-BFGS-B again after gradient steps
    for beta in [5000, 20000, 80000]:
        rem = budget_s - (time.time() - t0)
        if rem < 8:
            break
        cur = lbfgs_opt(best_h, n_cur, target, beta, 500.0, maxiter=1000)
        c_try = compute_c5(cur)
        if c_try < best_c5:
            best_h = cur
            best_c5 = c_try

    # Phase 5: SLSQP high-precision
    for beta in [50000, 200000, 500000]:
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        best_h = slsqp(best_h, n_cur, target, beta, maxiter=10000)
        best_c5 = compute_c5(best_h)

    # Phase 6: Try different resolutions
    for n_alt in [320, 448, 512]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        tgt = n_alt / 2.0
        cur = project_h(resample(best_h, n_alt), tgt)
        for beta in [500, 3000, 15000]:
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            cur = lbfgs_opt(cur, n_alt, tgt, beta, 200.0, maxiter=600)
            cur = slsqp(cur, n_alt, tgt, beta, maxiter=4000)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h = cur
                best_c5 = c_try
                n_cur = n_alt
                target = tgt

    # Phase 7: Perturbation restarts
    rng = np.random.default_rng(seed + 7)
    for att in range(10):
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        sc = [0.015, 0.008, 0.02, 0.005, 0.01, 0.012, 0.006, 0.018, 0.004, 0.009][att]
        hp = best_h + rng.standard_normal(n_cur) * sc
        hp = project_h(hp, target)
        hp = lbfgs_opt(hp, n_cur, target, 3000, 200.0, maxiter=500)
        hp = slsqp(hp, n_cur, target, 10000, maxiter=4000)
        hp, _ = subgrad(hp, target, n_cur, 3000, t0, budget_s, top_k=10, lr_init=0.02)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try

    # Phase 8: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 8:
        best_h = slsqp(best_h, n_cur, target, 500000, maxiter=10000)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 5:
        best_h, best_c5 = coord_polish(best_h, target, n_cur, 150, t0, budget_s)

    # Final project to ensure validity
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_cur
