def fft_convolve(a, b):
    """Linear convolution using FFT"""
    n = len(a) + len(b) - 1
    fsize = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:n]

import numpy as np
from scipy.optimize import minimize
import time

_state = {'beta': 100.0, 'n': 0}

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    fsize = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_fft(h):
    g = 1.0 - h
    return fft_convolve(h, g[::-1])

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_fft(h)
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    
    P = np.zeros(2 * n)
    P[1:2*n-1] = np.cumsum(w)
    C = P[n:2*n] - P[:n]
    
    w0 = w[n - 1]
    s = np.zeros(n)
    s[1:] = w[n:] + w[:n-1][::-1]
    
    S1 = fft_convolve(s, h)[:n]
    S2 = fft_convolve(s, h[::-1])[:n][::-1]
    
    grad = (C - 2.0 * w0 * h - S1 - S2) * dx
    return val, grad

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _penalty_obj(x):
    beta = _state['beta']
    n = _state['n']
    val, grad_sm = smooth_max_val_grad(x, beta)
    pen = 10.0 * (x.sum() - n / 2.0) ** 2
    return val + pen

def _penalty_grad(x):
    beta = _state['beta']
    n = _state['n']
    val, grad_sm = smooth_max_val_grad(x, beta)
    grad_pen = 20.0 * (x.sum() - n / 2.0) * np.ones(n)
    return grad_sm + grad_pen

def do_lbfgsb(h0, beta, maxiter, time_left):
    n = len(h0)
    _state['beta'] = beta
    _state['n'] = n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penalty_obj, h0, method='L-BFGS-B',
                      bounds=bounds, jac=_penalty_grad,
                      options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def do_subgrad(h, target, n_iter, time_left, top_k, steps):
    n = len(h)
    no_imp = 0
    c5_cur = compute_c5(h)
    for it in range(n_iter):
        if time_left() < 2:
            break
        corr = compute_corr_fft(h)
        nk = min(top_k, max(3, n // 20))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            w = dx
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif 0 < lag < n:
                h_shift = np.roll(h, -lag)
                mask = np.zeros(n, dtype=bool)
                mask[:n-lag] = True
                grad[:n-lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n-lag]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n-m2] += w * (1.0 - h[m2:])
                grad[m2:] -= w * h[:n-m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        found = False
        for step in steps:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-15:
                h = ht
                c5_cur = c5t
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % 100 == 99:
                sc = 0.01 * min(5, no_imp // 100 + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(ht) < c5_cur:
                    h = ht
                    c5_cur = compute_c5(ht)
                    no_imp = 0
            if no_imp > 2000:
                break
    return h, c5_cur

def upscale(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0, 2, old_n)
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, new_n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: L-BFGS-B with smooth max, increasing beta
    for beta in [30, 80, 200, 500, 1000, 3000, 8000, 20000, 50000]:
        if time_left() < 15:
            break
        h_new, c5_new = do_lbfgsb(best_h, beta, maxiter=300, time_left=time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    
    # Phase 2: Multi-resolution
    for n_res in [100, 150, 200, 300, 400, 500, 700, 1000]:
        if time_left() < 60:

        h_res = upscale(best_h, n_res)
        for beta in [100, 500, 2000, 10000, 50000]:
            if time_left() < 15:
                break
            h_new, c5_new = do_lbfgsb(h_res, beta, maxiter=200, time_left=time_left)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        # subgradient polish at this resolution
        if time_left() > 30:
            tgt = len(best_h) / 2.0
            best_h, best_c5 = do_subgrad(best_h, tgt, 5000, time_left, top_k=8,
                                          steps=[0.04, 0.02, 0.01, 0.005, 0.002])
    
    # Phase 3: Final L-BFGS-B with high beta
    for beta in [20000, 100000, 200000]:
        if time_left() < 15:
            break
        h_new, c5_new = do_lbfgsb(best_h, beta, maxiter=500, time_left=time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

        best_h, best_c5 = do_subgrad(best_h, tgt, 20000, time_left, top_k=5,
                                      steps=[0.03, 0.01, 0.005, 0.002, 0.001])

    # Final projection
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
