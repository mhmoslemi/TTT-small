import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = compute_corr(h)
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqp_optimize(h, n, deadline, seed, max_beta=50000):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    betas = [50, 200, 1000, 5000, 20000, max_beta]
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.005, 0.015]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except:
                pass
    return h, c5

def nesterov_subgrad(h, n, deadline, seed, top_k_init=5, top_k_max=20):
    np.random.seed(seed)
    ts = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    
    velocity = np.zeros(n)
    lr = 0.003
    alpha_w = 8.0
    stagnant = 0
    top_k = top_k_init
    beta_mom = 0.85
    
    # Barzilai-Borwein state
    prev_h = h.copy()
    prev_grad = np.zeros(n)
    
    for it in range(20000):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h)
        c_max = corr.max()
        c_std = corr.std()
        
        # Adaptive active set
        top_k = min(top_k_max, max(top_k_init, int(3 + 0.05 * it)))
        active = np.argsort(corr)[-top_k:]
        
        # Weighted subgradient
        w = corr[active]
        w = np.exp(alpha_w * (w - w.max()))
        w /= w.sum()
        
        grad = np.zeros(n)
        for j, k in enumerate(active):
            lag = k - (n - 1)
            grad += w[j] * subgrad_for_lag(h, lag, n)
        
        # BB step size
        if it > 2:
            s = h - prev_h
            y = grad - prev_grad
            sy = np.dot(s, y)
            if abs(sy) > 1e-10:
                ss = np.dot(s, s)
                yy = np.dot(y, y)
                if ss > 1e-10:
                    alpha_bb = ss / sy
                    if alpha_bb > 0 and alpha_bb < 10 * lr:
                        lr = 0.5 * lr + 0.5 * min(alpha_bb, 2 * lr)
                if yy > 1e-10:
                    beta_bb = sy / yy
                    if 0 < beta_bb < 2:
                        beta_mom = 0.7 * beta_mom + 0.3 * beta_bb
        
        prev_h = h.copy()
        prev_grad = grad.copy()
        
        # Nesterov step
        velocity = beta_mom * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        
        if c_new < c5 - 1e-13:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.001, 0.01)
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.92
            if stagnant % 50 == 0:
                alpha_w = max(alpha_w * 0.95, 2.0)
        
        if stagnant > 1000 and lr < 1e-7:
            break
    
    return h, c5

def structured_perturb(h, n, seed):
    """Generate structured perturbations to escape local optima."""
    np.random.seed(seed)
    ts = n / 2.0
    candidates = [h.copy()]
    
    # Sinusoidal perturbations at different frequencies
    x = np.arange(n)
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.01, 0.03, 0.05]:
            pert = amp * np.sin(2 * np.pi * freq * x / n)
            h_p = project_feasible(h + pert, ts)
            candidates.append(h_p)
    
    # Random perturbations
    for amp in [0.01, 0.02, 0.04]:
        pert = np.random.randn(n) * amp
        h_p = project_feasible(h + pert, ts)
        candidates.append(h_p)
    
    # Antisymmetric perturbation
    pert = np.zeros(n)
    pert[:n//2] = 0.02
    pert[n//2:] = -0.02
    candidates.append(project_feasible(h + pert, ts))
    
    return candidates

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqp_optimize(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    betas = [50, 200, 1000, 5000, 20000, 80000]
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.004, 0.01]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 4000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def nesterov_subgrad(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    momentum = 0.82
    alpha_w = 8.0
    stagnant = 0
    top_k = 5
    prev_h = h.copy()
    prev_grad = np.zeros(n)

    for it in range(30000):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        top_k = min(25, top_k + (1 if it % 200 == 0 else 0))
        active = np.argsort(corr)[-top_k:]
        w = corr[active]
        w = np.exp(alpha_w * (w - w.max()))
        w /= w.sum()

        grad = np.zeros(n)
        for j, k in enumerate(active):
            if w[j] > 1e-8:
                lag = k - (n - 1)
                grad += w[j] * subgrad_for_lag(h, lag, n)

        if it > 3:
            s = h - prev_h
            y = grad - prev_grad
            sy = np.dot(s, y)
            ss = np.dot(s, s)
            if sy > 1e-12 and ss > 1e-12:
                alpha_bb = ss / sy
                if 0 < alpha_bb < 20 * lr:
                    lr = 0.6 * lr + 0.4 * alpha_bb
                yy = np.dot(y, y)
                if yy > 1e-12:
                    beta_bb = sy / yy
                    if 0.3 < beta_bb < 0.98:
                        momentum = 0.8 * momentum + 0.2 * beta_bb

        prev_h = h.copy()
        prev_grad = grad.copy()
        velocity = momentum * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)

            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.002, 0.015)
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.93
            if stagnant > 100:
                alpha_w = max(alpha_w * 0.97, 2.0)
        if stagnant > 1500 and lr < 1e-8:
            break
    return h, c5

def make_perturbations(h, n, seed):

    ts = n / 2.0
    x = np.arange(n, dtype=float)
    cands = [h.copy()]
    for freq in [2, 3, 5, 7, 11]:

    half = n // 2
    pert = np.zeros(n)
    pert[:half] = 0.015
    pert[half:] = -0.015
    cands.append(project_feasible(h + pert, ts))
    return cands
