    peaks = (h_copy > 0.7) & ((h_copy > np.roll(h_copy, 1)) | (np.roll(h_copy, 1) < 0.1))
    # Increase peak amplitude
    h_copy[peaks] *= 1.05
    h_copy = project_to_constraints(h_copy, 200)
    return h_copy

def smooth_transition(h):
    """Smooth sharp transitions to reduce overlap"""
    h_copy = h.copy()
    # Find transition points
    transitions = np.diff(np.sign(h_copy - 0.5)).astype(bool)
    # Create smoother ramps
    for i in np.flatnonzero(transitions):
        # Simple smooth: linear interpolation
        width = 5 if i % (len(h_copy)//3) == 0 else 3
        left = max(0, i-width//2)
        right = i+width//2
        if right >= len(h_copy):
            continue
        start_val = h_copy[i]
        end_val = h_copy[i+1]
        if np.abs(start_val-end_val) > 0.2:
            t = np.arange(left, right+1)
            v = 0.5 * (1 + np.tanh((t - (left+right)/2) * 10) * (end_val - start_val))
            h_copy[t.astype(int)] = v
    return project_to_constraints(h_copy, 200)

def optimize_subset(h, step_size=0.01):
    """Perform local optimization on a subset"""
    orig = h.copy()
    try:
        n = len(h)
        bounds = np.vstack([np.zeros(n), np.ones(n)])
        obj = lambda x: compute_c5(x)
        # Simple conjugate gradient approach
        pos = orig.copy()
        best = pos.copy()
        for i in range(20):  # Local optimization steps
            grad = compute_subgradient(pos, mode="overlap")
            m = min(step_size, 0.1 * np.linalg.norm(grad))
            pos_new = pos - m * grad
            pos_new = project_to_constraints(pos_new, 200)
            val_new = obj(pos_new)
            if val_new < obj(best):
                best = pos_new.copy()
            else:
                # Backward step
                best = best * 0.95
            pos = 0.9*pos + 0.1*best
        h_cur_val = obj(best)
        if h_cur_val < compute_c5(orig):
            orig = best.copy()
    except:
        pass
    return orig
