import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_lp_index(n):
    """Pre-compute index arrays for LP constraints."""
    idx_list = []
    for k in range(2 * n - 1):
        off = k - (n - 1)
        m_lo = max(0, -off)
        m_hi = min(n, n - off)
        ms = np.arange(m_lo, m_hi)
        mq = ms + off
        idx_list.append((ms, mq))
    return idx_list

def lp_solve(h_old, n, idx_list, solver=cp.ECOS):
    h = cp.Variable(n)
    t = cp.Variable()
    dx = 2.0 / n
    cons = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    for ms, mq in idx_list:
        if len(ms) == 0:
            continue
        if np.array_equal(ms, mq):
            lin = cp.sum((1 - 2 * h_old[ms]) * h[ms])
            const = float(np.sum(h_old[ms] ** 2))
        else:
            lin = cp.sum((1 - h_old[mq]) * h[ms]) - cp.sum(h_old[ms] * h[mq])
            const = float(np.sum(h_old[ms] * h_old[mq]))
        cons.append(lin * dx + const * dx <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, max_iters=50000, verbose=False)
        if h.value is not None and not np.any(np.isnan(h.value)):
            return project_feasible(np.array(h.value).flatten(), n)
    except Exception:
        pass
    return h_old.copy()
