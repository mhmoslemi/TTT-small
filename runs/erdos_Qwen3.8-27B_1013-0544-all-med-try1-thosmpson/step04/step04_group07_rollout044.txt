    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def obj_grad_lbfgs(h, beta, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exponents = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(exponents)
    wsum = w.sum()
    val = (m + np.log(wsum / len(corr)) / beta) * dx
    s = h.sum()
    pen = 1e4 * (s - target) ** 2
    val += pen
    idx = np.where(w / wsum > 1e-6)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(corr)])
    g = np.zeros(n)
    for k in idx:
        lag = k - (n - 1)
        wk = w[k] / wsum
        if lag >= 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l = -lag
            g[:n - l] += wk * (1.0 - h[l:])
            g[l:] -= wk * h[:n - l]
    g *= dx
    g += 2e4 * (s - target)
    return val, g

def run_lbfgs(h0, n, target, beta, maxiter, t0, budget):
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj_grad_lbfgs, h0, args=(beta, target), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8,
                       'maxls': 50})
        return project_h(res.x, target)
    except Exception:
        return h0

def sl_obj(h, beta, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def sl_eq(x, t):
    return x.sum() - t

def sl_eq_jac(x, t):
    return np.ones(x.shape)

def run_slsqp(h0, n, target, beta, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sl_eq, 'jac': sl_eq_jac, 'args': (target,)}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(sl_obj, h0, args=(beta, target), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x, target)
    except Exception:
        return h0

def spectral_smooth(h, keep):
    n = len(h)
    H = np.fft.rfft(h)
    if keep < len(H):
        H[keep:] = 0
    out = np.fft.irfft(H, n=n)
    return np.clip(out, 0, 1)

def coord_descent(cur, target, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sw in range(sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for j in range(len(cur)):
            if time.time() - t0 > budget - 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def pair_descent(cur, target, sweeps, t0, budget, seed=0):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(seed)
    n = len(cur)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for _ in range(n // 2):
            if time.time() - t0 > budget - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.03, -0.03, 0.015, -0.015, 0.008, -0.008):
                trial = cur.copy()
                trial[j] += d
                trial[k] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 4:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(384, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: L-BFGS-B with smooth max (fast, good for initialization)
    for beta in [200, 500, 1000]:
        rem = budget_s - (time.time() - t0)
        if rem < 30:
            break
        cur = run_lbfgs(best_h, n, target, beta, 3000, t0, budget_s)
        cur = project_h(cur, target)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 2: SLSQP with equality constraint (more accurate)
    for beta in [400, 800, 2000]:
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        cur = run_slsqp(best_h, n, target, beta, 5000, 1e-15)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 3: Spectral smoothing + re-optimization (escape local optima)
    for keep in [4, 6, 8, 12]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        smoothed = spectral_smooth(best_h, keep)
        smoothed = project_h(smoothed, target)
        cur = run_lbfgs(smoothed, n, target, 500, 2000, t0, budget_s)
        cur = run_slsqp(cur, n, target, 1000, 3000, 1e-15)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 4: Random perturbation restarts
    for trial in range(5):
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        rng = np.random.RandomState(seed + 100 + trial)
        pert = best_h + rng.randn(n) * 0.03
        pert = project_h(pert, target)
        cur = run_lbfgs(pert, n, target, 500, 2000, t0, budget_s)
        cur = run_slsqp(cur, n, target, 800, 3000, 1e-15)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 5: Coordinate + pair descent polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h, best_c5 = coord_descent(best_h, target, 20, t0, budget_s)
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = pair_descent(best_h, target, 15, t0, budget_s, seed=seed)

    # Phase 6: Final SLSQP polish with high beta
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        cur = run_slsqp(best_h, n, target, 5000, 8000, 1e-15)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 7: Try slightly different resolution
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        n2 = 512
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_h(h2, target2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_h, best_c5, n, target = h2.copy(), c2, n2, target2
        cur = run_lbfgs(h2, n, target, 500, 3000, t0, budget_s)
        cur = run_slsqp(cur, n, target, 1000, 5000, 1e-15)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5, n, target = cur.copy(), c, n, target

    # Final cleanup
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
