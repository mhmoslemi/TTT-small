    best_c5 = compute_c5(best_h)
    n_best = len(best_h)
    
    # Phase 1: SLSQP with increasing beta
    h1, c5_1 = slsqp_opt(best_h, [50, 150, 500, 1500, 5000], 5000, time_left)
    if c5_1 < best_c5:
        best_h, best_c5, n_best = h1, c5_1, len(h1)

    for n_res in [400, 500, 600, 800, 1000]:

            break

        h_res = project_h(h_res, n_res / 2.0)
        h_nr, c5_nr = slsqp_opt(h_res, [200, 1000, 5000, 20000], 3000, time_left)
        if c5_nr < best_c5:

    # Phase 3: Local search (block coordinate)
    if time_left() > 30:
        h_ls, c5_ls = local_search(best_h, time_left, 100, 5)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls
    
    # Phase 4: SLSQP polish
    if time_left() > 20:
        h4, c5_4 = slsqp_opt(best_h, [10000, 30000, 100000], 10000, time_left)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4
    
    # Phase 5: More local search
    if time_left() > 15:
        h_ls2, c5_ls2 = local_search(best_h, time_left, 200, 3)
        if c5_ls2 < best_c5:
            best_h, best_c5 = h_ls2, c5_ls2
    
    # Phase 6: Final SLSQP
    if time_left() > 10:
        h6, c5_6 = slsqp_opt(best_h, [50000, 100000, 200000], 5000, time_left)
        if c5_6 < best_c5:

    return best_h, best_c5, len(best_h)

Hmm, the local_search function might be too slow because it calls compute_c5 many times. Let me estimate: for n=500, block_size=5, we have 100 blocks, and for each block we try 6 deltas. Each compute_c5 call is O(n) for the correlate (actually O(n^2) for np.correlate with mode="full"... no wait, np.correlate for 1D is O(n*m) where n and m are the lengths of the two inputs. Here both are length n, so it's O(n^2). For n=500, that's 250,000 operations per call. With 600 calls per round (100 blocks * 6 deltas), that's 150 million operations per round. At maybe 100M ops/sec in Python (with numpy), that's about 1.5 seconds per round. With 100 rounds, that's 150 seconds. That's too much.

Let me reduce the local search or make it smarter. Actually, I can precompute the correlation once and only update it when h changes. But that's complex.

Alternatively, I can make the local search less aggressive: fewer blocks per round, fewer deltas.

Or I can skip the local search and focus on the SLSQP with better parameters.

Let me reconsider. The main advantage of the local search is that it can make small improvements that gradient methods miss (because the gradient of the max is only defined at the argmax). But SLSQP with high beta should capture most of this.

Let me simplify: skip the local search, and instead focus on:
1. More SLSQP iterations
2. Higher beta values
3. Better multi-resolution strategy
4. A few stochastic restarts

import numpy as np
from scipy.optimize import minimize
import time

_beta_box = [100.0]
_target_box = [250.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    exp_x = np.exp(x)
    sum_exp = np.sum(exp_x)
    weights = exp_x / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki] * dx
        lag = int(ki) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def project_h(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    target = n / 2.0
    _target_box[0] = target
    bounds = [(0.0, 1.0)] * n
