def frank_wolfe_step(h, g, n):
    """Solve min g^T h s.t. 0<=h<=1, sum(h)=n/2"""
    # The optimum is h*=1 for the n/2 indices with smallest g values
    idx = np.argsort(g)
    h_star = np.zeros(n)
    h_star[idx[:n//2]] = 1.0
    return h_star

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def fw_step(h, g, n):
    """Frank-Wolfe: solve min g^T h s.t. 0<=h<=1, sum(h)=n/2"""
    idx = np.argsort(g)
    h_star = np.zeros(n)
    h_star[idx[:n // 2]] = 1.0
    return h_star

def frank_wolfe(h, n, deadline, max_iter=5000):
    ts = n / 2.0
    c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = 3
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(8.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        g = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            g += w[j] * subgrad_for_lag(h, lag, n)
        h_star = fw_step(h, g, n)
        # Backtracking line search
        best_h = h.copy()
        best_c5 = c5
        for t in np.linspace(1.0, 0.01, 20):
            h_new = (1 - t) * h + t * h_star
            h_new = project_feasible(h_new, ts)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new
        h = best_h
        c5 = best_c5
    return h, c5

def subgradient_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    for it in range(20000):

        corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def fw_linear_subproblem(g, n):
    idx = np.argsort(g)
    h_star = np.zeros(n)
    h_star[idx[:n // 2]] = 1.0
    return h_star

def frank_wolfe(h, n, deadline, seed, max_iter=5000):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_c5 = c5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(3, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(6.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        g = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            g += w[j] * subgrad_for_lag(h, lag, n)
        h_star = fw_linear_subproblem(g, n)
        improved = False
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
            h_new = (1.0 - t) * h + t * h_star
            h_new = project_feasible(h_new, ts)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def subgrad_top1(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    lr = 0.002
    stagnant = 0

        if time.time() > deadline:
            break

        k_max = np.argmax(corr)
        lag = k_max - (n - 1)
        grad = subgrad_for_lag(h, lag, n)
        h_new = project_feasible(h - lr * grad, ts)
        c_new = compute_c5(h_new)

def slsqp_pass(h, n, deadline, seed, betas, perts):
