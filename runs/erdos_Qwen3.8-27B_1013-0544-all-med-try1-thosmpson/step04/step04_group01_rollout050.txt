    dx = 2.0 / n
    target_sum = n / 2.0
    
    def eval_R_at(h, t, i, j, s):
        """Compute all R[k] with h[i]=t, h[j]=s-t, others fixed."""
        h[i] = t
        h[j] = s - t
        c = np.correlate(h, 1 - h, mode='full')
        return c * dx
    
    def find_optimal_pair(h, i, j, s):
        """Find optimal t for pair (i,j) with h[i]+h[j]=s."""
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-15:
            return h[i]  # No room to move
        
        # Compute slopes: R[k](t) = a_k * t + b_k
        # R[k](t) changes because h[i] and h[j] change
        # ∂R[k]/∂t = ∂R[k]/∂h[i] - ∂R[k]/∂h[j] (since h[j] = s-t, dh[j]/dt = -1)

        hi_save, hj_save = h[i], h[j]
        
        # Evaluate at t = lo
        h[i], h[j] = lo, s - lo
        R_lo = np.correlate(h, 1 - h, mode='full') * dx
        
        # Evaluate at t = hi
        h[i], h[j] = hi, s - hi
        R_hi = np.correlate(h, 1 - h, mode='full') * dx
        
        h[i], h[j] = hi_save, hj_save

        denom = hi - lo
        slopes = (R_hi - R_lo) / denom

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_R(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def exact_pair_cd(h, n, deadline, rng, max_passes=500):
    """Exact pair coordinate descent with line search."""
    best_c = compute_c5(h, n)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = list(range(n))
        rng.shuffle(indices)
        for ii in range(n):
            i = indices[ii]
            for jj in range(ii + 1, n):
                j = indices[jj]
                if time.time() > deadline:
                    return h, best_c
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                span = hi - lo
                if span < 1e-12:
                    continue
                
                # Quick check: current value
                u_cur = h[i] - lo  # current u = h[i] - lo ∈ [0, span]
                
                # Compute R at boundaries
                old_i, old_j = h[i], h[j]
                h[i], h[j] = lo, s - lo
                R_lo = np.correlate(h, 1 - h, mode='full')
                h[i], h[j] = hi, s - hi
                R_hi = np.correlate(h, 1 - h, mode='full')
                h[i], h[j] = old_i, old_j
                
                slopes = (R_hi - R_lo) / span
                # R[k](u) = slopes[k]*u + R_lo[k], u ∈ [0, span]
                
                # Bisection
                u_lo_b, u_hi_b = 0.0, span
                for _ in range(45):
                    if u_hi_b - u_lo_b < 1e-13 * span:
                        break
                    u_mid = (u_lo_b + u_hi_b) / 2
                    R_mid = slopes * u_mid + R_lo
                    M_mid = R_mid.max()
                    active = R_mid >= M_mid - 1e-12
                    if not active.any():
                        break
                    max_s = slopes[active].max()
                    min_s = slopes[active].min()
                    if min_s > 1e-10:
                        u_hi_b = u_mid
                    elif max_s < -1e-10:
                        u_lo_b = u_mid
                    else:
                        break
                
                u_opt = (u_lo_b + u_hi_b) / 2
                t_opt = lo + u_opt
                t_opt = max(lo, min(hi, t_opt))
                
                if abs(t_opt - old_i) > 1e-12:
                    h[i] = t_opt
                    h[j] = s - t_opt
                    c_new = float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
        if not improved:
            break
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def multi_scale_optimize(h, n, deadline, rng):
    """Multi-scale: coarse → fine"""
    h_work = h.copy()
    # Coarsen
    n_coarse = n // 4
    h_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        h_coarse[i] = h_work[i*4:(i+4)].mean()
    h_coarse = project_feasible(h_coarse, n_coarse)
    
    # Optimize at coarse scale
    t_coarse = time.time() + (deadline - time.time()) * 0.3
    h_coarse, _ = exact_pair_cd(h_coarse, n_coarse, t_coarse, rng, max_passes=200)
    
    # Interpolate back
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n)
    h_interp = np.interp(x_fine, x_coarse, h_coarse)
    h_interp = project_feasible(h_interp, n)
    
    # Optimize at fine scale
    h_opt, c_opt = exact_pair_cd(h_interp, n, deadline, rng, max_passes=500)
    
    # Also try without the coarse phase
    h_opt2, c_opt2 = exact_pair_cd(h_work, n, deadline, rng, max_passes=500)
    
    if c_opt2 < c_opt:
        return h_opt2, c_opt2
    return h_opt, c_opt
