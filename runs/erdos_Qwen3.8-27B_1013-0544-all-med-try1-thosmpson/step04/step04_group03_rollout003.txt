import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def smax_lse(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta) * (2.0 / n)

def smax_lp(h_in, p, n):
    c = np.maximum(np.correlate(h_in, 1.0 - h_in, mode="full"), 1e-15)
    return (np.sum(c**p))**(1.0/p) * (2.0 / n)

def do_slsqp_lse(h0, n, target, beta, maxiter=10000):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(smax_lse, h0, args=(beta, n), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def do_slsqp_lp(h0, n, target, p, maxiter=8000):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(smax_lp, h0, args=(p, n), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def weighted_subgrad(h, target, n, n_iter, t0, budget, top_k=32, lr_init=0.012, mu=0.9):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = lr_init
    momentum = np.zeros(n)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full") * (2.0/n)
        max_val = np.max(corr)
        top_idx = np.argsort(corr)[::-1][:top_k]
        excess = np.maximum(corr[top_idx] - 0.6 * max_val, 0)
        w_sum = excess.sum()
        if w_sum < 1e-12:
            excess = np.ones(top_k) / top_k
            w_sum = 1.0
        else:
            excess /= w_sum
        grad = np.zeros(n)
        for j, idx in enumerate(top_idx):
            w = excess[j]
            if w < 1e-12:
                continue
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n-lag] += w * (1.0 - cur[lag:])
                    grad[lag:] -= w * cur[:n-lag]
            else:
                la = -lag
                grad[la:] += w * (1.0 - cur[:n-la])
                grad[:n-la] -= w * cur[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        momentum = mu * momentum - step * grad
        trial = project_h(cur + momentum, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            step = min(step * 1.04, 0.06)
        else:
            step *= 0.72
            momentum *= 0.4
            trial = project_h(cur + momentum, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
            else:
                momentum = np.zeros(n)
    return cur.copy(), best_c5

def adam_subgrad(h, target, n, n_iter, t0, budget, lr=0.006, p_start=6, p_max=80):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.75, 0.999, 1e-8
    top_k = 24
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        p = min(p_start * (1.06 ** (it // 150)), p_max)
        corr = np.correlate(cur, 1.0 - cur, mode="full") * (2.0/n)
        top_idx = np.argsort(corr)[::-1][:top_k]
        pw = np.maximum(corr[top_idx], 1e-15) ** (p - 1)
        pw /= pw.sum()
        grad = np.zeros(n)
        for j, idx in enumerate(top_idx):
            w = pw[j]
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n-lag] += w * (1.0 - cur[lag:])
                    grad[lag:] -= w * cur[:n-lag]
            else:
                la = -lag
                grad[la:] += w * (1.0 - cur[:n-la])
                grad[:n-la] -= w * cur[la:]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**(it + 1))
        vh = v / (1 - b2**(it + 1))
        trial = project_h(cur - lr * mh / (np.sqrt(vh) + eps), target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
    return best_h, best_c5

def block_cd(h, target, n, max_sweeps, t0, budget, bs=3):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    nb = n // bs
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for b in range(nb):
            if budget - (time.time() - t0) < 1:
                break
            s, e = b * bs, (b + 1) * bs
            for sc in [0.04, -0.04, 0.02, -0.02, 0.01, -0.01]:
                trial = cur.copy()
                trial[s:e] += sc
                trial = project_h(trial, target)
                ct = compute_c5(trial)
                if ct < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = ct
                    best_h = cur.copy()
                    improved = True
                    break
        if not improved and sw > 4:
            break
    return best_h, best_c5

def multi_restart(h0, n, target, t0, budget, rng, n_restarts=6):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    scales = [0.05, 0.025, 0.04, 0.015, 0.035, 0.02, 0.06, 0.01]
    for att in range(n_restarts):
        if t0 and budget - (time.time() - t0) < 20:
            break
        sc = scales[att % len(scales)]
        hp = best_h + rng.standard_normal(n) * sc
        hp = project_h(hp, target)
        hp = do_slsqp_lse(hp, n, target, 1500, maxiter=5000)
        hp, c1 = weighted_subgrad(hp, target, n, 3000, t0, budget, top_k=24, lr_init=0.01)
        if c1 < best_c5:
            best_h = hp.copy()
            best_c5 = c1
        if t0 and budget - (time.time() - t0) < 10:
            break
        hp2 = do_slsqp_lse(hp, n, target, 8000, maxiter=5000)
        c2 = compute_c5(hp2)
        if c2 < best_c5:
            best_h = hp2.copy()
            best_c5 = c2
    return best_h, best_c5
