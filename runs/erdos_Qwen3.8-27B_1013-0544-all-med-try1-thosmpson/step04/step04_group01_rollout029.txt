    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # If already close, just clip
    if abs(h.sum() - target) < 1e-12 and h.min() >= -1e-12 and h.max() <= 1 + 1e-12:
        return np.clip(h, 0, 1)

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

    exp_shifted = np.exp(shifted - mx)
    weights = exp_shifted / exp_shifted.sum()
    val = m + (mx + np.log(exp_shifted.sum())) / alpha

    # M2[i,k] = h[i-n+1+k] if valid
    idx1 = i_idx[:, None] + (n - 1) - k_idx[None, :]
    idx2 = i_idx[:, None] - (n - 1) + k_idx[None, :]

        else:

        if c_new < best_c:

        return m + (mx + np.log(es.sum())) / alpha

    def obj_grad(x):
        return grad_smooth_max(x, n, alpha)

    def fsum(x):
        return x.sum() - n / 2.0

    def fsum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': fsum, 'jac': fsum_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    for sgn in (1, -1):
                        hi = h[i] + sgn * d
                        hj = h[j] - sgn * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sgn * d, hj + sgn * d
            if not improved:
                break
    return h, best_c

def subspace_optimize(h, n, alpha, n_sub, n_iter, deadline, rng):
    best_c = compute_c5(h, n)
    h_copy = h.copy()
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        idx = rng.choice(n, size=n_sub, replace=False)
        sub = h_copy[idx].copy()
        sub_pgd, sub_c = pgm(sub, n_sub, alpha, 30, deadline, use_momentum=False)
        h_copy[idx] = sub_pgd
        c = compute_c5(h_copy, n)
        if c < best_c - 1e-15:
            best_c = c
        else:
            h_copy[idx] = sub
    return h_copy, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    h_best = project_feasible(h_init, n)
    c_best = compute_c5(h_best, n)

    # Phase 1: Continuation PGM (45% budget)
    t1 = t0 + (deadline - t0) * 0.45
    h_cur = h_best.copy()
    alphas = [50, 200, 1000, 5000, 20000, 100000]
    iters_per = max(50, int(1600 * 0.45 / len(alphas)))
    for a in alphas:
        if time.time() > t1 - 3:
            break
        h_cur, c_cur = pgm(h_cur, n, a, iters_per, t1, use_momentum=True)
        if c_cur < c_best:
            h_best, c_best = h_cur.copy(), c_cur

    # Phase 2: SLSQP polish (10% budget)
    t2 = t0 + (deadline - t0) * 0.55
    if time.time() < t2 - 5:
        ht = slsqp_polish(h_best, n, alpha=50000, maxiter=1500)
        ct = compute_c5(ht, n)
        if ct < c_best:
            h_best, c_best = ht, ct

    # Phase 3: Restarts (30% budget)
    t3 = t0 + (deadline - t0) * 0.85
    n_restarts = 10
    for r in range(n_restarts):
        if time.time() > t3 - 20:
            break
        scale = 0.03 + 0.2 * (r / max(1, n_restarts - 1))
        hp = h_best + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = min(40, (t3 - time.time()) * 0.4)
        if alloc < 8:
            break
        hp, cp = pgm(hp, n, 30000, int(alloc * 8), time.time() + alloc, use_momentum=True)
        if cp < c_best:
            h_best, c_best = hp.copy(), cp
        if time.time() < t3 - 10:
            hp2 = slsqp_polish(hp, n, alpha=100000, maxiter=800)
            cp2 = compute_c5(hp2, n)
            if cp2 < c_best:
                h_best, c_best = hp2, cp2

    # Phase 4: Final polish (15% budget)
    if time.time() < deadline - 15:
        h_best = slsqp_polish(h_best, n, alpha=200000, maxiter=2500)
        c_best = compute_c5(h_best, n)

    if time.time() < deadline - 5:
        h_best, c_best = pair_search(h_best, n, deadline - 3,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005], radius=60)

    if time.time() < deadline - 2:
        h_best = slsqp_polish(h_best, n, alpha=500000, maxiter=1500)
        c_best = compute_c5(h_best, n)

    h_best = project_feasible(h_best, n)
    c_best = compute_c5(h_best, n)
    return h_best, c_best, n
