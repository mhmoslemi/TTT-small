import numpy as np, time, math, random
from itertools import repeat

def compute_c5(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def project_to_constraints(arr, target_sum):
    # Clip to [0,1] then adjust to hit exact sum by adding/subtracting uniformly
    arr = np.clip(arr, 0.0, 1.0)
    diff = target_sum - arr.sum()
    if abs(diff) < 1e-12:
        return arr
    # distribute diff proportionally to distance from bounds
    free = (arr > 0) & (arr < 1)
    if not free.any():
        # fallback: uniform shift
        arr += diff / len(arr)
        return np.clip(arr, 0.0, 1.0)
    step = diff / free.sum()
    arr[free] += step
    # final clip to ensure bounds
    arr = np.clip(arr, 0.0, 1.0)
    # if still off, scale whole vector
    if abs(arr.sum() - target_sum) > 1e-9:
        arr *= target_sum / arr.sum()
    return arr
