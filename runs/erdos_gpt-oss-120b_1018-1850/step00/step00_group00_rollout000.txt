import numpy as np, time, math, random

def _project_to_feasible(x, target_sum):
    """Clip to [0,1] and adjust to hit target_sum."""
    x = np.clip(x, 0.0, 1.0)
    # iterative redistribution
    for _ in range(10):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, distribute proportionally
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            break
        add = diff / n_free
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
    # final scaling if slight error remains
    cur_sum = x.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        x *= target_sum / cur_sum
        x = np.clip(x, 0.0, 1.0)
    return x

def _evaluate(h, dx):
