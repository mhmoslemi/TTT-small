import time
import numpy as np

def _project_bounds_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=100):
    """Project vector v onto the set {x | lo<=x<=hi, sum(x)=target_sum}."""
    x = np.clip(v, lo, hi)
    for _ in range(max_iter):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = (x > lo) & (x < hi)
        if not np.any(free):
            # all at bounds, distribute remaining diff uniformly and clip again
            x += diff / len(x)
            x = np.clip(x, lo, hi)
            continue
        adjust = diff / free.sum()
        x[free] += adjust
        x = np.clip(x, lo, hi)
    return x

def _score(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())
