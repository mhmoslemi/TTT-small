import numpy as np
import time
import math

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _random_binary(n, ones):
    """Generate a binary vector of length n with exactly `ones` ones."""
    arr = np.zeros(n, dtype=np.float64)
    arr[:ones] = 1.0
    np.random.shuffle(arr)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # Choose number of discretization points.
    # Prefer a modest size for speed; 140 gives finer resolution.
    n_points = 140
    dx = 2.0 / n_points
    required_ones = n_points // 2  # because sum(h)*dx = 1

    # Initialise h.
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, copy=False).astype(np.float64)
        # If the provided vector has different length, upscale/downscale by tiling.
        if init.size != n_points:
            # Tile or trim while preserving total mass.
            repeats = math.ceil(n_points / init.size)
            tiled = np.tile(init, repeats)[:n_points]
            # Adjust to exact number of ones by fixing rounding errors.
            current_sum = tiled.sum()
            diff = required_ones - round(current_sum / dx)
            if diff != 0:
                # flip diff entries to satisfy the count
                idx = np.where(tiled < 0.5)[0]
                if diff > 0:
                    chosen = np.random.choice(idx, min(len(idx), diff), replace=False)
                    tiled[chosen] = 1.0
                else:
                    idx = np.where(tiled > 0.5)[0]
                    chosen = np.random.choice(idx, min(len(idx), -diff), replace=False)
                    tiled[chosen] = 0.0
            h_current = tiled
        else:
            h_current = init.copy()
            # Ensure binary and correct mass
            h_current = (h_current > 0.5).astype(np.float64)
            current_ones = int(round(h_current.sum()))
            if current_ones != required_ones:
                # fix by random swaps
                diff = required_ones - current_ones
                if diff > 0:
                    zeros = np.where(h_current == 0)[0]
                    flip = np.random.choice(zeros, diff, replace=False)
                    h_current[flip] = 1.0
                elif diff < 0:
                    ones = np.where(h_current == 1)[0]
                    flip = np.random.choice(ones, -diff, replace=False)
                    h_current[flip] = 0.0
    else:
        h_current = _random_binary(n_points, required_ones)

    best_h = h_current.copy()
    best_score = _evaluate(best_h, dx)

    # Simulated annealing parameters
    max_iters = 2000000  # safety cap
    temp_start = 1.0
    temp_end = 1e-4
    # linear cooling based on elapsed time
    def temperature(elapsed):
        frac = elapsed / budget_s
        return max(temp_end, temp_start * (1 - frac))

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s or iteration >= max_iters:
            break
        iteration += 1

        # Propose a move: swap a 1 and a 0 to keep count constant
        ones_idx = np.where(h_current == 1)[0]
        zeros_idx = np.where(h_current == 0)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # perform swap
        h_candidate = h_current.copy()
        h_candidate[i] = 0.0
        h_candidate[j] = 1.0

        cand_score = _evaluate(h_candidate, dx)

        # Accept with SA probability
        T = temperature(elapsed)
        delta = cand_score - best_score
        if cand_score < best_score or np.random.rand() < math.exp(-delta / T):
            h_current = h_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()

        # occasional intensive local refinement
        if iteration % 5000 == 0:
            # try a small random perturbation multiple times
            for _ in range(20):
                # pick two random positions (could be same type)
                a, b = np.random.randint(0, n_points, size=2)
                if h_current[a] == h_current[b]:
                    continue
                h_tmp = h_current.copy()
                h_tmp[a], h_tmp[b] = h_tmp[b], h_tmp[a]
                sc = _evaluate(h_tmp, dx)
                if sc < best_score:
                    best_score = sc
                    best_h = h_tmp.copy()
                    h_current = h_tmp

    # final evaluation to be safe
    final_score = _evaluate(best_h, dx)
    return (best_h, final_score, n_points)
