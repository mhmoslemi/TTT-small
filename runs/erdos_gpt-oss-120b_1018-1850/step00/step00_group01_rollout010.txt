import time
import numpy as np
from scipy import interpolate
from scipy.optimize import minimize

def _c5_value(h):
    """Compute C5 = max convolution integral for given h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _make_start_vector(base_h, target_n):
    """Resize base_h (array) to length target_n via linear interpolation."""
    src_n = base_h.size
    x_src = np.linspace(0.0, 2.0, src_n, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, target_n, endpoint=False)
    f = interpolate.interp1d(x_src, base_h, kind="linear", fill_value="extrapolate")
    return f(x_tgt)

def _perturb(h, rng, scale=0.1):
    """Add small random noise and re‑project onto constraints."""
    n = h.size
    noise = rng.uniform(-scale, scale, size=n)
    h_new = h + noise
    h_new = np.clip(h_new, 0.0, 1.0)
    # enforce integral exactly 1 => sum = n/2
    cur_sum = h_new.sum()
    target = n / 2.0
    if cur_sum == 0:
        h_new[:] = target / n
    else:
        h_new *= target / cur_sum
        # after scaling some entries may leave bounds; fix iteratively
        for _ in range(5):
            viol_low = h_new < 0
            viol_high = h_new > 1
            if not (viol_low.any() or viol_high.any()):
                break
            # push violations back into [0,1] and renormalize the free mass
            excess = h_new[viol_high] - 1.0
            deficit = -h_new[viol_low]
            h_new[viol_high] = 1.0
            h_new[viol_low] = 0.0
            mass = excess.sum() - deficit.sum()
            free = ~(viol_low | viol_high)
            if free.any():
                h_new[free] += mass / free.sum()
        h_new = np.clip(h_new, 0.0, 1.0)
    # final renorm to exact sum (tiny drift)
    h_new *= (n / 2.0) / h_new.sum()
    return h_new

def _optimize(start_h, budget, start_time):
    """Run a single SLSQP optimization with time guard."""
    n = start_h.size
    dx = 2.0 / n
    cons = ({
        "type": "eq",
        "fun": lambda x: x.sum() - n / 2.0
    },)

    bounds = [(0.0, 1.0)] * n

    def obj(x):
        # early exit if time exhausted
        if time.time() - start_time > budget:
            raise TimeoutError
        return _c5_value(x)

    options = {"maxiter": 200, "ftol": 1e-9, "disp": False}
    try:
        res = minimize(obj, start_h, method="SLSQP", bounds=bounds,
                       constraints=cons, options=options)
    except TimeoutError:
        return None, np.inf
    if not res.success:
        return None, np.inf
    return res.x, _c5_value(res.x)
