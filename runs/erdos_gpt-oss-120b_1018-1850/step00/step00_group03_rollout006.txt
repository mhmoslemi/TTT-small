import numpy as np
import time
import math

def evaluate_score(h, dx):
    """Return max overlap C5 for binary step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_binary_state(n, rng):
    """Generate binary array of length n with exactly n/2 ones."""
    ones = n // 2
    arr = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=ones, replace=False)
    arr[idx] = 1.0
    return arr

def improve_state(initial, n, budget_s, rng):
    """Simulated annealing search for lower C5."""
    dx = 2.0 / n
    cur = initial.copy()
    cur_score = evaluate_score(cur, dx)
    best = cur.copy()
    best_score = cur_score

    start_time = time.time()
    end_time = start_time + budget_s - 0.5  # safety margin

    # temperature schedule parameters
    max_iter = 200000
    temp_start = 0.02
    temp_end = 1e-5

    it = 0
    while time.time() < end_time and it < max_iter:
        it += 1
        # linear temperature decay
        t = temp_start + (temp_end - temp_start) * (it / max_iter)

        # propose a swap between a 1 and a 0
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # perform swap
        cur[i], cur[j] = 0.0, 1.0
        new_score = evaluate_score(cur, dx)

        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / t):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert swap
            cur[i], cur[j] = 1.0, 0.0

    return best, best_score

        # If length differs, resize by tiling then fixing count

            init = np.tile(init, repeats)[:n_points]
        # Ensure binary and correct sum
        init_binary = (init > 0.5).astype(np.float64)
        cur_sum = init_binary.sum()
        target_sum = n_points // 2
        if cur_sum != target_sum:
            # adjust by random flips preserving binary nature
            diff = int(target_sum - cur_sum)
            if diff > 0:
                zeros = np.where(init_binary == 0)[0]

        initial = init_binary
