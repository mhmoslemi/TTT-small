import numpy as np, time, math, random

def _evaluate_erdos(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(corr.max())

def _repair(h, target_sum):
    # clip and renormalize to target sum while staying in [0,1]
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # fallback to uniform distribution
        h[:] = target_sum / len(h)
        return h
    # scale to meet sum constraint, then clip again
    h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    # final adjustment for any residual error
    err = target_sum - h.sum()
    if abs(err) > 1e-12:
        # distribute error uniformly (small adjustments keep bounds)
        h += err / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h
