import time
import numpy as np

def _project_h(h, target_sum):
    """Project vector h onto [0,1] box with given sum via iterative clipping."""
    h = np.clip(h, 0.0, 1.0)
    # simple iterative redistribution
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # distribute equally to non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # all at bounds, distribute to all then clip again
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        add = diff / free.sum()
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
    # final correction if slight off due to numerical issues
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        # adjust uniformly then re‑clip once
        h += (target_sum - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Compute C5 for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()
