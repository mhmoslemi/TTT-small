import numpy as np, time, math

def _max_overlap(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def _project(h, target_sum):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    curr = h.sum()
    if abs(curr - target_sum) < 1e-12:
        return h
    # adjust uniformly while respecting bounds
    diff = target_sum - curr
    # simple redistribution: add diff / n to all, then reclip iteratively
    n = h.size
    h += diff / n
    # iterate to enforce bounds
    for _ in range(10):
        over = h - 1.0
        under = -h
        if over.max() <= 0 and under.max() <= 0:
            break
        # bring back excess mass from overfull entries
        excess = over.clip(min=0).sum()
        deficit = (-under).clip(min=0).sum()
        if excess > 0:
            h[over > 0] -= excess / (over > 0).sum()
        if deficit > 0:
            h[under > 0] += deficit / (under > 0).sum()
        h = np.clip(h, 0.0, 1.0)
    # final scaling to exact sum if tiny error remains
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        h *= target_sum / cur
    return h

    # parameters for local search

        # propose a move: shift epsilon from i to j
        i_idx, j_idx = rng.integers(0, n_points, size=2)
        if i_idx == j_idx:
            continue
        # max amount we can move without breaking bounds
        max_move = min(h[i_idx], 1.0 - h[j_idx])
        if max_move <= 0:
            continue
        epsilon = rng.random() * max_move
        h_new = h.copy()
        h_new[i_idx] -= epsilon

        # no need to reproject sum is preserved
        val = _max_overlap(h_new, dx)
