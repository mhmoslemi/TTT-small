import time
import numpy as np

def _adjust_binary(vec, target):
    """Flip random bits until sum equals target (maintains binary)."""
    v = vec.copy()
    cur = int(v.sum())
    rng = np.random.default_rng()
    while cur != target:
        i, j = rng.choice(len(v), size=2, replace=False)
        # flip both
        v[i] = 1 - v[i]
        v[j] = 1 - v[j]
        cur = int(v.sum())
    return v

def _c5_score(h):
    """Compute C5 for binary array h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * dx
