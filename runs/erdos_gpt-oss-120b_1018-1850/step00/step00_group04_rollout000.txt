import time, math, numpy as np, random
# evaluate_erdos_solution and initial_h_values are provided in the environment

def evaluate_erdos(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    dx = 2.0 / len(h)
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_feasible(h):
    """Clip to [0,1] and renormalize so sum(h)=n/2."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = n / 2.0
    if total == 0:
        # avoid division by zero; distribute uniformly
        h[:] = target / n
    else:
        h *= target / total
    # after scaling some entries may leave bounds; a few passes fix this
    for _ in range(3):
        over = h > 1.0
        under = h < 0.0
        if not over.any() and not under.any():
            break
        excess = (h[over] - 1.0).sum()
        deficit = (-h[under]).sum()
        # move excess to deficit proportionally
        if excess > 0 and deficit > 0:
            h[over] = 1.0
            h[under] = 0.0
            # redistribute excess+deficit uniformly
            total = h.sum()
            h *= target / total
        else:
            # simple clip and renorm
            h = np.clip(h, 0.0, 1.0)
            total = h.sum()
            h *= target / total
    return h

def neighbor(h, step):
    """Create a neighbor by perturbing two random positions."""
    n = len(h)
    i, j = random.sample(range(n), 2)
    nh = h.copy()
    delta = (random.random() * 2 - 1) * step
    nh[i] = np.clip(nh[i] + delta, 0.0, 1.0)
    nh[j] = np.clip(nh[j] - delta, 0.0, 1.0)  # keep sum roughly constant
    return project_feasible(nh)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    random.seed(seed)

    best_h = h0
    best_score = evaluate_erdos(best_h)

    # Simple simulated annealing / hill climbing
    max_iter = 1000000
    step = 0.2  # initial perturb size
    decay = 0.9995
    for it in range(max_iter):

    best_h = project_feasible(best_h)
    best_score = evaluate_erdos(best_h)
