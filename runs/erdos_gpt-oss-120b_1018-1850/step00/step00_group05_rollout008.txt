import numpy as np, time, random, math

def compute_c5(h):
    """Return C5 for a step function given by array h on [0,2]."""
    dx = 2.0 / h.size
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return conv.max()

def project_bounds_sum(v, target_sum):
    """Project v onto [0,1] box and adjust to have given sum (if possible)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    # Simple redistribution of deficit/excess uniformly within free entries
    diff = target_sum - cur
    free = (v > 0.0) & (v < 1.0)
    if not free.any():
        # fallback: scale all values
        return np.clip(v * (target_sum / cur), 0.0, 1.0)
    # distribute diff equally among free entries, iterating if limits hit
    while abs(diff) > 1e-12 and free.any():
        add = diff / free.sum()
        v[free] += add
        # enforce bounds again
        over = v > 1.0
        under = v < 0.0
        if over.any():
            diff = v[over].sum() - 1.0 * over.sum()
            v[over] = 1.0
        elif under.any():
            diff = v[under].sum()  # negative
            v[under] = 0.0
        else:
            diff = target_sum - v.sum()
        free = (v > 0.0) & (v < 1.0)
    return np.clip(v, 0.0, 1.0)

def random_adjust(h, target_sum, step=0.05):
    """Make a small random perturbation that respects bounds and sum."""
    n = h.size
    i, j = random.sample(range(n), 2)
    # transfer amount from i to j
    max_transfer = min(step, h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h.copy()
    delta = random.random() * max_transfer
    newh = h.copy()
    newh[i] -= delta
    newh[j] += delta
    # ensure sum preserved (numerical)

    return newh

    """Simple simulated‑annealing style optimizer."""

def upscale(h, new_n):
    """Upsample by linear interpolation to new_n points preserving integral."""
    old_n = h.size
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(x_new, x_old, h, period=2.0)
    # adjust to exact integral
    target_sum = new_n / 2.0
    h_new = project_bounds_sum(h_new, target_sum)
    return h_new
