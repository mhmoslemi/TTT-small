import time
import numpy as np

def _evaluate(h, dx):
    """Return C5 value for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(candidate, target_sum):
    """Clamp to [0,1] and adjust to meet the target sum."""
    # clip
    np.clip(candidate, 0.0, 1.0, out=candidate)
    # iterative scaling to hit sum
    for _ in range(10):
        s = candidate.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        # distribute diff proportionally to distance from bounds
        free = np.where((candidate > 0.0) & (candidate < 1.0), 1.0, 0.0)
        free_count = free.sum()
        if free_count == 0:
            # fallback: add equally then clip
            candidate += diff / candidate.size
            np.clip(candidate, 0.0, 1.0, out=candidate)
            continue
        candidate += diff * free / free_count
        np.clip(candidate, 0.0, 1.0, out=candidate)
    # final force exact sum by tiny adjustment on non‑boundary entries
    s = candidate.sum()
    if abs(s - target_sum) > 1e-12:
        adjust = (target_sum - s) / max(1, (candidate > 0.0).sum())
        mask = candidate > 0.0
        candidate[mask] += adjust
        np.clip(candidate, 0.0, 1.0, out=candidate)
    return candidate
