import numpy as np, time, math, random

def _project(h, target_sum):
    """Project h to [0,1] and exact sum target_sum."""
    # simple iterative projection
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        h += (target_sum - cur) / h.size
    # final clip to ensure bounds
    h = np.clip(h, 0.0, 1.0)
    # tiny correction to hit exact sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly and re‑clip
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Compute max overlap integral for step function h."""
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return conv.max()
