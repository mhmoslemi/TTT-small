import numpy as np, time, math, random
from scipy.optimize import minimize

def _max_overlap(h, dx):
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_bounds_sum(h, target_sum):
    # simple projection: clip then adjust uniformly, repeat a few times
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # final correction by distributing remaining diff to non‑boundary entries
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h
