def compute_c5(h):
    """Return max overlap integral for step function h."""
    dx = 2.0 / h.size
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def project_feasible(v):

    if abs(cur_sum - target_sum) < 1e-12:
