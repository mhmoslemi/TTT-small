import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def _c5_of_h(h, dx):
    """Compute C5 = max convolution of h with (1-h)."""
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(conv)

def _make_start_vectors(base_h, n, rng, n_start=10, perturb_scale=0.05):
    """Generate start vectors of length n from base_h (length may differ)."""
    if len(base_h) != n:
        xs_src = np.linspace(0, 2, len(base_h), endpoint=False)
        xs_tgt = np.linspace(0, 2, n, endpoint=False)
        base_h = np.interp(xs_tgt, xs_src, base_h)
    starts = []
    # ensure exact integral
    target_sum = n / 2.0
    base_h = np.clip(base_h, 0.0, 1.0)
    base_h = base_h * (target_sum / base_h.sum())
    starts.append(base_h.copy())
    for _ in range(n_start - 1):
        pert = base_h + perturb_scale * rng.standard_normal(n)
        pert = np.clip(pert, 0.0, 1.0)
        # enforce sum constraint by scaling the deviation

        # final adjustment to meet sum exactly
        cur_sum = pert.sum()
        if cur_sum != 0:
            pert *= target_sum / cur_sum
        starts.append(pert)
    return starts
