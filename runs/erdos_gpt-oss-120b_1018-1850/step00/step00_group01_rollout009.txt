import numpy as np
import time
from scipy.optimize import minimize

def _c5_of_h(h, dx):
    """Return the C5 value for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _objective(h, dx):
    return _c5_of_h(h, dx)

def _run_one_resolution(n_points, start_vecs, target_sum, dx, time_limit, start_time):
    best_h = None
    best_c5 = np.inf
    bounds = [(0.0, 1.0)] * n_points
    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    opts = {"maxiter": 200, "ftol": 1e-9, "disp": False}

    for start in start_vecs:
        if time.time() - start_time > time_limit:
            break
        res = minimize(
            _objective,
            start,
            args=(dx,),
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options=opts,
        )
        if not res.success:
            continue
        h_opt = np.clip(res.x, 0.0, 1.0)
        # enforce exact sum by a tiny scaling (doesn't affect bounds much)
        scale = target_sum / np.sum(h_opt)
        h_opt = np.clip(h_opt * scale, 0.0, 1.0)
        c5 = _c5_of_h(h_opt, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    return best_h, best_c5
