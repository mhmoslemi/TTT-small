import numpy as np
import time

def _correlation_score(h, dx):
    # max_{k} ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _repair(h, target_sum):
    # enforce 0<=h<=1 and sum == target_sum
    h = np.clip(h, 0.0, 1.0)
    # simple iterative proportional adjustment
    for _ in range(20):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        # redistribute diff uniformly over non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # all at bounds, distribute to all then clip again
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        adjust = diff / free.sum()
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling if still off
    s = h.sum()
    if abs(s - target_sum) > 1e-9:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _init_individual(n, target_sum, rng):
    # random feasible vector
    x = rng.random(n)
    x = _repair(x, target_sum)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = initial_h_values.astype(float)
        n_points = len(h0)
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        base = _repair(h0, target_sum)
        population = [base.copy()]
        # fill rest of population randomly
        pop_size = 30
        while len(population) < pop_size:
            population.append(_init_individual(n_points, target_sum, rng))
    else:
        n_points = 45
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        pop_size = 30
        population = [_init_individual(n_points, target_sum, rng) for _ in range(pop_size)]

    best_h = None
    best_score = np.inf

    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        generation += 1
        # evaluate
        scores = np.empty(len(population))
        for i, ind in enumerate(population):
            scores[i] = _correlation_score(ind, dx)
        # keep best
        min_idx = scores.argmin()
        if scores[min_idx] < best_score:
            best_score = float(scores[min_idx])
            best_h = population[min_idx].copy()
        # selection (tournament)
        new_pop = []
        while len(new_pop) < pop_size:
            # tournament
            idx = rng.choice(len(population), size=5, replace=False)
            champ = idx[scores[idx].argmin()]
            # parent copy
            parent = population[champ].copy()
            # mutation
            mutant = parent + rng.normal(scale=0.1, size=n_points)
            mutant = _repair(mutant, target_sum)
            # crossover with another random individual
            mate = population[rng.integers(len(population))]
            mask = rng.random(n_points) < 0.5
            child = np.where(mask, mutant, mate)
            child = _repair(child, target_sum)
            new_pop.append(child)
        population = new_pop

    # final evaluation of best (in case best_h None)
    if best_h is None:
        best_h = population[0]
