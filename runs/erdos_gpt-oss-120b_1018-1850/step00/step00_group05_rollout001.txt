import numpy as np, time, random, math

def _c5_bound(h):
    """Compute C5 bound for step vector h."""
    dx = 2.0 / h.size
    conv = np.correlate(h, 1 - h, mode="full") * dx
    return conv.max()

def _project(h, target_sum):
    """Project h onto [0,1] box and adjust to exact sum."""
    # clip
    np.clip(h, 0.0, 1.0, out=h)
    # adjust sum by adding/subtracting uniformly while staying in bounds
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # simple proportional correction on interior points
    mask = (h > 0) & (h < 1)
    if not np.any(mask):
        # fallback: distribute equally
        h[:] = np.clip(target_sum / h.size, 0.0, 1.0)
        return h
    adjust = diff / mask.sum()
    h[mask] += adjust
    # clamp again and iterate a few times

            break
        mask = (h > 0) & (h < 1)
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
    return h
