import numpy as np
import time

def compute_c5(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_bounds_sum(h, target_sum):
    # clip to [0,1] then adjust to meet sum by adding/subtracting uniformly
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur == target_sum:
        return h
    # simple iterative scaling: shift all values proportionally within bounds
    diff = target_sum - cur
    # distribute diff uniformly, then re‑clip and repeat a few times
    for _ in range(5):
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            break
        delta = diff / np.count_nonzero(free)
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
    # final tiny correction on a random free index if needed
    if abs(diff) > 1e-12:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size:
            i = np.random.choice(idx)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_initial(n):
    # start with binary 0/1 having exactly n/2 ones
    ones = n // 2
    h = np.zeros(n)
    h[:ones] = 1.0
    np.random.shuffle(h)
    return h.astype(float)

def neighbor(h, eps=0.05):
    n = h.size
    a, b = np.random.choice(n, 2, replace=False)
    # move amount from a to b
    amount = min(eps, h[a])                     # cannot go below 0 at a
    amount = min(amount, 1.0 - h[b])            # cannot exceed 1 at b
    if amount == 0:
        return h.copy()
    h_new = h.copy()
    h_new[a] -= amount
    h_new[b] += amount
    return h_new
