import numpy as np
import time
from scipy.optimize import minimize

def _c5_objective(h, dx):
    """Return max overlap value for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_start(base_h, n):
    """Interpolate base_h to length n and enforce sum constraint."""
    x_old = np.linspace(0.0, 2.0, len(base_h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, base_h)
    # clip and renormalize to exact integral
    h_new = np.clip(h_new, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h_new.sum()
    if cur_sum == 0:
        h_new[:] = target_sum / n
    else:
        h_new *= target_sum / cur_sum
    return h_new

def _perturb(h, scale=0.05):
    """Add small noise and renormalize while staying in [0,1]."""
    n = len(h)
    noise = np.random.randn(n) * scale
    h2 = h + noise
    h2 = np.clip(h2, 0.0, 1.0)
    # enforce sum constraint
    target = n / 2.0
    cur = h2.sum()
    if cur > 0:
        h2 *= target / cur
    else:
        h2[:] = target / n
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = np.inf
    best_n = None

    # Determine available initial sample if present
    try:
        init_h = initial_h_values  # provided by harness
        n0 = len(init_h)
    except NameError:
        init_h = None
        n0 = None

    # Candidate resolutions
    n_list = []
    if n0 is not None:
        n_list.append(n0)
    n_list.extend([64, 96, 128])

    # Main search loop
    for n in n_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n

        # Build a base function for this resolution
        if init_h is not None:
            base = _make_start(init_h, n)
        else:
            # Uniform start with correct integral
            base = np.full(n, 0.5)

        starts = [base]
        # add perturbed starts
        for _ in range(8):
            starts.append(_perturb(base, scale=0.07))

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            # constraints: sum = n/2, bounds [0,1]
            cons = ({
                "type": "eq",
                "fun": lambda h, target=n/2.0: h.sum() - target,
            })
            bounds = [(0.0, 1.0)] * n

            res = minimize(
                fun=lambda h: _c5_objective(h, dx),
                x0=h0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )

            if not res.success:
                continue

            val = _c5_objective(res.x, dx)
            if val < best_val:
                best_val = val
                best_h = res.x.copy()
                best_n = n

    # Fallback if nothing succeeded
    if best_h is None:
        if init_h is not None:
            best_h = init_h.copy()
            best_n = len(best_h)
            best_val = _c5_objective(best_h, 2.0 / best_n)
        else:
            raise RuntimeError("Optimization failed without any initial data.")

    return best_h, float(best_val), int(best_n)
