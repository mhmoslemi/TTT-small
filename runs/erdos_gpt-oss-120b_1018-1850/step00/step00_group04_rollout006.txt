def _project_feasible(h, target_sum):
    """Project h onto [0,1] box and enforce sum == target_sum."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # simple redistribution loop
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:

def _neighbor(h, rng, step_scale=0.2):
    """Create a nearby feasible vector by swapping mass between two positions."""
    n = h.size
    # pick two distinct indices
    i, j = rng.choice(n, 2, replace=False)
    # amount to move, limited by remaining capacity at each side
    max_move = min(h[i], 1.0 - h[j])

    delta = np.clip(delta, -h[i], 1.0 - h[j])
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2
