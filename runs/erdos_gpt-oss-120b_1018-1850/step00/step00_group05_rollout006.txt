import numpy as np, time, math, random
from scipy.signal import correlate

def _project(h, target_sum):
    """
    Project h onto box [0,1] and enforce sum == target_sum.
    Simple iterative scaling/clipping.
    """
    n = h.size
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum by uniform shift, then re‑clip, repeat a few times
    for _ in range(5):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final correction if still off (distribute residual greedily)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # add/subtract small amount to/from entries not at bounds
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size == 0:
            # fallback: distribute to all
            idx = np.arange(n)
        inc = diff / idx.size
        h[idx] += inc
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Compute C5 = max convolution overlap for step function h."""
    conv = correlate(h, 1.0 - h, mode='full') * dx
    return conv.max()
