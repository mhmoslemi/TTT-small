import time, numpy as np, warnings
from scipy.optimize import minimize

# Helper: compute C5 for a given h (array) and dx
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # target integral condition
    def sum_constraint(x, target):
        return np.sum(x) - target

    # generate starting points for a given resolution n
    def make_start_points(base_h, n, num=10):
        base = np.interp(
            np.linspace(0, 2, n, endpoint=False),
            np.linspace(0, 2, len(base_h), endpoint=False),
            base_h,
        )
        starts = [base.copy()]
        target_sum = n / 2.0
        for _ in range(num - 1):
            noisy = base + rng.normal(scale=0.05, size=n)
            noisy = np.clip(noisy, 0.0, 1.0)
            # Adjust to satisfy sum constraint (simple scaling + shift)
            cur = np.sum(noisy)
            if cur != 0:
                noisy *= target_sum / cur
            noisy = np.clip(noisy, 0.0, 1.0)
            starts.append(noisy)
        return starts

    # initial base vector (if provided)
    try:
        base_vec = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: uniform function with correct integral
        base_vec = np.full(47, 0.5)
        base_vec *= (47 / 2.0) / np.sum(base_vec)

    resolutions = [len(base_vec), 64, 100]
    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
        starts = make_start_points(base_vec, n, num=10)
        target_sum = n / 2.0
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = {"type": "eq", "fun": sum_constraint, "args": (target_sum,)}

        for x0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            def obj(x):
                return compute_c5(x, 2.0 / n)

            # use SLSQP with modest iteration limit
            with warnings.catch_warnings():
                warnings.simplefilter("ignore")
                res = minimize(
                    obj,
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 200, "ftol": 1e-9, "disp": False},
                )
            if not res.success:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)
            # enforce sum exactly (tiny numerical drift)
            diff = target_sum - np.sum(h_opt)
            if abs(diff) > 1e-8:
                h_opt += diff / n
                h_opt = np.clip(h_opt, 0.0, 1.0)

            c5_val = compute_c5(h_opt, 2.0 / n)
            if c5_val < best_c5:
                best_c5, best_h, best_n = c5_val, h_opt.copy(), n

    # fallback if nothing succeeded
    if best_h is None:
        best_h = base_vec.copy()
        best_c5 = compute_c5(best_h, 2.0 / len(best_h))
        best_n = len(best_h)

    return (best_h, float(best_c5), int(best_n))
