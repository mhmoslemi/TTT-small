import numpy as np
import time
from math import inf

def evaluate_h(h, dx):
    """Return max overlap C5 for given h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    """Clip to [0,1] and then adjust linearly to hit target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if cur_sum == 0:
        # distribute uniformly
        return np.full_like(h, target_sum / h.size)
    # scale to hit target sum while staying in [0,1] as much as possible
    # simple linear scaling; if it pushes outside bounds, clip and repeat

        h = np.clip(h * scale, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) < 1e-12:
            break
    return h
