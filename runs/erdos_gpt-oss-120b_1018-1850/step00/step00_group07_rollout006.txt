import numpy as np
import time
import math
import random

def _adjust_to_sum(binary_vec, desired):
    """Flip pairs to achieve exact sum while keeping values binary."""
    vec = binary_vec.copy()
    cur = int(vec.sum())
    if cur == desired:
        return vec
    # get index lists
    ones = np.flatnonzero(vec == 1)
    zeros = np.flatnonzero(vec == 0)
    while cur != desired:
        if cur > desired:
            # need to turn a 1 into 0 and a 0 into 1
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
        else:
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
        vec[i] = 1 - vec[i]
        vec[j] = 1 - vec[j]
        # update lists and count
        cur = int(vec.sum())
        ones = np.flatnonzero(vec == 1)
        zeros = np.flatnonzero(vec == 0)
    return vec

def _evaluate_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
