import numpy as np
import time

def _evaluate(h, dx):
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_swap(state, rng):
    # pick two indices with different values and swap them
    idx = rng.choice(len(state), size=2, replace=False)
    if state[idx[0]] != state[idx[1]]:
        state[idx[0]], state[idx[1]] = state[idx[1]], state[idx[0]]
        return True
    return False
