import time
import numpy as np
from scipy.optimize import minimize

def _c5_value(h, dx):
    """Compute C5 for a given step vector h."""
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(conv)

def _project(h, target_sum):
    """Project h onto box [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        return np.full_like(h, target_sum / h.size)
    # Scale to meet the sum while staying in bounds
    diff = target_sum - cur
    # Distribute diff proportionally to distance from bounds
    free = ((h > 0) & (h < 1)).astype(float)
    if free.sum() == 0:
        # all at bounds, just add uniformly and re‑clip
        h += diff / h.size
        return np.clip(h, 0.0, 1.0)
    adjust = diff * free / free.sum()
    h += adjust
    # final clip (might push sum slightly off, iterate once)
    h = np.clip(h, 0.0, 1.0)
    return h

    n = h0.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility of the starting point
    h0 = _project(h0, target_sum)

    # Objective for optimizer (returns scalar C5)
    def obj(vec):
        return _c5_value(vec, dx)

    # Equality constraint: sum(vec) = target_sum
    cons = ({
        'type': 'eq',
        'fun': lambda v: v.sum() - target_sum
    },)

    bounds = [(0.0, 1.0) for _ in range(n)]

    # Run SLSQP optimization with a time guard
    max_iter = 300
    best_h = h0.copy()
    best_val = obj(best_h)

    def callback(xk):
        nonlocal best_h, best_val, start_time
        # keep best solution seen so far
        val = obj(xk)
