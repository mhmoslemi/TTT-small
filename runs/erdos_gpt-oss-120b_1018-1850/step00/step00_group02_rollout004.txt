import numpy as np
import time

# evaluation function provided by the harness
# from pre-imported: evaluate_erdos_solution(h) returns max overlap value
# initial_h_values may be defined by the harness

def _make_initial(n):
    """Create a feasible binary start vector of length n with sum n/2."""
    # try to use provided initial_h_values if available
    global initial_h_values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # ensure correct length
        if h0.size != n:
            # truncate or pad by repeating pattern
            repeats = (n + h0.size - 1) // h0.size
            h0 = np.tile(h0, repeats)[:n]
        # project to binary while keeping total mass
        h_bin = (h0 > 0.5).astype(float)
        curr = int(h_bin.sum())
        target = n // 2
        # repair count
        diff = target - curr
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            idx = np.random.choice(zeros, diff, replace=False)
            h_bin[idx] = 1.0
        elif diff < 0:
            ones = np.where(h_bin == 1)[0]
            idx = np.random.choice(ones, -diff, replace=False)
            h_bin[idx] = 0.0
        return h_bin
    # no initial provided: random binary with exact sum
    h_bin = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h_bin[ones] = 1.0
    return h_bin

def _evaluate(h):

def _swap_move(h):
    """Return a new vector after swapping a random 1 with a random 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return h.copy()
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function for the Erdős minimum overlap problem.

    Returns:
        h_values (np.ndarray): optimized values in [0,1], length n_points
        c5_bound (float): achieved max overlap
        n_points (int): number of discretisation points
    """
    np.random.seed(seed)
    start_time = time.time()

    # choose a modest size to stay fast; 200 points gives dx=0.01
    n_points = 200
    dx = 2.0 / n_points

    # initial feasible vector
    h = _make_initial(n_points)
    best_h = h.copy()
    best_val = _evaluate(best_h)

    # simple simulated annealing / hill climbing
    T0 = 0.05
    T_min = 1e-5
    decay = 0.995

    T = T0
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:  # leave tiny margin
            break

        # generate neighbor
        h_candidate = _swap_move(h)
        cand_val = _evaluate(h_candidate)

        # accept if better or by probability
        if cand_val < best_val:
            best_h, best_val = h_candidate, cand_val

        accept = False
        if cand_val <= _evaluate(h):
            accept = True
        else:
            prob = np.exp(-(cand_val - _evaluate(h)) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h = h_candidate

        # temperature schedule
        T = max(T * decay, T_min)
        iteration += 1

    # final sanity projection (should already be binary)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_val), n_points
