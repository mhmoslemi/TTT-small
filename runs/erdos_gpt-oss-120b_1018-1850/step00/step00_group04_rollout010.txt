import numpy as np
import time

def evaluate_erdos(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(v):
    """Project vector v onto the feasible set: 0<=v<=1 and sum(v)=n/2."""
    n = v.shape[0]
    # clip first
    v = np.clip(v, 0.0, 1.0)
    # adjust to meet sum constraint by adding a constant then re‑clip
    target = n / 2.0
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # distribute diff proportionally to distance from bounds
    free = (v > 0.0) & (v < 1.0)
    if free.any():
        adjust = diff / free.sum()
        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
    # if still off, scale whole vector linearly
    if abs(v.sum() - target) > 1e-9:
        v = v * (target / v.sum())
    return v

def random_neighbor(h, scale=0.05):
    """Create a neighboring feasible vector by perturbing two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
