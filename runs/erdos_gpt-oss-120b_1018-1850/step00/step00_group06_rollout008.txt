import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Return the maximal overlap integral for a discretised h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def feasibility_projection(h, target_sum):
    """Project h onto the box [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if current == 0:
        return np.full_like(h, target_sum / h.size)
    # simple scaling then clipping
    scale = target_sum / current
    h = np.clip(h * scale, 0.0, 1.0)
    # If still off, adjust uniformly on free entries
    diff = target_sum - h.sum()
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    Returns:
        h_values (np.ndarray): optimized values on the chosen grid.
        c5_bound (float): achieved maximal overlap integral.
        n_points (int): number of discretisation points used.

    np.random.seed(seed)
    start_time = time.time()

    # Determine number of points: use the size of initial_h_values if present,
    # otherwise a modest default (e.g., 120) that respects the <1k rule.

    dx = 2.0 / n
    target_sum = n / 2.0  # ensures integral equals 1

    # Ensure initial feasibility
    h0 = feasibility_projection(init, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Helper for the optimiser
    def objective(vec):
        return compute_c5(vec, dx)

    # Equality constraint: sum(vec) == target_sum
    eq_cons = {"type": "eq", "fun": lambda v: v.sum() - target_sum}
    bounds = [(0.0, 1.0)] * n

    # Main optimisation loop – perform several restarts within the time budget
    while time.time() - start_time < budget_s * 0.95:

        x0 = np.clip(best_h + perturb, 0.0, 1.0)
        x0 = feasibility_projection(x0, target_sum)

        res = minimize(
            objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
