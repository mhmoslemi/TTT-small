import numpy as np
import time
import math

def _evaluate(h, dx):
    """Return C5 for binary step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_initial(n_points, rng):
    """Create an initial feasible binary vector of length n_points."""
    # Try to use the provided parent if it exists
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
        # Tile or trim to target length
        if len(parent) == n_points:
            h = parent.copy()
        else:
            repeats = int(np.ceil(n_points / len(parent)))
            tiled = np.tile(parent, repeats)[:n_points]
            h = tiled.copy()
        # Round to 0/1 while preserving total mass as much as possible
        h = np.rint(h)
    except NameError:
        # No parent: start from a random balanced 0/1 vector
        h = np.zeros(n_points, dtype=np.float64)

    # Ensure binary
    h = np.clip(h, 0, 1)

    # Enforce exact sum = n_points/2
    target = n_points // 2
    current = int(h.sum())
    idxs = np.arange(n_points)
    rng.shuffle(idxs)
    if current < target:
        # flip zeros to ones
        zero_idxs = idxs[h[idxs] == 0]
        need = target - current
        h[zero_idxs[:need]] = 1.0
    elif current > target:
        # flip ones to zeros
        one_idxs = idxs[h[idxs] == 1]
        need = current - target
        h[one_idxs[:need]] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.

    Returns
    -------
    h_values : np.ndarray
        Binary array of length n_points satisfying the constraints.
    c5_bound : float
        Achieved upper bound C5.
    n_points : int
        Number of discretization points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose discretization size; 140 balances speed and quality
    n_points = 140
    dx = 2.0 / n_points

    # Initial feasible state
    h_curr = _make_initial(n_points, rng)
    best_h = h_curr.copy()
    best_score = _evaluate(best_h, dx)

    # Simulated annealing parameters
    max_iter = 2000000  # safety upper bound
    temp_initial = 0.2
    temp_final = 1e-4

    for it in range(max_iter):
        # Time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # Linear temperature schedule based on elapsed fraction
        frac = (time.time() - start_time) / budget_s
        temp = temp_initial * (1 - frac) + temp_final * frac

        # Propose a swap between a 1 and a 0
        ones = np.flatnonzero(h_curr == 1)
        zeros = np.flatnonzero(h_curr == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # Perform swap
        h_new = h_curr.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_score = _evaluate(h_new, dx)

        # Acceptance test
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
        if new_score < _evaluate(h_curr, dx) or rng.random() < math.exp(-(new_score - _evaluate(h_curr, dx)) / max(temp, 1e-12)):
            h_curr = h_new

    return best_h, best_score, n_points
