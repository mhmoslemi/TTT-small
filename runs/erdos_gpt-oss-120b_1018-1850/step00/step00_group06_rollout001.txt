import numpy as np, time, random, math

def _compute_c5(h):
    """Return C5 for a given discretized h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_bounds_and_sum(x, target_sum):
    """Project vector x onto [0,1] box and then adjust to hit target_sum."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    if cur == 0:
        # equally distribute
        return np.full_like(x, target_sum / x.size)
    # scale to hit target sum, then re‑clip and iterate a few times
    for _ in range(5):
        factor = target_sum / cur
        x *= factor
        np.clip(x, 0.0, 1.0, out=x)
        cur = x.sum()
    # final tiny correction
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly on non‑saturated entries
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            np.clip(x, 0.0, 1.0, out=x)
    return x
