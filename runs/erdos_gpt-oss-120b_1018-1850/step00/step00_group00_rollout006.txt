import numpy as np
import time
import random

def _repair_bounds_and_sum(h, target_sum):
    """Project h onto [0,1] box and enforce sum == target_sum."""
    # clip to bounds
    h = np.clip(h, 0.0, 1.0)
    # simple iterative redistribution
    max_iter = 10
    for _ in range(max_iter):
        current_sum = h.sum()
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound
        free = (h > 0.0) & (h < 1.0)
        free_cnt = free.sum()
        if free_cnt == 0:
            # all at bounds, distribute to all proportionally
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        # distribute equally among free variables
        delta = diff / free_cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling the free part
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-9:
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0.0, 1.0)
        # if still off, linearly adjust the first element
        residual = target_sum - h.sum()
        h[0] = np.clip(h[0] + residual, 0.0, 1.0)
    return h

def _evaluate(h, dx):
    """Return C5 value for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _random_candidate(n, target_sum, rng):
    """Generate a random feasible candidate."""
    # start with uniform random numbers
    h = rng.random(n)
    h = _repair_bounds_and_sum(h, target_sum)
    return h

    best_idx = int(fitness.argmin())
    best_h = population[best_idx].copy()

    tournament_k = 5

    while time.time() - start_time < budget_s:
        # --- Selection ---

        for _ in range(pop_size):
            participants = rng.choice(pop_size, size=tournament_k, replace=False)
            chosen = participants[int(fitness[participants].argmin())]
            parents.append(population[chosen])

        # --- Crossover & Mutation ---
        next_pop = []
        for i in range(0, pop_size, 2):
