import numpy as np
import time
import math
import random

# helper: evaluate max overlap
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# repair: clip to [0,1] and enforce sum = target_sum
def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # simple proportional adjustment while respecting bounds
    for _ in range(10):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # distribute uniformly and clip again
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        adjust = diff / free.sum()
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to match sum exactly
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # distribute residual to free slots
        free = (h > 0.0) & (h < 1.0)
        if np.any(free):
            h[free] += (target_sum - cur_sum) / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: scale whole vector and clip
            h = np.clip(h * (target_sum / cur_sum), 0.0, 1.0)
    return h

def tournament_select(pop, scores, k, rng):
    idx = rng.choice(len(pop), size=k, replace=False)
    best = idx[0]
    for i in idx[1:]:
        if scores[i] < scores[best]:
            best = i
    return pop[best]

def crossover(parent1, parent2, rng):
    mask = rng.random(parent1.size) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(child, rng, sigma):
    mut = rng.standard_normal(child.size) * sigma
    child = child + mut
    child = np.clip(child, 0.0, 1.0)
    return child

def run(seed=42, budget_s=1100, **kwargs):
