import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h, dx):
    """Maximum overlap integral via full correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def project_feasible(h, target_sum, max_iter=10):
    """Clip to [0,1] and enforce sum == target_sum by simple scaling+clipping."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) < 1e-12:
            break
        # scale towards target sum
        scale = target_sum / cur_sum if cur_sum > 0 else 0.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment if needed
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over non‑boundary elements
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def initial_vector(n_points):
    """Create a feasible starting vector, using the global initial_h_values if present."""
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # upscale / downscale by linear interpolation
        x_old = np.linspace(0, 2, len(base), endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_new, x_old, base)
        h = project_feasible(h, target_sum)
        return h
    # random feasible start: Dirichlet then scale into [0,1]
    rand = np.random.rand(n_points)
    h = rand / rand.sum() * target_sum
    h = project_feasible(h, target_sum)
    return h

def neighbor(h, step_scale=0.05):
    """Generate a nearby feasible vector by adding small noise and projecting."""
    n = h.size
    noise = (np.random.rand(n) - 0.5) * step_scale
    cand = h + noise
    return project_feasible(cand, h.sum())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # choose a moderate resolution; 200 gives decent accuracy and speed
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # initial candidate
    best_h = initial_vector(n_points)
    best_c5 = compute_c5(best_h, dx)

    # simple simulated‑annealing / hill‑climb loop
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # adapt step size over time
        progress = (time.time() - start_time) / budget_s
        step = 0.1 * (1 - progress) + 0.01 * progress
        cand = neighbor(best_h, step_scale=step)
        cand_c5 = compute_c5(cand, dx)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
        # occasional random restart to escape local minima
        if iteration % 500 == 0:
            rand_h = initial_vector(n_points)
            rand_c5 = compute_c5(rand_h, dx)
            if rand_c5 < best_c5:
                best_h, best_c5 = rand_h, rand_c5

    # final polishing with a small local optimizer (COBYLA) starting from best_h
    def obj(vec):
        return compute_c5(project_feasible(vec, target_sum), dx)

    cons = ({
        'type': 'eq',
        'fun': lambda v: v.sum() - target_sum
    },)
    bounds = [(0.0, 1.0)] * n_points
    res = minimize(
        obj,
        best_h,
        method='COBYLA',
        constraints=cons,
        bounds=bounds,
        options={'maxiter': 2000, 'tol': 1e-6, 'rhobeg': 0.05}
    )
    final_h = project_feasible(res.x, target_sum)
    final_c5 = compute_c5(final_h, dx)

    return final_h, float(final_c5), n_points
