import numpy as np, time, random, math
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the C5 value for a discretized h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _make_feasible(h, target_sum):
    """Project h onto [0,1] box and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # distribute uniformly
        return np.full_like(h, target_sum / h.size)
    # scale to meet sum, then re‑clip
    h = h * (target_sum / s)
    # occasional small drift outside [0,1] after scaling
    return np.clip(h, 0.0, 1.0)

def _objective(x, target_sum):
    """Objective for COBYLA: the max overlap."""
    h = _make_feasible(x, target_sum)
    return _compute_c5(h)

def _eq_constraint(x, target_sum):
    """Equality constraint: sum == target_sum."""
    return x.sum() - target_sum

    bounds = [(0.0, 1.0)] * n
