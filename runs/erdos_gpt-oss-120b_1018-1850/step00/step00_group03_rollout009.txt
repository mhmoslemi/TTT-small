import numpy as np, time, math, random

def _evaluate(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _project_to_feasible(x, target_sum):
    """Project a real vector onto [0,1] with a given sum (simple clipping then redistribution)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return x
    # distribute the difference uniformly over non‑boundary entries
    mask = (x > 0.0) & (x < 1.0)
    if mask.any():
        adjust = diff / mask.sum()
        x[mask] += adjust
        # clip again to stay feasible
        x = np.clip(x, 0.0, 1.0)
    else:
        # fallback: set a random subset to 1 until sum matches
        idx = np.arange(len(x))
        np.random.shuffle(idx)
        needed = int(round(target_sum))
        x[:] = 0.0
        x[idx[:needed]] = 1.0
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function approximation h, its C5 bound, and number of points."""
    rnd = np.random.default_rng(seed)
    start_time = time.time()
    # --------------------------------------------------------------
    # choose discretisation size
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2  # because integral must be 1
    # --------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        # upscale if needed
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] != n_points:
            if n_points % init.shape[0] == 0:
                factor = n_points // init.shape[0]
                init = np.repeat(init, factor)
            else:
                # fallback to random binary with correct sum
                init = None
        if init is not None:
            # force feasibility
            init = _project_to_feasible(init, target_sum)
            h = init.copy()
        else:
            h = np.zeros(n_points)
            ones = rnd.choice(n_points, size=int(target_sum), replace=False)
            h[ones] = 1.0
    else:
        # random binary feasible start
        h = np.zeros(n_points)
        ones = rnd.choice(n_points, size=int(target_sum), replace=False)
        h[ones] = 1.0

    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # pre‑compute index sets
    def _indices(arr):
        return np.where(arr > 0.5)[0], np.where(arr <= 0.5)[0]

    # annealing schedule parameters
    max_iters = 2000000  # safe upper bound; will break on time
    temp_start = 0.5
    temp_end = 1e-4

    it = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        it += 1
        # temperature decreases logarithmically
        t = temp_start * (temp_end / temp_start) ** (it / max_iters)

        ones_idx, zeros_idx = _indices(h)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        # propose a swap of one 1 and one 0
        i = rnd.choice(ones_idx)
        j = rnd.choice(zeros_idx)
        h[i], h[j] = 0.0, 1.0

        score = _evaluate(h, dx)
        delta = score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if rnd.random() < math.exp(-delta / max(t, 1e-12)):
                accept = True

        if accept:
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

        # occasional large‑scale perturbation to escape plateaus
        if it % 5000 == 0:
            # pick 5 random pairs and flip them
            for _ in range(5):
                ones_idx, zeros_idx = _indices(h)
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                i = rnd.choice(ones_idx)
                j = rnd.choice(zeros_idx)
                h[i], h[j] = 0.0, 1.0
    # final evaluation of best_h (ensure feasibility)
    best_h = _project_to_feasible(best_h, target_sum)
    best_score = _evaluate(best_h, dx)
    return best_h, float(best_score), int(n_points)
