def _c5_score(h, dx):
    """Compute C5 for binary vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Parameters
    n_points = 80               # keep relatively small for speed
    desired_sum = n_points // 2
    dx = 2.0 / n_points

    # Initial feasible binary vector
    best_h = _make_binary_initial(n_points, desired_sum)
    best_score = _c5_score(best_h, dx)

    # Simulated annealing schedule
    T0 = 5.0                     # moderate initial temperature
    T_min = 1e-4
    cooling_factor = 0.995
    T = T0

    # Main loop
    while True:
        # time guard
        if time.time() - start_time > budget_s:
            break
        # propose a two‑bit flip that preserves the sum
        idx_one = np.random.choice(np.where(best_h == 1)[0])
        idx_zero = np.random.choice(np.where(best_h == 0)[0])
        new_h = best_h.copy()
        new_h[idx_one] = 0
        new_h[idx_zero] = 1
        new_score = _c5_score(new_h, dx)

        # acceptance
        if new_score < best_score or np.random.rand() < np.exp(-(new_score - best_score) / max(T, 1e-12)):

        # cool
        T = max(T * cooling_factor, T_min)

    return (best_h.astype(float), best_score, n_points)
