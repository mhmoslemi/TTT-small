import numpy as np, time, math, random

def _adjust_to_sum(vec, desired):
    """Make binary vector with exact sum=desired by flipping random bits."""
    v = np.array(vec, dtype=np.int8)
    cur = int(v.sum())
    n = len(v)
    while cur != desired:
        i = random.randrange(n)
        j = random.randrange(n-1)
        if j >= i:
            j += 1
        # flip both
        vi, vj = v[i], v[j]
        v[i] = 1 - vi
        v[j] = 1 - vj
        cur = cur - vi + v[i] - vj + v[j]
    return v

def _evaluate_c5(h, dx):
    """Return max overlap for binary/real h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
