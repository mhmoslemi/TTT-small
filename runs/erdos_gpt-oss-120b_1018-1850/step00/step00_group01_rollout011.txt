import time
import numpy as np
from scipy import optimize

def _correlation_max(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _optimize_start(start, n):
    bounds = [(0.0, 1.0)] * n
    cons = ({
        "type": "eq",
        "fun": lambda x: x.sum() - n / 2.0,
    },)

    def fun(x):
        return _correlation_max(x)

    res = optimize.minimize(
        fun,
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-12, "disp": False},
    )
    if not res.success:
        return None, np.inf
    x = np.clip(res.x, 0.0, 1.0)  # safety
    return x, _correlation_max(x)

def _generate_starts(base_h, n, rng, count=10):
    # interpolate base_h to size n
    x0 = np.linspace(0, 2, base_h.size, endpoint=False)
    x1 = np.linspace(0, 2, n, endpoint=False)
    interp = np.interp(x1, x0, base_h)
    # enforce sum constraint
    interp = interp * (n / 2.0) / interp.sum()
    starts = [interp]
    for _ in range(count - 1):
        pert = interp + rng.normal(scale=0.05, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = pert * (n / 2.0) / pert.sum()
        starts.append(pert)
    return starts
