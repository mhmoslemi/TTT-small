import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h, dx):
    # correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _make_start_vectors(base_h, n, num=10, rng=None):
    rng = np.random.default_rng(rng)
    starts = []
    # interpolate base to target size
    x_src = np.linspace(0, 2, len(base_h), endpoint=False)
    x_tgt = np.linspace(0, 2, n, endpoint=False)
    base_interp = np.interp(x_tgt, x_src, base_h)
    starts.append(base_interp.copy())
    for _ in range(num - 1):
        pert = base_interp + rng.normal(scale=0.05, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        starts.append(pert)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine available initial vector
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_init = init_h.size

    # Ensure integral constraint
    init_h = np.clip(init_h, 0.0, 1.0)
    init_h *= (n_init / 2) / init_h.sum()

    resolutions = [n_init, 64, 100]
    best_c5 = np.inf
    best_h = None
    best_n = None

    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        starts = _make_start_vectors(init_h, n, num=10, rng=rng)

        for x0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            # constraint: sum == n/2
            cons = ({
                "type": "eq",
                "fun": lambda h, target=n/2: h.sum() - target,
            })
            bounds = [(0.0, 1.0) for _ in range(n)]

            def obj(h):
                return _evaluate_c5(h, dx)

            res = minimize(
                obj,

                method="SLSQP",
                bounds=bounds,
