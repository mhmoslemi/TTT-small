import time
import numpy as np

def _make_binary_initial(n, desired_sum, seed=0):
    rng = np.random.default_rng(seed)
    # start from uniform random 0/1 then adjust
    h = rng.integers(0, 2, size=n, dtype=np.int8)
    cur = h.sum()
    while cur != desired_sum:
        # pick one index to flip from 1->0 and another 0->1
        if cur > desired_sum:
            ones = np.where(h == 1)[0]
            i = rng.choice(ones)
            h[i] = 0
            cur -= 1
        else:
            zeros = np.where(h == 0)[0]
            i = rng.choice(zeros)
            h[i] = 1
            cur += 1
    return h

def _round_and_fix(initial, n, desired_sum):
    # round to nearest integer (0 or 1)
    h = np.where(initial >= 0.5, 1, 0).astype(np.int8)
    cur = h.sum()
    rng = np.random.default_rng()
    while cur != desired_sum:
        # flip a pair to keep count parity
        i, j = rng.choice(n, size=2, replace=False)
        if cur > desired_sum:
            # need to turn a 1 into 0
            if h[i] == 1:
                h[i] = 0
                cur -= 1
            if cur > desired_sum and h[j] == 1:
                h[j] = 0
                cur -= 1
        else:
            # need to turn a 0 into 1
            if h[i] == 0:
                h[i] = 1
                cur += 1
            if cur < desired_sum and h[j] == 0:
                h[j] = 1
                cur += 1
    return h

def _c5_score(h, n):
    # correlation of h with 1-h, full mode
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    n_points = 80
    desired = n_points // 2
    rng = np.random.default_rng(seed)

    # initialise solution
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, copy=False)
        if init.size != n_points:
            # fallback to random if size mismatches
            h_current = _make_binary_initial(n_points, desired, seed)
        else:
            h_current = _round_and_fix(init, n_points, desired)
    else:
        h_current = _make_binary_initial(n_points, desired, seed)

    best_h = h_current.copy()
    best_score = _c5_score(best_h, n_points)
    current_score = best_score

    # annealing schedule parameters
    T0 = 1.0
    Tmin = 1e-5
    total_iters = 200000  # upper bound, will break on time
    for it in range(total_iters):
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        # geometric temperature decay
        T = T0 * (0.9995 ** it)
        if T < Tmin:
            T = Tmin

        # propose a two‑flip move preserving the sum
        i, j = rng.choice(n_points, size=2, replace=False)
        new_h = h_current.copy()
        new_h[i] = 1 - new_h[i]
        new_h[j] = 1 - new_h[j]

        new_score = _c5_score(new_h, n_points)

        # Metropolis acceptance
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / T):
            h_current = new_h
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()

    # final check to ensure constraint (should hold)
    if best_h.sum() != desired:
        # minor correction by fixing excess ones/zeros
        diff = best_h.sum() - desired
        if diff > 0:
            idx = np.where(best_h == 1)[0][:diff]
            best_h[idx] = 0
        elif diff < 0:
            idx = np.where(best_h == 0)[0][: -diff]
            best_h[idx] = 1

    return (best_h.astype(float), float(best_score), n_points)
