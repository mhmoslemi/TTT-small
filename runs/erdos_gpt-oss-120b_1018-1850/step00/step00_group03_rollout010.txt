import numpy as np
import time
import math

def _evaluate(h, dx):
    # correlation of h with (1-h)
    # mode='full' gives length 2n-1, max corresponds to all shifts
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _random_binary_state(n, ones, rng):
    # generate a random binary vector with exactly `ones` entries equal to 1
    arr = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=ones, replace=False)
    arr[idx] = 1.0
    return arr

def _swap_move(state, rng):
    """Swap a 1 and a 0 to keep the total sum constant."""
    ones_idx = np.flatnonzero(state == 1.0)
    zeros_idx = np.flatnonzero(state == 0.0)
    if ones_idx.size == 0 or zeros_idx.size == 0:
        return False  # nothing to swap
    i = rng.choice(ones_idx)
    j = rng.choice(zeros_idx)
    state[i], state[j] = 0.0, 1.0
    return True
