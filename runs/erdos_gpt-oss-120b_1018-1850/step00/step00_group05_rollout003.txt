import numpy as np
import time
from math import inf

def compute_c5(h):
    """Return max overlap C5 for step vector h."""
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(conv)

def project(h, target_sum):
    """Project h onto [0,1] box and adjust to meet target sum."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if current == target_sum:
        return h
    # Simple proportional redistribution of deficit/excess while staying in bounds
    deficit = target_sum - current
    # indices that are not at bounds in the direction we need to move
    if deficit > 0:
        free = (h < 1.0)
    else:
        free = (h > 0.0)
    if not np.any(free):
        return h
    # distribute evenly among free entries
    delta = deficit / np.count_nonzero(free)
    h[free] += delta
    # clip again and recurse if needed (rare)
    return project(h, target_sum)

def improve_h(h, budget, start_time):
    """Simple stochastic local search within given budget."""
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        # propose swap of small amount between two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        amt = rng.random() * 0.1  # move up to 0.1
        # direction: move from i to j
        new_h = best_h.copy()
        delta = min(amt, new_h[i])                     # cannot go below 0
        delta = min(delta, 1.0 - new_h[j])             # cannot exceed 1 at j
        new_h[i] -= delta
        new_h[j] += delta
        new_h = project(new_h, target_sum)
        val = compute_c5(new_h)
        if val < best_val:
            best_h, best_val = new_h, val
    return best_h, best_val
