import time
import numpy as np
from scipy import interpolate
from scipy.optimize import minimize, Bounds, LinearConstraint

def compute_c5(h, dx):
    """Return max overlap integral for given h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def objective(h, dx):
    """Objective for the optimizer: max overlap."""
    return compute_c5(h, dx)

def make_starting_vectors(base_h, base_n, target_n, rng, count=10):
    """Interpolate base_h to target_n and produce perturbed starts."""
    if target_n == base_n:
        base = base_h.copy()
    else:
        x_old = np.linspace(0, 2, base_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        interp = interpolate.interp1d(x_old, base_h, kind='linear', fill_value='extrapolate')
        base = interp(x_new)
        # ensure bounds and exact integral
        base = np.clip(base, 0.0, 1.0)
        base *= (target_n / 2) / base.sum()
    starts = [base]
    for _ in range(count - 1):
        pert = base + rng.uniform(-0.1, 0.1, size=target_n)
        pert = np.clip(pert, 0.0, 1.0)
        pert *= (target_n / 2) / pert.sum()
        starts.append(pert)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # possible resolutions
    resolutions = [47, 64, 100]

    # obtain initial vector if provided
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_n = len(base_h)
        # ensure it respects constraints (might be slightly off due to rounding)
        base_h = np.clip(base_h, 0.0, 1.0)
        base_h *= (base_n / 2) / base_h.sum()
    else:
        # simple uniform start
        base_n = 47
        base_h = np.full(base_n, 0.5)  # sum = base_n/2 already

    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        starts = make_starting_vectors(base_h, base_n, n, rng, count=10)

        # constraints: sum(h) = n/2, bounds [0,1]
        lin_cons = LinearConstraint(np.ones((1, n)), [n / 2], [n / 2])
        bounds = Bounds(0.0, 1.0)

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break
            res = minimize(
                objective,
                h0,
                args=(dx,),
                method='SLSQP',
                bounds=bounds,
                constraints=[lin_cons],
                options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
            )
            if not res.success:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)
            # enforce exact integral after rounding errors
            h_opt *= (n / 2) / h_opt.sum()
            c5 = compute_c5(h_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n

    # fallback if nothing succeeded
    if best_h is None:
        # use uniform function as last resort
        best_n = 47
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, float(best_c5), int(best_n)
