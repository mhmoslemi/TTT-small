import numpy as np
import time
from math import inf

def evaluate_erdos(h, dx):
    # correlation of h and 1-h
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def repair(h, target_sum):
    """Project h onto [0,1] box and adjust to meet target sum."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if abs(current - target_sum) < 1e-12:
        return h
    # Redistribute the difference while staying in bounds
    diff = target_sum - current
    # Simple proportional adjustment
    if diff > 0:
        # add to those not at 1
        free = 1.0 - h
        cap = free.sum()
        if cap > 0:
            h += diff * free / cap
    else:
        # remove from those not at 0
        free = h
        cap = free.sum()
        if cap > 0:
            h += diff * free / cap
    # final clip
    return np.clip(h, 0.0, 1.0)
