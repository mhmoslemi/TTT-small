import numpy as np
import time
import random

def _evaluate(h, dx):
    """Return max overlap C5 for a candidate h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(candidate, target_sum):
    """Clamp to [0,1] and enforce exact sum = target_sum."""
    # clip to bounds
    np.clip(candidate, 0.0, 1.0, out=candidate)
    # adjust sum
    diff = target_sum - candidate.sum()
    if abs(diff) < 1e-12:
        return candidate
    # distribute diff proportionally to free room
    # free room = (1 - x) for positive diff, (x) for negative diff
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - candidate
            cap = room.sum()
            if cap == 0:
                break
            inc = diff * (room / cap)
            candidate += inc
        else:
            room = candidate
            cap = room.sum()
            if cap == 0:
                break
            dec = diff * (room / cap)
            candidate += dec
        np.clip(candidate, 0.0, 1.0, out=candidate)
        diff = target_sum - candidate.sum()
    # final tiny correction by adding uniform offset and clipping again
    if abs(diff) > 1e-12:
        candidate += diff / candidate.size
        np.clip(candidate, 0.0, 1.0, out=candidate)
        # renorm last time
        correction = target_sum - candidate.sum()
        candidate += correction / candidate.size
        np.clip(candidate, 0.0, 1.0, out=candidate)
    return candidate

def _random_candidate(n, target_sum, rng):
    """Generate a random feasible candidate."""
    # start with Dirichlet then scale
    probs = rng.random(n)
    probs /= probs.sum()
    cand = probs * target_sum
    # ensure within bounds by simple clipping and repair
    return _repair(cand, target_sum)
