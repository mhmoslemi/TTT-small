def compute_c5(h):
    """Return max overlap C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # full convolution length = 2n-1, correlate gives sum_i h[i] * (1-h[i+k])
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust to hit target_sum, staying in bounds."""
    h = np.clip(h, 0.0, 1.0)
    # simple redistribution of the residual using free slack
    for _ in range(10):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # slack where we can move values without leaving [0,1]
        lower_slack = h - 0.0
        upper_slack = 1.0 - h
        total_slack = lower_slack.sum() + upper_slack.sum()
        if total_slack == 0:
            break
        # proportionally adjust
        adjust = diff * (lower_slack - upper_slack) / total_slack
        h += adjust
        h = np.clip(h, 0.0, 1.0)
    # final small correction by scaling the free part
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb(initial_h, time_limit):
    """Simple stochastic hill climbing within the time limit."""
    n = initial_h.size
    target_sum = n / 2.0
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
