import numpy as np
from scipy.optimize import minimize

def _correlation_max(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_to_constraints(arr, target_sum):
    # simple projection: clip, then shift to meet sum, repeat a few times
    arr = np.clip(arr, 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - arr.sum()
        arr += diff / arr.size
        arr = np.clip(arr, 0.0, 1.0)
    # final tiny correction if needed
    diff = target_sum - arr.sum()
    if abs(diff) > 1e-12:
        # distribute proportionally to slack
        slack = np.where(arr < 1.0, 1.0 - arr, 0.0)

        else:
            arr += diff / arr.size
    return np.clip(arr, 0.0, 1.0)
