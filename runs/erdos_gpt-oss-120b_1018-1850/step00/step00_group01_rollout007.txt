import numpy as np
import time
from scipy.optimize import minimize

def _c5_objective(h, dx):
    # compute max overlap integral
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _constraint_sum(n):
    # equality: sum(h) = n/2
    return {"type": "eq", "fun": lambda h: h.sum() - n / 2.0}

def _optimize_start(start, n, dx, time_deadline):
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [_constraint_sum(n)]
    opts = {"maxiter": 200, "disp": False}
    # callback to stop if out of time
    def _callback(xk):
        if time.time() > time_deadline:
            raise TimeoutError()
    try:
        res = minimize(
            _c5_objective,
            start,
            args=(dx,),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
            callback=_callback,
        )
    except TimeoutError:
        return None, None
    if not res.success:
        return None, None
    val = _c5_objective(res.x, dx)
    return res.x, val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # possible resolutions
    resolutions = [47, 64, 100]
    best_h = None
    best_c5 = np.inf
    best_n = None

    # retrieve initial if present
    init_available = "initial_h_values" in globals()
    init_h = globals().get("initial_h_values", None)

    for n in resolutions:
        dx = 2.0 / n
        starts = []

        if n == 47 and init_available and init_h is not None:
            base = np.array(init_h, dtype=float)
            starts.append(base.copy())
            # perturbations of the base
            for _ in range(9):
                pert = base + 0.05 * (np.random.rand(n) - 0.5)
                pert = np.clip(pert, 0.0, 1.0)
                # enforce sum constraint by scaling the deviation
                diff = pert.sum() - n / 2.0
                pert -= diff / n
                pert = np.clip(pert, 0.0, 1.0)
                starts.append(pert)
        else:
            # for other n, interpolate from initial if possible, else random
            if init_available and init_h is not None:
                x_old = np.linspace(0, 2, len(init_h), endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                base = np.interp(x_new, x_old, init_h)
                # adjust to satisfy sum exactly
                base = np.clip(base, 0.0, 1.0)
                diff = base.sum() - n / 2.0
                base -= diff / n
                base = np.clip(base, 0.0, 1.0)
                starts.append(base)
            else:
                # pure random feasible start
                rand = np.random.rand(n)
                rand = np.clip(rand, 0.0, 1.0)
                diff = rand.sum() - n / 2.0
                rand -= diff / n
                rand = np.clip(rand, 0.0, 1.0)
                starts.append(rand)

            # add random perturbations around the base/random
            for _ in range(9):
                pert = starts[0] + 0.05 * (np.random.rand(n) - 0.5)
                pert = np.clip(pert, 0.0, 1.0)
                diff = pert.sum() - n / 2.0
                pert -= diff / n
                pert = np.clip(pert, 0.0, 1.0)
                starts.append(pert)

        # run optimizations
        for s in starts:
            if time.time() > deadline:
                break
            h_opt, c5_val = _optimize_start(s, n, dx, deadline)
            if h_opt is None:
                continue

                best_c5 = c5_val

        best_h = h
        best_c5 = _c5_objective(h, dx)
        best_n = n

    return best_h, float(best_c5), int(best_n)
