import numpy as np, time, math, random

def evaluate_h(h):
    """Return C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def random_feasible(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    ones = n // 2
    arr = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=ones, replace=False)
    arr[idx] = 1.0
    return arr

def swap_move(state, rng):
    """Swap a 1 and a 0 in the state, preserving the sum."""
    n = state.size
    ones_idx = np.flatnonzero(state > 0.5)
    zeros_idx = np.flatnonzero(state < 0.5)
    if ones_idx.size == 0 or zeros_idx.size == 0:
        return state.copy()
    i = rng.choice(ones_idx)
    j = rng.choice(zeros_idx)
    new_state = state.copy()
    new_state[i] = 0.0
    new_state[j] = 1.0
    return new_state

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑C5 step function."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of points (prefer 140, fallback to 70)
    n_points = 140
    if n_points % 2 != 0:
        n_points += 1
    # use initial_h_values if present and compatible
    init = globals().get('initial_h_values', None)
    if init is not None and init.size == n_points:
        h_cur = init.astype(np.float64).copy()
    elif init is not None and init.size * 2 == n_points:
        # upscale by simple repetition
        h_cur = np.repeat(init, 2).astype(np.float64)
    else:
        h_cur = random_feasible(n_points, rng)

    best_h = h_cur.copy()
    best_score = evaluate_h(best_h)

    # annealing schedule parameters
    T0 = 0.5
    Tf = 1e-4
    total_iters = 200000  # upper bound; will stop earlier by time
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
            break
        if iteration >= total_iters:
            break

        # temperature decays linearly in log‑time
        progress = iteration / total_iters
        T = T0 * (1 - progress) + Tf * progress

        # propose a swap
        h_new = swap_move(h_cur, rng)
        new_score = evaluate_h(h_new)

        # accept if better or with probability exp(-(Δ)/T)
        delta = new_score - best_score
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
            h_cur = h_new
        else:
            if rng.random() < math.exp(-delta / max(T, 1e-12)):
                h_cur = h_new

        iteration += 1

    return best_h, float(best_score), int(n_points)
