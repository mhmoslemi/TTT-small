import numpy as np
import time
from scipy.optimize import minimize

def _c5_value(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _objective(x, dx):
    return _c5_value(x, dx)

def _make_start(n, base_h):
    """Interpolate base_h (of length len(base_h)) to length n and add noise."""
    m = len(base_h)
    if m == n:
        interp = base_h.copy()
    else:
        src_x = np.linspace(0, 2, m, endpoint=False)
        tgt_x = np.linspace(0, 2, n, endpoint=False)
        interp = np.interp(tgt_x, src_x, base_h)
    # ensure sum constraint before noise
    target_sum = n / 2.0
    interp *= target_sum / interp.sum()
    # add small random perturbation
    noise = np.random.uniform(-0.02, 0.02, size=n)
    pert = interp + noise
    pert = np.clip(pert, 0.0, 1.0)
    # re‑project to satisfy sum constraint
    pert *= target_sum / pert.sum()
    return pert

    best_h = None
    best_val = np.inf
    best_n = None

    # possible resolutions (prefer small)
    resolutions = [47, 64, 100]

        target_sum = n / 2.0

        # generate a pool of starting points
        starts = []
        if base_vec is not None:
            starts.append(_make_start(n, base_vec))
        # additional random starts
        for _ in range(8):
            rnd = np.random.rand(n)
            rnd *= target_sum / rnd.sum()
            starts.append(rnd)

        # optimization settings
        bounds = [(0.0, 1.0)] * n
        eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

        for x0 in starts:
            if time.time() - start_time > budget_s:
                break

            res = minimize(_objective, x0, args=(dx,),
                           method='SLSQP',
                           bounds=bounds,
                           constraints=[eq_cons],
                           options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})

            if not res.success:

            # enforce sum precisely (minor numerical drift)
            h_opt *= target_sum / h_opt.sum()
