import numpy as np
import time

def _c5_of_h(h, dx):
    # correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return conv.max()

def _project_move(h, i, j, delta):
    """Move delta from i to j, staying in [0,1]. Return new array or None."""
    hi, hj = h[i], h[j]
    if delta > 0:
        # moving from i to j
        max_delta = min(delta, hi, 1 - hj)
    else:
        # moving from j to i
        max_delta = max(delta, -hj, -(1 - hi))
    if max_delta == 0:
        return None
    h2 = h.copy()
    h2[i] -= max_delta
    h2[j] += max_delta
    return h2

    best_h = h.copy()
