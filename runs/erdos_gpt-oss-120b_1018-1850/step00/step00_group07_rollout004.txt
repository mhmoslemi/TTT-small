import numpy as np
import time

def _make_binary_initial(n, desired_sum):
    """Create a feasible binary start vector of length n with exactly desired_sum ones."""
    # start from provided initial_h_values if present, else random
    if 'initial_h_values' in globals():
        arr = np.array(initial_h_values, dtype=float)
        if arr.size != n:

            x_old = np.linspace(0, 2, arr.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            arr = np.interp(x_new, x_old, arr)
    else:
        arr = np.random.rand(n)
    # binarize by threshold 0.5
    bin_vec = (arr >= 0.5).astype(int)

    # adjust count to desired_sum by swapping bits
    cur = bin_vec.sum()
    while cur != desired_sum:
        # pick a 1 to turn off and a 0 to turn on
        if cur > desired_sum:
            ones = np.where(bin_vec == 1)[0]
            zeros = np.where(bin_vec == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            bin_vec[i] = 0
            bin_vec[j] = 1
            cur -= 1
        else:  # cur < desired_sum
            zeros = np.where(bin_vec == 0)[0]
            ones = np.where(bin_vec == 1)[0]
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            bin_vec[i] = 1
            bin_vec[j] = 0
            cur += 1
    return bin_vec

def _c5_score(h, dx):
    """Compute C5 for a binary vector h with spacing dx."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    current = _make_binary_initial(n_points, desired_sum)
    best = current.copy()
    best_score = _c5_score(best, dx)

    # SA parameters
    T0 = 1.0
    Tmin = 1e-6
    decay = 0.995
    T = T0

    # main loop respecting time budget
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # generate neighbor by swapping two bits (keeps sum)
        i, j = np.random.choice(n_points, 2, replace=False)
        neighbor = current.copy()
        neighbor[i], neighbor[j] = neighbor[j], neighbor[i]  # swap values
        # compute score
        neigh_score = _c5_score(neighbor, dx)

        # accept if better or by Metropolis
        if neigh_score < best_score:
            best = neighbor.copy()
            best_score = neigh_score
        if neigh_score < _c5_score(current, dx):
            current = neighbor
        else:
            prob = np.exp(-(neigh_score - _c5_score(current, dx)) / max(T, 1e-12))
            if np.random.rand() < prob:
                current = neighbor

        # temperature update
        T = max(T * decay, Tmin)

    # final evaluation
    final_score = _c5_score(best, dx)
    return (best.astype(float), float(final_score), n_points)
