import numpy as np
import time

def _adjust_to_sum(arr, target):
    """Adjust binary array to have exactly target ones by swapping bits."""
    a = arr.astype(np.int8).copy()
    cur = a.sum()
    n = len(a)
    # If already correct, return
    if cur == target:
        return a
    # Determine indices of ones and zeros
    ones = np.flatnonzero(a == 1)
    zeros = np.flatnonzero(a == 0)
    # While sum not correct, flip a one to zero and a zero to one
    while cur != target:
        if cur > target:
            # need to turn a 1 into 0 and a 0 into 1 (net -0)
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            a[i] = 0
            a[j] = 1
            # update lists
            ones = np.setdiff1d(ones, [i], assume_unique=True)
            zeros = np.setdiff1d(zeros, [j], assume_unique=True)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            # sum unchanged, but we changed composition; repeat until correct count
        else:  # cur < target
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            a[i] = 1
            a[j] = 0
            zeros = np.setdiff1d(zeros, [i], assume_unique=True)
            ones = np.setdiff1d(ones, [j], assume_unique=True)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
        cur = a.sum()
    return a

def _c5_score(h, dx):
    """Compute C5 for binary (or real) vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
