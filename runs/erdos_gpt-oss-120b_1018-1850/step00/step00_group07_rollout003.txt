import time, numpy as np, math, random

def _make_feasible(vec, desired):
    """Adjust binary vector to have exactly desired sum by flipping bits."""
    cur = int(vec.sum())
    if cur == desired:
        return vec
    # Change bits until sum matches
    while cur != desired:
        # pick indices of opposite values to flip
        if cur > desired:
            # need to turn some 1s to 0s
            ones = np.where(vec == 1)[0]
            i = random.choice(ones)
            vec[i] = 0
            cur -= 1
        else:
            zeros = np.where(vec == 0)[0]
            i = random.choice(zeros)
            vec[i] = 1
            cur += 1
    return vec

def _initial_binary(n):
    """Random binary vector with exactly n/2 ones."""
    vec = np.zeros(n, dtype=int)
    vec[:n//2] = 1
    np.random.shuffle(vec)
    return vec

def _evaluate_c5(h, dx):
    """Return max correlation scaled by dx."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx
