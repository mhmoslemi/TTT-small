import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def _evaluate_c5(h):
    """Compute max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)
