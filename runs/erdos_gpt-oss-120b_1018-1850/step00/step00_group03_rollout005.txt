import time
import numpy as np

def evaluate_h(h):
    """Return C5 for the discretized step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def random_binary_state(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=np.float64)
    arr[:half] = 1.0
    rng.shuffle(arr)
    return arr

def upscale_initial(initial, target_n):
    """Resize initial vector to target length by linear interpolation and then binarize preserving total mass."""
    src_n = initial.size
    x_src = np.linspace(0, 2, src_n, endpoint=False)
    x_tgt = np.linspace(0, 2, target_n, endpoint=False)
    interp = np.interp(x_tgt, x_src, initial)
    # project to binary with same sum = target_n/2
    # take largest entries to be 1
    thresh = np.partition(interp, -target_n // 2)[-target_n // 2]
    binary = (interp > thresh).astype(np.float64)
    # adjust possible off‑by‑one due to ties
    while binary.sum() > target_n // 2:
        idx = np.where(binary == 1)[0][0]
        binary[idx] = 0
    while binary.sum() < target_n // 2:
        idx = np.where(binary == 0)[0][0]
        binary[idx] = 1
    return binary

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    h_values : numpy array of shape (n_points,)
    c5_bound : float, the smallest C5 found
    n_points : int, number of discretization points used
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Decide on discretization size
    DEFAULT_N = 140
    if 'initial_h_values' in globals():
        init = globals()['initial_h_values']
        n = init.size
        if n != DEFAULT_N:
            # upscale/downscale to DEFAULT_N for a fair search
            h_cur = upscale_initial(init, DEFAULT_N)
            n = DEFAULT_N
        else:
            h_cur = init.astype(np.float64).copy()
    else:
        n = DEFAULT_N
        h_cur = random_binary_state(n, rng)

    # Ensure feasibility
    half = n // 2
    if int(h_cur.sum()) != half:
        # project to feasible binary state
        h_cur = random_binary_state(n, rng)

    best_h = h_cur.copy()
    best_score = evaluate_h(best_h)

    # Simulated annealing parameters
    max_iter = 200000
    temp_start = 0.5
    temp_end = 1e-4

    for it in range(max_iter):
        # time stop check
        if time.time() - start_time > budget_s * 0.95:
            break

        # temperature schedule (linear in log‑time)
        progress = it / max_iter
        temp = temp_start * (temp_end / temp_start) ** progress

        # propose a swap of a 0 and a 1
        zeros = np.where(h_cur == 0)[0]
        ones = np.where(h_cur == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)

        # apply swap
        h_cur[i], h_cur[j] = 1.0, 0.0
        new_score = evaluate_h(h_cur)

        # acceptance test
        delta = new_score - best_score
        if new_score < best_score:
            best_score = new_score
            best_h = h_cur.copy()

            prob = np.exp(-delta / temp) if temp > 0 else 0.0
            if rng.random() >= prob:
                # reject: revert swap
                h_cur[i], h_cur[j] = 0.0, 1.0

    return best_h, best_score, n
