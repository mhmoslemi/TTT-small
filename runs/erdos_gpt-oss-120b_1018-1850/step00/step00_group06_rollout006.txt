import numpy as np
import time
from scipy.optimize import minimize

def _objective(h, dx):
    """Return the maximal overlap integral for a given h."""
    # correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _make_feasible(h, total):
    """Project h onto the box [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        return np.full_like(h, total / h.size)
    # simple affine scaling to hit the required sum while staying in bounds
    # if scaling would violate bounds, fall back to iterative clipping
    scale = total / cur
    h = h * scale
    # clip again and correct the residual iteratively (few iterations suffice)
    for _ in range(5):
        excess = h.sum() - total
        if abs(excess) < 1e-12:
            break
        # move excess proportionally from interior points
        mask = (h > 0.0) & (h < 1.0)
        if not np.any(mask):
            # all at bounds, distribute uniformly
            h = np.clip(h - excess / h.size, 0.0, 1.0)
            break
        adjust = excess / mask.sum()
        h[mask] -= adjust
        h = np.clip(h, 0.0, 1.0)
    return h
