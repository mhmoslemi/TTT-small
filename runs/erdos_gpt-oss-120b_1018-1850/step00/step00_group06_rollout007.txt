import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 value for step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def make_initial(n):
    """Create a feasible starting vector of length n."""

        parent = initial_h_values
        if len(parent) == n:

            # enforce bounds and integral
            h = np.clip(h, 0.0, 1.0)
            s = h.sum()
            target = n / 2.0
            if s != 0:
                h *= target / s
            return h
    except NameError:
        pass
    # fallback: two blocks of 1's and 0's giving integral 1
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    # smooth a little to avoid strict 0/1 which can stall COBYLA
    eps = 1e-6
    h[:half] = np.clip(h[:half] - eps, 0.0, 1.0)
    h[half:] = np.clip(h[half:] + eps, 0.0, 1.0)
    # adjust sum exactly
    cur = h.sum()
    target = n / 2.0
    if cur != 0:
        h *= target / cur
    return h

    h0 = make_initial(n_points)

    # constraints for COBYLA
    cons = []

    # sum == n/2  -> two inequalities
    target_sum = n_points / 2.0
    cons.append(lambda h: np.sum(h) - target_sum)          # >=0
    cons.append(lambda h: target_sum - np.sum(h))          # >=0

    # bounds 0<=h<=1
    cons += [lambda h, i=i: h[i] for i in range(n_points)]          # h[i] >=0
    cons += [lambda h, i=i: 1.0 - h[i] for i in range(n_points)]   # h[i] <=1

    def callback(xk):

            best_c5 = val
            best_h = xk.copy()

    def obj(h):
        return compute_c5(h)
