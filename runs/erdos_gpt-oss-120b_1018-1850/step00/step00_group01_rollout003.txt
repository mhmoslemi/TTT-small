import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    """Return the C5 value for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_sum(h, target):
    """Scale h to meet sum constraint while keeping within [0,1]."""
    cur = h.sum()
    if cur == target:
        return h
    # simple linear scaling and clipping, then adjust residual by adding/subtracting tiny amount
    h = np.clip(h * (target / cur), 0.0, 1.0)
    diff = target - h.sum()
    # distribute residual uniformly over non‑boundary entries
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    if free.size > 0:
        h[free] += diff / free.size
        h = np.clip(h, 0.0, 1.0)
    return h

def generate_starts(base, n, rng, count=5):
    """Create a list of starting vectors of length n."""
    starts = []
    # interpolate base to length n
    x_old = np.linspace(0, 2, base.size, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    interp = np.interp(x_new, x_old, base)
    interp = np.clip(interp, 0.0, 1.0)
    interp = project_sum(interp, n / 2)
    starts.append(interp)

    for _ in range(count - 1):
        pert = interp + rng.normal(scale=0.05, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = project_sum(pert, n / 2)
        starts.append(pert)
    return starts

def run(seed=42, budget_s=1100, **kwargs):

        init = project_sum(init, n0 / 2)

    candidate_ns = [init.size, 64, 100]
    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break
        starts = generate_starts(init, n, rng, count=6)
        for x0 in starts:
            if time.time() - start_time > budget_s * 0.95:

            # SLSQP with box bounds and equality sum constraint
            cons = {"type": "eq", "fun": lambda h, target=n/2: h.sum() - target}
            bounds = [(0.0, 1.0) for _ in range(n)]

            res = minimize(
