import time, numpy as np
from scipy.optimize import minimize

def _c5_objective(h, dx):
    # maximum overlap integral
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _constraint_sum(h, target):
    return np.sum(h) - target

def _optimize_start(start, n, dx, time_limit, start_time):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {'type': 'eq', 'fun': _constraint_sum, 'args': (target_sum,)}
    opts = {'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    res = minimize(_c5_objective, start, args=(dx,), method='SLSQP',
                   bounds=bounds, constraints=[cons], options=opts)
    if not res.success:
        return None, None
    h_opt = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-12:
        h_opt += diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5 = _c5_objective(h_opt, dx)
    return h_opt, c5

def _generate_starts(base_h, n, num=10, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    starts = []
    # interpolate base to target size
    x_base = np.linspace(0, 2, len(base_h), endpoint=False)
    x_target = np.linspace(0, 2, n, endpoint=False)
    interp = np.interp(x_target, x_base, base_h)
    starts.append(interp.copy())
    for _ in range(num - 1):
        pert = interp + rng.normal(scale=0.05, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        # renormalize to exact sum
        pert *= (n / 2.0) / pert.sum()
        starts.append(pert)
    return starts
