import time
import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated via correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def repair(h, target_sum):
    """Clip to [0,1] and renormalize to exact sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # fallback to uniform feasible vector
        h[:] = target_sum / h.size
    else:
        h *= target_sum / s
    # after scaling may exceed bounds slightly; clip again and renorm once more
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s != 0:
        h *= target_sum / s
    return h

def random_feasible(n, target_sum, rng):
    """Generate a random feasible vector."""
    # start with Dirichlet then scale
    x = rng.random(n)
    x = np.clip(x, 0.0, 1.0)
    x = repair(x, target_sum)
    return x

def local_refine(x0, target_sum, budget, start_time):
    """Run a short L-BFGS-B refinement respecting bounds and equality via penalty."""
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        # penalty for sum deviation
        penalty = 1e3 * (x.sum() - target_sum) ** 2
        return evaluate_erdos(x) + penalty
    opts = {"maxfun": 2000, "ftol": 1e-9}
    res = minimize(obj, x0, method="L-BFGS-B", bounds=bounds, options=opts)
    x = repair(res.x, target_sum)
    return x, evaluate_erdos(x)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        # Ensure feasibility of given start
        h0 = repair(h0, target_sum)
    else:
        n = 84  # moderate size <1k
        target_sum = n / 2.0
        h0 = np.full(n, 0.5)  # uniform satisfies sum
    best_h = h0.copy()
    best_c = evaluate_erdos(best_h)

    # Simple stochastic search with occasional local refinement
    population = [best_h]
    pop_size = 12
    while len(population) < pop_size:
        population.append(random_feasible(n, target_sum, rng))

    while time.time() - start < budget_s * 0.95:
        # select two parents
        i, j = rng.choice(len(population), size=2, replace=False)
        p1, p2 = population[i], population[j]
        # crossover (averaging + noise)
        alpha = rng.random()
        child = alpha * p1 + (1 - alpha) * p2
        child += 0.02 * rng.standard_normal(n)  # small mutation
        child = repair(child, target_sum)
        c_val = evaluate_erdos(child)

        # keep best in population
        worst_idx = np.argmax([evaluate_erdos(ind) for ind in population])
        if c_val < evaluate_erdos(population[worst_idx]):
            population[worst_idx] = child

        # update global best
        if c_val < best_c:
            best_c, best_h = c_val, child.copy()

                # replace worst with refined

                population[worst_idx] = refined
