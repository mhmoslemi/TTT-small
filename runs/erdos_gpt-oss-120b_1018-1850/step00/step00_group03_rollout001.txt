import numpy as np
import time
import math

def _evaluate(h, dx):
    # correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _feasible(h, target_sum):
    # clamp to [0,1] and adjust to meet sum exactly (integer counts)
    h = np.clip(h, 0.0, 1.0)
    # round to nearest 0/1 while preserving total count
    # compute current integer counts
    int_h = np.rint(h).astype(int)
    diff = int_h.sum() - target_sum
    if diff != 0:
        # adjust smallest/largest fractional parts
        frac = h - np.floor(h)
        idx = np.argsort(frac)  # ascending
        if diff > 0:
            # need to turn some 1s into 0s
            to_zero = idx[:diff]
            int_h[to_zero] = 0
        else:
            # need to turn some 0s into 1s
            to_one = idx[diff:]
            int_h[to_one] = 1
    return int_h.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine discretisation size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # ensure correct sum; if not, project
        target_sum = n // 2
        if not np.isclose(h0.sum(), target_sum):
            h0 = _feasible(h0, target_sum)
    else:
        n = 140
        target_sum = n // 2
        # start with half ones, half zeros randomly permuted
        h0 = np.zeros(n, dtype=float)
        h0[:target_sum] = 1.0
        np.random.shuffle(h0)

    dx = 2.0 / n
    best_h = h0.copy()
    best_score = _evaluate(best_h, dx)

    # Simulated annealing parameters
    max_iter = 2000000  # large upper bound, will be cut by time
    temp_start = 0.5
    temp_end = 1e-4

    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if iter_count >= max_iter:
            break

        # linear temperature schedule based on elapsed fraction
        frac = elapsed / budget_s
        temp = temp_start * (1 - frac) + temp_end * frac

        # propose a swap of two positions with different values
        i, j = np.random.choice(n, 2, replace=False)
        if best_h[i] == best_h[j]:
            continue  # no effect

        # create candidate by swapping
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_score = _evaluate(cand, dx)

        # acceptance test
        delta = cand_score - best_score
        if delta < 0 or np.random.rand() < math.exp(-delta / max(temp, 1e-12)):
            best_h = cand
            best_score = cand_score

        iter_count += 1

    # final feasibility check (ensures exact sum)
    best_h = _feasible(best_h, n // 2)

    return best_h, best_score, n
