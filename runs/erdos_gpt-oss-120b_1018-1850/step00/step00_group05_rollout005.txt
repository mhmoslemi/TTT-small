import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return corr.max()

def project_solution(h, target_sum):
    """Project onto box [0,1] and adjust sum to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return np.full_like(h, target_sum / h.size)
    # scale to achieve the required sum while staying in bounds
    diff = target_sum - s
    # distribute diff proportionally to distances from bounds
    free = ((h > 0) & (h < 1)).astype(float)
    if free.sum() == 0:
        # all at bounds, just shift uniformly and reclip
        h += diff / h.size
        return np.clip(h, 0.0, 1.0)
    adjust = diff / free.sum()
    h += free * adjust
    # ensure bounds again
    h = np.clip(h, 0.0, 1.0)
    # final correction if slight error remains
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = np.random.choice(np.where((h > 0) & (h < 1))[0])
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def optimize_h(initial_h, budget_s):
    """Run a series of SLSQP optimizations within the time budget."""
    n = initial_h.size
    target_sum = n / 2.0
    best_h = project_solution(initial_h.copy(), target_sum)
    best_val = compute_c5(best_h)
    start_time = time.time()
    # SLSQP options
    opts = {"maxiter": 200, "ftol": 1e-9, "disp": False}
    cons = {"type": "eq", "fun": lambda x: x.sum() - target_sum}
    bounds = [(0.0, 1.0)] * n

    while time.time() - start_time < budget_s * 0.9:
        # random perturbation to escape local minima
        pert = np.random.randn(n) * 0.05
        x0 = project_solution(best_h + pert, target_sum)

        def obj(x):
            return compute_c5(x)

        res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                       constraints=cons, options=opts)
        if not res.success:
            continue
        cand = project_solution(res.x, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val, best_h = val, cand
        # occasional reset to initial to diversify
        if np.random.rand() < 0.1:
            best_h = project_solution(initial_h.copy(), target_sum)

    return best_h, best_val
