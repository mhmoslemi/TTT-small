import time
import numpy as np

def _evaluate(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # determine problem size
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
        # ensure feasibility: project to binary with correct sum if needed
        target_sum = n // 2
        cur_sum = int(round(np.sum(h)))
        if cur_sum != target_sum:
            # simple fix: threshold to get required number of ones
            idx = np.argsort(-h)  # descending
            h[:] = 0.0
            h[idx[:target_sum]] = 1.0
    else:
        # fallback size
        n = 100  # must be even
        target_sum = n // 2
        h = np.zeros(n, dtype=np.float64)
        ones_idx = rng.choice(n, target_sum, replace=False)
        h[ones_idx] = 1.0

    dx = 2.0 / n
    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # annealing parameters
    start_temp = 0.1
    min_temp = 1e-4

    # pre‑compute indices split by value for quick swaps
    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s:
            break

        # linear cooling
        t = max(min_temp, start_temp * (1.0 - elapsed / budget_s))

        # propose a swap between a 0 and a 1
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)

        # perform swap
        h[i], h[j] = 1.0, 0.0
        score = _evaluate(h, dx)
        delta = score - best_score

        if delta < 0 or rng.random() < np.exp(-delta / t):
            # accept move
            if score < best_score:
                best_score = score
                best_h = h.copy()
                # update index lists
                zeros = np.flatnonzero(h == 0)
                ones = np.flatnonzero(h == 1)
            else:
                # keep updated lists after accepted worse move
                zeros = np.flatnonzero(h == 0)
                ones = np.flatnonzero(h == 1)
        else:
            # reject: revert swap
            h[i], h[j] = 0.0, 1.0

    return best_h, float(best_score), int(n)
