import time
import numpy as np
from scipy.optimize import minimize

def _c5_value(h, dx):
    """Maximum overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_start_vectors(base_h, n, rng):
    """Generate a few perturbed starts of length n from base_h."""
    # interpolate base_h to length n
    x_old = np.linspace(0.0, 2.0, len(base_h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(x_new, x_old, base_h)
    starts = [interp.copy()]
    for _ in range(9):
        pert = interp + rng.normal(scale=0.05, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        starts.append(pert)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 constant.

    Returns:
        h_values (np.ndarray): optimized step values on [0,2].
        c5_bound (float): corresponding C5 upper bound.
        n_points (int): number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine available base vector
    try:
        base_h = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: simple rectangular function (half ones, half zeros)
        n0 = 50
        base_h = np.concatenate([np.ones(n0 // 2), np.zeros(n0 // 2)])
    base_n = len(base_h)

    # Candidate resolutions
    resolutions = [base_n, 64, 100]
    # Ensure unique and reasonable
    resolutions = [n for n in sorted(set(resolutions)) if n <= 200]

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Equality constraint: integral = 1  => sum(h) = n/2
    def make_constraints(n):
        return ({
            'type': 'eq',
            'fun': lambda h: h.sum() - n / 2.0
        },)

    # Objective wrapper for optimizer (minimize max overlap)
    def make_objective(dx):
        return lambda h: _c5_value(h, dx)

    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        starts = _make_start_vectors(base_h, n, rng)

        bounds = [(0.0, 1.0)] * n
        constraints = make_constraints(n)

        for init in starts:
            if time.time() - start_time > budget_s * 0.95:
                break
            # SLSQP may struggle if initial point violates the equality;
            # project the sum by adding a constant then clipping.
            init = np.clip(init, 0.0, 1.0)
            # Adjust mean to satisfy sum constraint (ignoring bounds temporarily)
            delta = (n / 2.0 - init.sum()) / n
            init = np.clip(init + delta, 0.0, 1.0)

            res = minimize(
                make_objective(dx),
                init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            )
            if not res.success:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)
            c5_val = _c5_value(h_opt, dx)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                best_n = n

    # If nothing succeeded, fall back to interpolated base
    if best_h is None:
        best_h = np.interp(np.linspace(0, 2, base_n, endpoint=False),
                           np.linspace(0, 2, base_n, endpoint=False),
                           base_h)
        best_c5 = _c5_value(best_h, 2.0 / base_n)
        best_n = base_n

    return best_h, best_c5, int(best_n)
