import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint, Bounds

def _c5_of_h(h, dx):
    # compute max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation
    # mode='full' gives length 2n-1, shift indices correspond to k = -(n-1)...(n-1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _make_initial(n):
    # simple symmetric start: half ones, half zeros scaled to satisfy integral
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    # adjust to exact sum = n/2
    current = h.sum()
    target = n / 2.0
    if current != target:
        h[:half] *= target / current
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 50
        h0 = _make_initial(n)

    dx = 2.0 / n
    target_sum = n / 2.0

    # constraints: sum(h)=target_sum, 0<=h<=1
    linear_cons = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])
    bounds = Bounds(0.0, 1.0)

    best_h = None
    best_c5 = np.inf

    # evaluate initial
    c5 = _c5_of_h(h0, dx)
    if c5 < best_c5:
        best_c5, best_h = c5, h0.copy()

    # local refinement helper
    def _local_opt(x0):
        def obj(x):
            return _c5_of_h(x, dx)
        res = minimize(obj, x0, method='SLSQP',
                       bounds=bounds,
                       constraints=[linear_cons],
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
        return res.x, res.fun

    # Main loop: random restarts with occasional local refinement
    while time.time() - start_time < budget_s * 0.95:
        # random feasible point: sample uniform then project to sum constraint
        x = rng.random(n)
        x = np.clip(x, 0.0, 1.0)
        # scale to meet sum constraint while staying in [0,1]
        excess = x.sum() - target_sum
        if abs(excess) > 1e-12:
            # simple redistribution: subtract excess uniformly then clip
            delta = excess / n
            x -= delta
            x = np.clip(x, 0.0, 1.0)
            # re‑adjust final sum precisely
            cur = x.sum()
            if cur != target_sum:
                diff = target_sum - cur
                # add diff to a random index that can accept it
                idx = rng.choice(n)
                x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        # local refinement
        x_opt, c5_opt = _local_opt(x)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, x_opt.copy()
        # occasional perturbation of best solution
        if rng.random() < 0.1:
            pert = rng.normal(scale=0.05, size=n)
            y = best_h + pert
            y = np.clip(y, 0.0, 1.0)
            # renormalize to sum constraint
            diff = y.sum() - target_sum
            if abs(diff) > 1e-12:
                y -= diff / n
                y = np.clip(y, 0.0, 1.0)
                cur = y.sum()
                if cur != target_sum:
                    adjust = target_sum - cur
                    idx = rng.choice(n)
                    y[idx] = np.clip(y[idx] + adjust, 0.0, 1.0)
            y_opt, c5_y = _local_opt(y)
            if c5_y < best_c5:
                best_c5, best_h = c5_y, y_opt.copy()
        # break if very good
        if best_c5 < 0.3808:
            break

    # final ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h *= target_sum / best_h.sum() if best_h.sum() != 0 else 0.0

    return best_h, float(best_c5), n
