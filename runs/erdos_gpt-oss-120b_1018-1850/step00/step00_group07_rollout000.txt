import numpy as np
import time
import math
import random

def _adjust_to_target_sum(vec, target):
    """Flip random bits until sum(vec)==target. Works for binary vectors."""
    cur = int(vec.sum())
    n = len(vec)
    while cur != target:
        # pick a 1 and a 0 to swap
        ones = np.where(vec == 1)[0]
        zeros = np.where(vec == 0)[0]
        if cur > target:
            # need fewer ones: turn a 1 to 0 and a 0 to 1
            i = random.choice(ones)
            j = random.choice(zeros)
        else:
            # need more ones
            i = random.choice(zeros)
            j = random.choice(ones)
        vec[i] = 1 - vec[i]
        vec[j] = 1 - vec[j]
        cur = int(vec.sum())
    return vec

def _c5_score(h, dx):
    """Maximum overlap integral for binary (or real) h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx
