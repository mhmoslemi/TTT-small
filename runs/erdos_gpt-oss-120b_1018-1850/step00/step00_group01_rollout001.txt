import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def make_starting_vectors(base_h, target_n, rng):
    """Interpolate base_h to target_n and create perturbed copies."""
    # linear interpolation
    x_src = np.linspace(0, 2, base_h.size, endpoint=False)
    x_tgt = np.linspace(0, 2, target_n, endpoint=False)
    interp = np.interp(x_tgt, x_src, base_h)
    # enforce integral constraint
    interp *= (target_n / 2) / interp.sum()
    starts = [interp.copy()]
    for _ in range(9):
        pert = interp + rng.normal(scale=0.05, size=target_n)
        pert = np.clip(pert, 0.0, 1.0)
        pert *= (target_n / 2) / pert.sum()
        starts.append(pert)
    return starts
