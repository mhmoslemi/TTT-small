import numpy as np
import time

def evaluate_erdos(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    # correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # dx factor
    return conv.max() * (2.0 / h.size)

def repair_h(h):
    """Project h to [0,1] and renormalize to satisfy sum = n/2."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = n / 2.0
    if total == 0:
        # fallback to uniform
        h[:] = 0.5
    else:
        h *= target / total
    # clipping may have broken sum again; one more fix
    h = np.clip(h, 0.0, 1.0)
    excess = h.sum() - target
    if abs(excess) > 1e-12:
        # distribute excess uniformly respecting bounds
        mask = (h > 0) & (h < 1)
        if mask.any():
            adjust = excess / mask.sum()
            h[mask] -= adjust
            h = np.clip(h, 0.0, 1.0)
    return h

def random_neighbor(h, scale=0.1, rng=None):
    """Generate a neighboring feasible h by adding gaussian noise and repairing."""
    if rng is None:
        rng = np.random
    n = h.size
    delta = rng.normal(0, scale, size=n)
    h_new = h + delta
    return repair_h(h_new)
