def _project_to_simplex(h, target_sum):
    """Project h onto box [0,1] and adjust to have given sum."""
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Iteratively adjust to meet sum while staying in bounds
    for _ in range(10):
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) < 1e-12:
            break
        # Scale uniformly to hit the target sum
        scale = target_sum / cur_sum if cur_sum != 0 else 0.0
        h *= scale
        # Re‑clip after scaling
        h = np.clip(h, 0.0, 1.0)
    # Final correction of any residual error by distributing uniformly
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h += residual / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h
