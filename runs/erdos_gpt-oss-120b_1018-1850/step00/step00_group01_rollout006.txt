import numpy as np
import time
from scipy.optimize import minimize

def max_overlap(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def project_bounds(x):
    """Clip to [0,1] for feasibility of start points."""
    return np.clip(x, 0.0, 1.0)

def random_start(base, scale=0.2, rng=None):
    """Create a perturbed start from base vector."""
    if rng is None:
        rng = np.random.default_rng()
    pert = base + rng.normal(scale=scale, size=base.shape)
    return project_bounds(pert)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_val = np.inf
    best_n = None

    # candidate resolutions
    n_list = [47, 64, 100]

    for n in n_list:
        # prepare base vector for this resolution
        if n == 47 and 'initial_h_values' in globals():
            base = np.array(initial_h_values, dtype=float)
        else:
            # interpolate from the 47-point initial if available, else uniform
            if 'initial_h_values' in globals():
                src = np.array(initial_h_values, dtype=float)
                src_x = np.linspace(0.0, 2.0, src.size, endpoint=False)
                tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
                base = np.interp(tgt_x, src_x, src)
            else:
                base = np.full(n, 0.5)  # simple guess (integral 0.5*n*dx =1)

        # ensure base respects bounds
        base = project_bounds(base)

        # generate several starts
        starts = [base]
        for _ in range(6):
            starts.append(random_start(base, scale=0.15, rng=rng))

        # optimization for each start
        for x0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            # equality constraint: sum(h) = n/2  (since dx = 2/n)
            cons = ({
                'type': 'eq',
                'fun': lambda x, nn=n: np.sum(x) - nn / 2.0
            },)

            bounds = [(0.0, 1.0)] * n

            res = minimize(
                fun=lambda x: max_overlap(x),
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            )

            if not res.success:
                continue

            val = max_overlap(res.x)
            if val < best_val:
                best_val = val
                best_h = res.x.copy()
                best_n = n

        if time.time() - start_time > budget_s * 0.95:
            break

    # fallback if nothing succeeded
    if best_h is None:
        # simple uniform function satisfying integral
        n = 47
        h = np.full(n, 0.5)
        best_h = h
        best_val = max_overlap(h)
        best_n = n

    # final sanity clip
    best_h = project_bounds(best_h)
    # enforce exact integral by scaling a tiny amount if needed
    current_sum = best_h.sum()
    target_sum = best_n / 2.0
    if abs(current_sum - target_sum) > 1e-8:
        # distribute difference uniformly within bounds
        diff = target_sum - current_sum
        best_h += diff / best_n
        best_h = project_bounds(best_h)

    return (best_h, float(best_val), int(best_n))
