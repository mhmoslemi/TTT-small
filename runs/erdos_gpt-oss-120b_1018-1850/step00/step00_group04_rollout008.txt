def _evaluate(h, dx):
    """Return C5 for a given h vector."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project(feas):
    """Project a vector onto the feasible set: 0<=x<=1 and sum(x)=target."""
    n = feas.size
    target = n / 2.0
    # clip to [0,1]
    x = np.clip(feas, 0.0, 1.0)
    # Adjust sum by simple scaling and clipping loop
    for _ in range(5):
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
        # scale towards target
        if s == 0:
            x[:] = target / n
        else:
            scale = target / s
            x *= scale
        # re‑clip after scaling
        x = np.clip(x, 0.0, 1.0)
    # final renormalization if still off by tiny amount
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly respecting bounds
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            inc = diff / mask.sum()
            x[mask] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    # Simple stochastic hill‑climbing / simulated annealing
    # Parameters tuned for the time budget
    max_iter = 2000000
    temp0 = 0.1
    temp_end = 0.001
    for it in range(max_iter):
        # time check
        if time.time() - start > budget_s * 0.97:
            break
        # linear cooling
        t = temp0 + (temp_end - temp0) * (it / max_iter)

        # propose a small random move on a random subset
        cand = best_h.copy()
        # pick 2‑3 indices to modify

        # accept if better or by Metropolis criterion
        if val < best_val or rng.random() < math.exp((best_val - val) / max(t, 1e-12)):

    # final polishing with a short L‑BFGS‑like line search on the best point
