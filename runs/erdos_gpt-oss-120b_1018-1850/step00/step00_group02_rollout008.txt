import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.size
    # convolution via FFT (full length = 2n-1)
    m = 1 << (2 * n - 1).bit_length()
    fh = np.fft.rfft(h, m)
    f1mh = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(fh * f1mh, m)[: 2 * n - 1]
    return conv.max() * (2.0 / n)  # dx = 2/n

def project_h(h):
    """Project h onto feasible set: 0<=h<=1 and sum(h)=n/2."""
    n = h.size
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # enforce sum using simple scaling and redistribution
    target = n / 2.0
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # shift uniformly then clip again
    shift = (target - cur) / n
    h += shift
    h = np.clip(h, 0.0, 1.0)
    # final correction of any residual using water-filling
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        # distribute deficit on entries not at bounds
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # fallback: adjust first element
            h[0] += deficit
        else:
            add = deficit / free.size
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    return h

def random_binary_start(n, rng):
    """Return binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def hill_climb(initial_h, budget_s, rng):
    """Simple hill‑climbing with pairwise swaps."""
    n = initial_h.size
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s * 0.95:
        # propose a tiny move: pick two indices i,j with different values
        i, j = rng.choice(n, 2, replace=False)
        if best_h[i] == best_h[j]:
            continue
        # swap their values
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand = project_h(cand)
        val = compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val
    return best_h, best_val
