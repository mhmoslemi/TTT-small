import time
import numpy as np
import random

def _repair(h, target_sum, tol=1e-9, max_iter=10):
    """Project h onto [0,1] and enforce sum == target_sum."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # distribute diff equally among non‑border elements
        free = (h > 0.0) & (h < 1.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # all at bounds, just spread over all and re‑clip
            delta = diff / n
            h += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            delta = diff / free_idx.size
            h[free_idx] += delta
            h = np.clip(h, 0.0, 1.0)
    # final tiny correction by adjusting a single element within bounds
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > tol:
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def _score(h, dx):
    """Return the C5 value for candidate h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _initialize_population(pop_size, n, target_sum, rng):
    """Create initial population respecting constraints."""
    pop = []
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        if base.size == n:
            base = _repair(base, target_sum)
            pop.append(base)
    while len(pop) < pop_size:
        # random Dirichlet then clip to [0,1]
        vec = rng.random(n)
        vec = _repair(vec, target_sum)
        pop.append(vec)
    return np.stack(pop)

def run(seed=42, budget_s=1100, **kwargs):
    """Genetic‑algorithm search for a low C₅ step function.

    Returns
    -------
    h_values : np.ndarray
        Optimized values on the uniform grid.
    c5_bound : float
        Corresponding maximal overlap integral.
    n_points : int
        Number of discretisation points (grid size).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # grid size – keep modest for speed, but >=45 for compatibility
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫h =1 ⇒ sum(h)*dx =1 ⇒ sum = n/2

    pop_size = 60
    population = _initialize_population(pop_size, n_points, target_sum, rng)
    scores = np.array([_score(ind, dx) for ind in population])

    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # GA parameters
    tournament_k = 5
    crossover_prob = 0.5
    mutation_rate = 0.2
    mutation_scale = 0.15  # std of Gaussian perturbation

    while time.time() - start < budget_s:
        # selection
        parents_idx = []
        for _ in range(pop_size):
            cand = rng.choice(pop_size, tournament_k, replace=False)
            chosen = cand[np.argmin(scores[cand])]
            parents_idx.append(chosen)
        parents = population[parents_idx]

        # crossover
        offspring = np.empty_like(parents)
        for i in range(0, pop_size, 2):
            a, b = parents[i], parents[i + 1 if i + 1 < pop_size else 0]
            mask = rng.random(n_points) < crossover_prob
            child1 = np.where(mask, a, b)
            child2 = np.where(mask, b, a)
            offspring[i] = child1
            if i + 1 < pop_size:
                offspring[i + 1] = child2

        # mutation
        mutate_mask = rng.random(offspring.shape) < mutation_rate
        noise = rng.normal(0.0, mutation_scale, offspring.shape)
        offspring = offspring + mutate_mask * noise

        # repair to feasibility
        for i in range(pop_size):
            offspring[i] = _repair(offspring[i], target_sum)

        # evaluate
        offspring_scores = np.array([_score(ind, dx) for ind in offspring])

        # elitist replacement: keep the best of parents+offspring
        combined = np.vstack((population, offspring))
        combined_scores = np.concatenate((scores, offspring_scores))
        best_indices = np.argpartition(combined_scores, pop_size)[:pop_size]
        population = combined[best_indices]
        scores = combined_scores[best_indices]

        # track global best
        cur_best_idx = np.argmin(scores)
        cur_best_score = scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_h = population[cur_best_idx].copy()

        # early stop if we reach target bound
        if best_score <= 0.3808:
            break

    return best_h, float(best_score), n_points
