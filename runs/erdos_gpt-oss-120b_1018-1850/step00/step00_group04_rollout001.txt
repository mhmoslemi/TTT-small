import numpy as np
import time

def _evaluate(h):
    # pre-imported helper, but fallback to manual if missing
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        # manual computation: full correlation of h and (1-h)
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / h.size
        return np.max(corr) * dx

def _project(h, target_sum):
    # Clip to [0,1] and then adjust to meet sum constraint by scaling the interior
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if cur_sum == 0:
        # distribute uniformly
        h[:] = target_sum / h.size
        return h
    # Scale deviations from 0.5 to meet target sum while staying in bounds
    delta = target_sum - cur_sum
    # distribute delta proportionally to distance from bounds
    free = np.where((h > 0) & (h < 1))[0]
    if free.size == 0:
        # fallback: uniform distribution
        h[:] = target_sum / h.size
        return h
    weights = np.ones_like(h)
    weights[h == 0] = 0
    weights[h == 1] = 0
    total_w = weights.sum()
    if total_w == 0:
        h[:] = target_sum / h.size
        return h
    adjust = delta * weights / total_w
    h += adjust
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # small residual correction
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # add residual to a random free index
        idx = np.random.choice(free)
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # obtain initial vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        h = np.full(n_points, 0.5)
    n = h.size
    target_sum = n / 2.0  # because integral =1, dx=2/n => sum*dx=1 => sum=n/2
    h = _project(h, target_sum)

    best_h = h.copy()
    best_c = _evaluate(best_h)

    # parameters
    max_iter = 200000
    perturb_frac = 0.2  # fraction of entries to perturb each step
    sigma = 0.1  # magnitude of Gaussian noise

    it = 0
    while time.time() - start_time < budget_s and it < max_iter:
        it += 1
        # generate candidate
        cand = best_h.copy()
        k = max(1, int(perturb_frac * n))
        idx = np.random.choice(n, k, replace=False)
        noise = np.random.randn(k) * sigma
        cand[idx] += noise
        cand = _project(cand, target_sum)

        cand_c = _evaluate(cand)
        # accept if better
        if cand_c < best_c:
            best_h, best_c = cand, cand_c

        # occasional re‑anneal: reduce sigma over time
        if it % 1000 == 0:
            sigma *= 0.99
            if sigma < 1e-4:
                sigma = 0.1

    return best_h, float(best_c), n
