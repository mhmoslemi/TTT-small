import numpy as np
import time
import random

def _evaluate(h, dx):
    """Return max overlap C5 for a candidate h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(candidate, target_sum):
    """Clamp to [0,1] and adjust to meet exact sum target."""
    # clip
    np.clip(candidate, 0.0, 1.0, out=candidate)
    # current sum
    cur = candidate.sum()
    if abs(cur - target_sum) < 1e-12:
        return candidate
    # distribute the difference uniformly while staying in bounds
    diff = target_sum - cur
    # simple proportional scaling may break bounds, so use iterative redistribution
    # add diff equally
    inc = diff / candidate.size
    candidate += inc
    # fix any out‑of‑bounds after addition
    over = candidate > 1.0
    under = candidate < 0.0
    if over.any() or under.any():
        # bring back to bounds and recompute remaining diff
        candidate[over] = 1.0
        candidate[under] = 0.0
        # recompute remaining diff
        remaining = target_sum - candidate.sum()
        # distribute remaining uniformly among free variables
        free = (~over) & (~under)
        if free.any():
            candidate[free] += remaining / free.sum()
        # final clip
        np.clip(candidate, 0.0, 1.0, out=candidate)
    return candidate

def _mutate(parent, mutation_rate, sigma, rng):
    """Gaussian mutation with given rate and standard deviation."""
    child = parent.copy()
    mask = rng.random(parent.shape) < mutation_rate
    if mask.any():
        perturb = rng.normal(0.0, sigma, size=parent.shape)
        child[mask] += perturb[mask]
    return child

def _crossover(a, b, rng):
    """Uniform crossover."""
    mask = rng.random(a.shape) < 0.5
    child = np.where(mask, a, b)
    return child

def run(seed=42, budget_s=1100, **kwargs):
    """
    Genetic search for a step function h on [0,2] with ∫h=1.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # grid size – use 45 if initial_h_values exists, else try 60 for richer space
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 60
        # simple deterministic start: half ones, half zeros, shuffled
        h0 = np.zeros(n, dtype=float)
        h0[:n//2] = 1.0
        rng.shuffle(h0)

    dx = 2.0 / n
    target_sum = n / 2.0  # because sum(h)*dx = 1

    # ensure feasible start
    h0 = _repair(h0, target_sum)

    pop_size = 30
    population = np.empty((pop_size, n), dtype=float)
    # seed first individual with the given start
    population[0] = h0
    # fill rest randomly
    for i in range(1, pop_size):
        cand = rng.random(n)
        cand = _repair(cand, target_sum)
        population[i] = cand

    # GA parameters
    tournament_k = 5
    mutation_rate = 0.15
    sigma = 0.1
    best_h = None
    best_score = np.inf

    generation = 0
    while time.time() - start_time < budget_s:
        generation += 1
        # evaluate all
        scores = np.empty(pop_size, dtype=float)
        for i, ind in enumerate(population):
            scores[i] = _evaluate(ind, dx)

        # keep best
        idx_best = int(np.argmin(scores))
        if scores[idx_best] < best_score:
            best_score = float(scores[idx_best])
            best_h = population[idx_best].copy()

        # early stop if already good enough
        if best_score <= 0.3808:
            break

        # create next generation
        new_pop = np.empty_like(population)
        # elitism: carry best forward
        new_pop[0] = best_h

        for i in range(1, pop_size):
            # tournament selection for two parents
            participants = rng.integers(0, pop_size, size=tournament_k)
            p1 = participants[np.argmin(scores[participants])]
            participants = rng.integers(0, pop_size, size=tournament_k)
            p2 = participants[np.argmin(scores[participants])]

            child = _crossover(population[p1], population[p2], rng)
            child = _mutate(child, mutation_rate, sigma, rng)
            child = _repair(child, target_sum)
            new_pop[i] = child

        population = new_pop

        # adapt mutation sigma slowly
        if generation % 50 == 0:
            sigma = max(0.02, sigma * 0.95)

    # final evaluation for returned best
    final_score = _evaluate(best_h, dx)
    return best_h.tolist(), final_score, n
