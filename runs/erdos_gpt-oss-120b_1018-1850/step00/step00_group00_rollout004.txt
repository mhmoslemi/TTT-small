import numpy as np, time, math, random
# evaluate_erdos_solution and initial_h_values are provided in the environment

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def repair_candidate(h, target_sum):
    """Clamp to [0,1] and adjust to satisfy sum == target_sum."""
    # clamp
    np.clip(h, 0.0, 1.0, out=h)
    # adjust sum preserving bounds
    current = h.sum()
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h
    # distribute diff proportionally to distance from bounds
    # compute how much each element can move without leaving [0,1]
    free_up = 1.0 - h
    free_down = h
    if diff > 0:
        capacity = free_up.sum()
        if capacity < 1e-12:
            return h
        scale = diff / capacity
        h += free_up * scale
    else:
        capacity = free_down.sum()
        if capacity < 1e-12:
            return h
        scale = diff / capacity
        h += free_down * scale
    # final clamp (numerical safety)
    np.clip(h, 0.0, 1.0, out=h)
    return h

def tournament_select(pop, scores, k):
    """Select index via tournament of size k."""
    participants = random.sample(range(len(pop)), k)
    best = min(participants, key=lambda i: scores[i])
    return best

def crossover(p1, p2):
    """Uniform crossover."""
    mask = np.random.rand(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(child, rate, sigma):
    """Gaussian mutation with per‑gene probability `rate`."""
    mutate_mask = np.random.rand(child.size) < rate
    noise = np.random.randn(child.size) * sigma
    child[mutate_mask] += noise[mutate_mask]
    return child

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # grid size – use 45 as baseline (fast enough) but allow slight increase if time permits
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫h =1 ⇒ sum(h)*dx =1 ⇒ sum = 1/dx = n/2

    # initialise population
    pop_size = 60
    population = []
    if 'initial_h_values' in globals():
        # use provided vector (may have different length); resample to our n_points

            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = repair_candidate(init, target_sum)
        population.append(init)
    # fill rest randomly respecting constraints
    while len(population) < pop_size:
        cand = np.random.rand(n_points)
        cand = repair_candidate(cand, target_sum)
        population.append(cand)

    # evaluate initial scores
    scores = [compute_c5(ind, dx) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    # GA hyper‑parameters
    tournament_k = 5
    mutation_rate = 0.15
    sigma = 0.12

    while time.time() - start < budget_s * 0.95:  # leave margin

            # reduce mutation strength gradually
            sigma = max(0.02, sigma * 0.95)

    # final evaluation using provided helper for consistency
    c5_bound = compute_c5(best_h, dx)
    return (best_h.tolist(), c5_bound, n_points)
