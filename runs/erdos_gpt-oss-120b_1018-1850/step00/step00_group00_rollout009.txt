def _evaluate(h, dx):
    """Return max overlap for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(candidate, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum target."""
    # initial clipping
    cand = np.clip(candidate, 0.0, 1.0)

    # simple iterative redistribution to hit target sum
    for _ in range(5):
        cur_sum = cand.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # distribute proportionally to free space
        free_low = cand - 0.0
        free_high = 1.0 - cand
        total_free = free_low + free_high
        # avoid division by zero
        if total_free == 0:
            break
        # move towards target respecting bounds
        delta = diff * (free_high - free_low) / total_free
        cand = np.clip(cand + delta, 0.0, 1.0)
    # final scaling if still off (rare)
    cur_sum = cand.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        scale = target_sum / cur_sum if cur_sum != 0 else 0.0
        cand = np.clip(cand * scale, 0.0, 1.0)

def _mutate(parent, mutation_rate, sigma, rng):
    """Gaussian mutation with per‑gene probability."""
    child = parent.copy()
    mask = rng.random(parent.shape) < mutation_rate
    noise = rng.normal(0.0, sigma, size=parent.shape)
    child[mask] += noise[mask]
    return child
