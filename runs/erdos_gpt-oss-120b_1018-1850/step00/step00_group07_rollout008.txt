import time
import numpy as np

def _adjust_binary(vec, target_sum):
    """Flip random bits until sum equals target_sum (maintains binary values)."""
    v = vec.copy()
    cur = int(v.sum())
    n = len(v)
    rng = np.random.default_rng()
    while cur != target_sum:
        i, j = rng.choice(n, 2, replace=False)
        # flip both bits
        v[i] = 1 - v[i]
        v[j] = 1 - v[j]
        cur = int(v.sum())
    return v

def _c5_score(h):
    """Return C5 for binary/real vector h on [0,2] with spacing dx = 2/n."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    # ----- simulated annealing parameters -----------------------------------

        if time.time() - start > budget_s:
            break
        if T < T_min or it >= max_iter:

        i, j = rng.choice(n_points, 2, replace=False)
        cand = h_current.copy()
        cand[i] = 1 - cand[i]
