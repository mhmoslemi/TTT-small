import numpy as np
import time

def _c5_value(h):
    """Compute C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _project_bounds_and_sum(v, target_sum):
    """Clamp to [0,1] and then shift uniformly to hit the target sum."""
    v = np.clip(v, 0.0, 1.0)
    cur_sum = v.sum()
    if cur_sum == 0:
        # avoid division by zero – distribute uniformly
        v[:] = target_sum / v.size
        return v
    # shift by constant to meet sum, then re‑clip again
    delta = (target_sum - cur_sum) / v.size
    v += delta
    v = np.clip(v, 0.0, 1.0)
    # final adjustment to guarantee exact sum (tiny residual)
    residual = target_sum - v.sum()

        idx = np.argsort(v)[-min(10, v.size):]  # safe small set
        v[idx] += residual / len(idx)
        v = np.clip(v, 0.0, 1.0)
    return v
