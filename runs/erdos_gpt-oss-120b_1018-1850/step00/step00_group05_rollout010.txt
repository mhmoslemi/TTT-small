def compute_c5(h, dx):
    """Return the C5 value for step function h (array) with spacing dx."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def project_onto_constraints(h, target_sum, eps=1e-9, max_iter=100):
    """Clip to [0,1] and adjust to meet the exact sum using a simple water-filling."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < eps:
        return h
    # indices that are not at the bounds
    free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
    if free.size == 0:
        # all at bounds, cannot adjust; fallback to uniform scaling
        return np.clip(h * (target_sum / cur), 0.0, 1.0)
    # distribute diff uniformly over free entries, repeat if some hit bounds
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
        if free.size == 0:
            break
        add = diff / free.size
        h[free] += add
        # re‑clip and recompute diff
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
    return h

def local_search(h0, dx, budget, target_sum):
    """Simple stochastic improvement with projection."""
    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    n = len(h0)

            if val_j < best_val:
