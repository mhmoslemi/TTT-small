import numpy as np, time, math

def max_overlap(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_sum_one(h, target_sum):
    # adjust uniformly then clip, iterate a few times
    for _ in range(5):
        diff = target_sum - h.sum()
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    # final fix if still off by tiny amount
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff to first element that can absorb it
        for i in range(h.size):
            avail = 1.0 - h[i] if diff > 0 else h[i]
            take = max(min(diff, avail), -avail)
            h[i] += take
            diff -= take
            if abs(diff) < 1e-12:
                break
    return h

def initialize_h(n_points):
    # use provided initial_h_values if present, otherwise random
    try:
        init = initial_h_values  # noqa: F821
        init = np.asarray(init, dtype=float)
        # resize via linear interpolation
        x_old = np.linspace(0, 2, init.size, endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
        # ensure feasibility
        h = np.clip(h, 0.0, 1.0)
        target = n_points / 2.0
        h = project_sum_one(h, target)
        return h
    except NameError:
        # random feasible vector
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        target = n_points / 2.0
        h = project_sum_one(h, target)
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Choose a modest size for speed; can be increased if time permits
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initialize_h(n_points)
    best_h = h.copy()
    best_val = max_overlap(h, dx)

    # Simulated annealing parameters
    max_iter = 200000
    T0 = 0.1
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break
        # temperature schedule
        T = T0 * (1 - it / max_iter)

        # propose a move: transfer epsilon between two indices
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue
        # max amount we can move from i to j while respecting bounds
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (np.random.rand() - 0.5) * 2 * max_eps * 0.5  # scaled random
        eps = np.clip(eps, -max_eps, max_eps)

        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        # evaluate
        val = max_overlap(h_new, dx)

        # accept if better or with Metropolis probability
        if val < best_val or np.random.rand() < math.exp((best_val - val) / (T + 1e-12)):
            h = h_new
            if val < best_val:
                best_val = val
                best_h = h_new.copy()

    # final projection to ensure exact sum (should already hold)
    best_h = project_sum_one(best_h, target_sum)

    return best_h, best_val, n_points
