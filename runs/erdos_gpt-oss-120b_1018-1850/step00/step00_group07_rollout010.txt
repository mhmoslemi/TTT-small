import time
import numpy as np

def _binary_adjust(arr, desired):
    """Flip random pairs until sum equals desired."""
    arr = arr.astype(np.int8)
    cur = arr.sum()
    n = len(arr)
    rng = np.random.default_rng()
    while cur != desired:
        i, j = rng.choice(n, size=2, replace=False)
        # swapping two bits changes sum by (1-2*arr[i]) + (1-2*arr[j])
        delta = (1 - 2 * arr[i]) + (1 - 2 * arr[j])
        if cur + delta == desired or delta == 0:
            arr[i] = 1 - arr[i]
            arr[j] = 1 - arr[j]
            cur += delta
        else:
            # perform a flip that moves sum towards desired
            if cur < desired:
                # need more ones: turn a 0 into 1, turn a 1 into 0 elsewhere
                if arr[i] == 0:
                    arr[i] = 1
                    cur += 1
                if cur < desired and arr[j] == 0:
                    arr[j] = 1
                    cur += 1
                if cur > desired:
                    # overshoot, revert one of them
                    if arr[i] == 1:
                        arr[i] = 0
                        cur -= 1
                    elif arr[j] == 1:
                        arr[j] = 0
                        cur -= 1
            else:
                # need fewer ones
                if arr[i] == 1:
                    arr[i] = 0
                    cur -= 1
                if cur > desired and arr[j] == 1:
                    arr[j] = 0
                    cur -= 1
                if cur < desired:
                    if arr[i] == 0:
                        arr[i] = 1
                        cur += 1
                    elif arr[j] == 0:
                        arr[j] = 1
                        cur += 1
    return arr

def _c5_score(h, dx):
    """Maximum overlap integral for binary/real h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx
