import numpy as np, time, math, random
# evaluate_erdos_solution and initial_h_values are provided by the harness

def compute_c5(h):
    """Return C5 value for given discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_h(h, target_sum):
    """Clamp h to [0,1] and scale to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        return np.full_like(h, target_sum / h.size)
    # simple affine scaling then clip again
    scale = target_sum / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # final correction for any residual sum error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly while respecting bounds
        inc = diff / h.size
        h += inc
        h = np.clip(h, 0.0, 1.0)
        # tiny residual fix by adjusting first element
        residual = target_sum - h.sum()
        if abs(residual) > 1e-12:
            h[0] = min(1.0, max(0.0, h[0] + residual))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # Determine discretization size
    if 'initial_h_values' in globals():

        n = h0.size

    target_sum = n / 2.0  # because integral =1 -> sum = n/2
    h0 = project_h(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Simple stochastic local search
    max_iter = 200000
    iter_cnt = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break
        if iter_cnt >= max_iter:
            break
        iter_cnt += 1

        # propose perturbation
        cand = best_h.copy()
        # pick a random block to modify
        i = random.randrange(n)
        delta = (random.random() - 0.5) * 0.2  # up to +-0.1
        cand[i] += delta
        # small spread to keep total sum
        # adjust another random index oppositely to keep sum roughly
        j = random.randrange(n)
        if i != j:
            cand[j] -= delta
        cand = project_h(cand, target_sum)

        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand
            # early exit if already below target
            if best_c5 <= 0.3808:

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
