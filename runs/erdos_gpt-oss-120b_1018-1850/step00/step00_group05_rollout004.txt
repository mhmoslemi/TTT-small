import numpy as np
import time

def compute_c5(h):
    """Return max overlap integral for step function h."""
    dx = 2.0 / h.size
    # full correlation length = 2N-1, index shift corresponds to shift k*dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust to achieve exact sum."""
    h = np.clip(h, 0.0, 1.0)
    # Simple affine correction to meet sum, then re‑clip iteratively
    for _ in range(5):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute proportionally to distance from bounds
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # fallback: distribute equally
            h += diff / h.size
        else:
            step = diff / free.size
            h[free] += step
        h = np.clip(h, 0.0, 1.0)
    # final exact correction on a single variable (if needed)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
