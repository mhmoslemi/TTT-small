from itertools import islice

def _evaluate(h):
    # correlate h with 1-h, full mode, then multiply by dx
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_feasible(n):
    # uniform feasible start: half the entries 1, half 0 (or as close as possible)
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    # if n is odd, adjust one element to 0.5 to keep integral =1
    if n % 2:
        h[ones] = 0.5
    np.random.shuffle(h)
    return h
