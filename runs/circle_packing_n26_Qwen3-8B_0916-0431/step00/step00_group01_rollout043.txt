import numpy as np
import cvxpy as cp
import random

def run_packing():
    # Step 1: Define an initial placement grid
    grid_size = 5
    grid_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            grid_centers.append([x, y])
    # Add additional centers to reach 26
    for _ in range(26 - len(grid_centers)):
        x = random.uniform(0, 1)
        y = random.uniform(0, 1)
        grid_centers.append([x, y])
    centers = np.array(grid_centers)

    # Step 2: Define optimization variables
    r = cp.Variable(26)
    obj = cp.Maximize(cp.sum(r))

    # Step 3: Define constraints
    constraints = []
    for i in range(26):
        # Ensure circle i is inside the square
        constraints.append(centers[i, 0] - r[i] >= -1e-12)
        constraints.append(centers[i, 0] + r[i] <= 1 + 1e-12)
        constraints.append(centers[i, 1] - r[i] >= -1e-12)
        constraints.append(centers[i, 1] + r[i] <= 1 + 1e-12)

    # Step 4: Ensure no two circles overlap
    for i in range(26):
        for j in range(i + 1, 26):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-12)

    # Step 5: Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    try:
        problem.solve(solver=cp.SCS, max_iter=1000)
    except:
        # Fallback to default if optimization fails
        r = np.full(26, 0.05)

    # Step 6: Return the final packing
    radii = r.value
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
