import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial grid-based placement
    n_circles = 26
    rows = 5
    cols = 6

    # Grid spacing
    x_interval = 1.0 / (cols - 1)
    y_interval = 1.0 / (rows - 1)

    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # Initial grid-based placement
    for i in range(n_circles):
        col = i % cols
        row = i // cols
        x = col * x_interval
        y = row * y_interval
        centers[i] = [x, y]
        radii[i] = 0.05  # initial small radius

    # Optimization parameters
    max_iter = 100
    learning_rate = 0.01

    for _ in range(max_iter):

            idx = np.random.randint(n_circles)
            x, y = centers[idx]
            r = radii[idx]

            # Try moving the circle to the nearest boundary
            dx = 0
            dy = 0
            min_dist = float('inf')

            # Check all other circles for overlap
            for j in range(n_circles):
                if j == idx:
                    continue
                xj, yj = centers[j]
                rj = radii[j]
                dist = np.hypot(x - xj, y - yj)
                if dist < r + rj - 1e-12:
                    # Need to move this circle

                    dir_y = (yj - y) / dist
                    move_dist = (r + rj - dist - 1e-12)
                    dx += dir_x * move_dist
                    dy += dir_y * move_dist
                    min_dist = min(min_dist, dist)

            # Check boundaries

            if x + r > 1 + 1e-12:

                dy = max(dy, r - y)
            if y + r > 1 + 1e-12:
                dy = min(dy, -r + y)

            # Move circle

            # Update radius if possible

    # Final validation
    centers = np.round(centers, 10)

    return centers, radii, np.sum(radii)
