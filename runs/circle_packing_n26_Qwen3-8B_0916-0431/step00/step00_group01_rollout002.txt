from scipy.spatial.distance import pdist, squareform
from typing import Tuple

def compute_overlap_penalty(centers: np.ndarray, radii: np.ndarray, epsilon: float = 1e-6) -> float:
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            if dist < radii[i] + radii[j] - epsilon:
                penalty += (radii[i] + radii[j] - dist - epsilon) ** 2
    return penalty

def compute_boundary_penalty(centers: np.ndarray, radii: np.ndarray, epsilon: float = 1e-6) -> float:

    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -epsilon or x + r > 1 + epsilon or y - r < -epsilon or y + r > 1 + epsilon:
            penalty += (max(0, x - r + epsilon) + max(0, x + r - 1 - epsilon) + max(0, y - r + epsilon) + max(0, y + r - 1 - epsilon)) ** 2
    return penalty
