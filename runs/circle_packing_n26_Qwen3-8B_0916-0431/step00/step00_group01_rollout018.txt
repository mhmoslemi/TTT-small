import numpy as np
from scipy.optimize import minimize
from numpy import sqrt, pi, where, ones, arange, diag, dot, outer, kron
import matplotlib.pyplot as plt

def run_packing():
    # Define a hexagonal grid inside the unit square
    n_rows = 3
    n_cols = 9
    max_radius = 0.1  # Initial guess for radius

    # Generate the hexagonal grid points
    centers = []
    radii = []
    for row in range(n_rows):
        for col in range(n_cols):
            x = col / (n_cols - 1)
            y = row / (n_rows - 1)
            centers.append([x, y])
            radii.append(max_radius)

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative of sum to maximize sum

    def constraints_func(radii_vec):
        n = len(radii_vec)
        r = radii_vec

        # Ensure radii are non-negative
        for i in range(n):
            if r[i] < 0:
                return False, f"Circle {i} has negative radius {r[i]}"

        # Ensure circles are within the unit square
        for i in range(n):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r[i]} outside unit square"

        # Ensure circles are non-overlapping
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    # Initialize with the initial radii
    initial_radii = radii.copy()

    # Set up the optimization problem
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=[(0, 1)] * len(radii),
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda x: 0,
                'jac': None
            }
        ],
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract the optimized centers and radii
    optimized_radii = result.x
    optimized_centers = centers

    # Validate the packing after optimization
    try:
        valid, msg = constraints_func(optimized_radii)
        if not valid:

        optimized_radii = np.zeros(len(optimized_radii))
        optimized_centers = np.zeros((len(optimized_radii), 2))

    sum_radii = np.sum(optimized_radii)
    return (optimized_centers, optimized_radii, sum_radii)
