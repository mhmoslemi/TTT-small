import cvxpy as cp

            y = row + 0.5 * (row % 2)
            centers.append([x, y])

    centers = np.array(centers)

    centers = (centers - np.min(centers)) * scale

    n = len(centers)
    radii = cp.Variable(n)

    # Objective: maximize total radius sum
    objective = cp.Maximize(cp.sum(radii))

        constraint = (x - radii[i]) >= 0
        constraint = constraint + (x + radii[i]) <= 1
