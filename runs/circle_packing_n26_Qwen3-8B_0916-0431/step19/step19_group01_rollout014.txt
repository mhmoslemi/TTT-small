import scipy.optimize as optimize

def run_packing():

    rows = 5
    cols = 5

    vertical_spacing = 0.4
    offset = 0.05  # Stagger offset for alternating rows

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n - 1):

        col = i % cols

        y += np.random.uniform(-0.02, 0.02)

        y = np.clip(y, 0, 1)
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.05  # Start with some small initial radius

    # Place the last circle at a corner to utilize full boundary
    centers[-1, 0] = 0.99
    centers[-1, 1] = 0.99
    radii[-1] = 0.05

    # Objective: maximize sum of radii
    def objective(x):

        centers_opt = x[n:].reshape((n, 2))
        return -np.sum(radii_opt)  # Minimize negative sum to maximize total

    # Non-overlap and boundary constraints
    def constraints(x):
        radii_opt = x[:n]
        centers_opt = x[n:].reshape((n, 2))

        # Boundary constraints: circle must be inside square with margin
        boundary = []
        for i in range(n):
            boundary.append(centers_opt[i, 0] - radii_opt[i] >= -1e-10)
            boundary.append(1.0 - centers_opt[i, 0] - radii_opt[i] >= -1e-10)
            boundary.append(centers_opt[i, 1] - radii_opt[i] >= -1e-10)
