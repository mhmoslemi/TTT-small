import scipy.optimize as optimize

def run_packing():

    # Define an adaptive hexagonal grid with rows and columns
    rows = 5
    cols = 5
    spacing_x = 0.23
    spacing_y = 0.22
    offset_x = 0.05
    offset_y = 0.05

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):
        row = i // cols
        col = i % cols

        # Introduce staggered positioning
        if row % 2 == 0:
            x = col * spacing_x + offset_x
        else:
            x = col * spacing_x + offset_x + spacing_x / 2

        y = row * spacing_y + offset_y

        # Clip to the unit square
        x = np.clip(x, 0.01, 0.99)
        y = np.clip(y, 0.01, 0.99)

        # Initial radius based on the minimum spacing
        r = min(spacing_x / 2, spacing_y / 2)
        radii[i] = r
        centers[i, 0] = x
        centers[i, 1] = y

    # Define the objective function to maximize total sum of radii
    def objective(x):
        radii_guess = x[:n]
        centers_guess = x[n:].reshape((n, 2))
        return -np.sum(radii_guess)  # We minimize the negative sum to maximize it

    # Define constraints
    def constraints(x):
        radii_guess = x[:n]
        centers_guess = x[n:].reshape((n, 2))

        # Boundary constraints
        boundary_constraints = []
        for i in range(n):
            boundary_constraints.append(centers_guess[i, 0] - radii_guess[i])  # Left edge
            boundary_constraints.append(1.0 - centers_guess[i, 0] - radii_guess[i])  # Right edge
            boundary_constraints.append(centers_guess[i, 1] - radii_guess[i])  # Bottom edge
            boundary_constraints.append(1.0 - centers_guess[i, 1] - radii_guess[i])  # Top edge

        # Non-overlap constraints
        non_overlap = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_guess[i, 0] - centers_guess[j, 0]
                dy = centers_guess[i, 1] - centers_guess[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                non_overlap.append(dist - (radii_guess[i] + radii_guess[j]))

        return boundary_constraints + non_overlap

    # Bounds for all variables
    bounds = []
    for i in range(n):

    x0 = np.concatenate((radii, centers.flatten()))

    result = optimize.minimize(
