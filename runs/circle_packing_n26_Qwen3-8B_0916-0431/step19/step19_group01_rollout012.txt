                # Edge circles
                radii[i] = radius_edge
            else:
                # Center circles
                radii[i] = radius_base
            centers[i, 0] = x

        return -np.sum(radii_opt)  # Minimize negative sum to maximize total

    def constraints(x):

        for i in range(n):
