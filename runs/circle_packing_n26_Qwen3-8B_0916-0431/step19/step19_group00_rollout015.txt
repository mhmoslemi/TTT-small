import scipy.optimize as optimize

def run_packing():

    # Place the last circle near a corner
    centers[25, 0] = 0.9
    centers[25, 1] = 0.9

    def objective(x):

        centers_guess = x[n:].reshape((n, 2))
        return -np.sum(radii_guess)  # Maximize the sum of radii

    # Define constraints function
    def constraints(x):
        radii_guess = x[:n]
        centers_guess = x[n:].reshape((n, 2))

        # Boundary constraints: circle must be within the square
        boundary_constraints = []
        for i in range(n):
            boundary_constraints.append(centers_guess[i, 0] - radii_guess[i])
