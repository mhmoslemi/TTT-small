import scipy.optimize as optimize

def run_packing():

    # This allows more flexibility and uses both center and corner spaces

    rows = 5
    cells_per_row = 5
    horizontal_spacing = 0.22
    vertical_spacing = 0.22
    row_offset = 0.05

    for i in range(n):
        row = i // cells_per_row
        col = i % cells_per_row
        # Introduce staggered positions for even and odd rows
        x = col * horizontal_spacing + (row % 2) * horizontal_spacing / 2
        y = row * vertical_spacing + row_offset

        # Clip to [0, 1] to ensure all circles are inside the unit square
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)

        # Initial radius based on spacing and boundary constraints
        initial_radius = min(horizontal_spacing / 2, vertical_spacing / 2, 0  # Adjustable based on domain
                             )
        radii[i] = initial_radius
        centers[i, 0] = x
        centers[i, 1] = y

    # Objective function: maximize the sum of radii
    def objective(x):
        radii_opt = x[:n]
        centers_opt = x[n:].reshape((n, 2))
        return -np.sum(radii_opt)  # Minimize negative sum to maximize total

    # Constraints: boundary and non-overlap
    def constraints(x):
        radii_opt = x[:n]
        centers_opt = x[n:].reshape((n, 2))

        # Boundary constraints (x - r >= 0, 1 - x - r >= 0, same for y)
        boundary_constraints = []
        for i in range(n):
            boundary_constraints.append(centers_opt[i, 0] - radii_opt[i])
            boundary_constraints.append(1.0 - centers_opt[i, 0] - radii_opt[i])

            boundary_constraints.append(1.0 - centers_opt[i, 1] - radii_opt[i])

        # Non-overlap constraints (distance between centers >= sum of radii)
        non_overlap = []
        for i in range(n):

                dx = centers_opt[i, 0] - centers_opt[j, 0]

    result = optimize.minimize(
