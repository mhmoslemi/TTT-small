import scipy.optimize as optimize

def run_packing():

    row_spacing = 1 / (7)
    col_spacing = 1 / (5)

    centers = []
    radii = []

    # Define the angles of the staggered rows
    row_offset = np.sqrt(3) / 2 * row_spacing  # vertical offset between rows
    col_offset = 0.5 * col_spacing  # horizontal offset for staggered rows

    for row in range(7):
        if row % 2 == 0:
            # Even rows (left-aligned)
            x = np.arange(0, 5) * col_spacing
            y = row * row_spacing
        else:
            # Odd rows (right-aligned)
            x = np.arange(0, 5) * col_spacing + col_offset
            y = row * row_spacing + row_offset

        for i, (xi, yi) in enumerate(zip(x, y)):
            # Calculate radius based on hexagonal grid
            # Midpoint between adjacent circles in this row
            if i > 0:
                dist_to_left = np.sqrt((x[i] - x[i-1])**2 + (y[i] - y[i-1])**2)
                radius = dist_to_left / 2
            else:
                radius = 1 / 10  # Default small radius for border circles

            # Define center
            center = np.array([xi, yi])
            centers.append(center)
            radii.append(radius)

    # Convert to NumPy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # --- Step 2: Refinement Using Optimization ---
    # We'll use scipy's optimization to maximize the sum of radii,
    # subject to non-overlapping and staying within the square

    def objective(radii):
        # We aim to maximize the sum of radii
        return -np.sum(radii)  # Minimize the negative sum

    # Define constraints:

    # 2. Distance between circles must be at least the sum of their radii

    def constraint1(radii):
        # Ensure all circles are within the unit square
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1
        return 0

    def constraint2(radii):
        # Ensure circles do not overlap
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:

        return 0

    radii_guess = np.array([r for r in radii])

    # Bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(len(radii_guess))]

    # Optimization
    result = optimize.minimize(
        objective,
        radii_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ]
    )

    # Update radii
    radii = result.x

    # Final values

    return (centers, radii, sum_radii)
