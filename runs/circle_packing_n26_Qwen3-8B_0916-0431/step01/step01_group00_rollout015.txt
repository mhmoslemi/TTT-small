import scipy.optimize as optimize

def run_packing():

    # We'll use a hexagonal packing with 25 circles and one more placed in the corner
    def hex_grid(n_rows, n_cols, spacing):
        centers = []
        for i in range(n_rows):
            for j in range(n_cols):
                x = j * spacing
                y = i * spacing * (np.sqrt(3) / 2)
                centers.append([x, y])
        return np.array(centers)

    # Initial guess from a hexagonal packing with spacing 0.25
    def initial_centers():
        return hex_grid(5, 5, 0.25)

    # Define initial configuration for 25 circles in a hexagonal grid
    centers_grid = initial_centers()
    # Add one extra circle at (0.05, 0.05) to include corner spacing
    extra_circle = np.array([[0.05, 0.05]])
    centers = np.vstack([centers_grid, extra_circle])
    # Ensure total is 26 circles
    assert centers.shape[0] == 26, "Must have exactly 26 circles"

    radii = np.full(26, 0.08)

    # Define the target constraint function
    def constraint_func(radii):
        constraints = []
        for i in range(26):
            for j in range(i + 1, 26):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Optimization objective
    def objective(radii):
        return -np.sum(radii)

    # Set bounds for each radius, ensuring no negative values
    bounds = [(0, 0.5) for _ in range(26)]

    # Perform optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': constraint_func
        },
        method='SLSQP'
    )

    radii_opt = result.x
    centers_opt = centers.copy()

    # Return the centers, radii, and sum of radii
