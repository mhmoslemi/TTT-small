import random
import math
import copy

    best_radii = np.zeros(n)

        return np.sum(radii)

        # Check boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -boundary_tolerance or x + r > 1 + boundary_tolerance or

                return False

                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - overlap_tolerance:
                    return False
        return True
