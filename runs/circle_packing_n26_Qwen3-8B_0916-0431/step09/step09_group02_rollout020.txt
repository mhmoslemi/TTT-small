import scipy.optimize as optimize
from copy import deepcopy

def run_packing():

        radii = np.random.rand(n) * 0.1

    def initialize_hexagonal():
        centers = np.zeros((n, 2))
        radii = np.zeros(n)

        row_count = 6
        col_count = 5
        spacing = 0.2
        offset = 0.1

        for i in range(n):
            row = i // col_count
            col = i % col_count
            x = offset + col * spacing + spacing / 2
            y = offset + row * spacing + spacing / 2
            centers[i] = [x, y]
            radii[i] = (spacing / 2) * 0.9

        return centers, radii

    def evaluate_layout(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or
                y - r < 0 or y + r > 1):
                return False, "Circle out of bounds"

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Circles overlap"

        return True, "Valid layout"

    def optimize_layout(centers, radii):
        x0 = np.concatenate((radii, centers.flatten()))

        def objective(x):
            return -np.sum(x[:n])  # To maximize total radii, minimize negative of sum

            centers = x[n:].reshape((n, 2))
