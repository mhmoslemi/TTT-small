import scipy.optimize as optimize

    cols = 6  # Slightly asymmetric layout for better packing

    radii = np.zeros(n)

    offset = 0.01

        col = i % cols
        row = i // cols

        # Radius cannot exceed half of the minimum distance to boundary and neighbor
        dist_to_left = x
        dist_to_right = 1 - x
        dist_to_bottom = y
        dist_to_top = 1 - y
        min_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)

        # Compute distance to nearest neighbor (left or right depends on row)
        if row % 2 == 0:
            if col > 0:
                neighbor_x = centers[i - 1, 0]
                dist_to_neighbor = abs(x - neighbor_x)
            else:
                dist_to_neighbor = np.inf
        else:
            if col < cols - 1:
                neighbor_x = centers[i - cols, 0]
                dist_to_neighbor = abs(x - neighbor_x)
            else:
                dist_to_neighbor = np.inf

        # Radius initialized to half the minimum of boundary distance and neighbor distance
        radii[i] = min(min_dist / 2, dist_to_neighbor / 2)

        centers[i, 0] = x
        centers[i, 1] = y

    # Define the objective: maximize total radius sum
    def objective(x):

        centers = x[n:].reshape((n, 2))
        return -np.sum(radii)  # Negative for minimization

    # Define constraints: boundary and non-overlap
    def constraints(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))

        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, similar for y
        boundary = []
        for i in range(n):
            boundary.append(centers[i, 0] - radii[i])
