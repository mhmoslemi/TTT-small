import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Hexagonal-like staggered grid with dynamic adjustment
    col_count = 5
    row_count = 6
    col_spacing = 0.2
    row_spacing = 0.25
    offset = 0.02
    corner_enlargement = 0.1
    edge_enlargement = 0.05

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Build staggered grid
    for i in range(n):
        col = i % col_count
        row = i // col_count

        # Staggered grid
        if row % 2 == 1:
            x = col * col_spacing + offset
        else:
            x = col * col_spacing

        y = row * row_spacing

        # Expand radii for corner and edge circles
        is_corner = (col == 0 or col == col_count - 1) and (row == 0 or row == row_count - 1)
        is_edge = (col == 0 or col == col_count - 1) or (row == 0 or row == row_count - 1)

        if is_corner:
            radii[i] = min(col_spacing / 2, row_spacing / 2) * (1 + corner_enlargement)
        elif is_edge:
            radii[i] = min(col_spacing / 2, row_spacing / 2) * (1 + edge_enlargement)
        else:
            radii[i] = min(col_spacing / 2, row_spacing / 2)

        # Clip to ensure circles are inside the unit square
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)

        centers[i, 0] = x
        centers[i, 1] = y

    # Define objective function
    def objective(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))
        return -np.sum(radii)  # Negative for minimization

    # Define constraints for non-overlap and boundary
    def constraints(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))

        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, similar for y
        boundary = []
        for i in range(n):
            boundary.append(centers[i, 0] - radii[i])
            boundary.append(1.0 - centers[i, 0] - radii[i])
            boundary.append(centers[i, 1] - radii[i])
            boundary.append(1.0 - centers[i, 1] - radii[i])

        # Non-overlap constraints: distance >= r_i + r_j
        non_overlap = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                non_overlap.append(dist - (radii[i] + radii[j]))

        return boundary + non_overlap

    # Bounds: (radius, x, y) all between 0 and 1
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1)] * 3)  # (radii, x, y)

    # Initial guess: take seed with reasonable radii and centers
    x0 = np.concatenate((radii, centers.flatten()))

    # Run the optimization with tighter tolerances and more iterations
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        method='SLSQP',
        tol=1e-12,
        options={
            'maxiter': 50000,
            'eps': 1e-9,
            'ftol': 1e-12,
            'gtol': 1e-12,
            'iprint': 0,
            'disp': False
        }
    )

    # Return results
    radii_opt = result.x[:n]
    centers_opt = result.x[n:].reshape((n, 2))

    return (centers_opt, radii_opt, np.sum(radii_opt))
