import scipy.optimize as optimize

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Hexagonal grid dimensions
    rows = 5
    cols = 6
    horizontal_spacing = 0.3
    vertical_spacing = 0.3
    offset = 0.03

    # Generate staggered hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols

        # Staggered offset for hexagonal pattern
        x = col * horizontal_spacing + (row % 2) * offset
        y = row * vertical_spacing

        # Ensure within bounds
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)

        # Initialize radius based on spacing
        initial_radius = min(horizontal_spacing / 2, vertical_spacing / 2)
        radii[i] = initial_radius

        centers[i, 0] = x
        centers[i, 1] = y

    # Objective to maximize the sum of radii
    def objective(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))
        return -np.sum(radii)  # Minimize negative sum to maximize total

    # Constraints for non-overlap and boundary
    def constraints(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))

        # Boundary constraints
        boundary_constraints = []
        for i in range(n):
            boundary_constraints.append(centers[i, 0] - radii[i])
            boundary_constraints.append(1.0 - centers[i, 0] - radii[i])
            boundary_constraints.append(centers[i, 1] - radii[i])
            boundary_constraints.append(1.0 - centers[i, 1] - radii[i])

        non_overlap = []

    bounds = []

        bounds.extend([(0, 1)] * 3)

    x0 = np.concatenate((radii, centers.flatten()))
