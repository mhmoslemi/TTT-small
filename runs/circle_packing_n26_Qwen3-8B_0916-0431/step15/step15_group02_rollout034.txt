import scipy.optimize as optimize

def run_packing():

    row_count = 6
    col_spacing = 0.2
    row_spacing = 0.25
    offset = 0.02

    # Compute initial positions in a staggered grid
    for i in range(n):
        col = i % col_count
        row = i // col_count

        # Hexagonal arrangement with alternating offset
        x = col * col_spacing + (row % 2) * offset
        y = row * row_spacing

        # Ensure it's within [0, 1]
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)

        # Set initial radius based on spacing; can be adjusted in optimization
        radii[i] = min(col_spacing / 2, row_spacing / 2)

        centers[i, 0] = x
        centers[i, 1] = y

    # Define the objective: maximize total radius sum
    def objective(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))
        return -np.sum(radii)  # Negative for minimization

    # Define constraints for non-overlap and boundary
    def constraints(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))

        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, similar for y
        boundary = []
        for i in range(n):
            boundary.append(centers[i, 0] - radii[i])
