import numpy as np
import random
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    # Genetic algorithm parameters
    pop_size = 100
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    initial_pop = []

    # Function to check constraints
    def check_valid(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False, f"Circle {i} is out of bounds"
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return False, f"Circles {i} and {j} overlap"
        return True, "Valid configuration"

    # Initialize population with structured starting points
    for _ in range(pop_size):
        centers = np.random.rand(n, 2)
        radii = np.random.rand(n)
        initial_pop.append((centers, radii))

    for gen in range(generations):
        # Evaluate fitness
        def fitness(individual):
            centers, radii = individual
            valid, reason = check_valid(centers, radii)
            if not valid:
                return 0  # Invalid configuration
            return np.sum(radii)

        scores = [fitness(ind) for ind in initial_pop]
        ranked = sorted(zip(scores, initial_pop), reverse=True)

        # Select elites
        elites = ranked[:elite_size]
        new_pop = [elite[1] for elite in elites]

        # Generate offspring
        while len(new_pop) < pop_size:
            parent1, parent2 = random.choices(ranked, k=2)
            child_centers, child_radii = parent1[1], parent2[1]

                child_centers[:split] = parent1[1][0][:]
                child_centers[split:] = parent2[1][:]
                child_radii[:split] = parent1[1][:]
                child_radii[split:] = parent2[1][:]
            else:
                child_centers = parent1[1][:]
                child_radii = parent2[1][:]
