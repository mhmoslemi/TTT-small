import numpy as np
import scipy.optimize as optimize

def run_packing():
    n = 26

    # Strategy: Hybrid packing with optimized corner and edge placements
    # We place 24 circles in a hexagonal lattice with dynamic spacing,
    # and 2 additional circles near two opposite corners to utilize boundary space

    # Define the number of rows and columns for the hexagonal lattice
    hex_row_count = 5
    hex_col_count = 5
    horizontal_spacing = 0.26
    vertical_spacing = 0.26
    offset = 0.02
    row_offset = 0.02

    # Initialize centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 24 circles in a hexagonal pattern in the center
    for i in range(24):
        row = i // hex_col_count
        col = i % hex_col_count

        # Introduce staggered positions for even and odd rows
        x = col * horizontal_spacing + (row % 2) * offset
        y = row * vertical_spacing + row_offset

        # Ensure within bounds with some padding for dynamic adjustment
        x = np.clip(x, 0.05, 0.95)
        y = np.clip(y, 0.05, 0.95)

        # Initial radius based on spacing and boundary constraints
        horizontal_buffer = min(0.95 - x, x - 0.05)
        vertical_buffer = min(0.95 - y, y - 0.05)
        initial_radius = min(horizontal_buffer, vertical_buffer, horizontal_spacing / 2, vertical_spacing / 2)
        initial_radius = min(initial_radius, 0.15)

        radii[i] = initial_radius
        centers[i, 0] = x
        centers[i, 1] = y

    # Add the 25th circle near the top-left corner
    x = 0.05 + 0.1
    y = 0.95 - 0.1
    initial_radius = 0.1  # Smaller to prevent overlap
    radii[24] = initial_radius
    centers[24, 0] = x
    centers[24, 1] = y

    # Add the 26th circle near the bottom-right corner
    x = 0.95 - 0.1
    y = 0.05 + 0.1
    initial_radius = 0.1
    radii[25] = initial_radius
    centers[25, 0] = x
    centers[25, 1] = y

    # Objective to maximize the sum of radii
    def objective(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))
        return -np.sum(radii)  # Minimize negative sum to maximize total

    # Constraints for non-overlap and boundary
    def constraints(x):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))

        # Boundary constraints (distance from edge must be >= radius)
        boundary_constraints = []
        for i in range(n):
            boundary_constraints.append(centers[i, 0] - radii[i])
            boundary_constraints.append(1.0 - centers[i, 0] - radii[i])
            boundary_constraints.append(centers[i, 1] - radii[i])
            boundary_constraints.append(1.0 - centers[i, 1] - radii[i])

        # Non-overlap constraints (distance between centers >= sum of radii)
        non_overlap = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                non_overlap.append(dist - (radii[i] + radii[j]))

        return boundary_constraints + non_overlap

    # Bounds: (radius, x, y) for each circle
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1)] * 3)

    # Initial guess
    x0 = np.concatenate((radii, centers.flatten()))

    # Optimization parameters
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        method='SLSQP',
        tol=1e-12,
        options={
            'maxiter': 100000,  # Increased iterations for better convergence
            'eps': 1e-10,
            'ftol': 1e-12,
            'gtol': 1e-12,
            'iprint': 0,
            'disp': False
        }
    )

    # Extract results
    radii_opt = result.x[:n]
    centers_opt = result.x[n:].reshape((n, 2))

    return (centers_opt, radii_opt, np.sum(radii_opt))
