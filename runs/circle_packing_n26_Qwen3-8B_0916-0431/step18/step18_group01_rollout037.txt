            centers[i, 1] = y

        def objective(x):
            radii_guess = x[:n]
            centers_guess = x[n:].reshape((n, 2))
            return -np.sum(radii_guess)  # Maximize sum of radii

        # Constraints function
        def constraints(x):
            radii_guess = x[:n]
            centers_guess = x[n:].reshape((n, 2))
            boundary_constraints = []
            for i in range(n):
                boundary_constraints.append(centers_guess[i, 0] - radii_guess[i])
                boundary_constraints.append(1.0 - centers_guess[i, 0] - radii_guess[i])
