import numpy as np
import scipy.optimize as optimize

def run_packing():
    n = 26

    # Define a hybrid packing: two rows of circles along the top and bottom
    # with 5 circles per row, and one central column with 6 circles
    # Total rows = 12 (top and bottom rows) + 5 (central column) = 17
    # Total = 2*5 (top and bottom) + 6 (central) = 16; need 26 -> adjust

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 2 rows (top and bottom) with 5 circles each
    for i in range(2):
        for j in range(5):
            idx = i * 5 + j
            x = j / 4.0
            y = 1.0 if i == 0 else 0.0
            centers[idx] = [x, y]
            radii[idx] = 0.1  # Initial guess

    # Place a central column of 6 circles
    for i in range(6):
        idx = 10 + i
        x = 0.5
        y = i / 5.0
        centers[idx] = [x, y]
        radii[idx] = 0.1  # Initial guess

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(radii):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Boundary constraint: circle within [0,1] x [0,1]
    def boundary_constraints(radii):
        constraints = []
        for i in range(n):
            # Left boundary
            constraints.append(centers[i, 0] - radii[i])
            # Right boundary
            constraints.append(1.0 - centers[i, 0] - radii[i])
            # Bottom boundary
            constraints.append(centers[i, 1] - radii[i])
            # Top boundary
            constraints.append(1.0 - centers[i, 1] - radii[i])
        return constraints

    # Initial guess
    x0 = radii.copy()

    # Bounds: radii cannot be negative
    bounds = [(0, 1.0) for _ in range(n)]

    # Set up the optimization problem
    cons = []
    cons.append({'type': 'ineq', 'fun': non_overlap_constraints})
    cons.append({'type': 'ineq', 'fun': boundary_constraints})

    # Use a constrained optimization method
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-8,
        options={'maxiter': 2000, 'eps': 1e-8}
    )

    radii_opt = result.x
    centers_opt = centers.copy()

    return (centers_opt, radii_opt, np.sum(radii_opt))
