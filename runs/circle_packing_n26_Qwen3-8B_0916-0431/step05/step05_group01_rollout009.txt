import scipy.optimize as optimize

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    spacing = 0.2
    for i in range(num_rows):
        for j in range(num_cols):
            idx = i * num_cols + j
            if i % 2 == 0:
                # Even rows, regular grid
                x = j * spacing + spacing / 2
                y = i * spacing + spacing / 2
            else:
                # Odd rows, offset by half spacing

                y = i * spacing + spacing / 2
            centers[idx] = [x, y]
            radii[idx] = spacing / 2

    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

            # Left boundary

            # Right boundary

            # Bottom boundary

            # Top boundary
