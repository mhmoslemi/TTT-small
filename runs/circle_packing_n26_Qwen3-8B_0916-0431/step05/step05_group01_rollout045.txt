import scipy.optimize as optimize
import random

def run_packing():

    centers = np.random.rand(n, 2)
    radii = np.zeros(n)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

    def non_overlap_constraints(radii):
        constraints = []

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Boundary constraint: circle within [0,1] x [0,1]
    def boundary_constraints(radii):
        constraints = []
        for i in range(n):

            constraints.append(centers[i, 0] - radii[i])
            # Right boundary
            constraints.append(1.0 - centers[i, 0] - radii[i])
            # Bottom boundary
            constraints.append(centers[i, 1] - radii[i])
            # Top boundary
            constraints.append(1.0 - centers[i, 1] - radii[i])
        return constraints

    bounds = [(0, 1.0) for _ in range(n)]
