import scipy.optimize as optimize

def run_packing():

    spacing = 0.2

    cols = 5

    radii = np.zeros(n)

        return x, y

            idx = i * cols + j

    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize
