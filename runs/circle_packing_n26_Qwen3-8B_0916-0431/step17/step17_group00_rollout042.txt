import scipy.optimize as optimize

def run_packing():

    # and fill the remaining with boundary circles to utilize empty corners and sides.

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Parameters for hexagonal grid
    rows = 6
    cols = 5
    horizontal_spacing = 0.25
    vertical_spacing = 0.25
    row_offset = 0.05
    offset = 0.02

    # Place inner circles in a hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols

        # Use staggered positions for hexagonal grid
        x = col * horizontal_spacing + (row % 2) * offset
        y = row * vertical_spacing + row_offset

        # Ensure within bounds
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)

        # Assign initial radius based on spacing
        initial_radius = min(horizontal_spacing / 2, vertical_spacing / 2)
        initial_radius = min(initial_radius, 0.15)
        radii[i] = initial_radius
        centers[i, 0] = x
        centers[i, 1] = y

    # Objective function: maximize the sum of radii
    def objective(x):
        radii_opt = x[:n]
        centers_opt = x[n:].reshape((n, 2))
        return -np.sum(radii_opt)  # Minimize negative sum to maximize total

    # Constraints: non-overlap and boundary
    def constraints(x):
        radii_opt = x[:n]
        centers_opt = x[n:].reshape((n, 2))

        # Boundary constraints (circle must fit inside square)
        boundary_constraints = []
        for i in range(n):
            boundary_constraints.append(centers_opt[i, 0] - radii_opt[i])
            boundary_constraints.append(1.0 - centers_opt[i, 0] - radii_opt[i])
            boundary_constraints.append(centers_opt[i, 1] - radii_opt[i])
            boundary_constraints.append(1.0 - centers_opt[i, 1] - radii_opt[i])

        # Non-overlap constraints
        non_overlap = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)

    x0 = np.concatenate((radii, centers.flatten()))

    result = optimize.minimize(
