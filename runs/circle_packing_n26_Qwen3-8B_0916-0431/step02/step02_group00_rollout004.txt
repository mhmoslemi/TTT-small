import scipy.optimize as optimize

def run_packing():

    # Generate an initial grid-based configuration
    # Use a 6x5 grid with some spacing between circles
    # This gives more flexibility for optimization
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    spacing = 0.2  # Initial spacing between centers

    # Create a grid of centers
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * spacing + spacing / 2
        y = row * spacing + spacing / 2
        centers[i] = [x, y]
        radii[i] = spacing / 2

    # Objective function: maximize total radius
    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

        constraints = []

                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Constraints for circles staying within [0,1] x [0,1]
    def boundary_constraints(radii):
        constraints = []

            constraints.append(centers[i, 0] - radii[i])

            constraints.append(1.0 - centers[i, 0] - radii[i])
            # Bottom boundary
