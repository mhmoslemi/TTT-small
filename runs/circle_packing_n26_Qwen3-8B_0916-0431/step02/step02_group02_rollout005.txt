import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based positions with spacing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Create a grid layout with spacing
    spacing = 0.35
    for i in range(num_circles):
        row = i // 7
        col = i % 7
        x = col * spacing
        y = row * spacing
        centers[i] = [x, y]
        radii[i] = 0.12  # Initial radius estimate
    
    # Define objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative total radius to maximize

    # Define pairwise distance constraint
    def pairwise_distance_constraints(x):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (x[i] + x[j]))
        return constraints

    # Add boundary constraints (each circle must stay within the unit square)
    def boundary_constraints(x):
        constraints = []
        for i in range(num_circles):
            radius = x[i]
            x_cent, y_cent = centers[i]
            # Left boundary
            constraints.append(x_cent - radius)
            # Right boundary
            constraints.append(1.0 - x_cent - radius)
            # Bottom boundary
            constraints.append(y_cent - radius)
            # Top boundary
            constraints.append(1.0 - y_cent - radius)
        return constraints

    # Set up optimization problem
    x0 = radii.copy()

    # Optimization bounds
    bounds = [(0, 1.0) for _ in range(num_circles)]

    # Constraints
    cons_pairwise = {'type': 'ineq', 'fun': pairwise_distance_constraints}
    cons_boundary = {'type': 'ineq', 'fun': boundary_constraints}

    # Run the minimization
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=[cons_pairwise, cons_boundary],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    radii_opt = result.x
    centers_opt = centers.copy()

    return (centers_opt, radii_opt, np.sum(radii_opt))
