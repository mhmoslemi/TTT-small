    ])

    centers = centers[:26]

    # Initial radii: uniform for safety and to allow optimization
    radii = np.full(26, 0.05)  # Start small to prevent overlap

    def total_radius(radii):
        return np.sum(radii)

    def objective(x):
        return -total_radius(x)  # We minimize negative to maximize total

    def overlap_constraints(x):
        constraints = []
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (x[i] + x[j]))
        return constraints

    def boundary_constraints(x):
        constraints = []
        for i in range(26):
            x_dist = centers[i, 0] - x[i]
            y_dist = centers[i, 1] - x[i]
            constraints.append(x_dist)
            constraints.append(y_dist)
            constraints.append(1 - centers[i, 0] - x[i])

    bounds = [(0, 1.0) for _ in range(26)]

    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=[cons_overlap, cons_boundary],

    )

    centers_opt = centers.copy()

    return (centers_opt, radii_opt, np.sum(radii_opt))
