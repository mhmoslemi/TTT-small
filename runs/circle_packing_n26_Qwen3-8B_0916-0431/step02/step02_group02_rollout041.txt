import scipy.optimize as optimize

def run_packing():

    N = 26

    radii = np.zeros(N)

    # Define radii such that they are initially set to a small value for each circle
    # and a more regular grid of centers

    # 2nd layer: 10 circles
    # 3rd layer: 10 circles (total 6+10+10=26)

    # Define positions to be spaced more carefully in concentric layers
    angles = np.linspace(0, 2 * np.pi, N, endpoint=False)
    r_values = np.linspace(0.15, 0.3, N)

        angle = angles[i]
        r = r_values[i]
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers[i] = [x, y]
        radii[i] = 0.15  # Initial estimate for radii

    # Define objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative total radius to maximize

    # Define overlap constraints

        constraints = []
        for i in range(N):

                constraints.append(dist - (x[i] + x[j]))

            y_cent = centers[i][1]
            constraints.append(x_cent - x_i)
