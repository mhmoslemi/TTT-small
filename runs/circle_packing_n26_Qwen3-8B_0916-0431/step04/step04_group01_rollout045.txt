def run_packing():

    spacing = 0.23

    # Generate hexagonal grid centers

    radii = np.zeros(n)
    idx = 0

                x = j * spacing

                x = (j - 0.5) * spacing

            idx += 1

    radii = radii * scale

    def objective(radii):
        return -np.sum(radii)  # Minimize to maximize

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(radii):
