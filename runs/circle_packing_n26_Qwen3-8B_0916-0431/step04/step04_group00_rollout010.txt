import numpy as np
import scipy.optimize as optimize

def run_packing():
    n = 26

    # Initial positions: Hexagonal packing approach
    # We divide the unit square into a near-hexagonal grid with 6 rows (3 circles per row) and 5 columns
    # This allows for denser and more efficient circle positioning

    # Define a hexagonal grid
    grid = []
    for row in range(3):  # 3 rows of circles
        for col in range(5):  # 5 columns of circles
            # Offset alternate rows to form a hexagonal grid
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            grid.append([x, y])
    
    # Compact the grid to fit into [0,1]^2
    scaling = 0.8
    grid = np.array(grid) * scaling

    # Shuffle to allow for random growth opportunities
    np.random.shuffle(grid)

    # Assign weights based on proximity to boundaries
    radii = np.zeros(n)
    for i in range(n):
        # Distance from boundaries
        dist_left = grid[i, 0]
        dist_right = 1.0 - grid[i, 0]
        dist_top = grid[i, 1]
        dist_bottom = 1.0 - grid[i, 1]
        max_dist = max(dist_left, dist_right, dist_top, dist_bottom)
        radii[i] = max_dist * 0.5

    centers = grid

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(radii):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Boundary constraint: circles must be inside the unit square
    def boundary_constraints(radii):
        constraints = []
        for i in range(n):
            # Left boundary
            constraints.append(centers[i, 0] - radii[i])
            # Right boundary
            constraints.append(1.0 - centers[i, 0] - radii[i])
            # Bottom boundary
            constraints.append(centers[i, 1] - radii[i])
            # Top boundary
            constraints.append(1.0 - centers[i, 1] - radii[i])
        return constraints

    # Bounds: radii cannot be negative
    bounds = [(0, 1.0) for _ in range(n)]

    # Initial guess
    x0 = radii.copy()

    # Set up the optimization problem
    cons = []
    cons.append({'type': 'ineq', 'fun': non_overlap_constraints})
    cons.append({'type': 'ineq', 'fun': boundary_constraints})

    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-8,
        options={'maxiter': 300, 'eps': 1e-8}
    )

    radii_opt = result.x
    centers_opt = centers.copy()

    return (centers_opt, radii_opt, np.sum(radii_opt))
