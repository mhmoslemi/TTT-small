def run_packing():

            centers[i] = [0.5, 0.1]   # Top-middle
        elif i - 4 == 2:
            centers[i] = [0.75, 0.1]  # Top-right
        radii[i] = 0.1

    for i in range(8, 12):
        if i - 8 == 0:
            centers[i] = [0.25, 0.9]  # Bottom-left
        elif i - 8 == 1:
            centers[i] = [0.5, 0.9]   # Bottom-middle
        elif i - 8 == 2:
            centers[i] = [0.75, 0.9]  # Bottom-right
        radii[i] = 0.1

        col = (i - 12) % 5

        centers[i] = [x, y]
        radii[i] = 0.1  # Initial estimate

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(radii):
        constraints = []

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))

    def boundary_constraints(radii):
