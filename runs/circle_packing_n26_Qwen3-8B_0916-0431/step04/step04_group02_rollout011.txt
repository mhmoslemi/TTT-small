import scipy.optimize as optimize

def run_packing():

    spacing = 0.2

    # This reduces overlap and allows for better radius expansion
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    counter = 0

    row_count = 5
    col_offset = 0.5 * spacing  # To better align with square grid

            y = i * spacing + spacing / 2

                x += spacing / 2

                radii[counter] = 0.05
                counter += 1
            else:
                # Fill remaining positions
                centers[counter] = [x, y]
                radii[counter] = spacing / 2
                counter += 1

        return -np.sum(radii)

        return constraints
