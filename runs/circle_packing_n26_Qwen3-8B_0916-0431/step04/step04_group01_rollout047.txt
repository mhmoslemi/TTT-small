            y = row_heights[1] / 2

            y = row_heights[3] / 2

            y = row_heights[4] / 2

            x = col_spacing + col_spacing

            x = col_spacing + col_spacing + col_spacing / 2

        return -np.sum(radii)  # Minimize negative total to maximize
