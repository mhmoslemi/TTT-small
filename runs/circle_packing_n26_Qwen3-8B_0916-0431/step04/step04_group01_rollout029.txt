    # This is a more adaptive hexagonal-like pattern

    # Distribute them across the square to allow for more optimal spacing

    row_count = 5

                x = j * col_spacing + 0.1

    def objective(radii):

    def non_overlap_constraints(radii):

                constraints.append(dist - (radii[i] + radii[j]))
        return constraints
