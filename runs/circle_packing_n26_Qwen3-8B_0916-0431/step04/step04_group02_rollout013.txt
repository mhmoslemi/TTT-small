import scipy.optimize as optimize

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    rows = 5
    cols = 5
    spacing = 0.2  # initial hub spacing

    # Define a better hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2 + (i % 2) * (spacing / 2)
            centers[i * cols + j] = [x, y]

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Non-overlap constraint function: enforce non-overlapping between circles
    def non_overlap_constraints(radii):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Boundary constraint function: enforce circles within [0, 1]^2
    def boundary_constraints(radii):
        constraints = []
