import scipy.optimize as optimize

def run_packing():

    # Hexagonal grid with two rows of 13 circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Extend spacing for better packing
    spacing = 0.22
    row_height = spacing * np.sqrt(3) / 2  # Height for hexagonal packing

    # First row of 13 circles
    for i in range(13):
        centers[i] = [i * spacing + spacing / 2, spacing / 2]
        radii[i] = spacing / 2

    # Second row, offset by spacing/2 in x and row_height in y
    for i in range(13):
        centers[13 + i] = [i * spacing + spacing / 2 + spacing / 2, spacing / 2 + row_height]
        radii[13 + i] = spacing / 2

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative total to maximize

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(radii):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Boundary constraint: circle within [0,1] x [0,1]
    def boundary_constraints(radii):
        constraints = []
        for i in range(n):
            # Left boundary
            constraints.append(centers[i, 0] - radii[i])
            # Right boundary
            constraints.append(1.0 - centers[i, 0] - radii[i])
            # Bottom boundary
            constraints.append(centers[i, 1] - radii[i])
            # Top boundary
            constraints.append(1.0 - centers[i, 1] - radii[i])
        return constraints

    # Bounds: radii cannot be negative
    bounds = [(0, 1.0) for _ in range(n)]

    x0 = radii.copy()

    # Use a more robust optimization method with higher iterations
    result = optimize.minimize(

        x0=x0,

        constraints=cons,
