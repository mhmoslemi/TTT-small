        dist = np.sqrt(x*x + y*y)

            # Place in the corner
            pass  # Already in the corner

            # Close to corner, keep as-is
            adjusted_positions[i] = [x, y]

            adjusted_positions[i] = [x + (1 - x) * factor, y + (1 - y) * factor]
