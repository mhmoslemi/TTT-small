import numpy as np
import scipy.optimize as optimize

def run_packing():
    n = 26

    # Start with a hexagonal-like arrangement
    # Define a hexagonal packing with rows of circles
    # We aim for a structure with 5 rows: 6, 5, 5, 5, 5
    # Columns vary to maximize packing

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Define a hexagonal packing
    num_rows = 5
    num_cols = [6, 5, 5, 5, 5]
    spacing = 0.2

    # Define accurate, staggered hexagonal placement
    for i in range(num_rows):
        for j in range(num_cols[i]):
            row = i
            col = j
            if i % 2 == 0:
                # Even row: offset left
                x = col * spacing + spacing / 2
                y = row * spacing + spacing / 2
            else:
                # Odd row: offset right
                x = (col + 0.5) * spacing + spacing / 2
                y = row * spacing + spacing / 2

            idx = i * sum(num_cols[:i]) + j
            centers[idx] = [x, y]
            radii[idx] = spacing / 2

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint to ensure circles do not overlap
    def non_overlap_constraints(radii):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

    # Ensure circles are within [0,1]^2
    def boundary_constraints(radii):
        constraints = []
        for i in range(n):
            # Left boundary
            constraints.append(centers[i, 0] - radii[i])
            # Right boundary
            constraints.append(1.0 - centers[i, 0] - radii[i])
            # Bottom boundary
            constraints.append(centers[i, 1] - radii[i])
            # Top boundary
            constraints.append(1.0 - centers[i, 1] - radii[i])
        return constraints

    # Bounds: radii must be non-negative
    bounds = [(0, 1.0) for _ in range(n)]

    # Initial guess
    x0 = radii.copy()

    # Set up the optimization problem
    cons = []
    cons.append({'type': 'ineq', 'fun': non_overlap_constraints})
    cons.append({'type': 'ineq', 'fun': boundary_constraints})

    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-8,
        options={'maxiter': 2000, 'eps': 1e-8}
    )

    radii_opt = result.x

    return (centers_opt, radii_opt, np.sum(radii_opt))
