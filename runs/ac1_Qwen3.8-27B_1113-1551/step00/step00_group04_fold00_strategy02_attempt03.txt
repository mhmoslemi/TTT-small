def evaluate_helper(params, n_val=3809):
    """
    Helper function to compute the score without the full evaluator's checks.
    Does the following:
      computes sum_a and max_b from build_sequence and returns np.inf if invalid.

    sequence = build_sequence(params, n_val)
    seq_sum = np.sum(sequence)
    convolve_result = np.convolve(sequence, sequence)
    max_value = np.max(convolve_result)
    if seq_sum < 0.01 or not np.all(np.isfinite(sequence)):
        return np.inf
        
    return 2 * n_val * max_value / (seq_sum ** 2)
