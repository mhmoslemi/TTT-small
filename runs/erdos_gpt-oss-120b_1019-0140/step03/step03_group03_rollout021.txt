    target_sum = n / 2.0

    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        grad = compute_grad_h_max(h)
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad**2)
        lr = step0 * (np.sqrt(1 - beta2**t) / (1 - beta1**t))
        h_new = h - lr * m / (np.sqrt(v) + eps)

    return best_h, best_score

def upsample(arr, new_n):
    if arr.shape[0]==new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, arr.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)   # same using interpolation

    h = np.zeros(n, dtype=float)

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_raw(h):
    """Full correlation of h with (1-h) without scaling, length 2*len(h)-1."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_raw(h)
    return corr.max() * (2.0 / n)


def adam_projected(h0, n_iter, step0=0.5, beta1=0.9, beta2=0.999, eps=1e-8, eps_shift=1e-12):
    """Adam optimizer with projection onto box & sum constraint for the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, n_iter + 1):
        corr = compute_corr_raw(h)
        max_corr = corr.max()
        # identify shifts attaining the max
        idxs = np.where(np.abs(corr - max_corr) <= eps_shift)[0]
        shifts = idxs - (n - 1)

        grad = np.zeros_like(h)
        for s in shifts:
            s_abs = abs(s)
            if s_abs < n:
                grad[:n - s_abs] += (1.0 - h[s_abs:])
                grad[s_abs:] -= h[:n - s_abs]
        if len(shifts):
            grad /= len(shifts)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        lr = step0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
        h_new = h - lr * m / (np.sqrt(v) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate new point
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def upsample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_perturb(h, rng, scale=0.05):
    v = h + scale * rng.standard_normal(h.shape)
    v = np.clip(v, 0.0, 1.0)
    return project_onto_box_sum(v, h.size / 2.0)


def slsqp_optimize(h0, max_iter=300):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_raw(h) * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)

    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

        best_score, best_h = sc_med, h_med.copy()

    if time_left() > 15.0:
        iters = int(min(20000, time_left() * 1000))
        h_ad, sc_ad = adam_projected(best_h, n_iter=iters, step0=0.6)
        if sc_ad < best_score:
            best_score, best_h = sc_ad, h_ad.copy()

    # ---------- Phase 3: large scale (n=512) ----------
    n_large = 512
    h_large = upsample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    sc_large = compute_c5(h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, h_large.copy()

    if time_left() > 20.0:
        iters = int(min(50000, time_left() * 800))
        h_ad, sc_ad = adam_projected(best_h, n_iter=iters, step0=0.5)
        if sc_ad < best_score:
            best_score, best_h = sc_ad, h_ad.copy()

    # occasional random perturbations on large scale
    if time_left() > 10.0:
        for _ in range(5):
            h_pert = random_perturb(best_h, rng, scale=0.03)
            iters = int(min(5000, time_left() * 800))
            h_ad, sc_ad = adam_projected(h_pert, n_iter=iters, step0=0.5)
            if sc_ad < best_score:
                best_score, best_h = sc_ad, h_ad.copy()

    # ---------- Phase 4: extra-large (n=1024) ----------
    if time_left() > 30.0:
        n_xl = 1024
        h_xl = upsample(best_h, n_xl)
        h_xl = project_onto_box_sum(h_xl, n_xl / 2.0)
        sc_xl = compute_c5(h_xl)
        if sc_xl < best_score:
            best_score, best_h = sc_xl, h_xl.copy()
        if time_left() > 10.0:
            iters = int(min(20000, time_left() * 400))
            h_ad, sc_ad = adam_projected(best_h, n_iter=iters, step0=0.4)
            if sc_ad < best_score:
                best_score, best_h = sc_ad, h_ad.copy()

    # ---------- Phase 5: binary simulated annealing ----------
    if time_left() > 5.0:
        n_cur = best_h.shape[0]
        idx = np.argsort(best_h)[-n_cur // 2:]
        h_bin = np.zeros(n_cur, dtype=float)
        h_bin[idx] = 1.0

        max_iter_ann = int(time_left() * 500)
        temp_start = 0.02
        temp_end = 1e-5
        cur_h = h_bin.copy()
        cur_score = compute_c5(cur_h)
        best_h_bin = cur_h.copy()
        best_score_bin = cur_score
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)

        for it in range(max_iter_ann):
            if ones.size == 0 or zeros.size == 0:
                break
            t = temp_start * ((temp_end / temp_start) ** (it / max_iter_ann))
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i], cur_h[j] = 0.0, 1.0
            new_score = compute_c5(cur_h)
            delta = new_score - cur_score
            if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
                cur_score = new_score
                if new_score < best_score_bin:
                    best_score_bin = new_score
                    best_h_bin = cur_h.copy()
                ones = np.flatnonzero(cur_h > 0.5)
                zeros = np.flatnonzero(cur_h < 0.5)
            else:
                cur_h[i], cur_h[j] = 1.0, 0.0

        if best_score_bin < best_score:
            best_score, best_h = best_score_bin, best_h_bin.copy()

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
