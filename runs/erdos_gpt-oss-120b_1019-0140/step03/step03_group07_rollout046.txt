def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # using numpy float, corr is array length 2n-1.

### compute_overlap_vector

def compute_overlap_vector(h):
    n = h.shape[0]
    M = np.sum(h)
    ip = np.correlate(h, h, mode='full')
    overlap = M - ip
    # return overlap (numpy array)
    return overlap

def continuous_refine_slsqp(h0, time_limit, rng):
    # h0 binary but we treat as float initial.
    n = h0.shape[0]
    M = n // 2
    dx = 2.0 / n
    # initial t bound from compute_c5
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0.astype(float), [t0]])
    # define constraints
    # Sum equality
    def sum_eq(x):
        return np.sum(x[:n]) - M
    constraints = [{'type': 'eq', 'fun': sum_eq}]
    # Overlap constraints for each shift k != 0
    # define closure for each k
    # careful about capturing k each iteration; define function factory
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_fun(kk):
            def overlap_con(x, kk=kk):
                h = x[:n]
                # compute overlap for shift kk
                if kk >= 0:
                    overl = np.dot(h[:n-kk], 1.0 - h[kk:])
                else:
                    k2 = -kk
                    overl = np.dot(h[k2:], 1.0 - h[:n-k2])
                return x[-1] - overl * dx
            return overlap_con
        constraints.append({'type': 'ineq', 'fun': make_fun(k)})
    # bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # options
    # call minimize
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
    else:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum exactly
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt
