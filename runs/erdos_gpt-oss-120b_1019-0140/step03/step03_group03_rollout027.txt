def project_onto_box_sum(v, s, lo=0.0, hi=1.0, tol=1e-9, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda such that sum(np.clip(v - lam, lo, hi)) = s
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam_mid, lo, hi)
        cur = proj.sum()
        if np.abs(cur - s) <= tol:
            return proj
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

Potential issue: lam_low/lam_high set might not bracket solution if clipping saturates. But binary search works as long as f(lam) monotonic decreasing; we can get lam_low and lam_high as large extremes: lam_low = -1e6, lam_high=1e6 to guarantee.

def get_active_shifts(h, eps=1e-12):

    L = 2 * n

def gradient_descent_adam(h0, time_left, max_iters=100000, alpha=0.1, beta1=0.9, beta2=0.999, eps=1e-12):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # compute active shifts
        shifts = get_active_shifts(h, eps=1e-9)
        # compute gradient as average over shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s = -shift
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[s:] = h[:n-s]
                term2[:n-s] = h[s:]
            # gradient component: (1 - term1 - term2)
            grad += (1.0 - term1 - term2)
        grad /= len(shifts) if len(shifts) > 0 else 1.0

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        # projection onto box sum
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute score
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

        if remaining <= 0.0:

def gradient_descent_momentum_fast(h0, time_left_func, max_iters=10000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left_func() < 0.005:
            break
        # compute correlation via FFT to identify max
        n = h.size
        L = 2 * n
        H = np.fft.rfft(h, n=L)
        G = np.fft.rfft(1.0 - h, n=L)
        corr = np.fft.irfft(H * np.conj(G), n=L)
        max_corr = np.max(corr)
        # identify active shifts
        # Use tolerance relative
        eps = 1e-9
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient average over shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s = -shift
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[s:] = h[:n-s]
                term2[:n-s] = h[s:]
            grad += (1.0 - term1 - term2)
        if shifts.size > 0:
            grad /= shifts.size
        # update momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto feasibility
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute score
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = -1e6
    lam_high = 1e6
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam_mid, lo, hi)
        cur = proj.sum()
        if abs(cur - s) <= tol:
            return proj
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def get_active_shifts(h, rel_eps=1e-9):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    max_corr = np.max(corr)
    eps = max_corr * rel_eps + 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    return idxes - (n - 1)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1: h shifted left by shift
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[:shift] = 0.0
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[:s] = 0.0
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_momentum_fast(h0, time_left_func, max_iters=20000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        if time_left_func() < 0.005:
            break
        # compute active shifts using FFT
        shifts = get_active_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low, lam_high = -1e6, 1e6
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        cur = proj.sum()
        if abs(cur - s) <= tol:
            return proj
        if cur > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return np.max(corr) * (2.0 / n)


def get_active_shifts(h, rel_eps=1e-9):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    max_corr = np.max(corr)
    eps = max_corr * rel_eps + 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    return idxes - (n - 1)


def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[:shift] = 0.0
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[:s] = 0.0
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def gradient_descent_momentum_fast(h0, time_left_func, max_iters=20000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_score = h.copy(), compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left_func() < 0.005:
            break
        shifts = get_active_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def anneal_continuous(h0, time_left_func, rng):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h, best_score = h.copy(), cur_score
    t_start, t_end = 0.1, 1e-5
    total_time = time_left_func()
    if total_time <= 0.0:
        return best_h, best_score
    while True:
        remaining = time_left_func()
        if remaining <= 0.0:
            break
        progress = (total_time - remaining) / total_time
        t = t_start * ((t_end / t_start) ** progress)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        direction = rng.random() < 0.5  # True: move from j to i
        if direction:
            max_inc = min(1.0 - h[i], h[j])
            if max_inc <= 0:
                continue
            delta = rng.random() * max_inc
            h[i] += delta
            h[j] -= delta
        else:
            max_inc = min(1.0 - h[j], h[i])
            if max_inc <= 0:
                continue
            delta = rng.random() * max_inc
            h[j] += delta
            h[i] -= delta
        new_score = compute_c5_fft(h)
        delta_score = new_score - cur_score
        accept = delta_score < 0 or rng.random() < np.exp(-delta_score / (t + 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            if direction:
                h[i] -= delta
                h[j] += delta
            else:
                h[j] -= delta
                h[i] += delta
    return best_h, best_score
