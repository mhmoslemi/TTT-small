def adam_optimize(h0, time_left_func, max_iter=200000, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8, tol=1e-12):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while t < max_iter and time_left_func() > 0:
        t += 1
        # compute correlation and shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # can also consider top shifts for smoothing: take all shift where corr >= max_corr - delta, or take top K
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[shift:] += 1.0 - h[:n - shift]
                grad[:n - shift] -= h[shift:]
            else:
                s_abs = -shift
                grad[:n - s_abs] += 1.0 - h[s_abs:]
                grad[s_abs:] -= h[:n - s_abs]
        if len(shifts) > 0:
            grad /= len(shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step size schedule: lr / sqrt(t) maybe:
        step = lr / np.sqrt(t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto constraints
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure interval frames s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _grad_for_shift(h, shift):
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        grad[s:] += 1.0 - h[:n - s]
        grad[:n - s] -= h[s:]
    else:
        s = -shift
        grad[:n - s] += 1.0 - h[s:]
        grad[s:] -= h[:n - s]
    return grad

def adam_optimize(h0, time_left_func, max_iter=200000, lr=5.0, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while t < max_iter and time_left_func() > 0.0:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for sh in shifts:
            grad += _grad_for_shift(h, sh)
        if shifts.size > 0:
            grad /= shifts.size
        grad *= (2.0 / n)  # scale by dx
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / np.sqrt(t)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
