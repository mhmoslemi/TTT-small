def projected_gradient_descent(h, start_time, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    alpha = 0.5
    min_alpha = 1e-6

    while time.time() < deadline - 0.01:
        # compute corr and worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = int(np.argmax(corr))
        k = idx - (n-1)
        # compute gradient g
        g = np.zeros_like(cur_h)
        if k >= 0:
            if n - k > 0:
                g[:n - k] += (1.0 - cur_h[k:])
                if k > 0:
                    g[k:] -= cur_h[:n - k]
        else:
            kk = -k
            if n - kk > 0:
                g[kk:] += (1.0 - cur_h[:n - kk])
                g[:n - kk] -= cur_h[kk:]
        # step
        step = alpha * g
        new_h = cur_h - step
        # clip
        new_h = np.clip(new_h, 0.0, 1.0)
        # enforce sum exactly
        new_h = enforce_sum(new_h, target, rng)
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                if best_val <= 0.3808:
                    break
            # maybe increase alpha a bit
            alpha = min(alpha * 1.05, 1.0)
        else:
            # reject, reduce alpha
            alpha = max(alpha * 0.5, min_alpha)
            # maybe try a random move if alpha too small?
            if alpha <= min_alpha:
                # perform a random pair transfer to escape
                src = np.where(cur_h > 1e-12)[0]

                        if cur_val < best_val:
                            best_val = cur_val
                            best_h = cur_h.copy()
        # loop
    return best_h, best_val

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    diff = target - total
    # Distribute diff over entries that can be adjusted.
    # For diff >0, add to entries <1; for diff<0, subtract from entries>0.
    # Use iterative random picks.
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            candidates = np.where(v < 1.0 - 1e-12)[0]
            if len(candidates) == 0:
                break
            idx = rng.choice(candidates)
            add = min(diff, 1.0 - v[idx])
            v[idx] += add
            diff -= add
        else:
            candidates = np.where(v > 1e-12)[0]
            if len(candidates) == 0:
                break
            idx = rng.choice(candidates)
            sub = min(-diff, v[idx])
            v[idx] -= sub
            diff += sub
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    diff = target - total
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if not cand.size:
                break
            idx = rng.choice(cand)
            add = min(diff, 1.0 - v[idx])
            v[idx] += add
            diff -= add
        else:
            cand = np.where(v > 1e-12)[0]
            if not cand.size:
                break
            idx = rng.choice(cand)
            sub = min(-diff, v[idx])
            v[idx] -= sub
            diff += sub
    return np.clip(v, 0.0, 1.0)

            x_old = np.linspace(0, 2, m0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = random_continuous_vector(n, target, rng)

    best_h, best_val = projected_gradient_descent(h0, start, deadline, rng)

    if time.time() < deadline - 0.5:
        best_h, best_val = anneal_targeted(best_h, start, deadline, rng)

    if time.time() < deadline - 0.2:
        best_h, best_val = final_cleanup(best_h, best_val, deadline, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
