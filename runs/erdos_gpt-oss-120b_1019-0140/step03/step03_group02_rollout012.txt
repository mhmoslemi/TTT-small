def binary_local_optimize(h, time_left, target=None):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    while time_left() > 0.01:
        grad = compute_gradients_for_shifts(best_h, 
                                            np.where(np.correlate(best_h, 1-best_h, mode='full') == np.max(np.correlate(best_h, 1-best_h, mode='full')))[0] - (n-1))
        # gather candidates:
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        # compute scores for zeros and ones
        zero_vals = grad[zeros]
        one_vals = grad[ones]
        # we want increase at zeros (positive) and decrease at ones (negative)
        # sort descending for zeros, ascending for ones.
        zero_idx = zeros[np.argsort(-zero_vals)]  # largest positive first
        one_idx = ones[np.argsort(one_vals)]  # most negative first (since we want negative)
        improved = False
        for i in zero_idx:
            if grad[i] <= 0:
                # no positive gradient left
                break
            for j in one_idx:
                if grad[j] >= 0:
                    # no negative gradient left
                    break
                # propose swap
                h_new = best_h.copy()
                h_new[i] = 1.0
                h_new[j] = 0.0
                score_new = compute_c5(h_new)
                if score_new < best_score - 1e-12:
                    best_h, best_score = h_new, score_new
                    improved = True
                    break  # break inner loop to recompute grads
            if improved:
                break
        if not improved:
            break
        if target is not None and best_score <= target:
            break
    return best_h, best_score
```

    n = h.size

def binary_local_optimize(h, time_left, target=None):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 0.01:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0] - (n-1)
        grad = compute_gradients_for_shifts(best_h, idxes)
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        # find most negative gradient in zeros and most positive in ones
        i = zeros[np.argmin(grad[zeros])]
        j = ones[np.argmax(grad[ones])]
        # Check if swap could improve
        if grad[i] >= 0 or grad[j] <= 0:
            break
        # propose swap
        h_new = best_h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-12:
            best_h, best_score = h_new, new_score
            if target is not None and best_score <= target:
                break
            continue
        else:
            # no improvement despite direction, break to avoid infinite loop
            break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def upsample(arr, new_n):
    """Linear interpolation from arr (domain [0,2]) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shifts run from -(n-1) to (n-1)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the (average) max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, n_iter, step0, alpha_start, alpha_end, time_left,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
    return best_h, best_score

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):
    """Projected gradient descent with momentum on the true max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, idxes)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_local_optimize(h, time_left, target=None):
    """Greedy swap of a zero with a one using max‑correlation gradient."""
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 0.01:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        shifts = np.where(corr >= maxc - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(best_h, shifts)
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = zeros[np.argmin(grad[zeros])]   # most negative gradient among zeros
        j = ones[np.argmax(grad[ones])]    # most positive gradient among ones
        if grad[i] >= 0 or grad[j] <= 0:
            break  # no beneficial swap direction
        h_new = best_h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-12:
            best_h, best_score = h_new, new_score
            if target is not None and best_score <= target:
                break
            continue
        else:
            break
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style swap search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(300000, max(1, time_left() * 2500)))
    T0 = 0.006
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # select a zero and a one uniformly
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
