def linear_corr(a, b):
    nfft = 1 << ((len(a) + len(b) - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    corr = np.fft.irfft(fa * np.conj(fb), n=nfft)
    return corr[:len(a) + len(b) - 1]

import time
import numpy as np

# ----- projection onto box and sum constraint -----
def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = target}."""
    n = v.size
    if target <= lo * n:
        return np.full_like(v, lo)
    if target >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < target:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - target) < 1e-12:
            break
        if fm > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

# ----- linear correlation via FFT (full mode) -----
def linear_corr(a, b):
    """Linear correlation a ⋆ b (full) using FFT."""
    n = len(a) + len(b) - 1
    nfft = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, nfft)
    fb = np.fft.rfft(b, nfft)
    corr = np.fft.irfft(fa * np.conj(fb), nfft)[:n]
    return corr

def compute_corr_fft(h):
    """Correlation of h with (1‑h) of length 2·n‑1."""
    return linear_corr(h, 1.0 - h)

def compute_c5(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

# ----- utilities -----
def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

# ----- momentum gradient descent (subgradient) -----
def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left, step0=0.5,
                                        beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

# ----- binary swap annealing -----
def binary_swap_anneal(h0, rng, time_left,

        else:
            h[i] = 1.0

    return best_h, best_score

# ----- smoothed (log‑sum‑exp) gradient descent -----
def smoothed_gradient_descent(h0, time_left, max_iter=3000,
                             eps_start=0.03, eps_end=0.001):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        eps = eps_start * ((eps_end / eps_start) ** (it / max_iter))
        corr = compute_corr_fft(h)               # length 2n‑1
        max_corr = np.max(corr)
        w = np.exp((corr - max_corr) / eps)      # stable weights
        w /= w.sum()
        v = 1.0 - h
        # g1 = correlation(w, v) ; g2 = correlation(w, h)
        g1_full = linear_corr(w, v)               # length 3n‑2
        g2_full = linear_corr(w, h)
        offset = n - 1
        g1 = g1_full[offset:offset + n]
        g2 = g2_full[offset:offset + n]
        grad = g1 - g2
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
            if sc <= 0.3808:
                break
        h = h_new
    return best_h, best_sc
