def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # parameters
    n_small = 64
    n_mid = 128
    n_large = 256

    # initial candidates list
    candidates = []

    # if initial_h_values provided
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # Downsample if needed
        if init_full.size != n_small:
            init_small = downsample(init_full, n_small)
        else:
            init_small = init_full.copy()
        init_small = project_onto_box_sum(init_small, n_small/2.0)
        candidates.append(init_small)

    # add Thue-Morse
    candidates.append(thue_morse(n_small).astype(float))

    # random binary balanced (2)
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))

    # random continuous (2)
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small/2.0))

    # best trackers
    best_h = None
    best_score = np.inf

    # Phase: small grid smooth grad descent
    for h0 in candidates:
        if time_left() < 5.0:
            break
        # random hyperparam variations for robustness
        # Use default values or vary
        step0 = rng.uniform(0.4, 0.8)
        alpha_start = rng.uniform(0.5, 2.0)
        alpha_end = rng.uniform(5.0, 20.0)
        beta = 0.9
        n_iter = int(min(5000, max(2000, time_left()*200)))
        h_opt, sc_opt = smooth_grad_descent(h0, n_iter=n_iter, step0=step0,
                                            alpha_start=alpha_start,
                                            alpha_end=alpha_end, beta=beta,
                                            time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Ensure we have a best_h
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small/2.0)
        best_score = compute_c5(best_h)

    # Phase: upsample to mid resolution and momentum descent
    if time_left() > 2.0:
        h_mid = upsample(best_h, n_mid)
        h_mid = project_onto_box_sum(h_mid, n_mid/2.0)
        # momentum descent
        n_iter_mid = int(min(20000, max(5000, time_left()*800)))
        h_mid_opt, sc_mid_opt = momentum_grad_descent(h_mid, n_iter=n_iter_mid,
                                                      step0=0.5, beta=0.9,
                                                      time_left=time_left)
        if sc_mid_opt < best_score:
            best_score, best_h = sc_mid_opt, h_mid_opt.copy()

    # Phase: upsample to large resolution and momentum descent
    if time_left() > 2.0:
        h_large = upsample(best_h, n_large)
        h_large = project_onto_box_sum(h_large, n_large/2.0)
        n_iter_large = int(min(30000, max(8000, time_left()*1000)))
        h_large_opt, sc_large_opt = momentum_grad_descent(h_large,
                                                          n_iter=n_iter_large,
                                                          step0=0.5, beta=0.9,
                                                          time_left=time_left)
        if sc_large_opt < best_score:
            best_score, best_h = sc_large_opt, h_large_opt.copy()

    # Phase: binary hill climb SA at current best size (maybe large)
    if time_left() > 3.0:
        n_curr = best_h.shape[0]
        # convert to binary by ranking
        order = np.argsort(best_h)
        binary = np.zeros(n_curr, dtype=float)
        binary[order[n_curr//2:]] = 1.0
        # ensure exactly half ones; if ties, random selection
        # SA
        # target bound: original goal 0.3808; may tighten dynamically as best_score
        target = min(0.3808, best_score)
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=target)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

        # small continuous refinement after binary search
        if time_left() > 2.0:
            # momentum descent for some iters
            n_iter_refine = int(min(8000, max(2000, time_left()*500)))
            h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=n_iter_refine,
                                                  step0=0.3, beta=0.9,
                                                  time_left=time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    # final projection and compute final_score
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
