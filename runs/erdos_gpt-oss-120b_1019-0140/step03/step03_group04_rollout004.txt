def gradient_for_shift(h, s):
    n = h.size
    term1 = np.ones_like(h)
    if s >= 0:
        term1[:n-s] -= h[s:]
    else:
        s_neg = -s
        term1[s_neg:] -= h[:n-s_neg]
    term2 = np.zeros_like(h)
    if s >= 0:
        term2[s:] = h[:n-s]
    else:
        s_neg = -s
        term2[:n-s_neg] = h[s_neg:]
    return term1 - term2

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the interval brackets the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue {0,1} sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

    n = h.size
    a = h
    b = 1.0 - h
    revb = b[::-1]

    fb = np.fft.rfft(np.concatenate([revb, np.zeros(n, dtype=np.float64)]))
    conv = np.fft.irfft(fa * fb, n=N)
    return conv[:2 * n - 1]

    """Sub‑gradient of max‑correlation term for given shift list."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:

            term2[:n - sneg] = h[sneg:]
        grad += (term1 - term2)

        grad /= shifts.size
    return grad

    candidates_small = []
    if "initial_h_values" in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), target_small)
        candidates_small.append(cand)
    candidates_small.append(thue_morse_binary(n_small))
    for _ in range(4):
        candidates_small.append(random_binary_balanced(n_small, rng))
    # Legendre sequence of length 256 (p=257 is prime)
    leg = legendre_sequence(257)  # length 256
    candidates_small.append(project_onto_box_sum(leg, target_small))

    best_h_small = None
    best_score_small = np.inf

    for cand in candidates_small:
        if time_left() < 6.0:
            break
        t_end = start + min(time_left() - 0.5, 15.0)
        h_opt, sc_opt = gd_momentum(cand, t_end, beta=0.9, step0=1.0)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()

    if best_h_small is None:
        best_h_small = random_binary_balanced(n_small, rng)
        best_score_small = compute_c5_and_corr_fft(best_h_small)[0]

    # ---------- up‑sample ----------
    n_big = 1024
    target_big = n_big / 2.0
    h_big = resample(best_h_small, n_big)
    h_big = project_onto_box_sum(h_big, target_big)

    best_score_big = compute_c5_and_corr_fft(h_big)[0]
    best_h_big = h_big.copy()

    # ---------- large‑scale refinement ----------
    if time_left() > 5.0:
        t_end = start + budget_s - 0.4
        h_opt, sc_opt = gd_momentum(best_h_big, t_end, beta=0.9, step0=0.6)
        if sc_opt < best_score_big:
            best_score_big = sc_opt
            best_h_big = h_opt.copy()

    # ---------- optional binary local search ----------
    if time_left() > 3.0:
        # binarize by taking the largest entries
        bin_h = np.zeros_like(best_h_big)
        idx = np.argpartition(-best_h_big, int(target_big) - 1)[:int(target_big)]
        bin_h[idx] = 1.0
        c_bin = compute_c5_and_corr_fft(bin_h)[0]
        if c_bin < best_score_big:
            best_h_big = bin_h.copy()
            best_score_big = c_bin
        # simple swap hill‑climbing
        max_swaps = min(2000, int(time_left() * 5000))
        for _ in range(max_swaps):
            ones = np.where(best_h_big > 0.5)[0]
            zeros = np.where(best_h_big < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            best_h_big[i] = 0.0
            best_h_big[j] = 1.0
            c_new = compute_c5_and_corr_fft(best_h_big)[0]
            if c_new < best_score_big:
                best_score_big = c_new
            else:
                # revert
                best_h_big[i] = 1.0
                best_h_big[j] = 0.0
            if time_left() < 0.5:
                break

    # final safety projection & evaluation
    best_h_big = project_onto_box_sum(best_h_big, target_big)
    best_score_big = compute_c5_and_corr_fft(best_h_big)[0]

    return best_h_big, best_score_big, n_big
