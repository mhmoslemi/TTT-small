def transfer_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, 2, replace=False)
    # move from j to i
    max_transfer = min(1.0 - h[i], h[j])
    if max_transfer <= 1e-12:
        return None  # no move
    delta = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, delta

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_sequence(p: int) -> np.ndarray:

    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq
