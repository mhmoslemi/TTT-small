def random_exchange_search(h, rng, time_end, max_no_improve=2000):
    best_h = h.copy()
    best_score = compute_c5(h)
    n = len(h)
    no_improve = 0
    while time.time() < time_end and no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # ensure we can move some mass
        delta_max = min(0.01, best_h[i], 1 - best_h[j])
        if delta_max <= 0:
            # try opposite direction
            delta_max = min(0.01, best_h[j], 1 - best_h[i])
            if delta_max <= 0:
                continue
            i, j = j, i
        delta = rng.random() * delta_max
        # perform move
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # No need to project sum because sum preserved.
        # Ensure bounds
        # Already ensured delta respects bounds.
        score = compute_c5(h_new)
        if score < best_score:
            best_h = h_new
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the interval contains a solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, step0, beta, time_end, max_iter=200000, target=0.3808):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, max_iter + 1):
        if time.time() > time_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= target:
                break
        h = h_new
    return best_h, best_score

def random_exchange_search(h, rng, time_end, max_no_improve=500):
    """Small stochastic exchange moves preserving the sum."""
    best_h = h.copy()
    best_score = compute_c5(h)
    n = len(h)
    no_improve = 0
    while time.time() < time_end and no_improve < max_no_improve:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # try moving mass from i to j
        delta_max = min(0.02, best_h[i], 1.0 - best_h[j])
        if delta_max <= 0:
            # reverse direction
            delta_max = min(0.02, best_h[j], 1.0 - best_h[i])
            if delta_max <= 0:
                continue
            i, j = j, i
        delta = rng.random() * delta_max
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h = h_new
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_score
