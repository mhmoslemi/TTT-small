def compute_c5(h):
    n = h.size
    N = 2*n - 1
    # compute convolution using FFT
    # rfft of length N
    fft_h = np.fft.rfft(h, n=N)
    fft_b = np.fft.rfft(np.flip(1.0 - h), n=N)
    conv = np.fft.irfft(fft_h * fft_b, n=N)
    # conv length N = 2n-1, matches full correlation. conv[k] corresponds to shift k-(n-1)
    max_val = conv.max()
    return max_val * (2.0 / n)
```

import time, math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def random_binary_balanced(n, rng):
    b = np.zeros(n, dtype=float)
    b[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return b

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.size
    N = 2 * n - 1
    fft_h = np.fft.rfft(h, n=N)
    fft_b = np.fft.rfft(np.flip(1.0 - h), n=N)
    conv = np.fft.irfft(fft_h * fft_b, n=N)
    return conv.max() * (2.0 / n)

def simulated_annealing(h0, time_end, rng):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    start = time.time()
    while time.time() < time_end:
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        # swap bits
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        now = time.time()
        t = max(1e-7, 0.005 * (time_end - now) / (time_end - start))
        if delta < 0 or rng.random() < math.exp(-delta / t):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_end = start + budget_s - 0.5            # safety margin
    n = 256
    best_h = None
    best_score = np.inf

    # build initial candidates
    cand_list = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = resample(init_full, n)
        cand = binarize_from_real(cand)
        cand_list.append(cand)
    cand_list.append(thue_morse_binary(n))
    cand_list.append(random_binary_balanced(n, rng))

    for cand in cand_list:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
            if best_score <= 0.3808:
                return best_h, best_score, best_h.size

    # main simulated‑annealing loop with random restarts
    while time.time() < time_end:
        remaining = time_end - time.time()
        if remaining <= 0:
            break
        chunk = min(4.0, remaining)
        h0 = random_binary_balanced(n, rng)
        h_opt, sc_opt = simulated_annealing(h0, time.time() + chunk, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

    # optional refinement at higher resolution
    if time.time() < time_end and best_score > 0.3808:
        n2 = 512
        up = resample(best_h, n2)
        up = project_onto_box_sum(up, n2 / 2.0)
        up_bin = binarize_from_real(up)
        h_opt, sc_opt = simulated_annealing(up_bin, time_end, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            n = n2

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
