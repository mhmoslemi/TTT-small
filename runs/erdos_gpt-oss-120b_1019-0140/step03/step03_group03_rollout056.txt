def continuous_anneal(h, max_iter, temp_start, temp_end, rng):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # pick two indices i,j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i: j += 1
        # propose moving delta from i to j
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        # delta magnitude random
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        delta_score = new_score - cur_score
        accept = False
        if delta_score < 0 or rng.random() < np.exp(-delta_score / (t + 1e-12)):
            accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

def gradient_descent_momentum_timebounded(h0, time_budget, step0=0.5, beta=0.9, min_iter=1000, max_iter=200000):
    start = time.time()
    h = h0.copy()
    n = h.size
    target_sum = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while True:
        it += 1
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        # compute gradient for these shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[:n - shift] = h[shift:]
                term1[n - shift:] = 0  # not needed
                term2[shift:] = h[:n - shift]
                term2[:shift] = 0
            else:
                s = -shift
                term1 = np.empty_like(h)
                term2 = np.empty_like(h)
                term1[s:] = h[:n - s]
                term1[:s] = 0
                term2[:n - s] = h[s:]
                term2[n - s:] = 0
            grad += (1 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # ensure feasible
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # time check
        if time.time() - start > time_budget:
            break
        if it >= max_iter:
            break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_momentum_timebounded(h0, time_budget, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    end = time.time() + time_budget
    it = 0
    while time.time() < end and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def continuous_anneal_timebounded(h0, time_budget, rng, temp_start=0.1, temp_end=1e-5):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    end = time.time() + time_budget
    while time.time() < end:
        # linear schedule in log scale
        elapsed = time.time() - (end - time_budget)
        t = temp_start * ((temp_end / temp_start) ** (elapsed / time_budget))
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:

        if max_delta <= 0:

        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        delta_score = new_score - cur_score

                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_momentum_timebounded(h0, time_budget, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    end = time.time() + time_budget
    it = 0
    while time.time() < end and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def continuous_anneal_timebounded(h0, time_budget, rng, temp_start=0.1, temp_end=1e-5):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    end = start + time_budget
    while time.time() < end:
        elapsed = time.time() - start
        t = temp_start * ((temp_end / temp_start) ** (elapsed / time_budget))
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        dscore = new_score - cur_score
        if dscore < 0 or rng.random() < np.exp(-dscore / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score

def anneal_binary_timebounded(h0, time_budget, rng, temp_start=0.01, temp_end=1e-5):
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    max_iter = int(time_budget * 150) + 1
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def local_restart_optimize(h_start, time_budget, rng, step0=0.8):
    end = time.time() + time_budget
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    while time.time() < end:
        sigma = 0.05
        h_pert = best_h + rng.normal(scale=sigma, size=best_h.shape)

        h_pert = project_onto_box_sum(h_pert, best_h.size / 2.0)
        remain = end - time.time()
        gd_budget = min(2.0, remain)
        if gd_budget <= 0:

        h_opt, sc_opt = gradient_descent_momentum_timebounded(h_pert, time_budget=gd_budget, step0=step0)
        if sc_opt < best_score:

            best_h = h_opt.copy()
    return best_h, best_score
