def anneal_continuous(h0, max_iter, temp_start, temp_end, rng):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    total = h.sum()
    target_sum = total  # sum constraint: should equal n/2 (but after project ensures)
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # pick two indices where we can move
        i, j = rng.choice(n, size=2, replace=False)
        # propose delta
        max_delta_i = min(1 - h[i], target_sum - (h[i] - h[j]) - (h[j] + h[i] - target_sum))? Wait we need to preserve sum: if we change h[i] by -d, and h[j] by +d, sum constant.
        # Actually propose add d to i and subtract d from j, preserving sum.
        # Choose d in range [-dmax, dmax] where dmax = min(1-h[i], h[j]).
        dmax = min(1 - h[i], h[j])  # can increase i, decrease j
        dmin = -min(h[i], 1 - h[j])  # can decrease i, increase j
        if dmax <= 0 and dmin >= 0:
            continue
        d = rng.uniform(dmin, dmax)
        if d == 0:
            continue
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        # compute score
        new_score = compute_c5(h_new)
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (t + 1e-12)):
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_continuous(h0, max_iter, temp_start, temp_end, rng):
    """Simulated annealing on continuous vectors preserving sum and bounds."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i, j = rng.choice(n, size=2, replace=False)
        # move delta from j to i (preserve sum)
        dmax = min(1.0 - h[i], h[j])
        dmin = -min(h[i], 1.0 - h[j])
        if dmax <= 0 and dmin >= 0:
            continue
        d = rng.uniform(dmin, dmax)
        if abs(d) < 1e-15:
            continue
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (t + 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP to minimize max correlation on a small scale."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def de_optimize_candidate(n, rng):
    """Differential evolution with penalty for sum constraint."""
    bounds = [(0.0, 1.0)] * n
    penalty = 1000.0

    def obj(x):
        return compute_c5(x) + penalty * (np.sum(x) - n / 2.0) ** 2

    seed_int = int(rng.integers(0, 2**32 - 1))
    res = opt.differential_evolution(obj, bounds,
                                     seed=seed_int,
                                     maxiter=35,
                                     popsize=20,
                                     polish=False,
                                     updating='deferred')
    x = project_onto_box_sum(res.x, n / 2.0)
    return x, compute_c5(x)
