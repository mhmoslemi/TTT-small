def rudin_shapiro(n):
    arr = np.empty(n, dtype=float)
    for i in range(n):
        # compute number of (i & (i>>1)) bits mod 2
        b = i & (i>>1)
        # popcount of b
        # can compute parity by bin(b).count('1')%2
        parity = bin(b).count('1') & 1
        val = -1 if parity else 1
        arr[i] = (val+1)/2.0
    return arr

def binary_sa_search(n, target, time_left, rng):
    best_global_h = None; best_global_score = inf
    while time_left() > 5:
        # start from random binary balanced
        h = random_binary_balanced(n, rng).astype(float)
        # optionally adapt initial via greedy improvement
        # compute current score
        curr_score = compute_c5(h)
        # SA
        max_iters = int(min(200000, max(1, time_left() * 3000))) # each iter moderate
        T0 = 0.02
        for it in range(1, max_iters+1):
            if time_left() < 0.01:
                break
            # pick random zero and one
            zero_idx = rng.integers(0, n)
            while h[zero_idx] != 0.0:
                zero_idx = rng.integers(0, n)
            one_idx = rng.integers(0, n)
            while h[one_idx] != 1.0:
                one_idx = rng.integers(0, n)
            # swap
            h_new = h.copy()
            h_new[zero_idx] = 1.0
            h_new[one_idx] = 0.0
            new_score = compute_c5(h_new)
            delta = new_score - curr_score
            T = T0 * (1 - it / max_iters)
            if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
                h, curr_score = h_new, new_score
                if curr_score < best_global_score:
                    best_global_score, best_global_h = curr_score, h.copy()
                    if best_global_score <= target:
                        return best_global_h, best_global_score
        # optional greedy improvement after SA
        # perform a limited pass
        improved = True

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds bracket s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def adam_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta1, beta2, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def rudin_shapiro(n):
    # Generate Rudin–Shapiro binary sequence (0/1) via parity of (i & (i>>1))
    arr = np.empty(n, dtype=float)
    for i in range(n):
        b = i & (i >> 1)
        parity = bin(b).count('1') & 1
        val = -1 if parity else 1
        arr[i] = (val + 1) / 2.0
    return arr

                h, curr_score = h_new, new_score
                if curr_score < best_score:
                    best_score, best_h = curr_score, h.copy()
                    if best_score <= target:
                        return best_h, best_score
        improved = True
        while improved and time_left() > 0.01:
            improved = False
            for _ in range(200):
                i = rng.integers(0, n)
                if h[i] != 0.0:
                    continue
                j = rng.integers(0, n)
                if h[j] != 1.0:
                    continue
                h_new = h.copy()
                h_new[i] = 1.0
                h_new[j] = 0.0
                new_score = compute_c5(h_new)
                if new_score + 1e-12 < curr_score:
                    h, curr_score = h_new, new_score
                    improved = True
                    if curr_score < best_score:
                        best_score, best_h = curr_score, h.copy()
                        if best_score <= target:
                            return best_h, best_score
    return best_h, best_score

def rudin_shapiro(n):
    arr = np.empty(n, dtype=float)
    for i in range(n):
        b = i & (i >> 1)
        parity = bin(b).count('1') & 1
        val = -1 if parity else 1
        arr[i] = (val + 1) / 2.0
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_initial = 64
    target_bound = 0.3808

    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_initial:
            init_small = downsample(init_full, n_initial)
        else:
            init_small = init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_initial / 2.0))
    candidates.append(thue_morse(n_initial).astype(float))
    candidates.append(rudin_shapiro(n_initial))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_initial, rng).astype(float))
    for _ in range(2):
        v = rng.random(n_initial)
        candidates.append(project_onto_box_sum(v, n_initial / 2.0))

    best_h = None
    best_score = np.inf

    hyper_params = [
        (2000, 0.5, 0.5, 20.0, 0.9),
        (4000, 0.5, 0.5, 30.0, 0.95),
        (3000, 0.4, 0.1, 15.0, 0.9)
    ]
    for h0 in candidates:
        if time_left() < 2.0:
            break
        for n_iter, step0, a_start, a_end, beta in hyper_params:
            h_opt, sc_opt = smooth_grad_descent(
                h0, n_iter=n_iter, step0=step0,
                alpha_start=a_start, alpha_end=a_end,
                beta=beta, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            if time_left() < 1.0:
                break
        if time_left() < 1.0:
            break

    if best_h is not None and time_left() > 5.0:
        h_opt2, sc_opt2 = adam_grad_descent(
            best_h, n_iter=4000, step0=0.3,
            alpha_start=0.5, alpha_end=30.0,
            beta1=0.9, beta2=0.999,
            time_left=time_left)
        if sc_opt2 < best_score:
            best_score, best_h = sc_opt2, h_opt2.copy()

    if time_left() > 10.0:
        bin_h, bin_sc = binary_sa_search(n_initial, target_bound, time_left, rng)
        if bin_h is not None and bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
        if time_left() > 5.0:
            h_ref, sc_ref = smooth_grad_descent(
                best_h, n_iter=2000, step0=0.4,
                alpha_start=0.5, alpha_end=20.0,
                beta=0.9, time_left=time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    if time_left() > 12.0:
        n_up = 128
        h_up = upsample(best_h, n_up)
        h_up = project_onto_box_sum(h_up, n_up / 2.0)
        h_up_opt, sc_up = smooth_grad_descent(
            h_up, n_iter=3000, step0=0.4,
            alpha_start=0.5, alpha_end=20.0,
            beta=0.95, time_left=time_left)
        if sc_up < best_score:
            best_score, best_h = sc_up, h_up_opt.copy()

    if time_left() > 15.0:
        n_up2 = 256
        h_up2 = upsample(best_h, n_up2)
        h_up2 = project_onto_box_sum(h_up2, n_up2 / 2.0)
        h_up2_opt, sc_up2 = smooth_grad_descent(
            h_up2, n_iter=2000, step0=0.3,
            alpha_start=0.5, alpha_end=20.0,
            beta=0.95, time_left=time_left)
        if sc_up2 < best_score:
            best_score, best_h = sc_up2, h_up2_opt.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
