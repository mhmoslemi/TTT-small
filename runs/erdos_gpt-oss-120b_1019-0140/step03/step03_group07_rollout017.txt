def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute auto-correlation
    conv = np.correlate(h, h, mode='full')
    # exclude zero shift at index n-1
    conv_excl = np.concatenate([conv[:n-1], conv[n:]])
    min_conv = np.min(conv_excl)
    return 1.0 - min_conv * dx

def hill_climb_for_n(n, time_budget, rng, init_h=None):
    M = n // 2
    if init_h is not None and len(init_h)==n:
        h = np.array(init_h, dtype=float)
        # Ensure binary? may be fractional. If not binary, we can round maybe.
        # If not binary, we can keep as is but our swaps rely on binary. So convert to binary:
        # Count ones needed = M. Choose indices with highest values? Or random.
        # Simpler: if init_h not binary, we disregard and use random balanced binary.
        if not np.all(np.isin(h, [0.0,1.0])):  # fractional values?
            # convert to binary by selecting top M indices
            idx = np.argsort(h)[-M:]
            h = np.zeros(n, dtype=float)
            h[idx] = 1.0
    else:
        # random binary balanced
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, M, replace=False)] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)

    current_h = h.copy()
    current_c5 = best_c5

    zeros = np.where(current_h == 0)[0]
    ones = np.where(current_h == 1)[0]

    start = time.time()
    # schedule for temperature: start_temp = 0.01 maybe; decays every 1000 iter
    iter_count = 0
    start_temp = 0.02  # high enough to accept worse occasionally
    temp = start_temp
    while time.time() - start < time_budget - 0.01:
        # pick random zero, one
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        # swap
        # Instead of copy the whole array each time (cost O(n) per iteration) we can modify in place and revert if not accepted.
        # We'll modify current_h in place:
        current_h[i] = 1.0
        current_h[j] = 0.0

        c5_new = compute_c5(current_h)

        # acceptance
        if c5_new < current_c5:
            accept = True
        else:
            # probability
            # delta = c5_new - current_c5 >0 (worse)
            prob = np.exp(-(c5_new - current_c5) / temp) if temp > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            # update current state and zero/one lists
            # zeros becomes zeros w/out i plus j (if j now zero)
            zeros = np.where(current_h == 0)[0]
            ones = np.where(current_h == 1)[0]
            current_c5 = c5_new
            # Update best
            if c5_new + 1e-12 < best_c5:
                best_c5 = c5_new
                best_h = current_h.copy()
        else:
            # revert swap
            current_h[i] = 0.0
            current_h[j] = 1.0
            # no change to lists or current_c5

        iter_count += 1
        # temperature schedule: decay exponential per iteration perhaps
        if iter_count % 2000 == 0:
            # halve temperature
            temp *= 0.9   # or 0.95
            # also maybe random restart if stuck: after large number of iterations no improvement
        # optionally restart if stuck: but we will keep exploring.

    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap C5 for binary step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_sequence(p):
    """Legendre symbol sequence of length p (p prime). Returns 0/1 values."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def binary_initial(n, M, rng):
    """Construct a balanced binary vector of length n with M ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, M, replace=False)] = 1.0
    return h

def hill_climb_for_n(n, time_budget, rng, init_h=None):
    """Stochastic hill climbing / simulated annealing for a given n."""
    M = n // 2
    # Start from init_h if usable, otherwise balanced random or Legendre for n=80.
    if init_h is not None and len(init_h) == n:
        h = np.array(init_h, dtype=float)
        # Force binary while preserving as many ones as possible.
        ones = np.where(h > 0.5)[0]
        if len(ones) > M:
            drop = rng.choice(ones, len(ones) - M, replace=False)
            h[drop] = 0.0
        elif len(ones) < M:
            zeros = np.where(h <= 0.5)[0]
            add = rng.choice(zeros, M - len(ones), replace=False)
            h[add] = 1.0
        h = np.where(h > 0.5, 1.0, 0.0)
    else:
        if n == 80:
            # Legendre seed (p = 79)
            leg = legendre_sequence(79)
            h = np.concatenate([leg, [1.0]])
            # random rotate to break symmetry
            h = np.roll(h, rng.integers(n))
        else:
            h = binary_initial(n, M, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    current_h = h.copy()
    current_c5 = best_c5

    start = time.time()
    iter_cnt = 0
    no_improve = 0
    temp = 0.03  # initial temperature for occasional uphill moves

    while time.time() - start < time_budget - 0.01:
        # pick a random zero and a random one to swap
        zeros = np.where(current_h == 0)[0]
        ones = np.where(current_h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = zeros[rng.integers(zeros.size)]
        j = ones[rng.integers(ones.size)]

        # perform swap
        current_h[i] = 1.0
        current_h[j] = 0.0
        c5_new = compute_c5(current_h)

        # acceptance test
        if c5_new < current_c5:
            accept = True
        else:
            delta = c5_new - current_c5
            prob = np.exp(-delta / temp) if temp > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            current_c5 = c5_new
            if c5_new + 1e-12 < best_c5:
                best_c5 = c5_new
                best_h = current_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert swap
            current_h[i] = 0.0
            current_h[j] = 1.0
            no_improve += 1

        iter_cnt += 1

        # temperature decay
        if iter_cnt % 2000 == 0:
            temp *= 0.90
            if temp < 1e-5:
                temp = 1e-5

        # occasional random restart if stuck
        if no_improve >= 5000:
            current_h = binary_initial(n, M, rng)
            current_c5 = compute_c5(current_h)
            no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_all = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (even numbers, modest size)
    candidates = [80, 96, 112, 128, 144, 160]
    # place parent construction first if present
    if 'initial_h_values' in globals():
        n_parent = len(initial_h_values)
        if n_parent % 2 == 0:
            if n_parent in candidates:
                candidates.remove(n_parent)
            candidates = [n_parent] + candidates

    for idx, n in enumerate(candidates):
        elapsed = time.time() - start_all
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed
        # allocate time: more to the first (usually the parent) and less to later ones
        if idx == 0:
            alloc = min(remaining * 0.55, 600.0)
        else:
            alloc = min(remaining * 0.45 / (len(candidates) - idx), 300.0)

        init_h = None
        if 'initial_h_values' in globals() and n == len(initial_h_values):
            init_h = np.array(initial_h_values, dtype=float)

        h_opt, c5_opt = hill_climb_for_n(n, alloc, rng, init_h=init_h)

        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # fallback if nothing found
    if best_h is None:
        best_n = 80
        best_h = binary_initial(best_n, best_n // 2, rng)
        best_c5 = compute_c5(best_h)

    # final sanity: enforce constraints exactly
    n = best_n
    M = n // 2
    h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-9:
        diff = M - cur
        if diff > 0:
            zeros = np.where(h < 0.5)[0]
            if zeros.size >= diff:
                h[zeros[:int(diff)]] = 1.0
        else:
            ones = np.where(h > 0.5)[0]
            if ones.size >= -diff:
                h[ones[:int(-diff)]] = 0.0
    h = np.where(h > 0.5, 1.0, 0.0)

    final_c5 = compute_c5(h)
    return h, final_c5, n
