def compute_c5_fft(h):
    n = h.size
    L = n + n - 1
    H = np.fft.rfft(np.concatenate([h, np.zeros(n-1)]), n=L)
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n-1)]), n=L)
    corr = np.fft.irfft(H * G, n=L)
    return np.max(corr) * (2.0 / n)
```

def anneal_swap(h, time_end, max_iters):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    cur_h = h.copy()
    cur_score = best_score
    # Precompute indices of ones and zeros.
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    # Keep updating list after each swap.
    T0 = 0.01  # can be tuned
    for it in range(max_iters):
        if time.time() >= time_end:
            break
        # cooling schedule
        T = T0 * (1.0 - it / max_iters)
        # pick random indices to swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        idx1 = ones[i]
        idx2 = zeros[j]
        # swap
        cur_h[idx1] = 0.0
        cur_h[idx2] = 1.0
        # compute new score
        new_score = compute_c5_fft(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            # accept
            cur_score = new_score
            # update ones/zeros lists
            # swap elements in lists: replace idx1 in ones with idx2, idx2 in zeros with idx1
            ones[i] = idx2
            zeros[j] = idx1
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # reject: revert swap
            cur_h[idx1] = 1.0
            cur_h[idx2] = 0.0
        # maybe occasionally recompute ones/zeros to avoid drift if many rejects; but updates handled.
    return best_h, best_score
```

                best_h = h.copy()
        else:

            h[idx1] = 1.0
            h[idx2] = 0.0

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(np.concatenate([h, np.zeros(n - 1)]), n=L)
    G = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n - 1)]), n=L)
    corr = np.fft.irfft(H * G, n=L)
    return np.max(corr) * (2.0 / n)

compute_c5 = compute_c5_fft  # alias for convenience

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, time_end, max_iters, rng):
    n = h0.size
    h = h0.copy()
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    T0 = max(0.001, (cur_score - 0.38) * 5.0)
    for it in range(max_iters):
        if time.time() >= time_end:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        T = T0 * (1.0 - it / max_iters)
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        idx1 = ones[i]
        idx2 = zeros[j]
        h[idx1] = 0.0
        h[idx2] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_score = new_score
            ones[i] = idx2
            zeros[j] = idx1
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        else:
            h[idx1] = 1.0
            h[idx2] = 0.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512
    target = 0.3808
    best_h = None
    best_score = np.inf

    # initialize candidates
    cand_list = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init, n), n / 2.0)
        cand_list.append(cand)
    cand_list.append(thue_morse_binary(n))
    leg = legendre_sequence(257)[:n // 2]
    leg_big = np.tile(leg, 2)[:n]
    cand_list.append(leg_big)
    for _ in range(3):
        cand_list.append(random_binary_balanced(n, rng))

    # evaluate basic candidates and optional local GD
    for cand in cand_list:
        if time_left() < 0.5:
            break
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
        if time_left() > 5.0:
            h_opt, sc_opt = gradient_descent_momentum(
                cand,
                step0=0.5,
                beta=0.9,
                time_end=start + budget_s - 0.5,
                max_iter=20000
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_score = compute_c5(best_h)

    # binary annealing phase
    bin_start = binarize_from_real(best_h)
    while time_left() > 5.0 and best_score > target:
        remaining = time_left()
        max_iters = int(min(50000, max(2000, remaining * 400)))
        h_ann, sc_ann = anneal_binary(bin_start, start + remaining - 0.2, max_iters, rng)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()
            bin_start = binarize_from_real(best_h)
        else:
            bin_start = random_binary_balanced(n, rng)

    # final continuous refinement
    if time_left() > 2.0:
        h_fin, sc_fin = gradient_descent_momentum(
            best_h,
            step0=0.3,
            beta=0.9,
            time_end=start + budget_s - 0.1,
            max_iter=100000
        )
        if sc_fin < best_score:

    return best_h, best_score, n
