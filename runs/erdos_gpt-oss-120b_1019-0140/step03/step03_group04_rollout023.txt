def compute_c5(h):
    n = h.size
    # Use FFT based correlation
    fft_size = 1 << (2*n - 1).bit_length()
    # pad arrays
    h_pad = np.zeros(fft_size, dtype=float)
    h_pad[:n] = h
    rev = np.zeros(fft_size, dtype=float)
    rev[:n] = (1 - h)[::-1]  # reversed complement
    H = np.fft.rfft(h_pad)
    R = np.fft.rfft(rev)
    conv = np.fft.irfft(H * R, n=fft_size)
    # correlation indices: length = 2*n -1; conv array index k corresponds to correlation at lag (k - (n-1)) if using convolution? Let's verify:
    # If conv = convolution (h, rev), then conv[k] = sum_i h[i] * rev[k - i] = sum_i h[i] * (1 - h)[n-1 - (k - i)] = sum_i h[i] * (1 - h)[(i - (k - (n-1))]. So shift = k - (n-1).
    # So max over k of conv[k] corresponds to max over shift in [-(n-1), n-1].
    max_corr = conv.max()
    return max_corr * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    # compute convolution via FFT
    m = 1 << ((2 * n - 1).bit_length())
    # pad arrays
    a = np.zeros(m, dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    a[:n] = h
    b[:n] = (1.0 - h)[::-1]
    # compute FFT
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    conv = np.fft.irfft(fa * fb, n=m)
    # correlation length 2n-1 is conv[:2*n-1]? Actually conv will contain length m; relevant part is indices 0..2*n-2 inclusive.
    max_corr = conv[:2*n-1].max()
    return max_corr * (2.0 / n)

    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]
    # compute lists of indices for 1s and 0s
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    # precompute temperature schedule parameters
    T0 = 0.001  # initial temperature
    start = time.time()
    while time.time() < time_end:
        # possibly recompute ones/zeros each time for simplicity
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        elapsed = time.time() - start

            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        # break early if best_score <= target (0.3808)
        # but target may be slightly lower than actual bound.
        # check after each swap
        if best_score <= 0.3808:
            break
    return best_h, best_score

def compute_c5(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(m, dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    a[:n] = h
    b[:n] = (1.0 - h)[::-1]
    Fa = np.fft.rfft(a)

    conv = np.fft.irfft(Fa * Fb, n=m)
    max_corr = conv[:2*n-1].max()

def compute_c5(h):
    n = h.shape[0]
    m = next_pow2(2 * n - 1)
    a = np.zeros(m, dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    a[:n] = h
    b[:n] = (1.0 - h)[::-1]
    fa = np.fft.rfft(a)

def hill_climb_binary(h0, time_end, rng):
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]
    T0 = 0.01
    start = time.time()
    while time.time() < time_end:
        # choose random indices
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        elapsed = time.time() - start
        total = time_end - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * (1.0 - frac)
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score <= 0.3808:
                    break
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        # optional early break if time tight
    return best_h, best_score

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=np.float64)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def compute_c5(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    m = next_pow2(2 * n - 1)
    a = np.zeros(m, dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    a[:n] = h
    b[:n] = (1.0 - h)[::-1]
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    conv = np.fft.irfft(fa * fb, n=m)
    max_corr = conv[: 2 * n - 1].max()
    return max_corr * (2.0 / n)

def resample(arr: np.ndarray, new_n: int) -> np.ndarray:
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def legendre_sequence(p: int) -> np.ndarray:
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=np.float64)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_balanced(arr: np.ndarray) -> np.ndarray:
    n = arr.shape[0]
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=np.float64)
    binar[idx] = 1.0
    return binar

def hill_climb_binary(h0: np.ndarray, time_end: float, rng: np.random.Generator):
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    T0 = 0.01  # initial temperature for simulated annealing
    start = time.time()
    n = cur_h.shape[0]
    while time.time() < time_end:
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        elapsed = time.time() - start
        frac = elapsed / (time_end - start) if time_end > start else 1.0
        T = T0 * (1.0 - frac)
        accept = False
        if delta <= 0.0:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score

                if best_score <= 0.3808:

        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
