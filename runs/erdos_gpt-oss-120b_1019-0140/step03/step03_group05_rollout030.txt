def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation: length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Equivalent to projection onto hyperplane sum=v_sum to target while respecting bounds.
    # Use bisection on lambda offset.
    low = -1e10
    high = 1e10

    # For lambda = -inf, v+lambda = -inf => clip to lo -> sum = lo * n

        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(v + (low + high) * 0.5, lo, hi)

def local_descent(h_start, target_sum, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    max_iter = 10000
    it = 0
    while it < max_iter and time.time() < time_limit:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # get worst shift
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                # random perturb
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    # final check
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def slsqp_refine(h_start, target_sum, time_limit):
    n = h_start.shape[0]
    sum_con = lambda x: np.sum(x[:n]) - target_sum
    # Create constraints for each shift except zero
    # OverlapConstraint returns t - overlap*dx >=0
    # We'll include t as extra variable.
    class OverlapConstraint:
        def __init__(self, k):
            self.k = k
            self.dx = 2.0 / n
        def __call__(self, x):
            h = x[:n]
            k = self.k
            if k >= 0:
                overlap = np.dot(h[:n-k], 1.0 - h[k:])
            else:
                kk = -k
                overlap = np.dot(h[kk:], 1.0 - h[:n-kk])
            return x[-1] - overlap * self.dx
    overlap_constraints = [OverlapConstraint(k) for k in range(-n+1, n) if k != 0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [compute_c5(h_start)]])
    # Use SLSQP with some limited maxiter e.g., 500.
    res = minimize(lambda x: x[-1], x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=([{'type':'eq','fun':sum_con}] +
                                [{'type':'ineq','fun':oc} for oc in overlap_constraints]),
                   options={'maxiter':500, 'ftol':1e-9, 'disp':False})
    if not res.success:
        return h_start, compute_c5(h_start)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)

            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

# SLSQP refinement (introduce slack variable t)
class OverlapConstraint:
    def __init__(self, k: int, n: int):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_refine(h_start: np.ndarray, target_sum: float, remaining_time: float):
    n = h_start.shape[0]
    sum_con = lambda x: np.sum(x[:n]) - target_sum
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [compute_c5(h_start)]])
    # limit maxiter based on remaining time (heuristic)
    maxiter = 500
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method="SLSQP",
        bounds=bounds,
        constraints=([{"type": "eq", "fun": sum_con}]
                     + [{"type": "ineq", "fun": oc} for oc in overlap_cons]),
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        return h_start, compute_c5(h_start)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)

# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose discretization size (balance resolution vs speed)
    n = 120
    target_sum = n / 2.0

    # Initialise from parent if available, otherwise random feasible vector
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0, 2, h0.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    best_h, best_c5 = h0, compute_c5(h0)

    # ------------------------------------------------------------------
    # Phase 1: Differential Evolution (≈60 % of budget)
    de_deadline = start_time + budget_s * 0.6
    best_h, best_c5 = differential_evolution(compute_c5, n, target_sum, rng, de_deadline)

    # ------------------------------------------------------------------
    # Phase 2: Local gradient descent (≈30 % of budget)
    ld_deadline = start_time + budget_s * 0.9
    best_h, best_c5 = local_descent(best_h, target_sum, ld_deadline)

    # ------------------------------------------------------------------
    # Phase 3: Final SLSQP refinement (remaining time)
    if time.time() < start_time + budget_s - 1.0:
        remaining = start_time + budget_s - time.time()
        best_h, best_c5 = slsqp_refine(best_h, target_sum, remaining)

    # Ensure feasibility and final evaluation
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
