def adam_optimize(h0, time_left, max_iters=100000, alpha=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    while t < max_iters and time_left() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # projection
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # optional: early stop if best_score <= target? We'll handle outside
    return best_h, best_score

    h = h0.copy()

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):

    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            if shift < n:
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
        else:

                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000):
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h, best_sc = h.copy(), cur
    T, T_min, decay = 0.1, 1e-5, 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
