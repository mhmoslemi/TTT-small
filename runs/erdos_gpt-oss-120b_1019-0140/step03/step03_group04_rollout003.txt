def gradient_descent_momentum(h0, step0, beta, time_end, max_iter):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n-s]
                term2[:n-s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad = grad / len(shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            # Optionally reset momentum
        h = h_new
    return best_h, best_score
```

import time

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

                term2 = np.zeros_like(h)
                term1[s2:] = h[:n - s2]
                term2[:n - s2] = h[s2:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_refine(h, rng, time_left_func, max_iter=200000):
    # h is binary (0/1) and already sums to half
    best_score = compute_c5(h)
    it = 0
    while time_left_func() > 0.01 and it < max_iter:
        it += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
        else:
            h[i] = 1.0
            h[j] = 0.0
        if it % 1000 == 0 and time_left_func() < 0.2:
            break
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small = 256
    n_large = 512
    best_h = None
    best_score = np.inf

    # ----- small‑scale candidates -----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small / 2.0)
        candidates.append(cand)
    # Legendre (p = n_small+1 is prime)
    cand = legendre_sequence(n_small + 1)
    candidates.append(project_onto_box_sum(cand, n_small / 2.0))
    for _ in range(3):
        candidates.append(random_balanced_binary(n_small, rng))

    # Optimize each candidate with limited GD time
    for h0 in candidates:
        if time_left() < 10:
            break
        # Reserve a safety margin of 120 s for later stages
        rem = time_left() - 120.0
        gd_time = max(5.0, min(200.0, rem))
        end = time.time() + gd_time
        h_opt, sc_opt = gradient_descent_momentum(
            h0, step0=0.8, beta=0.9, time_end=end, max_iter=300000)
        # binary refinement
        h_bin = binarize_from_real(h_opt)
        h_ref, sc_ref = binary_swap_refine(h_bin, rng, time_left, max_iter=150000)
        # keep best
        for cand_h, cand_sc in [(h_opt, sc_opt), (h_ref, sc_ref)]:
            if cand_sc < best_score:
                best_score, best_h = cand_sc, cand_h.copy()
        if best_score <= 0.3808:
            break

    if best_h is None:
        best_h = random_balanced_binary(n_small, rng)
        best_score = compute_c5(best_h)

    # ----- upsample to larger resolution -----
    h_up = resample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)

    # gradient descent on large size (use remaining time)
    if time_left() > 30.0:
        end = time.time() + time_left() - 5.0
        h_opt, sc_opt = gradient_descent_momentum(
            h_up, step0=0.8, beta=0.9, time_end=end, max_iter=500000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # binary refinement of large candidate
        if time_left() > 5.0:
            h_bin = binarize_from_real(h_opt)
            h_ref, sc_ref = binary_swap_refine(h_bin, rng, time_left, max_iter=200000)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    # ----- additional random restarts if still above target -----
    while time_left() > 10.0 and best_score > 0.3808:
        v = rng.random(n_large)
        h0 = project_onto_box_sum(v, n_large / 2.0)
        end = time.time() + min(20.0, time_left() - 5.0)
        h_opt, sc_opt = gradient_descent_momentum(
            h0, step0=0.8, beta=0.9, time_end=end, max_iter=200000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if time_left() > 5.0:
            h_bin = binarize_from_real(h_opt)
            h_ref, sc_ref = binary_swap_refine(h_bin, rng, time_left, max_iter=150000)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= 0.3808:
            break

    # final safety projection and score recompute
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
