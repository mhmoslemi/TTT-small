import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def solve_milp(n, target, dx, deadline):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target, t >= 0]
    for k in range(1, n):
        # w variables
        w = cp.Variable(n - k, boolean=True)
        constraints += [
            w <= h[:n - k],
            w <= h[k:],
            w >= h[:n - k] + h[k:] - 1,
        ]
        # constraint on t
        constraints.append(t >= dx * (target - cp.sum(w)))
    prob = cp.Problem(cp.Minimize(t), constraints)
    remaining = max(deadline - time.time() - 0.5, 0.0)
    # if remaining is tiny, just skip
    try:
        prob.solve(solver=cp.CBC, maxSeconds=remaining, verbose=False)
    except Exception:
        return None, None
    if h.value is None:
        return None, None
    h_sol = np.round(np.array(h.value)).astype(float)  # round to 0/1
    # enforce sum exactly
    if abs(h_sol.sum() - target) > 1e-6:
        # adjust by adding/subtracting small amounts? but binary, sum should match.
        # In case rounding differs, fix:
        diff = int(round(target - h_sol.sum()))
        # if diff >0, set some zeros to 1
        zeros = np.where(h_sol < 0.5)[0]
        if diff > 0 and len(zeros) >= diff:
            idx = zeros[:diff]
            h_sol[idx] = 1.0
        elif diff < 0:
            ones = np.where(h_sol > 0.5)[0]
            if len(ones) >= -diff:
                idx = ones[: -diff]
                h_sol[idx] = 0.0
    t_val = t.value if t.value is not None else None
    return h_sol, t_val
```

            t >= dx * (target - cp.sum(w))
        ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    remaining = deadline - time.time() - 0.2
    if remaining <= 0:
        return None, None
    try:
        prob.solve(solver=solver, verbose=False, maxSeconds=remaining)
    except Exception:
        return None, None

    if h.value is None:
        return None, None
    h_bin = np.rint(np.array(h.value)).astype(float)  # round to 0/1
    return h_bin, t.value

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine number of points
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80

    target = n / 2.0
    dx = 2.0 / n

    # Try binary MILP first
    h_sol, _ = solve_binary_mip(n, target, dx, deadline)
    if h_sol is None:
        # Fallback random start
        h_sol = random_continuous_vector(n, target, rng)

    # Ensure feasibility
    h_sol = enforce_sum(h_sol, target, rng)
    best_h = h_sol.copy()
    best_val = compute_c5(best_h)

    # Simple hill‑climbing / simulated annealing using swaps
    cur_h = best_h.copy()
    cur_val = best_val

    while time.time() < deadline - 0.2:
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = compute_c5(cur_h)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # accept with a tiny temperature schedule
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # Final safety enforcement
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
