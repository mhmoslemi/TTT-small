import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    ... (same as before)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, arr.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)  # same approach

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[:n-shift] = h[shift:]
            term1[n-shift:] = 0.0
            term2[shift:] = h[:n-shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[s:] = h[:n-s]
            term1[:s] = 0.0
            term2[:n-s] = h[s:]
            term2[n-s:] = 0.0
        # gradient of 1 - h(x) h(x+shift) w.r.t h is - (h(x+shift) + h(x-shift))? The previous used (1 - term1 - term2) which is derivative of correlation w.r.t h? We'll reuse.
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, n_iter, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=400):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0

    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]

    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
        h_opt = res.x[:n]
        h_opt = project_onto_box_sum(h_opt, n/2.0)
        t_opt = compute_c5(h_opt)
        return h_opt, t_opt
    except Exception:
        return h0, compute_c5(h0)

def smooth_h(h, width):
    if width <= 1:
        return h
    kernel = np.ones(width) / width
    sm = np.convolve(h, kernel, mode='same')
    return sm

def anneal_continuous(h0, max_iter, temp_start, temp_end, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i, j = rng.choice(n, size=2, replace=False)

        # determine max delta to keep bounds
        max_inc = min(1.0 - h[i], h[j])
        max_dec = min(h[i], 1.0 - h[j])
        if max_inc + max_dec <= 0:
            continue
        # choose move direction
        if rng.random() < 0.5:
            # move from j to i
            delta = rng.random() * max_inc
        else:
            delta = -rng.random() * max_dec
        if delta == 0.0:
            continue
        h[i] += delta
        h[j] -= delta
        new_score = compute_c5(h)
        delta_score = new_score - cur_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_score / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bounds bracket the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, arr.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max‑correlation w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter, step0=0.5, beta=0.9):
    """Projected sub‑gradient descent with momentum on max‑correlation."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP refinement on small scale."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12,
                                    'disp': False})
        h_opt = res.x[:n]
        h_opt = project_onto_box_sum(h_opt, n / 2.0)
        t_opt = compute_c5(h_opt)
        return h_opt, t_opt
    except Exception:
        return h0, compute_c5(h0)


def smooth_h(h, width):
    """Uniform moving‑average smoothing."""
    if width <= 1:
        return h
    kernel = np.ones(width) / width
    return np.convolve(h, kernel, mode='same')


def anneal_continuous(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on continuous variables preserving sum."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i, j = rng.choice(n, size=2, replace=False)

        max_inc = min(1.0 - h[i], h[j])   # increase i, decrease j
        max_dec = min(h[i], 1.0 - h[j])   # decrease i, increase j
        if max_inc + max_dec <= 0:
            continue
        if rng.random() < 0.5:
            delta = rng.random() * max_inc
        else:
            delta = -rng.random() * max_dec
        if delta == 0.0:
            continue
        h[i] += delta
        h[j] -= delta
        new_score = compute_c5(h)
        delta_score = new_score - cur_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_score / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_score = np.inf

    # --------------------------------------------------
    # Phase 1 – small‑scale exploration (n = 128)
    # --------------------------------------------------
    n_small = 128
    small_cands = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)

        small = project_onto_box_sum(small, n_small / 2.0)
        small_cands.append(small)

    small_cands.append(thue_morse(n_small))
    small_cands.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        small_cands.append(project_onto_box_sum(v, n_small / 2.0))

    # SLSQP refinements of the seeds
    for cand in small_cands:
        if time_left() < 3.0:

        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score:

    # Additional random SLSQP runs
    while time_left() > 7.0:

                                                    step0=0.5, beta=0.9)

    # --------------------------------------------------
    if time_left() > 6.0:

    # Phase 5 – continuous simulated annealing
    # --------------------------------------------------
