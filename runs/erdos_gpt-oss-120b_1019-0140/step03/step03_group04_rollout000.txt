def hill_climb_binary(h, time_left, max_iters=50000):
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # Random swap
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        # pick random indices
        i = rng.choice(ones)
        j = rng.choice(zeros)

            best_h = h_new

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated on the uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence of length n (n a power of two)."""
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)
    return seq

def legendre_sequence(p):
    """Quadratic‑residue (Legendre) binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_top_half(arr):
    """Return binary vector with ones at the positions of the largest half of arr."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binvec = np.zeros(n, dtype=float)
    binvec[idx] = 1.0
    return binvec

def hill_climb_binary_with_time(h0, rng, time_limit_s):
    """Simple hill‑climbing on binary vectors under the sum constraint."""
    n = h0.size
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = h0.copy()
    cur_score = best_score
    end_time = time.time() + time_limit_s
    max_swaps = int(min(50000, max(2000, time_limit_s * 3000)))
    swaps = 0
    while swaps < max_swaps and time.time() < end_time:
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        swaps += 1
    return best_h, best_score
