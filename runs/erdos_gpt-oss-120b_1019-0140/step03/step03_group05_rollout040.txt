class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n-k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n-kk])
        return x[-1] - overlap * self.dx
```

def slsqp_optimize(h0, t0, sum_con, overlap_cons, maxiter=300):
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h0, [t0]])
    res = minimize(objective_slack, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': maxiter, 'ftol':1e-9, 'disp':False})
    return res
```

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # shift index
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                # add random perturbation to escape stagnation
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    # final check
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # overlap = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])? Actually earlier function gradient uses derivative of overlap = sum h[i] (1 - h[i+k]), derivative w.r.t h: for i in [0,n-k-1] derivative w.r.t h[i] = 1 - h[i+k]; for i in [k, n-1] derivative w.r.t h[i] = -h[i-k]
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing sum(v)=target via water-filling."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def gradient_for_shift(h, k):
    """Gradient of the worst overlap w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad


def gradient_descent(h_start, target_sum, max_iter, time_limit, rng=None):
    """Projected gradient descent on max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                if rng is not None:
                    noise = rng.normal(scale=0.02, size=n)
                else:
                    noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    final_c5 = compute_c5(best_h)
    return best_h, final_c5


# ---------- Constraints for SLSQP ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[-1] - overlap * self.dx


def objective_slack(x):
    """Objective for SLSQP: minimise the slack variable t."""
    return x[-1]


def slsqp_optimize(h0, t0, sum_con, overlap_cons, maxiter=500):
    """Run SLSQP refinement from h0 with initial slack t0."""
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h0, [t0]])
    res = minimize(
        fun=objective_slack,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Problem size (increase resolution for better bounds)
    n = 200
    target_sum = n / 2.0

    # ----- Initialise population -----
    pop_size = 12
    population = []

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        # linear interpolation to new grid
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        up_h = np.interp(x_new, x_old, h0)
        up_h = np.clip(up_h, 0.0, 1.0)
        up_h = project_to_box_sum(up_h, target_sum)
        population.append(up_h)

    while len(population) < pop_size:
        h = rng.random(n)
        h = project_to_box_sum(h, target_sum)
        population.append(h)

    # ----- Fitness evaluation -----
    fitness = [compute_c5(h) for h in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    safety = 5.0  # seconds left for final steps
    generation = 0

    # ----- Evolutionary loop -----
    while time.time() < start_time + budget_s - safety:
        generation += 1

        # Tournament selection for two parents
        i1, i2 = rng.integers(0, pop_size, size=2)
        p1 = population[i1] if fitness[i1] <= fitness[i2] else population[i2]
        i3, i4 = rng.integers(0, pop_size, size=2)
        p2 = population[i3] if fitness[i3] <= fitness[i4] else population[i4]

        # Crossover
        alpha = rng.random()
        child = alpha * p1 + (1.0 - alpha) * p2

        # Mutation
        if rng.random() < 0.8:
            child += rng.normal(scale=0.07, size=n)

        # Projection to feasible region
        child = np.clip(child, 0.0, 1.0)
        child = project_to_box_sum(child, target_sum)

        child_f = compute_c5(child)

        # Replace worst if better
        worst_idx = int(np.argmax(fitness))
        if child_f < fitness[worst_idx]:
            population[worst_idx] = child
            fitness[worst_idx] = child_f
            if child_f < best_c5:
                best_c5 = child_f
                best_h = child.copy()
                if best_c5 < 0.3808:
                    break

        # Periodic local refinement of the current best
        if generation % 50 == 0:
            remaining = start_time + budget_s - safety - time.time()
            if remaining > 10.0:
                gd_h, gd_c5 = gradient_descent(
                    best_h, target_sum, max_iter=2000,
                    time_limit=time.time() + remaining, rng=rng
                )
                if gd_c5 < best_c5:
                    best_h = gd_h
                    best_c5 = gd_c5
                    # Insert refined candidate into population
                    worst_idx = int(np.argmax(fitness))
                    population[worst_idx] = best_h.copy()
                    fitness[worst_idx] = best_c5

    # ----- Final SLSQP refinement if time permits -----
    time_left = start_time + budget_s - safety - time.time()
    if time_left > 10.0:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # ----- Final safety projection and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
