def compute_T_and_grad_softmax(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')  # length 2n-1
    # Compute T_k for all shifts:
    T = 1 - (2.0 / n) * corr
    # Softmax weights
    exps = np.exp(alpha * T - np.max(alpha * T))  # stability
    w = exps / np.sum(exps)
    # Compute gradient
    grad = np.zeros_like(h)
    for idx, shift in enumerate(range(-n+1, n)):
        p = w[idx]
        if p == 0:
            continue
        # shift positive: k>0 => h[j] interacts with h[j+k] (if j+k<n)
        # We'll compute contributions:
        if shift > 0:
            # j from 0 to n-1-shift: derivative from h[j] * h[j+shift]
            grad[:n-shift] += p * (-(2.0/n) * h[shift:])
            grad[shift:] += p * (-(2.0/n) * h[:n-shift])
        elif shift < 0:
            s = -shift
            # j from s to n-1: derivative from h[j-s] * h[j]
            grad[s:] += p * (-(2.0/n) * h[:n-s])
            grad[:n-s] += p * (-(2.0/n) * h[s:])
        else:
            # shift == 0
            grad += p * (-(4.0/n) * h)
    return T, grad
```

def linear_conv_fft(a, b):
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    fc = fa * fb
    c = np.fft.irfft(fc, n=N)
    return c[:n]
```

def grad_softmax_from_weights(h, w):
    # w: weight vector of shape (2*n-1,) corresponding to shifts -n+1..n-1.
    n = h.shape[0]
    w_rev = w[::-1]
    w_sym = w + w_rev
    w_sym_rev = w_sym[::-1]
    conv = np.convolve(w_sym_rev, h, mode='full')
    # conv indices for j = 0..n-1 start at 2*n-2
    start = 2*n-2
    grad = -(2.0 / n) * conv[start:start+n]
    return grad
```

Now function to compute w and gradient:

```
def compute_softmax_weights_and_grad(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    T = 1 - (2.0 / n) * corr
    aT = alpha * T
    max_aT = np.max(aT)
    exp_aT = np.exp(aT - max_aT)
    w = exp_aT / np.sum(exp_aT)
    grad = grad_softmax_from_weights(h, w)
    max_T = np.max(T)
    return max_T, grad, w
```

def adam_optimize(h0, max_iters, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, alpha_start=5.0, alpha_end=80.0, alpha_schedule='linear'):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    # schedule for alpha: linear from start to end over max_iters
    for it in range(1, max_iters+1):
        # compute alpha
        if alpha_schedule == 'linear':
            alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iters - 1) if max_iters > 1 else alpha_end
        else:
            alpha = alpha_end
        max_T, grad, _ = compute_softmax_weights_and_grad(h, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project back
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate actual C5
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Exact C5 for the discretized h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on a balanced binary vector."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h, best_score = cur_h.copy(), cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    for it in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def adam_optimize_time(h0, time_budget, max_iters=200000,
                       lr=0.01, beta1=0.9, beta2=0.999,
                       eps=1e-8, alpha_start=5.0, alpha_end=80.0):
    """Adam on the soft‑max approximation of max‑correlation."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h, best_score = h.copy(), compute_c5(h)
    it = 0
    while it < max_iters and (time.time() - start) < time_budget:
        it += 1
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, max_iters - 1)
        corr = np.correlate(h, h, mode='full')
        T = 1.0 - (2.0 / n) * corr
        aT = alpha * T
        max_aT = np.max(aT)
        exp_aT = np.exp(aT - max_aT)
        w = exp_aT / np.sum(exp_aT)                 # soft‑max weights
        w_rev = w[::-1]
        w_sym = w + w_rev
        w_sym_rev = w_sym[::-1]
        conv = np.convolve(w_sym_rev, h, mode='full')
        grad = -(2.0 / n) * conv[2 * n - 2 : 2 * n - 2 + n]
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # ------------------- small‑scale start -------------------
    n_small = 128
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        small = downsample(init_full, n_small) if init_full.size != n_small else init_full.copy()
    else:
        small = random_binary_balanced(n_small, rng)
    small = project_onto_box_sum(small, n_small / 2.0)

    t_small = min(time_left() * 0.08, 30.0)
    best_h_small, best_score_small = adam_optimize_time(small, t_small)

    # ------------------- upscale to medium -------------------
    n_large = 512
    best_h_large = upsample(best_h_small, n_large)
    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    best_score_large = compute_c5(best_h_large)

    # ------------------- medium optimisation (two starts) -----------
    remaining = time_left()
    if remaining > 12.0:
        t1 = remaining * 0.55
        h_opt1, sc1 = adam_optimize_time(best_h_large, t1)
        remaining = time_left()
        if remaining > 5.0:
            t2 = remaining * 0.65
            rand_start = random_binary_balanced(n_large, rng)
            rand_start = project_onto_box_sum(rand_start, n_large / 2.0)
            h_opt2, sc2 = adam_optimize_time(rand_start, t2)
            if sc2 < sc1:
                best_h_large, best_score_large = h_opt2, sc2
            else:
                best_h_large, best_score_large = h_opt1, sc1
        else:
            if sc1 < best_score_large:
                best_h_large, best_score_large = h_opt1, sc1

    # ------------------- binary annealing -------------------
    remaining = time_left()
    if remaining > 5.0:
        # Build binary from current best
        idx = np.argsort(best_h_large)[-n_large // 2:]
        h_bin = np.zeros(n_large, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(remaining * 600)  # ~600 proposals per sec
        temp_start, temp_end = 0.015, 1e-5
        h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann, temp_start, temp_end, rng)
        if sc_ann < best_score_large:
            best_h_large, best_score_large = h_ann, sc_ann

    # final projection and scoring
    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    final_score = compute_c5(best_h_large)
    return best_h_large, final_score, n_large
