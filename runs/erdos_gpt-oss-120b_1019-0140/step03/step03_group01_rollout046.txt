def compute_c5(h):
    n = h.size
    g = 1.0 - h
    h_pad = np.concatenate([h, np.zeros(n)])
    g_rev_pad = np.concatenate([g[::-1], np.zeros(n)])
    corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.fft.fft(g_rev_pad)))
    # The shifts correspond to k from -(n-1) to (n-1). In np.correlate output indexing: 0..2n-2.
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def gradient_descent_momentum_with_time(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        # compute corr via FFT
        n = h.size
        g = 1.0 - h
        N = 2*n
        H = np.fft.rfft(h, N)
        G_rev = np.fft.rfft(g[::-1], N)
        corr = np.fft.irfft(H * G_rev, N)
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # update momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=500000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

            return best_h, best_score, best_h.size

    # ---------- Phase 5: greedy binary hill climb ----------
    if best_h is not None and best_score > target and time_left() > 2.0:
        h_hc, sc_hc = hill_climb_binary(best_h, rng, time_left, max_trials=200000)
        if sc_hc < best_score:
            best_score, best_h = sc_hc, h_hc.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 6: upscale and further refinement ----------
    if best_h is not None and time_left() > 5.0:
        n_large = 512
        h_up = upsample(best_h, n_large)
        h_up = project_onto_box_sum(h_up, n_large / 2.0)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score, best_h = sc_up, h_up.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

        # gradient descent at larger scale
        if time_left() > 5.0 and best_score > target:
            h_gd2, sc_gd2 = gradient_descent_momentum_with_time(best_h, time_left, max_iter=200000)
            if sc_gd2 < best_score:
                best_score, best_h = sc_gd2, h_gd2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

        # binary conversion at large scale
        if best_score > target:
            n = best_h.size
            k = n // 2
            idx = np.argpartition(-best_h, k)[:k]
            h_bin2 = np.zeros_like(best_h)
            h_bin2[idx] = 1.0
            sc_bin2 = compute_c5(h_bin2)
            if sc_bin2 < best_score:
                best_score, best_h = sc_bin2, h_bin2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

        # binary annealing at large scale
        if best_score > target and time_left() > 3.0:
            h_ann2, sc_ann2 = binary_swap_anneal(best_h, rng, time_left, max_iters=300000)
            if sc_ann2 < best_score:
                best_score, best_h = sc_ann2, h_ann2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

        # final continuous refinement
        if best_score > target and time_left() > 2.0:
            h_final, sc_final = gradient_descent_momentum_with_time(best_h, time_left, max_iter=100000)
            if sc_final < best_score:
                best_score, best_h = sc_final, h_final.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 7: random continuous restarts ----------
    while time_left() > 5.0 and best_score > target:
        n = best_h.size if best_h is not None else 256
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        h_tmp, sc_tmp = gradient_descent_momentum_with_time(v, time_left, max_iter=5000)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
        if best_score <= target:
            break

    # final guard
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
