def compute_corr_fft(h):

    b_rev = (1.0 - h)[::-1]

    return corr

def gradient_descent_momentum_with_time(h0, time_left_func, max_iter=200000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.01:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # find shifts achieving max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        term1_sum = np.zeros_like(h)
        term2_sum = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1_sum[:n - shift] += h[shift:]
                term2_sum[shift:] += h[:n - shift]
            else:
                sft = -shift
                term1_sum[sft:] += h[:n - sft]
                term2_sum[:n - sft] += h[sft:]
        grad = np.ones_like(h) * len(shifts) - term1_sum - term2_sum
        if len(shifts) > 0:
            grad = grad / len(shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

    return np.clip(v - lam_mid, lo, hi)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_fft(h):
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, n=L)
    fb = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    conv = conv[:2 * n - 1]
    return conv[::-1]  # shifts from -(n-1) to +(n-1)

def compute_c5_fft(h):
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        max_iter=500000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.01:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        term1 = np.zeros_like(h)
        term2 = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1[:n - shift] += h[shift:]
                term2[shift:] += h[:n - shift]
            else:
                sft = -shift
                term1[sft:] += h[:n - sft]
                term2[:n - sft] += h[sft:]
        lsh = len(shifts)
        grad = np.ones_like(h) * lsh - term1 - term2
        if lsh:
            grad = grad / lsh
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=300000):
    h = h0.copy()
    n = h.shape[0]
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # ----- Phase 1 : small problem -----
    n_small = 128
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n_small)
        cand = project_onto_box_sum(cand, n_small / 2.0)
        candidates.append(cand)
    candidates.append(thue_morse(n_small))
    for _ in range(3):
        h_bin = np.zeros(n_small)
        idx = rng.choice(n_small, n_small // 2, replace=False)
        h_bin[idx] = 1.0
        candidates.append(h_bin.copy())
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))
    phi = (1 + np.sqrt(5)) / 2
    seq = (np.mod(np.arange(n_small) * phi, 1) < 0.5).astype(float)
    diff = int(seq.sum() - n_small / 2)
    if diff > 0:
        ones = np.where(seq == 1)[0]
        seq[ones[:diff]] = 0.0
    elif diff < 0:
        zeros = np.where(seq == 0)[0]
        seq[zeros[: -diff]] = 1.0
    candidates.append(seq)

    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = gradient_descent_momentum_with_time(
            h0, time_left, max_iter=200000, step0=0.7, beta=0.9)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 2 : upscale -----
    n_large = 512
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5_fft(best_h)
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    sc_up = compute_c5_fft(h_up)
    if sc_up < best_score:
        best_score, best_h = sc_up, h_up.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ----- Phase 3 : large continuous refinement -----
    if time_left() > 5.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(
            best_h, time_left, max_iter=600000, step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 4 : binary conversion -----
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5_fft(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ----- Phase 5 : binary annealing -----
    if time_left() > 2.0:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=400000)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 6 : final continuous refinement -----
    if time_left() > 1.0:
        h_ref2, sc_ref2 = gradient_descent_momentum_with_time(
            best_h, time_left, max_iter=200000, step0=0.3, beta=0.9)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # final projection guard
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
