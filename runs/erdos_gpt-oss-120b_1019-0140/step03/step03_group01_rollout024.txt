def smoothmax_and_grad(h, alpha):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    # compute softmax weights
    w = np.exp(alpha * corr)
    w_sum = np.sum(w)
    w_norm = w / w_sum
    term1_sum = np.zeros_like(h)
    term2_sum = np.zeros_like(h)
    # iterate shifts
    for idx, w_i in enumerate(w_norm):
        s = idx - (n - 1)  # shift
        if w_i == 0:
            continue
        if s >= 0:
            # term1: h[s:] placed at [:n-s]
            term1_sum[:n - s] += w_i * h[s:]
            # term2: h[:n-s] placed at [s:]
            term2_sum[s:] += w_i * h[:n - s]
        else:
            t = -s
            term1_sum[t:] += w_i * h[:n - t]
            term2_sum[:n - t] += w_i * h[t:]
    grad = 1.0 - term1_sum - term2_sum
    smooth_max = np.log(np.sum(w)) / alpha
    return smooth_max, grad
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx on the discretized grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max_{shift∈shifts} corr w.r.t. h (average over active shifts)."""
    n = h.size
    if len(shifts) == 0:
        return np.zeros_like(h)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            end = n - shift
            grad[:end] += 1.0 - h[shift:]
            grad[shift:] += -h[:end]
        else:
            s = -shift
            end = n - s
            grad[s:] += 1.0 - h[:end]
            grad[:end] += -h[s:]
    grad /= len(shifts)
    return grad

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def gradient_adam_optimize(h0, time_left_func, max_iters=50000, lr=0.2,
                           beta1=0.9, beta2=0.999, eps=1e-8, target=0.3808):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    while it <= max_iters:
        if time_left_func() <= 0.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= target:
                return best_h, best_score
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000,
                      target=0.3808):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Phase 1: small SLSQP on n=64

    candidates = []

    candidates.append(thue_morse(n_small))

    for _ in range(3):

    for cand in candidates:

            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=1500)
        if sc_opt < best_score:

        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 2: upsample to n=256
    n_large = 256
    if best_h is None:

    if best_score <= target:

                                               max_iters=40000, lr=0.2,
                                               target=target)

        if best_score <= target:

    # Phase 4: binary conversion (top half values)

    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)

            return best_h, best_score, best_h.size

    while time_left() > 12.0:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
