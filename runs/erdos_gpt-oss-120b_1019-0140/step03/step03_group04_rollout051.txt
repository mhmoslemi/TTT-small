def hill_climb_binary(initial_h, time_limit):
    h = initial_h.copy().astype(float)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    rng = np.random.default_rng()
    temp_start = 0.001  # maybe low
    # we may not use temperature, just accept improvements
    while time.time() < time_limit:
        # attempt swap
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        # perform swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        # Accept if improve
        if new_score <= current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_score
```

def hill_climb_binary(initial_h, time_end):
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    rng = np.random.default_rng()
    while time.time() < time_end:
        # sample swap
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_score
```

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts achieving max
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient subgradient
        grad = subgradient_max_corr(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box and sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def adam_optimize(h0, step0=0.5, beta1=0.9, beta2=0.999, eps=1e-8, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        if time_end is not None and time.time() >= time_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n-1)
        grad = subgradient_max_corr(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def legendre_sequence(p):
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def subgradient_max_corr(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift == 0:
            grad += 1.0 - 2.0 * h
        elif shift > 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] += 1.0 - h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += 1.0 - h[s:]
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = subgradient_max_corr(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(initial_h, time_end):
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    rng = np.random.default_rng()
    while time.time() < time_end:
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

        run_end = time.time() + per_restart

    # If no candidate yet, fall back to a random binary

    # Prepare seeds for binary hill climbing

    # Seed from initial_h_values if available
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
