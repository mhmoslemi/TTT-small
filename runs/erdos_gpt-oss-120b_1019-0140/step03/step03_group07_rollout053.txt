import time, numpy as np, cvxpy as cp

def solve_lp(n, time_limit):
   # Build problem quickly
   M = n // 2
   dx = 2.0 / n
   h = cp.Variable(n)
   t = cp.Variable()
   # constraints list
   constraints = [h >= 0, h <= 1, cp.sum(h) == M]
   # Build w variables as a list of cp.Variable for each shift
   # We'll keep a list w_k_vars per shift for convenience
   w_vars = []
   for k in range(1, n):
        size = n - k
        w_k = cp.Variable(size)
        # associate pairs h_i and h_{i+k}
        # create constraints linking w_k[i] to h_i, h_{i+k}
        constraints += [w_k >= 0,
                        w_k >= h[:size] + h[k:] - 1,
                        w_k <= h[:size],
                        w_k <= h[k:]]
        # overlapping constraint t >= 1 - dx * sum(w_k)
        constraints += [t >= 1 - dx * cp.sum(w_k)]
        w_vars.append(w_k)
   # objective
   objective = cp.Minimize(t)
   prob = cp.Problem(objective, constraints)
   # Solve with solver options to limit time
   # we can pass 'max_iters' or 'max_seconds' if solver supports.
   # ECOS supports max_iters; we can set max_iters = 50000 maybe.
   try:
        prob.solve(solver=cp.ECOS, max_iters=50000, verbose=False)
   except Exception:
        # fallback to default solver
        prob.solve(solver=cp.SCS, verbose=False)
   # after solving, get values
   if h.value is None or t.value is None:
        return None, None
   h_val = np.clip(np.array(h.value).flatten(), 0.0, 1.0)
   # adjust sum to exactly M via projection onto feasible set (may be minor numeric errors)
   cur = np.sum(h_val)
   if abs(cur - M) > 1e-6:
        # project onto hyperplane with box constraints
        h_val = project_onto_feasible(h_val, M)
   c5 = float(t.value)
   return h_val, c5

```python
def project_onto_feasible(y, M, tol=1e-9, max_iter=100):
    lo = -5.0
    hi = 5.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - M) <= tol:
            return x
        if s > M:
            lo = lam
        else:
            hi = lam
    # final
    return np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
```

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(y, M, tol=1e-9, max_iter=100):
    """Projection of y onto {0<=x<=1, sum(x)=M} using bisection on the shift."""
    lo, hi = -5.0, 5.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - M) <= tol:
            return x
        if s > M:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def solve_lp(n):
    """Solve the exact linear program for a given discretisation size."""
    M = n // 2
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    cons = [
        h >= 0,
        h <= 1,
        cp.sum(h) == M,
        t >= 0,
    ]
    for k in range(1, n):
        size = n - k
        w = cp.Variable(size)
        cons += [
            w >= 0,
            w >= h[:size] + h[k:] - 1,
            w <= h[:size],
            w <= h[k:],
            t >= 1 - dx * cp.sum(w),
        ]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iters=200000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None
    if h.value is None:
        return None, None
    h_val = np.clip(np.array(h.value).flatten(), 0.0, 1.0)
    cur = h_val.sum()
    if abs(cur - M) > 1e-8:
        h_val = project_onto_feasible(h_val, M)
    t_val = float(t.value) if t.value is not None else None
    return h_val, t_val

def hill_climb(h, cur_c5, time_limit, rng):
    n = h.shape[0]
    best_h, best_c5 = h.copy(), cur_c5
    end = time.time() + time_limit
    while time.time() < end:
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            max_tr = min(best_h[i], 1.0 - best_h[j])
            if max_tr <= 0.0:
                continue
            delta = rng.random() * max_tr
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_tr = min(best_h[j], 1.0 - best_h[i])
            if max_tr <= 0.0:
                continue
            delta = rng.random() * max_tr
            h_new = best_h.copy()
            h_new[j] -= delta
            h_new[i] += delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h, best_c5 = h_new, c5_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # consider the parent construction if present
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        if n0 % 2 == 0:
            M0 = n0 // 2
            cur = init.sum()
            if abs(cur - M0) > 1e-9:
                init = init + (M0 - cur) / n0
                init = np.clip(init, 0.0, 1.0)
            c5_init = compute_c5(init)
            best_h, best_c5, best_n = init, c5_init, n0

    # primary discretisation size
    candidates = [80, 96, 112, 128]
    for n in candidates:
        if time.time() - start > budget_s - 5:
            break
        # solve LP
        h_lp, _ = solve_lp(n)
        if h_lp is None:
            continue
        c5_lp = compute_c5(h_lp)
        if c5_lp + 1e-12 < best_c5:
            best_h, best_c5, best_n = h_lp, c5_lp, n
        # local refinement within remaining budget
        remaining = budget_s - (time.time() - start)
        if remaining <= 0:
            break
        # allocate at most half of the remaining time for hill climbing
        hc_time = min(remaining * 0.5, 30.0)
        h_ref, c5_ref = hill_climb(best_h, best_c5, hc_time, rng)
        if c5_ref + 1e-12 < best_c5:
            best_h, best_c5, best_n = h_ref, c5_ref, n
            if best_c5 < 0.3808:
                break

    # final safety fallback
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        M = best_n // 2
        cur = best_h.sum()
        if abs(cur - M) > 1e-8:
            best_h = project_onto_feasible(best_h, M)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
