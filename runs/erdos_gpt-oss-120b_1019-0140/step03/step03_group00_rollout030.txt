def local_search(h, n, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 2:
        # choose two indices
        i, j = rng.choice(n, size=2, replace=False)
        # amount to transfer
        max_delta = min(best_h[i], 1 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < best_score - 1e-6:
            best_h = h_new
            best_score = sc
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Start from initial_h_values if present.
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    # Multi-scale loop
    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    for n in scales:
        if time_left() < 5:
            break
        # Upsample base to n
        if base.shape[0] != n:
            base = upsample(base, n)
        # Project to feasible region
        base = project_onto_box_sum(base, n/2.0)

        # DE optimization - allocate some time
        if time_left() > 10:
            h_de, sc_de = de_optimize(n, rng, time_left)
            if sc_de < best_score:
                best_h = h_de
                best_score = sc_de
            # Update base for next stage
            base = h_de

        # Gradient descent refinement
        if time_left() > 5:
            h_gd, sc_gd = gradient_descent_momentum_custom(base, 
                                                            n_iter=int(min(80000, max(20000, time_left()*30))),
                                                            step0=0.2, beta=0.9,
                                                            rng=rng,
                                                            time_left=time_left)
            if sc_gd < best_score:
                best_h = h_gd
                best_score = sc_gd
            base = h_gd

        # Prepare for next scale
        if n != scales[-1]:
            base = upsample(base, scales[scales.index(n)+1])

    # After finishing scales, final refinement on best_h (largest n)
    if best_h is None:
        best_h = project_onto_box_sum(base, base.size/2.0)
        best_score = compute_c5(best_h)

    # final local search and random restart
    n_final = best_h.size
    # random balanced binary restarts
    while time_left() > 8:
        # create random binary candidate (balanced)
        h_rand = random_binary_balanced(n_final, rng)
        # refine with GD (short)
        h_rand_ref, sc_rand_ref = gradient_descent_momentum_custom(h_rand,
                                                                    n_iter=int(min(20000, max(5000, time_left()*15))),
                                                                    step0=0.5, beta=0.9,
                                                                    rng=rng,
                                                                    time_left=time_left)
        if sc_rand_ref < best_score:
            best_h = h_rand_ref
            best_score = sc_rand_ref
        # local search around best_h
        h_ls, sc_ls = local_search(best_h, n_final, rng, time_left)
        if sc_ls < best_score:
            best_h = h_ls
            best_score = sc_ls
        if best_score <= 0.3808:
            break

    # final projection
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_corr_fft(h):
    """Linear correlation of h with (1-h) via FFT, length 2n‑1."""
    n = h.shape[0]
    L = 2 * n
    f_h = np.fft.rfft(h, n=L)
    f_1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(f_h * np.conj(f_1mh), n=L)
    return corr[:2 * n - 1]


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def de_optimize(n, rng, time_left):
    """Differential evolution on the feasible region (via projection)."""
    if time_left() < 10:
        maxiter = 5
    elif time_left() < 30:
        maxiter = 10
    else:
        maxiter = 20
    popsize = 8
    bounds = [(0.0, 1.0)] * n
    result = opt.differential_evolution(
        lambda x: compute_c5(project_onto_box_sum(x, n / 2.0)),
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=int(rng.integers(1_000_000_000)),
        polish=False,
        updating='deferred')
    h = project_onto_box_sum(result.x, n / 2.0)
    return h, compute_c5(h)


def local_search(h, n, rng, time_left):
    """Small random mass‑transfer hill climbing."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while time_left() > 2:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5(h_new)
        if sc < best_score:
            best_h, best_score = h_new, sc
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    for n in scales:
        if time_left() < 5:
            break

        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Global DE search
        if time_left() > 10:
            h_de, sc_de = de_optimize(n, rng, time_left)
            if sc_de < best_score:
                best_h, best_score = h_de, sc_de
            base = h_de

        # Local momentum GD refinement
        if time_left() > 5:
            max_iter_map = {64: 40000, 128: 25000, 256: 15000, 512: 8000}
            max_iter = max_iter_map.get(n, 20000)
            # Scale iterations to remaining time proportionally
            est_iter = int(min(max_iter, max(2000, time_left() * (max_iter / budget_s))))
            step0 = 0.5 if n <= 128 else 0.2
            h_gd, sc_gd = gradient_descent_momentum_custom(
                base, n_iter=est_iter, step0=step0, beta=0.9, rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = h_gd, sc_gd
            base = h_gd

    # Fallback in case nothing was found
    if best_h is None:
        best_h = project_onto_box_sum(base, base.size / 2.0)
        best_score = compute_c5(best_h)

    # Final random‑restart hill climbing at the best resolution
    n_final = best_h.size
    while time_left() > 8:
        # Random balanced binary start
        h_rand = random_binary_balanced(n_final, rng)
        max_iter_rand_map = {64: 10000, 128: 8000, 256: 5000, 512: 3000}
        max_iter_rand = max_iter_rand_map.get(n_final, 5000)
        est_rand_iter = int(min(max_iter_rand, max(1000, time_left() * (max_iter_rand / budget_s))))
        h_rand_ref, sc_rand_ref = gradient_descent_momentum_custom(
            h_rand, n_iter=est_rand_iter, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
        if sc_rand_ref < best_score:
            best_h, best_score = h_rand_ref, sc_rand_ref

        # Small local search around current best
        h_ls, sc_ls = local_search(best_h, n_final, rng, time_left)
        if sc_ls < best_score:
            best_h, best_score = h_ls, sc_ls

        if best_score <= 0.3808:
            break

    # Final safety projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
