def binary_hill_climb_sa(h0, rng, time_left, target):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # initial temperature
    T0 = 0.01
    max_iters = int(min(500000, max(1, time_left()*2000)))  # e.g., about 2000 per second 
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # pick zero and one indices
        zeros = np.where(h == 0.0)[0]
        ones = np.where(h == 1.0)[0]
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        # swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:]      -= h[:n - k]
        else:
            s = -k
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style hill‑climb on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    T0 = 0.02
    max_iters = int(min(500000, max(1, time_left() * 1500)))
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        zeros = np.where(h == 0.0)[0]
        ones = np.where(h == 1.0)[0]
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ---- initial candidate pool (64 points) ----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != 64:
            init_small = downsample(init_full, 64)
        else:
            init_small = init_full.copy()
        init_small = project_onto_box_sum(init_small, 32.0)
        candidates.append(init_small)
    candidates.append(thue_morse(64).astype(float))
    for _ in range(2):
        candidates.append(random_binary_balanced(64, rng).astype(float))
    for _ in range(2):
        v = rng.random(64)
        candidates.append(project_onto_box_sum(v, 32.0))

    best_h = None
    best_score = np.inf

    # ---- Phase 1: small‑scale soft‑max gradient descent ----
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = smooth_grad_descent(
            h0,
            n_iter=3500,
            step0=0.5,
            alpha_start=1.0,
            alpha_end=30.0,
            beta=0.9,
            time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(64, 0.5), 32.0)
        best_score = compute_c5(best_h)

    # ---- Multi‑resolution refinement (64 → 128 → 256) ----
    for target_n in (128, 256):
        if time_left() < 5.0:
            break
        best_h = upsample(best_h, target_n)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)

        # momentum refinement
        iters = int(min(25000, max(8000, time_left() * 700)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                               step0=0.5, beta=0.9,
                                               time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

        # soft‑max refinement
        h_ref2, sc_ref2 = smooth_grad_descent(best_h, n_iter=2000,
                                              step0=0.5,
                                              alpha_start=1.0,
                                              alpha_end=30.0,
                                              beta=0.9,
                                              time_left=time_left)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()

    # ---- Binary hill‑climb at final resolution ----
    if time_left() > 4.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
        # final momentum polishing
        if time_left() > 2.0:
            h_ref3, sc_ref3 = momentum_grad_descent(best_h, n_iter=5000,
                                                   step0=0.3, beta=0.9,
                                                   time_left=time_left)
            if sc_ref3 < best_score:
                best_score, best_h = sc_ref3, h_ref3.copy()

    # ---- Additional random binary restarts if time permits ----
    while time_left() > 20.0:
        n = best_h.shape[0]
        rand_bin = random_binary_balanced(n, rng).astype(float)
        bin_h, bin_sc = binary_hill_climb_sa(rand_bin, rng, time_left, target=0.3808)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
            # polish the new incumbent
            if time_left() > 5.0:
                h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=4000,
                                                      step0=0.4, beta=0.9,
                                                      time_left=time_left)
                if sc_ref < best_score:
                    best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= 0.3808:
            break

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
