def projected_gradient_descent(h0, target_sum, max_iter, time_limit):
    n = len(h0)
    dx = 2.0/n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    beta = 0.9  # momentum factor
    step = 0.5  # initial step size
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation
        corr = np.fft.ifft(np.fft.fft(h, 2*n-1) * np.conj(np.fft.fft(1.0 - h, 2*n-1))).real
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        # update best
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # compute gradient of worst overlap shift (k)
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # Nesterov: compute lookahead
        # Actually we can compute momentum update: momentum = beta * momentum + grad
        # Update h with step * momentum
        momentum = beta * momentum + grad
        new_h = h - step * momentum
        # Project onto constraints
        new_h = project_to_box_sum(new_h, target_sum)
        # Evaluate new C5
        new_c5 = compute_c5_fft(new_h)
        if new_c5 < cur_c5:
            # Accept and maybe increase step
            h = new_h
            step = min(step * 1.1, 2.0)
        else:
            # Reject, reduce step, reset momentum maybe
            step = max(step * 0.5, 1e-4)
            momentum = np.zeros_like(h)  # Reset
            # Optionally add random perturbation
            if step <= 1e-4:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                # keep step moderate
                step = 0.2
    return best_h, best_c5

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        h = x[:self.n]
        return np.sum(h) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def gradient_descent_h(h_start, target_sum, max_iter, time_deadline):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5_fft(h)[0]   # compute c5 and ignore other returns
    momentum = np.zeros_like(h)
    beta = 0.9
    step = 0.5
    min_step = 1e-6
    for it in range(max_iter):
        if time.time() > time_deadline:
            break
        corr = compute_corr_fft(h)
        max_idx = int(np.argmax(corr))
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # Get shift k
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        momentum = beta * momentum + grad

                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                step = 0.2

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Projection ----------
def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while fixing the sum to target."""
    low = -10.0
    high = 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Gradient for a given shift ----------
def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

# ---------- Constraint classes ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

# ---------- C5 evaluation ----------
def compute_c5_info(h):
    """Return (c5, max_idx, full correlation array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    c5 = corr[max_idx] * dx
    return c5, max_idx, corr

# ---------- Projected gradient descent ----------
def gradient_descent(h_start, target_sum, max_iter, deadline):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5_info(h)[0]
    momentum = np.zeros_like(h)
    beta = 0.9
    step = 0.5
    min_step = 1e-6
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c5, max_idx, _ = compute_c5_info(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        momentum = beta * momentum + grad
        new_h = h - step * momentum
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5_info(new_h)[0]
        if new_c5 < c5:
            h = new_h
            step = min(step * 1.1, 2.0)
        else:
            step = max(step * 0.5, min_step)
            momentum = np.zeros_like(h)
            if step <= min_step:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                step = 0.2
    return best_h, best_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --- Initialise ---
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 120                     # larger grid when no parent is provided
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5_info(best_h)[0]

    # --- Prepare constraints for occasional SLSQP refinements ---
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # --- Quick SLSQP if time permits ---
    quick_deadline = start_time + budget_s * 0.04
    if time.time() < quick_deadline:
        t0 = best_c5
        try:
            res = minimize(lambda x: x[-1],
                           np.concatenate([best_h, [t0]]),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * n + [(0.0, None)],
                           constraints=[{'type': 'eq', 'fun': sum_con}]
                                        + [{'type': 'ineq', 'fun': oc} for oc in overlap_cons],
                           options={'maxiter': 300, 'ftol': 1e-9, 'disp': False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5_info(h_opt)[0]
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass

    # --- Main stochastic search loop ---
    buffer = 5.0  # safety margin before hard timeout
    while time.time() < start_time + budget_s - buffer:
        # Choose a starting point
        if rng.random() < 0.6:
            h_start = best_h + rng.normal(scale=0.04, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        # Allocate iterations based on remaining time
        remaining = start_time + budget_s - buffer - time.time()
        max_iter = int(min(8000, remaining * 180))  # around 180 iterations per second
        gd_h, gd_c5 = gradient_descent(h_start, target_sum, max_iter,
                                       start_time + budget_s - buffer)

        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
            if best_c5 < 0.3808:
                break

        # Optional short SLSQP refinement
        if time.time() < start_time + budget_s - buffer - 2.0:
            t_start = compute_c5_info(gd_h)[0]
            try:
                res = minimize(lambda x: x[-1],
                               np.concatenate([gd_h, [t_start]]),
                               method='SLSQP',
                               bounds=[(0.0, 1.0)] * n + [(0.0, None)],
                               constraints=[{'type': 'eq', 'fun': sum_con}]
                                            + [{'type': 'ineq', 'fun': oc} for oc in overlap_cons],
                               options={'maxiter': 250, 'ftol': 1e-9, 'disp': False})
                if res.success:
                    h_opt = np.clip(res.x[:n], 0.0, 1.0)
                    h_opt = project_to_box_sum(h_opt, target_sum)
                    c5_opt = compute_c5_info(h_opt)[0]
                    if c5_opt < best_c5:
                        best_h, best_c5 = h_opt, c5_opt
                        if best_c5 < 0.3808:
                            break
            except Exception:
                pass

    # --- Final projection and evaluation ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5_info(best_h)[0]

    return best_h, best_c5, n
