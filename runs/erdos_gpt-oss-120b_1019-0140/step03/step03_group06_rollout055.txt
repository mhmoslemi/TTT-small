    use_n2 = False
    n2 = 120
    if time.time() < deadline - 30 and n2 != n:

        old_x = np.linspace(0, 2, n, endpoint=False)  # points
        new_x = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(new_x, old_x, best_h)
        target2 = n2/2.0
        h2 = enforce_sum(h2, target2, rng)
        # We'll run annealing on h2 separately, store best there
        use_n2 = True

    # Define annealing function that works on given h, n, target
    def anneal(h, n, target):
        best_loc_h = h.copy()
        best_loc_val = compute_c5(best_loc_h)
        cur_h = h.copy()
        cur_val = best_loc_val
        # Temperature schedule
        T0 = 0.01  # initial temperature
        Tf = 1e-7
        # compute total allowed iterations based on remaining time
        while time.time() < deadline - 0.2:
            # recompute temperature based on elapsed fraction
            elapsed = time.time() - start
            total = deadline - start
            tau = elapsed / total
            T = T0 * (Tf/T0) ** tau  # exponential decay
            # find worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)
            # compute source indices candidates
            if k >= 0:
                src_idxs = np.arange(n - k)
                src = src_idxs[cur_h[src_idxs] > 1e-12]
                dst = src_idxs + k
                # need to ensure dst < n and h[dst] < 1-eps
                valid = src[cur_h[dst[src]] < 1.0 - 1e-12]

                i_candidates = np.where((cur_h > 1e-12) & (cur_h[(np.arange(n) + k) % n] < 1.0 - 1e-12))[0]
            else:
                kk = -k
                i_candidates = np.where((cur_h > 1e-12) & (cur_h[(np.arange(n) - kk) % n] < 1.0 - 1e-12))[0]
            if i_candidates.size == 0:

                dst = np.where(cur_h < 1.0 - 1e-12)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                delta = min(cur_h[i], 1.0 - cur_h[j]) * rng.random()
                cur_h[i] -= delta
                cur_h[j] += delta
            else:

                cur_h[j] += delta

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute convolution: h convolve (1 - h) reversed
    # correlation = convolve h with (1-h) reversed
    # using FFT:
    a = np.fft.rfft(h, n*2)
    b = np.fft.rfft(1.0 - h, n*2)
    corr = np.fft.irfft(a * np.conj(b), n*2)
    # The convolution length = 2n-1, but in irfft return length 2n
    # We'll take first 2n-1 entries
    corr = corr[:2*n-1]
    return np.max(corr) * dx
```

    conv = np.fft.ifft(A * np.conj(B)).real

def full_correlation(h):
    n = h.shape[0]
    N = 2 * n
    A = np.fft.fft(h, N)
    B = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(A * np.conj(B)).real
    return conv[:2*n-1]

def anneal(h, n, target, deadline, rng):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    # parameters
    T0 = 0.02
    Tf = 1e-8
    eps = 1e-12
    while time.time() < deadline - 0.1:
        elapsed = time.time() - start  # we need start defined globally. We'll pass start_time.
        total = deadline - start  # need start_time
        tau = (elapsed) / (total) if total > 0 else 0.0
        T = T0 * (Tf/T0) ** tau  # exponential cooling
        # find worst shift
        k, _ = find_worst_shift(cur_h)
        # candidates for transfer from i to i+k (mod n)
        i_candidates = np.where((cur_h > eps) & (cur_h[(np.arange(n) + k) % n] < 1.0 - eps))[0]
        if i_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = (i + k) % n
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            # perform move
            cur_h[i] -= delta
            cur_h[j] += delta
        else:
            # fallback random transfer between any two indices
            src = np.where(cur_h > eps)[0]
            dst = np.where(cur_h < 1.0 - eps)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            cur_h[i] -= delta
            cur_h[j] += delta

        new_val = compute_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            # accept with probability
            if T > 0 and np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert move
                cur_h[i] += delta
                cur_h[j] -= delta
    return best_h, best_val

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return v
    # iterate distributing diff
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v
```

import time
import numpy as np

def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust to achieve exact sum=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v

def random_continuous_vector(n, target, rng):
    """Uniform random vector in [0,1]^n projected to exact sum."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    """Binary vector with exactly target ones (target integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def interpolate_h(parent_h, n_new):
    """Linear interpolation of parent_h (length m) to length n_new."""
    m = parent_h.shape[0]
    x_parent = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_parent, parent_h)

def full_correlation(h):
    """Linear correlation of h with (1-h) via zero‑padded FFT."""
    n = h.shape[0]
    N = 2 * n
    A = np.fft.fft(h, N)
    B = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(A * np.conj(B)).real
    return conv[:2 * n - 1]

def compute_c5(h):
    """Maximum overlap integral approximated discretely."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = full_correlation(h)
    return np.max(corr) * dx

def anneal(h, n, target, deadline, start_time, rng):
    """Simulated annealing that transfers mass to reduce the worst overlap."""
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)

    # initial correlation
    corr = full_correlation(h)
    max_idx = np.argmax(corr)
    cur_val = corr[max_idx] * dx

    eps = 1e-12
    T0 = 0.02
    Tf = 1e-8

    while time.time() < deadline - 0.1:
        k = max_idx - (n - 1)  # shift giving current worst overlap
        # determine candidate transfer based on worst shift
        if k == 0:
            # fallback random transfer
            src = np.where(h > eps)[0]
            dst = np.where(h < 1.0 - eps)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
        else:
            if k > 0:
                i_range = np.arange(n - k)
            else:
                i_range = np.arange(-k, n)
            dest = i_range + k
            prod = h[i_range] * (1.0 - h[dest])
            i_local = np.argmax(prod)
            i = i_range[i_local]
            j = dest[i_local]
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                # fallback random transfer
                src = np.where(h > eps)[0]
                dst = np.where(h < 1.0 - eps)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
            else:
                delta = max_delta * rng.random()

        # apply transfer
        h[i] -= delta
        h[j] += delta

        # new correlation and value
        new_corr = full_correlation(h)
        new_max_idx = np.argmax(new_corr)
        new_val = new_corr[new_max_idx] * dx

        # temperature (linear decay)
        remaining = deadline - time.time()
        total = deadline - start_time
        T = max(Tf, T0 * (remaining / total))

        if new_val < cur_val:
            # accept improvement
            cur_val = new_val
            max_idx = new_max_idx
            corr = new_corr
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            # Metropolis criterion
            if T > 0 and np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
                max_idx = new_max_idx
                corr = new_corr
            else:
                # revert move
                h[i] += delta
                h[j] -= delta
                # keep old corr, cur_val, max_idx
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # target discretisation
    n = 120
    target = n / 2.0

    # initialise from parent if available, else random
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values, dtype=float)
        h0 = interpolate_h(parent_h, n)
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = random_continuous_vector(n, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # primary annealing run
    best_h, best_val = anneal(best_h, n, target, deadline, start_time, rng)

    # additional stochastic restarts while time permits
    while time.time() < deadline - 5.0:
        if rng.random() < 0.5:
            start_h = random_binary_vector(n, target, rng)
        else:
            start_h = random_continuous_vector(n, target, rng)
        h_local, val_local = anneal(start_h, n, target, deadline, start_time, rng)
        if val_local < best_val:
            best_val = val_local
            best_h = h_local
            if best_val < 0.3808:
                break

    # final feasibility enforcement and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
