    h = np.clip(h, 0, 1)

    cur = h.sum()

    if cur != M:

        h += delta
        # clip again
        h = np.clip(h, 0, 1)
        # recalc
        cur2 = h.sum()
        # adjust again if needed

            # compute amount left to adjust
            diff = M - cur2
            # distribute proportionally to those not at bounds
            mask = (h > 0) & (h < 1)
            if mask.any():

                h = np.clip(h, 0, 1)
            else:

                h = np.clip(h * (M / cur2), 0, 1)
                # final clip may need extra adjust

        # Use max directly
        return np.max(corr) * dx

    best_h_local = np.copy(h_init)
    best_c5_local = compute_c5_arr(best_h_local)

    h = np.copy(h_init)

def project_onto_bounds_and_sum(h, M):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    cur = h.sum()
    diff = M - cur
    if abs(diff) > 1e-12:
        # distribute diff over indices not at bounds (0 or 1)
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free > 0:
            h[free] += diff / n_free
            h = np.clip(h, 0.0, 1.0)
            # If still diff not zero due to clipping, adjust recursively
            cur2 = h.sum()
            if abs(cur2 - M) > 1e-12:
                # secondary adjust by scaling all entries
                h = h * (M / cur2)
                h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds; scale uniformly
            h = h * (M / cur)
            h = np.clip(h, 0.0, 1.0)
    return h

    best_c5 = np.max(O) * dx

    h = h_init.copy()

def project_onto_bounds_and_sum(h, M):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = M - cur
    if abs(diff) < 1e-12:
        return h
    # Distribute diff proportionally to free variables
    free = (h > 0) & (h < 1)
    n_free = free.sum()
    if n_free > 0:
        # limit each addition to not exceed bounds
        # compute possible increase/decrease for each free variable
        if diff > 0:
            caps = 1.0 - h[free]
            total_cap = caps.sum()
            if total_cap < diff:
                # cannot fill diff, set all free to 1
                h[free] = 1.0
                diff -= total_cap
                # distribute remaining diff uniformly
                h += diff / len(h)
            else:
                # fill diff proportionally
                h[free] += diff * caps / total_cap
        else:
            caps = h[free]  # amount can decrease
            total_cap = caps.sum()
            if total_cap < -diff:
                h[free] = 0.0
                diff += total_cap
                h += diff / len(h)
            else:
                h[free] += diff * caps / total_cap
        # clip again just in case
        h = np.clip(h, 0.0, 1.0)
        # final adjustment
        cur = h.sum()
        diff = M - cur
        if abs(diff) > 1e-12:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    else:
        # all at bounds; adjust uniformly
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def _project_bounds_sum(h, target_sum):
    """Clip to [0,1] and adjust to hit target_sum (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def _full_c5_fft(h, L, const_fft_ones):
    """C5 computed via zero‑padded FFT (exact full correlation)."""
    n = h.size
    H = np.fft.fft(h, L)
    B = const_fft_ones - H                      # FFT(1 - h)
    corr = np.real(np.fft.ifft(H * np.conj(B)))[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def _optimize_h(n, h0, time_budget, rng):
    """Gradient descent on a soft‑max surrogate of the max overlap."""
    M = n // 2
    dx = 2.0 / n

    # FFT size for full (zero‑padded) correlation
    L = 1
    while L < 2 * n:
        L <<= 1
    const_fft_ones_L = np.fft.fft(np.ones(L))
    const_fft_ones_n = np.fft.fft(np.ones(n))

    h = _project_bounds_sum(h0, M)

    # initial best (full C5)
    best_h = h.copy()
    best_c5 = _full_c5_fft(best_h, L, const_fft_ones_L)

    start = time.time()
    it = 0
    max_iters = 10_000_000

    while time.time() - start < time_budget and it < max_iters:
        # ----- cyclic overlap O (smooth surrogate) -----
        FH = np.fft.fft(h)                     # length n
        Bc = const_fft_ones_n - FH             # FFT(1 - h) on length n
        O_c = np.real(np.fft.ifft(FH * np.conj(Bc)))   # cyclic overlap vector

        # soft‑max weights, slowly increasing β
        beta = min(500.0, 0.5 + it * 0.0025)
        Omax = O_c.max()
        exp_vals = np.exp(beta * (O_c - Omax))
        w = exp_vals / exp_vals.sum()

        # ----- gradient of weighted sum of O -----
        FW = np.fft.fft(w)
        conv_b = np.real(np.fft.ifft(FH * FW))          # Σ w_k h_{i-k}
        conv_f = np.real(np.fft.ifft(FH * np.conj(FW))) # Σ w_k h_{i+k}
        grad = 1.0 - conv_f - conv_b

        # ----- gradient step with diminishing step size -----
        step = 0.25 / (1.0 + it * 0.001)
        h_new = h - step * grad
        h = _project_bounds_sum(h_new, M)

        # ----- evaluate true C5 (full correlation) -----
        c5 = _full_c5_fft(h, L, const_fft_ones_L)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 < 0.3808:
                break

        # occasional random perturbation to escape flat regions
        if it % 5000 == 0:
            h = best_h + 0.02 * (rng.random(n) - 0.5)
            h = _project_bounds_sum(h, M)

        it += 1

    return best_h, best_c5

    for idx, n in enumerate(candidates):
