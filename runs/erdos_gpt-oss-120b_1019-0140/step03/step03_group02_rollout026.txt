def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds: v - lam is clipped to [lo, hi]
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)
```

def grad_softmax_corr(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # enumerate shifts
    for idx_shift, k in enumerate(range(-(n-1), n)):
        w = weights[idx_shift]
        if w == 0.0:
            continue
        if k >= 0:
            # positions i from 0 to n-k-1: term w*(1 - h[i+k]) adds to grad[i]
            grad[:n-k] += w * (1.0 - h[k:])
            # positions i from k to n-1: term -w*h[i-k] subtracts for grad[i]
            grad[k:] -= w * h[:n-k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad
```

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n-shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n-s]
            grad[:n-s] -= h[s:]
    return grad / len(shifts) if shifts else grad
```

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    arr = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def resample(arr, new_n):
    old_n = arr.size
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, arr)

def grad_softmax_corr(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx_shift, k in enumerate(range(-(n - 1), n)):
        w = weights[idx_shift]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    if not shifts:
        return np.zeros_like(h)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    return grad / len(shifts)

def smooth_grad_descent(h0, n_iter=5000, step0=0.5, alpha_start=0.5,
                        alpha_end=20.0, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

                    best_score, best_h = sc_ref, h_ref.copy()
            # binary hill climbing
            if time_left() > 4.0:
                order = np.argsort(best_h)
                bin_vec = np.zeros(n, dtype=float)
                bin_vec[order[n // 2:]] = 1.0
                h_bin, sc_bin = binary_hill_climb_sa(bin_vec, rng, time_left,
                                                     target=0.3808)
                if sc_bin < best_score:
                    best_score, best_h = sc_bin, h_bin.copy()
                # final small momentum refinement
                if time_left() > 2.0:
                    max_iters = int(min(6000, time_left() * 5))
                    h_ref2, sc_ref2 = momentum_grad_descent(best_h, n_iter=max_iters,
                                                            step0=0.3, beta=0.9,
                                                            time_left=time_left)
                    if sc_ref2 < best_score:
                        best_score, best_h = sc_ref2, h_ref2.copy()

    # final safety projection
    if best_h is None:
        best_h = project_onto_box_sum(np.full(64, 0.5), 32)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
