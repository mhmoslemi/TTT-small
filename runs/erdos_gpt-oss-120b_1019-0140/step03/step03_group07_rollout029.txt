    # initial h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:

        h = project_box_sum(h, M)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    max_iter = 50000  # enough
    # compute allowed iterations based on time budget
    # each iteration ~ O(n^2) cheap; approximate time per iteration ~ 0.0002 s for n=128.
    # We'll just run until time left.

    iter = 0
    while time.time() - start < time_budget - 0.1 and iter < max_iter:

        # Compute correlation
        g = 1.0 - h
        corr = np.convolve(h, g, mode='full')
        # compute softmax weights
        # schedule alpha
        t_frac = iter / max_iter
        alpha = 0.5 + (50.0 - 0.5) * (t_frac ** 2)
        # Use stable softmax
        scaled = alpha * corr
        # logsumexp for scaling
        lse = np.max(scaled) + np.log(np.sum(np.exp(scaled - np.max(scaled))))

        grad2 = np.convolve(w, h, mode='full')[n-1:n-1+n]

        # step size schedule
        lr = 0.08 * (1 - t_frac) + 0.001 * t_frac
        # gradient descent step
        step = lr * grad / (np.linalg.norm(grad) + 1e-12)

import time
import numpy as np
from scipy.special import logsumexp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(x, M):
    # Find lambda so that sum(clip(x+lambda,0,1)) = M via bisection
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < M:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, M, rng):
    x = rng.random(n)
    return project_box_sum(x, M)

def optimise_n(n, rng, time_budget):
    M = n / 2.0
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        h = project_box_sum(h, M)
    else:
        h = random_feasible(n, M, rng)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    last_improve = start

    while time.time() - start < time_budget - 0.1:
        elapsed = time.time() - start
        t_frac = elapsed / time_budget

        g = 1.0 - h

    best_n = None
