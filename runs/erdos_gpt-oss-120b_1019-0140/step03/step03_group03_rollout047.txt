def upsample(arr, new_n):
    m = arr.size
    if m == new_n: return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_momentum(h0, max_iters, step0, beta, time_left):
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        conv = correlation_vals(h)
        max_conv = conv.max()
        # find shifts where conv close to max:
        eps = 1e-12
        idxes = np.where(conv >= max_conv - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        # compute score
        score = np.max(correlation_vals(h_new)) * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
