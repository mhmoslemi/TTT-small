def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.8, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        # full correlation using convolution
        conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
        max_corr = np.max(conv)
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

        if best_score <= target:
            break

    if best_h is not None:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:

            return best_h, best_score, best_h.size

    if best_h is not None and np.all((best_h == 0) | (best_h == 1)):
        # best_h is binary candidate
        h_bin = best_h.copy()
    else:
        # ensure binary starting point
        if best_h is None:

            h_bin = random_binary_balanced(n_target, rng).astype(float)
        else:
            # already binary analog above
            h_bin = best_h.copy()
    # Hill climbing
    if time_left() > 0.5:
        h_bin_ref, sc_bin_ref = binary_hill_climb(h_bin, rng, time_left, max_trials=500, max_no_improve=5)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Random binary restarts with hill climbing
    while time_left() > 5.0:
        h_rand = random_binary_balanced(n_target, rng).astype(float)
        h_ref, sc_ref = binary_hill_climb(h_rand, rng, time_left, max_trials=500, max_no_improve=5)
        if sc_ref < best_score:

            if best_score <= target:

    if best_h is not None and not np.all((best_h == 0) | (best_h == 1)):
        best_h = project_onto_box_sum(best_h, best_h.size/2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return np.max(conv) * dx

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.8, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
        max_corr = np.max(conv)
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_hill_climb(h0, rng, time_left_func,
                      max_trials=200, max_no_improve=5):
    """Greedy hill climbing on a balanced binary vector."""
    h = h0.copy()
    current_score = compute_c5(h)
    no_improve = 0
    while time_left_func() > 0.0 and no_improve < max_no_improve:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        improved = False
        for _ in range(max_trials):
            if time_left_func() <= 0.0:
                break
            i = ones[rng.integers(ones.size)]
            j = zeros[rng.integers(zeros.size)]
            h[i] = 0.0
            h[j] = 1.0
            new_score = compute_c5(h)
            if new_score < current_score:
                current_score = new_score
                improved = True
                break
            h[i] = 1.0
            h[j] = 0.0
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return h, current_score

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)
