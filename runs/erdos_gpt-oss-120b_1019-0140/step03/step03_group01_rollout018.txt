def continuous_anneal(h0, rng, time_left_func, T0=0.01, T_min=1e-5, decay=0.99995, max_iters=200000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        # randomly select i,j distinct
        i, j = rng.choice(n, size=2, replace=False)
        # compute feasible delta interval
        lower = max(-h[i], h[j] - 1.0)
        upper = min(1.0 - h[i], h[j])
        if lower >= upper:
            it += 1
            continue
        delta = rng.uniform(lower, upper)
        h[i] += delta
        h[j] -= delta
        # compute new_score
        new_score = compute_c5(h)
        delta_score = new_score - current_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta_score / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        T = max(T_min, T * decay)
        it += 1
        # optional early break if T becomes very small and no improvements
        # but keep going
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

            term2 = h[:n - s]           # h[i-s]
            grad[s:] += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def continuous_anneal(h0, rng, time_left_func, T0=0.01, T_min=1e-5, decay=0.99995, max_iters=200000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        i, j = rng.choice(n, size=2, replace=False)
        lower = max(-h[i], h[j] - 1.0)
        upper = min(1.0 - h[i], h[j])
        if lower < upper:
            delta = rng.uniform(lower, upper)
            h[i] += delta
            h[j] -= delta
            new_score = compute_c5(h)
            dscore = new_score - current_score
            accept = False
            if dscore < 0.0:
                accept = True
            else:
                if T > 0.0 and rng.random() < np.exp(-dscore / T):
                    accept = True
            if accept:
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=800):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # ---------- Phase 1: small-grid SLSQP ----------
    n0 = 64
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        candidates.append(project_onto_box_sum(downsample(init_full, n0), n0 / 2.0))
    candidates.append(thue_morse(n0))
    candidates.append(rudin_shapiro(n0))
    candidates.append(sinusoid_initial(n0))
    for _ in range(2):
        candidates.append(random_binary_balanced(n0, rng))
    for _ in range(2):
        candidates.append(random_continuous_balanced(n0, rng))
    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=1200)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 2: medium grid upsample ----------
    n_mid = 256
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n0, 0.5), n0 / 2.0)
        best_score = compute_c5(best_h)
    h_mid = upsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    sc_mid = compute_c5(h_mid)
    if sc_mid < best_score:
        best_score, best_h = sc_mid, h_mid.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 3: momentum gradient descent ----------
    if time_left() > 5.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left,
                                                            step0=0.5, beta=0.9,
                                                            max_iter=200000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 4: continuous annealing ----------
    if time_left() > 5.0:
        T0 = 0.02 * best_score
        h_ann, sc_ann = continuous_anneal(best_h, rng, time_left,
                                         T0=T0, T_min=1e-5, decay=0.9999,
                                         max_iters=300000)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 5: fine grid refinement ----------
    if time_left() > 10.0:
        n_fine = 512
        h_fine = upsample(best_h, n_fine)
        h_fine = project_onto_box_sum(h_fine, n_fine / 2.0)
        h_ref2, sc_ref2 = gradient_descent_momentum_with_time(h_fine, time_left,
                                                              step0=0.3, beta=0.9,
                                                              max_iter=150000)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 6: random continuous restarts ----------
    while time_left() > 8.0:
        n_rand = 256
        v = rng.random(n_rand)
        v = project_onto_box_sum(v, n_rand / 2.0)
        h_tmp, sc_tmp = gradient_descent_momentum_with_time(v, time_left,
                                                             step0=0.4, beta=0.9,
                                                             max_iter=40000)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
