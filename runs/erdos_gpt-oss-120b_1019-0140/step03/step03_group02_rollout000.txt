def grad_softmax_corr(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    shifted = corr - maxc
    exp_vals = np.exp(alpha * shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # For each shift k (range -(n-1) to n-1):
    # index s in corr corresponds to k = s - (n-1)
    for s, k in enumerate(range(-(n-1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            # h[0:n-k] term
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h, best_score = h.copy(), curr_score
    # generate list of zero and ones positions for quick random selection
    zeros = np.flatnonzero(h == 0.0)
    ones = np.flatnonzero(h == 1.0)
    # set max_iters based on time left
    max_iters = int(min(500000, max(1, time_left() * 3000)))  # rough
    T0 = 0.01  # maybe higher temperature
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        # random selection
        i = rng.integers(0, zeros.size)
        j = rng.integers(0, ones.size)
        idx0 = zeros[i]
        idx1 = ones[j]
        # Swap
        h[idx0], h[idx1] = 1.0, 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / (T+1e-12))
        if accept:
            curr_score = new_score
            # update zeros/ones arrays
            zeros[i] = idx1
            ones[j] = idx0
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
                if best_score <= target:
                    break
        else:
            # revert swap
            h[idx0], h[idx1] = 0.0, 1.0
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)  # use same linear interpolation.

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # Clip sum lower/upper bound
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Determine lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Define f(lambda)
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Adjust bounds so that f(lam_low) >= s >= f(lam_high)
    # Compute f at lows
    f_low = f(lam_low)
    f_high = f(lam_high)
    # If f_low < s, need lower lam to increase f: extend lam_low.
    if f_low < s:
        # decrease lam_low
        lam_low = np.min(v) - 10*hi
        f_low = f(lam_low)

        lam_high = np.max(v) - 10*lo

        if f_mid > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    f_low = f(lam_low)
    f_high = f(lam_high)
    if f_low < s:
        lam_low = np.min(v) - 10 * hi
        f_low = f(lam_low)
    if f_high > s:
        lam_high = np.max(v) - 10 * lo
        f_high = f(lam_high)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def grad_softmax_corr(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    shifted = corr - maxc
    exp_vals = np.exp(alpha * shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=4000, step0=0.8,
                       alpha_start=1.0, alpha_end=30.0,
                       beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    mom = np.zeros_like(h)
    best_h, best_score = h.copy(), compute_c5(h)
    n = h.size
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent(h0, n_iter=5000, step0=0.5,
                         alpha=30.0, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    mom = np.zeros_like(h)
    best_h, best_score = h.copy(), compute_c5(h)
    n = h.size
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr = compute_c5(h)
    best_h, best = h.copy(), curr
    zeros = np.flatnonzero(h == 0.0)
    ones = np.flatnonzero(h == 1.0)
    max_iters = int(min(500000, max(1, time_left() * 3000)))
    T0 = 0.01
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        i = rng.integers(0, zeros.size)
        j = rng.integers(0, ones.size)
        iz, jo = zeros[i], ones[j]
        h[iz], h[jo] = 1.0, 0.0
        new = compute_c5(h)
        delta = new - curr
        T = T0 * (1.0 - it / max_iters)
        accept = delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12))
        if accept:
            curr = new
            zeros[i], ones[j] = jo, iz
            if new < best:
                best, best_h = new, h.copy()
                if best <= target:
                    break
        else:
            h[iz], h[jo] = 0.0, 1.0
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small = 64
    n_large = 384
    target = 0.3808

    # ----- generate candidates at small scale -----
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_small = init if init.size == n_small else downsample(init, n_small)
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small).astype(float))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h, best_score = None, np.inf

    # ----- small‑scale smooth descent -----
    for h0 in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = smooth_grad_descent(h0,
                                            n_iter=4000,
                                            step0=0.8,
                                            alpha_start=1.0,
                                            alpha_end=30.0,
                                            beta=0.9,
                                            time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5(best_h)

    # ----- upsample to large scale -----
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    up_sc = compute_c5(h_up)
    if up_sc < best_score:
        best_score, best_h = up_sc, h_up.copy()

    # ----- large‑scale momentum descent -----
    if time_left() > 3.0:
        max_iters = int(min(50000, max(2000, time_left() * 800)))
        h_cont, sc_cont = momentum_grad_descent(best_h,
                                                n_iter=max_iters,
                                                step0=0.6,
                                                alpha=30.0,
                                                beta=0.9,
                                                time_left=time_left)
        if sc_cont < best_score:
            best_score, best_h = sc_cont, h_cont.copy()

    # ----- binary hill‑climbing -----
    if time_left() > 2.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=target)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
        # final continuous fine‑tuning
        if time_left() > 1.0:
            h_fine, fine_sc = momentum_grad_descent(best_h,
                                                    n_iter=5000,
                                                    step0=0.3,
                                                    alpha=30.0,
                                                    beta=0.9,
                                                    time_left=time_left)
            if fine_sc < best_score:
                best_score, best_h = fine_sc, h_fine.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
