    v = np.clip(v, 0.0, 1.0)

    total = v.sum()

        return v

    # Approach: we want to solve min ||y - v||^2 s.t. sum(y)=target, 0<=y<=1.
    # Use method: Initialize y=v; while True: compute diff = sum(y)-target.
    # If diff>0, need to reduce sum; can reduce values where y>0.
    # If diff<0, need to increase sum; can increase values where y<1.
    # Compute set of free variables: those not at bound 0 (if diff>0) or not at bound 1 (if diff<0).
    # Then distribute diff uniformly among free variables. Then clip again. Iterate.
    # This converges fast, because each iteration adds bounds.

def gradient_worst_shift(h, k):
    n = h.shape[0]
    grad = np.zeros(n)
    if k >= 0:

        grad[k:] -= h[:n - k]

        l = -k
        grad[l:] = 1 - h[:n - l]
        grad[:n - l] -= h[l:]
    return grad

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_onto_capped_simplex(v, target):
    """Euclidean projection of v onto {0<=y<=1, sum(y)=target}."""
    low, high = -1.0, 1.0  # enough to drive all entries to 0 or 1
    for _ in range(50):
        lam = (low + high) / 2.0
        y = np.clip(v + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = lam
        else:
            low = lam
    y = np.clip(v + (low + high) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = np.where((y > 0) & (y < 1))[0]
        if free.size:
            y[free] += diff / free.size
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_worst_shift(h, k):
    """Gradient of the overlap for the shift k that currently attains the max."""
    n = h.shape[0]
    grad = np.zeros(n)
    if k >= 0:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        l = -k
        grad[l:] = 1.0 - h[: n - l]
        grad[: n - l] -= h[l:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # problem size (increase for finer discretization)
    n = 200
    target = n / 2.0

    # initialise h (use provided parent if present)
    if "initial_h_values" in globals():
        h_small = np.array(initial_h_values, dtype=float)
        n0 = h_small.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h_small)
    else:
        h = rng.random(n)

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_capped_simplex(h, target)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # parameters for the descent/annealing
    alpha0 = 0.20
    alpha_min = 0.001
    no_improve = 0

    while time.time() < deadline - 0.1:
        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # gradient step
        grad = gradient_worst_shift(cur_h, k)
        elapsed = time.time() - start
        tfrac = elapsed / budget_s
        alpha = max(alpha_min, alpha0 * (1.0 - tfrac))
        cand = cur_h - alpha * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_onto_capped_simplex(cand, target)

        new_val = compute_c5(cand)

        # acceptance
        if new_val < cur_val:
            cur_h, cur_val = cand, new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_h = new_val, cand.copy()
                if best_val <= 0.3808:
                    break
        else:
            # simulated‑annealing style acceptance
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_h, cur_val = cand, new_val
                no_improve = 0
            else:
                no_improve += 1

        # occasional random jitter to escape plateaus
        if rng.random() < 0.01:
            i, j = rng.choice(n, size=2, replace=False)
            delta = (rng.random() - 0.5) * 0.10
            if 0.0 <= cur_h[i] + delta <= 1.0 and 0.0 <= cur_h[j] - delta <= 1.0:
                cur_h[i] += delta
                cur_h[j] -= delta
                cur_h = project_onto_capped_simplex(cur_h, target)
                cur_val = compute_c5(cur_h)

        # restart if stuck
        if no_improve >= 5000:
            if rng.random() < 0.5 and best_val < np.inf:
                cur_h = best_h + rng.normal(scale=0.02, size=n)
            else:
                cur_h = rng.random(n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = project_onto_capped_simplex(cur_h, target)
            cur_val = compute_c5(cur_h)
            no_improve = 0

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
