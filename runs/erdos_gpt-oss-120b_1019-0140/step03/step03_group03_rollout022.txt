def sa_binary(h_start, rng, time_limit, start_temp=0.1, end_temp=1e-5):
    h = h_start.copy()
    best = h.copy()
    best_score = compute_c5_fast(best)
    cur_score = best_score

    # Determine number of total iterations: roughly time*X per iteration. But easier: while time remaining > 0, do loops.

    start = time.time()
    elapsed = time.time() - start
    total_time = time_limit
    # use log-uniform temperature schedule: T = start_temp * (end_temp/start_temp)^(elapsed/total_time)
    while time.time() - start < total_time:
        # compute current temperature
        t_frac = (time.time() - start) / total_time
        T = start_temp * (end_temp / start_temp) ** t_frac

        # choose random i from ones, j from zeros
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_fast(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            if rng.random() < np.exp(-delta / (T + 1e-12)):

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = fftconvolve(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def upsample_linear(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_time_limited(h0, max_time):
    """Projected gradient descent with momentum, limited by wall‑clock time."""
    n = h0.shape[0]
    h = h0.copy()
    target = n / 2.0
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.5
    best_h = h.copy()
    best_sc = compute_c5_fast(h)
    start = time.time()
    it = 0
    while time.time() - start < max_time:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5_fast(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def sa_binary(start_h, rng, duration, start_temp=0.5, end_temp=1e-4):
    """Simulated annealing on binary vectors."""
    h = start_h.copy()
    cur_sc = compute_c5_fast(h)
    best_h, best_sc = h.copy(), cur_sc
    start = time.time()
    while time.time() - start < duration:
        t_frac = (time.time() - start) / duration
        T = start_temp * (end_temp / start_temp) ** t_frac
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_sc = compute_c5_fast(h)
        delta = new_sc - cur_sc
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (T + 1e-12)):
                accept = True
        if accept:
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_h = new_sc, h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    n = 512
    candidates = []

    # initial construction, if provided
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_up = upsample_linear(init, n)
        init_up = project_onto_box_sum(init_up, n / 2.0)
        candidates.append(init_up)

    # random continuous seeds
    for _ in range(2):
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        candidates.append(v)

    # random binary seeds
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng).astype(float))

    best_h = None
    best_sc = np.inf

    def consider(h):
        nonlocal best_h, best_sc
        sc = compute_c5_fast(h)
        if sc < best_sc:
            best_sc, best_h = sc, h.copy()

    # Phase 1: continuous refinement (gradient descent) and binary rounding
    for cand in candidates:
        if time_left() < 5:
            break
        if np.all((cand == 0) | (cand == 1)):
            consider(cand)
            continue
        max_gd = min(time_left() * 0.2, 25.0)
        h_opt, sc_opt = gradient_descent_time_limited(cand, max_gd)
        consider(h_opt)

        # binary rounding of the refined continuous vector
        idx = np.argpartition(h_opt, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n, dtype=float)
        bin_vec[idx] = 1.0
        consider(bin_vec)

    # Fallback if nothing succeeded yet
    if best_h is None:
        best_h = random_binary_balanced(n, rng).astype(float)
        best_sc = compute_c5_fast(best_h)

    # Phase 2: repeated simulated annealing runs
    while time_left() > 10:
        t_alloc = min(time_left() * 0.5, 20.0)
        start_vec = best_h if rng.random() < 0.7 else random_binary_balanced(n, rng).astype(float)
        h_sa, sc_sa = sa_binary(start_vec, rng, t_alloc, start_temp=0.5, end_temp=1e-4)
        if sc_sa < best_sc:
            best_sc, best_h = sc_sa, h_sa.copy()
        if time_left() < 5:
            break

    # Phase 3: greedy hill climbing (only improvements)
    while time_left() > 2:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i] = 0
        best_h[j] = 1
        new_sc = compute_c5_fast(best_h)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            best_h[i] = 1
            best_h[j] = 0

    # Ensure exact binary form and correct sum
    best_h = np.where(best_h > 0.5, 1.0, 0.0)
    cur_sum = int(best_h.sum())
    half = n // 2
    if cur_sum > half:
        excess = cur_sum - half
        idx_one = np.flatnonzero(best_h == 1)
        to_zero = rng.choice(idx_one, size=excess, replace=False)
        best_h[to_zero] = 0.0
    elif cur_sum < half:
        deficit = half - cur_sum
        idx_zero = np.flatnonzero(best_h == 0)
        to_one = rng.choice(idx_zero, size=deficit, replace=False)
        best_h[to_one] = 1.0

    best_sc = compute_c5_fast(best_h)
    return best_h, float(best_sc), n
