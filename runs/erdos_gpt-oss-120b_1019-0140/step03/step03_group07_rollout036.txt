import time
import math

import numpy as np
from scipy.optimize import minimize


def compute_c5(h):
    """Return C5 for a given step function h (array of samples)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_grads(h):
    """Return array of gradients of each non‑zero shift overlap w.r.t. h."""
    n = h.shape[0]
    num_shifts = 2 * n - 2
    grads = np.zeros((num_shifts, n), dtype=float)
    idx = 0
    for k in range(-n + 1, n):
        if k == 0:
            continue
        if k > 0:
            grads[idx, : n - k] = 1.0 - h[k:]
            grads[idx, k:] = -h[: n - k]
        else:
            kk = -k
            grads[idx, kk:] = 1.0 - h[: n - kk]
            grads[idx, : n - kk] = -h[kk:]
        idx += 1
    return grads


def project_onto_capped_simplex(v, target_sum, tol=1e-12, max_iter=60):
    """
    Euclidean projection of v onto the set
        {w : 0 <= w_i <= 1, sum w_i = target_sum}.
    Uses bisection on the threshold lambda.
    """
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(w)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


class SumEquality:
    """Equality constraint: sum(h) == target_sum."""

    def __init__(self, n):
        self.n = n
        self.target = n / 2.0

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target


class OverlapInequality:
    """Inequality constraint: t >= overlap(k) * dx."""

    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


def slsqp_objective(x):
    return x[-1]


def slsqp_refine(h_initial, n, dx, time_limit):
    """Refine a candidate using SLSQP (if time permits)."""
    if time.time() > time_limit - 1.0:
        return h_initial, compute_c5(h_initial)
    M = n // 2
    t0 = compute_c5(h_initial)
    x0 = np.concatenate([h_initial, [t0]])
    sum_con = SumEquality(n)
    overlap_cons = [OverlapInequality(k, n, dx) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": c} for c in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = minimize(
            slsqp_objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 800, "ftol": 1e-12, "disp": False},
        )
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt = project_onto_capped_simplex(h_opt, M)
            c5_opt = compute_c5(h_opt)
            return h_opt, c5_opt
    except Exception:
        pass
    return h_initial, compute_c5(h_initial)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = math.inf
    best_n = None
    target = 0.3808

    candidates = [80, 96, 112, 128]

    # Give priority to a parent construction if present
    if "initial_h_values" in globals():
        n_parent = len(initial_h_values)
        if n_parent % 2 == 0:
            if n_parent in candidates:
                candidates.remove(n_parent)
            candidates = [n_parent] + candidates

    for n in candidates:
        if time.time() - start_time > budget_s - 3.0:
            break
        M = n // 2
        dx = 2.0 / n

        # allocate a slice of the remaining budget to this n
        remaining = budget_s - (time.time() - start_time)
        alloc = remaining * 0.3  # 30 % of what is left
        n_restart = max(1, int(alloc // 6.0))  # roughly 6 s per restart

        best_h_n = None
        best_c5_n = math.inf

        for _ in range(n_restart):
            if time.time() - start_time > budget_s - 2.0:
                break

            # ----- initialise candidate -----
            if (
                "initial_h_values" in globals()
                and len(initial_h_values) == n
            ):
                h = np.array(initial_h_values, dtype=float)
                h = project_onto_capped_simplex(h, M)
            else:
                h = np.zeros(n)
                idx = rng.choice(n, M, replace=False)
                h[idx] = 1.0

            # ----- gradient‑descent stage -----
            max_iter = 4000
            step0 = 0.5
            beta = 8.0
            center = n - 1

            local_best_h = h.copy()
            local_best_c5 = compute_c5(h)

            for it in range(max_iter):
                if time.time() - start_time > budget_s - 1.5:
                    break

                corr = np.correlate(h, 1.0 - h, mode="full")
                c5 = np.max(corr) * dx
                if c5 + 1e-12 < local_best_c5:
                    local_best_c5 = c5
                    local_best_h = h.copy()
                    if c5 < target:
                        break

                grads = compute_grads(h)

                # soft‑max over overlaps (zero out the k=0 term)
                max_corr = np.max(corr)
                w = np.exp(beta * (corr - max_corr))
                w[center] = 0.0
                w = np.delete(w, center)  # now length 2n‑2
                w_sum = np.sum(w)
                if w_sum > 0.0:
                    w /= w_sum
                else:
                    w = np.full_like(w, 1.0 / w.size)

                grad_f = np.dot(w, grads)

                step = step0 * (0.999 ** it)
                h_new = h - step * grad_f
                h = project_onto_capped_simplex(h_new, M)

            # ----- optional SLSQP refinement -----
            if time.time() - start_time < budget_s - 1.0:
                h_ref, c5_ref = slsqp_refine(local_best_h, n, dx, budget_s - (time.time() - start_time))
                if c5_ref + 1e-12 < local_best_c5:
                    local_best_c5 = c5_ref
                    local_best_h = h_ref

            # ----- update best for this n -----
            if local_best_c5 < best_c5_n:
                best_c5_n = local_best_c5
                best_h_n = local_best_h.copy()
                if best_c5_n < target:
                    break

        # ----- compare with global best -----
        if best_c5_n < best_c5:
            best_c5 = best_c5_n
            best_h = best_h_n.copy()
            best_n = n
            if best_c5 < target:
                break

    # ----- final safety net -----
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Ensure final feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, best_n // 2)

    return best_h, best_c5, best_n
