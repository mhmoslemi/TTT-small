def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # binary search for lambda such that sum(clip(v+lambda, lo, hi)) = target
    lo_lambda = -np.max(v) - 5.0
    hi_lambda = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (lo_lambda + hi_lambda) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            lo_lambda = mid
        else:
            hi_lambda = mid
    return np.clip(v + (lo_lambda + hi_lambda) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)

        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]

    return grad

from scipy.optimize import minimize

def slsqp_optimize(h_start, t_start, n, target_sum, dx, maxiter=500):
    sum_con = lambda x: np.sum(x[:n]) - target_sum
    overlap_cons = []
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_con(k):
            def con(x):
                h = x[:n]
                if k >= 0:
                    ov = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    ov = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - ov * dx
            return con
        overlap_cons.append({'type': 'ineq', 'fun': make_con(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_con}] + overlap_cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst‑shift overlap w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def slsqp_optimize(h_start, t_start, n, target_sum, dx, maxiter=300):
    """Refine a feasible vector using SLSQP with a slack variable t."""
    # equality sum constraint
    def sum_con(x):
        return np.sum(x[:n]) - target_sum

    # inequality t >= overlap(k)*dx for each shift k≠0
    def make_ineq(k):
        def con(x, k=k):
            h = x[:n]
            if k >= 0:
                ov = np.dot(h[:n - k], 1.0 - h[k:])
            else:
                kk = -k
                ov = np.dot(h[kk:], 1.0 - h[:n - kk])
            return x[-1] - ov * dx
        return con

    ineq = [{'type': 'ineq', 'fun': make_ineq(k)} for k in range(-n + 1, n) if k != 0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])

    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': sum_con}] + ineq,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise problem size and feasible start -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- quick gradient‑descent refinement (≈20% of time) -----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    alpha = 0.2
    min_alpha = 1e-6
    gd_iters = 3000
    gd_deadline = start + budget_s * 0.2
    for _ in range(gd_iters):
        if time.time() > gd_deadline:
            break
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = gradient_for_shift(cur_h, k)
        new_h = cur_h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                cur_h = project_to_box_sum(
                    np.clip(cur_h + rng.normal(scale=0.02, size=n), 0.0, 1.0),
                    target_sum
                )
                alpha = 0.2
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
            if best_c5 < 0.3808:
                break

    # update from final gd state
    cur_c5 = compute_c5(cur_h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = cur_h.copy()

    # ----- simulated annealing phase -----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    dx = 2.0 / n
    step = 0.2      # max mass transferred per move
    T0 = 0.001
    Tmin = 1e-7
    accept_counter = 0
    iter_cnt = 0

    def corr_increment(h, i, delta_i, j, delta_j):
        inc = np.zeros(2 * n - 1)
        for k in range(-n + 1, n):
            idx = k + n - 1
            ipk = i + k
            if 0 <= ipk < n:
                inc[idx] += delta_i * (1.0 - h[ipk])
            imk = i - k
            if 0 <= imk < n:
                inc[idx] -= delta_i * h[imk]
            jpk = j + k
            if 0 <= jpk < n:
                inc[idx] += delta_j * (1.0 - h[jpk])
            jmk = j - k
            if 0 <= jmk < n:
                inc[idx] -= delta_j * h[jmk]
        return inc

    while time.time() < start + budget_s - 5:
        iter_cnt += 1
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        max_up = min(step, 1.0 - cur_h[i], cur_h[j])
        max_down = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_up + max_down == 0.0:
            continue
        delta = rng.random() * (max_up + max_down) - max_down
        if abs(delta) < 1e-12:
            continue
        delta_i = delta
        delta_j = -delta
        inc = corr_increment(cur_h, i, delta_i, j, delta_j)
        new_corr = corr + inc
        new_c5 = np.max(new_corr) * dx
        T = max(Tmin, T0 * (0.9995 ** iter_cnt))
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            # accept move
            cur_h[i] += delta_i
            cur_h[j] += delta_j
            # enforce exact bounds (numerical safety)
            cur_h[i] = min(1.0, max(0.0, cur_h[i]))
            cur_h[j] = min(1.0, max(0.0, cur_h[j]))
            corr = new_corr
            cur_c5 = new_c5
            accept_counter += 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 < 0.3808:
                    break
        # adapt step every 1000 iterations
        if iter_cnt % 1000 == 0:
            if accept_counter > 600:
                step = min(step * 1.2, 0.5)
            elif accept_counter < 200:
                step = max(step * 0.8, 0.01)
            accept_counter = 0

    # ----- final SLSQP refinement if time permits -----
    remaining = start + budget_s - 5 - time.time()
    if remaining > 10.0:
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, n, target_sum, dx, maxiter=300)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
