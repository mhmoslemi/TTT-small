import time
import numpy as np

# ------------------- helper functions -------------------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Project v into [0,1]^n with exact sum == target (within ~1e-12)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute the tiny residual using a few random adjustments
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 500:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def smooth_c5_and_grad(h, alpha, top_m):
    """Smooth approximation of max overlap and its gradient (only top_m shifts)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")          # length 2n-1
    O = conv * dx                                        # actual overlap values
    Omax = np.max(O)
    w = np.exp(alpha * (O - Omax))                      # softmax weights (stable)
    w_sum = np.sum(w)
    w /= w_sum

    # select top_m shifts (by weight) to compute gradient efficiently
    if top_m < w.size:
        idx = np.argpartition(-w, top_m - 1)[:top_m]
    else:
        idx = np.arange(w.size)

    g = np.zeros(n)
    for s in idx:
        wk = w[s]
        k = s - (n - 1)  # actual shift
        if k >= 0:
            nk = n - k
            if nk > 0:
                g[:nk] += wk * (1.0 - h[k:])   # derivative of the linear term
                g[k:]  -= wk * h[:nk]          # derivative of the quadratic term
        else:
            kk = -k
            nk = n - kk
            if nk > 0:
                g[kk:] += wk * (1.0 - h[:nk])
                g[:nk] -= wk * h[kk:]
    g *= dx
    smooth_val = np.dot(w, O)   # softmax-weighted overlap
    return smooth_val, g

# ------------------- main entry point -------------------
