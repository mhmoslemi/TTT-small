Implementation of gradient_max_corr:

```
def gradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    # indices where corr is within epsilon of max
    eps = 1e-12
    idxes = np.where(corr >= maxc - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad
```

def momentum_descent(h0, max_iter, step0, beta, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() < 0.01:
            break
        grad = gradient_max_corr(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    curr = compute_c5(h)
    best = h.copy()
    best_score = curr
    # determine max_iters
    max_iters = int(min(2_000_000, max(1000, time_left() * 5000)))
    T0 = 0.02
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # pick random zero index
        zero_idx = rng.integers(0, n)
        while h[zero_idx] != 0.0:
            zero_idx = rng.integers(0, n)
        one_idx = rng.integers(0, n)
        while h[one_idx] != 1.0:
            one_idx = rng.integers(0, n)
        # swap
        h_new = h.copy()
        h_new[zero_idx] = 1.0
        h_new[one_idx] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr
        T = T0 * max(0.0, 1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr = new_score
            if new_score < best_score:
                best_score = new_score
                best = h_new.copy()
                if best_score <= target:
                    break
    return best, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def gradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= maxc - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad

def momentum_descent(h0, max_iter, step0, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() < 0.01:
            break
        grad = gradient_max_corr(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def continuous_sa(h0, rng, time_left, target=0.3808):

    n = h.size
    curr = compute_c5(h)
    best = h.copy()
    best_score = curr
    max_iters = int(min(200000, max(1000, time_left() * 4000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        delta = (rng.random() - 0.5) * 0.2
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        new_score = compute_c5(h_new)
        dscore = new_score - curr

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def gradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= maxc - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad

def momentum_descent(h0, max_iter, step0, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() < 0.01:
            break
        grad = gradient_max_corr(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def continuous_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    curr = compute_c5(h)
    best = h.copy()
    best_score = curr
    max_iters = int(min(200000, max(1000, time_left() * 4000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        delta = (rng.random() - 0.5) * 0.2
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        new_score = compute_c5(h_new)
        dscore = new_score - curr
        T = T0 * max(0.0, 1.0 - it / max_iters)
        if dscore < 0.0 or (T > 0.0 and rng.random() < np.exp(-dscore / T)):
            h, curr = h_new, new_score
            if new_score < best_score:
                best_score, best = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr = compute_c5(h)
    best = h.copy()
    best_score = curr
    max_iters = int(min(500000, max(2000, time_left() * 5000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:

        zero_idx = rng.integers(0, n)
        while h[zero_idx] != 0.0:
            zero_idx = rng.integers(0, n)

        while h[one_idx] != 1.0:

        h_new = h.copy()
        h_new[zero_idx] = 1.0
        h_new[one_idx] = 0.0
        new_score = compute_c5(h_new)
        dscore = new_score - curr
        T = T0 * max(0.0, 1.0 - it / max_iters)

            h, curr = h_new, new_score
            if new_score < best_score:

                if best_score <= target:

def upsample(arr, new_n):
    if arr.size == new_n:
        return arr.copy()
    old_x = np.linspace(0, 1, arr.size, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(new_x, old_x, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
