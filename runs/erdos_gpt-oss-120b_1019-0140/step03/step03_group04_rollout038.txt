import time
import numpy as np

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n (balanced when n is a power of two)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def simulated_annealing(h_start, time_budget, rng):
    """Binary simulated‑annealing on C5 (minimization)."""
    n = h_start.size
    h = h_start.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    end = start + time_budget
    while time.time() < end:
        # propose a swap of a 1 and a 0
        i = rng.integers(n)
        j = rng.integers(n)
        while i == j or h[i] == h[j]:
            i = rng.integers(n)
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        # linear temperature schedule
        frac = (time.time() - start) / time_budget
        T = max(1e-4, 0.05 * (1.0 - frac))
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # resolution
    n = 256  # even, offers enough granularity while staying fast

    # ----- build initial candidates -------------------------------------------------
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = resample(init_full, n)
        cand = binarize_from_real(cand)
        candidates.append(cand)

    # Legendre (n+1 must be prime; 257 is prime)
    try:
        leg = legendre_sequence(n + 1)
        candidates.append(leg)
    except Exception:
        pass

    # Thue‑Morse when n is a power of two
    if (n & (n - 1)) == 0:
        candidates.append(thue_morse_binary(n))

    # a few random starters
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))

    best_h = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # ----- main simulated‑annealing phase -----------------------------------------
    # allocate most remaining time to a long annealing run
    while time_left() > 15.0:
        # try a fresh start if improvement stalls
        start_h = best_h.copy() if rng.random() < 0.7 else random_binary_balanced(n, rng)
        # give about 80 % of the remaining budget to this run
        alloc = time_left() * 0.8
        h_opt, sc_opt = simulated_annealing(start_h, alloc, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # stop early if we already beat the target
        if best_score <= 0.3808:
            break

    # ----- greedy improvement (hill climbing, accept only better) ------------------
    while time_left() > 0.5:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or best_h[i] == best_h[j]:
            continue
        best_h[i], best_h[j] = best_h[j], best_h[i]
        new_sc = compute_c5(best_h)
        if new_sc < best_score:
            best_score = new_sc
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]

    # final sanity check
    best_h = best_h.astype(float)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
