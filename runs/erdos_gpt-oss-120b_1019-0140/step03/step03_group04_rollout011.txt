def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # lambda search
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] += -h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def smooth_h(h, k):
    """Uniform smoothing with kernel size k followed by projection."""
    kernel = np.ones(k) / k
    smooth = np.convolve(h, kernel, mode='same')
    smooth = np.clip(smooth, 0.0, 1.0)
    return project_onto_box_sum(smooth, h.size / 2.0)

def generate_base_candidates(rng, n):
    """Construct a diverse set of initial vectors."""
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n), n / 2.0)
        candidates.append(cand)
    candidates.append(thue_morse_binary(n))
    # Legendre sequence for n=512 (length 256 upsampled)
    if n == 512:
        leg = legendre_sequence(257)  # 256‑length binary
        leg_up = resample(leg, n)
        leg_up = project_onto_box_sum(leg_up, n / 2.0)
        candidates.append(leg_up)
    # Random balanced binaries
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))
    # Random continuous vectors
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))
    return candidates

def slsqp_optimize_candidate(h0, max_iter=400):
    """Short SLSQP to improve a candidate (continuous variables)."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512  # target resolution
    best_h = None
    best_score = np.inf

    # ---------- 1. Base candidates + simple smoothing ----------
    bases = generate_base_candidates(rng, n)
    kernel_sizes = [3, 5, 7, 9, 11]
    max_passes = 2
    for base in bases:
        if time_left() < 1.0:
            break
        sc = compute_c5(base)
        if sc < best_score:
            best_score, best_h = sc, base.copy()
        for k in kernel_sizes:
            for p in range(1, max_passes + 1):
                if time_left() < 0.5:
                    break
                h = base.copy()
                for _ in range(p):
                    h = smooth_h(h, k)
                sc = compute_c5(h)
                if sc < best_score:
                    best_score, best_h = sc, h.copy()

    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_score = compute_c5(best_h)

    # ---------- 2. Adam‑style projected gradient descent ----------
    h_curr = best_h.copy()
    m = np.zeros_like(h_curr)
    v = np.zeros_like(h_curr)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    it = 0
    best_sc = best_score
    best_h_curr = h_curr.copy()
    while time_left() > 0.5:
        it += 1
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        g = compute_gradients_for_shifts(h_curr, shifts)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h_new = h_curr - lr * m_hat / (np.sqrt(v_hat) + eps)
        if it % 2000 == 0:
            h_new = h_new + rng.uniform(-0.005, 0.005, size=n)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h_curr = h_new.copy()
        if best_sc <= 0.3808:
            break
        h_curr = h_new

    best_h = best_h_curr
    best_score = best_sc

    # ---------- 3. Optional small‑scale SLSQP refinement ----------
    if time_left() > 5.0:
        n_small = 64
        h_small = resample(best_h, n_small)
        h_opt, sc_opt = slsqp_optimize_candidate(h_small, max_iter=300)
        if sc_opt < best_score:
            h_up = resample(h_opt, n)
            h_up = project_onto_box_sum(h_up, n / 2.0)
            best_score = compute_c5(h_up)
            best_h = h_up

    # Final safety projection
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
