def compute_c5_and_k(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    c5 = corr[idx] * (2.0 / len(h))
    k = idx - (len(h) - 1)  # shift index
    return c5, k

def subgradient(h):
    # compute overlap vector for each shift
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    k_star = np.argmax(corr) - (len(h) - 1)

    n = len(h)
    grad = np.full(n, 1.0)   # from sum h term
    for i in range(n):
        if i + k_star < n:
            grad[i] -= h[i + k_star]
        if i - k_star >= 0:
            grad[i] -= h[i - k_star]
    grad = grad * (2.0 / n)  # dx factor but irrelevant for direction
    return grad, k_star

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_k(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    c5 = corr[idx] * dx
    k = idx - (n - 1)
    return c5, k, corr

def gradient_for_shift(h, k):

        grad[k:] -= h[:n-k]
    else:

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_k(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    c5 = corr[idx] * dx
    k = idx - (n - 1)
    return c5, k, corr

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] -= h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def project_to_constraints(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target_sum) < 1e-12:
        return v
    for _ in range(15):
        nonbound = (v > 1e-12) & (v < 1.0 - 1e-12)
        cnt = np.count_nonzero(nonbound)
        if cnt == 0:
            if s == 0:
                v = np.full_like(v, target_sum / v.size)
                v = np.clip(v, 0.0, 1.0)
                s = np.sum(v)
                break
            scale = target_sum / s
            v = np.clip(v * scale, 0.0, 1.0)
            s = np.sum(v)
            if abs(s - target_sum) < 1e-12:
                break
        else:
            delta = (target_sum - s) / cnt
            v[nonbound] += delta
            v = np.clip(v, 0.0, 1.0)
            s = np.sum(v)
            if abs(s - target_sum) < 1e-12:
                break
    return v

def optimise_h(n, time_budget, init_h, rng):
    M = n // 2
    target_sum = M
    # projection helper
    proj = lambda v: project_to_constraints(v, target_sum)

    # initial solution
    if init_h is not None:
        h = proj(init_h.copy())
    else:
        h = proj(np.full(n, 0.5))
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Legendre seed for n==80
    if n == 80:
        p = 79
        leg = np.zeros(p, dtype=float)
        for i in range(1, p):
            leg[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
        leg = np.concatenate([leg, [1.0]])
        leg = proj(leg)
        c5_leg = compute_c5(leg)
        if c5_leg < best_c5:
            best_h, best_c5 = leg, c5_leg

    start = time.time()
    cur_h = best_h.copy()
    cur_c5, cur_k, _ = compute_c5_and_k(cur_h)
    T = 0.02

    while time.time() - start < time_budget - 0.1:
        # move selection
        if rng.random() < 0.65:
            grad = gradient_for_shift(cur_h, cur_k)
            step = 0.6 / (1.0 + (time.time() - start) / time_budget * 15.0)
            h_new = proj(cur_h - step * grad)
        else:
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 1e-12:
                continue
            delta = rng.random() * max_transfer * 0.5
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        new_c5, new_k, _ = compute_c5_and_k(h_new)

        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / max(T, 1e-12)):
            cur_h, cur_c5, cur_k = h_new, new_c5, new_k
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cur_h.copy(), cur_c5
        T *= 0.9999

        # occasional SLSQP refinement
        if rng.random() < 0.004:
            x0 = np.concatenate([cur_h, [cur_c5]])
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}]
            for kk in range(-n + 1, n):
                if kk == 0:
                    continue
                def ineq(x, k=kk):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk2 = -k
                        overl = np.dot(h[kk2:], 1.0 - h[:n - kk2])
                    return x[-1] - overl * (2.0 / n)
                cons.append({'type': 'ineq', 'fun': ineq})
            try:
                res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
                if res.success:
                    h_opt = proj(np.clip(res.x[:n], 0.0, 1.0))
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5 - 1e-12:
                        best_h, best_c5 = h_opt, c5_opt
                        cur_h, cur_c5, cur_k = h_opt, c5_opt, compute_c5_and_k(h_opt)[1]
            except Exception:
                pass

    return best_h, best_c5
