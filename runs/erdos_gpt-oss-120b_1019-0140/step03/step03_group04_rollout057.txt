def hill_climb_binary(h0, rng, time_end):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_binary(h)
    # precompute positions of ones and zeros
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    it = 0
    while time.time() < time_end:
        # pick random swap from lists
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # perform swap
        h[i_one], h[i_zero] = 0.0, 1.0
        # compute new score
        score = compute_c5_binary(h)
        # acceptance
        if score < best_score or rng.random() < np.exp((best_score - score) / max(1e-9, (1 - it/1e6))):
            # accept update (keep new h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
                # update ones/zeros sets
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            # else accepted as uphill, no need to update best
        else:
            # revert swap
            h[i_one], h[i_zero] = 1.0, 0.0
        it += 1
        # optionally occasionally reshuffle ones/zeros lists
        if it % 1000 == 0:
            # refresh
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return best_h, best_score
```

    n = p - 1

def hill_climb_binary_sa(h0, rng, time_end):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    it = 0
    init_temp = 0.02
    decay = 0.9995
    while time.time() < time_end:
        it += 1
        # select random 1 and 0
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h <= 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # swap
        h[i_one] = 0.0
        h[i_zero] = 1.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = init_temp * (decay ** it)
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_h = h.copy()
        else:
            # revert
            h[i_one] = 1.0
            h[i_zero] = 0.0
    return best_h, best_score

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def legendre_sequence(p):
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

        h[i_one] = 0.0

        new_score = compute_c5(h)

            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_h = h.copy()
        else:
            h[i_one] = 1.0
            h[i_zero] = 0.0
    return best_h, best_score
