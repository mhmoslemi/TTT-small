def find_swap_pair(h, k, rng):
    n = h.shape[0]
    if k >= 0:
        i_range = np.arange(n - k)
        cond_i = (h[i_range] == 0) & (h[i_range + k] == 1)

    i = int(rng.choice(i_candidates))
    j = int(rng.choice(j_candidates))
    return i, j

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Scale all entries proportionally (if total > 0)
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, target, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        dot = np.dot(h[: n - k], h[k:])
    else:
        kk = -k
        dot = np.dot(h[kk:], h[: n - kk])
    overlap = (target - dot) * dx
    return t - overlap

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append(
            {"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, target, dx)}
        )
    return cons

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        target = n / 2.0
        h0 = enforce_sum(init_arr, target, rng)
    else:
        n = 80
        target = n / 2.0
        h0 = random_continuous_vector(n, target, rng)

    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Build constraints for SLSQP
    constraints = build_constraints(n, target, dx)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # ---------- SLSQP phase ----------
    while time.time() < deadline - 15.0:
        # random start: binary 60% of time, else perturbed best
        if rng.random() < 0.6:
            start_h = random_binary_vector(n, target, rng)
        else:
            start_h = best_h.copy()
            # slight random redistribution preserving sum
            for _ in range(4):
                src = np.where(start_h > 1e-8)[0]
                dst = np.where(start_h < 1.0 - 1e-8)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 2000, "ftol": 1e-10, "disp": False},
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val <= 0.3808:
                break

    # ---------- Annealing phase ----------
    if time.time() < deadline - 0.5:
        best_h, best_val = anneal_improve(best_h, best_val, start, deadline, rng)

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
