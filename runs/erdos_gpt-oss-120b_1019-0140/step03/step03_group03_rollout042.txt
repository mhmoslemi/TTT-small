def gradient_descent_momentum_timebound(h0, step0=0.5, beta=0.9, time_limit=10.0):
    h = h0.copy()
    n = len(h)
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    start_iter = time.time()
    it = 0
    while True:
        now = time.time()
        if now - start_iter > time_limit:
            break
        it += 1
        # compute correlation and max shift(s)
        corr = compute_corr_fft(h) # returns array length 2n-1
        max_val = np.max(corr)
        idxes = np.where(corr >= max_val - 1e-12)[0]
        shifts = idxes - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta*momentum + (1-beta)*grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary_timebound(h0, time_limit, rng):
    n = len(h0)
    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # We'll maintain sets of ones and zeros indices
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    # Determine schedule: exponential cooling? We'll just use simple linear cooling in log scale.

    # We'll decide number of total steps = int(time_limit * 2000) (maybe 2000 proposals per second) requiring fast evaluation.
    total_steps = int(time_limit * 1000)  # conservative
    if total_steps <= 0:
        return best_h, best_score
    # compute start and end temperature
    temp_start = 0.01
    temp_end = 1e-5
    for it in range(total_steps):
        # compute temperature
        t = temp_start * ((temp_end / temp_start) ** (it / total_steps))

        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update index lists
            # after swap, i becomes zero, j becomes one
            zeros = np.flatnonzero(cur_h < 0.5)
            ones = np.flatnonzero(cur_h > 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0  # revert

    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)
    return corr[:2 * n - 1]

def compute_c5_fast(h):
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        grad += 1.0
        if shift >= 0:
            if shift < n:
                grad[:n-shift] -= h[shift:]
                grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            if s < n:
                grad[s:] -= h[:n-s]
                grad[:n-s] -= h[s:]
    return grad / len(shifts)

def gradient_descent_momentum_timebound(h0, step0=0.5, beta=0.9, time_limit=10.0):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    it = 0
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        it += 1
        corr = compute_corr_fft(h)
        max_val = np.max(corr)
        idxes = np.where(corr >= max_val - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal_binary_timebound(h0, time_limit, rng):
    n = len(h0)
    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # precompute index sets as arrays
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    total_steps = int(time_limit * 1000)
    if total_steps <= 0:
        return best_h, best_score
    temp_start = 0.01
    temp_end = 1e-5
    for it in range(total_steps):
        t = temp_start * ((temp_end / temp_start) ** (it / total_steps))

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)
    return corr[:2 * n - 1]

def compute_c5_fast(h):
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        grad += 1.0
        if shift >= 0:
            if shift < n:
                grad[:n - shift] -= h[shift:]
                grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            if s < n:
                grad[s:] -= h[:n - s]
                grad[:n - s] -= h[s:]
    return grad / len(shifts)

def gradient_descent_momentum_timebound(h0, step0=0.5, beta=0.9, time_limit=10.0):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    it = 0
    start = time.time()
    while True:
        if time.time() - start >= time_limit:
            break
        it += 1
        corr = compute_corr_fft(h)
        max_val = np.max(corr)
        idxes = np.where(corr >= max_val - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
