def simulated_annealing(h0, rng, time_left, max_iter=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_count(h)
    curr_score = best_score
    start = time.time()
    # Determine max_iter based on time budget
    if max_iter is None:
        max_iter = int(1e7) # large; will break early by time
    # temperature schedule parameters
    T0 = 0.5  # initial temperature
    Tmin = 1e-4
    for it in range(1, max_iter+1):
        if time_left() < 0.001:
            break
        # linear cooling: T = max(T0 * (1 - it / max_iter), Tmin)

        # ensure h[i] == 0, else find zero
        if h[i] == 0:
            zero = i
        else:
            # find random zero index

    start = time.time()
    def time_left():
        return max(budget_s - (time.time() - start), 0.0)

    target_c5 = 0.3808
    # choose n = 256 (or maybe 128 for faster)
    n = 256
    target_max_count = int(np.floor(target_c5 * n / 2))

    # Initialize best
    best_h = None
    best_score = n  # large

    # If initial_h_values exists, generate binary by sorting
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # need to downsample or upsample to n?
        # if size != n, we can repeat or upsample
        if init_full.size != n:
            # repeat to match length if possible integer factor
            factor = n // init_full.size
            if factor * init_full.size == n:
                init_bin = (init_full > np.median(init_full)).astype(np.int8)
                h0 = np.repeat(init_bin, factor)
            else:

                                          np.linspace(0,1,init_full.size,endpoint=False),

                h0[order[n//2:]] = 1

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = max_overlap_count(h)
    curr_score = best_score

    # How many iterations we aim to do based on time budget?
    # We can approximate a max_iter: maybe 10 million but break early.
    max_iter = int(5e7)  # large
    # temperature schedule parameters
    T0 = 0.5
    Tmin = 1e-4
    # compute factor for linear cooling
    total_iters_est = max_iter
    for it in range(1, max_iter+1):
        if time_left() < 0.001:
            break
        # linear cooling
        T = max(T0 * (1 - it/total_iters_est), Tmin)
        # generate a swap
        # pick two distinct indices i and j until bits differ
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            # get opposite bits via while loop but we can just continue to next iteration (skip)
            continue
        # Determine zero and one
        if h[i] == 0:
            zero, one = i, j
        else:
            zero, one = j, i
        # perform swap
        h_new = h.copy()
        h_new[zero] = 1
        h_new[one] = 0
        new_score = max_overlap_count(h_new)
        delta = new_score - curr_score
        # acceptance
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_max_count:
                    break
        # optionally, restart temperature after long stagnation? Not needed maybe
    return best_h, best_score

def binary_anneal(h0, rng, time_left, target_max):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = max_overlap_count(h)
    curr_score = best_score
    # Compute max number of iterations based on time left
    # allow high number (maybe 10 million)
    max_iters = int(1e7)  # huge
    # but break early if time ends
    for it in range(1, max_iters+1):
        if time_left() < 0.001:
            break
        # Temperature schedule: linear cooling from T0 to Tmin
        # set T0 depending on target? maybe 0.5
        T0 = 0.5
        Tmin = 1e-4
        T = max(T0 * (1 - it / max_iters), Tmin)
        # choose random swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # perform swap
        # Update h in-place
        # Save bits
        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi
        new_score = max_overlap_count(h)
        delta = new_score - curr_score
        accept = False
        if delta < 0 or rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_max:
                    break
        else:
            # revert swap
            h[i], h[j] = hi, hj
    return best_h, best_score
```

import time
import numpy as np

def max_overlap_count(h):
    """Return the maximal count of (1,0) pairs over all shifts for binary h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return int(corr.max())

def binary_anneal(h0, rng, time_left, target_max):
    """Simulated annealing on a balanced binary vector."""
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_score = max_overlap_count(h)
    curr_score = best_score

    # allocate iterations based on remaining time (≈3000 per second)
    max_iters = int(time_left() * 3000) + 1
    T0 = 0.5
    Tmin = 1e-4
    for it in range(1, max_iters + 1):
        if time_left() < 0.001:
            break
        T = max(T0 * (1 - it / max_iters), Tmin)

        # pick a random zero and a random one to swap
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        if h[i] == 0:
            zero, one = i, j
        else:
            zero, one = j, i

        # perform swap
        h[zero] = 1
        h[one] = 0
        new_score = max_overlap_count(h)
        delta = new_score - curr_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_max:
                    break
        else:
            # revert swap
            h[zero] = 0
            h[one] = 1

    return best_h, best_score

def final_improve(h, rng, time_left, target_max, current_score):
    """Local greedy improvement using random swaps."""
    n = h.size
    best_h = h.copy()
    best_score = current_score
    curr_score = current_score
    while time_left() > 0.1:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        if h[i] == 0:
            zero, one = i, j
        else:
            zero, one = j, i
        # swap
        h[zero] = 1
        h[one] = 0
        new_score = max_overlap_count(h)
        if new_score <= curr_score:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_max:
                    break
        else:
            # revert
            h[zero] = 0
            h[one] = 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(budget_s - (time.time() - start), 0.0)

    n = 256  # number of samples
    target_c5 = 0.3808
    target_max = int(np.floor(target_c5 * n / 2))

    # Construct initial binary vector
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.size != n:
            x_old = np.linspace(0, 1, init_arr.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init_resampled = np.interp(x_new, x_old, init_arr)
        else:
            init_resampled = init_arr
        order = np.argsort(init_resampled)
        h0 = np.zeros(n, dtype=np.int8)
        h0[order[n // 2 :]] = 1
    else:
        h0 = np.zeros(n, dtype=np.int8)
        h0[rng.choice(n, size=n // 2, replace=False)] = 1

    # First annealing run
    best_h, best_score = binary_anneal(h0, rng, time_left, target_max)

    # Additional restarts with perturbations if time permits
    while time_left() > 30.0 and best_score > target_max:
        h_start = best_h.copy()
        # perturb by swapping a few bits (≈5% of entries)
        num_swaps = max(1, n // 20)
        zeros = np.where(h_start == 0)[0]
        ones = np.where(h_start == 1)[0]
        for _ in range(num_swaps):
            if zeros.size == 0 or ones.size == 0:
                break
            zi = rng.choice(zeros)
            oi = rng.choice(ones)
            h_start[zi] = 1
            h_start[oi] = 0
            # update index lists
            zeros = np.where(h_start == 0)[0]
            ones = np.where(h_start == 1)[0]
        h_new, score_new = binary_anneal(h_start, rng, time_left, target_max)
        if score_new < best_score:
            best_h, best_score = h_new, score_new

    # Final greedy refinement
    best_h, best_score = final_improve(best_h, rng, time_left, target_max, best_score)

    # Compute final C5 bound
    c5_bound = 2.0 / n * best_score
    h_out = best_h.astype(float)
    return h_out, c5_bound, n
