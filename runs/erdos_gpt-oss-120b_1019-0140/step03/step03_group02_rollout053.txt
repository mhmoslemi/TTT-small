def smooth_adam(h0, n_iter, step0, alpha_start, alpha_end, beta1, beta2, time_left):
    h = h0.copy()
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start)*(it-1)/(max(1, n_iter-1))
        grad = grad_softmax_corr(h, alpha)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        score = compute_c5(h_new)

            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, start_time, budget, target=0.3808):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # Determine max_iters based on time left
    max_iters = int(min(300000, max(1000, (budget - (time.time() - start_time)) * 2000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time.time() - start_time > budget - 0.01:
            break
        # pick random zero and one
        # we can use while loops
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
def downsample(arr, new_n):
    return upsample(arr, new_n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    # compute stabilized exponentials
    exp_vals = np.exp(alpha * (corr - maxc))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        weights = np.zeros_like(exp_vals)
    else:
        weights = exp_vals / sum_exp
    grad = np.zeros_like(h)
    # shifts range from -(n-1) to (n-1)
    # index s corresponds to shift = s - (n-1)
    for s, w in enumerate(weights):
        if w == 0.0:
            continue
        k = s - (n - 1)
        if k >= 0:
            # h[i] * (1 - h[i+k]) contributions
            # derivative w.r.t h[i] is (1 - h[i+k])
            # derivative w.r.t h[i+k] is -h[i]
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad
```

def smooth_adam(h0, n_iter, step0, alpha_start, alpha_end,
                beta1, beta2, start_time, budget):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    for it in range(1, n_iter + 1):
        if time.time() - start_time > budget - 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / np.sqrt(it)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, start_time, budget, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # Estimate max iterations from remaining time
    max_iters = int(min(300000, max(1000, (budget - (time.time() - start_time)) * 2000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time.time() - start_time > budget - 0.01:
            break
        # randomly select a zero and a one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # adjust bounds
    # compute sum at lam_low
    sum_low = np.clip(v - lam_low, lo, hi).sum()
    if sum_low < s:
        lam_low = np.min(v) - 10 * hi
    sum_high = np.clip(v - lam_high, lo, hi).sum()
    if sum_high > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        sum_mid = np.clip(v - lam_mid, lo, hi).sum()
        if abs(sum_mid - s) < 1e-12:
            break
        if sum_mid > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def time_left(start_time, budget):
    return max(0.0, budget - (time.time() - start_time))

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    sum_low = np.clip(v - lam_low, lo, hi).sum()
    if sum_low < s:
        lam_low = np.min(v) - 10 * hi
    sum_high = np.clip(v - lam_high, lo, hi).sum()
    if sum_high > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        sum_mid = np.clip(v - lam_mid, lo, hi).sum()
        if abs(sum_mid - s) < 1e-12:
            break
        if sum_mid > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    t = np.array([0.0])
    while t.size < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp if sum_exp != 0 else np.zeros_like(exp_vals)
    grad = np.zeros_like(h)
    for s, w in enumerate(weights):
        if w == 0.0:
            continue
        k = s - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def smooth_adam(h0, n_iter, step0, alpha_start, alpha_end,
                beta1, beta2, start_time, budget):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    for it in range(1, n_iter + 1):
        if time_left(start_time, budget) < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / np.sqrt(it)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, start_time, budget, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(300000, max(1000, (budget - (time.time() - start_time)) * 2000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left(start_time, budget) < 0.01:
            break
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n_small = 128
    n_mid = 256
    n_big = 512
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = downsample(init_full, n_small) if init_full.size != n_small else init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small).astype(float))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))
    best_h = None
    best_score = np.inf
    for h0 in candidates:
        if time_left(start_time, budget_s) < 5.0:
            break
        remaining = time_left(start_time, budget_s)
        iters = int(min(8000, max(2000, remaining * 800)))
        h_opt, sc_opt = smooth_adam(h0, n_iter=iters, step0=0.3,
                                    alpha_start=1.0, alpha_end=12.0,
                                    beta1=0.9, beta2=0.999,
                                    start_time=start_time, budget=budget_s)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5(best_h)
    if time_left(start_time, budget_s) > 6.0:
        h_mid = upsample(best_h, n_mid)
        h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
        remaining = time_left(start_time, budget_s)
        iters_mid = int(min(12000, max(5000, remaining * 600)))
        h_mid_opt, sc_mid_opt = smooth_adam(h_mid, n_iter=iters_mid, step0=0.2,
                                            alpha_start=5.0, alpha_end=18.0,
                                            beta1=0.9, beta2=0.999,
                                            start_time=start_time, budget=budget_s)
        if sc_mid_opt < best_score:
            best_score = sc_mid_opt
            best_h = h_mid_opt.copy()
        else:
            best_h = h_mid.copy()
    if time_left(start_time, budget_s) > 8.0:
        h_big = upsample(best_h, n_big)
        h_big = project_onto_box_sum(h_big, n_big / 2.0)
        remaining = time_left(start_time, budget_s)
        iters_big = int(min(15000, max(7000, remaining * 400)))
        h_big_opt, sc_big_opt = smooth_adam(h_big, n_iter=iters_big, step0=0.15,
                                            alpha_start=10.0, alpha_end=30.0,
                                            beta1=0.9, beta2=0.999,
                                            start_time=start_time, budget=budget_s)
        if sc_big_opt < best_score:
            best_score = sc_big_opt
            best_h = h_big_opt.copy()
        else:
            best_h = h_big.copy()
    if time_left(start_time, budget_s) > 4.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, start_time, budget_s, target=0.3808)
        if bin_sc < best_score:
            best_score = bin_sc
            best_h = bin_h.copy()
        if time_left(start_time, budget_s) > 2.0:
            remaining = time_left(start_time, budget_s)
            iters_final = int(min(6000, max(2000, remaining * 300)))
            h_ref, sc_ref = smooth_adam(best_h, n_iter=iters_final, step0=0.1,
                                        alpha_start=15.0, alpha_end=30.0,
                                        beta1=0.9, beta2=0.999,
                                        start_time=start_time, budget=budget_s)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref.copy()
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
