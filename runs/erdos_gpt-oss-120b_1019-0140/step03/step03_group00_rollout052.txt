def mass_transfer_step(h, g, max_transfer=0.01):
    # find i with max positive g, j with min g
    i = np.argmax(g)
    j = np.argmin(g)
    # compute possible delta
    delta = min(h[i], 1.0 - h[j], max_transfer)
    if delta <= 0:
        return h
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def gradient_descent_with_mass_transfer(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get shift(s) attaining max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)   # returns avg gradient for active shifts
        # momentum update
        momentum = beta*momentum + (1-beta)*grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta*momentum
        # occasional jitter
        if it % 500 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # project onto box+sum
        h_new = project_onto_box_sum(h_new, s)
        # periodic mass transfer
        if it % 200 == 0:
            # compute gradient after projection? Could compute again
            grad2 = compute_gradients_for_shifts(h_new, shifts) # approximate same
            i = np.argmax(grad2)
            j = np.argmin(grad2)
            delta = min(h_new[i], 1.0 - h_new[j], 0.01)  # small
            if delta > 0:
                h_new[i] -= delta
                h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of C(k) over the given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_with_mass_transfer(h0, n_iter, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and periodic mass‑transfer."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    # initial score
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break

        # correlation and active shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)

        grad = compute_gradients_for_shifts(h, shifts)

        # momentum update and step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional jitter
        if it % 500 == 0:
            h_new += rng.normal(scale=0.005, size=n)

        # projection onto box and sum
        h_new = project_onto_box_sum(h_new, s)

        # periodic targeted mass transfer
        if it % 200 == 0:
            grad2 = compute_gradients_for_shifts(h_new, shifts)
            i = np.argmax(grad2)
            j = np.argmin(grad2)
            delta = min(h_new[i], 1.0 - h_new[j], 0.01)
            if delta > 0:
                h_new[i] -= delta
                h_new[j] += delta

        # evaluate new score
        score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0 / n)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score
