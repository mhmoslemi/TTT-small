def propose_move(h):
    n = len(h)
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    # Determine max positive delta for i: up = 1 - h[i]
    # max negative delta for j: down = h[j]
    max_delta = min(0.02, 1 - h[i], h[j])
    if max_delta <= 0:
        return None
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap (without dx factor) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected gradient descent on the max‑overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 < 0.3808:
                break
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = project_box_sum(h - alpha * grad, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

class SumConstraint:
    """Equality constraint: sum(h) == target."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k)*dx for a given shift k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=300):
    """Run a single SLSQP refinement."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def de_optimize(n, target_sum, deadline, rng):
    """Differential evolution with projection to feasibility."""
    bounds = [(0.0, 1.0)] * n
    def obj(v):
        h = project_box_sum(v, target_sum)
        return compute_c5(h)
    def callback(xk, convergence):

        func=obj,

        strategy='best1bin',
        maxiter=500,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=rng,
        polish=False,
        callback=callback,
        disp=False
    )
    h_opt = project_box_sum(result.x, target_sum)
    return h_opt, compute_c5(h_opt)
