def compute_corr_fft(h):
    n = h.size
    N = 2*n - 1
    u_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=N)
    U = np.fft.rfft(u_rev, n=N)
    conv = np.fft.irfft(H * U, n=N)
    return conv

def gradient_descent_momentum_fft(h0, time_end, step0=0.5, beta=0.9, max_iter=...):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sshift = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sshift:] = h[:n - sshift]
                term2[:n - sshift] = h[sshift:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_corr_fft(h):
    n = h.shape[0]
    N = 2 * n - 1
    # reverse of (1-h)
    u_rev = (1.0 - h)[::-1]
    # use rfft
    H = np.fft.rfft(h, n=N)
    U = np.fft.rfft(u_rev, n=N)
    conv = np.fft.irfft(H * U, n=N)
    return conv

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def resample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, n // 2, replace=False)] = 1.0
    return a

def compute_corr_fft(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    N = 2 * n - 1
    u_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=N)
    U = np.fft.rfft(u_rev, n=N)
    conv = np.fft.irfft(H * np.conj(U), n=N)
    return conv

def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return compute_corr_fft(h).max() * (2.0 / n)

def gradient_descent_momentum_fft(h0, time_end, step0=0.5, beta=0.9, max_iter=200000):
    """Projected GD with momentum on max‑correlation objective using FFT."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    eps = 1e-12
    while it < max_iter and time.time() < time_end:
        it += 1
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        idxes = np.flatnonzero(corr >= max_corr - eps)
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for sft in shifts:
            if sft >= 0:
                grad[:n - sft] += 1.0 - h[sft:]
                grad[sft:] += 1.0 - h[:n - sft]
            else:
                s = -sft
                grad[s:] += 1.0 - h[:n - s]
                grad[:n - s] += 1.0 - h[s:]
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_big = 1024                     # final resolution
    best_h = None
    best_score = np.inf

    # ----- seed candidates -------------------------------------------------
    candidates = []

    if 'initial_h_values' in globals():
        try:

            cand = project_onto_box_sum(resample(init, n_big), n_big / 2.0)

        except Exception:

    # deterministic patterned candidates
    cand_tm = thue_morse_binary(n_big)

    candidates.append(cand_tm)

    # a few random binary starters
    for _ in range(2):
        candidates.append(random_binary_balanced(n_big, rng))

    # ----- evaluate deterministic candidates --------------------------------
    for cand in candidates:
        if time_left() < 5.0:
            break
        end = time.time() + min(30.0, time_left() - 2.0)
        h_opt, sc_opt = gradient_descent_momentum_fft(cand, time_end=end,
                                                      step0=0.5, beta=0.9,
                                                      max_iter=200000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # ----- random restarts -------------------------------------------------
    while time_left() > 15.0:
        v = rng.random(n_big)
        cand = project_onto_box_sum(v, n_big / 2.0)

        h_opt, sc_opt = gradient_descent_momentum_fft(cand, time_end=end,
                                                      step0=0.5, beta=0.9,
                                                      max_iter=200000)
        if sc_opt < best_score:

        if best_score <= 0.3808:

    if best_h is None:

        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # final safety projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
