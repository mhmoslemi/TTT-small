def adam_optimize(h0, time_end, max_iter=1e7, step0=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while time.time() < time_end and t < max_iter:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get active shifts
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        step = step0 / (np.sqrt(t)) # optional adapt
        # Actually typical Adam uses step0 * m_hat / (sqrt(v_hat)+eps)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat)+eps)
        # Project
        h_new = project_onto_box_sum(h_new, h.size/2.0)
        # Evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Set sizes
    n_small = 128
    n_large = 512

    # candidate generator
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small/2.0)
        candidates.append(cand)

    candidates.append(thue_morse_binary(n_small))
    candidates.append(legendre_sequence(n_small + 1)[:n_small])  # length n_small

    # random balanced binary
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))

    # smoothed binary candidates
    kernel = np.array([0.2, 0.6, 0.2], dtype=float)
    for cand in list(candidates):  # for each existing binary candidate
        smooth = np.convolve(cand, kernel, mode='same')
        smooth = project_onto_box_sum(smooth, n_small/2.0)
        candidates.append(smooth)

    best_h = None
    best_score = np.inf

    # SLSQP optimization on small scale
    for cand in candidates:
        if time_left() < 1.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=600)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Additional random GD on small scale
    while time_left() > 12.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small/2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=400)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Upsample to large
    h_large = resample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large/2.0)
    best_score = compute_c5(h_large)
    best_h = h_large

    # Gradient descent refinement on large
    # Use Adam
    time_end = start + budget_s - 0.2
    h_ref, sc_ref = adam_optimize(best_h, time_end=time_end, max_iter=10**9, step0=0.05)
    if sc_ref < best_score:
        best_score, best_h = sc_ref, h_ref.copy()

    # Additional random restarts if time left
    while time_left() > 8.0:
        v = rng.random(n_large)
        h0 = project_onto_box_sum(v, n_large/2.0)
        time_end_restart = start + budget_s - 0.2
        h_ref, sc_ref = adam_optimize(h0, time_end=time_end_restart, max_iter=10**9, step0=0.05)
        if sc_ref < best_score:

    best_h = project_onto_box_sum(best_h, best_h.size/2.0)

    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation between uniform grids."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_gradients_for_shifts(h, shifts):
    """Gradient of correlation for given shifts (average over them)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            L = n - shift
            grad[:L] += 1.0 - h[shift:]
            grad[shift:] += -h[:L]
        else:
            s = -shift
            L = n - s
            grad[s:] += 1.0 - h[:L]
            grad[:L] += -h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def slsqp_optimize_candidate(h0, max_iter=400):
    """Small‑scale SLSQP refinement using a slack variable for max‑correlation."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def adam_optimize(h0, time_end, max_iter=10**9, step0=0.05,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam‑style projected optimizer on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = step0 / np.sqrt(it)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, n / 2.0)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # resolutions
    n_small = 128
    n_large = 512

    # build candidate list
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small / 2.0)
        candidates.append(cand)
    candidates.append(thue_morse_binary(n_small))
    leg_binary = legendre_sequence(257)[:n_small]  # slice to needed length
    candidates.append(leg_binary)
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    # a continuous random start
    candidates.append(project_onto_box_sum(rng.random(n_small), n_small / 2.0))

    # smoothing of binary candidates
    kernel = np.array([0.2, 0.6, 0.2], dtype=float)
    for cand in list(candidates):
        smooth = np.convolve(cand, kernel, mode='same')
        smooth = project_onto_box_sum(smooth, n_small / 2.0)
        candidates.append(smooth)

    best_h = None
    best_score = np.inf

    # small‑scale SLSQP refinement
    for cand in candidates:
        if time_left() < 1.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=600)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # extra random SLSQP attempts if time permits
    while time_left() > 12.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=400)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = random_binary_balanced(n_small, rng)
        best_score = compute_c5(best_h)

    # upsample to larger resolution
    h_large = resample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_h = h_large
    best_score = compute_c5(best_h)

    # primary Adam refinement
    time_end = start + budget_s - 0.3
    h_ref, sc_ref = adam_optimize(best_h, time_end=time_end,
                                  step0=0.05, max_iter=10**9)
    if sc_ref < best_score:
        best_score, best_h = sc_ref, h_ref.copy()

    # additional random restarts if needed
    while time_left() > 8.0 and best_score > 0.3808:
        v = rng.random(n_large)
        h0 = project_onto_box_sum(v, n_large / 2.0)
        restart_end = start + budget_s - 0.3
        h_ref, sc_ref = adam_optimize(h0, time_end=restart_end,
                                      step0=0.05, max_iter=10**9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= 0.3808:
            break

    # final safety projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
