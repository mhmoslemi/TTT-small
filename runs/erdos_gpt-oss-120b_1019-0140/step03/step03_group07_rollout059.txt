def project_sum_and_box(h, M):
    # ensure box constraints
    h = np.clip(h, 0.0, 1.0)
    # adjust sum to M
    cur = np.sum(h)
    if abs(cur - M) < 1e-12:
        return h
    # compute difference
    diff = M - cur
    # find indices not at bound 0 or 1
    free = np.where((h > 0) & (h < 1))[0]
    if free.size == 0:
        # no free variables, just scale globally but might violate bounds
        # fallback: add diff equally and clip
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # adjust sum by scaling if needed
        cur = np.sum(h)
        if abs(cur - M) > 1e-12:
            # distribute remaining diff greedily
            for i in range(len(h)):
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break
        return h
    # distribute diff equally among free indices
    amt = diff / free.size
    # add and clip again, but may cause some go out of bounds; iterate until diff small
    for i in free:
        h[i] += amt
    # clip again
    h = np.clip(h, 0.0, 1.0)
    # recursively adjust if sum still off
    cur = np.sum(h)
    if abs(cur - M) > 1e-9:
        # fallback simple scaling
        h *= M / cur
        h = np.clip(h, 0.0, 1.0)
    return h

def project(h, M):
    # clip
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    if abs(cur - M) > 1e-12:
        # adjust uniformly

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(np.sqrt(p))
    for i in range(3, r+1, 2):
        if p % i == 0:
            return False
    return True

def legendre_binary(p):
    # length p; 0 at index 0, rest as Legendre residues
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        seq[i] = 1.0 if pow(i, (p - 1)//2, p) == 1 else 0.0
    return seq

        return p == 2
    r = int(np.sqrt(p))
    for i in range(3, r + 1, 2):
        if p % i == 0:
            return False
    return True


def legendre_binary(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        seq[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
    return seq


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [200, 160, 128, 112, 96, 80]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    total = len(candidates)
    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.4, 400.0)
        else:
            alloc = min(remaining * 0.3 / (total - idx), 200.0)

        M = n // 2

        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        elif is_prime(n - 1):
            p = n - 1
            leg = legendre_binary(p)
            h0 = np.concatenate([leg, [1.0]])
            h0 = np.roll(h0, rng.integers(n))
        else:
            idxs = rng.choice(n, M, replace=False)
            h0 = np.zeros(n, dtype=float)
            h0[idxs] = 1.0

        h_opt, c5_opt = gradient_sa_opt(h0, M, alloc, rng)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
            if best_c5 < 0.3808:
                break

    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        M = n // 2
        cur = best_h.sum()
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
