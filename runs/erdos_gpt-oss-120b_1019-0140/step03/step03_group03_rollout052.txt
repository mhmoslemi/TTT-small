def local_search_continuous(h0, max_iter, rng, start_temp=1e-3, end_temp=1e-7):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = len(h)
    for it in range(max_iter):
        # temperature schedule
        t = start_temp * ((end_temp / start_temp) ** (it / max_iter))
        # choose two indices. Ensure they are different.
        i, j = rng.choice(n, size=2, replace=False)
        # compute max delta based on bounds
        # we will shift mass from i to j
        max_delta = min(h[i] - lo, hi - h[j]) if rng.random() < 0.5 else min(h[j] - lo, hi - h[i])
        # Actually we need to decide direction: either i -> j or j -> i
        # Choose direction randomly.
        if rng.random() < 0.5:
            # direction: i -> j
            max_delta = min(h[i] - lo, hi - h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            old_i, old_j = h[i], h[j]
            h[i] -= delta
            h[j] += delta
        else:
            max_delta = min(h[j] - lo, hi - h[i])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            old_i, old_j = h[i], h[j]
            h[j] -= delta
            h[i] += delta
        new_score = compute_c5(h)
        # decide acceptance
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # acceptance prob exp(- (new - best)/t)
            if rng.random() < np.exp(-(new_score - best_score) / (t + 1e-12)):
                accept = True
        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = old_i
            h[j] = old_j
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import minimize

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_correlation_fft(h):
    """Linear correlation of h with (1-h) using FFT, returns full array."""
    n = h.shape[0]
    y_rev = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    Y = np.fft.rfft(y_rev, n=size)
    conv = np.fft.irfft(H * Y, n=size)
    return conv[: 2 * n - 1]


def compute_c5(h):
    """Maximum overlap integral C5 for vector h."""
    n = h.shape[0]
    corr = compute_correlation_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    target = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_correlation_fft(h)
        max_corr = np.max(corr)
        eps = 1e-12
        shifts = np.where(corr >= max_corr - eps)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    """SLSQP refinement of a candidate (small n)."""
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        hh = x[:n]
        t = x[-1]
        corr = compute_correlation_fft(hh)
        return t - np.max(corr) * (2.0 / n)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simulated annealing on binary balanced vectors."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

                                              rng=rng, time_left=time_left)
        if sc_ls < best_score:
            best_score, best_h = sc_ls, h_ls.copy()

    # ------------------------------------------------------------------
    # Incremental up‑sampling stages
    # ------------------------------------------------------------------
    for n in (128, 256, 512):
        if best_h is None:
            best_h = random_binary_balanced(n, rng)
        else:
            best_h = resample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)

        # gradient descent (large iteration budget, interrupted by time)
        if time_left() > 5.0:
            h_gd, sc_gd = gradient_descent_momentum(best_h,
                                                    n_iter=50000,
                                                    step0=0.5,
                                                    beta=0.9,
                                                    time_left=time_left)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

        # continuous local search
        if time_left() > 5.0:
            max_iter_ls = int(min(30000, time_left() * 30))
            h_ls, sc_ls = local_search_continuous(best_h,
                                                  max_iter=max_iter_ls,
                                                  rng=rng,
                                                  time_left=time_left)
            if sc_ls < best_score:
                best_score, best_h = sc_ls, h_ls.copy()

        # binary annealing at final resolution
        if n == 512 and time_left() > 5.0:
            idx = np.argsort(best_h)[-n // 2:]
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx] = 1.0
            max_iter_ann = int(time_left() * 500)
            h_ann, sc_ann = anneal_binary(h_bin,
                                          max_iter=max_iter_ann,
                                          temp_start=0.01,
                                          temp_end=1e-5,
                                          rng=rng)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, float(best_score), best_h.size
