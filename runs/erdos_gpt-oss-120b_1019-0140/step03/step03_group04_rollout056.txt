def adam_optimize(h0, time_end, max_iters=100000):
    h = h0.copy()
    n = h.size
    sum_target = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    it = 0
    while time.time() < time_end and it < max_iters:
        it += 1
        grad = subgradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # step schedule: base learning rate = 0.5 * sqrt(2/n)? Not sure.
        # Use step = 0.5 / np.sqrt(it)
        step = 0.5 / np.sqrt(it)
        h = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, sum_target)
        score = compute_c5_fft(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the bounds bracket the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def full_correlation(h):
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(1.0 - h, n=N)
    prod = fa * np.conj(fb)
    corr = np.fft.irfft(prod, n=N)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
