import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto {lo<=x_i<=hi, sum x_i = s} using bisection on Lagrange multiplier."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x) (1-h(x+k)) dx on the uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation upsampling from length len(arr) to new_n points on [0,2)."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue-Morse sequence of length n (balanced only when n is a power of 2)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_shifts(corr, tol_factor=1e-8):
    """Return list of integer shifts where correlation is within tolerance of its maximum."""
    n = (corr.size + 1) // 2
    max_corr = np.max(corr)
    tol = max_corr * tol_factor
    idxes = np.where(corr >= max_corr - tol)[0]
    return idxes - (n - 1)

def gradient_for_shifts(h, shifts):
    """Gradient of max correlation w.r.t h for given shift set."""
    if len(shifts) == 0:
        return np.zeros_like(h)
    grad = np.zeros_like(h)
    for s in shifts:
        # term1 = h_{j+shift}, term2 = h_{j-shift}
        grad += 1.0 - np.roll(h, -s) - np.roll(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    """Momentum gradient descent with projection onto box-sum constraint."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        shifts = compute_shifts(corr)
        grad = gradient_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=400000):
    """Simulated annealing by swapping a 1 and a 0."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.2
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1200):
    """SLSQP to minimize the maximal overlap directly."""
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr  # t >= each correlation
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

def binarize_top_half(h):
    """Return a binary vector with exactly half entries 1 (the largest values)."""
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

        return best_h, best_score, best_h.size

    # -------------------- Phase 2: continuous refinement (gradient) --------------------
    while time_left() > 30.0:
        # either start from current best or fresh random
        start_from_best = rng.random() < 0.5
        if start_from_best:
            h0 = best_h.copy()
        else:
            h0 = project_onto_box_sum(rng.random(n_mid), n_mid / 2.0)
        h_ref, sc_ref = gradient_descent_momentum(h0, time_left,
                                                  step0=0.5, beta=0.9,
                                                  max_iter=200000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size
        # limit number of restarts to avoid long loops
        if time_left() < 20.0:
            break

    # -------------------- Phase 3: binarize --------------------
    h_bin = binarize_top_half(best_h)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # -------------------- Phase 4: binary annealing --------------------
    if time_left() > 5.0:
        h_anneal, sc_anneal = binary_swap_anneal(h_bin, rng, time_left,
                                                max_iters=400000)
        if sc_anneal < best_score:
            best_score, best_h = sc_anneal, h_anneal.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # -------------------- Phase 5: final polish (continuous) --------------------
    if time_left() > 2.0:
        h_ref2, sc_ref2 = gradient_descent_momentum(best_h, time_left,
                                                    step0=0.3, beta=0.9,
                                                    max_iter=100000)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # -------------------- Return best found --------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
