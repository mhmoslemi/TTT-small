import time, numpy as np

def compute_c5(h):
    n = h.shape[0]; dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def binary_local_search(h, budget, start_time):
    n = h.shape[0]; dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # Precompute ones and zeros positions
    # main loop
    while time.time() < start_time + budget:
        # compute overlaps
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        max_val = overlaps[max_idx]
        if max_val < best_val:
            best_val = max_val
            best_h = best_h.copy()
        # Identify shift
        k = max_idx - (n - 1)
        # Find mismatches
        if k >= 0:
            mism_mask = (best_h[:n - k] == 1) & (best_h[k:] == 0)
            zeros_mask = (best_h[:n - k] == 0) & (best_h[k:] == 1)
        else:
            kk = -k
            mism_mask = (best_h[kk:] == 1) & (best_h[:n - kk] == 0)
            zeros_mask = (best_h[kk:] == 0) & (best_h[:n - kk] == 1)
        mism_idxs = np.where(mism_mask)[0]
        zeros_idxs = np.where(zeros_mask)[0]
        if mism_idxs.size > 0 and zeros_idxs.size > 0:
            i = np.random.choice(mism_idxs)
            j = np.random.choice(zeros_idxs)
            # swap i and j
            new_h = best_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_val = compute_c5(new_h)
            # Accept if improvement or sometimes probabilistically
            if new_val < best_val or np.random.rand() < 0.001:
                best_h = new_h
                best_val = new_val
                if best_val < 0.3808:
                    break
        else:
            # No direct improving swap for this shift. Do generic random swap
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = best_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_val = compute_c5(new_h)
            if new_val < best_val or np.random.rand() < 0.001:
                best_h = new_h
                best_val = new_val
    return best_h, best_val

def binary_local_search(h, time_limit, start_time, rng):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    while time.time() < start_time + time_limit:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        max_val = overlaps[max_idx]
        k = max_idx - (n - 1)
        # find mismatched positions
        if k >= 0:
            mism = (cur_h[:n - k] == 1) & (cur_h[k:] == 0)
            zeros = (cur_h[:n - k] == 0) & (cur_h[k:] == 1)
        else:
            kk = -k
            mism = (cur_h[kk:] == 1) & (cur_h[:n - kk] == 0)
            zeros = (cur_h[kk:] == 0) & (cur_h[:n - kk] == 1)
        mism_idxs = np.where(mism)[0]
        zero_idxs = np.where(zeros)[0]
        # for k>=0 mism_idxs correspond to i; zero_idxs correspond to j in the same range
        # For k<0 the indices correspond to i in the second block; need to map to full index.
        # For simplicity we can convert indexes to full positions:
        if k >= 0:
            i_candidates = mism_idxs
            j_candidates = zero_idxs
        else:
            i_candidates = mism_idxs + kk
            j_candidates = zero_idxs + kk
        if i_candidates.size > 0 and j_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            # fallback generic swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_val = compute_c5(new_h)
        # Accept if better or with small prob
        if new_val < cur_val or rng.random() < 0.001:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h, new_val
                if best_val < 0.3808:
                    return best_h, best_val
        # maybe also include occasional random swaps (not from mism)
        # optionally also break if time is low.
    return best_h, best_val

def binary_local_search(h, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    while time.time() < deadline - 1e-6:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        max_val = overlaps[max_idx]
        # Save best if improvement
        if max_val < best_val:
            best_val = max_val
            best_h = cur_h.copy()
            if best_val < 0.3808:
                break
        # Determine shift
        k = max_idx - (n - 1)
        # Compute mismatched and zero positions
        if k >= 0:
            mism_mask = (cur_h[:n - k] == 1) & (cur_h[k:] == 0)
            zero_mask = (cur_h[:n - k] == 0) & (cur_h[k:] == 1)
            i_candidates = np.where(mism_mask)[0]
            j_candidates = np.where(zero_mask)[0]
        else:
            kk = -k
            mism_mask = (cur_h[kk:] == 1) & (cur_h[:n - kk] == 0)
            zero_mask = (cur_h[kk:] == 0) & (cur_h[:n - kk] == 1)
            i_candidates = np.where(mism_mask)[0] + kk
            j_candidates = np.where(zero_mask)[0] + kk
        if i_candidates.size > 0 and j_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            # generic swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # Perform swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_val = compute_c5(new_h)
        # Accept if better or with small probability
        if new_val < cur_val or rng.random() < 0.001:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                if best_val < 0.3808:
                    break
    return best_h, best_val

def continuous_sa(h, deadline, rng):
    n = h.shape[0]; dx = 2.0 / n
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    total_time = deadline - time.time()
    while time.time() < deadline - 1e-6:
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        max_val = overlaps[max_idx]
        # track best
        if max_val < best_val:
            best_val = max_val
            best_h = cur_h.copy()
            if best_val < 0.3808:
                break
        k = max_idx - (n - 1)
        # compute s array
        s = np.zeros(n)
        if k >= 0:
            if k < n:
                s[:n - k] += 1.0 - cur_h[k:]
            if k > 0:
                s[k:] -= cur_h[:n - k]
        else:
            kk = -k
            if kk < n:
                s[kk:] += 1.0 - cur_h[:n - kk]
            if kk > 0:
                s[:n - kk] -= cur_h[kk:]
        # source candidate: h_i > epsilon and s_i is max
        eps = 1e-12
        src_mask = cur_h > eps
        if not np.any(src_mask):
            break
        s_src = s.copy()
        s_src[~src_mask] = -np.inf
        i = int(np.argmax(s_src))
        # destination candidate: h_j < 1 - epsilon and s_j is min
        dst_mask = cur_h < 1.0 - eps
        if not np.any(dst_mask):
            break
        s_dst = s.copy()
        s_dst[~dst_mask] = np.inf
        j = int(np.argmin(s_dst))
        # if move cannot improve (i.e., s[i] <= s[j]), fallback to random
        if s[i] <= s[j]:
            # random move
            src_idxs = np.where(src_mask)[0]
            dst_idxs = np.where(dst_mask)[0]
            i = rng.choice(src_idxs)
            j = rng.choice(dst_idxs)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # cannot move, break
            break
        # Determine delta: use some fraction
        delta = max_delta * rng.random()
        if delta <= 0:
            continue
        # Apply move
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5(cur_h)
        # Temperature schedule
        remaining = deadline - time.time()
        # scale temperature relative to initial target range ~0.2
        T = max(0.01 * (remaining / (deadline - start)), 1e-12)  # note start defined globally
        # acceptance
        if new_val < cur_val or np.exp(-(new_val - cur_val) / T) > rng.random():
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
        # maybe occasional perturbation
    return best_h, best_val

            j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_val = compute_c5(new_h)
        if new_val < cur_val or rng.random() < 0.001:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                if best_val < 0.3808:
                    break
    return best_h, best_val

# ---------- continuous directed simulated annealing ----------
def continuous_sa(h_start, deadline, start_time, rng):
    n = h_start.shape[0]
    dx = 2.0 / n
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    total_budget = deadline - start_time
    while time.time() < deadline - 1e-6:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        max_val = overlaps[max_idx]
        if max_val < best_val:
            best_val = max_val
            best_h = cur_h.copy()
            if best_val < 0.3808:
                break
        k = max_idx - (n - 1)
        s = np.zeros(n)
        if k >= 0:
            if k < n:
                s[:n - k] += 1.0 - cur_h[k:]
            if k > 0:
                s[k:] -= cur_h[:n - k]
        else:
            kk = -k
            if kk < n:
                s[kk:] += 1.0 - cur_h[:n - kk]
            if kk > 0:
                s[:n - kk] -= cur_h[kk:]
        eps = 1e-12
        src_mask = cur_h > eps
        if not src_mask.any():
            break
        s_src = s.copy()
        s_src[~src_mask] = -np.inf
        i = int(np.argmax(s_src))
        dst_mask = cur_h < 1.0 - eps
        if not dst_mask.any():
            break
        s_dst = s.copy()
        s_dst[~dst_mask] = np.inf
        j = int(np.argmin(s_dst))
        if s[i] <= s[j]:
            src_idxs = np.where(src_mask)[0]
            dst_idxs = np.where(dst_mask)[0]
            i = rng.choice(src_idxs)
            j = rng.choice(dst_idxs)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            break
        delta = max_delta * rng.random()
        if delta <= 0:
            continue
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5(cur_h)
        remaining = deadline - time.time()
        T = max(0.01 * (remaining / total_budget), 1e-12)
        if new_val < cur_val or np.exp(-(new_val - cur_val) / T) > rng.random():
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = enforce_sum_continuous(h0, target, rng)
        best_h = h0.copy()
        best_val = compute_c5(best_h)
    else:
        n = 80
        target = n / 2.0
        best_h = random_binary_vector(n, target, rng)
        best_val = compute_c5(best_h)

    # ---------- binary local search phase ----------
    while time.time() < deadline - 5.0:
        hb = random_binary_vector(n, target, rng)
        hb_best, hb_val = binary_local_search(hb, deadline, rng)
        if hb_val < best_val:
            best_val, best_h = hb_val, hb_best.copy()
            if best_val < 0.3808:
                break

    # ---------- continuous annealing phase ----------
    if time.time() < deadline - 1.0:
        # start from the best binary or the original continuous if any
        cont_start = best_h.copy()
        cont_start = enforce_sum_continuous(cont_start, target, rng)
        cont_best, cont_val = continuous_sa(cont_start, deadline, start_time, rng)
        if cont_val < best_val:
            best_h, best_val = cont_best, cont_val

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_continuous(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
