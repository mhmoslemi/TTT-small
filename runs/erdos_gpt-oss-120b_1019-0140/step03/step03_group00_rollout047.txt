        new_h[j] += delta

        # Evaluate
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def random_pairwise_hill_climb(h, attempts, rng, time_left):
    """Random mass‑transfer hill climbing that respects the sum constraint."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = h.shape[0]
    for _ in range(attempts):
        if time_left() <= 0:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        if delta == 0:
            continue
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float).copy()
    else:
        base = thue_morse(64).astype(float)

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512]
    for idx, n in enumerate(scales):
        if time_left() <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent refinement
        if time_left() > 5:
            # iteration budget grows with available time and shrinks with n
            iters = int(min(100000, max(5000, time_left() * (18000.0 / n))))
            step0 = 0.5 if n <= 256 else 0.3
            base, sc = gradient_descent_momentum_custom(
                base, iters, step0=step0, beta=0.9, rng=rng, time_left=time_left)
            if sc < best_score:
                best_score, best_h = sc, base.copy()

        # Random pairwise hill climbing
        if time_left() > 5:
            attempts = int(min(20000, max(2000, time_left() * 20)))
            base, sc = random_pairwise_hill_climb(base, attempts, rng, time_left)
            if sc < best_score:
                best_score, best_h = sc, base.copy()

        # Prepare for next scale
        if idx < len(scales) - 1:
            # keep refined base for upsampling
            pass

    # Intensive final refinement on the best found vector
    if best_h is not None and time_left() > 5:
        n_final = best_h.size
        iters = int(min(150000, max(20000, time_left() * (30000.0 / n_final))))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, iters, step0=0.2, beta=0.9, rng=rng, time_left=time_left)
        attempts = int(min(50000, max(5000, time_left() * 30)))
        best_h, best_score = random_pairwise_hill_climb(best_h, attempts, rng, time_left)

    # Final random binary search as a safety net
    n_final = best_h.shape[0] if best_h is not None else base.shape[0]

        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:

        if best_score <= 0.3808:
            break

    # Ensure feasibility before returning
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
