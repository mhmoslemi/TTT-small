def solve_mip_balanced(n, time_limit):
    # n must be even.
    m = n // 2
    x = cvxpy.Variable(n, boolean=True)
    t = cvxpy.Variable(integer=True)
    constraints = [cvxpy.sum(x) == m]
    # for each shift
    for k in range(1, n):
        pairs = [(i, i+k) for i in range(n-k)]
        # build vector of y variables for this shift
        y = cvxpy.Variable(len(pairs), boolean=True)
        # each y_i <= x_i, <= x_j, >= x_i + x_j - 1
        for idx, (i,j) in enumerate(pairs):
            constraints.append(y[idx] <= x[i])
            constraints.append(y[idx] <= x[j])
            constraints.append(y[idx] >= x[i] + x[j] - 1)
        # sum y >= t
        constraints.append(cvxpy.sum(y) >= t)
    # t lower bound maybe 0
    constraints.append(t >= 0)
    prob = cvxpy.Problem(cvxpy.Maximize(t), constraints)
    prob.solve(solver='CBC', maxSeconds=time_limit, verbose=False)
    # after solve
    if x.value is None:
        return None, None
    x_sol = np.rint(x.value).astype(float) # ensure binary
    t_val = int(round(t.value))
    return x_sol, t_val
```

def solve_mip(n, time_limit):
    import cvxpy as cp
    m = n // 2
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == m]
    constraints.append(t >= 0)
    for k in range(1, n):
        num_pairs = n - k
        y = cp.Variable(num_pairs, boolean=True)

            constraints.append(y[i] <= x[i])
            constraints.append(y[i] <= x[i + k])
            constraints.append(y[i] >= x[i] + x[i + k] - 1)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad


def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad


def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=0.1, alpha_end=5.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def solve_mip_balanced(n, time_limit=300):

        return None, None
    m = n // 2
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == m, t >= 0]

    for k in range(1, n):
        cnt = n - k
        y = cp.Variable(cnt, boolean=True)
        for i in range(cnt):
            constraints.append(y[i] <= x[i])
            constraints.append(y[i] <= x[i + k])
            constraints.append(y[i] >= x[i] + x[i + k] - 1)

        prob.solve(solver=cp.GLPK_MI, verbose=False)
    if x.value is None:
        return None, None
    x_sol = np.rint(x.value).astype(float)
    t_val = int(round(t.value))
