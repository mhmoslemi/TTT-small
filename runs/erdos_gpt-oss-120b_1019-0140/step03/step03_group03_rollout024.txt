def compute_c5(h):
    n = h.shape[0]
    f = np.fft.fft(h, n*2-1)
    g = np.fft.fft(1.0 - h, n*2-1)
    corr = np.fft.ifft(f * np.conj(g)).real
    # due to non-circular correlation, we want the correlation at all shifts, same as np.correlate full. This corr length = n*2-1
    # The scaling factor dx = 2.0 / n
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    n = 1024
    # initial vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # upsample to n using linear interpolation
        if init.size != n:
            x_old = np.linspace(0, 1, init.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = init.copy()
        # Convert to binary balanced: pick top n/2 indices as 1
        idx = np.argpartition(h0, -n//2)[-n//2:]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # annealing parameters
    max_iters = int(min(5_000_000, max(1000, time_left() * 1500)))  # ~1500 iters/sec
    temp_start = 0.005
    temp_end = 1e-6

    # Pre-compute exponent factor
    # We'll compute t = temp_start * (temp_end/temp_start) ** (it/max_iters)
    log_ratio = np.log(temp_end / temp_start)

    it = 0
    while it < max_iters and time_left() > 0.0:
        # compute temperature
        t = temp_start * np.exp(log_ratio * (it / max_iters))

        # pick random different pair
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            # skip this iteration
            it += 1
            continue

        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5_fft(cur_h)
        delta = new_score - cur_score
        accept = delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        it += 1
        # periodic time check can be done every many iterations
        if it % 10000 == 0 and time_left() < 0.01:
            break

    # final projection: ensure sum= n/2 (should be exactly)
    # Due to binary swaps, sum unchanged

    # final compute using compute_c5 (maybe more precise using np.correlate)
    final_score = compute_c5_fft(best_h)

    return best_h, final_score, n

import time
import numpy as np

def compute_c5_fft(h):
    n = h.shape[0]
    m = 2 * n - 1
    f = np.fft.rfft(h, n=m)
    rev = (1.0 - h)[::-1]
    g = np.fft.rfft(rev, n=m)
    corr = np.fft.irfft(f * g, n=m)
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 1024

    # initial binary vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # upsample to n
        h_cont = upsample(init, n)
        # take top n/2 entries as ones
        idx = np.argpartition(h_cont, -n//2)[-n//2:]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # simulated annealing
    max_iters = int(max(1000, time_left() * 1200))
    temp_start = 0.02
    temp_end = 1e-6
    log_ratio = np.log(temp_end / temp_start)

    i = 0
    while i < max_iters and time_left() > 0.0:
        # temperature
        t = temp_start * np.exp(log_ratio * (i / max_iters))

        # choose random different positions
        a = rng.integers(n)

        if cur_h[a] == cur_h[b]:
