def max_shifts_gradient(h):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    mask = corr >= max_corr - 1e-12
    idxes = np.where(mask)[0]
    shifts = idxes - (n-1)
    grad = np.zeros(n, dtype=h.dtype)
    for shift in shifts:
        # Build shifted arrays
        if shift >= 0:
            h_right = np.zeros(n, dtype=h.dtype)
            h_right[:n-shift] = h[shift:]
            h_left = np.zeros(n, dtype=h.dtype)
            h_left[shift:] = h[:n-shift]
        else:
            s = -shift
            h_right = np.zeros(n, dtype=h.dtype)
            h_right[s:] = h[:n-s]
            h_left = np.zeros(n, dtype=h.dtype)
            h_left[:n-s] = h[s:]
        grad += (1.0 - h_right - h_left)
    grad /= len(shifts)
    return grad
```

def adam_optimize(h0, max_iter, lr, beta1, beta2, eps, time_left):

    v = np.zeros_like(h)

    t = 0

        t += 1
        grad = max_shifts_gradient(h)
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        score = compute_c5(h_new)

            best_h = h_new.copy()
        h = h_new

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def max_shifts_gradient(h):
    """Average subgradient of the max‑correlation term over active shifts."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    mask = corr >= max_corr - 1e-12
    idxes = np.where(mask)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            right = np.zeros_like(h)
            left = np.zeros_like(h)
            right[:n - shift] = h[shift:]
            left[shift:] = h[:n - shift]
        else:
            s = -shift
            right = np.zeros_like(h)
            left = np.zeros_like(h)
            right[s:] = h[:n - s]
            left[:n - s] = h[s:]
        grad += (1.0 - right - left)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def adam_optimize(h0, lr, beta1, beta2, eps, max_iter, time_left):
    """Adam optimizer with projection onto box‑sum constraints."""
    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        grad = max_shifts_gradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb(h0, n_iter, time_left, rng):
    """Simple hill‑climbing by moving mass between two positions."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(n_iter):
        if time_left() <= 0:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move from i to j
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
    return best_h, best_score


def random_balanced_binary(n, rng):
    """Return a balanced binary vector (exactly n/2 ones)."""
    arr = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    arr[ones] = 1.0
    return arr
