def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Use bisection to find shift lambda such that sum(clip(v+lambda, lo, hi)) = target
    lo_shift = -1e6
    hi_shift = 1e6
    # adjust bounds far enough
    for _ in range(60):
        mid = (lo_shift + hi_shift) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            lo_shift = mid
        else:
            hi_shift = mid
    return np.clip(v + (lo_shift + hi_shift) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def projected_gradient_descent(h_start, target, max_iter, time_deadline):
    # adaptive step size
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    n = h.shape[0]
    for it in range(max_iter):
        if time.time() > time_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                # random perturb
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target)
                alpha = 0.1
    # evaluate final best
    current_c5 = compute_c5(best_h)
    return best_h, current_c5

    rng = np.random.default_rng()

    cur_c5 = compute_c5(h)

    n = h.shape[0]
    # schedule
    start_time = time.time()
    # define parameters
    init_epsilon = 0.05
    epsilon_decay = 0.9995  # per iteration
    epsilon = init_epsilon
    init_temp = 0.001
    temp_decay = 0.9995
    temp = init_temp
    iter_count = 0

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v)=target via bisection."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap for shift k (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def projected_gradient_descent(h_start, target, max_iter, time_deadline, rng):
    """Projected gradient descent on the worst-shift objective."""
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target)
                alpha = 0.1
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def simulated_annealing(h_start, target, deadline, rng):
    """Pairwise epsilon-swap annealing with occasional gradient refinement."""
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n = h.shape[0]

    # Schedules
    eps = 0.05
    eps_decay = 0.9995
    temp = 0.005
    temp_decay = 0.9995
    iter_cnt = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if rng.random() < 0.5:
            max_eps = min(eps, h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            new_h = h.copy()
            new_h[i] -= max_eps
            new_h[j] += max_eps
        else:
            max_eps = min(eps, h[j], 1.0 - h[i])
            if max_eps <= 0:
                continue
            new_h = h.copy()
            new_h[j] -= max_eps
            new_h[i] += max_eps

        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

        eps *= eps_decay
        temp *= temp_decay
        iter_cnt += 1

        if iter_cnt % 5000 == 0:
            # short gradient refinement
            gd_deadline = deadline - 0.1
            gd_h, gd_c5 = projected_gradient_descent(h, target, max_iter=200,
                                                    time_deadline=gd_deadline, rng=rng)
            if gd_c5 < cur_c5:
                h = gd_h
                cur_c5 = gd_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    return best_h, best_c5

def binary_best(h, target):
    """Return binary vector with exactly round(target) ones having largest entries."""
    n = h.shape[0]
    t_int = int(round(target))
    idx = np.argpartition(h, -t_int)[-t_int:]
    bin_vec = np.zeros_like(h)
    bin_vec[idx] = 1.0
    return bin_vec

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Quick gradient descent to polish initial point
    gd_deadline = start_time + budget_s * 0.1
    gd_h, gd_c5 = projected_gradient_descent(best_h, target_sum, max_iter=1000,
                                            time_deadline=gd_deadline, rng=rng)
    if gd_c5 < best_c5:
        best_h, best_c5 = gd_h, gd_c5

    # Main annealing phase
    remaining_deadline = start_time + budget_s - 5.0  # safety buffer
    sa_h, sa_c5 = simulated_annealing(best_h, target_sum, remaining_deadline, rng)
    if sa_c5 < best_c5:
        best_h, best_c5 = sa_h, sa_c5

    # Final refinement
    final_deadline = start_time + budget_s - 2.0
    fin_h, fin_c5 = projected_gradient_descent(best_h, target_sum, max_iter=2000,
                                               time_deadline=final_deadline, rng=rng)
    if fin_c5 < best_c5:
        best_h, best_c5 = fin_h, fin_c5

    # Try binary rounding
    bin_h = binary_best(best_h, target_sum)
    bin_c5 = compute_c5(bin_h)
    if bin_c5 < best_c5:
        best_h, best_c5 = bin_h, bin_c5

    # Ensure feasibility before return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
