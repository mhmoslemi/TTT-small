# ---------- utilities ----------
def compute_c5_raw(h):
    """Maximum overlap (raw, without dx) for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full'))


def generate_random_binary(n, M, rng):
    """Binary vector with exactly M ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h


def legendre_sequence(p):
    """0/1 Legendre sequence of prime length p (index 0 stays 0)."""
    seq = np.zeros(p, dtype=float)
    half = (p - 1) // 2
    for i in range(1, p):
        leg = pow(i, half, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

        return False
    i = 5
    while i * i <= num:
        if num % i == 0 or num % (i + 2) == 0:
            return False
        i += 6
    return True

            i_local = rng.choice(len(i_idx), p=probs)
            i = i_idx[i_local]
            j = j_idx[i_local]
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta > 1e-12:
                delta = rng.random() * max_delta
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                move_made = True

        if not move_made:
            i, j = rng.choice(n, 2, replace=False)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 1e-12:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_raw = compute_c5_raw(new_h)
        if new_raw <= cur_raw or rng.random() < np.exp(-(new_raw - cur_raw) / (T + 1e-12)):
            cur_h, cur_raw = new_h, new_raw
            if cur_raw < best_raw:
                best_h, best_raw = cur_h.copy(), cur_raw
                if best_raw * (2.0 / n) < 0.3808:
                    break
        T = max(T * alpha, 1e-12)

    # ---- deterministic greedy refinement ----
    for _ in range(2000):
        if time.time() - start > time_budget - 0.05:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)
        if k >= 0:
            i_idx = np.arange(0, n - k)
        else:
            i_idx = np.arange(-k, n)
        j_idx = i_idx + k
        contrib = best_h[i_idx] * (1.0 - best_h[j_idx])
        if np.max(contrib) < 1e-12:
            break
        i_local = int(np.argmax(contrib))
        i = i_idx[i_local]
        j = j_idx[i_local]
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * 0.5
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_raw = compute_c5_raw(new_h)
        if new_raw < best_raw - 1e-12:
            best_h, best_raw = new_h, new_raw

    return best_h, best_raw * (2.0 / n)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80, 96, 112, 128]
    # prioritize parent construction if present
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)

        h_opt, c5_opt = optimise_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # final safety adjustments
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5_raw(best_h) * (2.0 / n)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
