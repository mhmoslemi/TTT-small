## Problem

Find a step function h: [0, 2] → [0, 1] that **minimizes** the overlap integral:

$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

**Constraints**:
1. h(x) ∈ [0, 1] for all x

**Discretization**: Represent h as n_points samples over [0, 2].

- 0 ≤ h[i] ≤ 1 for all i
- sum(h) * dx = 1 (equivalently: sum(h) == n_points / 2 exactly)

The evaluation computes: C₅ = max(np.correlate(h, 1-h, mode="full") * dx)

Smaller sequences with less than 1k samples are preferred - they are faster to optimize and evaluate.

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval contains a solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def slsqp_optimize_candidate(h0, max_iter=500):
    """Run a short SLSQP to improve a candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [
        {'type': 'eq', 'fun': eq_sum},
        {'type': 'ineq', 'fun': ineq_fun}
    ]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional jitter to escape flats
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
