def simulated_annealing(h0, target_sum, time_deadline, max_iter=200000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = h.size
    # set parameters
    init_temp = 0.01  # initial temperature in terms of C5 difference; maybe start higher.
    final_temp = 1e-5
    t0 = time.time()
    for i in range(max_iter):
        if time.time() > time_deadline:
            break
        # temperature schedule
        frac = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** frac
        # propose perturbation: select random subset size s (maybe 5% of n)
        s = max(1, n // 20)  # 5% of n
        idx = np.random.choice(n, size=s, replace=False)
        # add noise sigma decreasing over time
        sigma = 0.1 * (1 - frac) + 0.01
        h_new = h.copy()
        # Gaussian noise to selected entries
        h_new[idx] += np.random.normal(scale=sigma, size=s)
        # clip and project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_box_sum(h_new, target_sum)
        # evaluate
        new_c5 = compute_c5(h_new)
        # acceptance
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        if new_c5 < compute_c5(h) or np.random.rand() < np.exp(-(new_c5 - compute_c5(h)) / (temp + 1e-12)):
            h = h_new
    return best_h, best_c5

def gradient_descent_momentum(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    beta = 0.9
    v = np.zeros_like(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n-1)
        grad = gradient_for_shift(h, k)
        # momentum update
        v = beta * v - alpha * grad
        new_h = h + v
        new_h = project_to_box_sum(np.clip(new_h, 0.0, 1.0), target_sum)
        new_c5 = compute_c5(new_h)

            h = new_h
            alpha = min(alpha * 1.03, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-6)
            # reset velocity if too small?
            v = np.zeros_like(h)
    return best_h, best_c5

        n = h0.size

    else:
        n = 80

    target_sum = n / 2.0

    h0 = project_to_box_sum(h0, target_sum)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

# ---------- Constraint classes ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

# ---------- Gradient descent with momentum ----------
def gradient_descent_momentum(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent with momentum on the max‑overlap objective."""
    n = h_start.shape[0]
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = 0.2          # step size
    beta = 0.8           # momentum coefficient
    v = np.zeros_like(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # momentum update
        v = beta * v - alpha * grad
        new_h = h + v
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        # record best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        # acceptance for current point
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-6)
            v = np.zeros_like(h)
    return best_h, best_c5

# ---------- Simulated annealing ----------
def simulated_annealing(h_start, target_sum, time_limit):
    """Simple SA that respects box and sum constraints."""
    cur_h = h_start.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    n = cur_h.size
    init_temp = max(0.01, (cur_c5 - 0.35) * 0.5)
    final_temp = 1e-6
    max_iter = 200000
    i = 0
    while time.time() < time_limit and i < max_iter:
        i += 1
        frac = i / max_iter
        if frac > 1.0:
            frac = 1.0
        temp = init_temp * (1.0 - frac) + final_temp * frac
        # perturb a subset
        s = max(1, n // 15)                     # about 6‑7 % of entries
        idx = np.random.choice(n, size=s, replace=False)
        sigma = 0.05 * (1.0 - frac) + 0.005
        h_new = cur_h.copy()
        h_new[idx] += np.random.normal(scale=sigma, size=s)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_box_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        # Metropolis acceptance
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (temp + 1e-12)):
            cur_h = h_new
            cur_c5 = new_c5
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            if best_c5 < 0.3808:
                break
    return best_h, best_c5

                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # ---- Ensure feasibility -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(n)
