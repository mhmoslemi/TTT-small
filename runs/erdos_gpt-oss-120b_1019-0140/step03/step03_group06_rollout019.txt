import time
import numpy as np

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlate h with (1-h); mode='full' gives all integer shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_binary_vector(n, target, rng):
    """Binary vector with exactly `target` ones."""
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, target, replace=False)
    v[idx] = 1.0
    return v

def hill_climb(h, rng, deadline):
    """Deterministic hill climbing with single 1↔0 swaps."""
    n = h.shape[0]
    cur_val = compute_c5(h)
    # early exit if already excellent
    if cur_val < 0.3808:
        return h.copy(), cur_val
    while True:
        if time.time() >= deadline - 0.01:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_swap = None
        best_swap_val = cur_val
        # examine all possible swaps
        for i in ones:
            if time.time() >= deadline - 0.01:
                break
            for j in zeros:
                # tentative swap
                h[i] = 0.0
                h[j] = 1.0
                val = compute_c5(h)
                # revert
                h[i] = 1.0
                h[j] = 0.0
                if val + 1e-12 < best_swap_val:
                    best_swap_val = val
                    best_swap = (i, j)
                    if val < 0.3808:
                        break
            if best_swap and best_swap_val < 0.3808:
                break
        if best_swap:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            cur_val = best_swap_val
            if cur_val < 0.3808:
                break
        else:
            break
    return h.copy(), cur_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
    else:
        n = 80
        init_arr = None
    target = n // 2  # exactly half ones (n is even in all test cases)

    # Prepare initial candidate (binary, correct sum)
    best_h = None
    best_val = np.inf
    if init_arr is not None:
        # round to nearest integer and repair sum if needed
        cand = np.rint(init_arr).astype(int)
        cur_sum = cand.sum()
        if cur_sum != target:
            diff = target - cur_sum
            if diff > 0:
                zeros = np.where(cand == 0)[0]
                rng.shuffle(zeros)
                cand[zeros[:diff]] = 1
            else:
                ones = np.where(cand == 1)[0]
                rng.shuffle(ones)
                cand[ones[: -diff]] = 0
        h0 = cand.astype(float)
        h_opt, val_opt = hill_climb(h0, rng, deadline)
        best_h, best_val = h_opt, val_opt

    # Main random‑restart loop
    while time.time() < deadline - 0.5:
        h_start = random_binary_vector(n, target, rng)
        h_opt, val_opt = hill_climb(h_start, rng, deadline)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt
            if best_val < 0.3808:
                break

    # Fallback in the unlikely event no candidate was produced
    if best_h is None:
        best_h = random_binary_vector(n, target, rng)
        best_val = compute_c5(best_h)

    # Ensure feasibility (numerical safety)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Sum must be exactly target; binary vectors already satisfy it.
    # Return as required
    return best_h.astype(float), float(best_val), int(n)
