def adam_optimize(h0, time_left_func, max_iter=200000):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    s_target = h.size / 2
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iter+1):
        if time_left_func() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            grad += 1 - shift_array(h, shift) - shift_array(h, -shift)
        if len(shifts) > 0:
            grad /= len(shifts)
        # Adam update
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - (alpha / (np.sqrt(v_hat) + eps)) * m_hat
        # projection onto box and sum constraint
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    m = arr.shape[0]

    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Find lambda such that sum of clip(v - lam) = s
    # Use bisection.
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        total = w.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        # compute forward and backward shifted arrays
        if s >= 0:
            fwd = np.concatenate((np.zeros(s, dtype=h.dtype), h[:n - s]))
            bwd = np.concatenate((h[s:], np.zeros(s, dtype=h.dtype)))
        else:
            s_abs = -s
            fwd = np.concatenate((h[s_abs:], np.zeros(s_abs, dtype=h.dtype)))
            bwd = np.concatenate((np.zeros(s_abs, dtype=h.dtype), h[:n - s_abs]))
        grad += 1.0 - fwd - bwd
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

Now define adam_optimize:

def adam_optimize(h0, time_left, max_iter=200000, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 0
    while t < max_iter and time_left() > 0:
        t += 1
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # choose shifts near max
        threshold = max_corr - max(1e-12, 0.001 * max_corr)
        # shift indices offset: index - (n -1)
        shift_idxs = np.where(corr >= threshold)[0] - (n - 1)
        grad = compute_gradient(h, shift_idxs)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr / (np.sqrt(v_hat) + eps)
        h = h - step * m_hat
        # project
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, s_target)
        score = compute_c5_fft(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # early exit if target reached
        if best_score <= 0.3808:
            break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam_mid, lo, hi)
        total = w.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear up‑sampling of a periodic array from its current length to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def shift_array(a, s):
    """Return a shifted view of a with zero padding (a[i+s] for each i)."""
    n = a.shape[0]
    out = np.empty_like(a)
    if s >= 0:
        out[: n - s] = a[s:]
        if s:
            out[n - s :] = 0.0
    else:
        s = -s
        out[s:] = a[: n - s]
        out[:s] = 0.0
    return out


def compute_corr_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (length 2*len(h))."""
    n = h.shape[0]
    H = np.fft.rfft(h, n=2 * n)
    G = np.fft.rfft(1.0 - h, n=2 * n)
    conv = np.fft.irfft(H * np.conj(G), n=2 * n)
    return conv[: 2 * n - 1]  # length 2n‑1, lag –(n‑1)…+(n‑1)


def compute_c5_fft(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)


def compute_gradient(h, shifts):
    """Sub‑gradient of max‑correlation term for the given shift list."""
    grad = np.zeros_like(h)
    for s in shifts:
        fwd = shift_array(h, s)
        bwd = shift_array(h, -s)
        grad += 1.0 - fwd - bwd
    if shifts:
        grad /= len(shifts)
    return grad


def smooth(h, width=7):
    """Simple moving‑average smoothing (odd width)."""
    if width < 3:
        return h
    k = width
    kernel = np.ones(k) / k
    return np.convolve(h, kernel, mode='same')


def adam_optimize(h0, time_left, max_iter=200000, lr=0.05,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Projected Adam optimiser minimising C₅."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 0
    while t < max_iter and time_left() > 0.0:
        t += 1
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # pick up to 5 shifts with largest correlation values
        k = min(5, corr.size)
        top_idx = np.argpartition(-corr, k - 1)[:k]
        shifts = top_idx - (n - 1)
        grad = compute_gradient(h, shifts)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        h = h - step * m_hat

        # occasional smoothing to escape shallow plateaus
        if t % 1000 == 0:
            h = smooth(h, width=9)

        # projection onto box and sum constraint
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target_sum)

        score = compute_c5_fft(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if best_score <= 0.3808:
            break
    return best_h, best_score
