def gradient_descent_momentum_with_best(h0, max_iters, step0, beta, K, deadline):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time.time() > deadline - 2.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find top K shifts
        K_curr = K if K < corr.size else corr.size
        # argpartition may give unsorted order, that's okay
        idxes = np.argpartition(-corr, K_curr-1)[:K_curr]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                end = n - shift
                # first term
                grad[:end] += (1.0 - h[shift:])
                # second term
                grad[shift:] -= h[:end]
            else:
                s = -shift
                end = n - s
                grad[s:] += (1.0 - h[:end])
                grad[:end] -= h[s:]
        grad /= K_curr
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box and sum constraint
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, max_iter, temp_start, temp_end, rng, deadline):
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # maintain index sets of ones and zeros
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    for it in range(max_iter):
        if time.time() > deadline - 0.5:
            break
        # temperature schedule, geometric
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # pick random indices
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update ones, zeros after swap
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score
```

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # Find lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return the discrete approximation of C5 for vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue-Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Return a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_descent_momentum_with_best(h0, max_iters, step0, beta, K, deadline):
    """Projected gradient descent with momentum, keeping best encountered."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iters + 1):
        if time.time() > deadline - 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        # Determine top K shifts (indices of largest correlations)
        Kcurr = K if K < corr.size else corr.size
        idxes = np.argpartition(-corr, Kcurr - 1)[:Kcurr]
        shifts = idxes - (n - 1)

        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                end = n - s
                grad[:end] += (1.0 - h[s:])
                grad[s:] -= h[:end]
            else:
                sabs = -s
                end = n - sabs
                grad[sabs:] += (1.0 - h[:end])
                grad[:end] -= h[sabs:]
        grad /= Kcurr

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def anneal_binary(h0, max_iter, temp_start, temp_end, rng, deadline):
    """Simulated annealing on a balanced binary vector."""
    h = h0.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    # Initial index sets
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)

    for it in range(max_iter):
        if time.time() > deadline - 0.5:
            break
        # Geometric temperature schedule
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score

        accept = False
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update index sets
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_score

# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    n = 512  # target resolution

    # ----------------------------------------------------------------
    # Collect initial candidates
    # ----------------------------------------------------------------
    candidates = []

    # Parent construction, if available
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            init = upsample(init_full, n)
        else:
            init = init_full.copy()
        init = project_onto_box_sum(init, n / 2.0)
        candidates.append(init)

    # Thue-Morse binary
    candidates.append(thue_morse(n))

    # Some random balanced binaries
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    # Random continuous vectors (projected)
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    best_h = None
    best_score = np.inf

    for cand in candidates:
        if time.time() > deadline - 2.0:
            break
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # ----------------------------------------------------------------
    # Phase 1: Continuous gradient descent (if time permits)
    # ----------------------------------------------------------------
    if best_h is None:
        # fallback random start
        best_h = project_onto_box_sum(rng.random(n), n / 2.0)
        best_score = compute_c5(best_h)

    time_left = deadline - time.time()
    if time_left > 100.0:  # allocate at least ~100 s for gradient work
        max_iters = int(min(50000, time_left * 150))  # ~150 it/s estimate
        if max_iters > 0:
            h_gd, sc_gd = gradient_descent_momentum_with_best(
                best_h, max_iters, step0=0.5, beta=0.95, K=3, deadline=deadline
            )
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # ----------------------------------------------------------------
    # Phase 2: Convert to binary and anneal (dominant phase)
    # ----------------------------------------------------------------
    # Binary version from best continuous vector
    idx_top = np.argsort(best_h)[-n // 2 :]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx_top] = 1.0

    cur_h = h_bin.copy()
    cur_score = compute_c5(cur_h)
    if cur_score < best_score:
        best_score, best_h = cur_score, cur_h.copy()

    # Simulated annealing on binary
    time_left = deadline - time.time()
    if time_left > 30.0:
        max_iter_ann = int(time_left * 1000)  # ~1000 proposals/s
        if max_iter_ann > 0:
            h_ann, sc_ann = anneal_binary(
                cur_h,
                max_iter_ann,
                temp_start=0.02,
                temp_end=1e-6,
                rng=rng,
                deadline=deadline,
            )
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()

    # ----------------------------------------------------------------
    # Phase 3: Additional random binary sampling (fallback search)
    # ----------------------------------------------------------------
    while time.time() < deadline - 1.0:
        h_rand = random_binary_balanced(n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()

    # Final projection (ensure exact feasibility) and scoring
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
