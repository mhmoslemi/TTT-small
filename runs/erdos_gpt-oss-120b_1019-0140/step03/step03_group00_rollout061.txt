import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure interval
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term1[n-shift:] = 0.0
            term2[shift:] = h[:n-shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term1[:s] = 0.0
            term2[:n-s] = h[s:]
            term2[n-s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def de_optimize(base, n, rng, time_left, max_gen=5000, pop_size=None, F=0.8, CR=0.9):
    if pop_size is None:
        pop_size = max(15, int(10 + n / 20))
    pop = np.empty((pop_size, n), dtype=float)
    pop[0] = base.copy()
    for i in range(1, pop_size):
        v = rng.random(n)
        pop[i] = project_onto_box_sum(v, n/2)
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    gen = 0
    idx_all = np.arange(pop_size)
    while gen < max_gen and time_left() > 0.0:
        for i in range(pop_size):
            if time_left() <= 0.0:
                break
            # pick three distinct indices different from i
            candidates = idx_all[idx_all != i]
            a, b, c = rng.choice(candidates, size=3, replace=False)
            Fi = F if isinstance(F, float) else (0.5 + rng.random() * 0.5)
            mutant = pop[a] + Fi * (pop[b] - pop[c])
            mutant = project_onto_box_sum(mutant, n/2)
            # crossover
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_onto_box_sum(trial, n/2)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    else:
        base = thue_morse(64)

    best_h = base.copy()
    best_score = compute_c5(best_h)
    best_n = best_h.size

    # progressive refinement scales
    scales = [128, 256, 512]
    for n in scales:
        if time_left() <= 0:
            break
        # upsample current best to target size
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n/2.0)

        # allocate time slice for this scale
        remaining = time_left()
        if remaining <= 5.0:
            break
        # use half of remaining time for DE, half for local refinement
        de_time = remaining * 0.55
        de_end = time.time() + de_time
        de_best_h, de_best_score = de_optimize(
            base, n, rng,
            time_left=lambda: de_end - time.time(),
            max_gen=5000,
            pop_size=None,
            F=0.8,
            CR=0.9
        )
        if de_best_score < best_score:
            best_score, best_h = de_best_score, de_best_h.copy()
            best_n = n

        # local refinement with momentum gradient descent
        if time_left() > 2.0:
            iters = int(min(40000, max(3000, time_left() * (2000 if n <= 256 else 1000))))
            gd_h, gd_score = gradient_descent_momentum_custom(
                best_h, n_iter=iters, step0=0.5, beta=0.9,
                rng=rng, time_left=time_left)
            if gd_score < best_score:
                best_score, best_h = gd_score, gd_h.copy()
                best_n = n

    # final stochastic improvement loop
    while time_left() > 1.0:
        # random balanced binary perturbation
        h_rand = random_binary_balanced(best_n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        # small mass transfer between two positions
        i, j = rng.choice(best_n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def de_optimize(base, n, rng, time_left, max_gen=5000, pop_size=None, F=0.8, CR=0.9):
    if pop_size is None:
        pop_size = max(15, int(10 + n / 20))
    pop = np.empty((pop_size, n), dtype=float)
    pop[0] = base.copy()
    for i in range(1, pop_size):
        v = rng.random(n)
        pop[i] = project_onto_box_sum(v, n / 2.0)
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = compute_c5(pop[i])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_all = np.arange(pop_size)
    gen = 0
    while gen < max_gen and time_left() > 0.0:
        for i in range(pop_size):
            if time_left() <= 0.0:
                break
            candidates = idx_all[idx_all != i]
            a, b, c = rng.choice(candidates, size=3, replace=False)
            Fi = 0.5 + rng.random() * 0.5
            mutant = pop[a] + Fi * (pop[b] - pop[c])
            mutant = project_onto_box_sum(mutant, n / 2.0)
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_onto_box_sum(trial, n / 2.0)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)
    best_n = best_h.size

    scales = [128, 256, 512]
    for n in scales:
        if time_left() <= 0:
            break
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        remaining = time_left()
        if remaining <= 5.0:
            break
        de_time = remaining * 0.55
        de_end = time.time() + de_time
        de_h, de_score = de_optimize(
            base, n, rng,
            time_left=lambda: de_end - time.time(),
            max_gen=5000,
            pop_size=None,
            F=0.8,
            CR=0.9
        )
        if de_score < best_score:
            best_score, best_h, best_n = de_score, de_h.copy(), n

        if time_left() > 2.0:
            iters = int(min(40000, max(3000, time_left() * (2000 if n <= 256 else 1000))))
            gd_h, gd_score = gradient_descent_momentum_custom(
                best_h, n_iter=iters, step0=0.5, beta=0.9,
                rng=rng, time_left=time_left
            )
            if gd_score < best_score:
                best_score, best_h, best_n = gd_score, gd_h.copy(), n

    while time_left() > 1.0:
        h_rand = random_binary_balanced(best_n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        i, j = rng.choice(best_n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
