import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        seq[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
    return seq

def project_onto_feasible(h, M):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum by uniform shift
    err = M - h.sum()
    if abs(err) > 1e-12:
        h += err / h.size
        h = np.clip(h, 0.0, 1.0)
        # Second correction if needed
        err2 = M - h.sum()
        if abs(err2) > 1e-12:
            # distribute remaining error proportionally to free space
            # simple: add/subtract uniformly again
            h += err2 / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_step(h, M, step):
    n = h.size
    # compute correlation to get shift with max overlap
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argmax(corr) - (n - 1)
    k = k_idx
    g = np.zeros_like(h)
    if k >= 0:
        # first term: (1 - h[i+k])
        g[:n - k] += (1.0 - h[k:])
        # second term: -h[i - k] -> -h[i - k] for i >= k
        g[k:] -= h[:n - k]
    else:
        kk = -k
        # first term: (1 - h[i-kk])
        g[kk:] += (1.0 - h[:n - kk])
        # second term: -h[i+kk] for i < n-kk
        g[:n - kk] -= h[kk:]
    # gradient descent
    h_new = h - step * g
    return project_onto_feasible(h_new, M)

def random_fractional_move(h, M, rng):
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    # transfer from i to j
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        # try opposite direction
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return h
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
        return h_new
    delta = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def binary_swap_move(h, rng):
    n = h.size
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h > 0.5)[0]
    if zeros.size == 0 or ones.size == 0:
        return h
    i = rng.choice(zeros)
    j = rng.choice(ones)
    h_new = h.copy()

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        seq[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
    return seq

def project_onto_feasible(h, M):
    h = np.clip(h, 0.0, 1.0)
    err = M - h.sum()
    if abs(err) > 1e-12:
        h += err / h.size
        h = np.clip(h, 0.0, 1.0)
        err2 = M - h.sum()
        if abs(err2) > 1e-12:
            h += err2 / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_step(h, M, step):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = np.argmax(corr) - (n - 1)
    g = np.zeros_like(h)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    h_new = h - step * g
    return project_onto_feasible(h_new, M)

def random_fractional_move(h, rng):
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return h
        delta = rng.random() * max_transfer
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
        return h_new
    delta = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
