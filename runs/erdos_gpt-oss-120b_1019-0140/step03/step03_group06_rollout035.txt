def compute_c5(h):

def enforce_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Scale to target proportionally
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # Distribute diff iteratively
    rng = np.random.default_rng()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 0.0)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None

    target = n / 2.0

    # GA parameters
    pop_size = 200
    max_generations = 1000000
    mutation_sigma = 0.2
    mutation_prob = 0.3

    # Initialize population
    population = []
    if h0 is not None:
        population.append(enforce_sum(h0.copy(), target, rng))

    # Add random individuals
    while len(population) < pop_size:
        v = rng.random(n)
        v = enforce_sum(v, target, rng)
        population.append(v)

    population = np.array(population)

    # Pre-evaluate fitness
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]

    generation = 0
    # Main GA loop
    while time.time() < deadline - 10.0 and generation < max_generations:
        generation += 1
        # Generate offspring
        offspring = []
        for _ in range(pop_size):
            # select parents via tournament
            p1 = tournament_select(population, fitness, rng)
            p2 = tournament_select(population, fitness, rng)
            child = crossover(p1, p2, rng)
            child = mutate(child, target, rng, sigma=mutation_sigma, mutation_prob=mutation_prob)
            offspring.append(child)
        offspring = np.array(offspring)
        # Evaluate offspring fitness
        offspring_fitness = np.array([compute_c5(ind) for ind in offspring])

        # Combine and select best pop_size
        combined = np.concatenate((population, offspring), axis=0)
        combined_fitness = np.concatenate((fitness, offspring_fitness), axis=0)

        # Sort indices by fitness
        idx_sort = np.argsort(combined_fitness)
        population = combined[idx_sort[:pop_size]]
        fitness = combined_fitness[idx_sort[:pop_size]]

        # Update best
        if fitness[0] < best_val:
            best_val = fitness[0]
            best_h = population[0].copy()
            # Early exit if target reached
            if best_val <= 0.3808:
                break

        # Adaptive: if no improvement for many generations, increase mutation chance
        if generation % 100 == 0:
            # maybe random restart a fraction
            if best_val > 0.382:
                # replace worst 30% with random
                n_replace = int(0.3 * pop_size)
                for i in range(pop_size - n_replace, pop_size):
                    v = rng.random(n)
                    v = enforce_sum(v, target, rng)
                    population[i] = v
                    fitness[i] = compute_c5(v)
    # After GA, local refinement via annealing
    # Define annealing
    def anneal(h, cur_val):
        n = h.shape[0]
        dx = 2.0 / n
        while time.time() < deadline - 1.0:
            # Compute product matrix for worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)
            # Determine indices contributing to max product
            if k >= 0:
                prod = h[: n - k] * (1.0 - h[k:])
                i_range = np.arange(n - k)
            else:
                kk = -k
                prod = h[kk:] * (1.0 - h[: n - kk])
                i_range = np.arange(kk, n)
            i_local = np.argmax(prod)
            i = i_range[i_local]

            # Choose j to receive mass, prefer where (1-h) is larger and h is small
            # Sample candidate pool
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            j = rng.choice(cand)

            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
            # Temperature schedule: remaining fraction of time
            remaining = deadline - time.time()
            total = deadline - start_time
            T = remaining / total  # in [0,1]
            # Propose delta scaled by temperature
            delta = rng.random() * max_transfer * T
            if delta <= 0:
                continue
            # Apply move
            h[i] -= delta
            h[j] += delta
            new_val = compute_c5(h)
            if new_val < cur_val or T > rng.random():
                # Accept
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                    if best_val <= 0.3808:
                        return best_h, best_val
            else:
                # Reject: revert
                h[i] += delta
                h[j] -= delta
        return h, cur_val

    # Run anneal
    best_h, best_val = anneal(best_h.copy(), best_val)

    # Ensure final constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n

def local_anneal(h, cur_val, start_time, deadline, rng):
    n = h.shape[0]
    while time.time() < deadline - 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # candidate for destination
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:
            break
        j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # Temperature schedule: linear decreasing to 0
        remaining = deadline - time.time()
        total = deadline - start_time
        T = remaining / total
        # delta scaled
        delta = rng.random() * max_transfer * T
        if delta <= 0:
            continue
        # Apply move
        h[i] -= delta
        h[j] += delta
        new_val = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
        if new_val < cur_val or T > rng.random():
            cur_val = new_val
            if new_val < 0.3808:
                # early exit

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Clip v to [0,1] and adjust so that sum(v) == target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return v
    if total > 0.0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v

def tournament_select(pop, fitness, rng, tournament_size=2):
    """Select one individual by tournament."""
    idx = rng.choice(len(pop), tournament_size, replace=False)
    best = idx[0]
    for i in idx[1:]:
        if fitness[i] < fitness[best]:
            best = i
    return pop[best]

def crossover(p1, p2, rng):
    """Convex combination of two parents."""
    alpha = rng.random()
    return alpha * p1 + (1.0 - alpha) * p2

def mutate(child, target, rng, sigma=0.2, prob=0.3):
    """Additive noise mutation with sum repair."""
    if rng.random() < prob:
        noise = sigma * (rng.random(child.shape) - 0.5)
        child = child + noise
    child = enforce_sum(child, target, rng)
    return child

def local_anneal(h, cur_val, start_time, deadline, rng):
    """Simple simulated annealing that moves a small mass between points."""
    n = h.shape[0]
    total_time = deadline - start_time
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where((h < 1.0 - 1e-12) & (np.arange(n) != i))[0]
        if cand.size == 0:
            break
        j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        remaining = deadline - time.time()
        T = remaining / total_time
        delta = rng.random() * max_transfer * T
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        new_val = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < T:
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:

            h[i] += delta
            h[j] -= delta

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = int(h0.shape[0])
    else:
        n = 80
        h0 = None

    target = n / 2.0
    pop_size = 200
    mutation_sigma = 0.2
    mutation_prob = 0.3

    # Initialise population
    population = []
    if h0 is not None:
        population.append(enforce_sum(h0.copy(), target, rng))
    while len(population) < pop_size:
        v = rng.random(n)
        v = enforce_sum(v, target, rng)
        population.append(v)
    population = np.array(population)
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]

    gen_no_improve = 0
    generation = 0
    max_gens = 10_000_000  # effectively unlimited within time
    while time.time() < deadline - 10.0 and generation < max_gens:

        # generate offspring

        for _ in range(pop_size):

            p2 = tournament_select(population, fitness, rng)

                           sigma=mutation_sigma,

        fitness = combined_fitness[idx_sort[:pop_size]]

            gen_no_improve += 1

        # occasional diversification
        if generation % 100 == 0 and gen_no_improve > 200:
            n_replace = int(0.3 * pop_size)
            for i in range(pop_size - n_replace, pop_size):
                v = rng.random(n)
                v = enforce_sum(v, target, rng)
                population[i] = v
                fitness[i] = compute_c5(v)
            gen_no_improve = 0

    # Local continuous refinement
    best_h, best_val = local_anneal(best_h.copy(),
                                    best_val,
                                    start_time,
                                    deadline,
                                    rng)

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
