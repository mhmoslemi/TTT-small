def gradient_descent_refine(h_start, target_sum, max_iter, deadline):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.default_rng().normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    # finally return best h and best_c5
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------- Global target sum for DE objective -------------
GLOBAL_TARGET_SUM = 0.0

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent_refine(h_start, target_sum, max_iter, deadline):
    """Projected gradient descent on the max-overlap objective, respecting time."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    rng_local = np.random.default_rng()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng_local.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def objective_t(x):
    return x[-1]

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def de_objective(v):
    """Objective for DE: compute C5 after projection onto feasible set."""
    h = project_to_box_sum(np.clip(v, 0.0, 1.0), GLOBAL_TARGET_SUM)
    return compute_c5(h)

class DEState:
    """Callable state object for DE callback, tracking best feasible solution."""
    def __init__(self, start_time, budget_s):
        self.start_time = start_time
        self.budget_s = budget_s
        self.best_h = None
        self.best_c5 = float('inf')
    def __call__(self, xk, convergence):
        now = time.time()
        if now - self.start_time > self.budget_s - 5:
            return True
        h = project_to_box_sum(np.clip(xk, 0.0, 1.0), GLOBAL_TARGET_SUM)
        c5 = compute_c5(h)
        if c5 < self.best_c5:
            self.best_c5 = c5
            self.best_h = h.copy()
            if c5 < 0.3808:
                return True
        return False

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Initialise problem size and feasible start vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    global GLOBAL_TARGET_SUM
    GLOBAL_TARGET_SUM = target_sum
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Quick local refinement ----
    deadline = start_time + budget_s * 0.1
    gd_h, gd_c5 = gradient_descent_refine(best_h, target_sum,
                                          max_iter=2000, deadline=deadline)
    if gd_c5 < best_c5:
        best_h, best_c5 = gd_h, gd_c5

    # ---- SLSQP from current best ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    t0 = best_c5
    res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=300)
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # ---- Main optimisation loop (DE + local) ----
    while time.time() < start_time + budget_s - 5:
        remaining = start_time + budget_s - 5 - time.time()
        if remaining < 5:
            break

        # Differential Evolution batch
        de_state = DEState(start_time, budget_s)
        try:
            differential_evolution(
                de_objective,
                bounds=[(0.0, 1.0)] * n,
                seed=seed,
                maxiter=80,
                popsize=5,
                polish=False,
                init='random',
                callback=de_state,
                disp=False
            )
        except Exception:
            pass

        if de_state.best_h is not None and de_state.best_c5 < best_c5:
            best_h, best_c5 = de_state.best_h, de_state.best_c5

        # Local gradient descent refinement
        deadline = time.time() + min(remaining, 30)
        gd_h, gd_c5 = gradient_descent_refine(best_h, target_sum,
                                              max_iter=1500, deadline=deadline)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5

        # SLSQP refinement if time permits
        remain = start_time + budget_s - 5 - time.time()
        if remain > 5:
            t_start = compute_c5(best_h)
            res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=200)
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

        if best_c5 < 0.3808:
            break

    # ---- Final safety projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
