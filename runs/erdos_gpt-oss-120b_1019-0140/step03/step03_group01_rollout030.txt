def binary_anneal_fast(h0, rng, time_left_func, max_iters=10**9):
    h = h0.copy()
    n = len(h)
    offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    if init_score is None:
        best_score = np.max(corr) * (2.0 / n)
    else:
        best_score = init_score
    best_h = h.copy()
    current_score = best_score
    T = 0.1
    T_min = 1e-6
    decay = 0.99995
    it = 0
    # precompute 1 - h and maybe h_rev, but they change each swap.
    while it < max_iters and time_left_func() > 0.0:
        # choose random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        d_i = -1.0
        d_j = 1.0
        # compute delta array
        delta = np.zeros_like(corr)
        # term A for i
        delta[offset - i : offset - i + n] += d_i * (1.0 - h)
        # term A for j
        delta[offset - j : offset - j + n] += d_j * (1.0 - h)
        # term B for i
        delta[i : i + n] += -d_i * h[::-1]
        # term B for j
        delta[j : j + n] += -d_j * h[::-1]
        # term C
        delta[offset] += -(d_i*d_i + d_j*d_j)  # -2
        delta[offset + (j - i)] += - (d_i * d_j)  # -(-1) = +1? Wait: subtract product = -(d_i*d_j)
        # Let's compute correctly:
        delta[offset + (j - i)] += - (d_i * d_j)
        delta[offset + (i - j)] += - (d_j * d_i)
        # the second is same value.
        # Since d_i * d_j = -1, -(-1) = +1.
        # So correct.

        # update corr
        corr += delta

        # swap values in h
        h[i] = 0.0
        h[j] = 1.0

        new_score = np.max(corr) * (2.0 / n)
        delta_score = new_score - current_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta_score / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # revert corr
            corr -= delta
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

import time
import math
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.5, beta=0.9,
                                        max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=800):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def binary_anneal_fast(h0, rng, time_left_func, max_iters=10 ** 9):
    """Swap-anneal on a binary vector using O(n) updates of the full correlation."""
    h = h0.astype(float).copy()
    n = h.shape[0]
    offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    current_score = np.max(corr) * dx
    best_score = current_score
    best_h = h.copy()
    T = 0.1
    T_min = 1e-6
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        d_i = -1.0
        d_j = 1.0
        delta = np.zeros_like(corr)

        # term A: d_p * (1 - h_{p+shift})
        delta[offset - i: offset - i + n] += d_i * (1.0 - h)
        delta[offset - j: offset - j + n] += d_j * (1.0 - h)

        # term B: -h_{p-shift} * d_p  (implemented via reversed h)
        h_rev = h[::-1]
        delta[i: i + n] += -d_i * h_rev
        delta[j: j + n] += -d_j * h_rev

        # term C: -d_p d_{p+shift}
        delta[offset] += -(d_i * d_i + d_j * d_j)          # shift 0
        delta[offset + (j - i)] += - (d_i * d_j)           # shift j-i
        delta[offset + (i - j)] += - (d_j * d_i)           # shift i-j

        # apply update
        corr += delta
        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        new_score = np.max(corr) * dx
        delta_score = new_score - current_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta_score / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            corr -= delta
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

        return max(0.0, budget_s - (time.time() - start))

                                                            step0=0.4, beta=0.9,
                                                            max_iter=300000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 4: binary conversion ----------
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ---------- Phase 5: fast binary annealing ----------
    if time_left() > 0.5:
        h_bin_ref, sc_bin_ref = binary_anneal_fast(best_h, rng, time_left)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 6: upsample to finer grid and anneal ----------
    if time_left() > 120.0 and best_score > target:
        n_fine = 512
        h_fine = upsample(best_h, n_fine)
        h_fine = project_onto_box_sum(h_fine, n_fine / 2.0)
        idx_f = np.argpartition(-h_fine, n_fine // 2)[:n_fine // 2]
        h_bin_f = np.zeros_like(h_fine)
        h_bin_f[idx_f] = 1.0
        sc_f = compute_c5(h_bin_f)
        if sc_f < best_score:
            best_score, best_h = sc_f, h_bin_f.copy()
        if time_left() > 0.5:
            h_bin_ref2, sc_bin_ref2 = binary_anneal_fast(best_h, rng, time_left)
            if sc_bin_ref2 < best_score:
                best_score, best_h = sc_bin_ref2, h_bin_ref2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 7: final continuous refinement ----------
    if time_left() > 0.5:
        h_ref2, sc_ref2 = gradient_descent_momentum_with_time(best_h, time_left,
                                                              step0=0.3, beta=0.9,
                                                              max_iter=200000)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # final projection guard
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
