def anneal(start_h, start_val, deadline, rng):
    n = len(start_h)
    h = start_h.copy()
    cur_val = start_val
    best_h = h.copy()
    best_val = cur_val
    total_iters = 0
    while time.time() < deadline - 0.1:
        # compute temperature based on time left
        rem = deadline - time.time()
        frac = rem / (deadline - start_time)
        T = max_temp * (frac ** 1.0)  # linear cooling
        # find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # worst index
        if k >= 0:
            prod = h[: n - k] * (1 - h[k:])
            i_local = np.argmax(prod)
            src = i_local
        else:
            kk = -k
            prod = h[kk:] * (1 - h[: n - kk])
            i_local = np.argmax(prod)
            src = i_local + kk
        # choose destination with room
        cand = np.where(h < 1 - 1e-12)[0]
        if cand.size == 0:
            break
        # ensure destination not same as src and has room
        dest_candidates = cand[cand != src]
        if dest_candidates.size == 0:
            break
        dst = rng.choice(dest_candidates)
        max_transfer = min(h[src], 1 - h[dst])
        if max_transfer <= 0:
            continue
        # step factor based on T
        step_factor = T / max_temp if max_temp > 0 else 1.0
        delta = step_factor * max_transfer * rng.random()
        # if delta is zero, continue
        if delta <= 0:
            continue
        h[src] -= delta
        h[dst] += delta
        new_val = compute_c5(h)
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                # optionally break if target reached
                if best_val < target_bound: break
        else:
            # revert move
            h[src] += delta
            h[dst] -= delta
        total_iters += 1
    return best_h, best_val
```

def enforce_sum(v, target, rng, eps=1e-12):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= eps:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute diff
    attempts = 0
    while abs(diff) > eps and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - eps)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > eps)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def anneal(start_h, start_val, overall_start, overall_deadline, sub_deadline, rng, early_stop):
    h = start_h.copy()
    cur_val = start_val
    best_h = h.copy()
    best_val = cur_val
    max_temp = 0.02
    n = h.shape[0]
    while time.time() < sub_deadline:
        # temperature linearly decreasing with overall progress
        progress = (time.time() - overall_start) / (overall_deadline - overall_start)
        T = max_temp * max(0.0, 1.0 - progress)
        # identify worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # find index contributing most
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_local = np.argmax(prod)
            src = i_local
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_local = np.argmax(prod)
            src = i_local + kk
        # choose destination with room
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:
            break
        # exclude src
        dest_cand = cand[cand != src]
        if dest_cand.size == 0:
            # if no other, pick any with room (maybe src also)
            dest_cand = cand
        dst = rng.choice(dest_cand)
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0:
            continue
        # step size factor
        step_factor = T / max_temp if max_temp > 0 else 1.0
        delta = step_factor * max_transfer * rng.random()
        if delta <= 0:
            continue
        h[src] -= delta
        h[dst] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < early_stop:
                    return best_h, best_val
        else:
            # accept with probability exp((cur_val - new_val)/T)
            if T > 0 and rng.random() < np.exp((cur_val - new_val) / (T + 1e-12)):
                cur_val = new_val
            else:
                # revert
                h[src] += delta
                h[dst] -= delta
    return best_h, best_val

def hill_climb_binary(start_h, start_val, overall_start, overall_deadline, rng, early_stop):
    h = start_h.copy()

    best_val = cur_val

        T = max_temp * max(0.0, 1.0 - progress)
        # list indices of 1s and 0s
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        hv = best_h.copy()
        hv[i] = 0.0
        hv[j] = 1.0

                best_val = new_val
                best_h = hv
                cur_val = new_val
            # else reject, keep best_h unchanged
    return best_h, best_val
