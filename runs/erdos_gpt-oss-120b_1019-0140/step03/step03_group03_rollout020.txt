def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < 2000000 and time_left() > 0.01:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]

        # compute gradient for these shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = h[shift:]
                grad[:n-shift] -= term1  # subtract because derivative w.r.t h[0:n-shift] includes -h[i+shift]

def anneal_binary(h0, time_left, temp_start=0.1, temp_end=1e-6):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = list(np.flatnonzero(cur_h > 0.5))
    zeros = list(np.flatnonzero(cur_h < 0.5))
    it = 0
    max_iter = 10**9  # large, break using time
    while time_left() > 0 and it < max_iter:
        it += 1
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # sample i (index in ones)
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = delta <= 0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            # update lists
            ones[idx_i] = j
            zeros[idx_j] = i
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        # optional early break if best_score is below target and time left small.
    return best_h, best_score

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() < 1e-6:
            break
        # compute gradient for worst shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')

def anneal_binary(h0, time_left, rng, temp_start=0.1, temp_end=1e-6):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = list(np.flatnonzero(cur_h > 0.5))
    zeros = list(np.flatnonzero(cur_h < 0.5))
    it = 0
    # compute total allowed iterations based on time (example 2000 per sec)
    max_iters_est = int(time_left() * 3000) + 1
    while it < max_iters_est and time_left() > 0.001:
        it += 1
        # temperature schedule
        t = temp_start * ((temp_end / temp_start) ** (it / max_iters_est))
        # random indices
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = (delta <= 0) or (rng.random() < np.exp(-delta / (t + 1e-12)))
        if accept:
            cur_score = new_score
            # update index lists
            ones[idx_i] = j
            zeros[idx_j] = i
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h[shift:] added to positions 0:n-shift
            grad[:n - shift] -= h[shift:]
            # term2 = h[:n - shift] added to positions shift:n
            grad[shift:] -= h[:n - shift]

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
