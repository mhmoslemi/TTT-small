```python
def project_cap_simplex(x, M):
    # projection onto {0 <= y <= 1, sum(y)=M}
    # Using bisection for lambda
    n = x.shape[0]
    # define function sum_clipped(lam)
    def sum_clipped(lam):
        return np.clip(x - lam, 0, 1).sum()
    lo = -np.max(x) - 1.0  # ensure sum at lo >= M (most entries 1)
    hi = np.max(x) + 1.0   # ensure sum at hi <= M (most entries 0)
    # adjust bounds to guarantee sign
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = sum_clipped(mid)
        if s > M:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, 0, 1)
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p ** 0.5) + 1
    for i in range(3, r, 2):
        if p % i == 0:
            return False
    return True

def legendre_sequence(p):
    """Return 0/1 Legendre sequence of length p (prime)."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def project_cap_simplex(x, M, tol=1e-12, max_iter=60):
    """Project vector x onto {0<=y<=1, sum(y)=M}."""
    # Bisection on lambda for y_i = clip(x_i - lam, 0, 1)
    lo = -np.max(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > M:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    for i, n in enumerate(ns):
        if time.time() - start_total > budget_s - 5.0:
            break
        remaining_total = budget_s - (time.time() - start_total)
        # allocate a share of remaining time (more for earlier n)
        alloc = min(150.0, remaining_total * 0.6)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # safety fallback
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        # final projection to enforce constraints exactly
        best_h = np.clip(best_h, 0.0, 1.0)
        M = best_n // 2
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / best_n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
