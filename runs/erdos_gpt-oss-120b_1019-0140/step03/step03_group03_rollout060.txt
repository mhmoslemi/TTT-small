def solve_milp_binary(n, time_limit=120):
   h = cp.Variable(n, boolean=True)
   t = cp.Variable()
   # Define y matrix for product of h_i * h_j
   Y = cp.Variable((n, n), boolean=True)
   constraints = []
   constraints.append(cp.sum(h) == n/2)
   constraints.append(t >= 0)
   # linking Y with h
   for i in range(n):
       constraints.append(Y[i,i] == h[i])
   for i in range(n):
       for j in range(i+1, n):
           constraints.append(Y[i,j] <= h[i])
           constraints.append(Y[i,j] <= h[j])
           constraints.append(Y[i,j] >= h[i] + h[j] - 1)
   # shift constraints
   for k in range(1, n):
       # sum_i h_i - sum_{i} Y[i,i+k] <= t
       # Use slice indices
       idx_i = np.arange(n - k)
       idx_j = idx_i + k
       constraints.append(cp.sum(h[idx_i]) - cp.sum(Y[idx_i, idx_j]) <= t)
   prob = cp.Problem(cp.Minimize(t), constraints)
   # Solve with CBC/ECOS_BB
   try:
       prob.solve(solver=cp.CBC, timeLimit=time_limit, warm_start=True, verbose=False)
   except Exception:
       try:
           prob.solve(solver=cp.ECOS_BB, max_iters=500000, verbose=False)
       except Exception:
           # fallback
           prob.solve(solver=cp.SCS, verbose=False)
   if h.value is None:
       # fallback: random binary
       rng = np.random.default_rng()
       h_val = np.zeros(n)
       idx = rng.choice(n, size=n//2, replace=False)

def simulated_annealing_binary(h0, max_iter, rng):
    h = h0.copy()
    cur_max = max_corr_count(h)
    best_h = h.copy()
    best_max = cur_max

    # set temperature schedule
    start_temp = 2.0  # need to calibrate
    end_temp = 1e-4

    for it in range(max_iter):
        # linear schedule in log scale
        t = start_temp * ((end_temp / start_temp) ** (it / max_iter))
        # pick random 1 and 0 indices
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_max = max_corr_count(h)
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
    return best_h, best_max

def anneal_binary_timed(h0, time_budget, rng):
    start = time.time()
    h = h0.copy()
    cur_max = max_corr_count(h)
    best_h = h.copy()
    best_max = cur_max
    start_temp = 2.0
    end_temp = 1e-4

        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # schedule
        frac = elapsed / time_budget
        temp = start_temp * ((end_temp / start_temp) ** frac)
        # choose swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)

        j = rng.choice(zeros)

            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def max_corr_count(h):
    """Maximum integer count of h[i]*(1−h[i+k]) over shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.max(corr) + 0.5)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary_timed(h0, time_budget, rng):
    """Simulated annealing on a binary balanced vector for a fixed time."""
    start = time.time()
    h = h0.copy()
    cur_max = max_corr_count(h)
    best_h = h.copy()
    best_max = cur_max
    start_temp = 2.0
    end_temp = 1e-4

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        temp = start_temp * ((end_temp / start_temp) ** frac)

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i], h[j] = 0.0, 1.0
        new_max = max_corr_count(h)
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_max


def hill_climb_binary(h0, max_swaps, rng):
    """Simple hill climbing on binary vector (accept only improving swaps)."""
    h = h0.copy()
    cur_max = max_corr_count(h)
    for _ in range(max_swaps):
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_max = max_corr_count(h)
        if new_max <= cur_max:
            cur_max = new_max
        else:
            h[i], h[j] = 1.0, 0.0
    return h, cur_max
