import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def smooth_gradient_descent(h_start, target, max_iter, time_limit,
                           beta_start=1.0, beta_max=15.0, beta_step=0.03,
                           alpha_start=0.5, momentum=0.9):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = h.shape[0]
    vel = np.zeros_like(h)
    alpha = alpha_start
    beta = beta_start
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        cur_c5 = max_corr * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # weights for smoothing
        delta = corr - max_corr
        w = np.exp(beta * delta)
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.zeros_like(w)
        else:
            w_norm = w / w_sum
        # weighted gradient
        grad = np.zeros_like(h)
        for idx, weight in enumerate(w_norm):
            if weight == 0.0:
                continue
            k = idx - (n - 1)
            grad += weight * gradient_for_shift(h, k)
        # momentum update
        vel = momentum * vel + grad
        step = -alpha * vel
        new_h = project_to_box_sum(h + step, target)
        new_c5 = compute_c5(new_h)
        if new_c5 <= cur_c5:
            h = new_h
            alpha = min(alpha * 1.02, 2.0)
        else:
            alpha = max(alpha * 0.5, 1e-4)
        if beta < beta_max:
            beta = min(beta + beta_step, beta_max)
    return best_h, best_c5

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(objective_t, x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    safety = 5.0
    end_time = start_time + budget_s - safety

    # dimensionality
    n = 200
    target = n / 2.0

    # initialise from parent if available, upsample if needed
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != n:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target)
    else:
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # main stochastic search
    while time.time() < end_time:
        # generate start point
        if rng.random() < 0.6:
            start = best_h + rng.normal(scale=0.03, size=n)
        else:
            start = rng.random(n)
        start = project_to_box_sum(np.clip(start, 0.0, 1.0), target)

        # smooth gradient descent
        max_iters = 5000
        cand_h, cand_c5 = smooth_gradient_descent(start, target, max_iters, end_time)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
            if best_c5 < 0.3808:
                break

    # final SLSQP refinement if time permits
    if time.time() < start_time + budget_s - safety - 5.0:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=800)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
