def gradient_descent_subgrad(h0, max_iter, time_left_func, step0=1.0, beta=0.9, eps=1e-12):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        # check time
        if time_left_func() <= 0:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # Identify active shifts
        active = np.where(corr >= max_corr - eps)[0]
        shifts = active - (n - 1)
        # compute gradient for active shifts
        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                grad[:n-s] += 1 - h[s:]
                grad[s:] += -h[:n-s]
            else:
                s2 = -s
                grad[s2:] += 1 - h[:n-s2]
                grad[:n-s2] += -h[s2:]
        # average over number of shifts:
        if len(shifts) > 0:
            grad = grad / len(shifts)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        # step size schedule
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral on the discretised grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def resample(arr, new_n):
    """Linear interpolation of a vector to a new length."""
    if arr.size == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, arr.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Binary Thue‑Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    return h


def gradient_descent_subgrad(h0, max_iter, time_left_func,
                            step0=1.0, beta=0.9, eps=1e-12):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)

    for it in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        active = np.where(corr >= max_corr - eps)[0]
        shifts = active - (n - 1)

        grad = np.zeros_like(h)
        for s in shifts:
            if s >= 0:
                grad[:n - s] += 1.0 - h[s:]
                grad[s:] += -h[:n - s]
            else:
                s2 = -s
                grad[s2:] += 1.0 - h[:n - s2]
                grad[:n - s2] += -h[s2:]
        if shifts.size:
            grad /= shifts.size

        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary(h_bin, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on a balanced binary vector."""
    cur = h_bin.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        ones = np.flatnonzero(cur > 0.5)
        zeros = np.flatnonzero(cur < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i], cur[j] = 0.0, 1.0
        new_score = compute_c5(cur)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
    return best, best_score
