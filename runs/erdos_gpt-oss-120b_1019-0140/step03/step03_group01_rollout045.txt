def solve_milp_binary(n, time_left):
    import cvxpy as cp
    # Create binary variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2, t >= 0]
    # Symmetry break
    constraints.append(h[0] == 1)

    # For each shift k
    for k in range(1, n):
        # overlapping length = n - k
        length = n - k
        # define product variables and constraints
        # Use cp.Variable for product?
        # cp.Variable(boolean?) Actually product of two binaries is also binary, but we can define integer variable
        # with constraints as described.
        # We'll allocate z_i_k as cp.Variable(length, boolean=True) maybe not needed boolean, but can be integer.
        z = cp.Variable(length, boolean=True)  # product variables also binary

        # constraints to enforce z = h[i] * h[i + k] for each overlapping index i
        constraints.append(z <= h[:length])
        constraints.append(z <= h[k:])
        constraints.append(z >= h[:length] + h[k:] - 1)
        # Not necessary to add lower bound for z as boolean ensures >=0

        # correlation constraint: sum_i h[i] - sum_i z_i <= t
        constraints.append(cp.sum(h[:length]) - cp.sum(z) <= t)

    # objective
    obj = cp.Minimize(t)

    # Solve
    prob = cp.Problem(obj, constraints)

    # solver selection; try CBC, fallback ECOS_BB
    try:
        prob.solve(solver=cp.CBC, timeLimit=int(time_left()))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS_BB, max_iters=200000)
        except Exception:
            prob.solve(solver=cp.SCS)  # fallback, but might be continuous relaxation
    # Retrieve solution
    if h.value is None:
        return None, None

    h_opt = np.rint(h.value).astype(float)  # binary values
    # Ensure constraints (in case rounding).
    h_opt = project_onto_box_sum(h_opt, n/2)
    # compute score
    sc = compute_c5(h_opt)
    return h_opt, sc

import time
import numpy as np
import cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # Find suitable lambda range
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Alias for upsample when reducing size (linear interpolation)."""
    return upsample(arr, new_n)


def thue_morse(n):
    """Balanced Thue–Morse sequence (binary 0/1) of length n (n power of two)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max correlation w.r.t. h (averaged over shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    """Momentum gradient descent on continuous h with projection onto box+sum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score


def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    """Simulated annealing on binary vectors (swap a 1 with a 0)."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score


def solve_milp_binary(n, time_budget):
    """
    Solve the binary optimization for C5 using a MILP formulation.
    Returns (h_opt, score) or (None, None) if infeasible or timed out.
    """
    # time_budget is the maximum seconds allocated to the solver.
    # Build variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n / 2.0, t >= 0.0, h[0] == 1]  # symmetry break

    # For each shift k >= 1, introduce product variables and constraints.
    for k in range(1, n):
        length = n - k
        z = cp.Variable(length, boolean=True)  # product h[i] * h[i+k]
        constraints.append(z <= h[:length])
        constraints.append(z <= h[k:])
        constraints.append(z >= h[:length] + h[k:] - 1)
        constraints.append(cp.sum(h[:length]) - cp.sum(z) <= t)

    obj = cp.Minimize(t)

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_budget)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS_BB, max_iters=200000)
        except Exception:
            # fallback to continuous relaxation (not useful)
            prob.solve(solver=cp.SCS, max_iters=2000)

    if h.value is None:
        return None, None

    h_opt = np.rint(h.value).astype(float)
    # Ensure exact sum constraint (project if tiny drift)
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    score = compute_c5(h_opt)
    return h_opt, score

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

        return max(0.0, budget_s - (time.time() - start))

    best_h = None
    best_score = np.inf

    def consider(candidate):
        nonlocal best_h, best_score
        if candidate is None:
            return
        sc = compute_c5(candidate)

            best_score = sc

    # -------------------------------------------------

        h_milp, sc_milp = solve_milp_binary(n_milp, milp_time)
        consider(h_milp)

    # -------------------------------------------------

    # -------------------------------------------------
    n_small = 64
    candidates = []
    # deterministic candidates

    # few random continuous seeds

        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    # also include best from previous phases if size matches

    for cand in candidates:
        if time_left() < 5.0:

    # Phase 3: Upsample to larger grid and refine
    # -------------------------------------------------
    n_large = 256
    if best_h is None:

    consider(h_up)

    # Further gradient refinement on large grid
    if time_left() > 2.0:
        h_ref_big, sc_ref_big = gradient_descent_momentum_with_time(
            best_h, time_left, step0=0.4, beta=0.85, max_iter=300000)
        consider(h_ref_big)

    # Phase 4: Binary rounding and annealing
    # -------------------------------------------------
    if best_h is not None:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        consider(h_bin)

        if time_left() > 1.0:
            h_bin_ref, sc_bin_ref = binary_swap_anneal(
                h_bin, rng, time_left, max_iters=400000)
            consider(h_bin_ref)

    # -------------------------------------------------
    # Phase 5: Final continuous refinement from best binary
    # -------------------------------------------------
    if best_h is not None and time_left() > 0.5:
        h_final_ref, sc_final_ref = gradient_descent_momentum_with_time(
            best_h, time_left, step0=0.3, beta=0.8, max_iter=150000)
        consider(h_final_ref)

    # Ensure final projection and compute exact score
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
