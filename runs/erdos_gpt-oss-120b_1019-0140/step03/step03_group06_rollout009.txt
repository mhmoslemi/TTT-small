def enforce_sum(v, target, rng):

    total = v.sum()
    if abs(total - target) < 1e-12:

    # Scale to target

    # Clip again
    v = np.clip(v, 0.0, 1.0)

    n = h.shape[0]
    # decide number of transfer moves
    moves = max(1, int(mutation_rate * n))
    for _ in range(moves):
        # pick source, dest
        src_candidates = np.where(h > 1e-12)[0]
        dst_candidates = np.where(h < 1.0 - 1e-12)[0]

def hill_climb(h, target, rng, max_iter, cur_val):
    h = h.copy()
    val = cur_val
    n = h.shape[0]
    for _ in range(max_iter):
        # random transfer
        src = np.where(h > 1e-12)[0]
        dst = np.where(h < 1.0 - 1e-12)[0]
        if len(src)==0 or len(dst)==0: break
        i = rng.choice(src); j = rng.choice(dst)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0: continue
        delta = rng.random() * max_transfer
        h[i] -= delta; h[j] += delta
        new_val = compute_c5(h)
        if new_val < val:
            val = new_val
            # keep change
        else:
            # revert
            h[i] += delta; h[j] -= delta
        # optionally break if val < threshold
        if val < 0.3808:
            break
    return h, val

def binary_mutate(v, rng, swap_rate=0.2):
    n = v.shape[0]
    # Determine number of swaps
    swaps = max(1, int(swap_rate * n))
    for _ in range(swaps):
        ones = np.where(v > 0.5)[0]  # where v is 1
        zeros = np.where(v < 0.5)[0] # where v is 0
        if len(ones)==0 or len(zeros)==0:

        j = rng.choice(zeros)
        v[i] = 0.0
        v[j] = 1.0
    return v

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust to reach exact sum."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    """Uniform random vector projected onto the sum constraint."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    """Binary vector with exactly target ones."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, size=ones, replace=False)] = 1.0
    return v

def tournament_select(pop, scores, size, rng):
    """Select one individual via tournament."""
    idx = rng.choice(len(pop), size=size, replace=False)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]

def blend_crossover(p1, p2, rng):
    """Linear blend of two parents."""
    alpha = rng.random()
    child = alpha * p1 + (1.0 - alpha) * p2
    return child

def mutate_transfer(h, target, rng, rate=0.2):
    """Transfer mass between two indices while preserving sum."""
    n = h.shape[0]
    num = max(1, int(rate * n))
    h_new = h.copy()
    for _ in range(num):
        src = np.where(h_new > 1e-12)[0]
        dst = np.where(h_new < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_transfer = min(h_new[i], 1.0 - h_new[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h_new[i] -= delta
        h_new[j] += delta
    return enforce_sum(h_new, target, rng)

def guided_mutation(h, target, rng):
    """Targeted transfer to reduce the worst overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr) - (n - 1)
    if k >= 0:
        i_range = np.arange(n - k)
        prod = h[i_range] * (1.0 - h[i_range + k])
        i_local = np.argmax(prod)
        i = i_range[i_local]
        j = i + k
    else:
        kk = -k
        i_range = np.arange(kk, n)
        prod = h[i_range] * (1.0 - h[i_range - kk])
        i_local = np.argmax(prod)
        i = i_range[i_local]
        j = i - kk
    if 0 <= j < n:
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer > 0:
            delta = rng.random() * max_transfer
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            return enforce_sum(h_new, target, rng)
    # fallback to generic transfer mutation
    return mutate_transfer(h, target, rng, rate=0.2)

def sum_eq(x, target):
    return np.sum(x[:-1]) - target

def overlap_constraint(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 80
        target = n / 2.0
        h0 = random_continuous_vector(n, target, rng)

    # Population initialization
    pop_size = 40
    population = [h0]
    while len(population) < pop_size:
        if rng.random() < 0.5:
            h = random_binary_vector(n, target, rng)
        else:
            h = random_continuous_vector(n, target, rng)
        population.append(h)

    # Initial scoring
    scores = [compute_c5(h) for h in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_val = scores[best_idx]

    generation = 0
    max_gen = 5000
    while time.time() < deadline - 2.0 and generation < max_gen:
        # Elitism: keep current best
        new_pop = [best_h.copy()]
        # Produce offspring
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, scores, size=3, rng=rng)
            p2 = tournament_select(population, scores, size=3, rng=rng)
            child = blend_crossover(p1, p2, rng)
            # Apply guided mutation with higher probability
            if rng.random() < 0.7:
                child = guided_mutation(child, target, rng)
            else:
                child = mutate_transfer(child, target, rng, rate=0.1)
            new_pop.append(child)
        population = new_pop
        scores = [compute_c5(h) for h in population]
        cur_best_idx = int(np.argmin(scores))
        cur_best_val = scores[cur_best_idx]
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_h = population[cur_best_idx].copy()
            if best_val < 0.3808:
                break
        # Periodic local refinement on the elite
        if generation % 12 == 0:
            # simple hill climb using guided moves
            steps = 200
            if time.time() + steps * 0.0005 > deadline:
                steps = max(1, int((deadline - time.time()) * 2000))
            for _ in range(steps):
                cand = guided_mutation(best_h, target, rng)
                cand_val = compute_c5(cand)
                if cand_val < best_val:
                    best_h, best_val = cand, cand_val
                    if best_val < 0.3808:
                        break
        generation += 1

    # Final SLSQP refinement if time permits
    remaining = deadline - time.time()
    if remaining > 30.0:
        dx = 2.0 / n
        t0 = best_val
        x0 = np.concatenate([best_h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        cons = [{"type": "eq", "fun": sum_eq, "args": (target,)}]
        for k in range(-n + 1, n):
            if k == 0:
                continue
            cons.append({"type": "ineq", "fun": overlap_constraint, "args": (k, n, dx)})
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 500, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = enforce_sum(h_opt, target, rng)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best_h, best_val = h_opt, val_opt
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
