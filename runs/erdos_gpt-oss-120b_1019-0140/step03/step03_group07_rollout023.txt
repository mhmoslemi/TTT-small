    start = time.time()
    rng = np.random.default_rng(seed)

    # define n candidates
    candidates = [80, 96, 112, 128, 160]
    best_h, best_c5, best_n = None, np.inf, None

    # if initial_h_values present, prioritize that n
    if 'initial_h_values' in globals():

                candidates = [n0] + candidates
            else:
                # move to front
                candidates = [n0] + [n for n in candidates if n != n0]

    # allocate time per n
    total_budget = budget_s
    for idx, n in enumerate(candidates):
        elapsed = time.time() - start

    if abs(cur - M) > 1e-9:

        h += (M - cur)/n

        # re-check sum

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5_direct(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_fft(h):
    """Fast C5 using FFT (linear correlation)."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 2 * n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n)]))
    corr = np.fft.irfft(H * np.conj(G), n=size)[:2 * n - 1]
    return np.max(corr) * dx

def random_binary_h(n, rng):
    """Balanced 0/1 vector of length n."""
    M = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h

def enforce_sum_and_bounds(h, M):
    """Clip to [0,1] and adjust to exact sum M."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        return h
    # uniform adjustment
    h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idxs = np.argsort(h)  # smallest first
        for i in idxs:
            inc = min(diff, 1.0 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idxs = np.argsort(-h)  # largest first
        for i in idxs:
            dec = min(diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h

# ---------- constraints for SLSQP ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def objective_t(x):
    return x[-1]

# ---------- local refinement ----------
def refine_with_slsqp(h, time_budget, rng):
    n = h.shape[0]
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5_direct(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(fun=objective_t,
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h, compute_c5_direct(h)
    if not res.success:
        return h, compute_c5_direct(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5_direct(h_opt)

# ---------- simulated annealing ----------
def optimise_sa_for_n(n, time_budget, rng, init_h=None):
    M = n // 2
    if init_h is not None and len(init_h) == n:
        h_cur = enforce_sum_and_bounds(np.array(init_h, dtype=float), M)
    else:
        h_cur = random_binary_h(n, rng)
    c5_cur = compute_c5_fft(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    start = time.time()
    T0 = 0.05
    T_end = 1e-5
    idx = np.arange(n)

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.2:
            break
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + T_end * frac
        r = rng.random()
        # ----- move selection -----
        if r < 0.01:

        elif r < 0.16:

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    candidates = [80, 96, 112, 128, 160]
    init_h_global = globals().get('initial_h_values')
    if init_h_global is not None:
        n0 = len(init_h_global)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates = [n0] + candidates
            else:
                candidates = [n0] + [n for n in candidates if n != n0]

    best_h = None
    best_c5 = np.inf
    best_n = None
    total_budget = budget_s

    for idx, n in enumerate(candidates):
        elapsed = time.time() - start
        if elapsed >= total_budget - 5:
            break
        remaining = total_budget - elapsed
        # weight decreasing with index
        weights = [1.0 / (i + 1) for i in range(idx, len(candidates))]
        weight_sum = sum(weights)
        alloc = remaining * (weights[0] / weight_sum)
        alloc = max(alloc, 30.0)  # minimum per candidate
        init_h = init_h_global if (init_h_global is not None and len(init_h_global) == n) else None
        h_opt, c5_opt = optimise_sa_for_n(n, alloc, rng, init_h)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5_direct(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        n = best_n
        M = n // 2
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    # final exact C5 using the reference evaluator if available
    evaluator = globals().get('evaluate_erdos_solution')
    if evaluator is not None:
        best_c5 = evaluator(best_h)
    else:
        best_c5 = compute_c5_direct(best_h)

    return best_h, best_c5, n
