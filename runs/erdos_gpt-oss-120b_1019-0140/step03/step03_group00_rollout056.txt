            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        idxes = np.where(corr >= max_val - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h0, time_left, rng, max_iters=5000):
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        ones = np.flatnonzero(best_h > 0.5)
        zeros = np.flatnonzero(best_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i1] = 0.0
        h_new[i0] = 1.0
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_h, best_score = h_new, sc_new
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    for idx, n in enumerate(scales):
        if time_left() <= 0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if idx == 0:
            max_iters = int(min(20000, max(5000, time_left() * 5)))
        else:
            max_iters = int(min(40000, max(8000, time_left() * 10)))
        max_iters = max(2000, max_iters)

        step0 = 0.5 if n <= 128 else (0.2 if n <= 256 else 0.1)
        h_opt, sc_opt = gradient_descent_momentum_custom(
            base, n_iter=max_iters, step0=step0, beta=0.95,
            rng=rng, time_left=time_left)

        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

        # binary rounding
        sorted_idx = np.argsort(-h_opt)
        h_bin = np.zeros(n, dtype=float)
        h_bin[sorted_idx[:n // 2]] = 1.0
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()

        # binary local search
        if time_left() > 5:
            h_bin_ref, sc_bin_ref = binary_hill_climb(
                h_bin, time_left, rng,
                max_iters=int(min(5000, time_left() * 15)))
            if sc_bin_ref < best_score:
                best_score, best_h = sc_bin_ref, h_bin_ref.copy()

        base = best_h.copy() if best_h.size == n else h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)

    # final random binary hill climbs
    while time_left() > 2.0:
        n = best_h.shape[0]
        h_rand = random_binary_balanced(n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        if time_left() > 1.0:
            h_rand_ref, sc_rand_ref = binary_hill_climb(
                h_rand, time_left, rng,
                max_iters=int(min(3000, time_left() * 10)))
            if sc_rand_ref < best_score:
                best_score, best_h = sc_rand_ref, h_rand_ref.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
