def compute_c5(h):
    n = h.size
    N = 2*n - 1
    # Choose next power of 2 >= N for efficient FFT (maybe use np.fft.rfft with nfft=N)
    nfft = 1 << ((N-1).bit_length())

    G = np.fft.rfft(np.pad(1 - h, (0, nfft - n)))
    corr = np.fft.irfft(H * np.conj(G), n=nfft)[:N]
    return np.max(corr) * (2.0 / n)

    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    """Full cross-correlation of h with (1-h) using FFT."""
    n = h.size
    N = 2 * n - 1
    nfft = 1 << ((N - 1).bit_length())
    pad_h = np.pad(h, (0, nfft - n))
    pad_g = np.pad(1.0 - h, (0, nfft - n))
    H = np.fft.rfft(pad_h)
    G = np.fft.rfft(pad_g)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)[:N]
    return corr

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def compute_c5_slow(h):
    """Reference C5 using np.correlate (exact same scaling)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation (averaged over active shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            d = -s
            grad[d:] += 1.0 - h[:n - d]
            grad[:n - d] -= h[d:]
    if shifts:
        grad /= len(shifts)
    return grad

def projected_gradient_descent(h0, time_left_func, max_iter=20000,
                               step0=0.5, beta=0.9):
    """Momentum projected gradient descent on C5."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.1:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_arr(m):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(m)),
                       dtype=float, count=m)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    n = 512  # higher resolution for finer optimization

    # ------------------------------------------------------------------
    # Initial candidate pool
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n)
        candidates.append(project_onto_box_sum(init_up, n / 2.0))

    candidates.append(thue_morse_arr(n))

    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))

    for _ in range(2):
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        candidates.append(v)

    # ------------------------------------------------------------------
    # Optimize each candidate (moderate budget)
    for cand in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = projected_gradient_descent(cand, time_left,
                                                   max_iter=12000,
                                                   step0=0.6, beta=0.9)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, n

    # ------------------------------------------------------------------
    # Random restarts with short refinements
    while time_left() > 5.0:
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        h_opt, sc_opt = projected_gradient_descent(v, time_left,
                                                   max_iter=4000,
                                                   step0=0.6, beta=0.9)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    # ------------------------------------------------------------------
    # Final long refinement on the current best
    if best_h is not None and time_left() > 2.0:
        h_ref, sc_ref = projected_gradient_descent(best_h, time_left,
                                                   max_iter=25000,
                                                   step0=0.5, beta=0.9)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # ------------------------------------------------------------------
    # Safety fallback
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, n / 2.0)

    # Ensure final feasibility and compute exact bound
    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5_slow(best_h)
    return best_h, final_score, n
