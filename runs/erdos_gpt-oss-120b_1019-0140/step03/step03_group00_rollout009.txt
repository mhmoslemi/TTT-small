def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional jitter at intervals, but we may skip
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import numpy as np
import time

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = target_sum}."""
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return approximated C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float array."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones) using RNG."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    rng = np.random.default_rng(seed)

    # base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    best_h = None
    best_score = np.inf

    # low‑resolution exploration
    for n in (64, 128):
        if time_left() < 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # differential evolution
        if time_left() > 5.0:
            cand = de_optimize(base, n, rng, time_left)
            sc = compute_c5(cand)
            if sc < best_score:
                best_score, best_h = sc, cand.copy()
            base = cand

        # momentum gradient descent refinement
        if time_left() > 5.0:
            n_iter = int(min(20000, max(2000, time_left() * 1500)))
            cand, sc = gradient_descent_momentum_custom(
                base, n_iter, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
            if sc < best_score:
                best_score, best_h = sc, cand.copy()
            base = cand

    # high‑resolution refinement
    high_n = 256
    if time_left() > 5.0:
        if base.shape[0] != high_n:
            base = upsample(base, high_n)
        base = project_onto_box_sum(base, high_n / 2.0)
        n_iter = int(min(80000, max(20000, time_left() * 30)))
        cand, sc = gradient_descent_momentum_custom(
            base, n_iter, step0=0.2, beta=0.9, rng=rng, time_left=time_left)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
        base = cand

    # final random search pushes
    while time_left() > 5.0:
        h_rand = random_balanced_binary(high_n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        i, j = rng.choice(high_n, size=2, replace=False)
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max > 0.0:
            delta = delta_max * rng.uniform(0.1, 0.5)
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    # final safety projection and evaluation
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(thue_morse(high_n).astype(float), high_n / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
