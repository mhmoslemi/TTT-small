import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0., hi=1., max_iter=50):
    # return projection onto [lo,hi]^n intersect sum = s
    n = v.size
    v = np.asarray(v, dtype=float)
    # handle trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if np.abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)
```

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def local_pairwise_adjust(h, n_iter, step, rng):
    best = h.copy()
    best_score = compute_c5(best)
    n = h.size
    for _ in range(n_iter):
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-step, step)
        # propose add delta to i and subtract delta from j
        new_h = best.copy()
        new_h[i] = np.clip(new_h[i] + delta, 0, 1)
        new_h[j] = np.clip(new_h[j] - delta, 0, 1)
        # adjust sum if clipping changed sum: simply project
        new_h = project_onto_box_sum(new_h, n/2)
        sc = compute_c5(new_h)
        if sc < best_score:
            best_score, best = sc, new_h
    return best, best_score
```

import time
import numpy as np
import scipy.optimize as opt

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s} """
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    low_lam, high_lam = -1e6, 1e6
    for _ in range(max_iter):
        lam = (low_lam + high_lam) * 0.5
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            low_lam = lam
        else:
            high_lam = lam
    return np.clip(v - lam, lo, hi)


def compute_c5(h):
    """ Approximate C5 = max_k ∫ h(x)(1−h(x+k))dx on the grid. """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def resample(arr, new_n):
    """ Linear interpolation / decimation to new_n points (periodic). """
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """ Binary Thue‑Morse sequence (0/1). """
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """ Random {0,1} vector with exactly n/2 ones. """
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """ Gradient of max‑correlation term for given integer shifts. """
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]       -= h[:n - s]
            grad[:n - s]   -= h[s:]
        grad += 1.0
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """ Projected gradient descent with momentum, respecting a time budget. """
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)

        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new

    return best_h, best_score


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """ Simple simulated annealing on a balanced binary vector. """
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0  # revert

    return best_h, best_score


def local_pairwise_adjust(h0, n_iter, step, rng):
    """ Small continuous adjustments by moving mass between two coordinates. """
    best = h0.copy()
    best_score = compute_c5(best)
    n = best.size
    target = n / 2.0

    for _ in range(n_iter):
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-step, step)
        new_h = best.copy()
        new_h[i] = np.clip(new_h[i] + delta, 0.0, 1.0)
        new_h[j] = np.clip(new_h[j] - delta, 0.0, 1.0)
        new_h = project_onto_box_sum(new_h, target)
        sc = compute_c5(new_h)
        if sc < best_score:
            best_score, best = sc, new_h
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Grid sizes
    n_small = 64
    n_large = 512

    # --------------------------------------------------------------------
    # Initial candidates on the small grid
    # --------------------------------------------------------------------
    candidates_small = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        small = resample(init_full, n_small)
        small = project_onto_box_sum(small, n_small / 2.0)
        candidates_small.append(small)

    candidates_small.append(thue_morse(n_small))
    candidates_small.append(random_binary_balanced(n_small, rng))
    # a few random continuous seeds
    for _ in range(2):
        rand = rng.random(n_small)
        candidates_small.append(project_onto_box_sum(rand, n_small / 2.0))

    best_h = None
    best_score = np.inf

    # Evaluate initial small candidates
    for cand in candidates_small:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()

    # --------------------------------------------------------------------
    # Global search on the small grid using Differential Evolution
    # --------------------------------------------------------------------
    if time_left() > 30.0:
        bounds = [(0.0, 1.0)] * n_small

        def obj_small(v):
            h = project_onto_box_sum(v, n_small / 2.0)
            return compute_c5(h)

        de_opts = dict(
            bounds=bounds,
            maxiter=200,
            popsize=5,
            seed=rng,
            polish=False,
            updating='deferred',
            disp=False,
        )
        res = opt.differential_evolution(obj_small, **de_opts)
        h_de = project_onto_box_sum(res.x, n_small / 2.0)
        sc_de = compute_c5(h_de)
        if sc_de < best_score:
            best_score, best_h = sc_de, h_de.copy()

    # --------------------------------------------------------------------
    # Local refinement on the small grid (momentum GD)
    # --------------------------------------------------------------------
    if best_h is not None and time_left() > 5.0:
        gd_h, gd_sc = gradient_descent_momentum(
            best_h,
            n_iter=3000,
            step0=0.5,
            beta=0.9,
            time_left=time_left,
        )
        if gd_sc < best_score:
            best_score, best_h = gd_sc, gd_h.copy()

    # --------------------------------------------------------------------
    # Upsample to the large grid
    # --------------------------------------------------------------------
    if best_h is None:
        best_h_large = random_binary_balanced(n_large, rng)
    else:
        best_h_large = resample(best_h, n_large)
    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    sc_large = compute_c5(best_h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, best_h_large.copy()

    # --------------------------------------------------------------------
    # Large‑scale momentum gradient descent (time‑controlled)
    # --------------------------------------------------------------------
    if time_left() > 10.0:
        # empirical 1200 iterations per second for n=512
        max_iters_large = int(min(200000, max(0, time_left() * 1200)))
        gd_hL, gd_scL = gradient_descent_momentum(
            best_h,
            n_iter=max_iters_large,
            step0=0.5,
            beta=0.9,
            time_left=time_left,
        )
        if gd_scL < best_score:
            best_score, best_h = gd_scL, gd_hL.copy()

    # --------------------------------------------------------------------
    # Small continuous pairwise adjustments (fast hill climbing)
    # --------------------------------------------------------------------
    if time_left() > 5.0:
        iter_pair = int(time_left() * 250)  # ~250 proposals per second
        h_pair, sc_pair = local_pairwise_adjust(best_h, iter_pair, step=0.05, rng=rng)
        if sc_pair < best_score:
            best_score, best_h = sc_pair, h_pair.copy()

    # --------------------------------------------------------------------
    # Binary simulated annealing as a final polish
    # --------------------------------------------------------------------
    if time_left() > 5.0:
        n = best_h.size
        idx = np.argsort(best_h)[-n // 2 :]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(time_left() * 500) + 1
        h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                      temp_start=0.01,
                                      temp_end=1e-5,
                                      rng=rng)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # --------------------------------------------------------------------
    # Final projection and result
    # --------------------------------------------------------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
