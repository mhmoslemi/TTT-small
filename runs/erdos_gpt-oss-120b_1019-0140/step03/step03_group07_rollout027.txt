def targeted_move(h, k, step):
    n = len(h)
    if k >= 0:
        idx_range = np.arange(0, n - k)
        hi = h[idx_range]
        hj = h[idx_range + k]
    else:
        kk = -k
        idx_range = np.arange(kk, n)
        hi = h[idx_range]
        hj = h[idx_range - kk]
    # we want small hi high hj and large hi low hj
    # compute score = hi - hj? Actually want hi low and hj high for increase; and hi high and hj low for decrease.
    # For increase candidate:
    inc_candidates = idx_range[np.argsort(hi)][:10]  # but also need hj high: we will sort by hj descending.
    # Instead find i that maximizes (hj - hi). Use difference.
    diff = hj - hi
    i = idx_range[np.argmax(diff)]  # best potential inc
    # For decrease candidate:
    diff2 = hi - hj
    j = idx_range[np.argmax(diff2)]  # best potential dec
    # Ensure i and j are not same.
    if i == j:
        # fallback random
        i, j = np.random.choice(idx_range, 2, replace=False)
    # delta limited by step and bounds
    max_inc = min(step, 1 - h[i], h[j])
    if max_inc <= 0:
        return h
    h_new = h.copy()
    h_new[i] += max_inc
    h_new[j] -= max_inc
    return h_new

import time, numpy as np, math, scipy.optimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift
    return k, corr[idx] * dx

def generate_initials(n, M, rng):
    # yield several candidates
    yield np.full(n, M / n)  # uniform
    # linear decreasing from 1 to 0
    lin = np.linspace(1, 0, n)
    scale = M / lin.sum()
    h = np.clip(lin * scale, 0, 1)
    cur = h.sum()
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0, 1)
    yield h
    # random uniform, scaled to sum
    h = rng.random(n)
    h = h / h.sum() * M
    h = np.clip(h, 0, 1)
    cur = h.sum()
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0, 1)
    yield h
    # random binary
    idx = rng.choice(n, M, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    yield h

def targeted_move(h, k, step, rng):
    n = h.shape[0]
    if k >= 0:
        i_range = np.arange(0, n - k)
        hi = h[i_range]
        hj = h[i_range + k]
    else:
        kk = -k
        i_range = np.arange(kk, n)
        hi = h[i_range]
        hj = h[i_range - kk]
    diff = hj - hi
    if diff.size == 0:
        return h
    i_rel = np.argmax(diff)
    i = i_range[i_rel]
    # For decrease candidate, diff2 = hi - hj
    diff2 = hi - hj
    j_rel = np.argmax(diff2)
    j = i_range[j_rel]
    if i == j:
        # choose random distinct indices
        if i_range.size < 2:
            return h
        i, j = rng.choice(i_range, 2, replace=False)
    max_inc = min(step, 1.0 - h[i], h[j])
    if max_inc <= 0:
        return h
    h_new = h.copy()
    h_new[i] += max_inc
    h_new[j] -= max_inc
    return h_new

import time
import numpy as np
from scipy.optimize import minimize

# ---------- objective ----------
def objective_t(x):
    return x[-1]

# ---------- C5 computation ----------
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G)[:2 * n - 1]
    return np.max(conv) * dx

def worst_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G)[:2 * n - 1]
    idx = int(np.argmax(conv))
    k = idx - (n - 1)
    return k, conv[idx] * dx

# ---------- constraints for SLSQP ----------
class SumEQ:
    def __init__(self, n, M):
        self.n = n
        self.M = M
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.M

class OverlapIneq:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- moves ----------
def targeted_move(h, k, step, rng):
    n = h.shape[0]
    if k >= 0:
        idxs = np.arange(0, n - k)
        hi = h[idxs]
        hj = h[idxs + k]
    else:
        kk = -k
        idxs = np.arange(kk, n)
        hi = h[idxs]
        hj = h[idxs - kk]
    if idxs.size == 0:
        return h
    diff_inc = hj - hi
    i_rel = int(np.argmax(diff_inc))
    i = int(idxs[i_rel])
    diff_dec = hi - hj
    j_rel = int(np.argmax(diff_dec))
    j = int(idxs[j_rel])
    if i == j:
        if idxs.size < 2:
            return h
        i, j = rng.choice(idxs, 2, replace=False)
    max_inc = min(step, 1.0 - h[i], h[j])
    if max_inc <= 0:
        return h
    h_new = h.copy()
    h_new[i] += max_inc
    h_new[j] -= max_inc
    return h_new

def random_move(h, step, rng):
    n = h.shape[0]
    i, j = rng.choice(n, 2, replace=False)
    max_transfer = min(step, 1.0 - h[i], h[j])
    if max_transfer <= 0:
        return h
    delta = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

# ---------- initial generators ----------
def generate_initials(n, M, rng):
    # uniform
    h = np.full(n, M / n)
    yield h.copy()
    # linear decreasing from 1 to 0
    lin = np.linspace(1.0, 0.0, n)
    scale = M / np.sum(lin)
    h = np.clip(lin * scale, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    yield h
    # random uniform, scaled
    h = rng.random(n)
    h = h / np.sum(h) * M
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    yield h
    # random binary (if M integer)
    if abs(M - round(M)) < 1e-9:
        idx = rng.choice(n, int(round(M)), replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
        yield h
    # parent if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        cur = np.sum(h)
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
        yield h

# ---------- final projection ----------
def finalize_solution(h, c5, n):
    h = np.clip(h, 0.0, 1.0)
    M = n / 2.0
    cur = np.sum(h)
    if abs(cur - M) > 1e-9:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h, c5, n

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (even numbers for exact sum)
    candidates = [80, 96, 112, 128, 160]

    # prioritize parent size if present
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 in candidates:
            candidates = [n0] + [n for n in candidates if n != n0]

    for n in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        M = n / 2.0

        # ---------- obtain a quick good seed ----------
        local_best_h = None
        local_best_c5 = np.inf
        for h0 in generate_initials(n, M, rng):
            c5 = compute_c5(h0)
            if c5 < local_best_c5:
                local_best_h, local_best_c5 = h0, c5
                if c5 < best_c5:
                    best_h, best_c5, best_n = h0, c5, n
                    if best_c5 < 0.3808:
                        return finalize_solution(best_h, best_c5, best_n)

        # ---------- hill climbing with targeted moves ----------
        hill_start = time.time()
        step = 0.25  # initial transfer amount
        while time.time() - hill_start < (budget_s - (time.time() - start_time) - 1):
            k, _ = worst_shift(local_best_h)
            # try targeted improvement on worst shift
            h_new = targeted_move(local_best_h, k, step, rng)
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < local_best_c5:
                local_best_h, local_best_c5 = h_new, c5_new
                if c5_new < best_c5:
                    best_h, best_c5, best_n = h_new, c5_new, n
                    if best_c5 < 0.3808:
                        return finalize_solution(best_h, best_c5, best_n)
                step = min(step * 1.1, 0.5)
                continue
            # occasional random move
            if rng.random() < 0.3:
                h_rand = random_move(local_best_h, step, rng)
                c5_rand = compute_c5(h_rand)
                if c5_rand + 1e-12 < local_best_c5:
                    local_best_h, local_best_c5 = h_rand, c5_rand
                    if c5_rand < best_c5:
                        best_h, best_c5, best_n = h_rand, c5_rand, n
                        if best_c5 < 0.3808:
                            return finalize_solution(best_h, best_c5, best_n)
            # decay step size gradually
            step = max(step * 0.99, 0.05)

        # ---------- SLSQP refinement ----------
        try:
            x0 = np.concatenate([local_best_h, [local_best_c5]])
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            eq_con = SumEQ(n, M)
            ineq_cons = [OverlapIneq(k, n) for k in range(-n + 1, n) if k != 0]
            cons = [{'type': 'eq', 'fun': eq_con}]
            cons += [{'type': 'ineq', 'fun': ic} for ic in ineq_cons]
            res = minimize(objective_t, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                cur = np.sum(h_opt)
                if abs(cur - M) > 1e-9:
                    h_opt += (M - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt + 1e-12 < local_best_c5:
                    local_best_h, local_best_c5 = h_opt, c5_opt
                    if c5_opt < best_c5:
                        best_h, best_c5, best_n = h_opt, c5_opt, n
                        if best_c5 < 0.3808:
                            return finalize_solution(best_h, best_c5, best_n)
        except Exception:
            pass

    # ---------- fallback ----------
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    return finalize_solution(best_h, best_c5, best_n)
