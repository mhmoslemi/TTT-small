def adam_optimize(h0, n_iter, lr, alpha_start, alpha_end, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    for t in range(1, n_iter+1):
        if time_left() < 0.01: break
        # schedule alpha linearly from start to end
        alpha = alpha_start + (alpha_end - alpha_start)*(t-1)/(max(1, n_iter-1))
        grad = grad_softmax_corr(h, alpha)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        # project
        h = project_onto_box_sum(h, s)
        # compute score
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512
    s = n/2.0

    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_n = upsample(init_full, n) if init_full.size != n else init_full.copy()
        candidates.append(project_onto_box_sum(init_n, s))

    # Thue-Morse at n
    candidates.append(thue_morse(n).astype(float))

    # random continuous
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, s))

    best_h = None
    best_score = np.inf

    # Phase 1: multiple restarts of Adam gradient descent
    # allocate up to maybe 0.5 seconds per candidate? Let's just loop while time permits.

    for h0 in candidates:
        if time_left() < 1.0: break
        # compute some iterations based on remaining time
        max_iters = int(min(20000, max(2000, time_left() * 10)))  # rough
        h_opt, sc_opt = adam_optimize(h0, n_iter=max_iters, lr=0.1,
                                      alpha_start=1.0, alpha_end=20.0,
                                      time_left=time_left)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # Phase 2: final refinement using momentum gradient descent (similar to previous)
    if best_h is not None and time_left() > 2.0:
        # allocate remaining time for many iterations
        max_iters = int(min(50000, max(10000, time_left() * 20)))
        h_ref, sc_ref = momentum_grad_descent(best_h,
                                               n_iter=max_iters,
                                               step0=0.3,
                                               beta=0.9,
                                               time_left=time_left)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Final projection and compute bound
    if best_h is None:
        # fallback: uniform
        best_h = np.full(n, 0.5)
        best_h = project_onto_box_sum(best_h, s)
        best_score = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, s)
    final_score = compute_c5(best_h)

    return best_h, final_score, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Binary Thue‑Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[: n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[: n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[: n - p])
            grad[: n - p] -= w * h[p:]
    return grad


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for a set of integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += (1.0 - h[shift:])
            grad[shift:]      -= h[: n - shift]
        else:
            s = -shift
            grad[s:]         += (1.0 - h[: n - s])
            grad[: n - s]    -= h[s:]
    grad /= len(shifts)
    return grad


def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def adam_optimize(h0, n_iter, lr, alpha_start, alpha_end, time_left):
    """Adam optimizer on the soft‑max approximation of max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
    return best_h, best_score

    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    n = 512                     # grid size

    target = 0.3808

    # ----- candidate generation -----

    best_h = None

    # ----- Phase 1: multiple Adam restarts -----
    for h0 in candidates:
        if time_left() < 2.0:
