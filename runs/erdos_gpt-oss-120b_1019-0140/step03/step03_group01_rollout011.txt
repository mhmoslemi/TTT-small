    return np.interp(x_new, x_old, arr)

    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-6
    decay = 0.99997
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation to new length."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_and_score(h):
    """Linear correlation of h with (1-h) and max‑scaled score."""
    n = h.shape[0]
    g = 1.0 - h
    n2 = 1 << ((2 * n - 1).bit_length())
    a = np.empty(n2, dtype=np.float64)
    b = np.empty(n2, dtype=np.float64)
    a[:n] = h
    a[n:] = 0.0
    b[:n] = g[::-1]
    b[n:] = 0.0
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    conv = np.fft.irfft(fa * fb, n2)[:2 * n - 1]
    score = np.max(conv) * (2.0 / n)
    return conv, score

def compute_c5(h):
    """Fast C5 evaluation."""
    _, sc = compute_corr_and_score(h)
    return sc

def compute_gradients_for_shifts(h, shifts):
    """Subgradient for the max‑correlation shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty_like(h)
            term2 = np.empty_like(h)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_with_time(h0, time_left_func, step0=0.2, beta=0.9, max_iter=200000):
    """Projected momentum gradient descent with time guard."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        conv, _ = compute_corr_and_score(h)
        max_corr = np.max(conv)
        shifts = np.where(conv >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    # final check
    best_score = compute_c5(best_h)
    return best_h, best_score

def random_binary_balanced(n, rng):

    idx = rng.choice(n, size=n // 2, replace=False)

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000):
    """Swap‑based simulated annealing for binary vectors."""
    h = h0.copy()
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-6
    decay = 0.99997
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score
