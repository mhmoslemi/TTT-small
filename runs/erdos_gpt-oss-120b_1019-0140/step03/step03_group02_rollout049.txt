import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT."""
    g = 1.0 - h
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(g[::-1], size)
    conv = np.fft.irfft(H * G, size)[:2 * n - 1]
    return conv

def compute_c5(h):
    n = h.size
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:]      -= h[:n - k]
        else:
            s = -k
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    return grad

def grad_softmax_corr_fft(h, alpha):
    """Gradient of soft‑max approximation of max correlation."""
    n = h.size
    corr = compute_corr_fft(h)
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    shifts = np.arange(-(n - 1), n)
    for w, k in zip(weights, shifts):
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            s = -k
            grad[s:]      += w * (1.0 - h[:n - s])
            grad[:n - s]  -= w * h[s:]
    return grad

def smooth_grad_descent_fft(h0, n_iter=2000, step0=0.5,
                            alpha_start=1.0, alpha_end=10.0,
                            beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    corr = compute_corr_fft(h)
    best_score = np.max(corr) * (2.0 / n)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr_fft(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        corr = compute_corr_fft(h_new)
        score = np.max(corr) * (2.0 / n)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent_fft(h0, n_iter=5000, step0=0.5,
                              beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    corr = compute_corr_fft(h)
    best_score = np.max(corr) * (2.0 / n)
    mom = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = compute_corr_fft(h)
        maxc = np.max(corr)
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        corr_new = compute_corr_fft(h_new)
        score = np.max(corr_new) * (2.0 / n)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(500000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # select a zero and a one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score
