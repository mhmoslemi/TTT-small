def compute_corr(h):
    # h: float array (0 or 1)
    g = 1 - h
    # compute linear correlation using FFT
    n = h.size
    nfft = 1 << ((2*n - 1).bit_length())
    # compute convolution of h and g_rev
    g_rev = g[::-1]
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(g_rev, n=nfft)
    conv = np.fft.irfft(H * G, n=nfft)[:2*n-1]
    # correlation indices: shift from -(n-1) to (n-1)
    # conv index i corresponds to shift i - (n-1)
    return conv

def hill_climb(h, max_iters, target_score=None, time_end=None):
    n = h.size
    current_score = compute_max_overlap(h)
    best_h = h.copy()
    best_score = current_score
    # precompute positions of ones and zeros
    ones = np.flatnonzero(h > 0.5)  # ints
    zeros = np.flatnonzero(h < 0.5)
    # Use while loop.
    for it in range(max_iters):
        if time_end is not None and time.time() > time_end:
            break
        # find shift(s) with max overlap
        corr = compute_corr(h)
        max_val = int(np.max(corr) + 0.5)
        if max_val <= target_score:
            best_h = h.copy()
            best_score = max_val
            break
        # shift indices where corr == max_val
        idxs = np.where(corr >= max_val - 1e-6)[0]
        # choose a random such shift
        idx = rng.choice(idxs)
        shift = idx - (n - 1)
        # find i where h_i=1 and i+shift in range and h_{i+shift}=0
        i_candidates = np.where((h == 1) & (np.arange(n) + shift >= 0) & (np.arange(n) + shift < n) & (h[np.arange(n) + shift] == 0))[0]
        if i_candidates.size == 0:  # shouldn't but maybe because shift large => alignment trivial; fallback random swap
            # random swap
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(i_candidates)

            # compute condition for j: h_j=0 and (j+shift out of range or h_{j+shift}=1)

            if j_candidates.size == 0:
                # fallback random zero
                j = rng.choice(zeros)
            else:
                j = rng.choice(j_candidates)
        # perform swap
        h[i] = 0
        h[j] = 1
        # update ones/zeros arrays
        # update only those two entries maybe recompute after each swap.
        # compute new score
        new_score = compute_max_overlap(h)
        if new_score <= current_score:
            # accept

def compute_corr_int(h):
    n = h.shape[0]
    g = 1.0 - h
    # length of linear convolution = 2n-1
    nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(g[::-1], n=nfft)
    conv = np.fft.irfft(H * G, n=nfft)[:2*n-1]
    # round
    return np.rint(conv).astype(np.int64)

def resample_to_n(arr, n):
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_from_float(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    bin_arr = np.zeros(n, dtype=np.float64)
    bin_arr[idx] = 1.0
    return bin_arr

def hill_climb(h, max_iters, target_score, time_end):
    n = h.shape[0]
    # initial correlation
    corr = compute_corr_int(h)
    # exclude zero shift from consideration
    zero_idx = n - 1
    corr[zero_idx] = -1
    curr_score = int(np.max(corr))
    curr_shift = int(np.argmax(corr) - zero_idx)
    # best so far
    best_h = h.copy()
    best_score = curr_score
    # compute indices arrays (maybe used later)
    indices = np.arange(n)
    it = 0
    while it < max_iters and time.time() < time_end:
        it += 1
        # If current score already meets target, break
        if target_score is not None and curr_score <= target_score:
            best_h, best_score = h.copy(), curr_score
            break
        # find candidates for swap
        shift = curr_shift
        # compute i_candidates
        if shift >= 0:
            partner_idx = indices + shift
            valid = partner_idx < n
        else:
            partner_idx = indices + shift
            valid = partner_idx >= 0
        # i are positions where h=1 and partner is 0
        i_mask = (h == 1) & valid & (h[partner_idx] == 0)
        i_candidates = indices[i_mask]
        # j_candidates: positions where h=0 and (partner out of range OR partner is 1)
        j_mask = (h == 0)
        if shift >= 0:
            partner_idx_j = indices + shift
            out_range_j = partner_idx_j >= n
            partner_is_one = (partner_idx_j < n) & (h[partner_idx_j] == 1)
        else:
            partner_idx_j = indices + shift
            out_range_j = partner_idx_j < 0
            partner_is_one = (partner_idx_j >= 0) & (h[partner_idx_j] == 1)
        j_mask = j_mask & (out_range_j | partner_is_one)
        j_candidates = indices[j_mask]
        # If no candidates, fallback to random swap of any 1 and 0
        if i_candidates.size == 0 or j_candidates.size == 0:
            i = rng.choice(np.flatnonzero(h == 1))
            j = rng.choice(np.flatnonzero(h == 0))
        else:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new correlation
        new_corr = compute_corr_int(h)
        new_corr[zero_idx] = -1
        new_score = int(np.max(new_corr))
        new_shift = int(np.argmax(new_corr) - zero_idx)
        # accept if improvement or not worse
        if new_score <= curr_score:
            # accept
            curr_score = new_score
            curr_shift = new_shift
            corr = new_corr
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # keep old corr, curr_score unchanged
            # optionally perform random restart after some failures, but leave as is
            pass
    return best_h, best_score

import time, numpy as np

def compute_corr_int(h):
    n = h.shape[0]
    g = 1.0 - h
    nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(g[::-1], n=nfft)
    conv = np.fft.irfft(H * G, n=nfft)[:2*n - 1]
    return np.rint(conv).astype(np.int64)

def compute_max_overlap(h):
    corr = compute_corr_int(h)
    # exclude zero shift
    zero_idx = h.shape[0] - 1
    corr[zero_idx] = -1
    return int(np.max(corr))

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=np.float64, count=n)

def random_balanced(n, rng):
    arr = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n//2, replace=False)
    arr[idx] = 1.0
    return arr

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=np.float64)
    exp = (p - 1) // 2
    for i in range(1, p):
        if pow(i, exp, p) == 1:
            seq[i - 1] = 1.0
    return seq

def resample_to_n(arr, n):
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_from_float(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    out = np.zeros(n, dtype=np.float64)
    out[idx] = 1.0
    return out

def hill_climb(h, max_iters, target_score, time_end, rng):
    n = h.shape[0]
    zero_idx = n - 1
    # initial correlation
    corr = compute_corr_int(h)
    corr[zero_idx] = -1
    curr_score = int(np.max(corr))
    curr_shift = int(np.argmax(corr) - zero_idx)
    best_h = h.copy()
    best_score = curr_score
    # iteration loop
    for it in range(1, max_iters + 1):
        if time.time() >= time_end:
            break
        if target_score is not None and curr_score <= target_score:
            best_h, best_score = h.copy(), curr_score
            break
        shift = curr_shift
        # generate i candidates (1->0 contributions at this shift)
        if shift >= 0:
            i_candidates = np.where((h[:n - shift] == 1) & (h[shift:] == 0))[0]
            # j candidates (0 positions that won't increase O(k))
            j_range = np.where((h[:n - shift] == 0) & (h[shift:] == 1))[0]
            j_out = np.arange(n - shift, n, dtype=np.int64)
        else:
            s = -shift
            i_candidates = np.where((h[s:] == 1) & (h[:n - s] == 0))[0] + s
            j_range = np.where((h[s:] == 0) & (h[:n - s] == 1))[0] + s
            j_out = np.arange(0, s, dtype=np.int64)
        j_candidates = np.concatenate((j_range, j_out))
        if i_candidates.size == 0 or j_candidates.size == 0:
            # fallback random swap
            i = rng.choice(np.flatnonzero(h == 1))
            j = rng.choice(np.flatnonzero(h == 0))
        else:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # recompute correlation
        new_corr = compute_corr_int(h)
        new_corr[zero_idx] = -1
        new_score = int(np.max(new_corr))
        new_shift = int(np.argmax(new_corr) - zero_idx)
        # simulated annealing acceptance
        if new_score <= curr_score:
            accept = True
        else:
            # temperature schedule
            temp = max(1e-6, (max_iters - it) / max_iters) * 5.0
            prob = np.exp(-(new_score - curr_score) / temp)
            accept = rng.random() < prob
        if accept:
            curr_score = new_score
            curr_shift = new_shift
            corr = new_corr
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

import time
import numpy as np

def compute_corr_int(h):
    """Linear correlation of h and (1-h) as integer counts."""
    n = h.shape[0]
    g = 1.0 - h
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(g[::-1], n=nfft)
    conv = np.fft.irfft(H * G, n=nfft)[: 2 * n - 1]
    return np.rint(conv).astype(np.int64)


def compute_max_overlap(h):
    """Maximum of h·(1‑h) over all shifts (excluding shift 0)."""
    n = h.shape[0]
    corr = compute_corr_int(h)
    corr[n - 1] = -1  # ignore zero shift (value is always 0)
    return int(np.max(corr))


def thue_morse_binary(n):
    """Thue‑Morse binary sequence of length n (n power of two)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=np.float64, count=n)


def random_balanced(n, rng):
    """Random binary vector with exactly n/2 ones."""
    v = np.zeros(n, dtype=np.float64)
    v[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return v


def legendre_sequence(p):
    """Legendre‑symbol binary sequence of length p‑1 (p prime)."""
    n = p - 1

    for i in range(1, p):
        if pow(i, exp, p) == 1:
            seq[i - 1] = 1.0
    return seq
