            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # overlapping indices: i in [0, n‑1‑shift]
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=1.0, beta=0.9, time_end=None, max_iter=200000):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue 0/1 sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def rudin_shapiro_binary(n):
    """Rudin‑Shapiro binary sequence (0/1) of length n."""
    a = np.empty(n, dtype=int)
    a[0] = 1
    for i in range(1, n):
        if i & 1 == 0:            # even index
            a[i] = a[i >> 1]
        else:                     # odd index
            k = (i - 1) >> 1
            a[i] = a[k] if (k & 1) == 0 else -a[k]
    return (a + 1) // 2.0

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small, n_large = 128, 512
    s_small, s_large = n_small / 2.0, n_large / 2.0

    best_h = None
    best_score = np.inf

    # --- build small‑scale candidates ---------------------------------
    small_cands = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), s_small)
        small_cands.append(cand)

    small_cands.append(thue_morse_binary(n_small))
    # Legendre of length 256 down‑sampled
    leg256 = legendre_sequence(257)          # length 256
    small_cands.append(resample(leg256, n_small))
    small_cands.append(rudin_shapiro_binary(n_small))
    # a few random starts
    for _ in range(4):
        v = rng.random(n_small)
        small_cands.append(project_onto_box_sum(v, s_small))

    # --- optimise each small candidate --------------------------------
    for cand in small_cands:
        if time_left() < 8.0:
            break
        t_end = start + budget_s - 3.0
        h_opt, sc_opt = gradient_descent_momentum(
            cand, step0=1.0, beta=0.9, time_end=t_end, max_iter=250000)
        # try binarising the result
        bin_h = binarize_from_real(h_opt)
        sc_bin = compute_c5(bin_h)
        if sc_bin < sc_opt:
            h_opt, sc_opt = bin_h, sc_bin
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # fallback if nothing found
    if best_h is None:
        best_h = project_onto_box_sum(rng.random(n_small), s_small)
        best_score = compute_c5(best_h)

    # --- upsample to large resolution --------------------------------
    h_large = resample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, s_large)
    sc_large = compute_c5(h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, h_large.copy()

    # --- long gradient descent on the large candidate -----------------
    remaining = time_left()
    if remaining > 6.0:
        max_iters = int(min(800000, remaining * 1300))
        t_end = start + budget_s - 0.5
        h_ref, sc_ref = gradient_descent_momentum(
            best_h, step0=0.5, beta=0.9, time_end=t_end, max_iter=max_iters)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # --- additional random large‑scale restarts -----------------------
    while time_left() > 5.0 and best_score > 0.3808:
        v = rng.random(n_large)
        h0 = project_onto_box_sum(v, s_large)
        iters = int(min(200000, time_left() * 600))
        t_end = start + budget_s - 0.2
        h_ref, sc_ref = gradient_descent_momentum(
            h0, step0=0.5, beta=0.9, time_end=t_end, max_iter=iters)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= 0.3808:
            break

    # final safety projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
