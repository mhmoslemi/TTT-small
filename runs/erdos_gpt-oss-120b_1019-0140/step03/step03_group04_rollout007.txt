def rudin_shapiro_binary(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        # count overlapping '11' in binary of i.
        x = i & (i >> 1)
        # number of 1 bits in x (popcount)
        parity = bin(x).count('1') & 1
        seq[i] = parity  # 0 if even, 1 if odd? Standard RS uses +1 for even parity, -1 for odd.
    # Actually we want 1 for even parity (i.e., +1) and 0 for odd parity.
    # According to definition: a_n = (-1)^popcount(i & (i>>1)). So value +1 if popcount even, else -1.
    # Map +1->1, -1->0.
    seq = 1 - seq # Wait: parity=0 => even => a_n=+1 => map to 1. parity=1 => odd => -1 => map to 0.
    return seq

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    # handle boundaries
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # find lambda via bisection
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # final
    return np.clip(v - lam_mid, lo, hi)
```

import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def binary_random_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    """Thue–Morse binary sequence of length n."""
    arr = np.empty(n, dtype=float)
    for i in range(n):
        arr[i] = (bin(i).count('1') & 1)
    return arr

def rudin_shapiro_binary(n):
    """Rudin–Shapiro binary sequence of length n."""
    arr = np.empty(n, dtype=float)
    for i in range(n):
        parity = bin(i & (i >> 1)).count('1') & 1
        arr[i] = 1.0 - parity
    return arr

def hill_climb_binary(h0, rng, time_end):
    """Greedy hill climbing on binary vectors with fixed sum."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size
    while time.time() < time_end:
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def gradient_descent_continuous(h0, rng, time_end):
    """Projected gradient descent with momentum on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    it = 0
    while time.time() < time_end:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = 0.5 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def local_refine(h0, rng, time_end):
    """Small coordinated adjustments preserving the sum."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    while time.time() < time_end:
        i, j = rng.choice(n, size=2, replace=False)
        max_eps = min(1.0 - h[i], h[j], 0.02)
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h, h = score, h_new.copy(), h_new
    return best_h, best_score

        candidates.append(cand)
    candidates.append(thue_morse_binary(n_small))
    candidates.append(rudin_shapiro_binary(n_small))
    for _ in range(5):
        candidates.append(binary_random_balanced(n_small, rng))

    best_h_small = None
    best_score_small = np.inf
    for cand in candidates:
        if time_left() < 5.0:
            break
        t_end = time.time() + min(25.0, time_left())
        h_opt, sc_opt = hill_climb_binary(cand, rng, t_end)
        if sc_opt < best_score_small:
            best_score_small, best_h_small = sc_opt, h_opt.copy()
    if best_h_small is None:
        best_h_small = binary_random_balanced(n_small, rng)
        best_score_small = compute_c5(best_h_small)

    # upsample to large resolution
    x_old = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_large, endpoint=False)
    h_large = np.interp(x_new, x_old, best_h_small)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_score_large = compute_c5(h_large)
    best_h_large = h_large.copy()

    # continuous refinement
    if time_left() > 5.0:
        t_end = time.time() + min(400.0, time_left())
        h_cont, sc_cont = gradient_descent_continuous(best_h_large, rng, t_end)
        if sc_cont < best_score_large:
            best_score_large, best_h_large = sc_cont, h_cont.copy()

    # local coordinate refinement
    if time_left() > 3.0:
        t_end = time.time() + time_left()
        h_ref, sc_ref = local_refine(best_h_large, rng, t_end)
        if sc_ref < best_score_large:
            best_score_large, best_h_large = sc_ref, h_ref.copy()

    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    best_score_large = compute_c5(best_h_large)
    return best_h_large, best_score_large, n_large
