```
def compute_c5(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    nfft = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A*B, nfft)
    conv = conv[:2*n-1]
    return np.max(conv) * (2.0 / n)
```

```python
def smooth_grad_descent(h0, n_iter=2000, step0=0.5, alpha_start=1.0, alpha_end=10.0, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left()<0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start)*(it-1)/(max(1,n_iter-1))
        grad = grad_softmax_corr(h, alpha)
        mom = beta*mom + (1-beta)*grad
        eta = step0/np.sqrt(it)
        h_new = h - eta*mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(500000, max(1, time_left()*2000)))  # more attempts
    T0 = 0.01  # higher initial temperature for larger n
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # pick random zero and one
        zero_idx = rng.choice(np.where(h == 0.0)[0])
        one_idx = rng.choice(np.where(h == 1.0)[0])
        h_new = h.copy()
        h_new[zero_idx] = 1.0
        h_new[one_idx] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = (new_score < curr_score) or (rng.random() < np.exp(-delta / max(T, 1e-12)))
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = (1.0 - h)[::-1].astype(np.float64)
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    conv = np.fft.irfft(A * B, nfft)
    return conv[: 2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    corr = compute_corr(h)
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[: n - s])
            grad[: n - s] -= h[s:]
    return grad

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = compute_corr(h)
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    h = h0.copy()
    n = h.shape[0]

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.arange(old_n) / old_n
    x_new = np.arange(new_n) / new_n
    return np.interp(x_new, x_old, arr)

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(400000, max(1, time_left() * 1500)))
    T0 = 0.05
    zero_mask = (h == 0.0)
    one_mask = (h == 1.0)
    zero_idxs = np.flatnonzero(zero_mask)
    one_idxs = np.flatnonzero(one_mask)
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        if zero_idxs.size == 0 or one_idxs.size == 0:
            break
        i = rng.choice(zero_idxs)
        j = rng.choice(one_idxs)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = (new_score < curr_score) or (rng.random() < np.exp(-delta / max(T, 1e-12)))
        if accept:
            h, curr_score = h_new, new_score
            # update index sets
            zero_idxs = np.flatnonzero(h == 0.0)
            one_idxs = np.flatnonzero(h == 1.0)
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    # Base resolution
    n0 = 256
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = upsample(init_full, n0) if init_full.size != n0 else init_full.copy()
        candidates.append(project_onto_box_sum(cand, n0 / 2.0))
    candidates.append(thue_morse_binary(n0))
    for _ in range(3):

    for _ in range(2):
        v = rng.random(n0)
        candidates.append(project_onto_box_sum(v, n0 / 2.0))

    # Quick evaluation
    for h in candidates:
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()

    # Smooth gradient descent on small scale
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = smooth_grad_descent(
            h0, n_iter=2500, step0=0.6,
            alpha_start=0.5, alpha_end=8.0,
            beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Upsample to final resolution
    n_final = 512
    if time_left() > 10.0:
        h_up = upsample(best_h, n_final)
        h_up = project_onto_box_sum(h_up, n_final / 2.0)
        # Momentum descent on max-correlation gradient
        iters = int(min(30000, max(5000, time_left() * 800)))
        h_ref, sc_ref = momentum_grad_descent(
            h_up, n_iter=iters, step0=0.5, beta=0.9, time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Binary hill-climbing (simulated annealing)
    if time_left() > 5.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
        # Small continuous refinement after binary search
        if time_left() > 2.0:
            h_ref2, sc_ref2 = smooth_grad_descent(
                best_h, n_iter=1500, step0=0.3,
                alpha_start=1.0, alpha_end=6.0,
                beta=0.9, time_left=time_left)
            if sc_ref2 < best_score:
                best_score, best_h = sc_ref2, h_ref2.copy()

    # Additional random restarts if time permits
    while time_left() > 20.0:
        binary = random_binary_balanced(n_final, rng).astype(float)
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:

        if time_left() > 5.0:

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
