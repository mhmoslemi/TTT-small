def run(seed=42, budget_s=1100, **kwargs):

        if elapsed >= budget_s - 5:
            break
        rem = budget_s - elapsed

        else:
            alloc = min(0.2 * rem, 250)

            best_h, best_c5, best_n = h_opt, c5_opt, n

import time
import numpy as np

# ---------------------------------------------------------
# Helper functions
# ---------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a given discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_sequence(p):
    """Quadratic residue sequence of length p (prime). Returns 0/1 array."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for residue, p-1 otherwise
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def random_binary(n, M, rng):
    """Uniform random binary vector with exactly M ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h

# ---------------------------------------------------------
# Simulated annealing for a fixed n
# ---------------------------------------------------------
def anneal_for_n(n, time_budget, rng, init_h=None):
    """Returns best binary h (sum = n/2) found within time_budget seconds."""
    M = n // 2
    # -----------------------------------------------------------------
    # initialise current state
    # -----------------------------------------------------------------
    if init_h is not None:
        cur = np.array(init_h, dtype=float)
        # enforce binary and correct sum
        cur = np.where(cur >= 0.5, 1.0, 0.0)
        cur_sum = cur.sum()
        if cur_sum != M:
            # adjust by flipping random bits
            diff = int(M - cur_sum)
            if diff > 0:
                zero_idx = np.where(cur == 0)[0]
                flip = rng.choice(zero_idx, diff, replace=False)
                cur[flip] = 1.0
            elif diff < 0:
                one_idx = np.where(cur == 1)[0]
                flip = rng.choice(one_idx, -diff, replace=False)
                cur[flip] = 0.0
    else:
        cur = random_binary(n, M, rng)

    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5

    # annealing schedule
    T0 = max(1e-4, cur_c5 * 0.02)
    decay = 0.9995
    iter_cnt = 0
    last_improve = 0
    restart_gap = 5000  # iterations without improvement before a restart
    start = time.time()

    while time.time() - start < time_budget - 0.01:
        # temperature for this iteration
        T = T0 * (decay ** iter_cnt)
        # propose a move
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            # fallback to a fresh random state
            cur = random_binary(n, M, rng)
            cur_c5 = compute_c5(cur)
            iter_cnt += 1
            continue

        move_type = rng.integers(2)  # 0: single swap, 1: double swap
        if move_type == 0:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            new = cur.copy()
            new[i] = 0.0
            new[j] = 1.0
        else:
            if ones.size < 2 or zeros.size < 2:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                new = cur.copy()
                new[i] = 0.0
                new[j] = 1.0
            else:
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                new = cur.copy()
                new[i1] = 0.0
                new[i2] = 0.0
                new[j1] = 1.0
                new[j2] = 1.0

        new_c5 = compute_c5(new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur.copy()
                best_c5 = cur_c5
                last_improve = iter_cnt

        iter_cnt += 1
        if iter_cnt - last_improve > restart_gap:
            # restart from a fresh random state (keep global best)
            cur = random_binary(n, M, rng)
            cur_c5 = compute_c5(cur)
            last_improve = iter_cnt

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_all = time.time()
    rng = np.random.default_rng(seed)

    # candidate resolutions
    candidates = [80, 96, 112, 128]
    # prioritize any parent construction if present
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    best_h = None
    best_c5 = np.inf
    best_n = None

    for idx, n in enumerate(candidates):
        elapsed = time.time() - start_all
        if elapsed > budget_s - 5:
            break
        remaining = max(0.0, budget_s - elapsed)
        # allocate time: more for the first (typically smallest) n
        if idx == 0:
            alloc = min(remaining * 0.45, 500.0)
        else:
            alloc = min(remaining * 0.55 / (len(candidates) - idx), 300.0)

        # initialise using upscaled previous best if applicable
        init_h = None
        if best_h is not None and best_n is not None and n > best_n:
            # linear interpolation of the best known solution to a finer grid
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init_h = np.interp(x_new, x_old, best_h)
            # enforce binary and sum constraint
            init_h = np.where(init_h >= 0.5, 1.0, 0.0)
            cur_sum = init_h.sum()
            M = n // 2
            if cur_sum != M:
                diff = int(M - cur_sum)
                if diff > 0:
                    zero_idx = np.where(init_h == 0)[0]
                    flip = rng.choice(zero_idx, diff, replace=False)
                    init_h[flip] = 1.0
                elif diff < 0:
                    one_idx = np.where(init_h == 1)[0]
                    flip = rng.choice(one_idx, -diff, replace=False)
                    init_h[flip] = 0.0

        # if a parent matches this n, use it directly
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            init_h = np.array(initial_h_values, dtype=float)

        # for n=80 we also try a Legendre seed
        if n == 80 and init_h is None:
            leg = legendre_sequence(79)
            h_leg = np.concatenate([leg, [1.0]])  # length 80, sum = 40
            shift = rng.integers(n)
            h_leg = np.roll(h_leg, shift)
            init_h = h_leg

        # run annealing
        h_opt, c5_opt = anneal_for_n(n, alloc, rng, init_h)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 <= 0.3808:
                break

    # Fallback if nothing found (should not happen)
    if best_h is None:
        best_n = 80
        best_h = random_binary(best_n, best_n // 2, rng)
        best_c5 = compute_c5(best_h)

    # Final safety projection onto constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    M = best_n // 2
    cur_sum = best_h.sum()
    if abs(cur_sum - M) > 1e-9:
        best_h += (M - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
