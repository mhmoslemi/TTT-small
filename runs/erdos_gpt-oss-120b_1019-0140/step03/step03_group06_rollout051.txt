def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation length 2*n-1
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(v, target, rng):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute diff
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if cand.size == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

    ones = int(round(target))

    v[idx] = 1.0
    return v

    dx = 2.0 / n
    best_h = h.copy()
    best_val = cur_val
    # We'll use a simple temperature proportional to remaining time
    while time.time() < deadline - 0.5:
        # locate worst shift for heuristic move
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        i_local = np.argmax(prod)
        i = i_range[i_local]

        # pick a destination j with room
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0:
            break
        # ensure j != i
        cand = cand[cand != i]
        if cand.size == 0:
            break
        j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # temperature based on remaining time
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert move
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated discretely."""

def enforce_sum(v, target, rng):
    """Project v onto [0,1]^n with sum exactly target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()

        return v
