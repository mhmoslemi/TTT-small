def simulated_anneal(h, cur_val, deadline, rng):
    N = h.shape[0]
    dx = 2.0 / N
    target = N/2
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (N - 1)
        # compute the product for overlapping positions
        if k >= 0:
            prod = h[:N - k] * (1.0 - h[k:])
            i_range = np.arange(N - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[:N - kk])
            i_range = np.arange(kk, N)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # pick j where moving mass reduces mismatch
        # try random j with headroom
        candidates = np.where(h < 1.0 - 1e-12)[0]
        if len(candidates) == 0 or h[i] <= 1e-12:
            # pick random source and dst
            src_candidates = np.where(h > 1e-12)[0]
            dst_candidates = np.where(h < 1.0 - 1e-12)[0]
            if len(src_candidates) == 0 or len(dst_candidates) == 0:
                break
            i = rng.choice(src_candidates)
            j = rng.choice(dst_candidates)
        else:
            j = rng.choice(candidates)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # choose delta as fraction of max_transfer, maybe proportional to temperature
        remaining = deadline - time.time()
        T = max(remaining / (deadline - start) * 0.01, 1e-6) # temperature scaling may need start var
        # compute delta
        delta = rng.random() * max_transfer
        # Or use delta = max_transfer * min(1.0, T) maybe
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # acceptance prob
            prob = np.exp((cur_val - new_val) / max(T, 1e-12))
            if rng.random() < prob:
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def solve_milp(N, rng, time_limit=None):
    import cvxpy as cp
    target = N / 2
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target, t >= 0]
    for k in range(-N+1, N):
        if k == 0:
            continue
        L = N - abs(k)
        start_i = max(0, -k)
        i_idx = np.arange(start_i, start_i + L)
        j_idx = i_idx + k
        w = cp.Variable(L, boolean=True)
        constraints += [w <= h[i_idx],
                        w <= h[j_idx],
                        w >= h[i_idx] + h[j_idx] - 1]
        constraints.append(cp.sum(h[i_idx] - w) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve
    if time_limit is not None:
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
        except Exception:
            prob.solve(solver=cp.CBC, verbose=False)
    else:
        prob.solve(solver=cp.CBC, verbose=False)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE, cp.FEASIBLE_INACCURATE]:
        raise RuntimeError("MILP not solved")
    h_val = np.rint(h.value).astype(float)
    h_val = enforce_sum(h_val, target, rng)
    c5_val = compute_c5(h_val)
    return h_val, c5_val

def simulated_anneal(h, cur_val, deadline, rng):
    N = h.shape[0]
    dx = 2.0 / N
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    while time.time() < deadline - 0.01:
        # compute correlation

        max_idx = np.argmax(corr)
        k = max_idx - (N - 1)
        if k >= 0:
            prod = h[:N - k] * (1.0 - h[k:])
            i_range = np.arange(N - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[:N - kk])
            i_range = np.arange(kk, N)

        dst = np.where(h < 1.0 - 1e-12)[0]
        if len(dst) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]

                break

            j = rng.choice(dst)

            j = rng.choice(dst)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust continuous vector v∈[0,1] so its sum equals target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_binary_vector(n, target, rng):
    """Binary vector of length n with exactly target ones (target integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def solve_milp(N, rng, time_limit=None):
    """Mixed‑integer linear program to minimise the worst overlap for binary h."""
    import cvxpy as cp
    target = N / 2.0
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target, t >= 0]

    for k in range(-N + 1, N):
        if k == 0:
            continue
        L = N - abs(k)
        start_i = max(0, -k)
        i_idx = np.arange(start_i, start_i + L)
        j_idx = i_idx + k
        w = cp.Variable(L, boolean=True)
        constraints += [
            w <= h[i_idx],
            w <= h[j_idx],
            w >= h[i_idx] + h[j_idx] - 1,
            cp.sum(h[i_idx] - w) <= t,
        ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    solve_kwargs = {"verbose": False}
    if time_limit is not None:
        # most solvers accept 'max_seconds' or 'TimeLimit'; try both
        solve_kwargs.update({"max_seconds": time_limit})
    try:
        prob.solve(solver=cp.CBC, **solve_kwargs)
    except Exception:
        prob.solve(solver=cp.CBC, verbose=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE,
                           cp.FEASIBLE, cp.FEASIBLE_INACCURATE):
        raise RuntimeError("MILP did not converge")

    h_val = np.rint(h.value).astype(float)
    h_val = enforce_sum(h_val, target, rng)
    c5_val = compute_c5(h_val)
    return h_val, c5_val

    N = h.shape[0]
    start = time.time()
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.01:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (N - 1)

        if k >= 0:
            prod = h[: N - k] * (1.0 - h[k:])
            i_range = np.arange(N - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: N - kk])
            i_range = np.arange(kk, N)

        i = i_range[i_local]

        dst = np.where(h < 1.0 - 1e-12)[0]

            if len(src) == 0 or len(dst) == 0:

            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(dst)

        h[j] += delta
