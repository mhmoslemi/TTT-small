def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf
    best_n = None
    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates
    for idx, n in enumerate(candidates):
        elapsed = time.time() - start
        if elapsed > budget_s - 5:
            break
        remaining = budget_s - elapsed
        if idx == 0:
            # allocate 60% of remaining
            alloc = min(remaining * 0.6, 600.0)
        else:
            # allocate equally among remaining
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)
        h_opt, c5_opt = optimise_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break
    # final refinement
    if best_h is not None:
        # Run a final gradient descent for some time
        refine_time = max(0.0, budget_s - (time.time() - start) - 0.5)
        if refine_time > 0.0:
            h_ref, c5_ref = refine_h(best_h, refine_time, rng)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref
    # ensure feasibility
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        # enforce sum exactly M
        M = n // 2
        cur = best_h.sum()
        if abs(cur - M) > 1e-9:
            best_h = project_onto_box_and_sum(best_h, M)
    return best_h, best_c5, n

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_onto_box_and_sum(y, M):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > M:
            hi = lam
        else:
            lo = lam
    x = np.clip(y + lo, 0.0, 1.0)
    diff = M - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def optimise_n(n, time_budget, rng):

    best_h = None

    start = time.time()

        if len(iv) == n:
            h0 = np.array(iv, dtype=float)
            h0 = project_onto_box_and_sum(h0, M)
            c5 = compute_c5(h0)
            best_h, best_c5 = h0, c5
            parent_used = True
    while time.time() - start < time_budget:
        # generate start point
        if not parent_used and rng.random() < 0.2:
            h = rng.random(n)
            h = project_onto_box_and_sum(h, M)
