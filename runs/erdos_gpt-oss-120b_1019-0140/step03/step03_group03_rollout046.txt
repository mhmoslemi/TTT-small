def anneal_binary(cur_h, max_iters, t0, t_end, rng, time_left_func):
    n = cur_h.size
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    for it in range(max_iters):
        if time_left_func() < 0.1:
            break
        # compute temperature
        t = t0 * ((t_end / t0) ** (it / max_iters))
        # pick i,j
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update one/zero sets
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    if arr.shape[0] == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_correlation_fast(h):
    """Linear correlation of h with (1-h) via FFT, length 2n-1."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr[:2 * n - 1]


def compute_c5_fast(h):
    """Maximum overlap integral approximation on the grid."""
    n = h.shape[0]
    corr = compute_correlation_fast(h)
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑overlap term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, n_iter, step0, beta, time_left):
    """Projected momentum gradient descent on max‑overlap."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_correlation_fast(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fast(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=350):
    """SLSQP refinement on a small scale."""
    n = h0.shape[0]
    t0 = compute_c5_fast(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_correlation_fast(h) * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5_fast(h_opt)
    return h_opt, t_opt


def anneal_binary(cur_h, max_iters, t0, t_end, rng, time_left):
    """Simulated annealing on a binary balanced vector."""
    n = cur_h.size
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iters):
        if time_left() <= 0.0:
            break
        t = t0 * ((t_end / t0) ** (it / max_iters))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score

        accept = delta < 0.0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score
