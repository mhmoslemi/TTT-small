def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def contributions_shift(h, shift):
    """Number of autocorrelation contributions of a 1 at each position for given shift."""
    n = h.shape[0]
    u = np.zeros(n, dtype=int)
    if shift >= 0:
        k = shift
        u[:n - k] += h[k:]
        u[k:] += h[:n - k]
    else:
        k = -shift
        u[k:] += h[:n - k]
        u[:n - k] += h[k:]
    return u

def upsample(arr, new_n):
    """Linear up‑sampling of a 1‑D array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def greedy_improve(h, start_time, budget_s):
    """Repeatedly improve the worst shift by a single beneficial 1↔0 swap."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while True:
        if time.time() - start_time > budget_s - 0.5:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = corr.max()
        if max_corr == 0:
            break
        shift = int(np.argmax(corr) - (n - 1))
        u = contributions_shift(best_h, shift)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i_star = ones[np.argmin(u[ones])]
        j_star = zeros[np.argmax(u[zeros])]
        delta = int(u[j_star]) - int(u[i_star])
        if delta <= 0:
            break
        # perform swap
        best_h[i_star] = 0
        best_h[j_star] = 1
        cur_score = compute_c5(best_h)
        if cur_score < best_score:
            best_score = cur_score
        else:
            # revert if not an improvement
            best_h[i_star] = 1
            best_h[j_star] = 0
            break
    return best_h, best_score

def anneal(h0, start_score, start_time, budget_s, rng, target=0.3808):
    """Simulated‑annealing on binary vectors."""
    n = h0.shape[0]
    h = h0.copy()
    score = start_score
    best_h = h.copy()
    best_score = score
    T = 0.08
    T_min = 1e-6
    decay = 0.99997
    while True:
        if time.time() - start_time > budget_s - 0.5:
            break
        if best_score <= target:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        delta = new_score - score
        accept = False
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        T = max(T_min, T * decay)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 256  # grid size
    # ----- initial candidate -------------------------------------------------
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_full = upsample(init_full, n) if init_full.shape[0] != n else init_full.copy()
        idx = np.argsort(-init_full)[:n // 2]
        h0 = np.zeros(n, dtype=int)
        h0[idx] = 1
    else:
        h0 = np.zeros(n, dtype=int)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    target = 0.3808

    # ----- deterministic greedy improvement ---------------------------------
    h_g, sc_g = greedy_improve(best_h, start_time, budget_s)
    if sc_g < best_score:
        best_h, best_score = h_g, sc_g
    if best_score <= target:
        return best_h.astype(float), best_score, n

    # ----- annealing from current best --------------------------------------
    h_a, sc_a = anneal(best_h, best_score, start_time, budget_s, rng, target)
    if sc_a < best_score:
        best_h, best_score = h_a, sc_a
    if best_score <= target:
        return best_h.astype(float), best_score, n

    # ----- multiple random restarts -----------------------------------------
    while True:
        if time.time() - start_time > budget_s - 10.0:
            break
        if best_score <= target:
            break
        # new random start
        h0 = np.zeros(n, dtype=int)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1
        # quick greedy
        h_g, sc_g = greedy_improve(h0, start_time, budget_s)
        # annealing
        h_a, sc_a = anneal(h_g, sc_g, start_time, budget_s, rng, target)
        if sc_a < best_score:
            best_h, best_score = h_a, sc_a
    # final output (binary vector already satisfies the integral constraint)
    return best_h.astype(float), best_score, n
