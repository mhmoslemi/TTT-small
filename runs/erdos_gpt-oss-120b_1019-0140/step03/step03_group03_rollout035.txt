import time
import numpy as np

def compute_c5_exact(h):
    """Exact C5 using linear correlation (np.correlate)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_c5_fast(h):
    """Fast C5 using FFT (zero‑padded linear convolution)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())        # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=N)
    C = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(H * C, n=N)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def upsample(arr, new_n):
    """Linear up‑sampling from [0,1) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binary_from_continuous(v):
    """Turn a real vector into a balanced binary vector (top half ones)."""
    n = v.size
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def thue_morse(n):
    """Binary Thue‑Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def anneal_binary_h(h0, budget_s, rng):
    """Simulated annealing on a binary vector (swap moves)."""
    n = h0.size
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    T_start = 0.5
    T_end = 1e-5
    # rough iterations per second estimate ~150 for n=1024
    max_iters = int(budget_s * 150) + 1
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # temperature schedule (geometric)
        t = T_start * ((T_end / T_start) ** (it / max_iters))

        # propose a random 1↔0 swap
        i = rng.integers(n)
        while cur_h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] == 1:
            j = rng.integers(n)

        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score


def hill_climb(h0, budget_s, rng):
    """Greedy local improvement: accept any improving swap."""
    n = h0.size
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    while time_left() > 0.5:
        # random swap
        i = rng.integers(n)
        while cur_h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] == 1:
            j = rng.integers(n)

        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5_fast(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score

    n = 1024  # target resolution
