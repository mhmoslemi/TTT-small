import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Compute the discrete C5 bound for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Grid size (balance between resolution and ILP size)
    n = 64
    half_n = n // 2

    # ------------------------------------------------------------------
    # Warm‑start from parent construction if supplied
    init_h = None
    if 'initial_h_values' in globals():
        arr = np.asarray(initial_h_values, dtype=float)
        if arr.size != n:
            x_old = np.linspace(0, 1, arr.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            arr = np.interp(x_new, x_old, arr)
        # turn into a binary vector with exactly half_n ones
        order = np.argsort(arr)
        binary = np.zeros(n, dtype=float)
        binary[order[half_n:]] = 1.0
        init_h = binary

    # ------------------------------------------------------------------
    # Integer linear program: maximize the minimum overlap of ones across shifts
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    constraints = [
        cp.sum(h) == half_n,
        t >= 0,
        t <= half_n,
    ]
    if init_h is not None:
        h.value = init_h

    # For each shift k, introduce w variables that realise the product h_i * h_{i+k}
    for k in range(-(n - 1), n):
        i_start = max(0, -k)
        i_end = min(n, n - k)
        length = i_end - i_start
        if length <= 0:
            continue
        idx = np.arange(i_start, i_end, dtype=int)
        w = cp.Variable(length, boolean=True)
        constraints.append(cp.sum(w) >= t)               # enforce minimum overlap
        constraints.append(w <= h[idx])                  # w <= h_i
        constraints.append(w <= h[idx + k])              # w <= h_{i+k}
        constraints.append(w >= h[idx] + h[idx + k] - 1) # w >= h_i + h_{i+k} - 1

    prob = cp.Problem(cp.Maximize(t), constraints)

    # Solve (try CBC first, fall back to GLPK_MI)
    solved = False
    try:
        prob.solve(solver=cp.CBC, warm_start=True)
        solved = prob.status in ("optimal", "optimal_inaccurate")
    except Exception:
        solved = False
    if not solved:
        try:
            prob.solve(solver=cp.GLPK_MI, warm_start=True)
            solved = prob.status in ("optimal", "optimal_inaccurate")
        except Exception:
            solved = False

    # ------------------------------------------------------------------
    # Extract a feasible binary vector
    if solved:
        h_raw = np.ravel(h.value)
        # rounding should already give 0/1, but enforce binary and exact sum
        h_bin = (h_raw > 0.5).astype(float)
        # fix possible sum deviation by taking top half_n entries of raw values
        if int(h_bin.sum()) != half_n:
            order = np.argsort(h_raw)
            h_bin[:] = 0.0
            h_bin[order[-half_n:]] = 1.0
        h_vals = h_bin
    else:
        # fallback: random balanced binary vector
        rng = np.random.default_rng(seed)
        h_vals = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=half_n, replace=False)
        h_vals[idx] = 1.0

    # Compute the bound
    best_h = h_vals.copy()
    best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Quick local improvement if time permits and bound not yet good enough
    if time_left() > 1.0 and best_score > 0.3808:
        rng = np.random.default_rng(seed + 1)
        while time_left() > 0.2:
            # propose a swap of a zero and a one
            i0 = rng.integers(0, n)
            while best_h[i0] != 0.0:
                i0 = rng.integers(0, n)
            i1 = rng.integers(0, n)
            while best_h[i1] != 1.0:
                i1 = rng.integers(0, n)
            cand = best_h.copy()
            cand[i0] = 1.0
            cand[i1] = 0.0
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
                if best_score <= 0.3808:
                    break

    final_h = best_h
    final_score = best_score

    return final_h, final_score, n
