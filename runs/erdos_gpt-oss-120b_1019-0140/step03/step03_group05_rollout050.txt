def propose_swap(h, step, rng):
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # Determine direction: we can transfer from i to j or from j to i. For simplicity, pick direction randomly.
    direction = rng.choice([1, -1]) # 1 means move from i to j
    if direction == 1:
        max_delta = min(step, h[i], 1 - h[j])
    else:
        max_delta = min(step, h[j], 1 - h[i])
    if max_delta <= 1e-12:
        return h, 0.0, i, j, direction
    delta = rng.random() * max_delta
    new_h = h.copy()
    if direction == 1:
        new_h[i] -= delta
        new_h[j] += delta
    else:
        new_h[i] += delta
        new_h[j] -= delta
    return new_h, delta, i, j, direction

def local_refine(h, target_sum, max_iter, time_limit):
    n = len(h)
    dx = 2.0/n
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    alpha = 0.1
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # find worst shift
        corr = np.correlate(cur_h, 1-cur_h, mode='full')
        idx = np.argmax(corr)
        max_overlap = corr[idx]
        cur_c5 = max_overlap * dx
        k = idx - (n-1)
        # gradient
        grad = np.zeros_like(cur_h)
        if k >= 0:
            grad[:n-k] += 1 - cur_h[k:]
            grad[k:] -= cur_h[:n-k]
        else:
            kk = -k
            grad[kk:] += 1 - cur_h[:n-kk]
            grad[:n-kk] -= cur_h[kk:]
        # update with step size alpha, preserve constraints via projection
        new_h = cur_h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            # increase step
            alpha = min(alpha * 1.05, 2.0)
        else:
            alpha = max(alpha * 0.5, 1e-6)
    return cur_h, compute_c5(cur_h)
```

```
def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # v: numpy array, shape (n,)
    # returns projected vector.
    # Ensure lo <= hi
    # We find shift s such that sum(clip(v+s, lo, hi)) = target
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(100):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)
```

        grad = np.zeros_like(cur_h)

            grad[:n - k] += 1.0 - cur_h[k:]
            grad[k:] -= cur_h[:n - k]

            kk = -k

            cur_h = new_h
            cur_c5 = new_c5

            alpha = max(alpha * 0.5, min_alpha)

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def propose_swap(h, max_step, rng):
    """Propose a feasible swap move that preserves the total sum."""
    n = h.shape[0]
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # try moving mass from i to j
        max_delta = min(max_step, h[i], 1.0 - h[j])
        if max_delta > 0.0:
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h, delta
        # try moving mass from j to i
        max_delta = min(max_step, h[j], 1.0 - h[i])
        if max_delta > 0.0:
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            return new_h, delta
    return h, 0.0

    n = h_start.shape[0]
    cur_h = h_start.copy()

    min_alpha = 1e-6

        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = np.argmax(corr)
        max_overlap = corr[idx]
        cur_c5 = max_overlap * dx
        k = idx - (n - 1)
        grad = np.zeros_like(cur_h)
        if k >= 0:
            grad[:n - k] += 1.0 - cur_h[k:]

            kk = -k
            grad[kk:] += 1.0 - cur_h[:n - kk]
            grad[:n - kk] -= cur_h[kk:]

            cur_c5 = new_c5
            alpha = min(alpha * 1.05, 2.0)

    return cur_h, cur_c5

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

    """Run a single SLSQP optimisation from (h_start, t_start)."""
