def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # enlarge interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def full_corr_fft(h):
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return corr  # length N, contains all lags

def compute_c5(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    max_corr = full_corr_fft(h).max()
    return max_corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of the worst‑shift correlation."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func,
                                        step0=0.5, beta=0.9, max_iter=200000):
    """Projected momentum GD, stops when time runs out."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = full_corr_fft(h)
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes.copy()
        shifts[shifts > n - 1] -= corr.shape[0]   # wrap to negative lags
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    """Swap‑based simulated annealing on binary vectors."""
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    T, T_min, decay = 0.1, 1e-5, 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        delta = new - cur
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best
