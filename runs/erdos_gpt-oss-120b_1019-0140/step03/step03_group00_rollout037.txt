def compute_c5_and_shifts(h):
    n = h.shape[0]
    L = 2*n - 1
    # FFT
    fft_h = np.fft.fft(h, L)
    fft_rev = np.fft.fft(np.flip(1.0 - h), L)
    corr = np.fft.ifft(fft_h * fft_rev).real
    # Multiplying by dx factor later.
    max_corr = corr.max()
    shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (n - 1)
    c5 = max_corr * (2.0 / n)
    return corr, c5, shifts

def gradient_descent_momentum_proj(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation and shifts
        L = 2*n -1
        fft_h = np.fft.fft(h, L)
        fft_rev = np.fft.fft(np.flip(1.0 - h), L)
        corr = np.fft.ifft(fft_h * fft_rev).real
        max_corr = corr.max()
        # find shifts:
        shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (n - 1)
        # gradient
        grad = compute_gradients_for_shifts_fast(h, shifts)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # jitter:
        if it % 250 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # score
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)).real
    return np.max(corr) * (2.0 / n)


def compute_corr_and_shifts(h):
    """Return correlation array, its maximum value, and all shifts attaining it."""
    n = h.shape[0]
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)).real
    max_corr = corr.max()
    shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (n - 1)
    return corr, max_corr, shifts


def compute_gradients_for_shifts_fast(h, shifts):
    """Subgradient of max correlation w.r.t. h, averaged over given shifts."""
    n = h.shape[0]
    m = len(shifts)
    if m == 0:
        return np.zeros_like(h)
    grad = np.full_like(h, float(m))  # add 1 for each shift
    for shift in shifts:
        if shift >= 0:
            s = shift
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
        else:
            s = -shift
            grad[s:] -= h[:n - s]      # term1
            grad[:n - s] -= h[s:]      # term2
    grad /= m
    return grad


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_descent_momentum_proj(h0, n_iter, step0, beta, rng, time_left):
    """Momentum projected gradient descent on the C5 objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        _, _, shifts = compute_corr_and_shifts(h)
        grad = compute_gradients_for_shifts_fast(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 250 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    for idx, n in enumerate(scales):
        if time_left() <= 0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Step size per scale
        if n <= 128:
            step0 = 1.0
        elif n <= 256:
            step0 = 0.5
        else:
            step0 = 0.2
        beta = 0.9

        # Allocate many iterations based on remaining time
        factor = 30000.0 / n
        max_iters = int(min(200000, max(20000, time_left() * factor)))
        h_opt, sc_opt = gradient_descent_momentum_proj(
            base, max_iters, step0, beta, rng, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

        # A few short random restarts at this resolution
        for _ in range(3):
            if time_left() <= 0:
                break
            h_rand = random_binary_balanced(n, rng)
            short_iters = int(min(5000, time_left() * factor / 10))
            h_rand_opt, sc_rand_opt = gradient_descent_momentum_proj(
                h_rand, short_iters, step0, beta, rng, time_left)
            if sc_rand_opt < best_score:
                best_score, best_h = sc_rand_opt, h_rand_opt.copy()

    # Final refinement on the best found vector
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)
    else:
        if time_left() > 5:
            n_final = best_h.shape[0]
            factor_f = 15000.0 / n_final
            final_iters = int(min(80000, max(20000, time_left() * factor_f)))
            best_h, best_score = gradient_descent_momentum_proj(
                best_h, final_iters, 0.15, 0.9, rng, time_left)

    # Random binary hill climbing while time permits
    n_final = best_h.shape[0]
    while time_left() > 5:
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.5)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, n_final / 2.0)
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
