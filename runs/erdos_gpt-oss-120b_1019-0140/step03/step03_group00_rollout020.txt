import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x : lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_corr_fft(h):
    """Linear correlation of h with 1-h using FFT (full length)."""
    n = h.shape[0]
    N = 2 * n - 1
    g = 1.0 - h
    g_rev = g[::-1]
    H = np.fft.fft(h, N)
    G = np.fft.fft(g_rev, N)
    conv = np.fft.ifft(H * G).real
    return conv


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    max_corr = np.max(compute_corr_fft(h))
    return max_corr * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

                best_h = h_opt.copy()
            base = h_opt

        # Gradient descent
        if time_left() > 5:
            est_iters = int(min(20000, max(2000, time_left() * (2000.0 / n))))
            h_gd, sc_gd = gradient_descent_momentum(
                base, n_iter=est_iters, step0=0.5, beta=0.9,
                rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_score = sc_gd
                best_h = h_gd.copy()
            base = h_gd

    if best_h is None:
        best_h = base.copy()
        best_score = compute_c5(best_h)

    n_final = best_h.shape[0]
    if n_final % 2 != 0:
        target_n = n_final + 1
        best_h = upsample(best_h, target_n)
        n_final = target_n

    # Convert to balanced binary via top values
    idx_top = np.argpartition(-best_h, n_final // 2 - 1)[:n_final // 2]
    binary_h = np.zeros(n_final, dtype=float)
    binary_h[idx_top] = 1.0
    binary_score = compute_c5(binary_h)
    if binary_score < best_score:
        best_score = binary_score
        best_h = binary_h.copy()

    # Binary hill climbing stage
    while time_left() > 5:
        # Random binary start
        h_rand = random_balanced_binary(n_final, rng)
        rand_score = compute_c5(h_rand)
        if rand_score < best_score:
            best_score = rand_score
            best_h = h_rand.copy()
        max_iters = int(min(200000, max(5000, time_left() * 2000)))
        h_hc, sc_hc = binary_hill_climb(h_rand, rand_score, rng, time_left,
                                        max_iter=max_iters)
        if sc_hc < best_score:
            best_score = sc_hc
            best_h = h_hc.copy()

        # Hill climb from current best (ensure binary)
        if not np.all(np.logical_or(best_h == 0.0, best_h == 1.0)):
            idx_top = np.argpartition(-best_h, n_final // 2 - 1)[:n_final // 2]
            h_bin = np.zeros(n_final, dtype=float)
            h_bin[idx_top] = 1.0
        else:
            h_bin = best_h.copy()
        max_iters = int(min(200000, max(5000, time_left() * 2000)))
        h_hc2, sc_hc2 = binary_hill_climb(h_bin, compute_c5(h_bin),
                                          rng, time_left, max_iter=max_iters)
        if sc_hc2 < best_score:
            best_score = sc_hc2
            best_h = h_hc2.copy()

        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
