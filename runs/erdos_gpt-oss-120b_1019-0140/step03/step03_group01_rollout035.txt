def binary_anneal(h, rng, time_left, max_iters=1000000):
    n = h.size
    O = np.correlate(h, h, mode='full')[n-1:] # overlap counts for shifts 0..n-1
    # O_k = sum_i h_i * h_{i+k}
    # compute min O
    best_h = h.copy()
    best_min = O.min()
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # pick random 1 and random 0
        i = rng.choice(np.where(h == 1)[0])
        j = rng.choice(np.where(h == 0)[0])
        # compute delta O for shift k: -h_{i+k} - h_{i-k} + h_{j+k} + h_{j-k}
        # implement using roll
        delta = - (np.roll(h, -i) + np.roll(h, i)) + (np.roll(h, -j) + np.roll(h, j))
        O_new = O + delta
        new_min = O_new.min()
        # Accept if improves min (or maybe also accept if same with probability)
        if new_min > best_min:
            h[i] = 0; h[j] = 1
            O = O_new
            best_min = new_min
            best_h = h.copy()
        else:
            # maybe accept with some probability to escape local minima
            # use temperature schedule based on iteration
            # compute diff = new_min - best_min (negative)
            # accept with prob exp(diff/T)
            T = max(0.01, 0.5 * (1 - it / max_iters)) # linear cooling
            if rng.random() < np.exp((new_min - best_min) / T):
                h[i] = 0; h[j] = 1
                O = O_new
    return best_h, best_min

def mip_binary_maxmin_overlap(n, time_limit):
    import cvxpy as cp
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    # overlap variables w_ik
    # To reduce variables, we can define w as Matrix n x n (maybe use cp.Variable((n, n), boolean=True)).
    w = cp.Variable((n, n), boolean=True)

    constraints = []
    constraints.append(cp.sum(x) == n // 2)
    constraints.append(t >= 0)
    constraints.append(t <= n // 2)

    # linking constraints
    for k in range(n):
        # compute vector j indices
        # Indeed we can create a list of indices
        j = (np.arange(n) + k) % n
        # w[:,k] <= x
        constraints.append(w[:, k] <= x)
        # w[:,k] <= x[j]  (x indexed by array)
        constraints.append(w[:, k] <= x[j])
        # w[:,k] >= x + x[j] - 1
        constraints.append(w[:, k] >= x + x[j] - 1)
        # Sum >= t
        constraints.append(cp.sum(w[:, k]) >= t)

    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    # solve
    try:
        prob.solve(solver=cp.CBC, timeLimit=int(time_limit), verbose=False)
    except Exception:
        # fallback to GLPK_MI
        prob.solve(solver=cp.GLPK_MI, verbose=False)
    if x.value is None:
        return None, None
    x_bin = np.rint(x.value).astype(float)
    t_val = int(round(t.value))
    return x_bin, t_val

def binary_anneal(h, rng, time_left_func, max_iters=500000):
    n = h.size
    # compute overlap O_k for k=0..n-1 using circular correlation. Use np.correlate with mode='full' then take slice n-1:.
    corr_full = np.correlate(h, h, mode='full')
    O = corr_full[n-1:]  # length n
    best_h = h.copy()
    best_min = O.min()
    # accept with temperature schedule
    for it in range(max_iters):
        if time_left_func() <= 0:
            break
        # sample random swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # compute delta to O
        delta = -(np.roll(h, -i) + np.roll(h, i)) + (np.roll(h, -j) + np.roll(h, j))
        O_new = O + delta
        new_min = O_new.min()
        # acceptance
        if new_min > best_min:
            # accept improvement
            h[i] = 0
            h[j] = 1
            O = O_new
            best_min = new_min
            best_h = h.copy()
        else:
            # probabilistic acceptance
            # Temperature schedule: linear decreasing from T0 to Tmin
            T0 = 0.5
            Tmin = 0.01
            # compute progress proportion
            progress = it / max_iters
            T = max(Tmin, T0 * (1 - progress))
            # compute acceptance probability
            # delta can be negative; compute exp(diff/T)
            if rng.random() < np.exp((new_min - best_min) / T):
                h[i] = 0
                h[j] = 1
                O = O_new
                # best_min unchanged
    # compute final min overlap
    final_min = best_min
    return best_h, final_min

import time
import math
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_anneal(h, rng, time_left_func, max_iters=300000):
    """Simulated annealing on binary vectors to raise the minimal self‑overlap."""
    n = h.size
    # circular correlation h with itself
    O = np.correlate(h, h, mode='full')[n - 1:]  # length n, O[k] = sum_i h_i h_{i+k}
    best_h = h.copy()
    best_min = O.min()
    T0, Tmin = 0.5, 0.01
    for it in range(max_iters):
        if time_left_func() <= 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # change in O for each shift k
        delta = -(np.roll(h, -i) + np.roll(h, i)) + (np.roll(h, -j) + np.roll(h, j))
        O_new = O + delta
        new_min = O_new.min()
        if new_min > best_min:
            # strict improvement
            h[i] = 0.0
            h[j] = 1.0
            O = O_new
            best_min = new_min
            best_h = h.copy()
        else:
            # probabilistic acceptance
            progress = it / max_iters
            T = max(Tmin, T0 * (1.0 - progress))
            if rng.random() < math.exp((new_min - best_min) / T):
                h[i] = 0.0
                h[j] = 1.0
                O = O_new
    return best_h, best_min

def mip_binary_maxmin_overlap(n, time_limit=None):
    """Solve a binary max‑min overlap MIP for given n."""
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    w = cp.Variable((n, n), boolean=True)

    constraints = [
        cp.sum(x) == n // 2,
        t >= 0,
        t <= n // 2,
    ]

    rng_idx = np.arange(n)
    for k in range(n):
        j = (rng_idx + k) % n
        constraints.append(w[:, k] <= x)
        constraints.append(w[:, k] <= x[j])
        constraints.append(w[:, k] >= x + x[j] - 1)
        constraints.append(cp.sum(w[:, k]) >= t)

    prob = cp.Problem(cp.Maximize(t), constraints)
    solve_kwargs = {"verbose": False}
    if time_limit is not None:
        # CBC understands maxSeconds; GLPK_MI uses tm_lim (ms) – omitted for simplicity
        solve_kwargs["maxSeconds"] = time_limit
    try:
        prob.solve(solver=cp.CBC, **solve_kwargs)
    except Exception:
        prob.solve(solver=cp.GLPK_MI, **solve_kwargs)

    if x.value is None:
        return None, None
    h_bin = np.rint(x.value).astype(float)
    t_val = int(round(t.value))
    return h_bin, t_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # ---------- Phase 1: integer MIP (n=128) ----------
    n_mip = 128
    if time_left() > 30:
        h_bin, t_val = mip_binary_maxmin_overlap(n_mip, time_limit=min(20, time_left()))
        if h_bin is not None:
            score = compute_c5(h_bin)
            if score < best_score:
                best_score, best_h = score, h_bin.copy()
            if best_score <= target:
                return best_h, best_score, n_mip

    # ---------- Phase 2: binary annealing ----------
    # Choose a working size (256 if enough time, else 128)
    n_ann = 256 if time_left() > 60 else 128
    if best_h is None or best_h.size != n_ann:
        best_h = random_binary_balanced(n_ann, rng)
        best_score = compute_c5(best_h)

    # Multiple annealing restarts until time expires
    while time_left() > 5.0:
        # allocate a chunk of iterations proportional to remaining time
        chunk = int(min(400000, time_left() * 5000))
        h_cur, min_overlap = binary_anneal(best_h.copy(), rng, time_left, max_iters=chunk)
        score = 1.0 - (2.0 * min_overlap) / n_ann
        if score < best_score:
            best_score, best_h = score, h_cur.copy()
            if best_score <= target:
                break

    # final verification
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
