    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256
    s = n / 2.0

    # Initialize h
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            # upscale to n using upsample
            h0 = upsample(init_full, n)
        else:
            h0 = init_full.copy()
        h0 = project_onto_box_sum(h0, s)
    else:
        # random balanced continuous vector
        v = rng.random(n)
        h0 = project_onto_box_sum(v, s)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Phase A: softmax Adam
    if time_left() > 30.0:
        n_iter_a = int(min(20000, max(5000, time_left() * 15)))  # approx 15 iters per second
        # ensure at least some
        alpha_start = 0.1

            if score < best_score:

                best_h = h_new.copy()

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(300000, max(1, time_left() * 2500)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # Select random ones and zeros for swap
        zero_idx = rng.integers(0, n)
        while h[zero_idx] != 0.0:
            zero_idx = rng.integers(0, n)
        one_idx = rng.integers(0, n)
        while h[one_idx] != 1.0:
            one_idx = rng.integers(0, n)
        h_new = h.copy()
        h_new[zero_idx] = 1.0
        h_new[one_idx] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    # compute weights: exp(alpha*(corr - maxc))
    exp_vals = np.exp(alpha * (corr - maxc))
    total = np.sum(exp_vals)
    if total == 0:
        weights = np.zeros_like(exp_vals)
    else:
        weights = exp_vals / total
    grad = np.zeros_like(h)
    # iterate over shifts
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # broaden interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term (averaged over given integer shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of soft‑max approximation to max correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    total = np.sum(exp_vals)
    if total == 0:
        weights = np.zeros_like(exp_vals)
    else:
        weights = exp_vals / total
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def momentum_grad_descent(h0, n_iter, step0, beta, deadline):
    """Projected gradient descent on the max‑correlation objective using momentum."""
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    n = h.size
    for it in range(1, n_iter + 1):
        if time.time() > deadline - 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, deadline, target=0.3808):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(300000, max(1, (deadline - time.time()) * 2500)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time.time() > deadline - 0.01:
            break
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1.0 - it / max_iters)
        if new_score < curr_score or (T > 0.0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # --- problem size ---
    n = 256
    s = n / 2.0

    # --- initial solution ---
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init_full, n) if init_full.size != n else init_full.copy()
    else:
        h0 = rng.random(n)
    h0 = project_onto_box_sum(h0, s)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # ---------- Phase A: soft‑max Adam ----------
    if deadline - time.time() > 30.0:
        n_iter_a = int(min(20000, max(5000, (deadline - time.time()) * 15)))
        alpha_start, alpha_end = 0.1, 30.0
        step0 = 0.5
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        h = best_h.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        for it in range(1, n_iter_a + 1):
            if time.time() > deadline - 0.5:
                break
            alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter_a - 1)
            grad = grad_softmax_corr(h, alpha)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            eta = step0 / np.sqrt(it)
            h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_onto_box_sum(h_new, s)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
            h = h_new

    # ---------- Phase B: momentum refinement ----------
    if deadline - time.time() > 10.0:
        n_iter_b = int(min(20000, max(5000, (deadline - time.time()) * 800)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=n_iter_b,
                                              step0=0.5, beta=0.9,
                                              deadline=deadline)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # ---------- Phase C: binary hill climbing ----------
    if deadline - time.time() > 5.0:
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, deadline, target=0.3808)
        if bin_sc < best_score:
            best_score = bin_sc
            best_h = bin_h.copy()
        if deadline - time.time() > 2.0:
            h_ref2, sc_ref2 = momentum_grad_descent(best_h, n_iter=5000,
                                                   step0=0.3, beta=0.9,
                                                   deadline=deadline)
            if sc_ref2 < best_score:
                best_score = sc_ref2
                best_h = h_ref2.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
