import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # define function that projects v-lambda onto [lo,hi] and sums
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10*hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10*lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate bits of Thue-Morse sequence
    # 0 for even parity, 1 for odd
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n/2.0)

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
        h_opt = project_onto_box_sum(res.x[:n], n/2.0)
        t_opt = compute_c5(h_opt)
        return h_opt, t_opt
    except Exception:
        return h0, compute_c5(h0)

def compute_active_shifts(h, tol=1e-12):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - tol)[0]
    shifts = idxes - (n - 1)
    return shifts, max_corr

def adagrad_descent_adam_like(h0, time_left_func, max_iters=200000, init_lr=0.5):
    # Adam optimizer for subgradient of max overlap
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    while time_left_func() > 0.0 and it <= max_iters:
        shifts, max_corr = compute_active_shifts(h)
        # subgradient: average of gradients for active shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad = grad / len(shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = init_lr / np.sqrt(it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

    # Final guard projection
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

_global_n = None  # used by SLSQP constraint functions


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


downsample = upsample  # alias for convenience


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def random_continuous(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def objective_last(x):
    return x[-1]


def eq_sum_global(x):
    n = _global_n
    return np.sum(x[:n]) - n / 2.0


def ineq_fun_global(x):
    n = _global_n
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - corr


def slsqp_optimize_candidate(h0, max_iter=1500):
    global _global_n
    n = h0.size
    _global_n = n
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': eq_sum_global},
            {'type': 'ineq', 'fun': ineq_fun_global}]
    try:
        res = opt.minimize(objective_last, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter,
                                    'ftol': 1e-12,
                                    'disp': False})
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)


def compute_active_shifts(h, tol=1e-12):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - tol)[0]
    shifts = idxes - (n - 1)
    return shifts, max_corr


def adam_subgradient_descent(h0, time_left_func,
                             max_iters=200000, init_lr=0.5):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    while it <= max_iters and time_left_func() > 0.0:
        shifts, _ = compute_active_shifts(h)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        if shifts:
            grad /= len(shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = init_lr / np.sqrt(it)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # ----- Phase 1: small SLSQP passes -----
    n_small = 64

    candidates.append(thue_morse(n_small))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=1200)

        if best_score <= target:
            return best_h, best_score, best_h.size

    if best_h is None:

    # ----- Phase 2: upscale to medium resolution -----
    n_mid = 256

    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    sc_mid = compute_c5(h_mid)
    if sc_mid < best_score:

    # ----- Phase 3: Adam refinement on current size -----
    remaining = time_left()
    if remaining > 5.0:

        def time_left_ref():
            return max(0.0, t_end - time.time())

    # ----- Phase 4: attempt larger resolution -----
    if time_left() > 30.0:
        n_large = 512
        h_large = upsample(best_h, n_large)
        h_large = project_onto_box_sum(h_large, n_large / 2.0)

        if sc_large < best_score:

            t_end = time.time() + remaining * 0.5

                return max(0.0, t_end - time.time())

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
