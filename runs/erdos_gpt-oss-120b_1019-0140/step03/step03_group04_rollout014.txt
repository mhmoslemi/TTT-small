def simulated_annealing(h0, max_iters, start_time, budget_s):
    rng = np.random.default_rng()
    n = h0.size
    half = n // 2
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    # maintain lists of indices for 1s and 0s
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    # Annealing parameters
    T0 = 0.02
    Tf = 1e-5
    # compute cooling factor
    if max_iters > 0:
        alpha = (Tf / T0) ** (1.0 / max_iters)
    else:
        alpha = 1.0
    T = T0
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s - 0.2:
            break
        # proposal
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            # update lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # reject: revert swap
            h[i] = 1.0
            h[j] = 0.0
        T *= alpha
    return best_h,best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9,
                              time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, max_iters, start, budget_s, rng):

    best_h = h.copy()

    Tf = 1e-6
    if max_iters <= 0:
        return best_h, best_score
    alpha = (Tf / T0) ** (1.0 / max_iters)
    T = T0
    for it in range(max_iters):

            h[j] = 0.0
