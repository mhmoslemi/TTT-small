import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_onto_capped_simplex(v, target):
    """
    Project vector v onto the capped simplex {w | 0<=w_i<=1, sum w_i = target}.
    Solves for lambda in w_i = clip(v_i - lambda, 0, 1) via bisection.
    """
    lo = np.min(v) - 1.0  # ensures all w_i = 1 yields sum > target
    hi = np.max(v)        # ensures all w_i = 0 yields sum < target
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_descent(h_init, target, dx, deadline, rng, max_iter=2000):
    """
    Subgradient descent on the max-overlap objective with projection.
    Returns the best h and its max overlap found during the run.
    """
    n = h_init.shape[0]
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(best_h)
    step0 = 5.0 * dx  # initial step scale

    for it in range(max_iter):
        if time.time() > deadline - 0.1:
            break

        # current overlaps
        O = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = np.argmax(O)
        cur_val = O[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break

        # gradient of the worst shift
        k = max_idx - (n - 1)
        grad = np.zeros_like(h)
        if k >= 0:
            grad[: n - k] += (1.0 - h[k:]) * dx
            grad[k:] += -h[: n - k] * dx
        else:
            kk = -k
            grad[kk:] += (1.0 - h[: n - kk]) * dx
            grad[: n - kk] += -h[kk:] * dx

        # step size decays like 1/sqrt(iter)
        alpha = step0 / np.sqrt(it + 1.0)

        h_new = h - alpha * grad
        h_new = project_onto_capped_simplex(h_new, target)

        # evaluate new candidate
        O_new = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
        new_val = np.max(O_new)

        if new_val <= cur_val:
            h = h_new
        else:
            # simple annealing acceptance
            T = max(0.001 * (deadline - time.time()), 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                h = h_new
            # else keep old h

    return best_h, best_val

    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector

        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_capped_simplex(h0, target)
    else:
        n = 120
        target = n / 2.0
        h0 = rng.random(n)
        h0 = project_onto_capped_simplex(h0, target)

    dx = 2.0 / n

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # First descent from the provided start
    h_opt, val_opt = subgradient_descent(best_h, target, dx, deadline, rng, max_iter=3000)
    if val_opt < best_val:
        best_val = val_opt
        best_h = h_opt.copy()
        if best_val < 0.3808:
            return best_h.astype(float), best_val, n

    # Random-restart loop
    while time.time() < deadline - 0.5:
        # generate a fresh random feasible start
        h_start = rng.random(n)
        h_start = project_onto_capped_simplex(h_start, target)

        # run a shorter descent
        h_opt, val_opt = subgradient_descent(h_start, target, dx, deadline, rng, max_iter=600)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # Final local refinement via random mass transfers
    while time.time() < deadline - 0.2:

        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:

    # Ensure feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
