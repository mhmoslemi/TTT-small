import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (as float 0./1.)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def compute_gradient(h):
    """Subgradient of the max‑correlation objective at h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr == max_corr)[0]
    shifts = idxes - (n - 1)
    return compute_gradients_for_shifts(h, shifts)


def adam_optimize(h0, n_iter, lr, beta1, beta2, rng, time_left):
    """Adam‑style projected optimization."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        grad = compute_gradient(h)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=h.shape)
        h_new = project_onto_box_sum(h_new, h_new.size / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    """Simple momentum GD with decreasing step."""
    h = h0.copy()
    n = h.size
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        grad = compute_gradient(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = np.inf

    # multiscale schedule
    scales = [128, 256, 512, 1024]

    for n in scales:
        if time_left() <= 5:
            break

        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # determine number of restarts based on remaining time
        max_restarts = 6
        restarts = 0
        while time_left() > 6 and restarts < max_restarts:
            restarts += 1
            # choose initialization
            if rng.random() < 0.5:
                h0 = random_binary_balanced(n, rng)
            else:
                h0 = base + rng.normal(scale=0.04, size=n)
            # adam optimization
            iters = int(min(20000, max(3000, time_left() * 30)))
            lr = 0.02 if n >= 512 else 0.05
            h_opt, sc_opt = adam_optimize(
                h0, n_iter=iters, lr=lr,
                beta1=0.9, beta2=0.999, rng=rng,
                time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                base = h_opt  # seed next restarts
            if best_score <= 0.3808:
                break

        if best_h is not None:
            base = best_h

    # final fine‑tuning with momentum GD
    if best_h is not None and time_left() > 5:
        n = best_h.size
        iters = int(min(50000, max(10000, time_left() * 40)))
        h_ref, sc_ref = gradient_descent_momentum(
            best_h, n_iter=iters, step0=0.3, beta=0.9,
            rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # fallback in case nothing succeeded
    if best_h is None:
        n = scales[-1]
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    # final safety projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
