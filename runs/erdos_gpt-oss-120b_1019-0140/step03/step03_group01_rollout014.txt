import cvxpy as cp

n = 64

x = cp.Variable(n, boolean=True)

# sum constraint
constraints = [cp.sum(x) == n/2]

t = cp.Variable(integer=True)   # t >= 0, could set lower bound
constraints.append(t >= 0)

# w variables dictionary
w = {} # map (i, s) -> var boolean

for s in range(0, n):  # s >= 0
    for i in range(0, n - s):
        var = cp.Variable(boolean=True)
        w[(i, s)] = var
        # linking constraints
        constraints.append(var <= x[i])
        constraints.append(var <= x[i + s])
        constraints.append(var >= x[i] + x[i + s] - 1)

# shift constraints
for s in range(1, n):  # s > 0
    # forward shift s: y_s = sum_{i=0}^{n-1-s} (x_i - w_{i,s}) <= t
    expr_fwd = cp.sum(x[:n-s]) - cp.sum([w[(i, s)] for i in range(n-s)])
    constraints.append(expr_fwd <= t)
    # backward shift -s: y_neg = sum_{i=s}^{n-1} x_i - sum_{i=0}^{n-1-s} w_{i,s}
    expr_bwd = cp.sum(x[s:]) - cp.sum([w[(i, s)] for i in range(n-s)])
    constraints.append(expr_bwd <= t)

# Note: s = 0 gives y_0 = 0 automatically; skip

# Solve
prob = cp.Problem(cp.Minimize(t), constraints)
prob.solve(solver=cp.GLPK_MI) # or cp.CBC

        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
        h_imp, sc_imp = improve_binary(h, rng, time_left, max_iters=2000)
        if sc_imp < best_score:
            best_h, best_score = h_imp, sc_imp
            if best_score <= target:
                return best_h, best_score, n

    # ---------- Phase 3: continuous gradient descent ----------
    if best_h is None or best_score > target:
        h0 = rng.random(n)
        h0 = project_onto_box_sum(h0, n / 2.0)
        h_ref, sc_ref = gradient_descent_momentum(h0, time_left, max_iter=40000)
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref

    # final projection and return
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
        return best_h, best_score, best_h.size

    # fallback uniform half‑value
    h_def = np.full(n, 0.5)
    h_def = project_onto_box_sum(h_def, n / 2.0)
    return h_def, compute_c5(h_def), n
