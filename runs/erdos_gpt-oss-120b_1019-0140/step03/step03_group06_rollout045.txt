def smooth_max_and_weights(overlaps, alpha):
    scaled = alpha * overlaps
    max_scaled = np.max(scaled)
    shifted = np.exp(scaled - max_scaled)
    sum_shifted = np.sum(shifted)
    w = shifted / sum_shifted
    smax = (np.log(sum_shifted) + max_scaled) / alpha
    return smax, w

opt_params = {}

def compute_c5(h):
    """Maximum overlap integral (discrete)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Project v onto [0,1] with exact sum=target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
