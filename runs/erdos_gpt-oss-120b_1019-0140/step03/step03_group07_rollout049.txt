def anneal_binary(h0, M, time_budget):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    start = time.time()
    # precompute indexes of ones and zeros for quick random selection
    while time.time() - start < time_budget:
        # maybe update ones/zeros lists each iteration
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = compute_c5(cur_h)
        # acceptance

        progress = (time.time() - start) / time_budget
        T = T0 * (1 - progress)  # linear cooling
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):

                    binary_h[one_idxs[order[: -diff]]] = 0.0
            h_start = binary_h
        else:
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, M, replace=False)
            h_start[idx] = 1.0

        cur_c5 = compute_c5(h_start)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = h_start.copy(), cur_c5, n

        # ---- simulated annealing on binary vectors ----
        T0 = cur_c5 * 0.2 if cur_c5 > 0 else 0.1
        cur_h = h_start.copy()
        cur_c5_local = cur_c5
        anneal_start = time.time()
        while time.time() - anneal_start < per_n_budget:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_c5 = compute_c5(cur_h)

            elapsed = time.time() - anneal_start
            progress = elapsed / per_n_budget
            T = T0 * (1.0 - progress)

            accept = False
            if new_c5 < cur_c5_local - 1e-12:
                accept = True
            elif T > 1e-12:
                prob = np.exp(-(new_c5 - cur_c5_local) / T)
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_c5_local = new_c5
                if new_c5 < best_c5 - 1e-12:
                    best_h, best_c5, best_n = cur_h.copy(), new_c5, n
                    if best_c5 < target:
                        break
            else:
                cur_h[i] = 1.0
                cur_h[j] = 0.0

        if best_c5 < target:
            break

    # ---- optional continuous refinement with SLSQP ----
    if best_h is not None and time.time() - start < budget_s - 2:
        n = best_n
        M = n // 2
        dx = 2.0 / n

        def sum_con(x):
            return np.sum(x[:n]) - M

        shifts = [k for k in range(-n + 1, n) if k != 0]
        overlap_constraints = []
        for k in shifts:
            def make_con(k):
                def con(x):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                    return x[-1] - overl * dx
                return con
            overlap_constraints.append({'type': 'ineq', 'fun': make_con(k)})

        constraints = [{'type': 'eq', 'fun': sum_con}] + overlap_constraints
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_c5]])

        try:
            res = minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                cur_sum = np.sum(h_opt)
                if abs(cur_sum - M) > 1e-9:
                    h_opt += (M - cur_sum) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass

    # ---- final safety and cleanup ----
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur_sum = np.sum(best_h)
        M = n // 2
        if abs(cur_sum - M) > 1e-9:
            best_h += (M - cur_sum) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
