import time, numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    # generate parity of bits for each index
    arr = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return arr

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def compute_full_corr(h):
    # return linear correlation array (full) using FFT for speed
    n = h.shape[0]
    u = 1.0 - h
    # reverse h
    h_rev = h[::-1]
    L = 2 * n
    H = np.fft.fft(h_rev, n=L)
    U = np.fft.fft(u, n=L)
    conv = np.fft.ifft(H * U).real
    corr = conv[:2*n-1]
    return corr

def hill_climb_binary_swap(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill climb on a balanced binary vector."""
    h = h0.copy()
    n = h.size
    u = 1.0 - h
    corr = compute_full_corr(h)
    cur_score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    # schedule: linear cooling, start temperature
    max_iters = int(min(500_000, max(1, time_left() * 800)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick random zero and one
        i0 = rng.integers(0, n)
        while h[i0] != 0.0:
            i0 = rng.integers(0, n)
        i1 = rng.integers(0, n)
        while h[i1] != 1.0:
            i1 = rng.integers(0, n)
        # compute new correlation via copy and slice updates
        new_corr = corr.copy()
        # contributions from delta_h: +u at i0, -u at i1
        start0 = n - 1 - i0
        new_corr[start0:start0 + n] += u
        start1 = n - 1 - i1
        new_corr[start1:start1 + n] -= u
        # contributions from delta_u: -h_rev at i0, +h_rev at i1
        h_rev = h[::-1]
        new_corr[i0:i0 + n] -= h_rev
        new_corr[i1:i1 + n] += h_rev
        # cross terms
        idx_center = n - 1
        new_corr[idx_center] -= 2
        diff = i1 - i0
        new_corr[idx_center + diff] += 1
        new_corr[idx_center - diff] += 1
        new_score = np.max(new_corr) * (2.0 / n)
        # acceptance
        T = T0 * (1.0 - it / max_iters)
        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)
        if accept:
            # commit swap
            h[i0], h[i1] = 1.0, 0.0
            u[i0], u[i1] = 0.0, 1.0
            corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def momentum_grad_descent(h0, n_iter, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent using smooth‑max approximation (same as before)."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        # soft‑max gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        alpha = 10.0 * it / n_iter  # linearly increase to 10
        exp_vals = np.exp(alpha * (corr - maxc))
        weights = exp_vals / np.sum(exp_vals)
        grad = np.zeros_like(h)
        for sft, w in zip(range(-(n - 1), n), weights):
            if w == 0.0:
                continue
            if sft >= 0:
                grad[:n - sft] += w * (1.0 - h[sft:])
                grad[sft:]      -= w * h[:n - sft]
            else:
                p = -sft
                grad[p:]      += w * (1.0 - h[:n - p])
                grad[:n - p]  -= w * h[p:]
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256  # final resolution
    # ---- generate initial candidates ----
    candidates = []
    if 'initial_h_values' in globals():

        init_up = upsample(init_full, n)
        init_proj = project_onto_box_sum(init_up, n/2.0)
        candidates.append(init_proj)
    # Thue‑Morse
    candidates.append(thue_morse(n).astype(float))
    # Random balanced binaries
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    # Random continuous vectors projected
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n/2.0))

    best_h = None
    best_score = float('inf')
    # evaluate each candidate
    for h0 in candidates:
        sc = compute_c5(h0)
        if sc < best_score:
            best_score = sc
            best_h = h0.copy()
    # ensure binary start for hill climb
    # convert best_h to binary by taking top half entries
    n_half = n // 2
    order = np.argsort(best_h)
    binary = np.zeros(n, dtype=float)
    binary[order[n_half:]] = 1.0

    # Hill climbing on binary
    if time_left() > 5.0:
        bin_h, bin_sc = hill_climb_binary_swap(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # Final continuous refinement (momentum) if time permits
    if time_left() > 3.0:

        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                              step0=0.5, beta=0.9,
                                              time_left=time_left)
        if sc_ref < best_score:

    # final projection to satisfy exact sum
    best_h = project_onto_box_sum(best_h, n/2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def compute_full_corr(h):
    n = h.shape[0]
    u = 1.0 - h
    h_rev = h[::-1]
    L = 2 * n
    H = np.fft.fft(h_rev, n=L)
    U = np.fft.fft(u, n=L)
    conv = np.fft.ifft(H * U).real
    return conv[:2 * n - 1]

def hill_climb_binary_swap(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    u = 1.0 - h
    corr = compute_full_corr(h)
    cur_score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    max_iters = int(min(500_000, max(1, time_left() * 800)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i0 = rng.integers(0, n)
        while h[i0] != 0.0:
            i0 = rng.integers(0, n)
        i1 = rng.integers(0, n)
        while h[i1] != 1.0:
            i1 = rng.integers(0, n)
        new_corr = corr.copy()
        start0 = n - 1 - i0
        new_corr[start0:start0 + n] += u
        start1 = n - 1 - i1
        new_corr[start1:start1 + n] -= u
        h_rev = h[::-1]
        new_corr[i0:i0 + n] -= h_rev
        new_corr[i1:i1 + n] += h_rev
        idx_center = n - 1
        new_corr[idx_center] -= 2
        diff = i1 - i0
        new_corr[idx_center + diff] += 1
        new_corr[idx_center - diff] += 1
        new_score = np.max(new_corr) * (2.0 / n)
        T = T0 * (1.0 - it / max_iters)
        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)
        if accept:
            h[i0], h[i1] = 1.0, 0.0
            u[i0], u[i1] = 0.0, 1.0
            corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def momentum_grad_descent(h0, n_iter, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        alpha = 10.0 * it / n_iter
        exp_vals = np.exp(alpha * (corr - maxc))
        weights = exp_vals / np.sum(exp_vals)
        grad = np.zeros_like(h)
        for shift, w in zip(range(-(n - 1), n), weights):
            if w == 0.0:
                continue
            if shift >= 0:
                grad[:n - shift] += w * (1.0 - h[shift:])
                grad[shift:]      -= w * h[:n - shift]
            else:
                p = -shift
                grad[p:]      += w * (1.0 - h[:n - p])
                grad[:n - p]  -= w * h[p:]
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n)
        candidates.append(project_onto_box_sum(init_up, n / 2.0))
    candidates.append(thue_morse(n).astype(float))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    best_h = None
    best_score = float('inf')
    for h0 in candidates:
        sc = compute_c5(h0)
        if sc < best_score:
            best_score = sc
            best_h = h0.copy()

    # binary conversion of the current best
    half = n // 2
    order = np.argsort(best_h)
    binary = np.zeros(n, dtype=float)
    binary[order[half:]] = 1.0

    if time_left() > 5.0:
        bin_h, bin_sc = hill_climb_binary_swap(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    if time_left() > 3.0:
        iters = int(min(20000, max(5000, time_left() * 800)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                              step0=0.5, beta=0.9,
                                              time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
