    return best_h, best_c5

def slsqp_refine(h, budget, rng):
    n = h.shape[0]
    M = n // 2
    bounds = [(0.0, 1.0)] * n
    start = time.time()
    while time.time() - start < budget:
        t0 = compute_c5(h)
        x0 = np.concatenate([h, [t0]])
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - M}]
        prob = {'type': 'ineq', 'fun': lambda x: np.full(2 * n - 1, 0.0)}  # placeholder
        # use simple bounds and trust the optimizer to enforce constraints via penalty
        try:
            res = minimize(
                lambda x: x[-1],
                x0,
                method='SLSQP',
                bounds=bounds + [(0.0, None)],
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            break
        if not res.success:
            break
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < compute_c5(h):
            h = h_opt
            if c5_opt < 0.3808:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    best_h = None
    best_c5 = np.inf
    best_n = None

    for idx, n in enumerate(candidates):
        if time.time() - start_time > budget_s - 10:
            break
        remaining = budget_s - (time.time() - start_time)
        per_n_budget = min(remaining * 0.5, 300.0)

        # start with parent if matching size
        if best_h is None and 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            cur = np.sum(h0)
            M = n // 2
            if abs(cur - M) > 1e-12:
                h0 += (M - cur) / n
                h0 = np.clip(h0, 0.0, 1.0)
            best_h, best_c5 = h0, compute_c5(h0)
            best_n = n
            continue

        # LP relaxation
        try:
            h_lp = solve_lp_n(n)
            M = n // 2
            cur = np.sum(h_lp)
            if abs(cur - M) > 1e-9:
                h_lp += (M - cur) / n
                h_lp = np.clip(h_lp, 0.0, 1.0)
            c5_lp = compute_c5(h_lp)
            if c5_lp < best_c5:
                best_h, best_c5, best_n = h_lp.copy(), c5_lp, n
        except Exception:
            # fall back to random binary start
            pass

        # time for local refinement
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 5:
            break
        local_time = min(per_n_budget, time_left * 0.6)
        if best_h is not None and best_n == n:
            h_ref, c5_ref = local_refinement(best_h, local_time, rng)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref
            # final SLSQP touch if time permits
            remaining_after_local = budget_s - (time.time() - start_time)
            if remaining_after_local > 5 and best_c5 > 0.3808:
                h_slsqp = slsqp_refine(best_h, min(30.0, remaining_after_local), rng)
                c5_slsqp = compute_c5(h_slsqp)
                if c5_slsqp < best_c5:
                    best_h, best_c5 = h_slsqp, c5_slsqp
        if best_c5 < 0.3808:
            break

    # final safety fallback
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # final projection to feasibility
    n = best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    M = n // 2
    cur = np.sum(best_h)
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
