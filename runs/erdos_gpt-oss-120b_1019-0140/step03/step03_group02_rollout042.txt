    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')

def get_max_shifts(h, eps=1e-12):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    idx = np.where(corr >= maxc - eps)[0]
    return idx - (n - 1)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, idx in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if idx >= 0:
            grad[:n - idx] += w * (1.0 - h[idx:])
            grad[idx:]      -= w * h[:n - idx]
        else:
            p = -idx
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def get_max_shifts(h, eps=1e-12):
    """Shifts achieving the maximal correlation (subgradient set)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    idx = np.where(corr >= maxc - eps)[0]
    return idx - (n - 1)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max correlation w.r.t. h for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of soft‑max approximation of max correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def thue_morse(n):
    """Thue–Morse binary sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def smooth_grad_descent_custom(h0, n_iter=3000, step0=0.3,
                               alpha_start=1.0, alpha_end=30.0,
                               beta=0.9, time_left=lambda: 1e9):
    """Soft‑max based descent with momentum."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def adam_optimize(h0, max_iter, step0, time_left,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam with projection onto box and sum constraint."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for t in range(1, max_iter + 1):
        if time_left() < 0.01:
            break
        shifts = get_max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = step0 / np.sqrt(t)
        h_new = h - eta * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa_custom(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill climbing on binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    cur_sc = compute_c5(h)
    best_h, best_sc = h.copy(), cur_sc
    max_iters = int(min(500000, max(1, time_left() * 2000)))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        zeros = np.where(h == 0.0)[0]
        ones = np.where(h == 1.0)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_sc = compute_c5(h_new)
        delta = new_sc - cur_sc
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_sc < cur_sc:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            h, cur_sc = h_new, new_sc
            if new_sc < best_sc:
                best_sc, best_h = new_sc, h_new.copy()
                if best_sc <= target:
                    break
    return best_h, best_sc

    tm = thue_morse(n).astype(float)

    best_h = None
    best_score = float('inf')

    # explore each candidate
    for h0 in candidates:
        if time_left() < 8.0:

        # smooth soft‑max descent

    # final Adam polishing
