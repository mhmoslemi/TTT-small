def compute_c5(h):
    n = h.shape[0]
    hp = np.concatenate([h, np.zeros(n)])
    qp = np.concatenate([1.0 - h, np.zeros(n)])
    H = np.fft.rfft(hp)

    corr = np.fft.irfft(H * np.conj(Q), n=2*n)
    max_corr = np.max(corr[:2*n-1])  # ignore last zero
    return max_corr * (2.0 / n)

def gradient_descent_momentum(h0, n_iter, step0, beta, time_left):
    h = h0.copy()
    s = h.size / 2
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute gradient for current max shifts
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts indices achieving max
        indices = np.where(np.abs(corr - max_corr) <= 1e-12)[0]  # floating tolerance
        shifts = indices - (h.size - 1)
        # compute gradient as average across shifts
        grad = np.zeros_like(h)
        n = h.size
        for shift in shifts:
            if shift >= 0:
                term1 = np.empty(n, dtype=h.dtype)
                term2 = np.empty(n, dtype=h.dtype)
                term1[:n-shift] = h[shift:]
                term1[n-shift:] = 0
                term2[shift:] = h[:n-shift]
                term2[:shift] = 0
            else:
                sft = -shift
                term1 = np.empty(n, dtype=h.dtype)
                term2 = np.empty(n, dtype=h.dtype)
                term1[sft:] = h[:n-sft]
                term1[:sft] = 0
                term2[:n-sft] = h[sft:]
                term2[n-sft:] = 0
            grad += (1 - term1 - term2)
        if shifts.size > 0:
            grad /= shifts.size
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box sum
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

#### compute_c5_fast

def compute_c5_fast(h):
    n = h.shape[0]
    hp = np.concatenate([h, np.zeros(n)])
    qp = np.concatenate([1.0 - h, np.zeros(n)])
    H = np.fft.rfft(hp)
    Q = np.fft.rfft(qp)
    corr = np.fft.irfft(H * np.conj(Q), n=2 * n)
    max_corr = np.max(corr[:2 * n - 1])
    return max_corr * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def project_binary(arr):
    """Return a binary vector with exactly half entries 1 (top half of arr)."""
    n = arr.shape[0]
    k = n // 2
    idx = np.argpartition(-arr, k)[:k]
    binary = np.zeros_like(arr)
    binary[idx] = 1.0
    return binary


def compute_c5(h):
    """Exact C5 using np.correlate."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)


def compute_c5_fast(h):
    """Fast C5 using FFT (linear correlation)."""
    n = h.shape[0]
    hp = np.concatenate([h, np.zeros(n)])
    qp = np.concatenate([1.0 - h, np.zeros(n)])
    H = np.fft.rfft(hp)
    Q = np.fft.rfft(qp)
    corr = np.fft.irfft(H * np.conj(Q), n=2 * n)
    max_corr = np.max(corr[:2 * n - 1])
    return max_corr * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def simulated_annealing_binary_fixed_time(h0, time_limit, target, rng):
    """Binary simulated‑annealing for a fixed wall‑time."""
    start = time.time()
    n = h0.shape[0]
    h = h0.copy()
    cur_score = compute_c5_fast(h)
    best_h = h.copy()
    best_score = cur_score
    max_temp = 0.0015
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        remaining = time_limit - (now - start)
        T = max_temp * (remaining / time_limit)
        if T < 1e-12:
            T = 1e-12
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fast(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score


def hill_climb_binary(h0, max_no_improve, time_limit, rng):
    """Greedy hill‑climbing on binary vectors (swap 1↔0)."""
    start = time.time()
    h = h0.copy()
    best_score = compute_c5_fast(h)
    no_improve = 0
    while no_improve < max_no_improve and (time.time() - start) < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fast(h)
        if new_score < best_score:
            best_score = new_score
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_score


def gradient_descent_refine(h0, time_limit, rng):
    """Momentum gradient descent on max‑correlation objective (continuous)."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    step0 = 0.1
    it = 0
    while True:
        if time.time() - start >= time_limit:
            break
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
