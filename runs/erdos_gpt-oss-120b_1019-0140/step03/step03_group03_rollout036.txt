def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()

    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_continuous_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fa = np.fft.rfft(h, n=L)
    fb = np.fft.rfft((1.0 - h)[::-1], n=L)
    return np.fft.irfft(fa * fb, n=L)

def compute_c5(h):
    n = h.shape[0]
    max_corr = np.max(compute_corr_fft(h))
    return max_corr * (2.0 / n)

def compute_subgradient(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    cnt = len(shifts)
    if cnt == 0:
        return np.zeros_like(h)
    grad = np.full_like(h, float(cnt))
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= cnt
    return grad

        best_score = sc_bin
        best_h = binary_h.copy()

    # final projection and output
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
