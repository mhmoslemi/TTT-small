def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def binary_local_optimize(h, rng, t_start, t_budget):
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # compute initial O
    O_full = np.correlate(best_h, best_h, mode='full')
    center = n-1
    # iterate
    it = 0
    while True:
        # time check
        if time.time() - t_start > t_budget - 0.1:
            break
        # compute O and find min overlap
        O = np.correlate(best_h, best_h, mode='full')
        O[center] = np.inf
        min_overlap = O.min()
        # compute current C5
        # we can compute directly: c5 = 1 - (2/n)*min_overlap
        c5 = 1 - (2.0/n) * min_overlap
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            # best_h already same vector
        # find ks where O == min_overlap
        ks = np.where(O == min_overlap)[0] - center
        if ks.size == 0:
            break
        # try each k randomly for limited attempts
        improved = False
        ks_shuffled = rng.permuted(ks) if hasattr(rng, 'permuted') else ks  # rng.permuted may not exist; use rng.shuffle on copy
        # Use list conversion
        ks_list = ks.tolist()
        rng.shuffle(ks_list)
        for k in ks_list:
            # find i candidates
            if k >= 0:
                i_range = np.arange(0, n - k)
                i_zero = i_range[(best_h[i_range] == 0) & (best_h[i_range + k] == 1)]
                j_one = i_range[(best_h[i_range] == 1) & (best_h[i_range + k] == 0)]
            else:
                k_neg = -k
                i_range = np.arange(k_neg, n)
                i_zero = i_range[(best_h[i_range] == 0) & (best_h[i_range - k_neg] == 1)]
                j_one = i_range[(best_h[i_range] == 1) & (best_h[i_range - k_neg] == 0)]
            if i_zero.size and j_one.size:
                i = rng.choice(i_zero)
                j = rng.choice(j_one)
                # perform swap
                best_h[i] = 1.0
                best_h[j] = 0.0
                # after swap recompute overlap; can break out
                improved = True
                break
        if not improved:
            # no improvement possible
            break
        it += 1
        if it >= 2000:
            break

    final_c5 = compute_c5(best_h)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_sequence(p):
    """Legendre symbol sequence of length p (prime). Returns 0/1 array."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def generate_initial_legendre(n, rng):
    """Construct a binary seed using a Legendre sequence (requires n-1 prime)."""
    p = n - 1
    seq = legendre_sequence(p)
    h = np.concatenate([seq, [1.0]])  # length n, sum = n/2
    shift = rng.integers(n)
    return np.roll(h, shift)

def binary_local_optimize(h, rng, time_limit):
    """Greedy improvement of binary h to increase minimal self‑overlap."""
    n = h.shape[0]
    center = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    it = 0
    while True:
        if time.time() > time_limit - 0.01:
            break
        O = np.correlate(best_h, best_h, mode='full')
        O[center] = np.inf                     # exclude zero shift
        min_overlap = O.min()
        # C5 for binary case follows from min_overlap
        c5 = 1.0 - (2.0 / n) * min_overlap
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        ks = np.where(O == min_overlap)[0] - center
        if ks.size == 0:
            break
        rng.shuffle(ks)
        improved = False
        for k in ks:
            if k >= 0:
                i_range = np.arange(0, n - k)
                i_zero = i_range[(best_h[i_range] == 0) & (best_h[i_range + k] == 1)]
                j_one = i_range[(best_h[i_range] == 1) & (best_h[i_range + k] == 0)]
            else:
                kn = -k
                i_range = np.arange(kn, n)
                i_zero = i_range[(best_h[i_range] == 0) & (best_h[i_range - kn] == 1)]
                j_one = i_range[(best_h[i_range] == 1) & (best_h[i_range - kn] == 0)]
            if i_zero.size and j_one.size:
                i = rng.choice(i_zero)
                j = rng.choice(j_one)
                best_h[i] = 1.0
                best_h[j] = 0.0
                improved = True
                break
        if not improved:
            break
        it += 1
        if it >= 5000:
            break
    return best_h, compute_c5(best_h)
