def project_to_simplex_box(h, M):
    # Ensure 0<=h<=1 and sum==M.
    # We'll use algorithm: clip to [0,1], compute excess/deficit, and adjust accordingly with iterative water-filling.
    # At most two passes needed.

    # Clip first
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) <= 1e-12:
        return h
    if cur > M:
        # need to reduce sum
        excess = cur - M
        # sort indices by h descending
        idx = np.argsort(-h)  # descending
        for i in idx:
            dec = min(h[i], excess)
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
        # after reduction may have slight negative due to rounding, clip again
        h = np.clip(h, 0.0, 1.0)
    else:
        deficit = M - cur
        idx = np.argsort(h)  # ascending
        for i in idx:
            inc = min(1.0 - h[i], deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
        h = np.clip(h, 0.0, 1.0)
    # after adjusting, might still be off due to rounding; distribute remaining small residue
    cur = np.sum(h)
    if abs(cur - M) > 1e-9:
        # distribute small epsilon uniformly across non-boundary indices
        diff = M - cur
        # find indices that are not at bounds
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: add diff to first index within bounds
            for i in range(len(h)):
                if h[i] + diff <= 1.0 and h[i] + diff >= 0.0:
                    h[i] += diff
                    break
    return h

def project_feasible(h):
    n = h.shape[0]
    M = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) <= 1e-12:
        return h
    if cur > M:
        excess = cur - M
        idx = np.argsort(-h)  # descending
        for i in idx:
            dec = min(h[i], excess)
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    else:
        deficit = M - cur
        idx = np.argsort(h)  # ascending
        for i in idx:
            inc = min(1.0 - h[i], deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    # final tiny correction
    cur = np.sum(h)
    diff = M - cur
    if abs(diff) > 1e-12:
        # distribute among non‑boundary entries
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # adjust first element safely
            i = 0
            h[i] = max(0.0, min(1.0, h[i] + diff))
    return h

def slsqp_refine(h, n, rng, time_limit):

    # objective: minimize t
    def obj(x):
        return x[-1]
    # bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # constraints
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = opt.minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter':5000, 'ftol':1e-12, 'disp':False})
    except Exception:
        return h, t0
    if not res.success:
        # fallback to original
        return h, t0
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_feasible(h_opt)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def hill_climb(h0, n, rng, time_budget):

# ---------- utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overl = np.dot(h[: n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: n - kk])
        return x[-1] - overl * self.dx
