import time, numpy as np

Define functions:

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    // as previous

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate using binary parity
    # Use vectorized: np.arange(n) >> compute popcount parity
    seq = np.arange(n, dtype=np.int64)
    # popcount parity: number of ones mod 2
    # use builtin bit_count in Python 3.8+ for integers: int.bit_count()
    # But vectorized: can't directly; we can loop or use np.unpackbits
    # Simpler: generate via XOR fold
    # Use Python loops for small n
    out = np.empty(n, dtype=float)
    for i in range(n):
        out[i] = (bin(i).count('1') & 1)
    return out

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += -h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] += (1.0 - h[:n-s_abs])
            grad[:n-s_abs] += -h[s_abs:]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    # compute frequency for time checking
    chk = max(1, max_iter // 1000)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        # compute correlation and max shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # indices of max with tolerance
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional small random perturbation
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if it % chk == 0 and time_left() < 0.1:
            break
    return best_h, best_score

def local_random_search(h, best_score, rng, time_left):
    n = h.size
    s = n/2.0
    cur_h = h.copy()
    cur_score = best_score
    # we will run while time left > small
    while time_left() > 0.2:
        # random swap of mass
        i, j = rng.choice(n, size=2, replace=False)
        # ensure we can move from i to j
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.uniform(0.05, 0.3)
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # projection not needed as we maintain sum and box
        new_score = compute_c5(new_h)
        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
        # occasionally random reset from best solution with jitter
        if rng.random() < 0.01:
            jitter = rng.normal(scale=0.02, size=n)
            new_h2 = cur_h + jitter
            new_h2 = project_onto_box_sum(new_h2, s)
            new_score2 = compute_c5(new_h2)
            if new_score2 < cur_score:
                cur_h, cur_score = new_h2, new_score2
        if cur_score <= 0.3808:
            break
    return cur_h, cur_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    out = np.empty(n, dtype=float)
    for i in range(n):
        out[i] = (bin(i).count('1') & 1)
    return out


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of correlation for given integer shifts (averaged)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] += -h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += (1.0 - h[:n - s_abs])
            grad[:n - s_abs] += -h[s_abs:]
    if shifts:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    chk = max(1, max_iter // 1000)

    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if it % chk == 0 and time_left() < 0.1:
            break
    return best_h, best_score


def local_random_search(h, cur_score, rng, time_left):
    """Simple hill‑climbing by transferring mass between two positions."""
    n = h.size
    s = n / 2.0
    cur_h = h.copy()
    while time_left() > 0.2:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.uniform(0.05, 0.3)
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
        if rng.random() < 0.01:
            jitter = rng.normal(scale=0.02, size=n)
            cand = cur_h + jitter
            cand = project_onto_box_sum(cand, s)
            cand_score = compute_c5(cand)
            if cand_score < cur_score:
                cur_h, cur_score = cand, cand_score
        if cur_score <= 0.3808:
            break
    return cur_h, cur_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = np.inf
    scales = [64, 128, 256, 512]

    for n in scales:
        if time_left() <= 2:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # iteration budget roughly proportional to remaining time and dimension
        max_iter = int(min(200000, max(10000, time_left() * (n / 2))))
        if n <= 128:
            step0 = 1.0
        elif n <= 256:
            step0 = 0.5
        else:
            step0 = 0.3

        h_opt, sc_opt = gradient_descent(base, max_iter, step0, beta=0.9,
                                         rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

        # local random improvements
        if time_left() > 1:
            h_ls, sc_ls = local_random_search(base, best_score, rng, time_left)
            if sc_ls < best_score:
                best_score, best_h = sc_ls, h_ls.copy()
                base = h_ls

        # for the finest grid try a couple of random restarts if time permits
        if n == scales[-1] and time_left() > 20:
            for _ in range(2):
                if time_left() < 5:
                    break
                rand_start = random_binary_balanced(n, rng).astype(float)
                h_rand, sc_rand = gradient_descent(rand_start,
                                                   max_iter=int(min(80000,
                                                                    time_left() * 4)),
                                                   step0=0.3, beta=0.9,
                                                   rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_score, best_h = sc_rand, h_rand.copy()
                    base = h_rand

        if best_score <= 0.3808:
            break

    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
