def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    rng = np.random.default_rng(seed)
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Stage 1: small SLSQP
    n_small = 64
    cand_list = []
    # if initial_h_values exists
    try:
        init_full = np.asarray(initial_h_values, dtype=float)
        # downsample to n_small
        cand_list.append(project_onto_box_sum(downsample(init_full, n_small), n_small/2.0))
    except NameError:
        pass
    cand_list.append(thue_morse(n_small))
    for i in range(4):
        cand_list.append(random_binary_balanced(n_small, rng))
    for i in range(5):
        v = rng.random(n_small)
        cand_list.append(project_onto_box_sum(v, n_small/2.0))

    for cand in cand_list:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=1000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    # Stage 2: upsample to n_large=512
    n_large = 512
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small/2.0)
        best_score = compute_c5(best_h)
    # use interpolation
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large/2.0)
    sc_up = compute_c5(h_up)
    if sc_up < best_score:
        best_score, best_h = sc_up, h_up.copy()
    # early exit
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Stage 3: gradient descent momentum at large n
    if time_left() > 2.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left, step0=0.5, beta=0.9, max_iter=200000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Stage 4: binary conversion
    n = best_h.size
    k = n//2
    idx_top = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx_top] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Stage 5: binary annealing (large n)
    if time_left() > 2.0:
        # if best_h is binary already, use that, else maybe use upsampled binary from previous small solution?
        h_bin_start = best_h.copy()
        # ensure binary
        if not np.all(np.logical_or(h_bin_start == 0, h_bin_start == 1)):
            # convert
            idx_top = np.argpartition(-h_bin_start, k)[:k]
            h_bin_start[:] = 0.0
            h_bin_start[idx_top] = 1.0
        h_bin_ref, sc_bin_ref = binary_swap_anneal(h_bin_start, rng, time_left, max_iters=2000000)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Stage 6: gradient refinement from binary (continuous)
    if time_left() > 2.0:
        # start from binary but use continuous gradient descent
        h_ref2, sc_ref2 = gradient_descent_momentum_with_time(best_h, time_left, step0=0.5, beta=0.9, max_iter=200000)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Stage 7: random restarts
    while time_left() > 5.0:
        # random continuous balanced start
        v = rng.random(n_large)
        v = project_onto_box_sum(v, n_large/2.0)
        h_tmp, sc_tmp = gradient_descent_momentum_with_time(v, time_left, step0=0.5, beta=0.9, max_iter=5000)
        if sc_tmp < best_score:

        if best_score <= target:

    # final projection guard
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure interval covers the target sum
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=10_000_000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.5
    T_min = 1e-6
    decay = 0.99996
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1200):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
