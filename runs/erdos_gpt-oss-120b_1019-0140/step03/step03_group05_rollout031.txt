import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Use bisection on shift
    lo_shift = -np.max(v) - 1.0
    hi_shift = hi - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo_shift + hi_shift) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            lo_shift = mid
        else:
            hi_shift = mid
    return np.clip(v + (lo_shift + hi_shift) * 0.5, lo, hi)
```

```python
def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad
```

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    min_alpha = 1e-6
    for i in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        cur_c5 = max_overlap * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        # accept if improvement
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            # random perturb occasionally
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    # binary search for the shift that makes the sum equal target
    low = -np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Sub‑gradient of the worst‑shift overlap (without dx)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

# ---------- SLSQP helpers ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, shift, n):
        self.shift = shift
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.shift
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
