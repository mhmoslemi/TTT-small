def project_to_feasible(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

Let's derive gradient precisely.

Define C_k = dx * sum_{i=0}^{n-1} h_i * (1 - h_{i+k}) where for out-of-range indices, we treat h_{j}=0. Equivalent to sum_{i: 0<=i< n, 0<=i+k < n} h_i - h_i h_{i+k}.

Thus define S_k = sum_{i: i in [0,n-1] and i+k in [0,n-1]} (h_i - h_i h_{i+k}). This sum is over i such that i and i+k are valid indices. For k >=0, i ranges 0.. n-1-k. For k<0, i ranges -k .. n-1.

Then C_k = dx * S_k.

Compute derivative w.r.t h_j:

Consider contribution to S_k that involves h_j. There are two possible types: When i = j (the first term) and when i+k = j (the second term). For k >=0:

- For i = j (requires j <= n-1-k). Then term (h_j - h_j h_{j+k}) derivative = 1 - h_{j+k}.

- For i where i+k = j -> i = j - k, requires i >=0 => j >= k, and i <= n-1-k => j-k <= n-1-k => j <= n-1. So for j >= k, term is -(h_{j-k}) (since derivative of -h_i h_{i+k} w.r.t h_{i+k} is -h_i). So derivative = -h_{j-k}.

Thus total derivative for k>=0: dC_k/dh_j = dx * ((1 - h_{j+k}) if j <= n-1-k else 0) - dx * (h_{j-k} if j >= k else 0). So we can express as:

g_j = (dx) * (cond1 * (1 - h_{j+k}) - cond2 * h_{j-k}).

Similarly for k < 0 (k = -kk):

Indices i must satisfy i >= 0, i+(-kk) = i-kk >=0, and i-kk < n, i < n => i >= kk ? Let's derive:

For k = -kk <0, i + k = i - kk must be >=0 and <n. So i >= kk and i <= n-1. Also i must be >=0 as usual. So i in [kk, n-1] (inclusive). So S_k = sum_{i=kk}^{n-1} (h_i - h_i h_{i-kk}).

Derivative w.r.t h_j:

- When i=j (i in [kk, n-1]), contributes (1 - h_{j-kk}).

- When i-kk = j (i = j+kk) (requires i <= n-1 => j+kk <= n-1 => j <= n-1-kk). And i >= kk => j+kk >= kk => j >= 0. So j in [0, n-1-kk]. Then term -h_i = -h_{j+kk}.

Thus for k negative, derivative = dx * ((1 - h_{j-kk}) if j >= kk else 0) - dx * (h_{j+kk} if j <= n-1-kk else 0).

def gradient_for_shift(h, k, dx):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if k >= 0:
        # first term (1 - h_{j+k}) for j <= n-1-k
        if k < n:  # always true for k < n

def slsqp_refine(h_start, n, M, dx, time_limit):
    # check if time limit >0
    start = time.time()
    if time.time() - start > time_limit:
        return None, np.inf
    sum_constraint = lambda x: np.sum(x[:n]) - M
    # constraints generation
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    # overlap constraints
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_fun(k_val):
            def fun(x):
                h = x[:n]
                if k_val >= 0:
                    overl = np.dot(h[:n - k_val], 1.0 - h[k_val:])
                else:
                    kk = -k_val
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * dx
            return fun
        constraints.append({'type': 'ineq', 'fun': make_fun(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])
    try:
        res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None, np.inf
    if not res.success:
        # could also try simplify but ignore
        return None, np.inf
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # adjust sum precisely
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt = project_to_feasible(h_opt, M)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolution list and time fractions
    n_list = [200, 160, 128, 96, 80]
    fractions = [0.5, 0.25, 0.15, 0.07, 0.03]  # sum ~1
    total_budget = max(0.0, budget_s - 5.0)  # reserve 5s
    cumulative = 0.0

    for n, frac in zip(n_list, fractions):
        cumulative += frac
        end_time = start + cumulative * total_budget
        if time.time() > end_time - 0.5:
            continue
        M = n // 2
        dx = 2.0 / n

        # try Legendre seed if applicable
        if is_prime(n - 1):
            p = n - 1
            leg = legendre_sequence(p)
            h_leg = np.concatenate([leg, [1.0]])
            h_leg = np.roll(h_leg, rng.integers(n))
            c5_leg = compute_c5(h_leg)
            if c5_leg < best_c5:
                best_h, best_c5, best_n = h_leg.copy(), c5_leg, n
                if best_c5 < 0.3808:
                    break

        # main search loop
        # allocate per iteration PGD runs
        while time.time() < end_time - 0.5:
            # allocate small budget for a PGD run
            remaining = end_time - time.time()
            if remaining < 2.0:
                break
            pgd_iters = 500  # fixed
            step0 = 5.0
            # random feasible start
            h0 = rng.random(n)
            h0 = project_to_feasible(h0, M)
            h_loc, c5_loc = run_pgd(h0, n, M, dx, pgd_iters, step0)
            if c5_loc < best_c5:
                best_h, best_c5, best_n = h_loc.copy(), c5_loc, n
                if best_c5 < 0.3808:
                    break
            # occasional hill climbing swaps (to add variability)
            # perform a few binary swaps
            if rng.random() < 0.2:
                # create binary version from h_loc by thresholding top M entries
                idx_sorted = np.argsort(-h_loc)[:M]

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_for_shift(h, k, dx):
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if k >= 0:
        g[:n - k] += 1.0 - h[k:]
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += 1.0 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    return g * dx

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(np.sqrt(p)) + 1
    for i in range(3, r, 2):
        if p % i == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def run_pgd(h0, n, M, dx, max_iters, step0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        k = np.argmax(corr) - (n - 1)
        g = gradient_for_shift(h, k, dx)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = project_to_feasible(h, M)
        step *= 0.9995
    return best_h, best_c5

def slsqp_refine(h_start, n, M, dx, time_limit):
    start = time.time()
    if time.time() - start > time_limit:
        return None, np.inf
    def sum_con(x):
        return np.sum(x[:n]) - M
    constraints = [{'type': 'eq', 'fun': sum_con}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        def make_fun(k_val):
            def fun(x):
                h = x[:n]
                if k_val >= 0:
                    overl = np.dot(h[:n - k_val], 1.0 - h[k_val:])
                else:
                    kk = -k_val
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * dx
            return fun
        constraints.append({'type': 'ineq', 'fun': make_fun(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])
    try:
        res = minimize(lambda x: x[-1], x0=x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None, np.inf
    if not res.success:
        return None, np.inf
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    if abs(np.sum(h_opt) - M) > 1e-9:
        h_opt = project_to_feasible(h_opt, M)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt
