def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def simulated_annealing(h0, time_budget, rng):
    n = len(h0)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = h0.copy()
    cur_val = best_val
    start = time.time()
    it = 0
    # temperature schedule
    T0 = 0.005
    # iterate
    while time.time() - start < time_budget:
        it += 1
        # choose move
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # max transfer from j to i
        max_transfer = min(1.0 - cur_h[i], cur_h[j])
        if max_transfer <= 0:
            # try opposite direction
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            # swap i and j roles
            i, j = j, i
        # delta random
        delta = rng.random() * max_transfer
        # propose new
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # optionally if near binary, we could do flip:
        # but above already handles fractional.

        new_val = compute_c5(new_h)
        # accept if better or with probability
        delta_val = new_val - cur_val
        if delta_val < 0:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h, new_val
        else:
            # temperature schedule
            # linear cooling
            T = T0 * max(0.0, 1.0 - (time.time() - start) / time_budget)
            if T > 0 and rng.random() < np.exp(-delta_val / T):
                cur_h, cur_val = new_h, new_val
        # possibly occasional random restart to avoid stuck
        if it % 10000 == 0 and rng.random() < 0.1:
            # reset cur_h to best_h (or random)
            cur_h = best_h.copy()
            cur_val = best_val
    return best_h, best_val

def slsqp_refine(h_init, t_init, time_budget, rng):
    n = len(h_init)
    M = n//2
    sum_con = lambda x: np.sum(x[:n]) - M
    # define overlap constraints
    dx = 2.0 / n
    def make_con(k):
        def fun(x):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[:n-k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[:n-kk])
            return x[n] - overl * dx
        return fun
    constraints = [{'type':'eq', 'fun': sum_con}]
    constraints += [{'type':'ineq', 'fun': make_con(k)} for k in range(-n+1, n) if k != 0]
    bounds = [(0.0,1.0)]*n + [(0.0,None)]
    x0 = np.concatenate([h_init, [t_init]])
    try:
        res = minimize(fun=lambda x: x[n],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter':5000, 'ftol':1e-12})
    except Exception:
        return h_init, t_init
    if not res.success:
        return h_init, t_init
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # adjust sum if needed
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def local_refine(h, time_budget, rng):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        if rng.random() < 0.5:
            # binary swap move
            zeros = np.where(best_h < 0.5)[0]
            ones = np.where(best_h > 0.5)[0]
            if zeros.size == 0 or ones.size == 0:
                continue
            i = rng.choice(zeros)
            j = rng.choice(ones)
            new_h = best_h.copy()
            new_h[i] = 1.0
            new_h[j] = 0.0
        else:
            # fractional transfer move
            i, j = rng.choice(n, 2, replace=False)

            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_val = compute_c5(new_h)
        if new_val + 1e-12 < best_val:
            best_h, best_val = new_h, new_val
    return best_h, best_val

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(math.isqrt(p))
    for d in range(3, r + 1, 2):
        if p % d == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def simulated_annealing(h0, time_budget, rng):
    n = len(h0)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    T0 = 5e-3  # initial temperature
    while time.time() - start < time_budget:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # decide direction of transfer
        if rng.random() < 0.5:
            max_transfer = min(1.0 - cur_h[i], cur_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h, new_val
        else:
            elapsed = time.time() - start
            T = T0 * max(0.0, 1.0 - elapsed / time_budget)
            if T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
    return best_h, best_val

def local_refine(h0, time_budget, rng):
    n = len(h0)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_budget:
        if rng.random() < 0.5:
            zeros = np.where(best_h < 0.5)[0]
            ones = np.where(best_h > 0.5)[0]
            if zeros.size == 0 or ones.size == 0:
                continue
            i = rng.choice(zeros)
            j = rng.choice(ones)
            new_h = best_h.copy()
            new_h[i] = 1.0
            new_h[j] = 0.0
        else:
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h[i], best_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_val = compute_c5(new_h)
        if new_val + 1e-12 < best_val:
            best_h, best_val = new_h, new_val
    return best_h, best_val

def slsqp_refine(h_init, t_init, time_budget, rng):
    n = len(h_init)
    M = n // 2
    dx = 2.0 / n
    # equality constraint: sum(h) = M
    def sum_eq(x):
        return np.sum(x[:n]) - M
    # inequality constraints: t >= overlap(k) * dx
    def make_con(k):
        def con(x):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[:n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[:n - kk])
            return x[n] - overl * dx
        return con
    cons = [{'type': 'eq', 'fun': sum_eq}]
    cons += [{'type': 'ineq', 'fun': make_con(k)} for k in range(-n + 1, n) if k != 0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_init, [t_init]])
    try:
        res = minimize(fun=lambda x: x[n],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h_init, t_init
    if not res.success:
        return h_init, t_init
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt = h_opt + (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # candidate resolutions
    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    best_h = None
    best_c5 = np.inf
    best_n = None

    for idx, n in enumerate(candidates):
        elapsed = time.time() - start
        if elapsed >= budget_s - 5:
            break
        remaining = budget_s - elapsed
        if idx == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)

        M = n // 2

        # seed generation
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            p = n - 1
            if is_prime(p):
                leg = legendre_sequence(p)
                h0 = np.concatenate([leg, [1.0]])
                h0 = np.roll(h0, rng.integers(n))
                h0 = np.clip(h0, 0.0, 1.0)
            else:
                indices = rng.choice(n, M, replace=False)
                h0 = np.zeros(n)
                h0[indices] = 1.0

        cur_sum = np.sum(h0)
        if abs(cur_sum - M) > 1e-12:
            h0 = h0 + (M - cur_sum) / n
            h0 = np.clip(h0, 0.0, 1.0)
            if abs(np.sum(h0) - M) > 1e-9:
                # fallback to binary if still off
                indices = rng.choice(n, M, replace=False)
                h0 = np.zeros(n)
                h0[indices] = 1.0

        # Simulated annealing
        h_sa, c5_sa = simulated_annealing(h0, alloc, rng)

        # Local deterministic refinement
        refine_time = min(5.0, max(0.0, budget_s - (time.time() - start) - 12.0))
        h_ref, c5_ref = local_refine(h_sa, refine_time, rng)

        # SLSQP refinement (limited by remaining time)
        slsqp_time = max(0.0, budget_s - (time.time() - start) - 2.0)
        if slsqp_time > 1.0:
            h_opt, c5_opt = slsqp_refine(h_ref, c5_ref, slsqp_time, rng)

            h_opt, c5_opt = h_ref, c5_ref

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
            if best_c5 < 0.3808:

    # final safety
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h = best_h + (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
