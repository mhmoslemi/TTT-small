def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # v is array
    # find lambda
    # Clip: x = np.clip(v + lam, lo, hi)
    # sum_x = target
    # Use binary search over lam
    lam_low = -np.max(v) - 1.0  # lower bound ensures all v+lam <= lo
    lam_high = hi - np.min(v) + 1.0  # upper bound ensures all v+lam >= hi
    for _ in range(60):
        lam_mid = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam_mid, lo, hi)
        s = x.sum()
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    x = np.clip(v + (lam_low + lam_high) / 2.0, lo, hi)
    return x

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo, hi]^n while fixing the sum to target."""
    # Simple bisection on the dual variable lambda.
    lam_low = lo - np.max(v) - 1.0
    lam_high = hi - np.min(v) + 1.0
    for _ in range(60):
        lam_mid = (lam_low + lam_high) * 0.5
        x = np.clip(v + lam_mid, lo, hi)
        s = x.sum()
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    x = np.clip(v + (lam_low + lam_high) * 0.5, lo, hi)
    return x

def gradient_for_shift(h, k):
    """Gradient of the worst shift overlap w.r.t h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        cur_c5 = max_overlap * (2.0 / n)
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def random_balanced_binary(n, rng):
    """Return a float array of length n with exactly n/2 ones."""
    ones = n // 2
    zeros = n - ones
    arr = np.concatenate([np.ones(ones), np.zeros(zeros)])
    rng.shuffle(arr)
    return arr.astype(float)
