def compute_c5_fft(h):
    n = h.size
    size = 2 * n
    # FFT of h and 1-h with zero padding
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    # cross-correlation using ifft of H * conj(G)
    corr = np.fft.ifft(H * np.conj(G)).real
    # Only need linear correlation values for shifts - may align with full result.
    # The correlation for shift k (0 <= k < n) corresponds to corr[n-1 + k] (when using full mode).
    # For full correlation matrix, max over all indices corresponds to max over corr (including negative shifts).
    # We'll take max of corr.
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)  # dx = 2/n

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        max_corr, shifts = compute_c5_and_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # clip in [0,1] and adjust sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    size = 2 * n
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    corr = np.fft.ifft(H * np.conj(G)).real
    corr = corr[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_c5_and_shifts(h, eps=1e-12):
    n = h.shape[0]
    size = 2 * n
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    corr = np.fft.ifft(H * np.conj(G)).real
    corr = corr[:2 * n - 1]
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    return max_corr * (2.0 / n), shifts

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts) if shifts.size else grad

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        max_corr, shifts = compute_c5_and_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        cand = resample(init, n_small)
        candidates.append(binarize_from_real(cand))
    try:
        candidates.append(legendre_sequence(n_small + 1))
    except Exception:
        pass
    candidates.append(thue_morse_binary(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))

    for cand in candidates:
        if time_left() < 0.5:
            break
        sc = compute_c5_fft(cand)
        if sc < best_score:
            best_score = sc
            best_h_small = cand.copy()
        if best_score <= target:
            break

    # Hill‑climbing / simulated annealing on binary vectors
    while time_left() > 5.0 and best_score > target:
        # occasional random restart
        if rng.random() < 0.3:
            h_cur = random_binary_balanced(n_small, rng)
            cur_score = compute_c5_fft(h_cur)
        else:
            h_cur = best_h_small.copy()
            cur_score = best_score
        ones = np.where(h_cur == 1)[0].tolist()
        zeros = np.where(h_cur == 0)[0].tolist()
        max_iters = int(min(200000, max(2000, time_left() * 4000)))
        for it in range(max_iters):
            elapsed = time.time() - start
            temp = max(1e-5, (budget_s - elapsed) / budget_s * 0.001)
            i_idx = rng.integers(len(ones))
            j_idx = rng.integers(len(zeros))
            i = ones[i_idx]
            j = zeros[j_idx]
            h_cur[i], h_cur[j] = 0.0, 1.0
            new_score = compute_c5_fft(h_cur)
            accept = False
            if new_score <= cur_score:
                accept = True
            else:
                if rng.random() < np.exp(-(new_score - cur_score) / temp):
                    accept = True
            if accept:
                cur_score = new_score
                ones.pop(i_idx)
                zeros.append(i)
                zeros.pop(j_idx)
                ones.append(j)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h_small = h_cur.copy()
                    if best_score <= target:
                        break
            else:
                h_cur[i], h_cur[j] = 1.0, 0.0
            if (it & 1023) == 0 and time_left() < 1.0:
                break
        if best_score <= target:
            break

    # Upsample to larger resolution and refine continuously
    h_up = resample(best_h_small, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    remaining = time_left()
    if remaining > 5.0:
        time_end = start + budget_s - 0.2
        h_ref, sc_ref = gradient_descent_momentum(
            h_up, step0=0.5, beta=0.9, time_end=time_end, max_iter=200000)
        if sc_ref < best_score:
            best_h = h_ref
            best_score = sc_ref
        else:
            best_h = h_up
    else:
        best_h = h_up

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5_fft(best_h)
    return best_h, final_score, best_h.size
