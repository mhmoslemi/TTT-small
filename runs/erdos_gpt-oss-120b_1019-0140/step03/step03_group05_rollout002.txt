import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # Binary search on offset to adjust sum after clipping
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # overlapping region: i from 0 to n-k-1
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-7
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # shift
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                # random perturbation to escape stagnation
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    # final check that best_h is feasible
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def differential_evolution(initial_h, target_sum, n, start_time, budget_s, rng):
    # DE with projection to box+sum
    popsize = max(10, min(30, n // 4))
    # Build initial population
    population = np.empty((popsize, n))
    # Include initial_h as first individual
    population[0] = project_to_box_sum(initial_h, target_sum)
    for i in range(1, popsize):
        vec = rng.random(n)
        population[i] = project_to_box_sum(vec, target_sum)
    pop_c5 = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(pop_c5)
    best_h = population[best_idx].copy()
    best_c5 = pop_c5[best_idx]
    # DE parameters
    CR = 0.7
    F_low = 0.5
    F_high = 0.9
    safety = 5.0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - safety:
            break
        for i in range(popsize):
            # select 3 distinct others
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            F = rng.uniform(F_low, F_high)
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)
            # crossover
            jrand = rng.integers(n)
            trial = np.empty_like(mutant)
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)
            trial_c5 = compute_c5(trial)
            if trial_c5 < pop_c5[i]:
                population[i] = trial
                pop_c5[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    if best_c5 < 0.3808:
                        return best_h, best_c5
    return best_h, best_c5

    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the box [lo, hi]^n while enforcing sum(v) == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the overlap for a given integer shift k (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected gradient descent on the worst shift."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-7
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def differential_evolution(init_h, target_sum, n, start_time, budget_s, rng):
    """Simple DE with projection to the feasible box."""
    popsize = max(15, min(30, n // 2))
    population = np.empty((popsize, n))
    population[0] = project_to_box_sum(init_h, target_sum)
    for i in range(1, popsize):
        vec = rng.random(n)
        population[i] = project_to_box_sum(vec, target_sum)
    pop_c5 = np.array([compute_c5(ind) for ind in population])
    best_idx = np.argmin(pop_c5)
    best_h = population[best_idx].copy()
    best_c5 = pop_c5[best_idx]

    CR = 0.7
    F_low, F_high = 0.5, 0.9
    safety = 5.0
    while True:
        if time.time() - start_time > budget_s - safety:
            break
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            F = rng.uniform(F_low, F_high)
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            jrand = rng.integers(n)
            trial = np.empty_like(mutant)
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)

            trial_c5 = compute_c5(trial)
            if trial_c5 < pop_c5[i]:
                population[i] = trial
                pop_c5[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    if best_c5 < 0.3808:
                        return best_h, best_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, target_sum)

    # Phase 1: differential evolution (≈80 % of budget)
    de_budget = budget_s * 0.8
    best_h, best_c5 = differential_evolution(h0, target_sum, n, start_time, de_budget, rng)

    # Phase 2: gradient descent refinement with remaining time
    remaining = budget_s - (time.time() - start_time) - 5.0
    if remaining > 0.5:
        gd_limit = time.time() + remaining
        best_h, best_c5 = gradient_descent(best_h, target_sum, max_iter=200000,
                                          time_limit=gd_limit, rng=rng)

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
