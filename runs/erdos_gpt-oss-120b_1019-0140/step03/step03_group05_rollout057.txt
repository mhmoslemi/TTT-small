def simulated_annealing(h_start, target_sum, time_deadline, init_temp=0.1, min_temp=1e-6, init_sigma=0.05, min_sigma=1e-4):

   current_h = h_start.copy()
   current_c5 = compute_c5(current_h)
   best_h = current_h.copy()
   best_c5 = current_c5

   T = init_temp
   sigma = init_sigma
   iteration = 0

   while time.time() < time_deadline:
       # adapt cooling based on iteration count using exponential decay
       # Equivalent to T = max(min_temp, T * 0.9995)
       # sigma decay slower: sigma = max(min_sigma, sigma * 0.9999)
       # Could compute factors outside loop.

       # propose new
       h_new = current_h + sigma * rng.standard_normal(n)
       # clip to bounds
       h_new = np.clip(h_new, 0.0, 1.0)
       # project onto sum
       h_new = project_to_box_sum(h_new, target_sum)

       new_c5 = compute_c5(h_new)

       delta = new_c5 - current_c5
       if delta < 0 or rng.random() < np.exp(-delta / T):
           current_h, current_c5 = h_new, new_c5
           if new_c5 < best_c5:
               best_h, best_c5 = h_new.copy(), new_c5

       # schedule update
       T = max(min_temp, T * 0.9995)
       sigma = max(min_sigma, sigma * 0.99995)
       iteration += 1

   return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        if x.sum() < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.standard_normal(n) * 0.02
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def simulated_annealing(h_start, target_sum, time_deadline, rng,
                        init_temp=0.1, min_temp=1e-6,
                        init_sigma=0.05, min_sigma=1e-4):
    n = h_start.shape[0]
    cur_h = h_start.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = init_temp
    sigma = init_sigma
    while time.time() < time_deadline:
        # propose
        h_new = cur_h + sigma * rng.standard_normal(n)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_box_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
        T = max(min_temp, T * 0.9995)
        sigma = max(min_sigma, sigma * 0.9999)
    return best_h, best_c5

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1],
                   x0=x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

            h_start = best_h + rng.standard_normal(n) * 0.03
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_to_box_sum(h_start, target_sum)
            max_iter = 4000
            gd_h, gd_c5 = gradient_descent(h_start, target_sum, max_iter, gd_deadline, rng)
            if gd_c5 < best_c5:
                best_h, best_c5 = gd_h, gd_c5
                if best_c5 < 0.3808:

    # Final SLSQP refinement if time permits
    final_deadline = start_time + budget_s - 5.0
    if time.time() < final_deadline:

        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        t_start = best_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        if x.sum() < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.standard_normal(n) * 0.02
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1

def objective_t(x):
    return x[-1]

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=objective_t,
                   x0=x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

def swap_improvement(h, cur_c5, deadline, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = cur_c5
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        h_new = best_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    return best_h, best_c5
