def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        # Exponential cooling (log space)
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        # if no possible swap break
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            # probability based on temperature
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update index sets after swap
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score
```

Note: this returns best. This may be slower due to recomputing ones and zeros each iteration (O(n) each). But we can maintain incremental but fine.

def hill_climb_binary(h0, max_swaps, rng):
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_score = cur_score
    best_h = cur_h.copy()
    n = h0.shape[0]
    for _ in range(max_swaps):
        # random swap
        i = rng.integers(n)
        # ensure opposite type
        if cur_h[i] > 0.5:
            # pick a zero index randomly
            zeros = np.flatnonzero(cur_h < 0.5)
            if zeros.size == 0:
                break
            j = rng.choice(zeros)
        else:
            ones = np.flatnonzero(cur_h > 0.5)
            if ones.size == 0:
                break
            j = rng.choice(ones)
        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

        return np.full_like(v, lo)

        return np.full_like(v, hi)
    # use bisection for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):

        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Fast C5 using FFT based linear correlation."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    R = np.fft.rfft(rev, n=N)
    conv = np.fft.irfft(H * R, n=N)
    shifts = np.arange(-(n - 1), n)
    corr = conv[n - 1 - shifts]
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)

def hill_climb_binary(h0, max_swaps, rng):
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = h0.shape[0]
    for _ in range(max_swaps):
        i = rng.integers(n)
        if cur_h[i] > 0.5:
            zeros = np.flatnonzero(cur_h < 0.5)
            if zeros.size == 0:
                break
            j = rng.choice(zeros)
        else:
            ones = np.flatnonzero(cur_h > 0.5)
            if ones.size == 0:
                break
            j = rng.choice(ones)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # Choose resolution based on available time
    n = 1024 if time_left() > 300 else 512

    best_h = None
    best_score = np.inf

    # Helper for resampling
    def resample(arr, new_n):
        old_n = arr.shape[0]
        if old_n == new_n:
            return arr.copy()
        x_old = np.linspace(0, 1, old_n, endpoint=False)
        x_new = np.linspace(0, 1, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

    # Parent solution if provided
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_h = resample(init_full, n)
        init_h = project_onto_box_sum(init_h, n / 2.0)
        sc = compute_c5(init_h)
        best_h, best_score = init_h, sc
        # Binary version of the parent
        idx = np.argsort(init_h)[-n // 2:]
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx] = 1.0
        sc_bin = compute_c5(bin_h)
        if sc_bin < best_score:
            best_h, best_score = bin_h, sc_bin

    # Thue-Morse seed
    tm = thue_morse(n)
    sc_tm = compute_c5(tm)
    if sc_tm < best_score:
        best_h, best_score = tm, sc_tm

    # Random binary seed as a fallback
    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_score = compute_c5(best_h)

    runs = 0
    while time_left() > 5.0 and runs < 10:
        remaining = time_left()
        max_iter = int(min(remaining * 500, 80000))
        if max_iter < 2000:
            break
        # random start for this anneal
        h0 = random_binary_balanced(n, rng)
        h_ann, sc_ann = anneal_binary(h0, max_iter,
                                      temp_start=0.08,
                                      temp_end=1e-7,
                                      rng=rng)
        # short deterministic refinement
        if time_left() > 2.0:
            h_ref, sc_ref = hill_climb_binary(h_ann, max_swaps=3000, rng=rng)
        else:
            h_ref, sc_ref = h_ann, sc_ann

        # update global best
        if sc_ref < best_score:
            best_h, best_score = h_ref, sc_ref
        elif sc_ann < best_score:
            best_h, best_score = h_ann, sc_ann
        runs += 1

    # Ensure feasibility and final evaluation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, float(best_score), n
