def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # choose n=96
    n = 96
    half = n // 2
    # initial vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # convert to binary by thresholding at median maybe
        # choose indices of top half values
        idx = np.argsort(init)[-half:]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1
    else:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, size=half, replace=False)] = 1
    # compute max overlap
    max_count = np.max(np.correlate(h, 1-h, mode='full')).astype(int)
    best_h = h.copy()
    best_count = max_count

    # target max count threshold
    target_max = int(np.floor(0.19 * n))
    # define time budget for loop
    safety = 5.0
    # temperature schedule
    T0 = 2.0  # initial temperature (in terms of count units)
    # maybe linear schedule
    iteration = 0
    while time.time() < start + budget_s - safety:
        iteration += 1
        # choose random 1 and 0 indices
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        # compute new max count
        new_max = np.max(np.correlate(h, 1-h, mode='full')).astype(int)
        delta = new_max - max_count
        # temperature schedule: T = T0 * (0.999 ** iteration)
        T = T0 * (0.995 ** iteration)
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            # accept
            max_count = new_max
            if max_count < best_count:
                best_count = max_count
                best_h = h.copy()
                if best_count <= target_max:
                    break
        else:
            # reject: revert swap
            h[i], h[j] = 1, 0
        # occasionally (every 1000 iterations) recompute best via local greedy
        if iteration % 1000 == 0:
            # attempt local greedy improvement: try random swaps that improve
            improved = True
            while improved and time.time() < start + budget_s - safety:

                        max_count = cand_max
                        h = best_h.copy()
                        improved = True
                        break

            h = best_h.copy()
            max_count = best_count

import time
import numpy as np

def _max_overlap_count(h):
    """Return integer max overlap count (without dx scaling)."""
    return int(np.max(np.correlate(h, 1 - h, mode='full')))

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    safety = 5.0                     # stop before the kill time
    n = 100                          # number of samples (even)
    half = n // 2
    target_max = int(np.floor(0.19 * n))   # max allowed count → bound ≤ 0.38
    best_h = None
    best_max = n + 1                 # larger than any possible count

    # main outer loop: random restarts
    while time.time() < start + budget_s - safety:
        # random binary start with exactly n/2 ones
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, size=half, replace=False)] = 1
        # inner deterministic improvement loop
        no_imp_iters = 0
        while time.time() < start + budget_s - safety:
            corr = np.correlate(h, 1 - h, mode='full')
            cur_max = int(corr.max())
            # record best solution seen
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                if best_max <= target_max:
                    break
            # if already at target, exit inner loop
            if cur_max <= target_max:
                break

            # try to reduce the current worst shift
            k_idx = int(corr.argmax())
            k = k_idx - (n - 1)

            # indices of 1's that contribute to the overlap at shift k
            if k >= 0:
                bad_i = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
            else:
                kk = -k
                bad_i = np.where((h[kk:] == 1) & (h[:n - kk] == 0))[0] + kk

            if bad_i.size == 0:
                # no direct improvement possible; random perturbation
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i, j = rng.choice(ones), rng.choice(zeros)
                h[i], h[j] = 0, 1
                no_imp_iters = 0
                continue

            i = rng.choice(bad_i)          # a 1 that hurts the worst shift
            zeros = np.where(h == 0)[0]

            improved = False
            # try a few candidate zeros
            for _ in range(12):
                j = rng.choice(zeros)
                h[i], h[j] = 0, 1
                new_max = _max_overlap_count(h)
                if new_max < cur_max:
                    improved = True
                    break
                h[i], h[j] = 1, 0  # revert

            if not improved:
                # small random move to escape possible stall
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i2, j2 = rng.choice(ones), rng.choice(zeros)
                h[i2], h[j2] = 0, 1
                no_imp_iters += 1
            else:
                no_imp_iters = 0

            if no_imp_iters >= 250:
                # stagnation, break to a fresh restart
                break

        if best_max <= target_max:
            break

    # final values
    if best_h is None:
        # fallback (should not happen)
        best_h = np.zeros(n, dtype=np.int8)
        best_h[:half] = 1
        best_max = _max_overlap_count(best_h)

    final_h = best_h.astype(float)
    c5 = best_max * (2.0 / n)
    return final_h, c5, n
