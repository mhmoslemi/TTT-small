def project_onto_box_sum(v, s, lo=0.0, hi=1.0):
    """Projects onto [lo, hi] with sum s."""
    n = len(v)
    sort_indices = np.argsort(v)
    v_sorted = np.sort(v)[::-1]
    s_remaining = s
    if s_remaining <= lo * n:
        return np.full_like(v, lo)

        return np.full_like(v, hi)
    
    # Initial value for lambda

    v_new = np.clip(v - lam, lo, hi)
    s_val = np.sum(v_new)
    
    max_iter = 100
    for _ in range(max_iter):

            return v_new
