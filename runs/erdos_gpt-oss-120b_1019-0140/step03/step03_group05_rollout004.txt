import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst-shift overlap w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.01, size=n)
                new_h = h + noise
                new_h = project_to_box_sum(np.clip(new_h, 0.0, 1.0), target_sum)
                h = new_h
                alpha = 0.1
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

            if trial_c5 < c5_vals[i]:
                pop[i] = trial
                c5_vals[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    if best_c5 <= target_bound:
                        break
        generation += 1

        # periodic local refinement
        if generation % 10 == 0 and time.time() - start_time < budget_s - safety:
            refined_h, refined_c5 = gradient_descent(
                best_h, target_sum, max_iter=2000,
                time_limit=start_time + budget_s - safety, rng=rng
            )
            if refined_c5 < best_c5:
                best_h, best_c5 = refined_h, refined_c5
                worst = int(np.argmax(c5_vals))
                pop[worst] = best_h.copy()
                c5_vals[worst] = best_c5

    # Final local refinement if time permits
    if time.time() - start_time < budget_s - safety:
        final_h, final_c5 = gradient_descent(
            best_h, target_sum, max_iter=5000,
            time_limit=start_time + budget_s - safety, rng=rng
        )
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    # Ensure feasibility and final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(n)
