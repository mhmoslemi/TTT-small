def anneal_binary(h0, time_budget, rng):
    start = time.time()
    n = h0.shape[0]
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # prepare indices
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    # initial temperature scaling
    T0 = cur_score * 0.2 + 1e-4  # avoid zero
    T_end = cur_score * 1e-4 + 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # cooling
        frac = elapsed / time_budget
        # exponential cooling
        T = T0 * ((T_end / T0) ** frac)
        # propose swap
        pos_i = rng.integers(ones.size)
        pos_j = rng.integers(zeros.size)
        i = ones[pos_i]
        j = zeros[pos_j]
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # accept with probability exp(-delta/T)
            if T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            cur_score = new_score
            # update ones/zeros arrays
            ones[pos_i] = j
            zeros[pos_j] = i
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # optional large perturbation
        # implement after certain iterations count? we could not
    return best_h, best_score

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated on a grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Generate Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def get_initial_binary(target_n, rng):
    """Create a binary start vector of length target_n using initial_h_values if possible."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base_bin = (base >= 0.5).astype(int)
        n0 = base_bin.size
        if target_n % n0 == 0:
            repeat = target_n // n0
            h = np.repeat(base_bin, repeat).astype(float)
            cur = int(h.sum())
            need = target_n // 2
            if cur != need:
                diff = need - cur
                if diff > 0:
                    zer = np.flatnonzero(h == 0)
                    idx = rng.choice(zer, size=diff, replace=False)
                    h[idx] = 1.0
                else:
                    one = np.flatnonzero(h == 1)
                    idx = rng.choice(one, size=-diff, replace=False)
                    h[idx] = 0.0
            return h
    # Fallback: repeat Thue–Morse of length 64
    base = thue_morse(64)
    repeat = target_n // 64
    h = np.repeat(base, repeat).astype(float)
    cur = int(h.sum())
    need = target_n // 2
    if cur != need:
        diff = need - cur
        if diff > 0:
            zer = np.flatnonzero(h == 0)
            idx = rng.choice(zer, size=diff, replace=False)
            h[idx] = 1.0
        else:
            one = np.flatnonzero(h == 1)
            idx = rng.choice(one, size=-diff, replace=False)
            h[idx] = 0.0
    return h


def anneal_binary(h0, time_budget, rng):
    """Simulated annealing on binary vectors preserving the sum constraint."""
    start = time.time()
    n = h0.shape[0]
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)

    # temperature schedule
    T0 = max(cur_score * 0.2, 1e-4)
    T_end = 1e-6

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        frac = elapsed / time_budget
        T = T0 * ((T_end / T0) ** frac)  # exponential cooling

        # pick a random 1 and a random 0 to swap
        pos_i = rng.integers(ones.size)
        pos_j = rng.integers(zeros.size)
        i = ones[pos_i]
        j = zeros[pos_j]

        # perform swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - cur_score

        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_score = new_score
            # update index lists
            ones[pos_i] = j
            zeros[pos_j] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    target_n = 512  # resolution (must be even)

    # initial binary pattern
    h0 = get_initial_binary(target_n, rng)

    # first annealing run (takes up to 60% of the budget)
    first_alloc = min(600.0, time_left() * 0.6)
    best_h, best_score = anneal_binary(h0, first_alloc, rng)

    # additional random restarts while time permits
    while time_left() > 30.0:
        # generate a new random balanced binary start
        h_start = random_balanced_binary(target_n, rng)
        alloc = min(300.0, time_left() * 0.4)
        cand_h, cand_score = anneal_binary(h_start, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
            if best_score <= 0.3808:
                break

    # final safety: ensure exact sum constraint
    cur_sum = int(best_h.sum())
    target_sum = target_n // 2
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zer = np.flatnonzero(best_h == 0)
            idx = rng.choice(zer, size=diff, replace=False)
            best_h[idx] = 1.0
        else:
            one = np.flatnonzero(best_h == 1)
            idx = rng.choice(one, size=-diff, replace=False)
            best_h[idx] = 0.0

    final_score = compute_c5(best_h)
    return best_h, final_score, target_n
