def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt

    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        init_h = None

    dx = 2.0 / n
    M = n / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        # Clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Bounded simplex projection
        # Use simple iterative approach
        # Compute current sum
        s = np.sum(h)
        # if sum already close, ok
        # Use water-filling: adjust while sum != M
        # We'll perform at most a few iterations
        for _ in range(10):
            if abs(s - M) <= 1e-12:
                break
            if s > M:
                # need to reduce excess
                excess = s - M
                # find positive entries to reduce
                pos = h > 0
                if not np.any(pos):
                    break
                # distribute excess uniformly among pos entries
                delta = excess / np.sum(pos)
                # subtract but not below 0
                new_h = h.copy()
                new_h[pos] = np.maximum(0.0, h[pos] - delta)
                h = new_h
            else:
                deficit = M - s
                # find entries <1
                free = h < 1
                if not np.any(free):
                    break
                delta = deficit / np.sum(free)
                new_h = h.copy()
                new_h[free] = np.minimum(1.0, h[free] + delta)
                h = new_h
            s = np.sum(h)
        # final correction for any residual using scaling if still off
        s = np.sum(h)
        if abs(s - M) > 1e-9:
            # do simple scaling then clip again
            h = h + (M - s) / n
            h = np.clip(h, 0.0, 1.0)
        return h

    # define penalty objective for unconstrained optimizer
    penalty_factor = 1e3

    def penalized_obj(h):
        # h is vector length n
        c5 = compute_c5(h)
        # constraint violations
        sum_violation = abs(np.sum(h) - M)
        lower_violation = np.sum(np.maximum(-h, 0.0))
        upper_violation = np.sum(np.maximum(h - 1.0, 0.0))
        penalty = penalty_factor * (sum_violation + lower_violation + upper_violation)
        return c5 + penalty

    best_h = None
    best_c5 = np.inf

    # incorporate initial usage
    if init_h is not None:
        h0 = project(init_h.copy())
        c5_0 = compute_c5(h0)
        best_h, best_c5 = h0, c5_0

    # create list of seeds for initial guesses
    # generate few heuristic seeds
    def random_feasible():
        # generate random numbers uniformly, then scale to sum M and clip
        x = rng.random(n)
        x = x / np.sum(x) * M
        # after scaling some may be >1, clip and renormalize iteratively
        x = np.clip(x, 0.0, 1.0)
        # re-balance to exact sum within bounds
        x = project(x)
        return x

    # try Legendre if n==80
    if n == 80:
        # compute Legendre for p=79
        p = 79
        seq = np.zeros(p, dtype=float)
        for i in range(1, p):
            leg = pow(i, (p - 1) // 2, p)
            seq[i] = 1.0 if leg == 1 else 0.0
        leg80 = np.concatenate([seq, [1.0]])
        # shift
        shift = rng.integers(n)
        leg80 = np.roll(leg80, shift)
        leg80 = project(leg80)
        c5_leg = compute_c5(leg80)
        if c5_leg < best_c5:
            best_h, best_c5 = leg80, c5_leg

    # allocate time for global search
    # Use differential evolution if available
    use_de = True
    # Determine time left
    # We'll run DE first for a chunk, then local refinements
    time_left = budget_s - (time.time() - start)
    # set DE time slice ~0.6 of remaining if enough
    if use_de and time_left > 30:
        de_time = time_left * 0.6
        # define bounds
        bounds = [(0.0, 1.0)] * n
        # define callback to stop early if time exceeded
        def de_callback(x, convergence):
            # x is current best solution
            # project to constraints and evaluate
            nonlocal best_h, best_c5, start, de_time
            if time.time() - start > start + de_time:
                return True  # stop
            # quick evaluate
            h = project(x)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
                # can also early stop if target reached
                if best_c5 < 0.3808:
                    return True
            return False

        # Run DE
        try:
            result = opt.differential_evolution(
                penalized_obj,
                bounds,
                seed=seed,
                updating='deferred',
                strategy='best1bin',
                popsize=10,
                maxiter=2000,  # large, will be truncated by callback
                callback=de_callback,
                polish=False,
                disp=False,
                atol=0,
                tol=0,
            )
        except Exception:
            result = None

        # After DE, if result.success (or not), get best candidate
        if result is not None:
            h_candidate = project(result.x)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h, best_c5 = h_candidate, c5_candidate

    # After DE, run local refinements using SLSQP from best_h and random seeds
    # function to run SLSQP
    def slsqp_optimize(h_start):
        # Ensure feasible start
        h_start = project(h_start)
        t_start = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t_start]])
        # constraints
        constr = []
        # sum equality
        def sum_eq(x):
            return np.sum(x[:n]) - M
        constr.append({'type': 'eq', 'fun': sum_eq})
        # overlap inequality for each shift k != 0
        # we define function for a given k
        for k in range(-n+1, n):
            if k == 0:
                continue
            def overlap_fun(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * dx
            constr.append({'type': 'ineq', 'fun': overlap_fun})
        # bounds: h in [0,1], t >=0
        bnds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = opt.minimize(
                lambda x: x[-1],
                x0,
                method='SLSQP',
                bounds=bnds,
                constraints=constr,
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False},
            )
        except Exception:
            return None, np.inf
        if not res.success:
            # try fallback: ignore constraints and project
            h_opt = project(res.x[:n])
            c5_opt = compute_c5(h_opt)
            return h_opt, c5_opt
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project(h_opt)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt

    # local refinement loop
    while time.time() - start < budget_s - 1.0:
        # random perturbation around best_h or random start based on probability
        if best_h is not None and rng.random() < 0.7:
            # perturb best
            h0 = best_h.copy()
            # perform a few random swaps or fractional moves
            for _ in range(5):
                i, j = rng.choice(n, 2, replace=False)
                # compute max transfer to keep in [0,1]
                max_transfer = min(1.0 - h0[i], h0[j])
                max_transfer2 = min(h0[i], 1.0 - h0[j])
                # randomly decide direction
                if rng.random() < 0.5:
                    # move from j to i
                    if max_transfer > 0:
                        delta = rng.random() * max_transfer
                        h0[i] += delta
                        h0[j] -= delta
                else:
                    # move from i to j
                    if max_transfer2 > 0:
                        delta = rng.random() * max_transfer2
                        h0[i] -= delta
                        h0[j] += delta
            # also small random noise then project
            h0 += rng.normal(scale=0.01, size=n)
            h0 = project(h0)
        else:
            h0 = random_feasible()
        # Run SLSQP
        h_opt, c5_opt = slsqp_optimize(h0)
        if h_opt is not None and c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # fallback if never found feasible
    if best_h is None:
        # simple uniform h=0.5 (feasible)
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    # Ensure final best_h is feasible
    best_h = project(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import numpy as np
import scipy.optimize as opt

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_boxed_simplex(h, M):
    """Project h onto {0<=h_i<=1, sum h_i = M}."""
    h = np.clip(h, 0.0, 1.0)
    # Simple water‑filling iteration
    for _ in range(10):
        s = np.sum(h)
        if abs(s - M) <= 1e-12:
            break
        if s > M:      # need to reduce
            excess = s - M
            pos = h > 0
            if not np.any(pos):
                break
            delta = excess / np.sum(pos)
            h[pos] = np.maximum(0.0, h[pos] - delta)
        else:          # need to increase
            deficit = M - s
            free = h < 1
            if not np.any(free):
                break
            delta = deficit / np.sum(free)
            h[free] = np.minimum(1.0, h[free] + delta)
    # final tiny correction
    s = np.sum(h)
    if abs(s - M) > 1e-9:
        h += (M - s) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- basic problem data -----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
        if n % 2 != 0:  # require even length
            init_h = None
            n = 80
        else:
            # ensure sum close to n/2
            M_target = n / 2.0
            init_h = _project_to_boxed_simplex(init_h, M_target)
    else:
        n = 80
        init_h = None

    dx = 2.0 / n
    M = n / 2.0

    # ----- utility functions -----
    def compute_c5(h):
        return _compute_c5(h, dx)

    def random_feasible():
        x = rng.random(n)
        x = x / np.sum(x) * M
        x = _project_to_boxed_simplex(x, M)
        return x

    # ----- start from known constructions -----
    best_h = None
    best_c5 = np.inf

    if init_h is not None:
        h0 = init_h.copy()
        c5 = compute_c5(h0)
        best_h, best_c5 = h0, c5

    # Legendre seed for n=80
    if n == 80:
        p = 79
        seq = np.zeros(p, dtype=float)
        for i in range(1, p):
            leg = pow(i, (p - 1) // 2, p)
            seq[i] = 1.0 if leg == 1 else 0.0
        leg80 = np.concatenate([seq, [1.0]])
        leg80 = np.roll(leg80, rng.integers(n))
        leg80 = _project_to_boxed_simplex(leg80, M)
        c5 = compute_c5(leg80)
        if c5 < best_c5:
            best_h, best_c5 = leg80, c5

    # ----- global search with differential evolution -----
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30:
        # modest DE settings to stay within time budget
        de_popsize = 10          # population = popsize * n
        de_maxiter = 200        # enough generations for exploration
        bounds = [(0.0, 1.0)] * n
        penalty = 1e3

        def penalized_obj(h):
            # h is a candidate vector
            c5 = compute_c5(h)
            sum_violation = abs(np.sum(h) - M)
            lower = np.sum(np.maximum(-h, 0.0))
            upper = np.sum(np.maximum(h - 1.0, 0.0))
            return c5 + penalty * (sum_violation + lower + upper)

        # initialise with a few good seeds if we have them
        init_pop = []
        if best_h is not None:
            init_pop.append(best_h)
        if n == 80:
            init_pop.append(leg80)
        while len(init_pop) < de_popsize:
            init_pop.append(random_feasible())
        init_array = np.vstack(init_pop[:de_popsize])

        try:
            result = opt.differential_evolution(
                penalized_obj,
                bounds,
                init=init_array,
                seed=seed,
                strategy='best1bin',
                popsize=de_popsize,
                maxiter=de_maxiter,
                polish=False,
                disp=False,
                atol=0,
                tol=0,
                updating='deferred',
            )
            h_candidate = _project_to_boxed_simplex(result.x, M)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h, best_c5 = h_candidate, c5_candidate
        except Exception:
            pass

    # ----- local refinement with SLSQP -----
    def slsqp_refine(h_start):
        h0 = _project_to_boxed_simplex(h_start, M)
        t0 = compute_c5(h0)
        x0 = np.concatenate([h0, [t0]])

        # constraints: sum == M and t >= overlap for each shift
        cons = []

        def sum_eq(x):
            return np.sum(x[:n]) - M
        cons.append({'type': 'eq', 'fun': sum_eq})

        def make_overlap(k):
            def fun(x):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * dx
            return fun

        for k in range(-n + 1, n):
            if k == 0:
                continue
            cons.append({'type': 'ineq', 'fun': make_overlap(k)})

        bnds = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = opt.minimize(
                lambda x: x[-1],
                x0,
                method='SLSQP',
                bounds=bnds,
                constraints=cons,
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False},
            )
        except Exception:
            return None, np.inf

        if not res.success:
            # fallback to projection of the result
            h_opt = _project_to_boxed_simplex(res.x[:n], M)
            return h_opt, compute_c5(h_opt)

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = _project_to_boxed_simplex(h_opt, M)
        return h_opt, compute_c5(h_opt)

    # main refinement loop
    while time.time() - start_time < budget_s - 1.0:
        if best_h is not None and rng.random() < 0.65:
            # perturb around the best known
            h_trial = best_h.copy()
            # a few random swaps / transfers
            for _ in range(4):
                i, j = rng.choice(n, 2, replace=False)
                # transfer from j to i
                max_tx = min(1.0 - h_trial[i], h_trial[j])
                if max_tx > 0:
                    delta = rng.random() * max_tx
                    h_trial[i] += delta
                    h_trial[j] -= delta
            h_trial += rng.normal(scale=0.008, size=n)  # tiny noise
        else:
            h_trial = random_feasible()

        h_opt, c5_opt = slsqp_refine(h_trial)
        if h_opt is not None and c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # ----- final safety net -----
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
    else:
        best_h = _project_to_boxed_simplex(best_h, M)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
