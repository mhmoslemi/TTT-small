def anneal_binary_full(h0, time_limit, rng):
    # h0 binary balanced
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    start = time.time()
    iter = 0
    # schedule parameters: T0 = 0.01, T_end = 1e-6
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        iter += 1
        t = 0.01 * (1e-6/0.01) ** (elapsed / time_limit)  # geometric schedule
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t+1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update indices
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

                                                   step0=0.2, beta=0.9)
        if gd_score < best_score:
            best_h, best_score = gd_h, gd_score

    # Phase 4: brief SLSQP refinement (rest of time)
    remaining = time_left()
    if remaining > 2.0:
        try:
            opt_h, opt_score = slsqp_optimize_candidate(best_h, max_iter=200)
            if opt_score < best_score:
                best_h, best_score = opt_h, opt_score
        except Exception:
            pass

    # Final projection and scoring
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
