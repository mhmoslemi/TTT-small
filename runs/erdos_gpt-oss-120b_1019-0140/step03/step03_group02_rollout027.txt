        x_large = np.linspace(0, 1, n_large, endpoint=False)
        h_large = np.interp(x_large, x_old, best_h)
        h_large = project_onto_box_sum(h_large, n_large / 2.0)
        h_large, sc_large = adam_softmax_corr(
            h_large, n_iter=4000, step0=0.4,
            alpha_start=20.0, alpha_end=80.0,
            time_left=time_left)
        if time_left() > 2.0:
            h_large, sc_large2 = momentum_grad_descent(
                h_large, n_iter=3000, step0=0.25, beta=0.9,
                time_left=time_left)
            if sc_large2 < sc_large:
                sc_large = sc_large2
        if sc_large < best_score:
            best_score, best_h = sc_large, h_large.copy()

    # ----- extra random restarts at large scale (if time permits) -----
    if time_left() > 5.0:
        for _ in range(3):
            h_rand = project_onto_box_sum(rng.random(n_large), n_large / 2.0)
            h_rand, sc_rand = adam_softmax_corr(
                h_rand, n_iter=1500, step0=0.2,
                alpha_start=20.0, alpha_end=80.0,
                time_left=time_left)
            if sc_rand < best_score:
                best_score, best_h = sc_rand, h_rand.copy()
            if time_left() < 1.0:
                break

    # ----- binary hill‑climb (swap SA) -----
    if time_left() > 3.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        cur = binary.copy()
        cur_sc = compute_c5(cur)
        best_bin = cur.copy()
        best_bin_sc = cur_sc
        max_iters = int(min(200000, max(1, time_left() * 2000)))
        T0 = 0.01
        for it in range(1, max_iters + 1):
            if time_left() < 0.01:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while cur[i] == cur[j]:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            nxt = cur.copy()
            nxt[i], nxt[j] = nxt[j], nxt[i]
            nxt_sc = compute_c5(nxt)
            delta = nxt_sc - cur_sc
            T = T0 * (1 - it / max_iters)
            accept = False
            if nxt_sc < cur_sc:
                accept = True
            elif T > 0:
                accept = rng.random() < math.exp(-delta / T)
            if accept:
                cur, cur_sc = nxt, nxt_sc
                if nxt_sc < best_bin_sc:
                    best_bin_sc, best_bin = nxt_sc, nxt.copy()
                    if best_bin_sc <= 0.3808:
                        break
        if best_bin_sc < best_score:
            best_score, best_h = best_bin_sc, best_bin.copy()

    # final projection and report
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
