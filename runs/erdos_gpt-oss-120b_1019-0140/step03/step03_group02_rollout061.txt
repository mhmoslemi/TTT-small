def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def grad_softmax_corr(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')

    grad = np.zeros_like(h)

    for s, k in enumerate(range(-(n - 1), n)):

        if w == 0.0:

            grad[k:] -= w * h[:n - k]

            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

    curr_score = best_score
    max_iter = int(500000) # large

    for it in range(1, max_iter+1):
        if time_left() < 0.01:

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr) - (n - 1)
        curr_score = max_corr * (2.0 / n)
        # target swap with probability maybe high
        if rng.random() < 0.8:
            # find i set

            if i_candidates.size > 0:

                # choose j random zero not equal to i
                zero_indices = np.where(h == 0.0)[0]
                # ensure j != i

    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)

import time
import numpy as np
import math

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((i.bit_count() & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr, new_n):
    """Linear interpolation to a larger grid."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def binarize_top_half(v):
    """Convert to binary by setting the top half entries to 1."""
    n = v.shape[0]
    order = np.argsort(v)
    b = np.zeros(n, dtype=float)
    b[order[n // 2:]] = 1.0
    return b


def binary_sa(h0, rng, time_left, target=None):
    """Simulated‑annealing on balanced binary vectors."""
    n = h0.shape[0]
    # Ensure binary
    if not np.all((h0 == 0.0) | (h0 == 1.0)):
        h = binarize_top_half(h0)
    else:
        h = h0.copy()
    # Enforce exact balance
    diff = int(h.sum() - n / 2)
    if diff > 0:
        ones = np.where(h == 1.0)[0]
        h[rng.choice(ones, size=diff, replace=False)] = 0.0
    elif diff < 0:
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(zeros, size=-diff, replace=False)] = 1.0

    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    max_iter = 500_000
    T0 = 0.005
    idxs = np.arange(n)

    for it in range(1, max_iter + 1):
        if time_left() < 0.01:
            break

        # correlation and dominant shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr) - (n - 1)
        curr_score = max_corr * (2.0 / n)

        # propose swap (targeted with probability 0.8)
        if rng.random() < 0.8:
            valid = (idxs + k >= 0) & (idxs + k < n)
            vi = idxs[valid]
            if vi.size:
                mask = (h[vi] == 1.0) & (h[vi + k] == 0.0)
                i_cand = vi[mask]
            else:
                i_cand = np.array([], dtype=int)

            if i_cand.size:
                i = rng.choice(i_cand)
                j = rng.choice(np.where(h == 0.0)[0])
            else:
                i = rng.choice(np.where(h == 1.0)[0])
                j = rng.choice(np.where(h == 0.0)[0])
        else:
            i = rng.choice(np.where(h == 1.0)[0])
            j = rng.choice(np.where(h == 0.0)[0])

        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)

        delta = new_score - curr_score
        T = T0 * (1.0 - it / max_iter)
        if delta < 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if target is not None and best_score <= target:
                    return best_h, best_score

    return best_h, best_score
