def generate_linear_seed(n, M, rng):
    # linear decreasing from 1 to 0
    lin = np.linspace(1.0, 0.0, n)
    sum_lin = np.sum(lin)
    if sum_lin == 0:
        return np.full(n, M/n)
    scaling = M / sum_lin
    h = lin * scaling
    # clip to [0,1], adjust sum
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        # distribute deficit/excess uniformly, then clip
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

Define hill_climb function:

def hill_climb(h0, time_budget, rng):
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    # parameters
    max_no_improve = 200  # after this many fails break
    no_improve_cnt = 0
    while time.time() - start < time_budget:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        # update best
        if c5 + 1e-12 < best_c5:
            best_c5 = c5
            no_improve_cnt = 0
        else:
            no_improve_cnt += 1
            if no_improve_cnt >= max_no_improve:
                break
        # find shift with max overlap
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)
        # compute neighbor diff
        neighbor = np.roll(h, -k)
        diff = 1.0 - neighbor
        # candidates
        # i: search among indices with h>0
        pos_idx = np.where(h > 1e-12)[0]
        if pos_idx.size == 0:
            break
        i = pos_idx[np.argmax(diff[pos_idx])]
        # j: search among indices with h<1
        neg_idx = np.where(h < 1.0 - 1e-12)[0]
        if neg_idx.size == 0:
            break
        j = neg_idx[np.argmin(diff[neg_idx])]
        # if diff[i] <= diff[j], then move won't improve this shift, try random
        if diff[i] <= diff[j]:
            # random small transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if h[i] < 1e-12 or h[j] > 1.0 - 1e-12:
                continue
            delta = min(h[i], 1.0 - h[j]) * rng.random()
            if delta < 1e-12:
                continue
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5:
                h = h_new
                best_c5 = c5_new
                no_improve_cnt = 0
            continue
        # make maximal transfer
        delta = min(h[i], 1.0 - h[j])
        if delta < 1e-12:
            # fallback to random
            continue
        # try reduce delta if no improvement
        move_accepted = False
        # try full delta then scaled down
        for scale in (1.0, 0.5, 0.25, 0.1):
            d = delta * scale
            h_new = h.copy()
            h_new[i] -= d
            h_new[j] += d
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5:
                h = h_new
                best_c5 = c5_new
                no_improve_cnt = 0
                move_accepted = True
                break
        if not move_accepted:
            # maybe try alternative i,j with next best diff values
            # try few alternative pairs
            alt_i_candidates = pos_idx[np.argsort(-diff[pos_idx])][:5]
            alt_j_candidates = neg_idx[np.argsort(diff[neg_idx])][:5]
            improved = False
            for ii in alt_i_candidates:
                for jj in alt_j_candidates:
                    if ii == jj:
                        continue
                    if diff[ii] <= diff[jj]:
                        continue
                    dmax = min(h[ii], 1.0 - h[jj])
                    if dmax < 1e-12:
                        continue
                    for scale2 in (1.0, 0.5, 0.25):
                        d = dmax * scale2
                        h_alt = h.copy()
                        h_alt[ii] -= d
                        h_alt[jj] += d
                        c5_alt = compute_c5(h_alt)
                        if c5_alt + 1e-12 < best_c5:
                            h = h_alt
                            best_c5 = c5_alt
                            no_improve_cnt = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                # nothing improved, continue loop (will increase no_improve_cnt)
                pass
    return h, best_c5

Now SLSQP refinement function:

def slsqp_refine(h, time_budget, rng):
    n = h.shape[0]
    dx = 2.0 / n
    M = n / 2.0
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def sum_eq(x):
        return np.sum(x[:n]) - M

    constraints = [{'type': 'eq', 'fun': sum_eq}]
    # add overlap constraints
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_con(k):
            def con(x):
                h_vars = x[:n]
                if k >= 0:
                    overl = np.dot(h_vars[:n - k], 1.0 - h_vars[k:])
                else:
                    kk = -k
                    overl = np.dot(h_vars[kk:], 1.0 - h_vars[:n - kk])
                return x[n] - overl * dx
            return con
        constraints.append({'type': 'ineq', 'fun': make_con(k)})

    # Options: maxiter scaled to time_budget
    maxiter = max(200, int(time_budget / 0.01))  # rough estimate, maybe >500
    # but limit to maybe 2000
    maxiter = min(maxiter, 2000)
    try:
        res = minimize(lambda x: x[n], x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_opt = compute_c5(h_opt)
            return h_opt, c5_opt
    except Exception:
        pass
    return h, compute_c5(h)

# ----- basic utilities -----
def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def generate_random_binary(n, M, rng):
    """Binary vector with exactly M ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h


def generate_linear_seed(n, M, rng):
    """Linear decreasing shape, adjusted to meet the sum constraint."""
    lin = np.linspace(1.0, 0.0, n)
    if np.sum(lin) == 0:
        h = np.full(n, M / n, dtype=float)
    else:
        scaling = M / np.sum(lin)
        h = lin * scaling

    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

    no_improve = 0

    while time.time() - start < time_budget:
        # full correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)

        # try a directed move that reduces the contribution at lag k
        neighbor = np.roll(h, -k)
        diff = 1.0 - neighbor  # contribution factor for each index

        pos_idx = np.where(h > 1e-12)[0]
        neg_idx = np.where(h < 1.0 - 1e-12)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break

        i = pos_idx[np.argmax(diff[pos_idx])]
        j = neg_idx[np.argmin(diff[neg_idx])]

        if diff[i] <= diff[j]:
            # no improving directed move; try a small random transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or h[i] < 1e-12 or h[j] > 1.0 - 1e-12:
                no_improve += 1
                if no_improve >= max_no_improve:
                    break
                continue
            d = min(h[i], 1.0 - h[j]) * rng.random()
            if d < 1e-12:
                no_improve += 1
                if no_improve >= max_no_improve:
                    break
                continue
            h_new = h.copy()
            h_new[i] -= d
            h_new[j] += d
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5:
                h, best_c5 = h_new, c5_new
                no_improve = 0
                continue
        else:
            # attempt to move the maximal feasible mass from i to j
            delta = min(h[i], 1.0 - h[j])
            improved = False
            for scale in (1.0, 0.5, 0.25, 0.1):
                d = delta * scale
                if d < 1e-12:
                    continue
                h_new = h.copy()
                h_new[i] -= d
                h_new[j] += d
                c5_new = compute_c5(h_new)
                if c5_new + 1e-12 < best_c5:
                    h, best_c5 = h_new, c5_new
                    no_improve = 0
                    improved = True
                    break
            if improved:
                continue

            # fallback: try a few alternative pairs around the top candidates
            alt_i = pos_idx[np.argsort(-diff[pos_idx])][:5]
            alt_j = neg_idx[np.argsort(diff[neg_idx])][:5]
            for ii in alt_i:
                for jj in alt_j:
                    if ii == jj or diff[ii] <= diff[jj]:
                        continue
                    dmax = min(h[ii], 1.0 - h[jj])
                    if dmax < 1e-12:
                        continue
                    for scale in (1.0, 0.5, 0.25):
                        d = dmax * scale
                        h_new = h.copy()
                        h_new[ii] -= d
                        h_new[jj] += d
                        c5_new = compute_c5(h_new)
                        if c5_new + 1e-12 < best_c5:
                            h, best_c5 = h_new, c5_new
                            no_improve = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

    return h, best_c5


# ----- constraint objects for SLSQP -----
class SumConstraintFun:
    def __init__(self, n, M):
        self.n = n
        self.M = M
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.M


class OverlapConstraintFun:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[self.n] - overl * self.dx


class ObjectiveT:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return x[self.n]


# ----- SLSQP refinement -----
def slsqp_refine(h, time_budget, rng):
    n = h.shape[0]
    dx = 2.0 / n
    M = n / 2.0
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    sum_con = SumConstraintFun(n, M)
    obj = ObjectiveT(n)
    constraints = [{'type': 'eq', 'fun': sum_con}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append({'type': 'ineq', 'fun': OverlapConstraintFun(k, n, dx)})

    maxiter = max(200, int(time_budget * 50))
    maxiter = min(maxiter, 2000)

    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = np.sum(h_opt)
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_opt = compute_c5(h_opt)
            return h_opt, c5_opt
    except Exception:
        pass
    return h, compute_c5(h)


# ----- main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80, 96, 112, 128, 144]
    # prioritize any parent construction if present
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates.insert(0, n0)

    for n in candidates:
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        # allocate a share of the remaining budget to this resolution
        if n == candidates[0]:
            alloc = min(remaining * 0.6, 600.0)
        else:
            left = len(candidates) - candidates.index(n)
            alloc = min(remaining * 0.4 / left, 300.0)
        if alloc < 5.0:
            continue

        M = n // 2
        seeds = []

        # parent if size matches
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            seeds.append(np.array(initial_h_values, dtype=float))

        # Legendre seed for n=80
        if n == 80:
            p = 79
            leg = np.zeros(p, dtype=float)
            for i in range(1, p):
                leg_val = pow(i, (p - 1) // 2, p)
                leg[i] = 1.0 if leg_val == 1 else 0.0
            h_leg = np.concatenate([leg, [1.0]])  # length 80
            cur = np.sum(h_leg)
            if abs(cur - M) > 1e-12:
                h_leg += (M - cur) / n
                h_leg = np.clip(h_leg, 0.0, 1.0)
            h_leg = np.roll(h_leg, rng.integers(n))
            seeds.append(h_leg)

        # binary random
        seeds.append(generate_random_binary(n, M, rng))
        # linear shape
        seeds.append(generate_linear_seed(n, M, rng))
        # extra random binaries
        for _ in range(3):
            seeds.append(generate_random_binary(n, M, rng))

        # explore each seed
        per_seed_alloc = max(5.0, alloc / max(1, len(seeds)))
        for h0 in seeds:
            if time.time() - start > budget_s - 2:
                break
            h_opt, c5_opt = hill_climb(h0, per_seed_alloc, rng)

            # slight SLSQP polish if time permits
            time_left = budget_s - (time.time() - start)
            if time_left > 5.0:
                slsq_time = min(5.0, time_left * 0.2)
                h_opt, c5_opt = slsqp_refine(h_opt, slsq_time, rng)

            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                if best_c5 < 0.3808:
                    break
        if best_c5 < 0.3808:
            break

    # fallback safety
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        M = n / 2.0
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
