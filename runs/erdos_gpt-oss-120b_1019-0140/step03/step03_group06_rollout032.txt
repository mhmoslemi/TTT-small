def gradient_overlap(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # indices i=0..n-k-1 contribute via h_i term
        grad[:n-k] += 1 - h[k:]
        # indices j = i+k: includes -h_i term
        grad[k:] += -h[:n-k]

        grad[:n-kk] += -h[kk:]
    # This gradient is for O(k) i.e., sum_i h_i (1 - h_{i+k}), not scaled by dx.
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_simplex_box(v, target, lower=0.0, upper=1.0, eps=1e-12):
    """
    Euclidean projection of v onto the polytope:
        lower <= x_i <= upper,  sum_i x_i = target
    Uses bisection on the Lagrange multiplier.
    """
    v = np.clip(v, lower, upper)
    n = v.shape[0]
    if target < lower * n - 1e-9 or target > upper * n + 1e-9:
        # Infeasible; fallback to simple scaling and clipping
        x = v.copy()
        total = x.sum()
        if total > 0:
            x *= target / total
        return np.clip(x, lower, upper)

    lam_low = -np.max(v) - 1.0   # yields all ones after clipping
    lam_high = np.max(v) + 1.0   # yields all zeros after clipping
    for _ in range(60):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return x

def gradient_overlap(h, k):
    """Gradient of the overlap sum for a fixed shift k."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] += -h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] += -h[kk:]
    return grad

def gradient_descent_refine(h, target, dx, deadline, rng):
    """Simple projected gradient descent targeting the worst shift."""
    best_h = h.copy()
    best_val = compute_c5(best_h)
    step = 0.5
    while time.time() < deadline - 0.1:
        O = np.correlate(h, 1 - h, mode="full") * dx
        k_max = int(np.argmax(O) - (len(h) - 1))
        grad = gradient_overlap(h, k_max)
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        direction = -grad / norm
        new_h = project_to_simplex_box(h + step * direction, target)
        new_val = compute_c5(new_h)
        if new_val < best_val:
            h = new_h
            best_h = new_h
            best_val = new_val
            step = min(step * 1.2, 1.0)
        else:
            step *= 0.5
            if step < 1e-5:
                # random shake to escape plateaus
                h = best_h + 0.01 * (rng.random(len(h)) - 0.5)
                h = project_to_simplex_box(h, target)
                step = 0.5
        if best_val < 0.3808:  # early exit if target reached
            break
    return best_h, best_val

    if "initial_h_values" in globals():
        old = np.asarray(initial_h_values, dtype=float)
        old_n = old.shape[0]
        xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(xs_new, xs_old, old)
    else:
        h0 = rng.random(n)

    h0 = project_to_simplex_box(h0, target)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # smooth approximation of the max (log-sum-exp)
    alpha = 30.0  # higher → tighter approximation
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        # stable log-sum-exp
        lse = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        return lse * dx

    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    # several SLSQP restarts while time permits
    while time.time() < deadline - 25.0:
        # perturb current best to diversify starting points
        init = best_h + 0.05 * (rng.random(n) - 0.5)
        init = np.clip(init, 0.0, 1.0)
        init = project_to_simplex_box(init, target)

        res = minimize(
            smooth_obj,
            init,
            method="SLSQP",
            bounds=bounds,
            constraints=eq_cons,
            options={"maxiter": 2000, "ftol": 1e-12, "disp": False},
        )
        if not res.success:
            continue
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_to_simplex_box(h_opt, target)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
            if best_val < 0.3808:
                break

    # local gradient descent refinement
    if time.time() < deadline - 5.0:
        best_h, best_val = gradient_descent_refine(
            best_h, target, dx, deadline, rng
        )

    # final short random jitter search
    while time.time() < deadline - 0.5:
        pert = best_h + 0.02 * (rng.random(n) - 0.5)
        pert = project_to_simplex_box(pert, target)
        val = compute_c5(pert)
        if val < best_val:
            best_val, best_h = val, pert.copy()
            if best_val < 0.3808:
                break

    # safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_simplex_box(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
