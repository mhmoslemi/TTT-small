import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    """Differential Evolution on C5 at resolution n."""
    target_sum = n / 2.0
    def cost(x):
        v = np.clip(x, 0.0, 1.0)
        v = project_onto_box_sum(v, target_sum)
        return compute_c5(v)
    bounds = [(0.0, 1.0)] * n
    # population and iterations tuned to remaining time
    if n <= 64:
        popsize = 10
        maxiter = max(20, int(min(200, time_left() * 10)))
    else:
        popsize = 5
        maxiter = max(10, int(min(50, time_left() * 2)))
    # build initial population: include init_h then random candidates
    pop = [init_h]
    needed = popsize * n - len(pop)
    for _ in range(needed):
        cand = rng.random(n)
        cand = project_onto_box_sum(cand, target_sum)
        pop.append(cand)
    init_pop = np.stack(pop[:popsize * n])
    try:
        res = opt.differential_evolution(cost, bounds, strategy='best1bin',
                                         maxiter=maxiter, popsize=popsize,
                                         init=init_pop, polish=False,
                                         seed=int(rng.integers(2**31 - 1)),
                                         disp=False)
        best = project_onto_box_sum(res.x, target_sum)
        return best, compute_c5(best)
    except Exception:
        return init_h, compute_c5(init_h)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    rng = np.random.default_rng(seed)

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256]
    for n in scales:
        if time_left() <= 0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)
        cur_score = compute_c5(best_h)
        if cur_score < best_score:
            best_score, best_h = cur_score, best_h.copy()

        # Differential Evolution
        if time_left() > 5:
            de_h, de_score = de_optimize(n, best_h, time_left, rng)
            if de_score < best_score:
                best_score, best_h = de_score, de_h.copy()

        # Gradient descent momentum
        if time_left() > 5:
            factor = 4000.0 / n
            n_iter = int(min(60000, max(15000, time_left() * factor)))
            gd_h, gd_score = gradient_descent_momentum_custom(
                best_h, n_iter=n_iter, step0=0.5, beta=0.9, rng=rng,
                time_left=time_left)
            if gd_score < best_score:
                best_score, best_h = gd_score, gd_h.copy()

        # Local random search
        while time_left() > 3:
            # random balanced binary candidate
            h_rand = random_binary_balanced(n, rng)
            s_rand = compute_c5(h_rand)
            if s_rand < best_score:
                best_score, best_h = s_rand, h_rand.copy()
                if best_score <= 0.3808:
                    break
            # small mass transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(best_h[i], 1.0 - best_h[j])
            if max_delta > 0:
                delta = max_delta * rng.uniform(0.05, 0.3)
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score, best_h = s_new, h_new.copy()
                    if best_score <= 0.3808:
                        break
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break

    # Optional refinement at final resolution
    if time_left() > 5:
        n = best_h.shape[0]
        # fine‑grained gradient descent
        fine_iter = int(min(120000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=fine_iter, step0=0.2, beta=0.9,
            rng=rng, time_left=time_left)

    # Final SLSQP minimax refinement (if time permits)
    if time_left() > 5:
        n = best_h.shape[0]
        t0 = best_score
        x0 = np.concatenate([best_h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        def eq_sum(x):
            return np.sum(x[:n]) - n / 2.0
        def ineq_fun(x):
            h = x[:n]
            t = x[-1]
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            return t - corr
        cons = [{'type': 'eq', 'fun': eq_sum},
                {'type': 'ineq', 'fun': ineq_fun}]
        try:
            res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
            h_opt = res.x[:n]
            h_opt = project_onto_box_sum(h_opt, n / 2.0)
            t_opt = compute_c5(h_opt)
            if t_opt < best_score:
                best_score, best_h = t_opt, h_opt.copy()
        except Exception:
            pass

    # final safety projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
