def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # region j in [0, n-s-1]
            if n - s > 0:
                grad[:n - s] += 1.0 - h[s:]
            # region j in [s, n-1]
            grad[s:] -= h[:n - s]
        else:
            t = -s
            # region j >= t
            if n - t > 0:
                grad[t:] += 1.0 - h[:n - t]
            # region j <= n-1 - t
            grad[:n - t] -= h[t:]
    # average over number of shifts
    if len(shifts) > 0:
        grad = grad / len(shifts)
    return grad

Double-check sign: For positive shift s, region j >= s includes -h[j - s] contributed; we subtract h[:n - s] from grad[s:], which matches -h[j - s] term. For j < s but j <= n-s-1, only first term: add (1 - h[j + s]). That we have via grad[:n - s] += 1 - h[s:]; note that for j < s, still included because slice [:n-s] includes those indices; the expression (1 - h[j+s]) for j from 0 to n-s-1 is correct. For j >= s, grad term includes both contributions: first part adds (1 - h[j+s]) (still part of grad[:n-s] which covers up to n-s-1; for j>=s, but j may be > n-s-1, then the slice doesn't cover; so for j in [s, n-s-1], both contributions are included: first via grad[:n-s] (since j in that range) plus second via grad[s:] (since j>=s). The sum yields correct (1 - h[j+s]) - h[j - s].

Now for j in [n-s, n-1], only second part triggers (grad[s:] includes them) → -h[j - s] indeed because first slice only goes to n-s-1.

Thus gradient correct.

Expression derived earlier: grad_j = I[j >= t] * (1 - h[j - t]) - I[j <= n-1 - t] * h[j + t].

Implement: grad[t:] += 1 - h[:n - t]; grad[:n - t] -= h[t:]; which matches.

def gradient_descent_momentum(h0, time_end, step0=0.5, beta=0.9, max_iter=10**7):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # find shifts where corr near max
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def legendre_sequence(p):
    n = p - 1
    exp = (p - 1) // 2
    seq = np.empty(n, dtype=float)
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to new length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    exp = (p - 1) // 2
    seq = np.empty(n, dtype=float)
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            if n - s > 0:
                grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            t = -s
            if n - t > 0:
                grad[t:] += 1.0 - h[:n - t]
            grad[:n - t] -= h[t:]
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_end, step0=0.5, beta=0.9, max_iter=10**7):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score
