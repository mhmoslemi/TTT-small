def smooth_max_gradient(h, alpha=10.0):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # shift indices
    weights = np.exp(alpha * (corr - max_corr))
    weights_sum = np.sum(weights)
    if weights_sum == 0:
        weights = np.ones_like(weights) / weights.size
    else:
        weights /= weights_sum
    grad = np.zeros_like(h)
    n_shift = n - 1
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        shift = idx - n_shift
        if shift >= 0:
            term1 = np.empty_like(h)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty_like(h)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty_like(h)
            term1[s:] = h[:n - s]

def smooth_max_gradient(h, alpha=10.0):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # w now sums to 1
    # conv with h to compute weighted sum of shifted h
    conv_pos = np.convolve(h, w, mode='full')[n-1:n-1+n]
    conv_neg = np.convolve(h, w[::-1], mode='full')[n-1:n-1+n]
    grad = 1.0 - conv_pos - conv_neg
    return grad

def adam_optimize(h0, time_left, max_iter=2000, alpha=10.0, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        if time_left() <= 0.0:
            break
        grad = smooth_max_gradient(h, alpha=alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        # Possibly also clip to [0,1] already enforced by projection
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def smooth_max_gradient(h, alpha=12.0):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # weighted convolutions
    conv_pos = np.convolve(h, w, mode='full')[n-1:n-1+n]
    conv_neg = np.convolve(h, w[::-1], mode='full')[n-1:n-1+n]
    grad = 1.0 - conv_pos - conv_neg
    return grad

def adam_optimize(h0, time_left, max_iter=3000, alpha=12.0, lr=0.3,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break
        grad = smooth_max_gradient(h, alpha=alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_anneal(h_bin, time_left, rng, max_iters=400000,
                  T0=0.2, decay=0.99996, Tmin=1e-6):
    h = h_bin.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # --- Phase 0: small candidates ---
    n_small = 64
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.shape[0] != n_small:
            x_old = np.linspace(0.0, 2.0, init_full.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_small, endpoint=False)
            init_small = np.interp(x_new, x_old, init_full)
        else:
            init_small = init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small))
    candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    # --- Phase 1: SLSQP on small grid ---
    for cand in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 2: upscale to medium grid and Adam refine ---
    n_large = 256
    if best_h is None:
        best_h = project_onto_box_sum(rng.random(n_small), n_small / 2.0)
        best_score = compute_c5(best_h)

    x_small = np.linspace(0.0, 2.0, best_h.shape[0], endpoint=False)
    x_large = np.linspace(0.0, 2.0, n_large, endpoint=False)
    h_up = np.interp(x_large, x_small, best_h)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    sc_up = compute_c5(h_up)
    if sc_up < best_score:
        best_score, best_h = sc_up, h_up.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 0.5:

                                      alpha=14.0, lr=0.3)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Phase 3: binary conversion ---
    n = best_h.shape[0]
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # --- Phase 4: binary annealing (multiple runs if time permits) ---
    if time_left() > 0.5:
        for _ in range(3):
            if time_left() < 2.0:
                break
            h_ann, sc_ann = binary_anneal(best_h.copy(), time_left, rng,
                                          max_iters=400000, T0=0.2,
                                          decay=0.99996, Tmin=1e-6)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # --- Phase 5: final Adam refinement ---
    if time_left() > 0.5:
        h_ref2, sc_ref2 = adam_optimize(best_h, time_left, max_iter=2000,
                                        alpha=18.0, lr=0.25)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # final sanity projection (should already satisfy constraints)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
