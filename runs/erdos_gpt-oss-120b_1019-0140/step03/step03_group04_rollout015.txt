def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h shifted left by shift; term2 = h shifted right? as earlier
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)
```

    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    start = time.time()
    while time.time() < time_end:
        # compute temperature
        elapsed = time.time() - start
        total = max(time_end - start, 1e-8)
        frac = elapsed / total
        T = max(1e-6, (1 - frac) * 0.02)  # start temperature 0.02
        # choose random swap
        idx_one = np.flatnonzero(h > 0.5)
        idx_zero = np.flatnonzero(h <= 0.5)
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # early stop if best_score <= target
        if best_score <= 0.3808:
            break
    return best_h, best_score

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9,
                              time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(h0, time_end, rng):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    start = time.time()
    TARGET = 0.3808
    while time.time() < time_end:
        elapsed = time.time() - start
        total = max(time_end - start, 1e-12)
        frac = elapsed / total
        T = max(1e-6, (1.0 - frac) * 0.02)  # temperature schedule
        idx_one = np.flatnonzero(h > 0.5)
        idx_zero = np.flatnonzero(h <= 0.5)
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        if best_score <= TARGET:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # resolution
    n = 256
    # gather baseline candidates
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = resample(init_full, n)
        candidates.append(binarize_from_real(cand))
    candidates.append(thue_morse_binary(n))
    try:
        candidates.append(legendre_sequence(n + 1))  # n+1 = 257 (prime)
    except Exception:
        pass
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))
    best_h = None
    best_score = np.inf
    # hill‑climb each seed
    for idx, base in enumerate(candidates):
        if time_left() < 2.0:
            break
        remaining = len(candidates) - idx
        alloc = min(20.0, time_left() / remaining)
        h_opt, s_opt = hill_climb_binary(base, time.time() + alloc, rng)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt.copy()
        if best_score <= 0.3808:
            break
    # random restarts while time permits
    while time_left() > 5.0 and best_score > 0.3808:
        h0 = random_binary_balanced(n, rng)
        dur = min(15.0, time_left())
        h_opt, s_opt = hill_climb_binary(h0, time.time() + dur, rng)
        if s_opt < best_score:
            best_score, best_h = s_opt, h_opt.copy()
        if best_score <= 0.3808:
            break
    # final continuous refinement
    if time_left() > 5.0 and best_h is not None:
        t_end = time.time() + time_left() - 0.2
        h_ref, s_ref = gradient_descent_momentum(best_h,
                                                 step0=0.5,
                                                 beta=0.9,
                                                 time_end=t_end,
                                                 max_iter=200000)
        if s_ref < best_score:
            best_score, best_h = s_ref, h_ref.copy()
    if best_h is None:
        best_h = random_binary_balanced(n, rng)
        best_score = compute_c5(best_h)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
