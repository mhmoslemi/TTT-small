import time
import numpy as np

def compute_c5(h):
    """Maximum of h(x)*(1-h(x+k)) integral approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _min_overlap(h):
    """Minimum linear autocorrelation (excluding zero shift)."""
    n = h.shape[0]
    mid = n - 1
    O = np.correlate(h, h, mode="full")
    O[mid] = np.inf
    return np.min(O)

def greedy_improve(h, time_budget, rng):
    """Local search that repeatedly tries to raise the worst autocorrelation."""
    n = h.shape[0]
    mid = n - 1
    start = time.time()
    it = 0
    T0 = 5e-3  # initial temperature for occasional uphill moves
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[mid] = False

    while time.time() - start < time_budget and it < 20000:
        O = np.correlate(h, h, mode="full")
        O[mask]  # ensure mask applied later
        O[mid] = np.inf
        idx = np.argmin(O)               # worst shift index
        k = idx - mid                    # shift value (negative allowed)

        # indices that participate in this shift
        if k >= 0:
            i_vals = np.arange(n - k)
            partner = h[i_vals + k]
        else:
            ka = -k
            i_vals = np.arange(ka, n)
            partner = h[i_vals - ka]

        # choose i to increase (high partner) and j to decrease (low partner)
        inc_mask = h[i_vals] < 1.0 - 1e-12
        dec_mask = h[i_vals] > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            # fallback random transfer
            i, j = rng.integers(n, size=2)
            while i == j:
                j = rng.integers(n)
            max_delta = min(1.0 - h[i], h[j])
            if max_delta > 1e-12:

                h[i] += delta
                h[j] -= delta
            it += 1
            continue

        inc_idx = i_vals[inc_mask]

def optimise_for_n(n, time_budget, rng):
    """Search for a good step function with n samples."""
    M = n // 2
    best_h = None
    best_c5 = np.inf
    start = time.time()
