def transfer_search(h, target_sum, time_limit, init_eps=0.05, min_eps=1e-5, decay=0.9):
    n = len(h)
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    eps = init_eps
    while time.time() < time_limit and eps > min_eps:
        # attempt a number of trials per epsilon
        # We'll allow maybe n attempts per epsilon
        improved = False
        for _ in range(n):
            # randomly pick i with cur_h[i] > eps (so can subtract)
            # pick j with cur_h[j] < 1 - eps
            i = rng.integers(n)
            if cur_h[i] <= eps:
                continue
            j = rng.integers(n)
            if cur_h[j] >= 1.0 - eps:
                continue
            # transfer eps from i to j
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            c5_new = compute_c5(new_h)
            if c5_new < cur_c5 - 1e-12:  # improvement
                cur_h = new_h
                cur_c5 = c5_new
                improved = True
                break
        if not improved:
            eps *= decay
    return cur_h, cur_c5
