import time
import numpy as np

def compute_c5_fft(h, dx):
    """Maximum overlap integral using FFT (full correlation)."""
    n = h.shape[0]
    L = 2 * n - 1
    rev = np.flip(1.0 - h)
    fft_h = np.fft.fft(h, n=L)
    fft_rev = np.fft.fft(rev, n=L)
    conv = np.fft.ifft(fft_h * fft_rev).real
    return np.max(conv) * dx

def random_binary_vector(n, target, rng):
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, target, replace=False)
    v[idx] = 1.0
    return v

def repair_sum(v, target, rng):
    """Project v onto [0,1] and enforce exact sum==target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-9 and attempts < 2000:
        if diff > 0:
            cand = np.where(v < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        attempts += 1
    return v

def hill_climb_binary(start_h, dx, rng, max_swaps, deadline):
    """Swap a 1 and a 0 to reduce max overlap; stops early if time expires."""
    cur_h = start_h.copy()
    cur_val = compute_c5_fft(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    for _ in range(max_swaps):
        if time.time() > deadline - 0.5:
            break
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_c5_fft(cur_h, dx)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_val

def continuous_anneal(h, cur_val, rng, max_iters, dx, deadline):
    """Small mass transfers between points, accepting only improvements."""
    n = h.shape[0]
    cur_h = h.copy()
    best_h = cur_h.copy()
    best_val = cur_val
    cur_best = cur_val
    for _ in range(max_iters):
        if time.time() > deadline - 0.5:
            break
        pos = np.where(cur_h > 1e-12)[0]
        neg = np.where(cur_h < 1.0 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5_fft(cur_h, dx)
        if new_val < cur_best:
            cur_best = new_val
            cur_h[i] = cur_h[i]  # already updated
            cur_h[j] = cur_h[j]
            if new_val < best_val:

                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_val
