def project_to_simplex_with_bounds(v, target):
    n = len(v)
    # ensure target within [0, n]
    low_lambda = np.min(v) - 1.0
    high_lambda = np.max(v) + 1.0

def gradient_descent(h0, target, dx, deadline, rng, max_iter=2000):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = len(h)
    # predefine step factors
    step_factors = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    iter_count = 0
    while time.time() < deadline - 0.5 and iter_count < max_iter:
        # compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        idx_max = np.argmax(overlaps)
        k = idx_max - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        if k >= 0:
            i1 = np.arange(0, n - k)
            g[i1] += (1 - h[i1 + k])
            i2 = np.arange(k, n)
            g[i2] -= h[i2 - k]
        else:
            kk = -k
            i_main = np.arange(kk, n)
            g[i_main] += (1 - h[i_main - kk])
            i_sec = np.arange(0, n - kk)
            g[i_sec] -= h[i_sec + kk]

        improved = False
        # try each step factor
        for factor in step_factors:
            h_candidate = h - factor * g
            # project to feasible set
            h_candidate = project_to_simplex_with_bounds(h_candidate, target)
            cand_val = compute_c5(h_candidate)
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_h = h_candidate.copy()
                h = h_candidate
                improved = True
                break
        if not improved:
            # random perturbation

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_simplex_with_bounds(v, target, eps=1e-12, max_iter=60):
    """Project v onto {w | sum(w)=target, 0<=w<=1} using bisection on λ."""
    lo = np.min(v) - 1.0  # ensures all clipped to 1 => sum=n
    hi = np.max(v) + 1.0  # ensures all clipped to 0 => sum=0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit target exactly
    diff = target - w.sum()
    if abs(diff) > eps:
        # distribute diff greedily to non‑saturated entries
        free = np.where((w > eps) & (w < 1.0 - eps))[0]
        if free.size:
            inc = diff / free.size
            w[free] += inc
            w = np.clip(w, 0.0, 1.0)
    return w

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return project_to_simplex_with_bounds(v, target)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def gradient_descent(h_start, target, dx, deadline, rng,
                     max_iter=2000):
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.shape[0]
    step_factors = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    it = 0
    while time.time() < deadline - 0.5 and it < max_iter:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        idx_max = np.argmax(overlaps)
        k = idx_max - (n - 1)

        # gradient of the worst overlap
        g = np.zeros_like(h)
        if k >= 0:
            i1 = np.arange(0, n - k)
            g[i1] += (1.0 - h[i1 + k])
            i2 = np.arange(k, n)
            g[i2] -= h[i2 - k]
        else:
            kk = -k
            i_main = np.arange(kk, n)
            g[i_main] += (1.0 - h[i_main - kk])
            i_sec = np.arange(0, n - kk)
            g[i_sec] -= h[i_sec + kk]

        improved = False
        for f in step_factors:
            cand = h - f * g
            cand = project_to_simplex_with_bounds(cand, target)
            val = compute_c5(cand)
            if val < best_val - 1e-12:
                best_val = val
                best_h = cand.copy()
                h = cand
                improved = True
                break
        if not improved:
            src = np.where(h > 1e-9)[0]
            dst = np.where(h < 1.0 - 1e-9)[0]
            if src.size and dst.size:
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_simplex_with_bounds(h, target)
        it += 1
        if best_val < 0.3808:
            break
    return best_h, best_val

def simulated_annealing(h_start, cur_val, target, dx, deadline, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        src = np.where(h > 1e-8)[0]
        dst = np.where(h < 1.0 - 1e-8)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
            cur_val = new_val
            if best_val < 0.3808:
                break
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():

    h0 = project_to_simplex_with_bounds(h0, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    restart = 0
    while time.time() < deadline - 2.0:
        # choose start type
        if restart == 0 and "initial_h_values" in globals():
            h_start = h0.copy()
        else:
            if rng.random() < 0.3:
                h_start = random_binary_vector(n, target, rng)
            else:
                h_start = random_continuous_vector(n, target, rng)
        # gradient descent phase
        h_opt, val_opt = gradient_descent(h_start, target, dx, deadline, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break
        restart += 1

    # short annealing refinement if time left
    if time.time() < deadline - 1.0:
        best_h, best_val = simulated_annealing(best_h, best_val, target, dx, deadline, rng)

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_simplex_with_bounds(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), float(final_val), int(n)
