import time
import numpy as np
import cvxpy as cp

# --- helper functions ---------------------------------------------------------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def resample(arr, new_n):
    """Linear resampling of a 1‑D array to a different length."""
    m = arr.shape[0]
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad


def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

# --- main entry point ---------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # --- parameters -----------------------------------------------------------
    n_small = 64                     # size for exact MILP search
    target_ratio = 0.3808            # desired upper bound for C5
    # target maximal mismatch (integer) for n_small
    target_t = int(np.floor(target_ratio * n_small / 2.0))

    # start from simple feasible point (all 0.5)
    best_h = np.full(n_small, 0.5, dtype=float)
    best_h = project_onto_box_sum(best_h, n_small / 2.0)
    best_score = compute_c5(best_h)

    # --- exact binary optimisation (MILP) ------------------------------------
    if time_left() > 20.0:
        h_var = cp.Variable(n_small, boolean=True)
        t_var = cp.Variable(integer=True)
        constraints = [cp.sum(h_var) == n_small // 2,
                       t_var >= 0]

        # both positive and negative shifts
        for k in range(1, n_small):
            length = n_small - k
            # positive shift k
            hi = h_var[:length]
            hj = h_var[k:]
            w_pos = cp.Variable(length, boolean=True)
            constraints += [w_pos <= hi,
                            w_pos <= hj,
                            w_pos >= hi + hj - 1]
            mism_pos = cp.sum(hi) - cp.sum(w_pos)
            constraints.append(mism_pos <= t_var)

            # negative shift -k (swap the slices)
            hi_n = h_var[k:]
            hj_n = h_var[:length]
            w_neg = cp.Variable(length, boolean=True)
            constraints += [w_neg <= hi_n,
                            w_neg <= hj_n,
                            w_neg >= hi_n + hj_n - 1]
            mism_neg = cp.sum(hi_n) - cp.sum(w_neg)
            constraints.append(mism_neg <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver='CBC')
        except Exception:
            try:
                prob.solve(solver='GLPK_MI')
            except Exception:
                prob = None

        if prob is not None and prob.status in (cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE,
                                                cp.FEASIBLE):
            h_opt = np.rint(h_var.value).astype(float)
            # repair rounding errors if any
            cur_sum = h_opt.sum()
            needed = n_small // 2 - int(cur_sum)
            if needed > 0:
                zero_idx = np.where(h_opt == 0)[0]
                if len(zero_idx) >= needed:
                    h_opt[rng.choice(zero_idx, needed, replace=False)] = 1
            elif needed < 0:
                one_idx = np.where(h_opt == 1)[0]
                if len(one_idx) >= -needed:
                    h_opt[rng.choice(one_idx, -needed, replace=False)] = 0
            score = compute_c5(h_opt)
            if score < best_score:
                best_score, best_h = score, h_opt.copy()

    # --- move to a larger grid for fractional refinement -----------------------
    n_large = max(256, n_small * 4)
    if n_large % 2 == 1:
        n_large += 1

    # candidate from the best binary (or any) up‑sampled to large grid
    h_up = resample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    up_score = compute_c5(h_up)
    if up_score < best_score:
        best_score, best_h = up_score, h_up.copy()

    # also incorporate the parent construction if available
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_large:
            init_up = resample(init_full, n_large)
        else:
            init_up = init_full.copy()
        init_up = project_onto_box_sum(init_up, n_large / 2.0)
        init_score = compute_c5(init_up)
        if init_score < best_score:
            best_score, best_h = init_score, init_up.copy()

    # --- smooth gradient descent refinement ------------------------------------
    if time_left() > 5.0:
        # allocate iterations proportional to remaining time
        iters = int(min(8000, max(2000, time_left() * 400)))
        h_ref, sc_ref = smooth_grad_descent(best_h,
                                            n_iter=iters,
                                            step0=0.5,
                                            alpha_start=1.0,
                                            alpha_end=10.0,
                                            beta=0.9,
                                            time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # --- final projection and reporting ----------------------------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
