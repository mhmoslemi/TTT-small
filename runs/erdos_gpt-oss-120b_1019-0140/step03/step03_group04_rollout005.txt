def compute_subgradient(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # first part
            grad[:n-s] += 1 - h[s:]
            # second part
            grad[s:] += -h[:n-s]
        else:
            t = -s
            grad[t:] += 1 - h[:n-t]
            grad[:n-t] += -h[t:]
    return grad / len(shifts)

def gradient_descent_momentum(h0, time_end, max_iter=200000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < max_iter and time.time() < time_end:
        it += 1
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_corr = np.max(corr)
        # active shifts
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_subgradient(h, shifts)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        # step size schedule
        eta = step0 / np.sqrt(it)
        # gradient step
        h_new = h - eta * momentum
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

            grad[:n-t] += -h[t:]
    # Average over active shifts for subgradient (convex combination)
    if len(shifts) > 1:
        grad /= len(shifts)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial bracket for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets contain solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap C5 for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue (Legendre) binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_subgradient(h, shifts):
    """Subgradient of max‑correlation for given shift indices."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] += -h[:n - s]
        else:
            t = -s
            grad[t:] += 1.0 - h[:n - t]
            grad[:n - t] += -h[t:]
    if len(shifts) > 1:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_end, max_iter=200000, step0=0.5, beta=0.9):
    """Projected GD with momentum for the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    eps = 1e-12
    while it < max_iter and time.time() < time_end:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_subgradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
