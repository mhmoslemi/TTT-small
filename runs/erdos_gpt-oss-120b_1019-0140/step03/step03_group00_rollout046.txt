def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        h_initial = np.asarray(initial_h_values, dtype=float)
        n0 = h_initial.size
    else:
        n0 = 64
        h_initial = thue_morse(n0)

    h_initial = project_onto_box_sum(h_initial, n0 / 2.0)
    best_h = h_initial
    best_score = compute_c5(best_h)

    scales = [64, 128, 256]
    for n in scales:
        if time_left() <= 0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)

        # --- Differential Evolution ---
        target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n

        def de_cost(x):
            h = project_onto_box_sum(x, target_sum)
            return compute_c5(h)

        # simple time‑aware maxiter
        remaining = time_left()
        max_gen = int(min(40, max(5, remaining * 0.04)))  # allocate ~4% of remaining time
        if max_gen < 1:
            max_gen = 1

        popsize = 2
        pop = rng.random((popsize * n, n)).astype(float)
        pop[-1] = best_h.copy()

        res = opt.differential_evolution(
            de_cost,
            bounds,
            strategy='best1bin',
            maxiter=max_gen,
            popsize=popsize,
            init=pop,
            polish=False,
            tol=1e-6,
            seed=int(rng.integers(2**31 - 1)),
            callback=lambda xk, convergence: time_left() <= 0
        )
        h_de = project_onto_box_sum(res.x, target_sum)
        sc_de = compute_c5(h_de)
        if sc_de < best_score:
            best_score, best_h = sc_de, h_de.copy()

        # --- Local gradient refinement ---
        if time_left() > 1.0:
            # allocate iterations proportional to remaining time and problem size
            itr_per_sec = 1500.0 / n
            n_iter = int(min(25000, max(3000, time_left() * itr_per_sec)))
            best_h, best_score = gradient_descent_momentum_custom(
                best_h, n_iter=n_iter, step0=0.3, beta=0.9,
                rng=rng, time_left=time_left
            )

    # --- Random binary search and small transfers ---
    n_final = best_h.shape[0]
    while time_left() > 0.6:
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
            if best_score <= 0.3808:
                break
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.3)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= 0.3808:
                    break

    # --- Binary hill‑climbing (if binary) ---
    if np.all((best_h == 0) | (best_h == 1)):
        while time_left() > 0.5:
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = best_h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= 0.3808:
                    break

    # final projection and return
    best_h = project_onto_box_sum(best_h, best_h.size / 2)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
