We can implement that targeted swap approach: each iteration:

    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()

        corr = np.correlate(cur_h, 1 - cur_h, mode='full')

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    v = np.asarray(v)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(total - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def mismatched_pairs(h, shift):
    n = h.shape[0]
    if shift >= 0:
        i = np.arange(n - shift)
        hi = h[i]
        hj = h[i + shift]
    else:
        s = -shift
        i = np.arange(s, n)
        hi = h[i]
        hj = h[i - s]
    M = i[(hi == 1) & (hj == 0)]
    N = i[(hi == 0) & (hj == 1)]
    return M, N

def hill_climb_binary(start_h, max_moves, time_up):
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    move = 0
    while move < max_moves and not time_up():
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_corr = np.max(corr)
        shift = np.argmax(corr) - (n - 1)
        M, N = mismatched_pairs(cur_h, shift)
        if M.size > 0 and N.size > 0:
            i = rng.choice(M)
            j = rng.choice(N)
        else:
            i = rng.integers(n)
            while cur_h[i] < 0.5:
                i = rng.integers(n)
            j = rng.integers(n)
            while cur_h[j] > 0.5:
                j = rng.integers(n)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        move += 1
    return best_h, best_score

def slsqp_optimize(h0, max_iter=200):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0

    def corr_vec(h):
        return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - corr_vec(h)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_opt = res.x[:n]
    except Exception:
        h_opt = h0.copy()
    # project to satisfy constraints
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    return h_opt

def resample_to_n(arr, n):
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float)

def binary_from_continuous(v):
    n = v.shape[0]
    binary = np.zeros(n, dtype=float)
    idx = np.argpartition(v, -n//2)[-n//2:]
    binary[idx] = 1.0
    return binary

def run(seed=42, budget_s=1100, **kwargs):
    import time
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_up():
        return (time.time() - start_time) >= budget_s

    n = 256
    best_h = None
    best_score = np.inf

    # initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_res = resample_to_n(init_full, n)
        candidates.append(binary_from_continuous(init_res))
    candidates.append(thue_morse_seq(n))
    for _ in range(3):
        v = rng.random(n)
        candidates.append(binary_from_continuous(v))

    # Phase 1: continuous refinement via SLSQP on some candidates
    for cand in candidates:
        if time_up():
            break
        h_cont = slsqp_optimize(cand, max_iter=200)
        h_bin = binary_from_continuous(h_cont)
        # hill climb
        if time_up():
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate a chunk of moves, e.g., 20000 or based on time
        moves = int(remaining * 4000)  # about 4000 moves per second approx
        if moves < 1000:
            moves = 1000
        h_opt, sc_opt = hill_climb_binary(h_bin, moves, time_up)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # Phase 2: pure hill climbing from random starts
    while not time_up():
        # generate random binary start
        v = rng.random(n)
        start_h = binary_from_continuous(v)
        remaining = budget_s - (time.time() - start_time)
        moves = int(remaining * 4000)
        if moves < 2000:
            moves = 2000
        h_opt, sc_opt = hill_climb_binary(start_h, moves, time_up)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ensure feasibility
    if best_h is None:
        # fallback simple split
        best_h = np.zeros(n, dtype=float)
        best_h[:n//2] = 1.0
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(total - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def mismatched_pairs(h, shift):
    """Indices i where (h[i],h[i+shift]) is (1,0) and where it is (0,1)."""
    n = h.shape[0]
    if shift >= 0:
        i = np.arange(n - shift)
        hi = h[i]
        hj = h[i + shift]
    else:
        s = -shift
        i = np.arange(s, n)
        hi = h[i]
        hj = h[i - s]
    M = i[(hi == 1) & (hj == 0)]
    N = i[(hi == 0) & (hj == 1)]
    return M, N

def hill_climb_binary(start_h, max_moves, time_up, rng):
    """Targeted hill climbing on binary vectors."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    cur_score = np.max(corr) * (2.0 / n)
    best_h = cur_h.copy()
    best_score = cur_score
    moves = 0
    while moves < max_moves and not time_up():
        max_corr = np.max(corr)
        shift = np.argmax(corr) - (n - 1)
        M, N = mismatched_pairs(cur_h, shift)
        if M.size > 0 and N.size > 0:
            i = rng.choice(M)
            j = rng.choice(N)
        else:
            i = rng.integers(n)
            while cur_h[i] < 0.5:
                i = rng.integers(n)
            j = rng.integers(n)
            while cur_h[j] > 0.5:
                j = rng.integers(n)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        new_score = np.max(corr) * (2.0 / n)
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]   # revert, corr unchanged
        moves += 1
    return best_h, best_score

def slsqp_optimize(h0, max_iter=200):
    """SLSQP refinement of a continuous vector (box and sum constraints)."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def corr_vec(h):
        return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    def ineq(x):
        h = x[:n]
        t = x[-1]
        return t - corr_vec(h)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h_opt = res.x[:n]
    except Exception:
        h_opt = h0.copy()
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt

def resample_to_n(arr, n):
    """Linear interpolation from len(arr) to n points (periodic interval)."""
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_seq(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float)

def binary_from_continuous(v):
    n = v.shape[0]
    binary = np.zeros(n, dtype=float)
    idx = np.argpartition(v, -n // 2)[-n // 2:]
    binary[idx] = 1.0
    return binary
