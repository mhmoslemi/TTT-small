def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # compute lambda via bisection on shift
    lower = -np.max(v) - 10.0
    upper = hi - np.min(v) + 10.0
    for _ in range(50):
        mid = (lower + upper) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            lower = mid
        else:
            upper = mid
    return np.clip(v + (lower+upper)/2, lo, hi)

def smooth_grad(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_term = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_term)
    probs = exp_term / sum_exp   # shape (2n-1)
    grad = np.zeros_like(h)
    # loop over each shift index
    for idx, prob in enumerate(probs):
        if prob == 0:
            continue
        k = idx - (n - 1)
        if prob != 0:
            grad += prob * gradient_for_shift(h, k)
    # Since smooth objective c5_smooth = dx * (max_corr + (1/alpha)*log(sum_exp))
    # gradient is dx * sum_{k} probs * grad_k

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + 0.5 * (low + high), lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the overlap for shift k (without dx)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def smooth_grad(h, alpha):
    """Gradient of the smoothed max-overlap objective."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_term = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_term)
    probs = exp_term / sum_exp
    grad = np.zeros_like(h)
    for idx, p in enumerate(probs):
        if p == 0.0:
            continue
        k = idx - (n - 1)
        grad += p * gradient_for_shift(h, k)
    return grad * (2.0 / n)   # multiply by dx

def smooth_descent(h_start, target, deadline, rng):
    """Projected gradient descent on the smoothed objective."""
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 5.0
    step = 0.2
    min_step = 1e-6
    while time.time() < deadline:
        if alpha < 200.0:
            alpha = min(200.0, alpha * 1.02)
        grad = smooth_grad(h, alpha)
        cur_c5 = compute_c5(h)
        improved = False
        for _ in range(10):
            new_h = h - step * grad
            new_h = project_to_box_sum(np.clip(new_h, 0.0, 1.0), target)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-10:
                h = new_h
                cur_c5 = new_c5
                improved = True
                break
            step *= 0.5
            if step < min_step:
                break
        if not improved:
            h = h + rng.normal(scale=0.02, size=n)
            h = project_to_box_sum(np.clip(h, 0.0, 1.0), target)
            step = max(step, 0.01)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        if improved:
            step = min(step * 1.1, 0.5)
    return best_h, best_c5

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    """SLSQP objective: minimise the slack variable t."""
    return x[-1]
