def greedy_improve_step(h, rng):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    overl = corr * (2.0 / n)
    max_idx = np.argmax(overl)
    max_over = overl[max_idx]
    k = max_idx - (n - 1)
    # find potential swap
    if k >= 0:
        i_start = 0
        i_end = n - k
        h_i = h[i_start:i_end]
        h_j = h[i_start + k:i_end + k]
    else:
        kk = -k
        i_start = kk
        i_end = n
        h_i = h[i_start:i_end]
        h_j = h[i_start - kk:i_end - kk]
    # indices for bad, good
    bad_mask = (h_i > 0.5) & (h_j < 0.5)
    good_mask = (h_i < 0.5) & (h_j > 0.5)
    bad_idx = np.where(bad_mask)[0]
    good_idx = np.where(good_mask)[0]
    if bad_idx.size == 0 or good_idx.size == 0:
        return h, False, max_over
    i = i_start + rng.choice(bad_idx)
    j = i_start + rng.choice(good_idx)
    # swap
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]  # swap
    # Ensure sum unchanged
    return h_new, True, max_over

    start = time.time()
    no_improve_steps = 0
    # max iterations maybe large, break when time runs out or no improvement over many steps
    while time.time() - start < time_budget:

import time

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p ** 0.5)
    for i in range(3, r + 1, 2):
        if p % i == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq
