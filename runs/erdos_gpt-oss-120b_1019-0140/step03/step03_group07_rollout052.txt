def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ------------------- helpers ------------------- #
def compute_c5(h):
    """C5 = max overlap * dx."""
    n = h.shape[0]
    overlaps = np.correlate(h, 1 - h, mode='full')
    return overlaps.max() * (2.0 / n)

def legendre_sequence(p):
    """Binary Legendre sequence of length p (prime)."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def project_onto_bounds_sum(v, target_sum):
    """Project v onto [0,1]^n with exact sum = target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        shifted = np.clip(v + lam, 0.0, 1.0)
        s = shifted.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)

def gradient_smoothed_max(h, alpha):
    """Gradient of (1/alpha) log-sum-exp of overlaps."""
    n = len(h)
    overlaps = np.correlate(h, 1 - h, mode='full')
    max_ov = overlaps.max()
    exp_vals = np.exp(alpha * (overlaps - max_ov))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        weight = w[s + n - 1]
        if weight == 0.0:
            continue
        if s >= 0:
            i = np.arange(n - s)
            grad[i] += weight * (1 - h[i + s])
            i2 = np.arange(s, n)
            grad[i2] -= weight * h[i2 - s]
        else:
            s_abs = -s
            i = np.arange(s_abs, n)
            grad[i] += weight * (1 - h[i - s_abs])
            i2 = np.arange(0, n - s_abs)
            grad[i2] -= weight * h[i2 + s_abs]
    return grad

def gradient_descent_h(n, rng, time_budget, init_h=None):
    """Simple smoothed‑max gradient descent on h."""
    target_sum = n / 2.0
    if init_h is not None:
        h = np.array(init_h, dtype=float)
    else:
        h = rng.random(n)
    h = project_onto_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    # schedule parameters
    alpha_start, alpha_max, alpha_inc = 5.0, 30.0, 0.02
    eta0 = 0.5
    while time.time() - start < time_budget:
        alpha = min(alpha_max, alpha_start + it * alpha_inc)
        grad = gradient_smoothed_max(h, alpha)
        eta = eta0 / np.sqrt(it + 1)
        h = h - eta * grad
        h = project_onto_bounds_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        it += 1
    return best_h, best_c5

def slsqp_refine(h, n, time_budget):
    """Final SLSQP refinement of (h,t) minimising t."""
    start = time.time()
    M = n / 2.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def overlap_k(x, k):
        vec = x[:n]
        if k >= 0:
            return np.dot(vec[:n - k], 1.0 - vec[k:])
        else:
            kk = -k
            return np.dot(vec[kk:], 1.0 - vec[:n - kk])

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - M}]
    cons += [{'type': 'ineq',
              'fun': (lambda x, k=k: x[-1] - overlap_k(x, k))}
             for k in range(-n + 1, n) if k != 0]

    raw_max = np.correlate(h, 1 - h, mode='full').max()
    x0 = np.concatenate([h, [raw_max]])

    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 2000,
                                'ftol': 1e-12,
                                'disp': False})
    except Exception:
        return h
    if not res.success:
        return h
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_onto_bounds_sum(h_opt, M)
    return h_opt

def local_search(h, time_limit, rng):
    """Simple random binary/fractional moves."""
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        if rng.random() < 0.5:
            # binary swap
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = best_h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
        else:
            # fractional transfer
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h[i], best_h[j])
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
    # final projection to satisfy sum exactly
    best_h = project_onto_bounds_sum(best_h, target_sum)
    return best_h, best_c5

            if r == 0 and 'initial_h_values' in globals() and len(initial_h_values) == n:

                p = n - 1
                leg = legendre_sequence(p)

                shift = rng.integers(n)
                init = np.roll(h_leg, shift)
            else:

            # time for this restart

        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
