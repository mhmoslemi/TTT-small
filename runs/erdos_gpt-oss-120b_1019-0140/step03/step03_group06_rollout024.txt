    total = v.sum()

        return v
    # scaling approach
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:

            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    if ones > 0:
        idx = rng.choice(n, ones, replace=False)
        v[idx] = 1.0
    return v

def continuous_anneal_improved(h, cur_val, best_h, best_val, start_time, deadline, rng):
    """Aggressive simulated annealing focusing on the worst shift."""
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        i_local = np.argmax(prod)
        i = i_range[i_local]

        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0:
            break

        if k >= 0:
            partner = cand + k
            mask = partner < n
            factor = np.empty_like(cand, dtype=float)
            factor[mask] = 1.0 - h[partner[mask]]
            factor[~mask] = 0.0
        else:
            kk = -k
            partner = cand - kk
            mask = partner >= 0
            factor = np.empty_like(cand, dtype=float)
            factor[mask] = 1.0 - h[partner[mask]]
            factor[~mask] = 0.0

        min_factor = np.min(factor)
        eps = 1e-9
        good = np.where(factor <= min_factor + eps)[0]
        if len(good) > 0:
            j = cand[rng.choice(good)]
        else:
            j = cand[rng.choice(len(cand))]

        if i == j:
            other = np.where(cand != i)[0]
            if len(other) == 0:
                break
            j = cand[rng.choice(other)]

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                break

        delta = max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

            def sum_eq(x):
                return np.sum(x[:n]) - target
            def overlap_ineq(x, k):
                hv = x[:n]
                if k >= 0:
                    overl = np.dot(hv[: n - k], 1.0 - hv[k:])
                else:
                    kk = -k
                    overl = np.dot(hv[kk:], 1.0 - hv[: n - kk])

            constraints = [{"type": "eq", "fun": sum_eq}]

                    h_opt = enforce_sum(h_opt, target, rng)
                    val_opt = compute_c5(h_opt)
                    if val_opt < best_val:

                        best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 2.0 and best_n <= 120:

        for k in range(-n + 1, n):
            if k == 0:
                continue
            constraints.append({"type": "ineq", "fun": overlap_ineq, "args": (k,)})
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])
        try:
            res = minimize(
                fun=lambda x: x[-1],
                x0=x0,
                method="SLSQP",
