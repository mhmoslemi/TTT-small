import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f = lambda lam: np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse of length n as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of max-correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    target = 0.3808

    for n in scales:
        if time_left() <= 0 or best_score <= target:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2)
            best_score = compute_c5(best_h)

        # Random balanced binary sampling
        max_samples = int(min(20000, max(1000, time_left() * (2000.0 / n))))
        for _ in range(max_samples):
            if time_left() <= 0 or best_score <= target:
                break
            h_rand = random_binary_balanced(n, rng)
            sc = compute_c5(h_rand)
            if sc < best_score:
                best_score, best_h = sc, h_rand.copy()

        # Local gradient descent
        if time_left() > 2 and best_score > target:
            factor = 12000.0 / n
            est_iters = int(min(80000, max(8000, time_left() * factor)))
            best_h, best_score = gradient_descent_momentum_custom(
                best_h, n_iter=est_iters, step0=0.5, beta=0.9,
                rng=rng, time_left=time_left)

        # Small mass-transfer hill climbing
        if time_left() > 1 and best_score > target:
            for _ in range(2000):
                if time_left() <= 0 or best_score <= target:
                    break
                i, j = rng.choice(n, size=2, replace=False)
                max_delta = min(best_h[i], 1.0 - best_h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.uniform(0.02, 0.3)
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new = compute_c5(h_new)
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()

        # If we already beat the target, stop early
        if best_score <= target:
            break

    # Final refinement if time permits
    if time_left() > 2 and best_score > target:
        final_iters = int(min(200000, max(40000, time_left() * 150.0 / best_h.size)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=final_iters, step0=0.2, beta=0.9,
            rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
