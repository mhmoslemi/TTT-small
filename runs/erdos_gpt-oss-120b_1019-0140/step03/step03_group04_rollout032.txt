def compute_c5(h):
    n = h.size
    # using FFT with zero padding to length 2*n
    H = np.fft.rfft(h, n=2*n)
    B = np.fft.rfft(1.0 - h, n=2*n)
    corr = np.fft.irfft(H * np.conj(B))
    # correlation length 2n-1, there may be small imaginary errors; take real
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)
```

def compute_c5(h):
    n = h.size
    H = np.fft.rfft(h, n=2*n)
    B = np.fft.rfft(1.0 - h, n=2*n)
    corr = np.fft.irfft(H * np.conj(B), n=2*n)
    # length = 2*n
    # correlation values of length 2*n-1 are corr[:2*n-1]
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        # compute correlation and max term indices
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shift indices attaining max
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
```

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n=v.size
    if s<=lo*n: return np.full_like(v, lo)
    if s>=hi*n: return np.full_like(v, hi)
    lam_low=np.min(v)-hi
    lam_high=np.max(v)-lo
    def f(l): return np.clip(v-l, lo, hi).sum()
    if f(lam_low)<s: lam_low=np.min(v)-10*hi
    if f(lam_high)>s: lam_high=np.max(v)-10*lo
    for _ in range(max_iter):
        lam_mid=(lam_low+lam_high)*0.5
        fm=f(lam_mid)
        if abs(fm-s)<1e-12: break
        if fm>s: lam_low=lam_mid
        else: lam_high=lam_mid
    return np.clip(v-lam_mid, lo, hi)

def compute_c5(h):
    n=h.size
    corr=np.correlate(h, 1.0-h, mode='full')
    return np.max(corr)*(2.0/n)

def compute_c5_fft(h):
    n=h.size
    H=np.fft.rfft(h, n=2*n)
    B=np.fft.rfft(1.0-h, n=2*n)
    corr=np.fft.irfft(H*np.conj(B), n=2*n)
    return np.max(corr[:2*n-1])*(2.0/n)

def resample(arr, new_n):
    m=arr.shape[0]
    if m==new_n: return arr.copy()
    x_old=np.linspace(0.0,1.0,m,endpoint=False)
    x_new=np.linspace(0.0,1.0,new_n,endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1')&1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h=np.zeros(n, dtype=float)
    idx=rng.choice(n, size=n//2, replace=False)
    h[idx]=1.0
    return h

def binarize_from_real(arr):
    n=arr.size
    half=n//2
    idx=np.argpartition(-arr, half-1)[:half]
    binar=np.zeros(n, dtype=float)
    binar[idx]=1.0
    return binar

def legendre_sequence(p):
    n=p-1
    seq=np.zeros(n, dtype=float)
    exp=(p-1)//2
    for i in range(1,p):
        seq[i-1]=1.0 if pow(i,exp,p)==1 else 0.0
    return seq

def slsqp_optimize_candidate(h0, max_iter=300):
    n=h0.size
    t0=compute_c5(h0)
    x0=np.concatenate([h0, [t0]])
    bounds=[(0.0,1.0)]*n+[(0.0,2.0)]
    def eq_sum(x): return np.sum(x[:n])-n/2.0
    def ineq_fun(x):
        h=x[:n]
        t=x[-1]
        corr=np.correlate(h,1.0-h,mode='full')*(2.0/n)
        return t-np.max(corr)
    cons=[{'type':'eq','fun':eq_sum},{'type':'ineq','fun':ineq_fun}]
    try:
        res=opt.minimize(lambda x:x[-1], x0, method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt=project_onto_box_sum(res.x[:n], n/2.0)
        return h_opt, compute_c5(h_opt)
    h_opt=res.x[:n]
    h_opt=project_onto_box_sum(h_opt, n/2.0)
    return h_opt, compute_c5(h_opt)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h=h0.copy()
    n=h.size
    s=n/2.0
    best_h=h.copy()
    best_score=compute_c5(h)
    momentum=np.zeros_like(h)
    it=0
    while (time_end is None or time.time()<time_end) and it<max_iter:
        corr=np.correlate(h,1.0-h,mode='full')
        max_corr=np.max(corr)
        eps=1e-12
        idxes=np.where(corr>=max_corr-eps)[0]
        shifts=idxes-(n-1)
        grad=np.zeros_like(h)
        for shift in shifts:
            if shift>=0:
                term1=np.zeros_like(h); term2=np.zeros_like(h)
                term1[:n-shift]=h[shift:]
                term2[shift:]=h[:n-shift]
            else:
                sft=-shift
                term1=np.zeros_like(h); term2=np.zeros_like(h)
                term1[sft:]=h[:n-sft]
                term2[:n-sft]=h[sft:]
            grad+= (1.0-term1-term2)
        grad/=len(shifts)
        momentum=beta*momentum+(1.0-beta)*grad
        eta=step0/np.sqrt(it+1)
        h_new=h-eta*momentum
        h_new=project_onto_box_sum(h_new, s)
        score=compute_c5(h_new)
        if score<best_score:
            best_score=score
            best_h=h_new.copy()
        h=h_new
        it+=1
    return best_h, best_score

def local_search_binary(h0, deadline, rng):
    h=h0.copy()
    best_h=h.copy()
    best_score=compute_c5_fft(h)
    while time.time()<deadline:
        ones=np.where(h==1)[0]
        zeros=np.where(h==0)[0]
        if ones.size==0 or zeros.size==0: break
        i=rng.choice(ones)
        j=rng.choice(zeros)
        h[i],h[j]=0.0,1.0
        score=compute_c5_fft(h)
        if score<best_score:
            best_score=score
            best_h=h.copy()
        else:
            h[i],h[j]=1.0,0.0
    return best_h, best_score

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_fft(h):
    n = h.size
    H = np.fft.rfft(h, n=2 * n)
    B = np.fft.rfft(1.0 - h, n=2 * n)
    corr = np.fft.irfft(H * np.conj(B), n=2 * n)
    return np.max(corr[:2 * n - 1]) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def local_search_binary(h0, deadline, rng):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        score = compute_c5_fft(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return budget_s - (time.time() - start)

    n_small = 256
    n_large = 512
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small / 2.0)
        candidates.append(cand)

    candidates.append(thue_morse_binary(n_small))
    candidates.append(legendre_sequence(257))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))

    processed = []
    for c in candidates:
        if c.sum() != n_small / 2:
            c = binarize_from_real(c)
        c = project_onto_box_sum(c, n_small / 2.0)
        processed.append(c)

    best_h_small = None
    best_score_small = np.inf

    for c in processed:
        if time_left() < 6.0:
            break
        allot = min(10.0, time_left() - 2.0)
        if allot <= 0:
            break
        deadline = time.time() + allot
        h_opt, sc_opt = local_search_binary(c, deadline, rng)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()

    extra = 5
    for _ in range(extra):
        if time_left() < 10.0:
            break
        c = random_binary_balanced(n_small, rng)
        allot = min(6.0, time_left() - 4.0)
        if allot <= 0:
            break
        deadline = time.time() + allot
        h_opt, sc_opt = local_search_binary(c, deadline, rng)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_h_small = h_opt.copy()
        if best_score_small <= 0.3808:
            break

    if best_h_small is None:
        best_h_small = random_binary_balanced(n_small, rng)
        best_score_small = compute_c5_fft(best_h_small)

    h_large = resample(best_h_small, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    best_h_large = h_large
    best_score_large = compute_c5(best_h_large)

    if time_left() > 12.0:
        deadline = start + budget_s - 0.5
        max_iter = int(max(20000, time_left() * 800))
        h_ref, sc_ref = gradient_descent_momentum(best_h_large,
                                                  step0=0.5, beta=0.9,
                                                  time_end=deadline,
                                                  max_iter=max_iter)
        if sc_ref < best_score_large:
            best_score_large = sc_ref
            best_h_large = h_ref.copy()

    if time_left() > 5.0:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h_large, max_iter=300)
        if sc_opt < best_score_large:
            best_score_large = sc_opt
            best_h_large = h_opt.copy()

    if best_score_small < best_score_large:
        final_h = best_h_small
        final_n = n_small
        final_score = compute_c5(final_h)
    else:
        final_h = best_h_large
        final_n = n_large
        final_score = compute_c5(final_h)

    final_h = project_onto_box_sum(final_h, final_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, final_n
