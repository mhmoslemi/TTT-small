def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # target n choose maybe 512 (or 256 if speed). Let's try 512.
    n = 512

    # Precompute index maps for updates
    pre_idx = np.empty((n, n), dtype=np.int32)
    pre_ridx = np.empty((n, n), dtype=np.int32)
    for i in range(n):
        ks = np.arange(-i, n - i)
        pre_idx[i] = ks + (n - 1)
        pre_ridx[i] = i + ks

    # define function to compute corr from h via convolution (full)
    def compute_corr(h):
        # using FFT for speed
        a = h.astype(np.float64)
        b = (1 - h)[::-1].astype(np.float64)
        size = 1 << ((2*n - 1).bit_length())
        conv = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), n=size)
        conv = np.rint(conv[:2*n-1]).astype(np.int32)
        return conv

    # define function to compute C5 scoring
    def score_from_corr(corr):
        return corr.max() * (2.0 / n)

    # Helper to binarize a real vector
    def binarize(arr):
        half = n // 2
        idx = np.argpartition(-arr, half-1)[:half]
        binar = np.zeros_like(arr, dtype=np.int8)
        binar[idx] = 1
        return binar

    # Initialize best solution
    best_h = None
    best_corr = None
    best_score = np.inf

    # Try initial candidates
    init_candidates = []

    # If pre-existing initial_h_values
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # resample to n using interpolation to maintain shape
        # Use linear interpolation function similar to previous code's resample
        def resample(arr, new_n):
            m = arr.shape[0]
            if m == new_n:
                return arr.copy()
            x_old = np.linspace(0.0, 1.0, m, endpoint=False)
            x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
            return np.interp(x_new, x_old, arr)
        init_resampled = resample(init_full, n)
        init_bin = binarize(init_resampled)
        init_candidates.append(init_bin)

    # Legendre sequence for p=257 gives length 256, need to upsample to 512 via repeat each element?
    # we can generate legendre of length n (must be even). Use p = n+1? However Legendre length p-1. For n=512, we can generate Legendre for p=521 (prime?) 521 -1 = 520, close; need exactly 512, not possible. Use length 256 and upsample via repeat each element twice.
    def legendre_seq(p):
        n = p - 1
        seq = np.zeros(n, dtype=np.int8)
        exp = (p - 1) // 2
        for i in range(1, p):
            if pow(i, exp, p) == 1:
                seq[i - 1] = 1
        return seq
    leg256 = legendre_seq(257)  # length 256
    # upsample by 2x
    leg512 = np.repeat(leg256, 2)
    # ensure sum equals n/2? leg256 had exactly 128 ones (since balanced). After repeat, count = 256 ones (since each 1 repeats). That's double needed. We need exactly 256 ones, since n/2 = 256. Actually 256 ones is correct. So leg512 is balanced.
    init_candidates.append(leg512)

    # Thue-Morse binary sequence length n
    def thue_morse_binary(n):
        return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=np.int8, count=n)
    init_candidates.append(thue_morse_binary(n))

    # Add a few random candidates
    for _ in range(3):
        rand_vec = rng.random(n)
        init_candidates.append(binarize(rand_vec))

    # Evaluate init candidates via fast corr
    for cand in init_candidates:
        corr = compute_corr(cand)
        sc = score_from_corr(corr)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()
            best_corr = corr.copy()

    # Hill climbing / simulated annealing
    # Parameters
    target_score = 0.3808
    target_max = int(np.floor(target_score * n / 2 + 1e-9))  # as integer
    # target_max maybe 97

    # Basic hill climbing loop
    while time_left() > 5.0:
        # If already at target, break
        if best_corr.max() <= target_max:
            break
        # Start from current best as current h and corr
        h = best_h.copy()
        corr = best_corr.copy()
        cur_max = corr.max()
        # We'll attempt a number of swaps, accept only improvements
        # Define no_improve_counter for restart
        no_improve = 0
        max_no_improve = 20000  # after this restart
        while time_left() > 2.0 and no_improve < max_no_improve:
            # random swap
            # pick i where h[i]==1, j where h[j]==0
            i = rng.integers(n)
            while h[i] == 0:
                i = rng.integers(n)
            j = rng.integers(n)
            while h[j] == 1:
                j = rng.integers(n)
            # compute deltas
            delta_i = -(1 - h[pre_ridx[i]])  # vector length n
            # compute new right values for j
            h_right_j = h[pre_ridx[j]].copy()
            # mask for i and j
            mask_i = (pre_ridx[j] == i)
            mask_j = (pre_ridx[j] == j)
            h_right_j[mask_i] = 0
            h_right_j[mask_j] = 1
            delta_j = 1 - h_right_j

            # Apply updates
            corr[pre_idx[i]] += delta_i
            corr[pre_idx[j]] += delta_j
            # swap h bits
            h[i] = 0
            h[j] = 1

            new_max = corr.max()
            if new_max < cur_max:
                cur_max = new_max
                no_improve = 0
                # Update best if better
                if new_max < best_corr.max():
                    best_corr = corr.copy()
                    best_h = h.copy()
                    best_score = score_from_corr(best_corr)
                    # if reached target, break
                    if best_corr.max() <= target_max:
                        break
            else:
                # revert
                corr[pre_idx[i]] -= delta_i
                corr[pre_idx[j]] -= delta_j
                h[i] = 1
                h[j] = 0
                no_improve += 1
        # after inner loop, maybe we have not improved from best. Loop back to try new restart if time left.

    # After hill climbing, if still not at target but time left, perform gradient descent refinement on real-valued representation to possibly improve.
    # Convert best_h to float array
    h_float = best_h.astype(np.float64)
    # Perform a few iterations of projected gradient descent (like previous algorithm) with small step.
    # reuse functions from previous code: project_onto_box_sum, compute_c5.
    def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
        n = v.size
        if s <= lo * n:
            return np.full_like(v, lo)
        if s >= hi * n:
            return np.full_like(v, hi)
        lam_low = np.min(v) - hi
        lam_high = np.max(v) - lo
        for _ in range(max_iter):
            lam_mid = 0.5 * (lam_low + lam_high)
            proj = np.clip(v - lam_mid, lo, hi)
            total = proj.sum()
            if total > s:
                lam_low = lam_mid
            else:
                lam_high = lam_mid
            if abs(total - s) < 1e-12:
                break
        return proj

    def compute_c5_float(h):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    # Simple gradient descent with momentum for a few seconds
    time_end = start + budget_s - 0.5
    h_cont = h_float.copy()
    s = n / 2.0
    momentum = np.zeros_like(h_cont)
    # precompute shifts for gradient as earlier code compute_gradients_for_shifts using max shifts
    def compute_shifts_of_max(corr):
        max_val = np.max(corr)
        eps = 1e-12
        return np.where(corr >= max_val - eps)[0] - (n - 1)
    # gradient routine
    while time.time() < time_end:
        # compute correlation
        corr = np.correlate(h_cont, 1.0 - h_cont, mode='full')
        max_corr = np.max(corr)
        # get shifts achieving max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient similar to compute_gradients_for_shifts but for continuous h
        grad = np.zeros_like(h_cont)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] -= (1 - h_cont[shift:])   # derivative of h_i*(1 - h_{i+shift}) w.r.t h_i is (1 - h_{i+shift})
                grad[shift:] += h_cont[:n - shift]        # derivative w.r.t h_{i+shift} is -h_i
            else:
                sft = -shift
                grad[sft:] -= (1 - h_cont[:n - sft])
                grad[:n - sft] += h_cont[sft:]
        grad /= len(shifts)
        beta = 0.9
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = 0.5 / np.sqrt(np.maximum(1, np.sum(np.abs(momentum))))  # simple step scaling
        h_new = h_cont - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5_float(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
            best_corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        h_cont = h_new
    # final projection
    # ensure binary? Not necessary, but final step must satisfy constraints (0-1 and sum). We'll return best_h (maybe float) but constraints require 0<=h[i]<=1 and sum = n/2 within tolerance. Float acceptable. Return h values as list maybe.

    # Ensure result meets constraints via projection
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5_float(best_h)

    return best_h, best_score, n

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(total - s) < 1e-12:
            break
    return proj

def compute_corr_fft(h):
    """Linear correlation h * (1-h) using FFT; returns integer counts."""
    n = h.size
    a = h
    b = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), n=size)
    counts = np.rint(conv[:2 * n - 1]).astype(np.int32)
    return counts

def c5_from_counts(cnt, n):
    return cnt.max() * (2.0 / n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    """Thue–Morse binary sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=np.float64, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=np.float64)
    exp = (p - 1) // 2
    for i in range(1, p):
        if pow(i, exp, p) == 1:
            seq[i - 1] = 1.0
    return seq

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros_like(arr, dtype=np.float64)
    binar[idx] = 1.0
    return binar
