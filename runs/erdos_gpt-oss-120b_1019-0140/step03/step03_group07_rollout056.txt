def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)

def compute_corr_array(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def init_h_random_fractional(n, M, rng):

    h = h / np.sum(h) * M
    # Clip and adjust to keep sum = M
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum: simple loop
    total = np.sum(h)
    diff = M - total
    # If diff > 0: add to smallest entries.
    # If diff < 0: subtract from largest.
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 0:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff >= 0:
                    break
    return h

def optimise_n(n, time_alloc, rng):
    M = n // 2
    # initial h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        # ensure sum exactly M
        total = np.sum(h)
        diff = M - total
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        # compute c5
    else:
        # choose method randomly
        if rng.random() < 0.5:
            h = init_h_random_binary(n, M, rng)
        else:
            h = init_h_random_fractional(n, M, rng)
        # ensure sum exactly M
        total = np.sum(h)
        diff = M - total
        if abs(diff) > 1e-12:
            h += diff / n
            # clip again
            h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # optional: if n==80 use Legendre sequence as candidate
    if n == 80:
        # generate Legendre binary
        p = 79
        leg = np.zeros(p, dtype=float)
        for i in range(1, p):
            leg[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
        h_leg = np.concatenate([leg, [1.0]])
        # h_leg sum may be not M (maybe M=40)
        total_leg = np.sum(h_leg)
        if total_leg != M:
            h_leg = h_leg * (M / total_leg)  # scale but may exceed 1; ensure clip?
            h_leg = np.clip(h_leg, 0.0, 1.0)
            # adjust to sum M
            cur = np.sum(h_leg)
            if cur != M:
                h_leg += (M - cur) / n
                h_leg = np.clip(h_leg, 0.0, 1.0)
        c5_leg = compute_c5(h_leg)
        if c5_leg < best_c5:
            best_h = h_leg
            best_c5 = c5_leg

    start = time.time()
    # Simulated annealing parameters
    T0 = 0.01  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed > time_alloc - 0.5:
            break
        # linear temperature decay
        T = T0 * max(0.0, 1.0 - elapsed / time_alloc)
        # decide move type
        move_type = rng.integers(5)
        if move_type == 0:
            # binary swap: pick i with h approx 0 and j with h approx 1
            zeros = np.where(best_h < 0.5)[0]
            ones = np.where(best_h > 0.5)[0]
            if zeros.size > 0 and ones.size > 0:
                i = rng.choice(zeros)
                j = rng.choice(ones)
                h_new = best_h.copy()
                h_new[i] = 1.0
                h_new[j] = 0.0
            else:
                # fallback to random transfer
                i, j = rng.choice(n, 2, replace=False)
                max_transfer = min(1.0 - best_h[i], best_h[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer
                h_new = best_h.copy()
                h_new[i] += delta
                h_new[j] -= delta
        elif move_type == 1:
            # random transfer
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h[i], best_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        elif move_type == 2:
            # small perturbation at single index
            i = rng.integers(n)
            # perturb within bounds
            delta = (rng.random() - 0.5) * 0.2  # up to +/-0.1
            h_new = best_h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            # sum may change, adjust by small epsilon across all
            diff = np.sum(h_new) - M
            if abs(diff) > 1e-12:
                h_new -= diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
        elif move_type == 3:
            # gradient step towards reducing max overlap
            corr = compute_corr_array(best_h)
            k_idx = np.argmax(corr)
            k_shift = k_idx - (n - 1)
            # compute gradient
            shift = k_shift
            grad = np.zeros(n)
            if shift >= 0:
                grad[:n-shift] = (1.0 - best_h[shift:])  # term1
                grad[shift:] -= best_h[:n-shift]       # term2
            else:
                s = -shift
                grad[s:] = (1.0 - best_h[:n-s])
                grad[:n-s] -= best_h[s:]
            # step size proportional to T or fixed
            alpha = T * 0.5  # scale
            h_new = best_h - alpha * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            # adjust sum
            h_new = project_sum(h_new, M)
        else:
            # random restart: generate new candidate
            if rng.random() < 0.5:
                h_new = init_h_random_binary(n, M, rng)
            else:
                h_new = init_h_random_fractional(n, M, rng)
        # Ensure sum constraint exactly?
        total = np.sum(h_new)
        if abs(total - M) > 1e-9:
            # project to sum
            h_new = project_sum(h_new, M)
        c5_new = compute_c5(h_new)
        # Accept if better or with probability
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        else:
            # accept with probability exp(-(c5_new - best_c5)/T) if T>0
            if T > 0 and rng.random() < np.exp(-(c5_new - best_c5) / (T + 1e-12)):
                best_h, best_c5 = h_new, c5_new
        # early stop if good enough
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def refine_with_slsqp(h, n, M, rng):
    dx = 2.0 / n
    # pre-define constraints
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq','fun': sum_con}]
    constraints += [{'type':'ineq','fun': oc} for oc in overlap_cons]
    bounds = [(0.0,1.0)]*n + [(0.0, None)]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter':2000,'ftol':1e-12,'disp':False})
    except Exception:
        return h, compute_c5(h)
    if not res.success:
        return h, compute_c5(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

class SumConstraint:
    def __init__(self, n):
        self.n=n
        self.target=n/2.0
    def __call__(self,x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self,k,n):
        self.k=k
        self.n=n
        self.dx=2.0/n
    def __call__(self,x):
        h = x[:self.n]
        k = self.k
        n=self.n
        if k>=0:
            overl = np.dot(h[:n-k], 1.0 - h[k:]) * self.dx
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:n-kk]) * self.dx
        return x[-1] - overl

import time
import numpy as np
from scipy.optimize import minimize

# ----- constraints for SLSQP -----
class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overl = np.dot(h[:n - k], 1.0 - h[k:]) * self.dx
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:n - kk]) * self.dx
        return x[-1] - overl

# ----- basic utilities -----
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_array(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_sum(h, M):
    h = np.clip(h, 0.0, 1.0)
    diff = M - np.sum(h)
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        add = diff / np.sum(mask)
        h[mask] += add
        h = np.clip(h, 0.0, 1.0)
        diff = M - np.sum(h)
    return h

def init_h_random_binary(n, M, rng):
    h = np.zeros(n)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h

def init_h_random_fractional(n, M, rng):
    h = rng.random(n)
    h = h / np.sum(h) * M
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    diff = M - total
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
                if diff <= 0:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
                if diff >= 0:

    return h

            h = np.clip(h, 0.0, 1.0)

        else:

    best_h = h.copy()

# ----- local refinement with SLSQP -----
def refine_with_slsqp(h, n, M):
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h, compute_c5(h)
    if not res.success:
        return h, compute_c5(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
