import time

Define functions:

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)  # returns 0/1

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def hill_climb_optimize(h0, time_end, max_iters, rng):
    h = h0.copy()
    n = h.size
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    it = 0

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        improved = False
        if shift >= 0:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_from_real(arr):
    """Set exactly half of the entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def hill_climb_optimize(h0, time_end, max_iters, rng):
    """
    Targeted hill‑climbing: repeatedly reduce the maximal overlap
    by swapping a 'bad' (1,0) pair with a complementary (0,1) pair.
    """
    h = h0.copy()
    n = h.size
    score = compute_c5(h)
    best_h, best_score = h.copy(), score
    it = 0
    while it < max_iters and time.time() < time_end:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        improved = False

        if shift >= 0:
            seg_len = n - shift
            h_i = h[:seg_len]
            h_j = h[shift:]
            bad = np.where((h_i == 1) & (h_j == 0))[0]
            good = np.where((h_i == 0) & (h_j == 1))[0]
            if bad.size and good.size:
                i_bad = bad[rng.integers(bad.size)]
                i_good = good[rng.integers(good.size)]
                h[i_bad], h[i_good] = h[i_good], h[i_bad]
                new_score = compute_c5(h)
                if new_score < best_score:
                    best_score, best_h = new_score, h.copy()
                if new_score <= score:
                    score = new_score
                    improved = True
                else:
                    h[i_bad], h[i_good] = h[i_good], h[i_bad]
        else:
            s = -shift
            seg_len = n - s
            h_i = h[s:]
            h_j = h[:seg_len]
            bad = np.where((h_i == 1) & (h_j == 0))[0]
            good = np.where((h_i == 0) & (h_j == 1))[0]
            if bad.size and good.size:
                i_bad = bad[rng.integers(bad.size)] + s
                i_good = good[rng.integers(good.size)] + s
                h[i_bad], h[i_good] = h[i_good], h[i_bad]
                new_score = compute_c5(h)
                if new_score < best_score:
                    best_score, best_h = new_score, h.copy()
                if new_score <= score:
                    score = new_score
                    improved = True
                else:
                    h[i_bad], h[i_good] = h[i_good], h[i_bad]

        if not improved:
            # random swap of different bits
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
            if new_score <= score:
                score = new_score
            else:
                h[i], h[j] = h[j], h[i]

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512                     # resolution
    candidates = []

    # 1) parent construction, if any
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        cand = binarize_from_real(resample(init, n))
        candidates.append(cand)

    # 2) Thue–Morse
    candidates.append(thue_morse_binary(n))

    # 3) Legendre (length 256) up‑sampled
    leg = legendre_sequence(257)                # 256‑long 0/1 sequence
    leg_up = binarize_from_real(resample(leg, n))
    candidates.append(leg_up)

    # 4) Random balanced seeds
    for _ in range(5):
        candidates.append(random_balanced_binary(n, rng))

    # Evaluate initial pool
    best_h = None
    best_score = np.inf
    for h in candidates:
        if time_left() < 0.5:
            break
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()

    # Main refinement loop – hill climbing with time budget
    while time_left() > 5.0:
        remaining = time_left()
        max_iters = int(min(300000, max(50000, remaining * 6000)))
        time_end = time.time() + remaining - 0.2

        # Focus on the current best
        h_ref, sc_ref = hill_climb_optimize(best_h, time_end, max_iters, rng)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

        # Occasionally explore a fresh random start
        if remaining > 30 and rng.random() < 0.3:
            h_rand = random_balanced_binary(n, rng)
            h_ref2, sc_ref2 = hill_climb_optimize(
                h_rand, time_end, max_iters // 2, rng)
            if sc_ref2 < best_score:
                best_score, best_h = sc_ref2, h_ref2.copy()

        if best_score <= 0.3808:
            break

    # Ensure exact half‑filled binary vector (safety)
    if best_h.sum() != n / 2:
        best_h = binarize_from_real(best_h.astype(float))

    best_score = compute_c5(best_h)
    return best_h, best_score, n
