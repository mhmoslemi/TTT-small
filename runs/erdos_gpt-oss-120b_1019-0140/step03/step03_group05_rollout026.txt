import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low+high)*0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low+high)*0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def differential_evolution_optimize(h0, target_sum, budget_seconds, start_time):
    # set parameters
    n = h0.shape[0]
    # popsize: maybe 20 or based on n
    popsize = max(10, min(30, n // 10))
    rng = np.random.default_rng()
    # Initialize population
    pop = np.empty((popsize, n))
    pop[0] = h0.copy()
    for i in range(1, popsize):
        v = rng.random(n)
        v = project_to_box_sum(v, target_sum)
        pop[i] = v
    # compute fitness
    fitness = np.array([compute_c5(pop[i]) for i in range(popsize)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]
    # DE parameters
    F = 0.8
    CR = 0.9
    # loop
    safety_margin = 5.0
    while time.time() < start_time + budget_seconds - safety_margin:
        # one generation
        for i in range(popsize):
            # random indices a,b,c distinct from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # clip and project to constraints
            mutant = np.clip(mutant, 0.0, 1.0)
            trial = project_to_box_sum(mutant, target_sum)
            # crossover
            cross = np.empty_like(trial)
            jrand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    cross[j] = trial[j]
                else:
                    cross[j] = pop[i][j]
            cross = project_to_box_sum(np.clip(cross, 0.0, 1.0), target_sum)
            f = compute_c5(cross)
            if f < fitness[i]:
                pop[i] = cross
                fitness[i] = f
                if f < best_f:
                    best_f = f
                    best_h = cross.copy()
        # possible early stop
        if best_f <= 0.3808:
            break
    return best_h, best_f

def slsqp_optimize(h_start, t_start, target_sum, max_iter=500):
    # Use SciPy minimize with SLSQP; need constraints.
    n = h_start.shape[0]
    # Define constraints similar to previous code
    # SumConstraint and OverlapConstraint can be defined locally
    import numpy as np
    from scipy.optimize import minimize

    # Sum equality
    def sum_eq(x):
        return np.sum(x[:n]) - target_sum

    # For each k generate inequality
    overlap_cons = []
    for k in range(-n + 1, n):
        if k == 0:
            continue
        def make_ineq(k):
            def ineq(x):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                # t variable at end
                return x[-1] - overlap * (2.0 / n)
            return ineq
        overlap_cons.append({'type': 'ineq', 'fun': make_ineq(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': sum_eq}] + overlap_cons
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        lambda x: x[-1],
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    )
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, target_sum)
    return h_opt

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def differential_evolution_optimize(h0, target_sum, budget_seconds, start_time):
    """Simple DE with box+sum projection."""
    n = h0.shape[0]
    popsize = max(10, min(30, n // 10))
    rng = np.random.default_rng()
    pop = np.empty((popsize, n))
    pop[0] = h0.copy()
    for i in range(1, popsize):
        v = rng.random(n)
        pop[i] = project_to_box_sum(v, target_sum)
    fitness = np.array([compute_c5(pop[i]) for i in range(popsize)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]
    F = 0.8
    CR = 0.9
    safety = 5.0
    while time.time() < start_time + budget_seconds - safety:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            trial = project_to_box_sum(mutant, target_sum)
            # binomial crossover
            jrand = rng.integers(n)
            cross = np.empty_like(trial)
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    cross[j] = trial[j]
                else:
                    cross[j] = pop[i][j]
            cross = project_to_box_sum(np.clip(cross, 0.0, 1.0), target_sum)
            f = compute_c5(cross)
            if f < fitness[i]:
                pop[i] = cross
                fitness[i] = f
                if f < best_f:
                    best_f = f
                    best_h = cross.copy()
        if best_f <= 0.3808:
            break
    return best_h, best_f

def slsqp_optimize(h_start, t_start, target_sum, max_iter=300):
    """Refine with SLSQP on the max-overlap formulation."""
    n = h_start.shape[0]
    from scipy.optimize import minimize
    def sum_eq(x):
        return np.sum(x[:n]) - target_sum
    overlap_cons = []
    for k in range(-n + 1, n):
        if k == 0:
            continue
        def make_ineq(k):
            def ineq(x, k=k):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overlap * (2.0 / n)
            return ineq
        overlap_cons.append({'type': 'ineq', 'fun': make_ineq(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': sum_eq}] + overlap_cons
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        lambda x: x[-1],
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    )
    if not res.success:
        return None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, target_sum)
    return h_opt

                                                      start_time)
        if c5_de < best_c5:
            best_h, best_c5 = h_de, c5_de

    # Final gradient descent polish
    final_limit = start_time + budget_s - 5.0
    if time.time() < final_limit:
        h_fin, c5_fin = gradient_descent(best_h, target_sum,
                                         max_iter=3000,
                                         time_limit=final_limit)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    # Optional SLSQP refinement if time permits
    time_left = start_time + budget_s - time.time()
    if time_left > 10.0:
        t0 = compute_c5(best_h)
        h_opt = slsqp_optimize(best_h, t0, target_sum, max_iter=300)
        if h_opt is not None:
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
