def weighted_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # compute weights w_i = exp(beta*(corr_i - max_corr)), high weight on max
    # subtract max_corr for numerical stability.
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    # Now compute weighted sum of gradients.
    # To compute efficiently, we can loop over shifts.
    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        k = idx - (n - 1)
        # compute gradient for shift k, multiply by weight, accumulate.
        # Use same code as gradient_for_shift but add weight factor.
        # gradient_for_shift: for k>=0:
        # grad[:n - k] += (1 - h[k:]) * w
        # grad[k:] -= h[:n - k] * w
        # For k<0, similar.
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n - kk])
            grad[:n - kk] -= w * h[kk:]
    return grad

def gradient_descent_softmax(h_start, target_sum, max_iter, time_limit, beta=10.0):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = np.copy(h_start)
    best_h = np.copy(h)
    best_c5 = compute_c5(h)
    step = 0.3  # initial step size
    min_step = 1e-6

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute weighted gradient
        grad = weighted_gradient(h, beta)
        # step
        new_h = h - step * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        # if improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        # adapt step size
        if new_c5 < compute_c5(h):
            h = new_h
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, min_step)
            # optionally add noise if step too small
            if step <= min_step:
                noise = rng.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                step = 0.1
    return best_h, best_c5

def slsqp_optimize(h_start, t_start, n, target_sum, max_iter=500):
    sum_con = lambda x: np.sum(x[:n]) - target_sum
    # generate constraints list
    cons = [{'type': 'eq', 'fun': sum_con}]
    dx = 2.0 / n
    for k in range(-n+1, n):
        if k == 0: continue
        def make_ineq(k):
            def ineq(x, k=k):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overlap * dx
            return ineq
        cons.append({'type': 'ineq', 'fun': make_ineq(k)})
    bounds = [(0.0,1.0)]*n + [(0.0, None)]

    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(objective_t, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol':1e-9})
    return res

def weighted_gradient(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # compute weights
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        weights = np.ones_like(exp_vals) / exp_vals.size
    else:
        weights = exp_vals / sum_exp
    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n - kk])
            grad[:n - kk] -= w * h[kk:]
    return grad

def slsqp_optimize(h_start, t_start, n, target_sum, max_iter=600):
    from scipy.optimize import minimize
    dx = 2.0 / n
    def obj(x):
        return x[-1]
    def sum_con(x):
        return np.sum(x[:n]) - target_sum
    cons = [{'type': 'eq', 'fun': sum_con}]
    # create inequality constraints for each shift
    for k in range(-n + 1, n):
        if k == 0:
            continue
        def make_ineq(k):
            def ineq(x, k=k):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overlap * dx
            return ineq
        cons.append({'type': 'ineq', 'fun': make_ineq(k)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    return res

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Gradient (softmax weighted) ----------
def weighted_gradient(h, beta):
    """Gradient of a softmax approximation of the max-overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0.0:
        weights = np.full_like(exp_vals, 1.0 / exp_vals.size)
    else:
        weights = exp_vals / sum_exp
    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n - kk])
            grad[:n - kk] -= w * h[kk:]
    return grad

def gradient_descent_softmax(h_start, target_sum, max_iter, time_limit, beta, rng):
    """Projected gradient descent using the softmax gradient."""
    n = h_start.shape[0]
    h = np.copy(h_start)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.2
    min_step = 1e-7
    cur_c5 = best_c5
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        grad = weighted_gradient(h, beta)
        new_h = h - step * grad

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5

                cur_c5 = compute_c5(h)
                step = 0.1

def slsqp_optimize(h_start, t_start, n, target_sum, max_iter=600):
    """Minimise the slack variable t with SLSQP."""
    dx = 2.0 / n

    def obj(x):
        return x[-1]

    def sum_con(x):
        return np.sum(x[:n]) - target_sum

    cons = [{'type': 'eq', 'fun': sum_con}]
    for k in range(-n + 1, n):
        if k == 0:
            continue

        def make_ineq(k):
            def ineq(x, k=k):
                h = x[:n]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overlap * dx

            return ineq

        cons.append({'type': 'ineq', 'fun': make_ineq(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    return res

    # Determine dimensionality
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
        target_sum = n / 2.0
        init_h = project_to_box_sum(np.clip(init_h, 0.0, 1.0), target_sum)
    else:
        n = 120
        target_sum = n / 2.0
        init_h = None

    # Population initialisation
    pop_size = 12
    population = []
    if init_h is not None:
        population.append(init_h)
    while len(population) < pop_size:
        h_rand = rng.random(n)
        h_rand = project_to_box_sum(h_rand, target_sum)
        population.append(h_rand)

    scores = [compute_c5(h) for h in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    safety = 5.0
    # Evolutionary loop
    while time.time() < start_time + budget_s - safety:
        i1, i2 = rng.integers(0, pop_size, size=2)
        p1 = population[i1]
        p2 = population[i2]

        # Crossover
        child = (p1 + p2) * 0.5
        # Mutation
        child_mut = child + 0.05 * rng.standard_normal(n)

        child = project_to_box_sum(np.clip(child, 0.0, 1.0), target_sum)
        child_mut = project_to_box_sum(np.clip(child_mut, 0.0, 1.0), target_sum)

        child_score = compute_c5(child)
        child_mut_score = compute_c5(child_mut)

        if child_score < child_mut_score:
            best_child, best_child_score = child, child_score
        else:
            best_child, best_child_score = child_mut, child_mut_score

        worst_idx = int(np.argmax(scores))
        if best_child_score < scores[worst_idx]:
            population[worst_idx] = best_child
            scores[worst_idx] = best_child_score
            if best_child_score < best_c5:
                best_c5 = best_child_score
                best_h = best_child.copy()
                if best_c5 < 0.3808:
                    break

        # Local refinement on the current global best
        if time.time() < start_time + budget_s - safety - 30.0:
            refine_deadline = start_time + budget_s - safety
            max_refine = 12.0
            time_limit = min(refine_deadline, time.time() + max_refine)
            refined_h, refined_c5 = gradient_descent_softmax(
                best_h, target_sum, max_iter=5000, time_limit=time_limit,
                beta=10.0, rng=rng)

            if refined_c5 < best_c5:
                t0 = refined_c5
                res = slsqp_optimize(refined_h, t0, n, target_sum, max_iter=800)
                if res.success:
                    h_opt = np.clip(res.x[:n], 0.0, 1.0)
                    h_opt = project_to_box_sum(h_opt, target_sum)
                    c5_opt = compute_c5(h_opt)
                else:
                    h_opt = refined_h
                    c5_opt = refined_c5

                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    worst_idx = int(np.argmax(scores))
                    population[worst_idx] = h_opt
                    scores[worst_idx] = c5_opt
                    if best_c5 < 0.3808:
                        break

    # Final SLSQP polish if time permits
    if time.time() < start_time + budget_s - safety - 5.0:
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, n, target_sum, max_iter=800)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            best_c5 = compute_c5(h_opt)
            best_h = h_opt.copy()

    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
