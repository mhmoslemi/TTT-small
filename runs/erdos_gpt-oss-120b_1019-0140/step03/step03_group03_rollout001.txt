def anneal_binary_continuous(h, rng, time_left_func):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = list(np.flatnonzero(cur_h > 0.5))
    zeros = list(np.flatnonzero(cur_h < 0.5))
    it = 0
    # set schedule: total_iters target = 500000 perhaps
    max_iters = 500000
    while time_left_func() > 2.0 and it < max_iters:
        # temperature schedule
        t = 1.0 * (1e-5 / 1.0) ** (it / max_iters)
        # pick random i, j
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 1.0, 0.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            # update ones/zeros
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 0.0, 1.0
        it += 1
    return best_h, best_score

def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximation on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to size new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on max‑correlation."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary_continuous(h_start, rng, time_left_func):
    """Simulated annealing on a balanced binary vector."""
    cur_h = h_start.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = list(np.flatnonzero(cur_h > 0.5))
    zeros = list(np.flatnonzero(cur_h < 0.5))
    max_iters = 500000
    it = 0
    while it < max_iters and time_left_func() > 5.0:
        # temperature schedule (geometric)
        t = 1.0 * (1e-5 / 1.0) ** (it / max_iters)
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 1.0, 0.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            # update index lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 0.0, 1.0
        it += 1
    return best_h, best_score


def slsqp_refine(h, max_iter=200):
    """SLSQP refinement of a continuous vector."""
    n = h.size
    target = n / 2.0
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - target

    def ineq_fun(x):
        hh = x[:n]
        tt = x[-1]
        corr = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
        return tt - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def small_phase(rng, time_left):
    n = 256
    best_h = None
    best_score = np.inf

    # Gather initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = upsample(init, n)
        # binary version using top half entries
        idx = np.argsort(init)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        candidates.append(h_bin)

    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    # Evaluate candidates
    for h in candidates:
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()

    # Main iterative improvement
    while time_left() > 20.0:
        start = best_h.copy() if (best_h is not None and rng.random() < 0.7) \
                else random_binary_balanced(n, rng)

        h_ann, s_ann = anneal_binary_continuous(start, rng, time_left)
        if s_ann < best_score:
            best_score, best_h = s_ann, h_ann.copy()

        if time_left() > 5.0:
            max_iters = int(min(20000, time_left() * 1500))
            if max_iters > 0:
                h_gd, s_gd = gradient_descent_momentum(best_h, n_iter=max_iters,
                                                       step0=0.5, beta=0.9)
                if s_gd < best_score:
                    best_score, best_h = s_gd, h_gd.copy()
    return best_h, best_score


def large_phase(best_small, rng, time_left):
    n = 512
    # Upsample to larger resolution
    h_large = upsample(best_small, n)
    h_large = project_onto_box_sum(h_large, n / 2.0)
    best_h = h_large
    best_score = compute_c5(best_h)

    # Gradient descent refinement
    if time_left() > 5.0:
        max_iters = int(min(20000, time_left() * 1500))
        if max_iters > 0:
            h_gd, s_gd = gradient_descent_momentum(best_h, n_iter=max_iters,
                                                   step0=0.5, beta=0.9)
            if s_gd < best_score:
                best_h, best_score = h_gd, s_gd

    # Binary annealing on large grid (optional)
    if time_left() > 5.0:
        idx = np.argsort(best_h)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        h_ann, s_ann = anneal_binary_continuous(h_bin, rng, time_left)
        if s_ann < best_score:
            best_h, best_score = h_ann, s_ann

    # SLSQP local refinement (light)
    if time_left() > 5.0:
        h_slsqp, s_slsqp = slsqp_refine(best_h, max_iter=200)
        if s_slsqp < best_score:
            best_h, best_score = h_slsqp, s_slsqp

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Phase 1: medium resolution search
    best_small, score_small = small_phase(rng, time_left)

    # Phase 2: upscale and fine‑tune
    if time_left() > 5.0:
        best_large, score_large = large_phase(best_small, rng, time_left)
    else:
        best_large, score_large = best_small, score_small

    # Final projection and return
    n_final = best_large.size
    best_large = project_onto_box_sum(best_large, n_final / 2.0)
    final_score = compute_c5(best_large)

    return best_large, final_score, n_final
