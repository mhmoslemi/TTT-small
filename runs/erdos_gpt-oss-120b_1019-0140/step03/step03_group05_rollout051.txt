   h0 = np.clip(h0, 0.0, 1.0)

   best_h = h0.copy()

   # Prepare constraints for DE:
   bounds = [(0.0, 1.0)] * n

import time
import numpy as np
from scipy.optimize import differential_evolution, LinearConstraint

# ---------- Global timing for callbacks ----------
GLOBAL_START_TIME = None
GLOBAL_BUDGET = None

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing sum to target via bisection."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-8
    rng = np.random.default_rng()
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.02, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

    """Simple hill climbing on binary vectors derived from h_cont."""
    n = h_cont.shape[0]
    idx_sorted = np.argsort(-h_cont)  # descending
    binary = np.zeros(n, dtype=float)
    binary[idx_sorted[:int(target_sum)]] = 1.0
    best_h = binary
    best_c5 = compute_c5(best_h)
    ones = np.flatnonzero(best_h == 1)
    zeros = np.flatnonzero(best_h == 0)
    rng = np.random.default_rng()
    while time.time() < time_limit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            # update listings efficiently
            ones = np.flatnonzero(best_h == 1)
            zeros = np.flatnonzero(best_h == 0)
    return best_h, best_c5

# ---------- DE callback ----------
def de_callback(xk, convergence):
    """Stop DE a few seconds before the overall budget expires."""
    if GLOBAL_START_TIME is None:
        return False
    # keep a safety margin of 25 seconds
    return time.time() > GLOBAL_START_TIME + GLOBAL_BUDGET - 25.0

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    global GLOBAL_START_TIME, GLOBAL_BUDGET
    GLOBAL_START_TIME = time.time()
    GLOBAL_BUDGET = budget_s

    rng = np.random.default_rng(seed)

    # ---- Initialise ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = rng.random(80)
    n = h0.shape[0]
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ---- Differential Evolution ----
    bounds = [(0.0, 1.0)] * n
    lin_con = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])

    pop_factor = 5  # total pop = 5 * n
    total_pop = pop_factor * n
    init_pop = np.empty((total_pop, n))
    init_pop[0] = best_h
    for i in range(1, total_pop):
        rnd = rng.random(n)
        init_pop[i] = project_to_box_sum(rnd, target_sum)

    # large maxiter, real stopping will be via callback
    result = differential_evolution(
        compute_c5,
        bounds,
        constraints=(lin_con,),
        init=init_pop,
        maxiter=5000,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=de_callback,
        polish=False,
        updating='deferred',
        disp=False
    )
    de_h = np.clip(result.x, 0.0, 1.0)
    de_h = project_to_box_sum(de_h, target_sum)
    de_c5 = compute_c5(de_h)
    if de_c5 < best_c5:
        best_h, best_c5 = de_h, de_c5

    # ---- Gradient descent refinement (if time permits) ----
    time_left = GLOBAL_START_TIME + GLOBAL_BUDGET - time.time()
    if time_left > 15.0:
        gd_deadline = GLOBAL_START_TIME + GLOBAL_BUDGET - 5.0
        gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter=5000, time_limit=gd_deadline)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5

    # ---- Binary hill climbing refinement (if still time) ----
    time_left = GLOBAL_START_TIME + GLOBAL_BUDGET - time.time()
    if time_left > 5.0:
        bhc_deadline = GLOBAL_START_TIME + GLOBAL_BUDGET - 2.0
        bhc_h, bhc_c5 = binary_hill_climb(best_h, target_sum, bhc_deadline)
        if bhc_c5 < best_c5:
            best_h, best_c5 = bhc_h, bhc_c5

    # ---- Final projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
