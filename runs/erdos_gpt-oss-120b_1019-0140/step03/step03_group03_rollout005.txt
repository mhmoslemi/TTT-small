def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    h = h0.astype(float) # binary
    best_h = h.copy()
    best_score = compute_c5_ fft(h)
    cur_score = best_score

    for it in range(max_iter):
        # schedule
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # pick random swap:
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if cur_score < best_score:

                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

def gradient_descent_momentum(h0, max_iter, step0, beta, time_left_func):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left_func() < 0.5:
            break
        # compute correlation and get shifts with max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for these shifts
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto the box [lo,hi] with a prescribed sum s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    m = arr.shape[0]

    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

        fraction = elapsed / time_limit
        t = temp_start * ((temp_end / temp_start) ** fraction)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = delta < 0.0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
            # update index lists
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # resolution
    n = 1024

    # initial construction
    if "initial_h_values" in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init_full, n)
    else:
        h0 = random_binary_balanced(n, rng).astype(float)
    h0 = project_onto_box_sum(h0, n / 2.0)
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Phase 1: gradient descent (multiple restarts if time permits)
    gd_runs = 0
    while time_left() > 10.0 and gd_runs < 3:
        max_iter = int(min(150000, max(2000, time_left() * 2000)))
        h_gd, sc_gd = gradient_descent_momentum(
            best_h, max_iter, step0=0.5, beta=0.9, time_left_func=time_left
        )
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()
        # occasional random restart
        if time_left() > 5.0:
            h_rand = random_binary_balanced(n, rng).astype(float)
            h_rand = project_onto_box_sum(h_rand, n / 2.0)
            best_h, best_score = (h_rand, compute_c5(h_rand)) if compute_c5(h_rand) < best_score else (best_h, best_score)
        gd_runs += 1

    # Phase 2: binary rounding of the best continuous solution
    idx = np.argpartition(best_h, -n // 2)[-n // 2 :]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    bin_score = compute_c5(h_bin)
    if bin_score < best_score:
        best_score, best_h = bin_score, h_bin.copy()

    # Phase 3: annealing on binary vectors (multiple cycles)
    temp_start = 0.1
    temp_end = 1e-6
    while time_left() > 6.0:
        per_run = min(60.0, time_left() - 2.0)
        # choose start: best binary or fresh random
        if rng.random() < 0.5 and np.all(np.logical_or(best_h <= 0.5, best_h >= 0.5)):
            start_h = best_h.copy()
        else:
            start_h = random_binary_balanced(n, rng).astype(float)
        # ensure start is binary and balanced
        if not np.all(np.logical_or(start_h == 0.0, start_h == 1.0)):
            idx2 = np.argpartition(start_h, -n // 2)[-n // 2 :]
            start_h[:] = 0.0
            start_h[idx2] = 1.0
        h_ann, sc_ann = anneal_binary(start_h, rng, per_run, temp_start, temp_end)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
