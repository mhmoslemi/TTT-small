def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # v: 1d array
    # Adjust v by adding constant lambda then clipping to [lo,hi] such that sum = target.
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(v + (low + high) * 0.5, lo, hi)
    # due to numeric issues, adjust slight residual error:
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff equally among non-bound entries if any
        mask = (x > lo + 1e-9) & (x < hi - 1e-9)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, lo, hi)
    return x

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)  # shift
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                # random perturb
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    # ensure best_h is feasible
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):
    n = h_start.shape[0]
    bounds = [(0.0,1.0)]*n + [(0.0,None)]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(objective_t, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter':maxiter,'ftol':1e-9,'disp':False})
    return res

import time
import numpy as np
from scipy.optimize import minimize

# ----- Utility functions -----
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(v + (low + high) * 0.5, lo, hi)
    # tiny adjustment for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > lo + 1e-9) & (x < hi - 1e-9)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, lo, hi)
    return x

def gradient_for_shift(h, k):
    """Gradient of the worst overlap (shift k) w.r.t h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

# ----- Constraints for SLSQP -----
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ----- Main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Dimensionality and feasible start
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = project_to_box_sum(rng.random(n), target_sum)

    # Population initialization
    pop_size = 30
    population = np.empty((pop_size, n))
    population[0] = h0
    for i in range(1, pop_size):
        population[i] = project_to_box_sum(rng.random(n), target_sum)

    fitness = np.array([compute_c5(population[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Constraints for later SLSQP
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # DE parameters
    F = 0.8
    CR = 0.9
    safety_margin = 1.0
    target_c5 = 0.3808
    stop = False

    # Differential Evolution main loop
    while time.time() < start_time + budget_s - safety_margin and not stop:
        for i in range(pop_size):
            # choose 3 distinct indices different from i
            idx_pool = list(range(pop_size))
            idx_pool.pop(i)
            a, b, c = rng.choice(idx_pool, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            # binomial crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = project_to_box_sum(trial, target_sum)

            trial_c5 = compute_c5(trial)
            if trial_c5 < fitness[i]:
                population[i] = trial
                fitness[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    if best_c5 < target_c5:
                        stop = True
                        break
        if stop:
            break

    # Gradient descent refinement on the current best
    remaining = start_time + budget_s - safety_margin - time.time()
    if remaining > 0.5:
        gd_deadline = time.time() + remaining
        best_h, best_c5 = gradient_descent(best_h, target_sum, max_iter=5000, time_limit=gd_deadline)

    # Optional final SLSQP refinement if time permits
    if time.time() < start_time + budget_s - safety_margin - 5.0 and best_c5 >= target_c5:
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=600)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Final projection and evaluation
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
