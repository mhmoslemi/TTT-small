def compute_c5_bin(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # set middle term (k=0) to -inf to ignore
    corr[n-1] = -1
    max_corr = np.max(corr)
    return max_corr * dx

def hill_climb(h, deadline, rng, target_max_count):
    n = h.shape[0]
    dx = 2.0 / n
    # compute initial corr
    corr = np.correlate(h, 1 - h, mode='full')
    # maximum correlation value (int)
    max_corr = np.max(np.delete(corr, n-1))
    best_h = h.copy()
    best_corr = max_corr
    # loop while improving and time left
    while time.time() < deadline - 0.01:
        if max_corr <= target_max_count:
            break
        # identify worst shift(s) where corr is max
        # get all indices with corr == max_corr (excluding center)
        max_inds = np.where(corr == max_corr)[0]
        # exclude middle index
        max_inds = max_inds[max_inds != n-1]
        # pick one shift randomly
        idx = rng.choice(max_inds)
        k = idx - (n - 1)  # shift
        # find mismatched i's for this shift
        if k >= 0:
            # i in [0, n-1-k] where h[i]==1 and h[i+k]==0
            i_candidates = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
        else:
            kk = -k
            # i in [kk, n-1] where h[i]==1 and h[i-kk]==0  (since shift negative)
            # Equivalently i runs from kk to n-1 with condition h[i]==1 and h[i-kk]==0
            i_candidates = np.where((h[kk:] == 1) & (h[:n-kk] == 0))[0] + kk
        if i_candidates.size == 0:
            # no mismatched positions? Should not happen if max_corr>0
            break
        i = rng.choice(i_candidates)
        # find destination j where h[j]==0 and moving would not increase overlap for shift k
        # condition: either out of range for shift (meaning j+k >= n if k>=0, or j+k <0 if k<0) OR (h[j+k]==1)
        if k >= 0:
            # candidate j where h[j]==0 and (j+k >= n or h[j+k]==1)
            # compute mask
            mask = (h == 0)
            out_of_range = np.arange(n) + k >= n
            mask = mask & (out_of_range | (h[np.minimum(np.arange(n)+k, n-1)] == 1))
            # For out_of_range indices, we don't need to reference h beyond bounds; treat as true.
            # But we need to handle indexing for h[j+k] only where in range.
            # We'll construct using boolean array:
            j_candidates = np.where(mask)[0]
        else:
            # k < 0
            kk = -k
            mask = (h == 0)
            out_of_range = np.arange(n) - kk < 0
            # For indices where not out_of_range, need h[j - kk] == 1
            # compute safe index array
            idx_in_range = ~out_of_range
            # For indices in range, test condition
            cond = np.zeros(n, dtype=bool)
            cond[out_of_range] = True
            cond[idx_in_range] = (h[np.arange(n)[idx_in_range] - kk] == 1)
            mask = mask & cond
            j_candidates = np.where(mask)[0]
        if j_candidates.size == 0:
            # fallback: any zero position
            j_candidates = np.where(h == 0)[0]
            if j_candidates.size == 0:
                break
        j = rng.choice(j_candidates)
        # perform the swap
        h[i] = 0
        h[j] = 1
        # recompute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_new = np.max(np.delete(corr, n-1))
        # accept if not worse (or maybe accept equal)

def hill_climb(h, deadline, rng, target_max_count):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_corr = np.max(np.delete(np.correlate(h, 1 - h, mode='full'), n-1))
    while time.time() < deadline - 0.01:
        if best_corr <= target_max_count:
            break
        # compute corr
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(np.delete(corr, n-1))
        if max_corr != best_corr:
            # update best
            best_corr = max_corr
            best_h = h.copy()
        # find shift(s) with this max
        max_inds = np.where(corr == max_corr)[0]
        max_inds = max_inds[max_inds != n-1]
        # try attempts for improvements
        improved = False
        for attempt in range(20):
            idx = rng.choice(max_inds)
            k = idx - (n - 1)
            # find mismatched i list
            if k >= 0:
                i_cand = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
            else:
                kk = -k
                i_cand = np.where((h[kk:] == 1) & (h[:n - kk] == 0))[0] + kk
            if i_cand.size == 0:
                continue
            i = rng.choice(i_cand)
            # candidate j list as earlier
            if k >= 0:
                mask = (h == 0)
                out_of_range = np.arange(n) + k >= n
                # for in-range indices: need h[j+k]==1
                in_range_idx = np.where(~out_of_range)[0]
                mask[in_range_idx] = mask[in_range_idx] & (h[in_range_idx + k] == 1)
                # out_of_range already set as mask=True
                j_cand = np.where(mask)[0]
            else:
                kk = -k
                mask = (h == 0)
                out_of_range = np.arange(n) - kk < 0
                in_range_idx = np.where(~out_of_range)[0]
                mask[in_range_idx] = mask[in_range_idx] & (h[in_range_idx - kk] == 1)
                # out_of_range also allowed
                j_cand = np.where(mask)[0]
            if j_cand.size == 0:
                # fallback any zero
                j_cand = np.where(h == 0)[0]
            if j_cand.size == 0:
                continue
            j = rng.choice(j_cand)
            # test move
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = np.max(np.delete(new_corr, n-1))
            if new_max < max_corr:
                # accept
                improved = True
                # update best if better
                if new_max < best_corr:
                    best_corr = new_max
                    best_h = h.copy()
                # update corr for next attempt
                corr = new_corr
                max_corr = new_max
                # continue attempts inside loop
                break
            else:
                # revert
                h[i] = 1
                h[j] = 0
        if not improved:
            # no improvements found for this max_corr; break
            break
    return best_h, best_corr * dx

def hill_climb_binary(h, deadline, rng, target_max_count, max_iters=3000):
    n = h.shape[0]
    dx = 2.0 / n
    # compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(np.delete(corr, n-1))
    best_h = h.copy()
    best_corr = max_corr
    it = 0
    while time.time() < deadline - 0.01 and it < max_iters:
        it += 1
        if max_corr <= target_max_count:
            break
        # get indices where corr attains max (excluding center)
        max_inds = np.where(corr == max_corr)[0]
        max_inds = max_inds[max_inds != n-1]
        # choose shift
        idx = rng.choice(max_inds)
        k = idx - (n - 1)
        # find mismatched i
        if k >= 0:
            i_cand = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
        else:
            kk = -k
            i_cand = np.where((h[kk:] == 1) & (h[:n - kk] == 0))[0] + kk
        if i_cand.size == 0:
            # should not happen but break to avoid infinite loop
            break
        i = rng.choice(i_cand)
        # candidate j as earlier:
        if k >= 0:
            mask = (h == 0)
            out_of_range = np.arange(n) + k >= n
            # For in-range indices: need h[j+k]==1
            in_range = ~out_of_range
            mask[in_range] = mask[in_range] & (h[np.arange(n)[in_range] + k] == 1)
            # out_of_range already allowed
            j_cand = np.where(mask)[0]
        else:
            kk = -k
            mask = (h == 0)
            out_of_range = np.arange(n) - kk < 0
            in_range = ~out_of_range
            mask[in_range] = mask[in_range] & (h[np.arange(n)[in_range] - kk] == 1)
            j_cand = np.where(mask)[0]
        if j_cand.size == 0:
            # fallback any zero
            j_cand = np.where(h == 0)[0]
        if j_cand.size == 0:
            break
        j = rng.choice(j_cand)
        # perform swap
        h[i] = 0
        h[j] = 1
        # recompute corr for new h
        corr_new = np.correlate(h, 1 - h, mode='full')
        max_corr_new = np.max(np.delete(corr_new, n-1))
        if max_corr_new <= max_corr:
            # accept
            corr = corr_new
            max_corr = max_corr_new
            if max_corr_new < best_corr:
                best_corr = max_corr_new
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
            # Might try alternative move? For now continue to next iteration.
            # Could try other i or j later
    return best_h, best_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Choose n
    n = 200
    dx = 2.0 / n
    target_c5 = 0.3808
    target_max_count = int(np.floor(target_c5 / dx))

    # Initialize best solution
    overall_best_h = None
    overall_best_val = np.inf

    # If appropriate initial values exist and have same length, use them
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        if init_arr.shape[0] == n:
            h0 = (init_arr > 0.5).astype(int)
            # enforce exactly n/2 ones
            current_ones = int(h0.sum())
            target_ones = n // 2
            if current_ones != target_ones:
                diff = current_ones - target_ones
                if diff > 0:
                    idx = rng.choice(np.where(h0==1)[0], diff, replace=False)
                    h0[idx] = 0
                elif diff < 0:
                    idx = rng.choice(np.where(h0==0)[0], -diff, replace=False)
                    h0[idx] = 1
            overall_best_h = h0.copy()
            overall_best_val = compute_c5(overall_best_h)
    # If not initialized, create random balanced binary vector
    if overall_best_h is None:
        target_ones = n // 2
        h0 = np.zeros(n, dtype=int)
        h0[rng.choice(n, target_ones, replace=False)] = 1
        overall_best_h = h0.copy()
        overall_best_val = compute_c5(overall_best_h)

    # initial hill climb on the current best
    h_opt, val_opt = hill_climb_binary(overall_best_h.copy(), deadline, rng, target_max_count, max_iters=3000)
    if val_opt < overall_best_val:
        overall_best_h = h_opt.copy()
        overall_best_val = val_opt

    # main restarts
    while time.time() < deadline - 1.0:
        # random start
        h = np.zeros(n, dtype=int)
        h[rng.choice(n, n//2, replace=False)] = 1
        h_opt, val_opt = hill_climb_binary(h, deadline, rng, target_max_count, max_iters=2000)
        if val_opt < overall_best_val:
            overall_best_h = h_opt.copy()
            overall_best_val = val_opt
            if overall_best_val <= target_c5:
                break
        # Optionally include start from best with slight perturbation
        if time.time() < deadline - 1.0 and overall_best_val > target_c5:
            h = overall_best_h.copy()
            # small perturbation: swap a few pairs
            for _ in range(rng.integers(1,4)):
                ones_idx = np.where(h==1)[0]
                zeros_idx = np.where(h==0)[0]
                if len(ones_idx)==0 or len(zeros_idx)==0:

                i = rng.choice(ones_idx)

                h[i]=0
                h[j]=1

            if val_opt < overall_best_val:
                overall_best_h = h_opt.copy()
                overall_best_val = val_opt
                if overall_best_val <= target_c5:
                    break

    final_h = overall_best_h.astype(float)
    final_val = compute_c5(final_h)
    return final_h, final_val, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # ignore the zero shift (center element)
    max_corr = np.max(np.delete(corr, n - 1))
    return max_corr * dx

def hill_climb_binary(h, deadline, rng, target_max_count, max_iters=5000):
    """
    Greedy hill climbing on a binary vector.
    Swaps a 1 involved in the worst shift with a 0 that does not worsen that shift.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # initial correlation
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(np.delete(corr, n - 1))
    best_h = h.copy()
    best_corr = max_corr
    it = 0
    indices = np.arange(n)

    while time.time() < deadline - 0.01 and it < max_iters:
        it += 1
        if max_corr <= target_max_count:
            break

        # indices (in full correlation) where the current maximum occurs (excluding centre)
        max_inds = np.where(corr == max_corr)[0]
        max_inds = max_inds[max_inds != n - 1]
        if max_inds.size == 0:
            break

        # pick one worst shift
        idx = rng.choice(max_inds)
        k = idx - (n - 1)  # shift amount

        # find a mismatched position i that contributes to this shift
        if k >= 0:
            i_cand = np.where((h[: n - k] == 1) & (h[k:] == 0))[0]
        else:
            kk = -k
            i_cand = np.where((h[kk:] == 1) & (h[: n - kk] == 0))[0] + kk
        if i_cand.size == 0:
            break
        i = rng.choice(i_cand)

        # find a destination j (currently 0) that does not increase the overlap for this shift
        if k >= 0:
            mask = (h == 0)
            out_of_range = indices + k >= n
            in_range = ~out_of_range
            # for in‑range positions require h[j+k] == 1
            mask[in_range] &= (h[indices[in_range] + k] == 1)
            j_cand = np.where(mask)[0]
        else:
            kk = -k
            mask = (h == 0)
            out_of_range = indices - kk < 0
            in_range = ~out_of_range
            mask[in_range] &= (h[indices[in_range] - kk] == 1)
            j_cand = np.where(mask)[0]

        if j_cand.size == 0:
            # fallback: any zero entry
            j_cand = np.where(h == 0)[0]
        if j_cand.size == 0:
            break
        j = rng.choice(j_cand)

        # perform the swap
        h[i] = 0
        h[j] = 1

        # recompute correlation
        corr_new = np.correlate(h, 1 - h, mode="full")
        max_corr_new = np.max(np.delete(corr_new, n - 1))

        if max_corr_new <= max_corr:
            # accept move
            corr = corr_new
            max_corr = max_corr_new
            if max_corr_new < best_corr:
                best_corr = max_corr_new
                best_h = h.copy()
        else:
            # revert move
            h[i] = 1
            h[j] = 0

    return best_h, best_corr * dx

    target_max_count = int(np.floor(target_c5 / dx))

    final_h = overall_best_h.astype(float)

    return final_h, final_val, n
