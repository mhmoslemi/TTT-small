def compute_c5_and_shifts(h, tol=1e-12):

    # convert to shift values: shift = idx - (n - 1)
    shifts = shift_idxs - (len(h) - 1)
    return max_corr * (2.0 / len(h)), shifts

def subgradient(h, shifts):
    n = len(h)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # (1 - h[i+s]) part
            grad[:n - s] += (1.0 - h[s:])
            # -h[i-s] part
            grad[s:] += -h[:n - s]
        else:
            t = -s
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] += -h[t:]
    return grad / len(shifts) if shifts else grad

def adam_optimize(h0, max_iters, time_check_func, alpha=0.5):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5_and_shifts(h)[0]
    t = 0
    while t < max_iters and time_check_func() > 2.0:
        t += 1
        c5, shifts = compute_c5_and_shifts(h)
        grad = subgradient(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target_sum)
        # evaluate new h
        c5_new = compute_c5_and_shifts(h)[0]
        if c5_new < best_score:
            best_score = c5_new
            best_h = h.copy()
        # occasional binary evaluation for heuristic improvement
        if t % 20000 == 0:
            b = binary_from_continuous(h)
            c5_bin = compute_c5_and_shifts(b)[0]
            if c5_bin < best_score:
                best_score = c5_bin
                best_h = b.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    lam = (lam_low + lam_high) * 0.5
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_and_shifts(h, tol=1e-12):

    idxs = np.where(corr >= max_corr - tol)[0]
    shifts = idxs - (n - 1)

        else:
            t = -s
            grad[t:] += (1.0 - h[:n - t])

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
