def smooth_grad_descent_fft(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    h = np.copy(h0)
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr_fft(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def grad_softmax_corr_fft(h, alpha):
    n = h.shape[0]
    # compute correlation via conv
    corr = conv_fft(h, (1.0 - h)[::-1])
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    # split weights
    mid = n - 1
    w_pos = weights[mid:]  # length n
    w_neg_rev = weights[:mid]  # length n-1
    # construct w_neg of length n where w_neg[0]=0 and rest reversed
    w_neg = np.empty(n, dtype=h.dtype)
    w_neg[0] = 0.0
    w_neg[1:] = w_neg_rev[::-1]  # w_neg[p] = weight for shift -p
    # compute convs
    # A = conv(w_pos, (1-h)[::-1]) reversed
    A_full = conv_fft(w_pos, (1.0 - h)[::-1])
    A = A_full[n-1 - np.arange(n)]
    # B = -conv(w_pos, h) first n entries
    B_full = conv_fft(w_pos, h)
    B = -B_full[:n]
    # C = conv(w_neg, (1-h)) first n entries
    C_full = conv_fft(w_neg, (1.0 - h))
    C = C_full[:n]
    # D = -conv(w_neg, h[::-1]) reversed
    D_full = conv_fft(w_neg, h[::-1])
    D = -D_full[n-1 - np.arange(n)]
    grad = A + B + C + D
    return grad
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def conv_fft(a, b):
    """Convolution via real FFT (full mode)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    c = np.fft.irfft(A * B, size)
    return c[:n]


def compute_c5_fft(h):
    """C5 value using FFT‑based correlation."""
    n = h.shape[0]
    corr = conv_fft(h, (1.0 - h)[::-1])
    return np.max(corr) * (2.0 / n)


def grad_softmax_corr_fft(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t. h using FFT."""
    n = h.shape[0]
    # correlation and soft‑max weights
    corr = conv_fft(h, (1.0 - h)[::-1])
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)

    mid = n - 1
    w_pos = weights[mid:]                     # shift 0..n‑1
    w_neg_rev = weights[:mid]                 # shifts -(n‑1)..-1, reversed order

    w_neg = np.empty(n, dtype=h.dtype)
    w_neg[0] = 0.0
    w_neg[1:] = w_neg_rev[::-1]               # w_neg[p] = weight for shift -p

    # A = conv(w_pos, (1‑h)[::-1]) reversed
    A_full = conv_fft(w_pos, (1.0 - h)[::-1])
    A = A_full[n - 1 - np.arange(n)]

    # B = -conv(w_pos, h) (first n entries)
    B_full = conv_fft(w_pos, h)
    B = -B_full[:n]

    # C = conv(w_neg, (1‑h)) (first n entries)
    C_full = conv_fft(w_neg, (1.0 - h))
    C = C_full[:n]

    # D = -conv(w_neg, h[::-1]) reversed
    D_full = conv_fft(w_neg, h[::-1])
    D = -D_full[n - 1 - np.arange(n)]

    return A + B + C + D


def smooth_grad_descent_fft(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    """Projected gradient descent on soft‑max approximation with momentum."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr_fft(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_continuous(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill‑climb on balanced binary vectors."""
    h = h0.copy()
    curr_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(400000, max(60000, time_left() * 300)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        zero_idx = np.where(h == 0)[0]
        one_idx = np.where(h == 1)[0]
        if zero_idx.size == 0 or one_idx.size == 0:
            break
        i = rng.choice(zero_idx)
        j = rng.choice(one_idx)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5_fft(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    small_n = 128
    large_n = 512
    target_bound = 0.3808

    # --- small‑scale candidates ---
    candidates_small = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = downsample(init_full, small_n) if init_full.size != small_n else init_full.copy()
        candidates_small.append(project_onto_box_sum(init_small, small_n / 2.0))
    candidates_small.append(thue_morse(small_n).astype(float))
    for _ in range(2):
        candidates_small.append(random_binary_balanced(small_n, rng).astype(float))
    for _ in range(2):
        candidates_small.append(random_continuous(small_n, rng))

    best_small_h = None
    best_small_score = np.inf

    # --- optimize small candidates ---
    for h0 in candidates_small:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = smooth_grad_descent_fft(
            h0,
            n_iter=3000,
            step0=0.5,
            alpha_start=1.0,
            alpha_end=12.0,
            beta=0.9,
            time_left=time_left
        )
        if sc_opt < best_small_score:
            best_small_score = sc_opt
            best_small_h = h_opt.copy()

    if best_small_h is None:
        best_small_h = random_continuous(small_n, rng)
        best_small_h = project_onto_box_sum(best_small_h, small_n / 2.0)
        best_small_score = compute_c5_fft(best_small_h)

    # --- upsample to large resolution ---
    best_h = upsample(best_small_h, large_n)
    best_h = project_onto_box_sum(best_h, large_n / 2.0)
    best_score = compute_c5_fft(best_h)

    # --- large‑scale restarts ---
    candidates_large = [best_h]
    for _ in range(2):
        candidates_large.append(random_binary_balanced(large_n, rng).astype(float))
    for _ in range(2):
        candidates_large.append(random_continuous(large_n, rng))

    for h0 in candidates_large:
        if time_left() < 8.0:
            break
        max_iters = int(min(8000, max(3000, time_left() * 100)))
        h_opt, sc_opt = smooth_grad_descent_fft(
            h0,
            n_iter=max_iters,
            step0=0.5,
            alpha_start=2.0,
            alpha_end=20.0,
            beta=0.9,
            time_left=time_left
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
            if best_score <= target_bound:
                break

    # --- binary hill climbing ---
    if time_left() > 3.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        binary = project_onto_box_sum(binary, n / 2.0)
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=target_bound)
        if bin_sc < best_score:
            best_score = bin_sc
            best_h = bin_h.copy()

        # brief continuous refinement after binary search
        if time_left() > 2.0:
            h_ref, sc_ref = smooth_grad_descent_fft(
                best_h,
                n_iter=2000,
                step0=0.3,
                alpha_start=5.0,
                alpha_end=15.0,
                beta=0.9,
                time_left=time_left
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref.copy()

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5_fft(best_h)

    return best_h, final_score, best_h.size
