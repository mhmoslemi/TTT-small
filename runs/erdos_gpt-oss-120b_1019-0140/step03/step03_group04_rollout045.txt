def compute_c5(h):
    n = h.shape[0]
    # Use FFT convolution
    # convolution of h with rev(1-h)
    rev = (1.0 - h)[::-1]
    m = 2 * n - 1

    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)

def adam_optimize(h0, time_end, max_iter=100000):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m1 = np.zeros_like(h)
    m2 = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        shifts, _ = compute_max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)
        step = 0.5 / (np.sqrt(m2_hat) + eps)  # step size base
        h = h - step * m1_hat
        # Project onto box and sum
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial brackets for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated via FFT."""
    n = h.shape[0]
    m = 2 * n - 1
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)
    return conv.max() * (2.0 / n)

def compute_max_shifts(h, eps=1e-12):
    """Return shifts attaining the maximal correlation."""
    n = h.shape[0]
    m = 2 * n - 1
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    R = np.fft.rfft(rev, n=m)
    conv = np.fft.irfft(H * R, n=m)
    max_corr = conv.max()
    idxes = np.where(conv >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    return shifts

def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max-correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    """Random vector in [0,1] projected to exact sum n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def adam_optimize(h0, time_end, max_iter=10**7):
    """Adam optimizer with projection onto box-sum."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m1 = np.zeros_like(h)
    m2 = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        shifts = compute_max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)
        step = 0.5 / (np.sqrt(m2_hat) + eps)
        h = h - step * m1_hat
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    """Brief SLSQP refinement; returns feasible point and its score."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # Initial candidate at smallest resolution
    n_small = 128
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = resample(init_full, n_small)
    else:
        h0 = random_binary_balanced(n_small, rng)
    h0 = project_onto_box_sum(h0, n_small / 2.0)
    best_h = h0
    best_score = compute_c5(best_h)
    # Stages of increasing resolution
    stages = [128, 256, 512]
    for n_target in stages:
        if time_left() < 5.0:
            break
        # Upsample current best to new resolution
        h = resample(best_h, n_target)
        h = project_onto_box_sum(h, n_target / 2.0)
        # Optimize with Adam until time runs low
        time_end = start + budget_s - 0.4
        h_opt, sc_opt = adam_optimize(h, time_end)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # Short SLSQP polish if enough time remains
        if time_left() > 4.0:
            h_opt2, sc_opt2 = slsqp_optimize_candidate(best_h, max_iter=200)
            if sc_opt2 < best_score:
                best_score, best_h = sc_opt2, h_opt2.copy()
    # Final random restarts at highest resolution if still time
    while time_left() > 8.0 and best_score > 0.3808:
        n_cur = best_h.size
        h_rand = random_continuous_balanced(n_cur, rng)
        time_end = start + budget_s - 0.3
        h_opt, sc_opt = adam_optimize(h_rand, time_end, max_iter=200000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break
    # Ensure feasibility before return
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
