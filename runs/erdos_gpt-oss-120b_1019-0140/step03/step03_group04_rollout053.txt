def gd_momentum(h0, time_end, init_step=0.5, beta=0.9, max_iter=1000000):
    h = h0.copy()
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = init_step / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search on Lagrange multiplier
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure search interval brackets the target sum
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def resample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n (no wrap)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse_binary(n):
    """Return Thue–Morse binary vector (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones (n even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term for given list of shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h[shift:] aligns with h[:n-shift]
            term1 = np.empty_like(h)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty_like(h)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty_like(h)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty_like(h)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gd_momentum(h0, time_end, init_step=0.5, beta=0.9, max_iter=200000):
    """Projected gradient descent with momentum, stopped by wall‑clock."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time.time() >= time_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = init_step / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def local_pairwise_search(h0, time_end, rng):
    """Random pairwise mass transfers keeping sum constant."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    while time.time() < time_end:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # feasible transfer from j to i
        max_up = 1.0 - h[i]
        max_down = h[j]
        if max_up <= 0 or max_down <= 0:
            continue
        # transfer amount limited to 10% of the smaller bound
        delta_max = min(max_up, max_down) * 0.1
        delta = (rng.random() - 0.5) * 2 * delta_max
        if delta > 0:
            delta = min(delta, max_up, max_down)
        else:
            delta = -min(-delta, h[i], 1.0 - h[j])
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # bounds already respected by construction
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
    return best_h, best_score
