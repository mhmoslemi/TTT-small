def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap (no dx factor) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def differential_evolution_optimize(n, target_sum, rng, deadline):
    """Simple DE with projection onto the feasible hyperplane."""
    pop_size = max(20, n // 10)
    F = 0.8
    CR = 0.9
    population = [project_to_box_sum(rng.random(n), target_sum) for _ in range(pop_size)]
    fitness = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # choose three distinct indices different from i
            idxs = rng.choice([j for j in range(pop_size) if j != i], size=3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            cross = population[i].copy()
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(0, n)] = True
            cross[mask] = mutant[mask]
            cross = np.clip(cross, 0.0, 1.0)
            cross = project_to_box_sum(cross, target_sum)

            f_cross = compute_c5(cross)
            if f_cross < fitness[i]:
                population[i] = cross
                fitness[i] = f_cross
                if f_cross < best_c5:
                    best_c5 = f_cross
                    best_h = cross.copy()
                    if best_c5 < 0.3808:
                        return best_h, best_c5
        if time.time() >= deadline:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    # Phase 1: Differential Evolution (≈50% of time)

            best_h, best_c5 = gd_h, gd_c5

        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
        if best_c5 < 0.3808:

    # Final projection and evaluation

    return best_h, best_c5, n
