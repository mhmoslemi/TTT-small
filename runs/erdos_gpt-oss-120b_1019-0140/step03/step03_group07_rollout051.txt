def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def anneal_binary(n, rng, time_budget, init_h=None):
    M = n // 2
    # initial binary vector
    if init_h is not None:
        # threshold init_h at 0.5 then adjust sum to M
        h = (init_h >= 0.5).astype(float)
        cur = int(np.sum(h))
        if cur < M:
            # need to set some zeros to 1
            zeros = np.flatnonzero(h == 0)
            # random choose needed
            need = M - cur
            if need > len(zeros):
                need = len(zeros)
            chosen = rng.choice(zeros, size=need, replace=False)
            h[chosen] = 1.0
        elif cur > M:
            ones = np.flatnonzero(h == 1)
            need = cur - M
            if need > len(ones):
                need = len(ones)
            chosen = rng.choice(ones, size=need, replace=False)
            h[chosen] = 0.0
    else:
        # random binary with M ones
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, M, replace=False)] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # choose move type: 0=swap,1=transfer
        if rng.random() < 0.2:
            # transfer (fractional)
            # choose random i, j
            i, j = rng.choice(n, 2, replace=False)
            if h[i] > 0 and h[j] < 1.0:
                # transfer from i to j
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                continue
        else:
            # binary swap: need i where h[i]==1, j where h[j]==0
            # find indices
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                continue
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0

        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        # temperature schedule linearly decreasing from T0 to Tf
        # T0 choose maybe 0.02, Tf 1e-4
        elapsed = time.time() - start
        T0 = 0.02
        Tf = 1e-4
        progress = elapsed / time_budget
        T = T0 * (1 - progress) + Tf * progress
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_c5 = c5_new
            if c5_new + 1e-12 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                if best_c5 < 0.3808:
                    break

    return best_h, best_c5

def try_rolls(h, rng, num=20):
    best_h = h
    best_c5 = compute_c5(h)
    n = len(h)
    for _ in range(num):
        shift = rng.integers(n)
        h_shift = np.roll(h, shift)
        c5 = compute_c5(h_shift)
        if c5 < best_c5:
            best_h, best_c5 = h_shift, c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_global = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80,96,112,128,160,200]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates = [n0] + candidates
            else:
                # Move to front
                candidates = [n0] + [c for c in candidates if c != n0]

    for idx, n in enumerate(candidates):
        # compute remaining time
        elapsed = time.time() - start_global
        remaining = budget_s - elapsed
        if remaining < 20: # too little time left
            break
        # allocate fraction of remaining time
        # maybe allocate more to earlier but ensure some min
        # allocate = max(10, remaining / (len(candidates) - idx + 1))
        # but keep under budget
        # For simplicity, allocate each candidate 0.2 * remaining
        # and ensure we don't exceed.
        alloc = min(remaining * 0.25, 300.0) #cap
        # get initial h for this n if parent available else None
        init_h = None
        if 'initial_h_values' in globals():
            if len(initial_h_values) == n:
                init_h = np.array(initial_h_values, dtype=float)

        h_opt, c5_opt = anneal_binary(n, rng, alloc, init_h)

        # try roll improvements
        h_opt, c5_opt = try_rolls(h_opt, rng, num=10)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # final refinement: maybe a few more annealing steps on best_n
    if best_h is not None:
        remaining = budget_s - (time.time() - start_global)
        if remaining > 5:
            # run additional annealing with smaller temperature and fraction of time
            h_ref, c5_ref = anneal_binary(best_n, rng, remaining, best_h)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

    # fallback if none found
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # ensure constraints: clip and sum adjust
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    M = best_n // 2
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for step function h (discretized)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def try_rolls(h, rng, max_rolls=50):
    """Search cyclic shifts for a lower C5 bound."""
    n = h.shape[0]
    best_h = h
    best_c5 = compute_c5(h)
    if n <= 128:
        for shift in range(1, n):
            h_shift = np.roll(h, shift)
            c5 = compute_c5(h_shift)
            if c5 < best_c5:
                best_c5, best_h = c5, h_shift
    else:
        for _ in range(max_rolls):
            shift = rng.integers(n)
            h_shift = np.roll(h, shift)
            c5 = compute_c5(h_shift)
            if c5 < best_c5:
                best_c5, best_h = c5, h_shift
    return best_h, best_c5

def anneal_binary(n, rng, time_budget, init_h=None):
    """Simulated annealing on binary/fractional vectors with sum=n/2."""
    M = n // 2
    dx = 2.0 / n

    if init_h is not None:
        h = np.clip(init_h, 0.0, 1.0).astype(float)
        cur = np.sum(h)
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, M, replace=False)] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    start = time.time()
    it = 0
    # temperature schedule parameters
    T0, Tf = 0.02, 1e-4
    target = 0.3808

    while time.time() - start < time_budget:
        it += 1
        if rng.random() < 0.15:
            # fractional transfer move
            i, j = rng.choice(n, 2, replace=False)
            if h[i] > 0.0 and h[j] < 1.0:
                max_tr = min(h[i], 1.0 - h[j])
                delta = rng.random() * max_tr
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            elif h[j] > 0.0 and h[i] < 1.0:
                max_tr = min(h[j], 1.0 - h[i])
                delta = rng.random() * max_tr
                h_new = h.copy()
                h_new[j] -= delta
                h_new[i] += delta
            else:
                continue
        else:
            # simple swap of two entries
            i, j = rng.choice(n, 2, replace=False)
            if i == j:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]

        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        elapsed = time.time() - start
        prog = elapsed / time_budget
        T = T0 * (1 - prog) + Tf * prog
        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            h = h_new
            cur_c5 = c5_new
            if c5_new + 1e-12 < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
                if best_c5 < target:
                    break
    return best_h, best_c5

def slsqp_refine(n, h_start, rng, time_budget):
    """Refine a solution with SLSQP minimizing the max overlap bound."""
    M = n // 2
    dx = 2.0 / n
    # Equality sum constraint
    sum_con = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - M}
    # Inequality constraints for each shift k (excluding k=0)
    cons = [sum_con]

    def make_con(k):
        if k >= 0:
            def con(x):
                h = x[:n]
                return x[-1] - np.dot(h[:n - k], 1.0 - h[k:]) * dx
        else:
            kk = -k
            def con(x):
                h = x[:n]
                return x[-1] - np.dot(h[kk:], 1.0 - h[:n - kk]) * dx
        return con

    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({'type': 'ineq', 'fun': make_con(k)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # initial point
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    # limit iterations based on remaining time
    max_iter = 2000 if time_budget > 30 else 800

    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h_start, compute_c5(h_start)

    if not res.success:
        return h_start, compute_c5(h_start)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    start_global = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate resolutions (even numbers)
    candidates = [80, 96, 112, 128, 160, 200]
    # prioritize parent if available

    target = 0.3808

    for idx, n in enumerate(candidates):
        elapsed = time.time() - start_global
        remaining = budget_s - elapsed
        if remaining < 15.0:
            break
        # allocate a fraction of the remaining time
        alloc = min(remaining * 0.25, 300.0)

        init_h = None
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            init_h = np.array(initial_h_values, dtype=float)

        h_opt, c5_opt = anneal_binary(n, rng, alloc, init_h)
        h_opt, c5_opt = try_rolls(h_opt, rng, max_rolls=30)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < target:
                break

    # final refinement with SLSQP if time permits
    if best_h is not None:
        remaining = budget_s - (time.time() - start_global)
        if remaining > 10.0:

            if c5_ref < best_c5:

            # one more roll search after refinement
            best_h, best_c5 = try_rolls(best_h, rng, max_rolls=30)

        best_h = np.full(best_n, 0.5, dtype=float)

    best_h = np.clip(best_h, 0.0, 1.0)

        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
