def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2./n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.1
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        # update best
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # gradient for worst shift
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            # maybe add noise if alpha minimal
            if alpha <= min_alpha:
                noise = rng.normal(scale=0.01, size=n)
                new_h = project_to_box_sum(np.clip(h + noise, 0., 1.), target_sum)
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5:
                    h = new_h
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap term (without dx)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.1
    min_alpha = 1e-7
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = rng.normal(scale=0.01, size=n)
                trial = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                if compute_c5(trial) < cur_c5:
                    h = trial
    return best_h, best_c5

# ----- SLSQP refinement helpers -----
class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=300):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Determine dimension and initial feasible vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = project_to_box_sum(rng.random(n), target_sum)

    # ---- Population for Differential Evolution ----
    pop_size = 200 if n <= 80 else max(200, n * 2)
    pop = np.empty((pop_size, n))
    fitness = np.empty(pop_size)

    # seed first individual with h0
    pop[0] = h0
    fitness[0] = compute_c5(pop[0])
    for i in range(1, pop_size):
        pop[i] = project_to_box_sum(rng.random(n), target_sum)
        fitness[i] = compute_c5(pop[i])

    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]

    # ---- Time allocation ----
    safety_margin = 5.0
    de_time = budget_s * 0.60
    de_deadline = start_time + de_time
    # ---- Differential Evolution Loop ----
    while time.time() < de_deadline:
        for i in range(pop_size):
            # select three distinct individuals not i
            while True:
                a = rng.integers(pop_size)
                b = rng.integers(pop_size)
                c = rng.integers(pop_size)
                if len({i, a, b, c}) == 4:
                    break
            F = rng.uniform(0.5, 0.9)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            CR = rng.uniform(0.1, 0.9)
            trial = np.where(rng.random(n) < CR, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)

            f_trial = compute_c5(trial)
            if f_trial <= fitness[i] + 1e-12:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c5:
                    best_c5 = f_trial
                    best_h = trial.copy()
                    if best_c5 < 0.3808:
                        break
        if best_c5 < 0.3808:
            break

    # ---- Gradient descent refinement ----
    gd_deadline = start_time + budget_s - safety_margin
    gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter=20000,
                                 time_limit=gd_deadline, rng=rng)
    if gd_c5 < best_c5:
        best_h, best_c5 = gd_h, gd_c5

    # ---- Optional final SLSQP refinement if time permits ----
    remaining = start_time + budget_s - safety_margin - time.time()
    if remaining > 5.0:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=300)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
