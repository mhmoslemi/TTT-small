def hill_climb_binary(n, rng, time_left):
    # start with random balanced binary
    h = random_binary_balanced(n, rng)
    best_h = h.copy()
    best_score = compute_c5(h)
    # maintain indexes
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    # avoid endless loops if n small.
    max_without_improve = max(1000, n*10)
    it = 0
    last_improve = 0
    while time_left() > 0.5:
        if len(ones) == 0 or len(zeros) == 0:
            break

        i0 = rng.choice(zeros)
        # swap

        h_new[i0] = 1.0
        score = compute_c5(h_new)
        it += 1
        if score < best_score:

            last_improve = it
        if it - last_improve > max_without_improve:

            best_score = min(best_score, compute_c5(h))

                best_score = compute_c5(h)

    if m == new_n:

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    corr = None
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # Determine shifts with max corr (within tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # Add jitter occasionally
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # project
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    # initial t = compute_c5(h0)
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)] * n + [(0.0,1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return t - corr
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    h_opt = res.x[:n]

    t_opt = compute_c5(h_opt)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure interval contains solution
    if f(lam_low) < s:

            lam_low = lam_mid
        else:
            lam_high = lam_mid

    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:

            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue‑Morse binary sequence of length n as floats (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb_binary(n, rng, time_left):
    """Greedy hill‑climbing on balanced binary vectors."""
    h = random_binary_balanced(n, rng)
    best_h = h.copy()
    best_score = compute_c5(h)
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    max_without_improve = max(1000, n * 10)
    it = 0
    last_improve = 0

    while time_left() > 0.5:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        h_new = h.copy()
        h_new[i1] = 0.0
        h_new[i0] = 1.0
        score = compute_c5(h_new)
        it += 1
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            last_improve = it
        if it - last_improve > max_without_improve:
            h = random_binary_balanced(n, rng)
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            score_restart = compute_c5(h)
            if score_restart < best_score:
                best_score = score_restart
                best_h = h.copy()
                last_improve = it
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    """SLSQP refinement of a candidate (together with auxiliary variable)."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial vector (may be supplied)
    if 'initial_h_values' in globals():
        init_vec = np.asarray(initial_h_values, dtype=float)
    else:
        init_vec = random_binary_balanced(64, rng)

    scales = [128, 256, 512]
    best_h = None
    best_score = np.inf

    for n in scales:
        # prepare starting point at this scale
        if best_h is None:
            base = upsample(init_vec, n)
        else:
            base = upsample(best_h, n)
        base = project_onto_box_sum(base, n / 2.0)

        # binary hill‑climbing search
        if time_left() > 8:
            h_bin, sc_bin = hill_climb_binary(n, rng, time_left)
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin.copy()

        # gradient‑descent refinement
        if best_h is not None and time_left() > 5:
            max_iter = int(min(80000, max(20000, time_left() * 30)))
            h_ref, sc_ref = gradient_descent_momentum(best_h, max_iter,
                                                      step0=0.2, beta=0.9,
                                                      rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref.copy()

        # optional SLSQP refinement
        if best_h is not None and time_left() > 12:
            slsqp_iter = int(min(1500, max(200, time_left() * 2)))
            try:
                h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=slsqp_iter)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

        # ensure feasibility after this scale
        if best_h is not None:
            best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
            best_score = compute_c5(best_h)

        if best_score <= 0.3808:
            break

    # fallback in the unlikely case no candidate was found
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(512), 512 / 2.0)
        best_score = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
