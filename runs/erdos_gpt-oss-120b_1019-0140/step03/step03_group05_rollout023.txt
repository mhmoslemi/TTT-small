def gradient_descent(h_start, target_sum, max_iter, deadline):
    h = np.copy(h_start)
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    eps = 1e-12
    for i in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        cur_c5 = max_corr * dx
        if cur_c5 < best_c5 - eps:
            best_c5 = cur_c5
            best_h = h.copy()
        # gradient sum for worst indices
        max_idxs = np.where(corr >= max_corr - eps)[0]
        grad = np.zeros_like(h)
        for idx in max_idxs:
            k = idx - (n - 1)
            grad += gradient_for_shift(h, k)
        # line search steps
        step = alpha
        improved = False
        for ls in range(10):
            new_h = h - step * grad
            new_h = project_to_box_sum(new_h, target_sum)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-12:

                alpha = min(step * 1.5, 2.0)

            # small random move
            noise = np.random.normal(scale=alpha, size=n)
            h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

def project_to_box_sum(v, target, lo=0., hi=1.):
    # Clip v to lo, hi but shift to satisfy sum
    # Solve for lambda such that sum(clipped(v+lambda)) = target.
    # Use binary search.
    low = lo - np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high)/2.0, lo, hi)
