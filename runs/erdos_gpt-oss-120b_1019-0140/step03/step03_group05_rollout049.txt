def smoothed_gradient(h, beta):
    n = h.shape[0]
    o = np.correlate(h, 1 - h, mode='full')
    # compute probabilities safely
    scaled = beta * o
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    probs = exp_scaled / np.sum(exp_scaled)
    grad = np.zeros_like(h)
    for idx, p in enumerate(probs):
        if p == 0:
            continue
        k = idx - (n - 1)
        grad += p * gradient_for_shift(h, k)
    return grad
```

    n = h.shape[0]

    best_h = h.copy()

    alpha_inc = 1.03
    alpha_dec = 0.7
    beta_inc = 1.02

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def smoothed_gradient(h, beta):
    n = h.shape[0]
    o = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * o
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    probs = exp_scaled / exp_scaled.sum()
    grad = np.zeros_like(h)
    for idx, p in enumerate(probs):
        if p == 0.0:
            continue
        k = idx - (n - 1)
        grad += p * gradient_for_shift(h, k)
    return grad

def gradient_descent_smooth(h_start, target_sum, max_iters, deadline):
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = 0.2
    beta = 0.5
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        grad = smoothed_gradient(h, beta)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5
            alpha = min(alpha * 1.03, 5.0)
        else:
            alpha = max(alpha * 0.7, 1e-6)
            if alpha <= 1e-6:
                h = project_to_box_sum(h + np.random.normal(scale=0.02, size=h.shape[0]), target_sum)
                cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        beta = min(beta * 1.02, 200.0)
    return best_h, best_c5

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h, t0, sum_con, overlap_cons, maxiter=200):
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{'type': 'eq', 'fun': sum_con}]
    cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h, [t0]])
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

    safety = 5.0

        if gd_c5 < best_c5:

                if c5_opt < best_c5:
