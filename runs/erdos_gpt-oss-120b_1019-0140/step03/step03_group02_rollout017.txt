def binary_hill_climb_sa(h0, rng, max_iters, target, time_left, T0=0.01, Tmin=1e-6):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    it = 0
    # precompute indices of zeros and ones for fast selection
    while it < max_iters and time_left() > 0.01:
        # recompute zero and one indices each iteration (fast)
        zero_idx = np.flatnonzero(h == 0)
        one_idx = np.flatnonzero(h == 1)
        if zero_idx.size == 0 or one_idx.size == 0:
            break
        i = rng.choice(zero_idx)
        j = rng.choice(one_idx)
        # swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        # temperature schedule: simple linear decay based on iteration fraction
        T = max(Tmin, T0 * (1.0 - it / max_iters))
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
                if target is not None and best_score <= target:
                    break
        it += 1
    return best_h, best_score

    s = n / 2.0

    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the bracketing interval really contains the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    if len(shifts):
        grad /= len(shifts)
    return grad


def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb_sa(h0, rng, max_iters, target, time_left, T0=0.05, Tmin=1e-6):
    """Simulated‑annealing style hill climbing on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    it = 0
    while it < max_iters and time_left() > 0.01:
        zero_idx = np.flatnonzero(h == 0)
        one_idx = np.flatnonzero(h == 1)
        if zero_idx.size == 0 or one_idx.size == 0:
            break
        i = rng.choice(zero_idx)
        j = rng.choice(one_idx)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        # linear temperature decay
        T = max(Tmin, T0 * (1.0 - it / max_iters))
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if target is not None and best_score <= target:
                    break
        it += 1
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    n_small, n_large = 64, 256
    target_bound = 0.3808

    # ---------- Small‑scale search ----------
    candidates_small = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = downsample(init_full, n_small) if init_full.size != n_small else init_full.copy()
        # enforce binary balance
        order = np.argsort(init_small)
        binary = np.zeros_like(init_small)
        binary[order[n_small // 2:]] = 1.0
        candidates_small.append(binary)
    else:
        candidates_small.append(thue_morse(n_small).astype(float))

    for _ in range(4):
        candidates_small.append(random_binary_balanced(n_small, rng).astype(float))

    best_h_small = None
    best_score_small = np.inf

    for h0 in candidates_small:
        if time_left() < 5.0:
            break
        max_iters_small = int(min(50000, max(1, time_left() * 2000)))
        h_opt, sc_opt = binary_hill_climb_sa(
            h0, rng, max_iters=max_iters_small,
            target=target_bound, time_left=time_left,
            T0=0.05, Tmin=1e-6)
        if sc_opt < best_score_small:
            best_h_small, best_score_small = h_opt.copy(), sc_opt
            if best_score_small <= target_bound:
                break

    if best_h_small is None:
        best_h_small = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score_small = compute_c5(best_h_small)

    # ---------- Upsample to large resolution ----------
    h_large_initial = upsample(best_h_small, n_large)
    order = np.argsort(h_large_initial)
    binary_large_initial = np.zeros_like(h_large_initial)
    binary_large_initial[order[n_large // 2:]] = 1.0

    # ---------- Large‑scale stochastic search ----------
    best_h_large = None
    best_score_large = np.inf

    # start with the upsampled candidate and a couple of random seeds
    large_candidates = [binary_large_initial]
    for _ in range(2):
        large_candidates.append(random_binary_balanced(n_large, rng).astype(float))

    for h0 in large_candidates:
        if time_left() < 10.0:
            break
        max_iters = int(min(20000, max(1, time_left() * 500)))
        h_opt, sc_opt = binary_hill_climb_sa(
            h0, rng, max_iters=max_iters,
            target=target_bound, time_left=time_left,
            T0=0.05, Tmin=1e-6)
        if sc_opt < best_score_large:
            best_h_large, best_score_large = h_opt.copy(), sc_opt
            if best_score_large <= target_bound:
                break

    # continue generating random starts until time runs low
    while time_left() > 15.0:
        h0 = random_binary_balanced(n_large, rng).astype(float)
        max_iters = int(min(10000, max(1, time_left() * 300)))
        h_opt, sc_opt = binary_hill_climb_sa(
            h0, rng, max_iters=max_iters,
            target=target_bound, time_left=time_left,
            T0=0.05, Tmin=1e-6)
        if sc_opt < best_score_large:
            best_h_large, best_score_large = h_opt.copy(), sc_opt
            if best_score_large <= target_bound:
                break

    # ---------- Extended refinement on the best large candidate ----------
    if best_h_large is not None and time_left() > 5.0:
        max_iters = int(min(200000, max(1, time_left() * 2000)))
        h_opt, sc_opt = binary_hill_climb_sa(
            best_h_large, rng, max_iters=max_iters,
            target=target_bound, time_left=time_left,
            T0=0.02, Tmin=1e-6)
        if sc_opt < best_score_large:
            best_h_large, best_score_large = h_opt.copy(), sc_opt

    # ---------- Continuous momentum refinement ----------
    if best_h_large is not None and time_left() > 5.0:
        n_iter = int(min(5000, max(1, time_left() * 2000)))
        h_ref, sc_ref = momentum_grad_descent(
            best_h_large, n_iter=n_iter,
            step0=0.3, beta=0.9,
            time_left=time_left)
        if sc_ref < best_score_large:
            best_h_large, best_score_large = h_ref.copy(), sc_ref

    # ---------- Final projection and result ----------
    if best_h_large is None:
        best_h_large = binary_large_initial
        best_score_large = compute_c5(best_h_large)

    best_h_large = project_onto_box_sum(best_h_large, best_h_large.size / 2.0)
    final_score = compute_c5(best_h_large)

    return best_h_large, final_score, best_h_large.size
