import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def upsample(arr, new_n):
    """Linear interpolation of arr (values in [0,1]) to length new_n."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced_binary(n, rng):
    a = np.zeros(n, dtype=float)
    a[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return a

def repair_balance_bin(v, rng):
    """Force binary vector v to have exactly n/2 ones."""
    n = v.size
    target = n // 2
    ones = int(v.sum())
    if ones > target:
        idx = np.flatnonzero(v)
        to_flip = rng.choice(idx, size=ones - target, replace=False)
        v[to_flip] = 0.0
    elif ones < target:
        idx = np.flatnonzero(1.0 - v)
        to_flip = rng.choice(idx, size=target - ones, replace=False)
        v[to_flip] = 1.0
    return v

def tournament_select(pop, scores, rng, k=2):
    """Select the best of k random members."""
    idx = rng.integers(0, len(pop), size=k)
    best = idx[0]
    best_score = scores[best]
    for i in idx[1:]:
        if scores[i] < best_score:
            best = i
            best_score = scores[i]
    return pop[best]

def crossover(p1, p2, rng):
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(child, rng, prob=0.2):
    if rng.random() < prob:
        zeros = np.flatnonzero(child == 0)
        ones = np.flatnonzero(child == 1)
        if zeros.size and ones.size:
            i = rng.choice(zeros)
            j = rng.choice(ones)
            child[i], child[j] = child[j], child[i]
    return child

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h (averaged over given integer shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            p = -s
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] -= h[p:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        shifts = np.where(corr >= maxc - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best = sc, h_new.copy()
        h = h_new
    return best, best_score

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best = sc, h_new.copy()
        h = h_new
    return best, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style swap search on balanced binary vectors."""
    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_sc = curr
    max_iters = int(min(300000, max(1, time_left() * 2500)))
    T0 = 0.008
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # choose a zero and a one uniformly
        z = rng.choice(np.flatnonzero(h == 0))
        o = rng.choice(np.flatnonzero(h == 1))
        h_new = h.copy()
        h_new[z], h_new[o] = 1.0, 0.0
        new_sc = compute_c5(h_new)
        delta = new_sc - curr
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_sc < curr:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr = h_new, new_sc
            if new_sc < best_sc:
                best_sc, best = new_sc, h_new.copy()
                if best_sc <= target:
                    break
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    def time_left():
        return budget_s - (time.time() - start)

    n = 256                     # final resolution
    popsize = 30
    target_bound = 0.3808

    # ----- initial population -----
    pop = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n)
        order = np.argsort(init_up)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        binary = repair_balance_bin(binary, rng)
        pop.append(binary)

    pop.append(thue_morse(n))

    while len(pop) < popsize:
        pop.append(random_balanced_binary(n, rng))

    scores = np.array([compute_c5(h) for h in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    # ----- Genetic algorithm (binary) -----
    while time_left() > 150.0:
        elite_idx = np.argsort(scores)[:2]
        elite = [pop[i].copy() for i in elite_idx]

        children = []
        # inject a few fresh random individuals each generation
        inject = max(1, (popsize - 2) // 10)
        for _ in range(inject):
            children.append(random_balanced_binary(n, rng))

        while len(children) < popsize - 2:
            p1 = tournament_select(pop, scores, rng, k=2)
            p2 = tournament_select(pop, scores, rng, k=2)
            child = crossover(p1, p2, rng)
            child = mutate(child, rng, prob=0.15)
            child = repair_balance_bin(child, rng)
            children.append(child)

        pop = elite + children
        scores = np.array([compute_c5(h) for h in pop])
        idx = np.argmin(scores)
        if scores[idx] < best_score:
            best_score = scores[idx]
            best_h = pop[idx].copy()

    # ----- continuous refinement -----
    best_h = project_onto_box_sum(best_h, n / 2.0)

    if time_left() > 30.0:
        iters = int(min(40000, max(5000, time_left() * 350)))
        best_h, sc = smooth_grad_descent(best_h, n_iter=iters,
                                         step0=0.5, alpha_start=1.0,
                                         alpha_end=18.0, beta=0.9,
                                         time_left=time_left)
        if sc < best_score:
            best_score, best_h = sc, best_h.copy()

    if time_left() > 10.0:
        iters = int(min(20000, max(5000, time_left() * 300)))
        best_h, sc = momentum_grad_descent(best_h, n_iter=iters,
                                           step0=0.5, beta=0.9,
                                           time_left=time_left)
        if sc < best_score:
            best_score, best_h = sc, best_h.copy()

    # ----- binary local search -----
    if time_left() > 5.0:
        order = np.argsort(best_h)
        bin_h = np.zeros(n, dtype=float)
        bin_h[order[n // 2:]] = 1.0
        bin_h = repair_balance_bin(bin_h, rng)
        bin_h, bin_sc = binary_hill_climb_sa(bin_h, rng, time_left, target=target_bound)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
        if time_left() > 3.0:
            iters = int(min(8000, max(2000, time_left() * 200)))
            best_h, sc = momentum_grad_descent(best_h, n_iter=iters,
                                               step0=0.3, beta=0.9,
                                               time_left=time_left)
            if sc < best_score:
                best_score, best_h = sc, best_h.copy()

    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
