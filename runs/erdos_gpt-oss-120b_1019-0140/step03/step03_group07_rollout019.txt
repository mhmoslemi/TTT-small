def legendre_binary(n):
    p = n - 1
    # check if p is prime; if not, fallback to random binary
    # simple primality test
    if p < 2:
        return random_binary
    for i in range(2, int(p**0.5)+1):
        if p % i == 0:
            return None
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p-1)//2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    # set seq[0] maybe 0
    # need total ones = (p-1)/2
    # we need one extra 1
    h = np.concatenate([seq, [1.0]])
    # maybe shuffle random rotation
    shift = rng.integers(n)
    h = np.roll(h, shift)
    return h
```

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        # constraint: t >= overl * dx => t - overl*dx >= 0
        return x[-1] - overl * self.dx

    M = n // 2
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    bounds = [(0.0,1.0)] * n + [(0.0, None)]
    res = minimize(fun=lambda x: x[-1], x0=x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def generate_random_binary(n: int, M: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly M ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0

    if p % 2 == 0:
        return p == 2
    r = int(p ** 0.5)

    return True

        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    # add one extra 1 to reach sum = n/2
    h = np.concatenate([seq, [1.0]])
    shift = rng.integers(n)
    return np.roll(h, shift)


# ---------- simulated annealing ----------
def anneal_h(initial_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simulated annealing on h, preserving sum exactly."""
    start = time.time()
    n = initial_h.shape[0]
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    it = 0
    while time.time() - start < time_budget:
        it += 1
        T = max(1e-7, 5e-4 * (0.9995 ** it))
        i, j = rng.choice(n, 2, replace=False)
        hi, hj = cur_h[i], cur_h[j]
        max_ij = min(hi, 1.0 - hj)  # transfer i -> j
        max_ji = min(hj, 1.0 - hi)  # transfer j -> i
        if max_ij > 1e-12 and max_ji > 1e-12:
            if rng.random() < 0.5:
                max_transfer, direction = max_ij, 0
            else:
                max_transfer, direction = max_ji, 1
        elif max_ij > 1e-12:
            max_transfer, direction = max_ij, 0
        elif max_ji > 1e-12:
            max_transfer, direction = max_ji, 1
        else:
            continue
        delta = rng.random() * max_transfer
        h_new = cur_h.copy()
        if direction == 0:      # i -> j
            h_new[i] -= delta
            h_new[j] += delta
        else:                   # j -> i
            h_new[i] += delta
            h_new[j] -= delta
        c5_new = compute_c5(h_new)
        diff = c5_new - cur_c5
        if diff < 0 or rng.random() < np.exp(-diff / T):
            cur_h, cur_c5 = h_new, c5_new
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new.copy(), c5_new
                if best_c5 < 0.3808:
                    break
    return best_h, best_c5


# ---------- SLSQP refinement ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target


class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            ov = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            ov = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - ov * self.dx


def objective_last(x):
    return x[-1]


def slsqp_refine(h: np.ndarray, n: int):
    """Local SLSQP refinement (may improve bound)."""
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = minimize(fun=objective_last, x0=x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h, compute_c5(h)
    if not res.success:
        return h, compute_c5(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    M = n // 2
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)


# ---------- optimisation for a given resolution ----------
def optimise_n(n: int, time_budget: float, rng: np.random.Generator):
    M = n // 2
    starts = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        starts.append(np.array(initial_h_values, dtype=float))
    leg = legendre_binary(n, rng)
    if leg is not None:
        starts.append(leg)
    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n)
    lin = lin * (M / np.sum(lin))
    lin = np.clip(lin, 0.0, 1.0)
    cur = np.sum(lin)
    if abs(cur - M) > 1e-12:
        lin = lin + (M - cur) / n
        lin = np.clip(lin, 0.0, 1.0)
    starts.append(lin)
    # random binary
    starts.append(generate_random_binary(n, M, rng))

    best_h = None
    best_c5 = np.inf
    per_start = max(0.1, time_budget / len(starts))
    for h0 in starts:
        h_opt, c5_opt = anneal_h(h0, per_start, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break
    return best_h, best_c5


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80, 96, 112, 128, 160]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.5, 600.0)
        else:
            alloc = min(remaining / (len(candidates) - idx), 300.0)
        h_opt, c5_opt = optimise_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # optional local SLSQP refinement if close to target
    if best_h is not None and best_c5 < 0.381:
        if time.time() - start < budget_s - 2:
            h_ref, c5_ref = slsqp_refine(best_h, best_n)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

    # final safety adjustments
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        M = n // 2
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h = best_h + (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
