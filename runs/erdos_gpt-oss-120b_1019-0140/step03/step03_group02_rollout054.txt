def adam_optimize(h0, n_iter, step0, beta1, beta2, eps, alpha_start, alpha_end, time_left):

   m = np.zeros_like(h)

   v = np.zeros_like(h)

   s = h.size / 2.0 # target sum

   best_h = h.copy(); best_score = compute_c5(h)

   for it in range(1, n_iter+1):
       if time_left() < 0.01: break
       # schedule alpha (softmax temperature)
       alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter -1)
       grad = grad_softmax_corr(h, alpha)
       # Adam update
       m = beta1*m + (1-beta1)*grad
       v = beta2*v + (1-beta2)*(grad*grad)
       m_hat = m / (1 - beta1**it)
       v_hat = v / (1 - beta2**it)
       step = step0 * m_hat / (np.sqrt(v_hat) + eps)
       h_new = h - step
       h_new = project_onto_box_sum(h_new, s)
       score = compute_c5(h_new)
       if score < best_score:
           best_score = score; best_h = h_new.copy()
       h = h_new

   return best_h, best_score

def compute_corr_for_shift(h, shift):
    """Return integer count of h[i] * (1 - h[i+shift]) for linear correlation (no wrap)."""
    n = h.shape[0]
    if shift >= 0:
        if shift >= n:
            return 0
        return np.dot(h[:n-shift], 1.0 - h[shift:])
    else:
        k = -shift
        if k >= n:
            return 0
        return np.dot(h[k:], 1.0 - h[:n-k])

def compute_gradients_for_shifts(h):
   n = h.size
   corr = np.correlate(h, 1.0 - h, mode='full')
   maxc = np.max(corr)
   eps = 1e-12
   idxes = np.where(corr >= maxc - eps)[0]
   shifts = idxes - (n - 1)
   grad = np.zeros_like(h)
   for shift in shifts:
       if shift >= 0:
           grad[:n-shift] += (1.0 - h[shift:])
           grad[shift:] -= h[:n-shift]
       else:
           s = -shift
           grad[s:] += (1.0 - h[:n-s])
           grad[:n-s] -= h[s:]
   grad /= len(shifts)
   return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max shift integral approximation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_continuous_balanced(n, rng):
    """Random continuous vector projected onto box-sum constraints."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def upsample(arr, new_n):
    """Linear interpolation from arr (length m) to new_n samples."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_corr_for_shift(h, shift):
    """Correlation count for a given integer shift (no wrap)."""
    n = h.shape[0]
    if shift >= 0:
        if shift >= n:
            return 0.0
        return np.dot(h[:n - shift], 1.0 - h[shift:])
    else:
        k = -shift
        if k >= n:
            return 0.0
        return np.dot(h[k:], 1.0 - h[:n - k])

def compute_gradients_for_shifts(h):
    """Gradient of max-correlation term (averaged over maximal shifts)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= maxc - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]      -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of softmax-approximated max correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def adam_optimize(h0, n_iter, step0, beta1, beta2, eps,
                  alpha_start, alpha_end, time_left):
    """Adam optimizer on softmax‑approximated objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill‑climb on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    best_c5 = maxc * (2.0 / n)
    best_h = h.copy()
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)
    max_iters = int(min(500000, max(50000, time_left() * 2000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        a = rng.choice(ones_idx)
        b = rng.choice(zeros_idx)
        shift = np.argmax(corr) - (n - 1)
        c_before = compute_corr_for_shift(h, shift)
        # temporary swap
        h[a], h[b] = 0.0, 1.0
        c_after = compute_corr_for_shift(h, shift)
        delta = c_after - c_before
        # revert
        h[a], h[b] = 1.0, 0.0
        T = T0 * (1 - it / max_iters)
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h[a], h[b] = 0.0, 1.0
            corr = np.correlate(h, 1 - h, mode='full')
            maxc = np.max(corr)
            cur_c5 = maxc * (2.0 / n)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if best_c5 <= target:
                    break
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256  # grid size
    candidates = []

    # use parent construction if provided
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n) if init_full.size != n else init_full.copy()
        candidates.append(project_onto_box_sum(init_up, n / 2.0))

    # deterministic and random starts
    candidates.append(thue_morse(n).astype(float))
    for _ in range(4):
        candidates.append(random_binary_balanced(n, rng).astype(float))
    for _ in range(4):
        candidates.append(random_continuous_balanced(n, rng))

    best_h = None
    best_score = np.inf

    for h0 in candidates:
        if time_left() < 5.0:
            break
        max_iters = int(min(30000, max(5000, time_left() * 2500)))
        h_opt, sc_opt = adam_optimize(
            h0, n_iter=max_iters, step0=0.5,
            beta1=0.9, beta2=0.999, eps=1e-8,
            alpha_start=1.0, alpha_end=12.0,
            time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), n / 2.0)
        best_score = compute_c5(best_h)

    # binary rounding of the best continuous vector
    order = np.argsort(best_h)
    binary = np.zeros(n, dtype=float)
    binary[order[n // 2:]] = 1.0

    # allocate time for binary local search
    remaining = time_left()
    time_for_binary = remaining * 0.6
    binary_start = time.time()
    def binary_time_left():
        return time_for_binary - (time.time() - binary_start)

    best_bin_h = None
    best_bin_score = np.inf
    for restart in range(3):
        if binary_time_left() < 1.0:
            break
        start_vec = binary.copy() if restart == 0 else random_binary_balanced(n, rng).astype(float)
        h_bin, sc_bin = binary_hill_climb(start_vec, rng, binary_time_left, target=0.3808)
        if sc_bin < best_bin_score:
            best_bin_score, best_bin_h = sc_bin, h_bin.copy()
            if best_bin_score <= 0.3808:
                break

    if best_bin_h is not None and best_bin_score < best_score:
        best_h, best_score = best_bin_h.copy(), best_bin_score

    # final short Adam refinement
    if time_left() > 2.0:
        max_iters = int(min(5000, max(500, time_left() * 300)))
        h_ref, sc_ref = adam_optimize(best_h, n_iter=max_iters, step0=0.3,
                                      beta1=0.9, beta2=0.999, eps=1e-8,
                                      alpha_start=5.0, alpha_end=20.0,
                                      time_left=time_left)
        if sc_ref < best_score:
            best_h, best_score = h_ref.copy(), sc_ref

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
