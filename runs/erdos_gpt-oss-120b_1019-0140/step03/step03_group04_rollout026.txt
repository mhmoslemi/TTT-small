def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

    t0 = compute_c5(h0)

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds bracket s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    # generate Thue-Morse 0/1 via parity of popcount
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def binarize_top_half(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    # average gradient over considered shifts
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climbing_binary(h0, time_end, rng):
    h = h0.copy()
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    start_local = time.time()
    # temperature schedule parameters
    T0 = max(1e-4, current_score * 0.01)
    T_end = 1e-6
    while time.time() < time_end:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start_local
        total = max(time_end - start_local, 1e-12)
        frac = elapsed / total
        T = T_end + (T0 - T_end) * (1.0 - frac)  # linear cooling
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5(new_h)
        delta = new_score - current_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            h = new_h
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    # parameters
    N_SMALL = 64
    N_LARGE = 256
    # binary search phase (approx 45% of budget)
    binary_end = start + budget_s * 0.45
    best_small_h = None
    best_small_score = np.inf
    # collect possible initial seeds
    init_candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # ensure length N_SMALL (may be same)
        if init_full.size != N_SMALL:
            init_full = resample(init_full, N_SMALL)
        cand = binarize_top_half(project_onto_box_sum(init_full, N_SMALL / 2.0))
        init_candidates.append(cand)
    init_candidates.append(thue_morse_binary(N_SMALL))
    # generate a few random seeds upfront for diversity
    for _ in range(3):
        init_candidates.append(random_binary_balanced(N_SMALL, rng))
    # binary search loop
    while time.time() < binary_end:
        # choose a start candidate
        if rng.random() < 0.2 and init_candidates:
            h0 = rng.choice(init_candidates)
        else:
            h0 = random_binary_balanced(N_SMALL, rng)
        # occasionally use thue-morse directly
        if rng.random() < 0.05:
            h0 = thue_morse_binary(N_SMALL)
        slice_end = min(time.time() + 0.6, binary_end)
        h_opt, sc_opt = hill_climbing_binary(h0, slice_end, rng)
        if sc_opt < best_small_score:
            best_small_score, best_small_h = sc_opt, h_opt.copy()
    # fallback if no candidate found
    if best_small_h is None:
        best_small_h = random_binary_balanced(N_SMALL, rng)
        best_small_score = compute_c5(best_small_h)
    # upsample to larger resolution
    h_up = resample(best_small_h, N_LARGE)
    h_up = project_onto_box_sum(h_up, N_LARGE / 2.0)
    best_h = h_up
    best_score = compute_c5(best_h)
    # gradient descent refinement if time permits
    remaining = time_left()
    if remaining > 6.0:
        gd_end = time.time() + remaining - 3.0  # keep a margin for final SLSQP
        h_gd, sc_gd = gradient_descent_momentum(best_h, step0=0.5, beta=0.9,
                                                time_end=gd_end, max_iter=200000)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd
    # final SLSQP refinement
    remaining = time_left()
    if remaining > 4.0:
        max_iter = int(min(500, max(50, remaining * 5)))
        h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=max_iter)
        if sc_slsqp < best_score:
            best_h, best_score = h_slsqp, sc_slsqp
    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo <= x_i <= hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral C5 for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def binarize_top_half(arr):
    """Set the half largest entries to 1, others to 0."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    """Projected GD with momentum on C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climbing_binary(h0, time_end, rng):
    """Simulated‑annealing style binary swap search."""
    h = h0.copy()
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    start_local = time.time()
    T0 = max(1e-3, current_score * 0.05)   # initial temperature
    T_end = 1e-6
    while time.time() < time_end:
        elapsed = time.time() - start_local
        total = max(time_end - start_local, 1e-12)
        frac = elapsed / total
        T = T_end + (T0 - T_end) * (1.0 - frac)   # linear cooling
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = compute_c5(new_h)
        delta = new_score - current_score
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            h = new_h
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    """Short SLSQP refinement of a candidate."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    N_SMALL = 64
    N_LARGE = 256

    # ---------- binary search phase ----------
    binary_end = start + budget_s * 0.45
    best_small_h = None
    best_small_score = np.inf
    init_candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != N_SMALL:
            init_full = resample(init_full, N_SMALL)
        cand = binarize_top_half(project_onto_box_sum(init_full, N_SMALL / 2.0))
        init_candidates.append(cand)
    init_candidates.append(thue_morse_binary(N_SMALL))
    for _ in range(3):
        init_candidates.append(random_binary_balanced(N_SMALL, rng))

    while time.time() < binary_end:
        if rng.random() < 0.2 and init_candidates:
            idx = rng.integers(len(init_candidates))
            h0 = init_candidates[idx]
        else:
            h0 = random_binary_balanced(N_SMALL, rng)
        if rng.random() < 0.05:
            h0 = thue_morse_binary(N_SMALL)
        slice_end = min(time.time() + 0.6, binary_end)
        h_opt, sc_opt = hill_climbing_binary(h0, slice_end, rng)
        if sc_opt < best_small_score:
            best_small_score, best_small_h = sc_opt, h_opt.copy()
    if best_small_h is None:
        best_small_h = random_binary_balanced(N_SMALL, rng)
        best_small_score = compute_c5(best_small_h)

    # ---------- upsample ----------
    h_up = resample(best_small_h, N_LARGE)
    h_up = project_onto_box_sum(h_up, N_LARGE / 2.0)
    best_h = h_up
    best_score = compute_c5(best_h)

    # ---------- gradient descent refinement ----------
    remaining = time_left()
    if remaining > 6.0:
        gd_end = time.time() + remaining - 3.0   # reserve time for final SLSQP
        h_gd, sc_gd = gradient_descent_momentum(best_h, step0=0.5, beta=0.9,
                                                time_end=gd_end, max_iter=200000)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd

    # ---------- final SLSQP ----------
    remaining = time_left()
    if remaining > 4.0:
        max_iter = int(min(500, max(50, remaining * 5)))
        h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=max_iter)
        if sc_slsqp < best_score:
            best_h, best_score = h_slsqp, sc_slsqp

    # final projection and return
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
