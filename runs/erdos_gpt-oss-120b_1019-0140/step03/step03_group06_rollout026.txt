def greedy_search(h, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    cur_val = compute_c5(h)
    # while time and improvement
    while time.time() < deadline - 0.1:
        # compute correlation and find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        k = max_idx - (n - 1)
        # compute derivative
        d = np.zeros(n)
        if k >= 0:
            # forward term
            d[:n-k] += 1.0 - h[k:]
            # backward term
            d[k:] -= h[:n-k]
        else:
            kk = -k
            d[kk:] += 1.0 - h[:n-kk]
            d[:n-kk] -= h[kk:]
        # identify candidate indices
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        if not np.any(pos_mask) or not np.any(neg_mask):
            # cannot improve this shift, break
            break
        # choose indices with max magnitude
        i_pos = np.argmax(d * pos_mask)  # because d positive else zero
        i_neg = np.argmin(d * neg_mask)  # d negative else zero
        # allowable delta
        max_delta = min(h[i_pos], 1.0 - h[i_neg])
        if max_delta < 1e-12:
            # try other pair using random
            # pick random pos and neg with some delta
            pos_candidates = np.where(pos_mask)[0]
            neg_candidates = np.where(neg_mask)[0]
            i_pos = rng.choice(pos_candidates)
            i_neg = rng.choice(neg_candidates)
            max_delta = min(h[i_pos], 1.0 - h[i_neg])
            if max_delta < 1e-12:
                break
        # choose delta factor (maybe reduce to avoid overshoot)
        delta = max_delta * rng.random()
        # apply
        h[i_pos] -= delta
        h[i_neg] += delta
        # compute new val
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_val = np.max(new_corr) * dx
        if new_val <= cur_val:
            cur_val = new_val
            # continue loop
            continue
        else:
            # revert move
            h[i_pos] += delta
            h[i_neg] -= delta
            # try a smaller delta maybe
            delta = max_delta * 0.1 * rng.random()
            if delta < 1e-12:
                break
            # apply smaller delta
            h[i_pos] -= delta
            h[i_neg] += delta
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_val = np.max(new_corr) * dx
            if new_val <= cur_val:
                cur_val = new_val
                continue
            else:
                h[i_pos] += delta
                h[i_neg] -= delta
                # cannot improve, break
                break
    return h, cur_val

            h = best_h.copy()
            # apply few random swaps
            for _ in range(5):
                pos = np.where(h > 1e-8)[0]
                neg = np.where(h < 1 - 1e-8)[0]
                if len(pos) == 0 or len(neg) == 0:
                    break
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_delta = min(h[i], 1 - h[j])
                if max_delta <= 0: continue
                delta = max_delta * rng.random() * 0.5
                h[i] -= delta
                h[j] += delta
        else:
            h = random_continuous_vector(n, target, rng)

        # ensure feasibility
        h = enforce_sum(h, target, rng)

        # greedy local search
        h, val = greedy_search(h, deadline, rng)

        if val < best_val:
            best_val = val
            best_h = h.copy()
            # early exit if good enough
            if best_val < 0.3808:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    """Random vector in [0,1]^n with exact sum==target."""
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def sum_constraint_eq(x, n, target):
    """Equality constraint: sum of first n entries equals target."""
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    """Inequality: t - overlap(k) >= 0."""
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    """Objective for SLSQP: minimize the auxiliary variable t."""
    return x[-1]

def build_constraints(n, target, dx):
    """Create SLSQP constraints (equality + one inequality per shift)."""
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)})
    return cons

def greedy_search(h, deadline, rng):
    """Targeted local search: transfer mass to reduce worst overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    cur_val = compute_c5(h)
    max_iters = 5000
    for _ in range(max_iters):
        if time.time() > deadline - 0.1:
            break
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        k = max_idx - (n - 1)

        # derivative of overlap_k w.r.t. each h_i
        d = np.zeros(n)
        if k >= 0:
            d[: n - k] += 1.0 - h[k:]
            d[k:] -= h[: n - k]
        else:
            kk = -k
            d[kk:] += 1.0 - h[: n - kk]
            d[: n - kk] -= h[kk:]

        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        if not np.any(pos_mask) or not np.any(neg_mask):
            break

        i_pos = np.argmax(d * pos_mask)          # largest positive derivative
        i_neg = np.argmin(d * neg_mask)          # most negative derivative

        max_delta = min(h[i_pos], 1.0 - h[i_neg])
        if max_delta <= 1e-12:
            # fallback to any admissible pair
            pos_idxs = np.where(pos_mask)[0]
            neg_idxs = np.where(neg_mask)[0]
            i_pos = rng.choice(pos_idxs)
            i_neg = rng.choice(neg_idxs)
            max_delta = min(h[i_pos], 1.0 - h[i_neg])
            if max_delta <= 1e-12:
                break

        # try a random fraction of the transferable mass
        delta = max_delta * rng.random()
        h[i_pos] -= delta
        h[i_neg] += delta
        new_val = compute_c5(h)
        if new_val <= cur_val:
            cur_val = new_val
            continue

        # revert and attempt a smaller random move
        h[i_pos] += delta
        h[i_neg] -= delta
        improved = False
        for _ in range(3):
            src = rng.choice(np.where(h > 1e-12)[0])
            dst = rng.choice(np.where(h < 1 - 1e-12)[0])
            max_delta2 = min(h[src], 1.0 - h[dst])
            if max_delta2 <= 1e-12:
                continue
            dlt = max_delta2 * rng.random() * 0.5
            h[src] -= dlt
            h[dst] += dlt

            if val2 < cur_val:
                cur_val = val2
                improved = True
                break
            h[src] += dlt

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # problem dimension
    n = 200                     # increase degrees of freedom
    target = n / 2.0
    dx = 2.0 / n

    # initial vector
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        if init_h.shape[0] != n:
            x_old = np.linspace(0, 2, init_h.shape[0], endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init_h)
        else:
            h0 = init_h.copy()
    else:
        h0 = random_continuous_vector(n, target, rng)
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # multi‑start local search
    while time.time() < deadline - 1.0:
        if best_h is not None and rng.random() < 0.5:
            h = best_h.copy()
            for _ in range(5):
                pos = np.where(h > 1e-8)[0]
                neg = np.where(h < 1 - 1e-8)[0]
                if len(pos) == 0 or len(neg) == 0:
                    break
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random() * 0.5
                h[i] -= delta
                h[j] += delta
        else:
            h = random_continuous_vector(n, target, rng)
        h = enforce_sum(h, target, rng)

        h, val = greedy_search(h, deadline, rng)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val < 0.3808:
                break

    # final SLSQP refinement if time allows
    if time.time() < deadline - 5.0:
        cons = build_constraints(n, target, dx)
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_val]])
        try:
            res = minimize(
                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 2000, "ftol": 1e-10, "disp": False},
            )
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = enforce_sum(h_opt, target, rng)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best_val = val_opt
                    best_h = h_opt.copy()
        except Exception:
            pass

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
