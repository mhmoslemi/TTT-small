def anneal_binary(h, corr, time_left, rng, max_iters=10**9):
    n = len(h)
    best_h = h.copy()
    best_score = corr.max() * (2.0 / n)
    best_corr = corr.copy()
    curr_h = h
    curr_corr = corr
    curr_score = best_score

    # Temperature schedule
    T0 = 0.02  # initial temperature
    T_min = 1e-5
    decay = 0.99995
    T = T0

    iter = 0
    while time_left() > 0.0:
        # pick random ones and zeros
        ones = np.flatnonzero(curr_h)
        zeros = np.flatnonzero(1 - curr_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        hi = curr_h[i]; hj = curr_h[j]
        # compute new_corr and new_score
        new_corr, new_score = apply_swap_delta(curr_corr, curr_h, i, j, hi, hj)
        delta = new_score - curr_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            # accept swap
            # update h
            curr_h[i] = hj
            curr_h[j] = hi
            curr_corr = new_corr
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = curr_h.copy()
                best_corr = curr_corr.copy()
        # else discard swap (curr_h, curr_corr unchanged)
        iter += 1
        # update temperature
        T = max(T_min, T * decay)
        # optionally break if iter large
        if iter % 10000 == 0 and time_left() < 0.1:  # perhaps break if time nearly out

def upsample(arr, new_n):
    m = len(arr)
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def apply_swap_delta(corr, h, i, j, hi, hj):
    n = len(h)
    # early exit if hi==hj
    if hi == hj:
        return corr.copy(), corr.max() * (2.0 / n)
    # mask for other indices
    all_idx = np.arange(n)
    mask = (all_idx != i) & (all_idx != j)
    b = all_idx[mask]
    hb = h[b]
    hi_new = hj
    hj_new = hi
    delta = np.zeros_like(corr)
    # i,b
    idx_i_b = (b - i) + (n - 1)
    delta[idx_i_b] += (hi_new - hi) * (1 - hb)
    idx_b_i = (i - b) + (n - 1)
    delta[idx_b_i] += hb * ((1 - hi_new) - (1 - hi))  # = -hb*(hi_new - hi)
    # j,b
    idx_j_b = (b - j) + (n - 1)
    delta[idx_j_b] += (hj_new - hj) * (1 - hb)
    idx_b_j = (j - b) + (n - 1)
    delta[idx_b_j] += hb * ((1 - hj_new) - (1 - hj))
    # pair i,j
    k_ij = (j - i) + (n - 1)
    delta[k_ij] += (hj * (1 - hi) - hi * (1 - hj))
    k_ji = (i - j) + (n - 1)
    delta[k_ji] += (hi * (1 - hj) - hj * (1 - hi))
    new_corr = corr + delta
    new_score = new_corr.max() * (2.0 / n)
    return new_corr, new_score

Now annealing function:

def anneal_binary(h_start, corr_start, time_left, rng):
    n = len(h_start)
    curr_h = h_start.copy()
    curr_corr = corr_start.copy()
    curr_score = curr_corr.max() * (2.0 / n)
    best_h = curr_h.copy()
    best_corr = curr_corr.copy()
    best_score = curr_score
    T = 0.05  # initial temperature
    T_min = 1e-6
    decay = 0.99996
    while time_left() > 0.0:
        # choose random 1 and 0 indices
        ones = np.flatnonzero(curr_h)
        zeros = np.flatnonzero(1.0 - curr_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        hi = curr_h[i]
        hj = curr_h[j]
        new_corr, new_score = apply_swap_delta(curr_corr, curr_h, i, j, hi, hj)
        delta = new_score - curr_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # apply swap to current state
            curr_h[i], curr_h[j] = hj, hi
            curr_corr = new_corr
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = curr_h.copy()
                best_corr = curr_corr.copy()
        # temperature decay
        T = max(T_min, T * decay)
    return best_h, best_corr, best_score

import time
import numpy as np

def compute_corr_full(h):
    """Full linear correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode='full').astype(float)


def upsample(arr, new_n):
    """Linear up‑sample arr (length m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def binary_from_continuous(arr):
    """Return a binary (0/1) vector with exactly half the entries equal to 1."""
    n = arr.shape[0]
    k = n // 2
    idx = np.argpartition(-arr, k)[:k]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def apply_swap_delta(corr, h, i, j, hi, hj):
    """Return updated correlation and C5 after swapping positions i and j (binary values hi,hj)."""
    n = h.shape[0]
    if hi == hj:
        # no change
        return corr.copy(), corr.max() * (2.0 / n)

    all_idx = np.arange(n)
    mask = (all_idx != i) & (all_idx != j)
    b = all_idx[mask]                     # all other indices
    hb = h[b]                             # their values

    hi_new = hj
    hj_new = hi
    delta = np.zeros_like(corr)

    # contributions involving i and other indices
    idx_i_b = (b - i) + (n - 1)
    delta[idx_i_b] += (hi_new - hi) * (1.0 - hb)
    idx_b_i = (i - b) + (n - 1)
    delta[idx_b_i] += hb * ((1.0 - hi_new) - (1.0 - hi))

    # contributions involving j and other indices
    idx_j_b = (b - j) + (n - 1)
    delta[idx_j_b] += (hj_new - hj) * (1.0 - hb)
    idx_b_j = (j - b) + (n - 1)
    delta[idx_b_j] += hb * ((1.0 - hj_new) - (1.0 - hj))

    # direct i–j pair
    k_ij = (j - i) + (n - 1)
    delta[k_ij] += (hj * (1.0 - hi) - hi * (1.0 - hj))
    k_ji = (i - j) + (n - 1)
    delta[k_ji] += (hi * (1.0 - hj) - hj * (1.0 - hi))

    new_corr = corr + delta
    new_score = new_corr.max() * (2.0 / n)
    return new_corr, new_score


def anneal_binary(h_start, corr_start, time_left, rng):
    """Simulated annealing on binary vectors using O(n) incremental updates."""
    n = h_start.shape[0]
    curr_h = h_start.copy()
    curr_corr = corr_start.copy()
    curr_score = curr_corr.max() * (2.0 / n)

    best_h = curr_h.copy()
    best_corr = curr_corr.copy()
    best_score = curr_score

    T = 0.5          # initial temperature (in C5 units)
    T_min = 1e-6
    decay = 0.99998

    while time_left() > 0.0:
        ones = np.flatnonzero(curr_h)
        zeros = np.flatnonzero(1.0 - curr_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        hi = curr_h[i]
        hj = curr_h[j]

        new_corr, new_score = apply_swap_delta(curr_corr, curr_h, i, j, hi, hj)
        delta = new_score - curr_score
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # commit swap
            curr_h[i], curr_h[j] = hj, hi
            curr_corr = new_corr
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = curr_h.copy()
                best_corr = curr_corr.copy()

        T = max(T_min, T * decay)

    return best_h, best_corr, best_score


def greedy_improvement(h, corr, time_left, rng):
    """Deterministic hill climbing: accept any improving swap."""
    n = h.shape[0]
    best_h = h.copy()
    best_corr = corr.copy()
    best_score = best_corr.max() * (2.0 / n)

    improved = True
    while improved and time_left() > 0.0:
        improved = False
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1.0 - best_h)
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            for j in zeros:
                new_corr, new_score = apply_swap_delta(best_corr, best_h, i, j,
                                                      best_h[i], best_h[j])
                if new_score < best_score - 1e-12:
                    # perform swap
                    best_h[i], best_h[j] = best_h[j], best_h[i]
                    best_corr = new_corr
                    best_score = new_score
                    improved = True
                    break
            if improved:
                break
    return best_h, best_corr, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # problem size (balance between search depth and resolution)
    n = 512

    # initialise binary vector
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_arr, n)
        h0 = binary_from_continuous(init_up)
    else:
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0

    corr0 = compute_corr_full(h0)

    # main simulated annealing phase
    best_h, best_corr, best_score = anneal_binary(h0, corr0, time_left, rng)

    # optional local greedy refinement if time permits
    if time_left() > 2.0:
        best_h, best_corr, best_score = greedy_improvement(best_h, best_corr,
                                                           time_left, rng)

    # final safety recomputation
    final_corr = compute_corr_full(best_h)
    final_score = final_corr.max() * (2.0 / n)

    return best_h, final_score, n
