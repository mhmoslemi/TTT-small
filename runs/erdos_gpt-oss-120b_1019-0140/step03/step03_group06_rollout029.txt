def propose_move(h, k, target, rng):
    n = h.shape[0]
    # get i_valid
    start = -k if k < 0 else 0
    end = n - k if k > 0 else n
    i_valid = np.arange(start, end)
    # compute product
    prod = h[i_valid] * (1.0 - h[i_valid + k])
    i_max = i_valid[np.argmax(prod)]
    # candidate destinations
    all_idx = np.arange(n)
    outside = np.setdiff1d(all_idx, i_valid, assume_unique=True)
    # find any with headroom
    headroom_out = outside[h[outside] < 1.0 - 1e-12]
    if headroom_out.size > 0:
        j = rng.choice(headroom_out)
    else:

        # compute d = 1 - h[j+k] for all j in i_valid where headroom
        d = 1.0 - h[i_valid + k]
        mask = h[i_valid] < 1.0 - 1e-12
        if not np.any(mask):
            return None
        idx_candidates = i_valid[mask]

    # compute max delta

        return None
    delta = max_delta * rng.random()
    return i_max, j, delta

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [ {"type":"eq", "fun": sum_constraint_eq, "args": (n, target)} ]
    for k in range(-n+1, n):
        if k == 0: continue
        cons.append({"type":"ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)})
    return cons

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def worst_shift(h):
    """Return (shift k with maximal correlation, raw maximal correlation)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    return k, corr[max_idx]

def propose_move(h, k, rng):
    """Select a mass-transfer move aimed at reducing the worst overlap."""
    n = h.shape[0]
    start = -k if k < 0 else 0
    end = n - k if k > 0 else n
    i_valid = np.arange(start, end)                 # indices contributing to overlap(k)
    prod = h[i_valid] * (1.0 - h[i_valid + k])
    i_max = i_valid[np.argmax(prod)]                # source index
    # destination candidates
    all_idx = np.arange(n)
    outside = np.setdiff1d(all_idx, i_valid, assume_unique=True)
    headroom_out = outside[h[outside] < 1.0 - 1e-12]
    if headroom_out.size > 0:
        j = rng.choice(headroom_out)
    else:
        # inside region: choose index with smallest (1 - h[j+k]) and headroom
        d = 1.0 - h[i_valid + k]
        mask = h[i_valid] < 1.0 - 1e-12
        if not np.any(mask):
            return None
        cand_idx = i_valid[mask]
        cand_d = d[mask]
        j = cand_idx[np.argmin(cand_d)]
    if j == i_max:
        return None
    max_delta = min(h[i_max], 1.0 - h[j])
    if max_delta <= 1e-15:
        return None
    delta = max_delta * rng.random()
    return i_max, j, delta

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq,
                     "args": (k, n, dx)})
    return cons

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # dimension and initial vector
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.shape[0]
        n = 120                           # higher resolution than parent
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
    else:
        n = 120
        h0 = rng.random(n)
    target = n / 2.0
    h0 = enforce_sum(h0, target, rng)

    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    dx = 2.0 / n
    constraints = build_constraints(n, target, dx)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    iter_count = 0
    no_improve = 0
    T_start, T_end = 0.01, 1e-6

    while time.time() < deadline - 30.0:
        k, _ = worst_shift(cur_h)
        mv = propose_move(cur_h, k, rng)
        if mv is None:
            # fallback random small transfer
            src = np.where(cur_h > 1e-9)[0]
            dst = np.where(cur_h < 1.0 - 1e-9)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            i, j, delta = mv
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        # correct minute sum drift
        residual = target - new_h.sum()
        if abs(residual) > 1e-12:
            if residual > 0:
                cand = np.where(new_h < 1.0 - 1e-12)[0]
            else:
                cand = np.where(new_h > 1e-12)[0]
            if cand.size > 0:
                idx = rng.choice(cand)
                new_h[idx] += residual
                new_h[idx] = np.clip(new_h[idx], 0.0, 1.0)
        new_h = np.clip(new_h, 0.0, 1.0)

        new_val = compute_c5(new_h)
        elapsed = time.time() - start
        total = deadline - start
        T = T_start * (1 - elapsed / total) + T_end * (elapsed / total)

        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            no_improve += 1

        iter_count += 1

        # occasional local SLSQP refinement
        if iter_count % 2000 == 0 and time.time() < deadline - 30.0:
            t0 = compute_c5(cur_h)
            x0 = np.concatenate([cur_h, [t0]])
            try:
                res = minimize(fun=objective_fun, x0=x0, method="SLSQP",
                               bounds=bounds, constraints=constraints,
                               options={"maxiter": 500, "ftol": 1e-10, "disp": False})
                if res.success:
                    h_opt = np.clip(res.x[:n], 0.0, 1.0)
                    h_opt = enforce_sum(h_opt, target, rng)
                    val_opt = compute_c5(h_opt)
                    cur_h, cur_val = h_opt, val_opt
                    if val_opt < best_val:
                        best_val, best_h = val_opt, h_opt.copy()
                        if best_val < 0.3808:
                            break
            except Exception:
                pass

        # restart if stagnated
        if no_improve >= 8000 and time.time() < deadline - 30.0:
            cur_h = best_h.copy()
            # add tiny random perturbation while keeping sum
            for _ in range(5):
                src = np.where(cur_h > 1e-9)[0]
                dst = np.where(cur_h < 1.0 - 1e-9)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(cur_h[i], 1.0 - cur_h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random() * 0.1
                cur_h[i] -= delta
                cur_h[j] += delta
            cur_h = enforce_sum(cur_h, target, rng)
            cur_val = compute_c5(cur_h)
            no_improve = 0

    # final refinement if time permits
    if time.time() < deadline - 5.0:
        t0 = compute_c5(best_h)
        x0 = np.concatenate([best_h, [t0]])
        try:
            res = minimize(fun=objective_fun, x0=x0, method="SLSQP",
                           bounds=bounds, constraints=constraints,
                           options={"maxiter": 2000, "ftol": 1e-12, "disp": False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = enforce_sum(h_opt, target, rng)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best_val, best_h = val_opt, h_opt.copy()
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), float(final_val), n
