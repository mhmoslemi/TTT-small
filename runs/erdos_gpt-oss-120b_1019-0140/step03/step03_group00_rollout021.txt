def compute_c5_fast(h):
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(H * B, n=L)

    max_val = conv[:2*n-1].max()

def hill_climb_binary(h, max_iters, rng, time_left):

    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    current_h = h.copy()
    current_score = best_score

    T0 = 0.01
    T = T0
    cooling = 0.9999

    for it in range(max_iters):
        if time_left() <= 0:
            break
        # pick a random 1-index
        i = rng.integers(n)
        while current_h[i] == 0:
            i = rng.integers(n)
        # pick a random 0-index
        j = rng.integers(n)
        while current_h[j] == 1:
            j = rng.integers(n)
        # swap
        current_h[i] = 0
        current_h[j] = 1

        new_score = compute_c5(current_h)

        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = np.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            current_score = new_score
            # update best if needed
            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # revert
            current_h[i] = 1
            current_h[j] = 0

        # cooling schedule
        T = T * cooling
        # optionally reset T for early iteration to maintain acceptance
        # maybe every 1000 iterations, reset T = max(T, T0 * (1 - it / max_iters))?

    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def hill_climb_binary(h0, max_iters, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    # Temperature schedule
    T = 0.02
    cooling = 0.9995
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # select random i with 1
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        # select random j with 0
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # avoid overflow
            if T > 1e-12:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # temperature decay
        T = T * cooling
        # optionally reset T every 10000 iters to avoid too low temperature
        if (it+1) % 10000 == 0:
            T = max(T, 0.005)  # keep T from dropping too low
    return best_h, best_score

import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def upsample(arr, new_n):
    """Linear interpolation to a larger array size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def hill_climb_binary(h0, max_iters, rng, time_left):

    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()

    cur_score = best_score

    T = 0.02
    cooling = 0.9995

        if time_left() <= 0:
            break

        while h[i] == 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1.0:
            j = rng.integers(n)

        h[i] = 0.0
        h[j] = 1.0

    # optional refinement at higher resolution if time permits
    if time_left() > 30 and best_score > 0.3808:
        n2 = 2048
        h_up = upsample(best_h, n2)
        idx_desc2 = np.argsort(-h_up)
        h2 = np.zeros(n2, dtype=float)
        h2[idx_desc2[:n2 // 2]] = 1.0
        it_budget2 = int(min(120000, max(5000, time_left() * 800)))
        h_opt2, sc_opt2 = hill_climb_binary(h2, it_budget2, rng, time_left)
        if sc_opt2 < best_score:
            best_score, best_h = sc_opt2, h_opt2.copy()

    # final projection to guarantee exact constraint
    if not np.isclose(best_h.sum(), best_h.size / 2):
        best_h = project_onto_box_sum(best_h, best_h.size / 2)

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
