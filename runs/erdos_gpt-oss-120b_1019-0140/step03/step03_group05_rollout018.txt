def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    n = 80
    target_sum = n/2
    dx = 2.0/n
    # initialize population
    pop_size = 40
    pop = np.empty((pop_size, n))
    # first individual = initial_h if present
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.shape[0] != n:
            # adjust by interpolation or random
            # use linear interpolation to length n
            # But we cannot guarantee shape equals n; could be different length
            # Check, if not match, we can resample via interpolation
            idx = np.linspace(0, len(h0)-1, n)
            h0 = np.interp(idx, np.arange(len(h0)), h0)
        h0 = project_to_box_sum(np.clip(h0,0.0,1.0), target_sum)
        pop[0] = h0
        start_idx = 1
    else:
        start_idx = 0
    for i in range(start_idx, pop_size):
        h = rng.random(n)
        h = project_to_box_sum(h, target_sum)
        pop[i] = h
    # evaluate
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    # parameters for DE
    F = 0.8
    CR = 0.9

    # allocate time
    safety = 2.0
    while time.time() < start + budget_s - safety:
        # array of indices for each target
        for i in range(pop_size):
            # stop check
            if time.time() >= start + budget_s - safety:
                break
            # select three distinct different from i
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Ensure in [0,1] and sum constraint
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)
            # crossover binomial
            cross_mask = rng.random(n) < CR
            # ensure at least one dimension from mutant
            if not cross_mask.any():
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            # project again to sum constraint (crossover may disrupt sum)
            trial = project_to_box_sum(trial, target_sum)
            # evaluate
            c5_trial = compute_c5(trial)
            # selection
            if c5_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = c5_trial
                # update global best
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
                    # early break if target achieved
                    if best_c5 < 0.3808:
                        break
        # local refinement on best every few generations
        # apply gradient descent for a few steps
        if time.time() < start + budget_s - safety:
            # gradient descent
            gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter=200, time_limit=start+budget_s-safety)
            if gd_c5 < best_c5:
                best_c5 = gd_c5
                best_h = gd_h.copy()
                # replace in population if better than worst
                worst_idx = np.argmax(fitness)
                if gd_c5 < fitness[worst_idx]:
                    pop[worst_idx] = gd_h
                    fitness[worst_idx] = gd_c5
                if best_c5 < 0.3808:
                    break
    # final refinement: SLSQP optimization
    if time.time() < start + budget_s - safety:
        # prepare constraints
        # SLSQP expects constraints functions defined; we can copy from previous code
        # but we need to pass n and target sum.
        class SumConstraint:
            def __call__(self, x):
                return np.sum(x[:n]) - target_sum
        class OverlapConstraint:
            def __init__(self, k):
                self.k = k
            def __call__(self, x):
                h = x[:n]
                k = self.k
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overlap * dx
        # setup constraints
        sum_con = SumConstraint()
        overlap_cons = [OverlapConstraint(k) for k in range(-n+1, n) if k != 0]
        # objective t
        def obj_t(x):
            return x[-1]
        # initial guess
        t0 = best_c5
        x0 = np.concatenate([best_h, [t0]])
        bounds = [(0.0,1.0)]*n + [(0.0, None)]
        cons = [{'type':'eq','fun':sum_con}]
        cons += [{'type':'ineq','fun':oc} for oc in overlap_cons]
        # run minimize
        from scipy.optimize import minimize
        res = minimize(obj_t, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter':400, 'ftol':1e-9, 'disp':False})
        if res.success:
            h_opt = np.clip(res.x[:n],0.0,1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    # final projection
    best_h = np.clip(best_h,0.0,1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    lo_vals = lo * np.ones_like(v)
    hi_vals = hi * np.ones_like(v)
    # Use bisection on lambda to find shift such that sum of clipped(v + lam) = target
    low = -np.max(v) - 5.0
    high = hi - np.min(v) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

# ---------- Constraint classes for SLSQP ----------
class SumConstraint:
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Problem dimensions -----
    n = 80
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- Initial population -----
    pop_size = 40
    pop = np.empty((pop_size, n))
    fit = np.empty(pop_size)

    # First individual from parent if available
    start_idx = 0
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.shape[0] != n:
            # interpolate to correct length
            old_x = np.linspace(0.0, 2.0, h0.shape[0], endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(new_x, old_x, h0)
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
        pop[0] = h0
        fit[0] = compute_c5(h0)
        start_idx = 1

    for i in range(start_idx, pop_size):
        h = rng.random(n)
        h = project_to_box_sum(h, target_sum)
        pop[i] = h
        fit[i] = compute_c5(h)

    best_idx = int(np.argmin(fit))
    best_h = pop[best_idx].copy()
    best_c5 = fit[best_idx]

    # ----- Differential Evolution -----
    F = 0.8
    CR = 0.9
    safety_margin = 5.0
    no_improve = 0
    gen = 0

    while time.time() < start_time + budget_s - safety_margin:
        gen += 1
        improved_gen = False
        for i in range(pop_size):
            if time.time() >= start_time + budget_s - safety_margin:
                break
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != i]
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_to_box_sum(trial, target_sum)

            trial_c5 = compute_c5(trial)
            if trial_c5 < fit[i]:
                pop[i] = trial
                fit[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
                    improved_gen = True
                    if best_c5 < 0.3808:
                        break
        if best_c5 < 0.3808:
            break

        if improved_gen:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 10:
            # Reinitialize half the population
            num_reinit = pop_size // 2
            for _ in range(num_reinit):
                idx = rng.integers(pop_size)
                h = rng.random(n)
                h = project_to_box_sum(h, target_sum)
                pop[idx] = h
                fit[idx] = compute_c5(h)
            no_improve = 0

        # ----- Local refinement on the best candidate -----
        remain = start_time + budget_s - safety_margin - time.time()
        if remain > 1.0:
            gd_h, gd_c5 = gradient_descent(best_h, target_sum,
                                           max_iter=300,
                                           time_limit=start_time + budget_s - safety_margin)
            if gd_c5 < best_c5:
                best_c5 = gd_c5
                best_h = gd_h.copy()
                # replace worst individual with refined one
                worst_idx = int(np.argmax(fit))
                if gd_c5 < fit[worst_idx]:
                    pop[worst_idx] = gd_h
                    fit[worst_idx] = gd_c5
                if best_c5 < 0.3808:
                    break

    # ----- Final SLSQP refinement -----
    final_time = time.time()
    if final_time < start_time + budget_s - safety_margin:
        sum_con = SumConstraint(n, target_sum)
        overlap_cons = [OverlapConstraint(k, n, dx) for k in range(-n + 1, n) if k != 0]
        def obj_t(x):
            return x[-1]
        x0 = np.concatenate([best_h, [best_c5]])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        cons = [{'type': 'eq', 'fun': sum_con}]
        cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
        res = minimize(obj_t, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400,
                                                  'ftol': 1e-9,
                                                  'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # ----- Ensure feasibility and final evaluation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
