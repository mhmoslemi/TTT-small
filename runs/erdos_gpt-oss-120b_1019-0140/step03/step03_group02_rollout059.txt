def binary_hill_climb(h0, rng, time_left, max_iters=None, target=None):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    # Determine max_iters if not given using time_left
    if max_iters is None:
        # use time_left * 2000 swaps per sec approx
        max_iters = int(min(2_000_000, max(1, time_left() * 2000)))
    # temperature schedule: start at T0 = 0.01? maybe 0.02.
    T0 = 0.02
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # random 0 index
        # Using rng.choice on indices of zeros and ones might be faster if we maintain lists.
        # We'll maintain positions arrays for zeros and ones.
        # To update after swap efficiently, we can maintain lists but might be ok to just find using while loops as before.
        i = rng.integers(0, n)
        while h[i] != 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1:
            j = rng.integers(0, n)
        # swap
        h_new = h.copy()
        h_new[i] = 1
        h_new[j] = 0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        # schedule temperature decreasing linearly
        T = T0 * (1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if target is not None and best_score <= target:
                    break
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

The original code:

```
def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad
```

def binary_hill_climb(h0, rng, time_left, max_iters=None, target=None):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score

    if max_iters is None:
        max_iters = int(min(2_000_000, max(1, time_left() * 2500)))

    T0 = 0.02

    for it in range(1, max_iters + 1):
        if it % 1000 == 0 and time_left() < 0.01:
            break
        # compute max shift
        max_c, shift, _ = max_shift_corr(h)
        # find mismatch indices
        if shift >= 0:
            n_shift = n - shift
            i_cand = np.where((h[:n_shift] == 1) & (h[shift:] == 0))[0]
            j_cand = np.where((h[:n_shift] == 0) & (h[shift:] == 1))[0]
        else:
            s = -shift
            n_shift = n - s
            i_cand = np.where((h[s:] == 1) & (h[:n_shift] == 0))[0] + s
            j_cand = np.where((h[s:] == 0) & (h[:n_shift] == 1))[0] + s

        if i_cand.size > 0 and j_cand.size > 0:
            i = int(rng.choice(i_cand))
            j = int(rng.choice(j_cand))
        else:
            # fallback random swap
            i = rng.integers(0, n)
            while h[i] != 0.0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[j] != 1.0:
                j = rng.integers(0, n)

        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0

        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if target is not None and best_score <= target:
                    break

    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_shift_corr(h):
    """Return (max correlation value, shift achieving it, full correlation)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    return np.max(corr), shift, corr

def thue_morse(n):
    """Thue–Morse binary sequence as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end,
                        beta, time_left):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h0, rng, time_left, max_iters=None, target=None):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score

    if max_iters is None:
        max_iters = int(min(2_000_000, max(1, time_left() * 2500)))
    T0 = 0.02

    for it in range(1, max_iters + 1):
        if it % 1000 == 0 and time_left() < 0.01:
            break
        _, shift, _ = max_shift_corr(h)

        # find mismatching index pairs for the worst shift
        if shift >= 0:
            n_shift = n - shift
            i_cand = np.where((h[:n_shift] == 1) & (h[shift:] == 0))[0]
            j_cand = np.where((h[:n_shift] == 0) & (h[shift:] == 1))[0]
        else:
            s = -shift
            n_shift = n - s
            i_cand = np.where((h[s:] == 1) & (h[:n_shift] == 0))[0] + s
            j_cand = np.where((h[s:] == 0) & (h[:n_shift] == 1))[0] + s

        if i_cand.size > 0 and j_cand.size > 0:
            i = int(rng.choice(i_cand))
            j = int(rng.choice(j_cand))
        else:
            # fallback random balanced swap
            i = rng.integers(0, n)
            while h[i] != 0.0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[j] != 1.0:
                j = rng.integers(0, n)

        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0

        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if target is not None and best_score <= target:
                    break
    return best_h, best_score
