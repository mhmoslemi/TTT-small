def project_to_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 10 # ensure enough range
    high = (hi - np.min(v)) + 10
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)
```

def simulated_annealing(h_start, target_sum, time_limit, seed):
    rng = np.random.default_rng(seed)
    n = h_start.size
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # SA parameters
    T0 = 0.02
    T_end = 1e-5
    max_iter = 1000000  # but we will stop by time
    start_time = time.time()
    it = 0
    while True:
        now = time.time()
        if now > time_limit:
            break
        # compute temperature as linear schedule
        elapsed = now - start_time
        frac = elapsed / (time_limit - start_time)
        T = T0 * (1 - frac) + T_end * frac  # linear cooling
        # propose transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: transfer from i to j
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            # try opposite direction
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer <= 0:
                continue
            i, j = j, i  # swap so transfer from i to j
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        dE = new_c5 - cur_c5
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        it += 1
        # occasional reheat or random perturb
        if it % 10000 == 0:
            # small random jitter to escape local minima while preserving sum
            # apply random redistribution of small mass
            for _ in range(5):
                a, b = rng.integers(0, n, size=2)
                if a == b:
                    continue
                max_tr = min(h[a], 1.0 - h[b])
                if max_tr <= 0:
                    continue
                d = min(max_tr, 0.01)
                h[a] -= d
                h[b] += d
            cur_c5 = compute_c5(h)  # recompute after jitter
    return best_h, best_c5
```

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent_refine(h_start, target_sum, max_iter, time_limit):
    n = h_start.size
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    start = time.time()
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        grad = gradient_for_shift(h, max_idx - (n - 1))
        # step
        new_h = h - alpha * grad
        new_h = project_to_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        cur_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            # maybe random perturb
            if alpha == min_alpha:
                noise = np.random.default_rng().normal(scale=0.01, size=n)
                new_h = project_to_sum(np.clip(h + noise, 0, 1), target_sum)
                h = new_h
                alpha = 0.1
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # early stop if best below threshold
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # choose n
    n = 120  # or maybe 150. Keep moderate
    target_sum = n / 2.0

    # initialize h0
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n0 = h_init.size
        if n0 != n:
            x0 = np.linspace(0, 2, n0, endpoint=False)
            x = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x, x0, h_init)
        else:
            h0 = h_init.copy()
    else:
        h0 = rng.random(n)
    h0 = project_to_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Simulated annealing phase
    sa_deadline = start_time + budget_s * 0.7  # use 70% time for SA
    h_sa, c5_sa = simulated_annealing(best_h, target_sum, sa_deadline, seed+1)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # Gradient descent refinement phase
    gd_deadline = start_time + budget_s * 0.95
    remaining = gd_deadline - time.time()
    if remaining > 2.0:
        # allocate some iterations based on remaining time
        # approximate iteration time uses small constant
        max_iter = int(remaining * 50)  # e.g., 50 iterations per second
        max_iter = max(200, min(max_iter, 5000))
        h_gd, c5_gd = gradient_descent_refine(best_h, target_sum, max_iter, gd_deadline)
        if c5_gd < best_c5:
            best_h, best_c5 = h_gd, c5_gd

    # final projection to ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
```

import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing sum(v) == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def simulated_annealing(h_start, target_sum, time_limit, seed):
    """SA with sum-conserving pairwise transfers."""
    rng = np.random.default_rng(seed)
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    n = h.shape[0]
    T0, T_end = 0.04, 1e-6
    start = time.time()
    it = 0
    while True:
        now = time.time()
        if now > time_limit:
            break
        # linear temperature schedule
        frac = (now - start) / (time_limit - start)
        T = T0 * (1 - frac) + T_end * frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0.0:
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer <= 0.0:
                continue
            i, j = j, i
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        dE = new_c5 - cur_c5
        accept = False
        if dE < 0.0:
            accept = True
        else:
            # avoid overflow in exp
            prob = np.exp(-dE / max(T, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        it += 1
        # occasional tiny jitter to escape flat regions
        if it % 15000 == 0:
            for _ in range(3):
                a, b = rng.integers(0, n, size=2)
                if a == b:
                    continue
                tr = min(h[a], 1.0 - h[b])
                if tr <= 0.0:
                    continue
                d = min(tr, 0.01)
                h[a] -= d
                h[b] += d
            cur_c5 = compute_c5(h)
    return best_h, best_c5

def gradient_descent_refine(h_start, target_sum, max_iter, time_limit, seed):
    """Projected gradient descent on the max-overlap objective."""
    rng = np.random.default_rng(seed)
    n = h_start.shape[0]
    h = h_start.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    start = time.time()
    cur_c5 = best_c5
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.01, size=n)
                h = project_to_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                cur_c5 = compute_c5(h)
                alpha = 0.1
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

def upsample_h(old_h, new_n):
    """Linearly interpolate old_h (size m) to size new_n."""
    m = old_h.size
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    n = 120                     # number of discretisation points
    target_sum = n / 2.0

    # --- initialise feasible h ---
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        if h_init.size != n:
            h0 = upsample_h(h_init, n)
        else:
            h0 = h_init.copy()
    else:
        h0 = rng.random(n)
    h0 = project_to_sum(h0, target_sum)

    best_h, best_c5 = h0.copy(), compute_c5(h0)

    # --- simulated annealing phase (≈70% of budget) ---
    sa_deadline = start_time + budget_s * 0.70
    h_sa, c5_sa = simulated_annealing(best_h, target_sum, sa_deadline, seed + 101)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # --- gradient descent refinement (≈20% of budget) ---
    gd_deadline = start_time + budget_s * 0.95
    remaining = gd_deadline - time.time()
    if remaining > 2.0:
        # allocate iterations roughly proportional to remaining time
        max_iters = int(min(5000, max(200, remaining * 40)))
        h_gd, c5_gd = gradient_descent_refine(best_h, target_sum,
                                               max_iters, gd_deadline,
                                               seed + 202)
        if c5_gd < best_c5:
            best_h, best_c5 = h_gd, c5_gd

    # --- final projection and evaluation ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
