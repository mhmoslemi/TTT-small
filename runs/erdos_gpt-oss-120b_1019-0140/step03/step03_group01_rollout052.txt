def anneal_binary(h, time_left, total_iters=1000000):
    n = h.size
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_local_h = cur_h.copy()
    best_local_score = cur_score
    # init temperature:
    T = 0.01
    decay = 0.99999
    it = 0
    while time_left() > 0.0 and (total_iters is None or it < total_iters):
        # propose swap
        i = rng.integers(0, n)
        while cur_h[i] != 1:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while cur_h[j] != 0:
            j = rng.integers(0, n)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        # cooling
        T = max(T * decay, 1e-6)
        it += 1
    return best_local_h, best_local_score
```

def compute_c5_fft(h):
    n = h.size
    # linear correlation (full): compute convolution of h and reversed (1-h)
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)[::-1]
    # length = 2n-1
    L = 2*n - 1
    A = np.fft.fft(a, L)
    B = np.fft.fft(b, L)
    conv = np.fft.ifft(A * B).real
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)
```

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.shape[0]
    current_score = compute_c5(h)
    best_h = h.copy()

    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

    v = np.zeros_like(h)
    beta1_pow = beta1
    beta2_pow = beta2
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        h_new = h - step_size * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000):
    h = h0.copy()
    n = h.shape[0]
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def upsample(arr, new_n):
    old_n = arr.shape[0]
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def to_binary(h):
    n = h.shape[0]
    idx = np.argpartition(-h, n // 2)[: n // 2]
    b = np.zeros_like(h)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    n = 384
    s = n / 2.0

    candidates = []
    if 'initial_h_values' in globals():
        init_small = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_small, n)
        candidates.append(project_onto_box_sum(init_up, s))
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, s))

    best_h = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
        if best_score <= target:
            return best_h, best_score, n

    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = adam_optimize(cand.copy(), time_left, max_iter=50000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, n

    # binary refinement
    bin_h = to_binary(best_h)
    sc_bin = compute_c5(bin_h)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, bin_h.copy()
    if best_score <= target:
        return best_h, best_score, n

    # binary annealing
    if time_left() > 2.0:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h.copy(), rng, time_left, max_iters=300000)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, n

    # random continuous restarts with limited Adam
    while time_left() > 5.0:
        v = rng.random(n)
        v = project_onto_box_sum(v, s)
        h_ref, sc_ref = adam_optimize(v, time_left, max_iter=20000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            break

    # optional upscale to 512 for final polishing
    if time_left() > 20.0:
        n2 = 512
        s2 = n2 / 2.0
        h_up = upsample(best_h, n2)
        h_up = project_onto_box_sum(h_up, s2)
        bin_up = to_binary(h_up)
        sc_up = compute_c5(bin_up)
        if sc_up < best_score:
            best_score, best_h = sc_up, bin_up.copy()
        if time_left() > 2.0:
            h_up_ref, sc_up_ref = binary_swap_anneal(best_h.copy(), rng, time_left, max_iters=200000)
            if sc_up_ref < best_score:
                best_score, best_h = sc_up_ref, h_up_ref.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
