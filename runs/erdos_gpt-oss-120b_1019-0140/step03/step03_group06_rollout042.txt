def cvals(x):

def jac_cvals(x):
    h = x[:n]
    # shape (2n-2, n+1)
    J = np.empty((2*n - 2, n+1))
    # t derivative
    J[:, n] = 1.0
    idx = 0
    for k in range(-n+1, n):
        if k == 0: continue
        row = np.zeros(n)
        if k > 0:
            # forward shift: h_{j+k} for j=0..n-k-1
            row[:n - k] += h[k:]                # h_{j+k}
            # backward shift: h_{j - k} for j=k..n-1
            row[k:] += h[:n - k]                # h_{j - k}

            row[:n - kk] += h[kk:]              # h_{j - k} = h_{j+kk}
        J[idx, :n] = row
        idx += 1
    return J

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

import time
import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint

# ----- global placeholders for problem size -----
N_GLOBAL = None          # number of points
DX_GLOBAL = None         # spacing 2 / N_GLOBAL
global_bounds = None     # bounds for optimizer variables
global_linear_eq = None  # linear equality constraint (sum h = target)
global_nonlin = None     # nonlinear constraints (t >= overlap_k)

# ----- basic utilities -----
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clamp v to [0,1] and adjust to achieve exact sum == target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with exact sum == target."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly target ones (target assumed integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

# ----- constraint functions for trust-constr -----
def cvals(x: np.ndarray) -> np.ndarray:
    """Vector of constraints: t - overlap_k >= 0 for all k != 0."""
    h = x[:N_GLOBAL]
    t = x[N_GLOBAL]
    overlaps = np.correlate(h, 1.0 - h, mode="full") * DX_GLOBAL
    c = np.empty(2 * N_GLOBAL - 2)
    c[: N_GLOBAL - 1] = t - overlaps[: N_GLOBAL - 1]
    c[N_GLOBAL - 1 :] = t - overlaps[N_GLOBAL:]
    return c

def jac_cvals(x: np.ndarray) -> np.ndarray:
    """Jacobian of cvals: each row corresponds to a shift k."""
    h = x[:N_GLOBAL]
    n = N_GLOBAL
    J = np.empty((2 * n - 2, n + 1))
    J[:, n] = 1.0                     # derivative w.r.t. t
    idx = 0
    for k in range(-n + 1, n):
        if k == 0:
            continue
        row = np.zeros(n)
        if k > 0:
            row[: n - k] += h[k:]          # h_{j+k}
            row[k:] += h[: n - k]          # h_{j-k}
        else:
            kk = -k
            row[kk:] += h[: n - kk]        # h_{j+k}
            row[: n - kk] += h[kk:]        # h_{j-k}
        J[idx, :n] = row
        idx += 1
    return J

def obj_fun(x: np.ndarray) -> float:
    """Objective: minimize auxiliary variable t."""
    return x[N_GLOBAL]

def obj_jac(x: np.ndarray) -> np.ndarray:
    """Gradient of objective."""
    g = np.zeros_like(x)
    g[N_GLOBAL] = 1.0
    return g

# ----- optimizer wrapper -----
def opt_from_start(start_h: np.ndarray, target: float, rng: np.random.Generator):
    """Run trust‑constr from a given feasible start vector."""
    t0 = compute_c5(start_h)
    x0 = np.concatenate([start_h, [t0]])
    try:
        res = minimize(
            fun=obj_fun,
            x0=x0,
            method="trust-constr",
            jac=obj_jac,
            hess="2-point",
            constraints=[global_linear_eq, global_nonlin],
            bounds=global_bounds,
            options={"maxiter": 600, "verbose": 0},
        )
    except Exception:
        return None, None
    if not res.success:
        return None, None
    n = N_GLOBAL
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = enforce_sum(h_opt, target, rng)
    val = compute_c5(h_opt)
    return h_opt, val

# ----- main entry point -----

    start_time = time.time()
    deadline = start_time + budget_s

    global N_GLOBAL, DX_GLOBAL, global_bounds, global_linear_eq, global_nonlin

    global_linear_eq = LinearConstraint(A_eq, [target], [target])
    global_nonlin = NonlinearConstraint(

    for _ in range(max_restarts):
        if time.time() > deadline - 5.0:

    if time.time() < deadline - 1.0:

        cur_val = best_val

            j = i + k
            if j < 0 or j >= n:

            if cur_h[i] > cur_h[j] + 1e-12:

                if max_delta <= 0:

                delta = rng.random() * max_delta
                cur_h[i] -= delta
                cur_h[j] += delta
            elif cur_h[j] > cur_h[i] + 1e-12:
