import time
import numpy as np
from scipy.optimize import minimize

# ---- Helper functions ----
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap term w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        max_overlap = corr[max_idx]
        cur_c5 = max_overlap * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

# ---- SLSQP helpers ----
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        if self.k >= 0:
            overlap = np.dot(h[:self.n - self.k], 1.0 - h[self.k:])
        else:
            kk = -self.k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    """Objective for SLSQP: minimise the slack variable t."""
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=400):
    """Run SLSQP from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ---- Pre‑compute constraints for SLSQP ----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ---- Quick SLSQP warm‑start (if time permits) ----
    if time.time() - start_time < budget_s * 0.05:
        res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=300)
        if res.success:
            h_opt = project_to_box_sum(np.clip(res.x[:n], 0.0, 1.0), target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # ---- Simulated‑annealing + occasional refinement ----
    temp = 0.12                     # initial temperature
    temp_min = 1e-4
    decay = 0.9996                  # exponential cooling
    iteration = 0
    safety_margin = 5.0

    while time.time() < start_time + budget_s - safety_margin:
        # propose a neighbour
        cand = best_h + rng.normal(scale=temp, size=n)
        cand = project_to_box_sum(np.clip(cand, 0.0, 1.0), target_sum)
        c5_cand = compute_c5(cand)

        if c5_cand < best_c5:
            best_h, best_c5 = cand, c5_cand
        else:
            # Metropolis acceptance
            prob = np.exp((best_c5 - c5_cand) / max(temp, 1e-12))
            if rng.random() < prob:
                best_h, best_c5 = cand, c5_cand

        # temperature schedule
        temp = max(temp * decay, temp_min)

        # periodic local refinements
        if iteration % 80 == 0 and time.time() < start_time + budget_s - safety_margin - 2.0:
            # gradient descent refinement
            gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter=2000,
                                          time_limit=start_time + budget_s - safety_margin)
            if gd_c5 < best_c5:
                best_h, best_c5 = gd_h, gd_c5

        if iteration % 200 == 0 and time.time() < start_time + budget_s - safety_margin - 2.0:
            # SLSQP refinement
            res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=400)
            if res.success:
                h_opt = project_to_box_sum(np.clip(res.x[:n], 0.0, 1.0), target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

        if best_c5 < 0.3808:
            break

        iteration += 1

    # ---- Final projection and evaluation ----
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), n
