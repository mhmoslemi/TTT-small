import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the box [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient (without dx) of the overlap for shift k."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def random_feasible_vector(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = rng.random(n)
    return project_to_box_sum(v, target_sum)

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected gradient descent minimizing the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def differential_evolution(initial_h, n, target_sum, rng, time_limit):
    """Simple DE (rand/1/bin) with projection onto the feasible set."""
    popsize = min(60, max(20, n // 2))
    population = np.empty((popsize, n))
    population[0] = initial_h.copy()
    for i in range(1, popsize):
        population[i] = random_feasible_vector(n, target_sum, rng)

    fitness = np.empty(popsize)
    for i in range(popsize):
        fitness[i] = compute_c5(population[i])

    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_f = fitness[best_idx]

    F = 0.8
    CR = 0.9
    indices = np.arange(popsize)

    while time.time() < time_limit:
        for i in range(popsize):
            if time.time() >= time_limit:
                break
            # select three distinct individuals different from i
            pool = indices[indices != i]
            a, b, c = rng.choice(pool, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = project_to_box_sum(mutant, target_sum)

            # binomial crossover
            j_rand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[j_rand] = True
            trial = np.where(mask, mutant, population[i])
            trial = project_to_box_sum(trial, target_sum)

            trial_f = compute_c5(trial)
            if trial_f < fitness[i]:
                population[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best_h = trial.copy()
                    if best_f < 0.3808:
                        return best_h, best_f, True
        # (no adaptive parameter updates for simplicity)
    return best_h, best_f, False

        h0 = project_to_box_sum(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = random_feasible_vector(n, target_sum, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Allocate time for global DE, keep a safety buffer
    time_for_de = start_time + budget_s - 10.0
    best_h, best_c5, early = differential_evolution(best_h, n, target_sum, rng, time_for_de)

    # Local refinement if time permits
    if not early:
        gd_deadline = start_time + budget_s - 2.0
        gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter=4000,
                                       time_limit=gd_deadline, rng=rng)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
