import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def project_feasible(h, M):
    """Clip to [0,1] and adjust sum to M by water-filling."""
    h = np.clip(h, 0.0, 1.0)
    delta = M - h.sum()
    for _ in range(15):
        if abs(delta) <= 1e-12:
            break
        if delta > 0:
            eligible = np.where(h < 1 - 1e-12)[0]
        else:
            eligible = np.where(h > 1e-12)[0]
        if eligible.size == 0:
            break
        inc = delta / eligible.size
        adj = np.clip(h[eligible] + inc, 0.0, 1.0) - h[eligible]
        h[eligible] += adj
        delta = M - h.sum()
    return h

def gradient_of_max_shift(h):
    """Return gradient of the shift that currently attains max overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    g = np.zeros(n, dtype=float)
    i_start = max(0, -k)
    i_end = min(n, n - k)
    if i_start < i_end:
        i_vals = np.arange(i_start, i_end)
        j_vals = i_vals + k
        g[i_vals] += 1.0 - h[j_vals]
        g[j_vals] -= h[i_vals]
    return g, k, max_idx, corr[max_idx] * (2.0 / n)

def optimise_n(n, rng, time_budget):
    start_local = time.time()
    M = n // 2
    best_h = None
    best_c5 = np.inf

    seeds = []

    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        seeds.append(np.array(initial_h_values, dtype=float))

    if n == 80:
        p = 79
        leg = legendre_sequence(p)
        h_leg = np.concatenate([leg, [1.0]])
        h_leg = np.roll(h_leg, rng.integers(n))
        seeds.append(h_leg.astype(float))

    # a few random binary seeds
    for _ in range(2):
        h_bin = np.zeros(n)
        h_bin[rng.choice(n, M, replace=False)] = 1.0
        seeds.append(h_bin)

    # a few random fractional seeds
    for _ in range(2):
        h_frac = rng.random(n)
        h_frac = h_frac / h_frac.sum() * M
        h_frac = project_feasible(h_frac, M)
        seeds.append(h_frac)

    for seed_h in seeds:
        if time.time() - start_local > time_budget - 0.05:
            break
        h = project_feasible(seed_h.copy(), M)

        step0 = 0.6
        max_iter = 2000
        for it in range(max_iter):
            if time.time() - start_local > time_budget - 0.05:
                break
            g, k, idx, cur_c5 = gradient_of_max_shift(h)
            if cur_c5 < best_c5 - 1e-12:
                best_h, best_c5 = h.copy(), cur_c5
                if best_c5 < 0.3808:
                    return best_h, best_c5
            alpha = step0 / np.sqrt(it + 1.0)
            h_new = h - alpha * g
            h = project_feasible(h_new, M)

        # local random refinement
        refine_iters = 800
        for _ in range(refine_iters):
            if time.time() - start_local > time_budget - 0.05:
                break
            if rng.random() < 0.5:
                i, j = rng.choice(n, 2, replace=False)
                h_new = h.copy()
                h_new[i], h_new[j] = h_new[j], h_new[i]  # swap, sum unchanged
            else:
                i, j = rng.choice(n, 2, replace=False)
                max_transfer = min(1.0 - h[i], h[j])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
                h = h_new
                if best_c5 < 0.3808:
                    return best_h, best_c5

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates = [n0] + candidates
            else:
                candidates = [n0] + [x for x in candidates if x != n0]

    for idx, n in enumerate(candidates):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 0.5:
            break
        # allocate a share; keep a small reserve for final steps
        share = remaining * 0.9 / (len(candidates) - idx)
        h_opt, c5_opt = optimise_n(n, rng, share)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        # final feasibility enforcement
        n = best_n
        M = n // 2
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, M)

    return best_h, best_c5, best_n
