def anneal_binary(h0, max_iter, temp_start, temp_end, rng, time_left):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()

    n = len(h)

    zeros = np.flatnonzero(h < 0.5)

        j = rng.choice(zeros)

            zeros = np.flatnonzero(h < 0.5)
        else:

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand search interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the (unscaled) correlation over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_eps(h0, n_iter, step0=0.5, beta=0.9,
                                 eps_factor=0.02, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        max_raw = corr_raw.max()
        tol = eps_factor * max_raw
        idxes = np.where(corr_raw >= max_raw - tol)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * momentum, target_sum)
        if it % 250 == 0:
            score = compute_c5(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=300):
    """SLSQP refinement for a single candidate."""
    n = h0.shape[0]
    t0 = compute_c5(h0)

    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    def objective(x):
        return x[-1]

    res = opt.minimize(objective, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def anneal_binary(h0, max_iter, temp_start, temp_end, rng, time_left=lambda: 1e9):
    """Simulated annealing on binary balanced vectors."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]

    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)

    for it in range(max_iter):
        if time_left() <= 0:
            break
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

            if sc_gd < best_score:

        h_bin = np.zeros(n_mid, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(time_left() * 600)  # ~600 proposals per second
        h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                      temp_start=0.05, temp_end=1e-6,
                                      rng=rng, time_left=time_left)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # -------------------- Phase 5: Optional fine‑grained refinement --------------------
    if time_left() > 12.0:
        # Further gradient descent on the (potentially binary) best solution
        extra_iters = int(min(200000, time_left() * 800))
        if extra_iters > 0:
            h_ref, sc_ref = gradient_descent_momentum_eps(best_h, n_iter=extra_iters,
                                                          step0=0.4, beta=0.9,
                                                          eps_factor=0.03,
                                                          time_left=time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    # -------------------- Phase 6: Optional upscale to higher resolution (n=1024) --------------------
    if time_left() > 15.0:
        n_big = 1024
        big_h = upsample(best_h, n_big)
        big_h = project_onto_box_sum(big_h, n_big / 2.0)
        big_score = compute_c5(big_h)
        if big_score < best_score:
            best_score, best_h = big_score, big_h.copy()
        # Gradient descent at high resolution
        if time_left() > 8.0:
            max_iters_big = int(min(200000, time_left() * 500))
            if max_iters_big > 0:
                h_big, sc_big = gradient_descent_momentum_eps(best_h, n_iter=max_iters_big,
                                                             step0=0.5, beta=0.9,
                                                             eps_factor=0.025,
                                                             time_left=time_left)
                if sc_big < best_score:
                    best_score, best_h = sc_big, h_big.copy()

    # Final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, int(best_h.size)
