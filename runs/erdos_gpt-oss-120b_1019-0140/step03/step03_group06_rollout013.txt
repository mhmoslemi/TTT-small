import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def gradient_worst_shift(k, h):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # first term: (1 - h[i+k]) for i=0..n-k-1
        grad[:n - k] += 1.0 - h[k:]
        # second term: -h[i - k] for i=k..n-1
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def subgradient_descent(start_h, target, dx, rng, deadline):
    """Run subgradient descent from start_h, return best (h,val)."""
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5(best_h)
    max_iter = 1500  # per descent
    step0 = 0.2  # initial step size
    for it in range(max_iter):
        if time.time() >= deadline - 0.5:
            break
        # compute worst shift
        conv = np.correlate(h, 1.0 - h, mode='full')
        overlaps = conv * dx
        max_idx = int(np.argmax(overlaps))
        cur_val = overlaps[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            # early exit if we beat target threshold
            if best_val < 0.3808:
                break
        # gradient
        k = max_idx - (n - 1)
        grad = gradient_worst_shift(k, h)
        # step size schedule
        step = step0 / np.sqrt(it + 1)
        # try backtracking line search
        improved = False
        for _ in range(10):
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
            new_val = compute_c5(h_new)
            if new_val < cur_val:
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # occasional random perturbation to escape
            if rng.random() < 0.05:
                src = np.where(h > 1e-8)[0]
                dst = np.where(h < 1.0 - 1e-8)[0]
                if len(src) > 0 and len(dst) > 0:
                    i = rng.choice(src)
                    j = rng.choice(dst)
                    max_delta = min(h[i], 1.0 - h[j])
                    if max_delta > 0:
                        delta = rng.random() * max_delta
                        h[i] -= delta
                        h[j] += delta
                        h = enforce_sum(h, target, rng)
    return best_h, best_val

def annealing_refine(h, best_h, best_val, target, dx, rng, deadline):
    """Simple annealing that tries to improve h."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_val = best_val
    # compute correlation only once per iteration
    while time.time() < deadline - 0.2:
        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        cur_val = overlaps[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
            if best_val < 0.3808:
                break
        # pick source i (high contribution) and dest j (room)
        k = max_idx - (n - 1)
        # get indices involved in worst shift product values
        if k >= 0:
            prod = cur_h[: n - k] * (1.0 - cur_h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = cur_h[kk:] * (1.0 - cur_h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # destination with headroom
        cand = np.where(cur_h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or cur_h[i] <= 1e-12:
            src = np.where(cur_h > 1e-12)[0]
            dst = np.where(cur_h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur_h[i] -= delta
        cur_h[j] += delta
        # compute new max overlap
        new_val = np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full') * dx)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            # Metropolis acceptance
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_h[i] += delta
                cur_h[j] -= delta
            else:
                cur_h[i] -= delta
                cur_h[j] += delta
    return best_h, best_val

    start = time.time()

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():

    # ensure feasible start
    h0 = enforce_sum(h0, target, rng)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # main loop: multiple restarts
    while time.time() < deadline - 5.0:
        # generate start point
        if rng.random() < 0.6:
            start_h = random_binary_vector(n, target, rng)
        else:
            start_h = random_continuous_vector(n, target, rng)
        # blend with current best occasionally
        if rng.random() < 0.2:
            start_h = (start_h + best_h) * 0.5
            start_h = enforce_sum(start_h, target, rng)
        # subgradient descent
        h_sgd, val_sgd = subgradient_descent(start_h, target, dx, rng, deadline)
        if val_sgd < best_val:
            best_val = val_sgd
            best_h = h_sgd.copy()
            if best_val < 0.3808:
                break
        # short annealing refinement
        if time.time() < deadline - 2.0:
            h_ref, val_ref = annealing_refine(best_h, best_h, best_val, target, dx, rng, deadline)
            if val_ref < best_val:
                best_val = val_ref
                best_h = h_ref.copy()
                if best_val < 0.3808:
                    break

    # final enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
