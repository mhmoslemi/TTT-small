def compute_c5(h):
    n=len(h); dx=2.0/n
    corr=np.correlate(h,1.0-h,mode='full')
    return np.max(corr)*dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # v is array n
    # We need to find lambda s.t. sum(clip(v+lambda, lo, hi)) = target
    # monotonic in lambda: as lambda increases, each entry increase until hi.
    # Use binary search.
    lo_lambda = -np.max(v) - 10 # ensure all entries <= lo bound after shift? Actually v+lambda <= lo?

        if s < target:

    # initialize population

    population = [h0]
    for _ in range(pop_size-1):
        vec = rng.random(n)
        vec = project_to_box_sum(vec, target_sum)
        population.append(vec)

    # evaluate each
    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # main evolutionary loop
    generation = 0
    # define parameters
    # use simple tournament selection: select 2 parents, produce child via blend crossover and mutation.
    # Basic parameters: crossover_rate=0.7, mutation_scale decays.

    while time.time() - start < budget_s - 5:
        generation += 1
        # produce offspring and replace worst individuals
        # number offspring per generation: pop_size // 2
        num_offspring = pop_size // 2
        offspring = []
        for _ in range(num_offspring):
            # tournament selection for two parents
            i, j = rng.choice(pop_size, size=2, replace=False)
            parent1, parent2 = population[i], population[j]
            # blend crossover
            alpha = rng.random()
            child = alpha * parent1 + (1-alpha) * parent2
            # mutation (Gaussian)
            # mutation strength decays with generations
            sigma = 0.2 * np.exp(-generation/200)  # adjust
            noise = rng.normal(scale=sigma, size=n)
            child = child + noise
            # project to feasible region
            child = np.clip(child, 0.0, 1.0)
            child = project_to_box_sum(child, target_sum)
            offspring.append(child)

        # evaluate offspring
        offspring_scores = [compute_c5(ind) for ind in offspring]
        # combine and keep best pop_size individuals
        combined = population + offspring
        combined_scores = scores + offspring_scores
        # select indices of lowest scores
        sorted_idx = np.argsort(combined_scores)
        population = [combined[i] for i in sorted_idx[:pop_size]]
        scores = [combined_scores[i] for i in sorted_idx[:pop_size]]
        # update best

                scores[-1] = best_score

def gradient_descent_start(h_start, target_sum, max_iter, time_limit):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    dx = 2.0/n
    alpha = 0.5
    min_alpha = 1e-5
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        # gradient for worst shift
        k = max_idx - (n-1)
        grad = np.zeros_like(h)
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[kk:] += 1.0 - h[:n - kk]
            grad[:n - kk] -= h[kk:]
        # update
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_score = compute_c5(new_h)
        # acceptance
        if new_score <= best_score:
            best_h, best_score = new_h, new_score
        # adjust step size
        if new_score < compute_c5(h):
            h = new_h
            alpha = min(alpha * 1.1, 2.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
        # optionally add small random perturbation if stuck
        if it % 200 == 0:
            # perturb
            noise = np.random.normal(scale=0.01, size=n)

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# Global placeholder for DE target sum
_global_target_sum = None

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0) -> np.ndarray:
    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the worst overlap shift k w.r.t. h (without dx)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start: np.ndarray, target_sum: float,
                     max_iter: int, time_limit: float) -> tuple:
    """Projected gradient descent on the max‑overlap objective."""
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    dx = 2.0 / n
    alpha = 0.5
    min_alpha = 1e-6
    rnd = np.random.default_rng()
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_score:
            best_score = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.1, 2.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = rnd.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_score

class SumConstraint:
    """Equality constraint: sum(h) == target."""
    def __init__(self, n: int, target: float):
        self.n = n
        self.target = target
    def __call__(self, x: np.ndarray) -> float:
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k: int, n: int):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x: np.ndarray) -> float:
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[self.n] - overlap * self.dx

def objective_t(x: np.ndarray) -> float:
    """Objective for SLSQP: minimise the slack variable t."""
    return x[-1]

def slsqp_optimize(h_start: np.ndarray, t_start: float,
                   sum_con: SumConstraint, overlap_cons: list,
                   maxiter: int = 500) -> object:
    """Run a single SLSQP optimisation from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=objective_t, x0=x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

def de_objective(x: np.ndarray) -> float:
    """DE objective that projects onto the feasible set before evaluation."""
    global _global_target_sum
    proj = project_to_box_sum(x, _global_target_sum)
    return compute_c5(proj)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --- initialise problem size and feasible start vector ---
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # --- Differential Evolution phase (if time permits) ---
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 30.0:
        global _global_target_sum
        _global_target_sum = target_sum
        bounds = [(0.0, 1.0)] * n
        # rough budget for DE: aim for ≤ time_left‑5 s
        maxiter = int(min(2000, max(10, (time_left - 5.0) / 0.04)))
        popsize = 20
        result = differential_evolution(de_objective, bounds,
                                        seed=seed, maxiter=maxiter,
                                        popsize=popsize, polish=False,
                                        updating='deferred')
        de_h = project_to_box_sum(result.x, target_sum)
        de_score = compute_c5(de_h)
        if de_score < best_score:
            best_h, best_score = de_h, de_score
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed

    # --- Gradient descent refinement ---
    gd_time_limit = start_time + budget_s - 5.0
    gd_h, gd_score = gradient_descent(best_h, target_sum,
                                      max_iter=5000, time_limit=gd_time_limit)
    if gd_score < best_score:
        best_h, best_score = gd_h, gd_score

    # --- Final SLSQP polishing if time remains ---
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 10.0:
        sum_con = SumConstraint(n, target_sum)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        t0 = best_score
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_score:
                best_h, best_score = h_opt, c5_opt

    # --- final feasibility fix and return ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
