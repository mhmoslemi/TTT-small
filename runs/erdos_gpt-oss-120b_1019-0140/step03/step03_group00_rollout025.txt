import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_balanced(64, rng)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    for idx, n in enumerate(scales):
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        if n == 64 and time_left() > 15:
            for _ in range(3):
                if time_left() < 5:
                    break
                cand = base.copy()
                h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=400)
                if sc_opt < best_score:
                    best_h, best_score = h_opt.copy(), sc_opt
                base = h_opt

        if time_left() > 5:
            step0 = {64: 1.0, 128: 0.6, 256: 0.4, 512: 0.2}[n]
            beta = {64: 0.9, 128: 0.9, 256: 0.95, 512: 0.95}[n]
            est_iters = int(min(80000, max(5000, time_left() * 25 / (n / 64))))
            if est_iters < 2000:
                est_iters = 2000
            h_ref, sc_ref = gradient_descent_momentum_custom(
                base, n_iter=est_iters, step0=step0, beta=beta,
                rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_h, best_score = h_ref.copy(), sc_ref
            base = h_ref

        if idx < len(scales) - 1:
            base = upsample(base, scales[idx + 1])

    n_final = scales[-1]
    if best_h is None:
        best_h = project_onto_box_sum(random_balanced(n_final, rng), n_final / 2.0)
        best_score = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, n_final / 2.0)

    while time_left() > 8:
        h0 = rng.random(n_final)
        h0 = project_onto_box_sum(h0, n_final / 2.0)
        iters = int(min(20000, max(5000, time_left() * 15)))
        h_new, sc_new = gradient_descent_momentum_custom(
            h0, n_iter=iters, step0=0.2, beta=0.96, rng=rng, time_left=time_left)
        if sc_new < best_score:
            best_h, best_score = h_new.copy(), sc_new
        if best_score <= 0.3808:
            break

    if time_left() > 12:
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=300)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
