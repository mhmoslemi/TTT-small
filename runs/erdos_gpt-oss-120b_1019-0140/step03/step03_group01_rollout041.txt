import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure initial interval brackets the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf

    # baseline from parent if available
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = project_onto_box_sum(h0, h0.size / 2.0)

        best_h, best_score = h0, sc0

    # Phase 1: differential evolution on low‑dimensional step functions
    for m in (8, 12, 16):
        if time_left() < 10.0:
            break

        n = m * seg_len
        def de_obj(a):

            a_proj = project_onto_box_sum(a, m / 2.0)
            h = np.repeat(a_proj, seg_len)
            return compute_c5(h)

                de_obj,

        a_opt = project_onto_box_sum(result.x, m / 2.0)

        h_opt = project_onto_box_sum(h_opt, n / 2.0)
        sc_opt = compute_c5(h_opt)
        if sc_opt < best_score:

        if best_score <= target:

        if time_left() > 2.0:
            h_ref2, sc_ref2 = gradient_descent_momentum_with_time(best_h, time_left,
                                                                step0=0.5, beta=0.9, max_iter=200000)
            if sc_ref2 < best_score:
                best_score, best_h = sc_ref2, h_ref2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # final guard projection
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
        return best_h, best_score, best_h.size

    # fallback simple construction
