def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the interval bracket s
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            return np.clip(v - lam_mid, lo, hi)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # after loop
    lam_mid = (lam_low + lam_high) * 0.5
    return np.clip(v - lam_mid, lo, hi)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shifts correspond to indices range -(n-1) to (n-1)
    offset = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n-shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n-s]
            grad[:n-s] -= h[s:]
    return grad / len(shifts)

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda:1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda:1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()

    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:

def random_binary_balanced(n, rng):

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n // 2
    # ensure binary and sum correct
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # Determine max_iters based on time left
    max_iters = int(min(5_000_000, max(1, time_left() * 20000)))
    T0 = max(0.001, (curr_score - target) * 0.01)
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick swap
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # swap
        h[i], h[j] = 1.0, 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            # revert swap
            h[i], h[j] = 0.0, 1.0
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            return np.clip(v - lam_mid, lo, hi)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    lam_mid = (lam_low + lam_high) * 0.5
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    return grad / len(shifts)

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_to_binary(v, target_sum):
    idx = np.argpartition(v, -target_sum)[-target_sum:]
    binary = np.zeros_like(v)
    binary[idx] = 1.0
    return binary

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(5_000_000, max(1, time_left() * 20000)))
    T0 = max(0.001, (curr_score - target) * 0.01)
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h[i], h[j] = 1.0, 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_score
