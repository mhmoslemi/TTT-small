import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512
    # initialize binary candidate
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            init_full = upsample(init_full, n)
        # choose top half indices
        idx = np.argpartition(init_full, -n//2)[-n//2:]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n//2, replace=False)] = 1.0

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # annealing parameters
    temp_start = 0.05
    temp_end = 1e-6

    # main annealing loop
    while time_left() > 2.0:
        # maximum iterations for this run
        max_iters = int(min(200000, time_left() * 4000))  # ~4000 it/s estimate
        cur_h = best_h.copy()
        cur_score = best_score
        for it in range(max_iters):
            # time check
            if it % 1000 == 0 and time_left() <= 0:
                break
            # temperature schedule (log-linear)
            t = temp_start * ((temp_end / temp_start) ** (it / max_iters))
            # pick random swap
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # perform swap
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_score = compute_c5(cur_h)
            delta = new_score - cur_score
            if delta <= 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_h = new_score, cur_h.copy()
                    if best_score <= 0.3808:
                        return best_h, best_score, n
            else:
                # revert swap
                cur_h[i] = 1.0
                cur_h[j] = 0.0

    # final ensure sum constraint (binary sum should be n/2)
    # If best_h not exactly balanced due to possible numeric, enforce
    sum_ones = best_h.sum()
    if sum_ones != n/2:
        # adjust by setting the smallest values to 0 or 1
        # but best_h should be binary; we just project to binary balanced
        idx = np.argpartition(best_h, -n//2)[-n//2:]
        best_h = np.zeros(n, dtype=float)
        best_h[idx] = 1.0
        best_score = compute_c5(best_h)

    return best_h, best_score, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512
    # ----- initial binary candidate -----
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            init_full = upsample(init_full, n)
        idx = np.argsort(init_full)[-n // 2 :]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # ----- simulated annealing on binary vectors -----
    temp_start = 0.05
    temp_end = 1e-6
    target = 0.3808

    while time_left() > 2.0:
        # allocate iterations for this run (roughly 4000 it/s estimate)
        max_iters = int(min(200_000, time_left() * 4000))
        if max_iters <= 0:
            break
        cur_h = best_h.copy()
        cur_score = best_score
        for it in range(max_iters):
            if it % 1000 == 0 and time_left() <= 0:
                break
            # temperature schedule (log‑linear decay)
            t = temp_start * ((temp_end / temp_start) ** (it / max_iters))
            # choose a random 1↔0 swap
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_score = compute_c5(cur_h)
            delta = new_score - cur_score
            if delta <= 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_h = new_score, cur_h.copy()
                    if best_score <= target:
                        return best_h, best_score, n
            else:
                cur_h[i] = 1.0
                cur_h[j] = 0.0

    # ----- final safeguard: enforce exact balance -----
    if best_h.sum() != n // 2:
        idx = np.argsort(best_h)[-n // 2 :]
        best_h = np.zeros(n, dtype=float)
        best_h[idx] = 1.0
        best_score = compute_c5(best_h)

    return best_h, best_score, n
