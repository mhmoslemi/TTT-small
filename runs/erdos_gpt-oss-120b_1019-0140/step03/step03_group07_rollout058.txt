def anneal(h, time_limit, rng):
    start = time.time()
    n = len(h); M = n//2; dx = 2.0/n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    cur_c5 = best_c5
    t0 = time_limit
    while time.time() - start < time_limit:
        # temperature schedule: linearly decreasing from T0 to 0
        elapsed = time.time() - start
        T = max(0.001, (1 - elapsed/t_limit) * 1.0) # adjust scale
        # propose move: random swap of 0 and 1
        zero_idxs = np.where(current_h == 0)[0]
        one_idxs = np.where(current_h == 1)[0]
        i = rng.choice(zero_idxs)
        j = rng.choice(one_idxs)
        # compute new C5
        cand = current_h.copy()
        cand[i] = 1
        cand[j] = 0
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5
```

def improve_targeted(h, max_iter, rng):
    n = len(h)
    dx = 2.0/n
    best_c5 = compute_c5(h)
    best_h = h
    it = 0
    while it < max_iter:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        overl_max = corr[max_idx]
        if overl_max == 0:
            # perfect
            break
        k = max_idx - (n - 1)  # shift
        # Identify bad positions (where h_i=1 and h_{i+k}=0)
        if k >= 0:
            bad = np.where((h[:n-k]==1) & (h[k:]==0))[0]
            good = np.where((h[:n-k]==0) & (h[k:]==1))[0]
        else:
            kk = -k
            bad = np.where((h[kk:]==1) & (h[:n-kk]==0))[0] + kk
            good = np.where((h[kk:]==0) & (h[:n-kk]==1))[0] + kk
        if bad.size == 0 or good.size == 0:
            # fallback random swap
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(bad)
            j = rng.choice(good)
        # swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
            it = 0
        else:
            # maybe accept random? For simplicity reject
            it += 1
    return best_h, best_c5
```

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic utilities
# -------------------------------------------------
def compute_c5(h):
    """C5 = max_k ∫ h(x) (1-h(x+k)) dx evaluated on the discrete grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def objective_t(x):
    """objective for SLSQP: minimise the auxiliary variable t."""
    return x[-1]


# -------------------------------------------------
# constraints for the continuous refinement
# -------------------------------------------------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


def make_constraints(n):
    """build the full constraint list for a given size n."""
    cons = [{"type": "eq", "fun": SumConstraint(n)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": OverlapConstraint(k, n)})
    return cons


# -------------------------------------------------
# targeted local improvement for binary vectors
# -------------------------------------------------
def improve_targeted(h, rng, max_stagnation=400):
    """
    Greedy hill‑climbing that always attacks the shift giving the
    current maximal overlap. The move swaps a "bad" 1 with a "good"
    0, decreasing that maximal overlap whenever possible.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stagnation = 0

    while stagnation < max_stagnation:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = int(np.argmax(corr))
        overl = corr[max_idx]
        if overl <= 0:
            break
        k = max_idx - (n - 1)   # actual shift value

        if k >= 0:
            bad = np.where((best_h[: n - k] == 1) & (best_h[k:] == 0))[0]
            good = np.where((best_h[: n - k] == 0) & (best_h[k:] == 1))[0]
        else:
            kk = -k
            bad = np.where((best_h[kk:] == 1) & (best_h[: n - kk] == 0))[0] + kk
            good = np.where((best_h[kk:] == 0) & (best_h[: n - kk] == 1))[0] + kk

        if bad.size == 0 or good.size == 0:
            zeros = np.where(best_h == 0)[0]
            ones = np.where(best_h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(bad)
            j = rng.choice(good)

        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        c5_cand = compute_c5(cand)

        if c5_cand + 1e-12 < best_c5:
            best_h = cand
            best_c5 = c5_cand
            stagnation = 0
        else:
            stagnation += 1

    return best_h, best_c5


# -------------------------------------------------
# search routine for a single resolution
# -------------------------------------------------
def search_n(n, time_budget, rng):
    M = n // 2
    best_h = None
    best_c5 = np.inf
    start = time.time()

    # ----- parent construction (if it matches this n) -----
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        h0 = np.array(initial_h_values, dtype=float)
        bin_h = (h0 >= 0.5).astype(float)
        diff = M - bin_h.sum()
        if diff > 0:
            zero_idx = np.where(bin_h == 0)[0]
            order = np.argsort(-h0[zero_idx])[: int(diff)]
            bin_h[zero_idx[order]] = 1.0
        elif diff < 0:
            one_idx = np.where(bin_h == 1)[0]
            order = np.argsort(h0[one_idx])[: int(-diff)]
            bin_h[one_idx[order]] = 0.0
        c5 = compute_c5(bin_h)
        if c5 < best_c5:
            best_h, best_c5 = bin_h, c5

    # ----- random restarts with targeted improvement -----
    while time.time() - start < time_budget - 0.5:
        h = np.zeros(n)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
        h_opt, c5_opt = improve_targeted(h, rng, max_stagnation=400)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # ----- final continuous refinement (SLSQP) -----
    if best_h is not None and time.time() - start < time_budget - 0.2:
        try:
            t0 = compute_c5(best_h)
            x0 = np.concatenate([best_h, [t0]])
            cons = make_constraints(n)
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            res = minimize(
                fun=objective_t,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 800, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                h_cont = np.clip(res.x[:n], 0.0, 1.0)
                cur = h_cont.sum()
                if abs(cur - M) > 1e-9:
                    h_cont += (M - cur) / n
                    h_cont = np.clip(h_cont, 0.0, 1.0)
                c5_cont = compute_c5(h_cont)
                if c5_cont < best_c5:
                    best_h, best_c5 = h_cont, c5_cont
        except Exception:
            pass

    return best_h, best_c5


# -------------------------------------------------
# main entry point
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80, 96, 112, 128]

    # prioritize the parent size if supplied
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 in candidates:
            candidates.remove(n0)
            candidates = [n0] + candidates

    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.6, 500.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)

        h_opt, c5_opt = search_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # ----- fallback if nothing succeeded -----
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = best_h.sum()
        M = best_n // 2
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
