def smooth_max_corr_and_grad(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    max_corr = np.max(corr)
    # avoid overflow: subtract max
    exp_vals = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    smooth_max = max_corr + np.log(sum_exp) / alpha
    weights = exp_vals / sum_exp  # shape (2*n-1,)
    grad = np.zeros_like(h)
    shift_center = n - 1
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        shift = idx - shift_center  # shift can be negative
        if shift >= 0:
            # term1: 1 - h[i+shift] for i+shift < n, else 1
            # term2: h[i-shift] for i-shift >= 0, else 0
            # fill term1
            term1 = np.empty_like(h)
            term1[:n - shift] = 1.0 - h[shift:]
            term1[n - shift:] = 1.0
            term2 = np.empty_like(h)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty_like(h)
            term1[s:] = 1.0 - h[:n - s]
            term1[:s] = 1.0
            term2 = np.empty_like(h)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += w * (term1 - term2)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def smooth_max_corr_and_grad(h, alpha):
    """Smooth approximation of max correlation and its gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))  # safe scaling
    sum_exp = np.sum(exp_vals)
    smooth_max = max_corr + np.log(sum_exp) / alpha
    weights = exp_vals / sum_exp
    grad = np.zeros_like(h)
    # padded view for easy shifted access (zeros outside)
    hpad = np.pad(h, (n, n), mode='constant')
    shift_center = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - shift_center
        term1 = 1.0 - hpad[n + shift : n + shift + n]
        term2 = hpad[n - shift : n - shift + n]
        grad += w * (term1 - term2)
    return smooth_max, grad

def smooth_gradient_descent(h0, time_left_func,
                           step0=0.5, beta=0.9,
                           alpha0=1.0, alpha_max=15.0,
                           max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.5:
        alpha = min(alpha_max, alpha0 * (1.0 + it / 5000.0))
        _, grad = smooth_max_corr_and_grad(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000):
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h = h.copy()
    best_score = cur
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        delta = new - cur
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur = new
            if new < best_score:
                best_score = new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    n_large = 256

    # initial vector
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init_full, n_large) if init_full.size != n_large else init_full.copy()
    else:
        h0 = random_binary_balanced(n_large, rng)
    h0 = project_onto_box_sum(h0, n_large / 2.0)

    best_h = h0
    best_score = compute_c5(best_h)

    # Continuous smooth‑gradient phase (multiple restarts if time permits)
    while time_left() > 10.0:
        # 50% chance to start from current best, else random
        start_vec = best_h if rng.random() < 0.5 else project_onto_box_sum(rng.random(n_large), n_large / 2.0)
        h_opt, sc_opt = smooth_gradient_descent(start_vec, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Convert to binary (take largest half entries)
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, n

    # Binary annealing refinement
    if time_left() > 5.0:
        h_ref, sc_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=300000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, n

    # Additional random restarts of the smooth‑gradient method
    while time_left() > 5.0:
        rand_vec = project_onto_box_sum(rng.random(n), n / 2.0)
        h_opt, sc_opt = smooth_gradient_descent(rand_vec, time_left, max_iter=5000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
