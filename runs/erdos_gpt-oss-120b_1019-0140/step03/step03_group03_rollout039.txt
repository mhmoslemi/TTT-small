def compute_delta_swap(i, j, ones, N):
    """Return delta array of length 2N-1 for autocorr change when swapping 1 at i with 0 at j."""
    size = 2 * N - 1
    delta = np.zeros(size, dtype=int)
    # removal at i
    # shifts for all p in ones (including i)
    shifts = ones - i
    delta[shifts + N - 1] -= 1
    # other direction
    shifts2 = i - ones
    delta[shifts2 + N - 1] -= 1

    # addition at j (exclude i)
    mask = ones != i
    ones_ex = ones[mask]
    shifts = ones_ex - j
    delta[shifts + N - 1] += 1
    shifts2 = j - ones_ex
    delta[shifts2 + N - 1] += 1

    # d*d contributions
    delta[N - 1] += 2
    delta[(j - i) + N - 1] -= 1
    delta[(i - j) + N - 1] -= 1
    return delta

    if 'initial_h_values' in globals():

        # upsample to n via repeat
        factor = n // init_full.size

        else:

            x_old = np.linspace(0, 1, init_full.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            up = np.interp(x_new, x_old, init_full)
        # convert to binary: pick top n/2 values
        idx = np.argpartition(up, -n//2)[-n//2:]

    for _ in range(3):

        h_rand[idx] = 1
        candidates.append(h_rand)

    # For each candidate, run hill climb with time allocation
    # allocate time for each candidate proportionally
    # Let's just iterate while time left > some threshold
    while time_left() > 5.0:
        # pick a candidate: either random new or from list

def delta_autocorr_swap(i, j, ones, N):
    size = 2*N - 1
    delta = np.zeros(size, dtype=int)

    # removal at i
    shifts = ones - i
    delta[shifts + N - 1] -= 1
    shifts2 = i - ones
    delta[shifts2 + N - 1] -= 1

    # addition at j (include all ones)
    shifts = ones - j
    delta[shifts + N - 1] += 1
    shifts2 = j - ones
    delta[shifts2 + N - 1] += 1

    # d*d contributions
    delta[N - 1] += 2
    delta[(j - i) + N - 1] -= 1
    delta[(i - j) + N - 1] -= 1

    return delta

import time
import numpy as np

def compute_autocorr_bin(h):
    """Autocorrelation of binary vector h (full mode)."""
    return np.correlate(h, h, mode='full')

def delta_autocorr_swap(i, j, ones, N):
    """Delta of autocorrelation when swapping 1 at i with 0 at j."""
    size = 2 * N - 1
    delta = np.zeros(size, dtype=int)

    # removal of the 1 at i
    shifts = ones - i
    delta[shifts + N - 1] -= 1
    shifts2 = i - ones
    delta[shifts2 + N - 1] -= 1

    # addition of a 1 at j (use original ones, including i)
    shifts = ones - j
    delta[shifts + N - 1] += 1
    shifts2 = j - ones
    delta[shifts2 + N - 1] += 1

    # d*d term for the two-point change
    delta[N - 1] += 2               # k = 0
    delta[(j - i) + N - 1] -= 1     # k = j-i
    delta[(i - j) + N - 1] -= 1     # k = i-j
    return delta

def hill_climb_binary(h0, budget, rng):
    """Local search on binary vectors to maximize minimal autocorrelation."""
    start = time.time()
    N = h0.shape[0]
    h = h0.copy()
    a = compute_autocorr_bin(h)
    min_a = a.min()
    best_h = h.copy()
    best_min = min_a

    while True:
        if time.time() - start >= budget:
            break

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break

        batch = 300
        best_improve = 0
        best_swap = None
        best_delta = None
        best_new_min = min_a

        for _ in range(batch):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            delta = delta_autocorr_swap(i, j, ones, N)
            new_min = (a + delta).min()
            improve = new_min - min_a
            if improve > best_improve:
                best_improve = improve
                best_swap = (i, j)
                best_delta = delta
                best_new_min = new_min
                if improve >= 1:
                    # cannot beat this improvement in this batch
                    pass

        if best_improve > 0:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            a += best_delta
            min_a = best_new_min
            if min_a > best_min:
                best_min = min_a
                best_h = h.copy()
        else:
            # no improving swap found in this batch
            break

    return best_h, best_min

def thue_morse_binary(n):
    """Binary Thue–Morse sequence of length n."""
    tm = np.empty(n, dtype=int)
    for k in range(n):
        tm[k] = bin(k).count('1') & 1
    return tm

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    n = 256                     # grid size (smaller grid allows deeper search)
    best_h = None
    best_min = -1

    candidates = []

    # candidate from parent construction, if available
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        factor = n // init_full.size
        if factor * init_full.size == n:
            up = np.repeat(init_full, factor)
        else:
            x_old = np.linspace(0, 1, init_full.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            up = np.interp(x_new, x_old, init_full)
        idx = np.argpartition(up, -n // 2)[-n // 2:]
        h_bin = np.zeros(n, dtype=int)
        h_bin[idx] = 1
        candidates.append(h_bin)

    # deterministic Thue‑Morse start
    candidates.append(thue_morse_binary(n))

    # a few random starts
    for _ in range(3):
        h_rand = np.zeros(n, dtype=int)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand[idx] = 1
        candidates.append(h_rand)

    # main search loop
    while time_left() > 5.0:
        if candidates:
            h0 = candidates.pop(0)
        else:
            h0 = np.zeros(n, dtype=int)
            idx = rng.choice(n, size=n // 2, replace=False)
            h0[idx] = 1

        remaining = time_left()
        # allocate up to 30 s (or half of remaining) for this hill‑climb
        run_budget = min(30.0, remaining * 0.5)
        h_opt, min_a_opt = hill_climb_binary(h0, run_budget, rng)

        if min_a_opt > best_min:
            best_min = min_a_opt
            best_h = h_opt.copy()

    # fallback in the unlikely event that no candidate succeeded
    if best_h is None:
        best_h = np.zeros(n, dtype=int)
        idx = rng.choice(n, size=n // 2, replace=False)
        best_h[idx] = 1
        best_min = compute_autocorr_bin(best_h).min()

    # final C5 bound from the maximal overlap formula
    c5 = 1.0 - (2.0 * best_min) / n

    return best_h.astype(float), c5, n
