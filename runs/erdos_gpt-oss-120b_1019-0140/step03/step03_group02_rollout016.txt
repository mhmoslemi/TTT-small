def genetic_algorithm(n, rng, time_left, target):
    popsize = 50
    elite_size = 2
    mut_rate = 0.02
    # Prepare initial population
    pop = []
    # Add best known candidate from init_h if present
    if initial_h_values present:
        init_binary = np.asarray(initial_h_values, dtype=float)
        # Convert to binary: top half 1? Actually initial could be fractional. We'll binarize: threshold 0.5 maybe or keep as is? We'll just round to nearest.
        init_binary = (init_binary >= 0.5).astype(float)
        # ensure balanced
        init_binary = repair_balance(init_binary, n, rng)
        pop.append(init_binary)
    # Add thue_morse
    tm = thue_morse(n).astype(float)
    pop.append(tm)
    # Add some random starts
    while len(pop) < popsize:
        v = random_balanced_binary(n, rng)
        pop.append(v)

    # Evaluate
    fitness = [compute_max_corr(v) for v in pop]
    best_idx = np.argmin(fitness)
    best_vec = pop[best_idx].copy()
    best_f = fitness[best_idx]

    generation = 0
    while time_left() > 0.5:  # leave time for SA
        generation += 1
        # select parents via tournament
        new_pop = []

        # elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_pop.append(pop[idx].copy())

        # generate rest
        while len(new_pop) < popsize:
            # tournament selection
            cand = rng.choice(popsize, size=3, replace=False)
            parent1 = pop[cand[np.argmin([fitness[i] for i in cand])]]
            cand2 = rng.choice(popsize, size=3, replace=False)
            parent2 = pop[cand2[np.argmin([fitness[i] for i in cand2])]]

            # uniform crossover
            mask = rng.random(n) < 0.5
            child = np.where(mask, parent1, parent2).astype(float)

            # repair balance
            child = repair_balance(child, n, rng)

            # mutation: flip each bit with prob mut_rate
            if rng.random() < mut_rate:
                # flip a random bit? Or multiple bits
                # We'll flip random bits: choose random indices, flip them, then repair.
                flip_count = max(1, int(mut_rate * n))
                flip_idx = rng.choice(n, size=flip_count, replace=False)
                child[flip_idx] = 1 - child[flip_idx]

                child = repair_balance(child, n, rng)

            new_pop.append(child)

        pop = new_pop
        # evaluate fitness
        fitness = [compute_max_corr(v) for v in pop]

        # update best
        idx = np.argmin(fitness)
        if fitness[idx] < best_f:
            best_f = fitness[idx]
            best_vec = pop[idx].copy()
            if best_f * (2 / n) <= target:
                break

    return best_vec, best_f

Important: repair_balance ensures sum = n/2 exactly. Implementation:

def repair_balance(v, n, rng):
    target = n // 2
    current = int(v.sum())
    if current == target:
        return v
    diff = target - current
    if diff > 0:
        # need more ones: flip diff zeros to 1
        zero_idx = np.where(v == 0)[0]
        flip = rng.choice(zero_idx, size=diff, replace=False)
        v[flip] = 1
    else:
        # need fewer ones: flip -diff ones to 0
        one_idx = np.where(v == 1)[0]
        flip = rng.choice(one_idx, size=-diff, replace=False)
        v[flip] = 0
    return v

def hill_climb_sa(h, rng, time_left, target):
    n = h.shape[0]
    best = h.copy()
    best_corr = compute_max_corr(best)
    cur = best.copy()
    cur_corr = best_corr
    it = 0
    # total allowed iters based on time, but we just loop until time.

        it += 1

            i = rng.integers(0, n)
        j = rng.integers(0, n)

            j = rng.integers(0, n)
        new = cur.copy()
        new[i] = 1
        new[j] = 0

        delta = new_corr - cur_corr
        # temperature schedule: start high and fade
        # compute fraction of total time used: maybe use it or remaining time.

def hill_climb_sa(h, rng, time_left, target_corr):
    best = h.copy()
    best_corr = compute_max_corr(best)
    cur = best.copy()
    cur_corr = best_corr
    n = h.size
    it = 0
    while time_left() > 0.01:
        it += 1
        # randomly pick a swap (0->1 and 1->0)
        i = rng.integers(0, n)
        while cur[i] != 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while cur[j] != 1:
            j = rng.integers(0, n)
        # propose
        new = cur.copy()
        new[i] = 1
        new[j] = 0
        new_corr = compute_max_corr(new)
        delta = new_corr - cur_corr
        # temperature schedule: start at 0.02 and decay slowly
        temp = max(0.001, 0.02 / np.sqrt(it))
        # accept if improvement or probabilistically
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur = new
            cur_corr = new_corr
            if cur_corr < best_corr:
                best_corr = cur_corr
                best = cur.copy()
                if best_corr <= target_corr:
                    break
    return best, best_corr

import time
import numpy as np

def compute_max_corr(h: np.ndarray) -> int:
    """Maximum of the overlap integral (without dx) for a binary vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(corr.max())

def thue_morse(n: int) -> np.ndarray:
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)

def random_balanced_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Balanced binary vector with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return v

def repair_balance(v: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """Force sum(v) == n/2 by flipping the minimal number of bits."""
    target = n // 2
    cur = int(v.sum())
    if cur == target:
        return v
    diff = target - cur
    if diff > 0:
        zero_idx = np.where(v == 0)[0]
        flip = rng.choice(zero_idx, size=diff, replace=False)
        v[flip] = 1.0
    else:
        one_idx = np.where(v == 1)[0]
        flip = rng.choice(one_idx, size=-diff, replace=False)
        v[flip] = 0.0
    return v

def init_from_fractional(v: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """From a (possibly non‑binary) vector create a balanced binary one."""
    if v.size != n:
        factor = n // v.size
        v = np.repeat(v, factor)
        if v.size < n:
            v = np.concatenate([v, v[: n - v.size]])
    idx = np.argsort(v)[::-1]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx[: n // 2]] = 1.0
    return repair_balance(bin_vec, n, rng)

def genetic_algorithm(n: int, rng: np.random.Generator,
                      time_left, target_corr: int):
    popsize = 50
    elite_size = 5
    mut_rate = 0.02
    random_replace_prob = 0.12

    pop = []

    # seed from initial_h_values if present
    if "initial_h_values" in globals():
        init_vec = init_from_fractional(
            np.asarray(initial_h_values, dtype=float), n, rng)
        pop.append(init_vec)

    # seed from Thue‑Morse
    pop.append(thue_morse(n))

    # fill the rest
    while len(pop) < popsize:
        pop.append(random_balanced_binary(n, rng))

    fitness = [compute_max_corr(v) for v in pop]
    best_idx = int(np.argmin(fitness))
    best_vec = pop[best_idx].copy()
    best_corr = fitness[best_idx]

    # main GA loop
    while time_left() > 5.0:
        # elitism
        elite_idx = np.argsort(fitness)[:elite_size]
        new_pop = [pop[i].copy() for i in elite_idx]

        # generate offspring
        while len(new_pop) < popsize:
            if rng.random() < random_replace_prob:
                child = random_balanced_binary(n, rng)
            else:
                p1 = pop[rng.integers(popsize)]
                p2 = pop[rng.integers(popsize)]
                mask = rng.random(n) < 0.5
                child = np.where(mask, p1, p2).astype(float)
                child = repair_balance(child, n, rng)

                if rng.random() < mut_rate:
                    flip_num = max(1, int(mut_rate * n))
                    flips = rng.choice(n, size=flip_num, replace=False)
                    child[flips] = 1.0 - child[flips]
                    child = repair_balance(child, n, rng)
            new_pop.append(child)

        pop = new_pop
        fitness = [compute_max_corr(v) for v in pop]

        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_corr:
            best_corr = fitness[cur_best]
            best_vec = pop[cur_best].copy()
            if best_corr <= target_corr:
                break

    return best_vec, best_corr

def local_optimize_max_shift(h: np.ndarray, rng: np.random.Generator,
                            time_left, target_corr: int):
    """Focused hill‑climbing that reduces the worst shift."""
    n = h.size
    best = h.copy()
    best_corr = compute_max_corr(best)

    while time_left() > 0.01 and best_corr > target_corr:
        corr = np.correlate(best, 1.0 - best, mode="full")
        max_val = int(corr.max())
        if max_val <= target_corr:
            break
        indices = np.where(corr == max_val)[0]
        k_offset = rng.choice(indices)
        shift = k_offset - (n - 1)

        # collect indices i where best[i]=1 and best[i+shift]=0
        contrib = []
        if shift >= 0:
            for i in range(0, n - shift):
                if best[i] == 1.0 and best[i + shift] == 0.0:
                    contrib.append(i)
        else:
            s = -shift
            for i in range(s, n):
                if best[i] == 1.0 and best[i - s] == 0.0:
                    contrib.append(i)

        if not contrib:
            # fallback to a random swap
            i0 = rng.integers(0, n)
            while best[i0] != 0.0:
                i0 = rng.integers(0, n)
            j0 = rng.integers(0, n)
            while best[j0] != 1.0:
                j0 = rng.integers(0, n)
            cand = best.copy()
            cand[i0] = 1.0
            cand[j0] = 0.0
        else:
            i = rng.choice(contrib)
            zero_idx = np.where(best == 0.0)[0]
            j = rng.choice(zero_idx)
            cand = best.copy()
            cand[i] = 0.0
            cand[j] = 1.0

        new_corr = compute_max_corr(cand)
        if new_corr < best_corr:
            best = cand
            best_corr = new_corr

    return best, best_corr

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    n = 256                     # number of discretisation points
    target_bound = 0.3808
    target_corr = int(np.floor(target_bound * (n / 2)))  # <=48 for n=256

    # Phase 1: genetic algorithm
    best_vec, best_corr = genetic_algorithm(
        n, rng, time_left, target_corr)

    # Phase 2: targeted local improvement
    if time_left() > 2.0:
        best_vec, best_corr = local_optimize_max_shift(
            best_vec, rng, time_left, target_corr)

    # final bound
    c5_bound = best_corr * (2.0 / n)

    # ensure output conforms to constraints
    best_vec = repair_balance(best_vec, n, rng)

    return best_vec, float(c5_bound), n
