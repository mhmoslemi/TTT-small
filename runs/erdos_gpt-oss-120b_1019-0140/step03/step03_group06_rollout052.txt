import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Scale
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute residual iteratively
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    # target integer
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def mutate(h, rng, max_delta=0.2):
    n = h.shape[0]
    # choose source and target
    src = rng.choice(np.where(h > 1e-12)[0])
    dst = rng.choice(np.where(h < 1.0 - 1e-12)[0])
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0:
        return h.copy()
    delta = rng.random() * max_transfer * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h

def crossover(p1, p2, rng):
    alpha = rng.random()
    child = alpha * p1 + (1-alpha) * p2
    return child

def ga_optimize(n, target, rng, deadline):
    pop_size = 30
    elite = 5
    # initialize population
    pop = []
    # include initial if available later only
    for _ in range(pop_size):
        # random continuous or binary mix
        if rng.random() < 0.5:
            v = random_continuous_vector(n, target, rng)
        else:
            v = random_binary_vector(n, target, rng)
        pop.append(v)
    pop = np.array(pop)
    # evaluate
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # Main loop
    while time.time() < deadline - 2.0:
        # select elites
        elite_idx = np.argsort(scores)[:elite]
        elites = pop[elite_idx]
        # generate new population
        new_pop = []
        # keep elites
        for e in elites:
            new_pop.append(e)
        # fill rest
        while len(new_pop) < pop_size:
            # select two parents from elite (or random)
            parents = rng.choice(elites, size=2, replace=False)
            child = crossover(parents[0], parents[1], rng)
            # mutation
            if rng.random() < 0.8:
                child = mutate(child, rng)
            # enforce feasibility
            child = enforce_sum(child, target, rng)
            new_pop.append(child)
        pop = np.array(new_pop)
        scores = np.array([compute_c5(ind) for ind in pop])
        cur_best_idx = np.argmin(scores)
        if scores[cur_best_idx] < best_score:
            best_score = scores[cur_best_idx]
            best_h = pop[cur_best_idx].copy()
            # early stop if good enough
            if best_score < 0.3808:
                break
    return best_h, best_score

def anneal_optimize(h0, cur_score, rng, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.2:
        # compute full correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx
        # compute shift
        k = max_idx - (n - 1)
        # compute contributions array for that shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            idx_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            idx_range = np.arange(kk, n)
        # choose i where contribution high
        i_local = np.argmax(prod)
        i = idx_range[i_local]
        # pick j random with headroom
        candidates = np.where(h < 1.0 - 1e-12)[0]
        if len(candidates) == 0:
            break
        j = rng.choice(candidates)
        # ensure j != i
        if j == i:
            # pick different
            cand2 = np.where(h > 1e-12)[0]
            if len(cand2) <= 1:
                break
            j = rng.choice(cand2[cand2 != i])
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        # acceptance
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            if best_score < 0.3808:
                break
        if new_score <= max_val:
            # accept if not worse than previous max?
            # simple Metropolis
            accept = True
        else:
            # Simulated annealing temperature schedule
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            accept = np.exp((max_val - new_score) / T) > rng.random()
        if accept:
            # keep new h
            pass
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)
    target = n / 2.0
    # Ensure feasibility
    h0 = enforce_sum(h0, target, rng)
    # Initial best
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    # Run GA
    ga_deadline = deadline * 0.6  # allocate 60% time
    if time.time() < ga_deadline:
        h_ga, score_ga = ga_optimize(n, target, rng, ga_deadline)
        if score_ga < best_score:
            best_h, best_score = h_ga, score_ga
    # Run annealing on best
    if time.time() < deadline - 0.5:
        h_ann, score_ann = anneal_optimize(best_h, best_score, rng, deadline)
        if score_ann < best_score:
            best_h, best_score = h_ann, score_ann
    # Final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_score = compute_c5(best_h)
    return best_h.astype(float), final_score, n

def anneal_optimize(h0, init_score, rng, deadline):
    h = h0.copy()
    cur_score = init_score
    best_h = h.copy()
    best_score = cur_score
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        # find worst shift and possibly its contribution
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_score = corr[max_idx] * dx
        # if cur_score > best_score maybe continue? We'll try move
        # Determine shift
        k = max_idx - (n - 1)
        # compute contributions via prod
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            idx_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            idx_range = np.arange(kk, n)
        # choose index with largest product
        i_local = np.argmax(prod)
        i = idx_range[i_local]
        # choose random destination with headroom
        candidates = np.where(h < 1.0 - 1e-12)[0]
        if len(candidates) == 0:
            break
        j = rng.choice(candidates)
        # ensure j != i
        if j == i:
            # pick other
            cand2 = np.where(h > 1e-12)[0]
            if len(cand2) == 0:
                break
            j = rng.choice(cand2)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        # acceptance
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            if best_score < 0.3808:
                break
        # Metropolis acceptance based on new_score vs cur_score (previous before move)
        if new_score <= cur_score:
            accept = True
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            accept = np.exp((cur_score - new_score) / T) > rng.random()
        if accept:
            cur_score = new_score
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_score
```

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Project v into [0,1] and adjust to have exact sum == target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v


def random_continuous_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return enforce_sum(v, target, rng)


def random_binary_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, ones, replace=False)] = 1.0
    return v


# ---------- mutation & crossover ----------
def mutate(h: np.ndarray, rng: np.random.Generator, max_frac: float = 0.2) -> np.ndarray:
    """Transfer a random fraction of mass from a positive entry to a non‑full entry."""
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h.copy()
    delta = rng.random() * max_transfer * max_frac
    out = h.copy()
    out[i] -= delta
    out[j] += delta
    return out


def crossover(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.random()
    return alpha * p1 + (1.0 - alpha) * p2


# ---------- genetic algorithm ----------
def ga_optimize(
    n: int,
    target: float,
    rng: np.random.Generator,
    deadline: float,
    init_h: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    pop_size = 30
    elite_cnt = 5

    # build initial population
    pop = []
    if init_h is not None:
        pop.append(init_h.copy())
    # constant 0.5 pattern (feasible)
    const = np.full(n, 0.5)
    const = enforce_sum(const, target, rng)
    pop.append(const)

    while len(pop) < pop_size:
        if rng.random() < 0.5:
            v = random_continuous_vector(n, target, rng)
        else:
            v = random_binary_vector(n, target, rng)
        pop.append(v)

    pop = np.array(pop[:pop_size])
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline - 0.5:
        elite_idx = np.argsort(scores)[:elite_cnt]
        elites = pop[elite_idx]

        new_pop = elites.tolist()
        while len(new_pop) < pop_size:
            parents = rng.choice(elites, size=2, replace=False)
            child = crossover(parents[0], parents[1], rng)
            if rng.random() < 0.9:
                child = mutate(child, rng)
            child = enforce_sum(child, target, rng)
            new_pop.append(child)

        pop = np.array(new_pop)
        scores = np.array([compute_c5(ind) for ind in pop])
        cur_best = np.argmin(scores)
        if scores[cur_best] < best_score:
            best_score = scores[cur_best]
            best_h = pop[cur_best].copy()
            if best_score < 0.3808:
                break

    return best_h, best_score


# ---------- annealing with smart move ----------
def anneal_optimize(
    h0: np.ndarray,
    init_score: float,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    h = h0.copy()
    best_h = h.copy()
    best_score = init_score
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        dx = 2.0 / n
        cur_score = corr[max_idx] * dx
        k = max_idx - (n - 1)

        # locate i with largest contribution for this shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = int(np.argmax(prod))
        i = i_range[i_local]

        # choose destination j
        headroom = np.where(h < 1.0 - 1e-12)[0]
        if headroom.size == 0:
            break

        if k >= 0:
            out_of_range = np.where(np.arange(n) >= n - k)[0]
        else:
            out_of_range = np.where(np.arange(n) < -k)[0]

        j_candidates = np.intersect1d(headroom, out_of_range, assume_unique=True)
        if j_candidates.size > 0:
            j = rng.choice(j_candidates)
        else:
            # pick j that maximises h[j+k] - h[i+k] (if defined)
            if k >= 0:
                mask = headroom + k < n
                cand = headroom[mask]
                if cand.size > 0:
                    diffs = h[cand + k] - h[i + k]
                    j = cand[int(np.argmax(diffs))]
                else:
                    j = rng.choice(headroom)
            else:
                mask = headroom + k >= 0
                cand = headroom[mask]
                if cand.size > 0:
                    diffs = h[cand + k] - h[i + k]
                    j = cand[int(np.argmax(diffs))]
                else:
                    j = rng.choice(headroom)

        if i == j:
            continue

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_score = compute_c5(h)

        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            if best_score < 0.3808:
                break

        # Metropolis acceptance
        if new_score <= cur_score:
            accept = True
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            accept = np.exp((cur_score - new_score) / T) > rng.random()

        if accept:
            cur_score = new_score
        else:
            h[i] += delta
            h[j] -= delta

    return best_h, best_score


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs) -> tuple[np.ndarray, float, int]:
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    h0 = enforce_sum(h0, target, rng)
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # GA phase (≈60% of time)
    ga_deadline = start + 0.6 * budget_s
    if time.time() < ga_deadline - 1.0:
        h_ga, s_ga = ga_optimize(n, target, rng, ga_deadline, init_h=best_h)
        if s_ga < best_score:
            best_h, best_score = h_ga, s_ga

    # annealing phase with remaining budget
    if time.time() < deadline - 0.5:
        h_ann, s_ann = anneal_optimize(best_h, best_score, rng, deadline)
        if s_ann < best_score:
            best_h, best_score = h_ann, s_ann

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_score = compute_c5(best_h)

    return best_h.astype(float), final_score, n
