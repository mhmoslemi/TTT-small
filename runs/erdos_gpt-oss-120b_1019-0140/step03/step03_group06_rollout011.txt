import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # simple iterative adjustment
    for _ in range(1000):
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    # final clip for safety
    v = np.clip(v, 0.0, 1.0)
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    # target must be integer or close
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def tournament_select(pop, fitness, rng, size=3):
    idx = rng.choice(len(pop), size=size, replace=False)
    best_idx = idx[0]
    best_fit = fitness[best_idx]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best_fit = fitness[i]
            best_idx = i
    return pop[best_idx]

def uniform_crossover(p1, p2, rng):
    mask = rng.random(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(child, target, rng, p_mut=0.2, max_shift=0.5):
    # with probability p_mut perform a mass-transfer mutation
    if rng.random() < p_mut:
        # find source with some mass, destination with headroom
        src = np.where(child > 1e-12)[0]
        dst = np.where(child < 1.0 - 1e-12)[0]
        if len(src) > 0 and len(dst) > 0:
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(child[i], 1.0 - child[j])
            if max_transfer > 0:
                delta = max_transfer * rng.random() * max_shift
                child[i] -= delta
                child[j] += delta
    # small Gaussian noise could also be added
    # noise = rng.normal(scale=0.01, size=child.shape)
    # child = np.clip(child + noise, 0.0, 1.0)
    child = enforce_sum(child, target, rng)
    return child

def directed_anneal(h, cur_val, target, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline:
        # compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # build product for this shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        j = i + k
        # fallback if cannot transfer
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                break
        delta = max_transfer * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if new_val < cur_val:
            cur_val = new_val
            # If we already beat target threshold, early exit
            if cur_val < 0.3808:
                break
        else:
            remaining = deadline - time.time()
            if remaining <= 0:
                # revert before break
                h[i] += delta
                h[j] -= delta
                break
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
    return h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)
    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n/2.0, rng)
    target = n / 2.0
    # Ensure initial h0 feasible
    h0 = enforce_sum(h0, target, rng)
    # Initialize population
    popsize = 60
    population = [h0.copy()]
    fitness = [compute_c5(h0)]
    # add random binaries and continuous
    while len(population) < popsize:
        if rng.random() < 0.5:
            v = random_binary_vector(n, target, rng)
        else:
            v = random_continuous_vector(n, target, rng)
        population.append(v)
        fitness.append(compute_c5(v))
    # track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]
    # GA loop
    # allocate time: leave ~30 seconds for directed annealing
    ga_deadline = deadline - 30.0
    while time.time() < ga_deadline:
        # produce offspring
        offspring = []
        offspring_fitness = []
        # generate popsize // 2 pairs => popsize children (or 2*popsize)
        # let's produce popsize children
        for _ in range(popsize):
            p1 = tournament_select(population, fitness, rng, size=3)
            p2 = tournament_select(population, fitness, rng, size=3)
            child = uniform_crossover(p1, p2, rng)
            child = mutate(child, target, rng, p_mut=0.3, max_shift=0.5)
            offspring.append(child)
            offspring_fitness.append(compute_c5(child))
        # combine and select next generation
        combined = population + offspring
        combined_fitness = fitness + offspring_fitness
        # select best popsize
        idx_sort = np.argsort(combined_fitness)
        population = [combined[i] for i in idx_sort[:popsize]]
        fitness = [combined_fitness[i] for i in idx_sort[:popsize]]
        # update best
        if fitness[0] < best_val:
            best_val = fitness[0]
            best_h = population[0].copy()
            if best_val < 0.3808:
                break
    # final directed annealing refinement
    if time.time() < deadline:
        cur_h = best_h.copy()
        cur_val = best_val
        cur_h, cur_val = directed_anneal(cur_h, cur_val, target, deadline, rng)
        best_h, best_val = cur_h, cur_val
    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust to meet exact sum target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(1000):
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return np.clip(v, 0.0, 1.0)

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def tournament_select(pop, fitness, rng, size=3):
    idx = rng.choice(len(pop), size=size, replace=False)
    best = idx[0]
    best_fit = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best = i
            best_fit = fitness[i]
    return pop[best]

def uniform_crossover(p1, p2, rng):
    mask = rng.random(p1.size) < 0.5
    return np.where(mask, p1, p2)

def mutate(child, target, rng, p_mut=0.4, max_shift=0.5):
    if rng.random() < p_mut:
        src = np.where(child > 1e-12)[0]
        dst = np.where(child < 1.0 - 1e-12)[0]
        if len(src) and len(dst):
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(child[i], 1.0 - child[j])
            if max_transfer > 0:
                delta = max_transfer * rng.random() * max_shift
                child[i] -= delta
                child[j] += delta
    return enforce_sum(child, target, rng)

def directed_anneal(h, cur_val, target, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        j = i + k
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                break
        delta = max_transfer * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        if new_val < cur_val:
            cur_val = new_val
            if cur_val < 0.3808:
                break
        else:
            remaining = deadline - time.time()
            if remaining <= 0:
                h[i] += delta
                h[j] -= delta
                break
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine number of points
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    h0 = enforce_sum(h0, target, rng)

    # Population initialization
    popsize = 80
    population = [h0.copy()]
    fitness = [compute_c5(h0)]
    while len(population) < popsize:
        if rng.random() < 0.5:
            v = random_binary_vector(n, target, rng)
        else:
            v = random_continuous_vector(n, target, rng)
        population.append(v)
        fitness.append(compute_c5(v))

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]

    # GA phase (reserve time for annealing)
    anneal_time = 120.0
    ga_deadline = deadline - anneal_time
    while time.time() < ga_deadline:
        # generate offspring
        offspring = []
        offspring_fitness = []
        for _ in range(popsize):
            p1 = tournament_select(population, fitness, rng, size=3)
            p2 = tournament_select(population, fitness, rng, size=3)
            child = uniform_crossover(p1, p2, rng)
            child = mutate(child, target, rng, p_mut=0.4, max_shift=0.5)
            offspring.append(child)
            offspring_fitness.append(compute_c5(child))
        # combine and select next generation
        combined = population + offspring
        combined_fitness = fitness + offspring_fitness
        combined_fitness_arr = np.array(combined_fitness)
        idx_sort = np.argsort(combined_fitness_arr)
        population = [combined[i] for i in idx_sort[:popsize]]
        fitness = [combined_fitness_arr[i] for i in idx_sort[:popsize]]
        # update best
        if fitness[0] < best_val:
            best_val = fitness[0]
            best_h = population[0].copy()
            if best_val < 0.3808:
                break

    # Directed annealing refinement
    if time.time() < deadline:
        cur_h = best_h.copy()
        cur_val = best_val
        cur_h, cur_val = directed_anneal(cur_h, cur_val, target, deadline, rng)
        best_h, best_val = cur_h, cur_val

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
