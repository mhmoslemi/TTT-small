def gradient_descent(h0, max_iters, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    for it in range(1, max_iters+1):
        grad = gradient_subgrad(h)
        m = beta*m + (1-beta)*grad
        eta = step0/np.sqrt(it)
        h_new = h - eta*m
        # add small random noise maybe

        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_refine(h0, max_iter=200):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0,1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h,1.0-h,mode='full') * (2.0/n)
        return t - np.max(corr)
    cons = [{'type':'eq','fun':eq_sum},{'type':'ineq','fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
        constraints=cons, options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    h_opt = res.x[:n]
    # enforce bounds and sum
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_subgrad(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        grad += 1.0
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iters, step0, beta, rng):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        grad = grad_subgrad(h)
        m = beta * m + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * m
        if it % 5000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def slsqp_refine(h0, max_iter=200):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

    if remaining > 12:
        n = best_h.size
        idx = np.argsort(best_h)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(min(200000, remaining * 300))
        if max_iter_ann > 0:
            h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                          temp_start=0.01, temp_end=1e-6,
                                          rng=rng)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()

    # final SLSQP refinement
    remaining = time_left()
    if remaining > 5:
        h_opt, sc_opt = slsqp_refine(best_h, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
