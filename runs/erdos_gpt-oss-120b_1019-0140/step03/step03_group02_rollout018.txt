def smooth_adam(h0, n_iter, step0, alpha_start, alpha_end, beta1, beta2, epsilon, time_left):

    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # enlarge interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term (averaged over given integer shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def smooth_adam(h0, n_iter, step0, alpha_start, alpha_end,
                beta1, beta2, epsilon, time_left):
    """Adam on the soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def max_corr_adam(h0, n_iter, step0, beta1, beta2, epsilon, time_left):
    """Adam on the exact max‑correlation gradient (averaged over maximal shifts)."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=None):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(500000, max(1, time_left() * 5000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick a zero and a one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * max(0.0, 1.0 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if target is not None and best_score <= target:
                    break
    return best_h, best_score

def generate_initial_candidates(n, rng):
    candidates = []
    if "initial_h_values" in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            init_up = upsample(init_full, n)
        else:
            init_up = init_full.copy()
        init_up = project_onto_box_sum(init_up, n / 2.0)
        candidates.append(init_up)
    candidates.append(thue_morse(n).astype(float))
    for _ in range(5):
        candidates.append(random_binary_balanced(n, rng).astype(float))
    for _ in range(5):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # parameters
    n_initial = 256
    n_final = 512
    target_bound = 0.3808

    # --- Phase 0: generate candidates on initial grid ---
    candidates = generate_initial_candidates(n_initial, rng)

    best_h = None
    best_score = float('inf')

    # --- Phase 1: continuous optimisation on initial grid ---
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_smooth, sc_smooth = smooth_adam(
            h0,
            n_iter=4000,
            step0=0.05,
            alpha_start=1.0,
            alpha_end=30.0,
            beta1=0.9,
            beta2=0.999,
            epsilon=1e-8,
            time_left=time_left)
        h_ref, sc_ref = max_corr_adam(
            h_smooth,
            n_iter=4000,
            step0=0.05,
            beta1=0.9,
            beta2=0.999,
            epsilon=1e-8,
            time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        # early exit if we already beat the target
        if best_score <= target_bound:
            break

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_initial, 0.5), n_initial / 2.0)
        best_score = compute_c5(best_h)

    # --- Phase 2: upscale to finer grid and refine ---
    if time_left() > 10.0 and n_final <= 1000:
        h_up = upsample(best_h, n_final)
        h_up = project_onto_box_sum(h_up, n_final / 2.0)
        if time_left() > 5.0:
            h_smooth2, sc_smooth2 = smooth_adam(
                h_up,
                n_iter=4000,
                step0=0.05,
                alpha_start=1.0,
                alpha_end=30.0,
                beta1=0.9,
                beta2=0.999,
                epsilon=1e-8,
                time_left=time_left)
            h_ref2, sc_ref2 = max_corr_adam(
                h_smooth2,
                n_iter=4000,
                step0=0.05,
                beta1=0.9,
                beta2=0.999,
                epsilon=1e-8,
                time_left=time_left)
            if sc_ref2 < best_score:
                best_score, best_h = sc_ref2, h_ref2.copy()

    # --- Phase 3: optional binary hill‑climb for final polishing ---
    if time_left() > 2.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=target_bound)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
