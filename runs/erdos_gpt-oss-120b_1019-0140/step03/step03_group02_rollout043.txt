import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[: n - s])
            grad[: n - s] -= h[s:]
    grad /= len(shifts)
    return grad


def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        exp_vals = np.exp(alpha * (corr - maxc))
        weights = exp_vals / np.sum(exp_vals)
        grad = np.zeros_like(h)
        for sft, w in enumerate(weights):
            if w == 0.0:
                continue
            k = sft - (n - 1)
            if k >= 0:
                grad[: n - k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[: n - k]
            else:
                p = -k
                grad[p:] += w * (1.0 - h[: n - p])
                grad[: n - p] -= w * h[p:]
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points (periodic not assumed)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # resolution
    n = 512  # finer grid for better approximation
    target_bound = 0.3808

    # build initial candidates
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init_full, n)
        candidates.append(project_onto_box_sum(init_up, n / 2.0))

    # sinusoidal shape
    x = np.arange(n) / n
    sinus = 0.5 * (1.0 + np.sin(np.pi * x))
    candidates.append(project_onto_box_sum(sinus, n / 2.0))

    # triangular shape
    tri = np.minimum(x * 2, (1.0 - x) * 2)
    candidates.append(project_onto_box_sum(tri, n / 2.0))

    # random continuous vectors
    for _ in range(3):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    best_h = None
    best_score = float('inf')

    # Phase 0: quick smooth descent on each candidate
    for h0 in candidates:
        if time_left() < 5.0:
            break
        nit = min(1500, int(max(200, time_left() * 1.5)))
        h_opt, sc_opt = smooth_grad_descent(
            h0, n_iter=nit, step0=0.7,
            alpha_start=1.0, alpha_end=20.0,
            beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), n / 2.0)
        best_score = compute_c5(best_h)

    # Phase 1: extensive momentum refinement
    if time_left() > 4.0:
        nit = int(min(30000, max(8000, time_left() * 900)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=nit,
                                              step0=0.5, beta=0.9,
                                              time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Phase 2: binary hill‑climb (optional)
    if time_left() > 2.5:
        n_cur = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n_cur, dtype=float)
        binary[order[n_cur // 2 :]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left,
                                             target=target_bound)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
        if time_left() > 1.0:
            h_ref2, sc_ref2 = momentum_grad_descent(best_h, n_iter=5000,
                                                   step0=0.3, beta=0.9,
                                                   time_left=time_left)
            if sc_ref2 < best_score:
                best_score, best_h = sc_ref2, h_ref2.copy()

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
