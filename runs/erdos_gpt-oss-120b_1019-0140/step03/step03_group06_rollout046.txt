def project_simplex_box(v, target):
    lo = -max(v) - 1  # because subtracting λ maybe.
    hi = max(v) + 1
    for _ in range(50):

        p = np.clip(v - mid, 0, 1)

        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - hi, 0, 1)
    return p

import time

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_to_sum(v, target):
    """Project vector v onto the box [0,1]^n with exact sum == target."""
    # Bisection on λ for p_i = clamp(v_i - λ, 0, 1)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):

        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0.0, 1.0)
