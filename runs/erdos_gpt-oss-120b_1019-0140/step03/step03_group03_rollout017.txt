def hill_climb_binary(b, max_iter, rng):
    n = b.size
    half = n // 2
    best = b.copy()
    best_score = compute_c5(b)
    for it in range(max_iter):
        # compute correlation and find shift(s) with minimal A (largest S)
        corr = np.correlate(b, b, mode='full')
        A = corr[len(b)-1:]  # shape n
        # S = half - A
        worst_shifts = np.where(A == np.min(A))[0]  # the smallest A
        # Choose one shift (maybe random among worst)
        shift = rng.choice(worst_shifts)
        # compute v for each index for this shift
        forward = np.concatenate([b[shift:], np.zeros(shift, dtype=b.dtype)])
        backward = np.concatenate([np.zeros(shift, dtype=b.dtype), b[:n-shift]])
        v = forward + backward
        # find best swap candidate
        ones_idx = np.flatnonzero(b > 0.5)  # b is 0/1
        zeros_idx = np.flatnonzero(b < 0.5)
        # if no ones or zeros: break
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        # find i with minimal v among ones
        i = ones_idx[np.argmin(v[ones_idx])]
        # find j with maximal v among zeros
        j = zeros_idx[np.argmax(v[zeros_idx])]
        if v[j] <= v[i]:
            # no improving swap for this shift, try random swap to escape
            # random swap
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
        # swap
        b[i], b[j] = 0, 1
        new_score = compute_c5(b)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best = b.copy()
        else:
            # revert if not improve (or maybe accept with small prob)
            # but we can revert to maintain monotonic improvement
            b[i], b[j] = 1, 0
        # early exit if best_score <= target maybe? but we just continue
    return best, best_score

    half = n // 2
    best = b.copy()
    best_score = compute_c5(b)
    for _ in range(max_iter):

        corr = np.correlate(b, b, mode='full')
        A = corr[n-1:]  # length n
        min_A = np.min(A)
        # shift(s) with minimal A
        shifts = np.where(A == min_A)[0]
        shift = rng.choice(shifts)

        # forward: b[i+shift] for i where i+shift < n else 0
        if shift == 0:

            backward = b.copy()

            backward = np.concatenate([np.zeros(shift, dtype=b.dtype), b[:n-shift]])

        # indices of ones and zeros
        ones_idx = np.flatnonzero(b > 0.5)
        zeros_idx = np.flatnonzero(b < 0.5)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        # candidate i: 1 with minimal v
        i = ones_idx[np.argmin(v[ones_idx])]
        # candidate j: 0 with maximal v
        j = zeros_idx[np.argmax(v[zeros_idx])]
        # If no improvement for this shift, try random swap
        if v[j] <= v[i]:
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
        # Perform swap
        b[i], b[j] = 0.0, 1.0
        new_score = compute_c5(b)
        if new_score < best_score - 1e-12:

            best = b.copy()
        else:
            # revert swap
            b[i], b[j] = 1.0, 0.0
        # early exit if best_score already very low to avoid waste
    return best, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient (subgradient) of the max‑correlation term."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def resample(arr, new_n):
    """Linear interpolation up‑ or down‑sample."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    b = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    b[idx] = 1.0
    return b


def hill_climb_binary(b, max_iter, rng):
    """Deterministic swap hill‑climbing for binary balanced vectors."""
    n = b.size
    best = b.copy()
    best_score = compute_c5(b)
    for _ in range(max_iter):
        corr = np.correlate(b, b, mode='full')
        A = corr[n - 1:]                    # linear self‑correlation
        min_A = np.min(A)
        shifts = np.where(A == min_A)[0]    # worst shifts (minimum overlap)
        shift = rng.choice(shifts)

        if shift == 0:
            forward = b.copy()
            backward = b.copy()
        else:
            forward = np.concatenate([b[shift:], np.zeros(shift, dtype=b.dtype)])
            backward = np.concatenate([np.zeros(shift, dtype=b.dtype), b[:n - shift]])
        v = forward + backward                # contribution of each index to this shift

        ones_idx = np.flatnonzero(b > 0.5)
        zeros_idx = np.flatnonzero(b < 0.5)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break

        i = ones_idx[np.argmin(v[ones_idx])]
        j = zeros_idx[np.argmax(v[zeros_idx])]

        if v[j] <= v[i]:
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        b[i], b[j] = 0.0, 1.0
        new_score = compute_c5(b)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    def consider(h):
        nonlocal best_h, best_score
        n = h.size
        h_proj = project_onto_box_sum(h, n / 2.0)
        sc = compute_c5(h_proj)
        if sc < best_score:
            best_score = sc
            best_h = h_proj.copy()

    n_vals = [128, 256, 512]

    # ----- Continuous gradient descent phase -----
    for n in n_vals:
        if time_left() < 5:
            break
        candidates = []

        # use initial_h_values if available
        if 'initial_h_values' in globals():
            init_full = np.asarray(initial_h_values, dtype=float)
            vec = resample(init_full, n)
            vec = project_onto_box_sum(vec, n / 2.0)
            candidates.append(vec)

        # a few random starts
        for _ in range(2):
            v = rng.random(n)
            v = project_onto_box_sum(v, n / 2.0)
            candidates.append(v)

        # gradient descent on each candidate
        for h0 in candidates:
            if time_left() < 2:
                break
            # allocate iterations loosely based on remaining time
            max_iters = int(min(12000, max(2000, time_left() * 1500)))
            if n == 512:
                max_iters = min(max_iters, 5000)
            elif n == 256:
                max_iters = min(max_iters, 8000)
            else:
                max_iters = min(max_iters, 12000)
            h_opt, _ = gradient_descent_momentum(h0, n_iter=max_iters, step0=0.5, beta=0.9)
            consider(h_opt)

    # ----- Binary hill‑climbing phase -----
    for n in n_vals:
        if time_left() < 2:
            break
        # random binary starts
        starts = 5
        for _ in range(starts):
            if time_left() < 2:
                break
            b = random_binary_balanced_vec(n, rng)
            max_iter = int(min(8000, time_left() * 2000))
            if n == 512:
                max_iter = min(max_iter, 5000)
            elif n == 256:
                max_iter = min(max_iter, 7000)
            else:
                max_iter = min(max_iter, 9000)
            b_opt, _ = hill_climb_binary(b, max_iter, rng)
            consider(b_opt)

        # hill‑climb from thresholded best continuous vector (if size matches)
        if best_h is not None and best_h.size == n:
            order = np.argsort(best_h)[-n // 2:]
            b = np.zeros(n, dtype=float)
            b[order] = 1.0
            max_iter = int(min(8000, time_left() * 2000))
            if n == 512:
                max_iter = min(max_iter, 5000)
            b_opt, _ = hill_climb_binary(b, max_iter, rng)
            consider(b_opt)

    # Final fallback (should rarely happen)
    if best_h is None:
        n = 128
        best_h = random_binary_balanced_vec(n, rng)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
