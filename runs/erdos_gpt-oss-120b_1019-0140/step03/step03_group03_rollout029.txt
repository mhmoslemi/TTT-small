import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def next_pow_two(x):
    return 1 << (x - 1).bit_length()

    m = 2 * n - 1
    L = next_pow_two(m)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b_rev, n=L)
    corr = np.fft.irfft(A * B, n=L)[:m]
    return np.max(corr) * (2.0 / n)


def compute_c5(h):
    """Wrapper for C5 (uses FFT for speed)."""
    return compute_c5_fft(h)


def autocorr_fft(h):
    """Autocorrelation of h using FFT (full length)."""
    n = h.shape[0]
    m = 2 * n - 1
    L = next_pow_two(m)
    H = np.fft.rfft(h, n=L)
    H_rev = np.fft.rfft(h[::-1], n=L)
    ac = np.fft.irfft(H * H_rev, n=L)[:m]
    return ac


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_grad_for_shift(h, shift):
    """Gradient that increases autocorrelation at given integer shift."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n - shift] += h[shift:]
        grad[shift:] += h[:n - shift]
    else:
        d = -shift
        grad[d:] += h[:n - d]
        grad[:n - d] += h[d:]
    return grad


# Global helper for SLSQP (avoids lambdas/closures)
_slsqp_n = None


def _slsqp_obj(x):
    return x[-1]


def _slsqp_eq(x):
    n = _slsqp_n
    return np.sum(x[:n]) - n / 2.0


def _slsqp_ineq(x):
    n = _slsqp_n
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - np.max(corr)


def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP refinement on a candidate."""
    global _slsqp_n
    n = h0.shape[0]
    _slsqp_n = n
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': _slsqp_eq},
            {'type': 'ineq', 'fun': _slsqp_ineq}]
    res = opt.minimize(_slsqp_obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    _slsqp_n = None
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on binary balanced vectors."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score


def min_shift_gradient_ascent(h0, max_iter, step0, start_time, budget_s):
    """Targeted ascent to increase minimal autocorrelation (reducing C5)."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iter + 1):
        if budget_s - (time.time() - start_time) <= 0:
            break
        ac = autocorr_fft(h)
        shift = int(np.argmin(ac) - (n - 1))
        grad = compute_grad_for_shift(h, shift)
        eta = step0 / np.sqrt(it)
        h_new = h + eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Small scale exploration (n = 64)
    # ------------------------------------------------------------------
    n_small = 64
    small_cands = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_small:
            cand = downsample(init_full, n_small)
        else:
            cand = init_full.copy()
        cand = project_onto_box_sum(cand, n_small / 2.0)
        small_cands.append(cand)

    small_cands.append(thue_morse(n_small))
    for _ in range(4):
        small_cands.append(random_binary_balanced(n_small, rng))
    for _ in range(4):
        v = rng.random(n_small)
        small_cands.append(project_onto_box_sum(v, n_small / 2.0))

    best_h = None
    best_score = np.inf

    # Quick SLSQP refinements (limited iterations)
    for cand in small_cands:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=200)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Targeted gradient ascent on small scale
    if best_h is not None and time_left() > 5.0:
        h_gd, sc_gd = min_shift_gradient_ascent(best_h, max_iter=5000,
                                                step0=0.5,
                                                start_time=start_time,
                                                budget_s=budget_s)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()

    # ------------------------------------------------------------------
    # Upsample to medium resolution (n = 256)
    # ------------------------------------------------------------------
    n_medium = 256
    if best_h is None:
        best_h = random_binary_balanced(n_medium, rng)
    else:
        best_h = upsample(best_h, n_medium)
    best_h = project_onto_box_sum(best_h, n_medium / 2.0)
    cur_score = compute_c5(best_h)
    if cur_score < best_score:
        best_score, best_h = cur_score, best_h.copy()

    # ------------------------------------------------------------------
    # Upsample to large resolution (n = 1024)
    # ------------------------------------------------------------------
    n_large = 1024
    best_h = upsample(best_h, n_large)
    best_h = project_onto_box_sum(best_h, n_large / 2.0)
    cur_score = compute_c5(best_h)
    if cur_score < best_score:
        best_score, best_h = cur_score, best_h.copy()

    # ------------------------------------------------------------------
    # Large‑scale targeted gradient ascent
    # ------------------------------------------------------------------
    if time_left() > 10.0:
        # estimate ~2500 iterations per second for FFT‑based steps
        max_iters = int(min(250000, time_left() * 2500))
        h_gd, sc_gd = min_shift_gradient_ascent(best_h, max_iter=max_iters,
                                                step0=0.5,
                                                start_time=start_time,
                                                budget_s=budget_s)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()

    # ------------------------------------------------------------------
    # Binary simulated annealing refinement
    # ------------------------------------------------------------------
    if time_left() > 5.0:
        idx = np.argsort(best_h)[-best_h.size // 2:]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        max_iter_ann = int(time_left() * 500)
        h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                      temp_start=0.01,
                                      temp_end=1e-5,
                                      rng=rng)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # Final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
