def binary_hill_climb_sa(h0, rng, time_left, target, max_iters=None):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    if max_iters is None:
        max_iters = int(min(500000, max(20000, time_left()*2000)))
    T0 = 0.01
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # swap a 0 and a 1
        i0 = rng.integers(n)
        while h[i0] != 0:
            i0 = rng.integers(n)
        i1 = rng.integers(n)
        while h[i1] != 1:
            i1 = rng.integers(n)
        h_new = h.copy()
        h_new[i0] = 1.0
        h_new[i1] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it/max_iters)  # linear cooling
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:]   -= h[:n-k]
        else:
            p = -k
            grad[p:]      += (1.0 - h[:n-p])
            grad[:n-p]    -= h[p:]
    return grad / len(shifts)
```

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha*(corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n-1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n-k] += w*(1.0 - h[k:])
            grad[k:]   -= w*h[:n-k]
        else:
            p = -k
            grad[p:]      += w*(1.0 - h[:n-p])
            grad[:n-p]    -= w*h[p:]
    return grad

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # enlarge bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral (discrete approximation)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def resample(arr, new_n):
    """Linear resampling of 1‑D array to new length."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:]      -= h[:n - k]
        else:
            p = -k
            grad[p:]      += (1.0 - h[:n - p])
            grad[:n - p]  -= h[p:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    """Projected gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * mom, s)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()
    return best_h, best_score

def proj_subgradient_descent(h0, n_iter, step0, beta, time_left):
    """Projected subgradient descent using max‑correlation shifts."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        # indices achieving (near) max
        shifts = np.where(corr >= maxc - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * mom, s)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808, max_iters=None):
    """Simulated‑annealing hill‑climb on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    if max_iters is None:
        max_iters = int(min(300000, max(20000, time_left() * 2000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # select a random zero and a random one
        i0 = rng.integers(n)
        while h[i0] != 0.0:
            i0 = rng.integers(n)
        i1 = rng.integers(n)
        while h[i1] != 1.0:
            i1 = rng.integers(n)
        h_new = h.copy()
        h_new[i0] = 1.0
        h_new[i1] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1.0 - it / max_iters)
        accept = False
        if delta < 0.0:
            accept = True
        elif T > 0.0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_large = 512
    target_bound = 0.3808

    # ----- build initial candidates -----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = resample(init_full, n_large)
        candidates.append(project_onto_box_sum(cand, n_large / 2.0))
    # Thue–Morse binary (already 0/1)
    candidates.append(project_onto_box_sum(resample(thue_morse(n_large).astype(float), n_large), n_large / 2.0))
    # Random balanced binaries
    for _ in range(2):
        candidates.append(random_binary_balanced(n_large, rng))
    # Random continuous vectors projected onto constraints
    for _ in range(2):
        v = rng.random(n_large)
        candidates.append(project_onto_box_sum(v, n_large / 2.0))

    best_h = None
    best_score = float('inf')

    # ----- Phase 1: continuous refinement (smooth + subgradient) -----
    for h0 in candidates:
        if time_left() < 5.0:
            break
        # smooth gradient descent
        n_iter_smooth = int(min(2500, max(800, time_left() * 8)))
        h_smooth, sc_smooth = smooth_grad_descent(
            h0, n_iter=n_iter_smooth,
            step0=0.5, alpha_start=1.0, alpha_end=12.0,
            beta=0.9, time_left=time_left)
        if sc_smooth < best_score:
            best_score, best_h = sc_smooth, h_smooth.copy()
        # follow with subgradient descent for fine tuning
        n_iter_sub = int(min(3000, max(600, time_left() * 6)))
        h_sub, sc_sub = proj_subgradient_descent(
            h_smooth, n_iter=n_iter_sub,
            step0=0.7, beta=0.9, time_left=time_left)
        if sc_sub < best_score:
            best_score, best_h = sc_sub, h_sub.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_large, 0.5), n_large / 2.0)
        best_score = compute_c5(best_h)

    # ----- Phase 2: binary conversion & simulated annealing -----
    n = best_h.shape[0]
    order = np.argsort(best_h)
    binary = np.zeros(n, dtype=float)
    binary[order[n // 2:]] = 1.0

    if time_left() > 3.0:
        max_sa_iters = int(min(250000, max(20000, time_left() * 1800)))
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left,
                                             target=target_bound,
                                             max_iters=max_sa_iters)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # ----- Phase 3: brief final continuous polishing -----
    if time_left() > 2.0:
        n_iter_final = int(min(2000, max(400, time_left() * 4)))
        h_final, sc_final = proj_subgradient_descent(
            best_h, n_iter=n_iter_final,
            step0=0.4, beta=0.9, time_left=time_left)
        if sc_final < best_score:
            best_score, best_h = sc_final, h_final.copy()

    # ----- Return result -----
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
