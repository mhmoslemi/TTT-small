class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            inner = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            inner = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - inner * self.dx

def compute_c5(h):
    """Exact C5 for a given h (binary or continuous)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def legendre_sequence(p):
    """0/1 Legendre symbol sequence of prime length p (p odd)."""
    seq = np.zeros(p, dtype=int)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1 if leg == 1 else 0
    return seq

# ---------- SLSQP refinement ----------
class SumConstraint:
    def __init__(self, n, target):
        self.n = n
        self.target = target
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target


class OverlapConstraint:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            inner = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            inner = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - inner * self.dx


def refine_slsqp(h_start, n, remaining_time):
    """Run a single SLSQP refinement (if time permits)."""
    if remaining_time < 1.0:
        return h_start, compute_c5(h_start)

    M = n // 2
    dx = 2.0 / n
    t0 = compute_c5(h_start)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': SumConstraint(n, M)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        constraints.append({'type': 'ineq', 'fun': OverlapConstraint(k, n, dx)})

    x0 = np.concatenate([h_start, [t0]])

    opts = {'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options=opts)
    except Exception:
        return h_start, compute_c5(h_start)

    if not res.success:
        return h_start, compute_c5(h_start)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)

        h_bin, c5_bin = optimise_binary_sa(n, alloc, rng)

        if h_bin is not None and c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin, c5_bin, n
            if best_c5 <= 0.3808:
                break

    # ----- final refinement with SLSQP -----
    elapsed = time.time() - start_total
    rem = budget_s - elapsed
    if best_h is not None:
        h_ref, c5_ref = refine_slsqp(best_h, best_n, rem)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
    else:
        # fallback simple construction
        n = 80
        M = n // 2
        best_h = np.concatenate([np.ones(M), np.zeros(n - M)])
        best_c5 = compute_c5(best_h)
        best_n = n

    # safety: enforce constraints exactly
    n = best_n
    M = n // 2
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = np.sum(best_h)
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
