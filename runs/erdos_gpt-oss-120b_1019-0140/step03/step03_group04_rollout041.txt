def hill_climb_local_optimum(h, rng, max_no_improve, time_end):
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    no_improve = 0
    while time.time() < time_end and no_improve < max_no_improve:
        # select random swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1-h)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        if new_score < current_score - 1e-12: # improvement
            current_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
    return best_h, best_score, h, current_score

import time
import numpy as np

# ----- basic utilities -------------------------------------------------
def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_to_half(arr):
    """Return a 0/1 array of the same length with exactly half 1‑entries."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    out = np.zeros(n, dtype=int)
    out[idx] = 1
    return out

def legendre_seq(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=int)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1 if pow(i, exp, p) == 1 else 0
    return seq

def thue_morse_binary(n):
    """Standard Thue–Morse 0/1 sequence of length n."""
    # parity of popcount for each index
    parity = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=int)
    return parity  # 0 for even, 1 for odd parity

# ----- local optimisation on binary vectors ---------------------------
def _min_cross_excl_center(cross, center):
    """minimum autocorrelation value excluding the centre (shift 0)."""
    left = cross[:center]
    right = cross[center + 1:]
    # np.min on empty slice would error; but centre is never at the ends for n>1
    return left.min() if left.size else np.inf, right.min() if right.size else np.inf

def local_optimize_binary(h_start, time_end, rng,
                          max_no_improve=50000):
    """
    Greedy local search on binary vectors with exact sum constraint.
    Returns the best vector found and its C5 bound.
    """
    h = h_start.astype(np.int8).copy()
    n = h.size
    center = n - 1
    # initial autocorrelation
    cross = np.correlate(h, h, mode='full').astype(np.int32)
    left_min, right_min = _min_cross_excl_center(cross, center)
    min_cross = left_min if left_min < right_min else right_min
    cur_score = 1.0 - (2.0 / n) * float(min_cross)

    best_h = h.copy()
    best_score = cur_score

    no_improve = 0
    # main loop
    while time.time() < time_end and no_improve < max_no_improve:
        # pick distinct indices with opposite bits
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            # no swap effect
            continue

        # prepare delta for this swap (i:0→1, j:1→0)
        delta = np.zeros_like(cross, dtype=np.int32)

        # slices for i (adding)
        delta[center - i: center - i + n] += h
        delta[center + i - (n - 1): center + i + 1] += h[::-1]

        # slices for j (removing)
        delta[center - j: center - j + n] -= h
        delta[center + j - (n - 1): center + j + 1] -= h[::-1]

        # term3 (pairwise interaction)
        delta[center] += 2
        shift = j - i
        delta[center + shift] -= 1
        delta[center - shift] -= 1

        # evaluate new autocorrelation
        new_cross = cross + delta
        lmin, rmin = _min_cross_excl_center(new_cross, center)
        new_min = lmin if lmin < rmin else rmin
        new_score = 1.0 - (2.0 / n) * float(new_min)

        if new_score + 1e-12 < cur_score:          # improvement
            # accept swap
            h[i], h[j] = h[j], h[i]
            cross = new_cross
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            no_improve += 1

    return best_h, best_score

        h_opt, sc_opt = local_optimize_binary(cand, t_end, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # extra random restarts
    while time_left() > 5.0 and (best_score > 0.3808):
        rnd = np.zeros(n, dtype=int)
        idx = rng.choice(n, size=n // 2, replace=False)
        rnd[idx] = 1
        t_end = start + budget_s - 0.5
        h_opt, sc_opt = local_optimize_binary(rnd, t_end, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # fallback in the unlikely event no candidate was produced
    if best_h is None:
        best_h = np.zeros(n, dtype=int)
        idx = rng.choice(n, size=n // 2, replace=False)
        best_h[idx] = 1
        best_score = compute_c5(best_h.astype(float))

    # final safety recomputation (ensures consistency with the public metric)
    best_h_float = best_h.astype(float)
    final_score = compute_c5(best_h_float)

    return best_h_float, final_score, best_h_float.size
