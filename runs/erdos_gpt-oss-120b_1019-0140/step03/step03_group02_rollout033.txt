def solve_binary_mip(n, time_limit):
    # n must be even
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable(integer=True, nonneg=True)
    constraints = [cp.sum(h) == n//2]
    # link w to h
    constraints.append(w <= h[:, None])
    constraints.append(w <= h[None, :])
    constraints.append(w >= h[:, None] + h[None, :] - 1)
    # min overlap constraints:
    for k in range(-(n-1), n):
        if k >= 0:
            constraints.append(cp.sum(w[0:n-k, k:n]) >= t)
        else:
            constraints.append(cp.sum(w[-k:n, 0:n+k]) >= t)
    # objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    # solve with GLPK_MI if available, else ECOS_BB
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(time_limit*1000)}, warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS_BB, warm_start=True, max_iters=500000, verbose=False)
        except Exception as e:
            raise
    # check status
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise ValueError(f"Solver failed: {prob.status}")
    h_val = np.array(h.value).flatten()
    return h_val

import time
import numpy as np

# Optional cvxpy for exact binary optimization
try:
    import cvxpy as cp
    _CVXPY = True
except Exception:
    _CVXPY = False


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """
    Euclidean projection of v onto the box [lo,hi]^n intersected with the hyperplane sum(x)=s.
    """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term w.r.t. h for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad


def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9,
                          time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        # identify shifts attaining the maximum correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick a random zero and a random one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score


def solve_binary_mip(n, time_limit=60.0):
    """Exact binary formulation via MILP (max‑min overlap). Returns real‑valued h."""
    if not _CVXPY:
        raise RuntimeError("cvxpy is not available")
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable(integer=True, nonneg=True)

    constraints = [cp.sum(h) == n // 2]

    # linking constraints for w = h_i * h_j
    for i in range(n):
        for j in range(n):
            constraints.append(w[i, j] <= h[i])
            constraints.append(w[i, j] <= h[j])
            constraints.append(w[i, j] >= h[i] + h[j] - 1)

    # each shift must have at least t overlapping ones
    for k in range(-(n - 1), n):
        if k >= 0:
            constraints.append(cp.sum(w[0:n - k, k:n]) >= t)
        else:
            constraints.append(cp.sum(w[-k:n, 0:n + k]) >= t)

    prob = cp.Problem(cp.Maximize(t), constraints)
    # try GLPK_MI first, fallback to ECOS_BB
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
    except Exception:
        prob.solve(solver=cp.ECOS_BB, warm_start=True, max_iters=200000, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"MILP solver failed: {prob.status}")

    h_val = np.array(h.value).flatten()
    return h_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n = 64                     # number of discretization points
    target = 0.3808
    best_h = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # 1. Use any provided initial construction (if present)
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            # simple size adjustment: repeat or truncate
            if init_full.size > n:
                idx = np.linspace(0, init_full.size - 1, n, dtype=int)
                init_small = init_full[idx]
            else:
                factor = int(np.ceil(n / init_full.size))
                init_small = np.repeat(init_full, factor)[:n]
        else:
            init_small = init_full.copy()
        init_small = project_onto_box_sum(init_small, n / 2.0)
        sc = compute_c5(init_small)
        if sc < best_score:
            best_h, best_score = init_small, sc

    # -----------------------------------------------------------------
    # 2. Exact binary optimization via MILP (if time permits)
    if time_left() > 30.0 and _CVXPY:
        try:
            h_bin = solve_binary_mip(n, time_limit=time_left())
            # threshold to binary and enforce exact balance
            h_bin = (h_bin >= 0.5).astype(float)
            s_needed = n // 2
            diff = int(h_bin.sum() - s_needed)
            if diff > 0:
                ones = np.where(h_bin == 1)[0]
                h_bin[rng.choice(ones, size=diff, replace=False)] = 0.0
            elif diff < 0:
                zeros = np.where(h_bin == 0)[0]
                h_bin[rng.choice(zeros, size=-diff, replace=False)] = 1.0
            sc = compute_c5(h_bin)
            if sc < best_score:
                best_h, best_score = h_bin, sc
        except Exception:
            pass  # fall back to heuristics

    # -----------------------------------------------------------------
    # 3. Heuristic binary improvement (simulated annealing)
    def try_hill(h0):
        nonlocal best_h, best_score
        if time_left() < 1.0:
            return
        h0b = (h0 >= 0.5).astype(float)
        # enforce exact balance
        s_needed = n // 2
        diff = int(h0b.sum() - s_needed)
        if diff > 0:
            ones = np.where(h0b == 1)[0]
            h0b[rng.choice(ones, size=diff, replace=False)] = 0.0
        elif diff < 0:
            zeros = np.where(h0b == 0)[0]
            h0b[rng.choice(zeros, size=-diff, replace=False)] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(h0b, rng, time_left, target=target)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # start hill climb from current best (if any)
    if best_h is not None:
        try_hill(best_h)

    # perform random restarts while time permits
    while time_left() > 5.0:
        h0 = random_binary_balanced(n, rng)
        try_hill(h0)
        if best_score <= target:
            break

    # -----------------------------------------------------------------
    # 4. Continuous refinement (momentum gradient descent) if time remains
    if best_h is not None and time_left() > 5.0:
        h_cont, cont_sc = momentum_grad_descent(
            best_h,
            n_iter=5000,
            step0=0.5,
            beta=0.9,
            time_left=time_left,
        )
        if cont_sc < best_score:
            best_score, best_h = cont_sc, h_cont.copy()

    # -----------------------------------------------------------------
    # Final projection to satisfy constraints exactly
    if best_h is None:
        # fallback trivial construction
        best_h = np.zeros(n, dtype=float)
        best_h[n // 2 :] = 1.0
    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
