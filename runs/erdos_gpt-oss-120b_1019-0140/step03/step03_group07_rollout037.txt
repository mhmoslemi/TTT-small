def project_to_sum_and_bounds(h, target_sum):
    # clip first
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if np.isclose(cur, target_sum):
        return h
    # if cur < target_sum: raise some values
    # Use water-filling
    # tolerance
    eps = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        if cur < target_sum:
            mask = h < 1 - eps
            if not mask.any():
                break
            deficit = target_sum - cur
            # distribute uniformly
            inc = deficit / mask.sum()
            # ensure not exceed 1
            addition = np.minimum(inc, 1 - h[mask])
            h[mask] += addition
        else:
            mask = h > eps
            if not mask.any():
                break
            excess = cur - target_sum
            dec = excess / mask.sum()
            subtraction = np.minimum(dec, h[mask])
            h[mask] -= subtraction
        cur = h.sum()
        if abs(cur - target_sum) <= 1e-12:
            break
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

        grad[kk:] += h[:n - kk]
        grad[:n - kk] += h[kk:]

def compute_c5(h):

def project_to_sum_and_bounds(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum (water-filling)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    eps = 1e-12
    # simple iterative water-filling
    for _ in range(10):
        if abs(cur - target_sum) <= eps:
            break
        if cur < target_sum:
            mask = h < 1 - eps
            if not mask.any():
                break
            deficit = target_sum - cur
            inc = deficit / mask.sum()
            addition = np.minimum(inc, 1 - h[mask])
            h[mask] += addition
        else:
            mask = h > eps
            if not mask.any():
                break
            excess = cur - target_sum
            dec = excess / mask.sum()
            subtraction = np.minimum(dec, h[mask])
            h[mask] -= subtraction
        cur = h.sum()
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidates:
            candidates = [n0] + candidates

    for n in candidates:
        M = n // 2
        dx = 2.0 / n
        # keep generating restarts until time runs out
        while time.time() - start < budget_s - 2:
            # ----- initialise h -----
            if best_h is not None and best_n == n and rng.random() < 0.3:
                h = best_h + rng.normal(scale=0.03, size=n)
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum_and_bounds(h, M)
            elif 'initial_h_values' in globals() and len(initial_h_values) == n and rng.random() < 0.1:
                h = np.array(initial_h_values, dtype=float)
                h = project_to_sum_and_bounds(h, M)
            else:
                h = rng.random(n)
                h = h / h.sum() * M
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum_and_bounds(h, M)

            # ----- gradient ascent on min-overlap -----
            step = 0.5
            for it in range(5000):
                if time.time() - start > budget_s - 1:
                    break
                conv = np.correlate(h, h, mode='full')
                mid = n - 1
                conv_no0 = conv.copy()
                conv_no0[mid] = np.inf          # ignore zero shift
                min_idx = np.argmin(conv_no0)
                min_overlap = conv_no0[min_idx]
                c5 = 1.0 - min_overlap * dx
                if c5 < best_c5 - 1e-12:
                    best_h, best_c5, best_n = h.copy(), c5, n
                    if best_c5 < 0.3808:
                        return best_h, best_c5, best_n
                k = min_idx - mid
                grad = np.zeros_like(h)
                if k >= 0:
                    if k < n:
                        grad[:n - k]   += h[k:]
                        grad[k:]       += h[:n - k]
                else:
                    kk = -k
                    grad[kk:]         += h[:n - kk]
                    grad[:n - kk]     += h[kk:]
                norm = np.linalg.norm(grad)
                if norm == 0:
                    break
                step_size = step / norm
                h = h + step_size * grad
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum_and_bounds(h, M)
                step *= 0.9995

            # ----- local refinement with simple moves -----
            for _ in range(200):
                if time.time() - start > budget_s - 1:
                    break
                if rng.random() < 0.5:
                    zeros = np.where(h < 0.5)[0]
                    ones  = np.where(h > 0.5)[0]
                    if zeros.size and ones.size:
                        i = rng.choice(zeros)
                        j = rng.choice(ones)
                        h_new = h.copy()
                        h_new[i] = 1.0
                        h_new[j] = 0.0
                    else:
                        continue
                else:
                    i, j = rng.choice(n, 2, replace=False)
                    max_transfer = min(1.0 - h[i], h[j])
                    if max_transfer <= 0:
                        continue
                    delta = rng.random() * max_transfer
                    h_new = h.copy()
                    h_new[i] += delta
                    h_new[j] -= delta
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_h, best_c5, best_n = h_new.copy(), c5_new, n
                    h = h_new
                    if best_c5 < 0.3808:
                        return best_h, best_c5, best_n
                elif c5_new < compute_c5(h):
                    h = h_new

    # ----- fallback if nothing improved -----
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum_and_bounds(best_h, best_n // 2)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
