import time
import numpy as np

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if cand.size == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def compute_overlaps(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.convolve(h, 1.0 - h, mode='full')
    return conv * dx

def compute_c5(h):
    return np.max(compute_overlaps(h))

def gradient_step(h, alpha, step, target, rng):
    n = h.shape[0]
    dx = 2.0 / n
    O = compute_overlaps(h)
    O_max = O.max()
    exps = np.exp(alpha * (O - O_max))
    w = exps / exps.sum()
    w_rev = w[::-1]
    conv_full = np.convolve(w, h, mode='full')
    conv2 = conv_full[n-1:n-1+n]
    conv_full_rev = np.convolve(w_rev, h, mode='full')
    conv1 = conv_full_rev[n-1:n-1+n]
    grad = dx * (1.0 - (conv1 + conv2))
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, target, rng)
    return h_new, compute_c5(h_new)

def continuous_anneal(cur_h, cur_val, best_h, best_val, start, deadline, rng):
    n = cur_h.shape[0]
    dx = 2.0 / n
    h = cur_h.copy()
    cur = cur_val
    best = best_val
    best_vec = best_h.copy()
    while time.time() < deadline - 0.1:
        # find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute contributions for this shift
        if k >= 0:
            prod = h[:n-k] * (1.0 - h[k:])
            i_range = np.arange(n-k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[:n-kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # choose destination
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur:
            cur = new_val
            if new_val < best:
                best = new_val
                best_vec = h.copy()
                if best < 0.3808:
                    break
        else:
            # Metropolis acceptance
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur - new_val) / T) > rng.random():
                cur = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_vec, best

import time
import numpy as np

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if cand.size == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def compute_overlaps(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.convolve(h, 1.0 - h, mode='full')
    return conv * dx

def compute_c5(h):
    return np.max(compute_overlaps(h))

def gradient_step(h, alpha, step, target, rng):
    n = h.shape[0]
    dx = 2.0 / n
    O = compute_overlaps(h)
    O_max = O.max()
    exps = np.exp(alpha * (O - O_max))
    w = exps / exps.sum()
    w_rev = w[::-1]
    conv_full = np.convolve(w, h, mode='full')
    conv2 = conv_full[n - 1 : n - 1 + n]
    conv_full_rev = np.convolve(w_rev, h, mode='full')
    conv1 = conv_full_rev[n - 1 : n - 1 + n]
    grad = dx * (1.0 - (conv1 + conv2))
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, target, rng)
    return h_new, compute_c5(h_new)

def continuous_anneal(cur_h, cur_val, best_h, best_val, start, deadline, rng):
    n = cur_h.shape[0]
    h = cur_h.copy()
    cur = cur_val
    best = best_val
    best_vec = best_h.copy()
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if cand.size == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]

            j = rng.choice(dst)
        else:
            j = rng.choice(cand)

            else:
