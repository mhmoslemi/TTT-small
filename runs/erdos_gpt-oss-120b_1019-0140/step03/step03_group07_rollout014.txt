def compute_c5_fft(h):
    n = h.shape[0]
    a_rev = h[::-1]
    b = 1.0 - h
    L = 1 << ((2*n-1).bit_length())
    # Compute FFT
    A = np.fft.rfft(a_rev, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    conv = np.fft.irfft(C, n=L)
    # conv length L; the linear convolution part is conv[:2*n-1]
    # we want maximum of conv[:2*n-1]
    max_val = np.max(conv[:2*n-1])
    return max_val * (2.0 / n)

import time
import numpy as np

# ---------- helper functions ----------
def is_prime(num: int) -> bool:
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5) + 1
    for f in range(3, r, 2):
        if num % f == 0:
            return False
    return True


def legendre_binary(p: int) -> np.ndarray:
    """Legendre symbol sequence of length p (p odd prime), values 0/1 with seq[0]=0."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for quadratic residue, p-1 otherwise
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq


def compute_c5_fft(h: np.ndarray) -> float:
    """Compute max overlap integral using FFT (linear correlation)."""
    n = h.shape[0]
    a_rev = h[::-1]
    b = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a_rev, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)
    max_val = np.max(conv[: 2 * n - 1])
    return max_val * (2.0 / n)


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones."""
    m = n // 2
    idx = rng.choice(n, size=m, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    best_h = None

    if best_h is None:

                dec = min(-diff, best_h[idx])
                best_h[idx] -= dec
                diff += dec
