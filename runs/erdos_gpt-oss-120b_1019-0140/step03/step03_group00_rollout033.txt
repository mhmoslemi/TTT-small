        if time_left() <= 0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # Determine worst shifts (max C = max_corr * dx)
        # Might identify shifts where correlation equals max
        idxes = np.where(corr >= max_corr - 1e-12)[0]

        grad = np.zeros_like(h)

            else:

                term1 = np.zeros(n, dtype=h.dtype)

                term1[s_shift:] = h[:n-s_shift]

            # gradient of C(k) = -(∂J/∂h) = -(h shifted forward + h shifted backward) maybe? but we # use approximate.
            # For simplicity, we take gradient = -(term1 + term2)
            grad += -(term1 + term2)
        if len(shifts) > 0:
            grad /= len(shifts)

        momentum = beta*momentum + (1-beta)*grad

    for it in range(1, n_iter+1):

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval contains a solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average sub‑gradient of max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            fwd = np.zeros(n, dtype=h.dtype)
            bwd = np.zeros(n, dtype=h.dtype)
            fwd[:n - shift] = h[shift:]
            bwd[shift:] = h[:n - shift]
        else:
            s = -shift
            fwd = np.zeros(n, dtype=h.dtype)
            bwd = np.zeros(n, dtype=h.dtype)
            fwd[s:] = h[:n - s]
            bwd[:n - s] = h[s:]
        grad += 1.0 - fwd - bwd
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_refine(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def simulated_annealing_refine(h0, corr0, n_iter, init_temp, final_temp,
                               rng, time_left):
    """SA on the max‑correlation objective using exact incremental updates."""
    h = h0.copy()
    corr = corr0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    cur_score = corr.max() * dx
    best_h, best_score = h.copy(), cur_score
    s_arr = np.arange(-(n - 1), n)  # length 2n‑1

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        # temperature schedule (exponential)
        t = init_temp * ((final_temp / init_temp) ** (it / n_iter))

        # pick source index with some mass
        i = rng.integers(n)
        attempts = 0
        while h[i] <= 1e-9 and attempts < 5:
            i = rng.integers(n)
            attempts += 1
        if h[i] <= 1e-9:
            continue

        # pick destination index with room
        j = rng.integers(n)
        attempts = 0
        while h[j] >= 1.0 - 1e-9 and attempts < 5:
            j = rng.integers(n)
            attempts += 1
        if h[j] >= 1.0 - 1e-9:
            continue

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        if delta == 0:
            continue

        # incremental delta correlation
        idx_i_plus = i + s_arr
        mask_i_plus = (idx_i_plus >= 0) & (idx_i_plus < n)
        term_i_plus = np.zeros_like(s_arr, dtype=float)
        term_i_plus[mask_i_plus] = h[idx_i_plus[mask_i_plus]]

        idx_j_plus = j + s_arr
        mask_j_plus = (idx_j_plus >= 0) & (idx_j_plus < n)
        term_j_plus = np.zeros_like(s_arr, dtype=float)
        term_j_plus[mask_j_plus] = h[idx_j_plus[mask_j_plus]]

        idx_i_minus = i - s_arr
        mask_i_minus = (idx_i_minus >= 0) & (idx_i_minus < n)
        term_i_minus = np.zeros_like(s_arr, dtype=float)
        term_i_minus[mask_i_minus] = h[idx_i_minus[mask_i_minus]]

        idx_j_minus = j - s_arr
        mask_j_minus = (idx_j_minus >= 0) & (idx_j_minus < n)
        term_j_minus = np.zeros_like(s_arr, dtype=float)
        term_j_minus[mask_j_minus] = h[idx_j_minus[mask_j_minus]]

        delta_corr = delta * (term_i_plus - term_j_plus + term_i_minus - term_j_minus)

        new_max = (corr + delta_corr).max()
        new_score = new_max * dx
        delta_score = new_score - cur_score

        if delta_score < 0 or rng.random() < np.exp(-delta_score / t):
            # accept move
            h[i] -= delta
            h[j] += delta
            corr += delta_corr
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # progressive refinement
    scales = [64, 128, 256, 512]
    for idx, n in enumerate(scales):
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # short gradient descent on coarse levels
        if n <= 256 and time_left() > 10:
            gd_iters = int(min(20000, max(2000, time_left() * (4000.0 / n))))
            base, gd_score = gradient_descent_refine(base, gd_iters, 0.5, 0.9,
                                                     rng, time_left)
            if gd_score < best_score:
                best_score, best_h = gd_score, base.copy()

        # simulated annealing on current scale
        if time_left() > 5:
            corr = np.correlate(base, 1.0 - base, mode='full')
            # parameters scale with n
            init_temp = 0.02
            final_temp = 0.001
            est_iters = int(min(200000,
                               max(20000, time_left() * (2000.0 / n))))
            base, sa_score = simulated_annealing_refine(base, corr,
                                                        est_iters,
                                                        init_temp, final_temp,
                                                        rng, time_left)
            if sa_score < best_score:
                best_score, best_h = sa_score, base.copy()

        # upsample for next scale
        if idx < len(scales) - 1:
            base = upsample(base, scales[idx + 1])

    # final fine‑tuning on best candidate
    if time_left() > 5:
        n = best_h.size
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        final_iters = int(min(200000,
                              max(50000, time_left() * (3000.0 / n))))
        best_h, best_score = simulated_annealing_refine(best_h, corr,
                                                        final_iters,
                                                        0.005, 1e-4,
                                                        rng, time_left)

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
