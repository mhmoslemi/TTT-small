import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def max_overlap_info(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    max_val = corr[idx]
    shift = idx - (n - 1)
    if shift >= 0:
        i_range = np.arange(0, n - shift)
        j_range = i_range + shift
    else:
        shift_abs = -shift
        i_range = np.arange(shift_abs, n)
        j_range = i_range - shift_abs
    return max_val, shift, i_range, j_range, corr

def random_binary(n, M, rng):
    h = np.zeros(n)
    h[rng.choice(n, M, replace=False)] = 1.0
    return h

def improve_h_local(h, rng, max_iters=5000):
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    for _ in range(max_iters):
        max_val, shift, i_idx, j_idx, _ = max_overlap_info(best_h)
        # Identify mismatched indices for this shift where contribution>0
        contrib = best_h[i_idx] * (1.0 - best_h[j_idx])
        # mismatched: best_h[i]>0 and best_h[j]<1
        mism = i_idx[(best_h[i_idx] > 1e-9) & (best_h[j_idx] < 1 - 1e-9)]
        # good spots where shifting counterpart is 1, can increase without hurting this shift
        good = i_idx[(best_h[i_idx] < 1 - 1e-9) & (best_h[j_idx] > 1 - 1e-9)]
        if mism.size == 0 or good.size == 0:
            break
        i = rng.choice(mism)
        j = rng.choice(good)
        # Transfer amount delta
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            break
        delta = max_delta * rng.random()
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_c5(new_h)
        if new_val + 1e-14 < best_val:
            best_h, best_val = new_h, new_val
        # else possibly accept with small probability?
    return best_h, best_val

def anneal_h(h0, rng, time_limit):
    start = time.time()
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    M = n // 2
    # Temperature schedule parameters:
    T0 = cur_val
    T_end = 1e-5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = (1 - frac) * T0 + frac * T_end
        # propose move: transfer between random indices i,j
        i, j = rng.choice(n, 2, replace=False)
        # Ensure i is donor (higher) and j receiver (lower) for potential improvement
        if h[i] < h[j]:
            i, j = j, i
        max_delta = min(h[i] - 0.0, 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random()
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_c5(new_h)
        dE = new_val - cur_val
        if dE <= 0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h, best_val = h.copy(), cur_val
        # optionally invoke local improvement occasionally
        if rng.random() < 0.01:
            # local refinement
            h_ref, val_ref = improve_h_local(h, rng, max_iters=200)
            if val_ref < best_val:
                h, cur_val = h_ref, val_ref
                best_h, best_val = h_ref.copy(), val_ref
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    best_h = None
    best_val = np.inf
    best_n = None

    n_candidates = [80, 96, 112, 128]
    # prioritize parent size if exists
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in n_candidates:
                n_candidates.remove(n0)
            n_candidates = [n0] + n_candidates

    # allocate time per candidate dynamically
    total_budget = budget_s
    for idx, n in enumerate(n_candidates):
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining < 10:
            break
        # allocate portion of remaining time for this n
        # start with larger allocation for first candidate
        if idx == 0:
            alloc = min(remaining * 0.5, 600.0)
        else:
            alloc = min(remaining * 0.5 / (len(n_candidates) - idx), 300.0)
        M = n // 2
        # generate initial h

            h0 = np.array(initial_h_values, dtype=float)
            cur_sum = np.sum(h0)
            if abs(cur_sum - M) > 1e-9:
                h0 += (M - cur_sum) / n
                h0 = np.clip(h0, 0.0, 1.0)
        else:
            h0 = random_binary(n, M, rng)

        # perform multiple restarts within allocated time
        start_n = time.time()
        while time.time() - start_n < alloc * 0.9:
            # random perturbation of h0 for restarts
            # start from h0 or random binary sometimes
            if rng.random() < 0.2:
                h_start = random_binary(n, M, rng)
            else:

                if best_h is not None and best_n == n:
                    h_start = best_h.copy()
                    # apply few random swaps

                        i, j = rng.choice(n, 2, replace=False)
                        # swap values
                        h_start[i], h_start[j] = h_start[j], h_start[i]
                else:
                    h_start = h0.copy()
            # annealing within remaining subbudget
            sub_alloc = min(alloc * 0.1, 30.0)  # each anneal limited to 30 sec

def anneal_h(h0, rng, time_limit):
    start = time.time()
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    T0 = cur_val
    T_end = 1e-5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit
        T = (1 - frac) * T0 + frac * T_end
        # choose move type
        if rng.random() < 0.4:
            # targeted move based on worst shift
            max_val, shift, i_idx, j_idx, _ = max_overlap_info(h)
            mism = i_idx[(h[i_idx] > 1e-9) & (h[j_idx] < 1 - 1e-9)]
            good = i_idx[(h[i_idx] < 1 - 1e-9) & (h[j_idx] > 1 - 1e-9)]
            if mism.size > 0 and good.size > 0:
                i = rng.choice(mism)
                j = rng.choice(good)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
            else:
                # fallback random transfer
                i, j = rng.choice(n, 2, replace=False)
                if h[i] < h[j]:
                    i, j = j, i
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
        else:
            # random transfer
            i, j = rng.choice(n, 2, replace=False)
            if h[i] < h[j]:
                i, j = j, i
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_c5(new_h)
        dE = new_val - cur_val
        if dE <= 0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h = h.copy()
                best_val = cur_val
                if best_val < 0.3808:
                    break
    return best_h, best_val

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def max_overlap_info(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (n - 1)
    if shift >= 0:
        i_idx = np.arange(0, n - shift)
        j_idx = i_idx + shift
    else:
        shift_abs = -shift
        i_idx = np.arange(shift_abs, n)
        j_idx = i_idx - shift_abs
    return max_val, shift, i_idx, j_idx

def random_binary(n, M, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, M, replace=False)] = 1.0
    return h

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(math.isqrt(num))
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def improve_h_local(h, rng, max_iters=500):
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    for _ in range(max_iters):
        max_val, shift, i_idx, j_idx = max_overlap_info(best_h)
        # mismatched positions (contribute positively to overlap)
        mism_mask = (best_h[i_idx] > 1e-9) & (best_h[j_idx] < 1 - 1e-9)
        good_mask = (best_h[i_idx] < 1 - 1e-9) & (best_h[j_idx] > 1 - 1e-9)
        mism = i_idx[mism_mask]
        good = i_idx[good_mask]
        if mism.size == 0 or good.size == 0:
            break
        i = rng.choice(mism)
        j = rng.choice(good)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def max_overlap_info(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (n - 1)
    if shift >= 0:
        i_idx = np.arange(0, n - shift)
        j_idx = i_idx + shift
    else:
        s = -shift
        i_idx = np.arange(s, n)
        j_idx = i_idx - s
    return max_val, shift, i_idx, j_idx

def random_binary(n, M, rng):
    h = np.zeros(n, dtype=float)

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(math.isqrt(num))
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True

def legendre_sequence(p):

    for i in range(1, p):

        seq[i] = 1.0 if leg == 1 else 0.0
    return seq
