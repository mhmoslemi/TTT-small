    # O constraints
    for i in range(n):
        # O[i,0] = 0 (unused)

        for k in range(1, n):

            if j < n:
                constraints.append(O[i,k] <= h[i])
                constraints.append(O[i,k] + h[j] <= 1)
                constraints.append(O[i,k] + h[j] >= h[i])

        return None, None

        return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation does not wrap; mode="full" matches problem definition
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Scale/adjust a vector in [0,1] to have exact sum == target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def solve_mip(n, target, dx, deadline):
    """Mixed‑integer model that minimises the worst overlap count."""
    # binary decision for h
    h = cp.Variable(n, boolean=True)
    # matrix O[i,k] = h[i] * (1 - h[i+k]) for i+k < n, else forced 0
    O = cp.Variable((n, n), boolean=True)
    t = cp.Variable(integer=True)

    cons = [cp.sum(h) == target, t >= 0, t <= n]

    # O constraints and out‑of‑range zeroing
    for i in range(n):
        cons.append(O[i, 0] == 0)  # unused column
        for k in range(1, n):
            j = i + k
            if j < n:
                cons.append(O[i, k] <= h[i])               # O <= h_i
                cons.append(O[i, k] + h[j] <= 1)           # O + h_{i+k} <= 1
                cons.append(O[i, k] + h[j] >= h[i])        # O + h_{i+k} >= h_i
            else:
                cons.append(O[i, k] == 0)

    # each shift must respect the auxiliary bound t
    for k in range(1, n):
        cons.append(cp.sum(O[:n - k, k]) <= t)

    prob = cp.Problem(cp.Minimize(t), cons)

    remaining = deadline - time.time()
    if remaining <= 0:
        return None, None
    try:
        prob.solve(solver=cp.CBC, maxSeconds=remaining, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        # even sub‑optimal solutions give a value for t and h
        if h.value is None:
            return None, None

    # round h to exact 0/1 and repair the sum if needed
    h_val = np.rint(np.array(h.value)).astype(int)
    cur = h_val.sum()
    diff = target - cur
    if diff > 0:
        zeros = np.where(h_val == 0)[0]
        if len(zeros) >= diff:
            h_val[zeros[:diff]] = 1
    elif diff < 0:
        ones = np.where(h_val == 1)[0]
        if len(ones) >= -diff:
            h_val[ones[: -diff]] = 0

    t_val = int(round(t.value)) if t.value is not None else None
    return h_val.astype(float), t_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # dimension and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)
        h0 = enforce_sum(h0, n / 2.0, rng)

    target = n // 2
    dx = 2.0 / n

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Try integer‑programming approach first
    mip_h, mip_t = solve_mip(n, target, dx, deadline)
    if mip_h is not None:
        val = compute_c5(mip_h)
        if val < best_val:
            best_h, best_val = mip_h, val

    # If we already beat the target we can return early
    if best_val <= 0.3808:
        return best_h.astype(float), best_val, n

    # Simple continuous local search as a fallback (few iterations)
    cur_h = best_h.copy()
    cur_val = best_val
    while time.time() < deadline - 0.2:
        # pick a random 1 and 0 and swap a tiny amount
        idx1 = np.where(cur_h > 1e-8)[0]
        idx0 = np.where(cur_h < 1.0 - 1e-8)[0]
        if len(idx1) == 0 or len(idx0) == 0:
            break
        i = rng.choice(idx1)
        j = rng.choice(idx0)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            break
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
