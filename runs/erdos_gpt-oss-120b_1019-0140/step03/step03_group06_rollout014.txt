def enforce_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return v
    # If diff>0, we need to add diff across indices not at 1.
    # If diff<0, subtract from indices not at 0.
    if diff > 0:
        indices = np.where(v < 1.0)[0]
        for idx in indices:
            can = 1.0 - v[idx]
            inc = min(can, diff)
            v[idx] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        indices = np.where(v > 0.0)[0]
        for idx in indices:
            can = v[idx]
            dec = min(can, diff)
            v[idx] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    # after loop diff may still remain due to numeric; distribute evenly; but likely not.
    # Return v
    return v

The order of indices is deterministic (ascending) and ensures sum target.

    dx = 2.0 / n
    # Ensure h is a copy? We'll modify h in-place; best_h maybe also copy.
    # We'll track best internally.

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # decide move type
        if rng.random() < 0.7:
            # continuous transfer

            # Determine i (source) as index with max contribution in worst shift
            if k >= 0:
                prod = h[: n - k] * (1.0 - h[k:])
                i_candidates = np.arange(n - k)
            else:
                kk = -k
                prod = h[kk:] * (1.0 - h[: n - kk])
                i_candidates = np.arange(kk, n)
            # choose i where prod is max and h_i > 0 (should be)
            i_local = np.argmax(prod)
            i = i_candidates[i_local]

            dst_idx = np.where(h < 1.0 - 1e-12)[0]
            if dst_idx.size == 0:
                # no headroom - skip

            # choose dst maybe random among those not equal to i
            if dst_idx.size == 1 and dst_idx[0] == i:
                continue
            dst = rng.choice(dst_idx[dst_idx != i] if dst_idx.size > 1 else dst_idx)
            src = i
            max_transfer = min(h[src], 1.0 - h[dst])
            if max_transfer <= 0:
                continue

            h[src] -= delta
            h[dst] += delta

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target):
    """Project v onto the box [0,1] while enforcing sum(v)=target (deterministic)."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        idx = np.where(v < 1.0)[0]
        for i in idx:
            inc = min(1.0 - v[i], diff)
            v[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.where(v > 0.0)[0]
        for i in idx:
            dec = min(v[i], diff)
            v[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return v

def random_binary_vector(n, target, rng):

    ones = int(round(target))

            h[src], h[dst] = 0.0, 1.0
            move_continuous = False

        # evaluate new value
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = new_corr.max() * dx

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            total = deadline - start_time
            T = base_temp * (remaining / total) if total > 0 else 0.0
            if T > 0.0 and np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                if move_continuous:
                    h[src] += delta
                    h[dst] -= delta
                else:
                    h[src], h[dst] = 1.0, 0.0
    return best_h, best_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # candidate sample sizes
    n_candidates = []
    n0 = None
    if "initial_h_values" in globals():
        n0 = int(initial_h_values.shape[0])
        n_candidates.append(n0)
    for n in (96, 112, 128):
        if n != n0:
            n_candidates.append(n)

    global_best_h = None
    global_best_val = np.inf
    global_best_n = None

    # allocate time roughly equally among remaining n values
    while time.time() < deadline - 0.5 and n_candidates:
        n = n_candidates.pop(0)
        target = n / 2.0
        dx = 2.0 / n

        # time slice for this n
        remaining_n = len(n_candidates) + 1
        slice_time = (deadline - time.time()) / remaining_n
        n_deadline = time.time() + slice_time

        # helper to obtain an initial vector for this n
        def init_h():
            if n == n0 and "initial_h_values" in globals():
                h = np.array(initial_h_values, dtype=float)
                h = enforce_sum(h, target)
                return h
            return random_binary_vector(n, target, rng).astype(float)

        best_h_n = None
        best_val_n = np.inf

        # multiple independent annealing attempts within the slice
        while time.time() < n_deadline - 0.5:
            # start point
            if best_h_n is not None and rng.random() < 0.5:
                h = best_h_n.copy()
                for _ in range(3):
                    src = np.where(h > 1e-8)[0]
                    dst = np.where(h < 1.0 - 1e-8)[0]
                    if src.size == 0 or dst.size == 0:
                        break
                    i = rng.choice(src)
                    j = rng.choice(dst)
                    max_delta = min(h[i], 1.0 - h[j])
                    if max_delta <= 0:
                        continue
                    delta = rng.random() * max_delta
                    h[i] -= delta
                    h[j] += delta
                h = enforce_sum(h, target)
            else:
                h = init_h()

            cur_val = compute_c5(h)
            best_h_n, best_val_n = anneal(
                h,
                cur_val,
                best_h_n if best_h_n is not None else h,
                best_val_n,
                start,
                n_deadline,
                rng,
                base_temp=0.02,
            )
            if best_val_n < global_best_val:
                global_best_val = best_val_n
                global_best_h = best_h_n.copy()
                global_best_n = n
            if global_best_val < 0.3808:
                # sufficient bound achieved
                final_h = enforce_sum(np.clip(global_best_h, 0.0, 1.0), global_best_n / 2.0)
                return final_h.astype(float), compute_c5(final_h), global_best_n

        # after exhausting slice for this n
        if best_val_n < global_best_val:
            global_best_val = best_val_n
            global_best_h = best_h_n.copy()
            global_best_n = n

    # fallback safety
    if global_best_h is None:
        n = 80
        target = n / 2.0
        global_best_h = random_binary_vector(n, target, rng).astype(float)
        global_best_n = n
        global_best_val = compute_c5(global_best_h)

    # final projection onto feasible set
    global_best_h = enforce_sum(np.clip(global_best_h, 0.0, 1.0), global_best_n / 2.0)
    final_val = compute_c5(global_best_h)
    return global_best_h.astype(float), final_val, global_best_n
