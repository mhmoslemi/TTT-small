def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num**0.5) + 1
    for p in range(3, r, 2):
        if num % p == 0:
            return False
    return True

def slsqp_refine(h0, n, rng, time_budget):
    M = n // 2
    # variable x = [h (n), t]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    # define constraints: equality sum(h) == M, inequality t >= overlap(k)
    # Overlap constraint function uses same pattern as before:
    def sum_constraint(x):
        return np.sum(x[:n]) - M
    # Overlap constraints for each k
    # we can precompute indices for each k to speed up? maybe okay.
    overlap_cons = []
    for k in range(-n+1, n):
        if k == 0:
            continue
        def make_fun(k):
            def fun(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * (2.0 / n)
            return fun
        overlap_cons.append({'type': 'ineq', 'fun': make_fun(k)})
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    constraints += overlap_cons
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # Use minimize
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None, None
    if not res.success:
        # try again maybe not needed
        return None, None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum exact
    cur = h_opt.sum()
    if abs(cur - M) > 1e-8:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

import time
import numpy as np
from scipy.optimize import minimize

def _is_prime(num: int) -> bool:
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5) + 1
    for p in range(3, r, 2):
        if num % p == 0:
            return False
    return True

def _legendre_seq(p: int) -> np.ndarray:
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def _legendre_initial(n: int, rng: np.random.Generator):
    # try n-1 prime (append 1) else n+1 prime (drop first)
    if _is_prime(n - 1):
        p = n - 1
        leg = _legendre_seq(p)
        h = np.concatenate([leg, [1.0]])
    elif _is_prime(n + 1):
        p = n + 1
        leg = _legendre_seq(p)
        h = leg[1:]  # drop zero at index 0
    else:
        return None
    shift = rng.integers(n)
    return np.roll(h, shift).astype(float)

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    m = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, m, replace=False)
    h[idx] = 1.0
    return h

def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    h = _legendre_initial(n, rng)
    if h is not None:
        return h
    return _random_binary(n, rng)

def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _project_sum_box(h: np.ndarray, target: float) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h += (target - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target) > 1e-12:
            res = target - cur
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size:
                inc = res / free.size
                h[free] += inc
                h = np.clip(h, 0.0, 1.0)
    return h

def _simulated_annealing(h0: np.ndarray, n: int, budget: float,
                         rng: np.random.Generator) -> tuple:
    M = n // 2
    cur = _project_sum_box(h0, M)
    cur_c5 = _compute_c5(cur)
    best_h, best_c5 = cur.copy(), cur_c5
    start = time.time()
    T_start, T_end = 0.1, 1e-7
    while True:
        now = time.time()
        if now - start > budget - 0.5:
            break
        frac = (now - start) / budget
        T = T_start * (T_end / T_start) ** frac
        r = rng.random()
        if r < 0.4:  # binary swap if possible
            i, j = rng.choice(n, 2, replace=False)
            if cur[i] < 0.5 <= cur[j]:
                h_new = cur.copy()
                h_new[i], h_new[j] = 1.0, 0.0
            elif cur[j] < 0.5 <= cur[i]:
                h_new = cur.copy()
                h_new[i], h_new[j] = 0.0, 1.0
            else:
                continue
        elif r < 0.8:  # fractional transfer
            i, j = rng.choice(n, 2, replace=False)
            max_tr = min(cur[i], 1.0 - cur[j])
            if max_tr <= 0:
                max_tr = min(cur[j], 1.0 - cur[i])
                if max_tr <= 0:
                    continue
                delta = rng.random() * max_tr
                h_new = cur.copy()
                h_new[j] -= delta
                h_new[i] += delta
            else:
                delta = rng.random() * max_tr
                h_new = cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
        else:  # small Gaussian jitter
            h_new = cur + rng.normal(scale=0.03, size=n)
        h_new = _project_sum_box(h_new, M)
        new_c5 = _compute_c5(h_new)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur, cur_c5 = h_new, new_c5
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
                if best_c5 <= 0.3808:
                    return best_h, best_c5
    return best_h, best_c5

def _slsqp_refine(h0: np.ndarray, n: int, time_left: float) -> tuple:
    if time_left < 5.0:
        return None, None
    M = n // 2
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    def sum_eq(x):
        return np.sum(x[:n]) - M

    overlap_cons = []
    dx = 2.0 / n
    for k in range(-n + 1, n):
        if k == 0:
            continue

        def make_fun(k):
            def fun(x, k=k):
                h = x[:n]
                if k >= 0:
                    o = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    o = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - o * dx
            return fun

        overlap_cons.append({'type': 'ineq', 'fun': make_fun(k)})

    constraints = [{'type': 'eq', 'fun': sum_eq}] + overlap_cons
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None, None
    if not res.success:
        return None, None
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = _compute_c5(h_opt)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h, best_c5, best_n = None, np.inf, None

    candidates = [128, 112, 96, 80]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates

    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)

        # initial vector
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            init = np.array(initial_h_values, dtype=float)
        else:
            init = _initial_sequence(n, rng)

        h_opt, c5_opt = _simulated_annealing(init, n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 <= 0.3808:
                break

    # optional local refinement
    if best_h is not None and best_c5 > 0.3808:
        time_left = budget_s - (time.time() - start)
        h_ref, c5_ref = _slsqp_refine(best_h, best_n, time_left)
        if h_ref is not None and c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = _compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        M = n // 2
        cur = best_h.sum()
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
