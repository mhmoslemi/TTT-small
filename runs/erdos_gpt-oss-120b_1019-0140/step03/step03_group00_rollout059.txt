def de_optimize(pop, n, s, max_time, rng):
    # pop shape (pop_size, n)
    pop_size = pop.shape[0]
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.8
    CR = 0.9
    while True:
        now = time.time()
        if now >= max_time:
            break
        for i in range(pop_size):
            # choose three distinct indices not i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # clip to [0,1] then project sum
            mutant = np.clip(mutant, 0.0, 1.0)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def adam_optimize(h0, n_iter, step0, beta1, beta2, eps, rng, max_time):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for it in range(1, n_iter + 1):
        now = time.time()
        if now >= max_time:
            break
        t += 1
        # compute current max correlation shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(t) + eps)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def de_optimize(pop, n, s, max_time, rng):
    pop_size, n = pop.shape
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.8
    CR = 0.9
    while True:
        now = time.time()
        if now >= max_time:
            break
        for i in range(pop_size):
            # pick distinct a,b,c
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, s)
            # Crossover
            jrand = rng.integers(n)
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, s)
            tr_score = compute_c5(trial)
            if tr_score <= scores[i]:
                pop[i] = trial
                scores[i] = tr_score
                if tr_score < best_score:
                    best_score = tr_score
                    best_h = trial.copy()
        # end for
    return best_h, best_score

def adam_optimize(h0, n_iter, step0, beta1, beta2, eps, rng, max_time):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for it in range(1, n_iter + 1):
        now = time.time()
        if now >= max_time:
            break
        t += 1
        # compute shifts giving max correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(t) + eps)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb(h, max_iters, rng, max_time):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    temp = 0.1
    cooling = 0.999
    for it in range(max_iters):
        now = time.time()
        if now >= max_time:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)
        delta = score - best_score
        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            best_h = h_new
            best_score = score
        temp *= cooling
    return best_h, best_score

import time
import numpy as np
import math

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def rudin_shapiro(n):
    seq = np.empty(n, dtype=float)
    for i in range(n):
        cnt = bin(i & (i >> 1)).count('1')
        seq[i] = 1.0 if (cnt % 2 == 0) else 0.0
    return seq

def random_balanced_vector(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def de_optimize(pop, s, max_time, rng):
    pop_size, n = pop.shape
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = scores.argmin()
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.8
    CR = 0.9
    while True:
        if time.time() >= max_time:
            break
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, s)
            jrand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial = pop[i].copy()
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_onto_box_sum(trial, s)
            tr_score = compute_c5(trial)
            if tr_score <= scores[i]:
                pop[i] = trial
                scores[i] = tr_score
                if tr_score < best_score:
                    best_score = tr_score
                    best_h = trial.copy()
    return best_h, best_score

def adam_optimize(h0, n_iter, step0, beta1, beta2, eps, rng, max_time):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for it in range(1, n_iter + 1):
        if time.time() >= max_time:
            break
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(t) + eps)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binarize_by_top(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    bh = np.zeros_like(h)
    bh[idx] = 1.0
    return bh

def binary_hill_climb(h, max_iters, rng, max_time):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    temp = 0.1
    cooling = 0.999
    for it in range(max_iters):
        if time.time() >= max_time:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        score = compute_c5(h_new)
        delta = score - best_score
        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            best_h = h_new
            best_score = score
        temp *= cooling
    return best_h, best_score
