def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=80):
    low = -np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2.0, lo, hi)
```

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.3
    beta = 0.5
    v = np.zeros_like(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # compute gradient for worst shift
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # momentum
        v = beta * v - alpha * grad
        new_h = h + v
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            # increase step size slightly
            alpha = min(alpha * 1.02, 1.0)
        else:
            # reject, reduce step size and reset momentum
            alpha = max(alpha * 0.5, 1e-6)
            v = np.zeros_like(h)
            # optionally add small random perturb
            if alpha < 1e-4:
                noise = np.random.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=300):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=lambda x: x[-1],
                   x0=x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
```

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    safety = 3.0  # seconds buffer

    # set dimension
    n = 200
    target_sum = n / 2.0
    # initial vector
    if 'initial_h_values' in globals():
        h_small = np.asarray(initial_h_values, dtype=float)
        # upsample
        x_small = np.linspace(0, 2, h_small.shape[0], endpoint=False)
        x_large = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x_large, x_small, h_small)
    else:
        h0 = rng.random(n)
    # project to feasible set
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # optional quick SLSQP to improve starting point (spend small time)
    if time.time() - start_time < budget_s * 0.03:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=200)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # prepare constraints for later refinements
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]

    # main loop
    while time.time() < start_time + budget_s - safety:
        # choose start: perturb best or random
        if rng.random() < 0.6:
            h_start = best_h + rng.normal(scale=0.02, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        # gradient descent
        gd_deadline = start_time + budget_s - safety
        gd_h, gd_c5 = gradient_descent(h_start, target_sum, max_iter=2500, time_limit=gd_deadline)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
            if best_c5 < 0.3808:
                break

        # occasional SLSQP refinement
        remain = start_time + budget_s - safety - time.time()
        if remain > 4.0:
            t_start = compute_c5(gd_h)
            res = slsqp_optimize(gd_h, t_start, sum_con, overlap_cons, maxiter=250)
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    if best_c5 < 0.3808:
                        break

    # final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=80):
    """Project v onto the box [lo,hi]^n while enforcing sum(v)=target."""
    low = -np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (ignoring dx)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality constraint: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=300):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.3
    beta = 0.5
    v = np.zeros_like(h)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        v = beta * v - alpha * grad
        new_h = h + v
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.02, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-6)
            v = np.zeros_like(h)
            if alpha < 1e-4:
                noise = rng.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

def upsample_h(h_small, n_large):
    n_small = h_small.shape[0]
    x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
    x_large = np.linspace(0.0, 2.0, n_large, endpoint=False)
    return np.interp(x_large, x_small, h_small)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    safety = 3.0

    n = 200
    target_sum = n / 2.0

    if 'initial_h_values' in globals():
        h_small = np.asarray(initial_h_values, dtype=float)
        h0 = upsample_h(h_small, n)
    else:
        h0 = rng.random(n)
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # quick initial SLSQP refinement (optional)
    if time.time() - start_time < budget_s * 0.03:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n)]
        t0 = best_c5
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=200)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n)]

    while time.time() < start_time + budget_s - safety:
        if rng.random() < 0.6:
            h_start = best_h + rng.normal(scale=0.02, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        gd_deadline = start_time + budget_s - safety
        gd_h, gd_c5 = gradient_descent(h_start, target_sum, max_iter=2500,
                                      time_limit=gd_deadline, rng=rng)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
            if best_c5 < 0.3808:
                break

        remain = start_time + budget_s - safety - time.time()
        if remain > 5.0:
            t_start = compute_c5(gd_h)
            res = slsqp_optimize(gd_h, t_start, sum_con, overlap_cons, maxiter=250)
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    if best_c5 < 0.3808:
                        break

    # final optional refinement if time permits
    if time.time() < start_time + budget_s - safety - 5.0:
        t_start = compute_c5(best_h)
        res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
