def subgradient_max_corr(h, eps=1e-12):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # indices of max
    idx = np.where(corr >= max_corr - eps)[0]
    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # 1 - h[i+shift] - h[i-shift] where applicable
            # create arrays term_fwd and term_bwd
            term_fwd = np.zeros(n)
            term_fwd[:n-shift] = h[shift:]
            term_bwd = np.zeros(n)
            term_bwd[shift:] = h[:n-shift]
        else:
            s = -shift
            term_fwd = np.zeros(n)
            term_fwd[s:] = h[:n-s]
            term_bwd = np.zeros(n)
            term_bwd[:n-s] = h[s:]
        grad += (1.0 - term_fwd - term_bwd)
    grad /= len(shifts)
    return grad
```

def adam_optimize(h0, time_end, max_iter=100000, lr=0.1):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        if time.time() > time_end:
            break
        grad = subgradient_max_corr(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - step
        h = project_onto_box_sum(h, h.size / 2.0)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def subgradient_max_corr(h, eps=1e-12):
    """Subgradient of max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term_fwd = np.zeros_like(h)
            term_fwd[:n - shift] = h[shift:]
            term_bwd = np.zeros_like(h)
            term_bwd[shift:] = h[:n - shift]
        else:
            s = -shift
            term_fwd = np.zeros_like(h)
            term_fwd[s:] = h[:n - s]
            term_bwd = np.zeros_like(h)
            term_bwd[:n - s] = h[s:]
        grad += (1.0 - term_fwd - term_bwd)
    grad /= len(shifts)
    return grad

def adam_optimize(h0, time_end, max_iter=200000, lr=0.2):
    """Adam‑style projected optimization for the max‑correlation objective."""
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while t < max_iter and time.time() < time_end:
        t += 1
        grad = subgradient_max_corr(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, h.size / 2.0)
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    global_end = start + budget_s - 0.3

    sizes = [256, 512, 1024]                # multiscale resolutions
    # --- smallest scale candidates ------------------------------------------------
    n0 = sizes[0]
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n0), n0 / 2.0)
        candidates.append(cand)
    candidates.append(thue_morse_binary(n0))
    candidates.append(legendre_sequence(257))          # length 256
    for _ in range(3):
        candidates.append(random_binary_balanced(n0, rng))
    for _ in range(2):
        v = rng.random(n0)
        candidates.append(project_onto_box_sum(v, n0 / 2.0))

    best_h = None
    best_score = np.inf
    # refine each candidate on the smallest grid
    for cand in candidates:
        if time.time() >= global_end - 1.0:

        h_opt, sc_opt = adam_optimize(cand, time_end=global_end, max_iter=5000, lr=0.2)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    if best_h is None:
        best_h = random_binary_balanced(n0, rng)
        best_score = compute_c5(best_h)

    # --- progressive upsampling and refinement -----------------------------------
    for size in sizes[1:]:
        if time.time() >= global_end - 1.0:
            break
        best_h = resample(best_h, size)
        best_h = project_onto_box_sum(best_h, size / 2.0)
        best_score = compute_c5(best_h)
        # allocate iterations proportional to remaining time
        remaining = global_end - time.time()
        max_iters = int(min(200000, max(20000, remaining * 800)))
        lr = 0.15 if size == 512 else 0.1
        h_opt, sc_opt = adam_optimize(best_h, time_end=global_end,
                                      max_iter=max_iters, lr=lr)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # --- final random restarts if time permits ------------------------------------
    while time.time() < global_end - 0.5 and best_score > 0.3808:
        v = rng.random(best_h.size)
        h0 = project_onto_box_sum(v, best_h.size / 2.0)
        h_opt, sc_opt = adam_optimize(h0, time_end=global_end,
                                      max_iter=5000, lr=0.05)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # final safety projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
