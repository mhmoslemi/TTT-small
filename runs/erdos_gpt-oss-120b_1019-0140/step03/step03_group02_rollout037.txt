import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[: n - s] += (1.0 - h[s:])
            grad[s:] -= h[: n - s]
        else:
            p = -s
            grad[p:] += (1.0 - h[: n - p])
            grad[: n - p] -= h[p:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    diff = alpha * (corr - maxc)
    diff = np.clip(diff, -700, 700)
    exp_vals = np.exp(diff)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[: n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[: n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[: n - p])
            grad[: n - p] -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxc = np.max(corr)
        eps = 1e-12
        shifts = np.where(corr >= maxc - eps)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s_target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def continuous_sa(h0, rng, time_left, target=None):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    while time_left() > 0.02:
        elapsed = time.time() - start
        frac = elapsed / (time_left() + elapsed + 1e-9)
        T = max(0.0, (1 - frac)) * 0.02
        max_step = max(0.001, (1 - frac) * 0.2)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2 * max_step
        if delta > 0:
            delta = min(delta, 1 - h[j], h[i])
        else:
            delta = -min(-delta, h[j], 1 - h[i])
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)

            if target is not None and best_score <= target:

        if new_score < cur_score or (T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T)):
            h, cur_score = h_new, new_score
    return best_h, best_score
