def objective_and_grad(h, n, target_sum, alpha, penalty_coeff):
    dx = 2.0 / n
    s = np.correlate(h, 1.0 - h, mode='full')
    mid = n-1
    # remove zero shift
    s_nozero = np.concatenate((s[:mid], s[mid+1:]))
    # compute smooth max
    max_s = np.max(s_nozero)
    diff = s_nozero - max_s
    exp_scaled = np.exp(alpha * diff)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    smooth_max = max_s + np.log(sum_exp) / alpha
    # compute gradient
    grad = np.zeros_like(h)
    w_idx = 0
    for idx in range(2*n-1):
        if idx == mid:
            continue
        shift = idx - mid
        weight = w[w_idx]
        w_idx += 1
        if shift >= 0:
            seg_len = n - shift
            grad[:seg_len] += weight * (1 - h[shift:])
            grad[shift:] += weight * (-h[:seg_len])
        else:
            kk = -shift
            seg_len = n - kk
            grad[kk:] += weight * (1 - h[:seg_len])
            grad[:seg_len] += weight * (-h[kk:])
    # final gradient with scaling
    grad = dx * grad
    # penalty
    sum_h = np.sum(h)
    diff_sum = sum_h - target_sum
    penalty = penalty_coeff * diff_sum * diff_sum
    grad += 2.0 * penalty_coeff * diff_sum
    obj = smooth_max * dx + penalty
    return obj, grad

def compute_c5(h):
    """Maximum overlap integral C5 for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo, hi]^n while fixing sum to target via binary search."""
    low = -np.max(v) - 10.0
    high = (hi - np.min(v)) + 10.0

        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid

    Returns (obj, grad) where grad is the derivative w.r.t. h.
