def grad_logsumexp(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    # compute weights
    max_corr = np.max(corr)
    # shift values for p
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)  # shape (2n-1)
    n = h.shape[0]
    grad = np.ones_like(h)  # base 1 term
    # term1: sum_{s} w_s * term1_s
    # term2: similarly
    # We can compute term1 and term2 contributions via loops.
    # For each index s from -(n-1) to (n-1): s = i - (n-1)
    # Equivalent to i enumerates w indices.
    for i, weight in enumerate(w):
        s = i - (n - 1)
        if s >= 0:
            # term1: h[s:] -> placed at indices 0 to n-s-1
            grad[:n - s] -= weight * h[s:]
            # term2: h[:n-s] -> placed at indices s to n-1
            grad[s:] -= weight * h[:n - s]
        else:
            t = -s
            # term1: h[:n-t] -> placed at indices t to n-1
            grad[t:] -= weight * h[:n - t]
            # term2: h[t:] -> placed at indices 0 to n-t-1
            grad[:n - t] -= weight * h[t:]
    return grad

Note that weight distribution sum to 1, so base 1 term is correct.

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        sm = proj.sum()
        if abs(sm - s) <= tol:
            return proj
        if sm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Discrete C5 = max_k ∫ h(x)(1−h(x+k))dx."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):

def grad_logsumexp(h, alpha):

            t = -s
            if n - t > 0:
                grad[t:] -= weight * h[:n - t]
            if t < n:
                grad[:n - t] -= weight * h[t:]
