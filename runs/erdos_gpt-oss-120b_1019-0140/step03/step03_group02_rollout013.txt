def delta_corr_swap(h, i, j):
    n = h.shape[0]
    shifts = np.arange(-n+1, n, dtype=int)
    start = np.maximum(0, -shifts)
    end = n - 1 - np.maximum(0, shifts)
    # bool masks
    mask_i = (i >= start) & (i <= end)
    mask_j = (j >= start) & (j <= end)
    mask_i_k = ((i - shifts) >= start) & ((i - shifts) <= end)
    mask_j_k = ((j - shifts) >= start) & ((j - shifts) <= end)

    # contributions
    delta = np.zeros_like(shifts, dtype=int)
    # compute indices arrays for efficient indexing
    # For mask_i, idx = i + shifts[mask_i]
    # (i + shifts) can be computed vectorized:
    idx_i = i + shifts[mask_i]
    delta[mask_i] -= (1 - h[idx_i])

    idx_j = j + shifts[mask_j]
    delta[mask_j] += (1 - h[idx_j])

    idx_i_k = i - shifts[mask_i_k]  # i - k = index
    delta[mask_i_k] += h[idx_i_k]

    idx_j_k = j - shifts[mask_j_k]
    delta[mask_j_k] -= h[idx_j_k]

    return delta

def delta_corr_swap(h, i, j):
    n = h.shape[0]
    shifts = np.arange(-n+1, n, dtype=int)
    start = np.maximum(0, -shifts)
    end = n - 1 - np.maximum(0, shifts)
    # Masks
    mask_i = (i >= start) & (i <= end)
    mask_j = (j >= start) & (j <= end)
    mask_i_k = ((i - shifts) >= start) & ((i - shifts) <= end)
    mask_j_k = ((j - shifts) >= start) & ((j - shifts) <= end)

    delta = np.zeros_like(shifts, dtype=int)

    if np.any(mask_i):
        idx_i = i + shifts[mask_i]
        delta[mask_i] -= (1 - h[idx_i])
    if np.any(mask_j):
        idx_j = j + shifts[mask_j]
        delta[mask_j] += (1 - h[idx_j])
    if np.any(mask_i_k):
        idx_i_k = i - shifts[mask_i_k]
        delta[mask_i_k] += h[idx_i_k]
    if np.any(mask_j_k):
        idx_j_k = j - shifts[mask_j_k]
        delta[mask_j_k] -= h[idx_j_k]
    return delta

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    shifts = np.arange(-(n - 1), n)
    for idx, k in enumerate(shifts):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def hill_climb_sa_binary(h0, rng, time_left, target_score=None):
    h = h0.copy()
    n = h.shape[0]
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = score

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    # set max_iters big
    max_iters = 2000000  # large
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # random swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        delta = delta_corr_swap(h, i, j)
        new_corr = corr + delta
        new_score = np.max(new_corr) * (2.0 / n)
        delta_score = new_score - score
        T = T0 * (1.0 - it / max_iters)
        if delta_score < 0.0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            # accept
            h[i] = 0
            h[j] = 1
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            corr = new_corr
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if target_score is not None and best_score <= target_score:
                    break
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def delta_corr_swap(h, i, j):
    """Return the correlation change vector when swapping h[i]=1 with h[j]=0."""
    n = h.shape[0]
    shifts = np.arange(-n + 1, n, dtype=int)
    start = np.maximum(0, -shifts)
    end = n - 1 - np.maximum(0, shifts)

    mask_i = (i >= start) & (i <= end)
    mask_j = (j >= start) & (j <= end)
    mask_i_k = ((i - shifts) >= start) & ((i - shifts) <= end)
    mask_j_k = ((j - shifts) >= start) & ((j - shifts) <= end)

    delta = np.zeros_like(shifts, dtype=int)

    if mask_i.any():
        idx_i = i + shifts[mask_i]
        delta[mask_i] -= (1 - h[idx_i])
    if mask_j.any():
        idx_j = j + shifts[mask_j]
        delta[mask_j] += (1 - h[idx_j])
    if mask_i_k.any():
        idx_i_k = i - shifts[mask_i_k]
        delta[mask_i_k] += h[idx_i_k]
    if mask_j_k.any():
        idx_j_k = j - shifts[mask_j_k]
        delta[mask_j_k] -= h[idx_j_k]
    return delta

def hill_climb_sa_binary(h0, rng, time_left, target_score=None):
    """Simulated‑annealing style binary local search."""
    h = h0.copy()
    n = h.shape[0]
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    score = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_score = score

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    max_iters = 1000000
    T0 = 0.05
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        delta = delta_corr_swap(h, i, j)
        new_corr = corr + delta
        new_score = np.max(new_corr) * (2.0 / n)
        dscore = new_score - score
        T = T0 * (1.0 - it / max_iters)
        if dscore < 0.0 or rng.random() < np.exp(-dscore / max(T, 1e-12)):
            h[i] = 0.0
            h[j] = 1.0
            ones[i_idx] = j
            zeros[j_idx] = i
            corr = new_corr
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if target_score is not None and best_score <= target_score:
                    break
    return best_h, best_score

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    shifts = np.arange(-(n - 1), n)
    for idx, k in enumerate(shifts):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 256                     # grid resolution
    target_bound = 0.3808

    candidates = []
    if 'initial_h_values' in globals():
