def anneal_binary(h_start, time_end):

    best_h = h_start.copy()
    best_c5 = compute_c5_fast(best_h)
    h = h_start.copy()

    # maybe precompute autocorrelation array for speed? Not needed.

        elapsed = time.time() - start
        # Temperature schedule: linear decay from T0 to Tf
        frac = elapsed / (time_end - start)
        T = max(1e-4, (1.0 - frac) * 0.05)  # maybe T0=0.05 ?

        # targeted move probability
        if rng.random() < 0.3:
            # targeted swap based on min shift
            # compute autocorrelation once
            corr = np.correlate(h, h, mode='full')
            # min shift index (excluding center)
            # We need shift s where A_s minimal.
            # Compute min for s>0
            # shift s corresponds to Corr[n-1 + s]
            # Seek s in 1..n-1 that yields min
            s_candidates = np.arange(1, n)
            A_vals = corr[n - 1 + s_candidates]
            s_min = s_candidates[np.argmin(A_vals)]
            # find potential positions to increase A_{s_min}
            # Option: find indices where h[i]==0 and h[i+s_min]==1 (i between 0 and n-1-s_min)
            zero_i = np.where((h[:n-s_min] == 0) & (h[s_min:] == 1))[0]
            # find one_i where h[i]==1 and h[i+s_min]==0
            one_i = np.where((h[:n-s_min] == 1) & (h[s_min:] == 0))[0]
            if zero_i.size > 0 and one_i.size > 0:
                i0 = rng.choice(zero_i)
                j0 = rng.choice(one_i)
            else:
                # fallback random
                i0 = rng.integers(0, n)
                j0 = rng.integers(0, n)
                while h[i0] == h[j0]:
                    i0 = rng.integers(0, n)
                    j0 = rng.integers(0, n)
        else:
            # random swap
            i0 = rng.integers(0, n)
            j0 = rng.integers(0, n)
            while h[i0] == h[j0]:
                i0 = rng.integers(0, n)
                j0 = rng.integers(0, n)

        # swap
        h[i0], h[j0] = h[j0], h[i0]
        new_c5 = compute_c5_fast(h)
        # acceptance

    h = h_start.copy()
    cur_c5 = compute_c5_fast(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start_local = time.time()

    Tf = 1e-5

    while time.time() < time_end:

        elapsed = time.time() - start_local
        total = max(time_end - start_local, 1e-9)
        frac = elapsed / total
        T = T0 * (1 - frac) + Tf * frac
        # Decide move type
        if rng.random() < 0.3:
            # targeted move
            # compute autocorrelation
            corr = np.correlate(h, h, mode='full')
            # ignore shift 0
            # find shift s for which corr[n:] is minimum
            # compute min without including shift 0
            # slice positive shifts
            pos_corr = corr[n:]  # length n-1
            s_min = np.argmin(pos_corr) + 1  # shift value
            # find candidate zero positions where h[i]==0 and h[i+s_min]==1
            # Ensure i+s_min < n
            zeros = np.where((h[:n - s_min] == 0) & (h[s_min:] == 1))[0]
            ones = np.where((h[:n - s_min] == 1) & (h[s_min:] == 0))[0]
            if zeros.size and ones.size:
                i0 = rng.choice(zeros)
                j0 = rng.choice(ones)
            else:
                # fallback random swap
                i0 = rng.integers(0, n)
                j0 = rng.integers(0, n)
                while h[i0] == h[j0]:
                    i0 = rng.integers(0, n)
                    j0 = rng.integers(0, n)
        else:
            # random swap
            i0 = rng.integers(0, n)
            j0 = rng.integers(0, n)
            while h[i0] == h[j0]:
                i0 = rng.integers(0, n)
                j0 = rng.integers(0, n)
        # Apply swap
        h[i0], h[j0] = h[j0], h[i0]
        new_c5 = compute_c5_fast(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):

            cur_c5 = new_c5

                best_c5 = new_c5
                best_h = h.copy()

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    """Exact C5 using correlation with (1-h)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')

            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # problem sizes
    n_small = 256
    n_large = 512

    # initial candidate pool
    candidates = []

    try:
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = resample(init_full, n_small)
        cand_bin = binarize_top_half(cand)
        candidates.append(cand_bin)
    except Exception:
        pass

    candidates.append(thue_morse_binary(n_small))
    for _ in range(2):
        candidates.append(random_balanced_binary(n_small, rng))

    best_h = None
    best_c5 = np.inf

    # allocate time for binary simulated annealing
    total_sa_time = min(0.6 * budget_s, max(30.0, time_left() - 200.0))
    sa_start = time.time()
    per_cand = total_sa_time / max(1, len(candidates))

    for cand in candidates:
        if time_left() < per_cand:
            break
        t_end = time.time() + per_cand
        h_opt, c5_opt = binary_anneal(cand, t_end, rng)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        if best_c5 <= 0.3808:
            break

    # extra random restarts while SA budget lasts
    while (time.time() - sa_start) < total_sa_time and time_left() > 20.0:
        cand = random_balanced_binary(n_small, rng)
        t_end = time.time() + min(per_cand, time_left() - 10.0)
        h_opt, c5_opt = binary_anneal(cand, t_end, rng)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        if best_c5 <= 0.3808:
            break

    if best_h is None:
        best_h = random_balanced_binary(n_small, rng)
        best_c5 = compute_c5_fast(best_h)

    # upsample to larger resolution and project
    h_large = resample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    c5_large = compute_c5_full(h_large)
    if c5_large < best_c5:
        best_c5, best_h = c5_large, h_large.copy()

    # gradient‑descent refinement on the current best
    remaining = time_left()
    if remaining > 10.0:
        time_end_gd = start + budget_s - 0.5
        max_iter = int(min(300000, max(20000, remaining * 800)))
        h_ref, c_ref = gradient_descent_momentum(best_h,
                                                 step0=0.5,
                                                 beta=0.9,
                                                 time_end=time_end_gd,
                                                 max_iter=max_iter)
        h_ref = project_onto_box_sum(h_ref, h_ref.size / 2.0)
        c_ref = compute_c5_full(h_ref)
        if c_ref < best_c5:
            best_c5, best_h = c_ref, h_ref.copy()

    # final safety projection and bound computation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_c5 = compute_c5_full(best_h)

    return best_h, float(best_c5), best_h.size
