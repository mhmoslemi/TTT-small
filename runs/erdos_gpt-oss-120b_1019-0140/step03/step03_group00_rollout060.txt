def smooth_max_corr_and_grad(h, T, dx):
    n = h.size
    m = 2*n - 1
    # compute correlation corr (full)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # compute softmax weights w
    max_c = corr.max()
    exp_vals = np.exp((corr - max_c) / T)
    w = exp_vals / exp_vals.sum()
    # compute FFT convs
    # compute conv1 = conv(1-h, w_rev)
    # conv2 = conv(h, w)
    # Prepare length L as next power of 2 >= n + m - 1
    L = 1 << ((n + m - 1).bit_length())
    # compute FFTs
    a = np.fft.rfft(1 - h, n=L)
    w_rev = w[::-1]
    b = np.fft.rfft(w_rev, n=L)
    conv1 = np.fft.irfft(a * b, n=L)  # real part
    c = np.fft.rfft(h, n=L)
    d = np.fft.rfft(w, n=L)

    grad = dx * (conv1[offset:offset + n] - conv2[offset:offset + n])

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    # clamp sum extremes
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def upsample(arr, new_n):
    old_n = arr.size
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

def smooth_max_corr_and_grad(h, T, dx):
    n = h.size
    # correlation
    corr = compute_corr_fft(h, dx)  # length 2n-1
    max_c = corr.max()
    # weights w using stable calculation
    # compute exp((corr - max_c)/T) to avoid overflow
    exp_vals = np.exp((corr - max_c) / T)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # smooth max
    smooth_max = max_c + T * np.log(sum_exp)
    # compute gradient via FFT
    m = 2*n - 1
    Lg = 1 << ((3*n - 2).bit_length())
    # FFTs of h and 1-h
    fft_h = np.fft.rfft(h, n=Lg)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=Lg)
    # FFTs of w and w_rev
    w_rev = w[::-1]
    fft_w = np.fft.rfft(w, n=Lg)
    fft_w_rev = np.fft.rfft(w_rev, n=Lg)
    conv1 = np.fft.irfft(fft_one_minus_h * fft_w_rev, n=Lg)
    conv2 = np.fft.irfft(fft_h * fft_w, n=Lg)
    offset = m - 1  # 2n-2
    grad = dx * (conv1[offset:offset + n] - conv2[offset:offset + n])
    return smooth_max, grad, max_c
```

def adam_optimize(h0, n, time_left, max_iters=50000):
    h = project_onto_box_sum(h0, n / 2.0)
    dx = 2.0 / n
    # initial T based on initial spread
    corr = compute_corr_fft(h, dx)
    max_c = corr.max()
    spread = max_c - corr.min()
    T = max(spread * 0.5, 1e-4)
    Tmin = 5e-5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = max_c
    it = 0
    while time_left() > 0.2 and it < max_iters:
        it += 1
        smooth_max, grad, cur_max = smooth_max_corr_and_grad(h, T, dx)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, n / 2.0)
        # evaluate actual max correlation (cur_max from this iteration)
        if cur_max < best_score:
            best_score = cur_max
            best_h = h.copy()
            if best_score <= 0.3808:
                break
        # anneal temperature
        T = max(T * 0.9995, Tmin)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    old_n = arr.size
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_c5(h):
    """Exact C5 using direct correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr_fft(h, dx):
    """Full correlation h * (1-h) via FFT, scaled by dx."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * B, n=L)[:2 * n - 1]
    return conv * dx

def smooth_max_corr_and_grad(h, T, dx):
    """
    Returns (smooth_max, gradient, max_corr) where gradient is
    d/dh of the smooth max of the correlation values.
    """
    n = h.size
    # correlation vector
    corr = compute_corr_fft(h, dx)          # length 2n-1
    max_c = corr.max()
    # softmax weights (stable)
    exp_vals = np.exp((corr - max_c) / T)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    smooth_max = max_c + T * np.log(sum_exp)

    m = 2 * n - 1
    Lg = 1 << ((3 * n - 2).bit_length())
    # FFTs of h and (1-h)
    fft_h = np.fft.rfft(h, n=Lg)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=Lg)
    # FFTs of weights
    w_rev = w[::-1]
    fft_w = np.fft.rfft(w, n=Lg)
    fft_w_rev = np.fft.rfft(w_rev, n=Lg)
    # convolutions
    conv1 = np.fft.irfft(fft_one_minus_h * fft_w_rev, n=Lg)
    conv2 = np.fft.irfft(fft_h * fft_w, n=Lg)
    offset = m - 1          # 2n-2
    grad = dx * (conv1[offset:offset + n] - conv2[offset:offset + n])
    return smooth_max, grad, max_c

def adam_optimize(h0, n, time_left, max_iters=50000):
    """Projected Adam on the smooth‑max objective."""
    h = project_onto_box_sum(h0, n / 2.0)
    dx = 2.0 / n
    # initial temperature based on spread of correlation
    corr = compute_corr_fft(h, dx)
    spread = corr.max() - corr.min()
    T = max(spread * 0.5, 1e-4)
    Tmin = 5e-5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = corr.max()
    it = 0
    while time_left() > 0.25 and it < max_iters:
        it += 1
        smooth_max, grad, cur_max = smooth_max_corr_and_grad(h, T, dx)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, n / 2.0)
        if cur_max < best_score:
            best_score = cur_max
            best_h = h.copy()
            if best_score <= 0.3808:
                break
        T = max(T * 0.9995, Tmin)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512, 1024]
    best_h = None
    best_score = np.inf

    for n in scales:
        if time_left() <= 2.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # allocate iteration budget proportional to remaining time and size
        est_iters = int(min(80000, max(4000, time_left() * (n / 64) * 15)))
        h_opt, sc_opt = adam_optimize(base, n, time_left, max_iters=est_iters)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

        # occasional random re‑starts at this resolution
        if time_left() > 5.0:
            for _ in range(3):
                rand = rng.random(n)
                rand = rand / rand.sum() * (n / 2.0)
                rand = project_onto_box_sum(rand, n / 2.0)
                h_r, sc_r = adam_optimize(rand, n, time_left, max_iters=3000)
                if sc_r < best_score:
                    best_score, best_h = sc_r, h_r.copy()
                    base = h_r

        # small local mass‑transfer perturbation
        if best_h is not None and time_left() > 2.0:
            h_pert = best_h.copy()
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(h_pert[i], 1.0 - h_pert[j]) * rng.uniform(0.1, 0.5)
            if delta > 0:
                h_pert[i] -= delta
                h_pert[j] += delta
                h_pert = project_onto_box_sum(h_pert, n / 2.0)
                cur = compute_corr_fft(h_pert, 2.0 / n).max()
                if cur < best_score:
                    best_score, best_h = cur, h_pert.copy()
                    base = h_pert

    # final intensive refinement on the best vector found
    if best_h is not None and time_left() > 4.0:
        best_h, best_score = adam_optimize(best_h, best_h.size, time_left,
                                           max_iters=200000)

    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, float(best_score), best_h.size
