def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    # Euclidean projection onto box and sum constraint
    n = v.size
    # Clipping to box
    v = np.clip(v, lo, hi)
    # If sum already equals s (within tolerance), return v
    if abs(v.sum() - s) <= 1e-12:
        return v
    # Use water-filling algorithm: find lambda such that sum clip(v - lambda) = s
    # Binary search lambda in suitable range
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # Ensure f(lam_low) >= s? Actually f(lam) = sum clip(v - lam)
    # Evaluate f at low and high
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Adjust bounds if needed
    while f(lam_low) > s:
        lam_low -= hi - lo + 1.0
    while f(lam_high) < s:
        lam_high += hi - lo + 1.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def gradient_descent(h0, step0=0.5, beta=0.9, max_iter=200000, time_end=None):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < max_iter and (time_end is None or time.time() < time_end):
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # gradient for this shift
        if shift >= 0:
            term1 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2 = np.zeros_like(h)
            term2[shift:] = h[:n - shift]
        else:
            sft = -shift
            term1 = np.zeros_like(h)
            term1[sft:] = h[:n - sft]
            term2 = np.zeros_like(h)
            term2[:n - sft] = h[sft:]
        grad = 1.0 - term1 - term2
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen bounds if needed
    while f(lam_low) > s:
        lam_low -= hi - lo + 1.0
    while f(lam_high) < s:
        lam_high += hi - lo + 1.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def next_prime(m):
    """Return the smallest prime >= m."""
    def is_prime(x):
        if x < 2:
            return False
        if x % 2 == 0:
            return x == 2
        r = int(np.sqrt(x)) + 1
        for d in range(3, r, 2):
            if x % d == 0:
                return False
        return True
    i = max(2, int(m))
    while not is_prime(i):
        i += 1
    return i

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def smooth_candidate(v, kernel):
    """Convolutional smoothing (kernel sums to 1)."""
    sm = np.convolve(v, kernel, mode='same')
    return sm

def slsqp_optimize_candidate(h0, max_iter=300):
    """Brief SLSQP refinement of a feasible vector."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def gradient_descent(h0, step0=0.5, beta=0.9, max_iter=200000, time_end=None):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < max_iter and (time_end is None or time.time() < time_end):
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = np.ones(n, dtype=float)
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:]   -= h[:n - shift]
        else:
            sft = -shift
            grad[sft:]     -= h[:n - sft]
            grad[:n - sft] -= h[sft:]
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = project_onto_box_sum(h - eta * momentum, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
