def project_feasible(z):
    z = np.clip(z, 0.0, 1.0)
    M = len(z) / 2.0
    for _ in range(10):
        s = np.sum(z)
        diff = M - s
        if abs(diff) < 1e-12:
            break
        # distribute diff equally among free slots
        mask = (z > 0.0) & (z < 1.0)
        if not np.any(mask):
            # all at bounds, can't adjust
            break
        delta = diff / np.count_nonzero(mask)
        z[mask] += delta
        # clip again for any overshoot
        np.clip(z, 0.0, 1.0, out=z)

    np.clip(z, 0.0, 1.0, out=z)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap C5 for a given h (including shift 0)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_k(h):
    """Return (C5, shift k achieving it) for current h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    return corr[idx] * dx, k

def project_feasible(z):
    """Project vector onto box [0,1]^n with exact sum M = n/2."""
    n = z.shape[0]
    M = n / 2.0
    z = np.clip(z, 0.0, 1.0)
    # iterative water-filling
    for _ in range(10):
        s = z.sum()
        diff = M - s
        if abs(diff) < 1e-12:
            break
        mask = (z > 0.0) & (z < 1.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            # all at bounds, distribute uniformly (should rarely happen)
            z += diff / n
            np.clip(z, 0.0, 1.0, out=z)
            continue
        delta = diff / cnt
        z[mask] += delta
        np.clip(z, 0.0, 1.0, out=z)
    # final tiny correction if needed
    s = z.sum()
    diff = M - s
    if abs(diff) > 1e-9:
        # adjust a single interior entry if possible
        idx = np.argmin(np.abs(z - 0.5))
        z[idx] = np.clip(z[idx] + diff, 0.0, 1.0)
    return z
