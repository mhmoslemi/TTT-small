    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()

        if time.time() > deadline:
            break
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # shift (could be negative)
        # consider positive k only
        if k < 0:
            k = -k
        # Compute mismatched set A and B for shift k
        length = n - k
        if length <= 0:
            continue
        h_i = cur_h[:length]
        h_ipk = cur_h[k:]
        A = np.where((h_i == 1) & (h_ipk == 0))[0]
        B = np.where((h_i == 0) & (h_ipk == 1))[0]
        if len(A) == 0 or len(B) == 0:
            # no direct improving swap for this shift, maybe try random swap
            # Perform random swap of 1 and 0
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(A)
            j = rng.choice(B)
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = compute_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                # early exit if meets target
                if best_val <= target_val:
                    break
        else:
            # undo swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

def solve_mip(n, time_limit, rng):
    import cvxpy as cp
    target = n // 2
    dx = 2.0 / n

    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    constraints = [cp.sum(h) == target]

    # Add overlap constraints
    for k in range(1, n):
        length = n - k
        h_i = h[:length]
        h_ipk = h[k:]
        w = cp.Variable(length, boolean=True)
        constraints += [w <= h_i, w <= h_ipk, w >= h_i + h_ipk - 1]
        constraints += [t >= (target - cp.sum(w)) * dx]

    constraints += [t >= 0]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        # try CBC if available
        prob.solve(solver=cp.CBC, verbose=False, seconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False, glpk_options=["--tmlim", str(time_limit)])
        except Exception:
            # fallback slow solve
            prob.solve(solver=cp.ECOS_BB, verbose=False, max_iters=200000)

    if t.value is None:
        return None, None

    h_val = np.rint(h.value).astype(int)
    # ensure exactly target ones
    cur_sum = h_val.sum()
    if cur_sum != target:
        # repair by flipping some bits
        diff = target - cur_sum
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            if len(zeros) >= diff:
                flip = rng.choice(zeros, size=diff, replace=False)
                h_val[flip] = 1
        elif diff < 0:
            ones = np.where(h_val == 1)[0]
            if len(ones) >= -diff:
                flip = rng.choice(ones, size=-diff, replace=False)
                h_val[flip] = 0
    c5 = compute_c5(h_val.astype(float))
    return h_val.astype(float), c5
```

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Maximum overlap integral for a step function h (binary or continuous)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Project v∈[0,1]^n onto the affine hyperplane sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(target)
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v

def solve_lp_initial(n, target, dx, rng):
    """Linear program with McCormick relaxation to obtain a good continuous start."""
    h = cp.Variable(n)
    t = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == target, t >= 0]
    for k in range(1, n):
        length = n - k
        w = cp.Variable(length)
        cons += [
            w <= h[:length],
            w <= h[k:],
            w >= h[:length] + h[k:] - 1,
            w >= 0,
            t >= (target - cp.sum(w)) * dx,
        ]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iters=200000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
            except Exception:
                return None, None
    if t.value is None or h.value is None:
        return None, None
    h_val = np.clip(h.value, 0.0, 1.0)
    h_val = enforce_sum(h_val, target, rng)
    return h_val, compute_c5(h_val)

def continuous_anneal(h, cur_val, best_h, best_val, start, deadline, rng):
    """Simulated annealing on continuous values with direct worst‑shift moves."""
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k < 0:
            k = -k
        if k == 0:
            continue
        length = n - k
        # positions that worsen this shift and positions that could improve it
        A = np.where((h[:length] > 0.5) & (h[k:] < 0.5))[0]
        B = np.where((h[:length] < 0.5) & (h[k:] > 0.5))[0]
        if A.size > 0 and B.size > 0:
            i = int(rng.choice(A))
            j = int(rng.choice(B)) + k
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer > 1e-12:
                delta = rng.random() * max_transfer
                h[i] -= delta
                h[j] += delta
                new_val = compute_c5(h)
                if new_val < cur_val:
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h.copy()
                        if best_val <= 0.3808:
                            break
                    continue
                # revert if not improving
                h[i] += delta
                h[j] -= delta
        # random mass transfer (standard SA step)
        src = np.where(h > 1e-12)[0]
        dst = np.where(h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = int(rng.choice(src))
        j = int(rng.choice(dst))
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Choose dimension
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None

    target = n // 2
    dx = 2.0 / n

    # Build an initial feasible vector
    if h0 is not None:
        h_start = np.clip(h0, 0.0, 1.0)
        h_start = enforce_sum(h_start, target, rng)
    else:
        # try LP‑based start, fall back to random if needed
        h_start, lp_val = solve_lp_initial(n, target, dx, rng)
        if h_start is None:
            h_start = random_continuous_vector(n, target, rng)

    best_h = h_start.copy()
    best_val = compute_c5(best_h)

    # Simulated annealing phase (continuous)
    if time.time() < deadline - 0.5:
        best_h, best_val = continuous_anneal(
            best_h, best_val, best_h, best_val, start, deadline, rng
        )

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
