def binary_hill_climb(h, cur_val, deadline, rng):
    # h binary with exact sum; cur_val = compute_c5(h)
    n = len(h)
    while time.time() < deadline:
        # select i with h[i]==1, j with h[j]==0
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new_val = compute_c5(h)

                cur_val = new_val
            else:
                h[i], h[j] = 1, 0

    return h, cur_val

def continuous_anneal_local(h, cur_val, start_time, deadline, rng):
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        # correlation
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)
        # compute product contributions for this shift
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        # find index i with max contribution
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # choose destination j with room
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            # fallback choose random src/dst
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # acceptance probability
            T = max((deadline - time.time()) * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- helper functions -------------------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Clip v to [0,1] and adjust to make its sum exactly target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    """Random vector in [0,1]^n with exact sum == target."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    """Binary vector with exactly round(target) ones."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    i = 3
    while i * i <= num:
        if num % i == 0:
            return False
        i += 2
    return True

def legendre_binary_sequence(p):
    """Legendre symbol sequence for prime p (length p-1) mapped to {0,1}."""
    seq = np.empty(p - 1, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        ls = pow(i, exp, p)
        seq[i - 1] = 1.0 if ls == 1 else 0.0
    return seq

def continuous_anneal_local(h, cur_val, start_time, deadline, rng):
    """Simple simulated annealing with mass transfers, no early stop."""
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    while time.time() < deadline - 0.05:
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)

        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        i_local = np.argmax(prod)
        i = i_range[i_local]

        cand = np.where((h < 1.0 - 1e-12) & (np.arange(n) != i))[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            T = max((deadline - time.time()) * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def binary_hill_climb(h_bin, cur_val, start_time, deadline, rng):
    """Swap-based hill climbing on binary vectors."""
    best_h = h_bin.copy()
    best_val = cur_val
    cur_h = h_bin.copy()
    cur_val = cur_val
    n = len(h_bin)
    while time.time() < deadline - 0.05:
        ones = np.where(cur_h == 1.0)[0]
        zeros = np.where(cur_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_val = compute_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            T = max((deadline - time.time()) * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = base_h.shape[0]
        n = 100  # target size (even)
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, base_h)
    else:
        n = 100
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    dx = 2.0 / n
    h0 = enforce_sum(h0, target, rng)
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Build constraints for SLSQP
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # Main iterative search
    while time.time() < deadline - 0.5:
        # choose a start configuration
        r = rng.random()
        if r < 0.3 and is_prime(n + 1):
            start_h = legendre_binary_sequence(n + 1)
        elif r < 0.6:
            if abs(target - round(target)) < 1e-9:
                start_h = random_binary_vector(n, target, rng)
            else:
                start_h = random_continuous_vector(n, target, rng)
        else:
            start_h = best_h.copy()
            for _ in range(5):
                i, j = rng.integers(0, n, size=2)
                while i == j:
                    j = rng.integers(0, n)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 3000, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            break
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        # Continuous annealing phase
        h_ann, val_ann = continuous_anneal_local(
            h_opt.copy(), val, start, deadline, rng
        )
        if val_ann < best_val:
            best_val = val_ann
            best_h = h_ann.copy()

        # Binary hill climbing if target is integer
        if time.time() < deadline - 0.2 and abs(target - round(target)) < 1e-9:
            k_ones = int(round(target))
            idx_desc = np.argsort(-best_h)
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx_desc[:k_ones]] = 1.0
            bin_val = compute_c5(h_bin)
            h_bin_ref, bin_val_ref = binary_hill_climb(
                h_bin, bin_val, start, deadline, rng
            )
            if bin_val_ref < best_val:
                best_val = bin_val_ref
                best_h = h_bin_ref.copy()

    # Final enforcement and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
