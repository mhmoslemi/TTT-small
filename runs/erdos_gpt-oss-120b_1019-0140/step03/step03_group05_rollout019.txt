import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # same as previous
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n-k] += 1.0 - h[k:]
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n-kk]
        grad[:n-kk] -= h[kk:]
    return grad

def gradient_descent(h0, target_sum, max_iter, time_deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = h.shape[0]
    dx = 2.0 / n
    alpha = 0.2
    min_alpha = 1e-8
    momentum = np.zeros_like(h)
    beta = 0.9
    for it in range(max_iter):
        if time.time() > time_deadline:
            break
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # momentum update
        momentum = beta * momentum + (1-beta) * grad
        new_h = h - alpha * momentum
        # projection
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-12:
            best_h, best_c5 = new_h.copy(), new_c5
        # adjust alpha
        if new_c5 < best_c5 + 1e-12:
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            # occasional random perturb if stuck
            if alpha <= min_alpha:
                noise = np.random.default_rng().normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                momentum = np.zeros_like(h)
                continue
        h = new_h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # Dimensionality
    n = 80
    target_sum = n / 2.0
    # Initialize population
    pop_size = 30
    # Use initial_h_values if available, upsample if needed
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        if init_h.size != n:
            # upsample via linear interpolation
            x_old = np.linspace(0, 2, init_h.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            init_h = np.interp(x_new, x_old, init_h)
        init_h = project_to_box_sum(np.clip(init_h, 0.0, 1.0), target_sum)
        # Seed population around it
        pop = []
        pop.append(init_h)
        for _ in range(pop_size-1):
            h = init_h + rng.normal(scale=0.1, size=n)
            h = project_to_box_sum(np.clip(h, 0.0, 1.0), target_sum)
            pop.append(h)
        pop = np.array(pop)
    else:
        pop = rng.random((pop_size, n))
        for i in range(pop_size):
            pop[i] = project_to_box_sum(pop[i], target_sum)
    # Evaluate fitness
    fitness = np.array([compute_c5(ind) for ind in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    # DE parameters
    F = 0.8
    CR = 0.9
    # Time allocation: we will run DE until some fraction left for final gradient.
    # set target for final gradient: at least 50 seconds or 5% of budget.
    final_grad_time = max(50.0, budget_s * 0.07)
    # main loop
    while time.time() < start_time + budget_s - final_grad_time:
        for i in range(pop_size):
            # Choose three distinct indices a,b,c != i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a = pop[a_idx]
            b = pop[b_idx]
            c = pop[c_idx]
            mutant = a + F * (b - c)
            # Crossover
            cross = rng.random(n) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            # Clip and project sum
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)
            # Evaluate
            trial_f = compute_c5(trial)
            # Selection
            if trial_f < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_c5:
                    best_c5 = trial_f
                    best_h = trial.copy()
                    if best_c5 < 0.3808:
                        # early exit
                        break
        # early exit check also after inner loop
        if best_c5 < 0.3808:
            break
    # Final local refinement on best_h
    remaining = start_time + budget_s - time.time()
    if remaining > 5.0:
        # allocate most of remaining to gradient descent
        gd_iters = int(min(20000, remaining * 1000))  # rough iterations per sec
        best_h, best_c5 = gradient_descent(best_h, target_sum, gd_iters, start_time + remaining)
    # Final projection and compute
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h0, target_sum, max_iter, time_deadline):
    """Projected gradient descent on the max-overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = h.shape[0]
    alpha = 0.2
    min_alpha = 1e-6
    momentum = np.zeros_like(h)
    beta = 0.9
    for _ in range(max_iter):
        if time.time() > time_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        momentum = beta * momentum + (1.0 - beta) * grad
        new_h = h - alpha * momentum
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        cur_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        if new_c5 < cur_c5:
            alpha = min(alpha * 1.05, 1.0)
            h = new_h
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = np.random.default_rng().normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                momentum = np.zeros_like(h)
    return best_h, best_c5

    best_c5 = fitness[best_idx]

    # ------------------------------------------------------------------
    # Differential Evolution
    # ------------------------------------------------------------------
    F = 0.8
    CR = 0.9
    final_grad_time = max(50.0, budget_s * 0.07)  # reserve for final refinement
    while time.time() < start_time + budget_s - final_grad_time:
        for i in range(pop_size):
            # Select three distinct donors different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a, b, c = pop[a_idx], pop[b_idx], pop[c_idx]
            mutant = a + F * (b - c)

            # Binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)

            trial_f = compute_c5(trial)
            if trial_f < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_f
                if trial_f < best_c5:
                    best_c5 = trial_f
                    best_h = trial.copy()
                    if best_c5 < 0.3808:
                        break
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final local refinement
    # ------------------------------------------------------------------
    remaining = start_time + budget_s - time.time()
    if remaining > 5.0:
        deadline = start_time + budget_s
        # large iteration count; actual loop stops on deadline
        best_h, best_c5 = gradient_descent(best_h, target_sum,
                                           max_iter=200000,
                                           time_deadline=deadline)

    # ------------------------------------------------------------------
    # Final projection and evaluation
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
