def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Handle both positive and negative lags
    autocorr = np.zeros((2*n-1, 2), dtype=np.float64)
    good_lags = []
    for i in range(1, n):
        # For positive shifts
        if i < n:
            shift_val = dx * np.dot(h[:n-i], 1.0 - h[i:])
            autocorr[n-1+i] = [1.0, shift_val]
            good_lags.append((n-1+i, shift_val))
        # For negative shifts
        if i < n:
            shift_val_neg = dx * np.dot(h[i:], 1.0 - h[:n-i])
            autocorr[n-1-i] = [1.0, shift_val_neg]
            good_lags.append((n-1-i, shift_val_neg))
            
    # Exclude 0 lag (which is always 0)
    for pos in range(2*n-2):
        if pos==n-1:  # center shift (k=0) always excluded
            autocorr[pos] = [0.0, 0.0]

    max_shift = 0.0
