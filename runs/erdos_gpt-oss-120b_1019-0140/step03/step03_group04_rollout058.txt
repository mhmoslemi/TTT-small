def gradient_ascent_min_corr(h, time_end, max_iter=100000):
    n = h.size
    s_target = n/2.0
    best_h = h.copy()
    best_score = compute_c5_from_autocorr(autocorr_fft(h))
    step0 = 0.5
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        # compute autocorr and find min shifts
        ac = autocorr_fft(h)
        zero_idx = n-1
        # find minimal value, maybe tie
        min_val = np.min(np.concatenate((ac[:zero_idx], ac[zero_idx+1:])))
        # get indices where equal
        min_idxs = np.where(ac == min_val)[0]
        # exclude zero_idx
        min_idxs = min_idxs[min_idxs != zero_idx]
        # compute gradient as sum of rolls
        grad = np.zeros_like(h)
        for idx in min_idxs:
            s = idx - zero_idx
            grad += np.roll(h, s) + np.roll(h, -s)
        # average
        grad = grad / max(1, len(min_idxs))
        # step, maybe clamp step size
        eta = step0 / np.sqrt(it)
        h_new = h + eta * grad
        # project onto [0,1] and sum = s_target
        h_new = project_onto_box_sum(h_new, s_target)
        # compute new score
        new_score = compute_c5_from_autocorr(autocorr_fft(h_new))
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
        if best_score <= target:
            break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def autocorr_fft(h):
    """Linear autocorrelation of h with itself (length 2n-1)."""
    n = h.shape[0]
    L = next_pow2(2 * n - 1)
    H = np.fft.fft(h, n=L)
    Hr = np.fft.fft(h[::-1], n=L)
    conv = np.fft.ifft(H * Hr).real
    return conv[: 2 * n - 1]

def compute_c5_from_autocorr(ac):
    """C5 = 1 - (2/n) * min_{k!=0} autocorr_k."""
    n = (ac.size + 1) // 2
    zero = n - 1
    left_min = ac[:zero].min()
    right_min = ac[zero + 1 :].min()
    min_val = left_min if left_min < right_min else right_min
    return 1.0 - (2.0 / n) * min_val

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_real_balanced(n, rng):
    """Real vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(np.sqrt(num))
    for i in range(3, r + 1, 2):
        if num % i == 0:
            return False
    return True

def next_prime(x):
    p = int(x)
    if p <= 2:
        return 2
    if p % 2 == 0:
        p += 1
    while not is_prime(p):
        p += 2
    return p

def legendre_sequence(p):
    """Legendre (quadratic‑residue) binary sequence of length p‑1."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def hill_climb_binary(h, time_end):
    """Swap‑based local improvement for binary vectors."""
    n = h.size
    zero = n - 1
    best_h = h.copy()
    ac = autocorr_fft(h)
    best_score = compute_c5_from_autocorr(ac)
    rng = np.random.default_rng()
    while time.time() < time_end:
        left_min = ac[:zero].min()
        right_min = ac[zero + 1 :].min()
        if left_min < right_min:
            min_idx = np.argmin(ac[:zero])
        else:
            min_idx = zero + 1 + np.argmin(ac[zero + 1 :])
        s = min_idx - zero
        h_shift = np.roll(h, -s)
        good_ones = np.where((h == 1) & (h_shift == 0))[0]
        good_zeros = np.where((h == 0) & (h_shift == 1))[0]
        if good_ones.size == 0 or good_zeros.size == 0:
            break
        i = rng.choice(good_ones)
        j = rng.choice(good_zeros)
        h[i] = 0.0
        h[j] = 1.0
        ac = autocorr_fft(h)
        score = compute_c5_from_autocorr(ac)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            if best_score <= 0.3808:
                break
    return best_h, best_score

def gradient_ascent_min_corr(h, time_end, max_iter=200000):
    """Gradient ascent on the worst autocorrelation shift(s)."""
    n = h.size
    zero = n - 1
    target_sum = n / 2.0
    best_h = h.copy()
    ac = autocorr_fft(h)
    best_score = compute_c5_from_autocorr(ac)
    step0 = 0.5
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        ac = autocorr_fft(h)
        left_min = ac[:zero].min()
        right_min = ac[zero + 1 :].min()
        if left_min < right_min:
            min_val = left_min
            min_idxs = np.where(ac[:zero] == min_val)[0]
        else:
            min_val = right_min
            min_idxs = np.where(ac[zero + 1 :] == min_val)[0] + zero + 1
        grad = np.zeros_like(h)
        for idx in min_idxs:
            s = idx - zero
            grad += np.roll(h, s) + np.roll(h, -s)
        if min_idxs.size:
            grad /= min_idxs.size
        eta = step0 / np.sqrt(it)
        h_new = h + eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_from_autocorr(autocorr_fft(h_new))
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= 0.3808:
                break
        h = h_new
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)
    target = 0.3808
    n = 1024
    best_h = None
    best_score = np.inf
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n), n / 2.0)

    candidates.append(thue_morse_binary(n))

    # Legendre from nearest prime > n
    p = next_prime(n + 1)
    leg = legendre_sequence(p)
    if leg.size >= n:
        leg = leg[:n]
    else:
        leg = resample(leg, n)
    candidates.append(project_onto_box_sum(leg, n / 2.0))

    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    for cand in candidates:
        if time_left() < 0.2:
            break
        ac = autocorr_fft(cand)
        score = compute_c5_from_autocorr(ac)
        if score < best_score:
            best_score, best_h = score, cand.copy()
            if best_score <= target:
                break

    # Random real starts with short gradient ascent
    for _ in range(4):
        if time_left() < 2.0:
