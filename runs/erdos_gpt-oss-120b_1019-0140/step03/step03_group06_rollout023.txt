def final_anneal(h, cur_val, start_time, deadline, rng):
    n = len(h); dx = 2.0/n; total_time = deadline-start_time
    while time.time() < deadline - 1e-6:
        # propose swap
        src = np.where(h > 1e-8)[0]
        dst = np.where(h < 1 - 1e-8)[0]
        if len(src)==0 or len(dst)==0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0: continue
        delta = max_delta * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        # temperature schedule
        elapsed = time.time() - start_time

                # revert

                h[j] -= delta
    return h, cur_val

def run(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None

    target = n / 2.0

    # Setup population
    popsize = 40
    population = []
    fitness = []

    # Use provided initial vector if any
    if h0 is not None:
        h0 = enforce_sum(h0, target, rng)
        population.append(h0)
        fitness.append(compute_c5(h0))

    # Fill rest
    while len(population) < popsize:
        # Random continuous vector or binary with 70% chance binary
        if rng.random() < 0.5:
            h = random_continuous_vector(n, target, rng)
        else:
            h = random_binary_vector(n, target, rng).astype(float)
        population.append(h)
        fitness.append(compute_c5(h))

    # Keep best
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]

    # GA loop
    while time.time() < deadline - 0.5:
        # Create offspring
        new_population = []
        new_fitness = []
        # generate children equal to popsize
        for _ in range(popsize):
            # tournament selection: pick two random candidates and choose better as parent
            idx1, idx2 = rng.integers(0, len(population), size=2)
            p1 = population[idx1] if fitness[idx1] <= fitness[idx2] else population[idx2]
            idx3, idx4 = rng.integers(0, len(population), size=2)
            p2 = population[idx3] if fitness[idx3] <= fitness[idx4] else population[idx4]
            # Crossover
            child = crossover_uniform(p1, p2, rng)
            # Mutation
            child = mutate_continuous(child, target, rng)
            # Evaluate
            val = compute_c5(child)
            # Add child
            new_population.append(child)
            new_fitness.append(val)
        # Combine old and new
        combined_pop = population + new_population
        combined_fit = fitness + new_fitness
        # Sort by fitness
        sorted_idx = np.argsort(combined_fit)
        population = [combined_pop[i] for i in sorted_idx[:popsize]]
        fitness = [combined_fit[i] for i in sorted_idx[:popsize]]
        # Update best
        if fitness[0] < best_val:
            best_val = fitness[0]
            best_h = population[0].copy()
            if best_val < 0.3808:
                break

import time
import numpy as np

# ------------------- helper functions ------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to make sum(v) == target (within 1e-12)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0.0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0.0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return np.clip(v, 0.0, 1.0)


def random_continuous_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1]^n with exact sum == target."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)


def random_binary_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly target ones (target assumed integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v


def crossover_uniform(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One‑point uniform crossover."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2)
    return child


def mutate_continuous(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount from one index to another while preserving the sum."""
    h_new = h.copy()
    src = np.where(h_new > 1e-8)[0]
    dst = np.where(h_new < 1.0 - 1e-8)[0]
    if len(src) == 0 or len(dst) == 0:
        return h_new
    i = rng.choice(src)
    j = rng.choice(dst)
    max_delta = min(h_new[i], 1.0 - h_new[j])
    if max_delta <= 0.0:
        return h_new
    delta = max_delta * rng.random()
    h_new[i] -= delta
    h_new[j] += delta
    return enforce_sum(h_new, target, rng)


def final_anneal(
    h: np.ndarray,
    cur_val: float,
    start_time: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simple simulated annealing that preserves the sum exactly."""
    n = h.shape[0]
    total_time = deadline - start_time
    init_T = 0.001
    while time.time() < deadline - 1e-6:
        src = np.where(h > 1e-8)[0]
        dst = np.where(h < 1.0 - 1e-8)[0]
        if len(src) == 0 or len(dst) == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        remaining = deadline - time.time()
        T = max(init_T * (remaining / total_time), 1e-12)
        if new_val < cur_val:
            cur_val = new_val
        else:
            if rng.random() < np.exp((cur_val - new_val) / T):
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return h, cur_val

        for _ in range(popsize):

        combined_fit = fitness + new_fit

        if fitness[0] < best_val:
