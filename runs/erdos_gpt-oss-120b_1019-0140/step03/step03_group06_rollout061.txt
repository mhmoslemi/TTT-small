def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    f = np.fft.rfft(h, n=2*n-1)
    g = np.fft.rfft(np.flip(1-h), n=2*n-1)
    corr = np.fft.irfft(f * g, n=2*n-1)
    # corr length = 2n-1; actual correlation entries corresponding to shifts -n+1..n-1; max at full range
    # Determine max
    max_corr = np.max(corr)
    return max_corr * dx

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    pad = 2*n - 1
    f = np.fft.rfft(h, pad)
    g = np.fft.rfft(np.flip(1.0 - h), pad)
    corr = np.fft.irfft(f * g, pad)
    max_corr = np.max(corr)
    return max_corr * dx

    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = h0.copy()
    cur_val = best_val
    attempts_without_improve = 0
    max_attempts_no_improve = 5000  # perhaps

    while time.time() < deadline - 0.01:
        # compute correlation and worst shift
        pad = 2*n - 1
        f = np.fft.rfft(cur_h, pad)
        g = np.fft.rfft(np.flip(1.0 - cur_h), pad)
        corr = np.fft.irfft(f * g, pad)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        k = max_idx - (n - 1)
        # Compute product vector for shift k
        if k >= 0:
            i_range = np.arange(n - k)
            prod = cur_h[i_range] * (1.0 - cur_h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = cur_h[i_range] * (1.0 - cur_h[i_range - kk])
        # find i_max
        i_local = np.argmax(prod)
        i = int(i_range[i_local])
        # destination j = i + k if valid
        j = i + k
        if not (0 <= j < n) or cur_h[j] >= 1 - 1e-12:
            # random destination
            candidate_dst = np.where(cur_h < 1 - 1e-12)[0]
            if len(candidate_dst) == 0:
                break
            j = int(rng.choice(candidate_dst))
        if cur_h[i] <= 1e-12:
            src_candidates = np.where(cur_h > 1e-12)[0]
            if len(src_candidates) == 0:
                break
            i = int(rng.choice(src_candidates))
        # ensure i!=j
        if i == j:
            # find different dest

            cand = cand[cand != i]
            if len(cand) == 0:
                break

        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:

            dst = np.where(cur_h < 1 - 1e-12)[0]
            if len(src)==0 or len(dst)==0:

            j = int(rng.choice(dst))

        else:

            remaining = deadline - time.time()
            total = deadline - start_time
            # proportion left

            T = 0.01 * prop + 1e-12  # adapt

import time
import numpy as np

# ---------- helper functions ----------
def enforce_sum(v, target, rng):
    """Scale and adjust vector v (in [0,1]) so that sum(v) == target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation (FFT version)."""
    n = h.shape[0]
    dx = 2.0 / n
    pad = 2 * n - 1
    f = np.fft.rfft(h, pad)
    g = np.fft.rfft(np.flip(1.0 - h), pad)
    corr = np.fft.irfft(f * g, pad)
    return np.max(corr) * dx

# ---------- binary local search ----------
def binary_local_search(n, target_ones, deadline, rng):
    best_b = None
    best_val = np.inf
    while time.time() < deadline - 1.0:
        b = np.zeros(n, dtype=float)
        ones = rng.choice(n, target_ones, replace=False)
        b[ones] = 1.0
        cur_val = compute_c5(b)
        max_swaps = 2000
        for _ in range(max_swaps):
            if time.time() >= deadline - 1.0:
                break
            idx_one = np.where(b == 1)[0]
            idx_zero = np.where(b == 0)[0]
            if len(idx_one) == 0 or len(idx_zero) == 0:
                break
            i = int(rng.choice(idx_one))
            j = int(rng.choice(idx_zero))
            b[i] = 0.0
            b[j] = 1.0
            new_val = compute_c5(b)
            if new_val < cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
        if best_val < 0.3808:
            break
    return best_b, best_val

# ---------- continuous local search ----------
def continuous_local_search(h0, deadline, rng, start_time):
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = h0.copy()
    cur_val = best_val
    attempts_no_improve = 0
    max_no_improve = 2000
    total_budget = deadline - start_time
    zero_idx = n - 1

    while time.time() < deadline - 0.01:
        pad = 2 * n - 1
        f = np.fft.rfft(cur_h, pad)
        g = np.fft.rfft(np.flip(1.0 - cur_h), pad)
        corr = np.fft.irfft(f * g, pad)
        # ignore zero shift
        corr[zero_idx] = -np.inf
        max_idx = int(np.argmax(corr))
        k = max_idx - zero_idx

        if k >= 0:
            i_range = np.arange(n - k)
            prod = cur_h[i_range] * (1.0 - cur_h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = cur_h[i_range] * (1.0 - cur_h[i_range - kk])

        i_local = int(np.argmax(prod))
        i = int(i_range[i_local])
        j = i + k

        # ensure valid destination with headroom
        if not (0 <= j < n) or cur_h[j] >= 1.0 - 1e-12:
            cand_dst = np.where(cur_h < 1.0 - 1e-12)[0]
            if len(cand_dst) == 0:
                break
            j = int(rng.choice(cand_dst))
        if cur_h[i] <= 1e-12:
            cand_src = np.where(cur_h > 1e-12)[0]
            if len(cand_src) == 0:

            i = int(rng.choice(cand_src))
        if i == j:
            cand_dst = np.where(cur_h < 1.0 - 1e-12)[0]
            cand_dst = cand_dst[cand_dst != i]
            if len(cand_dst) == 0:

        max_transfer = min(cur_h[i], 1.0 - cur_h[j])

            dst = np.where(cur_h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = int(rng.choice(src))
            j = int(rng.choice(dst))

        delta = max_transfer * rng.random() * delta_factor

        cur_h[j] += delta

        if new_val < cur_val:
            cur_val = new_val
            attempts_no_improve = 0

                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            T = max(0.001 * prop, 1e-12)
            accept_prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < accept_prob:
                cur_val = new_val
                attempts_no_improve = 0
            else:
                cur_h[i] += delta
                cur_h[j] -= delta
                attempts_no_improve += 1

                i2 = int(rng.choice(src))

                cur_h[i2] -= delta2

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ---- optional binary phase (20% time) ----
    binary_deadline = start + min(0.2 * budget_s, 200.0)
    if time.time() < binary_deadline - 5.0:
        best_bin, best_bin_val = binary_local_search(
            n, int(target), binary_deadline, rng
        )
        if best_bin is not None and best_bin_val < best_val:
            best_h = best_bin.astype(float).copy()
            best_val = best_bin_val

    # ---- continuous refinement phase ----
    best_h, best_val = continuous_local_search(best_h, deadline, rng, start)

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
