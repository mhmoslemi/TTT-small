def gradient_descent_with_restarts(h0, target_sum, start_time, budget_s, rng, n):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    max_iter = 2000
    # runtime limit = start + budget - 5 - other loops
    time_limit = start_time + budget_s - 5
    # Use gradient descent loop with random restarts.
    h_current = best_h.copy()
    alpha = 0.2
    beta = 0.9
    v = np.zeros_like(h_current)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # Compute worst shift
        corr = np.correlate(h_current, 1.0 - h_current, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        cur_c5 = corr[max_idx] * (2.0 / n)
        # Update best
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h_current.copy()
            # optionally reduce step size?
        # gradient
        g = gradient_for_shift(h_current, k)
        # momentum update
        v = beta * v + (1 - beta) * g
        # step
        new_h = h_current - alpha * v
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h_current = new_h
            alpha = min(alpha * 1.03, 1.0)  # slight increase
        else:
            # no improvement: reduce step size, random restart maybe
            alpha = max(alpha * 0.5, 1e-5)
            # if small step size, random perturb
            if alpha <= 1e-5:
                # random perturbation around best
                h_current = best_h + rng.normal(scale=0.02, size=n)
                h_current = project_to_box_sum(np.clip(h_current, 0.0, 1.0), target_sum)
                v = np.zeros_like(h_current)
                alpha = 0.2
    # At end, optionally do short SLSQP refinement from best_h
    remaining = time_limit - time.time()
    if remaining > 5.0:
        t0 = compute_c5(best_h)
        # prepare constraints as before
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k !=0]

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without the dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad


# ----------------------------------------------------------------------
# Constraint classes for SLSQP
# ----------------------------------------------------------------------
class SumConstraint:
    """Equality: sum(h) == n/2."""

    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""

    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overlap * self.dx


def objective_t(x):
    """Objective for SLSQP: minimise the slack variable t."""
    return x[-1]


def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=200):
    """Run a single SLSQP optimisation from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )
    return res

# Local optimisation primitives
# ----------------------------------------------------------------------
def gradient_descent_momentum(h_start, target_sum, max_iter, deadline, rng):
    """Projected gradient descent with momentum on the worst shift."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = 0.2
    beta = 0.9
    v = np.zeros_like(h)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        v = beta * v + (1 - beta) * grad
        new_h = h - alpha * v
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.02, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-6)
            if alpha <= 1e-6:
                h = best_h + rng.normal(scale=0.02, size=n)
                h = np.clip(h, 0.0, 1.0)
                h = project_to_box_sum(h, target_sum)
                v = np.zeros_like(h)
                alpha = 0.2
    return best_h, best_c5


def local_swap_search(h, max_moves, deadline, rng):
    """Greedy pairwise swap preserving the sum."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(max_moves):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:

        max_inc = 1.0 - best_h[i]
        max_dec = best_h[j]

        cand[j] -= delta
        c5cand = compute_c5(cand)
        if c5cand < best_c5 - 1e-12:
            best_h = cand
            best_c5 = c5cand
    return best_h, best_c5

            break

        # Initialise a feasible start vector
        if n == 80 and "initial_h_values" in globals():
            h0 = np.asarray(initial_h_values, dtype=float)
            if h0.size != n:
                h0 = rng.random(n)
        else:
            h0 = rng.random(n)

        h_sol, c5_sol = solve_for_n(n, h0, rng, start_time, budget_s)

        if c5_sol < best_c5:
            best_c5 = c5_sol
            best_h = h_sol.copy()
            best_n = n
            if best_c5 < 0.3808:
                break

    # Fallback in the unlikely event that nothing was found
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_h = project_to_box_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
