def gradient_descent_multi(h_start, target_sum, max_iter, time_limit, top_k=3):
    n = h_start.shape[0]
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.1
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find top_k indices
        if top_k >= len(corr):
            idxs = np.arange(len(corr))
        else:
            idxs = np.argpartition(corr, -top_k)[-top_k:]
        grads = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grads += gradient_for_shift(h, k)
        grads /= len(idxs)
        # step
        new_h = h - alpha * grads
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = new_h.copy()
        if new_c5 < compute_c5(h):
            h = new_h
            alpha = min(alpha * 1.05, 2.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                # random jitter
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0, 1), target_sum)
                alpha = 0.1
    return best_h, best_c5
```

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=objective_t, x0=x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res
```

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent_multi(h_start, target_sum, max_iter, time_limit, rng, top_k=3):
    n = h_start.shape[0]
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.1
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_k < corr.size:
            idxs = np.argpartition(corr, -top_k)[-top_k:]
        else:
            idxs = np.arange(corr.size)
        grads = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grads += gradient_for_shift(h, k)
        grads /= len(idxs)
        new_h = h - alpha * grads
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        cur_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = new_h.copy()
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 2.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def hill_climb_swaps(h_start, target_sum, max_iters, time_limit, rng):
    n = h_start.shape[0]
    h = h_start.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    p_worse = 0.01
    for _ in range(max_iters):
        if time.time() > time_limit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        if delta <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            cur_c5 = c5_new
            h = h_new
        elif c5_new < cur_c5:
            cur_c5 = c5_new
            h = h_new
        else:
            if rng.random() < p_worse:
                cur_c5 = c5_new
                h = h_new
        p_worse *= 0.9999
    return best_h, best_c5

class SumConstraint:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(fun=objective_t, x0=x0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    try:
        res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=200)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    except Exception:
        pass

    safety = 5.0
    while time.time() < start + budget_s - safety:
        if rng.random() < 0.4:
            h_start = best_h + rng.normal(scale=0.02, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        gd_deadline = start + budget_s - safety
        gd_h, gd_c5 = gradient_descent_multi(h_start, target_sum,
                                             max_iter=800,
                                             time_limit=gd_deadline,
                                             rng=rng,
                                             top_k=3)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h.copy(), gd_c5
            if best_c5 < 0.3808:
                break

        hc_deadline = start + budget_s - safety
        hc_h, hc_c5 = hill_climb_swaps(gd_h, target_sum,
                                       max_iters=3000,
                                       time_limit=hc_deadline,
                                       rng=rng)
        if hc_c5 < best_c5:
            best_h, best_c5 = hc_h.copy(), hc_c5
            if best_c5 < 0.3808:
                break

        remain = start + budget_s - safety - time.time()
        if remain > 15.0:
            try:
                res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=200)
                if res.success:
                    h_opt = np.clip(res.x[:n], 0.0, 1.0)
                    h_opt = project_to_box_sum(h_opt, target_sum)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_h, best_c5 = h_opt, c5_opt
                        if best_c5 < 0.3808:
                            break
            except Exception:
                pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
