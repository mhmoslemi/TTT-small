def greedy_swap_search(h, max_iter, rng):

    n = len(h)
    M = n // 2
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for iter in range(max_iter):
        # get indices of ones and zeros
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        # if no swaps possible, break
        improved = False
        best_swap = None
        best_swap_c5 = best_c5
        for i in ones:
            for j in zeros:
                # early exit if i == j? not needed.
                # swap
                # Instead of full copy each time, just compute c5 on swapped array.
                # Use copy
                h_candidate = best_h.copy()
                h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
                c5_cand = compute_c5(h_candidate)
                if c5_cand < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5_cand
                    best_swap = (i,j)
        if best_swap is None:
            break
        # perform swap
        i,j = best_swap
        best_h[i], best_h[j] = best_h[j], best_h[i]
        best_c5 = best_swap_c5
        improved = True
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

            def fun(x):

                if k >= 0:

            return fun

                       constraints=constraints,

        return h0, compute_c5(h0)

    if not res.success:
        return h0, compute_c5(h0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def is_prime(x: int) -> bool:
    if x < 2:
        return False
    if x % 2 == 0:
        return x == 2
    i = 3
    while i * i <= x:
        if x % i == 0:
            return False
        i += 2
    return True


def legendre_sequence(p: int) -> np.ndarray:
    """Binary Legendre sequence of length p (p prime). seq[0] = 0."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for quadratic residue, p-1 otherwise
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

    return h

        # final feasibility guard
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        target = n / 2.0
        if abs(cur - target) > 1e-9:
            best_h += (target - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
