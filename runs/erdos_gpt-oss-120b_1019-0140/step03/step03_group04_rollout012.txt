def adam_optimize(h0, max_iters, step0, beta1=0.9, beta2=0.999, eps=1e-8, time_end=None):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        if time_end is not None and time.time() > time_end:
            break
        # compute correlation and shifts
        n = h.size
        # compute correlation via FFT
        L = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(h, L)
        fb = np.fft.rfft(1.0 - h, L)
        corr = np.fft.irfft(fa * np.conj(fb), L)[:2*n-1]
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n-1)
        # gradient
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def adam_optimize(h0, max_iters, step0, time_limit):
    start_local = time.time()
    h = h0.copy()
    n = h.size
    target_sum = n/2.
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        if time.time() - start_local > time_limit:
            break
        # compute correlation via FFT
        n = h.size
        L = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(h, L)
        fb = np.fft.rfft(1.0 - h, L)

        max_corr = corr.max()

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral C5 on the linear (non‑circular) grid."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(fa * np.conj(fb), L)[: 2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑overlap term for given shift indices (full correlation)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def resample(arr, new_n):
    """Linear interpolation from size len(arr) to new_n (no endpoint duplication)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence of length n (0/1 values)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue (Legendre) binary sequence of length p‑1 for prime p."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones (balanced)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros_like(arr)
    binar[idx] = 1.0
    return binar

def adam_optimize(h0, max_iters, step0, time_limit):
    """Adam‑style projected optimization of C5."""
    start_local = time.time()
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters + 1):
        if time.time() - start_local > time_limit:
            break
        L = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(h, L)
        fb = np.fft.rfft(1.0 - h, L)
        corr = np.fft.irfft(fa * np.conj(fb), L)[: 2 * n - 1]
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def simulated_annealing_h(h0, max_iters, time_limit, rng, target=0.3808):
    """Binary swap‑based simulated annealing to lower C5."""
    start_local = time.time()
    cur = binarize_from_real(h0)
    cur_score = compute_c5(cur)
    best_h = cur.copy()
    best_score = cur_score
    n = cur.size
    half = n // 2
    for it in range(1, max_iters + 1):
        if time.time() - start_local > time_limit:
            break
        tau = it / max_iters
        T = 0.1 * (1 - tau) + 0.001 * tau
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = cur.copy()
        new[i] = 0
        new[j] = 1
        new_score = compute_c5(new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur, cur_score = new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, cur.copy()
        if best_score <= target:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # resolution choices
    n_small, n_large = 128, 512
    best_h = None
    best_score = np.inf

    # ------------------- small‑scale candidates -------------------
    small_cands = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small / 2.0)
        small_cands.append(cand)
    small_cands.append(thue_morse_binary(n_small))
    leg_small = legendre_sequence(131)                 # length 130
    small_cands.append(project_onto_box_sum(resample(leg_small, n_small), n_small / 2.0))
    for _ in range(3):
        small_cands.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        rand = rng.standard_normal(n_small)
        sigma = rng.uniform(0.5, 5.0)
        k = int(3 * sigma)
        x = np.arange(-k, k + 1)
        kernel = np.exp(-0.5 * (x / sigma) ** 2)
        kernel /= kernel.sum()
        smooth = np.convolve(rand, kernel, mode='same')
        small_cands.append(project_onto_box_sum(smooth, n_small / 2.0))

    # Optimize each small candidate (budget‑aware)
    for cand in small_cands:
        if time_left() < 2.0:
            break
        remaining = time_left()
        iters = int(min(8000, max(2000, remaining * 600)))
        tl = max(0.0, remaining - 10.0)  # keep a safety margin
        h_opt, sc_opt = adam_optimize(cand, iters, step0=0.5, time_limit=tl)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # Fallback if nothing succeeded
    if best_h is None:
        best_h = random_binary_balanced(n_small, rng)
        best_score = compute_c5(best_h)

    # ------------------- upsample to large resolution -------------------
    h_up = resample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    sc_up = compute_c5(h_up)
    if sc_up < best_score:
        best_score, best_h = sc_up, h_up.copy()

    # ------------------- large‑scale candidates -------------------
    large_cands = [best_h.copy(),
                   thue_morse_binary(n_large),
                   ]
    leg_large = legendre_sequence(521)                 # length 520
    leg_large = project_onto_box_sum(resample(leg_large, n_large), n_large / 2.0)
    large_cands.append(leg_large)
    for _ in range(3):
        large_cands.append(random_binary_balanced(n_large, rng))

    # Optimize large candidates
    for cand in large_cands:
        if time_left() < 5.0:
            break
        remaining = time_left()
        iters = int(min(50000, max(5000, remaining * 800)))
        tl = max(0.0, remaining - 8.0)
        h_opt, sc_opt = adam_optimize(cand, iters, step0=0.5, time_limit=tl)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # ------------------- simulated annealing refinement (binary) -------------------
    while time_left() > 6.0 and best_score > 0.3808:
        remaining = time_left()
        iters = int(min(30000, max(5000, remaining * 500)))
        tl = max(0.0, remaining - 2.0)
        h_ref, sc_ref = simulated_annealing_h(best_h, iters, tl, rng, target=0.3808)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= 0.3808:
            break

    # final safety projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
