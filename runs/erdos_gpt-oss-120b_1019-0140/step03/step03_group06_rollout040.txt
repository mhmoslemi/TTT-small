def local_descent(h, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.01:
        k, corr = worst_shift_and_corr(h)
        # produce i front
        if k >= 0:
            i_idxs = np.arange(n - k)
            j_idxs = i_idxs + k
        else:
            kk = -k
            i_idxs = np.arange(kk, n)
            j_idxs = i_idxs - kk
        # compute scores
        hi = h[i_idxs]
        hj = h[j_idxs]
        scores = hi + (1.0 - hj)
        idx = np.argmax(scores)
        i = i_idxs[idx]
        j = j_idxs[idx]
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            # cannot transfer, maybe random perturb
            # choose random src, dst
            src_candidates = np.where(h > 1e-12)[0]
            dst_candidates = np.where(h < 1.0 - 1e-12)[0]
            if src_candidates.size == 0 or dst_candidates.size == 0:
                break
            i = rng.choice(src_candidates)
            j = rng.choice(dst_candidates)
            max_transfer = min(h[i], 1.0 - h[j])
        delta = max_transfer * rng.random()
        # adjust
        h[i] -= delta
        h[j] += delta
        # compute new max overlap
        new_val = compute_c5(h)
        # hill climb
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            # possibly revert with probability, else revert
            # We'll revert to maintain search progress; chance to accept
            if rng.random() < 0.001:
                cur_val = new_val
            else:
                # revert move
                h[i] += delta
                h[j] -= delta
        # continue
    return best_h, best_val
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Maximum overlap integral via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust to exact target sum."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

# ---------- constraint helpers for SLSQP ----------
def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overlap = np.dot(h[: n - k], 1.0 - h[k:]) * dx
    else:
        kk = -k
        overlap = np.dot(h[kk:], 1.0 - h[: n - kk]) * dx
    return t - overlap

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq,
                     "args": (k, n, dx)})
    return cons

# ---------- local search ----------
def worst_shift_and_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    return k, corr

def local_descent(h, rng, start_time, deadline, max_steps=2000):
    n = h.shape[0]
    dx = 2.0 / n
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    steps = 0
    total_time = max(deadline - start_time, 1e-12)
    while time.time() < deadline - 0.01 and steps < max_steps:
        k, _ = worst_shift_and_corr(h)
        if k >= 0:
            i_idxs = np.arange(n - k)
            j_idxs = i_idxs + k
        else:
            kk = -k
            i_idxs = np.arange(kk, n)
            j_idxs = i_idxs - kk
        hi = h[i_idxs]
        hj = h[j_idxs]
        scores = hi + (1.0 - hj)
        idx = np.argmax(scores)
        i = i_idxs[idx]
        j = j_idxs[idx]
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_transfer = min(h[i], 1.0 - h[j])
        delta = max_transfer * rng.random()
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            T = 0.01 * (remaining / total_time)
            if T > 0 and np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
        steps += 1
    return best_h, best_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # choose number of points (even)
    n_target = 200
    n = n_target if n_target % 2 == 0 else n_target + 1
    target = n / 2.0

    # initial solution
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.shape[0]
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x, x0, base)
        h0 = np.clip(h0, 0.0, 1.0)
        best_h = enforce_sum(h0, target, rng)
    else:
        best_h = random_binary_vector(n, target, rng)

    best_val = compute_c5(best_h)

    # main randomized restart loop
    while time.time() < deadline - 1.0:
        r = rng.random()
        if r < 0.5:
            start_h = random_binary_vector(n, target, rng)
        elif r < 0.85:
            start_h = random_continuous_vector(n, target, rng)
        else:
            start_h = best_h.copy()
            for _ in range(3):
                src = np.where(start_h > 1e-12)[0]
                dst = np.where(start_h < 1.0 - 1e-12)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random()
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)

        cur_h, cur_val = local_descent(start_h, rng, start, deadline, max_steps=2000)
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
            if best_val < 0.3808:
                break

    # optional SLSQP refinement if time remains
    if time.time() < deadline - 0.5:
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        constraints = build_constraints(n, target, dx)
        x0 = np.concatenate([best_h, [best_val]])
        try:
            res = minimize(fun=objective_fun, x0=x0, method="SLSQP",
                           bounds=bounds, constraints=constraints,
                           options={"maxiter": 500, "ftol": 1e-11, "disp": False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = enforce_sum(h_opt, target, rng)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best_val, best_h = val_opt, h_opt
        except Exception:
            pass

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), float(final_val), n
