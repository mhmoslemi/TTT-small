def adam_optimize(h0, max_iter, time_left, step0=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, max_iter+1):
        if time_left() <= 0:
            break
        grad = compute_subgradient(h)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - step * m_hat
        # Ensure within box-sum:
        h_new = project_onto_box_sum(h_new, h.size/2)
        # Clip to [0,1] is already done in projection.
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_c5_fft(h):
    n = h.shape[0]
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    L = 2*n - 1
    # use rfft for efficiency
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b_rev, n=L)

    a = h
    b = 1.0 - h
    b_rev = b[::-1]

    max_val = np.max(conv)
    # get indices where within tolerance
    # tolerance = 1e-12 (or maybe relative tolerance)
    idxes = np.where(conv >= max_val - 1e-12)[0]
    # compute shifts: shift = idx - (n-1)
    shifts = idxes - (n - 1)

    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # term1: (1 - h_{i+s}) term from first argument
            # term1 array: for i in 0..n-s-1, term1[i] = h[i+s]; zero beyond
            term1 = np.zeros(n)

            term2 = np.zeros(n)

            term2 = np.zeros(n)
            term2[:n-ss] = h[ss:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad, max_val

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def compute_subgradient(h):
    """Return (gradient, max_corr) for the active shift(s) of h."""
    n = h.shape[0]
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    L = 2 * n - 1
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(A * B, n=L)
    max_val = np.max(conv)
    idxes = np.where(conv >= max_val - 1e-12)[0]
    shifts = idxes - (n - 1)

    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            term1[:n - s] = h[s:]
            term2[s:] = h[:n - s]
        else:
            ss = -s
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            term1[ss:] = h[:n - ss]
            term2[:n - ss] = h[ss:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad, max_val


def compute_c5(h):
    """C5 value for a given h (max correlation)."""
    _, max_val = compute_subgradient(h)
    return max_val


def slsqp_optimize_candidate(h0, max_iter=200):
    """Refine h0 using SLSQP (minimize max correlation)."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        # full correlation via FFT (fast)
        a = h
        b = 1.0 - h
        b_rev = b[::-1]
        L = 2 * n - 1
        A = np.fft.rfft(a, n=L)
        B = np.fft.rfft(b_rev, n=L)
        corr = np.fft.irfft(A * B, n=L) * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter,
                                'ftol': 1e-12,
                                'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def adam_optimize(h0, max_iter, time_left, step0):
    """Adam-style projected subgradient descent."""
    n = h0.shape[0]
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    grad, corr = compute_subgradient(h)
    best_h = h.copy()
    best_score = corr

    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - step * m_hat
        h_new = project_onto_box_sum(h_new, n / 2.0)

        grad, corr = compute_subgradient(h_new)
        if corr < best_score:
            best_score = corr
            best_h = h_new.copy()

        h = h_new

    return best_h, best_score


def optimize_at_scale(h, n, time_left):
    """Upsample to n and perform iterative Adam+SLSQP refinement."""
    if h.shape[0] != n:
        h = upsample(h, n)
    h = project_onto_box_sum(h, n / 2.0)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # step size (constant works well across dimensions)
    step0 = 0.2

    total_iters = 0
    max_total_iters = 50000
    chunk = 3000

    while time_left() > 5 and total_iters < max_total_iters:
        remaining = max_total_iters - total_iters
        iters = min(chunk, remaining)
        h_opt, sc_opt = adam_optimize(best_h, iters, time_left, step0)
        total_iters += iters
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

        # occasional short SLSQP refinement
        if time_left() > 8:
            h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=150)
            if sc_slsqp < best_score:
                best_score, best_h = sc_slsqp, h_slsqp.copy()
        best_h = project_onto_box_sum(best_h, n / 2.0)

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base vector: use provided initial_h_values if present, else Thue–Morse
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float, copy=False)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    for n in scales:
        if time_left() <= 5:
            break
        base, score = optimize_at_scale(base, n, time_left)
        if score < best_score:
            best_score, best_h = score, base.copy()
        # prepare for next scale
        base = upsample(base, n)  # ensure continuity when upsampling later

    # final high‑resolution polish (if time permits)
    if best_h is not None and time_left() > 5:
        n_final = best_h.size
        best_h, best_score = adam_optimize(best_h, 20000, time_left, step0=0.15)
        if time_left() > 5:
            h_slsqp, sc_slsqp = slsqp_optimize_candidate(best_h, max_iter=250)
            if sc_slsqp < best_score:
                best_h, best_score = h_slsqp, sc_slsqp

    # random binary neighbourhood search as last push
    if best_h is not None:
        n_final = best_h.size
        while time_left() > 5:
            h_rand = random_binary_balanced(n_final, rng)
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score, best_h = sc_rand, h_rand.copy()

            i, j = rng.choice(n_final, size=2, replace=False)
            delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.4)
            if delta > 0:
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new = compute_c5(h_new)
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()

            if best_score <= 0.3808:
                break

    # final feasibility projection
    if best_h is None:
        # fallback to a simple balanced Thue–Morse at max scale
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
