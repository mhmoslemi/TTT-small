import time
import numpy as np
from scipy.optimize import minimize

# ----- constraints -----
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ----- utilities -----
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_t(x):
    return x[-1]

def legendre_sequence(p):
    """0/1 Legendre sequence of length p (prime)."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def random_fractional_h(n, rng):
    M = n // 2
    h = rng.random(n)
    h = h / np.sum(h) * M
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

    """Clip to [0,1] and enforce sum = n/2."""
    M = n // 2
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

                break
        if isp:
            primes.append(p)

    for p in primes:
        if time.time() - start > budget_s - 5:
            break
        n = p + 1
        M = n // 2
        leg = legendre_sequence(p)
        h_leg = np.concatenate([leg, [1.0]])  # length n, sum = M
        shift = rng.integers(n)
        h_leg = np.roll(h_leg, shift)
        c5 = compute_c5(h_leg)
        if c5 + 1e-12 < best_c5:
            best_h, best_c5, best_n = h_leg.copy(), c5, n
            if best_c5 < 0.3808:
                break

    # ----- random fractional seeds (limited) -----
    max_random = 800
    for _ in range(max_random):
        if time.time() - start > budget_s * 0.3:
            break
        n = rng.choice(np.arange(80, max_n + 1, 2))
        h = random_fractional_h(n, rng)
        c5 = compute_c5(h)
        if c5 + 1e-12 < best_c5:
            best_h, best_c5, best_n = h, c5, n
            if best_c5 < 0.3808:
                break

    # ----- collect candidates for SLSQP refinement -----
    candidates = []
    if best_h is not None:
        candidates.append((best_h.copy(), best_n, best_c5))
    # add a Legendre candidate for n=128 if not already best
    if best_n != 128 and 128 <= max_n:
        p = 127
        leg = legendre_sequence(p)
        h_leg = np.concatenate([leg, [1.0]])
        best_local = None
        best_local_c5 = np.inf
        for _ in range(5):
            h_tmp = np.roll(h_leg, rng.integers(128))
            c5_tmp = compute_c5(h_tmp)
            if c5_tmp < best_local_c5:
                best_local_c5 = c5_tmp
                best_local = h_tmp.copy()
        if best_local is not None:
            candidates.append((best_local, 128, best_local_c5))

    # limit to a few distinct candidates
    seen_n = set()
    uniq_candidates = []
    for h, n, c5 in candidates:
        if n not in seen_n:
            uniq_candidates.append((h, n, c5))
            seen_n.add(n)
    candidates = uniq_candidates[:5]

    # ----- SLSQP refinement -----
    for h0, n, _ in candidates:
        if time.time() - start > budget_s - 5:
            break
        M = n // 2
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        constraints = [{'type': 'eq', 'fun': sum_con}]
        constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        h_curr = h0.copy()
        # two refinement attempts per candidate
        for _ in range(2):
            if time.time() - start > budget_s - 5:
                break
            t0 = compute_c5(h_curr)
            x0 = np.concatenate([h_curr, [t0]])
            # small random perturbation
            x0[:n] = np.clip(x0[:n] + rng.normal(scale=0.01, size=n), 0.0, 1.0)
            cur_sum = np.sum(x0[:n])
            x0[:n] += (M - cur_sum) / n
            try:
                res = minimize(fun=objective_t,
                               x0=x0,
                               method='SLSQP',
                               bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
            except Exception:
                continue
            if not res.success:
                continue
            h_opt = project_h(res.x[:n], n)
            c5_opt = compute_c5(h_opt)
            if c5_opt + 1e-12 < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                if best_c5 < 0.3808:
                    break
            h_curr = h_opt  # use for next attempt
        if best_c5 < 0.3808:
            break

    # ----- local hill‑climbing refinement -----
    if best_h is not None:
        n = best_n
        while time.time() - start < budget_s - 0.5:
            # random fractional transfer move
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - best_h[i], best_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            h_new = best_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new + 1e-12 < best_c5:
                best_h, best_c5 = h_new, c5_new

    # ----- final safety fallback -----
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = project_h(best_h, best_n)
        best_n = best_n

    return best_h, best_c5, best_n
