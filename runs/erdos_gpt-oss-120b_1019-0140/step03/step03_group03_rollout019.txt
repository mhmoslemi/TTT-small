def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left(): return budget_s - (time.time() - start)

    n = 512
    best_h = None
    best_score = np.inf

    # get initial candidate from initial_h_values if present
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # upsample to n
        if init.size != n:
            init = upsample(init, n)
        init = project_onto_box_sum(init, n/2)
        # try gradient descent on this init
        if time_left() > 5:
            h_c, score_c = gradient_descent_momentum(init, time_left, rng)
            if score_c < best_score:
                best_h, best_score = h_c, score_c

    # random restarts
    while time_left() > 20:
        # random start: continuous vector with sum constraint
        v = rng.random(n)
        v = project_onto_box_sum(v, n/2)
        # gradient descent
        h_c, score_c = gradient_descent_momentum(v, time_left, rng)
        if score_c < best_score:
            best_h, best_score = h_c, score_c

    # binary improvement if time left > some threshold
    if best_h is not None and time_left() > 5:
        # Convert to binary by top half
        idx = np.argsort(best_h)[-n//2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        # evaluate
        score_bin = compute_c5(h_bin)
        if score_bin < best_score:
            best_h, best_score = h_bin, score_bin
        # local search
        h_bin, score_bin = binary_local_search(h_bin, time_left, rng)
        if score_bin < best_score:
            best_h, best_score = h_bin, score_bin

    # final projection (ensures sum)
    best_h = project_onto_box_sum(best_h, n/2)
    best_score = compute_c5(best_h)

    return best_h, best_score, n

def gradient_for_shift(h, s):
    n = h.size
    grad = np.zeros_like(h)
    if s >= 0:
        # forward: indices 0..n-s-1 have (1 - h[i+s])
        grad[:n - s] += 1.0 - h[s:]
        # backward: indices s..n-1 have -h[i-s]
        grad[s:] -= h[:n - s]
    else:
        k = -s
        # forward: indices k..n-1 have (1 - h[i-k])
        grad[k:] += 1.0 - h[:n - k]
        # backward: indices 0..n-k-1 have -h[i+k]
        grad[:n - k] -= h[k:]
    return grad

def project_onto_box_sum(v, s, lo=0.0, hi=1.0):
    # Clip v to [lo, hi]
    v_clipped = np.clip(v, lo, hi)
    # if sum matches within tolerance, return
    sum_v = v_clipped.sum()
    if abs(sum_v - s) <= 1e-12:
        return v_clipped
    # Need to adjust to meet sum s while staying within bounds.
    # Use algorithm: treat it as projecting onto intersection of box and hyperplane.
    # Equivalent to shifting v_clipped by lambda and then clipping again.
    # Use bisection on lambda for speed (since n small).
    # We'll use up to 30 iterations.

        w = np.clip(v - lam, lo, hi)
        sum_w = w.sum()
        if sum_w > s:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    # Clip initial to box
    w = np.clip(v, lo, hi)
    total = w.sum()
    if abs(total - s) <= 1e-12:
        return w
    # Bisection on lambda
    lam_low, lam_high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam, lo, hi)
        total = w.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)  # linear interpolation works both ways


def compute_c5_all(h):
    """Return all shifted overlap values (not scaled)."""
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    """Maximum overlap scaled by dx = 2/n."""
    n = h.shape[0]
    return np.max(compute_c5_all(h)) * (2.0 / n)


def compute_subgradient_max(h):
    """Subgradient of max_{k} correlation w.r.t. h."""
    n = h.size
    corr = compute_c5_all(h)
    max_val = np.max(corr)
    idxes = np.where(corr >= max_val - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
        else:
            k = -s
            grad[k:] += 1.0 - h[: n - k]
            grad[: n - k] -= h[k:]
    grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, max_iters, rng, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum using correct subgradient."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    momentum = np.zeros_like(h)
    best_h, best_score = h.copy(), compute_c5(h)
    for it in range(1, max_iters + 1):
        grad = compute_subgradient_max(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def binary_local_search(h0, time_left_func, rng, target_score=0.3808):
    """Focused swap search on binary vectors."""
    h = h0.copy()
    n = h.size
    best_score = compute_c5(h)
    best_h = h.copy()

            h_opt, sc_opt = gradient_descent_momentum(init, max_iters, rng)
            if sc_opt < best_score:
                best_h, best_score = h_opt, sc_opt

    # ---- Phase 1: many random restarts with gradient descent ----------
    while time_left() > 12.0:
        # balanced random start
        v = rng.random(n)
        v = project_onto_box_sum(v, target_sum)
        max_iters = min(25000, int(time_left() * 600))
        if max_iters < 1000:
            break
        h_opt, sc_opt = gradient_descent_momentum(v, max_iters, rng)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # ---- Phase 2: binary conversion and local improvement ------------
    if best_h is not None and time_left() > 5.0:
        # convert to balanced binary by taking top half entries
        idx = np.argsort(best_h)[-n // 2 :]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        bin_score = compute_c5(h_bin)
        if bin_score < best_score:
            best_h, best_score = h_bin, bin_score
        # focused binary local search
        best_h, best_score = binary_local_search(best_h, time_left, rng)

    # ---- Final projection and return ---------------------------------
    if best_h is None:
        # fallback: random balanced binary
        h_bin = np.zeros(n, dtype=float)
        h_bin[rng.choice(n, size=n // 2, replace=False)] = 1.0
        best_h = h_bin
        best_score = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
