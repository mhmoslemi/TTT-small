def binary_targeted_hill_climb(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score

    max_iters = int(min(500000, max(1, time_left() * 5000))) # e.g., 5k per sec
    T0 = 0.01
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # compute correlation
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        # find max shift index
        idx_max = np.argmax(corr_full)
        d = idx_max - (n - 1)
        # compute candidate I and J
        if d >= 0:
            # potential i's: from 0..n-1-d
            idx_range = np.arange(n - d)
            cond = (h[idx_range] == 1) & (h[idx_range + d] == 0)
            I = idx_range[cond]
        else:
            k = -d
            idx_range = np.arange(k, n)
            cond = (h[idx_range] == 1) & (h[idx_range - k] == 0)
            I = idx_range[cond]
        if I.size > 0:
            i = rng.choice(I)
            j = i + d
        else:
            # fallback random swap
            i = rng.integers(0, n)
            while h[i] != 1.0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[j] != 0.0:
                j = rng.integers(0, n)
        # propose swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def legendre_sequence(n):
    """Balanced binary Legendre sequence of length n (requires n+1 prime)."""
    p = n + 1
    seq = np.empty(n, dtype=float)
    for i in range(1, n + 1):
        leg = pow(i, (p - 1) // 2, p)
        seq[i - 1] = 1.0 if leg == 1 else 0.0
    return seq


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def resample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def binary_from_continuous(v):
    """Convert a vector to a balanced binary vector by thresholding at the median."""
    n = v.size
    order = np.argsort(v)
    b = np.zeros_like(v)
    b[order[n // 2:]] = 1.0
    return b


def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    shifts = np.arange(-(n - 1), n)
    for w, shift in zip(weights, shifts):
        if w == 0.0:
            continue
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n - shift]
        else:
            k = -shift
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
    return grad


def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end,
                       beta, time_left):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def binary_targeted_hill_climb(h0, rng, time_left, target=0.3808):
    """Swap‑based hill climbing focusing on the current worst shift."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score

    # dynamic iteration budget based on remaining time
    max_iters = int(min(300000, max(1, time_left() * 300)))
    T0 = 0.01

    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break

        corr_full = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr_full)
        d = idx_max - (n - 1)          # shift producing the max correlation

        # find 1‑0 pairs that realize this shift
        if d >= 0:
            valid_i = np.arange(n - d)
            pair_mask = (h[valid_i] == 1.0) & (h[valid_i + d] == 0.0)
            I = valid_i[pair_mask]
        else:
            k = -d
            valid_i = np.arange(k, n)
            pair_mask = (h[valid_i] == 1.0) & (h[valid_i - k] == 0.0)
            I = valid_i[pair_mask]

        if I.size > 0:
            i = rng.choice(I)
            j = i + d
        else:
            # fallback random swap
            i = rng.integers(0, n)
            while h[i] != 1.0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[j] != 0.0:
                j = rng.integers(0, n)

        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = max(T0 * (1.0 - it / max_iters), 1e-12)

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    target_bound = 0.3808
    n0 = 256                     # primary resolution

    # ----- generate initial candidates -----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init = resample(init_full, n0) if init_full.size != n0 else init_full.copy()
        candidates.append(project_onto_box_sum(init, n0 / 2.0))
    candidates.append(thue_morse(n0).astype(float))
    candidates.append(legendre_sequence(n0).astype(float))
    for _ in range(3):
        candidates.append(random_binary_balanced(n0, rng).astype(float))
    for _ in range(2):
        v = rng.random(n0)
        candidates.append(project_onto_box_sum(v, n0 / 2.0))

    # ----- evaluate candidates -----
    best_h = None
    best_score = float('inf')
    for h in candidates:
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h.copy()
        if best_score <= target_bound:
            break

    # ----- ensure binary for hill climbing -----
    if not np.all(np.logical_or(best_h == 0.0, best_h == 1.0)):
        best_h = binary_from_continuous(best_h)

    # ----- targeted binary hill climbing at n0 -----
    best_h, best_score = binary_targeted_hill_climb(best_h, rng, time_left, target=target_bound)

    # ----- occasional continuous refinement -----
    if time_left() > 30.0:
        n_iter_cont = int(min(5000, max(1, time_left() * 5)))
        best_h, sc_cont = smooth_grad_descent(best_h, n_iter=n_iter_cont,
                                              step0=0.5, alpha_start=5.0, alpha_end=30.0,
                                              beta=0.9, time_left=time_left)
        if sc_cont < best_score:
            best_score, best_h = sc_cont, best_h.copy()

    # ----- move to higher resolution if time permits -----
    if time_left() > 200.0:
        n_big = 512
        h_big = resample(best_h, n_big)
        h_big = project_onto_box_sum(h_big, n_big / 2.0)

        # continuous refinement at larger scale
        if time_left() > 80.0:
            n_iter_big = int(min(8000, max(1, time_left() * 4)))
            h_big, sc_big = smooth_grad_descent(h_big, n_iter=n_iter_big,
                                                step0=0.5, alpha_start=5.0, alpha_end=40.0,
                                                beta=0.9, time_left=time_left)
            if sc_big < best_score:
                best_score, best_h = sc_big, h_big.copy()

        # binary hill climbing at larger scale
        if time_left() > 30.0:
            bin_big = binary_from_continuous(h_big)
            bin_big, sc_bin_big = binary_targeted_hill_climb(bin_big, rng, time_left,
                                                             target=target_bound)
            if sc_bin_big < best_score:
                best_score, best_h = sc_bin_big, bin_big.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
