def adam_projected(h0, time_left, max_iter=200000, top_k=3, lr=0.1):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    while it <= max_iter and time_left() > 0:
        # compute gradient
        n = h.size
        # correlation compute
        corr_full = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        # Find top_k indices (largest)
        if top_k >= corr_full.size:
            shifts = np.arange(-n+1, n)
        else:
            top_idx = np.argpartition(corr_full, -top_k)[-top_k:]
            shifts = top_idx - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            # compute term1 and term2
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n-s]
                term2[:n-s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad /= top_k

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project onto box sum
        h_new = project_onto_box_sum(h_new, h.size/2.0)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo<=x_i<=hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap C5 for step function represented by array h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to size new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Alias for upsample (linear interpolation) when reducing size."""
    return upsample(arr, new_n)


def thue_morse(n):
    """Binary Thue–Morse sequence as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def generate_m_seq(m):
    """Generate binary m‑sequence (length 2**m-1) using a primitive polynomial."""
    taps_dict = {8: [8, 6, 5, 4]}  # primitive for m=8
    taps = taps_dict[m]
    state = np.ones(m, dtype=int)
    L = 2**m - 1
    seq = np.empty(L, dtype=int)
    for i in range(L):
        seq[i] = state[-1]
        fb = 0
        for t in taps:
            fb ^= state[-t]
        state[1:] = state[:-1]
        state[0] = fb
    return seq.astype(float)


def compute_gradient_topk(h, shifts):
    """Gradient of average of selected shifts (1 - h[i+shift] - h[i-shift])."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        grad += 1.0
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad


def adam_descent(h0, time_left, max_iter, top_k, lr):
    """Projected Adam optimizer on C5 (max shift overlap) using top‑k shifts."""
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    while it <= max_iter and time_left() > 0:
        # correlation to identify large shifts
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        if top_k >= corr_full.size:
            shifts = np.arange(-n + 1, n)
        else:
            idx = np.argpartition(corr_full, -top_k)[-top_k:]
            shifts = idx - (n - 1)
        grad = compute_gradient_topk(h, shifts)

        # Adam update with decaying step size
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step_scale = lr / np.sqrt(it)
        step = step_scale * (m_hat / (np.sqrt(v_hat) + eps))

        h_new = h - step
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)

        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score


def binary_swap_anneal(h0, rng, time_left,
                      max_iters=400000,
                      T0=0.2, Tmin=1e-5, decay=0.99996):
    """Simulated annealing on binary vectors via swapping one 1 with one 0."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h, best_score = h.copy(), current_score
    T = T0
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

    # large continuous refinement
    h_ref, sc_ref = adam_descent(best_h, time_left,
                                 max_iter=150000,
                                 top_k=7,
                                 lr=0.2)
    if sc_ref < best_score:
        best_score, best_h = sc_ref, h_ref.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ---------- Phase 3: Binary conversion ----------
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ---------- Phase 4: Continuous refinement from binary ----------
    h_ref2, sc_ref2 = adam_descent(best_h, time_left,
                                   max_iter=60000,
                                   top_k=5,
                                   lr=0.15)
    if sc_ref2 < best_score:
        best_score, best_h = sc_ref2, h_ref2.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # ---------- Phase 5: Binary annealing ----------
    if time_left() > 2.0:
        h_anneal, sc_anneal = binary_swap_anneal(best_h, rng, time_left,
                                                max_iters=500000,
                                                T0=0.15,
                                                decay=0.99997)
        if sc_anneal < best_score:
            best_score, best_h = sc_anneal, h_anneal.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 6: Random continuous restarts ----------
    while time_left() > 8.0:
        v = rng.random(n_large)
        v = project_onto_box_sum(v, n_large / 2.0)
        h_tmp, sc_tmp = adam_descent(v, time_left,
                                     max_iter=40000,
                                     top_k=5,
                                     lr=0.2)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
        if best_score <= target:
            break

    # ---------- Phase 7: Larger grid (optional) ----------
    if time_left() > 30.0:
        n_big = 512
        h_big = upsample(best_h, n_big)
        h_big = project_onto_box_sum(h_big, n_big / 2.0)
        h_big_ref, sc_big_ref = adam_descent(h_big, time_left,
                                             max_iter=120000,
                                             top_k=9,
                                             lr=0.15)
        if sc_big_ref < best_score:
            best_score, best_h = sc_big_ref, h_big_ref.copy()
        # try binary on big grid
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin_big = np.zeros_like(best_h)
        h_bin_big[idx] = 1.0
        sc_bin_big = compute_c5(h_bin_big)
        if sc_bin_big < best_score:
            best_score, best_h = sc_bin_big, h_bin_big.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 8: Extra deterministic binary candidate ----------
    # m‑sequence for m=8 (length 255) -> embed into 256
    seq = generate_m_seq(8)
    h_seq = np.zeros(256, dtype=float)
    h_seq[:255] = seq
    h_seq[255] = 1.0  # now exactly 128 ones
    sc_seq = compute_c5(h_seq)
    if sc_seq < best_score:
        best_score, best_h = sc_seq, h_seq.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # final guard projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
