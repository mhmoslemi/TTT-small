def compute_gradient_weighted(h, shifts, weights=None):
    n = h.size
    grad = np.zeros_like(h)
    if weights is None:
        weights = np.ones_like(shifts, dtype=float)
    for shift, w in zip(shifts, weights):
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += w * (1.0 - term1 - term2)
    if weights is not None and len(weights) > 0:
        grad /= weights.sum()
    return grad

def gradient_descent_topk(h0, time_left, step0=0.5, mu=0.9, top_frac=0.01, max_iter=200000):

    h = h0.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    n = h.size
    target_sum = n/2

    while it <= max_iter and time_left()>0:
        # Nesterov step
        y = h + mu * (h - h_prev)
        # compute top-k shifts
        corr = np.correlate(y, 1-y, mode='full')
        max_corr = corr.max()
        # Choose threshold: top fraction
        top_k = max(1, int(top_frac * corr.size))
        # use argpartition
        top_idx = np.argpartition(-corr, top_k-1)[:top_k]
        shifts = top_idx - (n-1)
        grad = compute_gradient_weighted(y, shifts) # equal weights
        # step size
        eta = step0 / np.sqrt(it)
        h_new = y - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        # update
        h_prev = h
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=500000):
    h = h0.copy()
    n = h.size
    # ensure exactly half ones
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.05
    T_min = 1e-5
    decay = 0.99995
    it = 0
    # precompute indices sets
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # attempt swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta/T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_full(h):
    n = h.size
    H = np.fft.fft(h, n=2 * n)
    G = np.fft.fft(1.0 - h, n=2 * n)
    corr = np.real(np.fft.ifft(H * np.conj(G)))
    return corr

def compute_c5(h):
    n = h.size
    return compute_corr_full(h).max() * (2.0 / n)

def compute_gradient_weighted(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_topk(h0, time_left, step0=0.5, mu=0.9, top_frac=0.01, max_iter=200000):
    h = h0.copy()
    h_prev = h.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        y = h + mu * (h - h_prev)
        corr = compute_corr_full(y)
        sz = corr.size
        top_k = max(1, int(top_frac * sz))
        top_idx = np.argpartition(-corr, top_k - 1)[:top_k]
        shifts = top_idx - (n - 1)

        h_prev, h = h, h_new
        it += 1
    return best_h, best_score

    x_new = np.linspace(0, 1, new_n, endpoint=False)

def thue_morse(n):
    return np.fromiter(((bin(int(i)).count('1') & 1) for i in range(n)), dtype=float, count=n)

def binary_swap_anneal(h0, rng, time_left, max_iters=800000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.05
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    n_small = 64
    n_large = 256
    best_h = None
    best_score = np.inf

    # ---- small‑scale candidates ----
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        candidates.append(project_onto_box_sum(downsample(init_full, n_small), n_small / 2.0))
    candidates.append(thue_morse(n_small))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in candidates:
        if time_left() < 1.0:
            break
        h_opt, sc_opt = gradient_descent_topk(cand, time_left,
                                              step0=0.8, mu=0.9,
                                              top_frac=0.02, max_iter=30000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    # ---- upscale to larger grid ----
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5(best_h)
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)
    h_opt_large, sc_opt_large = gradient_descent_topk(h_up, time_left,
                                                      step0=0.6, mu=0.9,
                                                      top_frac=0.01, max_iter=50000)
    if sc_opt_large < best_score:
        best_score, best_h = sc_opt_large, h_opt_large.copy()
    if best_score <= target:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        return best_h, compute_c5(best_h), best_h.size

    # ---- binarize ----
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        return best_h, compute_c5(best_h), best_h.size

    # ---- binary simulated annealing ----
    if time_left() > 0.5:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=800000)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
