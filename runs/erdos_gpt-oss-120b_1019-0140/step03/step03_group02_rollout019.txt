def adam_optimize(h0, n_iter, step0, time_left, alpha_start=1.0, alpha_end=100.0, beta1=0.9, beta2=0.999):

    v = np.zeros_like(h)

    for it in range(1, n_iter+1):

def adam_optimize(h0, n_iter, step0, time_left, alpha_start=1.0, alpha_end=100.0, beta1=0.9, beta2=0.999, eps=1e-8):
    n = h0.shape[0]
    s = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start)*(it-1)/max(1, n_iter-1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1*m + (1 - beta1)*grad
        v = beta2*v + (1 - beta2)*(grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h0, rng, time_left, max_iters=200000, target=0.3808):
    n = h0.size
    curr = h0.copy()
    curr_score = compute_c5_fft(curr)
    best = curr.copy()
    best_score = curr_score
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        while curr[i] != 1.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while curr[j] != 0.0:
            j = rng.integers(0, n)
        new = curr.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_score = compute_c5_fft(new)
        if new_score < curr_score:
            curr, curr_score = new, new_score
            if new_score < best_score:
                best, best_score = new.copy(), new_score
                if best_score <= target:
                    break
        else:
            T = T0 * (1 - it / max_iters)
            if T > 0 and rng.random() < np.exp(-(new_score - curr_score) / T):
                curr, curr_score = new, new_score
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n0 = 64
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n0:
            h0 = upsample_h(h0, n0)
    else:
        h0 = thue_morse(n0)
    h0 = project_onto_box_sum(h0, n0 / 2.0)

    best_h = h0.copy()
    best_score = compute_c5_fft(best_h)

    # Stage 1: small continuous refinement
    if time_left() > 20:
        iters = int(min(3000, max(500, time_left() * 5)))
        h_opt, sc_opt = adam_optimize(best_h, n_iter=iters, step0=0.5,
                                      time_left=time_left,
                                      alpha_start=1.0, alpha_end=50.0)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # Stage 2: upsample to 128 and refine
    n1 = 128
    if time_left() > 20:
        up1 = upsample_h(best_h, n1)
        up1 = project_onto_box_sum(up1, n1 / 2.0)
        candidate_score = compute_c5_fft(up1)
        if candidate_score < best_score:
            best_h, best_score = up1, candidate_score
        if time_left() > 20:
            iters = int(min(4000, max(500, time_left() * 5)))
            h_opt1, sc_opt1 = adam_optimize(best_h, n_iter=iters, step0=0.5,
                                            time_left=time_left,
                                            alpha_start=1.0, alpha_end=100.0)
            if sc_opt1 < best_score:
                best_h, best_score = h_opt1, sc_opt1

    # Stage 3: binary hill climbing on current best
    if time_left() > 20:
        cur_n = best_h.shape[0]
        half = cur_n // 2
        binary = np.zeros(cur_n, dtype=float)
        binary[np.argpartition(best_h, -half)[-half:]] = 1.0
        max_iters = int(min(1_000_000, max(200_000, time_left() * 3000)))
        bin_h, bin_sc = binary_hill_climb(binary, rng, time_left,
                                          max_iters=max_iters, target=0.3808)
        if bin_sc < best_score:
            best_h, best_score = bin_h, bin_sc

    # Stage 4: upscale to 256 and binary hill climbing
    if time_left() > 40:
        n2 = 256
        up2 = upsample_h(best_h, n2)
        up2 = project_onto_box_sum(up2, n2 / 2.0)
        half2 = n2 // 2
        binary2 = np.zeros(n2, dtype=float)
        binary2[np.argpartition(up2, -half2)[-half2:]] = 1.0
        max_iters2 = int(min(2_000_000, max(300_000, time_left() * 3000)))
        bin_h2, bin_sc2 = binary_hill_climb(binary2, rng, time_left,
                                            max_iters=max_iters2, target=0.3808)
        if bin_sc2 < best_score:
            best_h, best_score = bin_h2, bin_sc2

    best_h = project_onto_box_sum(best_h, best_h.shape[0] / 2.0)
    final_score = compute_c5_fft(best_h)
    return best_h, final_score, best_h.size
