    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

def refine_with_slsqp(h, rng):
    n = len(h)
    M = n // 2
    dx = 2.0 / n
    # compute initial t
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    # constraints
    sum_con = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - M}
    overlap_cons = []
    for k in range(-n+1, n):
        if k == 0: continue
        overlap_cons.append({'type': 'ineq',
                             'fun': lambda x, k=k: x[-1] - np.dot(x[:n][:n-k] if k>=0 else x[:n][ -k: ], 1.0 - (x[:n][k:] if k>=0 else x[:n][:n+k])) * dx })
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=[sum_con] + overlap_cons,
                       options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h, compute_c5(h)
    if not res.success:
        return h, compute_c5(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # reproject sum
    cur = h_opt.sum()
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

    M = n // 2
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()

            break
        it += 1

        if r < 0.04:  # big random restart
            h_new = generate_initial_h(n, M, rng)

            h_new[j] += delta

def slsqp_refine(h, rng):
    n = len(h)
    M = n // 2
    dx = 2.0 / n
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # equality constraint
    con_eq = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - M}
    # inequality constraints for each shift
    cons = [con_eq]
    for k in range(-n+1, n):
        if k == 0:
            continue
        def fun_factory(k):
            def fun(x):
                h_vec = x[:n]
                if k >= 0:
                    overl = np.dot(h_vec[:n - k], 1.0 - h_vec[k:])
                else:
                    kk = -k
                    overl = np.dot(h_vec[kk:], 1.0 - h_vec[:n - kk])
                return x[-1] - overl * dx
            return fun
        cons.append({'type': 'ineq', 'fun': fun_factory(k)})

    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h, compute_c5(h)
    if not res.success:
        return h, compute_c5(h)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # correct sum
    cur = h_opt.sum()
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def adjust_sum_to_target(h, target):
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc = min(inc, (1.0 - h[idx]).min())
            h[idx] += inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            dec = (-diff) / idx.size
            dec = min(dec, h[idx].min())
            h[idx] -= dec
        diff = target - h.sum()
    return h

def generate_initial_h(n, M, rng):
    h = rng.random(n)
    h = h * M / h.sum()
    h = np.clip(h, 0.0, 1.0)
    h = adjust_sum_to_target(h, M)
    return h

def generate_random_binary(n, M, rng):
    h = np.zeros(n)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def simulated_annealing(h_start, rng, time_budget):
    n = h_start.shape[0]
    M = n // 2
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    T = 0.02
    min_T = 1e-6
    decay = 0.9995
    start = time.time()
    it = 0
    target = 0.3808

    while True:
        if time.time() - start > time_budget:
            break
        it += 1
        r = rng.random()
        if r < 0.04:  # restart
            cur_h = generate_initial_h(n, M, rng)
            cur_c5 = compute_c5(cur_h)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
                if best_c5 < target:
                    break
            continue
        elif r < 0.34:  # transfer move (possibly large)
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:  # small perturbation
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer * 0.1
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-12)):
            cur_h, cur_c5 = h_new, c5_new

        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new
            if best_c5 < target:
                break

        T = max(min_T, T * decay)
        if it % 200000 == 0:
            T = 0.02

    return best_h, best_c5

def slsqp_refine(h, rng):
    n = len(h)
    M = n // 2
    dx = 2.0 / n
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def eq_sum(x):
        return np.sum(x[:n]) - M

    constraints = [{'type': 'eq', 'fun': eq_sum}]
    for k in range(-n + 1, n):
        if k == 0:
            continue

        def make_fun(k_shift):
            def fun(x):
                hv = x[:n]
                if k_shift >= 0:
                    overl = np.dot(hv[:n - k_shift], 1.0 - hv[k_shift:])
                else:
                    kk = -k_shift
                    overl = np.dot(hv[kk:], 1.0 - hv[:n - kk])
                return x[-1] - overl * dx
            return fun

        constraints.append({'type': 'ineq', 'fun': make_fun(k)})

    try:
        res = minimize(lambda x: x[-1],
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h, compute_c5(h)

    if not res.success:
        return h, compute_c5(h)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    n_vals = [80, 96, 112, 128, 160]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in n_vals:
                n_vals.remove(n0)
            n_vals = [n0] + n_vals

    for idx, n in enumerate(n_vals):
        if time.time() - start > budget_s - 5:
            break
        M = n // 2
        candidates = []

        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            candidates.append(np.array(initial_h_values, dtype=float))

        if n == 80:
            p = 79
            leg = legendre_sequence(p)
            h_leg = np.concatenate([leg, [1.0]])
            shift = rng.integers(n)
            candidates.append(np.roll(h_leg, shift))

        candidates.append(generate_random_binary(n, M, rng))
        candidates.append(generate_initial_h(n, M, rng))

        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc_n = min(remaining * 0.6, 600.0)
        else:
            alloc_n = min(remaining * 0.4 / (len(n_vals) - idx), 300.0)

        cand_alloc = alloc_n / max(1, len(candidates))
        best_h_n = None
        best_c5_n = np.inf

        for h0 in candidates:

                break
            h_sa, c5_sa = simulated_annealing(h0, rng, cand_alloc)
            if c5_sa < best_c5_n:
                best_h_n, best_c5_n = h_sa, c5_sa
                if c5_sa < best_c5:
                    best_h, best_c5, best_n = h_sa, c5_sa, n
                if best_c5 < 0.3808:
                    break
            if best_c5_n < 0.3808:
                break

        if best_h_n is not None and time.time() - start < budget_s - 5:
            h_ref, c5_ref = slsqp_refine(best_h_n, rng)
            if c5_ref < best_c5_n:
                best_h_n, best_c5_n = h_ref, c5_ref
                if c5_ref < best_c5:
                    best_h, best_c5, best_n = h_ref, c5_ref, n
            if best_c5_n < best_c5:
                best_h, best_c5, best_n = best_h_n, best_c5_n, n
            if best_c5 < 0.3808:
                break

    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = best_h.sum()
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
