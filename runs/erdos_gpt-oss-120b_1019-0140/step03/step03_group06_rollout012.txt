def anneal_optimize(h, cur_val, best_h, best_val, start_time, deadline, rng):
    n = len(h)
    while time.time() < deadline - 0.1:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # compute worst shift contributions
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        # find index i that has largest contribution (most to overlap)
        i_local = np.argmax(prod)
        i = i_range[i_local]

        # pick destination j randomly from those with h < 1
        possible_j = np.where(h < 1.0 - 1e-12)[0]
        if len(possible_j) == 0:
            break
        j = rng.choice(possible_j)

        # max amount can transfer
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer  # random fraction of max
        # apply
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            # record best overall
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                # early exit if close to target
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            # Accept with probability exp((cur_val - new_val) / T)
            remaining = deadline - time.time()
            # define temperature schedule; maybe use remaining ratio
            T = max(remaining / (deadline - start_time) * 0.02, 1e-12)  # simple descending
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break

            v[idx] += inc
            diff -= inc
        else:

            v[idx] -= dec

            i_range = np.arange(n - k)

        h[j] += delta

                    best_val = val
                    best_h = trial.copy()
                    if best_val < 0.3808:
                        return best_h, best_val
    return best_h, best_val

def anneal_optimize(h, cur_val, best_h, best_val, start, deadline, rng):
    n = h.shape[0]
    while time.time() < deadline - 0.2:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0:
            break
        j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    return best_h, best_val
        else:
            remaining = deadline - time.time()
            total = deadline - start
            T = max(remaining / total * 0.02, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

def binary_local_optimize(base_h, target, start, deadline, rng):
    n = len(base_h)
    idx_desc = np.argsort(-base_h)
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx_desc[:int(target)]] = 1.0
    cur_val = compute_c5(bin_h)
    best_h = bin_h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.1:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        trial = best_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        val = compute_c5(trial)
        if val < best_val:
            best_val = val
            best_h = trial
            if best_val < 0.3808:
                break
    return best_h, best_val

def binary_random_optimize(target, n, start, deadline, rng):
    best_h = None
    best_val = float('inf')
    int_target = int(target)
    while time.time() < deadline - 0.1:
        v = np.zeros(n, dtype=float)
        idx = rng.choice(n, int_target, replace=False)
        v[idx] = 1.0
        cur_val = compute_c5(v)
        no_improve = 0
        while time.time() < deadline - 0.1 and no_improve < 200:
            ones = np.where(v > 0.5)[0]
            zeros = np.where(v < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            trial = v.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            val = compute_c5(trial)
            if val < cur_val:
                v = trial
                cur_val = val
                no_improve = 0
            else:
                no_improve += 1
        if cur_val < best_val:
            best_val = cur_val
            best_h = v.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Differential Evolution phase (≈60% of total time)
    de_deadline = start + budget_s * 0.6
    if time.time() < de_deadline - 1.0:
        best_h, best_val = de_optimize(best_h, target, n, start, de_deadline, rng)

    # Simulated annealing refinement (remaining time)
    if time.time() < deadline - 5.0:
        cur_h = best_h.copy()
        cur_val = best_val
        best_h, best_val = anneal_optimize(cur_h, cur_val, best_h, best_val, start, deadline, rng)

    # Binary rounding + local swap improvement
    if time.time() < deadline - 2.0:
        bin_h, bin_val = binary_local_optimize(best_h, target, start, deadline, rng)
        if bin_val < best_val:
            best_h, best_val = bin_h, bin_val

    # Additional random binary search if time permits
    if time.time() < deadline - 2.0:
        rand_h, rand_val = binary_random_optimize(target, n, start, deadline, rng)
        if rand_h is not None and rand_val < best_val:
            best_h, best_val = rand_h, rand_val

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
