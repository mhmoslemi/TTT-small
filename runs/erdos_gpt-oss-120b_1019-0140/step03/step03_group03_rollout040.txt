def gradient_descent_momentum(h0, time_left_func, max_iter=100000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while it < max_iter:
        if time_left_func() <= 0:
            break
        it += 1
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        # indices where corr nearly max
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient as average over shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad = grad / len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, time_left_func, max_iter=None, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    if max_iter is None:
        it_per_sec = 1500.0 * 512.0 / n
        max_iter = int(min(300000 if n <= 512 else 150000,
                           max(0.0, time_left_func()) * it_per_sec))
    if max_iter < 1:
        return best_h, best_score

    for it in range(1, max_iter + 1):
        if time_left_func() <= 0.0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)

        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def local_swap_improvement(h, max_iters, rng):
    """Improve a binary vector by swapping to lower the maximum overlap."""
    h = h.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)

    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            i_vals = np.arange(n - k)
            hi = h[i_vals]
            hk = h[i_vals + k]
        else:
            s = -k
            i_vals = np.arange(s, n)
            hi = h[i_vals]
            hk = h[i_vals + k]

        I = i_vals[(hi > 0.5) & (hk < 0.5)]
        J = i_vals[(hi < 0.5) & (hk > 0.5)]

        if I.size == 0 or J.size == 0:
            break

        i_sel = rng.choice(I)
        j_sel = rng.choice(J)

        # perform swap
        h[i_sel], h[j_sel] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        else:
            # revert
            h[i_sel], h[j_sel] = 1.0, 0.0

    return best_h, best_score

    # ------------------------------------------------------------------

    n0 = 64

    return best_h, best_score, best_h.size
