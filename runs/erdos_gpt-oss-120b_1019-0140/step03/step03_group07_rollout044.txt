def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b = 1.0 - h
    n_fft = 1 << ((2*n-1).bit_length())
    # ensure n_fft >= 2*n
    if n_fft < 2*n:
        n_fft <<= 1
    fft_a = np.fft.rfft(a, n_fft)
    fft_b = np.fft.rfft(b, n_fft)
    corr = np.fft.irfft(fft_a * np.conj(fft_b), n_fft)
    # linear correlation from -n+1 to n-1 is first 2*n-1 entries
    corr = corr[:2*n-1]
    return np.max(corr) * dx
```

class SumConstraint:
    def __init__(self, n):
        self.n=n
        self.target = n/2.0
    def __call__(self,x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self,k,n):
        self.k=k; self.n=n; self.dx=2.0/n
    def __call__(self,x):
        h = x[:self.n]
        k=self.k
        if k>=0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def is_prime(num: int) -> bool:
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5)
    for i in range(3, r + 1, 2):
        if num % i == 0:
            return False
    return True


def legendre_sequence(p: int) -> np.ndarray:
    """0/1 Legendre sequence of length p (p prime)."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for quadratic residue, p-1 otherwise
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq


def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx using FFT based correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b = 1.0 - h
    n_fft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=n_fft)
    fb = np.fft.rfft(b, n=n_fft)
    corr = np.fft.irfft(fa * np.conj(fb), n=n_fft)[: 2 * n - 1]
    return np.max(corr) * dx


# ---------- constraints for SLSQP ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""

    def __init__(self, n: int):
        self.n = n
        self.target = n / 2.0

    def __call__(self, x):
        return np.sum(x[: self.n]) - self.target


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""

    def __init__(self, k: int, n: int):
        self.k = k
        self.n = n
        self.dx = 2.0 / n

    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overl * self.dx


# ---------- simulated annealing on binary vectors ----------
def anneal_binary(initial_h: np.ndarray, M: int, max_time: float, rng: np.random.Generator):
    """Hill‑climbing with Metropolis acceptance on a binary (0/1) vector."""
    n = initial_h.shape[0]
    # ensure exactly M ones
    h = np.rint(initial_h).astype(float)
    cur = h.sum()
    diff = M - cur
    if diff > 0:
        zeros = np.where(h < 0.5)[0]
        add = rng.choice(zeros, int(diff), replace=False)
        h[add] = 1.0
    elif diff < 0:
        ones = np.where(h > 0.5)[0]
        rem = rng.choice(ones, int(-diff), replace=False)
        h[rem] = 0.0
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5
    T = 0.004  # initial temperature
    it = 0
    start = time.time()
    while time.time() - start < max_time:
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        it += 1
        if it % 800 == 0:
            T *= 0.85
    return best_h, best_c5


# ---------- SLSQP refinement ----------
def slsqp_refine(h_start: np.ndarray, M: int, time_left: float):
    """One SLSQP run seeking lower C5, respecting constraints."""
    n = h_start.shape[0]
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])
    maxiter = 1200 if time_left > 5 else 400
    try:
        res = minimize(
            fun=lambda x: x[-1],
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h_start, t0
    if not res.success:
        return h_start, t0
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = h_opt.sum()
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

def search_n(n: int, budget: float, rng: np.random.Generator):
    M = n // 2
    best_h = None
    best_c5 = np.inf
    seeds = []

    # parent if present and matching size
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        parent = np.array(initial_h_values, dtype=float)
        cur = parent.sum()
        if abs(cur - M) > 1e-12:
            parent = parent + (M - cur) / n
            parent = np.clip(parent, 0.0, 1.0)
        seeds.append(parent)

    # Legendre seed when n-1 is prime
    p = n - 1
    if is_prime(p):
        leg = legendre_sequence(p)
        leg_full = np.concatenate([leg, [1.0]])  # sum = M
        leg_full = np.roll(leg_full, rng.integers(n))
        seeds.append(leg_full.astype(float))

    # few random binary seeds
    for _ in range(3):
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
        seeds.append(h)

    start = time.time()
    for idx, seed in enumerate(seeds):
        now = time.time()
        if now - start > budget - 1.0:
            break
        remaining = budget - (now - start)
        share = remaining / (len(seeds) - idx)
        # annealing
        sa_time = share * 0.7
        h_loc, c5_loc = anneal_binary(seed, M, sa_time, rng)

        if c5_loc < best_c5:

            if best_c5 < 0.3808:
                break
