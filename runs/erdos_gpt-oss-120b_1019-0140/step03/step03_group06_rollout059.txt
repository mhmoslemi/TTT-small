def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint, Bounds

    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n/2, rng)
    target = n / 2.0
    dx = 2.0 / n

    # enforce sum exactly
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # prepare constraints for trust-constr
    # linear equality sum constraint
    A_eq = np.concatenate([np.ones(n), [0.0]])[np.newaxis, :]
    lin_eq = LinearConstraint(A_eq, [target], [target])

    # nonlinear inequality t >= overlap
    def overlap_vec(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        t = x[n]
        return t - overlaps

    nl_ineq = NonlinearConstraint(overlap_vec, np.zeros(2*n-1), np.full(2*n-1, np.inf), jac='2-point')

    # bounds
    bounds = Bounds(np.concatenate([np.zeros(n), [0.0]]), np.concatenate([np.ones(n), [np.inf]]))

    # objective gradient
    def obj_grad(x):
        grad = np.zeros_like(x)
        grad[n] = 1.0
        return grad

    # start optimization restarts
    while time.time() < deadline - 5.0:
        # generate start point
        if rng.random() < 0.5 and best_h is not None:
            start_h = best_h.copy()
            # small perturbation
            for _ in range(5):
                src = np.where(start_h > 1e-8)[0]
                dst = np.where(start_h < 1.0 - 1e-8)[0]
                if len(src) == 0 or len(dst) == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)
        else:
            # random binary or continuous
            if rng.random() < 0.6:
                start_h = random_binary_vector(n, target, rng)
            else:
                start_h = random_continuous_vector(n, target, rng)
        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=lambda x: x[n],
                x0=x0,
                method='trust-constr',
                jac=obj_grad,
                hess=None,
                bounds=bounds,
                constraints=[lin_eq, nl_ineq],
                options={'maxiter': 2000, 'verbose': 0}
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)

        if val < best_val - 1e-12:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    if time.time() < deadline - 0.5:

    # final enforce and compute
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n

import time
import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint, Bounds

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def enforce_sum(v, target, rng):
    """Project v onto [0,1]^n with exact sum==target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, size=ones, replace=False)] = 1.0
    return v

def greedy_refine(h, target, dx, deadline, rng):
    """Greedy local search moving mass from worst product indices."""
    n = h.shape[0]
    cur_val = compute_c5(h)
    it = 0
    while time.time() < deadline - 0.01:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            i_range = np.arange(n - k)
            prod = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = h[i_range] * (1.0 - h[i_range - kk])
        i = i_range[np.argmax(prod)]

        # choose destination with headroom and large shifted partner
        if k >= 0:
            mask = (h[i_range] < 1.0 - 1e-12)
            cand = i_range[mask]
            if cand.size == 0:
                cand = np.where(h < 1.0 - 1e-12)[0]
                if cand.size == 0:
                    break
                j = rng.choice(cand)
            else:
                shift_vals = h[cand + k]
                j = cand[np.argmax(shift_vals)]
        else:
            kk = -k
            mask = (h[i_range] < 1.0 - 1e-12)
            cand = i_range[mask]
            if cand.size == 0:
                cand = np.where(h < 1.0 - 1e-12)[0]
                if cand.size == 0:
                    break
                j = rng.choice(cand)
            else:
                shift_vals = h[cand - kk]
                j = cand[np.argmax(shift_vals)]

        if i == j:
            other = np.where((h < 1.0 - 1e-12) & (np.arange(n) != i))[0]
            if other.size == 0:
                break
            j = rng.choice(other)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue

        improved = False
        for factor in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            delta = max_transfer * factor
            h[i] -= delta
            h[j] += delta
            new_val = compute_c5(h)
            if new_val < cur_val - 1e-14:
                cur_val = new_val
                improved = True
                break
            h[i] += delta
            h[j] -= delta
        if not improved:
            delta = max_transfer * 0.01
            if delta > 0:
                h[i] -= delta
                h[j] += delta
                new_val = compute_c5(h)
                if new_val < cur_val:
                    cur_val = new_val
                else:
                    h[i] += delta
                    h[j] -= delta

        if it % 1000 == 0:
            total = h.sum()
            diff = target - total
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, target, rng)
    return h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    dx = 2.0 / n
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- constraints for trust-constr -----
    A_eq = np.concatenate([np.ones(n), [0.0]])[np.newaxis, :]
    lin_eq = LinearConstraint(A_eq, [target], [target])

    def overlap_vec(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        t = x[n]
        return t - overlaps

    nl_ineq = NonlinearConstraint(overlap_vec,
                                  np.zeros(2 * n - 1),
                                  np.full(2 * n - 1, np.inf),
                                  jac='2-point')

    bounds = Bounds(np.concatenate([np.zeros(n), [0.0]]),
                    np.concatenate([np.ones(n), [np.inf]]))

    def obj_grad(x):
        g = np.zeros_like(x)
        g[n] = 1.0
        return g

    # ----- restart loop with trust-constr -----
    while time.time() < deadline - 5.0:
        # generate a feasible start
        if rng.random() < 0.5 and best_h is not None:
            start_h = best_h.copy()
            for _ in range(5):
                src = np.where(start_h > 1e-8)[0]
                dst = np.where(start_h < 1.0 - 1e-8)[0]
                if len(src) == 0 or len(dst) == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)
        else:
            if rng.random() < 0.6:
                start_h = random_binary_vector(n, target, rng)
            else:
                start_h = random_continuous_vector(n, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(fun=lambda x: x[n],
                           x0=x0,
                           method='trust-constr',
                           jac=obj_grad,
                           bounds=bounds,
                           constraints=[lin_eq, nl_ineq],
                           options={'maxiter': 2000, 'verbose': 0})
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)

        if val < best_val - 1e-12:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # ----- greedy refinement -----
    if time.time() < deadline - 0.5:
        best_h, best_val = greedy_refine(best_h, target, dx, deadline, rng)

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
