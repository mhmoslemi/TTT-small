def projected_gradient_descent(h_init, target_sum, max_iters, time_limit):
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    alpha = 0.5
    for i in range(max_iters):
        if time.time() > time_limit: break
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        cur_val = max_corr*dx
        # find all max idx (equal)
        eps = 1e-12
        max_idxs = np.where(np.abs(corr - max_corr) <= eps)[0]
        # compute gradient sum
        grad = np.zeros_like(h)
        for idx in max_idxs:
            k = idx - (n-1)
            grad += gradient_for_shift(h, k)
        # attempt step sizes
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            new_h = h - alpha*factor*grad
            new_h = project_to_box_sum(new_h, target_sum)
            val = compute_c5(new_h)
            if val < cur_val - 1e-12:
                h = new_h
                cur_val = val
                improved = True
                break
        if not improved:
            alpha *= 0.5
            if alpha < 1e-6:
                # jitter
                noise = np.random.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h+noise,0,1), target_sum)
                alpha = 0.5
        # record best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # dimension
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    # projection function

    # compute initial best
    h0 = project_to_box_sum(np.clip(h0,0,1), target_sum)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Population
    popsize = max(10, n // 8)  # 10 for n=80 => 10
    population = [best_h]
    # add other random individuals
    while len(population) < popsize:
        rand_h = rng.random(n)
        rand_h = project_to_box_sum(rand_h, target_sum)
        population.append(rand_h)
    # Evaluate population
    pop_vals = [compute_c5(ind) for ind in population]
    # update best
    for ind, val in zip(population, pop_vals):
        if val < best_val:
            best_val = val
            best_h = ind.copy()

    # DE parameters
    F = 0.8
    CR = 0.9

    # DE main loop
    safety = 5.0  # seconds buffer
    while time.time() < start_time + budget_s - safety:
        # Shuffle indices to random order
        idxs = list(range(popsize))
        rng.shuffle(idxs)
        for i in idxs:
            # select distinct a,b,c
            a, b, c = rng.choice([j for j in range(popsize) if j != i], size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            # clip to [0,1]
            mutant = np.clip(mutant, 0.0, 1.0)
            # project sum
            mutant = project_to_box_sum(mutant, target_sum)
            # crossover
            trial = population[i].copy()
            rand_j = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == rand_j:
                    trial[j] = mutant[j]
            # project after crossover (just in case)
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)
            # evaluate
            trial_val = compute_c5(trial)
            if trial_val < pop_vals[i]:
                population[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
            # early exit if target achieved
            if best_val < 0.3808:
                break
        if best_val < 0.3808:
            break
        # Occasionally do gradient refinement on best
        if (time.time() - start_time) % 30 < 0.5:  # every ~30 seconds, do GD
            # Run a short gradient descent from best
            gd_h, gd_val = gradient_descent(best_h, target_sum, max_iter=800, time_limit=start_time + budget_s - safety)
            if gd_val < best_val:
                best_val = gd_val
                best_h = gd_h.copy()
                # Replace worst individual with gd_h
                worst_idx = np.argmax(pop_vals)
                population[worst_idx] = gd_h
                pop_vals[worst_idx] = gd_val

    # Final refinement using SLSQP if time remains
    if time.time() < start_time + budget_s - safety*2:
        # prepare constraints as previously
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
        # initial t as best_val
        t0 = best_val
        # A bit of slack (like small buffer)
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            opt_val = compute_c5(h_opt)
            if opt_val < best_val:
                best_val = opt_val
                best_h = h_opt

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return best_h, best_val, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while forcing sum(v) == target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected gradient descent on max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    eps = 1e-12
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        cur_val = max_corr * dx
        max_idxs = np.where(np.abs(corr - max_corr) <= eps)[0]
        grad = np.zeros_like(h)
        for idx in max_idxs:
            k = idx - (n - 1)
            grad += gradient_for_shift(h, k)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05):
            new_h = h - alpha * factor * grad
            new_h = project_to_box_sum(new_h, target_sum)
            new_val = compute_c5(new_h)
            if new_val < cur_val - 1e-12:
                h = new_h
                cur_val = new_val
                improved = True
                break
        if not improved:
            alpha *= 0.5
            if alpha < min_alpha:
                h = project_to_box_sum(
                    np.clip(h + rng.normal(scale=0.01, size=n), 0.0, 1.0),
                    target_sum,
                )
                alpha = 0.2
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
    return best_h, best_val

class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[: self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[: self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[: self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[: self.n - kk])
        return x[-1] - overlap * self.dx

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=500):
    """SLSQP refinement with slack variable t."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{"type": "eq", "fun": sum_con}]
    constraints += [{"type": "ineq", "fun": oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Dimension & initial feasible point -----
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- Population for Differential Evolution -----
    popsize = max(10, n // 8)  # at least 10 individuals
    population = [best_h]
    while len(population) < popsize:
        cand = rng.random(n)
        cand = project_to_box_sum(cand, target_sum)
        population.append(cand)
    pop_vals = [compute_c5(ind) for ind in population]
    for ind, val in zip(population, pop_vals):
        if val < best_val:
            best_val = val
            best_h = ind.copy()

    # ----- DE parameters -----
    F = 0.8
    CR = 0.9
    safety = 5.0  # seconds reserved for final steps

    # ----- Main DE loop -----
    while time.time() < start_time + budget_s - safety:
        indices = list(range(popsize))
        rng.shuffle(indices)
        for i in indices:
            # pick three distinct donors different from i
            candidates = [j for j in range(popsize) if j != i]
            a, b, c = rng.choice(candidates, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            # binomial crossover
            trial = population[i].copy()
            jrand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)

            trial_val = compute_c5(trial)
            if trial_val < pop_vals[i]:
                population[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                if best_val < 0.3808:
                    break
        if best_val < 0.3808:
            break

        # occasional local refinement on the current best
        remaining = start_time + budget_s - safety - time.time()
        if remaining > 10.0:
            gd_h, gd_val = gradient_descent(
                best_h, target_sum, max_iter=600, time_limit=start_time + budget_s - safety, rng=rng
            )
            if gd_val < best_val:
                best_val = gd_val
                best_h = gd_h.copy()
                # replace worst individual with the refined one
                worst_idx = int(np.argmax(pop_vals))
                population[worst_idx] = gd_h
                pop_vals[worst_idx] = gd_val

    # ----- Final SLSQP refinement if time permits -----
    if time.time() < start_time + budget_s - safety * 2:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        res = slsqp_optimize(best_h, best_val, sum_con, overlap_cons, maxiter=500)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            opt_val = compute_c5(h_opt)
            if opt_val < best_val:
                best_val = opt_val
                best_h = h_opt

    # ----- Clean up and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val, n
