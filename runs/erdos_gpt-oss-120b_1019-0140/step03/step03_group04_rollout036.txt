def apply_swap_corr(corr, h, i, j):
    n = len(h)
    # define d_i = -1, d_j = +1
    di = -1.0
    dj = 1.0
    # compute shift indices t0 = 0..2n-2
    t = np.arange(2*n-1)
    s = t - (n-1)

    # term A:
    # for i: idx = i + s = i + (t - (n-1)) = t + (i - (n-1))
    # valid where 0 <= idx < n
    idx_i = t + (i - (n-1))
    term_A = np.zeros_like(corr)
    # mask_i
    mask_i = (idx_i >= 0) & (idx_i < n)
    term_A[mask_i] = di * (1.0 - h[idx_i[mask_i]])
    # j term
    idx_j = t + (j - (n-1))
    mask_j = (idx_j >= 0) & (idx_j < n)
    term_A[mask_j] += dj * (1.0 - h[idx_j[mask_j]])

    # term B:
    # for i: idx = i - s = i - (t - (n-1)) = (i + (n-1)) - t
    idx_iB = (i + (n-1)) - t
    term_B = np.zeros_like(corr)
    mask_iB = (idx_iB >= 0) & (idx_iB < n)
    # term_B contribution: -h[idx_iB] * di
    term_B[mask_iB] -= h[idx_iB[mask_iB]] * di
    # j term:
    idx_jB = (j + (n-1)) - t
    mask_jB = (idx_jB >= 0) & (idx_jB < n)
    term_B[mask_jB] -= h[idx_jB[mask_jB]] * dj

    # term C:
    term_C = np.zeros_like(corr)
    # shift s = i - j corresponds to t = (i - j) + (n-1)
    t_ij = i - j + (n-1)
    if 0 <= t_ij < len(corr):
        term_C[t_ij] -= di * dj
    t_ji = j - i + (n-1)
    if 0 <= t_ji < len(corr):
        term_C[t_ji] -= dj * di  # same as above, but include both
    # delta:
    delta = term_A + term_B + term_C
    return corr + delta

def apply_swap_corr(corr, h, i, j):
    n = h.size
    lo = -1.0
    hi = 1.0
    di = -1.0  # because swapping 1->0 at i
    dj = 1.0   # swapping 0->1 at j
    # t indices 0 .. 2n-2
    t = np.arange(2*n-1)
    # term A
    idx_i = t + (i - (n-1))
    term_A = np.zeros_like(corr)
    mask_i = (idx_i >= 0) & (idx_i < n)
    term_A[mask_i] = di * (1.0 - h[idx_i[mask_i]])
    idx_j = t + (j - (n-1))
    mask_j = (idx_j >= 0) & (idx_j < n)
    term_A[mask_j] += dj * (1.0 - h[idx_j[mask_j]])
    # term B
    idx_iB = (i + (n-1)) - t
    term_B = np.zeros_like(corr)
    mask_iB = (idx_iB >= 0) & (idx_iB < n)
    term_B[mask_iB] -= h[idx_iB[mask_iB]] * di
    idx_jB = (j + (n-1)) - t
    mask_jB = (idx_jB >= 0) & (idx_jB < n)
    term_B[mask_jB] -= h[idx_jB[mask_jB]] * dj
    # term C
    term_C = np.zeros_like(corr)
    t_ij = i - j + (n-1)
    if 0 <= t_ij < corr.size:
        term_C[t_ij] -= di * dj
    t_ji = j - i + (n-1)
    if 0 <= t_ji < corr.size:
        term_C[t_ji] -= dj * di
    delta = term_A + term_B + term_C
    return corr + delta

import time
import numpy as np

def compute_c5(h):
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def apply_swap_corr(corr, h, i, j):
    n = h.size
    di, dj = -1.0, 1.0
    t = np.arange(2 * n - 1)
    idx_i = t + (i - (n - 1))
    idx_j = t + (j - (n - 1))
    term_A = np.zeros_like(corr)
    mask_i = (idx_i >= 0) & (idx_i < n)
    term_A[mask_i] = di * (1.0 - h[idx_i[mask_i]])
    mask_j = (idx_j >= 0) & (idx_j < n)
    term_A[mask_j] += dj * (1.0 - h[idx_j[mask_j]])
    idx_iB = (i + (n - 1)) - t
    idx_jB = (j + (n - 1)) - t
    term_B = np.zeros_like(corr)
    mask_iB = (idx_iB >= 0) & (idx_iB < n)
    term_B[mask_iB] -= h[idx_iB[mask_iB]] * di
    mask_jB = (idx_jB >= 0) & (idx_jB < n)
    term_B[mask_jB] -= h[idx_jB[mask_jB]] * dj
    term_C = np.zeros_like(corr)
    t_ij = i - j + (n - 1)
    if 0 <= t_ij < corr.size:
        term_C[t_ij] -= di * dj
    t_ji = j - i + (n - 1)
    if 0 <= t_ji < corr.size:
        term_C[t_ji] -= dj * di
    return corr + term_A + term_B + term_C

def binarize(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def hill_climb_binary(h, time_end):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_score = np.max(corr) * (2.0 / n)
    rng = np.random.default_rng()
    while time.time() < time_end:
        i = rng.integers(n)
        if h[i] < 0.5:
            continue
        j = rng.integers(n)
        if h[j] > 0.5:
            continue
        new_corr = apply_swap_corr(corr, h, i, j)
        new_score = np.max(new_corr) * (2.0 / n)
        if new_score < best_score:
            h[i], h[j] = h[j], h[i]
            corr = new_corr
            best_h = h.copy()
            best_score = new_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 64
    w = n // 2
    t_target = int(np.floor(0.3808 * n / 2))

    # ----- integer programming attempt -----
    try:
        import cvxpy as cp
        solvers = cp.installed_solvers()
        solver_name = None
        for cand in ('CBC', 'GLPK_MI', 'ECOS_BB'):
            if cand in solvers:
                solver_name = cand
                break
        if solver_name is not None:
            x = cp.Variable(n, integer=True)
            constraints = [x >= 0, x <= 1, cp.sum(x) == w]
            t = cp.Variable(integer=True)
            constraints += [t >= 0, t <= w]
            for k in range(1, n):
                m = n - k
                idx1 = np.arange(m)
                idx2 = idx1 + k
                y = cp.Variable(m, integer=True)
                constraints += [y >= 0, y <= 1,
                                 y <= x[idx1],
                                 y <= x[idx2],
                                 y >= x[idx1] + x[idx2] - 1]
                O_k = cp.sum(x[idx1]) - cp.sum(y)
                constraints += [O_k <= t]
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                t_opt = int(round(t.value))
                if t_opt <= t_target:
                    x_opt = np.rint(x.value).astype(float)
                    c5 = compute_c5(x_opt)
                    return x_opt, c5, n
    except Exception:
        pass

    # ----- fallback: random search + hill climbing -----
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = np.resize(init, n)
        h0 = binarize(init)
    else:
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=w, replace=False)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # random sampling phase
    while time_left() > 2.0:
        cand = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=w, replace=False)
        cand[idx] = 1.0
        sc = compute_c5(cand)
        if sc < best_score:
            best_h, best_score = cand, sc
        if best_score <= 0.3808:
            break

    # hill climbing refinement
    if time_left() > 0.5:
        end_time = start + budget_s - 0.05
        refined_h, refined_sc = hill_climb_binary(best_h.copy(), end_time)
        if refined_sc < best_score:
            best_h, best_score = refined_h, refined_sc

    best_h = np.rint(best_h).astype(float)  # ensure binary
    best_score = compute_c5(best_h)
    return best_h, best_score, n
