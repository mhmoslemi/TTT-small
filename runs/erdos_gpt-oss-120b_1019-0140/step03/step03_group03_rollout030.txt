import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear up‑sample from len(arr) to new_n (periodic domain)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear down‑sample from len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (balanced for powers of two)."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Return a balanced 0/1 vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term w.r.t. h using the given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate_small(h0, max_iter=400):
    """SLSQP refinement for a small‑scale candidate (n≈64)."""
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def to_binary(v, k):
    """Return a balanced binary vector by selecting the k largest entries of v."""
    idx = np.argpartition(v, -k)[-k:]
    bin_vec = np.zeros_like(v, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec


def anneal_binary_custom(h0, max_iter, temp_start, temp_end, time_left, rng):
    """Simulated annealing on a binary balanced vector."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_local = h.copy()
    best_local_score = cur_score
    n = h.size

    for it in range(max_iter):
        if time_left() <= 0.0:
            break
        # exponential cooling
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = (delta < 0) or (rng.random() < np.exp(-delta / (t + 1e-12)))
        if accept:
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

    return best_local, best_local_score

    def time_left():
