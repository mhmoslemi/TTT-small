import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binary_from_real(arr):
    """Convert a real‑valued array to a balanced binary vector by top‑half threshold."""
    n = arr.shape[0]
    half = n // 2
    idx = np.argpartition(arr, -half)[-half:]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def hill_climb_binary(h0, max_iters, rng, time_left):
    """Greedy hill‑climb on balanced binary vectors."""
    h = h0.copy()
    score = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        if new_score < score:
            score = new_score
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # target number of samples (balance speed vs resolution)
    n_target = 256
    half_target = n_target // 2

    best_h = None
    best_score = np.inf
    target_score = 0.3808

    # --- use the provided construction as a seed (if any) ---
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.shape[0] != n_target:
            base = upsample(base, n_target)
        base_bin = binary_from_real(base)
        # modest hill‑climb on this seed
        max_iters_seed = 2500
        h_seed, sc_seed = hill_climb_binary(base_bin, max_iters_seed, rng, time_left)
        if sc_seed < best_score:
            best_score, best_h = sc_seed, h_seed.copy()
        if best_score <= target_score:
            best_h = project_onto_box_sum(best_h, half_target)
            return best_h, best_score, n_target

    # --- random binary hill‑climb restarts ---
    while time_left() > 200:  # preserve time for refinement
        max_iters = 1500
        h0 = random_binary_balanced(n_target, rng)
        h_opt, sc_opt = hill_climb_binary(h0, max_iters, rng, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target_score:
            break

    # fallback in case nothing was generated
    if best_h is None:
        best_h = random_binary_balanced(n_target, rng)
        best_score = compute_c5(best_h)

    # --- continuous refinement on the best binary candidate ---
    if time_left() > 5:
        # allocate generous iterations while respecting budget
        max_iters_refine = int(min(150000, max(40000, time_left() * 45)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters_refine, step0=0.2, beta=0.9,
            rng=rng, time_left=time_left)

    # final safety projection and score recomputation
    best_h = project_onto_box_sum(best_h, half_target)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_target
