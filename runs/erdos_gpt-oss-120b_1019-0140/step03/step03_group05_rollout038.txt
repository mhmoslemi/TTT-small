    start = time.time()
    rng = np.random.default_rng(seed)

    n = 100
    w = n // 2
    target_corr = int(np.floor(0.3808 * w))

    # define compute_max_corr
    ...

    best_h = None
    best_corr = None

    # We'll loop until time budget
    while time.time() - start < budget_s * 0.95:
        # initialize random h
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=w, replace=False)
        h[ones_idx] = 1.0
        cur_corr = compute_max_corr(h)

        # if current > best, maybe start with best
        # but we can also start from best_h if exists
        if best_h is not None:
            # with probability 0.2 start from best instead of random
            if rng.random() < 0.2:
                h = best_h.copy()
                cur_corr = best_corr

        # iter loop
        iter_no_improve = 0
        iter_total = 0
        while time.time() - start < budget_s * 0.95:
            iter_total += 1
            # propose swap
            # find indices of ones and zeros
            idx_one = np.where(h == 1)[0]
            idx_zero = np.where(h == 0)[0]
            if idx_one.size == 0 or idx_zero.size == 0:
                break
            a = rng.choice(idx_one)
            b = rng.choice(idx_zero)
            # make swap
            h[a] = 0.0
            h[b] = 1.0
            new_corr = compute_max_corr(h)

            # acceptance
            delta = new_corr - cur_corr
            # compute temperature
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.95)
            T = 2.0 * max(0.0, 1 - frac)  # linear cooling to 0
            accept = False
            if delta <= 0:
                accept = True
            else:
                if T > 0:
                    prob = np.exp(-delta / T)
                    if rng.random() < prob:
                        accept = True
            if accept:
                cur_corr = new_corr
                iter_no_improve = 0
                if cur_corr < best_corr if best_corr is not None else True:
                    best_corr = cur_corr
                    best_h = h.copy()
                    # if meets target
                    if best_corr <= target_corr:
                        # compute final bound and return
                        c5 = best_corr * (2.0 / n)
                        return best_h, c5, n
            else:
                # revert swap
                h[a] = 1.0
                h[b] = 0.0
                iter_no_improve += 1
            # stagnation handling
            if iter_no_improve >= 5000:
                # reinitialize random h (or perturb)
                break

        # after inner loop, maybe continue to next outer (new start)
    # after time limit
    if best_h is None:
        # fallback to simple random binary vector
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=w, replace=False)] = 1.0
        best_h = h
        best_corr = compute_max_corr(h)

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # problem size (feel free to change, but keep reasonable)
    n = 80                     # number of discretisation points
    w = n // 2                 # required sum of h (since ∫h=1 → sum = n/2)
    target_corr = int(np.floor(0.3808 * w))   # max allowed overlap count
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def max_overlap(vec):
        """Maximum value of ∑ h_i (1‑h_{i+k}) over all integer shifts k."""
        # correlation of h with (1‑h) gives exactly the needed sums
        return int(np.max(np.correlate(vec, 1.0 - vec, mode="full")))

    def c5_from_corr(corr):
        return corr * dx

    # ------------------------------------------------------------------
    # Starting points
    # ------------------------------------------------------------------
    def binary_from_continuous(cont):
        """Turn a continuous vector into a binary one with exactly w ones."""
        idx = np.argsort(-cont)[:w]
        hb = np.zeros_like(cont, dtype=float)
        hb[idx] = 1.0
        return hb

    start_vectors = []

    # 1) if a parent construction exists, use its binary projection
    if "initial_h_values" in globals():
        iv = np.asarray(initial_h_values, dtype=float)
        if iv.shape[0] == n:
            start_vectors.append(binary_from_continuous(iv))

    # 2) a few random binary vectors for diversity
    for _ in range(3):
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, size=w, replace=False)] = 1.0
        start_vectors.append(h0)

    # ------------------------------------------------------------------
    # Simulated‑Annealing search
    # ------------------------------------------------------------------
    best_h = None
    best_corr = None
    T0 = 8.0                    # initial temperature
    max_no_improve = 3000       # stagnation limit before restart

    # main restart loop
    while time.time() - start_time < budget_s * 0.95:
        # pick a starting vector (biased towards the current best)
        if best_h is not None and rng.random() < 0.2:
            h = best_h.copy()
        else:
            h = rng.choice(start_vectors)

        # maintain explicit lists of indices for fast swaps
        ones = [int(i) for i in np.where(h == 1.0)[0]]
        zeros = [int(i) for i in np.where(h == 0.0)[0]]
        cur_corr = max_overlap(h)
        no_improve = 0

        # inner SA loop
        while time.time() - start_time < budget_s * 0.95:
            # temperature decays linearly with elapsed time
            elapsed = time.time() - start_time
            frac = elapsed / (budget_s * 0.95)
            T = T0 * max(0.0, 1.0 - frac)

            # propose a swap of a 1 and a 0
            a = rng.choice(ones)
            b = rng.choice(zeros)
            h[a] = 0.0
            h[b] = 1.0

            new_corr = max_overlap(h)
            delta = new_corr - cur_corr

            accept = False
            if delta <= 0:
                accept = True
            elif T > 0.0:
                if rng.random() < np.exp(-delta / T):
                    accept = True

            if accept:
                # update index lists
                ones.remove(a)
                zeros.append(a)
                zeros.remove(b)
                ones.append(b)

                cur_corr = new_corr
                no_improve = 0

                # record global best
                if best_corr is None or cur_corr < best_corr:
                    best_corr = cur_corr
                    best_h = h.copy()
                    if best_corr <= target_corr:
                        return best_h, c5_from_corr(best_corr), n
            else:
                # revert the swap
                h[a] = 1.0
                h[b] = 0.0
                no_improve += 1

            # stagnation handling
            if no_improve >= max_no_improve:
                break

    # ------------------------------------------------------------------
    # Fallback in case time ran out without hitting the target
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[rng.choice(n, size=w, replace=False)] = 1.0
        best_corr = max_overlap(best_h)

    return best_h, c5_from_corr(best_corr), n
