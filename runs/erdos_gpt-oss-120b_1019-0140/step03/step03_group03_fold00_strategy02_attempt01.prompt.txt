def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:

    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]

            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()

def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP to minimize max correlation on a small scale."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on binary balanced vectors."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # pre‑compute index sets
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        # linear cooling in log‑temperature space
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:

            if new_score < best_score:

            # update index lists

            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score
