def enforce_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    # Early exit
    if abs(total - target) < 1e-12:
        return v
    # Scale
    if total > 0:
        scale = target / total
        v = v * scale
    # Clip again
    v = np.clip(v, 0.0, 1.0)
    # Distribute residual
    diff = target - v.sum()
    # while diff not small
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            # increase some element not at 1
            idx = np.where(v < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            i = idx[np.random.randint(len(idx))]
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
        else:
            idx = np.where(v > 1e-12)[0]
            if len(idx) == 0:
                break
            i = idx[np.random.randint(len(idx))]
            dec = min(-diff, v[i])
            v[i] -= dec
        diff = target - v.sum()
        attempts += 1
    return v

def refine_gradient(h, target, dx, deadline, rng):
    # loop until time limit or convergence
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    while time.time() < deadline - 0.05:
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = np.argmax(corr)
        k = idx_max - (len(cur_h) - 1)
        # compute term vector
        n = len(cur_h)
        term = np.zeros(n)
        if k >= 0:
            term[:n - k] += cur_h[k:]
            term[k:] += cur_h[:n - k]
        else:
            kk = -k
            term[kk:] += cur_h[:n - kk]
            term[:n - kk] += cur_h[kk:]
        grad = dx * (1.0 - term)  # gradient of C5 at worst shift
        d = -grad
        # make d sum zero
        d = d - np.mean(d)
        # compute max step
        pos = d > 0
        neg = d < 0
        # avoid division by zero
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min((1.0 - cur_h[pos]) / d[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min(cur_h[neg] / -d[neg]))
        if max_step == np.inf or max_step <= 0:
            break
        # step with factor
        alpha = max_step * 0.5  # cautious
        # optional line search:
        new_h = cur_h + alpha * d
        new_val = compute_c5(new_h)
        # if improvement, accept, else reduce alpha
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
        else:
            # try smaller alpha
            factor = 0.5
            improved = False
            while factor > 1e-4:
                alpha2 = max_step * factor * 0.5
                new_h2 = cur_h + alpha2 * d
                new_val2 = compute_c5(new_h2)
                if new_val2 < cur_val:
                    cur_h = new_h2
                    cur_val = new_val2
                    improved = True
                    break
                factor *= 0.5
            if not improved:
                # can't improve further
                break
        # early stop if good enough
        if cur_val < 0.3808:
            break
    return cur_h, cur_val

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust so that sum(v) == target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v *= target / total
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        attempts += 1
    return v

def random_binary_vector(n, target, rng):
    """Binary vector of length n with exactly target ones (target integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v

def gradient_refine(h, target, dx, deadline, rng):
    """Projected gradient descent on the worst shift."""
    cur = h.copy()
    cur_val = compute_c5(cur)
    n = len(cur)
    while time.time() < deadline - 0.05:
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)

        term = np.zeros(n)
        if k >= 0:
            term[: n - k] += cur[k:]
            term[k:]      += cur[: n - k]
        else:
            kk = -k
            term[kk:]     += cur[: n - kk]
            term[: n - kk] += cur[kk:]

        grad = dx * (1.0 - term)          # ∇C5 at worst shift
        d = -grad                         # direction to reduce C5
        d -= np.mean(d)                   # keep sum unchanged

        # maximal feasible step size
        pos = d > 0
        neg = d < 0
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min((1.0 - cur[pos]) / d[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min(cur[neg] / -d[neg]))
        if not np.isfinite(max_step) or max_step <= 0:
            break

        alpha = 0.5 * max_step
        new = cur + alpha * d
        new_val = compute_c5(new)

        if new_val < cur_val:
            cur, cur_val = new, new_val
            if cur_val < 0.3808:
                break
            continue

        # backtrack if no improvement
        improved = False
        factor = 0.5
        while factor > 1e-4:
            alpha2 = 0.5 * max_step * factor
            new2 = cur + alpha2 * d
            new_val2 = compute_c5(new2)
            if new_val2 < cur_val:
                cur, cur_val = new2, new_val2
                improved = True
                if cur_val < 0.3808:
                    break
                break
            factor *= 0.5
        if not improved:
            break
    return cur, cur_val

def hill_climb(start_h, target, dx, deadline, rng):
    """Simple greedy hill climbing on binary vectors."""
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val
    stagnation = 0
    stagnation_limit = 500

    while time.time() < deadline - 0.05:
        ones = np.flatnonzero(cur > 0.5)
        zeros = np.flatnonzero(cur < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i], cur[j] = cur[j], cur[i]   # swap
        new_val = compute_c5(cur)

        if new_val < cur_val - 1e-12:
            cur_val = new_val
            stagnation = 0
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
                if best_val < 0.3808:
                    return best_h, best_val, True
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert
            stagnation += 1
            if stagnation >= stagnation_limit:
                break
    return best_h, best_val, False

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_binary_vector(n, n / 2.0, rng)

    target = n / 2.0
    dx = 2.0 / n

    # Ensure feasibility
    h0 = enforce_sum(h0, target, rng)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Early gradient refinement on the initial candidate
    best_h, best_val = gradient_refine(best_h, target, dx, deadline, rng)

    # Main optimization loop
    while time.time() < deadline - 0.5:
        # Choose a starting point
        if rng.random() < 0.6:
            start_h = best_h.copy()
            swaps = rng.integers(1, 5)
            for _ in range(swaps):
                ones = np.flatnonzero(start_h > 0.5)
                zeros = np.flatnonzero(start_h < 0.5)
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                start_h[i], start_h[j] = start_h[j], start_h[i]
        else:
            start_h = random_binary_vector(n, target, rng)

        # Hill climbing
        local_h, local_val, finished = hill_climb(start_h, target, dx, deadline, rng)
        if local_val < best_val:
            best_h, best_val = local_h, local_val
            # Optional gradient refinement after improvement
            best_h, best_val = gradient_refine(best_h, target, dx, deadline, rng)
            if best_val < 0.3808:
                break
        if finished:
            break

    # Final brief refinement
    if time.time() < deadline - 0.1:
        best_h, best_val, _ = hill_climb(best_h, target, dx, deadline, rng)

    # Ensure exact feasibility before returning
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
