    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p**0.5)
    for i in range(3, r+1, 2):
        if p % i == 0:
            return False
    return True

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p ** 0.5)
    for i in range(3, r + 1, 2):
        if p % i == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for quadratic residue, p-1 otherwise
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Use parent construction if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_h, best_c5, best_n = h0.copy(), c5_0, len(h0)

    # ----- exhaustive Legendre search (binary) -----
    max_n = 200
    primes = [p for p in range(3, max_n) if p % 2 == 1 and is_prime(p)]
    for p in primes:
        if time.time() - start > budget_s - 3:
            break
        n = p + 1
        M = n // 2
        leg = legendre_sequence(p)                    # length p, sum = (p-1)/2 = M-1
        seq = np.concatenate([leg, np.array([1.0])])   # balanced binary of length n
        for base_seq in (seq, 1.0 - seq):
            for shift in range(n):
                if time.time() - start > budget_s - 3:
                    break
                h = np.roll(base_seq, shift)
                c5 = compute_c5(h)
                if c5 + 1e-12 < best_c5:
                    best_h, best_c5, best_n = h.copy(), c5, n
                    if best_c5 < 0.3808:
                        break
            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # ----- simple binary hill climbing on a few sizes -----
    # (useful if Legendre did not hit the target)
    for n in (80, 96, 112, 128):
        if time.time() - start > budget_s - 3:
            break
        M = n // 2
        # random balanced start
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, M, replace=False)
        h[ones_idx] = 1.0
        cur_c5 = compute_c5(h)
        if cur_c5 + 1e-12 < best_c5:
            best_h, best_c5, best_n = h.copy(), cur_c5, n
        ones = [int(i) for i in np.where(h == 1)[0]]
        zeros = [int(i) for i in np.where(h == 0)[0]]
        for _ in range(20000):
            if time.time() - start > budget_s - 3:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # swap
            h[i], h[j] = 0.0, 1.0
            new_c5 = compute_c5(h)
            if new_c5 + 1e-12 < cur_c5:
                cur_c5 = new_c5
                ones.remove(i)
                zeros.append(i)
                zeros.remove(j)
                ones.append(j)
                if new_c5 + 1e-12 < best_c5:
                    best_h, best_c5, best_n = h.copy(), new_c5, n
                    if best_c5 < 0.3808:
                        break
            else:
                # revert
                h[i], h[j] = 1.0, 0.0
        if best_c5 < 0.3808:
            break

    # ----- fractional hill climbing from the best binary ----------
    if best_h is not None:
        n = best_n
        h_cont = best_h.copy()
        cur_c5 = best_c5
        while time.time() - start < budget_s - 5:
            i_candidates = np.where(h_cont > 1e-12)[0]
            j_candidates = np.where(h_cont < 1.0 - 1e-12)[0]
            if i_candidates.size == 0 or j_candidates.size == 0:
                break
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            max_transfer = min(h_cont[i], 1.0 - h_cont[j])
            if max_transfer < 1e-12:
                continue
            delta = rng.random() * max_transfer
            h_cont[i] -= delta
            h_cont[j] += delta
            new_c5 = compute_c5(h_cont)
            if new_c5 + 1e-12 < cur_c5:
                cur_c5 = new_c5
                if new_c5 + 1e-12 < best_c5:
                    best_h, best_c5, best_n = h_cont.copy(), new_c5, n
                    if best_c5 < 0.3808:
                        break
            else:
                h_cont[i] += delta
                h_cont[j] -= delta

    # ----- SLSQP refinement (continuous) -----
    if best_h is not None and time.time() - start < budget_s - 5:
        n = best_n
        M = n // 2
        dx = 2.0 / n

        def obj(x):
            return x[-1]

        # equality constraint: sum(h) = M
        cons = [{'type': 'eq', 'fun': lambda x, M=M, n=n: np.sum(x[:n]) - M}]

        # inequality constraints: t >= overlap(k) for all k != 0
        for k in range(-n + 1, n):
            if k == 0:
                continue

            def con_factory(k):
                def con(x, k=k):
                    h = x[:n]
                    if k >= 0:
                        overl = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                    return x[-1] - overl * dx
                return con

            cons.append({'type': 'ineq', 'fun': con_factory(k)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_c5]])

        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000,
                                                      'ftol': 1e-12,
                                                      'disp': False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                cur = np.sum(h_opt)
                if abs(cur - M) > 1e-9:
                    h_opt += (M - cur) / n
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt + 1e-12 < best_c5:
                    best_h, best_c5, best_n = h_opt, c5_opt, n
        except Exception:
            pass

    # ----- final safety and normalization -----
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = best_h.sum()
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, n
