def adam_optimize(h0, n_iter, step0, alpha_start, alpha_end, beta1=0.9, beta2=0.999, eps=1e-8, time_left):

   h = h0.copy()
   m = np.zeros_like(h)
   v = np.zeros_like(h)
   s = h.size/2

   best_h = h.copy()
   best_score = compute_c5(h)

   for it in range(1, n_iter+1):
       if time_left() < 0.01: break
       alpha = alpha_start + (alpha_end-alpha_start)*(it-1)/(max(1,n_iter-1))
       grad = grad_softmax_corr(h, alpha)
       m = beta1*m + (1-beta1)*grad
       v = beta2*v + (1-beta2)*(grad**2)
       m_hat = m/(1-beta1**it)
       v_hat = v/(1-beta2**it)
       eta = step0 / (np.sqrt(v_hat) + eps)
       h_new = h - eta * m_hat
       h_new = project_onto_box_sum(h_new, s)
       score = compute_c5(h_new)
       if score < best_score:
           best_score, best_h = score, h_new.copy()
       h = h_new

   return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()  # binary 0/1
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score

    max_iters = int(min(500000, max(1, time_left() * 2000))) # adapt to time left

    T0 = 0.01  # initial temperature maybe higher
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # choose random swap 0 <->1
        # sample i from zeros positions and j from ones; to do efficiently, maintain arrays?
        # Simpler: get random i, random j, accept if they differ; else continue.
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # swap
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # swap between 0 and 1
        h_new[j] = 1 - h_new[j]
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it/max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds bracket the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation (preserving endpoint periodicity is not required)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    """Binary Thue–Morse sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def grad_softmax_corr(h, alpha):
    """Gradient of softmax of correlation w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def adam_optimize(h0, n_iter, step0=0.5, alpha_start=0.5, alpha_end=20.0,
                  beta1=0.9, beta2=0.999, eps=1e-8, time_left=lambda: 1e9):
    """Adam optimizer on the soft‑max approximation of max‑correlation."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        # schedule for smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        eta = step0 / (np.sqrt(v_hat) + eps)
        h_new = h - eta * m_hat
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill climb on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(500000, max(1, time_left() * 2000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        h_new = h.copy()
        h_new[i] = 1.0 - h_new[i]
        h_new[j] = 1.0 - h_new[j]
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if new_score < curr_score or (T > 0 and rng.random() < np.exp(-delta / T)):
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score
