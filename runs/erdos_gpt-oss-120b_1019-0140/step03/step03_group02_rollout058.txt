def momentum_subgradient_descent(h0, step0, beta, max_iters, time_left):

   h = h0.copy(); s = h.size / 2; best_h = h.copy(); best_score = compute_c5(h); mom = np.zeros_like(h)

   For it in range(1,...):

       if time_left() < 0.01: break

       corr = np.correlate(h, 1.0 - h, mode='full'); maxc = np.max(corr); tolerance = 1e-12

       # find shifts achieving maxc within tolerance

       idxes = np.where(corr >= maxc - tolerance)[0]; shifts = idxes - (h.size - 1)

       grad = compute_gradients_for_shifts(h, shifts)   # directly compute from h

       mom = beta * mom + (1 - beta) * grad

       eta = step0 / np.sqrt(it)

       h_new = h - eta * mom

       h_new = project_onto_box_sum(h_new, s)

       sc = compute_c5(h_new)

       if sc < best_score:
           best_score, best_h = sc, h_new.copy()

       h = h_new

   return best_h, best_score

```
def momentum_subgradient_descent(h0, step0, beta, max_iters, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # compute corr and max
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        # find shifts that achieve maxc within tolerance
        tol = 1e-12
        idxes = np.where(corr >= maxc - tol)[0]
        shifts = idxes - (h.size - 1)
        # compute subgradient
        grad = compute_gradients_for_shifts(h, shifts)
        # update momentum
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # choose an index with h=0
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score
```

def round_to_binary(v):
    n = v.size
    sorted_idx = np.argsort(v)  # ascending
    binary = np.zeros_like(v, dtype=float)
    binary[sorted_idx[n//2:]] = 1.0
    return binary
```

Now upsample function:

```
def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def random_continuous_balanced(n, rng):
    """Uniform random vector projected onto the box-sum constraint."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def sinusoid_initial(n, rng):
    """Sinusoidal pattern with random phase, then projected to exact sum."""
    phi = rng.random() * 2 * np.pi
    arr = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n + phi)
    return project_onto_box_sum(arr, n / 2.0)


def upsample(arr, new_n):
    """Linear interpolation to a new size, preserving endpoints as periodic."""
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max_k ∫ h(1‑h_shift) for given maximizing shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[: n - s])
            grad[: n - s] -= h[s:]
    grad /= len(shifts)
    return grad


def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[: n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[: n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[: n - p])
            grad[: n - p] -= w * h[p:]
    return grad


def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def momentum_subgradient_descent(h0, step0=0.5, beta=0.9,
                                 max_iters=20000, time_left=lambda: 1e9):
    """Projected subgradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idxes = np.where(corr >= maxc - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def round_to_binary(v):
    """Project v to a {0,1} vector with exactly half ones."""
    n = v.size
    idx = np.argsort(v)  # ascending
    binary = np.zeros_like(v, dtype=float)
    binary[idx[n // 2:]] = 1.0
    return binary


def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style hill‑climb on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():

        sizes.append(nxt)

    # ----- seed generation for the smallest grid -----
    seeds = []
    if 'initial_h_values' in globals():

        init_small = project_onto_box_sum(init_small, n0 / 2.0)

    seeds.append(thue_morse(n0).astype(float))
    for _ in range(2):

    for _ in range(2):
        seeds.append(random_continuous_balanced(n0, rng))
    for _ in range(2):
        seeds.append(sinusoid_initial(n0, rng))

    best_h = None
    best_score = np.inf

        if time_left() < 2.0:

        h_opt, sc_opt = momentum_subgradient_descent(
            h0, step0=1.0, beta=0.9, max_iters=20000, time_left=time_left)
        if time_left() > 1.0:
            h_opt2, sc_opt2 = smooth_grad_descent(
                h_opt, n_iter=2000, step0=0.5,

                beta=0.9, time_left=time_left)
            if sc_opt2 < sc_opt:
                h_opt, sc_opt = h_opt2, sc_opt2
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n0, 0.5), n0 / 2.0)
        best_score = compute_c5(best_h)

    # ----- multi‑resolution refinement -----
    cur_h = best_h
    for sz in sizes[1:]:
        if time_left() < 2.0:
            break
        cur_h = upsample(cur_h, sz)
        cur_h = project_onto_box_sum(cur_h, sz / 2.0)
        cur_h, cur_score = momentum_subgradient_descent(
            cur_h, step0=0.5, beta=0.9,
            max_iters=30000, time_left=time_left)
        if time_left() > 1.0:
            cur_h, cur_score = smooth_grad_descent(
                cur_h, n_iter=3000, step0=0.5,
                alpha_start=1.0, alpha_end=15.0,

    # ----- binary rounding and hill climbing -----
    if time_left() > 0.5:
        binary_h = round_to_binary(best_h)
        bin_hc, bin_sc = binary_hill_climb_sa(binary_h, rng, time_left, target=0.3808)
        if bin_sc < best_score:

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
