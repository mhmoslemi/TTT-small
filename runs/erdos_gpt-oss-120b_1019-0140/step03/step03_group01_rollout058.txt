def binary_anneal_fft(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.2
    T_min = 1e-6
    decay = 0.99996
    it = 0
    while it < max_iters and time_left_func() > 0.001:
        # randomly choose a one and a zero
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score
```

def compute_c5_fft(h):
    n = h.size
    L = 2*n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)
    # linear convolution result length 2n-1 is conv[:2*n-1]
    max_corr = np.max(conv[:2*n-1])
    return max_corr * (2.0 / n)

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        # compute corr via FFT for speed
        corr = np.fft.ifft(np.fft.fft(h) * np.fft.fft(1.0 - h)).real  # this might be circular conv; need linear
        # Instead compute full corr via np.correlate for correctness (slow)
        # Since gradient uses max_corr shifts, we can compute using np.correlate (n small enough)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral using FFT (returns normalized value)."""
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)
    max_corr = np.max(conv[: 2 * n - 1])
    return max_corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of C5 w.r.t. h for the worst shift(s)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    """Projected momentum gradient descent on the continuous problem."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_anneal_fft(h0, rng, time_left_func, max_iters=500000):
    """Simulated annealing on binary vectors using FFT for evaluation."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.2
    T_min = 1e-6
    decay = 0.99996
    it = 0
    while it < max_iters and time_left_func() > 0.001:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    best_h = None
    best_score = np.inf
    n_large = 512

    # --- Seed 1: upsample initial_h_values if present -----------------
    seed1 = None
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if n_large % init.size == 0:
            factor = n_large // init.size
            up = np.repeat(init, factor)
        else:
            x_old = np.linspace(0, 1, init.size, endpoint=False)
            x_new = np.linspace(0, 1, n_large, endpoint=False)
            up = np.interp(x_new, x_old, init)
        # binarize by taking the largest half values
        k = n_large // 2
        idx = np.argpartition(-up, k)[:k]
        bin_up = np.zeros_like(up)
        bin_up[idx] = 1.0
        seed1 = project_onto_box_sum(bin_up, n_large / 2.0)

    # --- Run annealing on seed1 --------------------------------------
    if seed1 is not None and time_left() > 5.0:
        h_opt, sc_opt = binary_anneal_fft(seed1, rng, time_left, max_iters=600000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Seed 2: random balanced binary --------------------------------
    if time_left() > 5.0:
        rand_bin = np.zeros(n_large, dtype=float)
        idx = rng.choice(n_large, size=n_large // 2, replace=False)
        rand_bin[idx] = 1.0
        rand_bin = project_onto_box_sum(rand_bin, n_large / 2.0)
        h_opt, sc_opt = binary_anneal_fft(rand_bin, rng, time_left, max_iters=600000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Long annealing on the best found ------------------------------
    if best_h is not None and time_left() > 10.0:
        h_opt, sc_opt = binary_anneal_fft(best_h, rng, time_left, max_iters=2000000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Optional continuous refinement ---------------------------------
    if best_h is not None and time_left() > 5.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(
            best_h.astype(float), time_left, step0=0.5, beta=0.9, max_iter=150000
        )
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # --- Final guard ----------------------------------------------------
    if best_h is None:
        best_h = np.full(n_large, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, n_large / 2.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
