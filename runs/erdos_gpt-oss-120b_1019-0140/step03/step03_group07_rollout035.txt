import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            # i from 0 to n-k-1
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def optimise_h(h_initial, n, time_budget, rng):
    M = n // 2
    # Ensure feasibility
    h = np.clip(h_initial.astype(float), 0.0, 1.0)
    cur_sum = np.sum(h)
    if abs(cur_sum - M) > 1e-12:
        h += (M - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    while time.time() - start < time_budget:
        # correlation to find worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        k_max = int(np.argmax(corr) - (n - 1))
        # contributions slice
        if k_max >= 0:
            i_idx = np.arange(0, n - k_max)
            j_idx = i_idx + k_max
        else:
            kk = -k_max
            i_idx = np.arange(kk, n)
            j_idx = i_idx - kk
        # compute contributions for those pairs
        contrib = cur_h[i_idx] * (1.0 - cur_h[j_idx])
        # choose max contribution
        max_pos = np.argmax(contrib)
        i = i_idx[max_pos]
        j = j_idx[max_pos]
        # attempt targeted move
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max > 1e-12 and rng.random() < 0.7:
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-12:
                cur_h = new_h
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                continue
        # random transfer move
        i_rand, j_rand = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            delta_max = min(cur_h[i_rand], 1.0 - cur_h[j_rand])
            if delta_max > 1e-12:
                delta = rng.random() * delta_max
                new_h = cur_h.copy()
                new_h[i_rand] -= delta
                new_h[j_rand] += delta
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5 - 1e-12:
                    cur_h = new_h
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
        else:
            delta_max = min(cur_h[j_rand], 1.0 - cur_h[i_rand])
            if delta_max > 1e-12:
                delta = rng.random() * delta_max
                new_h = cur_h.copy()
                new_h[i_rand] += delta
                new_h[j_rand] -= delta
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5 - 1e-12:
                    cur_h = new_h
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf
    best_n = None
    # candidate resolutions
    candidates = [80, 96, 112, 128, 160]
    # prioritize parent if present
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates
    # allocate time across candidates
    remaining_budget = budget_s - (time.time() - start)
    # simple loop: give each candidate a slice of time proportional
    for idx, n in enumerate(candidates):
        # check remaining time
        elapsed = time.time() - start
        if elapsed > budget_s - 5:
            break
        time_left = budget_s - elapsed
        # allocate at most e.g. 0.2 * remaining for early candidates, later less
        if idx == 0:
            alloc = min(time_left * 0.6, 600.0)
        else:
            alloc = min(time_left * 0.4 / (len(candidates) - idx), 300.0)
        # initialise h0
        if 'initial_h_values' in globals() and n == len(initial_h_values):
            h0 = np.array(initial_h_values, dtype=float)
        elif 'initial_h_values' in globals() and n > len(initial_h_values):
            # upsample via linear interpolation
            old_n = len(initial_h_values)
            x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, initial_h_values)
        else:
            # random binary with half ones
            M = n // 2
            h0 = np.zeros(n)
            idxs = rng.choice(n, M, replace=False)
            h0[idxs] = 1.0
        # run local optimisation
        h_opt, c5_opt = optimise_h(h0, n, alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        if best_c5 < 0.3808:
            break
    # final refinement via SLSQP if time permits
    if best_h is not None and time.time() - start < budget_s - 10:
        n = best_n
        M = n // 2
        # constraints
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        constraints = [{'type': 'eq', 'fun': sum_con}]
        constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([best_h, [best_c5]])
        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                h_ref = np.clip(res.x[:n], 0.0, 1.0)
                cur = np.sum(h_ref)
                if abs(cur - M) > 1e-9:
                    h_ref += (M - cur) / n
                    h_ref = np.clip(h_ref, 0.0, 1.0)
                c5_ref = compute_c5(h_ref)
                if c5_ref < best_c5:
                    best_h = h_ref
                    best_c5 = c5_ref
        except Exception:
            pass
    # safety fallback
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        # final clipping and sum adjustment
        n = best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        cur = np.sum(best_h)
        M = n // 2
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def optimise_h(h_initial, n, time_budget, rng):
    M = n // 2
    h = np.clip(h_initial.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - M) > 1e-12:
        h += (M - s) / n
        h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    while time.time() - start < time_budget:
        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        k_max = int(np.argmax(corr) - (n - 1))
        if k_max >= 0:
            i_idx = np.arange(0, n - k_max)
            j_idx = i_idx + k_max
        else:
            kk = -k_max
            i_idx = np.arange(kk, n)
            j_idx = i_idx - kk
        contrib = cur_h[i_idx] * (1.0 - cur_h[j_idx])
        pos = np.argmax(contrib)
        i = i_idx[pos]
        j = j_idx[pos]
        # temperature for occasional uphill moves
        elapsed = time.time() - start
        T = max(1e-6, (time_budget - elapsed) / time_budget * 0.01)
        # targeted transfer
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max > 1e-12 and rng.random() < 0.7:
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-12 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                cur_h, cur_c5 = new_h, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, new_h.copy()
                continue
        # random transfer
        i_rand, j_rand = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            delta_max = min(cur_h[i_rand], 1.0 - cur_h[j_rand])
            if delta_max > 1e-12:
                delta = rng.random() * delta_max
                new_h = cur_h.copy()
                new_h[i_rand] -= delta
                new_h[j_rand] += delta
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5 - 1e-12 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                    cur_h, cur_c5 = new_h, new_c5
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, new_h.copy()
        else:
            delta_max = min(cur_h[j_rand], 1.0 - cur_h[i_rand])
            if delta_max > 1e-12:
                delta = rng.random() * delta_max
                new_h = cur_h.copy()
                new_h[j_rand] -= delta
                new_h[i_rand] += delta
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5 - 1e-12 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                    cur_h, cur_c5 = new_h, new_c5
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, new_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf
    best_n = None
    candidates = [80, 96, 112, 128, 160]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates
    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:

        else:

        if 'initial_h_values' in globals() and n == len(initial_h_values):
            h0 = np.array(initial_h_values, dtype=float)
        elif 'initial_h_values' in globals() and n > len(initial_h_values):

            x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, initial_h_values)
        else:
            M = n // 2
            h0 = np.zeros(n)
            h0[rng.choice(n, M, replace=False)] = 1.0

    # final SLSQP refinement if time permits
    if best_h is not None and time.time() - start < budget_s - 10:
        n = best_n

        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        constraints = [{'type': 'eq', 'fun': sum_con}]
        constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]

    return best_h, best_c5, n
