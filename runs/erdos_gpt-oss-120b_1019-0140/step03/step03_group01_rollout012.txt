def binary_swap_anneal(h0, rng, time_left, max_iters=200000):

    h = h0.copy()
    n = h.shape[0]
    current_score = compute_c5_fast(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.2
    T_min = 1e-5
    decay = 0.99998
    it = 0

    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0; h[j] = 1.0
        new_score = compute_c5_fast(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0; h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

        v = project_onto_box_sum(v, target_sum)
        h_tmp, s_tmp = gradient_descent_momentum(v, time_left, max_iter=5000)
        if s_tmp < best_s:
            best_s, best_h = s_tmp, h_tmp.copy()
        if best_s <= target: break
    best_h = project_onto_box_sum(best_h, target_sum)
    best_s = compute_c5_fast(best_h)
    return best_h, best_s, n
