def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation to get max shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # indices where within tolerance
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # add jitter occasionally
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def binary_hill_climb(n, time_left, rng):
    # start with random balanced binary
    h = random_binary_balanced(n, rng)
    best_h = h.copy()
    best_score = compute_c5(h)
    # maintain sets
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # loop
    while time_left() > 0.5:
        # randomly pick a 1 and 0
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        # recompute score
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
            # update ones/zeros lists (we can recompute)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # optionally random restart
        if rng.random() < 0.001:
            h = random_binary_balanced(n, rng)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones) as float array."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(n, time_left, rng):
    """Hill climbing on balanced binary vectors."""
    # start from a random balanced binary vector
    h = random_binary_balanced(n, rng)
    best_h = h.copy()
    best_score = compute_c5(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    while time_left() > 0.5:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
        else:
            h[i] = 1.0
            h[j] = 0.0
        if rng.random() < 0.001:
            h = random_binary_balanced(n, rng)
            best_score = compute_c5(h)
            best_h = h.copy()
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf
    target_c5 = 0.3808

    for idx, n in enumerate(scales):
        if time_left() <= 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        target_sum = n / 2.0
        base = project_onto_box_sum(base, target_sum)

        # --- differential evolution (skip for largest scale to save time) ---
        if n <= 256 and time_left() > 15:
            de_attempts = 2
            for _ in range(de_attempts):
                if time_left() <= 5:

                maxiter = max(5, int(min(80, time_left() * 2)))
                de_res = opt.differential_evolution(
                    lambda x: compute_c5(project_onto_box_sum(x, target_sum)),
                    bounds=[(0.0, 1.0)] * n,
                    strategy='best1bin',
                    maxiter=maxiter,
                    popsize=10,
                    init='latinhypercube',
                    seed=rng,
                    callback=lambda xk, convergence: time_left() <= 0,
                    polish=False,
                    disp=False,

    if time_left() > 5:
        n = best_h.shape[0]
        bin_h, bin_score = binary_hill_climb(n, time_left, rng)
        if bin_score < best_score:
            best_score, best_h = bin_score, bin_h.copy()

    # ----- small random tweaks while time permits -----
    n_final = best_h.shape[0]
    while time_left() > 5:
        # random balanced binary neighbour
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
