    np.random.seed(random_seed + hash(n_points))
    
    # Setup
    p = 50  # Population size
    best_sol = None
    best_c5_val = float('inf')
    
    # Time tracking
    start_time = time.time()
    time_remaining = max_time
    
    # Set up initial population (mix of random and shifted parent)
    parent_feas = None
    shifted_parent = None
    parent_quality = float('inf')
    use_parent_seed = False
    
    # Mix parent variation based on initial_guess
    if initial_guess is not None:
        initial_vals = initial_guess.astype(float)
        if np.allclose(initial_vals, initial_guess, rtol=1e-5) and M - 0.1 < np.sum(initial_vals) < M + 0.1:
            repair = SumConstraintChecker(M)
            initial_vals = repair.project_to_feasible(initial_vals)
            parent_feas = initial_vals
            parent_quality = compute_overlap(parent_feas)
            use_parent_seed = True
    
    # Build initial population
    pop = [None] * p
    for i in range(p):
        if use_parent_seed and i < p//3:  # Inclusion: 30 parent seeds, 20% of population
            # Clone parent
            candidate = parent_feas.copy()
        else:
            candidate = generate_random_candidate(n_points)
        # Ensure sum condition and clip
        s_val = np.sum(candidate)
        dx = 2.0 / n_points
        dx_target = M  # Because sum(1) * dx = (n/2)/n *2 = 1, done separately
        if abs(s_val - M) > 0.1:
            repair = SumConstraintChecker(M)
            candidate = repair.project_to_feasible(candidate)
            if np.sum(candidate) < M - 0.1:
                adjust_val = M - np.sum(candidate)
                idx_min = np.argmin(candidate)
                candidate[idx_min] += adjust_val
            elif np.sum(candidate) > M + 0.1:
                adjust_val = - (np.sum(candidate) - M)
                idx_max = np.argmax(candidate)
                candidate[idx_max] += adjust_val
        # Scale to have integral 1.0
        # The function is defined to have integral 1, not just sum constraint.
        # But compute_overlap handles the dx factor. 
        # We'll evaluate with compute_overlap later.
        pop[i] = candidate

    gen_counter = 0
    improvement_counter = 0
    
    while time_remaining > 0.1:  # Until time is not just barely used
        new_pop = [None] * p
        for i in range(p):  # Generate a new population
            # Tournament selection: pick 5 random individuals and select best
            kwargs = {}
            selected = np.random.choice(p, size=5, replace=False)
            best_idx = selected[np.argmin([compute_overlap(pop[x]) for x in selected])]
            parent = pop[best_idx].copy()
            
            # Apply crossover or use mutation
            if np.random.random() < 0.8:  # Higher crossover chance to explore
                # Uniform crossover but only one point
                crossover_point = np.random.randint(0, n_points) if n_points > 0 else 0
                # Crossover with the best
                partner_idx = np.random.choice(p)
                if n_points > 0:
                    partner = pop[partner_idx]
                    new_par1 = np.concatenate([parent[:crossover_point], partner[crossover_point:]])
                    new_par2 = np.concatenate([partner[:crossover_point], parent[crossover_point:]])
                else:
                    new_par1 = parent.copy()
                    new_par2 = parent.copy()
            else:
                new_par1, new_par2 = parent.copy(), parent.copy()
            
            # Mutation and local search for each child
            child1 = new_par1.copy()
            child2 = new_par2.copy()
            
            # Mutation step (small)
            if n_points > 0:
                # Apply Gaussian perturbations to half the genes to avoid oscillation
                for i in range(n_points):
                    # 20% of genes might mutate
                    if np.random.random() < 0.2 and (abs(child1[i]-0.5) > 0.2 or improvement_counter < 5):
                        child1[i] += np.random.normal(0, 0.02)
                        child2[i] += np.random.normal(0, 0.02)
                # Repair box constraints and sum
                for i in range(2):
                    # Clip to [0,1]
                    valid_child = np.clip(child1, 0.0, 1.0)
                    total_valid = np.sum(valid_child)
                    if abs(total_valid - M) > 0.2:
                        # Repair sum by adjusting extreme values
                        repair = SumConstraintChecker(M)
                        valid_child = repair.project_to_feasible(valid_child)
                    child1 = valid_child
                    if np.random.random() < 0.7:
                        # Apply local search with fractional swap
                        try_frac_swap = np.random.random() < 0.3 * (1 + np.random.exponential(0.3) ** 2)
                        if try_frac_swap:
                            # Do fractional swap if parent is high quality
                            old_quality = compute_overlap(child1)
                            fracsafe_swap(child1)
                            new_quality = compute_overlap(child1)
                            if np.random.random() < 0.3 * (new_quality / old_quality if not np.isinf(old_quality) else 1):
                                # Keep swap if improved (stochastic)
                                pass
                    # Final repair and scale if needed ensure is integral?
                    final_quality = compute_overlap(child1)
                    if np.random.random() < 0.5:
                        old1 = final_quality
                        if n_points > 0:
                            low_indices = np.where(child1 < 0.01)[0]
                            high_indices = np.where(child1 > 0.99)[0]
                            if len(low_indices) > 0 and len(high_indices) > 0:
                                if len(low_indices) < 3 and len(high_indices) < 3:
                                    # Simple repair to allow local escape
                                    idx_low = np.random.choice(low_indices)
                                    idx_high = np.random.choice(high_indices)
                                    if child1[idx_low] < 0.009:
                                        child1[idx_low] = 0.01
                                        child1[idx_high] -= (child1[idx_high] - 0.01) * 0.02
                                    elif child1[idx_high] > 0.991:
                                        child1[idx_high] = 0.99
                                        child1[idx_low] += (1 - child1[idx_low]) * 0.02
                                    final_quality = compute_overlap(child1)
                        else:
                            # Safe repair
                            repair = SumConstraintChecker(M)
                            child1 = repair.project_to_feasible(child1)
                            final_quality = compute_overlap(child1)
                    candidate1 = child1
                    # For insertion into new_pop
                    new_pop[i] = candidate1
        
        # Selection 20-80% use the parent quality, not the previous iteration best
        # Compare current best and update
        for i in range(p):
            current_quality = compute_overlap(new_pop[i])
            if current_quality < best_c5_val or best_sol is None:
                best_c5_val = current_quality
                best_sol = new_pop[i]
                improvement_counter = max(0, improvement_counter + 1)
            if best_sol is not None:
                new_pop[i] = best_sol.copy()  # Use best solution for surviving to next generation
        
        # Limit to unique winners but allow parents to maintain clusters?
        current_time = time.time() - start_time
        time_remaining = max_time - current_time
        
    if best_sol is not None and best_c5_val < np.inf:
        repair = SumConstraintChecker(M)
        best_sol_repaired = repair.project_to_feasible(best_sol)
        if np.random.random() < 0.7:
            best_sol_repaired_quality = compute_overlap(best_sol_repaired)
            best_sol_final = best_sol_repaired
        else:
            best_sol_final = best_sol_repaired
    else:
        return None, float('inf'), None, n_points
    return best_sol_final, best_c5_val, gen_counter, n_points
