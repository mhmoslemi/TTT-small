def hill_climb_binary(h, time_limit, rng, max_iters=None):
    n = h.size
    # ensure binary (0/1)
    h = (h > 0.5).astype(np.int8)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_score = corr.max()
    best_h = h.copy()
    best_score = cur_score
    # temperature schedule if using simulated annealing
    start = time.time()
    it = 0
    # compute temperature schedule based on fraction of time left
    while time.time() - start < time_limit:
        # optionally break if max_iters reached
        if max_iters is not None and it >= max_iters:
            break
        # random swap: choose i from ones, j from zeros
        # sample i until h[i]==1
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # compute delta
        delta = np.zeros_like(corr, dtype=np.int32) # adjust dtype

        # contributions from i (1->0)
        len_src_i = n - i
        if len_src_i > 0:
            idx_src_i = np.arange(n-1, n-1 + len_src_i, dtype=np.int32)
            delta[idx_src_i] = h[i:i+len_src_i] - 1  # h[i+s] - 1
        if i > 0:
            s_i = np.arange(1, i+1, dtype=np.int32)
            idx_tgt_i = n-1 - s_i
            delta[idx_tgt_i] = h[i - s_i]
        # contributions from j (0->1)
        len_src_j = n - j
        if len_src_j > 0:
            idx_src_j = np.arange(n-1, n-1 + len_src_j, dtype=np.int32)
            delta[idx_src_j] += 1 - h[j:j+len_src_j]
        if j > 0:
            s_j = np.arange(1, j+1, dtype=np.int32)
            idx_tgt_j = n-1 - s_j
            delta[idx_tgt_j] += -h[j - s_j]
        # compute new score
        new_corr = corr + delta
        new_score = new_corr.max()
        # acceptance probability:
        # temperature declines linearly or exponential.
        # compute remaining fraction
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        # schedule: high at start, low at end.
        T0 = 0.1 * cur_score  # some base level
        T = T0 * (1 - fraction)  # linear cooling
        accept = False
        if new_score < cur_score:

        else:
            if T > 0:

                best_score = cur_score
                best_h = h.copy()
        # advance iteration
        it += 1
    return best_h, best_score

def rebalance_binary(h, target_sum, rng):
    # h integer 0/1.
    cur_sum = int(h.sum())
    if cur_sum == target_sum:
        return h
    elif cur_sum > target_sum:
        # need to flip diff ones to zero
        diff = cur_sum - target_sum
        idx_ones = np.flatnonzero(h)
        flip_idx = rng.choice(idx_ones, size=diff, replace=False)
        h[flip_idx] = 0
    else:
        diff = target_sum - cur_sum
        idx_zeros = np.flatnonzero(h == 0)
        flip_idx = rng.choice(idx_zeros, size=diff, replace=False)
        h[flip_idx] = 1
    return h

import time
import numpy as np

def _rebalance_binary(arr, target, rng):
    """Make a 0/1 array sum to target by flipping the minimal number of bits."""
    cur = int(arr.sum())
    if cur == target:
        return arr
    if cur > target:
        diff = cur - target
        idx = np.flatnonzero(arr)
        flip = rng.choice(idx, size=diff, replace=False)
        arr[flip] = 0
    else:
        diff = target - cur
        idx = np.flatnonzero(1 - arr)
        flip = rng.choice(idx, size=diff, replace=False)
        arr[flip] = 1
    return arr

def _hill_climb_binary(start_h, time_limit, rng):
    """Simulated‑annealing hill‑climb on a balanced binary vector."""
    n = start_h.shape[0]
    target = n // 2
    h = (start_h > 0.5).astype(np.int8)
    h = _rebalance_binary(h, target, rng)

    corr = np.correlate(h, 1 - h, mode='full')
    cur_score = int(corr.max())
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    # initial temperature proportional to the current worst correlation
    T0 = max(1.0, cur_score * 0.5)

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # pick a random 1 and a random 0 to swap
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)

        # compute delta of correlation caused by swapping i (1→0) and j (0→1)
        delta = np.zeros(2 * n - 1, dtype=np.int32)
        h_rev = h[::-1]

        # i : 1 → 0   (dh = -1)
        dh_i = -1
        delta[i:i + n] += dh_i * (1 - h_rev)        # source term
        delta[n - 1 - i:n - 1 - i + n] += -dh_i * h  # target term

        # j : 0 → 1   (dh = +1)
        dh_j = +1
        delta[j:j + n] += dh_j * (1 - h_rev)
        delta[n - 1 - j:n - 1 - j + n] += -dh_j * h

        # the central entry corresponds to shift 0 and must stay zero
        delta[n - 1] = 0

        new_score = int((corr + delta).max())

        # acceptance rule (simulated annealing)
        if new_score <= cur_score:
            accept = True
        else:
            T = T0 * (1.0 - elapsed / time_limit)
            accept = T > 0 and rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12))

        if accept:
            # apply the swap
            h[i] = 0
            h[j] = 1
            corr += delta
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # main resolution (must be even)
    n = 512
    target_sum = n // 2

    # initialise from parent if available, otherwise random
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 1.0, init_full.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        h_cont = np.interp(x_new, x_old, init_full)
        h_bin = (h_cont >= 0.5).astype(np.int8)
    else:
        h_bin = np.zeros(n, dtype=np.int8)
        idx = rng.choice(n, size=target_sum, replace=False)
        h_bin[idx] = 1

    h_bin = _rebalance_binary(h_bin, target_sum, rng)
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    best_h = h_bin.copy()
    best_score = int(corr.max())

    # main optimisation loop
    while time_left() > 6.0:
        # allocate a chunk of the remaining time to one hill‑climb run
        portion = min(30.0, time_left() * 0.6)

        # occasionally restart from a fresh random vector
        if rng.random() < 0.25:
            start_vec = np.zeros(n, dtype=np.int8)
            idx = rng.choice(n, size=target_sum, replace=False)
            start_vec[idx] = 1
        else:
            start_vec = best_h.copy()

        h_opt, sc_opt = _hill_climb_binary(start_vec, portion, rng)

        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    c5_bound = best_score * (2.0 / n)
    return best_h.astype(float), c5_bound, n
