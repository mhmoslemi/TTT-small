def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

This gradient is derived from derivative of correlation with respect to h? Let's verify.

Correlation corr(k) = sum h[j] * (1 - h[j + k]). gradient w.r.t h[i] contributed from terms where j=i (first factor) and where j + k = i (second factor). For each shift k:

first term: derivative of h[i] * (1 - h[i + k]) w.r.t h[i] is (1 - h[i + k]).

Second term: derivative of h[i - k] * (1 - h[i]) w.r.t h[i] is -h[i - k].

Thus gradient for shift k is (1 - h[i + k]) - h[i - k] = 1 - h[i+k] - h[i - k]. The code's term1 = h[shift:]; term2 = h[:n-shift]; then grad += (1.0 - term1 - term2).

If shift >=0, term1[i] = h[i+shift] (when i+shift < n). term2[i] = h[i - shift] (when i - shift >=0). So grad[i] = 1 - h[i+shift] - h[i - shift] (when needed). Works.

def gradient_descent_momentum(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)  # decreasing step
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

    thue = thue_morse(n)
    thue_score = compute_c5(thue)
    if thue_score < best_score:
        best_h, best_score = thue, thue_score
    if best_score <= target:
        return best_h, best_score, n

    # a few random binary seeds
    for _ in range(5):
        if time_left() < 5.0:
            break
        binary = random_binary_balanced(n, rng)
        score = compute_c5(binary)
        if score < best_score:
            best_h, best_score = binary, score
        if best_score <= target:
            return best_h, best_score, n

    # many continuous restarts with momentum GD
    while time_left() > 30.0:
        v = rng.random(n)
        v = project_onto_box_sum(v, n / 2.0)
        h_opt, sc_opt = gradient_descent_momentum(v, time_left,
                                                   max_iter=20000)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
        if best_score <= target:
            return best_h, best_score, n

    # binary conversion and annealing
    if best_h is not None:
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        bin_score = compute_c5(h_bin)
        if bin_score < best_score:
            best_h, best_score = h_bin, bin_score
        if best_score <= target:
            return best_h, best_score, n
        # annealing loop
        while time_left() > 5.0:
            # allocate iterations proportionally to remaining time
            max_iters = int(min(500000, time_left() * 2000))
            h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left,
                                                      max_iters=max_iters)
            if sc_bin_ref < best_score:
                best_h, best_score = h_bin_ref, sc_bin_ref
            if best_score <= target:
                return best_h, best_score, n

        # final continuous refinement
        if time_left() > 5.0:
            h_ref, sc_ref = gradient_descent_momentum(best_h, time_left,
                                                      max_iter=50000)
            if sc_ref < best_score:
                best_h, best_score = h_ref, sc_ref

    # final guard
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
