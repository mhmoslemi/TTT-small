def project_to_bounds_sum(h, M, lo=-1, hi=1, tol=1e-12, max_iter=60):
    # binary search lambda
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > M:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    return h_proj

import time
import numpy as np

# ---------- projection onto [0,1] with fixed sum ----------
def _project_sum_bounds(vec, target):
    """Project vec onto box [0,1]^n while enforcing sum(vec)=target."""
    lo, hi = -1.0, 1.0  # guarantees sum(lo) <= target <= sum(hi)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(vec + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)


# ---------- C5 computation ----------
def _c5_of(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# ---------- gradient of the worst overlap ----------
def _worst_lag_and_grad(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    zero = n - 1
    corr[zero] = -np.inf                     # ignore the zero shift
    idx = int(np.argmax(corr))
    lag = idx - zero
    # gradient of the raw overlap (without dx) for this lag
    g = np.zeros_like(h)
    if lag >= 0:
        up = n - lag
        g[:up] += 1.0 - h[lag:]
        g[lag:] += -h[:up]
    else:
        k = -lag
        up = n - k
        start = k
        g[start:] += 1.0 - h[:up]
        g[:up] += -h[start:]
    return lag, g, corr[idx] * (2.0 / n)      # also return current C5 value


# ---------- gradient descent for a fixed n ----------
def _optimize_n(n, time_budget, rng, init_vec=None):
    M = n // 2
    # start points
    starts = []
    if init_vec is not None and len(init_vec) == n:
        h0 = np.asarray(init_vec, dtype=float)
        h0 = _project_sum_bounds(h0, M)
        starts.append(h0)
    for _ in range(3):                         # a few random restarts
        h = rng.random(n)
        h = _project_sum_bounds(h, M)
        starts.append(h)

    best_h, best_c5 = None, np.inf
    start_time = time.time()
    for h0 in starts:
        h = h0.copy()
        best_local_h, best_local_c5 = h.copy(), _c5_of(h)
        alpha = 0.5
        no_imp = 0
        while time.time() - start_time < time_budget - 0.2:
            lag, grad, cur_c5 = _worst_lag_and_grad(h)
            if cur_c5 < best_local_c5 - 1e-12:
                best_local_c5 = cur_c5
                best_local_h = h.copy()
                no_imp = 0
                if cur_c5 < 0.3808:
                    return best_local_h, cur_c5
            else:
                no_imp += 1

            # line search on projected gradient step
            step = alpha
            improved = False
            for _ in range(10):
                h_new = _project_sum_bounds(h - step * grad, M)
                new_c5 = _c5_of(h_new)
                if new_c5 < cur_c5 - 1e-12:
                    h = h_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                # small random jitter to escape stagnation
                jitter = rng.uniform(-0.01, 0.01, size=n)
                h = _project_sum_bounds(h + jitter, M)

            alpha = max(alpha * 0.9995, 1e-4)

            if no_imp >= 5000:
                # fresh random start
                h = rng.random(n)
                h = _project_sum_bounds(h, M)
                no_imp = 0
                alpha = 0.5

        if best_local_c5 < best_c5:
            best_c5, best_h = best_local_c5, best_local_h
            if best_c5 < 0.3808:
                break
    return best_h, best_c5


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # decide which n to try
    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in candidates:
            candidates = [n0] + candidates
        elif n0 % 2 == 0:
            candidates.remove(n0)
            candidates = [n0] + candidates

    best_h, best_c5, best_n = None, np.inf, None

    for i, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        # allocate a fraction of remaining time for this n
        if i == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - i), 300.0)

        init = None
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            init = initial_h_values
        h_opt, c5_opt = _optimize_n(n, alloc, rng, init_vec=init)

        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # final safety fallback
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = _c5_of(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        M = best_n // 2
        if abs(best_h.sum() - M) > 1e-12:
            best_h = _project_sum_bounds(best_h, M)

    return best_h, best_c5, best_n
