def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def solve_mip_n(n, time_limit):
    import cvxpy as cp
    import numpy as np
    import time
    M = n // 2
    h = cp.Variable(n, boolean=True)
    # w matrix for products
    W = cp.Variable((n, n), boolean=True)
    constraints = []
    # product constraints
    # we only need i<j but enforce for all i,j to be safe? we can limit to i<j
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(W[i, j] <= h[i])
            constraints.append(W[i, j] <= h[j])
            # W[i,j] >= h[i] + h[j] -1
            constraints.append(W[i, j] >= h[i] + h[j] - 1)
    # sum constraint
    constraints.append(cp.sum(h) == M)
    # variable t
    t = cp.Variable()
    # min overlap constraints
    for k in range(1, n):
        indices = range(n - k)
        # sum_i W[i,i+k] >= t
        # Build sum expression
        expr = cp.sum([W[i, i+k] for i in indices])
        constraints.append(expr >= t)
    # t bounded
    constraints.append(t <= M)
    constraints.append(t >= 0)
    # objective maximize t

def solve_mip_n(n, time_limit):
    import cvxpy as cp
    M = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == M, t >= 0]
    W = []  # list of w vectors per shift
    for k in range(1, n):
        size = n - k
        w = cp.Variable(size, boolean=True)
        W.append(w)
        # product constraints
        for i in range(size):
            constraints.append(w[i] <= h[i])
            constraints.append(w[i] <= h[i + k])
            constraints.append(w[i] >= h[i] + h[i + k] - 1)
        # overlap constraint
        sum_h_k = cp.sum(h[:size])
        sum_w_k = cp.sum(w)
        constraints.append(sum_w_k >= sum_h_k - t)
    # objective minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit)
    except Exception:
        # try GLPK_MI
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000))
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h_val = np.rint(h.value).astype(float)  # binary rounding
    t_val = float(t.value)
    c5 = 2.0 / n * t_val
    return h_val, c5
```

import time
import numpy as np

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def solve_mip_n(n, time_limit):
    """
    Solve the binary max‑min overlap problem for a given resolution n.
    Returns (h_opt, c5_opt) or (None, None) on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    M = n // 2  # required sum of h

    # binary decision variables for h
    h = cp.Variable(n, boolean=True)
    # scalar t = maximal overlap (unscaled)
    t = cp.Variable()

    constraints = [
        cp.sum(h) == M,
        t >= 0
    ]

    # product variables w_{i,k} = h_i * h_{i+k}
    for k in range(1, n):
        size = n - k
        w = cp.Variable(size, boolean=True)  # w[i] corresponds to (i, i+k)
        # link w to h via McCormick constraints (exact for binary)
        constraints += [
            w <= h[:size],
            w <= h[k:],
            w >= h[:size] + h[k:] - 1
        ]
        # overlap constraint for this shift:
        # M_k - sum_i h_i*h_{i+k} <= t  =>  sum_i w_i >= sum_i h_i - t
        sum_hk = cp.sum(h[:size])
        constraints.append(cp.sum(w) >= sum_hk - t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Try CBC first; fall back to GLPK_MI if needed
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit * 1000))
        except Exception:
            return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(float)  # binary rounding
    t_val = float(t.value)
    c5 = (2.0 / n) * t_val
    return h_val, c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.default_rng(seed)  # seed unused but kept for compatibility

    best_h = None
    best_c5 = np.inf
    best_n = None

    # evaluate parent construction if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        if n0 % 2 == 0:
            M0 = n0 // 2
            cur = np.sum(h0)
            if abs(cur - M0) > 1e-9:
                h0 = h0 + (M0 - cur) / n0
                h0 = np.clip(h0, 0.0, 1.0)
            c0 = compute_c5(h0)
            best_h, best_c5, best_n = h0, c0, n0

    # candidate resolutions (even numbers)
    candidates = [80, 96, 112, 128]
    if best_n is not None and best_n in candidates:
        candidates.remove(best_n)
        candidates = [best_n] + candidates

    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        # allocate a generous chunk for the first resolution, smaller later
        if idx == 0:
            alloc = min(remaining * 0.6, 300.0)
        else:
            alloc = min(remaining * 0.3 / max(1, len(candidates) - idx - 1), 200.0)
        h_opt, c5_opt = solve_mip_n(n, alloc)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break

    # final fallback if nothing succeeded
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        n = best_n
        # enforce constraints exactly after integer rounding
        best_h = np.clip(best_h, 0.0, 1.0)
        M = n // 2
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h = best_h + (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
