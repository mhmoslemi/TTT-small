def project_feasibility(h):
    # clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # enforce sum
    s = h.sum()
    M = len(h) / 2
    if abs(s - M) > 1e-9:
        # adjust by distributing difference uniformly to interior entries (who are not at bounds)
        # Simple: add (M - s) / n to all entries, then clip again, maybe repeat few times
        diff = M - s
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
        # after clipping, recompute sum and if still off, do loop up to a few times
        s2 = h.sum()
        if abs(s2 - M) > 1e-9:
            # final correction: compute need = M - s2; adjust proportionally to entries not at bounds
            # find indices not at bounds
            free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
            if free.size > 0:
                h[free] += (M - s2) / free.size
                np.clip(h, 0.0, 1.0, out=h)
    # final ensure sum approx M
    return h

import time
import numpy as np

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, M):
    np.clip(h, 0.0, 1.0, out=h)
    diff = M - h.sum()
    n = h.size
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        diff = M - h.sum()
        if abs(diff) > 1e-12:
            free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if free.size:
                h[free] += diff / free.size
                np.clip(h, 0.0, 1.0, out=h)
    return h

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        seq[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
    return seq

def generate_random_binary(n, M, rng):
    h = np.zeros(n)
    h[rng.choice(n, M, replace=False)] = 1.0
    return h

def optimise_n(n, time_budget, rng):
    M = n // 2
    dx = 2.0 / n
    best_h = None
    best_c5 = np.inf
    start = time.time()
    inits = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        inits.append(np.array(initial_h_values, dtype=float))
    if n == 80:
        p = n - 1
        leg = legendre_sequence(p)
        h_leg = np.concatenate([leg, [1.0]]).astype(float)
        h_leg = np.roll(h_leg, rng.integers(n))
        inits.append(h_leg)
    inits.append(generate_random_binary(n, M, rng))
    per_init = max(time_budget / len(inits), 1.0)
    for init_h in inits:
        if time.time() - start > time_budget - 0.1:
            break
        h = project_feasible(init_h.copy(), M)
        overl = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(overl) * dx
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        sub_start = time.time()
        max_iter = 200000
        for it in range(max_iter):
            if time.time() - sub_start > per_init - 0.05:
                break
            overl = np.correlate(h, 1.0 - h, mode='full')
            C5_cur = np.max(overl) * dx
            if C5_cur < 0.3808:
                return h.copy(), C5_cur, n
            if C5_cur + 1e-12 < best_c5:
                best_c5, best_h = C5_cur, h.copy()
            # softmax temperature
            T = 1.0 + (0.01 - 1.0) * (it / max_iter)
            max_o = np.max(overl)
            w = np.exp((overl - max_o) / T)
            w /= w.sum()
            g = np.zeros_like(h)
            offset = n - 1
            for k in range(-n + 1, n):
                wk = w[k + offset]
                if wk == 0.0:
                    continue
                if k >= 0:
                    end = n - k
                    g[:end] -= dx * wk * h[k:k + end]
                    g[k:]   -= dx * wk * h[:end]
                else:
                    kk = -k
                    g[kk:] -= dx * wk * h[:n - kk]
                    g[:n - kk] -= dx * wk * h[kk:]
            lr = 0.5 + (0.01 - 0.5) * (it / max_iter)
            h = h - lr * g
            if rng.random() < 0.001:
                i = rng.integers(n)
                delta = (rng.random() - 0.5) * 0.1
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            np.clip(h, 0.0, 1.0, out=h)
            h = project_feasible(h, M)
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf
    best_n = None
    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates
    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            left = len(candidates) - idx
            alloc = min(remaining * 0.4 / left, 300.0)
        h_opt, c5_opt, n_used = optimise_n(n, alloc, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_used
            if best_c5 < 0.3808:
                break
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / best_n)
    else:
        best_h = project_feasible(best_h.copy(), best_n // 2)
        best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
