        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        cur_c5 = corr[idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def de_optimize(pop, target_sum, rng, deadline, best_h, best_c5,
                pop_fitness, F=0.5, CR=0.9, target=0.3808):
    """Classic DE/rand/1/bin with sum constraint."""
    n = pop.shape[1]
    popsize = pop.shape[0]
    while time.time() < deadline:
        for i in range(popsize):
            # choose three distinct other indices
            idxs = rng.choice([j for j in range(popsize) if j != i], size=3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = pop[i].copy()
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)
            c5_trial = compute_c5(trial)
            if c5_trial < pop_fitness[i]:
                pop[i] = trial
                pop_fitness[i] = c5_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
                    if best_c5 < target:
                        return pop, pop_fitness, best_h, best_c5
        if time.time() > deadline:
            break
    return pop, pop_fitness, best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Initialise problem size and feasible start -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ----- Quick gradient refinement of the starter -----
    gd_deadline = start_time + budget_s * 0.08
    best_h, best_c5 = gradient_descent(best_h, target_sum,
                                       max_iter=3000,
                                       deadline=gd_deadline)

    # ----- Prepare DE population -----
    popsize = 25
    pop = np.empty((popsize, n))
    pop_fitness = np.empty(popsize)
    pop[0] = best_h
    pop_fitness[0] = best_c5
    for i in range(1, popsize):
        vec = rng.random(n)
        vec = project_to_box_sum(vec, target_sum)
        pop[i] = vec
        pop_fitness[i] = compute_c5(vec)

    safety = 5.0
    target_bound = 0.3808

    # ----- Main search loop -----
    while time.time() < start_time + budget_s - safety:
        # allocate a short slice for DE work
        de_deadline = min(time.time() + 1.5,
                          start_time + budget_s - safety)
        pop, pop_fitness, best_h, best_c5 = de_optimize(
            pop, target_sum, rng, de_deadline,
            best_h, best_c5, pop_fitness,
            F=0.7, CR=0.9, target=target_bound)

        if best_c5 < target_bound:
            break

        # occasional local gradient refinement on the current best
        if time.time() < start_time + budget_s - safety - 5.0:
            gd_deadline = start_time + budget_s - safety
            h_gd, c5_gd = gradient_descent(best_h, target_sum,
                                           max_iter=4000,
                                           deadline=gd_deadline)
            if c5_gd < best_c5:
                best_h, best_c5 = h_gd, c5_gd
                if best_c5 < target_bound:
                    break

        # occasional random restart if stagnation detected
        if rng.random() < 0.02:
            idx = rng.integers(0, popsize)
            vec = rng.random(n)
            vec = project_to_box_sum(vec, target_sum)
            pop[idx] = vec
            pop_fitness[idx] = compute_c5(vec)

    # ----- Final projection and evaluation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(n)
