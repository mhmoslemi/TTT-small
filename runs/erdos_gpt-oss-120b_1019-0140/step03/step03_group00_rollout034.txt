def gradient_swap_local_search(h, max_iters, time_left):
    n = len(h)
    dx = 2.0 / n
    best_score = compute_c5(h)
    # while time left and iteration
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shift_idx = np.argmax(corr)
        shift = shift_idx - (n - 1)
        # compute gradient for shift
        grad = np.zeros_like(h)
        if shift >= 0:
            grad[:n-shift] += dx * (1.0 - h[shift:])
            grad[shift:] -= dx * h[:n-shift]
        else:
            s = -shift
            grad[s:] += dx * (1.0 - h[:n-s])
            grad[:n-s] -= dx * h[s:]
        # find i with max grad, j with min grad
        i_max = np.argmax(grad)  # largest positive gradient
        i_min = np.argmin(grad)  # most negative
        if grad[i_max] <= grad[i_min] + 1e-12:
            # no beneficial move
            break
        # ensure we can transfer
        max_transfer = min(h[i_max], 1 - h[i_min], 0.1)  # limit step
        if max_transfer <= 0:
            break
        # transfer
        h[i_max] -= max_transfer
        h[i_min] += max_transfer
        # compute new score maybe after each iteration or after some batch.
        score = compute_c5(h)
        if score < best_score:
            best_score = score
        # optional early break if under target
        if best_score <= 0.3808:
            break
    return h, best_score

def compute_c5(h):

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def grad_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad

def gradient_descent(h0, max_iter, step0, beta, rng, time_left):
    n = h0.size
    s = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_sc = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = grad_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def gradient_swap_local_search(h0, max_iters, time_left):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    best_sc = compute_c5(h)
    for _ in range(max_iters):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        grad = np.zeros_like(h)
        if shift >= 0:
            grad[:n - shift] += dx * (1.0 - h[shift:])
            grad[shift:] -= dx * h[:n - shift]
        else:
            s = -shift
            grad[s:] += dx * (1.0 - h[:n - s])
            grad[:n - s] -= dx * h[s:]
        i_max = np.argmax(grad)
        i_min = np.argmin(grad)
        if grad[i_max] <= grad[i_min] + 1e-12:
            break
        delta = min(h[i_max], 1.0 - h[i_min], 0.07)
        if delta <= 0:
            break
        h[i_max] -= delta
        h[i_min] += delta
        sc = compute_c5(h)
        if sc < best_sc:
            best_sc = sc
        if best_sc <= 0.3808:
            break
    return h, best_sc
