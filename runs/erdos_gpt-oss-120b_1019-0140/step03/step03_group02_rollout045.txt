def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shift indices from -(n-1) to (n-1)
    for idx, k in enumerate(range(-(n-1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            # i from 0..n-k-1 add (1-h[i+k])
            grad[:n - k] += w * (1.0 - h[k:])
            # i from k..n-1 subtract h[i-k]
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad
```

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # set iteration limit based on time left
    max_iters = int(min(300000, max(1, time_left() * 5000)))  # 5000 evaluations per sec
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick random zero and one
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = rng.choice(zeros)
        j = rng.choice(ones)
        # swap
        h[i], h[j] = 1, 0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        # cooling schedule
        T = T0 * (1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            # accept
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            # reject: swap back
            h[i], h[j] = 0, 1
    return best_h, best_score
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximation on the discrete grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_softmax_corr(h, alpha):
    """Gradient of the soft‑max surrogate of the max‑correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shifts range from -(n-1) to (n-1)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    """Projected gradient descent on soft‑max surrogate with momentum."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target):
    """Simulated‑annealing style hill‑climb on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h, best_score = h.copy(), curr_score
    max_iters = int(min(300000, max(1, time_left() * 5000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        zeros = np.flatnonzero(h == 0)
        ones  = np.flatnonzero(h == 1)
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h[i], h[j] = 1.0, 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            curr_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
                if best_score <= target:
                    break
        else:
            h[i], h[j] = 0.0, 1.0
    return best_h, best_score

def thue_morse(n):
    """Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    arr[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return arr

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)
