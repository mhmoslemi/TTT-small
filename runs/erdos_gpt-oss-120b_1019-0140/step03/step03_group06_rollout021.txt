def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine n
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        # start with random continuous vector
        h0 = random_continuous_vector(n, n/2.0, rng)

    target = n / 2.0
    dx = 2.0 / n
    # Ensure sum
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Create bounds for SLSQP
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # Build constraints for SLSQP
    constraints = build_constraints(n, target, dx)

    # Set number of SLSQP attempts based on remaining time
    while time.time() < deadline - 10.0:  # keep time for local search
        # Choose start vector
        if rng.random() < 0.5:
            # random binary
            start_h = random_binary_vector(n, target, rng)
        else:
            # perturb best
            start_h = best_h.copy()
            # apply few random transfers
            for _ in range(5):
                src = rng.choice(np.where(start_h > 1e-8)[0])
                dst = rng.choice(np.where(start_h < 1.0 - 1e-8)[0])
                max_delta = min(start_h[src], 1.0 - start_h[dst])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[src] -= delta
                start_h[dst] += delta
            start_h = enforce_sum(start_h, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 2000, "ftol": 1e-10, "disp": False},
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # Gradient-guided local search
    while time.time() < deadline - 0.5:
        # identify worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute gradient for shift k
        g = np.ones(n, dtype=float)
        if k >= 0:
            g[: n - k] -= best_h[k:]
            g[k:] -= best_h[: n - k]
        else:
            kk = -k
            g[kk:] -= best_h[: n - kk]
            g[: n - kk] -= best_h[kk:]
        # src where g>0, h>eps; tgt where g<0, h<1-eps
        eps = 1e-8
        src_cand = np.where((g > 1e-4) & (best_h > eps))[0]
        tgt_cand = np.where((g < -1e-4) & (best_h < 1.0 - eps))[0]
        if src_cand.size == 0 or tgt_cand.size == 0:
            break
        src = rng.choice(src_cand)
        tgt = rng.choice(tgt_cand)
        max_delta = min(best_h[src] - eps, 1.0 - best_h[tgt] - eps)
        if max_delta <= 0:
            break
        # delta scaled by magnitude of gradient difference
        delta = max_delta * min(1.0, abs(g[src] - g[tgt]) / 2.0) * rng.random()
        # ensure min step
        if delta < 1e-6:
            delta = min(max_delta, 1e-6)
        # apply
        best_h[src] -= delta
        best_h[tgt] += delta
        val = compute_c5(best_h)
        if val < best_val:
            best_val = val
            # keep changes
        else:
            # revert with probability based on temperature
            remaining = deadline - time.time()
            T = remaining * 0.001 + 1e-12
            if np.exp((best_val - val) / T) > rng.random():
                best_val = val
                # keep worse
            else:
                # revert
                best_h[src] += delta
                best_h[tgt] -= delta

    # final enforcement and compute final value
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Project v onto [0,1]^n with exact sum == target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Scale first
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    # Distribute the residual iteratively
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if cand.size == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)})
    return cons

# ---------- main entry point ----------

    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # --- initialise ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    dx = 2.0 / n
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # --- SLSQP optimisation phase ---
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = build_constraints(n, target, dx)

    while time.time() < deadline - 10.0:
        # random start: binary 60% of the time, else perturb current best
        if rng.random() < 0.6:
            start_h = random_binary_vector(n, target, rng)
        else:
            start_h = best_h.copy()
            for _ in range(5):
                src = rng.choice(np.where(start_h > 1e-8)[0])
                dst = rng.choice(np.where(start_h < 1.0 - 1e-8)[0])
                max_delta = min(start_h[src], 1.0 - start_h[dst])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[src] -= delta
                start_h[dst] += delta
            start_h = enforce_sum(start_h, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:

                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,

        except Exception:
            break

            if new_val < best_val:

                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:

            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
                # keep the uphill move
            else:
                # revert
                cur_h[src] += delta
                cur_h[dst] -= delta

    # --- final enforcement and return ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
