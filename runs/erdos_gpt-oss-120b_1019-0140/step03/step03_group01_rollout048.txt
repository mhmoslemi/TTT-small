def weighted_gradient(h, shifts, w):
    # w must sum to 1 (or we can normalize)
    n = h.size
    grad = np.zeros_like(h)
    for shift, wi in zip(shifts, w):
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += wi * (1.0 - term1 - term2)
    return grad

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    # generate 0/1 Thue-Morse pattern
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=20000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.01:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def continuous_swap_anneal(h0, rng, time_left_func, max_iters=200000, T0=0.2, T_min=1e-5, decay=0.9999):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # decide direction
        if rng.random() < 0.5:
            delta_max = min(1.0 - h[i], h[j])
            if delta_max <= 0:
                it += 1
                continue
            delta = delta_max * rng.random()
            i_target, j_target = i, j
        else:
            delta_max = min(1.0 - h[j], h[i])
            if delta_max <= 0:
                it += 1
                continue
            delta = delta_max * rng.random()
            i_target, j_target = j, i
        h[i_target] += delta
        h[j_target] -= delta
        new_score = compute_c5(h)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i_target] -= delta
            h[j_target] += delta
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000, T0=0.1, T_min=1e-5, decay=0.99995):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1500):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
