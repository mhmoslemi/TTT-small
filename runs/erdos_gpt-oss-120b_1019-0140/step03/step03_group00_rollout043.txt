def local_search_binary(h0, time_left, rng):
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # get indices lists
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    # Temperature schedule
    T = 0.001
    # number of iterations
    it = 0
    while time_left() > 0.2:
        it += 1
        # occasionally refresh ones/zeros if they changed
        if it % 1000 == 0:
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            cur_score = new_score
            # update lists
            # remove i from ones, add to zeros; etc.
            # we can just recompute after many swaps
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        # cooling
        T *= 0.9999  # cooling factor per iteration
        # ensure T not too small
        if T < 1e-10:
            T = 1e-10
    return best_h, best_score
```

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def binary_from_scores(scores):
    """Turn a real-valued vector into a balanced binary vector using top half."""
    n = scores.shape[0]
    k = n // 2
    idx = np.argpartition(-scores, k)[:k]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of the max‑correlation term averaged over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def local_search_binary(h0, time_left_func, rng):
    """Simulated‑annealing hill‑climb on binary vectors."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T = 0.01  # initial temperature
    n = h.shape[0]

    while time_left_func() > 0.1:
        # propose a swap of a 1 and a 0
        i = rng.integers(n)
        if h[i] != 1.0:
            continue
        j = rng.integers(n)
        if h[j] != 0.0:
            continue
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        T *= 0.9999
        if T < 1e-9:
            T = 1e-9
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 512  # number of sample points (preferred under 1000)

    # ---- build initial candidates ------------------------------------------------
    candidates = []

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base_up = upsample(base, n)
        candidates.append(binary_from_scores(base_up))

    tm64 = thue_morse(64)
    candidates.append(binary_from_scores(upsample(tm64, n)))

    for _ in range(4):
        candidates.append(random_binary_balanced(n, rng))

    # evaluate and keep the best two
    scored = [(compute_c5(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in scored[:2]]
    best_h = top_candidates[0].copy()
    best_score = compute_c5(best_h)

    # ---- intensive binary simulated annealing ------------------------------------
    reserve = 20.0  # seconds kept for final refinement
    while time_left() > reserve:
        remaining = time_left() - reserve
        # allocate a slice of the remaining time to this local search
        slice_time = min(remaining, 60.0)  # at most a minute per slice
        def slice_time_left():
            return slice_time - (time.time() - local_start)
        local_start = time.time()
        h_opt, sc_opt = local_search_binary(best_h, slice_time_left, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # update best_h for the next slice
        best_h = h_opt

    # ---- final continuous refinement --------------------------------------------
    if time_left() > 5.0:
        # generous iteration count; the function respects the time budget internally
        best_h, best_score = gradient_descent_momentum_custom(
            best_h,
            n_iter=200000,
            step0=0.2,
            beta=0.9,
            rng=rng,
            time_left=time_left)

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
