def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        # schedule alpha
        alpha = alpha_start + (alpha_end - alpha_start)*(it-1)/(max(1, n_iter-1))
        # compute gradient of softmax
        grad = grad_softmax_corr(h, alpha)
        # momentum
        mom = beta * mom + (1 - beta) * grad
        # step size
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # shift indices: shift = idx - (n-1)
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            s = -k
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    return grad
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808, max_swaps_total=2000000):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    iter_count = 0
    T0 = 0.02
    while time_left() > 0.01 and iter_count < max_swaps_total:
        # pick i zero, j one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - iter_count / max_swaps_total)
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
        iter_count += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # Initial lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Return Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    """Linear interpolation from size arr.shape[0] to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of sum of correlation terms for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:]      -= h[:n - k]
        else:
            s = -k
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    return grad

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            s = -k
            grad[s:]      += w * (1.0 - h[:n - s])
            grad[:n - s]  -= w * h[s:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end,
                        beta, time_left):
    """Gradient descent on soft‑max approximation of max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):
    """Projected gradient descent using only the maximal correlation shifts."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808, max_swaps_total=2000000):
    """Simulated‑annealing hill climb on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    iter_count = 0
    T0 = 0.02
    while time_left() > 0.01 and iter_count < max_swaps_total:
        # pick a zero and a one uniformly
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1.0 - iter_count / max_swaps_total)
        accept = False
        if delta < 0.0:
            accept = True
        elif T > 0.0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
        iter_count += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # grid sizes
    n_small = 128
    n_large = 512
    s_small = n_small / 2.0
    s_large = n_large / 2.0

    candidates_small.append(thue_morse(n_small).astype(float))
    for _ in range(6):

    for _ in range(4):

        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), s_small)

    # ---- Phase 2: upsample to large grid and diversify ----
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, s_large)

    candidates_large = [h_up]
    for _ in range(5):

            step0=0.5,
            alpha_start=0.5,
            alpha_end=20.0,
            beta=0.9,
            time_left=time_left
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # deeper refinement on current best large solution
    if time_left() > 4.0:
        h_ref, sc_ref = smooth_grad_descent(
            best_h,
            n_iter=6000,
            step0=0.5,
            alpha_start=0.5,
            alpha_end=30.0,
            beta=0.9,
            time_left=time_left
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # ---- Phase 3: binary rounding and simulated‑annealing hill climb ----
    n = best_h.shape[0]
    order = np.argsort(best_h)
    binary_h = np.zeros(n, dtype=float)
    binary_h[order[n // 2:]] = 1.0

            best_h = h_final.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
