```python
def binary_hill_climb(h0, rng, time_left, max_no_improve=2000, batch=100):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    # maintain list of positions of zeros and ones
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    no_improve_counter = 0
    while time_left() > 0.01:
        # sample batch candidate swaps
        i_candidates = rng.choice(zeros, size=batch, replace=True)
        j_candidates = rng.choice(ones, size=batch, replace=True)
        improved = False
        for i, j in zip(i_candidates, j_candidates):
            h_new = h.copy()
            h_new[i] = 1.0
            h_new[j] = 0.0
            new_score = compute_c5(h_new)
            if new_score < best_score - 1e-12:
                # accept
                h = h_new
                best_h, best_score = h_new.copy(), new_score
                # update zeros/ones
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                improved = True
                break
        if not improved:
            no_improve_counter += 1
            if no_improve_counter >= max_no_improve:
                # do random restart maybe
                # random perturbation: flip a few bits
                # choose k bits to flip: maybe 5.
                k_flip = 5
                # ensure equal numbers: pick k_flip zeros and k_flip ones
                zero_idx = zeros[rng.choice(len(zeros), size=k_flip, replace=False)]
                one_idx = ones[rng.choice(len(ones), size=k_flip, replace=False)]
                h[zero_idx] = 1.0
                h[one_idx] = 0.0
                # update zeros/ones
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                no_improve_counter = 0
        # ensure time left eventually
    return best_h, best_score
```

```python
        j_batch = rng.choice(ones, size=batch, replace=True)
```

```python
    m = arr.shape[0]

    x_old = np.linspace(0, 1, m, endpoint=False)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h0, rng, time_left, max_no_improve=2000):
    """Targeted hill climbing on binary vectors using max‑shift swaps."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    while time_left() > 0.02:
        corr = np.correlate(h, 1.0 - h, mode='full')
        shift = np.argmax(corr) - (n - 1)
        # determine mismatched (bad) and complementary (good) indices for this shift
        if shift >= 0:
            rng_idx = np.arange(0, n - shift)
            left = h[rng_idx]
            right = h[rng_idx + shift]
        else:
            k = -shift
            rng_idx = np.arange(k, n)
            left = h[rng_idx]
            right = h[rng_idx - k]
        bad_idx = rng_idx[(left == 1) & (right == 0)]
        good_idx = rng_idx[(left == 0) & (right == 1)]
        if bad_idx.size and good_idx.size:
            i = rng.choice(bad_idx)
            j = rng.choice(good_idx)
        else:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size == 0 or ones.size == 0:
                break
            i = rng.choice(zeros)
            j = rng.choice(ones)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-12:
            h = h_new
            best_h = h_new.copy()
            best_score = new_score
            no_improve = 0
            continue
        no_improve += 1
        if no_improve >= max_no_improve:
            kbits = min(5, n // 20)
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if zeros.size >= kbits and ones.size >= kbits:
                z_sel = rng.choice(zeros, size=kbits, replace=False)
                o_sel = rng.choice(ones, size=kbits, replace=False)
                h[z_sel] = 1.0
                h[o_sel] = 0.0
            no_improve = 0
    return best_h, best_score
