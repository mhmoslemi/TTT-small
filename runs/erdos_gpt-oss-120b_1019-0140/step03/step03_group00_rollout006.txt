def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # clamp sum feasible
    if total <= lo*n:
        return np.full_like(v, lo)
    if total >= hi*n:
        return np.full_like(v, hi)
    # initial lam bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # adjust bounds if needed
    # find lam such that f(lam) = total
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        sum_mid = f(lam_mid)
        if abs(sum_mid - total) < 1e-12:
            break
        if sum_mid > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count('1') & 1)
    return seq


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(h0, time_limit, rng):
    """Simple SA that preserves sum and box constraints."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    max_iters = int(1e8)
    init_temp = 0.006  # modest temperature

    for it in range(max_iters):
        if time.time() - start >= time_limit:
            break
        # linear temperature decay
        temp = init_temp * (1.0 - it / max_iters)
        if temp < 0:
            temp = 0.0

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        scale = min(1.0, (temp / init_temp) * 2.0)
        delta = max_delta * rng.random() * scale
        if delta <= 0.0:
            continue

        # apply move
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            if temp > 0.0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    cur_score = new_score
                else:
                    h[i] += delta
                    h[j] -= delta
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    scales = [64, 128, 256, 512]
    fractions = {64: 0.1, 128: 0.2, 256: 0.3, 512: 0.4}
    best_h = base.copy()
    best_score = compute_c5(best_h)

    for n in scales:
        if time_left() <= 0.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        alloc = fractions[n] * budget_s
        if alloc > time_left():
            alloc = time_left()
        if alloc <= 0.0:
            continue

        cand_h, cand_score = simulated_annealing(base, alloc, rng)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
        base = cand_h

    # final stochastic local improvements with remaining budget
    while time_left() > 0.6:
        n = best_h.size
        # random balanced binary perturbation
        h_rand = random_balanced_binary(n, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score = sc_rand
            best_h = h_rand.copy()
        # small mass transfer between two positions
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0.0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
