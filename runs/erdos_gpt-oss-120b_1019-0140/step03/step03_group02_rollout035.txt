- Define compute_c5_fft as:

    def compute_c5_fft(h):
        n = h.shape[0]
        # compute full linear correlation using FFT
        # pad to length 2*n
        size = 1 << ((2*n-1).bit_length())
        H = np.fft.rfft(h, n=size)
        G = np.fft.rfft(1.0 - h, n=size)
        corr = np.fft.irfft(H * np.conj(G), n=size)[:2*n-1]
        # compute max
        return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):

    # Ensure sum constraint
    init_h = project_onto_box_sum(init_h, n_target / 2.0)

    best_h = init_h.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: continuous smooth descent
    # maybe run several cycles
    while time_left() > 30.0:  # allocate at least 30 sec for rest
        # Determine iteration count dynamic based on time left
        remaining = time_left()

def compute_c5_fft(h):
    n = h.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)[:2*n - 1]
    return np.max(corr) * (2.0 / n)

def smooth_grad_descent(h0, n_iter=2000, step0=0.5, alpha_start=1.0, alpha_end=10.0, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        # indices of shifts achieving max
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient for max shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] += (1.0 - h[shift:])
                grad[shift:]   -= h[:n - shift]
            else:
                s = -shift
                grad[s:]      += (1.0 - h[:n - s])
                grad[:n - s]  -= h[s:]
        grad /= len(shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

        if time_left() < 0.01:

                best_score, best_h = new_score, h_new.copy()

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(budget_s - (time.time() - start), 0.0)

    # target resolution
    n_target = 512

    # initial candidate
    if 'initial_h_values' in globals():

    init_h = project_onto_box_sum(init_h, n_target / 2.0)
    best_h = init_h.copy()
    best_score = compute_c5_fft(best_h)

    # Phase 1: smooth descent loop
    while time_left() > 30.0:
        remaining = time_left()
        # dynamic iteration count for smooth descent, aim ~2000-6000
        max_iter = int(min(8000, remaining * 30))  # assuming 0.03 sec per iteration
        if max_iter < 5:
            break
        h_opt, sc_opt = smooth_grad_descent(best_h, n_iter=max_iter,
                                            step0=0.5,

                                            beta=0.9,
                                            time_left=time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt.copy(), sc_opt
        # momentum descent
        remaining2 = time_left()
        max_iter2 = int(min(20000, remaining2 * 100))
        if max_iter2 < 5:
            break
        h_opt2, sc_opt2 = momentum_grad_descent(best_h,
                                                n_iter=max_iter2,
                                                step0=0.5,
                                                beta=0.9,
                                                time_left=time_left)
        if sc_opt2 < best_score:
            best_h, best_score = h_opt2.copy(), sc_opt2
        if best_score <= 0.3808:
            break

    # Phase 2: binary hill climbing
    if time_left() > 10.0:
        # binary projection of best_h
        n = best_h.shape[0]
        order = np.argsort(best_h)  # ascending
        binary = np.zeros(n, dtype=float)
        binary[order[n//2:]] = 1.0

        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:

        # final continuous refinement

            h_ref, sc_ref = momentum_grad_descent(best_h,
                                                  n_iter=5000,
                                                  step0=0.3,
                                                  beta=0.9,
                                                  time_left=time_left)
            if sc_ref < best_score:
                best_h, best_score = h_ref.copy(), sc_ref

    # final projection (ensure sum)
    best_h = project_onto_box_sum(best_h, n_target / 2.0)
    final_score = compute_c5_fft(best_h)

    return best_h, final_score, n_target

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_c5_fft(h):
    """Fast max overlap integral using FFT; returns C5."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def grad_softmax_corr(h, alpha):
    """Gradient of soft‑max approximation of max correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on soft‑max of correlations."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent(h0, n_iter=5000, step0=0.5,
                          beta=0.9, time_left=lambda: 1e9):
    """Momentum projected gradient descent using only max‑correlation shifts."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] += (1.0 - h[shift:])
                grad[shift:]   -= h[:n - shift]
            else:
                sft = -shift
                grad[sft:]     += (1.0 - h[:n - sft])
                grad[:n - sft] -= h[sft:]
        grad /= len(shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill climbing on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(400000, max(1, time_left() * 3000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5_fft(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0.0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score
