import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust v∈[0,1] so that sum(v) == target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def gradient_descent_local(h, target, deadline, rng, best_h=None, best_val=None):
    n = h.shape[0]
    dx = 2.0 / n
    if best_h is None:
        best_h = h.copy()
        best_val = compute_c5(best_h)
    it = 0
    while time.time() < deadline - 0.1 and it < 5000:
        # correlation and current value
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        cur_val = max_corr * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break
        # indices of worst shift(s)
        mask = np.isclose(corr, max_corr, atol=1e-12)
        idxs = np.flatnonzero(mask)
        grad = np.zeros(n)
        for idx in idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                i = np.arange(0, n - shift)
                grad[i] += (1.0 - h[i + shift])
                grad[i + shift] += -h[i]
            else:
                s = -shift
                i = np.arange(s, n)
                grad[i] += (1.0 - h[i - s])
                grad[i - s] += -h[i]
        # descent direction respecting sum constraint
        d = -grad + grad.mean()
        pos = d > 1e-12
        neg = d < -1e-12
        step_max = np.inf
        if np.any(pos):
            step_max = min(step_max, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            step_max = min(step_max, np.min((-h[neg]) / d[neg]))
        if not np.isfinite(step_max) or step_max <= 0:
            # random small transfer to keep moving
            src = np.where(h > 1e-8)[0]
            dst = np.where(h < 1.0 - 1e-8)[0]
            if src.size and dst.size:
                i = rng.choice(src)
                j = rng.choice(dst)
                delta = rng.random() * min(h[i], 1.0 - h[j])
                h[i] -= delta
                h[j] += delta
                h = enforce_sum(h, target, rng)
            it += 1
            continue
        step = 0.5 * step_max
        h = h + step * d
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
        it += 1
        if it % 200 == 0:
            # occasional random nudge
            for _ in range(2):
                src = np.where(h > 1e-8)[0]
                dst = np.where(h < 1.0 - 1e-8)[0]
                if src.size == 0 or dst.size == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(h[i], 1.0 - h[j])
                delta = max_delta * rng.random() * 0.1
                h[i] -= delta
                h[j] += delta
            h = enforce_sum(h, target, rng)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None

    target = n / 2.0
    if h0 is not None:
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = random_continuous_vector(n, target, rng)

    best_h, best_val = gradient_descent_local(h0, target, deadline, rng)

    # Additional random restarts if time permits
    while time.time() < deadline - 5.0:
        h_rand = random_continuous_vector(n, target, rng)
        h_cand, val_cand = gradient_descent_local(h_rand, target, deadline, rng)
        if val_cand < best_val:
            best_val = val_cand
            best_h = h_cand.copy()
            if best_val < 0.3808:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
