def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):

    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximation on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation from arr (size m) to size new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=400):
    """Short SLSQP refinement for a candidate vector."""
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)

def time_remaining(start, budget):
    """Seconds left before budget runs out."""
    return max(0.0, budget - (time.time() - start))

def simulated_annealing(h0, start, budget, rng):
    """Simple SA that moves mass between two positions while preserving sum."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    T0, T1 = 0.2, 1e-4
    while True:
        remaining = time_remaining(start, budget)
        if remaining < 0.5:
            break
        T = T0 * (remaining / budget) + T1 * (1 - remaining / budget)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = min(max_delta, T * 5.0)
        if delta <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new.copy(), score
            h = h_new
        else:
            if rng.random() < np.exp((best_score - score) / (T + 1e-12)):
                h = h_new
        # occasional reprojection to avoid numeric drift
        if int(time.time() * 10) % 97 == 0:
            h = project_onto_box_sum(h, n / 2.0)
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- small scale stage -------------------------------------------------
    n_small = 64
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n_small), n_small / 2.0)
        candidates.append(cand)

    candidates.append(thue_morse_binary(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))

    best_h = None
    best_score = np.inf

    for cand in candidates:
        if time_remaining(start, budget_s) < 1.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    while time_remaining(start, budget_s) > 12.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=200)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = random_binary_balanced(n_small, rng)
        best_score = compute_c5(best_h)

    # --- multiscale refinement ---------------------------------------------
    sizes = [256, 512, 1024]
    for target_n in sizes:
        # upsample and project
        best_h = resample(best_h, target_n)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
        best_score = compute_c5(best_h)

        # gradient descent refinement
        if time_remaining(start, budget_s) > 6.0:
            max_iters = int(min(250000,
                                max(50000, time_remaining(start, budget_s) * 350)))
            time_end = start + budget_s - 0.4
            h_ref, sc_ref = gradient_descent_momentum(best_h,
                                                       step0=0.9,
                                                       beta=0.9,
                                                       time_end=time_end,
                                                       max_iter=max_iters)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

        # occasional simulated annealing if still time and bound not met
        if time_remaining(start, budget_s) > 8.0 and best_score > 0.3808:
            h_sa, sc_sa = simulated_annealing(best_h, start, budget_s, rng)
            if sc_sa < best_score:
                best_score, best_h = sc_sa, h_sa.copy()
            if best_score <= 0.3808:
                break

    # final safety projection
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
