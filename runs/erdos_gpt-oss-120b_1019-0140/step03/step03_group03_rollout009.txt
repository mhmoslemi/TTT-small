import time
import numpy as np
from scipy.signal import fftconvolve

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the interval brackets the target sum
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear up‑sampling from [0,1) domain."""
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear down‑sampling from [0,1) domain."""
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return the Thue‑Morse binary sequence of length n as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_c5(h):
    """C₅ = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    conv = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return conv.max() * (2.0 / n)


def gradient_descent_momentum(h0, max_iters, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    # initial score
    conv = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    best_score = conv.max() * (2.0 / n)
    momentum = np.zeros_like(h)
    eps = 1e-12

    for it in range(1, max_iters + 1):
        # correlation via FFT
        conv = fftconvolve(h, (1.0 - h)[::-1], mode='full')
        max_corr = conv.max()
        idxes = np.where(conv >= max_corr - eps)[0]
        shifts = idxes - (n - 1)

        # gradient of the max‑correlation term (averaged over tied shifts)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n - s]
                term2[:n - s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad = grad / len(shifts)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate new point
        conv_new = fftconvolve(h_new, (1.0 - h_new)[::-1], mode='full')
        score_new = conv_new.max() * (2.0 / n)

        if score_new < best_score:
            best_score, best_h = score_new, h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on a balanced binary vector."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # initial index sets
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        # exponential cooling in log‑temperature space
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, cur_h.copy()
            # update index lists
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score
