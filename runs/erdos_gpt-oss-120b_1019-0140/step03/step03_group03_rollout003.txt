def compute_gradient_max_corr(h):
    n = h.size
    raw_corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(raw_corr)
    shift = max_idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # i from 0 .. n-shift-1
        # grad[i] += 1 - h[i+shift]
        # grad[i+shift] += -h[i]
        i_range = slice(0, n - shift)
        grad[i_range] += 1 - h[i_range + shift]  # careful with broadcasting
        grad[i_range + shift] += -h[i_range]

def projected_gradient_descent(h0, max_iter, step0, beta):
    h = h0.copy()
    target_sum = h.size / 2
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        grad, shift, _ = compute_gradient_max_corr(h)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # ensure within [0,1], preserve sum via projection
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # handle trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initialise lam bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # define f(lam)
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure lam bounds enclose solution

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradient_max_corr(h):
    """Gradient of the maximal correlation term (scaled as in compute_c5)."""
    n = h.size
    raw = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(raw)
    shift = max_idx - (n - 1)
    grad = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(n - shift)
        grad[i] += 1.0 - h[i + shift]
        grad[i + shift] += -h[i]
    else:
        s = -shift
        i = np.arange(n - s)
        grad[i] += -h[i + s]
        grad[i + s] += 1.0 - h[i]

    grad *= (2.0 / n)
    return grad, shift, raw[max_idx] * (2.0 / n)


def projected_gradient_descent(h0, max_iter, step0, beta):
    """Projected gradient descent with momentum on C5."""
    h = h0.copy()
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iter + 1):
        grad, _, _ = compute_gradient_max_corr(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def upsample_arr(arr, new_n):

    return np.interp(x_new, x_old, arr)

            refined_h, refined_score = projected_gradient_descent(best_h,
                                                                 max_iter=max_iters,
                                                                 step0=0.3,
                                                                 beta=0.9)
            if refined_score < best_score:
                best_score = refined_score
                best_h = refined_h.copy()

    # Final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
