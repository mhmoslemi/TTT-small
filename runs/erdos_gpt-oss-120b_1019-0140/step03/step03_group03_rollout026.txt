def gradient_descent_momentum_smooth(h0, n_iter, step0=0.5, beta=0.9, epsilon=1e-3, alpha=50):
    h = h0.copy()
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1,...):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # active shifts within epsilon of max
        active = np.where(corr >= max_corr - epsilon)[0] - (h.size - 1)
        # compute weights via exp(alpha*(corr - max_corr))/ sum
        w_raw = np.exp(alpha * (corr[active + (h.size - 1)] - max_corr))
        w = w_raw / w_raw.sum()
        # compute gradient as weighted sum of per-shift gradients
        n = h.size
        grad = np.zeros_like(h)
        for idx_shift, w_shift in zip(active, w):
            s = idx_shift
            if s >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n - s] = h[s:]
                term2[s:] = h[:n - s]
            else:
                s2 = -s
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s2:] = h[:n - s2]
                term2[:n - s2] = h[s2:]
            grad += w_shift * (1.0 - term1 - term2)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def gradient_descent_weighted(h0, n_iter, step0=0.5, beta=0.9, eps_start=1e-3, eps_end=1e-12, alpha=50):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        # compute correlation (without scaling)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # epsilon schedule
        eps = eps_end + (eps_start - eps_end) * (1 - it / n_iter)
        active_idx = np.where(corr >= max_corr - eps)[0] - (n - 1)
        # weights
        weights_raw = np.exp(alpha * (corr[active_idx + (n - 1)] - max_corr))
        w_sum = weights_raw.sum()
        if w_sum == 0:
            w = np.full_like(weights_raw, 1.0 / len(weights_raw))
        else:
            w = weights_raw / w_sum
        # build w_shifts array
        w_shifts = np.zeros(2 * n - 1, dtype=float)
        w_shifts[active_idx + (n - 1)] = w
        # convolution
        conv1 = np.convolve(w_shifts, h, mode='full')
        conv2 = np.convolve(w_shifts[::-1], h, mode='full')
        conv1_slice = conv1[n-1:2*n-1]
        conv2_slice = conv2[n-1:2*n-1]
        # gradient = 1 - conv1 - conv2
        grad = 1.0 - conv1_slice - conv2_slice
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        # test if empty
        if ones.size == 0 or zeros.size == 0:
            break
        # temperature schedule (linear in log)
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update ones and zeros
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=600):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0,1.0)]*n + [(0.0,1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')*(2.0/n)
        return t - np.max(corr)
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as opt


def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def slsqp_optimize_candidate(h0, max_iter=600):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def gradient_descent_weighted(h0, n_iter, rng, step0=0.5, beta=0.9,
                             eps_start=1e-3, eps_end=1e-12,
                             alpha=80):
    """Projected gradient descent with weighted sub‑gradient of the max‑correlation."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        eps = eps_end + (eps_start - eps_end) * (1.0 - it / n_iter)
        idx = np.where(corr >= max_corr - eps)[0]
        weights_raw = np.exp(alpha * (corr[idx] - max_corr))
        w_sum = weights_raw.sum()
        if w_sum == 0:
            w = np.full_like(weights_raw, 1.0 / len(weights_raw))
        else:
            w = weights_raw / w_sum

        w_shifts = np.zeros_like(corr)
        w_shifts[idx] = w

        conv1 = np.convolve(w_shifts, h, mode='full')
        conv2 = np.convolve(w_shifts[::-1], h, mode='full')
        conv1_slice = conv1[n - 1:2 * n - 1]
        conv2_slice = conv2[n - 1:2 * n - 1]

        grad = 1.0 - conv1_slice - conv2_slice
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional random perturbation to escape plateaus
        if it % 1000 == 0:
            amp = 0.01 * (1.0 - it / n_iter)
            h_new += rng.normal(scale=amp, size=n)

        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on a balanced binary vector."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1: small‑scale exploration (n = 64)
    # ------------------------------------------------------------------
    n_small = 64
    small_cands = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_small:
            small = downsample(init_full, n_small)
        else:
            small = init_full.copy()
        small = project_onto_box_sum(small, n_small / 2.0)
        small_cands.append(small)

    small_cands.append(thue_morse(n_small))
    for _ in range(3):
        small_cands.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        small_cands.append(project_onto_box_sum(v, n_small / 2.0))

    # SLSQP refinements on the small scale
    for cand in small_cands:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=600)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Additional random SLSQP starts while time permits
    while time_left() > 5.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=500)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # ------------------------------------------------------------------
    # Phase 2: Upsample to medium resolution (n = 256)
    # ------------------------------------------------------------------
    n_mid = 256
    if best_h is None:
        h_mid = random_binary_balanced(n_mid, rng)
    else:
        h_mid = upsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    sc_mid = compute_c5(h_mid)
    if sc_mid < best_score:
        best_score, best_h = sc_mid, h_mid.copy()

    # ------------------------------------------------------------------
    # Phase 3: Gradient descent with weighted sub‑gradient on medium scale
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 6.0:
        max_iters = int(min(300000, remaining * 2000))
        if max_iters > 0:
            h_gd, sc_gd = gradient_descent_weighted(best_h, max_iters, rng,
                                                    step0=0.5, beta=0.9,
                                                    eps_start=1e-3, eps_end=1e-12,
                                                    alpha=80)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # Phase 4: Upsample to larger resolution (n = 512) if time allows
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 30.0:
        n_large = 512
        h_large = upsample(best_h, n_large)
        h_large = project_onto_box_sum(h_large, n_large / 2.0)
        sc_large = compute_c5(h_large)
        if sc_large < best_score:
            best_score, best_h = sc_large, h_large.copy()

        # Gradient descent on the large scale
        remaining = time_left()
        if remaining > 6.0:
            max_iters = int(min(250000, remaining * 1200))
            if max_iters > 0:
                h_gd2, sc_gd2 = gradient_descent_weighted(best_h, max_iters, rng,
                                                          step0=0.5, beta=0.9,
                                                          eps_start=1e-3, eps_end=1e-12,
                                                          alpha=80)
                if sc_gd2 < best_score:
                    best_score, best_h = sc_gd2, h_gd2.copy()

        # ------------------------------------------------------------------
        # Phase 5: Binary simulated annealing on the large scale
        # ------------------------------------------------------------------
        remaining = time_left()
        if remaining > 5.0:
            idx = np.argsort(best_h)[-n_large // 2:]
            h_bin = np.zeros(n_large, dtype=float)
            h_bin[idx] = 1.0
            max_iter_ann = int(remaining * 400)
            h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                          temp_start=0.05, temp_end=1e-5, rng=rng)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()

    # Final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
