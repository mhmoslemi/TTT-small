import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate thue-morse binary values 0/1 as floats
    seq = np.arange(n, dtype=np.uint64)
    bits = np.unpackbits(seq[:, None].view(np.uint8), axis=1)
    parity = bits.sum(axis=1) & 1
    return parity.astype(float)

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def compute_c5_fft(h, fft_len=None):
    n = h.shape[0]
    s = h.sum()
    if fft_len is None:
        fft_len = next_pow2(2 * n - 1)
    H = np.fft.rfft(h, fft_len)
    conv = np.fft.irfft(H * np.conj(H), fft_len)[:2 * n - 1]
    min_conv = conv.min()
    max_corr = s - min_conv
    return max_corr * (2.0 / n)

def gradient_descent_smooth(h0, n_iters, step0, alpha_start, alpha_end, rng, time_left):

    best_h = h0.copy()

    fft_len = next_pow2(3 * n - 2)
    for it in range(1, n_iters + 1):
        if time_left() <= 0:
            break
        progress = it / n_iters
        # increase smoothing parameter
        if alpha_end == alpha_start:
            alpha = alpha_start
        else:
            alpha = alpha_start * (alpha_end / alpha_start) ** progress
        # FFT of current h
        H = np.fft.rfft(h0, fft_len)
        conv = np.fft.irfft(H * np.conj(H), fft_len)[:2 * n - 1]
        s = h0.sum()
        corr = s - conv

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.arange(n, dtype=np.uint64)
    bits = np.unpackbits(seq[:, None].view(np.uint8), axis=1)
    parity = bits.sum(axis=1) & 1
    return parity.astype(float)

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def compute_c5_fft(h, fft_len=None):
    """Fast evaluation of C₅ = max_k ∫h(x)(1−h(x+k))dx on the grid."""
    n = h.shape[0]
    s = h.sum()
    if fft_len is None:
        fft_len = next_pow2(2 * n - 1)
    H = np.fft.rfft(h, fft_len)
    conv = np.fft.irfft(H * np.conj(H), fft_len)[:2 * n - 1]
    min_conv = conv.min()
    max_corr = s - min_conv
    return max_corr * (2.0 / n)
