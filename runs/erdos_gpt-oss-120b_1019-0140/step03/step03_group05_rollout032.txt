import time
import numpy as np

# ---------- Core utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    dx = 2.0 / h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def overlap_vector(h):
    """Raw overlap values (without dx) for all integer shifts."""
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

# ---------- Soft-max based gradient ----------
def softmax_weights(O, alpha):
    """Softmax weights for overlap vector O with temperature 1/alpha."""
    aO = alpha * O
    m = np.max(aO)
    w = np.exp(aO - m)
    w /= w.sum()
    return w

def gradient_softmax(h, w):
    """Gradient of the smooth max-overlap (softmax) objective."""
    n = h.size
    # convolution w * h  (shifted left) and w_rev * h (shifted right)
    conv_rev = np.convolve(w[::-1], h, mode='full')[n - 1:n - 1 + n]
    conv = np.convolve(w, h, mode='full')[n - 1:n - 1 + n]
    return 1.0 - (conv_rev + conv)

def gradient_descent_softmax(h_start, target_sum, deadline, max_iter=3000):
    """Projected gradient descent on the soft‑max approximation of max overlap."""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 5.0          # start temperature (low -> smooth)
    lr = 0.15            # learning rate
    for it in range(max_iter):
        if time.time() > deadline:
            break
        O = overlap_vector(h)
        cur_c5 = np.max(O) * (2.0 / h.size)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # soft‑max weights and gradient
        w = softmax_weights(O, alpha)
        grad = gradient_softmax(h, w)
        # SGD step with projection
        h_new = h - lr * grad
        h_new = project_to_box_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        # adapt learning rate
        if new_c5 < cur_c5:
            h = h_new
            lr = min(lr * 1.05, 1.0)
        else:
            lr = max(lr * 0.5, 1e-5)
        # slowly tighten the soft‑max
        alpha = min(alpha * 1.015, 500.0)
    return best_h, best_c5

def random_feasible(n, target_sum, rng):
    """Generate a random feasible vector."""
    v = rng.random(n)
    return project_to_box_sum(v, target_sum)

# ---------- Main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimensionality from parent if available
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = random_feasible(n, target_sum, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    safety = 5.0  # stop a bit early to avoid hard timeout
    while time.time() < start_time + budget_s - safety:
        # Generate a start point: perturb best or fresh random
        if rng.random() < 0.5:
            start = best_h + rng.normal(scale=0.025, size=n)
        else:
            start = rng.random(n)
        start = project_to_box_sum(np.clip(start, 0.0, 1.0), target_sum)

        # Run the soft‑max gradient descent
        deadline = start_time + budget_s - safety
        h_opt, c5_opt = gradient_descent_softmax(start, target_sum, deadline, max_iter=2500)

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break

    # Final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
