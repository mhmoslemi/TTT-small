import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def random_binary_vector(n, ones, rng):
    """Binary vector with exactly `ones` entries set to 1."""
    v = np.zeros(n, dtype=int)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1
    return v

def enforce_sum(v, target, rng):
    """Project `v` onto [0,1]^n while fixing the sum to `target` (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # simple proportional scaling
    if total > 0:
        v *= target / total
    v = np.clip(v, 0.0, 1.0)
    # final tiny correction on a random element
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(len(v))
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Use a moderately large discretisation for finer granularity
    n = 100
    target = n // 2
    dx = 2.0 / n
    target_val = 0.3808

    # Initialise a feasible binary vector
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.shape[0] == n:
            cur = (init > 0.5).astype(int)
            cur_sum = cur.sum()
            while cur_sum > target:
                i = rng.choice(np.where(cur == 1)[0])
                cur[i] = 0
                cur_sum -= 1
            while cur_sum < target:
                i = rng.choice(np.where(cur == 0)[0])
                cur[i] = 1
                cur_sum += 1
        else:
            cur = random_binary_vector(n, target, rng)
    else:
        cur = random_binary_vector(n, target, rng)

    cur = cur.astype(int)
    cur_val = compute_c5(cur)

    best = cur.copy()
    best_val = cur_val

    attempts_no_improve = 0
    max_no_improve = 3000
    total_iter = 0
    max_iter = 200000

    # Greedy / annealing loop focusing on the worst shift
    while time.time() < deadline - 0.2 and total_iter < max_iter:
        if best_val <= target_val:
            break
        corr = np.correlate(cur, 1 - cur, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        moved = False
        # Targeted swap: move a 1 to a 0 exactly k positions away
        if k >= 0:
            i_range = np.arange(n - k)
            mask_i = (cur[i_range] == 1) & (cur[i_range + k] == 0)
            i_cands = i_range[mask_i]
            if i_cands.size > 0:
                i = int(rng.choice(i_cands))
                j = i + k
                cur[i], cur[j] = cur[j], cur[i]
                new_val = compute_c5(cur)
                if new_val < cur_val - 1e-12:
                    cur_val = new_val
                    moved = True
                    if new_val < best_val - 1e-12:
                        best_val = new_val
                        best = cur.copy()
                        attempts_no_improve = 0
                else:
                    cur[i], cur[j] = cur[j], cur[i]
        else:
            shift = -k
            i_range = np.arange(shift, n)
            mask_i = (cur[i_range] == 1) & (cur[i_range - shift] == 0)
            i_cands = i_range[mask_i]
            if i_cands.size > 0:
                i = int(rng.choice(i_cands))
                j = i - shift
                cur[i], cur[j] = cur[j], cur[i]
                new_val = compute_c5(cur)
                if new_val < cur_val - 1e-12:
                    cur_val = new_val
                    moved = True
                    if new_val < best_val - 1e-12:
                        best_val = new_val
                        best = cur.copy()
                        attempts_no_improve = 0
                else:
                    cur[i], cur[j] = cur[j], cur[i]

        if not moved:
            # fallback random swap
            ones_idx = np.where(cur == 1)[0]
            zeros_idx = np.where(cur == 0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = int(rng.choice(ones_idx))
            j = int(rng.choice(zeros_idx))
            cur[i], cur[j] = cur[j], cur[i]
            new_val = compute_c5(cur)
            if new_val < cur_val - 1e-12:
                cur_val = new_val
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best = cur.copy()
                    attempts_no_improve = 0
            else:
                cur[i], cur[j] = cur[j], cur[i]
                attempts_no_improve += 1
        else:
            attempts_no_improve = 0

        if attempts_no_improve >= max_no_improve:
            cur = random_binary_vector(n, target, rng)
            cur_val = compute_c5(cur)
            attempts_no_improve = 0

        total_iter += 1

    # If time permits, run a single SLSQP refinement from the best binary vector
    if time.time() < deadline - 5.0 and best_val > target_val:
        h_start = best.astype(float)
        t0 = compute_c5(h_start)
        x0 = np.concatenate([h_start, [t0]])

        cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
        for k in range(-n + 1, n):
            if k == 0:
                continue
            cons.append({"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)})

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = minimize(fun=objective_fun, x0=x0, method="SLSQP",
                           bounds=bounds, constraints=cons,
                           options={"maxiter": 2000, "ftol": 1e-9, "disp": False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = enforce_sum(h_opt, target, rng)
                val_opt = compute_c5(h_opt)
                if val_opt < best_val:
                    best = h_opt.copy()
                    best_val = val_opt
        except Exception:
            pass

    final_h = best.astype(float)
    final_val = compute_c5(final_h)
    return final_h, float(final_val), n
