def de_optimize(h0, n, rng, time_left):
    # returns feasible h_best
    target_sum = n/2.0
    bounds = [(0.0,1.0)]*n

    # initial population: size popsize = 15
    popsize = 15
    # use Latin Hypercube initial + h0.
    # SciPy's DE will generate init automatically; we also want to include h0.
    init = np.empty((popsize, n))
    for i in range(popsize):
        init[i] = rng.random(n)  # uniform in [0,1]
    # replace first individual with h0
    init[0] = h0.copy()
    # Define cost:
    def cost(x):
        h = project_onto_box_sum(x, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * (2.0 / n)  # C5

    # To break early based on time left:
    def de_callback(xk, convergence):
        return time_left() <= 0

    # Determine max iterations based on time: each iteration is one generation (popsize evals). So approximate time per generation (popsize * cost eval time). We'll just set high maxiter, rely on callback.
    maxiter = 2000  # fallback high value.

    # Use differential_evolution:
    result = opt.differential_evolution(cost, bounds,
                                         strategy='best1bin',
                                         maxiter=maxiter,
                                         popsize=popsize,
                                         mutation=(0.5, 1.0),
                                         recombination=0.9,
                                         init=init,
                                         updating='deferred',  # maybe not needed
                                         polish=False,
                                         seed=int(rng.integers(2**32 - 1)),
                                         callback=de_callback,
                                         disp=False,
                                         tol=1e-7)
    h_best = project_onto_box_sum(result.x, target_sum)
    return h_best

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n=v.size
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10*hi

def gradient_descent_momentum(h0, n_iter, step0, beta, rng):

    n = h.size

    best_score = compute_c5(h)

        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
