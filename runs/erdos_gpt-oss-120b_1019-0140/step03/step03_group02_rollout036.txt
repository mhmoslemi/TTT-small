import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shifts from -(n-1) to (n-1)
    for idx, k in enumerate(range(-(n - 1), n)):
        wk = w[idx]
        if wk == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:]      -= wk * h[:n - k]
        else:
            p = -k
            grad[p:]      += wk * (1.0 - h[:n - p])
            grad[:n - p]  -= wk * h[p:]
    return grad

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    grad /= len(shifts)
    return grad

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # target bound

        init_up = upsample(init_full, n_target)

        candidates.append(init_up)
    candidates.append(thue_morse(n_target))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_target, rng))
    for _ in range(2):
        v = rng.random(n_target)
        candidates.append(project_onto_box_sum(v, n_target / 2.0))

    best_h = None
    best_score = float('inf')

    # Phase 1: smooth gradient descent on each candidate
    for h0 in candidates:
        if time_left() < 5.0:
            break
        # allocate iterations based on remaining time
        max_iters = int(min(8000, max(2000, time_left() * 6)))
        h_opt, sc_opt = smooth_grad_descent(
            h0, n_iter=max_iters, step0=0.7,
            alpha_start=0.5, alpha_end=30.0,
            beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= target_c5:
                break

    # If nothing found, start from a random balanced vector
    if best_h is None:
        h0 = random_binary_balanced(n_target, rng)
        h_opt, sc_opt = smooth_grad_descent(h0, n_iter=3000, step0=0.5,
                                            alpha_start=0.5, alpha_end=20.0,
                                            beta=0.9, time_left=time_left)
        best_h, best_score = h_opt, sc_opt

    # Phase 2: momentum refinement
    if time_left() > 2.0:
        iters = int(min(30000, max(5000, time_left() * 800)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                              step0=0.5, beta=0.9,
                                              time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # If still above target, try a smaller resolution and binary hill climb as fallback
    if best_score > target_c5 and time_left() > 1.0:
        # fallback to n=64 binary search
        n_small = 64
        # downsample best_h to small size
        h_small = upsample(best_h, n_small)
        h_small = project_onto_box_sum(h_small, n_small / 2.0)
        # binary hill climbing (simulated annealing)
        def binary_sa(h_init, rng, time_left, target_score):
            h = h_init.copy()
            n = h.size
            cur_score = compute_c5(h)
            best_h = h.copy()
            best_score = cur_score
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            T0 = 0.2
            start_local = time.time()
            while time_left() > 0.01:
                elapsed = time.time() - start_local
                prog = elapsed / max(1.0, time_left() + elapsed)
                T = max(1e-5, T0 * (1 - prog))
                if len(zeros) == 0 or len(ones) == 0:
                    break
                i = zeros[rng.integers(len(zeros))]
                j = ones[rng.integers(len(ones))]
                h_new = h.copy()
                h_new[i] = 1.0
                h_new[j] = 0.0
                new_score = compute_c5(h_new)
                delta = new_score - cur_score
                accept = False
                if delta < 0:
                    accept = True
                else:
                    accept = rng.random() < np.exp(-delta / T)
                if accept:
                    h, cur_score = h_new, new_score
                    zeros = np.where(h == 0)[0]
                    ones = np.where(h == 1)[0]
                    if new_score < best_score:
                        best_score, best_h = new_score, h_new.copy()
                        if best_score <= target_score:
                            break
            return best_h, best_score

        order = np.argsort(h_small)

        binary[order[n_small // 2:]] = 1.0
        bin_h, bin_sc = binary_sa(binary, rng, time_left, target_c5)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # final projection (ensures exact sum)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
