def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Scale all values proportionally (avoid division by zero)
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute diff iteratively
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v
```

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# Global placeholders for callbacks
__global_n = None
__global_target = None
__global_dx = None
__global_deadline = None
__global_start = None

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)})
    return cons

def de_obj(x):
    n = __global_n
    target = __global_target
    h = np.empty(n)
    h[:-1] = np.clip(x, 0.0, 1.0)
    sum_h = h[:-1].sum()
    h[-1] = target - sum_h
    low = h < 0.0
    high = h > 1.0
    penalty = 0.0
    if low.any():
        penalty += 1e6 * np.square(-h[low]).sum()
    if high.any():
        penalty += 1e6 * np.square(h[high] - 1.0).sum()
    diff = h.sum() - target
    penalty += 1e6 * diff**2
    c5 = compute_c5(np.clip(h, 0.0, 1.0))
    return c5 + penalty

def de_callback(xk, convergence):
    if time.time() > __global_deadline - 5:
        return True
    return False

def hill_climb(h, cur_val, rng, deadline):
    """Simple stochastic hill climbing with optional Metropolis acceptance."""
    global __global_start
    n = h.shape[0]
    target_local = n / 2.0
    best_h = h.copy()
    best_val = cur_val
    cur_h = h.copy()
    cur_val = cur_val
    while time.time() < deadline - 0.1:
        src = np.where(cur_h > 1e-12)[0]
        dst = np.where(cur_h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur_h[i] -= delta
        cur_h[j] += delta
        new_val = compute_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            elapsed = time.time() - __global_start
            total = deadline - __global_start
            remaining = total - elapsed
            T = max(remaining / total, 1e-6)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                cur_h[i] += delta
                cur_h[j] -= delta
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target_local, rng)
    return best_h, best_val

def local_optimize(start_h, n, target, dx, rng, deadline):

    if time.time() > deadline - 5:
        return None, None

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(start_h)
    x0 = np.concatenate([start_h, [t0]])
    try:

            fun=objective_fun,
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 1500, "ftol": 1e-10, "disp": False},
