def grad_logsumexp(h, alpha):
    n = len(h)
    # compute corr = correlate(h, 1 - h, mode='full') * (2/n)
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    # compute weights
    # To avoid overflow, subtract max before exp
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    w = exp_vals / np.sum(exp_vals)  # length 2n-1

    # compute A, B using convolution
    pad = np.zeros(n-1)
    a = np.convolve(w, np.concatenate([pad, 1 - h, pad]), mode='full')
    a = a[n-1:n-1+n]
    # reversed w
    w_rev = w[::-1]
    b = np.convolve(w_rev, np.concatenate([pad, h, pad]), mode='full')
    b = b[n-1:n-1+n]
    grad = a - b  # because B is - sum w_rev * h
    # gradient of smoothed objective f = dx * logsumexp(alpha * corr) / alpha:
    # f = (dx/alpha) * log(sum exp(alpha*corr))
    # df/dh = (dx/alpha) * sum_s w_s * d(corr_s)/dh = (dx/alpha) * grad
    # So we can compute gd = (dx/alpha) * grad
    return (2.0 / n) * (1.0 / alpha) * grad

def hill_climb_binary(h_bin, max_swaps, rng):
    n = len(h_bin)
    # Precompute best score initial
    best = h_bin.copy()
    best_score = compute_c5(best)
    # maintain lists of ones/zeros
    ones = np.flatnonzero(best)
    zeros = np.flatnonzero(~best.astype(bool))
    for _ in range(max_swaps):
        # random candidate
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        best[i] = 0.0
        best[j] = 1.0
        new_score = compute_c5(best)
        if new_score < best_score:
            best_score = new_score
            # update lists
            ones = np.flatnonzero(best)
            zeros = np.flatnonzero(~best.astype(bool))
        else:
            # revert
            best[i] = 1.0
            best[j] = 0.0
    return best, best_score

def gradient_descent_smooth(h0, n_iter, alpha, step0, beta):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    momentum = np.zeros_like(h)
    best_score = compute_c5(h)
    best_h = h.copy()
    for it in range(1, n_iter+1):

        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_grad(h, alpha):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    # compute weights
    maxc = np.max(corr)
    # shift scaling to avoid overflow
    exp_vals = np.exp(alpha * (corr - maxc))
    w = exp_vals / np.sum(exp_vals)
    # compute A & B using convolution via np.convolve
    pad = np.zeros(n - 1)
    a = np.convolve(w, np.concatenate([pad, 1 - h, pad]), mode='full')
    a = a[n-1:n-1+n]
    w_rev = w[::-1]
    b = np.convolve(w_rev, np.concatenate([pad, h, pad]), mode='full')
    b = b[n-1:n-1+n]
    grad = a - b
    # scale factor dx/alpha = (2/n)/alpha
    return (2.0 / n) * (1.0 / alpha) * grad

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    # Use linear interpolation on periodic domain [0,2):
    # but array is just sample points; simplistic use np.interp over linspace 0..1 with endpoint=False
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):

        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:

        if fm > s:
            lam_low = lam_mid
        else:

    x_new = np.linspace(0, 1, new_n, endpoint=False)

def hill_climb_binary(h, max_swaps, rng, start_time, budget_s):
    n = h.shape[0]
    best = h.copy()
    best_score = compute_c5(best)
    ones = np.flatnonzero(best > 0.5)
    zeros = np.flatnonzero(best < 0.5)
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s - 0.1:
            break
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best[i], best[j] = 0.0, 1.0
        new_score = compute_c5(best)
        if new_score < best_score:
            best_score = new_score
            ones = np.flatnonzero(best > 0.5)
            zeros = np.flatnonzero(best < 0.5)
        else:
            best[i], best[j] = 1.0, 0.0
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 512  # number of sample points

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init, n) if init.shape[0] != n else init.copy()
    else:
        h0 = rng.random(n)
    h0 = project_onto_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    alphas = [5, 10, 20, 40, 80]
    step0 = 0.5
    beta = 0.9
    for alpha in alphas:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break
        n_iter = int(min(2000, max(0, remaining * 1500)))
        if n_iter <= 0:
            break
        h_opt, sc_opt = gradient_descent_smooth(best_h, n_iter, alpha,
                                                step0, beta,
                                                start_time, budget_s)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        max_swaps = int(min(50000, remaining * 300))
        if max_swaps > 0:
            idx = np.argpartition(best_h, -n // 2)[-n // 2:]
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx] = 1.0
            h_bin, sc_bin = hill_climb_binary(h_bin, max_swaps, rng,
                                              start_time, budget_s)
            if sc_bin < best_score:
                best_score, best_h = sc_bin, h_bin.copy()

    best_h = project_onto_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
