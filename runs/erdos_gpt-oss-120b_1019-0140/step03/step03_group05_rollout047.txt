def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    n = 120
    target_sum = n / 2.0

    # generate initial h
    if 'initial_h_values' in globals():
        h0_small = np.asarray(initial_h_values, dtype=float)
        # interpolate to n points
        # use np.interp with points from 0..2
        xs_small = np.linspace(0, 2, h0_small.size, endpoint=False)
        xs = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(xs, xs_small, h0_small)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        h0 = project_to_box_sum(rng.random(n), target_sum)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # define constraints
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]

    # try initial SLSQP refinement
    t0 = best_c5
    res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=2000)
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # while time left, random restarts
    # allocate maybe up to 5 restarts each with SLSQP + small gradient descent
    while time.time() - start < budget_s - 5:
        # generate random start based on best_h with noise
        h_start = best_h + rng.normal(scale=0.05, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_to_box_sum(h_start, target_sum)

        # quick gradient descent (few steps)
        gd_h, gd_c5 = gradient_descent(h_start, target_sum, max_iter=2000,
                                       time_limit=start + budget_s - 5)
        # SLSQP refinement
        t_start = compute_c5(gd_h)
        res = slsqp_optimize(gd_h, t_start, sum_con, overlap_cons, maxiter=2000)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
        else:
            h_opt = gd_h
            c5_opt = gd_c5

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            # if meets target break
            if best_c5 < 0.3808:
                break

    # final projection ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)

    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    """
    Project v onto the hypercube [lo,hi]^n while fixing the sum to target.
    Uses a simple bisection on an additive offset.
    """
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without the dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality constraint: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    """Objective for SLSQP: minimise the slack variable t."""
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=2000):
    """Run a single SLSQP optimisation from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
    )
    return res

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """
    Projected gradient descent on the max-overlap objective.
    Uses the gradient of the worst shift only.
    """
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-15 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # gradient for the worst shift
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step and projection
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Choose discretisation size -----
    n = 120               # more points gives finer control
    target_sum = n / 2.0

    # ----- Initialise feasible h -----
    if 'initial_h_values' in globals():
        # interpolate the provided 80‑point construction to n points
        h_small = np.asarray(initial_h_values, dtype=float)
        xs_small = np.linspace(0.0, 2.0, h_small.size, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(xs, xs_small, h_small)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        h0 = project_to_box_sum(rng.random(n), target_sum)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ----- Prepare constraints for SLSQP -----
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    # ----- Quick initial refinement -----
    t0 = best_c5
    res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=2000)
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # ----- Main stochastic search loop -----
    safety_margin = 5.0
    while time.time() < start_time + budget_s - safety_margin:
        # Generate a new start point near the current best
        if rng.random() < 0.6:
            h_start = best_h + rng.normal(scale=0.04, size=n)
        else:
            h_start = rng.random(n)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_to_box_sum(h_start, target_sum)

        # Gradient descent phase (moderate iterations)
        gd_iters = 1500
        gd_deadline = start_time + budget_s - safety_margin
        gd_h, gd_c5 = gradient_descent(h_start, target_sum, gd_iters, gd_deadline)

        # SLSQP refinement from the gradient result
        t_start = compute_c5(gd_h)
        res = slsqp_optimize(gd_h, t_start, sum_con, overlap_cons, maxiter=2000)

        if res.success:
            h_candidate = np.clip(res.x[:n], 0.0, 1.0)
            h_candidate = project_to_box_sum(h_candidate, target_sum)
            c5_candidate = compute_c5(h_candidate)
        else:
            h_candidate = gd_h
            c5_candidate = gd_c5

        if c5_candidate < best_c5:
            best_h, best_c5 = h_candidate, c5_candidate
            if best_c5 < 0.3808:
                break  # goal achieved

    # ----- Final projection and evaluation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
