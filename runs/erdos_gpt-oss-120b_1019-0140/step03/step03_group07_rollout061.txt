def slsqp_refine(h, n, time_left):
    from scipy.optimize import minimize
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}] + [{'type':'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5_fft(h)
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(fun=lambda x: x[-1],
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return h, t0
    if not res.success:
        return h, t0
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # ensure sum constraint due to numeric errors

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    m = 1 << (2 * n - 1).bit_length()
    h_fft = np.fft.rfft(h, m)
    g_rev = (1.0 - h)[::-1]
    g_fft = np.fft.rfft(g_rev, m)
    conv = np.fft.irfft(h_fft * g_fft, m)[:2 * n - 1]
    corr = conv[::-1]
    return np.max(corr) * dx

class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:]) * self.dx
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk]) * self.dx
        return x[-1] - overl

def slsqp_refine(h, n, time_left):
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type':'eq','fun':sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]

                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 != 0:
            n0 = 80
    else:
        n0 = 80
    if n0 % 2 != 0:
        n0 += 1
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    alloc0 = remaining * 0.85
    init_h = None
    if 'initial_h_values' in globals() and len(initial_h_values) == n0:
        init_h = np.array(initial_h_values, dtype=float)
        init_h = np.clip(init_h, 0.0, 1.0)
        M0 = n0 // 2
        cur = init_h.sum()
        if abs(cur - M0) > 1e-12:
            init_h += (M0 - cur) / n0
            init_h = np.clip(init_h, 0.0, 1.0)
    best_h, best_c5 = search_n(n0, alloc0, rng, init_h)
    best_n = n0
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 15.0:
        n1 = best_n * 2
        if n1 > 200:
            n1 = 200
        if n1 % 2 != 0:
            n1 += 1
        h2, c5_2 = search_n(n1, remaining, rng, None)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n1
    return best_h, best_c5, best_n
