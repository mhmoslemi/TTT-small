def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n: return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    up = np.interp(x_new, x_old, arr)
    # Clip to [0,1] maybe
    return np.clip(up, 0, 1)
```

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n-shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n-shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n-s])
            grad[:n-s]    -= h[s:]
    grad /= len(shifts)
    return grad

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    up = np.interp(x_new, x_old, arr)
    return np.clip(up, 0.0, 1.0)
```

def thue_morse(n):
    # generate Thue-Morse sequence bits via parity of ones count in binary representation
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n-1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:]   -= w * h[:n-k]
        else:
            p = -k
            grad[p:]   += w * (1.0 - h[:n-p])

    return grad

        n += 1
    best_h = None
    best_score = np.inf
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        candidates.append(project_onto_box_sum(upsample(init, n), n / 2.0))
    base_len = 2 ** int(np.floor(np.log2(n)))
    candidates.append(project_onto_box_sum(upsample(thue_morse(base_len).astype(float), n), n / 2.0))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng).astype(float))
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))
    for h0 in candidates:
        if time_left() < 8:
            break
        h_opt, sc_opt = smooth_grad_descent(
            h0, n_iter=4000, step0=0.5,
            alpha_start=0.5, alpha_end=12.0,
            beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    if best_h is not None and time_left() > 12:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_sa(binary, rng, time_left, target=0.3808)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()
    if best_h is not None and time_left() > 2:
        iters = int(min(20000, max(2000, time_left() * 400)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                              step0=0.5, beta=0.9,
                                              time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
