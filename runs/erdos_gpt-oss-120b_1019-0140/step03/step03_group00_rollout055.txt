        if total > s:

        else:

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def simulated_annealing(h0, rng, start_time, time_left_func, total_budget):
    h_cur = h0.copy()
    best_h = h_cur.copy()
    cur_score = best_score = compute_c5(h_cur)
    n = h_cur.size
    T0 = 0.02
    Tmin = 1e-5
    k = -np.log(Tmin / T0) / total_budget
    base_max_move = 0.1
    it = 0
    while time_left_func() > 0.005:
        elapsed = time.time() - start_time
        T = max(Tmin, T0 * np.exp(-k * elapsed))
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_transfer = min(h_cur[i], 1.0 - h_cur[j])
        if max_transfer <= 0:
            continue
        max_delta = base_max_move * max_transfer * (T / T0)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        dE = new_score - cur_score
        if dE <= 0 or rng.random() < np.exp(-dE / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        it += 1
        if it % 20000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h_jit = np.clip(h_cur + jitter, 0.0, 1.0)
            h_jit = project_onto_box_sum(h_jit, n / 2.0)
            h_cur = h_jit
            cur_score = compute_c5(h_cur)
            if cur_score < best_score:
                best_h = h_cur.copy()
                best_score = cur_score
    return best_h, best_score

def improve_by_worst_shift(h, rng, time_left_func):
    n = h.size
    cur_score = compute_c5(h)
    while time_left_func() > 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        shift = shifts[0]
        if shift >= 0:
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n)
            term2 = np.zeros(n)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad = 1.0 - term1 - term2
        i_max = np.argmax(grad)
        i_min = np.argmin(grad)
        if grad[i_max] - grad[i_min] < 1e-10:
            break
        delta_max = min(h[i_max], 1.0 - h[i_min])
        if delta_max <= 0:
            break
        delta = delta_max * 0.5
        h[i_max] -= delta
        h[i_min] += delta
        new_score = compute_c5(h)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
        else:
            h[i_max] += delta
            h[i_min] -= delta
            break
    return h, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # base resolution
    base_n = 256
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    if base.shape[0] != base_n:
        base = upsample(base, base_n)
    base = project_onto_box_sum(base, base_n / 2.0)

    best_h = base
    best_score = compute_c5(best_h)

    # Simulated annealing on base grid
    if time_left() > 5:
        best_h, best_score = simulated_annealing(best_h, rng, start, time_left, budget_s)

    # Deterministic local improvement on worst shift
    if time_left() > 5:
        best_h, best_score = improve_by_worst_shift(best_h, rng, time_left)

    # Random balanced restarts if time permits
    n = best_h.shape[0]
    while time_left() > 30:
        h_rand = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand[idx] = 1.0
        h_rand += rng.normal(scale=0.02, size=n)
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = project_onto_box_sum(h_rand, n / 2.0)
        h_rand, sc_rand = simulated_annealing(h_rand, rng, start, time_left, budget_s)
        h_rand, sc_rand = improve_by_worst_shift(h_rand, rng, time_left)
        if sc_rand < best_score:
            best_h, best_score = h_rand.copy(), sc_rand
        if best_score <= 0.3808:
            break

    # Upsample to finer grid if time remains
    if time_left() > 20:
        finer_n = 512
        finer_h = upsample(best_h, finer_n)
        finer_h = project_onto_box_sum(finer_h, finer_n / 2.0)
        if time_left() > 5:
            finer_h, finer_score = simulated_annealing(finer_h, rng, start, time_left, budget_s)
            finer_h, finer_score = improve_by_worst_shift(finer_h, rng, time_left)
            if finer_score < best_score:
                best_h, best_score = finer_h, finer_score

    # Final safety projection and exact score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
