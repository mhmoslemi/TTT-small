def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    # compute linear correlation using FFT; pad to at least 2n
    size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    corr = np.fft.irfft(A * B.conj(), size)[:2*n-1]
    # due to floating errors may get ~1e-12 negative; clamp
    return np.max(corr) * dx
```

def anneal_binary(h_init, time_budget, rng):
    n = h_init.size
    M = n // 2
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    start = time.time()
    # Temperature schedule: we can compute total iterations approximate by budget / time_per_iter.
    # We'll implement while time left > some threshold.
    # Use exponential cooling: T = T0 * (1 - frac) where frac = (elapsed / time_budget)
    # T0 maybe set to (best_c5 - 0.37) maybe ~0.01; but safe set 0.02.
    while time.time() - start < time_budget - 0.01:
        # propose move
        i = rng.integers(n)
        # find a partner j of opposite value
        # try up to few attempts to find such j
        if cur_h[i] == 1:
            # pick a zero j
            zeros = np.where(cur_h == 0)[0]
            if zeros.size == 0: break
            j = rng.choice(zeros)
        else:
            ones = np.where(cur_h == 1)[0]
            if ones.size == 0: break
            j = rng.choice(ones)
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c5 = compute_c5(new_h)
        # acceptance
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(1e-6, 0.02 * (1 - frac))   # linear cooling
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
        # optionally occasionally do multi-swap if stuck
    return best_h, best_c5
```

def legendre_binary(n):
    p = n - 1
    # check primality quickly. Use simple trial division up to sqrt(p). n <=128, so fine.
    is_prime = True
    if p < 2:
        is_prime = False
    else:
        for d in range(2, int(p**0.5)+1):
            if p % d == 0:
                is_prime = False
                break
    if not is_prime:
        return None
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    seq[0] = 0.0  # Legendre sequence often defined with 0 for 0.
    # sum(seq) = (p-1)/2
    # Append a 1 to make length n and sum = ((p-1)/2)+1 = n/2
    h = np.concatenate([seq, [1.0]])
    # return as integer array
    return h.astype(np.int8)

def anneal_binary(h_init, time_budget, rng):
    n = h_init.shape[0]
    M = n // 2
    h = h_init.astype(np.int8).copy()
    # maintain list of indices
    ones = [i for i in range(n) if h[i] == 1]
    zeros = [i for i in range(n) if h[i] == 0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    # temperature schedule
    T0 = 0.02  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.01:
            break
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        # update lists
        i_pos = ones.index(i)
        j_pos = zeros.index(j)
        ones[i_pos] = j
        zeros[j_pos] = i
        new_c5 = compute_c5(h)
        # acceptance
        frac = elapsed / time_budget
        T = T0 * (1 - frac)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                # if best already below target, could break early? but we let continue
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            ones[i_pos] = i
            zeros[j_pos] = j
        # optionally occasional random reinitialization to avoid stagnation; if stuck for many iterations without improvement, do random restart
        # if we count iterations without improvement > threshold, reinitialize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1.0 - a
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    corr = np.fft.irfft(A * np.conj(B), size)[:2*n-1]
    return np.max(corr) * dx
```

    n = h_init.shape[0]
    h = h_init.astype(np.float64).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T0 = 0.05
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.01:
            break
        # choose two distinct indices
        i, j = rng.choice(n, 2, replace=False)
        # decide direction: try increase i, decrease j
        delta_max = min(1.0 - h[i], h[j])
        if delta_max > 0:
            delta = rng.random() * delta_max
            if delta == 0:
                continue
            h[i] += delta
            h[j] -= delta
            new_c5 = compute_c5(h)
            frac = elapsed / time_budget
            T = T0 * (1 - frac)
            if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):

                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

                h[i] -= delta
                h[j] += delta

        else:

def legendre_binary(n):
    p = n - 1
    # simple primality test
    if p < 2:
        return None
    for d in range(2, int(p**0.5) + 1):
        if p % d == 0:
            return None
    seq = np.zeros(p, dtype=np.float64)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    # seq[0] remains 0
    # sum(seq) = (p-1)/2 = (n-2)/2
    # append a 1 to reach n/2 ones
    h = np.concatenate([seq, [1.0]])
    return h.astype(np.int8)

def anneal_binary(h_init, time_budget, rng):
    n = h_init.shape[0]
    if time_budget <= 0.0:
        return h_init.copy(), compute_c5(h_init)
    h = h_init.astype(np.int8).copy()
    # maintain lists
    ones = [i for i, v in enumerate(h) if v == 1]
    zeros = [i for i, v in enumerate(h) if v == 0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T0 = 0.02  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.001:
            break
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        # update indices
        i_pos = ones.index(i)
        j_pos = zeros.index(j)
        ones[i_pos] = j
        zeros[j_pos] = i
        new_c5 = compute_c5(h)
        # acceptance
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if best_c5 < 0.3808:
                    # early exit: break loop
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            ones[i_pos] = i
            zeros[j_pos] = j
    return best_h, best_c5

def anneal_continuous(h_init, time_budget, rng):
    n = h_init.shape[0]
    if time_budget <= 0.0:
        return h_init.copy(), compute_c5(h_init)
    h = h_init.astype(np.float64).copy()
    # ensure sum constraint
    M = n / 2.0
    cur_sum = np.sum(h)
    if abs(cur_sum - M) > 1e-12:
        h += (M - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T0 = 0.05
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.001:
            break
        i, j = rng.choice(n, 2, replace=False)
        # attempt transfer from j to i
        delta_max = min(1.0 - h[i], h[j])
        if delta_max > 0:
            delta = rng.random() * delta_max
            if delta > 0:
                h[i] += delta
                h[j] -= delta
                new_c5 = compute_c5(h)
                frac = elapsed / time_budget
                T = T0 * (1.0 - frac)
                if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        if best_c5 < 0.3808:
                            break
                else:
                    h[i] -= delta
                    h[j] += delta
        else:
            # try opposite direction
            delta_max2 = min(1.0 - h[j], h[i])
            if delta_max2 > 0:
                delta = rng.random() * delta_max2
                if delta > 0:
                    h[i] -= delta
                    h[j] += delta
                    new_c5 = compute_c5(h)
                    frac = elapsed / time_budget
                    T = T0 * (1.0 - frac)
                    if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                        cur_c5 = new_c5
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                            if best_c5 < 0.3808:
                                break
                    else:
                        h[i] += delta
                        h[j] -= delta
        # occasional random restart if stagnant? Could skip
    return best_h, best_c5

    starts = []

    if leg_h is not None:
        shift = rng.integers(n)
        starts.append(np.roll(leg_h, shift))

    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h0 = np.array(initial_h_values, dtype=float)
        # ensure sum constraint
        cur = np.sum(h0)
        target = M
        if abs(cur - target) > 1e-12:
            h0 += (target - cur) / n
            h0 = np.clip(h0, 0.0, 1.0)
        starts.append(h0)
    # random binary
    rand_h = np.zeros(n, dtype=int)
    rand_h[rng.choice(n, M, replace=False)] = 1
    starts.append(rand_h)
    # allocate time for each start equally, but we will also allow restarts within each allocation
    num_starts = len(starts)
    # split total time into binary phase and continuous phase proportionate (e.g., 0.6 binary, 0.4 cont)
    binary_phase = time_budget * 0.6
    cont_phase = time_budget - binary_phase
    # binary phase: run anneal_binary with each start for a share of binary_phase
    per_start_budget = binary_phase / num_starts
    for h0 in starts:
        h_opt, c5_opt = annealing_binary_wrapper(h0, per_start_budget, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                return best_h, best_c5
    # continuous phase: start from best_h (if binary) or from a random float start if best_h is None
    if best_h is None:
        # fallback: random float start that satisfies sum M
        h0 = rng.random(n)
        h0 = h0 / np.sum(h0) * M
        h0 = np.clip(h0, 0.0, 1.0)
        best_h, best_c5 = anneal_continuous(h0, cont_phase, rng)
    else:
        # ensure sum is exactly M (binary already)
        best_h = best_h.astype(np.float64)
        best_h, best_c5 = anneal_continuous(best_h, cont_phase, rng)
    return best_h, best_c5

    M = n // 2
    start_time = time.time()
    # prepare binary starts
    binary_starts = []
    leg_h = legendre_binary(n)
    if leg_h is not None:
        binary_starts.append(np.roll(leg_h, rng.integers(n)))
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h_init = np.array(initial_h_values, dtype=float)
        # binarize
        h_bin = (h_init >= 0.5).astype(int)
        cur = h_bin.sum()
        if cur != M:
            diff = M - cur
            if diff > 0:
                zeros_idx = np.where(h_bin == 0)[0]
                chosen = rng.choice(zeros_idx, diff, replace=False)
                h_bin[chosen] = 1
            else:
                ones_idx = np.where(h_bin == 1)[0]
                chosen = rng.choice(ones_idx, -diff, replace=False)
                h_bin[chosen] = 0
        binary_starts.append(h_bin)
    # random binary start
    rand_h = np.zeros(n, dtype=int)
    rand_h[rng.choice(n, M, replace=False)] = 1
    binary_starts.append(rand_h)

    best_h = None
    best_c5 = np.inf

    # allocate time
    remaining = time_budget - (time.time() - start_time)
    if remaining <= 0:
        return np.full(n, 0.5), compute_c5(np.full(n, 0.5))
    binary_budget = remaining * 0.6
    cont_budget = remaining - binary_budget

    per_start = binary_budget / len(binary_starts)
    for h0 in binary_starts:
        h_opt, c5_opt = anneal_binary(h0, per_start, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                return best_h, best_c5

    # continuous phase
    # ensure best_h is float
    if best_h is None:
        # create random float start
        h0 = rng.random(n)
        h0 = h0 / np.sum(h0) * M
        h0 = np.clip(h0, 0.0, 1.0)

    h_opt, c5_opt = anneal_continuous(h0, cont_budget, rng)
    if c5_opt < best_c5:

                candidates.remove(n0)
            candidates = [n0] + candidates
    for idx, n in enumerate(candidates):
        now = time.time()
        if now - start > budget_s - 5:
            break
        remaining = budget_s - (now - start)
        # allocate fraction; earlier candidates get more
        if idx == 0:

        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break
    if best_h is None:
        # default solution
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        # final postprocess
        best_h = np.clip(best_h, 0.0, 1.0)
        n = best_n
        M = n // 2
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)

                zeros = np.where(h_bin == 0)[0]
                chosen = rng.choice(zeros, diff, replace=False)
                h_bin[chosen] = 1
            else:
                ones = np.where(h_bin == 1)[0]
                chosen = rng.choice(ones, -diff, replace=False)
                h_bin[chosen] = 0
        binary_starts.append(h_bin)
    rand = np.zeros(n, dtype=int)
    rand[rng.choice(n, M, replace=False)] = 1
    binary_starts.append(rand)

    best_h = None
    best_c5 = np.inf
    remaining = time_budget - (time.time() - start)
    if remaining <= 0:
        return np.full(n, 0.5), compute_c5(np.full(n, 0.5))
    binary_budget = remaining * 0.6
    cont_budget = remaining - binary_budget
    per_start = binary_budget / len(binary_starts)
    for h0 in binary_starts:
        h_opt, c5_opt = anneal_binary(h0, per_start, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                return best_h, best_c5
    if best_h is None:
        h0 = rng.random(n)
        h0 = h0 / np.sum(h0) * M
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        h0 = best_h.astype(np.float64)
    h_opt, c5_opt = anneal_continuous(h0, cont_budget, rng)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt
    cur = np.sum(best_h)
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None
    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 in candidates:
                candidates.remove(n0)
            candidates = [n0] + candidates
    for idx, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        if idx == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)
        h_opt, c5_opt = optimise_for_n(n, alloc, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        n = best_n
        M = n // 2
        cur = np.sum(best_h)
        if abs(cur - M) > 1e-9:
            best_h += (M - cur) / n
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), int(best_n)
