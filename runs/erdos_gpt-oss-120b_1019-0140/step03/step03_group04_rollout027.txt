```python
def compute_c5_fft(h):
    n = h.shape[0]
    f = 1.0 - h
    size = 2 * n
    a_fft = np.fft.rfft(h, size)   # real FFT half-size
    b_fft = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(a_fft * b_fft, size)
    # conv length = size = 2*n
    # correlation values length = 2*n-1 (ignore last element)
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

```python
def binary_local_search(h0, max_iters, rng, time_end):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    n = h.shape[0]
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)
    # maintain arrays of indices
    for it in range(max_iters):
        if time.time() > time_end:
            break
        # randomly pick swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0
        h[j] = 1
        # compute new score
        new_score = compute_c5_fft(h)
        if new_score < current_score:
            # accept improvement
            current_score = new_score
            # update index lists
            ones_idx[ones_idx == i] = j  # replace i with j
            zeros_idx[zeros_idx == j] = i
            # track best
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_score
```

```
def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    f = 1.0 - h
    size = 2 * n
    a = np.fft.rfft(h, size)
    b = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(a * b, size)
    corr = conv[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def correlation_fft(h):
    n = h.shape[0]
    f = 1.0 - h
    size = 2 * n
    a = np.fft.rfft(h, size)
    b = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(a * b, size)
    return conv[:2 * n - 1]

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = correlation_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def binary_local_search(h0, max_iters, rng, time_end):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    n = h.shape[0]
    for _ in range(max_iters):
        if time.time() > time_end:
            break
        while True:
            i = rng.integers(n)
            if h[i] == 1:
                break
        while True:
            j = rng.integers(n)
            if h[j] == 0:
                break
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score
