def smooth_gradient_descent(h0, n_iter, alpha_start, alpha_end, step_start, step_end):
    h = h0.copy()
    n = len(h)
    target_sum = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        # compute corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute weight exponent
        # linear schedule for alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it-1) / (n_iter-1)
        # compute weights
        w = np.exp(alpha * corr)
        w_sum = w.sum()
        # weighted forward/back
        term_fwd = np.correlate(w, h, mode='full')[(n-1):(n-1)+n]
        term_back = np.convolve(w, h, mode='full')[(n-1):(n-1)+n]
        grad = 1.0 - (term_fwd + term_back) / w_sum
        # step size schedule
        step = step_start + (step_end - step_start) * (it-1) / (n_iter-1)
        # gradient descent step (since we want minimize C5, but our gradient of smooth max objective is gradient of the smooth approximation; we want to go down gradient? The gradient of objective = d f / d h; we need to subtract gradient times step.
        h_new = h - step * grad
        # project onto box/sum
        h_new = project_onto_box_sum(h_new, target_sum)
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal_binary_improved(h0, max_iter, temp_start, temp_end, rng):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # precompute ones, zeros
    for it in range(max_iter):
        # cooling schedule: exponential
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # sample swap
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # broaden interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1‑h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear up‑sampling of a 1‑D array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue‑Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0

    alpha_start, alpha_end = 0.01, 0.5

        corr = np.correlate(h, 1.0 - h, mode='full')
        t_frac = (it - 1) / (max_iter - 1) if max_iter > 1 else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * t_frac
        step = step_start + (step_end - step_start) * t_frac

        # avoid overflow in exp
        cmax = corr.max()
        w = np.exp(alpha * (corr - cmax))
        w_sum = w.sum()

        term_fwd = np.correlate(w, h, mode='full')[(n - 1):(n - 1) + n]
        term_back = np.convolve(w, h, mode='full')[(n - 1):(n - 1) + n]

        grad = 1.0 - (term_fwd + term_back) / w_sum
        h = h - step * grad
        h = project_onto_box_sum(h, target_sum)

        if (it % 2000) == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()

    final_sc = compute_c5(h)
    if final_sc < best_score:
        best_score = final_sc
        best_h = h.copy()
    return best_h, best_score


def anneal_binary_improved(h0, max_iter, temp_start, temp_end, rng):
    """Simulated annealing on balanced binary vectors."""
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        elif rng.random() < np.exp(-delta / (t + 1e-12)):
            accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score


def binary_swap_hill_climb(h0, max_iter, rng):
    """Deterministic hill climbing on a balanced binary vector."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    n = h.size

    for _ in range(max_iter):
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    n = 512  # final resolution

    # -------------------- initial candidates --------------------
    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            cand = upsample(init, n)
        else:
            cand = init.copy()
        cand = project_onto_box_sum(cand, n / 2.0)
        candidates.append(cand)

    candidates.append(thue_morse(n))
    candidates.append(random_binary_balanced(n, rng))
    candidates.append(project_onto_box_sum(rng.random(n), n / 2.0))

    best_h = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # -------------------- smooth‑max gradient descent --------------------
    while time_left() > 30.0:
        # allocate iterations based on remaining time (≈800 it/s)
        iters = int(max(2000, time_left() * 800))
        start = best_h + rng.normal(scale=0.1, size=n)
        start = project_onto_box_sum(start, n / 2.0)
        h_opt, sc_opt = gradient_descent_h(start, iters, start_time, budget_s)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # -------------------- binary rounding & local improvement --------------------
    if time_left() > 5.0:
        idx = np.argsort(best_h)[-n // 2:]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()

        # hill climbing (deterministic)
        max_hc = int(time_left() * 400)
        if max_hc > 0:
            h_hc, sc_hc = binary_swap_hill_climb(best_h.copy(), max_hc, rng)
            if sc_hc < best_score:
                best_score = sc_hc
                best_h = h_hc.copy()

        # simulated annealing with a warmer start
        if time_left() > 10.0:
            max_ann = int(time_left() * 200)
            h_ann, sc_ann = anneal_binary_improved(best_h.copy(),
                                                   max_ann,
                                                   temp_start=0.5,
                                                   temp_end=1e-6,
                                                   rng=rng)
            if sc_ann < best_score:
                best_score = sc_ann
                best_h = h_ann.copy()

    # final safety projection and score
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
