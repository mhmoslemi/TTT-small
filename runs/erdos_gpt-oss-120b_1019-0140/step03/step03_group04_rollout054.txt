```
def hill_climb(h0, max_iters, time_end):
    h = h0.copy()
    n = h.size
    best_score = compute_c5(h)
    best_h = h.copy()
    # get indices of ones and zeros
    ones = np.flatnonzero(h==1)
    zeros = np.flatnonzero(h==0)
    rng_local = np.random.default_rng()
    for it in range(max_iters):
        if time.time() > time_end:
            break
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng_local.choice(ones)
        j = rng_local.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # compute score
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
            # update ones and zeros
            # after swap, i becomes 0, j becomes 1
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_score
```

```
def adam_optimize(h0, time_end, lr=0.1):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    while time.time() < time_end:
        t += 1
        # compute gradient for max shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps_corr = 1e-12
        idxes = np.where(corr >= max_corr - eps_corr)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
```

```
import time
import numpy as np

TARGET = 0.3808

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10*hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10*lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    # returns length p-1 binary sequence (0/1)
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        if pow(i, exp, p) == 1:
            seq[i - 1] = 1.0
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half-1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]

            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts) if shifts else grad

def adam_optimize(h0, time_end, lr0=0.2, target=TARGET):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while time.time() < time_end:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
        score = compute_c5(h)
        if score < best_score:

            best_h = h.copy()
            if best_score <= target:
                break
    return best_h, best_score

    return best_h, best_score
```
