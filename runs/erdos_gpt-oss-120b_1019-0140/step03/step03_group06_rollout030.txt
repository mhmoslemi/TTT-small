    if abs(total - target) > 1e-12:
        # scaling factor

        steps = 0

def gradient_step(h, target, lr, alpha, rng):
    n = h.shape[0]
    dx = 2.0 / n
    # compute corr full (without dx)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # overlaps with dx
    overlaps = corr * dx
    # compute softmax weights
    max_o = overlaps.max()
    exp_term = np.exp(alpha * (overlaps - max_o))
    weights = exp_term / exp_term.sum()
    # compute gradient
    grad = np.zeros_like(h)
    # shift range
    for s, w in enumerate(weights):
        k = s - (n - 1)
        if k >= 0:
            # first part
            if n - k > 0:
                grad[:n - k] += w * dx * (1.0 - h[k:])
                grad[k:] -= w * dx * h[:n - k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += w * dx * (1.0 - h[:n - kk])
                grad[:n - kk] -= w * dx * h[kk:]
    # step
    new_h = h - lr * grad
    # project
    new_h = enforce_sum(new_h, target, rng)
    return new_h

import time
import numpy as np

# ---------- helper utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Project v to [0,1] and adjust to have exact sum == target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # distribute the remaining discrepancy
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v


def random_continuous_vector(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with exact sum == target."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)


# ---------- gradient descent on smooth max ----------
def gradient_step(h: np.ndarray, target: float, lr: float, alpha: float,
                  rng: np.random.Generator) -> np.ndarray:
    """One projected gradient step for the smooth‑max objective."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation without dx
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx                     # include dx once
    # soft‑max weights
    m = overlaps.max()
    exp_term = np.exp(alpha * (overlaps - m))
    weights = exp_term / exp_term.sum()
    # gradient accumulation
    grad = np.zeros_like(h)
    for s, w in enumerate(weights):
        k = s - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += w * dx * (1.0 - h[k:])
                grad[k:]   -= w * dx * h[:nk]
        else:
            kk = -k
            nk = n - kk
            if nk > 0:
                grad[kk:]   += w * dx * (1.0 - h[:nk])
                grad[:nk]   -= w * dx * h[kk:]
    new_h = h - lr * grad
    return enforce_sum(new_h, target, rng)


# ---------- targeted refinement (simple annealing) ----------
def targeted_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    deadline: float) -> np.ndarray:
    """Move mass away from the worst shift to reduce the max overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.05:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)

        # locate the index i contributing most to this overlap
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i_local = np.argmax(prod)
        i = i_range[i_local]

        # choose a destination j with headroom
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0:
            break
        j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            break
        delta = rng.random() * max_transfer * 0.5  # modest move
        h[i] -= delta
        h[j] += delta
        h = enforce_sum(h, target, rng)

        if compute_c5(h) < 0.3808:
            break
    return h


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = random_continuous_vector(n, n / 2.0, rng)

    target = n / 2.0
    h0 = enforce_sum(h0, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # parameters for the smooth‑max gradient descent
    lr = 0.2                 # initial learning rate
    alpha = 1.0              # start with a soft approximation

    while time.time() < deadline - 0.5:
        # occasionally restart from a fresh random point
        if rng.random() < 0.1:
            h = random_continuous_vector(n, target, rng)
        else:
            h = best_h.copy()

        cur_val = compute_c5(h)

        # inner gradient descent loop
        inner_iters = 0
        while time.time() < deadline - 0.2 and inner_iters < 500:
            new_h = gradient_step(h, target, lr, alpha, rng)
            new_val = compute_c5(new_h)

            if new_val < cur_val - 1e-7:
                # successful step
                h = new_h
                cur_val = new_val
                lr = min(lr * 1.02, 1.0)   # slowly enlarge step if we are improving
            else:
                # reject, shrink step
                lr = lr * 0.5
                if lr < 1e-7:
                    break

            # tighten the smooth‑max
            alpha = min(alpha * 1.01, 200.0)
            inner_iters += 1

        if cur_val < best_val:
            best_h = h.copy()
            best_val = cur_val
            if best_val < 0.3808:
                break

    # final targeted refinement (light annealing)
    if time.time() < deadline - 0.2:
        best_h = targeted_refine(best_h, target, rng, deadline)

    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
