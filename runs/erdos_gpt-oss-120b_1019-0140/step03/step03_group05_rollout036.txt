def run(...):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    n = 80

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # hill climbing
    while time.time() - start_time < budget_s - 5.0:
        # perturbation magnitude
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 5.0
        # adapt noise scale: start big, shrink
        # compute scale factor based on elapsed fraction
        fraction = elapsed / (budget_s - 5.0)
        scale = (0.1 * (1 - fraction) + 0.005)   # between 0.005 and 0.1
        # generate noise that sums to zero (to keep sum target approx)
        noise = rng.normal(scale=scale, size=n)
        # project after adding
        cand = best_h + noise
        cand = np.clip(cand, 0.0, 1.0)
        # adjust sum to target via projection
        cand = project_to_box_sum(cand, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 + 1e-12 < best_c5:
            best_h, best_c5 = cand, cand_c5
            # if improved significantly, maybe do local gradient descent
            # run a few GD steps starting from cand
            gd_h, gd_c5 = gradient_descent(best_h, target_sum,
                                           max_iter=500, # small due time
                                           time_limit=time.time()+min(remaining,2.0))
            if gd_c5 < best_c5:

    best_c5 = compute_c5(best_h)

    n = h_start.shape[0]

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Euclidean projection of v onto [lo,hi]^n intersect {sum = target}."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Subgradient of the worst overlap w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

# ----------------------------------------------------------------------
# Gradient descent with momentum
# ----------------------------------------------------------------------
def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    n = h_start.shape[0]
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        alpha = 0.5 / np.sqrt(it + 1.0)  # decreasing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = -alpha * momentum
        new_h = h + step
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
        else:
            # occasional tiny random shake to get out of flat regions
            if rng.random() < 0.001:
                h = project_to_box_sum(
                    np.clip(h + rng.normal(scale=0.02, size=n), 0.0, 1.0),
                    target_sum,
                )
    return best_h, best_c5

# ----------------------------------------------------------------------
# Local swap search (preserves sum exactly)
# ----------------------------------------------------------------------
def local_swap_search(h, target_sum, max_swaps, time_limit, rng):
    n = h.shape[0]
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    eps_max = 0.05
    for s in range(max_swaps):
        if time.time() > time_limit:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = rng.uniform(-eps_max, eps_max)
        if not (0.0 <= cur_h[i] + eps <= 1.0 and 0.0 <= cur_h[j] - eps <= 1.0):
            continue
        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
    return cur_h, cur_c5

# ----------------------------------------------------------------------
# SLSQP helpers
# ----------------------------------------------------------------------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=800):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Dimension and initial feasible vector
    # ------------------------------------------------------------------
    n = 80
    target_sum = n / 2.0

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            # resample to required size if parent uses a different discretisation
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        h0 = project_to_box_sum(rng.random(n), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Search loop
    # ------------------------------------------------------------------
    safety = 5.0                     # keep a buffer before full timeout
    deadline = start_time + budget_s - safety

    # pre‑create constraint objects for occasional SLSQP refinements
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]

    while time.time() < deadline:
        # ----------------------------------------------------------------
        # Generate a fresh start point
        # ----------------------------------------------------------------
        if rng.random() < 0.5:
            h_start = best_h + rng.normal(scale=0.04, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        # ----------------------------------------------------------------
        # Subgradient descent
        # ----------------------------------------------------------------
        gd_h, gd_c5 = gradient_descent(
            h_start, target_sum,
            max_iter=2000,
            time_limit=deadline,
            rng=rng
        )

        # ----------------------------------------------------------------
        # Light local swap optimisation
        # ----------------------------------------------------------------
        sw_h, sw_c5 = local_swap_search(
            gd_h, target_sum,
            max_swaps=600,
            time_limit=deadline,
            rng=rng
        )

        # ----------------------------------------------------------------
        # Accept if better
        # ----------------------------------------------------------------
        if sw_c5 < best_c5:
            best_h, best_c5 = sw_h, sw_c5
            if best_c5 < 0.3808:
                break

        # ----------------------------------------------------------------
        # Occasionally run a cheap SLSQP refinement (if time permits)
        # ----------------------------------------------------------------
        if time.time() < deadline - 5.0:
            t_start = best_c5
            res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=300)
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    if best_c5 < 0.3808:
                        break

    # ------------------------------------------------------------------
    # Final polishing phase
    # ------------------------------------------------------------------
    # Final SLSQP refinement with higher iteration budget
    t_start = best_c5
    res = slsqp_optimize(best_h, t_start, sum_con, overlap_cons, maxiter=600)
    if res.success:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_box_sum(h_opt, target_sum)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final gradient descent polish (short time window)
    final_deadline = time.time() + 4.0
    final_h, final_c5 = gradient_descent(
        best_h, target_sum,
        max_iter=3000,
        time_limit=final_deadline,
        rng=rng
    )
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    # Ensure feasibility exactly before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
