def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Phase 0: determine base size (n0)
    n0 = 128

    # Build initial candidates list
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # downsample or upsample to n0
        if init_full.size != n0:
            # up/down
            candidates.append(project_onto_box_sum(downsample(init_full, n0), n0/2.0))
        else:
            candidates.append(project_onto_box_sum(init_full, n0/2.0))
    candidates.append(thue_morse(n0))
    for _ in range(3):
        candidates.append(random_binary_balanced(n0, rng))
    for _ in range(3):
        v = rng.random(n0)
        candidates.append(project_onto_box_sum(v, n0/2.0))

    # Phase 1: SLSQP with more iterations
    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=2000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= target:
            break

    # Phase 2: upsample to larger n
    n_large = 256
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n0, 0.5), n0/2.0)
        best_score = compute_c5(best_h)
    # upsample using np.interp
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large/2.0)
    sc_up = compute_c5(h_up)
    if sc_up < best_score:
        best_score = sc_up
        best_h = h_up.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Phase 3: continuous gradient descent with momentum
    if time_left() > 2.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left, max_iter=200000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 4: binary conversion (take top half)
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Phase 5: extended binary annealing
    if time_left() > 1.0:
        # schedule longer
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=800000)
        if sc_bin_ref < best_score:
            best_score = sc_bin_ref
            best_h = h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 6: greedy local improve
    if time_left() > 0.5:
        h_greedy, sc_greedy = greedy_binary_improve(best_h, time_left)
        if sc_greedy < best_score:
            best_score = sc_greedy
            best_h = h_greedy.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Phase 7: random continuous restarts with gradient descent
    n_current = best_h.size
    while time_left() > 5.0:
        v = rng.random(n_current)

        if sc_tmp < best_score:
            best_score = sc_tmp
            best_h = h_tmp.copy()
        if best_score <= target:
            break

    # final guard
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    if time_left() > 1.0:
        h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=800000)
        if sc_bin_ref < best_score:
            best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    if time_left() > 0.5:
        h_hc, sc_hc = hill_climbing_binary(best_h, rng, time_left, max_trials=200000)
        if sc_hc < best_score:
            best_score, best_h = sc_hc, h_hc.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    n_cur = best_h.size
    while time_left() > 5.0:
        h_rand = random_binary_balanced(n_cur, rng)
        h_rand, sc_rand = hill_climbing_binary(h_rand, rng, time_left, max_trials=20000)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
