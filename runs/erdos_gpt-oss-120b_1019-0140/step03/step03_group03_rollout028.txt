import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation of a periodic‑non‑wrapping array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones uniformly at random."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for the given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------
    # Choose resolution and initialise
    # ------------------------------------------------------------
    n = 512
    dx = 2.0 / n

    try:
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init_full, n) if init_full.size != n else init_full.copy()
    except NameError:
        h0 = random_binary_balanced(n, rng).astype(float)

    h0 = project_onto_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # binary rounding of the initial candidate
    idx = np.argpartition(best_h, -n // 2)[-n // 2 :]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    score_bin = compute_c5(h_bin)
    if score_bin < best_score:
        best_score, best_h = score_bin, h_bin.copy()

    # ------------------------------------------------------------
    # Smooth‑max gradient descent with restarts
    # ------------------------------------------------------------
    schedule = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    max_T = schedule[0]
    base_iter = 80                     # iterations for T = max_T
    step0 = 0.5

    while time_left() > 12.0:
        # starting point for this restart
        if rng.random() < 0.5:
            noise = rng.normal(scale=0.07, size=n)
            h_start = best_h + noise
            h_start = np.clip(h_start, 0.0, 1.0)
        else:
            h_start = random_binary_balanced(n, rng).astype(float)
        h_start = project_onto_box_sum(h_start, n / 2.0)

        h_cur = h_start.copy()
        it_counter = 1

        for T in schedule:
            it_per_T = int(base_iter * (max_T / T))
            for _ in range(it_per_T):
                if time_left() < 1.0:
                    break
                # correlation (raw) and softmax weights
                corr_raw = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                corr = corr_raw * dx
                max_corr = np.max(corr)
                exps = np.exp((corr - max_corr) / T)
                w = exps / np.sum(exps)

                # forward and backward convolution contributions
                forward = np.convolve(w[::-1], h_cur, mode='full')
                back = np.convolve(w, h_cur, mode='full')
                off = n - 1
                grad = 1.0 - forward[off : off + n] - back[off : off + n]

                eta = step0 * (T / max_T) / np.sqrt(it_counter)
                h_new = h_cur - eta * grad
                h_new = project_onto_box_sum(h_new, n / 2.0)
                h_cur = h_new
                it_counter += 1
            if time_left() < 1.0:
                break

        # evaluate continuous candidate
        cur_score = compute_c5(h_cur)
        if cur_score < best_score:
            best_score, best_h = cur_score, h_cur.copy()

        # evaluate its binary rounding
        idx = np.argpartition(h_cur, -n // 2)[-n // 2 :]
        h_bin = np.zeros_like(h_cur)
        h_bin[idx] = 1.0
        score_bin = compute_c5(h_bin)
        if score_bin < best_score:
            best_score, best_h = score_bin, h_bin.copy()

    # ------------------------------------------------------------
    # Final momentum gradient refinement (if time permits)
    # ------------------------------------------------------------
    remaining = time_left()
    if remaining > 6.0:
        max_iters = int(min(200000, remaining * 1500))
        h_ref, score_ref = gradient_descent_momentum(best_h, n_iter=max_iters,
                                                     step0=0.5, beta=0.9)
        if score_ref < best_score:
            best_score, best_h = score_ref, h_ref.copy()

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
