def subgradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # active shifts
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        # term1: 1 for all i
        # term2: -h[i+s] for i where i+s < n
        # term3: -h[i-s] for i where i-s >=0
        # So we can compute using vectorized ops
        # For shift s >=0
        if s >= 0:
            grad[:n] += 1.0
            grad[:n - s] -= h[s:]  # subtract h[i+s] term for i==0..n-s-1
            grad[s:] -= h[:n - s] # subtract h[i-s] term for i==s..n-1
        else:
            s2 = -s
            grad[:n] += 1.0
            # i+s >=0 -> i >= s2 -> i from s2..n-1, term -h[i+s] = -h[i - s2] (since s negative)
            grad[s2:] -= h[:n - s2] # subtract h[i+s]
            # i-s >=0 -> i >= s2? For s negative, i - s = i + s2, condition i + s2 < n => i < n - s2
            grad[:n - s2] -= h[s2:] # subtract h[i-s]
    grad /= len(shifts)  # average
    return grad

        zeros = np.where(current_h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        current_h[i] = 0.0
        current_h[j] = 1.0
        new_score = compute_c5(current_h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):

        if accept:
            current_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # revert
            current_h[i] = 1.0
            current_h[j] = 0.0
        T = max(T_min, T * decay)
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if s <= lo * n:

        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # binary search for lam to satisfy sum = s
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def binary_anneal(h0, rng, time_left, max_iters=500000):
    """Simulated‑annealing on binary vectors with exact sum constraint."""
    n = h0.size
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    T = 0.3
    T_min = 1e-6
    decay = 0.99995
    it = 0
    while it < max_iters and time_left() > 0.001:
        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] != 1.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0.0:
            j = rng.integers(n)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
        if it % 1000 == 0 and time_left() < 0.0:
            break
    return best_h, best_score

def interp_to_n(arr, n_target):
    """Linear interpolation of a 1‑D array to n_target points on [0,2)."""
    n_src = arr.shape[0]
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_tgt, x_src, arr)

def binary_from_continuous(v, n):
    """Convert a vector to a balanced binary vector of length n."""
    idx = np.argpartition(-v, n // 2)[:n // 2]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h
