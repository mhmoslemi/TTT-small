        max_corr = np.max(corr)

def compute_corr_full(h):

    fft_b = np.fft.fft((1 - h)[::-1], n_fft)
    corr = np.fft.ifft(fft_h * fft_b).real[:m]

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_full(h):
    """Linear correlation of h with (1-h) (full mode) using FFT."""
    n = h.size
    m = 2 * n - 1
    n_fft = 1 << (m - 1).bit_length()
    fft_h = np.fft.fft(h, n_fft)
    fft_b = np.fft.fft((1.0 - h)[::-1], n_fft)
    corr = np.fft.ifft(fft_h * fft_b).real[:m]
    # Numerical noise may introduce tiny negative values
    corr = np.maximum(corr, 0.0)
    return corr

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = compute_corr_full(h)
    return np.max(corr) * (2.0 / n)

def subgradient_max_corr(h):
    """Subgradient of the max‑correlation objective."""
    n = h.size
    corr = compute_corr_full(h)
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] += -h[:n - s]
        else:
            d = -s
            grad[d:] += 1.0 - h[:n - d]
            grad[:n - d] += -h[d:]
    grad = grad / len(shifts)
    return grad, max_corr

def gradient_descent_momentum(h0, step0=1.0, beta=0.9, max_iter=200000, time_end=None):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        grad, _ = subgradient_max_corr(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def simulated_annealing(h0, rng, time_end):
    """Simple SA that preserves the sum by moving mass between two coordinates."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # initial temperature based on current score
    T0 = max(0.01, cur_score * 0.5)
    it = 0
    while time.time() < time_end:
        it += 1
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        delta = min(0.02, h[i], 1.0 - h[j])
        if delta <= 0.0:
            continue
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        elapsed = time.time() - (time_end - (time_end - time.time()))
        # linear cooling
        T = max(1e-6, T0 * (1.0 - (time.time() - (time_end - (time_end - time.time()))) / (time_end - (time_end - (time_end - time.time())))))
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score
