def anneal_for_n(n, time_budget, rng):
    M = n // 2
    # initial binary
    if n == 80:
        # legendre with random shift
        p = 79
        leg = legendre_sequence(p)
        h = np.concatenate([leg, [1.0]])
        # ensure binary int
        h = (h > 0.5).astype(float)
    else:
        # random binary with exactly M ones
        idx = rng.choice(n, M, replace=False)
        h = np.zeros(n)
        h[idx] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    curr_h = h.copy()
    curr_c5 = best_c5

    start = time.time()
    # parameters
    iter = 0
    max_iter = int(1e6)  # just as cap
    # compute approximate scale for T
    T0 = 0.001  # may need adjust bigger?
    while time.time() - start < time_budget and iter < max_iter:
        # neighbor
        # pick random 1 and 0 indices
        ones = np.where(curr_h > 0.5)[0]
        zeros = np.where(curr_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Swap
        new_h = curr_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)
        delta = new_c5 - curr_c5
        # Acceptance
        # temperature schedule
        T = T0 * (1 - iter / max_iter)
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            curr_h = new_h
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        iter += 1
    return best_h, best_c5
```

        return h0, compute_c5(h0)
    M = n // 2

    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n+1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

# ---------- constraints for SLSQP ----------
class SumConstraint:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def objective_t(x):
    return x[-1]

# ---------- local refinement using SLSQP ----------
def slsqp_refine(h0, n, time_left):
    if time_left < 0.7:
        return h0, compute_c5(h0)
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    try:
        res = minimize(fun=objective_t,
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 1200, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        return h0, compute_c5(h0)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    M = n // 2
    if abs(cur - M) > 1e-9:
        h_opt += (M - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

# ---------- simulated annealing for a given resolution ----------
def solve_n(n, budget, rng):
    M = n // 2
    # initial construction
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        cur = np.sum(h)
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
    elif n == 80:
        p = 79
        leg = legendre_sequence(p)
        h = np.concatenate([leg, [1.0]])
        h = (h > 0.5).astype(float)
        h = np.roll(h, rng.integers(n))
    else:
        idx = rng.choice(n, M, replace=False)
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    T0 = 0.005  # initial temperature
    # main SA loop
    while time.time() - start < budget - 0.5:
        # temperature decays linearly with elapsed fraction
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / budget)

        # propose a binary swap (preserves sum)
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = compute_c5(h_new)
        delta = c5_new - best_c5
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            best_h, best_c5 = h_new, c5_new

        # occasional local SLSQP refinement
        if rng.random() < 0.02:
            h_ref, c5_ref = slsqp_refine(best_h, n, budget - (time.time() - start))
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

        if best_c5 < 0.3808:
            break

    # final refinement before returning
    h_opt, c5_opt = slsqp_refine(best_h, n, budget - (time.time() - start))
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    rng = np.random.default_rng(seed)

    # determine candidate resolutions
    candidates = [80, 96, 112, 128]
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates.insert(0, n0)
            else:
                candidates.remove(n0)
                candidates.insert(0, n0)

    # sort remaining (non‑parent) candidates descending to favour larger n
    parent_first = candidates[:1] if 'initial_h_values' in globals() else []
    rest = sorted(candidates[1:] if parent_first else candidates, reverse=True)
    ordered = parent_first + rest

    best_h = None
    best_c5 = np.inf
    best_n = None

    for idx, n in enumerate(ordered):
        elapsed = time.time() - start_total
        if elapsed > budget_s - 1.0:
            break
        remaining = budget_s - elapsed
        # allocate time: first candidate gets about half, rest share rest
        if idx == 0:
            alloc = min(remaining * 0.55, 600.0)
        else:
            alloc = min(remaining * 0.45 / (len(ordered) - idx), 300.0)
        h_n, c5_n = solve_n(n, alloc, rng)
        if h_n is not None and c5_n < best_c5:
            best_h, best_c5, best_n = h_n, c5_n, n
            if best_c5 < 0.3808:
                break

    # fallback if nothing found
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # final safety adjustment
    M = best_n // 2
    cur = np.sum(best_h)
    if abs(cur - M) > 1e-9:
        best_h += (M - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
