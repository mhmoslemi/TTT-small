def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts attaining max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for these shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[:n - shift] = h[shift:]
                term2[shift:] = h[:n - shift]
            else:
                sft = -shift
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[sft:] = h[:n - sft]
                term2[:n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        # update momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional small random jitter occasionally
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
```

    best_score = np.inf

    # Define scales
    scales = [128, 256, 512, 1024]
    for n in scales:

        if budget_s - (time.time() - start_time) < 5.0:
            break
        # upsample base
        base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Multiple gradient descent runs
        # Determine iteration count based on size and remaining time
        remaining = budget_s - (time.time() - start_time)
        # generous iteration count
        iters = int(min(30000, max(5000, remaining * (10 if n <= 512 else 5))))
        for _ in range(2):
            if budget_s - (time.time() - start_time) < 5.0:
                break
            h0 = base + rng.normal(scale=0.02, size=n)

                                             rng=rng, start_time=start_time, budget_s=budget_s)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        # occasional SLSQP for small n
        if n <= 256 and budget_s - (time.time() - start_time) > 5.0:
            h_opt, sc_opt = slsqp_optimize_candidate(base, max_iter=300)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

    # If no best found, fallback to target
    if best_h is None:
        n = scales[-1]
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    # Upsample to larger size if time permits

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type':'eq', 'fun':eq_sum},
            {'type':'ineq', 'fun':ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def gradient_descent(h0, n_iter, step0, beta, rng, start_time, budget_s):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        # time check
        if budget_s - (time.time() - start_time) <= 0:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient
        grad = compute_average_gradient(h, shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional jitter
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # projection
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def thue_morse(n):
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if budget_s - (time.time() - start_time) <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_average_gradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return t - corr
    cons = [{"type": "eq", "fun": eq_sum},
            {"type": "ineq", "fun": ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_local_search(h, rng, start_time, budget_s):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while budget_s - (time.time() - start_time) > 0.5:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_h, best_score = h_new, sc_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = None
    best_score = np.inf
    scales = [128, 256, 512, 1024]

    for n in scales:
        if budget_s - (time.time() - start_time) < 5.0:
            break
        base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        remaining = budget_s - (time.time() - start_time)
        iters = int(min(30000, max(5000, remaining * (10 if n <= 512 else 5))))

        for _ in range(2):
            if budget_s - (time.time() - start_time) < 5.0:
                break
            h0 = base + rng.normal(scale=0.02, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = gradient_descent(h0, n_iter=iters, step0=0.6,
                                             beta=0.9, rng=rng,
                                             start_time=start_time,
                                             budget_s=budget_s)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

        if n <= 256 and budget_s - (time.time() - start_time) > 5.0:
            h_opt, sc_opt = slsqp_optimize_candidate(base, max_iter=300)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            base = h_opt

    if best_h is None:
        n = scales[-1]
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5(best_h)

    if budget_s - (time.time() - start_time) > 15.0 and best_h.size < 2048:
        best_h = upsample(best_h, 2048)
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)

    if budget_s - (time.time() - start_time) > 5.0:
        remaining = budget_s - (time.time() - start_time)
        final_iters = int(min(60000, max(20000, remaining * 30)))
        best_h, best_score = gradient_descent(best_h, n_iter=final_iters,
                                              step0=0.3, beta=0.9,
                                              rng=rng,
                                              start_time=start_time,
                                              budget_s=budget_s)

    if budget_s - (time.time() - start_time) > 5.0:
        n = best_h.size
        idx_desc = np.argsort(best_h)[::-1]
        binary_h = np.zeros_like(best_h)
        binary_h[idx_desc[: n // 2]] = 1.0
        binary_h, bin_score = binary_local_search(binary_h, rng,
                                                  start_time, budget_s)
        if bin_score < best_score:
            best_score, best_h = bin_score, binary_h.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
