def gradient_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = corr.max()
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_softmax_shift(h, alpha):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # convs
    conv_back = np.convolve(w, h, mode='full')
    conv_for = np.convolve(w[::-1], h, mode='full')
    offset = n - 1
    grad_back = conv_back[offset:offset + n]
    grad_for = conv_for[offset:offset + n]
    grad = 1.0 - (grad_back + grad_for)
    return grad

def gradient_descent(h0, time_left_func, max_iters=200000, init_step=1.0, start_alpha=0.5, final_alpha=5.0):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    target_sum = n / 2.0
    for it in range(1, max_iters+1):
        if time_left_func() <= 0:
            break
        # schedule alpha linearly increase
        alpha = start_alpha + (final_alpha - start_alpha) * (it / max_iters)
        grad = gradient_softmax_shift(h, alpha)
        step = init_step / np.sqrt(it)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):

    return h

def gradient_softmax_shift(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_back = np.convolve(w, h, mode='full')
    conv_for = np.convolve(w[::-1], h, mode='full')
    offset = n - 1
    grad_back = conv_back[offset:offset + n]
    grad_for = conv_for[offset:offset + n]
    grad = 1.0 - (grad_back + grad_for)
    return grad

def gradient_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, time_left_func, max_iters=200000, init_step=1.0, start_alpha=0.5, final_alpha=5.0):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        if time_left_func() <= 0:
            break
        alpha = start_alpha + (final_alpha - start_alpha) * (it / max_iters)
        grad = gradient_softmax_shift(h, alpha)
        step = init_step / np.sqrt(it)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def refine_max_shift(h0, time_left_func, max_iters=200000, step0=0.5, beta=0.9):

    n = h.shape[0]

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_softmax_shift(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_back = np.convolve(w, h, mode='full')
    conv_for = np.convolve(w[::-1], h, mode='full')
    off = n - 1
    grad_back = conv_back[off:off + n]
    grad_for = conv_for[off:off + n]
    return 1.0 - (grad_back + grad_for)

def gradient_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    idxes = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent(h0, time_left, max_iters=200000,
                    init_step=0.8, start_alpha=0.5, final_alpha=15.0):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        alpha = start_alpha + (final_alpha - start_alpha) * (it / max_iters)
        grad = gradient_softmax_shift(h, alpha)
        step = init_step / np.sqrt(it)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def refine_max_shift(h0, time_left, max_iters=200000,
                     step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iters and time_left() > 0:
        grad = gradient_worst_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = h - step * mom
        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_sc

def binary_swap_anneal_simple(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur_sc = compute_c5(h)
    best_h = h.copy()
    best_sc = cur_sc
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_sc = compute_c5(h)
        delta = new_sc - cur_sc
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_h = new_sc, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_sc
