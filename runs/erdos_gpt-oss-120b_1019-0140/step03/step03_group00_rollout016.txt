def full_correlation_fft(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, N)
    B_rev = np.fft.rfft((1 - h)[::-1], N)
    corr = np.fft.irfft(H * B_rev, N)[:2*n - 1]
    return corr

def gradient_descent_fft(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = full_correlation_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n-1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional jitter maybe every 2000 iterations
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        # project onto box sum
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

# --- Helper functions ---------------------------------------------------------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    lam = (lam_low + lam_high) * 0.5
    return np.clip(v - lam, lo, hi)


def full_correlation_fft(h):
    """Full linear correlation corr[k] = sum_i h_i * (1-h_{i+k}) via FFT."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    B_rev = np.fft.rfft((1.0 - h)[::-1], N)
    corr = np.fft.irfft(H * B_rev, N)[: 2 * n - 1]
    return corr


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = full_correlation_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Average sub‑gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        term1 = np.zeros(n, dtype=h.dtype)
        term2 = np.zeros(n, dtype=h.dtype)
        if shift >= 0:
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_fft(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent using FFT correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    momentum = np.zeros_like(h)
    # initial score
    corr = full_correlation_fft(h)
    max_corr = np.max(corr)
    best_score = max_corr * (2.0 / n)
    best_h = h.copy()

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        # active shifts
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional jitter to escape flats
        if it % 2500 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        # feasibility projection
        h_new = project_onto_box_sum(h_new, s)

        # new correlation and score
        corr = full_correlation_fft(h_new)
        max_corr = np.max(corr)
        score = max_corr * (2.0 / n)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()

        h = h_new

    return best_h, best_score


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
