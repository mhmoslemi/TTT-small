def compute_smoothed_gradient(h, T):
    # h: vector length n
    n = h.size
    a = 1.0 - np.correlate(h, h, mode='full') * (2.0 / n)  # 2/n = dx
    # Weight w
    # For numerical stability, subtract max:
    max_a = np.max(a)
    exp_a = np.exp((a - max_a) / T)  # stable
    w = exp_a / np.sum(exp_a)
    # Compute gradient: -dx * (conv(w, h) + conv(w_rev, h))
    dx = 2.0 / n
    # Convolution via FFT
    # w length = 2n-1
    # zero-pad to length L >= 2n-1 + n -1 = 3n-2 => choose pow2
    L = 1 << ((3*n - 2).bit_length())
    # FFT.
    w_f = np.fft.rfft(w, n=L)
    h_f = np.fft.rfft(h, n=L)
    conv_wh = np.fft.irfft(w_f * h_f, n=L)[:n]  # first n values correspond to conv(w, h) indices 0..n-1
    # w_rev
    w_rev = w[::-1]
    w_rev_f = np.fft.rfft(w_rev, n=L)
    conv_wrev_h = np.fft.irfft(w_rev_f * h_f, n=L)[:n]
    grad = -dx * (conv_wh + conv_wrev_h)
    return grad

def optimize_h(h0, time_end, T_start=0.2, T_end=0.02, lr=0.5, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t0 = time.time()
    it = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    while time.time() < time_end and it < max_iter:
        it += 1
        # temperature schedule
        T = T_start * ((T_end / T_start) ** (it / max_iter))
        grad = compute_smoothed_gradient(h, T)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # Project onto box and sum
        h_new = project_onto_box_sum(h_new, s)
        # Evaluate C5
        score = compute_c5(h_new)
        if score < best_score:

def compute_smoothed_gradient(h, T):
    n = h.shape[0]
    # compute a = 1 - corr
    corr = np.correlate(h, h, mode='full') * (2.0 / n)
    a = 1.0 - corr
    # compute weights
    max_a = np.max(a)
    # avoid overflow
    w = np.exp((a - max_a) / T)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    dx = 2.0 / n
    # convolution via FFT
    L = 1 << ((3*n - 2).bit_length())
    w_f = np.fft.rfft(w, n=L)
    h_f = np.fft.rfft(h, n=L)
    conv_wh = np.fft.irfft(w_f * h_f, n=L)[:n]  # length n
    # reversed w
    w_rev = w[::-1]
    w_rev_f = np.fft.rfft(w_rev, n=L)
    conv_wrev_h = np.fft.irfft(w_rev_f * h_f, n=L)[:n]
    grad = -dx * (conv_wh + conv_wrev_h)
    return grad

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, h, mode='full') * (2.0 / n)
    return np.max(1.0 - corr)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def compute_smoothed_gradient(h, T):
    """Gradient of T‑scaled log‑sum‑exp of (1‑corr) w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full') * (2.0 / n)
    a = 1.0 - corr                     # a_k = 1 - corr_k
    max_a = np.max(a)
    w = np.exp((a - max_a) / T)
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    dx = 2.0 / n
    L = 1 << ((3 * n - 2).bit_length())      # power of two ≥ 3n‑2
    w_f = np.fft.rfft(w, n=L)
    h_f = np.fft.rfft(h, n=L)
    conv_wh = np.fft.irfft(w_f * h_f, n=L)[:n]
    w_rev = w[::-1]
    w_rev_f = np.fft.rfft(w_rev, n=L)
    conv_wrev_h = np.fft.irfft(w_rev_f * h_f, n=L)[:n]
    grad = -dx * (conv_wh + conv_wrev_h)
    return grad

def adam_optimize(h0, time_end, T_start=0.2, T_end=0.01,
                  lr=0.5, max_iter=20000):
    """Adam with projection onto box & sum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while time.time() < time_end and it < max_iter:
        it += 1
        T = T_start * ((T_end / T_start) ** (it / max_iter))
        grad = compute_smoothed_gradient(h, T)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(b, time_end, rng):
    """Simple hill climbing on binary vectors (swap 1↔0)."""
    n = b.size
    best_b = b.copy()
    best_score = compute_c5(b)
    while time.time() < time_end:
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b[i], b[j] = 0.0, 1.0
        score = compute_c5(b)
        if score < best_score:
            best_score = score
            best_b = b.copy()
            if best_score <= 0.3808:
                break
        else:
            b[i], b[j] = 1.0, 0.0
    return best_b, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ---- Phase 1 : coarse (64 points) ----
    n_small = 64
    candidates = []
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init, n_small), n_small / 2.0)
        candidates.append(cand)
    candidates.append(thue_morse_binary(n_small))
    for _ in range(4):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(4):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h = None
    best_score = np.inf
    for cand in candidates:
        if time_left() < 4.0:
            break
        t_end = time.time() + min(8.0, time_left() * 0.2)
        h_opt, sc_opt = adam_optimize(cand, t_end,
                                      T_start=0.3, T_end=0.02,
                                      lr=0.6, max_iter=6000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    if best_h is None:
        best_h = random_binary_balanced(n_small, rng)
        best_score = compute_c5(best_h)

    # ---- Phase 2 : medium (256 points) ----
    n_mid = 256
    if time_left() > 10.0:
        h_mid = resample(best_h, n_mid)
        h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
        t_end = time.time() + min(30.0, time_left() * 0.3)
        max_iter = int(min(40000, max(5000, (t_end - time.time()) * 800)))
        h_opt, sc_opt = adam_optimize(h_mid, t_end,
                                      T_start=0.25, T_end=0.015,
                                      lr=0.55, max_iter=max_iter)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # a couple random restarts at this resolution
        for _ in range(2):
            if time_left() < 6.0:
                break
            v = rng.random(n_mid)

            t_end2 = time.time() + min(8.0, time_left() * 0.15)
            h_r, sc_r = adam_optimize(h0, t_end2,
                                      T_start=0.25, T_end=0.015,
                                      lr=0.55, max_iter=8000)
            if sc_r < best_score:
                best_score, best_h = sc_r, h_r.copy()

    # ---- Phase 3 : fine (512 points) ----
    n_large = 512
    if time_left() > 15.0:
        h_large = resample(best_h, n_large)
        h_large = project_onto_box_sum(h_large, n_large / 2.0)
        t_end = time.time() + min(45.0, time_left() * 0.4)
        max_iter = int(min(80000, max(10000, (t_end - time.time()) * 800)))
        h_opt, sc_opt = adam_optimize(h_large, t_end,
                                      T_start=0.22, T_end=0.01,
                                      lr=0.5, max_iter=max_iter)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

        while time_left() > 6.0 and best_score > 0.3808:
            v = rng.random(n_large)
            h0 = project_onto_box_sum(v, n_large / 2.0)
            t_end2 = time.time() + min(6.0, time_left() * 0.1)
            h_r, sc_r = adam_optimize(h0, t_end2,

        bin_h = binarize_from_real(best_h)

    return best_h, best_score, best_h.size
