   - GA explores a wider search space, reduces local optimum trapping better

   def enforce_sum(v, target):
       v = np.clip(v, 0.0, 1.0)
       total = np.sum(v)
       if abs(total - target) < 1e-12:
           return v
       
       # Scale first
       scale = target / total
       scale_adjust = 1.0
       residual = target - total
       v = v * scale * (1 + scale_adjust)  # Add scaling buffer
       v = np.clip(v, 0.0, 1.0)
       residual = target - np.sum(v)
       # Distribute residual 20 times
       attempts = 0
       while abs(residual) > 0.5 and attempts < 20:
           if residual > 0:
               # Find index to increase
               valid_mask = v < 1.0 - 1e-12
               if not np.any(valid_mask):
                   break
               idx = np.where(valid_mask)[0][np.random.choice(len(valid_mask))]
               transfer = min(residual, 1.0 - v[idx])
               v[idx] += transfer
           else:
               # Find index to decrease
               valid_mask = v > 1e-12
               if not np.any(valid_mask):
                   break
               idx = np.where(valid_mask)[0][np.random.choice(len(valid_mask))]
               transfer = min(-residual, v[idx])
               v[idx] -= transfer
           residual = target - np.sum(v)
           attempts += 1
       # Final clip
       v = np.clip(v, 0.0, 1.0)
       return v
