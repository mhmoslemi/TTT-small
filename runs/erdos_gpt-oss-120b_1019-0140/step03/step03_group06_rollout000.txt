def enforce_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    # Recompute total, distribute residual
    total = v.sum()
    diff = target - total
    # Distribute diff
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0: break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])

            diff -= inc

            if len(cand) == 0: break

            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    # final clip
    v = np.clip(v, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation includes full length
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    v = np.clip(v, 0.0, 1.0)
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v

def uniform_crossover(p1, p2, rng):
    mask = rng.random(p1.shape[0]) < 0.5
    child1 = np.where(mask, p1, p2)
    child2 = np.where(mask, p2, p1)
    return child1, child2

def mass_transfer_mutate(h, rng):
    # move a random amount from a positive entry to a smaller one
    src_candidates = np.where(h > 1e-12)[0]
    dst_candidates = np.where(h < 1.0 - 1e-12)[0]
    if len(src_candidates) == 0 or len(dst_candidates) == 0:
        return h
    src = rng.choice(src_candidates)
    dst = rng.choice(dst_candidates)
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0:
        return h
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

def local_search(h, cur_val, deadline, rng):
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.05:
        src_candidates = np.where(best_h > 1e-12)[0]
        dst_candidates = np.where(best_h < 1.0 - 1e-12)[0]
        if len(src_candidates) == 0 or len(dst_candidates) == 0:

        if len(new_pop) > popsize:

        new_scores = [compute_c5(ind) for ind in new_pop]

        # Merge elites from previous generation

        order = np.argsort(combined_scores)[:popsize]

        cur_best_val = scores[cur_best_idx]
        if cur_best_val < best_val:

            worst_idx = int(np.argmax(scores))
            if best_val < scores[worst_idx]:

                scores[worst_idx] = best_val

    if time.time() < deadline - 0.2:

    best_h = enforce_sum(best_h, target, rng)
    best_h = np.clip(best_h, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    v = np.clip(v, 0.0, 1.0)
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v

def uniform_crossover(p1, p2, rng):
    mask = rng.random(p1.shape[0]) < 0.5
    child1 = np.where(mask, p1, p2)
    child2 = np.where(mask, p2, p1)
    return child1, child2

def mass_transfer_mutate(h, rng):
    src = np.where(h > 1e-12)[0]
    dst = np.where(h < 1.0 - 1e-12)[0]
    if len(src) == 0 or len(dst) == 0:
        return h
    i = rng.choice(src)
    j = rng.choice(dst)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def local_search(h, cur_val, deadline, rng):
    cur_h = h.copy()
    cur_val = cur_val
    best_h = cur_h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.05:
        src = np.where(cur_h > 1e-12)[0]
        dst = np.where(cur_h < 1.0 - 1e-12)[0]
        if len(src) == 0 or len(dst) == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = compute_c5(h_new)
        if new_val < cur_val:
            cur_h, cur_val = h_new, new_val
        else:
            remaining = deadline - time.time()
            T = max(remaining * 0.001, 1e-12)
            if np.exp(-(new_val - cur_val) / T) > rng.random():
                cur_h, cur_val = h_new, new_val
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    else:
        n = 80
        init_h = None
    target = n / 2.0

    popsize = 40
    population = []
    if init_h is not None:
        population.append(enforce_sum(init_h, target, rng))
    alt = np.tile([1.0, 0.0], n // 2)
    if alt.shape[0] < n:
        alt = np.concatenate([alt, np.ones(n - alt.shape[0])])
    population.append(enforce_sum(alt, target, rng))
    population.append(random_binary_vector(n, target, rng))
    while len(population) < popsize:
        population.append(random_continuous_vector(n, target, rng))

    scores = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_val = scores[best_idx]

    while time.time() < deadline - 1.0:
        def select():
            idxs = rng.choice(popsize, size=3, replace=False)
            vals = [scores[i] for i in idxs]
            return population[int(idxs[int(np.argmin(vals))])]

        new_pop = []
        while len(new_pop) < popsize:
            p1 = select()
            p2 = select()
            c1, c2 = uniform_crossover(p1, p2, rng)
            for _ in range(rng.integers(1, 4)):
                c1 = mass_transfer_mutate(c1, rng)
                c2 = mass_transfer_mutate(c2, rng)
            c1 = enforce_sum(c1, target, rng)
            c2 = enforce_sum(c2, target, rng)
            new_pop.extend([c1, c2])
        if len(new_pop) > popsize:
            new_pop = new_pop[:popsize]

        new_scores = [compute_c5(ind) for ind in new_pop]
        combined = population + new_pop
        combined_scores = scores + new_scores
        order = np.argsort(combined_scores)[:popsize]
        population = [combined[i] for i in order]
        scores = [combined_scores[i] for i in order]

            worst_idx = int(np.argmax(scores))
            if best_val < scores[worst_idx]:
                population[worst_idx] = best_h.copy()
                scores[worst_idx] = best_val

    if time.time() < deadline - 0.2:
        best_h, best_val = local_search(best_h, best_val, deadline, rng)

    best_h = enforce_sum(best_h, target, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
