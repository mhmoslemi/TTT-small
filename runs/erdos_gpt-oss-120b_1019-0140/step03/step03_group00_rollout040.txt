def compute_c5_fft(h):
    n = h.shape[0]
    # zero pad to 2*n
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    # corr length N, correlation for lag d = - (n-1) ... (n-1) corresponds to indices:
    # In linear correlation (full), length 2*n-1, but we have N=2*n.
    # The correlation is conv of h with (1-h) reversed? Actually np.correlate(a,b,mode='full') yields sum_i a[i] * b[i + shift], where shift = -(len(b)-1) .. len(a)-1.
    # Using fft: conv via FFT of a and reversed b for correlation; but since one of them is reversed? Let's verify: 
    # np.correlate(h, 1-h, mode='full')[k] = sum_i h[i] * (1-h)[i + shift] where shift = k - (len(b)-1).
    # That is exactly convolution of h and reversed(1-h) ?? Actually correlation can be computed by conv of h and reversed b.
    # Using FFT: corr = ifft( fft(h, N) * conj(fft(b, N)) )

def upsample_binary(arr, new_n):
    m = arr.shape[0]
    if new_n % m != 0:
        # Use interpolation and threshold
        x_old = np.linspace(0, 1, m, endpoint=False)
        x_new = np.linspace(0, 1, new_n, endpoint=False)
        arr_interp = np.interp(x_new, x_old, arr)
        # threshold at 0.5 (but ensure balanced)
        # Count ones after threshold, adjust.
        binary = (arr_interp >= 0.5).astype(float)
        # Ensure sum exactly new_n/2: adjust if needed.
        current_sum = binary.sum()
        target = new_n // 2
        diff = int(target - current_sum)
        if diff > 0:
            # turn some zeros to ones
            zero_idx = np.where(binary == 0)[0]
            idx = rng.choice(zero_idx, size=diff, replace=False)
            binary[idx] = 1.0
        elif diff < 0:
            one_idx = np.where(binary == 1)[0]
            idx = rng.choice(one_idx, size=-diff, replace=False)
            binary[idx] = 0.0
        return binary
    else:
        factor = new_n // m
        return np.repeat(arr, factor).astype(float)

import time
import numpy as np

def compute_corr_max(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr, corr

def compute_c5_fast(h):
    max_corr, _ = compute_corr_max(h)
    return max_corr * (2.0 / h.shape[0])

def compute_c5_full(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def repeat_binary(arr, new_n):
    m = arr.shape[0]
    factor = new_n // m
    return np.repeat(arr, factor).astype(float)

    h = start_h.copy()
    n = h.size
    max_corr, corr = compute_corr_max(h)
    current_score = max_corr * (2.0 / n)
    best_h = h.copy()
    best_score = current_score
    iter_count = 0
    # temperature parameters
    T0 = current_score * 0.2
    Tmin = 1e-8
    while time_left() > 0.05:
        iter_count += 1
        # update temperature
        elapsed = time.time() - start_time
        frac = elapsed / total_budget
        T = max(Tmin, T0 * (1 - frac))
        # decide targeted or random swap
        if rng.random() < 0.25:
            # targeted move
            # use last corr computed, else compute
            # ensure we have corr (full array)
            # we have corr from previous compute_corr_max
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)
            if shift >= 0:
                problematic = np.where((h[:n-shift] == 1) & (h[shift:] == 0))[0]
            else:
                d = -shift
                prob = np.where((h[d:] == 1) & (h[:n-d] == 0))[0]
                problematic = prob + d
            if problematic.size > 0:
                i = rng.choice(problematic)
                zeros = np.flatnonzero(1 - h)
                j = rng.choice(zeros)
            else:
                i = rng.choice(np.flatnonzero(h))
                j = rng.choice(np.flatnonzero(1 - h))
        else:
            i = rng.choice(np.flatnonzero(h))
            j = rng.choice(np.flatnonzero(1 - h))
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # evaluate
        max_corr_new, corr_new = compute_corr_max(h)
        new_score = max_corr_new * (2.0 / n)
        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_score = new_score
            corr = corr_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # optional break if best_score already below target
        if best_score <= 0.3808:

    return best_h, best_score, iter_count

import time
import numpy as np

def compute_corr_max(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr, corr

def compute_c5_full(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def repeat_binary(arr, new_n):
    m = arr.shape[0]
    factor = new_n // m
    return np.repeat(arr, factor).astype(float)

def hill_climb_binary(start_h, rng, time_left, start_time, total_budget, max_iters=10**9):
    h = start_h.copy()
    n = h.size
    max_corr, corr = compute_corr_max(h)
    current_score = max_corr * (2.0 / n)
    best_h = h.copy()
    best_score = current_score
    iter_count = 0
    T0 = current_score * 0.2
    Tmin = 1e-8
    while time_left() > 0.05 and iter_count < max_iters:
        iter_count += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_budget
        T = max(Tmin, T0 * (1 - frac))
        # decide move type
        if rng.random() < 0.25:
            # targeted move using current corr
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)
            if shift >= 0:
                problematic = np.where((h[:n-shift] == 1) & (h[shift:] == 0))[0]
            else:
                d = -shift
                prob = np.where((h[d:] == 1) & (h[:n-d] == 0))[0]
                problematic = prob + d
            if problematic.size > 0:
                i = rng.choice(problematic)
                zeros = np.flatnonzero(1 - h)
                j = rng.choice(zeros)
            else:
                i = rng.choice(np.flatnonzero(h))
                j = rng.choice(np.flatnonzero(1 - h))
        else:
            i = rng.choice(np.flatnonzero(h))
            j = rng.choice(np.flatnonzero(1 - h))
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # evaluate
        max_corr_new, corr_new = compute_corr_max(h)
        new_score = max_corr_new * (2.0 / n)
        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_score = new_score
            corr = corr_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            if best_score <= 0.3808:
                break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score, iter_count

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    total_budget = budget_s

    # target resolution
    n = 256
    # build initial binary pattern
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        m = base.shape[0]
        if n % m == 0:
            base = repeat_binary(base, n)
        else:
            base = random_binary_balanced(n, rng)
    else:
        base = random_binary_balanced(n, rng)
    # binarize and enforce exact balance
    base = (base >= 0.5).astype(float)
    cur_ones = int(base.sum())
    target_ones = n // 2
    if cur_ones > target_ones:
        idx = np.flatnonzero(base)[:cur_ones - target_ones]
        base[idx] = 0.0
    elif cur_ones < target_ones:
        idx = np.flatnonzero(1 - base)[:target_ones - cur_ones]
        base[idx] = 1.0

    best_h = base.copy()
    best_score = compute_c5_full(best_h)

    # primary hill climbing
    h_opt, score_opt, _ = hill_climb_binary(best_h, rng, time_left, start_time, total_budget, max_iters=200000)
    full_score_opt = compute_c5_full(h_opt)
    if full_score_opt < best_score:
        best_h = h_opt.copy()
        best_score = full_score_opt
    if best_score <= 0.3808:
        return best_h, best_score, best_h.size

    # random restarts
    while time_left() > 12:
        start_h = random_binary_balanced(n, rng)
        h_cand, _, _ = hill_climb_binary(start_h, rng, time_left, start_time, total_budget, max_iters=50000)
        cand_score = compute_c5_full(h_cand)
        if cand_score < best_score:
            best_h = h_cand.copy()
            best_score = cand_score
            if best_score <= 0.3808:
                break

    # upscale to finer grid if time permits
    if time_left() > 15:
        n2 = 512
        h_big = repeat_binary(best_h, n2)
        # ensure exact balance after repeat (should be balanced)
        h_big_opt, _, _ = hill_climb_binary(h_big, rng, time_left, start_time, total_budget, max_iters=150000)
        big_score = compute_c5_full(h_big_opt)
        if big_score < best_score:
            best_h = h_big_opt.copy()
            best_score = big_score

    # final verification
    best_score = compute_c5_full(best_h)
    return best_h, best_score, best_h.size
