def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def slsqp_optimize_continuous(h_start, t_start, sum_con, overlap_cons, maxiter=600):
    n = h_start.shape[0]

    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type':'ineq','fun':oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the hypercube [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def simulated_annealing(rng, h_start, target_sum, start_temp, cooling_rate,
                        max_iter, time_limit):
    """Simple simulated annealing with projection to box-sum."""
    n = h_start.shape[0]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    temp = start_temp
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        step = rng.normal(scale=0.05, size=n) * temp
        h_new = h_cur + step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur or rng.random() < np.exp((c5_cur - c5_new) / max(temp, 1e-12)):
            h_cur = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        temp *= cooling_rate
        if best_c5 < 0.3808:
            break
    return best_h, best_c5

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimensionality
    if 'initial_h_values' in globals():
