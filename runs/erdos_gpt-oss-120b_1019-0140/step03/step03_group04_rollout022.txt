def simulated_annealing(h_init, max_iter, time_end, rng):

    h = h_init.copy()
    n = h.size
    # compute autocorr via FFT
    autocorr = autocorr_fft(h)
    cur_score = (n/2 - autocorr).max() * (2/n)

    best_h = h.copy()
    best_score = cur_score

    # prepare lists of indices
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    # temperature schedule
    T0 = 0.01  # initial temperature maybe 0.01 (scale)
    T_min = 1e-5

    for it in range(1, max_iter+1):
        if time.time() >= time_end:
            break
        # linear cooling
        t_frac = it / max_iter
        T = T0 * (1 - t_frac)
        if T < T_min:
            T = T_min
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta
        delta_autocorr = compute_delta_autocorr(h, i, j)
        new_autocorr = autocorr + delta_autocorr
        new_score = (n/2 - new_autocorr).max() * (2/n)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # acceptance probability
            prob = np.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            # apply updates
            autocorr = new_autocorr
            cur_score = new_score
            # swap entries in h
            h[i] = 0
            h[j] = 1
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # record best
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        # optional: every some steps recompute autocorr via FFT for safety
        if (it % 5000) == 0:
            autocorr = autocorr_fft(h)
            cur_score = (n/2 - autocorr).max() * (2/n)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
