def gradient_descent_momentum_with_budget(h0, start, budget_s, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time.time() - start > budget_s:
            break
        # compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute grad using shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                sft = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[sft:] = h[:n-sft]
                term2[:n-sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        if shifts.size > 0:
            grad /= shifts.size
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto box-sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_anneal(h0, rng, start, budget_s, max_iters=500000, T0=0.5, decay=0.99997):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    for it in range(max_iters):
        if time.time() - start > budget_s:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(1e-6, T * decay)
    return best_h, best_score
```

def local_swap_improvement(h, start, budget_s):
    # deterministic hill climbing with best improving swap
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    while True:
        if time.time() - start > budget_s:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        improved = False
        for i in ones:
            for j in zeros:
                # quick check: if time left insufficient break
                if time.time() - start > budget_s:
                    break
                # swap
                trial = best_h.copy()
                trial[i] = 0.0
                trial[j] = 1.0

                    best_score = score
                    best_h = trial
                    improved = True
                    break  # break inner loop to restart
            if improved:
                break

            break

def milp_binary_opt(n, time_limit):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    constraints = []
    constraints.append(cp.sum(h) == n/2)
    # w constraints for i<j
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
        # optionally enforce w[i,i] = h[i] (not needed)
    t = cp.Variable()
    constraints.append(t >= 0)
    # shift constraints
    for s in range(-(n-1), n):
        if s >= 0:
            idx_i = np.arange(0, n-s)
            idx_j = idx_i + s
            sum_h = cp.sum(h[idx_i])
            sum_w = cp.sum(w[idx_i, idx_j])
        else:
            p = -s
            idx_i = np.arange(p, n)
            idx_j = idx_i - p
            sum_h = cp.sum(h[idx_i])
            sum_w = cp.sum(w[idx_j, idx_i])
        constraints.append(t >= sum_h - sum_w)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'tmlim': time_limit})
    except Exception:
        try:
            prob.solve(solver=cp.CBC, cbc={'seconds': time_limit})
        except Exception:
            return None
    if h.value is None:
        return None
    h_opt = np.rint(np.array(h.value)).astype(float)
    return h_opt
```

import time
import numpy as np
import scipy.optimize as opt
import cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            sft = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[sft:] = h[:n - sft]
            term2[:n - sft] = h[sft:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_momentum_with_budget(h0, start, budget_s,
                                          step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time.time() - start > budget_s:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_anneal(h0, rng, start, budget_s,
                      max_iters=800000, T0=0.5, decay=0.99997):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T0
    for it in range(max_iters):
        if time.time() - start > budget_s:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(1e-6, T * decay)
    return best_h, best_score

def local_swap_improvement(h, start, budget_s):
    """Deterministic hill climbing by swapping a 1 with a 0."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while True:
        if time.time() - start > budget_s:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        improved = False
        for i in ones:
            for j in zeros:
                if time.time() - start > budget_s:
                    break
                trial = best_h.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                score = compute_c5(trial)
                if score < best_score - 1e-12:
                    best_h = trial
                    best_score = score
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_score

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def milp_binary_opt(n, time_limit):
    """Solve the binary optimization exactly via MILP (if feasible)."""
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    constraints = [cp.sum(h) == n / 2]
    # linearization of products
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(w[i, j] <= h[i])
            constraints.append(w[i, j] <= h[j])
            constraints.append(w[i, j] >= h[i] + h[j] - 1)
    t = cp.Variable()
    constraints.append(t >= 0)
    for s in range(-(n - 1), n):
        if s >= 0:
            idx_i = np.arange(0, n - s, dtype=int)
            idx_j = idx_i + s
            sum_h = cp.sum(h[idx_i])
            sum_w = cp.sum(w[idx_i, idx_j])
        else:
            p = -s
            idx_i = np.arange(p, n, dtype=int)
            idx_j = idx_i - p
            sum_h = cp.sum(h[idx_i])
            sum_w = cp.sum(w[idx_j, idx_i])
        constraints.append(t >= sum_h - sum_w)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'tmlim': time_limit})
    except Exception:
        try:
            prob.solve(solver=cp.CBC, cbc={'seconds': time_limit})
        except Exception:
            return None
    if h.value is None:
        return None
    h_opt = np.rint(np.array(h.value)).astype(float)
    return h_opt
