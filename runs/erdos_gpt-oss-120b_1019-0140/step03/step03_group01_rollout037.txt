def anneal_binary(h_start, time_budget, rng, n):

    dx = 2.0 / n
    offset = n - 1
    len_overlap = n - np.abs(np.arange(-n+1, n))
    # compute conv start
    conv = np.correlate(h_start, h_start, mode='full')
    # compute initial score
    best_h = h_start.copy()
    best_conv = conv.copy()
    best_score = -dx * np.min(conv - len_overlap)
    # initial temperature maybe based on best_score
    T = 0.5
    decay = 0.9999
    min_T = 1e-6
    start = time.time()
    ones = list(np.where(h_start == 1)[0])
    zeros = list(np.where(h_start == 0)[0])
    h = h_start.copy()
    conv_cur = conv.copy()
    cur_score = best_score
    while time.time() - start < time_budget:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute conv_candidate via copy and updates

            idx2 = offset - d

                continue

            idx2 = offset - d

            conv_candidate[idx2] += 1
        new_score = -dx * np.min(conv_candidate - len_overlap)
        delta = new_score - cur_score

def anneal_binary(h0, time_budget, rng, n):
    dx = 2.0 / n
    offset = n - 1
    # initial conv arrays
    conv_hh = np.correlate(h0, h0, mode='full')
    conv_h1 = np.correlate(h0, np.ones(n, dtype=int), mode='full')
    cur_score = dx * np.max(conv_h1 - conv_hh)
    best_h = h0.copy()
    best_score = cur_score
    best_hh = conv_hh.copy()
    best_h1 = conv_h1.copy()
    # maintain ones list
    ones = list(np.where(h0 == 1)[0])
    zeros = list(np.where(h0 == 0)[0])
    start = time.time()
    T = 0.5
    decay = 0.9999
    min_T = 1e-7
    while time.time() - start < time_budget:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # copy conv arrays
        conv_hh_cand = conv_hh.copy()
        conv_h1_cand = conv_h1.copy()
        # Removal of i contributions from conv_hh
        # create array of other ones
        # To avoid creating new list each iteration, we can use loop but it's okay
        for k in ones:
            if k == i:
                continue
            d = k - i
            idx1 = d + offset
            idx2 = offset - d
            conv_hh_cand[idx1] -= 1
            conv_hh_cand[idx2] -= 1
        # Addition of j contributions
        for k in ones:
            if k == i:
                continue
            d = k - j
            idx1 = d + offset
            idx2 = offset - d
            conv_hh_cand[idx1] += 1
            conv_hh_cand[idx2] += 1
        # Update conv_h1 for i removal and j addition
        conv_h1_cand[offset - i : offset - i + n] -= 1
        conv_h1_cand[offset - j : offset - j + n] += 1
        # compute candidate score
        cand_score = dx * np.max(conv_h1_cand - conv_hh_cand)
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > min_T and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # commit changes
            h0[i] = 0
            h0[j] = 1
            # update ones and zeros
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            conv_hh = conv_hh_cand
            conv_h1 = conv_h1_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_h = h0.copy()
                best_hh = conv_hh.copy()
                best_h1 = conv_h1.copy()
            if best_score <= 0.3808:
                # early exit
                break
        # temperature decay
        T = max(min_T, T * decay)

import time
import numpy as np

def upsample(arr, new_n):
    """Linearly upsample a 1‑D array to length new_n."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_from_continuous(arr, k):
    """Return a binary vector of length len(arr) with exactly k entries equal to 1."""
    idx = np.argpartition(-arr, k)[:k]
    out = np.zeros_like(arr, dtype=np.int8)
    out[idx] = 1
    return out

def anneal_binary(h0, time_budget, rng, n):
    """Simulated‑annealing on binary vectors with exact updates of the
    two necessary convolutions.
    Returns the best vector found and its C₅ value."""
    dx = 2.0 / n
    offset = n - 1

    # initial convolutions
    conv_hh = np.correlate(h0, h0, mode='full')
    conv_h1 = np.correlate(h0, np.ones(n, dtype=int), mode='full')
    cur_score = dx * np.max(conv_h1 - conv_hh)

    best_h = h0.copy()
    best_score = cur_score
    best_hh = conv_hh.copy()
    best_h1 = conv_h1.copy()

    ones = list(np.where(h0 == 1)[0])
    zeros = list(np.where(h0 == 0)[0])

    start = time.time()
    T = 0.8
    decay = 0.99995
    min_T = 1e-7

    while time.time() - start < time_budget:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # candidate convolutions (copy‑on‑write)
        conv_hh_cand = conv_hh.copy()
        conv_h1_cand = conv_h1.copy()

        # update conv_hh for removal of i and addition of j
        for k in ones:
            if k == i:
                continue
            d = k - i
            idx1 = d + offset
            idx2 = offset - d
            conv_hh_cand[idx1] -= 1
            conv_hh_cand[idx2] -= 1
        for k in ones:
            if k == i:
                continue
            d = k - j
            idx1 = d + offset
            idx2 = offset - d
            conv_hh_cand[idx1] += 1
            conv_hh_cand[idx2] += 1

        # update conv_h1 (sum of h over the overlapped region)
        conv_h1_cand[offset - i : offset - i + n] -= 1
        conv_h1_cand[offset - j : offset - j + n] += 1

        cand_score = dx * np.max(conv_h1_cand - conv_hh_cand)
        delta = cand_score - cur_score

        accept = False
        if delta < 0.0:
            accept = True
        else:
            if T > min_T and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # commit the move
            h0[i] = 0
            h0[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)

            conv_hh = conv_hh_cand
            conv_h1 = conv_h1_cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_h = h0.copy()
                best_hh = conv_hh.copy()
                best_h1 = conv_h1.copy()
                if best_score <= 0.3808:
                    break

        T = max(min_T, T * decay)

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def remaining():
        return max(0.0, budget_s - (time.time() - start))

    # Choose a grid size that fits the remaining time.
    n = 512 if remaining() > 200 else 256
    half = n // 2

    # Build an initial binary vector.
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cont = upsample(init_full, n)
        h0 = binary_from_continuous(cont, half)
    else:
        h0 = np.zeros(n, dtype=np.int8)
        idx = rng.choice(n, size=half, replace=False)
        h0[idx] = 1

    # Run annealing for the bulk of the time, leaving a few seconds for final evaluation.
    time_for_anneal = remaining() - 5.0
    if time_for_anneal < 5.0:
        time_for_anneal = max(0.5, remaining())

    best_h, best_score = anneal_binary(h0, time_for_anneal, rng, n)

    # Final accurate evaluation (uses the pre‑imported helper if present).
    if 'evaluate_erdos_solution' in globals():
        final_score = evaluate_erdos_solution(best_h.astype(float))
    else:
        dx = 2.0 / n
        conv_hh = np.correlate(best_h, best_h, mode='full')
        conv_h1 = np.correlate(best_h, np.ones(n, dtype=int), mode='full')
        final_score = dx * np.max(conv_h1 - conv_hh)

    return best_h.astype(float), final_score, n
