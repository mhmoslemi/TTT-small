import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -1e3
    high = 1e3
    # binary search for lambda to achieve sum target after clipping
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def subgradient_max_overlap(h, dx):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    max_val = np.max(overlaps)
    # tolerance for “active” constraints
    eps = 1e-10
    active = np.abs(overlaps - max_val) <= eps
    ks = np.arange(-n+1, n)[active]
    # average gradient over active shifts
    grad = np.zeros_like(h)
    for k in ks:
        grad += gradient_for_shift(h, k)
    if ks.size > 0:
        grad /= ks.size
    return grad, max_val

# ---------- Core utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while forcing sum(v) == target."""
    low, high = -1e3, 1e3
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def subgradient_max_overlap(h, dx):
    """Subgradient of max_k overlap (average over all active shifts)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    max_val = np.max(overlaps)
    eps = 1e-10
    active = np.abs(overlaps - max_val) <= eps
    ks = np.arange(-n + 1, n)[active]
    grad = np.zeros_like(h)
    for k in ks:
        grad += gradient_for_shift(h, k)
    if ks.size:
        grad /= ks.size
    return grad, max_val

# ---------- Optimization primitives ----------
def gradient_descent(h_start, target_sum, max_iter, deadline, rng):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    min_alpha = 1e-6
    momentum = np.zeros_like(h)
    beta = 0.9
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad, cur_c5 = subgradient_max_overlap(h, dx)
        momentum = beta * momentum + (1 - beta) * grad
        new_h = h - alpha * momentum
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 - 1e-12:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                momentum.fill(0.0)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            if best_c5 < 0.3808:
                break
    return best_h, best_c5

def random_feasible(n, rng, target_sum):
    v = rng.random(n)
    return project_to_box_sum(v, target_sum)

def smooth_initial(n):
    """Cosine wave ranging from 0 to 1."""
    x = np.arange(n) / n * 2 * np.pi
    return 0.5 + 0.5 * np.cos(x)

def local_swap_improve(h, max_swaps, rng):
    """Tiny mass swaps between two coordinates to lower C5."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_swaps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.1
        if eps <= 0:
            continue
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
    return best_h, best_c5

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Determine dimensionality and target sum ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
    target_sum = n / 2.0

    # ---- Initial candidates ----
    h0 = random_feasible(n, rng, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # smooth start
    h_smooth = smooth_initial(n)
    h_smooth = project_to_box_sum(np.clip(h_smooth, 0.0, 1.0), target_sum)
    c5_smooth = compute_c5(h_smooth)
    if c5_smooth < best_c5:
        best_h, best_c5 = h_smooth, c5_smooth

    # quick random sampling
    deadline = start_time + budget_s - 5.0
    for _ in range(200):
        if time.time() > deadline:
            break
        h_rand = random_feasible(n, rng, target_sum)
        c5_rand = compute_c5(h_rand)
        if c5_rand < best_c5:
            best_h, best_c5 = h_rand, c5_rand
            if best_c5 < 0.3808:
                break

    # ---- Main stochastic search loop ----
    while time.time() < deadline:
        # generate a start point
        if rng.random() < 0.5:
            h_start = best_h + rng.normal(scale=0.04, size=n)
        else:
            h_start = random_feasible(n, rng, target_sum)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        # gradient descent phase
        gd_iters = 3000
        h_gd, c5_gd = gradient_descent(h_start, target_sum, gd_iters, deadline, rng)

        # local swap refinement
        h_sw, c5_sw = local_swap_improve(h_gd, 200, rng)

        if c5_sw < best_c5:
            best_h, best_c5 = h_sw, c5_sw
            if best_c5 < 0.3808:
                break

    # ---- Final projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
