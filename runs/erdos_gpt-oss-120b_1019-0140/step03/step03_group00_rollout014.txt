def gd_momentum(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr == max_corr)[0] - (n-1)
        # optionally restrict to near max within tolerance to avoid many shifts
        # but we can take all max as before.
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1-beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # jitter occasionally?
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h_bin, max_swaps, rng, time_left):
    h = h_bin.copy()
    n = h.shape[0]
    best_score = compute_c5(h)
    # create lists of indices
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    swaps = 0
    while swaps < max_swaps and time_left() > 0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        swaps += 1
    return h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.shape[0]
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low, lam_high = -1e5, 1e5
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lo, hi)
        tot = x.sum()
        if tot > s:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        term1 = np.zeros(n, dtype=h.dtype)
        term2 = np.zeros(n, dtype=h.dtype)
        if shift >= 0:
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gd_momentum(h0, max_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr == max_corr)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0 / n)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def binary_round(h):
    n = h.shape[0]
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    b[idx[:n//2]] = 1.0
    return b

def binary_hill_climb(h_bin, max_swaps, rng, time_left):
    h = h_bin.copy()
    n = h.shape[0]
    best_score = compute_c5(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    swaps = 0
    while swaps < max_swaps and time_left() > 0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
        swaps += 1
    return h, best_score
