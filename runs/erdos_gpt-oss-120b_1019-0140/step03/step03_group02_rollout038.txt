    n = h.shape[0]
    # copy corr to new
    new_corr = corr.copy()
    # old values:
    hi = 1.0
    hj = 0.0
    # for each shift k from -(n-1) to n-1:
    # compute delta contributions as per formulas.
    for k in range(-(n-1), n):

def binary_local_search_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    # ensure binary values exactly 0 and 1
    h = (h > 0.5).astype(float) # but if already binary fine.
    # ensure sum = n/2:
    # Use projection onto box sum maybe; but after binary there may be exactly n/2 due to generation.
    # Compute initial correlation and max
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    cur_score = max_corr * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score

    # temperature schedule
    # Determine iterations budget: can be based on time_left
    # We'll compute max_iters dynamic: big number but break early
    max_iters = int(min(500000, max(5000, time_left() * 1500)))  # approx 1500 iterations per sec
    T0 = 0.02  # maybe bigger? We'll use 0.03

    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # find shift(s) achieving max_corr
        shift_idxs = np.where(corr == max_corr)[0]
        # random select one shift
        shift_idx = rng.choice(shift_idxs)
        shift = shift_idx - (n - 1)

        # Determine candidate swap indices
        if shift >= 0:
            idx_range = np.arange(n - shift)
            A = idx_range[(h[idx_range] == 1) & (h[idx_range + shift] == 0)]
            B = idx_range[(h[idx_range] == 0) & (h[idx_range + shift] == 1)]
        else:
            s = -shift
            idx_range = np.arange(s, n)
            A = idx_range[(h[idx_range] == 1) & (h[idx_range - s] == 0)]
            B = idx_range[(h[idx_range] == 0) & (h[idx_range - s] == 1)]

        if A.size > 0 and B.size > 0:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            # fallback random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))

        # compute new corr using incremental update
        new_corr = update_corr_for_swap(corr, h, i, j)
        new_max_corr = np.max(new_corr)
        new_score = new_max_corr * (2.0 / n)

        # temperature
        T = T0 * (1 - it / max_iters)
        if T < 1e-6:
            T = 1e-6

        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)

        if accept:
            # perform swap in h
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            max_corr = new_max_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    # early termination if target reached
                    # but still we can continue to use it
                    pass

    return best_h, best_score

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust if f(lam_low) < s:
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def update_corr_for_swap(corr, h, i, j):
    n = h.shape[0]
    new_corr = corr.copy()
    # old values for i and j
    # hi = 1.0, hj = 0.0
    for k in range(-(n - 1), n):
        idx = k + (n - 1)
        # i as first
        if k >= 0 and i + k < n:
            new_corr[idx] -= (1.0 - h[i + k])
        # i as second
        if i - k >= 0 and i - k < n:
            new_corr[idx] += h[i - k]
        # j as first
        if k >= 0 and j + k < n:
            new_corr[idx] += (1.0 - h[j + k])
        # j as second
        if j - k >= 0 and j - k < n:
            new_corr[idx] -= h[j - k]
    new_corr[n - 1] = 0.0
    return new_corr

def binary_local_search_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    # ensure binary (0 or 1)
    h = (h > 0.5).astype(float)
    # ensure balanced sum: can project onto box sum if needed but swap preserves sum
    # compute initial correlation and max
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr[n - 1] = 0.0
    max_corr = np.max(corr)
    cur_score = max_corr * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    # temperature schedule based on remaining time
    max_iters = int(min(500000, max(5000, time_left() * 1500)))
    T0 = 0.05
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # find shifts achieving max_corr
        shift_idxs = np.where(corr == max_corr)[0]
        shift_idx = rng.choice(shift_idxs)
        shift = shift_idx - (n - 1)
        # determine swap candidate
        if shift >= 0:
            idx_range = np.arange(n - shift)
            A = idx_range[(h[idx_range] == 1) & (h[idx_range + shift] == 0)]
            B = idx_range[(h[idx_range] == 0) & (h[idx_range + shift] == 1)]
        else:
            s = -shift
            idx_range = np.arange(s, n)
            A = idx_range[(h[idx_range] == 1) & (h[idx_range - s] == 0)]
            B = idx_range[(h[idx_range] == 0) & (h[idx_range - s] == 1)]
        if A.size > 0 and B.size > 0:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        # compute new correlation incrementally
        new_corr = update_corr_for_swap(corr, h, i, j)
        new_max_corr = np.max(new_corr)
        new_score = new_max_corr * (2.0 / n)
        # temperature
        T = T0 * (1.0 - it / max_iters)
        if T < 1e-6:
            T = 1e-6
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            # perform swap in h
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            max_corr = new_max_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    # early exit if target reached
                    break
    return best_h, best_score

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def update_corr_for_swap(corr, h, i, j):
    n = h.shape[0]
    new_corr = corr.copy()
    for k in range(-(n - 1), n):
        idx = k + (n - 1)
        ii = i + k
        if 0 <= ii < n:
            new_corr[idx] -= (1.0 - h[ii])
        ii2 = i - k
        if 0 <= ii2 < n:
            new_corr[idx] += h[ii2]
        jj = j + k
        if 0 <= jj < n:
            new_corr[idx] += (1.0 - h[jj])
        jj2 = j - k
        if 0 <= jj2 < n:
            new_corr[idx] -= h[jj2]
    new_corr[n - 1] = 0.0
    return new_corr

def binary_local_search_sa(h0, rng, time_left, target=0.3808):
    h = (h0 > 0.5).astype(float)  # enforce binary
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr[n - 1] = 0.0
    max_corr = np.max(corr)
    cur_score = max_corr * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    max_iters = int(min(200000, max(2000, time_left() * 1500 / n)))
    T0 = 0.05
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        shift_idxs = np.where(corr == max_corr)[0]
        shift_idx = rng.choice(shift_idxs)
        shift = shift_idx - (n - 1)
        if shift >= 0:
            idx_range = np.arange(n - shift)
            A = idx_range[(h[idx_range] == 1) & (h[idx_range + shift] == 0)]
            B = idx_range[(h[idx_range] == 0) & (h[idx_range + shift] == 1)]
        else:
            s = -shift
            idx_range = np.arange(s, n)
            A = idx_range[(h[idx_range] == 1) & (h[idx_range - s] == 0)]
            B = idx_range[(h[idx_range] == 0) & (h[idx_range - s] == 1)]
        if A.size > 0 and B.size > 0:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        new_corr = update_corr_for_swap(corr, h, i, j)
        new_max_corr = np.max(new_corr)
        new_score = new_max_corr * (2.0 / n)
        T = T0 * (1.0 - it / max_iters)
        if T < 1e-6:
            T = 1e-6
        delta = new_score - cur_score
        accept = delta < 0 or rng.random() < np.exp(-delta / T)
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            max_corr = new_max_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_scales = [64, 128, 256, 512]
    best_h = None
    best_score = np.inf

    # --- smallest scale candidates ---
    n0 = n_scales[0]
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_n0 = upsample(init_full, n0) if init_full.size != n0 else init_full.copy()
        candidates.append(project_onto_box_sum(init_n0, n0 / 2.0))
        # also a binary version derived from it
        order = np.argsort(init_n0)
        binary = np.zeros(n0, dtype=float)
        binary[order[n0 // 2:]] = 1.0
        binary = project_onto_box_sum(binary, n0 / 2.0)
        candidates.append(binary)
    candidates.append(thue_morse(n0).astype(float))
    for _ in range(3):
        candidates.append(random_binary_balanced(n0, rng).astype(float))

    # binary SA on smallest scale
    for h0 in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = binary_local_search_sa(h0, rng, time_left, target=0.3808)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # optional continuous refinement at n0
    if time_left() > 1.0 and best_h is not None:
        iters = int(min(3000, max(500, time_left() * 5)))
        h_ref, sc_ref = smooth_grad_descent(best_h, n_iter=iters,
                                            step0=0.5, alpha_start=1.0,
                                            alpha_end=6.0, beta=0.9,
                                            time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # --- larger scales ---
    for n in n_scales[1:]:
        if time_left() < 2.0:
            break
        # upsample current best
        h_up = upsample(best_h, n)
        h_up = project_onto_box_sum(h_up, n / 2.0)

        # generate candidates for this scale
        cand = [h_up]
        # binary version of upsampled
        order = np.argsort(h_up)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        binary = project_onto_box_sum(binary, n / 2.0)
        cand.append(binary)
        cand.append(thue_morse(n).astype(float))
        for _ in range(2):
            cand.append(random_binary_balanced(n, rng).astype(float))

        for h0 in cand:
            if time_left() < 1.5:
                break
            h_opt, sc_opt = binary_local_search_sa(h0, rng, time_left, target=0.3808)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= 0.3808:
                    pass

        # continuous refinement at this scale
        if time_left() > 1.0:
            iters = int(min(4000, max(500, time_left() * 5)))
            h_ref, sc_ref = smooth_grad_descent(best_h, n_iter=iters,
                                                step0=0.5, alpha_start=1.0,
                                                alpha_end=6.0, beta=0.9,
                                                time_left=time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
