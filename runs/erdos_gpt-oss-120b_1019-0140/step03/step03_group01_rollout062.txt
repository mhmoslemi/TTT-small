def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    n = 1024
    target = 0.3808
    best_h = None
    best_score = np.inf
    # initial candidate from initial_h_values if exists
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # if length not n, upsample.
        if init_full.size != n:
            x_old = np.linspace(0, 1, init_full.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init_full = np.interp(x_new, x_old, init_full)
        cand = project_onto_box_sum(init_full, n/2.0)
        score = compute_c5(cand)
        best_h, best_score = cand, score
    # Also add some random continuous candidates
    for _ in range(5):
        if time_left() < 0.1:
            break
        v = rng.random(n)
        v = project_onto_box_sum(v, n/2.0)
        sc = compute_c5(v)
        if sc < best_score:
            best_h, best_score = v, sc
    # Random search over three-step parameters
    while time_left() > 0.1:
        # sample a and alpha
        a = rng.random()  # uniform [0,1)
        alpha = rng.random() * 0.99 + 0.01  # [0.01,1)
        # compute b
        b = a + (1.0 - a) / alpha
        if b > 2.0:
            continue
        # build h
        dx = 2.0 / n
        x = np.arange(n) * dx
        h_raw = np.where(x < a, 1.0, np.where(x < b, alpha, 0.0))
        # project onto sum constraint
        h = project_onto_box_sum(h_raw, n/2.0)
        sc = compute_c5(h)
        if sc < best_score:
            best_h, best_score = h, sc
            if best_score <= target:
                break
    # Local refinement using SLSQP if time left
    if time_left() > 5.0:
        # SLSQP on best_h
        h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=5000)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
    # final projection guard
    best_h = project_onto_box_sum(best_h, n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
