def binary_hill_climb(h_bin, max_iter, rng):
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    for it in range(max_iter):
        # select random pair i (1) and j (0)
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert with probability based on temperature
            # use cooling schedule: T = start * (end/start)^(it/max_iter)
            t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
            if rng.random() < np.exp(-(new_score - cur_score) / (t+1e-12)):
                cur_score = new_score
                # keep change
                if new_score < best_score:
                    best_score = new_score
                    best_h = cur_h.copy()
            else:
                cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def gradient_descent_continuous(h0, n_iter, step0, beta):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        # compute correlation array
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        # get indices of shifts near max
        eps = 1e-9
        idxes = np.where(corr_full >= max_corr - eps)[0]
        # convert to shifts
        shifts = idxes - (n - 1)
        # compute gradient based on those shifts
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def gradient_descent_continuous(h0, n_iter, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        # include shifts within small epsilon relative to max
        eps = max(1e-9, max_corr * 1e-3)
        idxes = np.where(corr_full >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        if shifts.size == 0:
            shifts = np.array([0], dtype=int)
        grad = compute_gradients_for_shifts(h, shifts) / max(1, shifts.size)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x : lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C₅ ≈ max_k ∫h(x)(1−h(x+k))dx on the discretised grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1 as float)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for the given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad

def gradient_descent_continuous(h0, n_iter, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr_full)
        eps = max(1e-9, max_corr * 1e-3)
        idxes = np.where(corr_full >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        if shifts.size == 0:
            shifts = np.array([0], dtype=int)
        grad = compute_gradients_for_shifts(h, shifts) / max(1, shifts.size)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP refinement of a candidate on the small grid."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simulated annealing on a balanced binary vector."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def time_remaining(start_time, budget):
    return max(0.0, budget - (time.time() - start_time))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------- small‑scale exploration ----------
    n_small = 64
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_small:
            small = downsample(init_full, n_small)
        else:
            small = init_full.copy()
        small = project_onto_box_sum(small, n_small / 2.0)
        candidates.append(small)
    candidates.append(thue_morse(n_small))
    candidates.append(random_binary_balanced(n_small, rng))
    candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h = None
    best_score = np.inf

    # SLSQP refinements on small grid
    for cand in candidates:
        if time_remaining(start, budget_s) < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=800)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Additional random SLSQP starts while time permits
    while time_remaining(start, budget_s) > 5.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=600)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Gradient descent on small scale
    if best_h is not None and time_remaining(start, budget_s) > 5.0:
        max_iter_small = int(min(20000, time_remaining(start, budget_s) * 1500))
        if max_iter_small > 0:
            h_gd, sc_gd = gradient_descent_continuous(best_h,
                                                      n_iter=max_iter_small,
                                                      step0=0.5, beta=0.9)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # -------- upscale to large grid ----------
    n_large = 512
    if best_h is None:
        best_h_large = random_binary_balanced(n_large, rng)
    else:
        best_h_large = upsample(best_h, n_large)
    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    best_score_large = compute_c5(best_h_large)
    if best_score_large < best_score:
        best_score, best_h = best_score_large, best_h_large.copy()

    # -------- continuous gradient descent on large grid ----------
    remaining = time_remaining(start, budget_s)
    if remaining > 6.0:
        # allocate roughly 70 % of remaining time to gradient descent
        time_for_gd = remaining * 0.7
        max_iters_large = int(min(500000, time_for_gd * 1500))
        if max_iters_large > 0:
            h_gd, sc_gd = gradient_descent_continuous(best_h,
                                                      n_iter=max_iters_large,
                                                      step0=0.5, beta=0.9)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # -------- binary simulated annealing ----------
    remaining = time_remaining(start, budget_s)
    if remaining > 5.0:
        # produce a balanced binary start from the current best
        idx = np.argsort(best_h)[-n_large // 2:]
        h_bin = np.zeros(n_large, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(min(4_000_000, remaining * 1500))
        temp_start = 0.1
        temp_end = 1e-6
        h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                      temp_start, temp_end, rng)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
