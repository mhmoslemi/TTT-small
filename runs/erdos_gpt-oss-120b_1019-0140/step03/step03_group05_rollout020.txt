def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    # find lambda such that sum(clip(v + lambda, lo, hi)) = target
    lo_lambda = -np.max(v) - 10.0
    hi_lambda = (hi - np.min(v)) + 10.0
    for _ in range(50):
        lam = (lo_lambda + hi_lambda) / 2.0
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s < target:
            lo_lambda = lam
        else:
            hi_lambda = lam
    return np.clip(v + (lo_lambda + hi_lambda) / 2.0, lo, hi)

def differential_evolution(pop, compute_c5, target_sum, max_iter, start_time, budget_s):
    n = pop.shape[1]
    best_idx = np.argmin([compute_c5(ind) for ind in pop])
    best = pop[best_idx].copy()
    best_score = compute_c5(best)
    # set parameters
    F = 0.5
    CR = 0.9
    # control loop
    generation = 0
    while time.time() < start_time + budget_s - 2.0:
        generation += 1
        # adapt F,CR optionally
        new_pop = pop.copy()
        for i in range(pop.shape[0]):
            # pick three distinct others
            idxs = list(range(pop.shape[0]))
            idxs.remove(i)
            a,b,c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # clip and project to sum
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)
            # crossover
            cross = np.random.rand(n) < CR
            if not cross.any():
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])
            # ensure feasibility
            trial = np.clip(trial,0.0,1.0)
            trial = project_to_box_sum(trial, target_sum)
            trial_score = compute_c5(trial)
            # selection
            parent_score = compute_c5(pop[i])
            if trial_score < parent_score:
                new_pop[i] = trial
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
        pop = new_pop
        # optionally break if target reached
        if best_score <= 0.3808:
            break
    return best, best_score, pop

import time
import numpy as np

# ----- Core utilities --------------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    lower = -np.max(v) - 10.0
    upper = (hi - np.min(v)) + 10.0
    for _ in range(60):
        lam = (lower + upper) * 0.5
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s < target:
            lower = lam
        else:
            upper = lam
    x = np.clip(v + (lower + upper) * 0.5, lo, hi)
    # tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < hi - 1e-12)[0]
        else:
            idx = np.where(x > lo + 1e-12)[0]
        if idx.size:
            x[idx[0]] += diff
            x = np.clip(x, lo, hi)
    return x


def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad


def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max‑overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5


# ----- Differential Evolution -----------------------------------------------
def differential_evolution(pop, target_sum, budget_s, start_time, rng):
    """Simple DE (best/1/bin) with box‑sum projection."""
    pop_size, n = pop.shape
    scores = np.array([compute_c5(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    F = 0.8
    CR = 0.9
    generation = 0
    safety = 2.0

    while time.time() < start_time + budget_s - safety:
        generation += 1
        new_pop = pop.copy()
        for i in range(pop_size):
            # choose two distinct indices not equal to i
            idxs = np.arange(pop_size)
            mask = idxs != i
            a, b = rng.choice(idxs[mask], 2, replace=False)
            mutant = best + F * (pop[a] - pop[b])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                new_pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
                    if best_score <= 0.3808:
                        return best, best_score, new_pop, scores
        pop = new_pop

        # occasional random injection to keep diversity
        if generation % 50 == 0:
            worst_idx = int(np.argmax(scores))
            rnd = rng.random(n)
            pop[worst_idx] = project_to_box_sum(rnd, target_sum)
            scores[worst_idx] = compute_c5(pop[worst_idx])
        # early exit if target reached
        if best_score <= 0.3808:
            break
    return best, best_score, pop, scores
