import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p ** 0.5)
    for i in range(3, r + 1, 2):
        if p % i == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def simulated_annealing(h_start, rng, time_limit):
    target_val = 0.3808
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur

    start_loc = time.time()
    T0 = 0.005  # initial temperature

    while True:
        elapsed = time.time() - start_loc
        if elapsed > time_limit - 0.01:
            break
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        if T < 1e-12:
            T = 1e-12

        zeros = np.where(h_cur == 0)[0]
        ones = np.where(h_cur == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)

        h_cur[i] = 1.0
        h_cur[j] = 0.0
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
                if best_c5 < target_val:
                    return best_h, best_c5
        else:
            h_cur[i] = 0.0
            h_cur[j] = 1.0

    # deterministic hill climb on the best found
    refine_start = time.time()
    while time.time() - refine_start < min(0.05 * time_limit, 1.0):
        improved = False
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        for i in zeros:
            for j in ones:
                best_h[i] = 1.0
                best_h[j] = 0.0
                c5_try = compute_c5(best_h)
                if c5_try + 1e-12 < best_c5:
                    best_c5 = c5_try
                    improved = True
                    break
                best_h[i] = 0.0
                best_h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5
