def compute_c5_fft(h):
    n = h.shape[0]
    # compute cyclic correlation; but we need linear correlation (full). For linear we can pad zeros length 2n.
    f_h = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    f_one_minus_h = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n)]))
    corr = np.fft.irfft(f_h * np.conj(f_one_minus_h))
    # corr length 2n-1 (valid range). The correlation starts at index 0 corresponding to shift - (n-1). The max over all is max(corr[:2*n-1]).
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

def binary_hill_climb(h_bin, max_iter, rng):
    n = h_bin.size
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    for it in range(max_iter):
        # random swap indices
        i = rng.integers(n)  # position with 1 maybe
        # pick i with 1
        if best_h[i] == 0:
            # choose a different index with 1
            ones = np.flatnonzero(best_h)
            if ones.size == 0: break
            i = rng.choice(ones)
        # pick j with 0
        zeros = np.flatnonzero(best_h == 0)
        if zeros.size == 0: break
        j = rng.choice(zeros)
        # swap
        best_h[i], best_h[j] = 0, 1
        new_score = compute_c5(best_h)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best_h[i], best_h[j] = 1, 0
    return best_h, best_score

def gradient_descent_momentum(h0, n_iter=2000, step0=0.5, beta=0.9):
   h = h0.copy()
   n = h.size
   target_sum = n / 2.0
   best_h = h.copy()
   best_score = compute_c5(h)
   momentum = np.zeros_like(h)
   for it in range(1, n_iter+1):
       # compute correlation
       corr = np.correlate(h, 1.0 - h, mode='full')
       max_corr = np.max(corr)
       eps = 1e-12
       idxes = np.where(corr >= max_corr - eps)[0]
       shifts = idxes - (n - 1)
       grad = compute_gradients_for_shifts(h, shifts)
       momentum = beta * momentum + (1 - beta) * grad
       eta = step0 / np.sqrt(it)
       h_new = h - eta * momentum
       h_new = project_onto_box_sum(h_new, target_sum)
       score = compute_c5(h_new)
       if score < best_score:
           best_score = score
           best_h = h_new.copy()
       h = h_new
   return best_h, best_score
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_target = 512
    # initial candidate
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init_full, n_target)
    else:
        # random binary balanced
        h0 = np.zeros(n_target, dtype=float)
        idx = rng.choice(n_target, size=n_target//2, replace=False)
        h0[idx] = 1.0
    # ensure sum constraint exact
    h0 = project_onto_box_sum(h0, n_target/2.0)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Also evaluate binary version of initial
    binary_h0 = np.zeros_like(best_h)
    binary_h0[np.argsort(best_h)[-n_target//2:]] = 1.0
    score_bin0 = compute_c5(binary_h0)
    if score_bin0 < best_score:
        best_h, best_score = binary_h0, score_bin0

    # Determine number of gradient descent starts
    # We'll loop until time runs out.
    while time_left() > 10.0:
        # Random start: either use slight perturbation of best_h or random binary
        if rng.random() < 0.3 and best_score < np.inf:
            # start near best_h with small noise
            h_start = best_h + rng.normal(scale=0.1, size=n_target)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_onto_box_sum(h_start, n_target/2.0)
        else:
            # random binary balanced
            h_start = np.zeros(n_target, dtype=float)
            idx = rng.choice(n_target, size=n_target//2, replace=False)
            h_start[idx] = 1.0
        # Gradient descent
        # Estimate max iterations based on time left: allocate maybe up to half remaining time.
        remaining = time_left()
        max_iters = int(min(6000, remaining * 1500 * 0.4))
        if max_iters < 1000:
            max_iters = int(min(2000, remaining * 1500 * 0.5))
        if max_iters < 500:
            break
        h_gd, sc_gd = gradient_descent_momentum(h_start, n_iter=max_iters,
                                                step0=0.5, beta=0.9)
        # Update best
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd

        # Binary projection of GD result
        binary_h = np.zeros_like(h_gd)
        binary_h[np.argsort(h_gd)[-n_target//2:]] = 1.0
        score_bin = compute_c5(binary_h)
        if score_bin < best_score:
            best_h, best_score = binary_h, score_bin

        # Binary hill climb
        # Determine hill iterations based on remaining time
        remaining = time_left()
        hill_iters = int(min(5000, remaining * 1500 * 0.25))
        if hill_iters >= 200:
            h_hc, sc_hc = binary_hill_climb(binary_h, max_iter=hill_iters, rng=rng)
            if sc_hc < best_score:
                best_h, best_score = h_hc, sc_hc

    # Final projection
    best_h = project_onto_box_sum(best_h, n_target/2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_target

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral approximated on the discrete grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_gradients_for_shifts(h, shifts):
    """Subgradient of the max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=2000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def binary_hill_climb(h_bin, max_iter, rng):
    """Simple hill climbing on binary balanced vectors (swap 1↔0)."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    for _ in range(max_iter):
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(best_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i], best_h[j] = 0.0, 1.0
        new_score = compute_c5(best_h)
        if new_score < best_score:
            best_score = new_score
        else:
            best_h[i], best_h[j] = 1.0, 0.0
    return best_h, best_score
