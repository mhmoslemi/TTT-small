def binary_sa_optimize(n, rng, time_left, target=None, max_restarts=10):
    best_h = None; best_score = inf
    # count restarts based on time_left proportion
    while time_left() > 1.0 and max_restarts > 0:
        max_restarts -= 1
        # initialize random balanced binary
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
        # maintain zero and one position lists
        zero_pos = list(set(range(n)) - set(idx))
        one_pos = list(idx)
        # compute initial score
        score = compute_c5(h)
        best_local_h = h.copy()
        best_local_score = score
        # SA parameters
        max_iters = int(min(200000, max(1, time_left() * 2000))) # similar to prior
        T0 = 0.02
        for it in range(1, max_iters+1):
            if time_left() < 0.01: break
            # propose swap
            i = zero_pos[rng.integers(len(zero_pos))]
            j = one_pos[rng.integers(len(one_pos))]
            # swap
            h[i] = 1.0; h[j] = 0.0
            # move i to one list, j to zero list
            zero_pos.remove(i); one_pos.append(i)
            one_pos.remove(j); zero_pos.append(j)
            new_score = compute_c5(h)
            delta = new_score - score
            T = T0 * (1 - it / max_iters)
            accept = False
            if delta < 0:
                accept = True
            else:
                if T > 0 and rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local_h = h.copy()
                    if target is not None and best_local_score <= target:
                        break
            else:
                # revert swap
                h[i] = 0.0; h[j] = 1.0
                # revert lists
                one_pos.remove(i); zero_pos.append(i)
                zero_pos.remove(j); one_pos.append(j)
        # update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_h = best_local_h.copy()
            if target is not None and best_score <= target:
                break
    return best_h, best_score
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # Determine iteration budget based on time left
    max_iters = int(min(400000, max(1, time_left() * 3000)))  # roughly 3000 swaps per sec
    T0 = 0.02
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # pick random zero index
        i = rng.integers(n)
        while h[i] != 0.0:
            i = rng.integers(n)
        # pick random one index
        j = rng.integers(n)
        while h[j] != 1.0:
            j = rng.integers(n)
        # propose swap
        h[i], h[j] = 1.0, 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = T0 * np.exp(-it / max_iters)   # exponential cooling
        accept = False
        if delta < 0:
            accept = True

                best_score, best_h = new_score, h.copy()
                if best_score <= target:
                    break
        else:
            # revert swap
            h[i], h[j] = 0.0, 1.0

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure the interval contains a root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation resampling (preserves shape, not sum)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter, step0, alpha_start, alpha_end,
                        beta, time_left):
    """Soft‑max (smooth) gradient descent with momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

    if time_left() > 5.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=target_bound)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # ----- final continuous refinement -----
    if time_left() > 5.0:
        h_ref, sc_ref = smooth_grad_descent(
            best_h,
            n_iter=4000,
            step0=0.3,
            alpha_start=0.5,
            alpha_end=10.0,
            beta=0.9,
            time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # ----- extra binary SA on the refined result (optional) -----
    if time_left() > 5.0:
        n = best_h.shape[0]
        order = np.argsort(best_h)
        binary = np.zeros(n, dtype=float)
        binary[order[n // 2:]] = 1.0
        bin_h2, bin_sc2 = binary_hill_climb_sa(binary, rng, time_left, target=target_bound)
        if bin_sc2 < best_score:
            best_score, best_h = bin_sc2, bin_h2.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
