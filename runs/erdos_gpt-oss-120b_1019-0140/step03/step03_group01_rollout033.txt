def anneal_binary(h, rng, time_left_func):
    n = len(h)
    total = n // 2
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    cur_h = h.copy()
    cur_score = best_score
    T = 1.0
    T_min = 1e-4
    decay = 0.9999
    # For each iteration, while time left > threshold
    while time_left_func() > 0.05:
        # choose random swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        # if degenerate, break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = compute_c5_fast(cur_h)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        # update temperature
        T = max(T_min, T * decay)
    return best_h, best_score

def binary_targeted_anneal(h, rng, time_left_func, target_corr):
    n = len(h)
    cur_h = h.copy()
    cur_corr = compute_corr_fast(cur_h)
    cur_max = int(np.max(cur_corr))
    best_h = cur_h.copy()
    best_max = cur_max
    T = 2.0
    T_min = 0.0001
    decay = 0.99995
    while time_left_func() > 0.1:
        # find shift indices where max
        idxes = np.where(cur_corr == cur_max)[0]
        # we may have many, random choose one
        idx = rng.choice(idxes)
        k = idx - (n - 1)
        # get mismatched positions for this shift
        start = max(0, -k)
        end = min(n, n - k)
        # slice
        h_slice = cur_h[start:end]
        h_shift = cur_h[start+k:end+k]
        mismatches = np.where((h_slice == 1) & (h_shift == 0))[0]
        if mismatches.size == 0:
            # no mismatches (should not happen if max>0)
            # try random swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = start + rng.choice(mismatches)
            j = i + k
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # swap 1 and 0
        # recompute new correlation
        new_corr = compute_corr_fast(cur_h)
        new_max = int(np.max(new_corr))
        # acceptance
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True
        if accept:
            cur_corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = cur_h.copy()
                # check target
                if best_max <= target_corr:
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # decay temperature
        T = max(T_min, T * decay)
    return best_h, best_max

def compute_corr_fast(h):
    n = h.shape[0]
    # compute convolution of h with reversed complement
    a = h
    b = (1.0 - h)[::-1]
    L = 2 * n
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * B, n=L)[:2*n-1]
    return c

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # 0/1 Thue-Morse sequence
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=int)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def compute_corr_fast(h):
    """Linear correlation of h with (1-h) via FFT. Returns full array length 2n-1."""
    n = h.shape[0]
    a = h.astype(float)
    b = (1.0 - h)[::-1].astype(float)
    L = 2 * n
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * B, n=L)[:2 * n - 1]
    return c

def binary_targeted_anneal(h0, rng, time_left_func, target_corr):
    """Hill‑climbing with simulated‑annealing style swaps targeting the maximal shift."""
    n = h0.shape[0]
    h = h0.copy()
    corr = compute_corr_fast(h)
    cur_max = int(np.rint(np.max(corr)))
    best_h = h.copy()
    best_max = cur_max

    T = 5.0
    T_min = 1e-4
    decay = 0.9999

    while time_left_func() > 0.1:
        # locate shifts attaining current maximum
        idxes = np.where(np.rint(corr) == cur_max)[0]
        k_idx = rng.choice(idxes)
        k = k_idx - (n - 1)

        # find a mismatched pair (1,0) for this shift, if any
        start = max(0, -k)
        end = min(n, n - k)
        h_slice = h[start:end]
        h_shift = h[start + k:end + k]
        mism = np.where((h_slice == 1) & (h_shift == 0))[0]
        if mism.size == 0:
            # fallback: random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = start + rng.choice(mism)
            j = i + k  # guaranteed 0
        # perform swap
        h[i], h[j] = h[j], h[i]

        new_corr = compute_corr_fast(h)
        new_max = int(np.rint(np.max(new_corr)))

        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                if best_max <= target_corr:
                    break
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        T = max(T_min, T * decay)

    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    target = 0.3808
    n = 512                     # number of sample points
    target_corr = int(np.floor(target * n / 2))

    # prepare starting candidates
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        up = upsample(init_full, n)
        # convert to binary while satisfying the sum constraint
        idx = np.argpartition(-up, n // 2)[: n // 2]
        bin_h = np.zeros(n, dtype=int)
        bin_h[idx] = 1
        candidates.append(bin_h)
    candidates.append(thue_morse(n).astype(int))
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    best_h = None
    best_corr = np.iinfo(np.int64).max

    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, max_corr = binary_targeted_anneal(cand, rng, time_left, target_corr)
        if max_corr < best_corr:
            best_corr = max_corr
            best_h = h_opt.copy()
        if best_corr <= target_corr:
            break

    # random restarts while time permits
    while time_left() > 5.0:
        cand = random_binary_balanced(n, rng)
        h_opt, max_corr = binary_targeted_anneal(cand, rng, time_left, target_corr)
        if max_corr < best_corr:
            best_corr = max_corr
            best_h = h_opt.copy()
        if best_corr <= target_corr:
            break

    # fallback in the unlikely case no candidate succeeded
    if best_h is None:
        best_h = random_binary_balanced(n, rng)

    c5_bound = (2.0 / n) * best_corr
    return best_h.astype(float), c5_bound, n
