import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # ensure float for 1-h
    hh = h.astype(np.float64)
    corr = np.correlate(hh, 1.0 - hh, mode="full")
    return np.max(corr) * dx

def enforce_sum_binary(v, target, rng):
    current = int(v.sum())
    if current > target:
        surplus = current - target
        ones = np.where(v == 1)[0]
        idx = rng.choice(ones, size=surplus, replace=False)
        v[idx] = 0
    elif current < target:
        deficit = target - current
        zeros = np.where(v == 0)[0]
        idx = rng.choice(zeros, size=deficit, replace=False)
        v[idx] = 1
    return v

def generate_random_binary(n, target, rng):
    v = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=target, replace=False)
    v[idx] = 1
    return v

def crossover_one_point(p1, p2, rng):
    n = p1.shape[0]
    point = rng.integers(1, n)  # 1..n-1
    child = np.concatenate([p1[:point], p2[point:]]).astype(np.int8)
    return child

def mutate_swap(v, rng):
    ones = np.where(v == 1)[0]
    zeros = np.where(v == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return v
    i = rng.choice(ones)
    j = rng.choice(zeros)
    v[i], v[j] = 0, 1
    return v

def tournament_select(scores, tournament_size, rng):
    cand = rng.integers(0, len(scores), size=tournament_size)
    best = cand[np.argmin(scores[cand])]
    return best

def hill_climb_improve(h, cur_val, rng, max_iter, deadline):
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i], best_h[j] = 0, 1
        new_val = compute_c5(best_h)
        if new_val < best_val:
            best_val = new_val
        else:
            # revert
            best_h[i], best_h[j] = 1, 0
    return best_h, best_val

                child = crossover_one_point(p1, p2, rng)
            else:
                child = p1.copy()
            # mutation
            if rng.random() < mutation_rate:
                child = mutate_swap(child, rng)
            # repair (just in case)
            child = enforce_sum_binary(child, target, rng)
            children.append(child)
        new_pop = np.vstack([elite] + children)
        pop = new_pop

        # Occasionally refine the current global best with hill climbing
        if time.time() < deadline - 5.0:
            # allocate small time slice
            climb_deadline = min(deadline, time.time() + 0.8)
            refined_h, refined_val = hill_climb_improve(best_h, best_val, rng, max_iter=200, deadline=climb_deadline)
            if refined_val < best_val:
                best_val = refined_val
                best_h = refined_h.copy()
                if best_val < 0.3808:
                    break

    # Final local improvement
    if best_h is not None and time.time() < deadline - 0.2:
        final_deadline = deadline
        best_h, best_val = hill_climb_improve(best_h, best_val, rng, max_iter=500, deadline=final_deadline)

    # Ensure feasibility
    best_h = enforce_sum_binary(best_h.astype(np.int8), target, rng)
    best_h = best_h.astype(np.float64)

    final_val = compute_c5(best_h)
    return best_h, final_val, n
