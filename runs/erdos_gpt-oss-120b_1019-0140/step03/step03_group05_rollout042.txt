def compute_c5(h):
    n = len(h)
    weight = h.sum()
    autocorr = np.correlate(h, h, mode='full')
    # remove zero shift (center)
    mid = n - 1
    # ids non-zero: [0:mid] + [mid+1:2n-1]
    min_autocorr = np.min(np.concatenate((autocorr[:mid], autocorr[mid+1:])))  # or np.min(autocorr[np.arange(len(autocorr))!=mid])
    dx = 2.0 / n
    return (weight - min_autocorr) * dx

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    # Compute initial best C5
    weight = h.sum()
    autocorr = np.correlate(h, h, mode='full')
    mid = n - 1
    min_autocorr = np.min(np.concatenate((autocorr[:mid], autocorr[mid+1:])))
    best_c5 = (weight - min_autocorr) * dx
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute worst shift (min autocorr)
        autocorr = np.correlate(h, h, mode='full')
        mid = n - 1
        autocorr[mid] = np.inf
        min_idx = np.argmin(autocorr)
        k = min_idx - mid
        min_autocorr = autocorr[min_idx]
        cur_c5 = (h.sum() - min_autocorr) * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5

            # break if below target

                break
        # gradient for worst shift (i.e., shift k)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        # compute new C5
        weight_new = new_h.sum()
        autocorr_new = np.correlate(new_h, new_h, mode='full')
        mid = n - 1
        min_autocorr_new = np.min(np.concatenate((autocorr_new[:mid], autocorr_new[mid+1:])))

                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

            low = mid
        else:

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # For i < n - k: gradient contribution = 1 - h[i + k]
        grad[:n - k] += 1.0 - h[k:]
        # For i >= k: subtract h[i - k]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    weight = h.sum()
    autocorr = np.correlate(h, h, mode='full')
    mid = n - 1
    # compute min over nonzero shifts
    min_autocorr = np.min(np.concatenate((autocorr[:mid], autocorr[mid+1:])))
    return (weight - min_autocorr) * (2.0 / n)

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    weight = h.sum()
    autocorr = np.correlate(h, h, mode='full')
    mid = n - 1
    min_autocorr = np.min(np.concatenate((autocorr[:mid], autocorr[mid+1:])))
    best_c5 = (weight - min_autocorr) * dx
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # worst shift = min autocorr (excluding zero)
        autocorr = np.correlate(h, h, mode='full')
        mid = n - 1
        autocorr[mid] = np.inf
        min_idx = np.argmin(autocorr)
        k = min_idx - mid
        min_autocorr = autocorr[min_idx]
        cur_c5 = (target_sum - min_autocorr) * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 < 0.3808:
                break
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        weight_new = new_h.sum()
        autocorr_new = np.correlate(new_h, new_h, mode='full')
        mid = n - 1
        min_autocorr_new = np.min(np.concatenate((autocorr_new[:mid], autocorr_new[mid+1:])))
        new_c5 = (weight_new - min_autocorr_new) * dx
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    safety = 5.0
    # Determine dimension
    if 'initial_h_values' in globals():
        n = int(initial_h_values.shape[0])
        target_sum = n / 2.0
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    # first local refinement
    time_limit = start_time + budget_s - safety
    if time.time() < time_limit:
        gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter=800, time_limit=time_limit)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
    # a few random restarts
    for _ in range(3):
        if time.time() >= time_limit:
            break
        if rng.random() < 0.5:
            h_start = best_h + rng.normal(scale=0.05, size=n)
        else:
            h_start = rng.random(n)
        h_start = project_to_box_sum(np.clip(h_start, 0.0, 1.0), target_sum)
        gd_time = min(time_limit, time.time() + 300.0)
        gd_h, gd_c5 = gradient_descent(h_start, target_sum, max_iter=1200, time_limit=gd_time)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
        if best_c5 < 0.3808:
            break
    # Global search via differential evolution
    if time.time() < time_limit - 10.0:
        de_deadline = start_time + budget_s - safety
        def obj_de(x):
            h = project_to_box_sum(np.clip(x, 0.0, 1.0), target_sum)
            return compute_c5(h)
        bounds = [(0.0, 1.0)] * n
        try:
            result = opt.differential_evolution(
                obj_de,
                bounds,
                maxiter=30,
                popsize=15,
                seed=seed,
                polish=False,
                callback=lambda xk, convergence: time.time() > de_deadline,
                updating='deferred',
                workers=1
            )
            h_de = project_to_box_sum(np.clip(result.x, 0.0, 1.0), target_sum)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_h, best_c5 = h_de, c5_de
        except Exception:
            pass
    # Final local refinement
    if time.time() < time_limit:
        final_h, final_c5 = gradient_descent(best_h, target_sum, max_iter=2000,
                                            time_limit=time_limit)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5
    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), n
