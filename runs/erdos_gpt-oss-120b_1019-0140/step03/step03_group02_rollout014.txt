    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)

def binary_to_top_half(arr):
    # arr float, convert to binary with exactly half ones based on largest values
    n = arr.size
    idx = np.argsort(arr)
    bin_arr = np.zeros(n, dtype=float)
    bin_arr[idx[n//2:]] = 1.0
    return bin_arr

def crossover(parent1, parent2, rng):
    mask = rng.random(parent1.shape[0]) < 0.5
    child = np.where(mask, parent1, parent2)
    # repair balance
    n = child.size
    half = n // 2
    cur = int(child.sum())
    if cur > half:
        # need to turn (cur-half) ones to zeros
        ones = np.where(child == 1)[0]
        to_flip = rng.choice(ones, size=cur-half, replace=False)
        child[to_flip] = 0.0
    elif cur < half:
        zeros = np.where(child == 0)[0]
        to_flip = rng.choice(zeros, size=half-cur, replace=False)
        child[to_flip] = 1.0
    return child

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shifts from -(n-1) to n-1

        if w == 0.0:

        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
    """Thue–Morse binary sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def to_binary_top_half(v):
    """Convert real vector to binary with exactly half ones (top half values)."""
    n = v.size
    idx = np.argsort(v)
    b = np.zeros(n, dtype=float)
    b[idx[n // 2:]] = 1.0
    return b

def crossover(parent1, parent2, rng):
    """Uniform crossover preserving the exact half‑ones balance."""
    mask = rng.random(parent1.shape[0]) < 0.5
    child = np.where(mask, parent1, parent2)
    # repair to exact half ones
    n = child.size
    half = n // 2
    cur = int(child.sum())
    if cur > half:
        ones = np.where(child == 1)[0]
        to_flip = rng.choice(ones, size=cur - half, replace=False)
        child[to_flip] = 0.0
    elif cur < half:
        zeros = np.where(child == 0)[0]
        to_flip = rng.choice(zeros, size=half - cur, replace=False)
        child[to_flip] = 1.0
    return child

def mutate_swap(vec, rng, swaps=1):
    """Perform a given number of balanced swaps (0↔1)."""
    n = vec.size
    for _ in range(swaps):
        i0 = rng.integers(n)
        while vec[i0] != 0.0:
            i0 = rng.integers(n)
        i1 = rng.integers(n)
        while vec[i1] != 1.0:
            i1 = rng.integers(n)
        vec[i0] = 1.0
        vec[i1] = 0.0
    return vec

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shifts from -(n-1) to n-1
    for idx, k in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=12.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Momentum soft‑max gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Swap‑based simulated annealing on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(500000, max(5000, time_left() * 2000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i0 = rng.integers(n)
        while h[i0] != 0.0:
            i0 = rng.integers(n)
        i1 = rng.integers(n)
        while h[i1] != 1.0:
            i1 = rng.integers(n)
        h_new = h.copy()
        h_new[i0] = 1.0
        h_new[i1] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n = 256                       # target resolution (even)
    half = n // 2
    pop_size = 30
    population = []
    scores = []

    # helper to add a candidate
    def add_candidate(v):
        v = np.asarray(v, dtype=float)
        if v.size != n:
            v = upsample(v, n)
        # ensure binary balance for GA population
        v_bin = to_binary_top_half(v)
        population.append(v_bin)
        scores.append(compute_c5(v_bin))

    # 1. initial construction from parent if present
    if 'initial_h_values' in globals():
        add_candidate(initial_h_values)

    # 2. Thue‑Morse sequence
    add_candidate(thue_morse(n))

    # 3. random balanced binary vectors
    while len(population) < pop_size:
        add_candidate(random_binary_balanced(n, rng))

    # track best overall
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # ------- Genetic‑style exploration -------
    while time_left() > 12.0:
        # tournament selection for two parents
        idxs = rng.integers(len(population), size=4)
        p1 = population[idxs[np.argmin([scores[i] for i in idxs])]]
        idxs = rng.integers(len(population), size=4)
        p2 = population[idxs[np.argmin([scores[i] for i in idxs])]]
        child = crossover(p1, p2, rng)
        # mutate a few balanced swaps
        child = mutate_swap(child, rng, swaps=max(1, int(0.02 * n)))
        child_score = compute_c5(child)

        # replace worst if better
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score, best_h = child_score, child.copy()

    # ------- Simulated annealing refinement -------
    if time_left() > 8.0:
        best_h, best_score = binary_hill_climb_sa(best_h, rng, time_left,
                                                  target=0.3808)

    # ------- Continuous smooth refinement -------
    if time_left() > 4.0:
        max_iter = int(min(20000, max(5000, time_left() * 800)))
        best_h, best_score = smooth_grad_descent(best_h, n_iter=max_iter,
                                                 step0=0.5, alpha_start=1.0,
                                                 alpha_end=12.0, beta=0.9,
                                                 time_left=time_left)

    # final projection and score
    best_h = project_onto_box_sum(best_h, half)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
