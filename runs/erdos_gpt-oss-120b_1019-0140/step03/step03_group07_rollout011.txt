def project_capped_simplex(v, M, ub=1.0, max_iter=50, tol=1e-10):
    low = np.min(v) - ub
    high = np.max(v)
    for _ in range(max_iter):
        tau = (low + high) / 2.0
        w = np.clip(v - tau, 0.0, ub)
        s = np.sum(w)
        if abs(s - M) <= tol:
            return w
        if s > M:
            low = tau
        else:
            high = tau
    return np.clip(v - tau, 0.0, ub)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_of_overlap(h, k):
    n = h.shape[0]
    grad = np.zeros(n, dtype=float)
    if k >= 0:
        # indices where i in [0, n-1-k]
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] += -h[:n-k]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n-kk])
        grad[:n-kk] += -h[kk:]
    return grad

def project_capped_simplex(v, M, ub=1.0, max_iter=100, tol=1e-12):
    # Solve w = clip(v - tau, 0, ub) such that sum(w) = M
    low = np.min(v) - ub
    high = np.max(v)
    tau = 0.0
    for _ in range(max_iter):
        tau = (low + high) / 2.0
        w = np.clip(v - tau, 0.0, ub)
        s = w.sum()
        if abs(s - M) <= tol:
            return w
        if s > M:
            low = tau
        else:
            high = tau
    return np.clip(v - tau, 0.0, ub)

def optimise_for_n(n, rng, time_budget, start_time, total_budget):
    M = n // 2
    # initial solution
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        h = project_capped_simplex(h, M, ub=1.0)
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # Legendre seed for n=80
    if n == 80:
        p = 79

        h_leg = np.concatenate([leg, np.array([1.0])])
        h_leg = project_capped_simplex(h_leg, M, ub=1.0)
        c5_leg = compute_c5(h_leg)
        if c5_leg < best_c5:
            best_h = h_leg.copy()
            best_c5 = c5_leg
            h = h_leg.copy()
    alpha = 0.05
    no_improve = 0
    it = 0
    target = 0.3808
    while True:
        # time check
        now = time.time()
        if now - start_time >= time_budget - 0.2:
            break
        # compute overlaps and gradients
        dx = 2.0 / n
        overl = np.correlate(h, 1.0 - h, mode='full') * dx

        eps = 1e-12
        worst_idx = np.where(overl >= max_val - eps)[0]

            grads.append(gradient_of_overlap(h, k))
        grad = np.mean(grads, axis=0)
        norm = np.linalg.norm(grad)

            norm = np.linalg.norm(grad)
        direction = -grad / norm
        step = direction * alpha
        h_temp = h + step
        h_new = project_capped_simplex(h_temp, M, ub=1.0)
        c5_new = compute_c5(h_new)
        # update best

            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        # accept move if non-worsening

        if c5_new <= cur_c5 + 1e-12:
            h = h_new
        # adaptation
        if no_improve > 500:
            alpha *= 0.5
            no_improve = 0
            if alpha < 1e-5:
                # random restart around best
                if rng.random() < 0.5:
                    h = np.zeros(n, dtype=float)
                    idx = rng.choice(n, M, replace=False)
                    h[idx] = 1.0
                else:
                    h = best_h + rng.normal(scale=0.02, size=n)
                    h = project_capped_simplex(h, M, ub=1.0)
                alpha = 0.05
        # occasional perturbation
        if it % 200 == 0 and it > 0:
            h = best_h + rng.normal(scale=0.03, size=n)
            h = project_capped_simplex(h, M, ub=1.0)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_of_overlap(h, k):
    n = h.shape[0]
    grad = np.zeros(n, dtype=float)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] += -h[:n - k]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n - kk])
        grad[:n - kk] += -h[kk:]
    return grad

def project_capped_simplex(v, M, ub=1.0, max_iter=100, tol=1e-12):
    low = np.min(v) - ub
    high = np.max(v)
    tau = 0.0
    for _ in range(max_iter):
        tau = (low + high) / 2.0
        w = np.clip(v - tau, 0.0, ub)
        s = w.sum()
        if abs(s - M) <= tol:
            return w
        if s > M:
            low = tau
        else:
            high = tau
    return np.clip(v - tau, 0.0, ub)

def optimise_for_n(n, rng, time_budget, run_start, total_budget):
    M = n // 2
    # initial vector
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=float)
        h = project_capped_simplex(h, M, ub=1.0)
    else:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, M, replace=False)
        h[idx] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # legendre seed for n=80
    if n == 80:
        p = 79
        leg = np.zeros(p, dtype=float)
        for i in range(1, p):
            leg[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
        h_leg = np.concatenate([leg, np.array([1.0])])
        h_leg = project_capped_simplex(h_leg, M, ub=1.0)
        c5_leg = compute_c5(h_leg)
        if c5_leg < best_c5:
            best_h = h_leg.copy()
            best_c5 = c5_leg
            h = h_leg.copy()

    alpha = 0.05
    no_improve = 0
    it = 0
    target = 0.3808
    local_start = time.time()
    while True:
        if time.time() - local_start >= time_budget - 0.2:
            break
        dx = 2.0 / n
        overl = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = overl.max()
        eps = 1e-12

        grads = [gradient_of_overlap(h, idx - (n - 1)) for idx in worst_idx]

        if norm < 1e-12:

        direction = -grad / norm

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    candidates = []
    if 'initial_h_values' in globals():

    for n in [80, 96, 112, 128]:
        if n not in candidates:
            candidates.append(n)

    best_h = None
    best_c5 = np.inf
    best_n = None
    total_cands = len(candidates)

    for idx, n in enumerate(candidates):
        elapsed = time.time() - run_start
        remaining = budget_s - elapsed
        if remaining <= 5:
            break
        if idx == 0:
            alloc = min(remaining * 0.6, 600.0)
        else:
            alloc = min(remaining * 0.4 / max(1, total_cands - idx - 1), 300.0)

            break

    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    best_h = project_capped_simplex(best_h, best_n // 2, ub=1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
