def binary_hill_climb(h_bin, time_budget):
    # h_bin: array of 0/1, balanced
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    start = time.time()
    # get indices of ones and zeros
    ones = np.flatnonzero(best_h > 0.5)
    zeros = np.flatnonzero(best_h < 0.5)
    while time.time() - start < time_budget:
        if len(ones) == 0 or len(zeros) == 0:
            break
        # try a set of random swaps, say 20 per outer iteration
        improved = False
        for _ in range(20):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # compute new score after swap
            tmp = best_h[i]
            best_h[i] = 0.0
            best_h[j] = 1.0
            new_score = compute_c5(best_h)
            if new_score < best_score - 1e-9:
                # accept
                best_score = new_score
                # update ones/zeros arrays
                ones = np.flatnonzero(best_h > 0.5)
                zeros = np.flatnonzero(best_h < 0.5)
                improved = True
                break
            else:
                # revert
                best_h[i] = 1.0
                best_h[j] = 0.0
        if not improved:
            # no improvement found after trials, break
            break
    return best_h, best_score

def gradient_descent_adam(h0, max_iter=100000, lr=0.1, beta1=0.9, beta2=0.999, epsilon=1e-8):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iter+1):
        # compute gradient (subgradient of max correlation)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # project onto box and sum
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=200):
    import scipy.optimize as opt
    n = h0.shape[0]
    target = n/2.0
    # initial guess includes t variable
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, np.max(np.correlate(h0, 1.0 - h0, mode='full') * (2.0/n)) + 0.1)] # t bound
    # equality constraint sum(h) = target
    def eq_sum(x):
        return np.sum(x[:n]) - target
    # inequality constraint t >= corr for all shifts
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        # compute max corr
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    # Project again onto box + sum (to fix any small violation)
    h_opt = project_onto_box_sum(h_opt, target)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

import time
import numpy as np
import scipy.optimize as _opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx on the uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Subgradient (average) of max‑correlation w.r.t. h for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_adam(h0, max_iter=100000, lr=0.3,
                         beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam‑style projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1_pow = 1.0
    b2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for t in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps_shift = 1e-12
        idxes = np.where(corr >= max_corr - eps_shift)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1_pow *= beta1
        b2_pow *= beta2
        m_hat = m / (1.0 - b1_pow)
        v_hat = v / (1.0 - b2_pow)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=200):
    """SLSQP refinement of a feasible vector."""
    n = h0.shape[0]
    target = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - target

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = _opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on a balanced binary vector."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0

    return best_h, best_score


def binary_hill_climb(h_bin, time_budget, rng):
    """Greedy hill climbing on a balanced binary vector."""
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)

    ones = list(np.flatnonzero(best_h > 0.5))
    zeros = list(np.flatnonzero(best_h < 0.5))
    start = time.time()

    while time.time() - start < time_budget:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        best_h[i], best_h[j] = 0.0, 1.0
        new_score = compute_c5(best_h)

        if new_score < best_score - 1e-12:
            best_score = new_score
            # update index lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
        else:
            best_h[i], best_h[j] = 1.0, 0.0
    return best_h, best_score

    start = time.time()

        return budget_s - (time.time() - start)

    n = 512
    target_sum = n / 2.0

    # Initialise h
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:

            x_new = np.linspace(0, 1, n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()

        h[idx] = 1.0

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP refinements (if time permits)

        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=200)
        if sc_opt < best_score:

        if time_left() < 15.0:

    if time_left() > 5.0:
        max_iters = int(time_left() * 2000)  # generous estimate

            if sc_gd < best_score:

    if time_left() > 10.0:
        # Convert to a balanced binary vector
        idx = np.argpartition(best_h, -n // 2)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0

        h_bin_opt, sc_bin_opt = binary_hill_climb(h_bin, hc_time, rng)
        if sc_bin_opt < best_score:

            if ann_iters > 0:

                if sc_ann_opt < best_score:

    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
