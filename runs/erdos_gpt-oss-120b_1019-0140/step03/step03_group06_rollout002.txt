def subgradient_optimize(h0, target, dx, max_iters, deadline):
    h = h0.copy()
    best_h_local = h.copy()
    best_val_local = compute_c5(h)
    # schedule: initial step size 0.5 maybe
    a0 = 0.5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute Overlaps
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        overlap_val = corr[max_idx]
        k = max_idx - (len(h) - 1)
        # compute gradient
        n = len(h)
        g = np.ones_like(h)
        if k >= 0:
            g[:n-k] -= h[k:]
            g[k:] -= h[:n-k]
        else:
            kk = -k
            g[kk:] -= h[:n-kk]
            g[:n-kk] -= h[kk:]
        # step size
        alpha = a0 / (1 + it * 0.01)  # gradually decrease
        v = h - alpha * g
        # project
        h = project_box_sum(v, target)
        # evaluate
        val = compute_c5(h)
        if val < best_val_local:
            best_val_local = val
            best_h_local = h.copy()
            # maybe reset step size?
        # Early stop if val < 0.3808
        if best_val_local < 0.3808:
            break
    return best_h_local, best_val_local

def project_box_sum(v, target):
    # clip v to [0, 1] after adding lambda shift? We'll solve for lambda such that sum(clipping) = target.
    # Use bisection on lambda. Find low and high bounds.
    # Since sum(clipped) is decreasing with lambda (we subtract lambda before clipping).
    # For lambda very negative, v - lambda huge, all values >1, clipped to 1 => sum=n.
    # For lambda very positive, v - lambda <0, all values 0 => sum=0.
    # So we can start low=-1, high=1 maybe, but need to ensure covers extremes.
    low = -1.0
    high = 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam

            high = lam
    return w

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize

    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)  # will be projected later

    target = n / 2.0
    dx = 2.0 / n

    # enforce sum
    h0 = project_box_sum(h0, target)

    # Compute initial best
    best_h = h0.copy()
    best_val = compute_c5(best_h)  # returns C5

    # Pre-build constraints for SLSQP
    def sum_eq(x):
        return np.sum(x[:n]) - target
    def overlap_ineq(x, k):
        h = x[:n]
        t = x[-1]
        # compute overlapping integral for shift k
        if k >= 0:
            overlap = np.dot(h[:n-k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n-kk])
        return t - overlap * dx

    constraints = [{'type':'eq','fun':sum_eq}]
    for k in range(-n+1, n):
        if k == 0: continue
        constraints.append({'type':'ineq','fun':overlap_ineq,'args':(k,)})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # Subgradient main loop
    max_iters_per_run = 2000
    # We may set subgradient runs until time left > some minimal for SLSQP if needed
    while time.time() < deadline - 25.0:
        # random start
        start_vec = rng.random(n)
        start_vec = project_box_sum(start_vec, target)
        # run subgradient
        h_opt, val_opt = subgradient_optimize(start_vec, target, dx, max_iters_per_run, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            # early stop if good enough
            if best_val < 0.3808:
                break

    # Final SLSQP refinement if time permits
    if time.time() < deadline - 5.0:
        t0 = compute_c5(best_h)
        x0 = np.concatenate([best_h, [t0]])
        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 5000, 'ftol':1e-10, 'disp':False})
            if res.success:
                h_opt = np.clip(res.x[:n], 0.0, 1.0)
                h_opt = project_box_sum(h_opt, target)
                val = compute_c5(h_opt)
                if val < best_val:
                    best_h = h_opt
                    best_val = val
        except Exception:
            pass

    # Final enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -----
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum=v_target.
    Uses bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return w


def enforce_exact_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry (if possible) to make the sum exactly target."""
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    for i in range(v.size):
        if 0.0 <= v[i] + diff <= 1.0:
            v[i] += diff
            break
    return v


# ----- subgradient optimizer -----
def subgradient_optimize(h0: np.ndarray, target: float, max_iters: int, deadline: float):
    """Run a simple sub‑gradient descent on the worst overlap shift."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    n = h.size
    a0 = 0.5  # initial step size

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Determine the shift with maximal raw overlap (dx factor irrelevant for argmax)
        raw_corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(raw_corr))
        k = max_idx - (n - 1)

        # Sub‑gradient of the worst overlap w.r.t. h
        g = np.ones_like(h)
        if k >= 0:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] -= h[: n - kk]
            g[: n - kk] -= h[kk:]

        # Diminishing step size
        alpha = a0 / (1.0 + it * 0.1)
        v = h - alpha * g
        h = project_box_sum(v, target)

        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val < 0.3808:
                break

    return best_h, best_val

def sum_constraint_eq(x: np.ndarray, n: int, target: float) -> float:
    return np.sum(x[:n]) - target


def overlap_constraint_ineq(x: np.ndarray, k: int, n: int, dx: float) -> float:
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

    rng = np.random.default_rng(seed)
