def binary_anneal(h0, rng, time_left, max_iters=1000000, init_temp=0.1, decay=0.9999, min_temp=1e-8):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    T = init_temp
    n = h.size
    it = 0
    while it < max_iters and time_left() > 0.01:
        # propose swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5(h)
        delta = new_score - current_score
        # Since we want smaller score, accept if delta < 0; else with prob exp(-delta/T)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        T = max(min_temp, T * decay)
        it += 1
    return best_h, best_score

def compute_c5_auto(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.fft(h, N)
    auto = np.fft.ifft(H * np.conj(H)).real
    # due to round-off negative small values, clip
    # ignore zero shift
    min_overlap = np.min(auto[1:n])
    # In case it's negative due to errors, clip at 0.
    min_overlap = max(0.0, min_overlap)
    return 1.0 - (2.0 * min_overlap) / n

import time
import numpy as np

# ---------- Helper utilities ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    # ensure the interval [lam_low, lam_high] brackets the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5_auto(h):
    """C5 = 1 - 2*min_overlap/n for any h with sum(h)=n/2."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    auto = np.fft.irfft(H * np.conj(H), N)
    # take linear overlaps for shifts 0..n-1
    overlaps = auto[:n]
    # ignore the zero shift
    min_ov = np.min(overlaps[1:]) if n > 1 else 0.0
    if min_ov < 0:
        min_ov = 0.0
    return 1.0 - (2.0 * min_ov) / n


def active_shifts_and_min_overlap(h):
    """Return indices (positive shifts) achieving minimal overlap and the min value."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, N)
    auto = np.fft.irfft(H * np.conj(H), N)
    overlaps = auto[:n]
    min_ov = np.min(overlaps[1:])
    tol = 1e-12
    idx = np.where(overlaps[1:] <= min_ov + tol)[0] + 1
    return idx, min_ov


def gradient_descent_momentum_with_time(h0, time_left, step0=0.5,
                                        beta=0.9, max_iter=200000):
    """Projected gradient descent on the continuous problem."""
    h = h0.astype(float).copy()
    n = h.size
    target_sum = n / 2.0
    active_shifts, _ = active_shifts_and_min_overlap(h)
    best_h = h.copy()
    best_score = compute_c5_auto(h)
    momentum = np.zeros_like(h)

    for it in range(1, max_iter + 1):
        if time_left() <= 0.0:
            break

        # compute gradient as average of (1 - shifted - opposite shifted)
        grad = np.zeros_like(h)
        for k in active_shifts:
            # term1: h[i+k] with zero padding
            grad[: n - k] -= h[k:]
            # term2: h[i-k] with zero padding
            grad[k:] -= h[: n - k]
        grad += len(active_shifts)          # add the constant 1 for each active shift
        grad /= len(active_shifts)

        # momentum & step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)

        # evaluate
        new_score = compute_c5_auto(h_new)
        if new_score < best_score:
            best_score, best_h = new_score, h_new.copy()

        h = h_new
        active_shifts, _ = active_shifts_and_min_overlap(h)

        # early stop if already very good
        if best_score <= 0.3808:
            break

    return best_h, best_score


def binary_anneal(h0, rng, time_left,
                 max_iters=500000, init_temp=0.2,
                 decay=0.9999, min_temp=1e-8):
    """Simulated annealing on the binary space (swap a 1 with a 0)."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5_auto(h)
    best_h = h.copy()
    best_score = current_score
    T = init_temp
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_auto(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        T = max(min_temp, T * decay)
        it += 1
    return best_h, best_score


def local_improve(h0, rng, time_left, max_trials=5000):
    """Greedy hill climbing on binary vectors."""
    h = h0.copy()
    n = h.size
    best_score = compute_c5_auto(h)
    trials = 0
    while time_left() > 0.0 and trials < max_trials:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_auto(h)
        if new_score < best_score:
            best_score = new_score
            trials = 0
        else:
            h[i] = 1
            h[j] = 0
            trials += 1
    return h, best_score


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=int, count=n)

        h_cont = np.repeat(best_h, factor).astype(float)
    else:
        # generic interpolation + threshold
        x_old = np.linspace(0, 1, best_h.size, endpoint=False)
        x_new = np.linspace(0, 1, n_large, endpoint=False)
        cont = np.interp(x_new, x_old, best_h)
        idx = np.argpartition(-cont, n_large // 2)[:n_large // 2]
        h_cont = np.zeros(n_large, dtype=float)
        h_cont[idx] = 1.0

    update_best(h_cont, compute_c5_auto(h_cont))

    # ----- continuous gradient descent (momentum) -----
    if time_left() > 2.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(
            h_cont, time_left, step0=0.6, beta=0.85, max_iter=400000)
        update_best(h_ref, sc_ref)
        if best_score <= 0.3808:
            return best_h, best_score, best_h.size

    # additional random continuous restarts if time permits
    while time_left() > 5.0:
        v = rng.random(n_large)
        v = project_onto_box_sum(v, n_large / 2.0)
        h_ref, sc_ref = gradient_descent_momentum_with_time(
            v, time_left, step0=0.6, beta=0.85, max_iter=250000)
        update_best(h_ref, sc_ref)
        if best_score <= 0.3808:
            break

    # final safeguard projection & score recompute
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_auto(best_h)

    return best_h, best_score, best_h.size
