def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    rng_local = np.random.default_rng()
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # optional noise
        if it % 20000 == 0:
            grad += rng_local.normal(scale=0.01, size=n)  # small noise
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.5
    T_min = 1e-5
    decay = 0.99997
    it = 0
    while it < max_iters and time_left_func() > 0.05:
        # choose i in ones and j in zeros
        ones = np.where(h > 0.5)[0]  # using > 0.5 to represent ones (they are 1.0)
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear up‑sampling from len(arr) to new_n points (periodic domain)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    return upsample(arr, new_n)


def thue_morse(n):
    """Return Thue‑Morse binary sequence of length n as float array."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation w.r.t. h for given shift list."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    """Projected sub‑gradient descent with momentum, obeying the time budget."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    rng_local = np.random.default_rng()
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        # occasional jitter to escape plateaus
        if it % 15000 == 0:
            grad += rng_local.normal(scale=0.01, size=n)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score


def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000):
    """Swap‑based simulated annealing on a binary vector."""
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.5
    T_min = 1e-5
    decay = 0.99997
    it = 0
    while it < max_iters and time_left_func() > 0.05:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score


def slsqp_optimize_candidate(h0, max_iter=800):
    """SLSQP refinement of a candidate on the small grid."""
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

            init_small = downsample(init_full, n_small)
        else:
            init_small = init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=2000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            break

    if best_h is None:
        best_h = np.full(n_small, 0.5)
        best_h = project_onto_box_sum(best_h, n_small / 2.0)
        best_score = compute_c5(best_h)

    # ---------- Phase 2: upsample to 256 ----------
    n_mid = 256
    h_mid = upsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    sc_mid = compute_c5(h_mid)
    if sc_mid < best_score:
        best_score, best_h = sc_mid, h_mid.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 3: gradient descent on 256 ----------
    if time_left() > 5.0:
        max_iter_mid = min(8000, int(time_left() * 10))
        if max_iter_mid > 0:
            h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left,
                                                                step0=0.5, beta=0.9,
                                                                max_iter=max_iter_mid)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
                if best_score <= target:
                    return best_h, best_score, best_h.size

    # ---------- Phase 4: upsample to 512 ----------
    n_large = 512
    h_large = upsample(best_h, n_large)
    h_large = project_onto_box_sum(h_large, n_large / 2.0)
    sc_large = compute_c5(h_large)
    if sc_large < best_score:
        best_score, best_h = sc_large, h_large.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 5: gradient descent on 512 ----------
    if time_left() > 5.0:
        max_iter_large = min(8000, int(time_left() * 10))
        if max_iter_large > 0:
            h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left,
                                                                step0=0.5, beta=0.9,
                                                                max_iter=max_iter_large)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
                if best_score <= target:
                    return best_h, best_score, best_h.size

    # ---------- Phase 6: binary conversion ----------
    n = best_h.size
    if n % 2 != 0:   # ensure even length for exact half‑ones
        best_h = best_h[:-1]
        n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score, best_h = sc_bin, h_bin.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ---------- Phase 7: binary simulated annealing ----------
    if time_left() > 5.0:
        h_bin_opt, sc_bin_opt = binary_swap_anneal(h_bin, rng, time_left,
                                                   max_iters=200000)
        if sc_bin_opt < best_score:
            best_score, best_h = sc_bin_opt, h_bin_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Phase 8: final continuous refinement ----------
    if time_left() > 2.0:
        h_ref2, sc_ref2 = gradient_descent_momentum_with_time(best_h, time_left,
                                                              step0=0.5, beta=0.9,
                                                              max_iter=4000)
        if sc_ref2 < best_score:
            best_score, best_h = sc_ref2, h_ref2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ---------- Final guard ----------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
