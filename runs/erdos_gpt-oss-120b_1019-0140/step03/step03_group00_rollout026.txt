def compute_avg_gradient(h, shifts):
    n = h.shape[0]
    if len(shifts) == 0:
        return np.zeros_like(h)
    grad = np.ones(n) * len(shifts)
    for k in shifts:
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:]   -= h[:n - k]
        else:
            s = -k
            grad[s:]   -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad

def gd_momentum(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        # compute correlation and shifts
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        # shift indices where at max (within tolerance)
        # tolerance maybe 1e-12 relative to max
        idxes = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        shifts = idxes - (n - 1)
        # gradient average
        grad = compute_avg_gradient(h, shifts)
        v = beta * v + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = h - step * v
        # occasional jitter
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_avg_gradient(h, shifts):
    n = h.shape[0]
    if len(shifts) == 0:
        return np.zeros_like(h)
    grad = np.ones(n) * len(shifts)
    for k in shifts:
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            s = -k
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_fft(h):
    """Correlation of h with (1-h) using FFT (linear, not circular)."""
    n = h.shape[0]
    L = 2 * n - 1
    f_h = np.fft.rfft(h, n=L)
    rev = (1.0 - h)[::-1]
    f_rev = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(f_h * f_rev, n=L)
    return corr


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_avg_gradient(h, shifts):
    """Average subgradient of the max‑correlation term over given integer shifts."""
    n = h.shape[0]
    if not shifts:
        return np.zeros_like(h)
    grad = np.ones(n) * len(shifts)
    for k in shifts:
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            s = -k
            grad[s:] -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= len(shifts)
    return grad


def gd_fixed_iters(h0, max_iters, step0, beta, rng, time_left):
    """Projected momentum gradient descent with a fixed iteration budget."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        idxes = np.where(np.abs(corr - max_corr) <= 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_avg_gradient(h, shifts)
        v = beta * v + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = h - step * v
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Target resolution
    n = 512

    # Base vector from parent or Thue-Morse, upsampled to n
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = upsample(base, n)
    base = project_onto_box_sum(base, n / 2.0)

    # Primary long run on the base vector
    remaining = time_left()
    max_iters = int(min(300000, max(50000, remaining * 2000.0 / n)))
    best_h, best_score = gd_fixed_iters(base, max_iters, step0=0.6,
                                        beta=0.9, rng=rng, time_left=time_left)

    # Additional stochastic restarts while time permits
    while time_left() > 30.0:
        # Random continuous start
        h_start = rng.random(n)
        h_start = project_onto_box_sum(h_start, n / 2.0)

        remaining = time_left()
        max_iters = int(min(200000, max(30000, remaining * 1500.0 / n)))
        h_opt, sc_opt = gd_fixed_iters(h_start, max_iters, step0=0.6,
                                       beta=0.9, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

        # Random balanced binary start (optional)
        if time_left() > 10.0:
            h_bin = random_binary_balanced(n, rng)
            h_bin = project_onto_box_sum(h_bin, n / 2.0)
            remaining = time_left()
            max_iters = int(min(150000, max(20000, remaining * 1200.0 / n)))
            h_opt, sc_opt = gd_fixed_iters(h_bin, max_iters, step0=0.6,
                                           beta=0.9, rng=rng, time_left=time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

    # Final local refinements: small mass transfers and jitter
    while time_left() > 2.0:
        # Small mass transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.uniform(0.01, 0.3)
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= 0.3808:
                    break

        # Random jitter
        h_jit = best_h + rng.normal(scale=0.01, size=n)
        h_jit = project_onto_box_sum(h_jit, n / 2.0)
        sc_jit = compute_c5(h_jit)
        if sc_jit < best_score:
            best_score, best_h = sc_jit, h_jit.copy()
            if best_score <= 0.3808:
                break

    # Final safety projection and score recomputation
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
