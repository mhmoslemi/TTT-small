def gradient_descent_local(h, dx, target_sum, max_iter, deadline):
    cur_h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(cur_h)
    alpha = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
        # gradient for worst shift:
        k = max_idx - (len(cur_h) - 1)
        grad = gradient_for_shift(cur_h, k)
        # step
        new_h = cur_h - alpha * grad
        # projection:
        new_h = project_to_box_sum(new_h, target_sum)
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h = new_h
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-5)
    return best_h, best_val

def gradient_for_shift(h, k):
    n = len(h)
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]

        now = time.time()
        if now >= start_time + total_time:

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = lo - np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent_local(h, target_sum, dx, max_iter, deadline):
    """Projected gradient descent on max-overlap objective."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    alpha = 0.1
    while max_iter > 0 and time.time() < deadline:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(cur_h, k)
        new_h = cur_h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-5)
        max_iter -= 1
    return best_h, best_val
