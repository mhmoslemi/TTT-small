def gradient_descent_momentum_fft(h0, n_iter, step0, beta, rng, time_left):

   h = h0.copy()

   n = h.size

   best_h = h.copy()

   best_score = compute_c5_fft(h) # compute max correlation

   momentum = np.zeros_like(h)

   for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # indices where close to max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = h - step * momentum
        # jitter occasionally
        if (it % 5000) == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, n/2.0)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
   return best_h, best_score

def compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n -1).bit_length())
    # zero pad
    h_pad = np.zeros(L, dtype=np.float64)
    one_minus_h_pad = np.zeros(L, dtype=np.float64)
    h_pad[:n] = h
    one_minus_h_pad[:n] = 1.0 - h
    fft_h = np.fft.fft(h_pad)
    fft_one = np.fft.fft(one_minus_h_pad)
    conv = np.real(np.fft.ifft(fft_h * np.conj(fft_one)))
    # first 2*n - 1 entries
    corr = conv[:2*n - 1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
