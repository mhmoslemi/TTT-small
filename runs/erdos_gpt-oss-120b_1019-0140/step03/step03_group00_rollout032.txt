import cvxpy as cp

def milp_optimize_binary(n, time_left):
    m = n // 2
    h = cp.Variable(n, boolean=True)
    # y dictionary for unordered pairs
    y = {}
    for i in range(n):
        for j in range(i+1, n):
            y[(i,j)] = cp.Variable(boolean=True)
    constraints = []
    # sum h = m
    constraints.append(cp.sum(h) == m)
    # linking constraints
    for (i,j), y_ij in y.items():
        constraints.append(y_ij <= h[i])
        constraints.append(y_ij <= h[j])
        constraints.append(y_ij >= h[i] + h[j] - 1)
    # t variable
    t = cp.Variable()
    constraints.append(t >= 0)
    # for each shift k from -(n-1) to (n-1)
    for k in range(-(n-1), n):
        # compute sum_i h[i] * h[i+k]
        # sum over i where i+k in [0,n-1]
        if k >= 0:
            # i from 0 to n-1-k
            sum_y = cp.sum([y[(i, i+k)] for i in range(n-k)])
        else:
            # k negative: let kk = -k >0, i from -k to n-1 => i > j
            kk = -k
            sum_y = cp.sum([y[(i-kk, i)] for i in range(kk, n)])
        corr_k = m - sum_y  # since m = sum h[i] constant
        constraints.append(corr_k <= t)
    # objective: minimize t
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    # set solver options, maybe have a time limit
    # cp.CBC can include "maxTime" param? Actually CBC uses "maxTime" parameter.
    solver_opts = {"maxTime": max(0.0, time_left() - 1)} # ensure some slack
    prob.solve(solver=cp.CBC, **solver_opts)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback if not solved
        return None, None
    h_opt = np.round(h.value).astype(float)  # ensure 0/1
    # compute c5
    c5 = compute_c5_np_i(h_opt)
    return h_opt, c5

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n: return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # compute binary Thue-Morse sequence as float {0,1}
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def compute_gradients_for_shifts(h, shifts):

import time

import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-9)[0]
        shifts = idxes - (corr.size // 2)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if best_score <= 0.3808:
            break
    return best_h, best_score

def try_milp_binary(n, time_left):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    m = n // 2
    h = cp.Variable(n, boolean=True)
    Y = {}
    cons = [cp.sum(h) == m]
    for i in range(n):
        for j in range(i + 1, n):
            y = cp.Variable(boolean=True)
            Y[(i, j)] = y
            cons.append(y <= h[i])
            cons.append(y <= h[j])
            cons.append(y >= h[i] + h[j] - 1)
    t = cp.Variable()
    cons.append(t >= 0)
    for k in range(-(n - 1), n):
        if k >= 0:
            exprs = [Y[(i, i + k)] for i in range(n - k)]
        else:
            kk = -k
            exprs = [Y[(i - kk, i)] for i in range(kk, n)]
        sum_y = cp.sum(exprs) if exprs else 0
        corr_k = m - sum_y
        cons.append(corr_k <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    remaining = max(0.0, time_left() - 0.5)
    try:
        prob.solve(solver=cp.CBC, maxTime=remaining, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, maxTime=remaining, verbose=False)
        except Exception:
            return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_opt = np.rint(np.array(h.value)).astype(float)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    if base.size == 64 and time_left() > 40:
        h_bin, sc_bin = try_milp_binary(64, time_left)
        if h_bin is not None and sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin

    scales = [64, 128, 256, 512]
    max_iters_map = {64: 80000, 128: 50000, 256: 30000, 512: 20000}
    step_map = {64: 0.8, 128: 0.5, 256: 0.35, 512: 0.2}

    for n in scales:
        if time_left() <= 5:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        max_iters = max_iters_map.get(n, 20000)
        step0 = step_map.get(n, 0.3)
        h_opt, sc_opt = gradient_descent_momentum(best_h, max_iters,
                                                  step0=step0, beta=0.9,
                                                  rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    while time_left() > 0.5:
        n = best_h.size
        # Gaussian jitter
        h_jit = best_h + rng.normal(scale=0.008, size=n)
        h_jit = project_onto_box_sum(h_jit, n / 2.0)
        sc_jit = compute_c5(h_jit)
        if sc_jit < best_score:
            best_score, best_h = sc_jit, h_jit.copy()
        # small mass transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random()
            h_sw = best_h.copy()
            h_sw[i] -= delta
            h_sw[j] += delta
            h_sw = project_onto_box_sum(h_sw, n / 2.0)
            sc_sw = compute_c5(h_sw)
            if sc_sw < best_score:
                best_score, best_h = sc_sw, h_sw.copy()
        # random binary swap (if time permits)
        if time_left() > 2 and n <= 256:
            h_rand = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            sc_rand = compute_c5(h_rand)
            if sc_rand < best_score:
                best_score, best_h = sc_rand, h_rand.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
