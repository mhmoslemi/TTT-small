    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        # compute correlation and shift of current h
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = corr.argmax()
        max_corr = corr[max_idx]  # integer
        max_val = max_corr * (2.0 / n)  # scaled
        # update best
        if max_val < best_score:
            best_score = max_val
            best_h = h.copy()
        # if max_corr == 0, done
        if max_corr == 0:
            break
        k = max_idx - (n - 1)
        # mismatched sets
        if k >= 0:
            i_range = np.arange(n - k)
            i_shift = i_range + k
        else:
            i_range = np.arange(-k, n)
            i_shift = i_range + k
        # bool masks
        mismatch_one = (h[i_range] == 1) & (h[i_shift] == 0)
        mismatch_zero = (h[i_range] == 0) & (h[i_shift] == 1)
        if mismatch_one.any() and mismatch_zero.any():
            i_candidates = i_range[mismatch_one]
            j_candidates = i_range[mismatch_zero]
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            # swap
            h[i], h[j] = h[j], h[i]
            # update current score using compute_c5
            curr_score = compute_c5(h)
            if curr_score < best_score:
                best_score = curr_score
                best_h = h.copy()
            continue
        else:
            # random swap with SA acceptance
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                # cannot swap further
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            # temperature schedule
            T = 0.001 * (1 - it / max_iters)
            accept = False
            if new_score <= curr_score:

            else:

                    best_score = new_score
                    best_h = h.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure bounds enclose solution
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10 + 1.0
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10 + 1.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    # binary Thue-Morse sequence (0/1)
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):

def grad_softmax_corr(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10 + 1.0
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10 + 1.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s_idx, k in enumerate(range(-(n - 1), n)):
        w = weights[s_idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=0.5, alpha_end=8.0,
                        beta=0.9, time_left=None):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left is not None and time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_local_improve(h0, max_iters, time_left, rng):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    for it in range(max_iters):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = corr.argmax()
        max_corr = corr[max_idx]
        max_val = max_corr * (2.0 / n)
        if max_val < best_score:
            best_score = max_val
            best_h = h.copy()
        if max_corr == 0:
            break
        k = max_idx - (n - 1)
        if k >= 0:
            i_range = np.arange(n - k)
            i_shift = i_range + k
        else:
            i_range = np.arange(-k, n)
            i_shift = i_range + k
        mismatch_one = (h[i_range] == 1) & (h[i_shift] == 0)
        mismatch_zero = (h[i_range] == 0) & (h[i_shift] == 1)
        if mismatch_one.any() and mismatch_zero.any():
            i_candidates = i_range[mismatch_one]
            j_candidates = i_range[mismatch_zero]
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            h[i], h[j] = h[j], h[i]
            curr_score = compute_c5(h)
            if curr_score < best_score:
                best_score = curr_score
                best_h = h.copy()
            continue
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        T = 0.001 * (1 - it / max_iters)
        accept = False
        if new_score <= curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-(new_score - curr_score) / T)
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    rng = np.random.default_rng(seed)
    n = 256
    half_n = n // 2
    target = 0.3808

    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)

        else:

            x_new = np.linspace(0, 1, n, endpoint=False)
            init_up = np.interp(x_new, x_old, init_full)
        order = np.argsort(init_up)
        binary = np.zeros_like(init_up)
        binary[order[half_n:]] = 1.0
        candidates.append(binary)

    candidates.append(thue_morse(n).astype(float))
    for _ in range(5):

    best_h = None
    best_score = float('inf')

    for h0 in candidates:
        if time_left() < 3.0:

        sc0 = compute_c5(h0)
        if sc0 < best_score:

        max_iters = int(min(200000, max(5000, time_left() * 4000)))
        h_opt, sc_opt = binary_local_improve(h0, max_iters, time_left, rng)

        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    best_h = project_onto_box_sum(best_h, half_n)
    final_score = compute_c5(best_h)
