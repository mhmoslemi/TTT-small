def compute_c5(h):
    n = h.shape[0]
    M = n // 2
    # autocorr via FFT
    H = np.fft.fft(h)
    ac = np.real(np.fft.ifft(H * np.conj(H)))
    # minimal autocorr
    min_ac = ac.min()
    C5 = (M - min_ac) * (2.0 / n)
    return C5

def improve_one_step(h, rng):
    n = h.shape[0]
    M = n // 2
    # compute autocorr and find worst shift
    H = np.fft.fft(h)
    ac = np.real(np.fft.ifft(H * np.conj(H)))
    min_idx = np.argmin(ac)
    # compute derivative scores
    idxs = np.arange(n)
    plus = (idxs + min_idx) % n
    minus = (idxs - min_idx) % n
    s = h[plus] + h[minus]
    # candidates
    inc_cands = np.where(h < 1 - 1e-12)[0]
    dec_cands = np.where(h > 1e-12)[0]
    if inc_cands.size == 0 or dec_cands.size == 0:
        return h.copy()
    inc = inc_cands[np.argmax(s[inc_cands])]
    dec = dec_cands[np.argmin(s[dec_cands])]
    if inc == dec:
        # pick different indices randomly
        inc = inc_cands[rng.integers(len(inc_cands))]
        dec = dec_cands[rng.integers(len(dec_cands))]
        if inc == dec:
            return h.copy()
    # compute transferable amount
    max_inc = 1.0 - h[inc]
    max_dec = h[dec]
    max_transfer = min(max_inc, max_dec)
    if max_transfer <= 1e-12:
        return h.copy()
    # step size factor, maybe random between 0.1 and 0.5
    factor = rng.random() * 0.4 + 0.1
    delta = min(max_transfer, factor)
    h_new = h.copy()
    h_new[inc] += delta
    h_new[dec] -= delta
    # sum preserved
    return h_new

            def maker(k):
                def con(x, k=k, n=n):
                    h = x[:n]
                    if k > 0:
                        overl = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                    return x[-1] - overl * (2.0 / n)
                return con
            cons.append({'type': 'ineq', 'fun': maker(k)})
        res = minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-9, 'disp': False})
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            cur = h_opt.sum()
            if abs(cur - M) > 1e-9:
                h_opt += (M - cur) / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    except Exception:
        pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h, best_c5, best_n = None, np.inf, None
    candidates = []
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            candidates.append(n0)
    candidates += [80, 96, 112, 128, 144]
    candidates = [n for n in candidates if n % 2 == 0]
    for i, n in enumerate(candidates):
        if time.time() - start > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start)
        alloc = min(remaining * 0.6 if i == 0 else remaining * 0.4 / (len(candidates) - i), 350.0)
        init_h = None
        if best_h is not None and n == best_n:
            init_h = best_h
        elif 'initial_h_values' in globals() and len(initial_h_values) == n:
            init_h = np.array(initial_h_values, dtype=float)
        elif legendre_h(n) is not None:
            init_h = legendre_h(n)
        else:
            init_h = random_binary_h(n, n // 2, rng).astype(float)
        h_opt, c5_opt = anneal_opt(n, rng, alloc, init_h)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n
            if best_c5 < 0.3808:
                break
    if best_h is None:
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    M = best_n // 2
    if abs(cur - M) > 1e-12:
        best_h += (M - cur) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5, best_n
