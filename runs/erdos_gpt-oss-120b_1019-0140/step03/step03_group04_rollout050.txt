def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1: h[i+shift] for i< n-shift else 0
            term1 = np.concatenate([h[shift:], np.zeros(shift, dtype=h.dtype)])
            # term2: h[i-shift] for i >= shift else 0
            term2 = np.concatenate([np.zeros(shift, dtype=h.dtype), h[:n-shift]])
        else:
            s = -shift
            term1 = np.concatenate([np.zeros(s, dtype=h.dtype), h[:n-s]])
            term2 = np.concatenate([h[s:], np.zeros(s, dtype=h.dtype)])
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def resample(arr, new_n):
    """Linear interpolation to change length (up‑ or down‑sample)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    b_rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, L)
    B = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(A * B, L)
    max_corr = conv[:2 * n - 1][::-1].max()
    return max_corr * (2.0 / n)

def compute_corr_and_shifts_fft(h, eps=1e-12):
    """Full linear correlation of h with (1‑h) and shifts attaining the max."""
    n = h.size
    b_rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, L)
    B = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(A * B, L)
    corr = conv[:2 * n - 1][::-1]          # same ordering as np.correlate(..., 'full')
    max_corr = corr.max()
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    return corr, shifts, max_corr

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts (zero‑padded)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.concatenate([h[shift:], np.zeros(shift, dtype=h.dtype)])
            term2 = np.concatenate([np.zeros(shift, dtype=h.dtype), h[:n - shift]])
        else:
            s = -shift
            term1 = np.concatenate([np.zeros(s, dtype=h.dtype), h[:n - s]])
            term2 = np.concatenate([h[s:], np.zeros(s, dtype=h.dtype)])
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_proj(h0, step0=0.5, beta=0.9, max_iter=200000, time_end=None):
    """Projected gradient descent with momentum on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        _, shifts, _ = compute_corr_and_shifts_fft(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        if best_score <= 0.3808:
            break
    return best_h, best_score
