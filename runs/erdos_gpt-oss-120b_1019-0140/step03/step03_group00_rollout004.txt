import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10.0 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10.0 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # target resolution
    n_final = 512

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    # ---------- Differential Evolution ----------
    pop_size = 30
    F = 0.8
    CR = 0.9

    # initialise population (first individual = current best)
    pop = np.empty((pop_size, n_final), dtype=float)
    pop[0] = best_h
    for i in range(1, pop_size):
        cand = rng.random(n_final)
        pop[i] = project_onto_box_sum(cand, n_final / 2.0)

    scores = np.array([compute_c5(ind) for ind in pop])
    # keep best in population
    best_idx = np.argmin(scores)
    best_h, best_score = pop[best_idx].copy(), scores[best_idx]

    # evolve while enough time remains for refinement later
    while time_left() > 30.0:
        for idx in range(pop_size):
            # choose three distinct individuals different from idx
            choices = rng.choice([i for i in range(pop_size) if i != idx],
                                 size=3, replace=False)
            a, b, c = choices
            mutant = pop[a] + F * (pop[b] - pop[c])
            np.clip(mutant, 0.0, 1.0, out=mutant)
            mutant = project_onto_box_sum(mutant, n_final / 2.0)

            crossover_mask = rng.random(n_final) < CR
            if not np.any(crossover_mask):
                crossover_mask[rng.integers(n_final)] = True
            trial = np.where(crossover_mask, mutant, pop[idx])
            trial = project_onto_box_sum(trial, n_final / 2.0)

            trial_score = compute_c5(trial)
            if trial_score <= scores[idx]:
                pop[idx] = trial
                scores[idx] = trial_score
                if trial_score < best_score:
                    best_score, best_h = trial_score, trial.copy()
            # early exit if target reached
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break

    # ---------- Local gradient refinement ----------
    remaining = time_left()
    if remaining > 5.0:
        # allocate iterations proportionally to remaining time
        n_iter = int(min(60000, max(8000, remaining * 120)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=n_iter, step0=0.5, beta=0.9,
            rng=rng, time_left=time_left)

    # final projection and score recompute
    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_final
