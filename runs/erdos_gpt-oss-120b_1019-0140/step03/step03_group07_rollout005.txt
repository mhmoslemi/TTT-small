import time
import numpy as np
from scipy.optimize import minimize

# ----- helper functions -----
def compute_c5(h):
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# ----- constraint classes for SLSQP -----
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality constraint: t >= overlap(k) * dx for a given shift k."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ----- simulated annealing optimizer -----
def anneal_optimize(n, init_h, init_c5, time_limit, rng):
    """Run a simple SA on binary/fractional h for up to time_limit seconds."""
    h = init_h.copy()
    cur_c5 = init_c5
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # temperature schedule parameters
    T_start = 0.02
    T_end = 5e-4
    # main loop
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T_start * (1 - frac) + T_end * frac
        # decide move type
        if rng.random() < 0.2:
            # fractional transfer move
            i, j = rng.choice(n, 2, replace=False)
            # move mass from i to j
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            h[i] -= delta
            h[j] += delta
            new_c5 = compute_c5(h)
            delta_c5 = new_c5 - cur_c5
            accept = (delta_c5 < 0) or (rng.random() < np.exp(-delta_c5 / max(T, 1e-12)))
            if accept:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = cur_c5
                # early success
                if best_c5 < 0.3808:
                    break
            else:
                # revert
                h[i] += delta
                h[j] -= delta
        else:
            # binary swap move
            i = rng.integers(n)
            if h[i] > 0.5:
                zeros = np.where(h < 0.5)[0]
                if zeros.size == 0:
                    continue
                j = rng.choice(zeros)
            else:
                ones = np.where(h > 0.5)[0]
                if ones.size == 0:
                    continue
                j = rng.choice(ones)
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi  # swap
            new_c5 = compute_c5(h)
            delta_c5 = new_c5 - cur_c5
            accept = (delta_c5 < 0) or (rng.random() < np.exp(-delta_c5 / max(T, 1e-12)))
            if accept:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = cur_c5
                if best_c5 < 0.3808:
                    break
            else:
                # revert swap
                h[i], h[j] = hi, hj
    return best_h, best_c5

# ----- SLSQP refinement -----
def slsqp_refine(h, c5, n, rng, time_remaining):
    """Attempt a few SLSQP refinements of h using the max-overlap formulation."""
    M = n // 2
    sum_con = SumConstraint(n)
    overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    best_h, best_c5 = h, c5
    # allocate up to three attempts or until time runs out
    attempts = 3
    for _ in range(attempts):
        if time.time() + 0.5 > time.time() + time_remaining:
            break
        x0 = np.concatenate([best_h, [best_c5]])
        try:
            res = minimize(fun=lambda x: x[-1],
                           x0=x0,
                           method='SLSQP',
                           bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
        except Exception:
            break
        if not res.success:
            break
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        cur_sum = np.sum(h_opt)
        if abs(cur_sum - M) > 1e-9:
            h_opt += (M - cur_sum) / n
            h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < 0.3808:
                break
    return best_h, best_c5
