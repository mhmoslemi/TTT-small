import time
import numpy as np
import scipy.optimize as opt
from scipy.optimize import LinearConstraint

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = target_sum}. """
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < target_sum:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > target_sum:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """ C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points on [0,1)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Generate Thue‑Morse binary sequence of length n as float array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector of length n (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on C5."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def de_optimize(n, base, time_left, seed):
    """Run a short DE on dimension n, starting from base (used only for RNG)."""
    if time_left() < 5.0:
        return base.copy()
    # allocate iterations proportionally to remaining time but keep modest
    maxiter = int(min(80, max(5, time_left() * 0.3)))
    popsize = 20
    bounds = [(0.0, 1.0)] * n
    lincon = LinearConstraint(np.ones((1, n)), [n / 2.0], [n / 2.0])

    def cost_func(h, _n):
        return compute_c5(h)

    result = opt.differential_evolution(cost_func, bounds, args=(n,),
                                        constraints=(lincon,),
                                        maxiter=maxiter,
                                        popsize=popsize,
                                        seed=seed,
                                        polish=False,
                                        updating='deferred')
    cand = result.x
    # final safety projection (should already satisfy)
    cand = project_onto_box_sum(cand, n / 2.0)
    return cand


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256]
    best_h = None
    best_score = np.inf

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Global DE on the coarsest scale only
        if n == scales[0]:
            cand = de_optimize(n, base, time_left, seed)
            cand_score = compute_c5(cand)
            if cand_score < best_score:
                best_score, best_h = cand_score, cand.copy()
            base = cand
        else:
            # Use upsampled best as base for finer scales
            if best_h is not None:
                base = upsample(best_h, n)
                base = project_onto_box_sum(base, n / 2.0)

        # Local momentum gradient descent refinement
        if time_left() > 5.0:
            # iteration budget scaling similar to previous approach
            factor = 4000.0 / n
            est_iters = int(min(80000, max(8000, time_left() * factor)))
            refined_h, refined_score = gradient_descent_momentum_custom(
                base, n_iter=est_iters, step0=0.5, beta=0.9,
                rng=rng, time_left=time_left)
            if refined_score < best_score:
                best_score, best_h = refined_score, refined_h.copy()
            base = refined_h

    # Final intensive refinement on the best vector (largest scale)
    if best_h is not None and time_left() > 5.0:
        n_final = best_h.size
        factor = 3000.0 / n_final
        final_iters = int(min(80000, max(20000, time_left() * factor)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=final_iters, step0=0.2, beta=0.9,
            rng=rng, time_left=time_left)

    # Random binary search and small swaps as a last push
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)

    n_best = best_h.size
    while time_left() > 5.0:
        # Random balanced binary candidate
        h_rand = random_binary_balanced(n_best, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        # Small mass transfer between two positions
        i, j = rng.choice(n_best, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.5)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    # ensure feasibility before returning
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
