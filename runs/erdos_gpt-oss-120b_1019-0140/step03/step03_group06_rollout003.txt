def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust vector v (in [0,1]) so its sum equals target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def objective_fun(x):
    return x[-1]

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append(
            {"type": "ineq", "fun": overlap_constraint_ineq, "args": (k, n, dx)}
        )
    return cons

    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")

        k = max_idx - (n - 1)

        if k >= 0:

        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        i_local = np.argmax(prod)
        i = i_range[i_local]

        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:

                if best_val < 0.3808:

            else:

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    n = 100                      # discretization points
    target = n / 2.0
    dx = 2.0 / n

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.shape[0] != n:
            x_old = np.linspace(0, 2, h0.shape[0], endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
    else:
        h0 = random_continuous_vector(n, target, rng)

    h0 = enforce_sum(h0, target, rng)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = build_constraints(n, target, dx)

    # ----- SLSQP phase -----
    while time.time() < deadline - 5.0:
        # Choose a start vector
        if rng.random() < 0.5:
            start_h = random_binary_vector(n, target, rng)
        else:
            start_h = best_h.copy()
            # small perturbations
            for _ in range(5):
                src = np.where(start_h > 1e-12)[0]
                dst = np.where(start_h < 1.0 - 1e-12)[0]
                if len(src) == 0 or len(dst) == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(start_h[i], 1.0 - start_h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                start_h[i] -= delta
                start_h[j] += delta
            start_h = enforce_sum(start_h, target, rng)

        t0 = compute_c5(start_h)
        x0 = np.concatenate([start_h, [t0]])

        try:
            res = minimize(
                fun=objective_fun,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 1500, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            break

        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = enforce_sum(h_opt, target, rng)
        val = compute_c5(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break

    # ----- Annealing phase -----
    if time.time() < deadline - 0.5:
        best_h, best_val = continuous_anneal(
            best_h, best_val, best_h, best_val, start, deadline, rng
        )

    # Final enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
