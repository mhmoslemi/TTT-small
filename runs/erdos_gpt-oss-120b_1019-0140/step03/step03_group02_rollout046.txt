def adam_optimize(h0, n_iter, lr, beta1, beta2, eps, alpha_start, alpha_end, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (t-1)/(max(1,n_iter-1))
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score
```

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    # shift k runs from -(n-1) to n-1, index s = k + (n-1)

        if w == 0.0:
            continue
        if k >= 0:
            end = n - k
            grad[:end] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:end]
        else:
            p = -k
            end = n - p
            grad[p:] += w * (1.0 - h[:end])
            grad[:end] -= w * h[p:]
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            end = n - k
            grad[:end] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:end]
        else:
            p = -k
            end = n - p
            grad[p:] += w * (1.0 - h[:end])
            grad[:end] -= w * h[p:]
    return grad

def adam_optimize(h0, n_iter, lr, beta1, beta2, eps,
                  alpha_start, alpha_end, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h - step, target)
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
    return best_h, best_score

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1) of length n."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
