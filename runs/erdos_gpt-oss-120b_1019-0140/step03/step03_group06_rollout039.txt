def propose_swap(h, k=None):
    n=len(h)
    if k is None:
        # random two indices: i where h[i]=1, j where h[j]=0
        ones_idx = np.where(h==1)[0]
        zeros_idx = np.where(h==0)[0]
        if len(ones_idx)==0 or len(zeros_idx)==0:
            return None
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        return i, j
    else:
        # try targeted swap for shift k
        if k>=0:
            # compute indices where h[i]==1 and h[i+k]==0 (i range valid)
            i_candidates = np.where((h[:n-k]==1) & (h[k:]==0))[0]
            # indices where h[j]==0 and h[j+k]==1
            j_candidates = np.where((h[:n-k]==0) & (h[k:]==1))[0]
        else:
            kk = -k
            i_candidates = np.where((h[kk:]==1) & (h[:n-kk]==0))[0] + kk  # original i index
            j_candidates = np.where((h[kk:]==0) & (h[:n-kk]==1))[0] + kk
        if len(i_candidates)>0 and len(j_candidates)>0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            return i,j
        else:
            # fallback
            return propose_swap(h, None)

def fix_binary_sum(v, target, rng):
    cur = v.sum()
    while cur < target:
        zeros = np.where(v == 0)[0]
        if len(zeros)==0: break
        idx = rng.choice(zeros)
        v[idx] = 1
        cur += 1
    while cur > target:
        ones = np.where(v == 1)[0]
        if len(ones)==0: break
        idx = rng.choice(ones)
        v[idx] = 0
        cur -= 1
    return v

    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    val = corr[idx] * dx
    shift = idx - (n - 1)
    return float(val), int(shift)

    val, _ = compute_c5_and_shift(h)

        return i, j
    return None

def hillclimb_binary(start_h, start_time, deadline, rng,
                     T0=0.015, max_iter=20000):
    h = start_h.copy()
    cur_val, cur_shift = compute_c5_and_shift(h)
    best_h = h.copy()
    best_val = cur_val
    it = 0
    while True:
        now = time.time()
        if now >= deadline - 0.2 or it >= max_iter:
            break
        frac = (deadline - now) / (deadline - start_time)
        T = max(1e-12, T0 * frac)
        pair = propose_swap(h, cur_shift, rng)
        if pair is None:
            pair = propose_swap(h, None, rng)
            if pair is None:
                break
        i, j = pair
        h[i], h[j] = h[j], h[i]
        new_val, new_shift = compute_c5_and_shift(h)
        delta = new_val - cur_val
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            cur_shift = new_shift
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            h[i], h[j] = h[j], h[i]
        it += 1
    return best_h, best_val

def sum_constraint_eq(x, n, target):
    return np.sum(x[:n]) - target

def overlap_constraint_ineq(x, k, n, dx):
    h = x[:n]
    t = x[-1]
    if k >= 0:
        overl = np.dot(h[: n - k], 1.0 - h[k:])
    else:
        kk = -k
        overl = np.dot(h[kk:], 1.0 - h[: n - kk])
    return t - overl * dx

def build_constraints(n, target, dx):
    cons = [{"type": "eq", "fun": sum_constraint_eq, "args": (n, target)}]
    for k in range(-n + 1, n):
        if k == 0:
            continue
        cons.append({"type": "ineq", "fun": overlap_constraint_ineq,
                     "args": (k, n, dx)})
    return cons

def objective_fun(x):
    return x[-1]

def final_refine_continuous(h_float, start_time, deadline, rng):
    n = h_float.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    t0 = compute_c5(h_float)
    x0 = np.concatenate([h_float, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = build_constraints(n, target, dx)
    remaining = deadline - time.time()
    max_iter = int(min(2000, max(10, remaining * 30)))
    options = {"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    try:
        res = minimize(fun=objective_fun,
                       x0=x0,
                       method="SLSQP",
                       bounds=bounds,
                       constraints=cons,
                       options=options)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = enforce_sum(h_opt, target, rng)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h_float, compute_c5(h_float)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    else:
        n = 80
        init_h = None

    target_sum = n / 2.0
    binary_target = int(round(target_sum))

    if init_h is not None:
        best_h_float = init_h.copy()
        best_val = compute_c5(best_h_float)
        best_h_binary = (init_h > 0.5).astype(int)
        best_h_binary = fix_binary_sum(best_h_binary, binary_target, rng)
    else:
        best_h_binary = random_binary_vector(n, binary_target, rng)
        best_h_float = best_h_binary.astype(float)
        best_val = compute_c5(best_h_float)

    restart = 0
    while time.time() < deadline - 1.0 and restart < 30:
        if rng.random() < 0.5:
            start_h = random_binary_vector(n, binary_target, rng)
        else:
            start_h = best_h_binary.copy()
            for _ in range(3):
                ones = np.where(start_h == 1)[0]
                zeros = np.where(start_h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                start_h[i], start_h[j] = 0, 1
        start_h = fix_binary_sum(start_h, binary_target, rng)

        local_h, local_val = hillclimb_binary(start_h, start, deadline, rng)
        if local_val < best_val:
            best_val = local_val
            best_h_binary = local_h.copy()
            best_h_float = best_h_binary.astype(float)

        if best_val < 0.3808:
            break
        restart += 1

    if time.time() < deadline - 0.5:
        best_h_float, best_val = final_refine_continuous(best_h_float, start, deadline, rng)

    best_h_float = np.clip(best_h_float, 0.0, 1.0)
    best_h_float = enforce_sum(best_h_float, target_sum, rng)
    final_val = compute_c5(best_h_float)

    return best_h_float.astype(float), final_val, n
