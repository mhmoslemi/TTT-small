def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return v
    # If diff > 0, need to increase sum: add to indices with v < 1.0
    # If diff < 0, need to decrease sum: subtract from indices with v > 0.0
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v

def optimize_h(n, target, h0, rng, deadline):
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    h = h0.copy()
    cur_val = best_val
    total_time = deadline - time.time()
    init_temp = 0.01
    while time.time() < deadline - 0.02:
        # identify worst shift
        n_local = n
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr_full)
        k = max_idx - (n_local - 1)
        if k >= 0:
            prod = h[:n_local - k] * (1.0 - h[k:])
            i_range = np.arange(n_local - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[:n_local - kk])
            i_range = np.arange(kk, n_local)
        i_local = np.argmax(prod)
        i = i_range[i_local]

        # choose j
        j = None
        if k == 0:
            # choose j with minimal product among those not at capacity
            cand = np.where(h < 1.0 - 1e-12)[0]
            if len(cand) > 0:
                j = cand[np.argmin(h[cand] * (1.0 - h[cand]))]
        else:
            # try directed partner i+k or i-k
            j_candidate = i + k
            if 0 <= j_candidate < n_local and h[j_candidate] < 1.0 - 1e-12:
                j = j_candidate
            else:
                j_candidate2 = i - k
                if 0 <= j_candidate2 < n_local and h[j_candidate2] < 1.0 - 1e-12:
                    j = j_candidate2
                else:
                    # fallback to lowest h
                    cand = np.where(h < 1.0 - 1e-12)[0]
                    if len(cand) > 0:
                        j = cand[np.argmin(h[cand])]
        if j is None or j == i:
            # fallback: random src and dst
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            if i == j:
                continue

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            # fallback: small random transfer
            max_transfer = min(h[i], 1.0 - h[j])
        delta = rng.random() * max_transfer
        if delta <= 0:
            continue
        # apply move
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # temperature based acceptance
            remaining = deadline - time.time()
            temp = init_temp * max(remaining / total_time, 0.0)
            if temp > 0 and np.exp((cur_val - new_val) / temp) > rng.random():
                cur_val = new_val
            else:
                # revert move
                h[i] += delta
                h[j] -= delta
        # optional occasional random perturbation
        # we skip for simplicity
    return best_h, best_val

import time
import numpy as np

# ----- basic helpers -------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust so that sum(v) == target exactly."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v


def random_continuous_vector(n, target, rng):
    """Uniform random vector in [0,1]^n with exact sum == target."""
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v


def compute_grad_overlap(h, k):
    """Gradient of overlap(k) = sum_i h_i (1-h_{i+k}) w.r.t. h."""
    n = len(h)
    grad = np.full(n, 1.0)
    if k >= 0:
        if n - k > 0:
            grad[: n - k] -= h[k:]          # subtract h_{i+k}
        if k < n:
            grad[k:] -= h[: n - k]          # subtract h_{i-k}
    else:
        kk = -k
        if n - kk > 0:
            grad[kk:] -= h[: n - kk]        # subtract h_{i+k}=h_{i-kk}
        if kk < n:
            grad[: n - kk] -= h[kk:]        # subtract h_{i-k}=h_{i+kk}
    return grad


# ----- optimizer -----------------------------------------------------
def optimize_h_gradient(n, target, h0, rng, deadline):
    """Projected gradient ascent on min correlation (i.e. descent on C5)."""
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    h = h0.copy()
    cur_val = best_val
    total_time = max(deadline - time.time(), 1e-8)
    init_temp = 0.01

    while time.time() < deadline - 0.02:
        # worst shift (max overlap)
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr_full)
        k = max_idx - (n - 1)

        # gradient of the worst overlap
        grad = compute_grad_overlap(h, k)
        d = -grad                         # direction that reduces overlap
        d -= d.mean()                     # zero‑sum for the equality constraint

        # feasible step size respecting bounds
        pos = d > 1e-12
        neg = d < -1e-12
        if np.any(pos):
            alpha_pos = np.min((1.0 - h[pos]) / d[pos])
        else:
            alpha_pos = np.inf
        if np.any(neg):
            alpha_neg = np.min((-h[neg]) / d[neg])
        else:
            alpha_neg = np.inf
        alpha_max = min(alpha_pos, alpha_neg)

        if not np.isfinite(alpha_max) or alpha_max <= 0:
            # fallback: simple random transfer
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            max_tr = min(h[i], 1.0 - h[j])
            if max_tr <= 0:
                continue
            delta = rng.random() * max_tr
            h[i] -= delta
            h[j] += delta
            new_val = compute_c5(h)
        else:
            # take a random fraction of the maximal step
            fraction = 0.2 + 0.6 * rng.random()   # between 0.2 and 0.8
            alpha = fraction * alpha_max
            h_new = h + alpha * d
            h_new = np.clip(h_new, 0.0, 1.0)

            # tiny correction to keep the sum exact (helps numerical stability)
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                free = np.where((h_new > 1e-12) & (h_new < 1.0 - 1e-12))[0]
                if len(free) > 0:
                    adj = diff / len(free)
                    h_new[free] += adj
                    h_new = np.clip(h_new, 0.0, 1.0)

            new_val = compute_c5(h_new)

        # acceptance
        if new_val < cur_val - 1e-12:
            h = h_new if 'h_new' in locals() else h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break
        else:
            # simulated‑annealing style acceptance for non‑improving moves
            remaining = deadline - time.time()
            temp = init_temp * max(remaining / total_time, 0.0)
            if temp > 0 and np.exp((cur_val - new_val) / temp) > rng.random():
                h = h_new if 'h_new' in locals() else h
                cur_val = new_val
            else:
                # revert (if we used the random‑fallback move, it is already reverted)
                if 'h_new' in locals():
                    pass  # h unchanged
        # cleanup locals for next iteration
        if 'h_new' in locals():
            del h_new

    return best_h, best_val


# ----- entry point ---------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # use n = 80 (preferred small size)
    n = 80
    target = n / 2.0

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # enforce exact sum just in case
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = random_continuous_vector(n, target, rng)

    # main optimization
    best_h, best_val = optimize_h_gradient(n, target, h0, rng, deadline)

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
