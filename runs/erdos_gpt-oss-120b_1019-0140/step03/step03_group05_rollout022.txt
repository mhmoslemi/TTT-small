def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=40):
    low = -1e6
    high = 1e6
    # initially find bounds that ensure sum below/above target
    # Let's use bisection on shift lambda:
    for _ in range(max_iter):
        mid = (low + high)/2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low+high)/2.0, lo, hi)

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad
```

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation and find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # gradient for worst shift
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # step
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            # maybe increase step
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                # random perturb to escape
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0, max_iter=30):
    """Project v onto [lo,hi]^n while fixing sum(v)=target."""
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (ignoring dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad


def gradient_descent(h_start, target_sum, max_iter, time_limit, rng):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5


# ---------- Constraint classes for SLSQP ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx


def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=200):
    """Run a single SLSQP optimisation from (h_start, t_start)."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=lambda x: x[-1],
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res


# ---------- Evolutionary optimisation for a given n ----------
def optimize_for_n(n, rng, budget_slice, init_h=None):
    start = time.time()
    target_sum = n / 2.0
    pop_size = 8
    population = []
    if init_h is not None and init_h.shape[0] == n:
        population.append(project_to_box_sum(np.clip(init_h, 0.0, 1.0), target_sum))
    while len(population) < pop_size:
        h = rng.random(n)
        h = project_to_box_sum(h, target_sum)
        population.append(h)
    pop = [(h, compute_c5(h)) for h in population]
    pop.sort(key=lambda x: x[1])

    child_counter = 0
    mutation_sigma = 0.07

    while time.time() - start < budget_slice - 2.0:
        p1 = pop[rng.integers(len(pop))][0]
        p2 = pop[rng.integers(len(pop))][0]
        alpha = rng.random()
        child = alpha * p1 + (1.0 - alpha) * p2
        child = child + rng.normal(scale=mutation_sigma, size=n)
        child = np.clip(child, 0.0, 1.0)
        child = project_to_box_sum(child, target_sum)
        child_score = compute_c5(child)
        child_counter += 1
        if child_score < pop[-1][1]:
            pop[-1] = (child, child_score)
            pop.sort(key=lambda x: x[1])

        if child_counter % 40 == 0:
            best_h = pop[0][0]
            gd_iter = 2000
            gd_deadline = start + budget_slice - 2.0
            refined_h, refined_score = gradient_descent(best_h, target_sum,
                                                        gd_iter, gd_deadline, rng)
            if refined_score < pop[0][1]:
                pop[0] = (refined_h, refined_score)
                pop.sort(key=lambda x: x[1])

    # final local refinement
    remaining = start + budget_slice - time.time()
    if remaining > 7.0:
        best_h = pop[0][0]
        refined_h, refined_score = gradient_descent(best_h, target_sum,
                                                    4000, time.time() + remaining - 1.0, rng)
        if refined_score < pop[0][1]:
            pop[0] = (refined_h, refined_score)

    # optional SLSQP refinement if time permits
    remaining = start + budget_slice - time.time()
    if remaining > 12.0:
        best_h, best_score = pop[0]
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        t0 = compute_c5(best_h)
        res = slsqp_optimize(best_h, t0, sum_con, overlap_cons, maxiter=150)
        if res.success:
            h_sl = np.clip(res.x[:n], 0.0, 1.0)
            h_sl = project_to_box_sum(h_sl, target_sum)
            score_sl = compute_c5(h_sl)
            if score_sl < best_score:

        pop[0] = (best_h, best_score)

    return pop[0][0], pop[0][1]


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate discretisations (all ≤ 200 for speed)
    n_list = [80, 96, 112, 128, 160]

    for idx, n in enumerate(n_list):
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 6.0:
            break
        # Allocate remaining time equally among pending n's
        slice_time = remaining / (len(n_list) - idx)

        init_h = None
        if 'initial_h_values' in globals():
            ih = globals()['initial_h_values']
            try:
                arr = np.asarray(ih, dtype=float)
                if arr.size == n:
                    init_h = arr
            except Exception:
                pass

        h_opt, c5_opt = optimize_for_n(n, rng, slice_time, init_h)

        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt, n
            if best_c5 < 0.3808:
                break

    if best_h is None:
        # Fallback simple construction
        n = 80
        best_n = n
        h = np.zeros(n)
        h[:n // 2] = 1.0
        best_h = project_to_box_sum(h, n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final safety projection and recompute exact C5
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(best_n)
