def greedy_improve_binary(h, time_end, max_iters=200):
    n = h.size
    a = np.correlate(h, h, mode='full')
    min_a = a.min()
    best_h = h.copy()
    best_min_a = min_a
    it = 0
    while time.time() < time_end and it < max_iters:
        it += 1
        idxs = np.where(a == min_a)[0]
        # choose random shift among those
        k_idx = rng.choice(idxs)
        shift = k_idx - (n - 1)
        # compute candidates
        if shift >= 0:
            # i where h[i]==0 and h[i+shift]==1 (i+shift < n)
            mask_i = (h == 0) & (np.arange(n - shift) >= 0) & (h[shift:] == 1)
            # This yields array for i indices with length n-shift.
            cand_i = np.where(mask_i)[0]
            # For j: h[j]==1 and h[j+shift]==0
            mask_j = (h == 1) & (h[shift:] == 0)
            cand_j = np.where(mask_j)[0]
        else:
            s = -shift
            mask_i = (h == 0) & (np.arange(s, n) >= 0) & (h[:n-s] == 1)  # i-s >=0 => i>=s; compare h[i-s] which is h[:n-s] for i>=s
            cand_i = np.where(mask_i)[0]
            mask_j = (h == 1) & (h[:n-s] == 0)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binarize_from_real(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    binar = np.zeros(n, dtype=float)
    binar[idx] = 1.0
    return binar

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def greedy_improve_binary(h, rng, time_end, max_iters=200):
    n = h.shape[0]
    a = np.correlate(h, h, mode='full')
    min_a = a.min()
    it = 0
    while time.time() < time_end and it < max_iters:
        it += 1
        idxs = np.where(a == min_a)[0]
        k_idx = rng.choice(idxs)
        shift = k_idx - (n - 1)
        if shift >= 0:
            length = n - shift
            cand_i = np.where((h[:length] == 0) & (h[shift:] == 1))[0]
            cand_j = np.where((h[:length] == 1) & (h[shift:] == 0))[0]
        else:
            s = -shift
            length = n - s
            cand_i = np.where((h[s:] == 0) & (h[:length] == 1))[0] + s
            cand_j = np.where((h[s:] == 1) & (h[:length] == 0))[0] + s
        if cand_i.size == 0 or cand_j.size == 0:
            break
        i_sel = rng.choice(cand_i)
        j_sel = rng.choice(cand_j)
        h_new = h.copy()
        h_new[i_sel] = 1.0
        h_new[j_sel] = 0.0
        a_new = np.correlate(h_new, h_new, mode='full')
        min_a_new = a_new.min()
        if min_a_new > min_a:
            h = h_new
            a = a_new
            min_a = min_a_new
    return h, min_a

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    n_small = 256
    n_large = 512

    # Build seed pool for small scale
    seed_vecs = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        seed_vecs.append(project_onto_box_sum(resample(init_full, n_small), n_small / 2.0))
    seed_vecs.append(thue_morse_binary(n_small))
    # legendre sequence works for n_small+1 prime (257)
    seed_vecs.append(legendre_sequence(n_small + 1))
    for _ in range(3):
        seed_vecs.append(random_binary_balanced(n_small, rng))

    # Process each seed
    for base in seed_vecs:
        if time_left() < 5.0:
            break
        # ensure binary representation
        if not np.all((base == 0) | (base == 1)):
            h0 = binarize_from_real(base)
        else:
            h0 = base.copy()
        # Greedy binary improvement
        greedy_end = start + budget_s - 4.0
        h_greedy, _ = greedy_improve_binary(h0, rng, greedy_end, max_iters=300)
        score = compute_c5(h_greedy)
        if score < best_score:
            best_score, best_h = score, h_greedy.copy()
        # Gradient descent refinement on upsampled version if time permits
        if time_left() > 5.0:
            h_up = resample(h_greedy, n_large)
            h_up = project_onto_box_sum(h_up, n_large / 2.0)
            gd_end = start + budget_s - 1.0
            h_ref, sc_ref = gradient_descent_momentum(h_up, step0=0.5, beta=0.9,
                                                      time_end=gd_end, max_iter=200000)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    # If best found is still small, upsample and refine
    if best_h is not None and best_h.size < n_large and time_left() > 5.0:
        h_up = resample(best_h, n_large)
        h_up = project_onto_box_sum(h_up, n_large / 2.0)
        gd_end = start + budget_s - 1.0
        h_ref, sc_ref = gradient_descent_momentum(h_up, step0=0.5, beta=0.9,
                                                  time_end=gd_end, max_iter=200000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Final safety projection
    if best_h is None:
        best_h = random_binary_balanced(n_small, rng)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
