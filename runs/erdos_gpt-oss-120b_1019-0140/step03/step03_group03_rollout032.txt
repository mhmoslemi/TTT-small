def compute_gradient_max_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    eps = 1e-12
    shifts = np.where(corr >= max_corr - eps)[0] - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n-shift] += (1.0 - h[shift:])
            grad[shift:] += -h[:n-shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += -h[s:]
    if shifts.size > 0:
        grad /= shifts.size

def adam_optimize(h0, max_iters, lr, beta1, beta2, eps, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        if time_left() <= 0:
            break
        grad = compute_gradient_max_corr(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)

def anneal_binary(h0, max_iter, temp_start, temp_end, rng, time_left):
    h = h0.copy()
    score = compute_c5(h)
    best_h, best_score = h.copy(), score
    n = len(h)
    # precompute list of indices
    # maintain sets of ones and zeros
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    for it in range(max_iter):
        if time_left() <= 0:
            break
        # temperature schedule
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        # choose a random swap
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update indices
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

    n = h.shape[0]
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()

    for it in range(1, max_iters+1):

        max_corr = corr.max()

        score = np.max(corr)*(2.0/n)

            best_score = score
            best_h = h.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return the discretized C5 for array h on [0,2]."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)


def upsample(arr, new_n):
    """Linear up‑sample using interpolation (no endpoint duplication)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear down‑sample using interpolation."""
    return upsample(arr, new_n)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def compute_grad_from_shifts(h, shifts):
    """Sub‑gradient of max‑correlation term given shift list."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:] += -h[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += -h[s:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def gradient_descent_adam(h0, max_iters, lr, beta1, beta2, eps,
                         budget_s, start_time):
    """Projected Adam on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    # initial correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_score = np.max(corr) * (2.0 / n)

    for it in range(1, max_iters + 1):
        if time.time() - start_time >= budget_s:
            break

        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)

        grad = compute_grad_from_shifts(h, shifts)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, n / 2.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        score = np.max(corr) * (2.0 / n)

        if score < best_score:
            best_score = score
            best_h = h.copy()

    return best_h, best_score


def slsqp_objective(x):
    return x[-1]


def slsqp_optimize_candidate(h0, max_iter=400):
    """SLSQP refinement for a given candidate."""
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(slsqp_objective, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12,
                                'disp': False})

    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


def anneal_binary(h0, max_iter, temp_start, temp_end, rng,
                  budget_s, start_time):
    """Simple Metropolis annealing on a balanced binary vector."""
    h = h0.copy()
    score = compute_c5(h)
    best_h, best_score = h.copy(), score
    n = len(h)

    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)

    for it in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        # exponential cooldown
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True

        if accept:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
        else:
            h[i], h[j] = 1.0, 0.0

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # ------------------------------------------------------------------
    # Phase 1: Small‑scale search (n = 64)
    # ------------------------------------------------------------------
    n_small = 64
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        small = init_full if init_full.size == n_small else downsample(init_full, n_small)
        small = project_onto_box_sum(small, n_small / 2.0)
        candidates.append(small)

    candidates.append(thue_morse(n_small))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h = None
    best_score = np.inf

    # SLSQP refinements on the small grid
    for cand in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=400)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Additional random SLSQP restarts while time permits
    while time_left() > 6.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=300)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # Adam optimization on the small grid
    if best_h is not None and time_left() > 5.0:
        max_iters_small = int(min(20000, time_left() * 1200))
        h_gd, sc_gd = gradient_descent_adam(
            best_h, max_iters_small, lr=0.5,
            beta1=0.9, beta2=0.999, eps=1e-8,
            budget_s=budget_s, start_time=start_time)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()

    # ------------------------------------------------------------------
    # Phase 2: Upsample to larger resolution (n = 512)
    # ------------------------------------------------------------------
    n_large = 512
    if best_h is None:
        best_h_large = random_binary_balanced(n_large, rng)
    else:
        best_h_large = upsample(best_h, n_large)
    best_h_large = project_onto_box_sum(best_h_large, n_large / 2.0)
    score_large = compute_c5(best_h_large)
    if score_large < best_score:
        best_score, best_h = score_large, best_h_large.copy()

    # ------------------------------------------------------------------
    # Phase 3: Large‑scale Adam optimization
    # ------------------------------------------------------------------
    if time_left() > 6.0:
        remaining = time_left()
        # Rough 1000–1500 iterations per second for n=512
        max_iters_large = int(min(250000, remaining * 1300))
        if max_iters_large > 0:
            h_gd, sc_gd = gradient_descent_adam(
                best_h, max_iters_large, lr=0.5,
                beta1=0.9, beta2=0.999, eps=1e-8,
                budget_s=budget_s, start_time=start_time)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

    # ------------------------------------------------------------------
    # Phase 4: Binary simulated annealing
    # ------------------------------------------------------------------
    if time_left() > 5.0:
        # Build a balanced binary start from the continuous best
        idx = np.argsort(best_h)[-n_large // 2:]
        h_bin = np.zeros(n_large, dtype=float)
        h_bin[idx] = 1.0
        remaining = time_left()
        max_iter_ann = int(remaining * 350)  # ~350 proposals per second
        if max_iter_ann > 0:
            h_ann, sc_ann = anneal_binary(
                h_bin, max_iter_ann,
                temp_start=0.1, temp_end=1e-5,
                rng=rng, budget_s=budget_s, start_time=start_time)
            if sc_ann < best_score:
                best_score, best_h = sc_ann, h_ann.copy()

    # Final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
