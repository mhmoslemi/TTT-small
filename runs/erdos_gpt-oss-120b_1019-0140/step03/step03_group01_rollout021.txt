def smooth_optimize(h0, time_left, n_iters=200000):
    h = h0.copy()
    n = h.shape[0]
    s = n/2
    best_h, best_score = h.copy(), compute_c5(h)
    # Adaptive step size:
    step = 1.0
    momentum = np.zeros_like(h)
    beta_mom = 0.9
    # Beta for smooth max schedule:
    beta = 0.5
    beta_inc = 1.05 # increase each iteration maybe
    # loop until time_left <= small or max_iters reached
    it=0
    while time_left() > 0 and it < n_iters:
        # compute correlation and smooth max
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_val = np.max(corr)
        # compute smooth max
        # Use log-sum-exp
        scaled = beta * (corr - max_corr_val)
        exp_scaled = np.exp(scaled)
        sum_exp = np.sum(exp_scaled)
        smooth_val = max_corr_val + np.log(sum_exp) / beta
        # compute weights
        weights = exp_scaled / sum_exp  # shape (2n-1)
        # gradient
        grad = np.zeros_like(h)
        # contributions:
        # For shift >=0:
        for shift in range(0, n):
            w = weights[shift + n - 1]
            if w == 0: continue
            # part1: (1 - h[i+shift]) for i in [0,n-shift)
            grad[:n-shift] += w * (1 - h[shift:])
            # part2: -h[i] for i+shift index i+shift in shift..n-1:
            grad[shift:] += -w * h[:n-shift]
        # For shift negative:
        for shift in range(-n+1, 0):
            w = weights[shift + n - 1]
            if w == 0: continue
            s = -shift
            # part1: -h[i+s] for i in [0,n-s)
            grad[:n-s] += -w * h[s:]
            # part2: (1 - h[i]) for i in [s,n)
            grad[s:] += w * (1 - h[:n-s])
        # momentum update
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        # step size update: maybe use decreasing step: step0 / sqrt(it+1)
        eta = step / np.sqrt(it + 1)
        h_new = h - eta * momentum
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate actual C5
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        # maybe accept new if score improved else keep; but we always set h = h_new (gradient descent).
        h = h_new
        # increase beta gradually
        beta = min(beta * beta_inc, 50.0)  # up to large
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=400000):
    h = h0.copy()
    n = h.size

    best_h = h.copy()

    T_min = 1e-6

        j = rng.choice(zeros)

        h[j] = 1.0

            h[j] = 0.0

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000):
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    T, T_min, decay = 0.5, 1e-6, 0.9999
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        delta = new - cur
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best

def smooth_obj_and_grad(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    scaled = beta * (corr - max_corr)
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    grad1 = np.correlate(w, 1.0 - h, mode='full')
    grad2 = np.correlate(w[::-1], h, mode='full')
    start = n - 1
    grad = grad1[start:start + n] - grad2[start:start + n]
    smooth_val = max_corr + np.log(sum_exp) / beta
    return smooth_val, grad
