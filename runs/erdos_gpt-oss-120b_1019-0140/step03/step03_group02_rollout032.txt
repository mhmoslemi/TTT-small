def smooth_softmax_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start)*(it-1)/(n_iter-1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1-beta) * grad
        eta = step0 / np.sqrt(it)
        h = h - eta * mom
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, max_swaps, target, time_left):
    h = h0.copy()
    n = h.shape[0]
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    T0 = 0.01
    for i in range(1, max_swaps+1):
        if time_left() < 0.01:
            break
        # choose random indices i0(0) and i1(1)
        zero_idx = rng.integers(n)
        while h[zero_idx] != 0.0:
            zero_idx = rng.integers(n)
        one_idx = rng.integers(n)
        while h[one_idx] != 1.0:
            one_idx = rng.integers(n)
        h_new = h.copy()
        h_new[zero_idx] = 1.0
        h_new[one_idx] = 0.0
        newc = compute_c5(h_new)
        delta = newc - curr
        T = T0 * (1 - i / max_swaps)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = h_new
            curr = newc
            if newc < best:
                best = newc
                best_h = h_new.copy()
                if best <= target:
                    break
    return best_h, best

def run(...):

    n_large = 256

    # generate initial candidates
    candidates = []

    init_small = get_initial_h(n_small)

        candidates.append(init_small)

    # thue-morse binary
    candidates.append(thue_morse(n_small).astype(float))
    # Thue-morse complement
    candidates.append(1.0 - thue_morse(n_small).astype(float))

    # random balanced binary vectors
    for _ in range(4):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))

    # random continuous projected vectors
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small/2))

    # Phase 1: small-scale smooth descent
    best_h = None
    best_score = np.inf
    # allocate time for Phase 1 maybe 10% of budget
    # iterates over candidates until time insufficient
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = smooth_softmax_descent(
            h0, n_iter=8000, step0=1.0, alpha_start=1.0, alpha_end=50.0,
            beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # fallback
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small/2)
        best_score = compute_c5(best_h)

    # Phase 2: upsample and gradient descent on large grid
    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large/2)
    up_score = compute_c5(h_up)
    if up_score < best_score:
        best_score, best_h = up_score, h_up.copy()
    # momentum descent
    if time_left() > 2.0:
        iters = int(min(50000, max(10000, time_left() * 800)))
        h_ref, sc_ref = momentum_grad_descent(best_h, n_iter=iters,
                                              step0=0.5, beta=0.9,
                                              time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Phase 3: binary hill climbing with multiple restarts
    # round continuous best to binary
    if time_left() > 4.0:
        n_big = best_h.shape[0]
        # produce binary by top half entries
        order = np.argsort(best_h)
        binary = np.zeros(n_big, dtype=float)
        binary[order[n_big//2:]] = 1.0
        # multiple SA runs
        num_runs = 3
        for run_idx in range(num_runs):
            if time_left() < 2.0:
                break
            # estimate max swaps based on time left
            max_swaps = int(min(400000, max(50000, time_left() * 6000)))
            bin_h, bin_sc = binary_hill_climb_sa(binary, rng, max_swaps,
                                                 target, time_left)
            if bin_sc < best_score:
                best_score, best_h = bin_sc, bin_h.copy()
                # update binary for next runs as new best (binary)
                binary = bin_h.copy()
        # After binary optimization, optionally do continuous refinement again
        if time_left() > 2.0:
            h_ref2, sc_ref2 = momentum_grad_descent(best_h, n_iter=8000,
                                                   step0=0.3, beta=0.9,
                                                   time_left=time_left)
            if sc_ref2 < best_score:

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size/2)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral for a discretized h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Thue–Morse binary sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max-correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                grad[:n - k] += (1.0 - h[k:])
                grad[k:]      -= h[:n - k]
            else:
                sft = -k
                grad[sft:]    += (1.0 - h[:n - sft])
                grad[:n - sft]-= h[sft:]
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def smooth_softmax_descent(h0, n_iter=2000, step0=0.5,
                           alpha_start=1.0, alpha_end=10.0,
                           beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = h - eta * mom
        h = project_onto_box_sum(h, s)
        score = compute_c5(h)
        if score < best_score:
            best_score, best_h = score, h.copy()
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, max_swaps, target, time_left):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    T0 = 0.01
    for it in range(1, max_swaps + 1):
        if time_left() < 0.01:
            break
        # select a random zero and a random one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1.0 - it / max_swaps)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

        if sc_opt < best_score:

    if time_left() > 4.0:
        n_cur = best_h.shape[0]
        order = np.argsort(best_h)

    # final projection & score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
