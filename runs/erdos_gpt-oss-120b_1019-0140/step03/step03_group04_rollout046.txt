Define compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n-s] += (1 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            sneg = -s
            grad[sneg:] += (1 - h[:n-sneg])  # wait careful
            # careful: shift negative: s = -sneg, meaning we consider correlation for shift = -sneg.
            # correlation term for negative shift: sum_{i=0}^{n-1-(-sneg)} h[i] * (1 - h[i - sneg]) where i - sneg >= 0.
            # Equivalent to sum_{j=sneg}^{n-1} h[j] * (1 - h[j - sneg]).
            # So term for gradient: for each j >= sneg, gradient at j (h_j) gets contribution (1 - h_{j - sneg}). For each j >= sneg, gradient at j - sneg gets contribution -h_j.
            # That's: grad[sneg:] += (1 - h[:n-sneg]) and grad[:n-sneg] -= h[sneg:].
            grad[sneg:] += (1 - h[:n-sneg])

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        sm = proj.sum()
        if abs(sm - s) < 1e-12:
            break
        if sm > s:
            lam_low = lam
        else:
            lam_high = lam
    return proj

def full_correlation_fft(h):
    """Full correlation c[k]=sum_i h[i]*(1-h[i+k]) for k=-(n-1)..(n-1) via FFT."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    rh = h[::-1]
    g = 1.0 - h
    fh = np.fft.rfft(rh, n=m)
    fg = np.fft.rfft(g, n=m)
    conv = np.fft.irfft(fh * fg, n=m)
    return conv[:2 * n - 1]

def compute_c5_fft(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    return np.max(full_correlation_fft(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            sneg = -s
            grad[sneg:] += (1.0 - h[:n - sneg])
            grad[:n - sneg] -= h[sneg:]
    if shifts:
        grad /= len(shifts)
    return grad

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def binarize_from_real(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def gradient_descent_momentum(h0, step0=0.5, beta=0.9,
                              time_end=None, max_iter=200000):
    """Projected GD with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 0
    eps = 1e-12
    while it < max_iter and (time_end is None or time.time() < time_end):
        it += 1
        corr = full_correlation_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad

        h_new = project_onto_box_sum(h_new, target)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
