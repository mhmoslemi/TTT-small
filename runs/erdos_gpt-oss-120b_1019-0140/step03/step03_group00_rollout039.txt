import cvxpy as cp

def solve_mip_binary(n, time_limit):
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n,n), boolean=True)
    t = cp.Variable()
    constraints = []

    # sum constraint
    constraints.append(cp.sum(h) == n//2)

    # w constraints
    for i in range(n):
        constraints.append(w[i,i] == h[i])
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
            # since symmetric, also enforce w[j,i] equals w[i,j]?
            # we can just ignore w lower triangle and not use it later.
            # But define constraints for j,i maybe:
            constraints.append(w[j,i] == w[i,j])
    # shift constraints
    for s in range(-(n-1), n):
        expr = 0
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                expr = expr + h[i] - w[i,j]
        constraints.append(expr <= t)
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit, verbose=False)
    except Exception as e:
        # fallback maybe
        prob.solve(solver=cp.CBC, verbose=False)

    if h.value is None:
        # fallback: return None
        return None, None

    h_sol = np.round(h.value).astype(float)  # as float for compute_c5
    t_val = t.value
    return h_sol, t_val

def hill_climb_binary(h, max_iters, time_left):
    best = h.copy()
    best_score = compute_c5(best)
    n = len(h)
    # precompute indices of 1s and 0s
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # pick random i from ones, j from zeros
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        candidate = best.copy()
        candidate[i] = 0
        candidate[j] = 1
        score = compute_c5(candidate)
        if score < best_score:
            best = candidate
            best_score = score
            # update ones/zeros maybe next iteration will recalc
    return best, best_score

def solve_mip_binary(n, time_limit, rng):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n//2]
    # product constraints for i<=j to reduce constraints
    for i in range(n):
        constraints.append(w[i,i] == h[i])
        for j in range(i+1, n):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    for s in range(-(n-1), n):
        expr = 0
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                if i <= j:
                    expr += h[i] - w[i,j]
                else:
                    expr += h[i] - w[j,i]  # using upper triangle
        constraints.append(expr <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, verbose=False, timeLimit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            return None, None
    if h.value is None:
        return None, None
    h_sol = np.rint(h.value).astype(float)  # rounding
    t_val = t.value
    return h_sol, t_val

def compute_gradients_for_shifts(h, shifts):

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    top_k = 5
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        if corr.size <= top_k:
            idxes = np.arange(corr.size)
        else:
            idxes = np.argpartition(-corr, top_k - 1)[:top_k]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def solve_mip_binary(n, time_limit, rng):
    """Return binary vector minimizing max‑correlation using a MILP."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n // 2]
    for i in range(n):
        constraints.append(w[i, i] == h[i])
        for j in range(i + 1, n):
            constraints.append(w[i, j] <= h[i])
            constraints.append(w[i, j] <= h[j])
            constraints.append(w[i, j] >= h[i] + h[j] - 1)
    for s in range(-(n - 1), n):
        expr = 0
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                if i <= j:
                    expr += h[i] - w[i, j]
                else:
                    expr += h[i] - w[j, i]
        constraints.append(expr <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, verbose=False, timeLimit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            return None, None
    if h.value is None:
        return None, None
    h_sol = np.rint(h.value).astype(float)
    t_val = t.value
    return h_sol, t_val

    start = time.time()

    # initial base vector

        if time_left() > 5:

    while time_left() > 10:

        pert = project_onto_box_sum(pert, best_h.size / 2.0)
        n_iter = int(min(50000, max(5000, time_left() * 10)))
        cand_h, cand_score = gradient_descent_momentum_custom(

        if cand_score < best_score:

        # small mass transfer between two positions
        i, j = rng.choice(best_h.size, size=2, replace=False)
        if best_h[i] > 0 and best_h[j] < 1:

            if delta > 0:

                new_h[i] -= delta

                new_h = project_onto_box_sum(new_h, best_h.size / 2.0)

                if sc < best_score:

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
