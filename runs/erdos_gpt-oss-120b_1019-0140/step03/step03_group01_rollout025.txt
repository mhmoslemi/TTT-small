def beatty_sequence(N, alpha):
    seq = ((np.arange(N) * alpha) % 1) < 0.5
    arr = seq.astype(float)
    diff = int(arr.sum() - N/2)
    if diff > 0:
        # too many ones, set diff largest values to 0? But all ones have equal; arbitrarily turn diff ones to 0 with smallest fractional part (i.e., largest mod value).
        # Actually we can compute fractional part values:
        fracs = (np.arange(N) * alpha) % 1
        # ones positions have fracs < 0.5. To reduce ones, turn those with fracs closest to 0.5 down.
        idx = np.where(seq)[0]
        # sort idx by fracs descending (closest to 0.5)
        order = idx[np.argsort(fracs[idx])[::-1]]
        arr[order[:diff]] = 0.0
    elif diff < 0:
        # too few ones, add diff zeros with smallest fractional part (closest to 0)
        idx = np.where(~seq)[0]
        fracs = (np.arange(N) * alpha) % 1
        order = idx[np.argsort(fracs[idx])[: -diff]]
        arr[order] = 1.0
    return arr

def compute_c5(h):
    n = h.shape[0]
    # autocorrelation via FFT
    H = np.fft.rfft(h, n=n)
    autocorr = np.fft.irfft(H * np.conj(H), n=n)
    # autocorr is sum_i h[i] * h[(i+k)%n] for k=0..n-1
    # ignoring shift 0
    min_autocorr = np.min(autocorr[1:])
    return 1.0 - (2.0 / n) * min_autocorr

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.shape[0]
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    # initial temperature maybe high relative to typical delta scale; set T = 0.01 maybe.
    T = 0.01
    T_min = 1e-6
    decay = 0.9999  # slower decay
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        # randomly pick a 1 and a 0
        ones = np.where(h > 0.5)[0]  # since h binary
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    # clamp sum alpha
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    # Use algorithm for projecting onto simplex with box constraints: bisection on lambda.
    # lambda in real numbers, compute sum_{i} clip(v_i - lambda, lo, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # bisection
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if np.abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def beatty_sequence(n, alpha):
    # generate fractional part
    fracs = (np.arange(n) * alpha) % 1.0
    seq = fracs < 0.5
    arr = seq.astype(float)
    diff = int(arr.sum() - n/2)
    if diff > 0:
        # too many ones, need to turn diff ones to zero
        idx = np.where(seq)[0]
        # sort idx by fracs descending (closest to 0.5)
        order = idx[np.argsort(fracs[idx])][::-1]  # descending
        arr[order[:diff]] = 0.0
    elif diff < 0:
        # need more ones
        idx = np.where(~seq)[0]
        order = idx[np.argsort(fracs[idx])]  # ascending fracs (closest to 0)
        arr[order[: -diff]] = 1.0
    return arr

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b[::-1], n=L)
    conv = np.fft.irfft(A * B, n=L)[:2 * n - 1]
    return np.max(conv) * (2.0 / n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    x = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    x[idx] = 1.0
    return x

def beatty_sequence(n, alpha):
    fracs = (np.arange(n, dtype=np.float64) * alpha) % 1.0
    seq = fracs < 0.5
    arr = seq.astype(float)
    diff = int(arr.sum() - n / 2)
    if diff > 0:
        idx = np.where(seq)[0]
        order = idx[np.argsort(fracs[idx])][::-1]
        arr[order[:diff]] = 0.0
    elif diff < 0:
        idx = np.where(~seq)[0]
        order = idx[np.argsort(fracs[idx])]
        arr[order[: -diff]] = 1.0
    return arr

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    L = 1 << ((2 * n - 1).bit_length())
    while it <= max_iter and time_left_func() > 0.0:
        a = h
        b = 1.0 - h
        A = np.fft.rfft(a, n=L)
        B = np.fft.rfft(b[::-1], n=L)
        conv = np.fft.irfft(A * B, n=L)[:2 * n - 1]
        max_corr = np.max(conv)
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.shape[0]
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    T = 0.005
    T_min = 1e-7
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5(h)
        delta = new - cur
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    N = 512
    best_h = None
    best_score = np.inf
    candidates = []

    if 'initial_h_values' in globals():

    candidates.append(thue_morse(N))
    alphas = [(np.sqrt(5) - 1) / 2, np.sqrt(2) - 1, (np.sqrt(3) - 1) / 2]
    for a in alphas:
        candidates.append(beatty_sequence(N, a))
    for _ in range(3):
        candidates.append(random_binary_balanced(N, rng))
    for _ in range(2):
        v = rng.random(N)
        candidates.append(project_onto_box_sum(v, N / 2.0))

    for cand in candidates:
        if time_left() < 5.0:
            break
        if not (np.all(cand == 0) or np.all(cand == 1)):
            idx = np.argpartition(-cand, N // 2)[: N // 2]
            bin_cand = np.zeros_like(cand)
            bin_cand[idx] = 1.0
        else:
            bin_cand = cand.astype(float)
        h_opt, sc_opt = binary_swap_anneal(bin_cand, rng, time_left, max_iters=200000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= target:
            best_h = project_onto_box_sum(best_h, N / 2.0)

    if best_h is None:
        best_h = np.full(N, 0.5)

    if time_left() > 1.0:
        h_ref, sc_ref = gradient_descent_momentum_with_time(best_h, time_left, max_iter=300000)
        if sc_ref < best_score:

        v = rng.random(N)

        h_tmp, sc_tmp = binary_swap_anneal(v, rng, time_left, max_iters=150000)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, N / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, N
