def compute_c5_fast(h):
    # using FFT correlation
    n = h.size
    L = 2**int(np.ceil(np.log2(2*n-1)))
    fh = np.fft.rfft(h, L)
    f1mh = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(fh * np.conj(f1mh), L)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def random_balanced_binary(n, rng):

def anneal_binary(h0, rng, time_end):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # maintain indices list
    ones = np.flatnonzero(cur_h).tolist()
    zeros = np.flatnonzero(1 - cur_h).tolist()
    # temperature schedule: start T = cur_score * 0.02 maybe, end T = cur_score * 0.0002.
    # Use linear schedule.
    start_T = cur_score * 0.02
    end_T = cur_score * 0.0002
    start_time = time.time()
    while True:
        now = time.time()
        if now >= time_end:
            break
        # compute fraction of elapsed relative to time_end - start_time
        frac = (now - start_time) / (time_end - start_time) if time_end > start_time else 1.0
        T = start_T + (end_T - start_T) * frac  # linear decreasing
        # choose random swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # apply swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            # avoid overflow when delta/T large; use exp(-delta/T)
            # Ensure T > 0
            if T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            cur_score = new_score
            # update ones/zeros lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score

def compute_c5_fast(h):
    n = h.size
    L = 1 << (2 * n - 1).bit_length() # power of two >= 2n-1
    # rfft requires shape L for real transform, returns L//2+1 length.
    fh = np.fft.rfft(h, n=L)
    f1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(f1mh), n=L)[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def gradient_descent_momentum(h0, rng, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.astype(float).copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        # compute correlation and shifts with max value
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts achieving near max (tolerance)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        # compute gradient as average of gradients of max shifts
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                # term1: h[i+shift] for i <= n-shift-1
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s_shift = -shift
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[s_shift:] = h[:n-s_shift]
                term2[:n-s_shift] = h[s_shift:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        # update momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto box sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # If s is lower than feasible range
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # find lambda by bisection: we need to find lam such that sum_i clip(v_i - lam, lo, hi) = s
    # The function f(lam) = sum_i clip(v_i - lam, lo, hi) is decreasing in lam.
    # Determine bounds for lam.
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # ensure f(lam_low) >= s >= f(lam_high)
    # Actually for low lam, clipping yields hi for many entries, sum possibly too high.
    # For safety, expand bounds a bit.
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if np.abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def anneal_binary(h0, rng, time_end):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # indices of ones and zeros.
    ones = np.flatnonzero(cur_h).tolist()
    zeros = np.flatnonzero(1 - cur_h).tolist()
    # temperature schedule.
    start_T = cur_score * 0.02 if cur_score > 0 else 1e-3
    end_T = cur_score * 0.0002 if cur_score > 0 else 1e-5
    t0 = time.time()
    while True:
        now = time.time()
        if now >= time_end:
            break
        # schedule fraction
        if time_end > t0:
            frac = (now - t0) / (time_end - t0)
        else:
            frac = 1.0
        T = start_T + (end_T - start_T) * frac
        # pick indices
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # apply swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    L = 1 << (2 * n - 1).bit_length()          # next power of two ≥ 2n‑1
    fh = np.fft.rfft(h, n=L)
    f1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fh * np.conj(f1mh), n=L)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_fft(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.size
    L = 1 << (2 * n - 1).bit_length()
    fh = np.fft.rfft(h, n=L)
    f1mh = np.fft.rfft(1.0 - h, n=L)
    return np.fft.irfft(fh * np.conj(f1mh), n=L)[:2 * n - 1]

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def legendre_sequence(p):
    """Binary sequence of length p‑1 (p prime) using quadratic residues."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def binarize_by_half(arr):
    """Set exactly half of entries to 1 (largest values)."""
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def resample_float(arr, new_n):
    """Linear interpolation to change length (real‑valued)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal_binary(h0, rng, time_end):
    """Simulated‑annealing hill‑climbing on binary vectors."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h).tolist()
    zeros = np.flatnonzero(1 - cur_h).tolist()
    start_T = max(cur_score * 0.02, 1e-4)
    end_T = max(cur_score * 0.0002, 1e-6)
    t0 = time.time()
    while True:
        now = time.time()
        if now >= time_end:
            break
        frac = (now - t0) / (time_end - t0) if time_end > t0 else 1.0
        T = start_T + (end_T - start_T) * frac
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5_fast(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score

def gradient_descent_momentum(h0, rng, step0=0.5, beta=0.9,
                              time_end=None, max_iter=200000):
    """Projected gradient descent (sub‑gradient) on the max‑correlation objective."""
    h = h0.astype(float).copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fast(h)
    momentum = np.zeros_like(h)
    it = 0
    t0 = time.time()
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = compute_corr_fft(h)
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[: n - shift] = h[shift:]
                term2[shift:] = h[: n - shift]
            else:
                sft = -shift
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                term1[sft:] = h[: n - sft]
                term2[: n - sft] = h[sft:]
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fast(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
