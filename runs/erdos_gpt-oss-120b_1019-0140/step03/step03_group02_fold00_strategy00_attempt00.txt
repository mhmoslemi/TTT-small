def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    if s <= lo*n:
        return np.full_like(v, lo)
    elif s >= hi*n:
        return np.full_like(v, hi)
    
    try:
        # Use scipy to solve quadratic projection
        _, _, x0 = np.linalg.svd(v)
        def f(x):
            return np.sum((x - v)**2)
        res = minimize(f, 0.5*np.ones_like(v), 
                      method='L-BFGS-B',
                      bounds=( (lo, hi) for _ in range(n) ),
                      constraints=[LinearConstraint([1]*n, [-s], [-s])],
                      options=dict(maxiter=max_iter, disp=False))
        return np.clip(res.x, lo, hi)
    except Exception as e:
        return np.clip(v + np.random.normal(0, 0.1, n), lo, hi)
