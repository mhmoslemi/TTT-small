def solve_binary_milp(n):

    t = cp.Variable()
    # y matrix variables: y_{i,j} binary?
    y = cp.Variable((n, n), boolean=True)

def solve_binary_milp(n, time_limit):
    import cvxpy as cp, numpy as np
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    y = cp.Variable((n,n), boolean=True)
    constraints = []
    # product constraints
    h_row = cp.reshape(h, (n,1))
    h_col = cp.reshape(h, (1,n))
    constraints.append(y <= h_row)
    constraints.append(y <= h_col)
    constraints.append(y >= h_row + h_col - 1)
    constraints.append(cp.sum(h) == n/2)
    # shift constraints
    for k in range(n):
        # sum_i y[i, (i+k)%n]
        idx_rows = np.arange(n)
        idx_cols = (np.arange(n) + k) % n
        # Use cp.sum(cp.hstack([y[i, col] for i,col in zip(idx_rows, idx_cols)]))
        # But better: build expression using cp.sum(cp.multiply(y, mask)), where mask is a binary matrix of same shape with 1 at those entries.
        # We'll build mask via numpy.
        mask = np.zeros((n,n))
        mask[idx_rows, idx_cols] = 1
        # Convert mask to cp.Constant
        constraints.append(t >= n/2 - cp.sum(cp.multiply(y, mask)))
    constraints.append(t >= 0)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    # set solver options with time limit if available
    try:
        prob.solve(solver=cp.CBC, timeLimit=time_limit)  # may support time limit param
    except Exception:
        # try GLPK_MI
        prob.solve(solver=cp.GLPK_MI)
    # extract
    if h.value is None:
        return None, None
    h_opt = np.round(h.value).astype(float)
    t_opt = t.value
    return h_opt, t_opt

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=50000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0:
        # compute correlation to get shifts where max achieved
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # compute grad as average of gradient for max shifts:
        # For each shift where corr == max_corr, gradient of corr w.r.t. h? The function C5 = max corr * dx; gradient is piecewise:
        # derivative of corr for particular shift index s0: d/dh[j] = (1 - h[(j + shift) % n]) - h[(j - shift) % n] (something)
        # Actually corr for shift k: sum_i h[i] * (1 - h[i+k]), derivative w.r.t. h[i] = (1 - h[i+k]) - h[i-k] (if we treat forward shift).
        # But easier: approximate gradient by finite differencing? That would be expensive.
        # Could reuse compute_gradients_for_shifts from previous code.
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s2 = -shift
                term1 = np.zeros(n, dtype=h.dtype)
                term2 = np.zeros(n, dtype=h.dtype)
                term1[s2:] = h[:n-s2]
                term2[:n-s2] = h[s2:]
            grad += (1.0 - term1 - term2)
        if len(shifts) > 0:
            grad /= len(shifts)
        # momentum
        momentum = beta * momentum + (1-beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # project onto box sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
```

import time
import numpy as np

# Optional import of cvxpy for MILP binary optimization
try:
    import cvxpy as cp
except Exception:
    cp = None

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term1[n - shift :] = 0.0
            term2[shift:] = h[: n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term1[:s] = 0.0
            term2[: n - s] = h[s:]
            term2[n - s :] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def solve_binary_milp(n, max_seconds):
    """Solve binary min‑max overlap MILP for given n. Returns (h_opt, t_opt) or (None, None)."""
    if cp is None:
        return None, None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    y = cp.Variable((n, n), boolean=True)
    cons = []
    # product linearization constraints
    for i in range(n):
        for j in range(n):
            cons.append(y[i, j] <= h[i])
            cons.append(y[i, j] <= h[j])
            cons.append(y[i, j] >= h[i] + h[j] - 1)
    # sum of h = n/2
    cons.append(cp.sum(h) == n / 2)
    # max‑overlap constraints
    for k in range(n):
        sum_y = cp.sum(cp.vstack([y[i, (i + k) % n] for i in range(n)]))
        cons.append(t >= n / 2 - sum_y)
    cons.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), cons)
    # Try CBC with a time limit; fall back to GLPK_MI without limit
    try:
        prob.solve(solver=cp.CBC, maxSeconds=max_seconds, verbose=False)
    except Exception:
        prob.solve(solver=cp.GLPK_MI, verbose=False)
    if h.value is None:
        return None, None
    h_opt = np.round(np.array(h.value)).astype(float)
    t_opt = t.value
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # ----- Phase 1: binary MILP (small to medium n) -----
    if cp is not None:
        for n in (64, 80, 96):
            if time_left() < 5.0:
                break
            max_sec = min(300.0, time_left())
            h_opt, _ = solve_binary_milp(n, max_sec)
            if h_opt is None:
                continue
            c5 = compute_c5(h_opt)
            if c5 < best_score:
                best_score, best_h = c5, h_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ----- Phase 2: refine best binary with continuous momentum -----
    if best_h is not None:
        h_ref, sc_ref = gradient_descent_momentum(best_h, time_left, max_iter=200000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 3: random continuous restarts with momentum -----
    n_large = 256
    while time_left() > 5.0:
        v0 = rng.random(n_large)
        v0 = project_onto_box_sum(v0, n_large / 2.0)
        h_tmp, sc_tmp = gradient_descent_momentum(v0, time_left, max_iter=5000)
        if sc_tmp < best_score:
            best_score, best_h = sc_tmp, h_tmp.copy()
        if best_score <= target:
            break

    # ----- Fallback: uniform 0.5 if nothing else succeeded -----
    if best_h is None:
        n = 64
        best_h = np.full(n, 0.5)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    # final guard
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
