    # Compute correlation c_raw scaled without dx: np.correlate(h, 1-h, mode='full')
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c_raw)

    w = np.exp(alpha * (c_raw - max_c))
    # apply threshold: mask
    mask = w > weight_thresh
    w = w * mask  # zero low ones
    W = np.sum(w)
    if W == 0:
        # fallback: gradient of all shifts with max
        return compute_gradients_for_shifts(h, np.where(c_raw == max_c)[0] - (h.size - 1))
    # Compute A = conv(w, h) at positions i
    # Using correlation: corr_wh = np.correlate(w, h, mode='full')
    corr_wh = np.correlate(w, h, mode='full')
    start = h.size - 1  # shift offset

def gradient_descent_hgd(h0, step0=0.5, beta=0.9, alpha=15.0, K=20, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    s_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        # Compute current C5
        # not needed every iteration for gradient; compute occasionally to check improvement
        # Compute gradient
        grad = weighted_gradient(h, alpha=alpha, K=K)
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size schedule: decreasing
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box and sum
        h_new = project_onto_box_sum(h_new, s_target)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def weighted_gradient(h, alpha=15.0, K=20):
    n = h.shape[0]
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c_raw)
    # select top K shifts
    total_shifts = c_raw.size
    if K < total_shifts:
        top_idx = np.argpartition(-c_raw, K-1)[:K]
    else:
        top_idx = np.arange(total_shifts)
    c_top = c_raw[top_idx]
    # compute weights via softmax (exp of (c - max_c))
    w = np.exp(alpha * (c_top - max_c))
    w_sum = np.sum(w)
    if w_sum == 0:
        # fallback to equal weights on top shifts
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    shifts = top_idx - (n - 1)
    grad = np.zeros_like(h)
    for shift, weight in zip(shifts, w):
        grad += weight
        if shift >= 0:
            end = n - shift
            if end > 0:
                grad[:end] -= weight * h[shift:]
            if shift < n:
                grad[shift:] -= weight * h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                grad[s:] -= weight * h[:end]
            if s < n:
                grad[:end] -= weight * h[s:]
    return grad

def gradient_descent_max(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def next_prime_ge(x):
    """Small utility to find a prime >= x (x integer)."""
    p = int(x)
    if p <= 2:
        return 2
    if p % 2 == 0:
        p += 1
    while True:
        is_prime = True
        r = int(np.sqrt(p)) + 1
        for d in range(3, r, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            return p
        p += 2

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            end = n - shift
            if end > 0:
                grad[:end] -= h[shift:]
            if shift < n:
                grad[shift:] -= h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                grad[s:] -= h[:end]
            if s < n:
                grad[:end] -= h[s:]
        grad += 1.0 / len(shifts)
    return grad

def weighted_gradient(h, alpha=15.0, K=20):
    """Smooth max gradient using top‑K shifts."""
    n = h.shape[0]
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c_raw)
    total = c_raw.size
    if K < total:
        top_idx = np.argpartition(-c_raw, K - 1)[:K]
    else:
        top_idx = np.arange(total)
    c_top = c_raw[top_idx]
    w = np.exp(alpha * (c_top - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    shifts = top_idx - (n - 1)
    grad = np.zeros_like(h)
    for shift, weight in zip(shifts, w):
        grad += weight
        if shift >= 0:
            end = n - shift
            if end > 0:
                grad[:end] -= weight * h[shift:]
            if shift < n:
                grad[shift:] -= weight * h[:end]
        else:
            s = -shift
            end = n - s
            if end > 0:
                grad[s:] -= weight * h[:end]
            if s < n:
                grad[:end] -= weight * h[s:]
    return grad

def gradient_descent_hgd(h0, step0=0.5, beta=0.9, alpha=15.0, K=20,
                         time_end=None, max_iter=200000):
    """Projected gradient descent using weighted‑max gradient."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        grad = weighted_gradient(h, alpha=alpha, K=K)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def gradient_descent_max(h0, step0=0.5, beta=0.9,
                         time_end=None, max_iter=200000):
    """Projected GD using exact max‑shift subgradient."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def generate_initial_candidates(n, rng):
    cand_list = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = resample(init_full, n)
        cand = project_onto_box_sum(cand, n / 2.0)
        cand_list.append(cand)
    cand_list.append(thue_morse_binary(n))
    # Legendre‑based candidate
    p = next_prime_ge(n + 1)
    leg = legendre_sequence(p).astype(float)
    leg_up = resample(leg, n)
    leg_up = project_onto_box_sum(leg_up, n / 2.0)
    cand_list.append(leg_up)
    # a few random balanced binary vectors
    for _ in range(3):
        cand_list.append(random_binary_balanced(n, rng))
    # a random continuous vector (projected)
    v = rng.random(n)
    cand_list.append(project_onto_box_sum(v, n / 2.0))
    return cand_list

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose resolution (higher resolution often yields a tighter bound)
    n = 1024

    # Initial candidates

        # Occasionally perturb the best solution; otherwise restart randomly

        if sc_opt < best_score:
