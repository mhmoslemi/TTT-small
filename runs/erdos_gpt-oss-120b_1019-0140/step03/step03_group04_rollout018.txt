def smooth_gradient(h, beta):
    n = h.size
    dx = 2.0 / n
    L = 2 * n
    # compute autocorrelation g via FFT
    H = np.fft.rfft(h, n=L)
    g_full = np.fft.irfft(H * np.conj(H), n=L) * dx
    g = g_full[:2*n-1]   # length 2*n-1
    # compute weights w
    exp_arg = -beta * g
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # compute convolution of w and h (linear)
    # Use FFT: need appropriate lengths: conv_len = len(w) + n - 1 = 2*n-1 + n - 1 = 3*n - 2
    conv_len = 3*n - 2
    Lc = 1 << (conv_len-1).bit_length()  # next power of two for speed
    # compute conv_f = w * h (linear)
    W = np.fft.rfft(w, n=Lc)
    Hc = np.fft.rfft(h, n=Lc)
    conv_f = np.fft.irfft(W * Hc, n=Lc)[:conv_len]
    # conv_r = reversed w * h
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=Lc)
    conv_r = np.fft.irfft(W_rev * Hc, n=Lc)[:conv_len]
    offset = n - 1  # index of zero lag in conv_f and conv_r
    grad = -dx * (conv_f[offset:offset+n] + conv_r[offset:offset+n])
    return grad, w

def smooth_gradient(h, beta):
    n = h.size
    dx = 2.0 / n
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    g_full = np.fft.irfft(H * np.conj(H), n=L) * dx
    g = g_full[:2*n-1]
    # compute weights
    exp_arg = -beta * g
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # convolution lengths
    M = 2*n - 1
    conv_len = M + n - 1  # 3n - 2
    Lc = 1 << ((conv_len - 1).bit_length())
    # FFT of h for conv
    Hc = np.fft.rfft(h, n=Lc)
    # FFT of w
    W = np.fft.rfft(w, n=Lc)
    conv_f = np.fft.irfft(W * Hc, n=Lc)[:conv_len]
    # reversed w
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=Lc)
    conv_r = np.fft.irfft(W_rev * Hc, n=Lc)[:conv_len]
    offset = n - 1
    grad = -dx * (conv_f[offset:offset+n] + conv_r[offset:offset+n])
    return grad, w

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bracket
    for _ in range(max_iter):
        if f(lam_low) <= s <= f(lam_high):
            break
        lam_low -= (hi - lo) * n
        lam_high += (hi - lo) * n
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Legendre symbol sequence of length p-1 (p prime). 1 for residues, 0 otherwise."""
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i-1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_c5_fast(h):
    """Maximum overlap ∫ h(x)*(1‑h(x+k))dx approximated by FFT."""
    n = h.size
    dx = 2.0 / n
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(B), n=L)[:2*n-1] * dx
    return np.max(corr)

def smooth_gradient(h, beta):
    """Gradient of the smooth‑min of autocorrelation (J = 1‑smooth_min(g))."""
    n = h.size
    dx = 2.0 / n
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    g_full = np.fft.irfft(H * np.conj(H), n=L) * dx
    g = g_full[:2*n-1]                     # autocorrelation for shifts -(n-1)…(n-1)

    # soft‑min weights
    exp_arg = -beta * g
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    M = 2*n - 1
    conv_len = M + n - 1               # = 3*n-2
    Lc = 1 << ((conv_len - 1).bit_length())   # next power of two
    Hc = np.fft.rfft(h, n=Lc)
    W = np.fft.rfft(w, n=Lc)
    conv_f = np.fft.irfft(W * Hc, n=Lc)[:conv_len]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=Lc)
    conv_r = np.fft.irfft(W_rev * Hc, n=Lc)[:conv_len]
    offset = n - 1
    grad = -dx * (conv_f[offset:offset+n] + conv_r[offset:offset+n])
    return grad, w

            best_h = h_new.copy()
        h = h_new
        beta = min(beta + beta_inc, beta_final)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n = 1024                               # resolution
    s = n / 2.0

    # ---- initial candidate pool ----
    candidates = []

    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(init_full, n), s)
        candidates.append(cand)

    # Legendre (prime 1031 → length 1030)
    try:
        leg_raw = legendre_sequence(1031)
        leg = resample(leg_raw, n)
        leg = project_onto_box_sum(leg, s)
        candidates.append(leg)
    except Exception:
        pass

    # Thue–Morse
    candidates.append(thue_morse_binary(n))

    # few random balanced binaries
    for _ in range(3):
        candidates.append(random_binary_balanced(n, rng))

    # ---- evaluate simple candidates ----
    best_h = None
    best_score = np.inf
    for cand in candidates:
        if time_left() < 0.2:
            break
        sc = compute_c5_fast(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # ---- gradient descent restarts ----
    while time_left() > 15.0:
        # pick start point
        if best_h is not None and rng.random() < 0.4:
            h0 = best_h.copy()
        else:
            v = rng.random(n)
            h0 = project_onto_box_sum(v, s)
        # allocate iterations based on remaining time
        max_iters = int(min(20000, max(2000, time_left() * 800)))
        h_opt, sc_opt = gradient_descent_smooth(
            h0, max_iters, time_end=start + budget_s - 0.5)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # ---- binarize best continuous solution ----
    if best_h is not None:
        bin_h = np.zeros_like(best_h)
        idx = np.argpartition(-best_h, n//2 - 1)[:n//2]
        bin_h[idx] = 1.0
        sc_bin = compute_c5_fast(bin_h)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, bin_h.copy()

    # ---- local binary swap search (if binary) ----
    if best_h is not None and np.all(np.logical_or(best_h == 0, best_h == 1)):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        swap_iters = int(min(50000, max(2000, time_left() * 500)))
        for _ in range(swap_iters):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            best_h[i], best_h[j] = 0.0, 1.0
            sc_new = compute_c5_fast(best_h)
            if sc_new < best_score:
                best_score = sc_new
                ones = np.where(best_h == 1)[0]
                zeros = np.where(best_h == 0)[0]
            else:
                best_h[i], best_h[j] = 1.0, 0.0
            if best_score <= 0.3808:
                break

    # final safety projection and scoring
    best_h = project_onto_box_sum(best_h, s)
    best_score = compute_c5_fast(best_h)
    return best_h, best_score, n
