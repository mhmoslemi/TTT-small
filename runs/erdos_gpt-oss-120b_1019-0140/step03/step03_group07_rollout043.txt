def solve_milp_binary(n, time_limit=60, init_h=None):
    M = n // 2
    x = cp.Variable(n, integer=True, name="x")
    a = cp.Variable(integer=True, name="a")
    constraints = [cp.sum(x) == M,
                   x >= 0,
                   x <= 1,
                   a >= 0,
                   a <= M]
    ws = []
    for k in range(1, n):
        size = n - k
        w = cp.Variable(size, integer=True, name=f"w_{k}")
        ws.append((k, w))
        # linking constraints
        constraints += [w <= x[:size],
                        w <= x[k:],
                        w >= x[:size] + x[k:] - 1,
                        w >= 0,
                        w <= 1,
                        cp.sum(w) >= a]
    # objective
    objective = cp.Maximize(a)
    prob = cp.Problem(objective, constraints)
    # warm start if init_h provided: set x.value
    if init_h is not None:
        # init_h may be fractional or binary; round to nearest int
        x0 = np.rint(init_h).astype(int)
        # If round changes sum, adjust to get exact M
        cur_sum = np.sum(x0)
        if cur_sum != M:
            diff = M - cur_sum
            # adjust by flipping zeros/ones with smallest distance to 0.5 maybe
            # but simply set first diff entries accordingly
            # Find indices 0..n-1 where x0[i]=0 and diff>0, set to 1
            # Or where x0[i]=1 and diff<0, set to 0
            # We can cheat: just set first diff zeros to 1
            zeros = np.where(x0 == 0)[0]
            ones = np.where(x0 == 1)[0]
            if diff > 0:
                x0[zeros[:diff]] = 1
            elif diff < 0:
                x0[ones[: -diff]] = 0
        x.value = x0
        # also assign w values maybe None; no need
    # Solve with time limit optional
    try:
        prob.solve(solver=cp.CBC, verbose=False)  # or cp.GLPK_MI if no CBC

        # fallback to GLPK_MI

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # maybe not solved; return best known via x.value? but may be partial.
        # We'll still try to extract.

    x_opt = np.array(x.value).flatten()

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def solve_feasible_milp(n, target_a, init_h=None, time_limit=60):
    import cvxpy as cp, numpy as np, time
    M = n // 2
    x = cp.Variable(n, integer=True)
    constraints = [cp.sum(x) == M,
                   x >= 0,
                   x <= 1]
    # For each k
    for k in range(1, n):
        size = n - k
        w = cp.Variable(size, integer=True)
        constraints += [w <= x[:size],
                        w <= x[k:],
                        w >= x[:size] + x[k:] - 1,
                        w >= 0,
                        w <= 1,
                        cp.sum(w) >= target_a]
    # dummy objective
    objective = cp.Minimize(0)
    prob = cp.Problem(objective, constraints)
    # warm start if init_h provided
    if init_h is not None:
        # Round init_h to integer 0/1
        x0 = np.rint(init_h).astype(int)
        cur_sum = np.sum(x0)
        if cur_sum != M:
            diff = M - cur_sum
            zeros = np.where(x0 == 0)[0]
            ones = np.where(x0 == 1)[0]
            if diff > 0:
                x0[zeros[:diff]] = 1
            elif diff < 0:
                x0[ones[: -diff]] = 0
        x.value = x0
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_iters=2000000, max_seconds=time_limit)
    except TypeError:
        # maybe older cvxpy version: use cbc_params
        try:
            prob.solve(solver=cp.CBC, verbose=False, cbc_params={'seconds': time_limit})
        except Exception:
            # fallback to GLPK_MI
            prob.solve(solver=cp.GLPK_MI, verbose=False)
    # Check status
    if prob.status not in ["optimal", "optimal_inaccurate", "feasible"]:
        return None, None
    x_val = np.rint(np.array(x.value).flatten()).astype(int)
    # In case not integer due to tolerance
    # Not compute a, we can compute actual min overlap
    # Compute a_val as min sum_i w (via h)

    # compute actual a from correlation: we can compute sum_i h[i] * h[i+k] for each k and take min.

        s = np.sum(h[:n-k] * h[k:])
        a_vals.append(s)
    a_val = min(a_vals) if a_vals else 0
    return h, a_val

import time
import math
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initial_h(n, rng):
    """Construct a feasible start: random continuous respecting sum = n/2."""
    M = n // 2
    h = rng.random(n)
    h = h / np.sum(h) * M
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur < M:
        deficit = M - cur
        for i in np.argsort(h):
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    elif cur > M:
        excess = cur - M
        for i in np.argsort(-h):
            dec = min(excess, h[i])
            h[i] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Choose resolution, preferring a parent construction if present
    if 'initial_h_values' in globals() and len(initial_h_values) % 2 == 0:
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float).copy()
        M = n // 2
        cur = np.sum(h)
        if abs(cur - M) > 1e-12:
            h += (M - cur) / n
            h = np.clip(h, 0.0, 1.0)
    else:
        n = 80
        h = _initial_h(n, rng)

    M = n // 2
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing parameters
    total_anneal = budget_s * 0.92
    anneal_end = start + total_anneal
    T0 = cur_c5 * 0.02 + 1e-4   # small positive start temperature
    min_T = 1e-9

    while time.time() < anneal_end:
        elapsed = time.time() - start
        t_frac = (elapsed) / total_anneal
