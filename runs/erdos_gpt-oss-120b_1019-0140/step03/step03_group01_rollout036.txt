def hill_climb_binary(h0, rng, time_left, max_no_improve_factor=5):
    n = len(h0)
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    max_no_improve = max_no_improve_factor * n
    while time_left() > 0.01 and no_improve < max_no_improve:
        # pick random index from ones and zeros
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # update ones and zeros arrays temporarily need to revert if reject, but we cannot update until accept.
        # We'll compute new score first.
        new = compute_c5_fft(h)
        if new < cur:
            cur = new
            no_improve = 0
            # accept swap: update ones/zeros arrays
            ones[i_idx] = j
            zeros[j_idx] = i
            if new < best_score:
                best_score = new
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        # early stop if best_score <= target
        if best_score <= target:
            break
    return best_h, best_score

import time, numpy as np

def compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[:2*n-1]
    # conv is real but may have tiny imaginary due to rounding; we take max of real part.
    return np.max(conv) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # generate using parity of bits
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def hill_climb_binary(h0, rng, time_left, target=0.3808, max_no_improve_factor=10):
    n = len(h0)
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    max_no_improve = max_no_improve_factor * n
    no_improve = 0
    while time_left() > 0.01 and no_improve < max_no_improve:
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5_fft(h)
        if new < cur:
            cur = new
            no_improve = 0
            # accept: update indices
            ones[i_idx] = j
            zeros[j_idx] = i
            if new < best_score:
                best_score = new
                best_h = h.copy()
            if best_score <= target:
                break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if time_left() <= 0.0:
            break
    return best_h, best_score

def simulated_annealing_binary(h0, rng, time_left, target=0.3808, T0=0.02, decay=0.9999, T_min=1e-6):
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur
    n = h.size
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    iter = 0
    while time_left() > 0.01:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5_fft(h)
        delta = new - cur
        T = max(T_min, T0 * (decay ** iter))
        accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
        if accept:
            cur = new
            # update index arrays: swap entries
            ones[i_idx] = j
            zeros[j_idx] = i
            if new < best_score:
                best_score = new
                best_h = h.copy()
            if best_score <= target:
                break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        iter += 1
        # optional break if iter hugely large
    return best_h, best_score

    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()

    momentum = np.zeros_like(h)

        eta = step0 / np.sqrt(it)

        # project onto box and sum

        score = compute_c5_fft(h_new)

            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # ensure bounds contain solution
    # iterate bisection
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # make sure the interval contains the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def compute_corr_fft(h):
    """Linear correlation of h with (1‑h) without scaling."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    return np.fft.irfft(H * G, n=L)[: 2 * n - 1]


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation objective for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum(h0, rng, time_left, step0=0.5, beta=0.9, max_iters=200000):
    """Momentum gradient descent on the continuous problem."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iters and time_left() > 0.01:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        # active shifts (tiny tolerance accounts for float noise)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def hill_climb_binary(h0, rng, time_left, target=0.3808, max_no_improve_factor=10):
    """Deterministic swap hill‑climb on binary vectors."""
    n = len(h0)
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    max_no_improve = max_no_improve_factor * n
    no_improve = 0
    while time_left() > 0.01 and no_improve < max_no_improve:
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5_fft(h)
        if new < cur:
            cur = new
            no_improve = 0
            # accept update of index tables
            ones[i_idx] = j
            zeros[j_idx] = i
            if new < best_score:
                best_score = new
                best_h = h.copy()
            if best_score <= target:
                break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_h, best_score


def simulated_annealing_binary(h0, rng, time_left, target=0.3808,
                               T0=0.02, decay=0.9999, T_min=1e-6):
    """Simple SA allowing uphill moves."""
    h = h0.copy()
    cur = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur
    n = h.size
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    while time_left() > 0.01:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5_fft(h)
        delta = new - cur
        T = max(T_min, T0 * (decay ** it))
        accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
        if accept:
            cur = new
            ones[i_idx] = j
            zeros[j_idx] = i
            if new < best_score:
                best_score = new
                best_h = h.copy()
            if best_score <= target:
                break
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best_score
