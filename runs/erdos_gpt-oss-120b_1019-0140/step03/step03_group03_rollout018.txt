def anneal_binary(h, max_iters, t_start, t_end, rng):
    n = len(h)
    ones = np.flatnonzero(h).tolist()
    zeros = np.flatnonzero(1 - h).tolist()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(1, max_iters + 1):
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        t = t_start * ((t_end / t_start) ** (it / max_iters))
        if new_score < cur_score or (t > 0 and rng.random() < np.exp(-(new_score - cur_score) / t)):

            # update lists

            zeros[idx_j] = i

                best_h = h.copy()

            h[i] = 1.0
            h[j] = 0.0
        if it % 10000 == 0:
            # optional recompute
            if rng.random() < 0.5:  # not needed maybe
                ones = np.flatnonzero(h).tolist()
                zeros = np.flatnonzero(1 - h).tolist()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def random_binary_balanced(n, rng):
    """Balanced binary vector with n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def thue_morse(n):
    """Thue–Morse sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def upsample(arr, new_n):
    """Linear interpolation up‑sampling."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_corr_fft(h):
    """Full linear correlation of h with (1-h) via FFT."""
    n = h.shape[0]
    g = 1.0 - h
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h_rev, N)
    G = np.fft.rfft(g, N)
    conv = np.fft.irfft(H * G, N)
    return conv[: 2 * n - 1]


def compute_c5_fft(h):
    """C₅ using FFT (fast)."""
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)


def compute_c5_np(h):
    """Exact C₅ using NumPy correlate (for final reporting)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of max‑correlation term for given active shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[: n - shift] = h[shift:]
            term2[shift:] = h[: n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[: n - s]
            term2[: n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum (continuous refinement)."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary(h0, max_iters, t_start, t_end, rng):
    """Simulated annealing on binary balanced vectors."""
    h = h0.copy()
    n = h.size
    ones = np.flatnonzero(h).tolist()
    zeros = np.flatnonzero(1.0 - h).tolist()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(1, max_iters + 1):
        # pick a random 1 and a random 0
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        # exponential cooling
        t = t_start * ((t_end / t_start) ** (it / max_iters))
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if t > 0 and rng.random() < np.exp(-(new_score - cur_score) / t):
                accept = True
        if accept:
            cur_score = new_score
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        # occasional clean‑up
        if it % 10000 == 0:
            ones = np.flatnonzero(h).tolist()
            zeros = np.flatnonzero(1.0 - h).tolist()
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # problem size – choose the largest power of two fitting the budget
    n = 1024

    # ------------------------------------------------------------------
    # Build a set of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # up‑sample parent if provided
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n:
            init_up = upsample(init_full, n)
        else:
            init_up = init_full.copy()
        # binarize by top half values
        order = np.argpartition(init_up, -n // 2)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[order] = 1.0
        candidates.append(h_bin)

    # Thue–Morse (balanced for powers of two)
    tm = thue_morse(n)
    if np.sum(tm) == n // 2:
        candidates.append(tm)
    else:
        order = np.argpartition(tm, -n // 2)[-n // 2:]
        h_tm = np.zeros(n, dtype=float)
        h_tm[order] = 1.0
        candidates.append(h_tm)

    # a few random balanced binaries
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))

    # random real vectors turned binary by threshold
    for _ in range(2):
        v = rng.random(n)
        order = np.argpartition(v, -n // 2)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[order] = 1.0
        candidates.append(h_bin)

    # pick the best of the cheap candidates
    best_h = None
    best_score = np.inf
    for h0 in candidates:
        sc = compute_c5_np(h0)
        if sc < best_score:
            best_score = sc
            best_h = h0.copy()

    # ------------------------------------------------------------------
    # Simulated annealing on the binary space
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 8.0:
        # Allocate ~80 % of remaining time to annealing
        anneal_time = remaining * 0.8
        # Heuristic: ~2000 swaps per second (depends on hardware)
        max_iters = int(min(500_000, anneal_time * 2000))
        max_iters = max(max_iters, 1000)
        best_h, best_score = anneal_binary(best_h, max_iters,
                                           t_start=0.5, t_end=1e-6, rng=rng)

    # ------------------------------------------------------------------
    # Continuous refinement via projected gradient descent
    # ------------------------------------------------------------------
    remaining = time_left()
    if remaining > 6.0:
        gd_iters = int(min(200_000, remaining * 2000))
        best_h, best_score = gradient_descent_momentum(best_h,
                                                       n_iter=gd_iters,
                                                       step0=0.5, beta=0.9)

    # ------------------------------------------------------------------
    # Final projection and exact evaluation
    # ------------------------------------------------------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5_np(best_h)

    return best_h, final_score, best_h.size
