def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)
    # parameters
    n = 256  # target resolution
    target_bound = 0.3808

    # initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        # upsample/downsample to n
        if init_full.size != n:
            # upsample using linear interpolation
            x_old = np.linspace(0, 1, init_full.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            init_n = np.interp(x_new, x_old, init_full)
        else:
            init_n = init_full.copy()
        init_n = project_onto_box_sum(init_n, n/2.0)
        candidates.append(init_n)
    # Thue-Morse of length n
    candidates.append(thue_morse(n).astype(float))
    # random binary balanced vectors
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng).astype(float))
    # random continuous vectors projected to sum constraint
    for _ in range(2):
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n/2.0))

    best_h = None
    best_score = np.inf

    # Phase: smooth descent on each candidate
    for h0 in candidates:
        if time_left() < 5.0:
            break
        # parameters based on remaining time
        # allocate maybe 2000 iterations or more
        n_iter = int(min(5000, max(2000, time_left() * 30))) # rough
        h_opt, sc_opt = smooth_descent(
            h0, n_iter=n_iter,
            step0=0.7,
            alpha_start=5.0,
            alpha_end=30.0,
            beta=0.95,
            time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    # If no best yet, fallback to random (maybe use projection of uniform)
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n, 0.5), n/2.0)
        best_score = compute_c5(best_h)

    # Binary hill climbing from rounding best continuous solution
    # Round to binary by top half
    if best_h is not None:
        binary = np.zeros_like(best_h)
        # Choose indices with top values
        idx_top = np.argpartition(best_h, -n//2)[-n//2:]
        binary[idx_top] = 1.0
        bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=target_bound)
        if bin_sc < best_score:
            best_score, best_h = bin_sc, bin_h.copy()

    # Additional random binary hill climbs if time
    # Perform a few more attempts
    attempts = 3
    while time_left() > 5.0 and attempts > 0:
        rand_bin = random_binary_balanced(n, rng).astype(float)
        b_h, b_sc = binary_hill_climb_sa(rand_bin, rng, time_left, target=target_bound)
        if b_sc < best_score:
            best_score, best_h = b_sc, b_h.copy()
        attempts -= 1

    # Final continuous refinement (optional)
    if time_left() > 2.0:
        # Use momentum descent on best solution (continuous) for a few thousand iterations
        h_ref, sc_ref = momentum_grad_descent(best_h,
                                               n_iter=int(min(10000, max(2000, time_left()*20))),
                                               step0=0.3,
                                               beta=0.95,
                                               time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, n/2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        # gradient of max corr: find shifts where corr is approx max
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = corr.max()
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    # maximum iterations estimated from time left
    max_iters = int(min(200000, max(1000, time_left()*1500)))
    T0 = 0.01
    for it in range(1, max_iters+1):
        if time_left() < 0.01:
            break
        # propose swapping a 0 and 1
        # Use random indices
        i = rng.integers(n)
        while h[i] != 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 1.0:
            j = rng.integers(n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta/T)):
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Project v onto {x: lo≤x_i≤hi, sum x_i = s} using bisection on lambda."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    # lambda bounds such that f(lambda) = sum clip(v - lambda, lo, hi)
    lam_low = np.min(v - hi) - 1.0
    lam_high = np.max(v - lo) + 1.0

    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if abs(total - s) <= 1e-12:
            return proj
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def grad_softmax_corr(h, alpha):
    """Gradient of (1/α)·log Σ exp(α·corr) w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_descent(h0, n_iter, step0, alpha_start, alpha_end, beta, time_left):
    """Projected gradient descent on soft‑max approximation of the max‑overlap."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha) * (2.0 / n)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:])
            grad[shift:]   -= h[:n - shift]
        else:
            s = -shift
            grad[s:]      += (1.0 - h[:n - s])
            grad[:n - s]  -= h[s:]
    return grad

def momentum_grad_descent(h0, n_iter, step0, beta, time_left):
    """Projected gradient descent on the true max‑overlap objective with momentum."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_sc = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = corr.max()
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts) * (2.0 / n)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_sc

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing hill‑climb on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_sc = compute_c5(h)
    best_h, best_sc = h.copy(), curr_sc
    max_iters = int(min(200000, max(1000, time_left() * 1500)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(n)
        while h[i] != 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 1.0:
            j = rng.integers(n)
        h_new = h.copy()
        h_new[i], h_new[j] = 1.0, 0.0
        new_sc = compute_c5(h_new)
        delta = new_sc - curr_sc
        T = T0 * (1.0 - it / max_iters)
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h, curr_sc = h_new, new_sc
            if new_sc < best_sc:
                best_sc, best_h = new_sc, h_new.copy()
                if best_sc <= target:
                    break
    return best_h, best_sc
