def de_optimize(n, time_left, pop=30, F=0.5, CR=0.9, max_gen=10000):
    rng = np.random.default_rng()
    # init pop
    pop_vec = []
    for i in range(pop):
        v = rng.random(n)
        v = project_onto_box_sum(v, n/2)
        pop_vec.append(v)
    pop_arr = np.stack(pop_vec)
    scores = np.array([compute_c5(v) for v in pop_arr])
    best_idx = np.argmin(scores)
    best_h = pop_arr[best_idx].copy()
    best_score = scores[best_idx]
    gen = 0
    while time_left() > 0.2 and gen < max_gen:
        for i in range(pop):
            # mutation
            idxs = list(range(pop))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop_arr[a] + F*(pop_arr[b] - pop_arr[c])
            # clip + project
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, n/2)
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0,n)] = True
            trial = np.where(cross, mutant, pop_arr[i])
            # evaluate
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop_arr[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
    return best_h, best_score

def de_optimize(n, rng, time_left, init_h=None, pop_size=30, F=0.5, CR=0.9, max_gen=10000):
    # initialize
    pop = []
    if init_h is not None:
        pop.append(init_h.copy())
    while len(pop) < pop_size:
        v = rng.random(n)
        v = project_onto_box_sum(v, n/2.0)
        pop.append(v)
    pop = np.stack(pop)
    scores = np.array([compute_c5(v) for v in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    gen = 0
    while time_left() > 0.2 and gen < max_gen:
        for i in range(pop.shape[0]):
            # if time left low, break loop
            if time_left() < 0.2:
                break
            # mutation
            idxs = list(range(pop.shape[0]))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, n/2.0)
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0,n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # shift right?
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find all shifts achieving max (within tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
```

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000, T_start=0.1, T_min=1e-5, decay=0.99995):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T_start
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score
```

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=200000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000,
                       T_start=0.1, T_min=1e-5, decay=0.99995):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T_start
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def de_optimize(n, rng, time_left, init_h=None,
                pop_size=30, F=0.8, CR=0.9, max_gen=10000):
    """Simple Differential Evolution with projection onto box+sum."""
    pop = []
    if init_h is not None:
        pop.append(project_onto_box_sum(init_h, n / 2.0))
    while len(pop) < pop_size:
        v = rng.random(n)
        pop.append(project_onto_box_sum(v, n / 2.0))
    pop = np.stack(pop)
    scores = np.array([compute_c5(v) for v in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    gen = 0
    while time_left() > 0.2 and gen < max_gen:
        for i in range(pop.shape[0]):
            if time_left() < 0.2:
                break
            idxs = list(range(pop.shape[0]))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_onto_box_sum(mutant, n / 2.0)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
    return best_h, best_score
