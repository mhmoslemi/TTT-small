def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    # if target sum out of bounds
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # initial lam region
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds to bracket s
    # Ensure f(lam_low) >= s and f(lam_high) <= s? Actually f(lam) is decreasing in lam.
    # At lam -> -inf, v - lam > hi => clip to hi => sum = hi * n (maximum)
    # At lam -> +inf, clip to lo => sum = lo * n (minimum)

    # Starting values from v might not satisfy; adjust:

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # add jitter occasionally
        if it % 500 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        # project to constraints
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # overlapping region i=0..n-shift-1
            # term1[i] = h[i+shift] for i in overlapping region
            # term2[i+shift] = h[i] for i in overlapping region
            # we can compute using slices
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # expand interval until it brackets the target sum
    while f(lam_low) < s:
        lam_low -= 1.0
    while f(lam_high) > s:
        lam_high += 1.0

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max correlation ∫ h(x)(1-h(x+k))dx approximated on grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from size len(arr) to new_n (periodic domain)."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Binary Thue–Morse sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    """Gradient (subgradient) of max correlation w.r.t. h for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected gradient descent with momentum on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def de_cost_factory(target_sum):
    """Factory returning a DE cost function with a heavy sum‑penalty."""
    def cost(x):
        s = np.sum(x)
        penalty = 0.0
        if abs(s - target_sum) > 1e-4:
            penalty = 1e6 * (s - target_sum) ** 2
        return compute_c5(x) + penalty
    return cost

def de_optimize(init_h, n, time_left, rng):
    """Global search via differential evolution."""
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    # adapt generations to remaining time
    max_iter = int(min(200, max(20, time_left() * 4)))
    popsize = max(5, int(2000 // n))      # total individuals = popsize * n
    cost_fun = de_cost_factory(target_sum)

    def callback(xk, convergence):
        return time_left() <= 0

    res = opt.differential_evolution(
        cost_fun,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=int(rng.integers(2**32)),
        callback=callback,
        polish=False,
        init='latinhypercube')
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256]
    for n in scales:
        if best_h.size != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)

        # Global DE search
        if time_left() > 5:
            cand = de_optimize(best_h, n, time_left, rng)
            cand = project_onto_box_sum(cand, n / 2.0)
            sc = compute_c5(cand)
            if sc < best_score:
                best_h, best_score = cand, sc

        # Local momentum gradient descent
        if time_left() > 5:
            iters = int(min(80000, max(8000, time_left() * 30)))
            step0 = 0.5 if n <= 128 else 0.3
            h_ref, sc_ref = gradient_descent_momentum_custom(
                best_h, n_iter=iters, step0=step0, beta=0.9,
                rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_h, best_score = h_ref, sc_ref

        # Small batch of random balanced binaries
        if time_left() > 5:
            for _ in range(5):
                h_rand = random_binary_balanced(n, rng)
                sc_rand = compute_c5(h_rand)
                if sc_rand < best_score:
                    best_h, best_score = h_rand, sc_rand

    # Final fine‑grained perturbation loop
    while time_left() > 5:
        i, j = rng.choice(best_h.size, size=2, replace=False)
        if best_h[i] > 0 and best_h[j] < 1:
            delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.1, 0.5)
            if delta > 0:
                h_new = best_h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc_new = compute_c5(h_new)
                if sc_new < best_score:
                    best_h, best_score = h_new, sc_new
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
