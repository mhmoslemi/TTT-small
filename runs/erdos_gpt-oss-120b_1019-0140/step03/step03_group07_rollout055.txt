def binary_anneal(n, rng, time_budget, start_time):
    M = n // 2
    # initial h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h0 = np.array(initial_h_values, copy=False)
        h0 = np.clip(h0, 0.0, 1.0)
        h = (h0 >= 0.5).astype(np.int8)
        cur = int(h.sum())
        diff = M - cur
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if zeros.size >= diff:
                idx = rng.choice(zeros, diff, replace=False)
                h[idx] = 1
        elif diff < 0:
            ones = np.where(h == 1)[0]
            if ones.size >= -diff:
                idx = rng.choice(ones, -diff, replace=False)
                h[idx] = 0
    else:
        h = initial_binary_sequence(n, rng)
    counts = compute_counts(h)
    cur_max = int(counts.max())
    best_h = h.copy()
    best_max = cur_max
    target_max = int(np.floor(0.3808 * n / 2))
    # SA loop
    while True:
        now = time.time()
        if now - start_time >= time_budget - 0.2:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        delta = swap_delta_counts(h, i, j)
        new_counts = counts + delta
        new_max = int(new_counts.max())
        # temperature schedule: linear based on elapsed fraction
        elapsed = now - start_time
        frac = elapsed / time_budget
        T0 = max(1.0, cur_max - target_max)
        T = T0 * (1 - frac)
        # acceptance
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            # exponential acceptance
            prob = np.exp(-(new_max - cur_max) / (T + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            # apply swap
            h[i] = 1
            h[j] = 0
            counts = new_counts
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                if best_max <= target_max:
                    break
    c5 = best_max * (2.0 / n)
    return best_h.astype(float), c5
```

import time
import numpy as np

# ---------- utilities ----------
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_counts(h):
    return np.correlate(h, 1 - h, mode="full")

def swap_delta_counts(h, i, j):
    n = h.shape[0]
    k = np.arange(-n + 1, n, dtype=np.int32)
    delta = np.zeros_like(k, dtype=np.int32)

    idx = i + k
    mask = (idx >= 0) & (idx < n)
    delta[mask] += (1 - h[idx[mask]])

    idx = j + k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= (1 - h[idx[mask]])

    idx = i - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]

    idx = j - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]

    return delta

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5)
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=np.int8)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1 if leg == 1 else 0
    return seq

        h[idx] = 1
