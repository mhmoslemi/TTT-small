def compute_c5(h):
    n = h.size
    # compute linear correlation of h and (1-h)
    a = h
    b = 1.0 - h
    size = 2*n - 1
    # FFT length as next power of 2
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    # correlation via convolution of a and reversed b?
    # cross-correlation: corr[k] = sum_i a[i] * b[i+k], with out-of-range b treated as zero.
    # This equals convolution of a and reversed b? Actually conv of a and rev(b) yields corr with shift index relative.
    # Let's compute conv = np.fft.ifft(np.fft.fft(a, m) * np.fft.fft(b[::-1], m))
    # conv is length m, we take first size terms.
    conv = np.fft.ifft(np.fft.fft(a, m) * np.fft.fft(b[::-1], m)).real
    corr = conv[:size]
    return np.max(corr) * (2.0/n)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Accurate subgradient of max‑correlation over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # (1 - h[i+shift]) part
            grad[:n - shift] += 1.0 - h[shift:]
            # -h[i-shift] part
            grad[shift:] -= h[:n - shift]
        else:
            s = -shift
            # (1 - h[i-s]) part
            grad[s:] += 1.0 - h[:n - s]
            # -h[i+s] part
            grad[:n - s] -= h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_time_limited(h0, time_budget, step0=0.8, beta=0.9):
    """Projected gradient descent with momentum, stops after time_budget seconds."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * (2.0 / n)
    momentum = np.zeros_like(h)
    it = 0
    start = time.time()
    while time.time() - start < time_budget:
        it += 1
        # current correlation and max shift set
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        # evaluate new point
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        max_corr_new = np.max(corr_new)
        score_new = max_corr_new * (2.0 / n)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simulated annealing on a balanced binary vector."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

        t_chunk = min(20.0, time_left() - 5.0)
        v = rng.random(n)
        start_h = project_onto_box_sum(v, n / 2.0)
        h_gd, sc_gd = gradient_descent_momentum_time_limited(start_h, t_chunk,
                                                             step0=0.8, beta=0.9)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd

    # Phase C: binary annealing at half resolution, then upsample
    if time_left() > 5.0:
        n_bin = n // 2  # 512
        h_bin_start = downsample(best_h, n_bin)
        idx = np.argpartition(h_bin_start, -n_bin // 2)[-n_bin // 2:]
        h_bin = np.zeros(n_bin, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(time_left() * 600)  # ~600 proposals per second
        h_ann_bin, sc_ann_bin = anneal_binary(h_bin, max_iter_ann,
                                              temp_start=0.02,
                                              temp_end=1e-6,
                                              rng=rng)
        h_ann = upsample(h_ann_bin, n)
        h_ann = project_onto_box_sum(h_ann, n / 2.0)
        sc_ann = compute_c5(h_ann)
        if sc_ann < best_score:
            best_h, best_score = h_ann, sc_ann

    # Phase D: final refinement with remaining time
    if time_left() > 5.0:
        h_gd, sc_gd = gradient_descent_momentum_time_limited(best_h,
                                                             time_left(),
                                                             step0=0.8,
                                                             beta=0.9)
        if sc_gd < best_score:
            best_h, best_score = h_gd, sc_gd

    # Final projection and score
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
