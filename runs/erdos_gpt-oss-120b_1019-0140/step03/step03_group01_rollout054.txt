def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=500000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5(h_new)
        if score_new < best_score:

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo≤x_i≤hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=500000):
    """Projected gradient descent with momentum on the C5 objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def improve_binary(h, rng, time_left_func, target_max_corr=None):
    """Attempt to lower the maximum correlation of a binary vector by swaps."""
    n = h.shape[0]
    while time_left_func() > 0.02:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = int(corr.max())
        if (target_max_corr is not None) and (max_corr <= target_max_corr):
            break
        shift_idxs = np.where(corr == max_corr)[0] - (n - 1)
        improved = False
        for s in shift_idxs:
            if s >= 0:
                i_range = np.arange(0, n - s)
                j_range = i_range + s
            else:
                s_abs = -s
                i_range = np.arange(s_abs, n)
                j_range = i_range - s_abs
            h_i = h[i_range]
            h_j = h[j_range]
            bad = i_range[(h_i == 1) & (h_j == 0)]
            good = i_range[(h_i == 0) & (h_j == 1)]
            if bad.size and good.size:
                i_bad = bad[rng.integers(bad.size)]
                i_good = good[rng.integers(good.size)]
                h[i_bad] = 0.0
                h[i_good] = 1.0
                new_corr = np.correlate(h, 1 - h, mode='full')
                new_max = int(new_corr.max())
                if new_max <= max_corr:
                    improved = True
                    break
                # revert if not better
                h[i_bad] = 1.0
                h[i_good] = 0.0
        if not improved:
            # random perturbation to escape stagnation
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = ones[rng.integers(ones.size)]
            j = zeros[rng.integers(zeros.size)]
            h[i], h[j] = 0.0, 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # ---- Initial construction -------------------------------------------------
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: balanced random binary of length 64
        n0 = 64
        init_full = np.zeros(n0, dtype=float)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        init_full[idx] = 1.0

    n0 = init_full.shape[0]
    n_target = 256
    if n_target % n0 != 0:
        n_target = n0  # use original size if not a divisor
    factor = n_target // n0
    if factor > 1:
        h0 = np.repeat(init_full, factor)
    else:
        h0 = init_full.copy()
    h0 = project_onto_box_sum(h0, n_target / 2.0)

    # ---- Phase 1: Continuous momentum descent ---------------------------------
    best_h, best_score = gradient_descent_momentum_with_time(
        h0, time_left, step0=0.5, beta=0.9, max_iter=500000
    )

    # ---- Phase 2: Binary conversion -------------------------------------------
    if time_left() > 5.0:
        k = n_target // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        bin_score = compute_c5(h_bin)
        if bin_score < best_score:
            best_h, best_score = h_bin.copy(), bin_score

        # ---- Phase 3: Targeted binary improvement ------------------------------
        target_c5 = 0.3808
        target_max_corr = int(np.floor(target_c5 * (n_target / 2)))
        h_bin_improved = improve_binary(best_h.copy(), rng, time_left, target_max_corr)
        imp_score = compute_c5(h_bin_improved)
        if imp_score < best_score:
            best_h, best_score = h_bin_improved.copy(), imp_score

        # ---- Phase 4: Final fine‑tune (continuous) -----------------------------
        if time_left() > 5.0:
            h_ref, ref_score = gradient_descent_momentum_with_time(
                best_h, time_left, step0=0.3, beta=0.9, max_iter=200000
            )
            if ref_score < best_score:
                best_h, best_score = h_ref.copy(), ref_score

    # ---- Guard against numerical drift ---------------------------------------
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
