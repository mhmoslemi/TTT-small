def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    # binary search for the shift lambda such that sum(clip(v+λ, lo, hi))=target
    low, high = -1e6, 1e6
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without dx factor)."""
    n = h.shape[0]

        grad[k:] -= h[:n - k]
    else:

        grad[kk:] += 1.0 - h[:n - kk]

        m_hat = mom / (1 - beta1 ** t)
        v_hat = vel / (1 - beta2 ** t)
        lr = lr0 / (1 + 0.0005 * t)   # mild decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        h = project_to_box_sum(h_new, target_sum)

        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # occasional random kick if stagnation
        if t % 400 == 0 and cur_c5 >= best_c5 - 1e-8:
            noise = np.random.default_rng().normal(scale=0.025, size=n)
            h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
    return best_h, best_c5

# ------------------- Differential Evolution -------------------
def de_optimize(init_h, target_sum, time_budget, n, seed):
    """Run a compact DE, projecting each trial onto the feasible set."""
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        h = project_to_box_sum(x, target_sum)
        return compute_c5(h)
    # allocate generations based on remaining time
    # popsize=4 gives population size 4*n (≈320 for n=80)
    maxiter = max(1, int(time_budget // 6))  # very rough budgeting
    try:
        res = optimize.differential_evolution(
            func=obj,
            bounds=bounds,
            seed=seed,
            maxiter=maxiter,
            popsize=4,
            polish=False,
            updating='deferred',
            disp=False)
        h_opt = project_to_box_sum(res.x, target_sum)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    except Exception:
        return init_h, compute_c5(init_h)

# ------------------- Main entry point -------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # initialise dimension and feasible start
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target)
    else:
        n = 80
        target = n / 2.0
        h0 = rng.random(n)
        h0 = project_to_box_sum(h0, target)

    best_h, best_c5 = h0.copy(), compute_c5(h0)

    # ---- Phase 1: Differential Evolution (≈20% of budget) ----
    time_de = budget_s * 0.20
    if time.time() - start_time < time_de:
        h_de, c5_de = de_optimize(best_h, target, time_de, n, seed)
        if c5_de < best_c5:
            best_h, best_c5 = h_de, c5_de

    # ---- Phase 2: Adam sub‑gradient descent (≈70% of budget) ----
    time_adam_end = start_time + budget_s * 0.90  # leave a small buffer
    h_adam, c5_adam = adam_descent(best_h, target, time_adam_end)
    if c5_adam < best_c5:
        best_h, best_c5 = h_adam, c5_adam

    # ---- Optional tiny polishing with a few SLSQP steps (≤5 s) ----
    if time.time() - start_time < budget_s - 5.0:
        # create simple constraints for SLSQP without slack variable
        def sum_con(x):
            return np.sum(x) - target
        cons = [{'type': 'eq', 'fun': sum_con}]
        # inequality: overlap(k) <= best_c5 for all k != 0
        dx = 2.0 / n
        for k in range(-n + 1, n):
            if k == 0:
                continue
            def make_con(k):
                def con(x):
                    h = x
                    if k >= 0:
                        ov = np.dot(h[:n - k], 1.0 - h[k:])
                    else:
                        kk = -k
                        ov = np.dot(h[kk:], 1.0 - h[:n - kk])
                    return best_c5 - ov * dx
                return con
            cons.append({'type': 'ineq', 'fun': make_con(k)})
        bounds = [(0.0, 1.0)] * n
        try:
            res = optimize.minimize(
                fun=lambda x: 0.0,
                x0=best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
            if res.success:
                h_pol = project_to_box_sum(np.clip(res.x, 0.0, 1.0), target)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_h, best_c5 = h_pol, c5_pol
        except Exception:
            pass

    # final safety projection
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
