import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    # Euclidean projection to box & sum constraint
    n = v.size
    # Simple if s <= lo*n or >= hi*n
    if s <= lo*n:
        return np.full_like(v, lo)
    if s >= hi*n:
        return np.full_like(v, hi)
    # Bisection on lambda to achieve sum constraint after clipping
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # Return binary sequence 0/1 of length n (not normalized)
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_balanced_binary(n, rng):
    # n must be even
    h = np.zeros(n, dtype=float)
    indices = rng.choice(n, size=n//2, replace=False)
    h[indices] = 1.0

def hill_climb_binary(start_h, time_left, rng, max_iters=None):

    n = h.size

    ones = np.flatnonzero(h > 0.5)

        if accept:

            # update ones/zeros lists efficiently: swap i and j

def momentum_gradient_descent(h0, time_left, rng, max_iters=None, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_score = compute_c5(h)
    best_h = h.copy()
    momentum = np.zeros_like(h)
    it = 0
    while time_left() > 0.0:
        if max_iters is not None and it >= max_iters:
            break
        # gradient: average over worst shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # get indices where correlation close to max
        # Compare with tolerance to capture potentially multiple shifts
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient for each shift: (1 - h_{i+shift} - h_{i-shift})? earlier code
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n-shift] += (1.0 - h[shift:])
                grad[shift:] += (1.0 - h[:n-shift])
            else:
                sneg = -shift
                grad[sneg:] += (1.0 - h[:n-sneg])
                grad[:n-sneg] += (1.0 - h[sneg:])
        grad /= len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it+1)
        h_new = h - eta * momentum
        # projection to feasible set
        h_new = project_onto_box_sum(h_new, s)

            h = project_onto_box_sum(h, s)
        # early exit
        if best_score <= 0.3808:
            break
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def hill_climb_binary(start_h, time_left, rng, max_iters=None):
    """Simulated‑annealing hill climbing on a binary vector (swap 0↔1)."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    target = 0.3808
    T0 = 0.05
    while time_left() > 0.0:
        if max_iters is not None and it >= max_iters:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]          # perform swap
        new_score = compute_c5(h)
        delta = new_score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            T = max(T0 * (1 - it / (max_iters or 1e9)), 1e-6)
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # keep swapped h
        else:
            h[i], h[j] = h[j], h[i]      # revert
        it += 1
        if best_score <= target:
            break
    return best_h, best_score


def momentum_gradient_descent(h0, time_left, rng, max_iters=None, step0=0.5, beta=0.9):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    target = 0.3808
    while time_left() > 0.0:
        if max_iters is not None and it >= max_iters:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] += 1.0 - h[shift:]
                grad[shift:]    += 1.0 - h[:n - shift]
            else:
                sneg = -shift
                grad[sneg:]     += 1.0 - h[:n - sneg]
                grad[:n - sneg] += 1.0 - h[sneg:]
        grad /= len(shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
        if it % 2000 == 0:
            h += rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h, s)
        if best_score <= target:
            break
    return best_h, best_score


def slsqp_refine(h0, time_left, rng):
    """Short SLSQP refinement of a feasible vector."""
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    max_iter = int(min(200, max(20, time_left() * 10)))
    if max_iter < 5:
        return h0, compute_c5(h0)
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    return h_opt, compute_c5(h_opt)
