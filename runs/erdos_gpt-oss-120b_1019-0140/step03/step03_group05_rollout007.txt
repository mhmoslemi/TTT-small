def gradient_descent(h_start, target_sum, max_iter, time_limit):
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        cur_c5 = max_overlap * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # gradient
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        # update
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                # small random perturbation
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0, iters=60):
    """Project v onto [lo,hi]^n while fixing the sum to target using bisection."""
    low = -np.max(v) - 2.0
    high = (hi - np.min(v)) + 2.0
    for _ in range(iters):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Subgradient of max overlap w.r.t. h for shift k."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, deadline):
    """Projected subgradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    rng = np.random.default_rng()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --- problem dimension ---
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)
    target_sum = n / 2.0

    # --- ensure feasibility of the initial vector ---
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_box_sum(h0, target_sum)

    # --- Differential Evolution parameters ---
    pop_size = 60
    # initialize population
    population = [h0.copy()]
    while len(population) < pop_size:
        h = rng.random(n)
        h = project_to_box_sum(h, target_sum)
        population.append(h)
    population = np.stack(population)  # shape (pop_size, n)
    fitness = np.array([compute_c5(ind) for ind in population])
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_f = fitness[best_idx]

    # time allocation
    safety = 5.0
    de_deadline = start_time + budget_s * 0.70  # about 70% for DE
    gen = 0
    while time.time() < de_deadline - safety:
        gen += 1
        for i in range(pop_size):
            if time.time() >= de_deadline - safety:
                break
            # mutation indices
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            F = rng.uniform(0.5, 0.9)
            CR = rng.uniform(0.7, 1.0)
            mutant = population[a] + F * (population[b] - population[c])
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)
            trial_f = compute_c5(trial)
            if trial_f <= fitness[i]:
                population[i] = trial
                fitness[i] = trial_f
                if trial_f < best_f:
                    best_f = trial_f
                    best_h = trial.copy()
                    if best_f < 0.3808:
                        # good enough, break all loops
                        de_deadline = start_time  # force exit
                        break
        # optional early stop if no improvement for many generations
        if gen > 200 and best_f < 0.3815:
            #continue a bit longer but not endless
            pass

    # --- Local refinement with gradient descent ---
    gd_deadline = start_time + budget_s - safety
    best_h, best_f = gradient_descent(best_h, target_sum, max_iter=8000,
                                      deadline=gd_deadline)

    # final projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
