def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, alpha):
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    # compute logsumexp in stable way
    shifted = overlaps - max_ov
    lse = max_ov + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return lse * (2.0/h.shape[0]) # dx factor

def smooth_gradient(h, alpha):
    n = h.shape[0]
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    # compute weights
    shifted = overlaps - max_ov
    # to avoid overflow, compute exp(alpha * shifted) maybe large negative values but ok.
    exp_val = np.exp(alpha * shifted)
    w = exp_val / np.sum(exp_val)
    # compute weighted gradient using gradient_smooth
    grad = gradient_smooth(h, w)
    return grad * (2.0 / n)  # include dx

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[-1] - overlap * self.dx

import time
import numpy as np
from scipy.optimize import minimize

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_smooth(h, w):
    n = h.shape[0]
    grad = np.zeros_like(h)
    # w length is 2n-1; index offset = n-1
    offset = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            # i: 0..n-1-k
            grad[:n - k] += weight * (1.0 - h[k:])
            # i: k..n-1
            grad[k:] -= weight * h[:n - k]
        else:
            kk = -k
            # i: kk..n-1
            grad[kk:] += weight * (1.0 - h[:n - kk])
            # i: 0..n-1-kk
            grad[:n - kk] -= weight * h[kk:]
    return grad

def smooth_gradient(h, alpha):
    # compute weighted gradient of smooth approximation of max overlap
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    shifted = overlaps - max_ov  # <=0
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    grad = gradient_smooth(h, w)
    n = h.shape[0]
    dx = 2.0 / n
    return grad * dx  # scale by dx to match objective scale

class OverlapConstraint:
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[-1] - overlap * self.dx
