def compute_c5(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]
    return np.max(corr) * (2.0 / n)

def hill_climb(h, time_end, max_iter=None):
    n = h.size
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    best_score = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng()
    it = 0
    while time.time() < time_end and (max_iter is None or it < max_iter):
        it += 1
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # swap indices in lists
        ones[i_idx] = j
        zeros[j_idx] = i
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones[i_idx] = i
            zeros[j_idx] = j
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n-s] = h[s:]
            term1[n-s:] = 0.0
            term2[s:] = h[:n-s]
            term2[:s] = 0.0
        else:
            sneg = -s
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[sneg:] = h[:n-sneg]
            term1[:sneg] = 0.0
            term2[:n-sneg] = h[sneg:]
            term2[n-sneg:] = 0.0
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gd_momentum(h0, step0, beta, time_end):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end:
        it += 1
        corr = compute_correlation(h)
        max_corr = np.max(corr)
        # find shifts with max
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box sum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def legendre_sequence(p):
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        if pow(i, exp, p) == 1:
            seq[i-1] = 1.0
        else:
            seq[i-1] = 0.0
    return seq

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    # quick bounds
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # Initialize lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # bisection
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) <= 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def gd_momentum(h0, step0, beta, time_end):
    h = h0.copy()
    n = h.size
    s = n * 0.5
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end:
        it += 1
        corr = compute_correlation(h)
        max_corr = np.max(corr)
        epsilon = max_corr * 1e-12 if max_corr != 0 else 1e-12
        idxes = np.where(corr >= max_corr - epsilon)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) <= 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_correlation(h):
    """Linear correlation h * (1-h) via FFT (no scaling issues)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    return corr

def compute_c5(h):
    """Maximum overlap C5 for discretized h."""
    n = h.size
    return np.max(compute_correlation(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max‑correlation term for given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            term1 = np.concatenate([h[s:], np.zeros(s, dtype=h.dtype)])
            term2 = np.concatenate([np.zeros(s, dtype=h.dtype), h[: n - s]])
        else:
            s2 = -s
            term1 = np.concatenate([np.zeros(s2, dtype=h.dtype), h[: n - s2]])
            term2 = np.concatenate([h[s2:], np.zeros(s2, dtype=h.dtype)])
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gd_momentum(h0, step0, beta, time_end, max_iter=None):
    """Projected gradient descent with momentum on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    s = n * 0.5
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time.time() < time_end:
        if max_iter is not None and it >= max_iter:
            break
        it += 1
        corr = compute_correlation(h)
        max_corr = np.max(corr)
        eps = max_corr * 1e-12 if max_corr != 0 else 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def hill_climb(h, time_end):
    """Simple binary hill‑climbing by swapping a 1 with a 0."""
    n = h.size
    ones = list(np.flatnonzero(h == 1.0))
    zeros = list(np.flatnonzero(h == 0.0))
    best_score = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng()
    while time.time() < time_end:
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones[i_idx] = j
        zeros[j_idx] = i
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones[i_idx] = i
            zeros[j_idx] = j
    return best_h, best_score

def primes_upto(limit):
    """Return array of primes <= limit."""
    sieve = np.ones(limit + 1, dtype=bool)
    sieve[:2] = False
    for i in range(2, int(limit ** 0.5) + 1):
        if sieve[i]:
            sieve[i * i :: i] = False
    return np.nonzero(sieve)[0]

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p odd prime)."""
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    best_h = None
    best_score = np.inf

    # 1. Use the provided initial construction if available

            h[idx] = 1.0
            sc = compute_c5(h)
            if sc < best_score:

    # 4. Gradient‑descent refinement on the current best
    if best_h is not None and time_left() > 5.0:
