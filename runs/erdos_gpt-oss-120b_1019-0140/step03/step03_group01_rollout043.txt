```python
import cvxpy as cp

def solve_milp_binary(n):
    # n must be even.
    h = cp.Variable(n, boolean=True)
    # t continuous
    t = cp.Variable()
    constraints = []
    # sum h = n/2
    constraints.append(cp.sum(h) == n/2)
    constraints.append(t >= 0)
    # For each shift k = 0..n-1
    for k in range(n):
        # number of pairs = n - k
        if n - k == 0:
            continue
        # y variables for this shift
        y = cp.Variable(n - k, boolean=True)
        # constraints linking y[i] = h[i] AND h[i+k]
        constraints.append(y <= h[:n - k])
        constraints.append(y <= h[k:])
        # y >= h_i + h_{i+k} - 1
        constraints.append(y >= h[:n - k] + h[k:] - 1)
        # sum y >= (n/2)*(1 - t)
        constraints.append(cp.sum(y) >= (n/2)*(1 - t))
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, warm_start=True, max_iters=100000)
    # retrieve values if solved
    if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
        h_opt = np.round(h.value).astype(float)  # already boolean maybe
        t_opt = t.value
        return h_opt, t_opt
    else:
        return None, None
```

    conv = np.fft.irfft(h_fft * y_fft, N)[:m]

import time, numpy as np, scipy.optimize as opt, cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low, lam_high = np.min(v) - hi, np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.size
    m = 2 * n - 1
    N = 1 << ((m - 1).bit_length())
    H = np.fft.rfft(h, N)
    Y = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * Y, N)[:m]
    return conv[::-1]

def compute_c5(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)

def upsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left_func, step0=0.5, beta=0.9, max_iter=500000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left_func() > 0.0:
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.size
    best_h, best_score = h.copy(), compute_c5(h)
    T, T_min, decay = 0.1, 1e-5, 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i, j = rng.choice(ones), rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - best_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            best_score, best_h = new_score, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=800):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = compute_corr_fft(h)
        return t - np.max(corr) * (2.0 / n)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

def milp_binary_optimize(n, time_left_func):
    if time_left_func() < 15:
        return None, None
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    cons = [cp.sum(h) == n / 2,
            t >= 0,
            t <= 1]
    for k in range(n):
        length = n - k
        if length == 0:
            continue
        y = cp.Variable(length, boolean=True)
        cons += [y <= h[:length],
                 y <= h[k:],
                 y >= h[:length] + h[k:] - 1,
                 cp.sum(y) >= (n / 2) * (1 - t)]
    prob = cp.Problem(cp.Minimize(t), cons)
    tl = min(time_left_func() * 0.9, 120)
    try:
        prob.solve(solver=cp.CBC, max_seconds=tl, warm_start=True)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, max_seconds=tl)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        return None, None
    h_opt = np.rint(h.value).astype(float)
    sc = compute_c5(h_opt)
    return h_opt, sc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h, best_score = None, np.inf

    n_small = 64
    candidates = []
    if 'initial_h_values' in globals():

        candidates.append(project_onto_box_sum(downsample(init_full, n_small), n_small / 2.0))
    candidates.append(thue_morse(n_small))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(3):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    for cand in candidates:
        if time_left() < 5:

        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=800)
        if sc_opt < best_score:

        if best_score <= target:

    if time_left() > 20:

        if h_milp is not None and sc_milp < best_score:
            best_score, best_h = sc_milp, h_milp.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    n_large = 256
    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5(best_h)
    if best_h.size == n_small:
        if np.all(np.isin(best_h, [0.0, 1.0])):
            factor = n_large // best_h.size
            h_up = np.repeat(best_h, factor)
        else:
            h_up = upsample(best_h, n_large)
        h_up = project_onto_box_sum(h_up, n_large / 2.0)

                                                            step0=0.5, beta=0.9, max_iter=500000)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()
            if best_score <= target:

            if best_score <= target:
