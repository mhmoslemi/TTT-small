def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project onto [lo,hi] and sum == target."""
    # If sum of clipped values equals target, return clipped.
    # else find lambda using bisection.
    # Implementation
    a = np.clip(v, lo, hi)
    s = a.sum()
    if np.isclose(s, target):
        return a
    # Determine bounds for lambda
    # shift s.t. v + lam is clipped; lambda_low ensures all v+lam <= lo => after clipping sum = lo*n.
    # lambda_low = lo - np.max(v)
    # lambda_high = hi - np.min(v)
    # Use binary search between low and high
    low = lo - np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s < target:
            low = lam
        else:
            high = lam
    return np.clip(v + (low+high)/2.0, lo, hi)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while enforcing sum(v) == target."""
    # Initial clipping
    x = np.clip(v, lo, hi)
    s = x.sum()
    if np.isclose(s, target):
        return x
    # Bounds for the shift λ
    low = lo - np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(60):
        lam = (low + high) * 0.5
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s < target:
            low = lam
        else:
            high = lam
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap (without dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Determine problem size and initial vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_sum(h0, target_sum)
    else:
        n = 200                       # a modestly fine discretization
        target_sum = n / 2.0
        h0 = project_to_box_sum(rng.random(n), target_sum)

    # ---- DE parameters ----
    pop_size = 60
    population = np.empty((pop_size, n), dtype=float)
    population[0] = h0
    for i in range(1, pop_size):
        population[i] = project_to_box_sum(rng.random(n), target_sum)

    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i])

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_f = fitness[best_idx]

    # allocate time for final gradient descent
    final_gd_time = max(5.0, budget_s * 0.2)

    # ---- Differential Evolution main loop ----
    while True:
        now = time.time()
        remaining = start_time + budget_s - now
        if remaining <= final_gd_time + 1.0:
            break

        for i in range(pop_size):
            # choose three distinct individuals
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            F = 0.8 + 0.2 * rng.random()
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_box_sum(mutant, target_sum)

            CR = 0.85 + 0.15 * rng.random()
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, mutant, population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_to_box_sum(trial, target_sum)

            f_trial = compute_c5(trial)
            if f_trial < fitness[i]:
                population[i] = trial
                fitness[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = trial.copy()

    # ---- Final local refinement (gradient descent) ----
    gd_deadline = start_time + budget_s
    h = best_h.copy()
    cur_c5 = best_f
    alpha = 0.2
    min_alpha = 1e-7

    while time.time() < gd_deadline - 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k_shift = max_idx - (n - 1)
        grad = gradient_for_shift(h, k_shift)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)

        if new_c5 < cur_c5 - 1e-12:
            h = new_h
            cur_c5 = new_c5
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.01, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1

    best_h = np.clip(h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
