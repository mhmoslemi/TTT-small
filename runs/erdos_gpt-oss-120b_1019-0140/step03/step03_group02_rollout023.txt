def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    nfft = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(g, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)
    # full correlation length = 2*n-1, corresponds to indices 0..2*n-2
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)
```

def binary_sa(h0, rng, time_left, max_iter=None, target=None):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    if max_iter is None:
        max_iter = int(min(500000, max(1, time_left() * 5000)))
    # initial temperature based on initial score maybe?
    T0 = 0.01  # small because differences may be small
    for it in range(1, max_iter+1):
        if time_left() < 0.01:
            break
        # random swap
        i = rng.integers(n)
        while h[i] != 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 1.0:
            j = rng.integers(n)
        # propose swap
        h[i] = 1.0
        h[j] = 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        # temperature schedule linearly decreasing to 0
        T = T0 * (1 - it / max_iter)
        accept = False
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            accept = True
        if not accept:
            # revert
            h[i] = 0.0
            h[j] = 1.0
        else:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if target is not None and best_score <= target:
                    break
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def full_corr_fft(h):
    """Linear full correlation of h with (1-h) via FFT."""
    n = h.shape[0]
    g = 1.0 - h
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(g, nfft)
    corr = np.fft.ifft(H * np.conj(G)).real
    return corr[: 2 * n - 1]

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return np.max(full_corr_fft(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += (1.0 - h[shift:])
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - h[: n - s])
            grad[: n - s] -= h[s:]
    grad /= len(shifts)
    return grad

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = full_corr_fft(h)
        maxc = np.max(corr)
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing local search on balanced binary vectors."""
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(300000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        i = rng.integers(n)
        while h[i] != 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 1.0:
            j = rng.integers(n)
        h[i] = 1.0
        h[j] = 0.0
        new_score = compute_c5(h)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score, best_h = new_score, h.copy()
                if best_score <= target:
                    break
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_score

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    """Linear up‑sampling of a periodic‑like array."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    n_small, n_mid, n_large = 64, 256, 512
    init_full = None
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        init_full = project_onto_box_sum(init_full, init_full.size / 2.0)

    # ----- Phase 1: small scale -----
    candidates = []
    if init_full is not None:
        if init_full.size != n_small:
            init_small = upsample(init_full, n_small)
        else:
            init_small = init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small).astype(float))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    best_h, best_score = None, np.inf
    for h0 in candidates:
        if time_left() < 5.0:
            break
        h_opt, sc_opt = momentum_grad_descent(
            h0, n_iter=3000, step0=0.5, beta=0.9, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5(best_h)

    # ----- Phase 2: medium resolution -----
    h_mid = upsample(best_h, n_mid)
    h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
    if time_left() > 4.0:
        it_mid = int(min(25000, max(8000, time_left() * 800)))
        h_mid_opt, sc_mid = momentum_grad_descent(
            h_mid, n_iter=it_mid, step0=0.5, beta=0.9, time_left=time_left)
        if sc_mid < best_score:
            best_score, best_h = sc_mid, h_mid_opt.copy()
    else:
        sc_mid = compute_c5(h_mid)
        if sc_mid < best_score:
            best_score, best_h = sc_mid, h_mid.copy()

    # ----- Phase 3: large resolution -----
    if time_left() > 20.0:
        h_large = upsample(best_h, n_large)
        h_large = project_onto_box_sum(h_large, n_large / 2.0)
        if time_left() > 5.0:
            it_large = int(min(40000, max(15000, time_left() * 800)))
            h_large_opt, sc_large = momentum_grad_descent(
                h_large, n_iter=it_large, step0=0.5, beta=0.9, time_left=time_left)
            if sc_large < best_score:
                best_score, best_h = sc_large, h_large_opt.copy()
        else:
            sc_large = compute_c5(h_large)
            if sc_large < best_score:
                best_score, best_h = sc_large, h_large.copy()

        # binary refinement
        if time_left() > 4.0:
            n = best_h.shape[0]
            order = np.argsort(best_h)
            binary = np.zeros(n, dtype=float)
            binary[order[n // 2:]] = 1.0
            bin_h, bin_sc = binary_hill_climb_sa(binary, rng, time_left, target=0.3808)
            if bin_sc < best_score:
                best_score, best_h = bin_sc, bin_h.copy()

        # final continuous polishing
        if time_left() > 2.0:
            h_fin, sc_fin = momentum_grad_descent(
                best_h, n_iter=8000, step0=0.3, beta=0.9, time_left=time_left)
            if sc_fin < best_score:
                best_score, best_h = sc_fin, h_fin.copy()

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
