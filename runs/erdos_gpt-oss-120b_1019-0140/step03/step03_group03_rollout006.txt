def adam_descent(h0, max_iters, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8, rel_tol=0.0, time_check=None):
    # time_check is a function returning True if still have time else break; but it's closure; not allowed. Instead we embed time checking outside; we can compute max_iters based on time before calling.
    # So we ignore time_check; just run max_iters exactly.
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    for it in range(1, max_iters+1):
        grad = compute_gradient_max_shifts(h, rel_tol=rel_tol)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        # step size schedule
        step = step0 / np.sqrt(it)
        h = h - step * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

# ---------- Helper functions (top level) ----------

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Discrete approximation of C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points (periodic on [0,1))."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points (periodic on [0,1))."""
    return upsample(arr, new_n)


def thue_morse(n):
    """Thue‑Morse binary sequence as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _slsqp_obj(x):
    return x[-1]

def _slsqp_eq_sum(x, n):
    return np.sum(x[:n]) - n / 2.0

def _slsqp_ineq_fun(x, n):
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - np.max(corr)

def slsqp_optimize_candidate(h0, max_iter=400):
    """Refine h0 using SLSQP to lower the max‑correlation objective."""
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_sum, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'args': (n,)}
    ]

    res = opt.minimize(
        _slsqp_obj,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt


# ----- Gradient computation -----

def compute_gradient_max_shifts(h, rel_tol=0.0):
    """Gradient of the max‑correlation term (averaged over near‑max shifts)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    if rel_tol > 0.0:
        thresh = max_corr - rel_tol * max_corr
        idxes = np.where(corr >= thresh)[0]
    else:
        idxes = np.where(corr == max_corr)[0]
    shifts = idxes - (n - 1)
    if shifts.size == 0:
        return np.zeros_like(h)
    grad = np.full_like(h, 1.0, dtype=float)
    for sh in shifts:
        if sh >= 0:
            grad[:n - sh] -= h[sh:]
            grad[sh:]   -= h[:n - sh]
        else:
            s = -sh
            grad[s:]    -= h[:n - s]
            grad[:n - s] -= h[s:]
    grad /= shifts.size
    return grad


# ----- Adam optimizer (no closures) -----

def adam_descent(h0, max_iters, step0=0.1, beta1=0.9, beta2=0.999,
                 eps=1e-8, rel_tol=0.0):
    """Adam‑style projected descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1_pow = 1.0
    beta2_pow = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        grad = compute_gradient_max_shifts(h, rel_tol=rel_tol)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = step0 / np.sqrt(it)
        h = h - step * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


# ----- Simulated annealing on binary vectors -----

def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Swap‑based simulated annealing for balanced binary vectors."""
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)

    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score
