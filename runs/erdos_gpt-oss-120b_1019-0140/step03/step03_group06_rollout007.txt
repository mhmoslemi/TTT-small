def weighted_gradient(h, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    # compute corr
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # compute weights
    # compute exponent stability: subtract max to avoid overflow
    max_corr = np.max(corr)
    exp_terms = np.exp(alpha * (corr - max_corr))
    w = exp_terms / np.sum(exp_terms)  # normalized weights, sum to 1
    # compute gradient contributions weighted by w
    grad = np.zeros_like(h)
    # loop over shifts
    idx_offset = n - 1
    # iterate all k
    for k in range(-n+1, n):
        if k == 0:
            continue
        weight = w[k + idx_offset]
        if weight == 0:
            continue
        if k > 0:
            i = np.arange(n - k)
            j = i + k
        else:
            i = np.arange(-k, n)
            j = i + k  # negative shift
        grad[i] += weight * (1.0 - h[j])
        grad[j] -= weight * h[i]
    return grad, np.max(corr)

def refine_by_local_moves(h, cur_val, steps, rng):
    n = len(h)
    dx = 2.0 / n
    for _ in range(steps):
        # compute correlation, find worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute product for that shift: h[i] * (1 - h[i+k])
        if k >= 0:
            i_range = np.arange(n - k)
            prod = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = h[i_range] * (1.0 - h[i_range + k])
        i_local = i_range[np.argmax(prod)]
        # attempt to move some mass from i_local to a j with room
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0:
            break
        j = rng.choice(cand)
        max_transfer = min(h[i_local], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i_local] -= delta
        h[j] += delta
        # evaluate new
        new_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        if new_val < cur_val:
            cur_val = new_val
            # keep changes
        else:
            # revert
            h[i_local] += delta
            h[j] -= delta
    return h, cur_val

import time

def compute_c5(h):
    """Maximum overlap integral (discrete)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Project v onto [0,1] box and adjust to have exact sum=target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def weighted_gradient(h, alpha):
    """Gradient of smoothed max overlap (log-sum-exp) w.r.t. h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    # stabilized weights
    exp_terms = np.exp(alpha * (corr - max_corr))
    w = exp_terms / np.sum(exp_terms)
    grad = np.zeros_like(h)
    offset = n - 1
    for k in range(-n + 1, n):
        weight = w[k + offset]
        if weight == 0:
            continue
        if k > 0:
            i = np.arange(n - k)
            j = i + k
