def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

```
def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s_idx, k in enumerate(range(-(n - 1), n)):
        w = weights[s_idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.005
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick a random zero and a random one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        # propose swap
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

downsample = upsample

def thue_morse(n):
    """Return Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s_idx, k in enumerate(range(-(n - 1), n)):
        w = weights[s_idx]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def smooth_grad_descent(h0, n_iter=2000, step0=0.5,
                        alpha_start=1.0, alpha_end=10.0,
                        beta=0.9, time_left=lambda: 1e9):
    """Gradient descent on a soft‑max approximation of the max‑correlation."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def momentum_grad_descent(h0, n_iter=5000, step0=0.5, beta=0.9, time_left=lambda: 1e9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= maxc - eps)[0]
        shifts = idxes - (n - 1)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] += (1.0 - h[shift:])
                grad[shift:]   -= h[:n - shift]
            else:
                s = -shift
                grad[s:]      += (1.0 - h[:n - s])
                grad[:n - s]  -= h[s:]
        grad /= len(shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    """Simulated‑annealing style local search on balanced binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(500000, max(1, int(time_left() * 4000))))
    T0 = 0.02
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick a random zero and a random one
        i = rng.integers(0, n)
        while h[i] != 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 1.0:
            j = rng.integers(0, n)
        h_new = h.copy()
        h_new[i] = 1.0
        h_new[j] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score, best_h = new_score, h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # grid sizes
    n_small, n_mid, n_large = 64, 128, 256
    target_bound = 0.3808

    # ----- build initial candidates -----
    candidates = []
    if 'initial_h_values' in globals():

        init_small = downsample(init_full, n_small) if init_full.size != n_small else init_full.copy()
        candidates.append(project_onto_box_sum(init_small, n_small / 2.0))
    candidates.append(thue_morse(n_small).astype(float))
    for _ in range(3):
        candidates.append(random_binary_balanced(n_small, rng).astype(float))
    for _ in range(3):
        v = rng.random(n_small)

    best_h = None

    for h0 in candidates:

            break
        iter_small = int(min(5000, max(500, time_left() * 5)))
        h_opt, sc_opt = smooth_grad_descent(

            beta=0.9, time_left=time_left)
        if sc_opt < best_score:

    if best_h is None:
        best_h = project_onto_box_sum(np.full(n_small, 0.5), n_small / 2.0)
        best_score = compute_c5(best_h)

    # ----- Phase 2: upsample to medium and refine -----
    if time_left() > 3.0:
        h_mid = upsample(best_h, n_mid)
        h_mid = project_onto_box_sum(h_mid, n_mid / 2.0)
        iter_mid = int(min(3000, max(300, time_left() * 5)))
        h_mid_opt, sc_mid_opt = smooth_grad_descent(
            h_mid, n_iter=iter_mid,
            step0=0.5, alpha_start=1.0, alpha_end=20.0,

        if sc_mid_opt < best_score:

    # ----- Phase 3: upsample to large and refine -----
    if time_left() > 3.0:
        h_large = upsample(best_h, n_large)

            h_large, n_iter=iter_large,

            beta=0.9, time_left=time_left)
        if sc_large_opt < best_score:
            best_score, best_h = sc_large_opt, h_large_opt.copy()
        # extra momentum refinement if time permits
        if time_left() > 2.0:
            it_mom = int(min(5000, max(200, time_left() * 3)))
            h_mom, sc_mom = momentum_grad_descent(
                best_h, n_iter=it_mom,
                step0=0.5, beta=0.9,

    # ----- Convert to binary (exactly half ones) -----
    n_final = best_h.shape[0]
    binary = np.zeros_like(best_h)
    order = np.argsort(best_h)

    if bin_score < best_score:

    # final projection and score
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, best_h.size
