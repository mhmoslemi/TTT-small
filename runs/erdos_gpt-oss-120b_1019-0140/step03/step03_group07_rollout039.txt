def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def max_overlap_and_shift(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    k = max_idx - (n - 1)
    return max_val, k

def gradient_for_shift(h, k):
    n = len(h)
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] = 1.0 - h[k:]
    else:
        grad[-k:] = 1.0 - h[:n + k]
    return grad

def hill_climb(h_init, time_limit, rng):

    n = len(h)

    # ensure sum = M
    cur = np.sum(h)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def max_overlap_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    val = float(corr[idx])
    shift = idx - (n - 1)
    return val, shift

def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.ones(n, dtype=float)
    if k >= 0:
        grad[: n - k] = 1.0 - h[k:]
    else:
        grad[-k:] = 1.0 - h[: n + k]
    return grad

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    i = 3
    while i * i <= num:
        if num % i == 0:
            return False
        i += 2
    return True

def legendre_sequence(p):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def random_binary_sequence(n, M, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, M, replace=False)] = 1.0
    return h

def hill_climb(h_init, time_limit, rng):
    h = h_init.copy()
    n = h.shape[0]
    M = n // 2
    cur = np.sum(h)
    if abs(cur - M) > 1e-12:
        h += (M - cur) / n
        h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    T0 = max(best_c5, 1.0)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        T = T0 * (1.0 - t_frac)
        move_type = rng.integers(3)
        if move_type == 0:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif move_type == 1:
            max_val, k = max_overlap_and_shift(h)
            grad = gradient_for_shift(h, k)
            ones_idx = np.where(h > 1e-9)[0]
            if ones_idx.size == 0:
                continue
            i = ones_idx[np.argmax(grad[ones_idx])]
            zeros_idx = np.where(h < 1.0 - 1e-9)[0]
            if zeros_idx.size == 0:
                continue
            j = zeros_idx[np.argmin(grad[zeros_idx])]
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 1e-12:
                continue
            delta = max_transfer * rng.random()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer > 1e-12:
                delta = max_transfer * rng.random()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                max_transfer = min(h[j], 1.0 - h[i])
                if max_transfer <= 1e-12:
                    continue
                delta = max_transfer * rng.random()
                h_new = h.copy()
                h_new[j] -= delta
                h_new[i] += delta
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            if best_c5 < 0.3808:
                break
        else:
            if T > 0.0:
                prob = np.exp(-(c5_new - best_c5) / T)
                if rng.random() < prob:
                    h = h_new
    return best_h, best_c5

def optimise_for_n(n, time_budget, rng):
