import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h):
    """Maximum overlap integral for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project vector onto [lo,hi]^n while fixing its sum to target."""
    low = -5.0 - np.max(v)
    high = 5.0 + hi - np.min(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of a single overlap term with respect to h (no dx)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        cur_c5 = max_overlap * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha <= min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- Initial vector and dimension -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 100
        h0 = rng.random(n)
    target_sum = n / 2.0
    h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ----- Early gradient descent (≈20% of budget) -----
    gd_deadline = start_time + budget_s * 0.2
    if time.time() < gd_deadline:
        gd_h, gd_c5 = gradient_descent(best_h, target_sum,
                                       max_iter=4000,
                                       time_limit=gd_deadline)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5

    # ----- Differential Evolution (≈60% of budget) -----
    de_deadline = start_time + budget_s * 0.8
    if time.time() < de_deadline - 5.0:
        def de_obj(x):
            h = project_to_box_sum(np.clip(x, 0.0, 1.0), target_sum)
            return compute_c5(h)

        bounds = [(0.0, 1.0)] * n

        def de_callback(xk, convergence):
            return time.time() >= de_deadline

        # Run DE with a generous maxiter; callback will stop it early.
        result = differential_evolution(de_obj,
                                        bounds,
                                        strategy='best1bin',
                                        maxiter=2000,
                                        popsize=15,
                                        seed=rng,
                                        polish=False,
                                        callback=de_callback,
                                        updating='deferred',
                                        disp=False)
        h_de = project_to_box_sum(np.clip(result.x, 0.0, 1.0), target_sum)
        c5_de = compute_c5(h_de)
        if c5_de < best_c5:
            best_h, best_c5 = h_de, c5_de

    # ----- Final local refinements until timeout -----
    final_deadline = start_time + budget_s - 3.0
    while time.time() < final_deadline:
        if rng.random() < 0.5:
            start_vec = best_h + rng.normal(scale=0.02, size=n)
        else:
            start_vec = rng.random(n)
        start_vec = project_to_box_sum(np.clip(start_vec, 0.0, 1.0), target_sum)
        gd_h, gd_c5 = gradient_descent(start_vec, target_sum,
                                       max_iter=3000,
                                       time_limit=final_deadline)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
            if best_c5 < 0.3808:
                break

    # ----- Final projection and return -----
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), n
