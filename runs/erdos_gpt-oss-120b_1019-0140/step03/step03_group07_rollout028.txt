def solve_binary_mip(n, time_limit):
    m = n // 2
    h = cp.Variable(n, boolean=True)
    z = cp.Variable()
    constraints = [cp.sum(h) == m]
    y_vars = []
    for k in range(1, n):
        n_k = n - k
        y = cp.Variable(n_k, boolean=True)
        y_vars.append(y)
        constraints += [y <= h[:n_k]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:n_k] + h[k:] - 1]
        constraints += [cp.sum(y) >= z]
    constraints += [z >= 0]
    prob = cp.Problem(cp.Maximize(z), constraints)
    try:
        prob.solve(solver=cp.CBC, cbc_params={"seconds": time_limit})
    except Exception:
        # fallback to GLPK
        prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(time_limit*1000)})
    if h.value is None:
        return None, None
    h_opt = np.array(h.value).astype(float)
    # rounding: due to boolean, values are 0 or 1 (or small tolerance)
    h_opt = np.rint(h_opt)
    z_opt = float(z.value)
    return h_opt, z_opt
```

def solve_mip_min_t(n, time_limit):
    import cvxpy as cp
    import numpy as np
    m = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == m, t >= 0]
    M = m
    for k in range(1, n):
        n_k = n - k
        y = cp.Variable(n_k, boolean=True)
        # product linearization
        constraints += [y <= h[:n_k],
                        y <= h[k:],
                        y >= h[:n_k] + h[k:] - 1]
        s_pos = cp.sum(h[:n_k])
        s_neg = M - cp.sum(h[:k])
        # we need sum(y) as expression
        sum_y = cp.sum(y)
        constraints += [s_pos - sum_y <= t,
                        s_neg - sum_y <= t]
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Attempt solve with CBC or GLPK
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(time_limit*1000), 'msg_lev': 'GLP_MSG_OFF'}, verbose=False)
        except Exception:
            return None, None
    if prob.status not in cp.settings.OPTIMAL and prob.status not in cp.settings.FEASIBLE:
        # If not optimal, still may have solution values
        pass
    if h.value is None or t.value is None:
        return None, None
    h_opt = np.rint(np.array(h.value).astype(float))
    t_opt = float(t.value)
    return h_opt, t_opt
```

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def solve_mip_min_t(n, time_limit):
    m = n // 2
    M = m
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == M, t >= 0]
    for k in range(1, n):
        nk = n - k
        y = cp.Variable(nk, boolean=True)
        # product linearization
        constraints += [y <= h[:nk],
                        y <= h[k:],
                        y >= h[:nk] + h[k:] - 1]
        s_pos = cp.sum(h[:nk])
        s_neg = M - cp.sum(h[:k])
        sum_y = cp.sum(y)
        constraints += [s_pos - sum_y <= t,
                        s_neg - sum_y <= t]
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Try CBC with time limit, fallback to GLPK_MI
    try:
        # CBC supports max_seconds argument
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI,
                       glpk={'tm_lim': int(time_limit * 1000), 'msg_lev': 'GLP_MSG_OFF'},
                       verbose=False)
        except Exception:
            return None, None
    # Retrieve values
    if h.value is None or t.value is None:
        return None, None
    h_opt = np.rint(np.array(h.value).astype(float))
    t_opt = float(t.value)
    return h_opt, t_opt
