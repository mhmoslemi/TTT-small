def binary_swap_local_search(h, target, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    eps = 1e-12
    while time.time() < deadline - 0.01:
        # compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute contributions array for this shift:
        if k >= 0:
            prod = best_h[:n - k] * (1 - best_h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = best_h[kk:] * (1 - best_h[:n - kk])
            i_range = np.arange(kk, n)
        # positions where contribution is 1 (since binary)
        contrib_idx = i_range[prod > 0.5]
        if contrib_idx.size == 0:
            # no contributions? break
            break
        i = rng.choice(contrib_idx)
        # Find a zero index to swap with, not equal to i
        zeros = np.where(best_h == 0)[0]

            best_h = new_h
            best_val = new_val
            if best_val < 0.3808:

def continuous_targeted_anneal(h, target, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    eps = 1e-12
    total_time = deadline - time.time()
    # We'll loop while time left
    while time.time() < deadline - 0.01:
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        # Determine candidate source indices
        if k >= 0:
            src_candidates = np.where(cur_h[:n - k] > eps)[0]
            if src_candidates.size == 0:
                break
            i = rng.choice(src_candidates)
            dest = i + k
        else:
            kk = -k
            src_candidates = np.where(cur_h[kk:] > eps)[0]
            if src_candidates.size == 0:
                break
            i = rng.choice(src_candidates) + kk
            dest = i - kk
        # check dest boundary
        if dest < 0 or dest >= n:
            continue
        if cur_h[dest] >= 1 - eps:
            # cannot increase dest, try another source maybe
            continue
        max_transfer = min(cur_h[i], 1 - cur_h[dest])
        if max_transfer <= 0:
            continue
        # transfer fraction
        delta = rng.random() * max_transfer
        # apply
        cur_h[i] -= delta
        cur_h[dest] += delta

        new_val = compute_c5(cur_h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            # accept with probability based on temperature
            remaining = deadline - time.time()
            # Simple linear schedule: T proportional to remaining_time
            T = max(1e-8, remaining / (deadline - start))
            # acceptance prob
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                # revert
                cur_h[i] += delta
                cur_h[dest] -= delta
    return best_h, best_val
```

def enforce_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        # Scale to match target proportionally
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute diff greedily
    attempts = 0
    while abs(diff) > 1e-10 and attempts < 1000:
        if diff > 0:
            idx = np.where(v < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            inc = min(diff, 1.0 - v[idx[0]])
            v[idx[0]] += inc
            diff -= inc
        else:
            idx = np.where(v > 1e-12)[0]
            if len(idx) == 0:
                break
            dec = min(-diff, v[idx[0]])
            v[idx[0]] -= dec
            diff += dec
        attempts += 1
    # final clip and return
    return np.clip(v, 0.0, 1.0)
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target):
    """Clip v to [0,1] and adjust so that sum(v) == target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    # distribute the residual greedily
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = cand[0]
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = cand[0]
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        attempts += 1
    return np.clip(v, 0.0, 1.0)

def random_binary_vector(n, target, rng):
    """Binary vector with exactly target ones (target assumed integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def binary_swap_local_search(h_bin, target, rng, deadline):
    """Improve a binary vector by swapping a 1 with a 0 to lower max overlap."""
    n = h_bin.shape[0]
    best_h = h_bin.copy()
    best_val = compute_c5(best_h)
    while time.time() < deadline - 0.01:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # contributions for this shift
        if k >= 0:
            prod = best_h[: n - k] * (1.0 - best_h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = best_h[kk:] * (1.0 - best_h[: n - kk])
            i_range = np.arange(kk, n)
        contrib_idx = i_range[prod > 0.5]
        if contrib_idx.size == 0:
            break
        i = rng.choice(contrib_idx)
        zeros = np.where(best_h == 0.0)[0]
        if zeros.size == 0:
            break
        j = rng.choice(zeros)
        # perform swap
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = compute_c5(new_h)
        if new_val < best_val - 1e-12:
            best_h = new_h
            best_val = new_val
            if best_val < 0.3808:
                break
    return best_h, best_val

def continuous_targeted_anneal(h, target, start, deadline, rng):
    """Simulated annealing that transfers mass from a source to its shifted partner."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    eps = 1e-12
    total_time = deadline - start
    while time.time() < deadline - 0.01:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # choose source index participating in the worst shift
        if k >= 0:
            src_cand = np.where(cur_h[: n - k] > eps)[0]
            if src_cand.size == 0:
                break
            i = rng.choice(src_cand)
            dest = i + k
        else:
            kk = -k
            src_cand = np.where(cur_h[kk:] > eps)[0]
            if src_cand.size == 0:
                break
            i = rng.choice(src_cand) + kk
            dest = i - kk
        if dest < 0 or dest >= n:
            continue
        if cur_h[dest] >= 1.0 - eps:
            continue
        max_transfer = min(cur_h[i], 1.0 - cur_h[dest])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur_h[i] -= delta
        cur_h[dest] += delta
        new_val = compute_c5(cur_h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                if best_val < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            T = max(1e-8, remaining / total_time) * 0.01
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                cur_h[i] += delta
                cur_h[dest] -= delta
    return best_h, best_val

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = rng.random(n)

    target = n / 2.0
    h0 = enforce_sum(h0, target)

    # Initial best (continuous)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Create a binary starting point from h0
    if n % 2 == 0:
        binary0 = np.zeros(n, dtype=float)
        idx_top = np.argpartition(best_h, -int(target))[-int(target):]
        binary0[idx_top] = 1.0
    else:
        # fallback: round and repair
        binary0 = (best_h > 0.5).astype(float)
        binary0 = enforce_sum(binary0, target)
    binary0_val = compute_c5(binary0)
    if binary0_val < best_val:
        best_h = binary0.copy()
        best_val = binary0_val

    # Keep track of best binary solution
    best_binary_h = binary0.copy()
    best_binary_val = binary0_val

    # Binary swap local search phase
    while time.time() < deadline - 30.0:
        # generate random binary seed
        seed_h = random_binary_vector(n, target, rng)
        per_deadline = time.time() + min(5.0, (deadline - time.time()) * 0.5)
        h_local, val_local = binary_swap_local_search(seed_h, target, rng, per_deadline)
        if val_local < best_binary_val:
            best_binary_h = h_local
            best_binary_val = val_local
            if best_binary_val < 0.3808:
                break

    # Update best_h with best binary result
    if best_binary_val < best_val:
        best_h = best_binary_h.copy()
        best_val = best_binary_val

    # Continuous targeted annealing phase
    if time.time() < deadline - 2.0:
        best_h, best_val = continuous_targeted_anneal(best_h, target, start, deadline, rng)

    # Final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
