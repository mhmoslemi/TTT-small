def binary_anneal(initial_h, target, start, deadline, rng):

    h = initial_h.copy()
    cur_max = compute_max_corr(h)
    best_h = h.copy()
    best_max = cur_max

    n = len(h)

    # Temperature schedule: starting temperature maybe 5.0.
    T0 = 5.0

    # number of iterations limit huge; just loop while time left.

    while time.time() < deadline:
        # compute current temperature
        progress = (time.time() - start) / (deadline - start)
        T = max(1e-6, T0 * (1 - progress))

        # select i with h[i]==1
        i = rng.integers(n)
        if h[i] == 0:
            # find a random one
            ones = np.where(h == 1)[0]
            if len(ones) == 0:
                break
            i = rng.choice(ones)
        # select j with h[j]==0
        j = rng.integers(n)
        if h[j] == 1:
            zeros = np.where(h == 0)[0]
            if len(zeros) == 0:
                break
            j = rng.choice(zeros)

        # perform swap
        h[i] = 0
        h[j] = 1

        new_max = compute_max_corr(h)
        delta = new_max - cur_max

        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                # early exit if bound reached
                if best_max * 2 / n <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0

    return best_h, best_max

import time
import numpy as np
import math

def compute_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return int(np.max(corr))

def compute_c5(h):
    n = h.shape[0]
    return compute_max_corr(h) * (2.0 / n)

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def binary_anneal(initial_h, target, start, deadline, rng):
    h = initial_h.copy()
    n = len(h)
    cur_max = compute_max_corr(h)
    best_h = h.copy()
    best_max = cur_max
    T0 = 5.0
    while time.time() < deadline:
        # Temperature schedule
        progress = (time.time() - start) / (deadline - start)
        T = max(1e-6, T0 * (1 - progress))
        # pick random ones and zeros for swap
        # find i with h[i]==1
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # swap
        h[i] = 0
        h[j] = 1
        new_max = compute_max_corr(h)
        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                # early stop if target reached
                if best_max * 2.0 / n <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, best_max

def continuous_anneal(h, cur_val, best_h, best_val, start, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute product contributions for worst shift to pick bad i
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        # Identify index i in overlapped region with largest contribution
        # If prod all zero, choose random i in region
        if prod.size == 0:
            continue
        i_local = np.argmax(prod)
        i = i_range[i_local]
        # Choose destination j with headroom (h[j] < 1)
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            # fallback: random swap
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    break

            T = max(remaining * 0.001, 1e-12)

                cur_val = new_val
            else:
                # revert
                h[i] += delta
                h[j] -= delta
    return best_h, best_val
