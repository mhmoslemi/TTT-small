def mismatched_indices(h, shift):
    n = h.shape[0]
    start = max(0, -shift)
    end = n - max(0, shift)
    # length overlapped = end-start
    h1 = h[start:end]
    h2 = h[start+shift:end+shift]  # shift may be negative; Python indexing handles
    # A: h1==1 and h2==0
    A = np.where((h1 == 1) & (h2 == 0))[0] + start
    # B: h1==0 and h2==1
    B = np.where((h1 == 0) & (h2 == 1))[0] + start
    return A, B
```

def improve_sequence(h, time_end, target_c5_norm=None):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    start = time.time()
    n = h.size
    while time.time() < time_end:
        cur_score = best_score
        if target_c5_norm is not None and cur_score <= target_c5_norm:
            break
        # compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        # find shift(s) achieving max
        idxes = np.where(corr == max_corr)[0]
        shift = idxes[0] - (n - 1)  # pick first
        A, B = mismatched_indices(best_h, shift)
        if A.size == 0 or B.size == 0:
            # need random swap
            # pick random i with 1 and random j with 0
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            # pick random from A and B
            i = rng.choice(A)
            j = rng.choice(B)
        # try swap
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score
            # continue

            # probability decreasing over time
            elapsed = time.time() - start

            # simple schedule: temp = 0.01 * (remaining / (time_end - start))
            # but we can ignore; just occasional random acceptance at low probability
            if rng.random() < 0.001:
                best_h, best_score = new_h, new_score

    return best_h, best_score

import time, numpy as np

def project_onto_box_sum(v, s, lo=0., hi=1., max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # lambda bounds
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds bracket s
    if f(lam_low) < s:
        lam_low -= (abs(s - f(lam_low)) + 1)
    if f(lam_high) > s:
        lam_high += (abs(s - f(lam_high)) + 1)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse_binary(n):
    # generate via bit parity
    arr = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return arr

def legendre_sequence(p):
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i-1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def mismatched_indices(h, shift):
    n = h.shape[0]
    start = max(0, -shift)
    end = n - max(0, shift)
    if start >= end:
        return np.array([], dtype=int), np.array([], dtype=int)
    h1 = h[start:end]
    h2 = h[start+shift:end+shift]
    A = np.where((h1 == 1) & (h2 == 0))[0] + start
    B = np.where((h1 == 0) & (h2 == 1))[0] + start
    return A, B

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)
    N = 512
    half_N = N // 2
    candidates = []
    # initial_h_values if exists
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if N % init_full.size == 0:
            cand = np.tile(init_full, N // init_full.size)
        else:
            # linear interpolation fallback
            x_old = np.linspace(0.0, 1.0, init_full.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, N, endpoint=False)
            cand = np.interp(x_new, x_old, init_full)
            cand = project_onto_box_sum(cand, half_N)
        cand = np.clip(cand, 0.0, 1.0)
        # ensure binary: if not integer, round to nearest while preserving sum
        if not np.all(np.isin(cand, [0.0, 1.0])):
            # keep as continuous; our optimizer can handle continuous
            pass
        candidates.append(cand)
    # legendre based
    leg = legendre_sequence(257)  # length 256
    leg2 = np.tile(leg, 2)  # length 512
    candidates.append(leg2)
    # thue-morse repeated
    tm = thue_morse_binary(N // 8)  # length 64
    tm2 = np.tile(tm, 8)  # length 512
    candidates.append(tm2)
    # random balanced
    candidates.append(random_binary_balanced(N, rng))
    # Evaluate initial candidates
    best_h = None
    best_score = np.inf
    for cand in candidates:
        # ensure sum half_N (may be off due to tile)
        cand = project_onto_box_sum(cand, half_N)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()
    # Simulated annealing / targeted swap search
    # Temperature schedule
    T0 = best_score * 0.3 + 1e-4  # small baseline
    iteration = 0
    while time_left() > 0.5:
        iteration += 1
        # recompute correlation and worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        # if already good enough, break
        cur_score = max_corr * (2.0 / N)
        if cur_score <= 0.3808:
            best_score = cur_score
            break
        # find all shifts achieving max
        idxes = np.where(corr == max_corr)[0]
        shift = rng.choice(idxes) - (N - 1)
        # propose swap
        A, B = mismatched_indices(best_h, shift)
        if A.size == 0 or B.size == 0:
            # fallback generic swap
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(A)
            j = rng.choice(B)
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score
        else:
            # probabilistic acceptance
            elapsed = time.time() - start
            frac = elapsed / budget_s
            T = T0 * (1.0 - frac)
            if T > 0 and rng.random() < np.exp(-(new_score - best_score) / T):
                best_h, best_score = new_h, new_score
        # occasional random restart if stagnation
        if iteration % 5000 == 0:
            # check improvement over last segment
            # simple heuristic: if no improvement in last 5000 iterations, restart
            # we approximate using best_score not changing
            # Use random new candidate
            rand_cand = random_binary_balanced(N, rng)
            rand_cand = project_onto_box_sum(rand_cand, half_N)
            rand_score = compute_c5(rand_cand)
            if rand_score < best_score:
                best_h, best_score = rand_cand, rand_score
    # final safety projection
    best_h = project_onto_box_sum(best_h, half_N)
    best_score = compute_c5(best_h)
    return best_h, best_score, N

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low -= (abs(s - f(lam_low)) + 1.0)
    if f(lam_high) > s:
        lam_high += (abs(f(lam_high) - s) + 1.0)
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float, count=n)


def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.empty(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq


def mismatched_indices(h, shift):
    """Indices i where h[i]=1,h[i+shift]=0 (A) and h[i]=0,h[i+shift]=1 (B)."""
    n = h.shape[0]
    start = max(0, -shift)
    end = n - max(0, shift)
    if start >= end:
        return np.array([], dtype=int), np.array([], dtype=int)
    h1 = h[start:end]
    h2 = h[start + shift : end + shift]
    A = np.where((h1 == 1) & (h2 == 0))[0] + start
    B = np.where((h1 == 0) & (h2 == 1))[0] + start
    return A, B


def slsqp_improve(h, max_iter=200):
    """Short SLSQP to improve a candidate (continuous version)."""
    n = h.size
    t0 = compute_c5(h)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h2 = x[:n]
        t = x[-1]
        corr = np.correlate(h2, 1.0 - h2, mode="full") * (2.0 / n)
        return t - np.max(corr)

    cons = [{"type": "eq", "fun": eq_sum}, {"type": "ineq", "fun": ineq_fun}]
    try:
        res = opt.minimize(
            lambda x: x[-1],
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h, compute_c5(h)
    if not res.success:
        h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    N = 512
    half_N = N // 2
    candidates = []

    # Use parent if available
    if "initial_h_values" in globals():

        else:

            x_new = np.linspace(0.0, 1.0, N, endpoint=False)
            cand = np.interp(x_new, x_old, init_full)
        candidates.append(cand)

    # Legendre based (length 256 repeated)
    leg = legendre_sequence(257)  # 256 entries, half ones
    candidates.append(np.tile(leg, 2))

    # Thue–Morse repeated
    tm = thue_morse_binary(N // 8)  # length 64

    # Random balanced
    candidates.append(random_binary_balanced(N, rng))

    # Evaluate initial pool
    best_h = None
    best_score = np.inf
    for cand in candidates:
        cand = project_onto_box_sum(cand, half_N)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # Simulated annealing with targeted swaps
    T0 = best_score * 0.3 + 1e-4
    iteration = 0
    while time_left() > 0.5:
        iteration += 1
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_corr = np.max(corr)
        cur_score = max_corr * (2.0 / N)
        if cur_score <= 0.3808:
            best_score = cur_score
            break
        idxes = np.where(corr == max_corr)[0]
        shift = rng.choice(idxes) - (N - 1)

        # 10% chance of a completely random swap
        if rng.random() < 0.1:
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            A, B = mismatched_indices(best_h, shift)
            if A.size == 0 or B.size == 0:
                ones = np.where(best_h == 1)[0]
                zeros = np.where(best_h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    continue
                i = rng.choice(ones)
                j = rng.choice(zeros)
            else:
                i = rng.choice(A)
                j = rng.choice(B)

        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_score = compute_c5(new_h)

        if new_score < best_score:
            best_h, best_score = new_h, new_score
        else:
            elapsed = time.time() - start
            frac = elapsed / budget_s
            T = T0 * (1.0 - frac)
            if T > 0 and rng.random() < np.exp(-(new_score - best_score) / T):
                best_h, best_score = new_h, new_score

        # occasional random restart if stagnating
        if iteration % 5000 == 0:
            rand_cand = random_binary_balanced(N, rng)
            rand_cand = project_onto_box_sum(rand_cand, half_N)
            rand_score = compute_c5(rand_cand)
            if rand_score < best_score:

    # Final short continuous refinement
    h_ref, sc_ref = slsqp_improve(best_h, max_iter=200)
    if sc_ref < best_score:
        best_h, best_score = h_ref, sc_ref

    best_h = project_onto_box_sum(best_h, half_N)
    best_score = compute_c5(best_h)
    return best_h, best_score, N
