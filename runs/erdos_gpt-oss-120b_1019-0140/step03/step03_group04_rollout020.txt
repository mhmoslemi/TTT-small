def random_swap_improve(h, time_end):
    n = h.size
    best_score = compute_c5(h)
    while time.time() < time_end:
        # pick random indices i1 (with h[i1]==1) and i0 (with h[i0]==0)
        i1 = rng.choice(np.where(h > 0.5)[0])
        i0 = rng.choice(np.where(h < 0.5)[0])
        h[i1], h[i0] = h[i0], h[i1]  # swap
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            # keep swap
        else:
            # revert with probability maybe p or always revert
            # we can accept with some small probability to escape local minima
            if rng.random() < 0.001:
                best_score = new_score
            else:
                h[i1], h[i0] = h[i0], h[i1]

def slsqp_optimize_candidate(h0, max_iter=300):
    n = h0.size
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n/2.0
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type':'eq','fun':eq_sum},
            {'type':'ineq','fun':ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        # fallback
        h_opt = project_onto_box_sum(res.x[:n], n/2.0)
        return h_opt, compute_c5(h_opt)
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n/2.0) # to ensure constraints
    return h_opt, compute_c5(h_opt)
```

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while (time_end is None or time.time() < time_end) and it < max_iter:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # max positions (within eps)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (h.size - 1)
        # compute gradient sum over these shifts
        grad = np.zeros_like(h)
        n = h.size
        for shift in shifts:
            if shift >= 0:
                term1 = np.concatenate([h[shift:], np.zeros(shift)])
                term2 = np.concatenate([np.zeros(shift), h[:n-shift]])
            else:
                s = -shift
                term1 = np.concatenate([np.zeros(s), h[:n-s]])
                term2 = np.concatenate([h[s:], np.zeros(s)])
            grad += (1.0 - term1 - term2)
        grad /= len(shifts)
        # update momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        # projection
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score
```

def hill_climb_binary(h0, time_end, rng):
    h = h0.copy()
    n = h.size
    half = n // 2
    # ensure binary (0/1)
    bin_h = np.zeros(n, dtype=float)
    idx = np.argpartition(-h, half-1)[:half]
    bin_h[idx] = 1.0
    cur_score = compute_c5(bin_h)
    best_h = bin_h.copy()
    best_score = cur_score
    while time.time() < time_end:
        # pick random 1 and 0 indices
        ones = np.where(bin_h == 1.0)[0]
        zeros = np.where(bin_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # swap
        bin_h[i1], bin_h[i0] = 0.0, 1.0
        new_score = compute_c5(bin_h)
        if new_score < best_score:
            best_score = new_score
            best_h = bin_h.copy()
        else:
            # revert with probability
            if rng.random() < 0.001:
                cur_score = new_score
            else:
                bin_h[i1], bin_h[i0] = 1.0, 0.0
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # enlarge bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * (hi - lo)
    if f(lam_high) > s:
        lam_high = np.max(v) + 10 * (hi - lo)

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral approximated on the uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue 0/1 sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑overlap term for given shift list."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.concatenate([h[shift:], np.zeros(shift)])
            term2 = np.concatenate([np.zeros(shift), h[:n - shift]])
        else:
            s = -shift
            term1 = np.concatenate([np.zeros(s), h[:n - s]])
            term2 = np.concatenate([h[s:], np.zeros(s)])
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, step0=0.5, beta=0.9, time_end=None, max_iter=200000):
    """Projected GD with momentum on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    eps = 1e-12
    while (time_end is None or time.time() < time_end) and it < max_iter:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it + 1)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def hill_climb_binary(h0, time_end, rng):
    """Swap‑based hill climbing on binary vectors."""
    n = h0.size
    half = n // 2
    idx = np.argpartition(-h0, half - 1)[:half]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    cur_score = compute_c5(bin_h)
    best_h = bin_h.copy()
    best_score = cur_score

    while time.time() < time_end:
        ones = np.where(bin_h == 1.0)[0]
        zeros = np.where(bin_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # swap
        bin_h[i1], bin_h[i0] = 0.0, 1.0
        new_score = compute_c5(bin_h)
        if new_score < best_score:
            best_score = new_score
            best_h = bin_h.copy()
        else:
            # occasional acceptance to escape plateaus
            if rng.random() < 0.001:
                cur_score = new_score
            else:
                bin_h[i1], bin_h[i0] = 1.0, 0.0
    return best_h, best_score
