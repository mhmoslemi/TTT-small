def binary_simulated_annealing(h_init, time_limit, dx):
    h = h_init.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c5 = corr.max() * dx
    # Precompute positions of 1s and 0s
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    # temperature schedule parameters
    T_start = 0.01
    T_end = 1e-5
    max_iter = int(1e7)  # not really used; will break on time
    it = 0
    while time.time() < time_limit:
        it += 1
        # choose random 1 and 0 positions
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # compute delta
        delta = np.zeros(2*n-1, dtype=np.int32)
        # i1 contribution
        delta[n-1-i1 : n-1-i1 + n] -= (1 - h)
        delta[i1 : i1 + n] += h
        # i0 contribution
        delta[n-1-i0 : n-1-i0 + n] += (1 - h)
        delta[i0 : i0 + n] -= h
        new_corr = corr + delta
        new_c5 = new_corr.max() * dx
        # acceptance
        if new_c5 < cur_c5:
            accept = True
        else:
            # temperature decay
            T = T_start * (T_end / T_start) ** (it / 1e6)  # 1e6 as approx max iters
            if rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True
            else:
                accept = False
        if accept:
            # swap h values
            h[i1] = 0
            h[i0] = 1
            # update ones and zeros arrays (could recompute each time costly)
            # Actually we can maintain sets or simply recompute after certain number of swaps.
            # Simpler: recompute after each swap
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            corr = new_corr
            cur_c5 = new_c5
        # optional early exit if cur_c5 <= target
        if cur_c5 <= target_c5:
            break
    return h, cur_c5
```

def binary_simulated_annealing(h_init, time_end, dx, rng, target_c5=0.3808):
    import math, time, numpy as np
    h = h_init.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c5 = corr.max() * dx
    # Both arrays of indices for ones and zeros
    # We will recompute after each accepted move
    # Temperature schedule parameters
    T_start = 0.01
    T_end = 1e-6
    it = 0
    while time.time() < time_end:
        it += 1
        # Choose random 1 and 0
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        # if none, break
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # compute delta
        delta = np.zeros(2*n - 1, dtype=np.int32)
        # i1 contribution (-1)
        delta[n-1-i1 : n-1-i1 + n] -= (1 - h)
        delta[i1 : i1 + n] += h
        # i0 contribution (+1)
        delta[n-1-i0 : n-1-i0 + n] += (1 - h)
        delta[i0 : i0 + n] -= h
        new_corr = corr + delta
        new_c5 = new_corr.max() * dx
        if new_c5 < cur_c5:
            accept = True
        else:
            # temperature decay based on iteration
            T = T_start * (T_end / T_start) ** (it / 1e6)
            # guard T lower bound
            if T < 1e-12:
                T = 1e-12
            # acceptance probability
            if rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True
            else:
                accept = False
        if accept:
            # perform swap
            h[i1] = 0
            h[i0] = 1
            corr = new_corr
            cur_c5 = new_c5
            if cur_c5 <= target_c5:
                break
    return h, cur_c5

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo, dtype=float)
    if s >= hi * n:
        return np.full_like(v, hi, dtype=float)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to change length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse 0/1 sequence of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=int, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=int)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1 if pow(i, exp, p) == 1 else 0
    return seq

def binary_simulated_annealing(h_init, time_end, dx, rng, target_c5=0.3808):
    """Annealing on balanced binary vectors."""
    h = h_init.copy()
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c5 = corr.max() * dx
    T_start, T_end = 0.01, 1e-6
    it = 0
    while time.time() < time_end:
        it += 1
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)

        delta = np.zeros(2 * n - 1, dtype=np.int32)
        # i1: 1->0
        delta[n - 1 - i1 : n - 1 - i1 + n] -= (1 - h)
        delta[i1 : i1 + n] += h
        # i0: 0->1
        delta[n - 1 - i0 : n - 1 - i0 + n] += (1 - h)
        delta[i0 : i0 + n] -= h

        new_corr = corr + delta
        new_c5 = new_corr.max() * dx

        if new_c5 < cur_c5:
            accept = True
        else:
            T = T_start * (T_end / T_start) ** (it / 1e6)
            if T < 1e-12:
                T = 1e-12
            accept = rng.random() < math.exp(-(new_c5 - cur_c5) / T)

        if accept:
            h[i1] = 0
            h[i0] = 1
            corr = new_corr
            cur_c5 = new_c5
            if cur_c5 <= target_c5:
                break
    return h, cur_c5

def compute_gradients_for_shifts_cont(h):
    """Gradient of max‑correlation term (continuous case)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    eps = 1e-12
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[: n - shift] += 1 - h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] += 1 - h[: n - s]
            grad[: n - s] -= h[s:]
    return grad / len(shifts)

def continuous_gradient_descent(h0, time_end):
    """Projected GD with momentum on continuous formulation."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

            3: [3, 2],
            4: [4, 3],
            5: [5, 3],
            6: [6, 5, 4, 1],
            7: [7, 6, 5, 4],
            8: [8, 6, 5, 4],
            9: [9, 5],
            10: [10, 7],
        }
        tap = taps[m]
        state = seed_state
        seq = np.empty(2 ** m - 1, dtype=int)
        for i in range(2 ** m - 1):
            seq[i] = state & 1
            fb = 0
            for t in tap:
                fb ^= (state >> (t - 1)) & 1
            state = (state >> 1) | (fb << (m - 1))
        return seq
    seq_lfsr = lfsr_seq(7)
    if seq_lfsr.size >= n_small:
        h_lfsr = seq_lfsr[:n_small]
    else:
        repeats = n_small // seq_lfsr.size + 1
        h_lfsr = np.tile(seq_lfsr, repeats)[:n_small]
    cur = h_lfsr.sum()
    if cur > n_small // 2:
        diff = cur - n_small // 2
        idx = np.flatnonzero(h_lfsr == 1)
        flip = rng.choice(idx, size=diff, replace=False)
        h_lfsr[flip] = 0
    elif cur < n_small // 2:
        diff = n_small // 2 - cur
        idx = np.flatnonzero(h_lfsr == 0)
        flip = rng.choice(idx, size=diff, replace=False)
        h_lfsr[flip] = 1
    candidates_small.append(h_lfsr.astype(int))

    for _ in range(2):
        h_rand = np.zeros(n_small, dtype=int)
        idx = rng.choice(n_small, size=n_small // 2, replace=False)
        h_rand[idx] = 1
        candidates_small.append(h_rand)

    # --- Phase 1: binary annealing on small scale ---
    best_h_small = None
    best_score_small = np.inf
    phase1_end = start + budget_s * 0.4
    for cand in candidates_small:
        if time.time() >= phase1_end:
            break
        h_opt, sc_opt = binary_simulated_annealing(cand, phase1_end, dx_small, rng, target_c5)
        if sc_opt < best_score_small:
            best_score_small, best_h_small = sc_opt, h_opt.copy()
        if best_score_small <= target_c5:
            break

    if best_h_small is None:
        h_rand = np.zeros(n_small, dtype=int)
        idx = rng.choice(n_small, size=n_small // 2, replace=False)
        h_rand[idx] = 1
        best_h_small = h_rand
        best_score_small = compute_c5(best_h_small.astype(float))

    # --- Upsample to large resolution ---
    h_up = resample(best_h_small.astype(float), n_large)
    h_up = project_onto_box_sum(h_up, n_large / 2.0)

    # --- Phase 2: continuous gradient descent ---
    gd_end = start + budget_s * 0.85
    if gd_end - time.time() > 5.0:
        h_ref, sc_ref = continuous_gradient_descent(h_up, gd_end)
    else:
        h_ref, sc_ref = h_up, compute_c5(h_up)

    if sc_ref < best_score_small:
        best_h, best_score = h_ref, sc_ref
    else:
        best_h, best_score = h_up, compute_c5(h_up)

    # --- Phase 3: binary annealing on large scale if needed ---
    if best_score > target_c5 and time_left() > 10.0:
        h_bin = np.zeros(n_large, dtype=int)
        idx = np.argpartition(-best_h, n_large // 2 - 1)[: n_large // 2]
        h_bin[idx] = 1
        bin_end = start + budget_s - 0.5
        h_bin_opt, sc_bin_opt = binary_simulated_annealing(h_bin, bin_end, dx_large, rng, target_c5)
        if sc_bin_opt < best_score:
            best_h = h_bin_opt.astype(float)
            best_score = sc_bin_opt

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
