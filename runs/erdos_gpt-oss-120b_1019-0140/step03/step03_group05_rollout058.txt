def smoothed_max_and_grad(h, alpha, lambda_sum):
    n = len(h)
    num_shifts = 2*n-2
    overlaps = np.empty(num_shifts, dtype=float)
    grads = np.zeros((num_shifts, n), dtype=float)
    idx = 0
    for k in range(-n+1, n):
        if k == 0: continue
        if k >=0:
            h1 = h[:n-k]
            h2 = h[k:]
            overlap = np.dot(h1, 1 - h2)  # scalar
            # compute gradient
            grad = np.zeros_like(h)
            grad[:n-k] += 1 - h2
            grad[k:] -= h1
        else:
            kk = -k
            h1 = h[kk:]   # corresponds to h_i for i>=kk
            h2 = h[:n-kk] # h_{i-kk}
            overlap = np.dot(h1, 1 - h2)
            grad = np.zeros_like(h)
            grad[kk:] += 1 - h2
            grad[:n-kk] -= h1
        overlaps[idx] = overlap
        grads[idx] = grad
        idx += 1
    # Convert to dx scaling:
    dx = 2.0 / n
    overlaps_dx = overlaps * dx
    # Softmax weights
    max_o = np.max(alpha * overlaps_dx)  # for numerical stability
    exp_vals = np.exp(alpha * overlaps_dx - max_o)  # subtract max
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    # smoothed max:
    smooth_max = (np.log(sum_exp) + max_o) / alpha
    # gradient: sum_k w_k * grad_k * dx
    grad_smooth = np.dot(w, grads) * dx
    # sum constraint penalty
    sum_h = h.sum()
    penalty = lambda_sum * (sum_h - target_sum)
    grad_penalty = lambda_sum * 2 * (sum_h - target_sum) * np.ones_like(h)  # derivative of penalty term lambda*(sum-h)^2 is 2*lambda*(sum-h)
    # Actually we used lambda_sum as weight for squared penalty: lambda * (sum - target)^2
    # Its gradient: 2*lambda*(sum - target)*1

    return obj, total_grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Core utilities ----------
def compute_c5(h):
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto [lo,hi]^n while fixing the sum to target."""
    low = -np.max(v) - 10.0
    high = hi - np.min(v) + 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)


def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (no dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad


def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max‑overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5


def local_swap_opt(h, target_sum, max_swaps, time_limit, rng):
    """Swap small mass between two coordinates while preserving the sum."""
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_swaps):
        if time.time() > time_limit:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        avail = best_h[i]
        space = 1.0 - best_h[j]
        max_move = min(avail, space)
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h)
        if new_c5 + 1e-12 < best_c5:
            best_h, best_c5 = new_h, new_c5
    return best_h, best_c5


# ---------- Constraint helpers for SLSQP ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0


class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overlap = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overlap * self.dx


def objective_t(x):
    """Slack variable objective for SLSQP."""
    return x[-1]


# ---------- Smoothed max objective ----------
class SmoothObjGrad:
    """Callable returning smoothed max value and its gradient."""
    def __init__(self, alpha, lam, target):
        self.alpha = alpha          # smoothing parameter
        self.lam = lam              # sum‑penalty weight
        self.target = target        # desired sum
    def __call__(self, h):
        n = h.shape[0]
        dx = 2.0 / n
        # compute overlaps and gradients for all non‑zero shifts
        overlaps = []
        grads = []
        for k in range(-n + 1, n):
            if k == 0:
                # zero shift handled directly
                overlap = np.dot(h, 1.0 - h)
                grad = 1.0 - 2.0 * h
            elif k > 0:
                h1 = h[:n - k]
                h2 = h[k:]
                overlap = np.dot(h1, 1.0 - h2)
                grad = np.empty_like(h)
                grad[:n - k] = 1.0 - h2
                grad[k:] = -h1
            else:  # k < 0
                kk = -k
                h1 = h[kk:]
                h2 = h[:n - kk]
                overlap = np.dot(h1, 1.0 - h2)
                grad = np.empty_like(h)
                grad[kk:] = 1.0 - h2
                grad[:n - kk] = -h1
            overlaps.append(overlap)
            grads.append(grad)
        overlaps = np.array(overlaps) * dx                     # shape (num_shifts,)
        grads = np.vstack(grads)                               # shape (num_shifts, n)

        # soft‑max weights
        a = self.alpha * overlaps
        a_max = np.max(a)
        exp_vals = np.exp(a - a_max)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp

        # smoothed max and gradient
        smooth_max = (np.log(sum_exp) + a_max) / self.alpha
        grad_smooth = np.dot(w, grads) * dx

        # sum‑penalty
        sum_h = h.sum()
        penalty = self.lam * (sum_h - self.target) ** 2
        grad_penalty = 2.0 * self.lam * (sum_h - self.target) * np.ones_like(h)

        obj = smooth_max + penalty
        grad = grad_smooth + grad_penalty
        return obj, grad


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- Determine dimension and initial feasible vector ----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        best_h = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 80
        target_sum = n / 2.0
        best_h = project_to_box_sum(rng.random(n), target_sum)

    best_c5 = compute_c5(best_h)

    safety = 5.0  # buffer before hard timeout

    # ---- Main stochastic‑optimization loop ----
    while time.time() < start_time + budget_s - safety:
        # random start: perturb current best or fresh
        if rng.random() < 0.6:
            h = best_h + rng.normal(scale=0.02, size=n)
        else:
            h = rng.random(n)
        h = project_to_box_sum(np.clip(h, 0.0, 1.0), target_sum)

        # ---- Smoothed‑max L‑BFGS‑B refinement (progressively tighter alphas) ----
        lam = 200.0
        for alpha in (5.0, 15.0, 30.0, 60.0):
            obj_fun = SmoothObjGrad(alpha=alpha, lam=lam, target=target_sum)
            res = minimize(obj_fun, h, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 40, 'ftol': 1e-9, 'disp': False})
            if not res.success:
                # ignore failures, keep current h
                pass
            h = np.clip(res.x, 0.0, 1.0)
            h = project_to_box_sum(h, target_sum)

        # ---- Gradient‑descent fine tuning ----
        gd_h, gd_c5 = gradient_descent(
            h, target_sum, max_iter=2000,
            time_limit=start_time + budget_s - safety
        )

        # ---- Local swap search to escape shallow traps ----
        gd_h, gd_c5 = local_swap_opt(
            gd_h, target_sum, max_swaps=2000,
            time_limit=start_time + budget_s - safety,
            rng=rng
        )

        if gd_c5 + 1e-12 < best_c5:
            best_h, best_c5 = gd_h, gd_c5
            if best_c5 < 0.3808:
                break

        # ---- Optional SLSQP refinement if time permits ----
        remaining = start_time + budget_s - safety - time.time()
        if remaining > 6.0:
            sum_con = SumConstraint(n)
            overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
            t0 = compute_c5(gd_h)
            x0 = np.concatenate([gd_h, [t0]])
            bounds = [(0.0, 1.0)] * n + [(0.0, None)]
            cons = [{'type': 'eq', 'fun': sum_con}]
            cons += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
            res2 = minimize(
                objective_t, x0, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            )
            if res2.success:
                h_opt = np.clip(res2.x[:n], 0.0, 1.0)
                h_opt = project_to_box_sum(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt + 1e-12 < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    if best_c5 < 0.3808:
                        break

    # ---- Final projection and evaluation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
