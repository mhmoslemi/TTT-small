def adam_optimize(h0, n_iter, lr0, alpha_start, alpha_end, time_left):
    h = h0.copy()
    n = h.size
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        if time_left() < 0.01:
            break
        # schedule alpha
        if n_iter > 1:
            alpha = alpha_start + (alpha_end - alpha_start)*(t-1)/(n_iter-1)
        else:
            alpha = alpha_start
        grad = grad_softmax_corr(h, alpha)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        # learning rate schedule
        lr = lr0/np.sqrt(t)
        h_new = h - lr * m_hat/(np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad
```

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        # pick random zero and one indices
        zero_idx = rng.integers(0, n)
        while h[zero_idx] != 0.0:
            zero_idx = rng.integers(0, n)
        one_idx = rng.integers(0, n)
        while h[one_idx] != 1.0:
            one_idx = rng.integers(0, n)
        # propose swap
        h_new = h.copy()
        h_new[zero_idx] = 1.0
        h_new[one_idx] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_softmax_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(alpha * (corr - maxc))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(h)
    for s, k in enumerate(range(-(n - 1), n)):
        w = weights[s]
        if w == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:]      -= w * h[:n - k]
        else:
            p = -k
            grad[p:]      += w * (1.0 - h[:n - p])
            grad[:n - p]  -= w * h[p:]
    return grad

def thue_morse_seq(m):
    i = np.arange(m)
    bits = np.vectorize(lambda x: bin(x).count('1') & 1)(i)
    return bits.astype(float)

def upsample_to(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def adam_optimize(h0, n_iter, lr0, alpha_start, alpha_end, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter + 1):
        if time_left() < 0.01:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (t - 1) / max(1, n_iter - 1)
        grad = grad_softmax_corr(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb_sa(h0, rng, time_left, target=0.3808):
    h = h0.copy()
    n = h.size
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    max_iters = int(min(200000, max(1, time_left() * 2000)))
    T0 = 0.01
    for it in range(1, max_iters + 1):
        if time_left() < 0.01:
            break
        zi = rng.integers(0, n)
        while h[zi] != 0.0:
            zi = rng.integers(0, n)
        oi = rng.integers(0, n)
        while h[oi] != 1.0:
            oi = rng.integers(0, n)
        h_new = h.copy()
        h_new[zi] = 1.0
        h_new[oi] = 0.0
        new_score = compute_c5(h_new)
        delta = new_score - curr_score
        T = T0 * (1 - it / max_iters)
        accept = False
        if new_score < curr_score:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            h, curr_score = h_new, new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if best_score <= target:
                    break
    return best_h, best_score
