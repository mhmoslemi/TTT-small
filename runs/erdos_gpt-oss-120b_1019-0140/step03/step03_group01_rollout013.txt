def compute_c5_fft(h):
    n = h.shape[0]
    # compute correlation full
    # use double length for linear convolution
    N = n*2
    H = np.fft.fft(h, N)
    G = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(H * np.conj(G)).real
    # shift to proper indexing? Not needed for max because all values are there.

    import numpy as np, time
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf

    # Phase: coarse SLSQP on n_small=128
    n_small = 128
    init_candidates = []
    if 'initial_h_values' in globals():
        # downsample or upsample to n_small and project
        init_full = np.asarray(initial_h_values, dtype=float)
        init_small = upsample(init_full, n_small)
        init_small = project_onto_box_sum(init_small, n_small/2)
        init_candidates.append(init_small)
    # Thue-Morse
    init_candidates.append(thue_morse(n_small))
    # Random balanced binary
    for _ in range(3):
        init_candidates.append(random_binary_balanced(n_small, rng))
    # Random continuous
    for _ in range(3):
        v = rng.random(n_small)
        init_candidates.append(project_onto_box_sum(v, n_small/2))
    # Run SLSQP on each candidate
    for cand in init_candidates:
        if time_left() < 5:
            break
        h_opt, sc_opt = slsqp_optimize_candidate(cand, max_iter=1500)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()
        if best_score <= target:
            break

    # Phase: upscale to larger n (e.g., n_large=512)
    n_large = 512
    if best_h is None:
        best_h = np.full(n_small, 0.5)
        best_h = project_onto_box_sum(best_h, n_small/2)
        best_score = compute_c5(best_h)

    h_up = upsample(best_h, n_large)
    h_up = project_onto_box_sum(h_up, n_large/2)
    sc_up = compute_c5(h_up)
    if sc_up < best_score:

    # Phase: continuous optimization at large n
    if time_left() > 5:
        # Use Adam optimizer
        h_cont, sc_cont = adam_optimize(best_h, time_left, max_iter=200000, alpha=0.2)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()
    if best_score <= target:
        return best_h, best_score, best_h.size

    # Phase: binary threshold
    # Convert current best to binary with half ones
    n = best_h.size
    k = n // 2
    idx = np.argpartition(-best_h, k)[:k]
    h_bin = np.zeros_like(best_h)
    h_bin[idx] = 1.0
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:

        best_h = h_bin.copy()

    # Phase: binary annealing
    if time_left() > 5:

        best_bin_h = best_h.copy()
        best_bin_score = best_score

                best_bin_h = h_annealed.copy()

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def slsqp_optimize_candidate(h0, max_iter=800):
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def adam_optimize(h0, time_left_func, max_iter=150000, alpha=0.2,
                  beta1=0.9, beta2=0.999, eps=1e-8, check_every=5000):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 0
    while t < max_iter and time_left_func() > 0.0:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, s)
        if t % check_every == 0:
            score = compute_c5(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
            if best_score <= 0.3808:
                break
    # final check
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score, best_h = final_score, h.copy()
    return best_h, best_score

                      max_iters=200000, T0=0.5, decay=0.99995, T_min=1e-5):
