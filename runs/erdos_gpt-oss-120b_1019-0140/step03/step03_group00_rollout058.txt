def upsample(arr, new_n):
    m = len(arr)
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            ns = n - s
            grad[:ns] += 1.0 - h[s:]
            grad[s:] -= h[:ns]
        else:
            s2 = -s
            ns = n - s2
            grad[s2:] += 1.0 - h[:ns]
            grad[:ns] -= h[s2:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def grad_descent_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    iter_count = 0
    while iter_count < max_iters:
        if time_left() <= 0:
            break
        # compute correlation
        conv = compute_corr_full(h)
        max_corr = conv.max()
        # find shifts achieving max
        # conv index offset = n-1 for zero shift
        # The shifts range from -(n-1) to (n-1)
        # shift = idx - (n-1)
        # tolerance
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(iter_count + 1)
        h_new = h - eta * momentum

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        iter_count += 1
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    low, high = -1e5, 1e5
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, lo, hi)
        cur = proj.sum()
        if cur > s:
            low = mid
        else:
            high = mid
    mid = (low + high) * 0.5
    return np.clip(v - mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_corr_full(h):
    """Full correlation of h with (1-h) as in np.correlate(h,1-h,'full')."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    g = (1.0 - h)[::-1]
    G = np.fft.rfft(g, n=size)
    conv = np.fft.irfft(H * G, n=size)
    return conv[: 2 * n - 1]

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    conv = compute_corr_full(h)
    return conv.max() * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            ns = n - s
            grad[s:] += 1.0 - h[:ns]
            grad[:ns] -= h[s:]
        else:
            s2 = -s
            ns = n - s2
            grad[:ns] += 1.0 - h[s2:]
            grad[s2:] -= h[:ns]
    if shifts.size:
        grad /= shifts.size
    return grad

def grad_descent_momentum(h0, max_iters, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        conv = compute_corr_full(h)
        max_corr = conv.max()
        idxes = np.where(conv >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial base vector
    if 'initial_h_values' in globals():

        # allocate iterations, respecting remaining time
        max_by_time = int(time_left() / 0.0001) if time_left() > 0 else 0
        est_iters = int(6000 * (n / 64.0))
        est_iters = max(2000, min(est_iters, max_by_time, 50000))

        h_opt, sc_opt = grad_descent_momentum(
            base, max_iters=est_iters, step0=0.5, beta=0.9,
            rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

        # occasional short random restarts

            for _ in range(2):

    n_final = best_h.shape[0]
    if time_left() > 3:
        final_iters = min(40000, int(time_left() / 0.0001))
        best_h, best_score = grad_descent_momentum(

    while time_left() > 2:
        # continuous mass transfer
        i, j = rng.choice(n_final, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 1e-12:
            delta = max_delta * rng.random()
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_onto_box_sum(h_new, n_final / 2.0)
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        # binary swap if already near binary
        if np.all(np.logical_or(best_h < 1e-6, best_h > 1 - 1e-6)):

        if best_score <= 0.3808:

    best_score = compute_c5(best_h)
