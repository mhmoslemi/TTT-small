def swap_descent(h0, n_iter, step_frac, rng, time_left):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(n_iter):
        if time_left() <= 0:
            break
        # compute worst shift indices
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        # compute gradient (averaged)
        grad = compute_gradients_for_shifts(h, shifts)
        # indices where grad>0 and grad<0
        pos_idx = np.where(grad > 1e-6)[0]
        neg_idx = np.where(grad < -1e-6)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            # fallback: random jitter
            h += rng.normal(scale=0.001, size=n)
            h = project_onto_box_sum(h, n/2.0)
            continue
        i = rng.choice(pos_idx)
        j = rng.choice(neg_idx)
        # max delta limited by bounds
        max_delta = min(h[i], 1.0 - h[j]) * step_frac
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # evaluate
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        # accept new h anyway? Might adopt greedy; but keep using current h for next iteration.
        h = h_new
        # occasional projection and jitter
        if (it+1) % 1000 == 0:
            h = project_onto_box_sum(h, n/2.0)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure interval
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    # returns binary sequence 0/1 as float
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n-s]
            term2[:n-s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional jitter
        if it % 1500 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def swap_descent(h0, n_iter, step_frac, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(n_iter):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        pos_idx = np.where(grad > 1e-6)[0]
        neg_idx = np.where(grad < -1e-6)[0]
        if pos_idx.size and neg_idx.size:
            i = rng.choice(pos_idx)
            j = rng.choice(neg_idx)
            max_delta = min(h[i], 1.0 - h[j]) * step_frac
            if max_delta > 0:
                delta = max_delta * rng.random()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score, best_h = sc, h_new.copy()
                h = h_new
        # periodic projection to correct numerical drift
        if (it + 1) % 2000 == 0:
            h = project_onto_box_sum(h, s)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    rng = np.random.default_rng(seed)

    # target resolution
    n_final = 512

    # initial base
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = upsample(base, n_final)
    base = project_onto_box_sum(base, n_final / 2.0)

    # Stage 1: coarse descent on base
    best_h, best_score = gradient_descent_momentum_custom(
        base, n_iter=60000, step0=0.5, beta=0.9, rng=rng, time_left=time_left)

    # Stage 2: random restarts
    while time_left() > 30.0:
        # allocate iterations proportionally to remaining time
        remain = time_left()
        iters = int(min(20000, max(5000, remain * 12)))
        h_rand = random_binary_balanced(n_final, rng)
        h_opt, sc_opt = gradient_descent_momentum_custom(
            h_rand, n_iter=iters, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # Stage 3: final intensive refinement
    if time_left() > 5.0:
        final_iters = int(min(180000, max(40000, time_left() * 30)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=final_iters, step0=0.2, beta=0.9, rng=rng, time_left=time_left)

    # Stage 4: swap-based fine tuning
    if time_left() > 5.0:
        swap_iters = int(min(60000, max(10000, time_left() * 10)))
        best_h, best_score = swap_descent(
            best_h, n_iter=swap_iters, step_frac=0.05, rng=rng, time_left=time_left)

    # final projection and evaluation
    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final

            h_rand, n_iter=iters, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    if time_left() > 5.0:
        final_iters = int(min(180000, max(40000, time_left() * 30)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=final_iters, step0=0.2, beta=0.9, rng=rng, time_left=time_left)

    if time_left() > 5.0:
        swap_iters = int(min(60000, max(10000, time_left() * 10)))
        best_h, best_score = swap_descent(
            best_h, n_iter=swap_iters, step_frac=0.05, rng=rng, time_left=time_left)

    if time_left() > 2.0:
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=20000, step0=0.05, beta=0.9, rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, n_final
