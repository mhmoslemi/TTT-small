def find_worst_pair(h):
    n = h.shape[0]
    best_corr = -np.inf
    best_k = 0
    best_i = 0
    for k in range(-n+1, n):
        if k == 0:
            continue
        if k > 0:
            i_range = np.arange(n - k)
            prod = h[i_range] * (1.0 - h[i_range + k])
        else:
            kk = -k
            i_range = np.arange(kk, n)
            prod = h[i_range] * (1.0 - h[i_range - kk])  # because shift left
        max_local = prod.max()
        if max_local > best_corr:
            best_corr = max_local
            best_k = k
            # get index of max
            i_local = prod.argmax()
            best_i = i_range[i_local]
    return best_k, best_i
```

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v
```

            # partner = i + k
            # candidate indices for j where h[j] < 1 and (1 - h[j+k]) minimal
            max_j = n - k
            idxs = np.arange(max_j)
            mask = h[idxs] < 1.0 - 1e-12
            if np.any(mask):
                contrib = 1.0 - h[idxs + k]
                j = idxs[mask][np.argmin(contrib[mask])]
            else:
                cand = np.where(h < 1.0 - 1e-12)[0]
                if len(cand) == 0:
                    break
                j = rng.choice(cand)
        else:
            kk = -k
            max_j = n - kk
            idxs = np.arange(kk, n)
            mask = h[idxs] < 1.0 - 1e-12
            if np.any(mask):
                contrib = 1.0 - h[idxs - kk]
                j = idxs[mask][np.argmin(contrib[mask])]
            else:
                cand = np.where(h < 1.0 - 1e-12)[0]
                if len(cand) == 0:
                    break
                j = rng.choice(cand)
        # Transfer
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            # fallback random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)

                best_h = h.copy()

                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_feasible(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def blend_crossover(p1, p2, alpha, rng):

    return child
