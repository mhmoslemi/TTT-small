def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)
```

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left_func, max_iters=200000):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    T = 0.5
    decay = 0.9999
    it = 0
    while it < max_iters and time_left_func() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(1e-5, T * decay)
        it += 1
    return best_h, best_score
```

    # Stage 1: MILP for n=64 binary
    if time_left() > 30:
        n = 64
        # Build MILP
        a = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True)
        constraints = [cp.sum(a) == n//2, t >= 0, t <= n//2]
        for k in range(1, n):
            length = n - k
            w = cp.Variable(length, boolean=True)
            idx = np.arange(length)
            constraints.append(w <= a[idx])
            constraints.append(w <= a[idx + k])
            constraints.append(w >= a[idx] + a[idx + k] - 1)
            constraints.append(cp.sum(w) >= t)
        prob = cp.Problem(cp.Maximize(t), constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK_MI, verbose=False)
            except Exception:
                prob = None
        if prob is not None and prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
            a_val = np.rint(a.value).astype(float)
            # ensure exact sum
            sum_a = int(a_val.sum())
            if sum_a != n//2:
                # adjust using fractional values
                # compute residuals = a.value - a_val (but a_val is round)
                residual = a.value - a_val
                diff = n//2 - sum_a
                if diff > 0:
                    # need to flip zeros to ones with largest residual (close to 1)
                    zero_idx = np.where(a_val == 0)[0]
                    if zero_idx.size:
                        order = zero_idx[np.argsort(-residual[zero_idx])]
                        a_val[order[:diff]] = 1.0
                elif diff < 0:
                    # flip ones to zeros with smallest residual (close to 0)
                    one_idx = np.where(a_val == 1)[0]
                    if one_idx.size:
                        order = one_idx[np.argsort(residual[one_idx])]
                        a_val[order[: -diff]] = 0.0
            score = compute_c5(a_val)
            best_h = a_val
            best_score = score
            if best_score <= target:
                return best_h, best_score, best_h.size

    # Stage 2: fallback heuristic
    # Use larger n if we have time
    n_fallback = 256
    # Build initial h
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_fallback:
            h0 = upsample(init_full, n_fallback)
        else:
            h0 = init_full.copy()

        h0 = project_onto_box_sum(h0, n_fallback/2.0)
    else:
        # start from random balanced binary
        h0 = random_binary_balanced(n_fallback, rng)
    # Convert to binary balanced (if not)
    # Already balanced due to project_onto_box_sum
    # Run binary annealing
    if time_left() > 5:
        # ensure binary
        h_bin = (h0 >= 0.5).astype(float)
        # rebalance if needed
        diff = int(h_bin.sum() - n_fallback/2)
        if diff != 0:
            # adjust
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if diff > 0:
                # too many ones, flip some to zero
                idxs = rng.choice(ones, size=diff, replace=False)
                h_bin[idxs] = 0.0
            else:
                idxs = rng.choice(zeros, size=-diff, replace=False)
                h_bin[idxs] = 1.0
        h_bin_ref, sc_bin_ref = binary_swap_anneal(h_bin, rng, time_left, max_iters=300000)
        if sc_bin_ref < best_score:
            best_h, best_score = h_bin_ref, sc_bin_ref
            if best_score <= target:
                return best_h, best_score, best_h.size

    # final guard projection
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size/2.0)
        best_score = compute_c5(best_h)
        return best_h, best_score, best_h.size
    else:
        # fallback: return uniform
        n = 64
        h = np.full(n, 0.5)
        h = project_onto_box_sum(h, n/2.0)
        return h, compute_c5(h), n

import time, numpy as np, scipy.optimize as opt, cvxpy as cp

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for sh in shifts:
        if sh >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - sh] = h[sh:]
            term2[sh:] = h[:n - sh]
        else:
            s = -sh
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_with_time(h0, time_left, step0=0.5, beta=0.9, max_iter=300000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = corr.max()
        shifts = np.where(corr >= maxc - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * mom
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=400000):
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h, best_sc = h.copy(), cur
    T, decay = 0.5, 0.9999
    i = 0
    while i < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i1, i0 = rng.choice(ones), rng.choice(zeros)
        h[i1], h[i0] = 0.0, 1.0
        new = compute_c5(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_sc:
                best_sc, best_h = new, h.copy()
        else:
            h[i1], h[i0] = 1.0, 0.0
        T = max(1e-5, T * decay)
        i += 1
    return best_h, best_sc

def slsqp_optimize_candidate(h0, max_iter=2000):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    if not res.success:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

        if best_score <= target:

    # ---- final projection guard ----

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
