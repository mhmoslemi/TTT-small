def compute_c5_fft(h):
    n = h.size
    # Use real FFT
    L = 1 << ((2*n - 1).bit_length())
    # Actually should be next power of 2 >= 2n
    L = 2 ** int(np.ceil(np.log2(2*n)))
    # Pad arrays to length L
    h_pad = np.zeros(L, dtype=float)
    one_minus_h_pad = np.zeros(L, dtype=float)
    h_pad[:n] = h
    one_minus_h_pad[:n] = 1.0 - h
    # Compute convolution via FFT: corr = ifft(fft(h) * fft(1-h)).real
    H = np.fft.rfft(h_pad)
    O = np.fft.rfft(one_minus_h_pad)
    corr = np.fft.irfft(H * O, n=L)[:2*n-1]
    # C5 = max(corr) * (2.0 / n)
    return np.max(corr) * (2.0 / n)

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    h_pad = np.zeros(L)
    one_minus_h_pad = np.zeros(L)
    h_pad[:n] = h
    one_minus_h_pad[:n] = 1.0 - h
    H = np.fft.rfft(h_pad)
    O = np.fft.rfft(one_minus_h_pad)
    corr = np.fft.irfft(H * O, n=L)[:2*n-1]
    return corr

def grad_of_max_shift(h, worst_idxs):

            # subtract h_{i-shift}
            grad[shift:] -= h[:n - shift]
        else:
            t = -shift
            grad[t:] -= h[:n - t]
            grad[:n - t] -= h[t:]
    # average
    grad /= len(worst_idxs)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_corr_fft(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n-1
    h_pad = np.zeros(L, dtype=np.float64)
    oh_pad = np.zeros(L, dtype=np.float64)
    h_pad[:n] = h
    oh_pad[:n] = 1.0 - h
    H = np.fft.rfft(h_pad)
    O = np.fft.rfft(oh_pad)
    corr = np.fft.irfft(H * O, n=L)[:2 * n - 1]
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)


def grad_of_max_shift(h, worst_idxs):
    """Gradient of the max‐overlap objective for given worst shift indices."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for idx in worst_idxs:
        shift = idx - (n - 1)
        grad += 1.0
        if shift >= 0:
            grad[:n - shift] -= h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            t = -shift
            grad[t:] -= h[:n - t]
            grad[:n - t] -= h[t:]
    grad /= len(worst_idxs)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        worst_idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        grad = grad_of_max_shift(h, worst_idxs)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        # occasional jitter to escape flats
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)

        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def simulated_annealing(h_start, n_iter, temp_start, temp_end, rng, time_left):
    """Simple SA that moves mass between two positions."""
    h = h_start.copy()
    n = h.shape[0]
    cur_score = compute_c5_fft(h)
    best_h, best_score = h.copy(), cur_score

    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        T = temp_start + (temp_end - temp_start) * (t / n_iter)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5_fft(h_new)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_score = h_new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h_new.copy()
        if t % 5000 == 0:  # occasional global jitter
            h_j = h + rng.normal(scale=0.02, size=n)
            h_j = project_onto_box_sum(h_j, n / 2.0)
            score_j = compute_c5_fft(h_j)
            if score_j < best_score:
                best_score, best_h = score_j, h_j.copy()
            if score_j < cur_score:
                h, cur_score = h_j, score_j
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Choose resolution based on available time
    if time_left() > 600:
        n_target = 1024
    else:
        n_target = 512

    # Base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    if base.shape[0] != n_target:
        base = upsample(base, n_target)
    base = project_onto_box_sum(base, n_target / 2.0)

    # Build a small pool of candidates
    candidates = [base]
    for _ in range(4):
        h = base + rng.normal(scale=0.08, size=n_target)
        h = project_onto_box_sum(h, n_target / 2.0)
        candidates.append(h)
    for _ in range(2):
        h = random_binary_balanced(n_target, rng).astype(float)
        h = project_onto_box_sum(h, n_target / 2.0)
        candidates.append(h)

    best_h = None
    best_score = np.inf

    # Phase 1: momentum gradient descent from each candidate
    for h0 in candidates:
        if time_left() < 10:
            break
        # allocate iterations proportional to remaining time
        iter_est = int(min(20000, max(4000, time_left() * (2000.0 / n_target))))
        h_opt, sc_opt = gradient_descent_momentum_custom(
            h0, n_iter=iter_est, step0=0.4, beta=0.9, rng=rng, time_left=time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
            if best_score <= 0.3808:
                break

    # Phase 2: simulated annealing from the best found point
    if best_h is not None and time_left() > 5:
        anneal_iters = int(min(50000, max(20000, time_left() * 30)))
        h_ann, sc_ann = simulated_annealing(best_h, anneal_iters,
                                            temp_start=0.2, temp_end=1e-4,
                                            rng=rng, time_left=time_left)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    # Phase 3: final refinement
    if best_h is not None and time_left() > 5:
        final_iters = int(min(40000, max(10000, time_left() * 20)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=final_iters, step0=0.2, beta=0.9,
            rng=rng, time_left=time_left)

    # Safety final projection and evaluation
    if best_h is None:
        best_h = base.copy()
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
