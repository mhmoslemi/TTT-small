    best_h = h.copy()

        if time.time() >= deadline - 0.5:
            break

        corr = np.correlate(h, 1.0-h, mode='full')
        overlaps = corr * dx

        max_ov = np.max(overlaps)
        exps = np.exp((overlaps - max_ov) / epsilon)
        sum_exps = np.sum(exps)
        w = exps / sum_exps  # shape (2*n-1)
        # gradient init
        grad = np.zeros(n)
        # loop k
        for k in range(-n+1, n):

                b = a + k

                # We'll handle negative k separately below.
                # continue to next else part.

                grad[a] += weight * (1.0 - h[b]) * dx
                grad[b] -= weight * h[a] * dx
            else:

                grad[b] += weight * (1.0 - h[a]) * dx
                grad[a] -= weight * h[b] * dx

import time
import numpy as np

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Adjust v∈[0,1] so that sum(v) == target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    """Random vector in [0,1]^n with exact sum==target."""
    v = rng.random(n)
    v = enforce_sum(v, target, rng)
    return v

# ---------- gradient descent on smooth max ----------
def smooth_descent(h, target, rng, deadline,
                  start_eps=0.05, lr=0.2, max_iter=2000, eps_decay=0.9995):
    """Gradient descent minimizing a smooth approximation of max overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    eps = start_eps
    best_h = h.copy()
    best_val = compute_c5(best_h)

    for it in range(max_iter):
        if time.time() >= deadline - 0.5:
            break

        # overlaps for all shifts
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        max_ov = np.max(overlaps)

        # softmax weights (stable)
        exps = np.exp((overlaps - max_ov) / eps)
        w = exps / np.sum(exps)

        # gradient of smooth max (vectorized over shifts)
        grad = np.zeros(n)
        for k in range(-n + 1, n):
            if k == 0:
                continue
            weight = w[k + n - 1]
            if weight == 0.0:
                continue
            if k > 0:
                a = np.arange(n - k)
                b = a + k
                grad[a] += weight * (1.0 - h[b]) * dx
                grad[b] -= weight * h[a] * dx
            else:
                kk = -k
                a = np.arange(n - kk)
                b = a + kk
                grad[b] += weight * (1.0 - h[a]) * dx
                grad[a] -= weight * h[b] * dx

        # step and projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)

        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            if best_val < 0.3808:
                h = h_new
                break
        h = h_new

        eps = max(eps * eps_decay, 0.001)
        lr *= 0.9995
    return best_h, best_val

# ---------- simple simulated annealing for final polishing ----------
def continuous_anneal(h, cur_val, best_h, best_val, start_time, deadline, rng):
    n = h.shape[0]
    dx = 2.0 / n
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)

        i_local = np.argmax(prod)
        i = i_range[i_local]

        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            T = max((deadline - time.time()) * 0.001, 1e-12)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Choose a moderately fine discretisation
    n = 120               # even number, 2 sec interval → dx = 0.01666…
    target = n / 2.0

    # Initialise from parent if available, otherwise random
    if "initial_h_values" in globals():
        small = np.array(initial_h_values, dtype=float)
        n0 = small.shape[0]
        x_small = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new   = np.linspace(0.0, 2.0, n,  endpoint=False)
        h0 = np.interp(x_new, x_small, small)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
    else:
        h0 = random_continuous_vector(n, target, rng)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Multi‑restart gradient descent
    restart_limit = 8
    restarts = 0
    while time.time() < deadline - 5.0 and restarts < restart_limit:
        if rng.random() < 0.4:
            h_start = random_continuous_vector(n, target, rng)
        else:
            h_start = best_h.copy()
            for _ in range(3):
                src = np.where(h_start > 1e-8)[0]
                dst = np.where(h_start < 1.0 - 1e-8)[0]
                if len(src) == 0 or len(dst) == 0:
                    break
                i = rng.choice(src)
                j = rng.choice(dst)
                max_delta = min(h_start[i], 1.0 - h_start[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                h_start[i] -= delta
                h_start[j] += delta
            h_start = enforce_sum(h_start, target, rng)

        h_opt, val_opt = smooth_descent(
            h_start, target, rng, deadline,
            start_eps=0.05, lr=0.2, max_iter=2000, eps_decay=0.9995,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val < 0.3808:
                break
        restarts += 1

    # Final annealing refinement
    if time.time() < deadline - 0.5:
        best_h, best_val = continuous_anneal(
            best_h, best_val, best_h, best_val, start, deadline, rng
        )

    # Ensure feasibility before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

    return best_h.astype(float), final_val, n
