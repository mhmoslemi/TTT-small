def simulated_annealing(h0, time_limit, rng):
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = h0.copy()
    cur_score = best_score
    start = time.time()
    n = len(h0)
    # Precompute sum constraint M = n/2
    while time.time() - start < time_limit:
        # propose move
        if rng.random() < 0.5:
            # binary swap move (if h is binary)
            zeros = np.where(cur_h < 0.5)[0]  # treat near 0 as zero
            ones = np.where(cur_h > 0.5)[0]
            if zeros.size == 0 or ones.size == 0:
                continue
            i = rng.choice(zeros)
            j = rng.choice(ones)
            new_h = cur_h.copy()
            # swap values (0<->1)
            new_h[i] = 1.0
            new_h[j] = 0.0
        else:
            # fractional transfer move
            i, j = rng.choice(n, 2, replace=False)
            max_inc = min(1.0 - cur_h[i], cur_h[j])
            if max_inc <= 0:
                continue
            delta = rng.random() * max_inc
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_score = compute_c5(new_h)
        # acceptance
        if new_score < cur_score:
            accept = True
        else:
            # temperature: schedule based on time fraction
            t_elapsed = time.time() - start
            fraction = t_elapsed / time_limit
            T = max(1e-6, (1.0 - fraction))  # maybe linear from 1 to 0
            # scale acceptance probability by difference (new - cur) / T
            # we need a scale; objective difference typical ~0.001, so T ~0.001? Let's use T_start=0.01.
            T_start = 0.01
            T = T_start * (1 - fraction) + 1e-8
            prob = math.exp(-(new_score - cur_score) / T)
            accept = rng.random() < prob
        if accept:
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
                # early exit if below target
                if best_score < 0.3808:
                    break
    return best_h, best_score

def legendre_binary(n, rng):

    # simple primality test
    if p < 2:
        return None
    for i in range(2, int(p**0.5) + 1):
        if p % i == 0:
            return None
    # generate legendre sequence of length p
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    # include position 0 maybe as 0 (not residue)
    # we need M = n/2 ones
    M = n // 2
    cur = int(seq.sum())
    # adjust by adding extra ones if needed
    # pad with extra element to reach length n
    seq = np.concatenate([seq, np.zeros(1)])
    # Now length n
    cur = int(seq.sum())
    if cur < M:
        # add ones to first positions not already 1
        zeros = np.where(seq == 0)[0]
        need = M - cur
        if need <= zeros.size:
            idxs = rng.choice(zeros, need, replace=False)
            seq[idxs] = 1.0
        else:

        if need <= ones.size:

            seq[idxs] = 0.0
        else:
            seq[:] = 0.0
    # shuffle
    seq = np.roll(seq, rng.integers(n))
    return seq

import time
import math
import numpy as np

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_binary(n, rng):
    """Return a binary sequence of length n based on a Legendre symbol (if n−1 is prime)."""
    p = n - 1
    if p < 2:
        return None
    # simple primality test
    for i in range(2, int(p**0.5) + 1):
        if p % i == 0:
            return None
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for quadratic residue, p‑1 otherwise
        seq[i] = 1.0 if leg == 1 else 0.0
    # seq[0] stays 0, giving M‑1 ones.
    h = np.concatenate([seq, np.array([0.0])])  # length n
    M = n // 2
    cur = int(h.sum())
    if cur < M:
        need = M - cur
        zero_idxs = np.where(h == 0)[0]
        if need <= zero_idxs.size:
            add = rng.choice(zero_idxs, need, replace=False)
            h[add] = 1.0
    elif cur > M:
        need = cur - M
        one_idxs = np.where(h == 1)[0]
        if need <= one_idxs.size:
            rem = rng.choice(one_idxs, need, replace=False)
            h[rem] = 0.0
    # random rotation to vary boundary effects
    h = np.roll(h, rng.integers(n))
    return h

def simulated_annealing(h0, time_limit, rng):
    """Simple SA that preserves the sum of h exactly."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    start = time.time()
    T0 = 0.02  # initial temperature
    while time.time() - start < time_limit:
        # propose move
        if rng.random() < 0.5:
            # swap two entries (preserves sum exactly)
            i, j = rng.choice(n, 2, replace=False)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
        else:
            # fractional transfer from j to i (also preserves sum)
            i, j = rng.choice(n, 2, replace=False)
            max_inc = min(1.0 - cur_h[i], cur_h[j])
            if max_inc <= 0.0:
                continue
            delta = rng.random() * max_inc
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_score = compute_c5(new_h)
        if new_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start
            fraction = elapsed / time_limit
            T = T0 * (1.0 - fraction) + 1e-12
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)
        if accept:
            cur_h = new_h
            cur_score = new_score
            if cur_score + 1e-12 < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
                if best_score < 0.3808:
                    break
    return best_h, best_score
