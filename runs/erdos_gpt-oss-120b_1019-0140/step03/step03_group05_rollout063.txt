def smoothmax_overlap(h, alpha):
    n = h.shape[0]; dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    # compute softmax weights
    max_O = np.max(alpha * O)
    exp_terms = np.exp(alpha * O - max_O)
    w = exp_terms / np.sum(exp_terms)
    # compute weighted gradient
    grad = np.zeros_like(h)
    # compute for each shift
    for idx, k in enumerate(range(-(n-1), n)):
        weight = w[idx]
        if weight == 0: continue
        # gradient_for_shift
        if k >= 0:
            grad[:n - k] += weight * (1.0 - h[k:])
            grad[k:] -= weight * h[:n - k]
        else:
            kk = -k
            grad[kk:] += weight * (1.0 - h[:n - kk])
            grad[:n - kk] -= weight * h[kk:]
    return np.dot(w, O), grad  # f = weighted sum (i.e., smoothmax)
```

def smoothmax_and_grad(h, alpha):
    n = len(h); dx = 2.0/n
    O = np.correlate(h, 1.0 - h, mode='full') * dx
    # compute softmax
    aO = alpha * O
    max_aO = np.max(aO)
    expa = np.exp(aO - max_aO)
    w = expa / np.sum(expa)
    # gradient
    grad = np.zeros_like(h)
    # iterate over k
    # idx 0 corresponds to shift -(n-1)
    for idx, k in enumerate(range(-(n-1), n)):
        weight = w[idx]
        if weight == 0.0:
            continue
        if k >= 0:
            grad[:n - k] += weight * (1.0 - h[k:])
            grad[k:]   -= weight * h[:n - k]
        else:
            kk = -k
            grad[kk:] += weight * (1.0 - h[:n - kk])
            grad[:n - kk] -= weight * h[kk:]
    grad *= dx
    f = np.dot(w, O) # weighted sum equals smoothmax
    return f, grad
```

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    low, high = -np.max(v) - 5.0, (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift_raw(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def smoothmax_and_grad(h, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode='full') * dx
    aO = alpha * O
    max_aO = np.max(aO)
    expa = np.exp(aO - max_aO)
    w = expa / np.sum(expa)
