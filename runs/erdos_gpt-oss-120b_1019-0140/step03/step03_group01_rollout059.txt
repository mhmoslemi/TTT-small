def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)

def random_binary_balanced(n, rng):

import time
import numpy as np

def upsample(arr, new_n):
    """Linear up‑sample a 1‑D array to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1
    return h

def min_overlap_from_autocorr(autocorr, n):
    """Minimum autocorrelation value over all non‑zero shifts."""
    m1 = autocorr[:n - 1].min()
    m2 = autocorr[n:].min()
    return m1 if m1 < m2 else m2

def score_from_min_overlap(min_ov, n):
    """C5 score derived from the minimal overlap."""
    return 1.0 - (2.0 * min_ov) / n

def swap_update(h, autocorr, i, j):
    """Perform a 1↔0 swap (i has 1, j has 0) and update autocorrelation."""
    n = h.shape[0]
    # remove contributions of i (currently 1)
    autocorr[i:i + n] -= h
    h[i] = 0
    # add contributions of j (currently 0)
    autocorr[j:j + n] += h
    h[j] = 1

def swap_revert(h, autocorr, i, j):
    """Undo the swap performed by swap_update."""
    n = h.shape[0]
    # remove contributions of j (currently 1)
    autocorr[j:j + n] -= h
    h[j] = 0
    # add back contributions of i (currently 0)
    autocorr[i:i + n] += h
    h[i] = 1

def simulated_annealing_binary(h, budget_s, start_time, rng):
    """Anneal binary vector h to minimise C5 within the time budget."""
    n = h.shape[0]
    autocorr = np.correlate(h, h, mode='full')
    cur_min = min_overlap_from_autocorr(autocorr, n)
    cur_score = score_from_min_overlap(cur_min, n)

    best_h = h.copy()
    best_score = cur_score
    best_min = cur_min

    T0 = 0.15  # initial temperature (in C5 units)
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 0.2:
            break
        # linear cooling
        T = T0 * (1.0 - elapsed / budget_s)

        # propose a random feasible swap
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)

        # apply swap
        swap_update(h, autocorr, i, j)
        new_min = min_overlap_from_autocorr(autocorr, n)
        new_score = score_from_min_overlap(new_min, n)

        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)

        if accept:
            cur_score = new_score
            cur_min = new_min
            if new_score < best_score:
                best_score = new_score
                best_min = new_min
                best_h = h.copy()
                if best_score <= 0.3808:
                    break
        else:
            # revert the rejected move
            swap_revert(h, autocorr, i, j)

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 128                      # number of samples (must be even)
    # initialise a binary vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init_up = upsample(init, n)
        idx = np.argpartition(-init_up, n // 2)[: n // 2]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1
    else:
        h = random_binary_balanced(n, rng)

    # optimise with simulated annealing
    best_h, best_score = simulated_annealing_binary(h, budget_s, start, rng)

    # final C5 evaluation (as defined in the problem)
    h_float = best_h.astype(float)
    corr = np.correlate(h_float, 1.0 - h_float, mode='full')
    final_score = np.max(corr) * (2.0 / n)

    return h_float, final_score, n
