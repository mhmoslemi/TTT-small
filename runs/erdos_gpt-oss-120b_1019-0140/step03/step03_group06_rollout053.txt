def compute_c5_and_corr(h):
    n = len(h)
    corr_full = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    O = corr_full * dx
    max_idx = np.argmax(O)
    max_val = O[max_idx]
    # shift k = max_idx - (n - 1)
    k_max = max_idx - (n - 1)
    return max_val, O, k_max

import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_and_k(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr * dx
    idx = np.argmax(O)
    k = idx - (n - 1)
    return O[idx], k

def compute_grad(h, k):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if k >= 0:
        i1 = np.arange(0, n - k)
        grad[i1] += dx
        grad[i1] -= dx * h[i1 + k]
        i2 = np.arange(k, n)
        grad[i2] -= dx * h[i2 - k]
    else:
        kk = -k
        i1 = np.arange(kk, n)
        grad[i1] += dx
        grad[i1] -= dx * h[i1 - kk]
        i2 = np.arange(0, n - kk)
        grad[i2] -= dx * h[i2 + kk]
    return grad

def project_to_feasible(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return project_to_feasible(v, target)

# ---------- subgradient phase ----------
def subgradient_descent(h0, target, deadline, rng):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step0 = 1.0
    it = 0
    while time.time() < deadline:
        val, k = compute_c5_and_k(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            if best_val < 0.3808:
                break
        grad = compute_grad(h, k)
        step = step0 / np.sqrt(it + 1)
        h = project_to_feasible(h - step * grad, target)
        it += 1
        if it % 300 == 0:
            noise = (rng.random(h.shape) - 0.5) * 0.02
            h = project_to_feasible(h + noise, target)
    return best_h, best_val

# ---------- annealing refinement ----------
def anneal_refine(h, cur_val, deadline, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_val
    while time.time() < deadline - 0.1:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            prod = h[: n - k] * (1.0 - h[k:])
            i_range = np.arange(n - k)
        else:
            kk = -k
            prod = h[kk:] * (1.0 - h[: n - kk])
            i_range = np.arange(kk, n)
        i = i_range[np.argmax(prod)]
        cand = np.where(h < 1.0 - 1e-12)[0]
        if len(cand) == 0 or h[i] <= 1e-12:
            src = np.where(h > 1e-12)[0]
            dst = np.where(h < 1.0 - 1e-12)[0]
            if len(src) == 0 or len(dst) == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
        else:
            j = rng.choice(cand)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val < 0.3808:
                    break
        else:
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            T = max(remaining * 0.01, 1e-6)
            if np.exp((cur_val - new_val) / T) > rng.random():
                cur_val = new_val
            else:
                h[i] += delta
                h[j] -= delta
    return best_h, best_val

    deadline = start + budget_s
    rng = np.random.default_rng(seed)

        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]

    target = n / 2.0
    h0 = project_to_feasible(h0, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # subgradient phase (≈65% of time)
    sub_deadline = start + budget_s * 0.65
    cur_h, cur_val = subgradient_descent(best_h, target, sub_deadline, rng)
    if cur_val < best_val:
        best_h, best_val = cur_h, cur_val
    if best_val < 0.3808:
        return best_h.astype(float), best_val, n

    # annealing refinement with remaining budget
    best_h, best_val = anneal_refine(best_h, best_val, deadline, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, target)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
