import time, numpy as np, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Using np.correlate
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_seed(p, rng):
    seq = np.zeros(p, dtype=float)
    for i in range(1,p):
        leg = pow(i, (p-1)//2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    # sum(seq)= (p-1)/2
    # add 1 at end
    h = np.concatenate([seq, [1.0]])
    # maybe random shift
    shift = rng.integers(p+1)
    h = np.roll(h, shift)
    return h

def random_binary_seed(n, rng):
    M = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, M, replace=False)
    h[idx] = 1.0
    return h

def anneal_opt(h0, budget, rng):
    # returns best_h, best_c5 found within budget seconds
    start = time.time()
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    # ensure sum constraint satisfied, adjust if needed
    target_sum = n/2
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust uniformly
        h += (target_sum - cur_sum) / n
        np.clip(h, 0.0, 1.0, out=h)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # temperature schedule: exponential decay from T0 to near 0
    T0 = 0.02  # could be larger
    # compute max iterations allowed based on budget
    # let's run while time left > small
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break
        # temperature
        T = T0 * (0.9995 ** it)  # decay
        # choose move type
        if rng.random() < 0.5:
            # binary swap
            # find indices with near 0 or near 1
            zeros = np.where(h < 0.5)[0]
            ones = np.where(h > 0.5)[0]

            i = rng.choice(zeros)

import time
import math
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def legendre_seed(p, rng):
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    h = np.concatenate([seq, [1.0]])  # length p+1, sum = (p+1)/2
    shift = rng.integers(p + 1)
    h = np.roll(h, shift)
    return h

def random_binary_seed(n, rng):
    m = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, m, replace=False)
    h[idx] = 1.0
    return h

def anneal_opt(h0, budget, rng):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target) > 1e-12:
        h += (target - cur_sum) / n
        np.clip(h, 0.0, 1.0, out=h)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    T0 = 0.05
    it = 0
    while True:
        if time.time() - start >= budget:
            break
        T = T0 * (0.9995 ** it)

        # choose move
        if rng.random() < 0.5:
            # binary swap
            zeros = np.where(h < 0.5)[0]
            ones = np.where(h > 0.5)[0]
            if zeros.size == 0 or ones.size == 0:
                continue
            i = rng.choice(zeros)
            j = rng.choice(ones)
            h_new = h.copy()
            h_new[i] = 1.0
            h_new[j] = 0.0
        else:
            # fractional transfer
            i, j = rng.choice(n, 2, replace=False)
            max_transfer = min(1.0 - h[i], h[j])
            if max_transfer <= 1e-12:
                continue
            delta = rng.random() * max_transfer
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

        c5_new = compute_c5(h_new)
        dE = c5_new - cur_c5
        if dE < 0 or rng.random() < math.exp(-dE / (T + 1e-12)):
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        it += 1
    return best_h, best_c5

def optimise_n(n, budget, rng):
    seeds = []

    p = n - 1
    # add Legendre seeds if p is prime
    if p > 2:
        is_prime = True
        for d in range(2, int(math.isqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            for _ in range(5):
                seeds.append(legendre_seed(p, rng))

    # random binary seeds
    for _ in range(3):
        seeds.append(random_binary_seed(n, rng))

    # parent seed, if compatible
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        seeds.append(np.array(initial_h_values, dtype=float))

    if not seeds:
        h0 = np.full(n, 0.5, dtype=float)
        return h0, compute_c5(h0)

    seed_budget = budget / len(seeds)
    best_h = None
    best_c5 = np.inf

        cur_sum = h0.sum()
        if abs(cur_sum - target) > 1e-12:

        if c5_opt < best_c5:
