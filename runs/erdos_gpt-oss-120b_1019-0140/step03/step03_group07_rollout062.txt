    # ensure feasible sums: M must be between lower*n and upper*n.
    lo = -1e5
    hi = 1e5
    # Actually λ range: we can set lo = min(v) - upper, hi = max(v) - lower.
    lo = np.min(v) - upper
    hi = np.max(v) - lower

        if s > M:

            hi = lam
    # after loop, return clip at final lam
    return np.clip(v - lam, lower, upper)

def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 evaluation as used by the harness."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_onto_capped_simplex(v: np.ndarray, M: float,
                                 lower: float = 0.0, upper: float = 1.0,
                                 tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {x | sum(x)=M, lower<=x<=upper} via binary search on λ."""
    n = v.size
    # ensure M feasible
    M = max(lower * n, min(upper * n, M))
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - M) <= tol:
            return x
        if s > M:
            lo = lam          # need larger λ to reduce sum
        else:
            hi = lam
    return np.clip(v - lam, lower, upper)

            # scale for numeric stability
            neg_beta_f = -beta * f_nz
            shift = np.max(neg_beta_f)
            w_nz = np.exp(neg_beta_f - shift)
            w_nz /= w_nz.sum()

            # split into positive/negative shifts and combine by magnitude
            pos = w_nz[n - 1:]                # shifts +1 … + (n‑1)
            neg_rev = w_nz[:n - 1][::-1]      # shifts -1 … - (n‑1)
            w_tot = pos + neg_rev              # length n‑1, w_tot[k‑1] for |k|=k

            # gradient of the soft‑min of overlaps
            g = np.zeros_like(h)
            for k in range(1, n):
                wk = w_tot[k - 1]
                if wk == 0.0:
                    continue
                g[:n - k] += wk * h[k:]      # term from h[i+k]
                g[k:] += wk * h[:n - k]      # term from h[i‑k]

            # ascent step + projection
            h += step * g
            h = _project_onto_capped_simplex(h, M)

            # occasional noise to escape plateaus
            if (it + 1) % 200 == 0:
                h += rng.normal(scale=0.01, size=n)
                h = _project_onto_capped_simplex(h, M)

            # decay / increase parameters
            step = max(step * step_factor, 1e-6)
            beta = min(beta * beta_factor, 500.0)

            # evaluate periodically
            if (it + 1) % eval_interval == 0:
                c5 = _compute_c5(h)
                if c5 + 1e-12 < best_c5:
                    best_h, best_c5 = h.copy(), c5
                    if best_c5 < 0.3808:
                        return best_h, best_c5, n

        # final evaluation of this restart
        c5 = _compute_c5(h)
        if c5 + 1e-12 < best_c5:
            best_h, best_c5 = h.copy(), c5
            if best_c5 < 0.3808:
                return best_h, best_c5, n

    # ----- fallback (should never occur) ------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _compute_c5(best_h)
    else:
        # final feasibility enforcement
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_onto_capped_simplex(best_h, M)
        best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
