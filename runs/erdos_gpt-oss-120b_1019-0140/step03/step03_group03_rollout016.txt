def gradient_descent_with_restart(h_init, time_budget, start_time):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    momentum = np.zeros_like(h)
    it = 0
    last_improve_it = 0
    step0 = 0.5
    beta = 0.9

    while time_left() > 0.01:
        it += 1
        # compute corr
        corr = np.correlate(h, 1-h, mode='full')
        max_corr = corr.max()
        # get indices within eps
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (len(h) - 1)

        # compute gradient for each shift
        n = len(h)
        grad = np.zeros_like(h)
        for shift in shifts:
            # shift positive: h_i h_{i+shift}
            if shift >= 0:
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[:n-shift] = h[shift:]
                term2[shift:] = h[:n-shift]
            else:
                s = -shift
                term1 = np.zeros_like(h)
                term2 = np.zeros_like(h)
                term1[s:] = h[:n-s]
                term2[:n-s] = h[s:]
            grad += (1.0 - term1 - term2)
        grad = grad / len(shifts)

        # momentum
        momentum = beta * momentum + (1-beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, len(h)/2)
        score_new = compute_c5(h_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_h = h_new.copy()
            last_improve_it = it
        else:
            # maybe after certain stagnation, random perturb
            if it - last_improve_it > 5000:
                # small random perturbation
                noise = (np.random.rand(len(h))-0.5)*0.01
                h_new = h_new + noise
                h_new = project_onto_box_sum(h_new, len(h)/2)
                last_improve_it = it

        h = h_new
```

def upsample_binary(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    # interpolation yields values in [0,1]; then threshold top half
    vals = np.interp(x_new, x_old, arr)
    # convert to binary balanced
    idx = np.argsort(vals)[-new_n//2:]
    bin_arr = np.zeros(new_n, dtype=float)
    bin_arr[idx] = 1.0
    return bin_arr
```

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

    corr = np.correlate(h, 1.0 - h, mode='full')

def upsample_binary(arr, new_n):
    old_n = arr.size
    if old_n == new_n:

    x_old = np.linspace(0, 1, old_n, endpoint=False)

    vals = np.interp(x_new, x_old, arr)

    out = np.zeros(new_n, dtype=float)

    # generate thue-morse binary pattern scaled to [0,1]: 0/1? original yields 0/1?
    # Use parity of bits: 0 if even parity, 1 if odd parity? previous code uses (bits count & 1)
    # That yields 0/1 pattern.
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, max_iters, step0=0.5, beta=0.9, time_left=None):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    eps = 1e-12
    for it in range(1, max_iters+1):
        if time_left is not None and time_left() <= 0.1:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, time_left, rng, temp_start=0.5, temp_end=1e-5):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    # schedule based on exponential decay: have a counter for total steps estimated
    # We'll just use iteration counter and compute progress as it / max_iter where max_iter approximated by time_left * 2000
    max_iter_est = int(time_left() * 2000) + 1
    it = 0
    while time_left() > 0.05 and it < max_iter_est:
        it += 1
        progress = it / max_iter_est
        if progress > 1.0:
            progress = 1.0
        t = temp_start * ((temp_end / temp_start) ** progress)
        # select random one and zero indices via rejection sampling
        i = rng.integers(n)
        while cur_h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5:
            j = rng.integers(n)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # continue with swapped configuration
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
