def enforce_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # adjust proportionally
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # distribute diff deterministically across indices
    # if diff >0, add to first indices not at 1; if diff <0, subtract from first indices not at 0
    i = 0
    n = len(v)
    while abs(diff) > 1e-12 and i < n:
        if diff > 0 and v[i] < 1.0 - 1e-12:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 1e-12:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        i += 1
    # if still diff leftover (due to all at bounds), distribute from beginning again (first pass) using a second loop
    if abs(diff) > 1e-12:
        # fallback: adjust all entries uniformly (small amount)
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v
```

import time
import numpy as np
import scipy.optimize as opt

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target):
    """Project v onto [0,1] and adjust to have exact sum==target (deterministic)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    # deterministic distribution of residual
    n = v.shape[0]
    i = 0
    while abs(diff) > 1e-12 and i < n:
        if diff > 0 and v[i] < 1.0 - 1e-12:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        elif diff < 0 and v[i] > 1e-12:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        i += 1
    if abs(diff) > 1e-12:  # fallback uniform correction
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v

def random_binary_vector(n, target, rng):
    """Binary vector with exactly target ones (target assumed integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def random_continuous_vector(n, target, rng):
    """Random vector in [0,1]^n with exact sum==target."""
    v = rng.random(n)
    return enforce_sum(v, target)
