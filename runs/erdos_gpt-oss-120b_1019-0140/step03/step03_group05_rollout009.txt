def gradient_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n-k] += 1.0 - h[k:]
        grad[k:] -= h[:n-k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n-kk]
        grad[:n-kk] -= h[kk:]

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """
    Project vector v onto the hypercube [lo, hi]^n while fixing the sum to target.
    Uses a bisection search for an additive offset.
    """
    low = -np.max(v) - 5.0
    high = (hi - np.min(v)) + 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of the worst overlap term (without dx factor)."""
    n = h.size
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_deadline, rng):
    """
    Projected gradient descent on the max-overlap objective.
    Returns the best h encountered and its C5 value.
    """
    n = h_start.size
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.2
    min_alpha = 1e-6

    for _ in range(max_iter):
        if time.time() > time_deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)

        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)

        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = rng.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1

    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def upscale_h(old_h, n_new):
    """Linear interpolation of a step vector to a larger discretisation."""
    n_old = old_h.size
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # Discretisation size (larger gives a finer bound)
    # -------------------------------------------------
    n = 200                         # can be changed, must be <1000 for speed
    target_sum = n / 2.0

    # -------------------------------------------------
    # Initialise population (use parent if available)
    # -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            h0 = upscale_h(h0, n)
        h0 = project_to_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        h0 = project_to_box_sum(rng.random(n), target_sum)

    pop_size = 16
    population = [h0]
    while len(population) < pop_size:
        cand = project_to_box_sum(rng.random(n), target_sum)
        population.append(cand)

    pop_c5 = [compute_c5(h) for h in population]
    best_idx = int(np.argmin(pop_c5))
    best_h = population[best_idx]
    best_c5 = pop_c5[best_idx]

    # -------------------------------------------------
    # Main evolutionary loop
    # -------------------------------------------------
    safety_margin = 5.0
    end_time = start_time + budget_s - safety_margin
    next_refine = start_time + 30.0          # first local refinement after 30 s
    sigma_mut = 0.05                         # mutation strength

    while time.time() < end_time:
        # ---- tournament selection ----
        idx_a, idx_b = rng.integers(pop_size, size=2)
        parent_a = population[idx_a] if pop_c5[idx_a] <= pop_c5[idx_b] else population[idx_b]

        # second parent (ensure not the same)
        idx_c = rng.integers(pop_size)
        if idx_c == idx_a:
            idx_c = (idx_c + 1) % pop_size
        parent_b = population[idx_c]

        # ---- Crossover child ----
        child_c = 0.5 * (parent_a + parent_b)
        child_c = project_to_box_sum(np.clip(child_c, 0.0, 1.0), target_sum)
        c_c5 = compute_c5(child_c)

        # ---- Mutation child ----
        child_m = parent_a + sigma_mut * rng.standard_normal(n)
        child_m = project_to_box_sum(np.clip(child_m, 0.0, 1.0), target_sum)
        c_m5 = compute_c5(child_m)

        # ---- Replace worst individuals if better ----
        worst_idx = int(np.argmax(pop_c5))
        if c_c5 < pop_c5[worst_idx]:
            population[worst_idx] = child_c
            pop_c5[worst_idx] = c_c5
            if c_c5 < best_c5:
                best_c5, best_h = c_c5, child_c

        worst_idx = int(np.argmax(pop_c5))
        if c_m5 < pop_c5[worst_idx]:
            population[worst_idx] = child_m
            pop_c5[worst_idx] = c_m5
            if c_m5 < best_c5:
                best_c5, best_h = c_m5, child_m

        # ---- Periodic local refinement on the current best ----
        now = time.time()
        if now >= next_refine:
            gd_h, gd_c5 = gradient_descent(
                best_h,
                target_sum,
                max_iter=3000,
                time_deadline=end_time,
                rng=rng
            )
            if gd_c5 < best_c5:
                best_h, best_c5 = gd_h, gd_c5
                # Replace the worst member with the improved best if beneficial
                worst_idx = int(np.argmax(pop_c5))
                if gd_c5 < pop_c5[worst_idx]:
                    population[worst_idx] = gd_h
                    pop_c5[worst_idx] = gd_c5
            next_refine = now + 30.0

        # Early stop if we already beat the target bound
        if best_c5 < 0.3808:
            break

    # -------------------------------------------------
    # Final projection & evaluation
    # -------------------------------------------------
    best_h = project_to_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
