def binary_swap_anneal_fft(h0, rng, time_left_func, max_iters=10**9):
    h = h0.copy()
    n = h.size
    s = n // 2
    # start with current C5
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    # temperature schedule - let's adapt initial T based on best_score magnitude; maybe 0.1 or 0.2
    T = 0.2
    T_min = 1e-5
    decay = 0.9999
    it = 0
    # precompute indices of ones and zeros for fast sampling
    while it < max_iters and time_left_func() > 0.05:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # temperature decay
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            # term1 = h shifted forward: h[shift:] aligns with h[:n-shift]
            # term2 = h shifted backward: h[:n-shift] aligns with h[shift:]
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_refine(h0, time_left_func, max_iter=50000, alpha=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    s = h.size / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    it = 0
    while it < max_iter and time_left_func() > 0.05:
        it += 1
        c5, shifts, _ = compute_c5_and_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # project onto constraints
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5_fft(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_swap_anneal_fft(h0, rng, time_left_func, max_iters=10**9, T0=0.2, decay=0.9999, Tmin=1e-5):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    current_score = best_score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.05:
        it += 1
        # quick ones/zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
    return best_h, best_score

    best_score = compute_c5_fft(best_h)
