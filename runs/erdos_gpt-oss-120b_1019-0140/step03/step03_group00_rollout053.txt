def gradient_smooth_corr(h, alpha):
    n = h.size
    corr_full = np.correlate(h, 1-h, mode='full')
    corr = corr_full * (2.0 / n)
    # compute softmax weights
    max_corr = np.max(corr)  # for numerical stability
    exp_terms = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_terms)
    weights = exp_terms / sum_exp
    # compute gradient
    grad = np.zeros_like(h)
    # shift index mapping: s = i - j + (n-1)? Actually correlation index s corresponds to shift k = s - (n-1)
    for s, w in enumerate(weights):
        if w == 0:
            continue
        k = s - (n - 1)
        if k >= 0:
            # overlap i from 0 to n-1-k
            i = np.arange(n - k)
            # grad_i += w * (1 - h[i+k])
            grad[i] += w * (1 - h[i + k])
            # grad_{i+k} += w * (-h[i]) = -w * h[i]
            grad[i + k] += -w * h[i]
        else:
            s_neg = -k
            i = np.arange(s_neg, n)
            # i-k ranges? For negative k, i + k = i - s_neg, overlapping index
            # Overlap i from -k to n-1
            # Compute contributions similarly:
            # grad_i += w * (1 - h[i+k])
            grad[i] += w * (1 - h[i + k])
            # grad_{i+k} += -w * h[i]
            grad[i + k] += -w * h[i]
    return grad

def smooth_gradient_descent(h0, n_iter, step0, beta, rng, time_left, alpha=30.0):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter+1):
        if time_left() <= 0:
            break
        grad = gradient_smooth_corr(h, alpha)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # optional jitter
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def gradient_smooth_corr(h, alpha):
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_full * (2.0 / n)
    max_corr = np.max(corr)
    exp_terms = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_terms)
    if sum_exp == 0:
        weights = np.zeros_like(exp_terms)
    else:
        weights = exp_terms / sum_exp
    grad = np.zeros_like(h)
    # iterate over all shifts
    for s, w in enumerate(weights):
        if w == 0.0:
            continue
        k = s - (n - 1)
        if k >= 0:
            length = n - k
            i = np.arange(length)
            # vectorized updates
            grad[i] += w * (1.0 - h[i + k])
            grad[i + k] += -w * h[i]
        else:
            s_neg = -k
            length = n - s_neg
            i = np.arange(s_neg, n)
            # i and i+k present.
            grad[i] += w * (1.0 - h[i + k])
            grad[i + k] += -w * h[i]
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure interval contains solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_smooth_corr(h, alpha):
    """Gradient of the soft‑max of the correlation vector."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_full * (2.0 / n)
    max_corr = np.max(corr)
    exp_terms = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_terms)
    if sum_exp == 0:
        weights = np.zeros_like(exp_terms)
    else:
        weights = exp_terms / sum_exp
    grad = np.zeros_like(h)
    for s, w in enumerate(weights):
        if w == 0.0:
            continue
        k = s - (n - 1)
        if k >= 0:
            length = n - k
            i = np.arange(length)
            grad[i] += w * (1.0 - h[i + k])
            grad[i + k] += -w * h[i]
        else:
            s_neg = -k
            length = n - s_neg
            i = np.arange(s_neg, n)
            grad[i] += w * (1.0 - h[i + k])
            grad[i + k] += -w * h[i]
    return grad

def smooth_momentum_descent(h0, n_iter, step0, beta, rng, time_left, alpha):
    """Projected momentum descent on the soft‑max correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        grad = gradient_smooth_corr(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            if best_score <= 0.3808:
                break
        h = h_new
    return best_h, best_score
