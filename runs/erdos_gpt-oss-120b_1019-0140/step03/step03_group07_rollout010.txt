import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_t(x):
    """Objective for SLSQP: minimise the auxiliary variable t."""
    return x[-1]

# ---------- constraints ----------
class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.target

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

# ---------- sequence generators ----------
def generate_balanced_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    ones = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Return Legendre (quadratic residue) sequence of length p (prime)."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def generate_legendre_h(n, rng):
    """For n = p+1 where p is prime, create a balanced binary Legendre vector."""
    p = n - 1
    if p < 2:
        return None
    # simple primality test
    for d in range(2, int(p**0.5) + 1):
        if p % d == 0:
            return None
    leg = legendre_sequence(p)          # length p, sum = (p-1)/2
    h = np.concatenate([leg, [1.0]])    # length n, sum = n/2
    # random cyclic shift to explore equivalent patterns
    shift = rng.integers(n)
    h = np.roll(h, shift)
    return h
