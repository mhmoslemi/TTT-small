def gradient_softmax(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')   # size 2n-1
    # scaling factor optional but not needed for gradient magnitude scaling because we later project.

    # compute softmax weights
    # to avoid overflow, subtract max
    m = np.max(alpha * corr)
    exp_scaled = np.exp(alpha * corr - m)
    p = exp_scaled / exp_scaled.sum()
    # compute fft conv
    L = 1 << ((3*n - 2).bit_length())
    # compute conv p_rev * h
    p_rev = p[::-1]
    H = np.fft.rfft(h, n=L)
    P_rev = np.fft.rfft(p_rev, n=L)
    conv1 = np.fft.irfft(P_rev * H, n=L)
    # conv1 length L, slice positions
    # conv1 index i corresponds to sum_{j} p_rev[j] * h[i - j]; we need indices (n-1) to (n-1)+(n-1) => i from n-1 to 2n-2 inclusive.
    a = conv1[n-1 : n-1 + n]
    # compute conv2 = p * h
    P = np.fft.rfft(p, n=L)
    conv2 = np.fft.irfft(P * H, n=L)
    b = conv2[n-1 : n-1 + n]
    grad = -(a + b)
    return grad

def gradient_softmax(h, alpha):
    n = h.shape[0]
    # compute raw corr
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    p = exp_vals / np.sum(exp_vals)
    # FFT convolution
    L = 1 << ((3*n - 2).bit_length())
    # h padded to L
    H = np.fft.rfft(h, n=L)
    # p_rev
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, n=L)
    conv1 = np.fft.irfft(P_rev * H, n=L)  # length L
    a = conv1[n-1: n-1+n]
    # conv2
    P = np.fft.rfft(p, n=L)
    conv2 = np.fft.irfft(P * H, n=L)
    b = conv2[n-1: n-1+n]
    grad = -(a + b)
    return grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx on the uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def gradient_softmax(h, alpha):
    """Gradient of soft‑max of raw correlations (no dx scaling)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (corr - max_corr))
    p = w / np.sum(w)                     # probabilities over shifts
    # FFT‑based convolution
    L = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2
    H = np.fft.rfft(h, n=L)
    P_rev = np.fft.rfft(p[::-1], n=L)
    conv1 = np.fft.irfft(P_rev * H, n=L)
    a = conv1[n - 1: n - 1 + n]
    P = np.fft.rfft(p, n=L)
    conv2 = np.fft.irfft(P * H, n=L)
    b = conv2[n - 1: n - 1 + n]
    return -(a + b)


def softmax_gradient_descent(h0, time_left, target_iter=150000,
                            step0=5.0, start_alpha=0.1,
                            final_alpha=30.0, alpha_ramp_iters=20000):
    """Projected gradient descent using soft‑max approximation of max."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)
    it = 0
    while it < target_iter and time_left() > 0.5:
        it += 1
        alpha = (start_alpha + (final_alpha - start_alpha) *
                 min(1.0, it / max(1, alpha_ramp_iters)))
        grad_raw = gradient_softmax(h, alpha)   # gradient of raw corr
        grad = grad_raw * dx                     # gradient of scaled corr
        step = step0 / np.sqrt(it)
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def anneal_binary(h0, max_iter, temp_start, temp_end, rng):
    """Simple simulated annealing on a balanced binary vector."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    for it in range(max_iter):
        if ones.size == 0 or zeros.size == 0:
            break
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

                                                  final_alpha=30.0,
                                                  alpha_ramp_iters=20000)

    # ---- occasional random restarts (short runs) -------------------------
    while time_left() > 30.0:
        h_rand = project_onto_box_sum(rng.random(n), n / 2.0)
        h_opt, sc_opt = softmax_gradient_descent(h_rand, time_left,
                                                 target_iter=5000,
                                                 step0=5.0,
                                                 start_alpha=0.1,
                                                 final_alpha=30.0,
                                                 alpha_ramp_iters=2000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

    # ---- binary simulated annealing ---------------------------------------
    if time_left() > 5.0:
        idx = np.argsort(best_h)[-n // 2:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(time_left() * 400)  # ~400 proposals per second
        if max_iter_ann > 0:
            h_ann, sc_ann = anneal_binary(h_bin, max_iter_ann,
                                          temp_start=0.1,
                                          temp_end=1e-5,
                                          rng=rng)
            if sc_ann < best_score:
                best_score = sc_ann
                best_h = h_ann.copy()

    # final projection and scoring
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, n
