        idx_ones = rng.choice(ones, size=s, replace=True)
        idx_zeros = rng.choice(zeros, size=s, replace=True)
        # evaluate each swap
        best_local_h = None
        best_local_c5 = best_c5

            h_new = best_h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0

def refine_with_slsqp(h0, M, rng):
    n = len(h0)
    dx = 2.0 / n
    # initial t guess
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # constraints:
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - M}]
    # overlap constraints
    for k in range(-n+1, n):
        if k == 0: continue
        def make_con(k):
            def con(x, k=k):
                h = x[:n]
                if k >= 0:
                    overl = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overl = np.dot(h[kk:], 1.0 - h[:n - kk])
                return x[-1] - overl * dx
            return con
        constraints.append({'type':'ineq','fun': make_con(k)})
    # objective: minimize t (last variable)
    def obj(x):
        return x[-1]
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 1500, 'ftol':1e-12, 'disp': False})
    if res.success:
        h_opt = np.clip(res.x[:n],0,1)
        # enforce sum exactly
        cur = np.sum(h_opt)
        if abs(cur - M) > 1e-9:
            h_opt += (M - cur)/n
            h_opt = np.clip(h_opt,0,1)
        return h_opt, compute_c5(h_opt)
    else:
        return h0, compute_c5(h0)

import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

# -------------------------------------------------
def generate_random_binary(n, m, rng):
    """Binary vector with exactly m ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, m, replace=False)
    h[idx] = 1.0
    return h

def binary_from_continuous(v, m):
    """Project a continuous vector to the nearest binary vector with sum=m."""
    idx = np.argsort(v)[::-1]
    h = np.zeros_like(v, dtype=float)
    h[idx[:m]] = 1.0
    return h

# -------------------------------------------------
def legendre_sequence(p):
    """Quadratic-residue (Legendre) 0/1 sequence of prime length p."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    i = 3
    while i * i <= p:
        if p % i == 0:
            return False
        i += 2
    return True

def largest_prime_leq(limit):
    p = limit
    while p > 2:
        if is_prime(p):
            return p
        p -= 1
    return 2

def local_search_binary(h0, max_moves, rng, time_budget):
    """Simple hill‑climbing on binary vectors using random swaps."""
    start = time.time()
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    for _ in range(max_moves):
        if time.time() - start > time_budget:
            break
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)

        if new_c5 + 1e-12 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 < 0.3808:
                    break
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c5

def refine_with_slsqp(h0, m, rng):
    """SLSQP refinement of a binary/continuous vector."""
    n = len(h0)
    dx = 2.0 / n
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "eq", "fun": lambda x, n=n, m=m: np.sum(x[:n]) - m}
    ]

    for k in range(-n + 1, n):
        if k == 0:
            continue

        def con(x, k=k, n=n, dx=dx):
            h = x[:n]
            if k >= 0:
                overl = np.dot(h[: n - k], 1.0 - h[k:])
            else:
                kk = -k
                overl = np.dot(h[kk:], 1.0 - h[: n - kk])
            return x[-1] - overl * dx

        constraints.append({"type": "ineq", "fun": con})

    def obj(x):
        return x[-1]

    try:
        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 1200, "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h0, compute_c5(h0)

    if not res.success:
        # fall back to the raw result
        pass

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    cur = np.sum(h_opt)
    if abs(cur - m) > 1e-9:
        h_opt += (m - cur) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

    start = time.time()
    m = n // 2
    best_h = None
    best_c5 = math.inf

    # ----- seed constructions -----
    seeds = []

    # parent if available
    if "initial_h_values" in globals() and len(initial_h_values) == n:
        seeds.append(np.array(initial_h_values, dtype=float))

    p = largest_prime_leq(n - 1)
    leg = legendre_sequence(p)

    h_leg[:p] = leg

    if cur_sum < m:
        zeros = np.where(h_leg == 0)[0]
        need = int(m - cur_sum)
        if need > 0 and len(zeros) >= need:
            add = rng.choice(zeros, size=need, replace=False)
            h_leg[add] = 1.0
    elif cur_sum > m:
        ones = np.where(h_leg == 1)[0]
        excess = int(cur_sum - m)
        if excess > 0 and len(ones) >= excess:
            rem = rng.choice(ones, size=excess, replace=False)
            h_leg[rem] = 0.0
    seeds.append(h_leg)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n)
    scale = m / np.sum(lin)
    h_lin = np.clip(lin * scale, 0.0, 1.0)
    cur = np.sum(h_lin)
    if abs(cur - m) > 1e-12:
        h_lin += (m - cur) / n
        h_lin = np.clip(h_lin, 0.0, 1.0)
    seeds.append(h_lin)

    # random binary seeds
    for _ in range(5):
        seeds.append(generate_random_binary(n, m, rng))

    # ----- search over seeds -----
    for seed_h in seeds:
        if time.time() - start > time_budget - 5:
            break

        # work in binary space
        h_bin = binary_from_continuous(seed_h, m)

        # allocate time for local search
        time_left = time_budget - (time.time() - start) - 1.5
        if time_left <= 0:
            break
        # heuristic move budget (large enough, limited by time)
        max_moves = 60000
        h_opt, c5_opt = local_search_binary(
            h_bin, max_moves, rng, time_left
        )
        if c5_opt + 1e-12 < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if best_c5 < target:
                return best_h, best_c5

        # optional SLSQP refinement if time permits
        if time.time() - start < time_budget - 2.0:
            h_ref, c5_ref = refine_with_slsqp(best_h, m, rng)
            if c5_ref + 1e-12 < best_c5:
                best_h, best_c5 = h_ref, c5_ref
                if best_c5 < target:
                    return best_h, best_c5

    return best_h, best_c5

# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Main entry point returning the best step function found."""
    start_total = time.time()
    rng = np.random.default_rng(seed)
    target = 0.3808

    # candidate resolutions (prefer parent size if present)
    candidates = [80, 96, 112, 128]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0:
            if n0 not in candidates:
                candidates = [n0] + candidates
            else:
                candidates = [n0] + [c for c in candidates if c != n0]

    overall_best_h = None
    overall_best_c5 = math.inf
    overall_best_n = None

    for idx, n in enumerate(candidates):
        if time.time() - start_total > budget_s - 5:
            break
        remaining = budget_s - (time.time() - start_total)
        # allocate a larger share to early candidates
        if idx == 0:
            alloc = min(remaining * 0.6, 550.0)
        else:
            alloc = min(remaining * 0.4 / (len(candidates) - idx), 300.0)

        h_n, c5_n = optimise_n(n, alloc, rng, target)
        if h_n is not None and c5_n < overall_best_c5:
            overall_best_h, overall_best_c5, overall_best_n = h_n, c5_n, n
            if overall_best_c5 < target:
                break

    # fallback if nothing succeeded
    if overall_best_h is None:
        n = 80
        m = n // 2
        overall_best_h = np.zeros(n, dtype=float)
        overall_best_h[:m] = 1.0
        overall_best_c5 = compute_c5(overall_best_h)
        overall_best_n = n

    # final projection onto constraints
    n = overall_best_n
    m = n // 2
    overall_best_h = np.clip(overall_best_h, 0.0, 1.0)
    cur_sum = np.sum(overall_best_h)
    if abs(cur_sum - m) > 1e-9:
        overall_best_h += (m - cur_sum) / n
        overall_best_h = np.clip(overall_best_h, 0.0, 1.0)

    return overall_best_h, float(overall_best_c5), int(n)
