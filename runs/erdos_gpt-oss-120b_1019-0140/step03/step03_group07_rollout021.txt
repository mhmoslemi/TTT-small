    if p < 2: return False
    if p % 2 == 0:
        return p == 2
    r = int(p**0.5)
    for i in range(3, r+1, 2):
        if p % i == 0:

    return True

def hill_climb(h_start, c5_start, time_limit, rng):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = c5_start
    n = len(h)
    # Precompute zero and one positions

        h_new[j] = 0.0
        c5_new = compute_c5(h_new)
        if c5_new + 1e-12 < best_c5:
            best_h = h_new
            best_c5 = c5_new
            # continue early maybe
        # optionally try random other swap for diversification
    return best_h, best_c5

import time

def compute_c5(h):
    """Return C5 for a (possibly fractional) step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p ** 0.5)
    for i in range(3, r + 1, 2):
        if p % i == 0:
            return False
    return True

def legendre_sequence(p):
    """Quadratic residue (0/1) sequence of length p (p prime)."""
    seq = np.zeros(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)  # 1 for residue, p-1 for non‑residue
        seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def best_legendre_h(n):
    """Return best cyclic shift of the Legendre construction for given n, else (None, inf)."""
    p = n - 1
