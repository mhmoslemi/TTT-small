import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # indices achieving max
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional jitter
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def de_optimize(h0, time_left, rng):
    n = h0.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def cost(x):
        h_proj = project_onto_box_sum(x, target_sum)
        return compute_c5(h_proj)
    def callback(xk, convergence):
        return time_left() <= 0
    # choose popsize modestly
    popsize = max(5, min(15, int(2000 // n)))
    # max generation estimate
    max_iter = max(5, int(min(30, time_left() / (0.08 * n))))
    result = opt.differential_evolution(cost, bounds,
                                        strategy='best1bin',
                                        maxiter=max_iter,
                                        popsize=popsize,
                                        mutation=(0.5, 1),
                                        recombination=0.9,
                                        seed=int(rng.integers(1 << 31)),
                                        callback=callback,
                                        polish=False,
                                        updating='deferred',
                                        disp=False)
    h_best = project_onto_box_sum(result.x, target_sum)
    score = compute_c5(h_best)
    return h_best, score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: small Thue-Morse or random balanced
        base = random_binary_balanced(64, rng)

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # define scales, ensure ascending and include base size
    scales = sorted(set([base.size, 128, 256, 512]))
    for n in scales:
        if time_left() <= 2:
            break
        if best_h.size != n:
            # upsample from current best to new size
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)
        # also keep base as starting point for DE (could be best_h)
        h_start = best_h.copy()
        # DE global search
        if time_left() > 5:
            h_de, sc_de = de_optimize(h_start, time_left, rng)
            if sc_de < best_score:
                best_score, best_h = sc_de, h_de.copy()
        # local refinement
        if time_left() > 5:
            n_iter = int(min(60000, max(15000, time_left() * (n / 8.0))))
            h_ref, sc_ref = gradient_descent_momentum_custom(
                best_h, n_iter=n_iter, step0=0.2, beta=0.9, rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
        # prepare for next scale: best_h already sized appropriately

    # final random tweaks
    n_final = best_h.size
    while time_left() > 4:
        # random balanced binary
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        # small mass transfer
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def de_optimize(h0, time_left, rng):
    """Differential Evolution with projection to feasible set."""
    n = h0.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def cost(x):
        h_proj = project_onto_box_sum(x, target_sum)
        return compute_c5(h_proj)

    def callback(xk, convergence):
        return time_left() <= 0

    popsize = max(5, min(15, int(2000 // n)))          # modest population
    max_iter = max(5, int(min(30, time_left() / (0.08 * n))))  # adaptive gens

    res = opt.differential_evolution(
        cost,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=popsize,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=int(rng.integers(1 << 31)),
        callback=callback,
        polish=False,
        updating='deferred',
        disp=False,
    )
    h_best = project_onto_box_sum(res.x, target_sum)
    score = compute_c5(h_best)
    return h_best, score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)

    # multi‑scale refinement
    scales = sorted(set([base.size, 128, 256, 512]))
    for n in scales:
        if time_left() <= 2:
            break
        if best_h.size != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)

        # global DE search
        if time_left() > 5:
            h_de, sc_de = de_optimize(best_h, time_left, rng)
            if sc_de < best_score:
                best_score, best_h = sc_de, h_de.copy()

        # local momentum gradient descent
        if time_left() > 5:
            n_iter = int(min(60000, max(15000, time_left() * (n / 8.0))))
            h_ref, sc_ref = gradient_descent_momentum_custom(
                best_h, n_iter=n_iter, step0=0.2, beta=0.9, rng=rng, time_left=time_left
            )
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    # final random tweaks and small swaps
    n_final = best_h.size
    tweak_cnt = 0
    while time_left() > 4 and tweak_cnt < 2000:
        # random balanced binary candidate
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
        # small mass transfer between two positions
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
        if delta > 0:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break
        tweak_cnt += 1

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
