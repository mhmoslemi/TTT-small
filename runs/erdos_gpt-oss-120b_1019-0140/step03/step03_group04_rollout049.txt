def compute_c5_fft(h):
    n = h.size
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)
    # correlation linear = conv(a, reverse(b))
    b_rev = b[::-1]

    B = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(A * B, size)[:2*n-1]

def linear_convolution_fft(a, b):
    n = a.size
    m = b.size
    size = 1 << ((n+m-1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)
    return conv[:n+m-1]
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def linear_convolution_fft(a, b):
    """Linear convolution using real FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    sz = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, sz)
    B = np.fft.rfft(b, sz)
    conv = np.fft.irfft(A * B, sz)
    return conv[:n + m - 1]

def compute_c5(h):
    """Exact C5 for discrete h using linear correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation to a new length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence (0/1) of length n (half ones if n is power of two)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime). 1 for residue, 0 otherwise."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def smooth_max_c5_and_grad(h, alpha):
    """Return smoothed max C5, raw correlation, and gradient w.r.t. h."""
    n = h.shape[0]
    # correlation h * (1-h) (linear, zero‑padded)
    corr = linear_convolution_fft(h, (1.0 - h)[::-1])
    # softmax weights over all shifts
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp                     # length 2n-1, sums to 1
    # conv_plus = sum_k w_k * h_{i+k}
    conv_plus_full = linear_convolution_fft(w, h)
    conv_plus = conv_plus_full[n-1:n-1+n]
    # conv_minus = sum_k w_k * h_{i-k}
    w_rev = w[::-1]
    conv_minus_full = linear_convolution_fft(w_rev, h)
    conv_minus = conv_minus_full[n-1:n-1+n]
    grad = 1.0 - conv_plus - conv_minus        # because sum w = 1
    smooth_val = max_corr + np.log(sum_exp) / alpha
    return smooth_val, corr, grad

def smooth_max_gd(h0, max_iters, time_end, alpha, eta0=0.5, beta=0.9, eval_every=20):
    """Projected gradient descent on smoothed max C5."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iters + 1):
        if time.time() > time_end:
            break
        _, _, grad = smooth_max_c5_and_grad(h, alpha)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        if it % eval_every == 0:
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                # early stop if already below target
                if best_score <= 0.3808:
                    break
        h = h_new
    # final evaluation
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score
