def adam_optimize(h0, n_iter, alpha=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target_sum = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, n_iter+1):
        # compute gradient for max shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def m_sequence(m, taps):
    n = 2**m - 1
    state = np.ones(m, dtype=int)  # non-zero initial state
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = state[-1]
        new = 0
        for t in taps:
            new ^= state[t]
        state = np.roll(state, 1)
        state[0] = new
    return seq

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)
```

def adam_optimize(h0, n_iter, alpha=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        step = -alpha * m / (np.sqrt(v) + eps)
        h = h + step  # note step negative already
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def momentum_descent_refine(h0, n_iter=2000, step0=0.2, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n/2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter+1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = h - eta * momentum
        h = project_onto_box_sum(h, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def m_sequence(m, taps):
    n = 2**m - 1
    state = np.ones(m, dtype=int)
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = state[-1]
        new = 0
        for t in taps:
            new ^= state[t]
        state = np.roll(state, 1)
        state[0] = new
    return seq

import time
import numpy as np

# ---------- basic utilities ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    # expand interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1‑h(x+k))dx on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


# ---------- m‑sequence generation ----------
def m_sequence(m, taps):
    """Generate a maximal‑length binary LFSR sequence of length 2**m‑1."""
    n = 2**m - 1
    state = np.ones(m, dtype=int)  # non‑zero start
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = state[-1]
        new = 0
        for t in taps:
            new ^= state[t]
        state = np.roll(state, 1)
        state[0] = new
    return seq


# ---------- gradient utilities ----------
def compute_gradients_for_shifts(h, shifts):
    """Sub‑gradient of the max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


# ---------- Adam optimizer ----------
def adam_optimize(h0, n_iter, alpha=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    """Projected Adam for the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        step = -alpha * m / (np.sqrt(v) + eps)

        h = h + step
        h = project_onto_box_sum(h, target)

        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


# ---------- optional momentum refinement ----------
def momentum_refine(h0, n_iter=5000, step0=0.4, beta=0.9):
    """Projected momentum gradient descent (fine tuning)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h = h - eta * momentum
        h = project_onto_box_sum(h, target)

        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score
