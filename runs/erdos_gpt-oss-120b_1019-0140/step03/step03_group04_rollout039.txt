import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # Ensure the interval brackets the solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    """Full linear correlation of h with (1-h) via FFT."""
    n = h.size
    L = 2 * n - 1
    hf = np.fft.rfft(h, n=L)
    gf = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(hf * gf, n=L)
    return corr

def compute_c5_fft(h):
    """Return C5 = (2/n) * max correlation."""
    n = h.size
    corr = compute_corr_fft(h)
    return corr.max() * (2.0 / n)

def resample(arr, new_n):
    """Linear interpolation from len(arr) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def legendre_sequence(p):
    """Quadratic-residue binary sequence of length p-1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def weighted_gradient(h, corr, alpha=10.0, top_k=15):
    """Gradient of a soft‑max approximation of max‑correlation."""
    n = h.size
    max_c = corr.max()
    # soft‑max weights (no overflow)
    w_raw = np.exp(alpha * (corr - max_c))
    # focus on a few dominant shifts
    if top_k < w_raw.size:
        idx = np.argpartition(w_raw, -top_k)[-top_k:]
        w = w_raw[idx]
        w_sum = w.sum()
        w = w / w_sum
        shifts = idx - (n - 1)
    else:
        w = w_raw / w_raw.sum()
        shifts = np.arange(w_raw.size) - (n - 1)
    grad = np.zeros_like(h)
    for shift, weight in zip(shifts, w):
        s = shift
        if s >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - s] = h[s:]
            term2[s:] = h[:n - s]
        else:
            s_abs = -s
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s_abs:] = h[:n - s_abs]
            term2[:n - s_abs] = h[s_abs:]
        grad += weight * (1.0 - term1 - term2)
    return grad

def adam_optimize(h0, time_end, alpha=10.0, top_k=15):
    """Adam optimizer with projection onto box and sum."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < time_end:
        t += 1
        corr = compute_corr_fft(h)
        grad = weighted_gradient(h, corr, alpha=alpha, top_k=top_k)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.7 / np.sqrt(t)  # decaying step
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= 0.3808:
                break
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Choose a resolution balancing accuracy and speed
    n = 2048  # 2048 points give dx≈0.001
    target_sum = n / 2.0

    # Build initial candidates
    candidates = []

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        cand = project_onto_box_sum(resample(base, n), target_sum)
        candidates.append(cand)

    # Legendre difference‑set candidate (prime 263 ≡ 3 mod 4)
    leg = legendre_sequence(263)                 # length 262, balanced
    cand = project_onto_box_sum(resample(leg, n), target_sum)
    candidates.append(cand)

    # Random balanced binary
    candidates.append(random_binary_balanced(n, rng))

    # Pure random continuous
    candidates.append(project_onto_box_sum(rng.random(n), target_sum))

    # Simple sinusoidal pattern
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    h_sin = 0.5 + 0.45 * np.cos(2 * np.pi * x)
    candidates.append(project_onto_box_sum(np.clip(h_sin, 0.0, 1.0), target_sum))

    best_h = None
    best_score = np.inf

    # Optimize each candidate with a modest time budget
    for cand in candidates:
        if time_left() < 8.0:
            break
        per = min(30.0, time_left() * 0.25)
        h_opt, sc_opt = adam_optimize(cand, start + budget_s - 0.3, alpha=10.0, top_k=15)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # Additional random restarts if time permits
    while time_left() > 12.0 and best_score > 0.3808:
        rand_vec = rng.random(n)
        cand = project_onto_box_sum(rand_vec, target_sum)
        h_opt, sc_opt = adam_optimize(cand, start + budget_s - 0.3, alpha=10.0, top_k=15)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # Final safety projection and evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_score = compute_c5_fft(best_h)

    return best_h, best_score, n
