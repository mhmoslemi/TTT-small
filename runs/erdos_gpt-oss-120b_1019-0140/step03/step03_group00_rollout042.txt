def compute_subgradient(h):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = corr.max()
    # tolerance
    tol = 1e-12
    # indices where corr >= max_corr - tol
    idxes = np.where(corr >= max_corr - tol)[0]
    shifts = idxes - (n - 1)
    # compute gradient sum across shifts
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n-shift] += 1 - h[shift:] - h[:n-shift]
        else:
            s = -shift
            grad[s:] += 1 - h[:n-s] - h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

Now compute_subgradient:

def compute_subgradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    tol = 1e-12
    idxes = np.where(corr >= max_corr - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:] - h[:n - shift]
        else:
            s = -shift
            grad[s:] += 1.0 - h[:n - s] - h[s:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = project_onto_box_sum(h0, h0.size / 2.0)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        grad = compute_subgradient(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            # jitter to escape plateaus
            h_new += rng.normal(scale=0.003, size=n)
        h_new = project_onto_box_sum(h_new, n / 2.0)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def simulated_annealing(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    start = time.time()
    total = time_left()
    T0 = 0.001
    # Ensure total >0
    if total <= 0:
        total = 1.0
    while time_left() > 0.5:
        elapsed = time.time() - start
        t_rem = time_left()
        T = T0 * max(t_rem / total, 0.01)
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < current_score:
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - current_score) / T):
                h = h_new
                current_score = new_score
        # occasional small noise
        if rng.random() < 0.01:
            h_noise = h + rng.normal(scale=0.004, size=n)
            h_noise = project_onto_box_sum(h_noise, n / 2.0)
            noise_score = compute_c5(h_noise)
            if noise_score < current_score:
                h = h_noise
                current_score = noise_score
                if noise_score < best_score:
                    best_score = noise_score
                    best_h = h_noise.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def gradient_descent_momentum(h0, max_iters, step0, beta, rng, time_left):
    h = project_onto_box_sum(h0, h0.size / 2.0)
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        grad = compute_subgradient(h)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum

        score = compute_c5(h_new)
        if score < best_score:

            best_h = h_new.copy()

        h_new[j] += delta

    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = None
    best_score = np.inf
    scales = [64, 128, 256, 512]
    iter_sched = {64: 15000, 128: 15000, 256: 20000, 512: 25000}

    for n in scales:
        if time_left() <= 2:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        max_iters = int(min(iter_sched.get(n, 20000), time_left() * 30))
        if max_iters < 1000:
            max_iters = int(min(2000, time_left() * 10))
        step0 = 1.0 if n <= 128 else 0.5
        h_opt, sc_opt = gradient_descent_momentum(base, max_iters, step0, 0.9, rng, time_left)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt

    if best_h is None:
        n = scales[-1]
        best_h = thue_morse(n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    n_final = best_h.size
    if time_left() > 5:
        max_iters = int(min(50000, time_left() * 40))
        if max_iters > 0:
            h_ref, sc_ref = gradient_descent_momentum(best_h, max_iters, 0.2, 0.9, rng, time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()

    if time_left() > 5:
        h_sa, sc_sa = simulated_annealing(best_h, time_left, rng)
        if sc_sa < best_score:
            best_score, best_h = sc_sa, h_sa.copy()

    while time_left() > 5:
        h_rand = random_binary_balanced(n_final, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_score, best_h = sc_rand, h_rand.copy()
            swaps = int(min(20000, time_left() * 3000))
            if swaps > 0:
                h_bin, sc_bin = binary_local_search(best_h, swaps, rng, time_left)
                if sc_bin < best_score:
                    best_score, best_h = sc_bin, h_bin.copy()
        if np.all((best_h == 0) | (best_h == 1)):
            swaps = int(min(20000, time_left() * 3000))
            if swaps > 0:
                h_bin, sc_bin = binary_local_search(best_h, swaps, rng, time_left)
                if sc_bin < best_score:
                    best_score, best_h = sc_bin, h_bin.copy()
        i, j = rng.choice(n_final, size=2, replace=False)
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max > 0:
            delta = delta_max * rng.random() * 0.2
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
