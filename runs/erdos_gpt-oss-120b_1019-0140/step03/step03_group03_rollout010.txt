def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    # Use next power of 2 for speed

    fft_h = np.fft.rfft(h, n=N)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(fft_h * np.conj(fft_one_minus_h), n=N)[:L]
    # The result matches np.correlate 'full'
    return np.max(corr) * (2.0 / n)

def compute_gradient_active_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:

        else:
            s = -shift
            grad[s:] += 1 - h[:n-s]
            grad[:n-s] += 1 - h[s:]

def gradient_descent_momentum(h0, max_iters, step0, beta, time_left, N):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h, N)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters+1):
        if it % 500 == 0 and time_left() < 0.01:
            break
        corr = correlation_full(h, N)
        max_corr = corr.max()
        # active shifts
        eps = 1e-12
        shift_idxs = np.where(corr >= max_corr - eps)[0]
        shifts = shift_idxs - (n - 1)
        # compute gradient
        grad = np.zeros_like(h)
        # iterate over shifts
        for s in shifts:
            if s >= 0:
                grad[:n-s] += 1 - h[s:]
                grad[s:] += 1 - h[:n-s]
            else:
                s_abs = -s
                grad[s_abs:] += 1 - h[:n-s_abs]
                grad[:n-s_abs] += 1 - h[s_abs:]
        grad /= len(shifts)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new, N)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, max_iter, temp_start, temp_end, rng, N):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h, N)
    best_h = cur_h.copy()
    best_score = cur_score
    # indices of ones and zeros
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    for it in range(max_iter):
        # linear cooling in log space
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h, N)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score
```

import time
import numpy as np

def get_fft_len(n):
    L = 2 * n - 1
    return 1 << (L - 1).bit_length()

def correlation_full(h, N):
    fft_h = np.fft.rfft(h, n=N)
    fft_1mh = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=N)
    return corr[:2 * h.size - 1]

def compute_c5(h, N=None):
    n = h.shape[0]
    if N is None:
        N = get_fft_len(n)
    corr = correlation_full(h, N)
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def gradient_descent_momentum(h0, max_iters, step0, beta, time_left, N):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h, N)
    momentum = np.zeros_like(h)
    for it in range(1, max_iters + 1):
        if it % 500 == 0 and time_left() < 0.01:
            break
        corr = correlation_full(h, N)
        max_corr = corr.max()
        idx = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = np.zeros_like(h)
        for s in idx:
            if s >= 0:
                grad[:n - s] += 1 - h[s:]
                grad[s:] += 1 - h[:n - s]
            else:
                a = -s
                grad[a:] += 1 - h[:n - a]
                grad[:n - a] += 1 - h[a:]
        grad /= len(idx)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new, N)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def anneal_binary(h0, max_iter, temp_start, temp_end, rng, N):
    n = h0.size
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h, N)
    best_h = cur_h.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur_h > 0.5)
    zeros = np.flatnonzero(cur_h < 0.5)
    for it in range(max_iter):
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h, N)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n_target = 1024
    N_fft = get_fft_len(n_target)

    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n_target:
            init_up = upsample(init, n_target)
        else:
            init_up = init.copy()
        init_up = project_onto_box_sum(init_up, n_target / 2.0)
        candidates.append(init_up)

    candidates.append(thue_morse(n_target))
    candidates.append(random_binary_balanced(n_target, rng))
    candidates.append(project_onto_box_sum(rng.random(n_target), n_target / 2.0))

    best_h = None
    best_score = np.inf

    for h0 in candidates:
        if time_left() < 5:
            break
        max_iters = int(min(300000, time_left() * 8000))
        h_opt, sc_opt = gradient_descent_momentum(
            h0, max_iters, step0=1.0, beta=0.9, time_left=time_left, N=N_fft)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()

    if best_h is not None and time_left() > 5:
        idx = np.argsort(best_h)[-n_target // 2:]
        h_bin = np.zeros(n_target, dtype=float)
        h_bin[idx] = 1.0
        max_iter_ann = int(time_left() * 600)
        h_ann, sc_ann = anneal_binary(
            h_bin, max_iter_ann, 0.02, 1e-6, rng, N_fft)
        if sc_ann < best_score:
            best_score, best_h = sc_ann, h_ann.copy()

    if best_h is None:
        best_h = random_binary_balanced(n_target, rng)
        best_h = project_onto_box_sum(best_h, n_target / 2.0)
        best_score = compute_c5(best_h, N_fft)
    else:
        best_h = project_onto_box_sum(best_h, n_target / 2.0)
        best_score = compute_c5(best_h, N_fft)

    return best_h, best_score, n_target
