import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_sum(v, target, lo=0.0, hi=1.0):
    """Project v onto the hypercube [lo,hi]^n with fixed sum = target."""
    low = -5.0 - np.max(v)
    high = 5.0 + hi - np.min(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, lo, hi)

def gradient_for_shift(h, k):
    """Gradient of overlap(k) w.r.t. h (without dx factor)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1.0 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad

def gradient_descent(h_start, target_sum, max_iter, time_limit):
    """Projected gradient descent on the max-overlap objective."""
    n = h_start.shape[0]
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = 0.2
    min_alpha = 1e-6
    for _ in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_c5 = corr[max_idx] * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        k = max_idx - (n - 1)
        grad = gradient_for_shift(h, k)
        new_h = h - alpha * grad
        new_h = project_to_box_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha = max(alpha * 0.5, min_alpha)
            if alpha == min_alpha:
                noise = np.random.normal(scale=0.02, size=n)
                h = project_to_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)
                alpha = 0.1
    return best_h, best_c5

class SumConstraint:
    """Equality constraint: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        n = self.n
        if k >= 0:
            overlap = np.dot(h[:n - k], 1.0 - h[k:])
        else:
            kk = -k
            overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
        return x[-1] - overlap * self.dx

def objective_t(x):
    return x[-1]

def slsqp_optimize(h_start, t_start, sum_con, overlap_cons, maxiter=200):
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [{'type': 'eq', 'fun': sum_con}]
    constraints += [{'type': 'ineq', 'fun': oc} for oc in overlap_cons]
    x0 = np.concatenate([h_start, [t_start]])
    res = minimize(
        fun=objective_t,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    return res

            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, mutant, population[i])

            trial_c5 = compute_c5(trial)
            if trial_c5 <= fitness[i]:
                new_pop[i] = trial
                new_fit[i] = trial_c5
            else:
                new_pop[i] = population[i]
                new_fit[i] = fitness[i]

        population = new_pop
        fitness = new_fit

        cur_best_idx = np.argmin(fitness)
        cur_best_c5 = fitness[cur_best_idx]
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = population[cur_best_idx].copy()
            if best_c5 < 0.3808:
                break

    # Local gradient refinement
    time_left = start_time + budget_s - buffer - time.time()
    if time_left > 2.0:
        max_iter = int(2000 * min(1.0, time_left / 100.0))
        gd_deadline = time.time() + time_left - 1.0
        gd_h, gd_c5 = gradient_descent(best_h, target_sum, max_iter, gd_deadline)
        if gd_c5 < best_c5:
            best_h, best_c5 = gd_h, gd_c5

    # Final SLSQP refinement
    time_left = start_time + budget_s - buffer - time.time()
    if time_left > 2.0:
        sum_con = SumConstraint(n)
        overlap_cons = [OverlapConstraint(k, n) for k in range(-n + 1, n) if k != 0]
        maxiter = int(min(200, max(10, time_left * 5)))
        res = slsqp_optimize(best_h, best_c5, sum_con, overlap_cons, maxiter=maxiter)
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            h_opt = project_to_box_sum(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
