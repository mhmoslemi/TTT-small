def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # Determine n, target
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        # compute C5 of init
        init_c5 = compute_c5(init)
        best_h = init.copy()
        best_val = init_c5
    else:
        n = 200
        best_h = random_binary_vector(n, n//2, rng)
        best_val = compute_c5(best_h)

    # ensure sum constraint exact for best_h
    target = n / 2.0
    # ensure binary and exact sum is satisfied; random_binary_vector already does.

    # We will also maintain a current solution candidate for local search.
    cur_h = best_h.copy()
    cur_val = best_val

    # Parameters for annealing
    initial_temp = 0.02  # maybe.
    # We'll update temp as decreasing linearly with remaining time.

    # Main loop
    while time.time() < deadline - 0.5:
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        cur_val = max_corr * (2.0 / n)
        k = max_idx - (n - 1)

        # find candidate swap
        i_candidates, j_candidates = find_swap_candidates(cur_h, k)

        if i_candidates.size > 0 and j_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            # random swap
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            if len(ones_idx)==0 or len(zeros_idx)==0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        # perform swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]

        # Evaluate new
        new_corr = np.correlate(new_h, 1.0 - new_h, mode='full')
        new_val = np.max(new_corr) * (2.0 / n)

        # acceptance
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(initial_temp * (deadline - time.time()) / (deadline - start), 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                # early exit if good enough
                if best_val <= 0.3808:
                    break
        # occasional random restart if no improvement for some iterations
        # we can track iterations without improvement
        # also may need to adjust temperature schedule inside loop

    # final safety: clip and ensure sum (should be binary and sum target)
    best_h = np.clip(best_h, 0.0, 1.0)

    # For safety run enforce_sum with small tolerance

    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)

def enforce_sum(v, target, rng, tol=1e-12):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= tol:
        return v
    diff = target - total
    # use while loop
    max_iter = 10000
    it = 0
    while abs(diff) > tol and it < max_iter:
        if diff > 0:
            # need to increase sum: increase some entries <1
            idxs = np.where(v < 1.0 - 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            idxs = np.where(v > 1e-12)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        it += 1
    return v

import time
import numpy as np

# ---------- helper functions ----------
def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2] (discrete)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng, tol=1e-12, max_iter=10000):
    """Project v onto [0,1]^n while fixing the sum to target (exact)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= tol:
        return v
    diff = target - total
    it = 0
    while abs(diff) > tol and it < max_iter:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        it += 1
    return v

def random_binary_vector(n, target, rng):
    """Binary vector of length n with exactly target ones (target is float, rounded)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    v[rng.choice(n, ones, replace=False)] = 1.0
    return v

def find_swap_candidates(h, k):
    """Return indices i (1->0) and j (0->1) that fix shift k."""
    n = h.shape[0]
    if k > 0:
        i_cand = np.where((h[:n - k] == 1.0) & (h[k:] == 0.0))[0]
        j_cand = np.where((h[:n - k] == 0.0) & (h[k:] == 1.0))[0]
    elif k < 0:
        s = -k
        i_cand = np.where((h[s:] == 1.0) & (h[:n - s] == 0.0))[0] + s
        j_cand = np.where((h[s:] == 0.0) & (h[:n - s] == 1.0))[0] + s
    else:
        i_cand = np.array([], dtype=int)
        j_cand = np.array([], dtype=int)
    return i_cand, j_cand
