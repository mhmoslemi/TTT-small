def subgradient_max_corr(h, corr_raw, max_corr, eps=1e-12):
    n = h.size
    idxes = np.where(corr_raw >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            sp = -s
            grad[sp:] += 1 - h[:n - sp]
            grad[:n - sp] -= h[sp:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

That's similar but faster than building term1, term2 arrays each time.

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def downsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def upsample(arr, new_n):
    return downsample(arr, new_n)  # same linear interp works both ways

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def subgradient_max_corr(h, corr_raw, max_corr, eps=1e-12):
    n = h.size
    idxes = np.where(corr_raw >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            sp = -s
            grad[sp:] += 1.0 - h[:n - sp]
            grad[:n - sp] -= h[sp:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_timebound(h0, time_left_func, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time_left_func() > 0.05:
        it += 1
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr_raw.max()
        grad = subgradient_max_corr(h, corr_raw, max_corr)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        # evaluate exact C5
        score = np.correlate(h_new, 1.0 - h_new, mode='full').max() * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_anneal(h_cont, time_left_func, rng):
    n = h_cont.size
    # create binary seed from top half
    idx = np.argsort(h_cont)[-n//2:]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    iter_num = 0
    t_start = 0.1
    t_end = 1e-5
    while time_left_func() > 0.01:
        iter_num += 1
        # exponential cooling based on iteration fraction (assume 100k max)
        frac = iter_num / 100000.0
        if frac > 1.0:
            frac = 1.0
        t = t_start * ((t_end / t_start) ** frac)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

def slsqp_optimize_small(cand, max_iter=300):
    n = cand.size
    target_sum = n / 2.0
    t0 = compute_c5(cand)
    x0 = np.concatenate([cand, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter,
                                                  'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)
    # small scale setup
    n_small = 64
    candidates = []
    # initial_h if exists
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        if init_full.size != n_small:
            init_small = downsample(init_full, n_small)
        else:
            init_small = init_full.copy()
        init_small = project_onto_box_sum(init_small, n_small / 2.0)
        candidates.append(init_small)
    # thue-morse
    candidates.append(thue_morse(n_small))
    # random binary balanced
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng))
    # random continuous
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))
    # SLSQP refinement (time-limited)
    best_small_h = None
    best_small_score = np.inf
    for cand in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = slsqp_optimize_small(cand, max_iter=300)
        if sc_opt < best_small_score:
            best_small_score = sc_opt
            best_small_h = h_opt.copy()
    # additional random SLSQP starts if time permits
    while time_left() > 5.0:
        v = rng.random(n_small)
        cand = project_onto_box_sum(v, n_small / 2.0)
        h_opt, sc_opt = slsqp_optimize_small(cand, max_iter=200)
        if sc_opt < best_small_score:
            best_small_score = sc_opt
            best_small_h = h_opt.copy()
    # fallback if no candidate improved
    if best_small_h is None:
        best_small_h = random_binary_balanced(n_small, rng)
    # upscale to large resolution
    n_large = 512
    best_h = upsample(best_small_h, n_large)
    best_h = project_onto_box_sum(best_h, n_large / 2.0)
    best_score = compute_c5(best_h)
    # large-scale gradient descent with time bound
    if time_left() > 5.0:
        best_h, best_score = gradient_descent_timebound(best_h, time_left,
                                                       step0=0.6, beta=0.9)
    # optional additional GD restarts if time remains
    restarts = 0
    while time_left() > 30.0 and restarts < 2:
        # start from random continuous vector
        h0 = rng.random(n_large)
        h0 = project_onto_box_sum(h0, n_large / 2.0)
        h_new, sc_new = gradient_descent_timebound(h0, time_left,
                                                   step0=0.6, beta=0.9)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        restarts += 1
    # binary annealing refinement
    if time_left() > 5.0:
        h_bin, sc_bin = binary_anneal(best_h, time_left, rng)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
    # final projection and scoring
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
        if abs(fm - s) < 1e-12:
            break
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def downsample(arr, new_n):
    m = arr.size
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def upsample(arr, new_n):
    return downsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def subgradient_max_corr(h, corr_raw, max_corr, eps=1e-12):
    n = h.size
    idxes = np.where(corr_raw >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += 1.0 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            sp = -s
            grad[sp:] += 1.0 - h[:n - sp]
            grad[:n - sp] -= h[sp:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def gradient_descent_timebound(h0, time_left_func, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 0
    while time_left_func() > 0.05:
        it += 1
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr_raw.max()
        grad = subgradient_max_corr(h, corr_raw, max_corr)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target)
        score = np.correlate(h_new, 1.0 - h_new, mode='full').max() * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def binary_anneal(h_cont, time_left_func, rng):
    n = h_cont.size
    idx = np.argsort(h_cont)[-n // 2:]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    iter_num = 0
    t_start = 0.1
    t_end = 1e-5
    while time_left_func() > 0.01:
        iter_num += 1
        frac = iter_num / 100000.0
        if frac > 1.0:
            frac = 1.0
        t = t_start * ((t_end / t_start) ** frac)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / (t + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score

def slsqp_optimize_small(cand, max_iter=300):
    n = cand.size
    target_sum = n / 2.0
    t0 = compute_c5(cand)
    x0 = np.concatenate([cand, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - target_sum
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter,
                                'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, target_sum)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ---- Small‑scale candidates ----

    if 'initial_h_values' in globals():

        candidates.append(init_small)
    candidates.append(thue_morse(n_small))
    for _ in range(2):
        candidates.append(random_binary_balanced(n_small, rng))
    for _ in range(2):
        v = rng.random(n_small)
        candidates.append(project_onto_box_sum(v, n_small / 2.0))

    # ---- SLSQP refinements (limited) ----
    best_small_h = None
    best_small_score = np.inf
    for cand in candidates:

        if sc_opt < best_small_score:
            best_small_score = sc_opt
            best_small_h = h_opt.copy()
    # a few extra random starts
    for _ in range(5):
        if time_left() < 2.0:

        v = rng.random(n_small)

        if sc_opt < best_small_score:

    if best_small_h is None:

    # ---- Upsample to large resolution ----
    n_large = 512
    best_h = upsample(best_small_h, n_large)
