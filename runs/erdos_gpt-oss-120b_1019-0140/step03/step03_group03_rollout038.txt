def solve_milp(n, time_limit):
    import cvxpy as cp, time
    start = time.time()
    h = cp.Variable(n, boolean=True)
    m = cp.Variable()
    constraints = [cp.sum(h) == n // 2]
    # store w variables in list for potential memory cleanup
    ws = []
    for k in range(-n+1, n):
        length = n - abs(k)
        if length <= 0: continue
        if k >= 0:
            hi = h[:length]
            hj = h[k:k+length]
        else:
            hi = h[-k: -k+length] # i.e., h[-k: n]
            hj = h[:length]
        w = cp.Variable(length, boolean=True)
        ws.append(w)
        constraints += [w <= hi,
                        w <= hj,
                        w >= hi + hj - 1]
        c_k = cp.sum(w)
        constraints += [c_k >= m]
    obj = cp.Maximize(m)
    prob = cp.Problem(obj, constraints)
    # solve with time limit as param if supported. For CBC: set "maxSeconds"? Not sure.
    try:
        prob.solve(solver=cp.CBC, warm_start=True, max_seconds=time_limit)
    except Exception as e:
        prob.solve(solver=cp.CBC, warm_start=True)  # fallback

    if h.value is None:
        # fallback: use random balanced binary
        rng = np.random.default_rng()
        h_bin = np.zeros(n)
        idx = rng.choice(n, size=n//2, replace=False)
        h_bin[idx] = 1.0
        return h_bin, None
    h_bin = np.round(h.value).astype(float) # ensure 0/1
    return h_bin, m.value

def solve_balanced_binary_milp(n, time_limit, rng):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    m = cp.Variable()
    constraints = [cp.sum(h) == n//2]
    for k in range(-n+1, n):
        length = n - abs(k)
        if length <= 0:
            continue
        if k >= 0:
            hi = h[:length]
            hj = h[k:k+length]
        else:
            kj = -k
            hi = h[kj:kj+length]
            hj = h[:length]
        w = cp.Variable(length, boolean=True)
        constraints += [w <= hi,
                        w <= hj,
                        w >= hi + hj - 1]
        constraints += [cp.sum(w) >= m]
    obj = cp.Maximize(m)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, max_seconds=time_limit)
    except Exception:
        # fallback without time limit
        prob.solve(solver=cp.CBC, warm_start=True)
    h_val = h.value
    m_val = m.value
    if h_val is None:
        # fallback random balanced binary
        h_bin = np.zeros(n)
        idx = rng.choice(n, size=n//2, replace=False)
        h_bin[idx] = 1.0
        return h_bin, None
    h_bin = np.round(h_val).astype(float)
    return h_bin, m_val

            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure the interval brackets the root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 = max_k ∫ h(x)(1-h(x+k))dx on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def resample(arr, new_n):
    """Linear interpolation between two uniform grids."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term w.r.t. h for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)


def gradient_descent_momentum(h0, n_iter=10000, step0=0.5, beta=0.9):
    """Projected gradient descent with momentum on the max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def local_swap_improvement(h, max_iters, rng):
    """Targeted hill‑climbing: swap a 1 and a 0 to reduce the worst shift."""
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)

    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # Determine overlapping region
        if k >= 0:
            length = n - k
            h_slice = h[:length]
            h_shift = h[k:k + length]
        else:
            k_abs = -k
            length = n - k_abs
            h_slice = h[k_abs:k_abs + length]
            h_shift = h[:length]

        bad1_local = np.where((h_slice == 1) & (h_shift == 0))[0]
        bad0_local = np.where((h_slice == 0) & (h_shift == 1))[0]

        if bad1_local.size and bad0_local.size:
            i = bad1_local[rng.integers(bad1_local.size)]
            j = bad0_local[rng.integers(bad0_local.size)]
            if k >= 0:
                i_glob = i
                j_glob = j + k
            else:
                i_glob = i + k_abs
                j_glob = j
        else:
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
            if ones.size == 0 or zeros.size == 0:
                break
            i_glob = rng.choice(ones)
            j_glob = rng.choice(zeros)

        # Perform the swap
        h[i_glob], h[j_glob] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
        else:
            # revert most of the non‑improving moves
            if rng.random() < 0.8:
                h[i_glob], h[j_glob] = 1.0, 0.0
    return best_h, best_score


def solve_balanced_binary_milp(n, time_limit, rng):
    """MILP: maximize the minimal autocorrelation for a binary balanced vector."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    h = cp.Variable(n, boolean=True)
    m = cp.Variable()
    constraints = [cp.sum(h) == n // 2]

    for k in range(-n + 1, n):
        length = n - abs(k)
        if length <= 0:
            continue
        if k >= 0:
            hi = h[:length]
            hj = h[k:k + length]
        else:
            k_abs = -k
            hi = h[k_abs:k_abs + length]
            hj = h[:length]

        w = cp.Variable(length, boolean=True)
        constraints += [w <= hi,
                        w <= hj,
                        w >= hi + hj - 1,
                        cp.sum(w) >= m]

    prob = cp.Problem(cp.Maximize(m), constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, max_seconds=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True)
        except Exception:
            return None, None

    if h.value is None:
        return None, None
    h_bin = np.round(h.value).astype(float)
    # Ensure exact balance
    h_bin = project_onto_box_sum(h_bin, n / 2.0)
    return h_bin, m.value
