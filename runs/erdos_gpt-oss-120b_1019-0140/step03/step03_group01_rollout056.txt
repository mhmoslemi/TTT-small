def gradient_descent_softmax(h0, time_left, max_iters=200000, step0=1.0, beta_init=1.0, beta_max=100.0, beta_factor=1.2, beta_step_iters=500):
    h = h0.copy()
    n = h.size
    sum_target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    beta = beta_init
    it = 1
    while it <= max_iters and time_left() > 0.0:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # softmax weights
        max_corr = np.max(corr)
        # avoid overflow
        exp_args = beta * (corr - max_corr)
        w = np.exp(exp_args)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(w) / w.size
            w_sum = 1.0
        w /= w_sum  # w sum = 1

        # compute convs
        L = 2 * n - 1
        # we have w length L
        conv_minus = np.convolve(w, h, mode='full')
        start = L - 1
        conv_minus = conv_minus[start:start + n]
        conv_plus = np.convolve(w[::-1], h, mode='full')
        conv_plus = conv_plus[start:start + n]

        grad = 1.0 - conv_plus - conv_minus

        # step size
        eta = step0 / np.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, sum_target)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

        # adjust beta
        if it % beta_step_iters == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left, max_iters=200000, T_init=0.1, T_min=1e-5, decay=0.9999):

    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = T_init
    while max_iters > 0 and time_left() > 0.01:
        # pick random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        max_iters -= 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=2000):

    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

        return np.sum(x[:n]) - n / 2.0

        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

                           bounds=bounds, constraints=cons,

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    # ensure f(lam_low) >= s? We define f(l) = sum clamp(v - l, lo, hi)
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Expand if needed
    # Usually f(lam_low) = sum hi = hi*n (since v - lam_low >= hi?), but we want f(lam_low) > s
    # Similarly f(lam_high) = sum lo = lo*n < s
    # Actually need lam_low s.t. f(lam_low) >= s, lam_high s.t. f(lam_high) <= s.
    # So choose lam_low = np.min(v) - hi (ensures v-lam_low >= hi => clamp=hi). So f(lam_low) = hi*n >= s.
    # lam_high = np.max(v) - lo => v-lam_high <= lo => clamp=lo => f(lam_high) = lo*n <= s.
    # Use binary search.
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return np.max(corr)

def upsample(arr, new_n):
    """Linear interpolation up‑sampling."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_descent_softmax(h0, time_left, max_iters=50000,
                            step0=0.5, beta_init=1.0, beta_max=64.0,
                            beta_factor=1.2, beta_step=1000):
    """Projected gradient descent on a soft‑max approximation of max‑correlation."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    beta = beta_init
    it = 1
    L = 2 * n - 1
    while it <= max_iters and time_left() > 0.0:
        # correlation (unscaled, used only for weights)
        corr = np.correlate(h, 1.0 - h, mode='full')
        # soft‑max weights
        max_corr = np.max(corr)
        w = np.exp(beta * (corr - max_corr))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # convolutions for gradient
        conv_full = np.convolve(w, h, mode='full')
        start = L - 1
        conv_minus = conv_full[start:start + n]               # sum_s w_s * h[i‑s]
        conv_plus = np.convolve(w[::-1], h, mode='full')
        conv_plus = conv_plus[start:start + n]               # sum_s w_s * h[i+s]
        grad = 1.0 - conv_plus - conv_minus
        # step and projection
        eta = step0 / np.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, target_sum)
        # score update
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        # beta schedule
        if it % beta_step == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left,
                      max_iters=300000, T_init=0.1,
                      T_min=1e-5, decay=0.9999):
    """Swap‑based simulated annealing on a binary vector."""
    h = h0.copy()
    n = h.size
    current = compute_c5(h)
    best_h, best_score = h.copy(), current
    T = T_init
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current
        accept = delta < 0.0 or (T > 0.0 and rng.random() < np.exp(-delta / T))
        if accept:
            current = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=1200):
    """SLSQP refinement of a candidate (continuous)."""
    n = h0.shape[0]
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter,
                                    'ftol': 1e-12,
                                    'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
