import time
import numpy as np

# ----------------- Helper Functions -----------------
def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_sum(v, target, rng):
    """Adjust v (in [0,1]) so that its sum equals target (exactly)."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
        else:
            cand = np.where(v > 1e-12)[0]
        if len(cand) == 0:
            break
        idx = rng.choice(cand)
        if diff > 0:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

# ----------------- Binary Annealing -----------------
def binary_annealing(n, rng, start_time, deadline):
    """Simulated annealing on binary vectors of length n, weight n/2."""
    target = n // 2
    # initial vector
    h = np.zeros(n, dtype=float)
    init_idx = rng.choice(n, target, replace=False)
    h[init_idx] = 1.0
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    T0 = 0.01
    Tmin = 1e-5
    total_time = deadline - start_time
    no_improve = 0

    while time.time() < deadline - 0.1:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        T = max(Tmin, T0 * (1.0 - progress))

        # choose move type
        if rng.random() < 0.3:
            # targeted move based on worst shift
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)

            if k >= 0:
                mask = (h[: n - k] == 1) & (h[k:] == 0)
                i_cands = np.where(mask)[0]
            else:
                kk = -k
                mask = (h[kk:] == 1) & (h[: n - kk] == 0)
                i_cands = np.where(mask)[0] + kk

            if i_cands.size == 0:
                i_cands = np.where(h == 1)[0]
            i = rng.choice(i_cands)

            zero_cands = np.where(h == 0)[0]
            j = rng.choice(zero_cands)
        else:

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val

                no_improve = 0

            else:

        else:
