import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Approximate C5 on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Return Thue-Morse binary sequence of length n as floats."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max correlation w.r.t. h (average over maximal shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad


def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional jitter
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.005, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score


def de_optimize(n, time_left, rng):
    """Differential evolution on feasible set (projection inside cost)."""
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0

    def cost(x):
        h = project_onto_box_sum(x, target_sum)
        return compute_c5(h)

    def callback(xk, convergence):
        return time_left() <= 0

    maxiter = 200  # generous; early stop via callback
    seed_int = int(rng.integers(0, 2**31 - 1))
    res = opt.differential_evolution(cost, bounds,
                                     strategy='best1bin',
                                     maxiter=maxiter,
                                     popsize=15,
                                     mutation=0.5,
                                     recombination=0.9,
                                     seed=seed_int,
                                     callback=callback,
                                     polish=False,
                                     disp=False,
                                     tol=1e-6)
    h_opt = project_onto_box_sum(res.x, target_sum)
    return h_opt, compute_c5(h_opt)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256]
    for s in scales:
        if time_left() <= 1.0:
            break

        # upscale current best to current scale
        if best_h.size != s:
            cur = upsample(best_h, s)
        else:
            cur = best_h.copy()
        cur = project_onto_box_sum(cur, s / 2.0)

        # global search via DE
        if time_left() > 10.0:
            h_de, sc_de = de_optimize(s, time_left, rng)
            if sc_de < best_score:
                best_score, best_h = sc_de, h_de.copy()
                cur = best_h.copy()

        # local refinement (gradient descent)
        if time_left() > 5.0:
            # allocate iterations proportionally to remaining time and problem size
            est_iters = int(min(80000, max(15000, time_left() * (1500.0 / s))))
            h_ref, sc_ref = gradient_descent_momentum_custom(
                cur, n_iter=est_iters, step0=0.2, beta=0.9,
                rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
                cur = best_h.copy()

        # small stochastic swaps / mass transfers
        while time_left() > 2.0:
            i, j = rng.choice(s, size=2, replace=False)
            max_delta = min(best_h[i], 1.0 - best_h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.uniform(0.05, 0.3)
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            # ensure feasibility
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_box_sum(h_new, s / 2.0)
            sc_new = compute_c5(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                cur = best_h.copy()
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break

    # final polish if time permits
    if time_left() > 5.0:
        final_iters = int(min(120000, max(40000, time_left() * 40)))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=final_iters, step0=0.15, beta=0.9,
            rng=rng, time_left=time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_h.size
