def compute_c5_fft(h):
    n = h.size
    L = 2*n
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(fft_h * np.conj(fft_1mh), n=L)
    # conv length L
    # correlation at shift k (negative shift) corresponds to conv[n-1 + k]
    # we need max over all k in -(n-1) .. (n-1)
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def compute_gradient_max(h, eps=1e-12):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    idxes = np.where(corr >= max_corr - eps)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        # compute term1 = np.roll(h, -k) ? For shift k: h_{i+k}
        term1 = np.roll(h, -k)  # h_{i+k}
        term2 = np.roll(h, k)   # h_{i-k}
        grad += 1.0 - term1 - term2
    grad /= len(shifts)
    return grad
```

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

def resample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradient_max(h, eps=1e-12):
    n = h.size
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=L)
    max_corr = np.max(corr)
    idx = np.where(corr >= max_corr - eps)[0]
    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        term1 = np.roll(h, -k)
        term2 = np.roll(h, k)
        grad += 1.0 - term1 - term2
    grad /= len(shifts)
    return grad
```

def adam_optimize(h0, learning_rate=0.2, beta1=0.9, beta2=0.999, eps=1e-8, time_end=None, max_iter=200000):
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 0
    while (time_end is None or time.time() < time_end) and t < max_iter:
        t += 1
        grad = compute_gradient_max(h)
        m = beta1 * m + (1 - beta1) * grad

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def resample(arr, new_n):
    """Linear interpolation from length len(arr) to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse_binary(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def legendre_sequence(p):
    """Quadratic‑residue binary sequence of length p‑1 (p prime)."""
    n = p - 1
    seq = np.zeros(n, dtype=float)
    exp = (p - 1) // 2
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid via FFT."""
    n = h.size
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=L)
    return np.max(corr) * (2.0 / n)

def compute_gradient_max(h, eps=1e-12):
    """Sub‑gradient of max‑correlation term."""
    n = h.size
    L = 2 * n
    fft_h = np.fft.rfft(h, n=L)
    fft_1mh = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=L)
    max_corr = np.max(corr)
    idx = np.where(corr >= max_corr - eps)[0]
    shifts = idx - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        term1 = np.roll(h, -k)   # h_{i+k}
        term2 = np.roll(h, k)    # h_{i-k}
        grad += 1.0 - term1 - term2
    grad /= len(shifts)
    return grad

def adam_optimize(h0, learning_rate=0.2, beta1=0.9, beta2=0.999,
                  eps=1e-8, time_end=None, max_iter=200000):
    """Projected Adam on the C5 objective."""
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    t = 0
    while (time_end is None or time.time() < time_end) and t < max_iter:
        t += 1
        grad = compute_gradient_max(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, n / 2.0)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
