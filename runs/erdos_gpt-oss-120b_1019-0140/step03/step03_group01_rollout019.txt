def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=500000):

    h = h0.copy()
    n = h.size
    s = n/2.0

    best_h = h.copy()
    best_score = compute_c5(h) # compute using conv

    momentum = np.zeros_like(h)

    it = 1
    while it <= max_iter and time_left() > 0.0:
        # compute correlation & max shift(s)
        max_corr, shifts = compute_c5_and_shifts(h)
        # gradient
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1-beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # projection onto box [0,1] and sum constraint
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        # choose random indices of 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

import time

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=30):
    """Euclidean projection onto {lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # Expand interval if necessary
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_c5_and_shifts(h):
    """Return max correlation (raw sum) and the list of integer shifts achieving it."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    conv = signal.fftconvolve(h, rev, mode='full')
    max_corr = np.max(conv)
    idxes = np.where(np.abs(conv - max_corr) < 1e-9)[0]
    shifts = (n - 1) - idxes
    return max_corr, shifts

def compute_c5(h):
    """C5 value = max_k ∫ h(x)(1−h(x+k)) dx approximated on the grid."""
    max_corr, _ = compute_c5_and_shifts(h)
    return max_corr * (2.0 / h.size)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for d in shifts:
        if d >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - d] = h[d:]
            term2[d:] = h[:n - d]
        else:
            s = -d
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=100000):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        max_corr, shifts = compute_c5_and_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_score

def binary_swap_anneal(h0, rng, time_left_func, max_iters=500000):
    h = h0.copy()
    n = h.size
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.1
    T_min = 1e-5
    decay = 0.99995
    it = 0
    while it < max_iters and time_left_func() > 0.0:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_score

def slsqp_refine(h0, max_iter=300):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr, _ = compute_c5_and_shifts(h)
        return t - corr
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h0, compute_c5(h0)
    h_opt = project_onto_box_sum(res.x[:n], n / 2.0)
    return h_opt, compute_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))
    target = 0.3808
    best_h = None
    best_score = np.inf
    n_target = 512

    # Prepare initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        init_full = np.asarray(initial_h_values, dtype=float)
        h0 = upsample(init_full, n_target)
        h0 = project_onto_box_sum(h0, n_target / 2.0)
        candidates.append(h0)
    # Thue‑Morse based candidate
    tm_small = thue_morse(64)
    h_tm = upsample(tm_small, n_target)
    h_tm = project_onto_box_sum(h_tm, n_target / 2.0)
    candidates.append(h_tm)
    # Random continuous candidates
    for _ in range(4):
        v = rng.random(n_target)
        v = project_onto_box_sum(v, n_target / 2.0)
        candidates.append(v)

    # Gradient descent on each candidate
    for cand in candidates:
        if time_left() < 2.0:
            break
        h_opt, sc_opt = gradient_descent_momentum(cand, time_left,
                                                  step0=0.5, beta=0.9, max_iter=100000)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        # Optional local SLSQP refinement
        if time_left() > 1.0:
            h_ref, sc_ref = slsqp_refine(best_h, max_iter=300)
            if sc_ref < best_score:
                best_score, best_h = sc_ref, h_ref.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Binary conversion and annealing if time permits
    if best_h is not None:
        n = best_h.size
        k = n // 2
        idx = np.argpartition(-best_h, k)[:k]
        h_bin = np.zeros_like(best_h)
        h_bin[idx] = 1.0
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
        if time_left() > 2.0:
            h_bin_ref, sc_bin_ref = binary_swap_anneal(best_h, rng, time_left, max_iters=500000)
            if sc_bin_ref < best_score:
                best_score, best_h = sc_bin_ref, h_bin_ref.copy()
        if time_left() > 1.0:
            h_final, sc_final = gradient_descent_momentum(best_h, time_left,
                                                          step0=0.5, beta=0.9, max_iter=100000)
            if sc_final < best_score:
                best_score, best_h = sc_final, h_final.copy()

    # Final safeguard projection and scoring
    if best_h is None:
        best_h = np.full(n_target, 0.5)
        best_h = project_onto_box_sum(best_h, n_target / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
