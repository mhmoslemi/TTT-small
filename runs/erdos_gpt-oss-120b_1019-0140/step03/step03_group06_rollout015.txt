def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # Scale
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    # Now adjust residual
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v
```

import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_binary_vector(n, target, rng):
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def random_continuous_vector(n, target, rng):
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def tournament_select(pop, fitness, rng, k=3):
    idx = rng.choice(len(pop), k, replace=False)
    best = idx[0]
    best_fit = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best = i
            best_fit = fitness[i]
    return pop[best].copy()

def crossover_uniform(p1, p2, rng):
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(child, target, rng, p_mut=0.08, delta_max=0.15):
    # per-gene mutation
    for i in range(child.shape[0]):
        if rng.random() < p_mut:
            delta = rng.uniform(-delta_max, delta_max)
            child[i] = np.clip(child[i] + delta, 0.0, 1.0)
    child = enforce_sum(child, target, rng)
    return child

def local_search(h, cur_val, target, rng, deadline):
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        src = np.where(h > 1e-12)[0]
        dst = np.where(h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:

            else:

                h[i] += delta
                h[j] -= delta

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust so that sum(v) == target exactly."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    if total > 0:
        v = v * (target / total)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        if diff > 0:
            cand = np.where(v < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        attempts += 1
    return v

def random_binary_vector(n, target, rng):
    """Binary vector with exactly target ones (target integer)."""
    ones = int(round(target))
    v = np.zeros(n, dtype=float)
    idx = rng.choice(n, ones, replace=False)
    v[idx] = 1.0
    return v

def random_continuous_vector(n, target, rng):
    """Random vector in [0,1]^n with exact sum == target."""
    v = rng.random(n)
    return enforce_sum(v, target, rng)

def tournament_select(pop, fitness, rng, k=3):
    """Select one parent via tournament of size k."""
    idx = rng.choice(len(pop), k, replace=False)
    best = idx[0]
    best_fit = fitness[best]
    for i in idx[1:]:
        if fitness[i] < best_fit:
            best = i
            best_fit = fitness[i]
    return pop[best].copy()

def crossover_uniform(p1, p2, rng):
    """Uniform crossover producing one child."""
    mask = rng.random(p1.shape[0]) < 0.5
    child = np.where(mask, p1, p2)
    return child

def mutate(child, target, rng, p_mut=0.08, delta_max=0.15):
    """Per‑gene mutation with optional swap and cyclic shift."""
    for i in range(child.shape[0]):
        if rng.random() < p_mut:
            delta = rng.uniform(-delta_max, delta_max)
            child[i] = np.clip(child[i] + delta, 0.0, 1.0)
    # occasional swap mutation
    if rng.random() < 0.02:
        i, j = rng.choice(child.shape[0], 2, replace=False)
        child[i], child[j] = child[j], child[i]
    # occasional cyclic shift
    if rng.random() < 0.05:
        k = rng.integers(1, child.shape[0])
        child = np.roll(child, k)
    child = enforce_sum(child, target, rng)
    return child

def local_search(h, cur_val, target, rng, deadline):
    """Hill‑climbing: move mass from a positive entry to a negative one."""
    n = h.shape[0]
    while time.time() < deadline - 0.1:
        src = np.where(h > 1e-12)[0]
        dst = np.where(h < 1.0 - 1e-12)[0]
        if src.size == 0 or dst.size == 0:
            break
        i = rng.choice(src)
        j = rng.choice(dst)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
        else:
            h[i] += delta
            h[j] -= delta
    return h, cur_val

def genetic_optimization(n, target, rng, deadline):
    """Population‑based search with elites, crossover, mutation and occasional hill‑climb."""
    pop_size = 70
    elite_size = 5
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.shape[0] == n:
            h0 = enforce_sum(h0, target, rng)
            population.append(h0)
    while len(population) < pop_size:
        if rng.random() < 0.5:
            v = random_binary_vector(n, target, rng)
        else:
            v = random_continuous_vector(n, target, rng)
        population.append(v)
    fitness = [compute_c5(v) for v in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]
    generation = 0
    while time.time() < deadline - 0.5:
        sorted_idx = np.argsort(fitness)
        new_pop = [population[i].copy() for i in sorted_idx[:elite_size]]
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, fitness, rng, k=3)
            p2 = tournament_select(population, fitness, rng, k=3)
            child = crossover_uniform(p1, p2, rng)
            child = mutate(child, target, rng)
            new_pop.append(child)
        population = new_pop
        fitness = [compute_c5(v) for v in population]
        cur_best_idx = int(np.argmin(fitness))
        cur_best_val = fitness[cur_best_idx]
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_h = population[cur_best_idx].copy()
            if best_val < 0.3808:
                break
        generation += 1
        if generation % 25 == 0:
            best_h, best_val = local_search(best_h, best_val, target, rng, deadline)
            worst_idx = int(np.argmax(fitness))
            population[worst_idx] = best_h.copy()
            fitness[worst_idx] = best_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)
    n = len(initial_h_values) if 'initial_h_values' in globals() else 80
    target = n / 2.0
    best_h, best_val = genetic_optimization(n, target, rng, deadline)
    # Final hill‑climb with remaining time
    if time.time() < deadline - 0.5:
        best_h, best_val = local_search(best_h, best_val, target, rng, deadline)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
