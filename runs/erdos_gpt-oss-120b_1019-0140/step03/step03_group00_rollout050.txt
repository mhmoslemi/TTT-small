import time
import numpy as np

# ---------- Helper functions ----------
def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}. """
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # adjust interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_weighted_gradient(h, w):
    """Gradient of soft‑max weighted max‑overlap objective."""
    n = h.size
    grad = np.ones_like(h)
    offset = n - 1
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = idx - offset
        if s >= 0:
            if s < n:
                grad[:n - s] -= ws * h[s:]
                grad[s:]      -= ws * h[:n - s]
        else:
            s = -s
            if s < n:
                grad[s:]      -= ws * h[:n - s]
                grad[:n - s] -= ws * h[s:]
    return grad


def smooth_gradient_descent(h0, max_iter, time_left,
                           eta0=0.5, beta=0.9,
                           alpha_start=1.0, alpha_end=40.0):
    """Projected gradient descent using soft‑max smoothing of the max."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    mom = np.zeros_like(h)
    for t in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        # correlation and C5
        corr = np.correlate(h, 1.0 - h, mode='full')
        L = corr * (2.0 / n)          # integral values for all shifts
        C5 = L.max()
        if C5 < best_score:
            best_score = C5
            best_h = h.copy()
        # soft‑max weights
        maxL = L.max()
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        w = np.exp(alpha * (L - maxL))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(L, 1.0 / L.size)
        else:
            w /= w_sum
        # gradient
        grad = compute_weighted_gradient(h, w)
        # momentum
        mom = beta * mom + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(t)
        h_new = h - eta * mom
        # occasional jitter to escape flats
        if t % 1500 == 0:
            h_new += np.random.default_rng().normal(scale=0.01, size=n)
        h = project_onto_box_sum(h_new, s)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # Starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    best_h = None
    best_score = np.inf

    # Multi‑scale optimisation
    scales = [64, 128, 256, 512]
    for idx, n in enumerate(scales):
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # allocate iterations based on remaining time and dimension
        remaining = time_left()
        # generous iteration budget but capped
        iters = int(min(15000, max(3000, remaining * (10.0 / n) * 1000)))
        h_opt, sc_opt = smooth_gradient_descent(base, iters, time_left)

        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        base = h_opt  # continue from improved point for next scale

    # If nothing found (unlikely), fall back to Thue‑Morse at finest scale
    if best_h is None:
        best_h = project_onto_box_sum(thue_morse(scales[-1]), scales[-1] / 2.0)
        best_score = compute_c5(best_h)

    # Final local stochastic refinement on the best vector
    n_final = best_h.size
    s_final = n_final / 2.0
    while time_left() > 5:
        # random balanced binary perturbation
        cand = random_binary_balanced(n_final, rng)
        sc = compute_c5(cand)
        if sc < best_score:
            best_score, best_h = sc, cand.copy()
        # small mass transfer between two positions
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.3)
        if delta > 0:
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_h = project_onto_box_sum(new_h, s_final)
            sc_new = compute_c5(new_h)
            if sc_new < best_score:
                best_score, best_h = sc_new, new_h.copy()
        # early exit if target reached
        if best_score <= 0.3808:
            break

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
