def compute_c5_fft(h):
    n = h.shape[0]
    a_rev = h[::-1]
    b = 1.0 - h
    m = 1 << ((2*n-1).bit_length())

    max_corr = conv[:2*n-1].max()
    return max_corr * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.size
    # clip to [lo,hi]
    v = np.clip(v, lo, hi)
    # if sum already s, return v
    cur = v.sum()
    if abs(cur - s) < 1e-12:
        return v
    # Use bisection on lambda to satisfy sum_i clip(v_i - lam, lo, hi) = s
    lam_low = -1e6
    lam_high = 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
        if abs(total - s) < 1e-12:
            break
    return proj
```

    best_h = h.copy()
    best_score = cur_score

    Tmin = 1e-6
    for it in range(max_iters):

            break
        # pick random distinct i,j with different values

        j = rng.integers(n-1)

            accept = True

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    v = np.clip(v, lo, hi)
    cur = v.sum()
    if abs(cur - s) < 1e-12:
        return v
    # find lambda by bisection
    lam_low, lam_high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam, lo, hi)
        total = proj.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
        if abs(total - s) < 1e-12:
            break
    return np.clip(v - lam, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap using FFT (matches np.correlate(...,mode='full'))."""
    n = h.size
    a_rev = h[::-1]
    b = 1.0 - h
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a_rev, m)
    fb = np.fft.rfft(b, m)
    conv = np.fft.irfft(fa * fb, m)
    max_corr = conv[: 2 * n - 1].max()
    return max_corr * (2.0 / n)

def resample_to_n(arr, n):
    m = arr.shape[0]
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binarize_to_half(arr):
    n = arr.size
    half = n // 2
    idx = np.argpartition(-arr, half - 1)[:half]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def random_binary_balanced(n, rng):
    out = np.zeros(n, dtype=float)
    out[rng.choice(n, n // 2, replace=False)] = 1.0
    return out

def thue_morse_binary(n):
    seq = np.array([0], dtype=float)
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]

def is_prime(p):
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p**0.5) + 1
    for i in range(3, r, 2):
        if p % i == 0:
            return False
    return True

def next_prime(start):
    p = max(2, int(start))
    while not is_prime(p):
        p += 1
    return p

def legendre_sequence(p):
    n = p - 1
    exp = (p - 1) // 2
    seq = np.empty(n, dtype=float)
    for i in range(1, p):
        seq[i - 1] = 1.0 if pow(i, exp, p) == 1 else 0.0
    return seq

def legendre_candidate(n, rng):
    p = next_prime(n + 1)
    seq = legendre_sequence(p)[:n]
    return binarize_to_half(seq)

def anneal(h0, start_time, budget_s, n, max_iters, rng):
    h = h0.copy()
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01
    Tmin = 1e-9
    for it in range(max_iters):
        if time.time() - start_time >= budget_s - 0.2:
            break
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_fft(h)
        t = it / max_iters
        T = T0 * (1 - t) + Tmin * t
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-15)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Choose resolution based on available time
    if time_left() > 900:
        n = 1024
    elif time_left() > 300:
        n = 512
    else:
        n = 256

    candidates = []
    if 'initial_h_values' in globals():

        cand = binarize_to_half(resample_to_n(init, n))
        candidates.append(cand)
    candidates.append(thue_morse_binary(n))
    leg = legendre_candidate(n, rng)
    if leg is not None:
        candidates.append(leg)
    for _ in range(2):
        candidates.append(random_binary_balanced(n, rng))

    # Evaluate initial pool
    best_h = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_c5_fft(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # Main annealing loop
    while time_left() > 5.0:
        # occasional fresh restart
        if rng.random() < 0.3:
            h0 = random_binary_balanced(n, rng)
        else:
            h0 = best_h.copy()
        remaining = time_left()
        # rough estimate: ~3000 iterations per second for n=1024

        h_opt, sc_opt = anneal(h0, start, budget_s, n, max_iters, rng)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score <= 0.3808:
            break

    # Final feasibility projection and exact scoring
    best_h = project_onto_box_sum(best_h, n / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, n
