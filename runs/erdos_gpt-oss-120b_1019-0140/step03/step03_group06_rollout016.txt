def compute_c5(h):

def greedy_refine(h, cur_val, target, rng, deadline):
    n = len(h)
    eps = 1e-12
    while time.time() < deadline - 0.01:
        k, i, raw_max = worst_shift_and_i(h)
        cur_val = raw_max * (2.0 / n)  # recompute (in case changed)
        # candidate dst
        dst = np.where(h < 1.0 - eps)[0]
        if len(dst) == 0:
            break
        # compute base constant
        base = 0.0
        i_shift = i + k
        if 0 <= i_shift < n:
            base -= (1.0 - h[i_shift])
        i_minus = i - k
        if 0 <= i_minus < n:
            base += h[i_minus]
        # compute term2, term4 for all dst
        j = dst
        term2 = np.zeros_like(j, dtype=float)
        j_shift = j + k
        mask = (j_shift >= 0) & (j_shift < n)
        term2[mask] = 1.0 - h[j_shift[mask]]
        term4 = np.zeros_like(j, dtype=float)
        j_minus = j - k
        mask2 = (j_minus >= 0) & (j_minus < n)
        term4[mask2] = -h[j_minus[mask2]]
        coeff = base + term2 + term4
        # choose best negative coeff
        neg = coeff < -1e-12
        if np.any(neg):
            j_best = j[np.argmin(coeff)]  # most negative
        else:
            # random move
            src_candidates = np.where(h > eps)[0]
            if len(src_candidates) == 0:
                break
            i = rng.choice(src_candidates)
            j_best = rng.choice(dst)
        delta = min(h[i], 1.0 - h[j_best])
        if delta <= 0:
            # cannot move
            # maybe try random small delta
            continue
        # apply move
        h[i] -= delta
        h[j_best] += delta
        new_val = compute_c5(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
        else:
            # revert
            h[i] += delta
            h[j_best] -= delta
            # try small random perturbation
            src_candidates = np.where(h > eps)[0]
            dst_candidates = np.where(h < 1.0 - eps)[0]
            if len(src_candidates) > 0 and len(dst_candidates) > 0:
                i2 = rng.choice(src_candidates)
                j2 = rng.choice(dst_candidates)
                max_delta = min(h[i2], 1.0 - h[j2])
                if max_delta > 0:
                    delta2 = max_delta * rng.random()
                    if delta2 > 0:
                        h[i2] -= delta2
                        h[j2] += delta2
                        val2 = compute_c5(h)
                        if val2 < cur_val:
                            cur_val = val2
                        else:
                            h[i2] += delta2
                            h[j2] -= delta2
        # Possibly break early if cur_val < target threshold
        if cur_val * (2.0 / n) < target: # no, target was bound; maybe we have global target?
            pass
    return h, cur_val

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def enforce_sum(v, target, rng):
    """Force vector v∈[0,1]^n to have sum exactly target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = target - total
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        else:
            cand = np.where(v > 0.0)[0]
            if len(cand) == 0:
                break
            idx = rng.choice(cand)
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
    return v

def worst_shift_and_i(h):
    """Return worst shift k, index i contributing most to that shift, and raw max value."""
    n = h.shape[0]
    raw = np.correlate(h, 1.0 - h, mode="full")
    max_idx = raw.argmax()
    k = max_idx - (n - 1)                     # shift value
    # indices i that actually appear in this shift
    if k >= 0:
        i_range = np.arange(0, n - k)
    else:
        i_range = np.arange(-k, n)
    prod = h[i_range] * (1.0 - h[i_range + k])
    i = i_range[np.argmax(prod)]
    return k, i, raw[max_idx]

def greedy_refine(h, cur_val, rng, deadline):
    """Local improvement by moving mass from worst contributors."""
    n = h.shape[0]
    dx = 2.0 / n
    eps = 1e-12
    stagnation = 0
    while time.time() < deadline - 0.01:
        k, i, raw_max = worst_shift_and_i(h)
        cur_val = raw_max * dx
        # destination candidates
        dst = np.where(h < 1.0 - eps)[0]
        if len(dst) == 0:
            break
        # constant part of coefficient (independent of destination)
        base = 0.0
        i_shift = i + k
        if 0 <= i_shift < n:
            base -= (1.0 - h[i_shift])
        i_minus = i - k
        if 0 <= i_minus < n:
            base += h[i_minus]
        # coefficients for all destinations
        j = dst
        term2 = np.zeros_like(j, dtype=float)
        j_shift = j + k
        mask = (j_shift >= 0) & (j_shift < n)
        term2[mask] = 1.0 - h[j_shift[mask]]
        term4 = np.zeros_like(j, dtype=float)
        j_minus = j - k
        mask2 = (j_minus >= 0) & (j_minus < n)
        term4[mask2] = -h[j_minus[mask2]]
        coeff = base + term2 + term4
        # pick destination giving most negative change
        if np.any(coeff < -1e-12):
            j_best = j[np.argmin(coeff)]
        else:
            # fallback random move
            src_cands = np.where(h > eps)[0]
            if len(src_cands) == 0:
                break
            i = rng.choice(src_cands)
            j_best = rng.choice(dst)
        delta = min(h[i], 1.0 - h[j_best])
        if delta <= 0:
            stagnation += 1
            if stagnation >= 50:
                break
            continue
        h[i] -= delta
        h[j_best] += delta
        new_val = compute_c5(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            stagnation = 0
        else:
            # revert
            h[i] += delta
            h[j_best] -= delta
            # small random perturbation for escape
            src_cands = np.where(h > eps)[0]
            dst_cands = np.where(h < 1.0 - eps)[0]
            if len(src_cands) and len(dst_cands):
                i2 = rng.choice(src_cands)
                j2 = rng.choice(dst_cands)
                max_delta = min(h[i2], 1.0 - h[j2])
                if max_delta > 0:
                    d2 = max_delta * rng.random()
                    h[i2] -= d2
                    h[j2] += d2
                    val2 = compute_c5(h)
                    if val2 < cur_val:
                        cur_val = val2
                    else:
                        h[i2] += d2
                        h[j2] -= d2
        stagnation += 1
        if stagnation >= 100:
            break
    return h, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s
    rng = np.random.default_rng(seed)

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 100
        target = n / 2.0
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    best_h = h0.copy()
    best_val = compute_c5(best_h)
    target_c5 = 0.3808

    first = True
    # main search loop
    while time.time() < deadline - 0.5:
        if first and "initial_h_values" in globals():
            cur_h = np.array(initial_h_values, dtype=float)
            cur_h = enforce_sum(cur_h, target, rng)
            first = False
        else:
            cur_h = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            cur_h[idx] = 1.0
        cur_val = compute_c5(cur_h)
        # local refinement
        cur_h, cur_val = greedy_refine(cur_h, cur_val, rng, deadline)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = cur_h.copy()
            if best_val < target_c5:
                break
    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_val = compute_c5(best_h)
    return best_h.astype(float), final_val, n
