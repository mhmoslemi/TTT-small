```python
import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # Ensure brackets contain a solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    return grad / len(shifts)

def gradient_descent_momentum(h0, n_iter=5000, step0=0.5, beta=0.9):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)

    for it in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        eps = 1e-12
        idxes = np.where(corr >= max_corr - eps)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def slsqp_optimize_candidate(h0, max_iter=400):
    n = h0.shape[0]
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - np.max(corr)

    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]

    res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def binary_from_continuous(v):

    idx = np.argpartition(v, -n // 2)[-n // 2:]

    h_bin[idx] = 1.0
    return h_bin

def hill_climb_binary(start_h, max_iters, start_temp, end_temp, rng):
    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h)
    best_h_local = cur_h.copy()
    best_score_local = cur_score
    n = cur_h.size

    for it in range(max_iters):
        # temperature schedule (exponential)
        temp = start_temp * ((end_temp / start_temp) ** (it / max_iters))
        # decide neighbor type
        if rng.random() < 0.2:
            # double swap
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, size=2, replace=False)
            j1, j2 = rng.choice(zeros, size=2, replace=False)
            cur_h[i1] = cur_h[i2] = 0.0
            cur_h[j1] = cur_h[j2] = 1.0
            swapped = (i1, i2, j1, j2)
        else:
            # single swap
            ones = np.flatnonzero(cur_h > 0.5)
            zeros = np.flatnonzero(cur_h < 0.5)
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            swapped = (i, j)

        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if temp > 0:
                accept = rng.random() < np.exp(-delta / temp)
        if accept:
            cur_score = new_score
            if new_score < best_score_local:
                best_score_local = new_score
                best_h_local = cur_h.copy()
        else:
            # revert swap
            if len(swapped) == 4:
                i1, i2, j1, j2 = swapped
                cur_h[i1] = cur_h[i2] = 1.0
                cur_h[j1] = cur_h[j2] = 0.0
            else:
                i, j = swapped
                cur_h[i] = 1.0
                cur_h[j] = 0.0
    return best_h_local, best_score_local

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {x: lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
