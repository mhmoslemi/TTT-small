def optimize_h(n, init_h, time_limit, start_time):
    h = init_h.copy()
    best_h = h.copy()
    best_c = evaluate_erdos_solution(h)
    iteration = 0
    while time.time() - start_time < time_limit:
        # compute correlation of h with (1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        c_vals = corr * (2.0/n)
        max_idx = np.argmax(c_vals)
        curr_c = c_vals[max_idx]
        if curr_c < best_c:
            best_c = curr_c
            best_h = h.copy()
        # if curr_c very low maybe break
        if iteration > 2000:
            break
        # compute shift k
        k = max_idx - (n - 1)
        # compute shifted contributions
        left = np.zeros_like(h)
        right = np.zeros_like(h)
        if k > 0:
            left[k:] = h[:-k]
            right[:-k] = h[k:]
        elif k < 0:
            k = -k
            left[:-k] = h[k:]
            right[k:] = h[:-k]
        # gradient
        grad = (2.0/n) * (1.0 - left - right)
        # step size schedule
        lr = 0.5 / (1 + iteration * 0.001)
        h = h - lr * grad
        # project onto bounds and sum
        h = project_simplex_box(h, n/2)
        iteration += 1
    return best_h, best_c
```

def project_onto_simplex_box(v, target_sum):
    """Project v onto the set {x | 0<=x_i<=1, sum(x)=target_sum}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):

        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
