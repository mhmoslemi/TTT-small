def project_onto_constraints(z, target):
    # find lambda via bisection such that sum(clip(z_i + lambda,0,1)) = target
    lower = -np.max(z) - 1.0  # large negative
    upper = 1.0 - np.min(z)   # large positive
    for _ in range(50):
        mid = (lower + upper)/2
        projected = np.clip(z + mid, 0, 1)
        s = projected.sum()
        if s > target:
            upper = mid
        else:
            lower = mid
    return np.clip(z + (lower+upper)/2, 0, 1)

def optimize(h_init, time_budget):
    n = h_init.shape[0]
    target = n / 2.0
    h = h_init.copy()
    cur_cost = evaluate(h)
    best = h.copy()
    best_cost = cur_cost
    start = time.time()
    rng = np.random.default_rng()
    iteration = 0

    while time.time() - start < time_budget:
        iteration += 1
        # decide move type
        if random.random() < 0.2:
            # random perturb
            step = 0.5 * (0.5 ** (iteration / 20000))  # decays
            cand = h + (rng.random(n) - 0.5) * step
        else:
            # gradient ascent to increase min auto-correlation => reduce max C5
            # compute self-correlation

                grad = 2 * h
            else:

                grad[s:] += h[:n-s]

def _project_onto_constraints(z, target):
    lo = -1.0
    hi = 1.0
    # ensure hi large enough
    while np.clip(z + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    while np.clip(z + lo, 0.0, 1.0).sum() > target:
        lo -= 2.0

        mid = (lo + hi) * 0.5
        proj = np.clip(z + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)

    if remaining > 0.5:
        best_fine, best_fine_c5 = _optimize_h(h_fine, remaining)
    else:
        best_fine, best_fine_c5 = h_fine, _c5_value(h_fine)

    # ------------------------------------------------------------------
    # 4️⃣  Return the best result obtained
    # ------------------------------------------------------------------
    if best_fine_c5 < best_small_c5:
        final_h, final_c5 = best_fine, best_fine_c5
    else:
        final_h, final_c5 = best_small, best_small_c5

    return final_h, float(final_c5), final_h.shape[0]
