    s = h.sum()

    # While difference > tolerance:
    diff = target_sum - s
    # If diff > 0, need to increase values where h < 1.
    # If diff < 0, need to decrease values where h > 0.
    # Use uniform distribution across eligible indices.
    # We'll limit iterations to, say, 10.
    for _ in range(10):

            break
        if diff > 0:

            eligible = np.where(h < 1 - 1e-12)[0]
            if eligible.size == 0:

def hill_climb(h, time_limit, start_time, rng):

   cur_h = h.copy()
   cur_c5 = compute_c5(cur_h)
   best_h = cur_h.copy()
   best_c5 = cur_c5

   T0 = 0.01

   # We'll set iterations to high, but break if time over.

   while time.time() - start_time < time_limit:
        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # Determine max delta
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        # step size factor
        delta = max_delta * rng.random() * 0.5 # step factor
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = compute_c5(new_h)
        diff = new_c5 - cur_c5
        # temperature schedule
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = T0 * (1 - frac)
        if diff <= 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

        # occasional restart: random binary each some iteration
        if rng.random() < 0.001:
            # reset cur_h to best_h or random?
            cur_h = best_h.copy()
            cur_c5 = best_c5

   return best_h, best_c5

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_sum_bounds(h, target_sum, rng):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    # Simple iterative distribution
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc = min(inc, 1 - h[idx].max())
            h[idx] += inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            dec = (-diff) / idx.size
            dec = min(dec, h[idx].min())
            h[idx] -= dec
        s = h.sum()
        diff = target_sum - s
    # leftover correction
    if abs(diff) > 1e-8:
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size > 0:
            i = rng.choice(idx)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def hill_climb(start_h, time_left, rng):
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start_time = time.time()
    T0 = 0.01
    step_factor = 0.5
    iteration = 0
    while True:
        elapsed_total = time.time() - start_time
        if elapsed_total >= time_left:
            break
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # direction: if h[i] > h[j], move from i to j else opposite
        if cur_h[i] > cur_h[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * step_factor
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        new_c5 = compute_c5(new_h)
        diff = new_c5 - cur_c5
        # temperature schedule
        frac = elapsed_total / time_left
        T = T0 * (1 - frac)
        if diff <= 0 or rng.random() < math.exp(-diff / (T + 1e-12)):

import time, math
import numpy as np
try:
    from scipy.optimize import minimize_scalar
except Exception:  # pragma: no cover
    minimize_scalar = None


def _c5_of(h: np.ndarray) -> float:
    """Compute C₅ for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_sum_bounds(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact sum (water‑filling style)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(12):
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc = min(inc, 1.0 - h[idx].max())
            h[idx] += inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            dec = (-diff) / idx.size
            dec = min(dec, h[idx].min())
            h[idx] -= dec
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        if diff > 0.0:
            cand = np.where(h < 1.0)[0]
        else:
            cand = np.where(h > 0.0)[0]
        if cand.size:
            i = rng.choice(cand)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def _hill_climb(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simple pairwise mass‑transfer hill climbing with a simulated‑annealing acceptance."""
    n = start_h.shape[0]
    target = n / 2.0
    cur = start_h.copy()
    cur_val = _c5_of(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.015
    step_factor = 0.6
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # propose a move: transfer a random amount from src to dst
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] >= cur[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(cur[src], 1.0 - cur[dst])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * step_factor
        nxt = cur.copy()
        nxt[src] -= delta
        nxt[dst] += delta
        nxt_val = _c5_of(nxt)
        # temperature declines linearly
        T = T0 * (1.0 - elapsed / time_budget)
        if nxt_val <= cur_val or rng.random() < math.exp(-(nxt_val - cur_val) / (T + 1e-18)):
            cur, cur_val = nxt, nxt_val
            if nxt_val < best_val:
                best, best_val = nxt.copy(), nxt_val
        # occasional tiny cooling of step size
        if it % 1000 == 0 and step_factor > 0.02:
            step_factor *= 0.99
        # occasional restart from best
        if it % 5000 == 0 and rng.random() < 0.07:
            cur, cur_val = best.copy(), best_val
        it += 1
    return best, best_val
