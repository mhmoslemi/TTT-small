def project_to_box_sum(v, S, tol=1e-8, max_iter=100):
    lo = -1.0
    hi = 1.0
    # Expand bounds until sum at lo > S and sum at hi < S or vice versa.
    # Actually for lo = -np.max(v) - 1 ensures all v_i - lo >> 1 => h_i=1 => sum=n
    # hi = np.max(v) + 1 ensures v_i - hi <= -1 => h_i=0 => sum=0.
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        h = np.clip(v - lam, 0, 1)
        s = h.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    h = np.clip(v - (lo + hi)/2, 0, 1)
    # If sum not exactly S due to clipping (e.g., some at bounds cannot adjust),
    # we can distribute residual across non-bound indices.
    diff = S - h.sum()
    if abs(diff) > 1e-8:
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
            # clip again in case of small overshoot
            h = np.clip(h, 0, 1)
    return h

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    # pad to next power of 2 for speed maybe.
    size = 2 * n
    H = np.fft.rfft(h, n=size)  # real FFT
    G = np.fft.rfft(1 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)  # length size
    # The correlation at shift indices from -(n-1) to +(n-1) corresponds to
    # part of conv result? Actually using convolution formula.
    # For correlation via FFT, we need to compute conv of h and reversed (1-h). But correlation defined as
    # (f ⋆ g)[k] = sum_i f[i] * g[i+k] (no reversed). np.correlate uses this definition.
    # Using FFT: corr = ifft(fft(f) * conj(fft(g))) gives correlation of f with g (circular) of length size.
    # For linear correlation, we need zero padding as we did size = 2*n.
    corr = corr[:2*n-1]
    return (corr * dx).max()

def max_shift_and_contrib(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    # compute contributions
    if k >= 0:
        p = h[:n - k] * (1 - h[k:])
        i_offset = 0
    else:
        k_abs = -k
        p = h[k_abs:] * (1 - h[:n - k_abs])
        i_offset = k_abs
    i_max_rel = np.argmax(p)
    i_max = i_offset + i_max_rel
    # j index is i_max + k
    j = i_max + k
    return k, i_max, j, corr[max_idx]
```

def project_to_box_sum(v, S, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):

        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual
    diff = S - h.sum()
    if abs(diff) > 1e-8:
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    return float(corr.max())

def project_to_box_sum(v: np.ndarray, S: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Project v onto the box [0,1] with exact sum S (∑h = S)."""
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-8:
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def random_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random feasible vector of length n (values in [0,1], sum = n/2)."""
    v = rng.random(n)
    return project_to_box_sum(v, n / 2)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    max_time = budget_s * 0.95  # safety margin

    # Initialise h
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_to_box_sum(h0, n / 2)
    except Exception:
        n = 120
        h0 = np.full(n, 0.5)
        h0 = project_to_box_sum(h0, n / 2)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Main optimisation loop
    while time.time() - start < max_time:
        # Occasionally start from a fresh random point
        if rng.random() < 0.1:
            h = random_vector(rng, n)
            cur_score = compute_c5(h)
        else:
            h = best_h.copy()
            cur_score = best_score

        T = 0.02
        step = 0.2
        it = 0
        # inner SA loop
        while time.time() - start < max_time and it < 3000:
            dx = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            idx = int(np.argmax(corr))
            k = idx - (n - 1)

            # contributions for the worst shift
            if k >= 0:
                p = h[: n - k] * (1.0 - h[k:])
                i0 = 0
            else:
                kk = -k
                p = h[kk:] * (1.0 - h[: n - kk])
                i0 = kk
            i_rel = int(np.argmax(p))
            i = i0 + i_rel
            j = i + k

            moved = False
            if 0 <= j < n:
                delta = min(step, h[i], 1.0 - h[j])
                if delta > 1e-12:
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    new_score = compute_c5(h_new)
                    dscore = new_score - cur_score
                    if dscore < 0 or rng.random() < math.exp(-dscore / T):
                        h = h_new
                        cur_score = new_score
                        moved = True
                        if cur_score < best_score:
                            best_h = h.copy()
                            best_score = cur_score

            if not moved:
                a, b = rng.integers(n, size=2)
                if a != b:
                    max_inc = min(step, 1.0 - h[a])
                    max_dec = min(step, h[b])
                    delta = rng.uniform(-max_dec, max_inc)
                    delta = max(-max_dec, min(max_inc, delta))
                    if abs(delta) > 1e-12:
                        h_new = h.copy()
                        h_new[a] += delta
                        h_new[b] -= delta
                        new_score = compute_c5(h_new)
                        dscore = new_score - cur_score
                        if dscore < 0 or rng.random() < math.exp(-dscore / T):
                            h = h_new
                            cur_score = new_score
                            if cur_score < best_score:
                                best_h = h.copy()
                                best_score = cur_score
            it += 1
            if it % 400 == 0:
                T *= 0.85
                step *= 0.9

        # early exit if we already beat the target
        if best_score <= 0.3808:
            break

    # final feasibility projection
    best_h = project_to_box_sum(best_h, n / 2)
    best_score = compute_c5(best_h)

    return (best_h.tolist(), float(best_score), n)
