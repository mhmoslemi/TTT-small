def project_onto_box_and_sum(x, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)

import time
import math
import random
import numpy as np

def _project_onto_box_and_sum(x: np.ndarray, target_sum: float, tol: float = 1e-12) -> np.ndarray:
    """Project x onto the box [0,1] with the linear equality sum(x)=target_sum."""
    # binary search for lambda such that sum(clip(x - lambda,0,1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function given by vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the inner product sum_i h_i * h_{i+shift}
    (used to increase the inner product and thus reduce the overlap)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        # contribution from h_i * h_{i+shift}
        grad[: n - shift] += h[shift:]
        # contribution from h_{i-shift} * h_i
        grad[shift:] += h[: n - shift]
    else:
        s = -shift
        grad[s:] += h[: n - s]
        grad[: n - s] += h[s:]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.

    Returns:
        h_values (list of float): the optimized step heights.
        c5_bound (float)       : the achieved C5 value.
        n_points (int)         : number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Obtain an initial vector.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)  # use the supplied construction
    else:
        # fallback: uniform half height
        h0 = np.full(45, 0.5, dtype=float)

    n = h0.shape[0]
    target_sum = n / 2.0
    h = _project_onto_box_and_sum(h0, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(h)

    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a little slack

    # ------------------------------------------------------------------
    # Subgradient descent on the worst shift.
    # ------------------------------------------------------------------
    max_iter = 50000
    step0 = 0.5
    no_improve = 0

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break

        # correlation (not yet scaled by dx)
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_c5 = corr[idx] * (2.0 / n)

        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # locate the shift that yields the worst overlap
        shift = idx - (n - 1)

        # gradient of the inner product for that shift
        grad = _gradient_worst_shift(h, shift)

        # diminishing step size
        step = step0 / math.sqrt(it + 1.0)

        # tentative update and projection onto the feasible set
        h_candidate = h + step * grad
        h = _project_onto_box_and_sum(h_candidate, target_sum)

        # occasional random perturbation to escape plateaus
        if (it + 1) % 2000 == 0:
            if random.random() < 0.5:
                jitter = (np.random.rand(n) - 0.5) * 0.02
                h = _project_onto_box_and_sum(h + jitter, target_sum)

        # if stuck for a while, re‑initialize randomly
        if no_improve >= 4000:
            h = _project_onto_box_and_sum(np.random.rand(n), target_sum)
            no_improve = 0

    # final check in case the loop finished without updating best_h
    final_c5 = _c5_value(best_h)
    return best_h.tolist(), float(final_c5), int(n)
