def repair_vector(h, n):
    # Clip values to [0,1]
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = np.sum(h)
    if cur_sum == 0:
        # set all to target / n = 0.5 maybe
        h[:] = 0.5
        return h
    # Scale to match sum
    scale = target_sum / cur_sum
    h = h * scale
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    # Re-adjust any residual sum error using simple water-filling:
    cur_sum = np.sum(h)
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute diff equally among entries not at bounds
        # Identify indices where also not at 0 or 1.
        not_fixed = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(not_fixed)
        if cnt > 0:
            h[not_fixed] += diff / cnt
            # Clip again to ensure bounds
            h = np.clip(h, 0.0, 1.0)
    return h

import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def repair_vector(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur_sum = np.sum(h)
    if cur_sum == 0:
        h[:] = target / n
        return h
    scale = target / cur_sum
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    # adjust residual sum via water-filling
    cur_sum = np.sum(h)
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        # find indices not at bounds
        free = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_descent_max(h, steps, step_size):
    n = h.shape[0]
    for _ in range(steps):
        corr_full = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr_full)
        lag = idx - (n - 1)
        grad = np.zeros_like(h)
        if lag >= 0:
            # (1 - h[j+lag])
            grad[:n - lag] += 1 - h[lag:]
            # -h[j - lag]
            grad[lag:] -= h[:n - lag]
        else:
            d = -lag
            grad[d:] += 1 - h[:n - d]
            grad[:n - d] -= h[d:]
        h = h - step_size * grad
        h = repair_vector(h, n)
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def repair_vector(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur_sum = np.sum(h)
    if cur_sum == 0.0:
        h[:] = target / n
        return h
    scale = target / cur_sum
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_descent_max(h, steps, step_size):
    n = h.shape[0]
    for _ in range(steps):
        corr_full = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr_full)
        lag = idx - (n - 1)
        grad = np.zeros_like(h)
        if lag >= 0:
            # (1 - h[j+lag])
            grad[:n - lag] += 1 - h[lag:]
            # -h[j-lag]
            grad[lag:] -= h[:n - lag]
        else:
            d = -lag
            grad[d:] += 1 - h[:n - d]
            grad[:n - d] -= h[d:]
        h = h - step_size * grad
        h = repair_vector(h, n)
    return h

def optimize_n(n, init_h, budget_remaining, rng):
    best_h = None
    best_score = float('inf')
    # Gradient descent phase
    gd_steps = 100
    step_size = 0.1
    h_current = init_h.copy()
    start = time.time()
    for i in range(gd_steps):
        if time.time() - start > budget_remaining * 0.5:
            break
        h_current = gradient_descent_max(h_current, steps=1, step_size=step_size)
    # L-BFGS-B refinement
    penalty = 1000.0
    def obj_fun(vec):
        return compute_c5(vec) + penalty * (np.sum(vec) - n/2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(obj_fun, h_current, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        if res.success:
            cand_h = repair_vector(res.x, n)
        else:
            cand_h = h_current
    except Exception:
        cand_h = h_current
    cand_score = compute_c5(cand_h)
    best_h, best_score = cand_h, cand_score
    # Random restarts
    elapsed = time.time() - start
    time_left = budget_remaining - elapsed
    restart_limit = 3
