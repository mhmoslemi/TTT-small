def project_to_feasible(h, target):
    # Project onto [0,1] and sum constraint exactly.
    # Use algorithm: first clip to [0,1]
    h = np.clip(h, 0, 1)
    # Compute sum diff
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # We will adjust entries that are not at bounds.
    # Determine free indices where 0 < h < 1
    free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
    # If no free indices (rare), just adjust a random one within bounds if possible.
    # For simplicity, distribute diff evenly across free indices, clipping as needed.
    # iterative loop:
    while abs(diff) > 1e-10 and len(free) > 0:
        # amount per index:
        delta = diff / len(free)
        # update each free index
        new_vals = h[free] + delta
        # Clip to [0,1]
        clipped = np.clip(new_vals, 0, 1)
        # compute actual change applied:
        actual_change = np.sum(clipped - h[free])
        # update
        h[free] = clipped
        # update diff
        diff -= actual_change
        # recompute free (those still interior)
        free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
    # If still diff not zero due to rounding or no free indices, adjust a single index within bound if possible
    if abs(diff) > 1e-8:
        # find index where we can add/subtract without violating bounds
        if diff > 0:
            idx = np.argmax(1 - h)  # largest capacity
            capacity = 1 - h[idx]
            add = min(diff, capacity)
            h[idx] += add
        else:
            idx = np.argmax(h)   # largest value, can decrease
            capacity = h[idx]
            sub = min(-diff, capacity)
            h[idx] -= sub
    # Clip final
    h = np.clip(h, 0, 1)
    # Now sum should be target (within tolerance)
    return h

import numpy as np
import time
from scipy.optimize import differential_evolution

def compute_c5(h, dx):
    # Compute max overlap integral using full correlation.
    # h can be any iterable, ensure np.array
    h_arr = np.asarray(h, dtype=float)
    corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, target_sum, tol=1e-12, max_iter=1000):
    # Project onto box [0,1] and linear equality sum(h) = target_sum.
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) <= tol:
        return h
    # Iterative water-filling style redistribution.
    for _ in range(max_iter):
        # Indices that can still move (strict interior)
        free = np.where((h > tol) & (h < 1.0 - tol))[0]
        if free.size == 0:
            # No interior points; adjust a bound that permits change
            if diff > 0:
                # Increase the one with greatest capacity
                caps = 1.0 - h
                idx = np.argmax(caps)
                delta = min(diff, caps[idx])
                h[idx] += delta
                diff -= delta
            else:
                caps = h
                idx = np.argmax(caps)
                delta = min(-diff, caps[idx])
                h[idx] -= delta
                diff += delta
            if abs(diff) <= tol:
                break
            continue
        # Distribute diff uniformly across free entries
        delta = diff / free.size
        # Apply with clipping
        new_vals = h[free] + delta
        # Clip to [0,1] and compute actual change applied
        clipped = np.clip(new_vals, 0.0, 1.0)
        actual = np.sum(clipped - h[free])
        h[free] = clipped
        diff -= actual
        if abs(diff) <= tol:
            break
    # Final tiny correction if necessary
    if abs(diff) > tol:
        # pick a index with capacity
        if diff > 0:
            caps = 1.0 - h
            idx = np.argmax(caps)
            delta = min(diff, caps[idx])
            h[idx] += delta
        else:
            caps = h
            idx = np.argmax(caps)
            delta = min(-diff, caps[idx])
            h[idx] -= delta
    # Ensure exactness (within float tolerance)
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    import time
    from scipy.optimize import differential_evolution

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
        has_initial = True
    except NameError:
        # No provided initial; use modest size
        n_points = 60
        init_h = None
        has_initial = False

    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Helper to evaluate objective with penalty
    penalty_coeff = 1e5  # large to enforce sum constraint

    def objective(x):
        # x is a candidate array
        c5 = compute_c5(x, dx)
        sum_err = np.sum(x) - target_sum
        if abs(sum_err) > 1e-8:
            c5 += penalty_coeff * abs(sum_err)
        return c5

    # Build bounds for each variable
    bounds = [(0.0, 1.0)] * n_points

    # Prepare initial population for differential evolution
    popsize = 20
    # generate random initial population
    pop = rng.random((popsize, n_points))
    if has_initial:
        # place the given construction as first individual (projected)
        pop[0] = np.clip(init_h, 0.0, 1.0)
    # Provide this population to DE
    # Early termination callback
    def stop_callback(xk, convergence):
        # stop if time budget about to expire
        return (time.time() - start_time) > budget_s * 0.85

    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=2000,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        init=pop,
        callback=stop_callback,
        polish=False,
        disp=False,
        updating='deferred',
    )
    # Use best solution from DE
    best_h = result.x
    # Project onto feasible region exactly
    best_h = project_to_feasible(best_h, target_sum)

    # Compute current bound
    best_c5 = compute_c5(best_h, dx)

    # Local refinement via pairwise exchanges
    # Preserve sum exactly by moving mass between two coordinates
    # Continue while time permits
    while (time.time() - start_time) < budget_s * 0.95:
        # Random pair
        i, j = rng.integers(0, n_points, size=2)
        if i == j:

        # Small random delta
        delta = (rng.random() - 0.5) * 0.02  # +-0.01 typical step

        if delta > 0:

            max_dec_j = best_h[j]
            delta = min(delta, max_inc_i, max_dec_j)
        else:
