        h_bin[turn_on] = 1

import time, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    init = globals().get('initial_h_values')
    if init is not None:
        n = int(np.asarray(init).size)
    else:
        n = 70
    M = n // 2

    # Helper to create a binary vector of length n with sum M
    def random_binary():
        vec = np.zeros(n, dtype=int)
        idx = rng.choice(n, M, replace=False)
        vec[idx] = 1
        return vec

    # Build initial candidate
    if init is not None:
        h0 = np.asarray(init, dtype=float)
        # threshold to binary
        h_bin = (h0 >= 0.5).astype(int)
        # adjust sum to exactly M
        cur_sum = h_bin.sum()
        if cur_sum > M:
            # flip some ones to zero
            excess = cur_sum - M
            ones = np.flatnonzero(h_bin)
            flip = rng.choice(ones, size=excess, replace=False)
            h_bin[flip] = 0
        elif cur_sum < M:
            deficit = M - cur_sum
            zeros = np.flatnonzero(h_bin == 0)
            flip = rng.choice(zeros, size=deficit, replace=False)
            h_bin[flip] = 1
        cur_best_h = h_bin.copy()
    else:
        cur_best_h = random_binary()

    # Evaluate function
    def evaluate(vec):
        # evaluate_erdos_solution expects array of floats
        return evaluate_erdos_solution(vec.astype(float))

    # Global best
    best_h = cur_best_h.copy()
    best_score = evaluate(best_h)

    # Parameters for inner SA
    decay = 0.9995
    min_temp = 1e-4
    reheat_temp = 1.0
    stagnation_limit = 20000  # iterations without improvement before restart

    # Main loop: keep trying new starts until time runs out
    while time.time() - start_time < budget_s - 0.5:
        # start from a random binary vector
        cur_h = random_binary()
        cur_score = evaluate(cur_h)

        temperature = reheat_temp
        it_without_improve = 0

        # inner SA loop
        while True:
            now = time.time()
            if now - start_time >= budget_s - 0.2:
                break

            # perform a swap (1 <-> 0)
            ones = np.flatnonzero(cur_h)
            zeros = np.flatnonzero(cur_h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i_one = rng.choice(ones)
            i_zero = rng.choice(zeros)
            # swap bits
            cur_h[i_one] = 0
            cur_h[i_zero] = 1

            new_score = evaluate(cur_h)
            delta = new_score - cur_score
            if delta < 0 or rng.random() < np.exp(-delta / temperature):
                # accept
                cur_score = new_score
                it_without_improve = 0 if delta < 0 else it_without_improve + 1
            else:
                # reject: revert swap
                cur_h[i_one] = 1
                cur_h[i_zero] = 0
                it_without_improve += 1

            # update global best
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
                it_without_improve = 0

            # temperature schedule
            temperature = max(min_temp, temperature * decay)
            if temperature <= min_temp * 10:
                temperature = reheat_temp

            # restart criteria
            if it_without_improve >= stagnation_limit:
                break

        # Continue to next random start (loop condition will check time)

    # Ensure output shape and type
    h_out = best_h.astype(float)
    final_score = evaluate(h_out)
    return (h_out, final_score, n)

import time
import numpy as np

def project_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the set {x | 0<=x_i<=1, sum x_i = target_sum}."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = 0.5 * (lo + hi)
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def triangular_initial(n: int) -> np.ndarray:
    """Create a triangular step shape (zero at ends, peak ~1 in middle)."""
    mid = n // 2
    i = np.arange(n)
    h = np.where(i <= mid, i / mid, (n - 1 - i) / mid).astype(float)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    init = globals().get('initial_h_values')
    if init is not None:
        n = int(np.asarray(init).size)
    else:
        n = 70
    M = n // 2  # required sum of h values

    # Helper to evaluate a candidate
    def evaluate(vec: np.ndarray) -> float:
        return evaluate_erdos_solution(vec.astype(float))

    # Build a set of initial candidates
    candidates = []

    # 1. Triangular shape, projected onto the feasible set
    tri = triangular_initial(n)
    tri = project_capped_simplex(tri, M)
    candidates.append(tri)

    # 2. If a parent construction is present, use it (projected if needed)
    if init is not None:
        h_parent = np.asarray(init, dtype=float)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        if not np.isclose(h_parent.sum(), M):
            h_parent = project_capped_simplex(h_parent, M)
        candidates.append(h_parent)

    # Evaluate initial candidates
    best_h = None
    best_score = np.inf
    for h in candidates:
        sc = evaluate(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    # Simulated annealing parameters
    temperature = 1.0
    decay = 0.9995
    min_temp = 1e-4
    reheat_temp = 1.0
    step_max = 0.2
    stagnation_limit = 30000
    it_without_improve = 0
    cur_h = best_h.copy()
    cur_score = best_score

    target_score = 0.3808  # stop early if we beat the target

    # Main SA loop
    while time.time() - start_time < budget_s - 0.2:
        # propose a move: transfer delta from j to i (maintain sum)
        i, j = rng.choice(n, size=2, replace=False)
        direction = rng.random() < 0.5  # True: increase i, decrease j
        if direction:
            max_delta = min(step_max, 1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            cur_h[i] += delta
            cur_h[j] -= delta
        else:
            max_delta = min(step_max, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            cur_h[i] -= delta
            cur_h[j] += delta

        new_score = evaluate(cur_h)
        delta_score = new_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta_score < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_score / temperature):
                accept = True

        if accept:
            cur_score = new_score
            it_without_improve = 0 if delta_score < 0 else it_without_improve + 1
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
                it_without_improve = 0
                if best_score <= target_score:
                    break
        else:
            # revert the move
            if direction:
                cur_h[i] -= delta
                cur_h[j] += delta
            else:
                cur_h[i] += delta
                cur_h[j] -= delta
            it_without_improve += 1

        # temperature schedule
        temperature = max(min_temp, temperature * decay)
        if temperature <= min_temp * 10:
            temperature = reheat_temp

        # restart if stagnated
        if it_without_improve >= stagnation_limit:
            cur_h = best_h.copy()
            cur_score = best_score
            temperature = reheat_temp
            it_without_improve = 0

    # Final projection to ensure feasibility (tiny numerical drift possible)
    best_h = project_capped_simplex(best_h, M)
    final_score = evaluate(best_h)

    return (best_h.astype(float), final_score, n)
