def anneal_refine(h, time_end):
    n = len(h)
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    steps = 0
    T0 = cur_score * 0.1 + 1e-4
    Tmin = 1e-6
    while time.time() < time_end:
        # propose small move
        i = np.random.randint(n)
        if cur_h[i] <= 0.0 or cur_h[i] >= 1.0:
            continue
        # propose small delta
        delta = (np.random.rand() - 0.5) * 0.2  # +/-0.1
        # find j different from i
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        # adjust i and j to keep sum unchanged
        hi = cur_h[i]
        hj = cur_h[j]
        # Determine feasible delta maxima
        delta_max = min(1 - hi, hj)  # for positive delta
        delta_min = -min(hi, 1 - hj) # for negative delta
        # Clip delta to feasible range
        delta = np.clip(delta, delta_min, delta_max)
        if delta == 0:
            continue
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_score = compute_c5(new_h)
        T = max(Tmin, T0 * (0.9995 ** steps))
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_score - cur_score) / T):
                accept = True
        if accept:
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        steps += 1
    return best_h, best_score

import time
import math
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _repair(h):
    n = h.shape[0]
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s == 0.0:
        h[:int(target)] = 1.0
        return h
    factor = target / s
    h = np.clip(h * factor, 0.0, 1.0)
    # tiny residual fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _random_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _optimize(initial_h, n, time_deadline):
    # objective
    def obj(x):
        return _c5_of(x)
    # equality constraint sum(h)=n/2
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0},)
    bounds = [(0.0, 1.0)] * n
    best_h = _repair(initial_h)
    best_val = obj(best_h)
    # single SLSQP run (quick)
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 150, 'ftol': 1e-9, 'disp': False})
        if res.success:
            cand = _repair(res.x)
            val = obj(cand)
            if val < best_val:
                best_val, best_h = val, cand
    except Exception:
        pass
    return best_h, best_val

def _anneal_refine(h, time_deadline):
    n = h.shape[0]
    cur = h.copy()
    cur_val = _c5_of(cur)
    best = cur.copy()
    best_val = cur_val
    steps = 0
    T0 = cur_val * 0.1 + 1e-4
    Tmin = 1e-6
    while time.time() < time_deadline:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        hi, hj = cur[i], cur[j]
        # propose a delta that keeps both within [0,1]
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = (np.random.rand() - 0.5) * (delta_max - delta_min) * 0.5
        delta = np.clip(delta, delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        new = cur.copy()
        new[i] = hi + delta
        new[j] = hj - delta
        new_val = _c5_of(new)
        T = max(Tmin, T0 * (0.9995 ** steps))
        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            cur, cur_val = new, new_val
            if cur_val < best_val:
                best, best_val = cur.copy(), cur_val
        steps += 1
    return best, best_val
