    hi = h[i]

    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new, (i, j, delta)

def adjust_sum(v, target):
    diff = target - np.sum(v)
    # if diff > 0: increase some elements that are <1
    # if diff < 0: decrease some elements that >0
    # We'll distribute diff evenly across all free entries.
    # Loop:
    free = (v > 0) & (v < 1)
    nfree = np.count_nonzero(free)
    if nfree == 0:
        # all entries saturated, fallback: adjust a random index within bounds
        idx = np.random.randint(len(v))
        v[idx] = np.clip(v[idx] + diff, 0, 1)
        return v
    adjustment = diff / nfree
    v[free] += adjustment
    # clip again and if overshoot, iterate.
    v = np.clip(v, 0, 1)
    # repeat few times
    for _ in range(10):
        cur = np.sum(v)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        adjustment = diff / nfree
        v[free] += adjustment
        v = np.clip(v, 0, 1)
    return v

import time
import numpy as np
from math import exp

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project the vector v onto the feasible set:
    0 <= v[i] <= 1 and sum(v) == target_sum.
    Simple iterative water‑filling algorithm.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(12):  # few passes are enough for small n
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            # all entries are at a bound; adjust a random free slot forcibly
            idx = np.random.randint(len(v))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
            break
        adjustment = diff / nfree
        x[free] += adjustment
        x = np.clip(x, 0.0, 1.0)
    return x

def _max_overlap(h: np.ndarray) -> float:
    """return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated discretely."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Build an initial feasible vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
        target_sum = n / 2.0
        h = _project_to_constraints(h0, target_sum)
    else:
        # a simple triangular “tent” shape, then project to the exact constraints
        n = 200                     # reasonable size (<1k)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        target_sum = n / 2.0
        h = _project_to_constraints(h, target_sum)

    dx = 2.0 / n
    best_h = h.copy()
    best_cost = _max_overlap(best_h)

    current_h = h.copy()
    current_cost = best_cost

    # ------------------------------------------------------------------
    # 2. Simple simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start_time = time.time()
    max_step_initial = 0.2          # maximal mass transferred in one move
    T0 = 0.02                       # initial temperature for occasional uphill moves

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.94:            # keep a safety margin
            break

        # linear cooling
        frac = elapsed / (budget_s * 0.94)
        max_step = max_step_initial * (1.0 - frac)
        temperature = max(T0 * (1.0 - frac), 1e-8)

        # --------------------------------------------------------------
        # 2a. Choose move type: targeted (60%) or simple two‑index (40%)
        # --------------------------------------------------------------
        if rng.random() < 0.6:
            # Targeted move: reduce the worst shift (maximum overlap)
            corr = np.correlate(current_h, 1.0 - current_h, mode='full')
            worst_idx = int(np.argmax(corr))
            lag = worst_idx - (n - 1)               # signed shift
            if lag == 0:
                # fall back to a simple two‑index transfer when lag=0
                i, j = rng.choice(n, size=2, replace=False)
                hi, hj = current_h[i], current_h[j]
                low = max(-max_step, -hi, hj - 1.0)
                high = min(max_step, 1.0 - hi, hj)
                if low >= high:
                    continue
                delta = rng.random() * (high - low) + low
                new_h = current_h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta
            else:
                # indices that actually overlap for this lag
                i_start = max(0, -lag)
                i_end = min(n, n - lag)           # exclusive
                if i_end <= i_start:
                    continue
                i = rng.integers(i_start, i_end)
                j = i + lag
                # third index to take the compensating mass from
                k = rng.integers(0, n)
                while k == i or k == j:
                    k = rng.integers(0, n)

                # feasible delta so that bounds stay respected
                delta_max = min(
                    max_step,
                    1.0 - current_h[i],
                    1.0 - current_h[j],
                    current_h[k] / 2.0
                )
                if delta_max <= 0.0:
                    continue
                delta = rng.random() * delta_max
                new_h = current_h.copy()
                new_h[i] += delta
                new_h[j] += delta
                new_h[k] -= 2.0 * delta
        else:
            # Simple two‑index exchange
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = current_h[i], current_h[j]
            low = max(-max_step, -hi, hj - 1.0)
            high = min(max_step, 1.0 - hi, hj)
            if low >= high:
                continue
            delta = rng.random() * (high - low) + low
            new_h = current_h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta

        # --------------------------------------------------------------
        # 2b. Evaluate the new candidate
        # --------------------------------------------------------------
        new_cost = _max_overlap(new_h)

        # --------------------------------------------------------------
        # 2c. Acceptance rule
        # --------------------------------------------------------------
        if new_cost < current_cost or rng.random() < exp(-(new_cost - current_cost) / temperature):
            current_h = new_h
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = new_h.copy()
                # early exit if we already beat the target bound
                if best_cost < 0.3808:
                    break

    # ------------------------------------------------------------------
    # 3. Final clean‑up – enforce exact sum (tiny floating error possible)
    # ------------------------------------------------------------------
    target_sum = n / 2.0
    total = best_h.sum()
    if abs(total - target_sum) > 1e-9:
        diff = target_sum - total
        # adjust a random non‑saturated entry
        for _ in range(10):
            idx = rng.integers(0, n)
            new_val = np.clip(best_h[idx] + diff, 0.0, 1.0)
            diff = diff - (new_val - best_h[idx])
            best_h[idx] = new_val
            if abs(diff) < 1e-12:
                break

    return best_h.tolist(), float(best_cost), n
