def anneal(initial_h):
    h = initial_h.copy()
    n = len(h)
    current_score = compute_c5(h)
    best_h_local = h.copy()
    best_score_local = current_score
    iteration = 0
    T0 = 0.01  # start temperature
    # Use schedule
    while time.time() - start_time < max_time:
        iteration += 1
        # temperature schedule
        T = T0 * (0.9995 ** iteration)
        # propose swap
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        delta = new_score - current_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            current_score = new_score
            if new_score < best_score_local:
                best_score_local = new_score
                best_h_local = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h_local, best_score_local

import time
import numpy as np
import random

def _compute_c5(h, dx):
    """
    Compute the C5 bound for a binary (or fractional) step function h.
    C5 = max_k ∫ h(x) (1 - h(x + k)) dx, approximated by discrete correlation.
    """
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _random_interval_pattern(n, total_ones, max_intervals=8):
    """
    Generate a binary pattern of length n with exactly total_ones ones,
    arranged as a random number (1..max_intervals) of non‑overlapping intervals.
    """
    k = np.random.randint(1, max_intervals + 1)  # number of intervals
    # ---- lengths of intervals (positive integers summing to total_ones) ----
    if k == 1:
        lengths = np.array([total_ones], dtype=int)
    else:
        # choose k-1 cut points among 1..total_ones-1
        cuts = np.sort(np.random.choice(total_ones - 1, k - 1, replace=False))
        lengths = np.diff(np.concatenate(([0], cuts, [total_ones])))
    # ---- gaps between intervals (including leading and trailing gaps) ----
    total_gap = n - total_ones
    # split total_gap into k+1 non‑negative parts
    if k == 0:
        gaps = np.array([total_gap], dtype=int)
    else:
        gap_cuts = np.sort(np.random.choice(total_gap + 1, k, replace=False))
        gaps = np.diff(np.concatenate(([0], gap_cuts, [total_gap])))
    # ---- construct binary pattern ----
    h = np.zeros(n, dtype=float)
    pos = gaps[0]
    for idx in range(k):
        L = int(lengths[idx])
        h[pos:pos + L] = 1.0
        pos += L + int(gaps[idx + 1])
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for an Erdős minimum overlap step function.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    max_time = budget_s * 0.95  # safe margin

    # ----- problem size -----
    n = 256                     # number of sample points (must be even)
    if n % 2:
        n += 1
    total_ones = n // 2
    dx = 2.0 / n

    # ----- helper to evaluate score -----
    def score(arr):
        return _compute_c5(arr, dx)

    # ----- initial solution -----
    # start from a random binary pattern
    best_h = np.zeros(n, dtype=float)
    best_h[:total_ones] = 1.0
    np.random.shuffle(best_h)
    best_score = score(best_h)

    # if a parent construction is provided, use it (converted to binary)
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_old, init)
        interp = np.clip(interp, 0.0, 1.0)
        # take the total_ones largest entries as ones
        idx = np.argsort(-interp)[:total_ones]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        s = score(h_bin)
        if s < best_score:
            best_h = h_bin
            best_score = s

    # ----- random interval search -----
    iter_count = 0
    # main loop: keep generating random interval patterns
    while time.time() - start_time < max_time:
        cand = _random_interval_pattern(n, total_ones, max_intervals=8)
        s = score(cand)
        if s < best_score:
            best_score = s
            best_h = cand.copy()
        iter_count += 1
        # occasional local refinement on the current best
        if iter_count % 1000 == 0:
            # simple hill‑climbing using swaps (fixed budget)
            h_local = best_h.copy()
            local_score = best_score
            for _ in range(200):
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                if h_local[i] == h_local[j]:
                    continue
                # swap
                h_local[i], h_local[j] = h_local[j], h_local[i]
                ns = score(h_local)
                if ns < local_score:
                    local_score = ns
                else:
                    h_local[i], h_local[j] = h_local[j], h_local[i]
            if local_score < best_score:
                best_score = local_score
                best_h = h_local.copy()

    # ----- final hill‑climbing phase (use remaining time) -----
    while time.time() - start_time < max_time:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if best_h[i] == best_h[j]:
            continue
        # swap
        best_h[i], best_h[j] = best_h[j], best_h[i]
        ns = score(best_h)
        if ns < best_score:
            best_score = ns
        else:
            # revert
            best_h[i], best_h[j] = best_h[j], best_h[i]

    # Ensure feasibility (exact sum constraint)
    # The algorithm preserves the number of ones, but a final safety check:
    ones = np.sum(best_h)
    if not np.isclose(ones, total_ones):
        # fix by adjusting the smallest/largest entries
        # (should rarely happen, but we correct it)
        diff = int(round(total_ones - ones))
        if diff > 0:
            zero_idx = np.where(best_h == 0)[0]
            chosen = np.random.choice(zero_idx, diff, replace=False)
            best_h[chosen] = 1.0
        elif diff < 0:
            one_idx = np.where(best_h == 1)[0]
            chosen = np.random.choice(one_idx, -diff, replace=False)
            best_h[chosen] = 0.0

    return best_h, best_score, n
