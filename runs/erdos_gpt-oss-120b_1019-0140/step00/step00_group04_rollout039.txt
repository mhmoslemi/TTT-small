def solve_milp(n):
    import cvxpy as cp, numpy as np
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(x) == n//2]
    # list to hold each shift sum constraint
    for k in range(-(n-1), n):
        start_i = max(0, -k)
        end_i = min(n, n - k)  # exclusive bound
        L = end_i - start_i
        if L == 0:
            continue
        # create y variables for this shift
        y = cp.Variable(L, boolean=True)
        # linking constraints
        for idx, i in enumerate(range(start_i, end_i)):
            j = i + k
            constraints.append(y[idx] <= x[i])
            constraints.append(y[idx] <= x[j])
            constraints.append(y[idx] >= x[i] + x[j] - 1)
        constraints.append(cp.sum(y) >= t)
    # maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
    # if feasible, extract x_vals as solution
    if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
        x_vals = np.round(x.value).astype(float)  # ensure 0/1
        t_val = int(round(t.value))
        # compute c = #ones - t (since common ones minimal t)
        # but careful: t is common1 for each shift, minimal intersection count
        # Actually t is min_{k} sum_i x_i x_{i+k}. So a typical shift may have intersection >= t.
        # Max mismatches = #ones - t (since #ones = n/2)
        max_mismatch = n//2 - t_val
        dx = 2.0 / n
        c5 = max_mismatch * dx
        return x_vals, c5, n
    else:
        return None, None, n

def local_search_binary(n, h_init, time_budget):
    rng = np.random.default_rng()
    h = h_init.copy()
    # ensure binary
    h = (h > 0.5).astype(float)
    # adjust weight
    target_ones = n//2
    cur_ones = int(h.sum())
    if cur_ones != target_ones:
        idx = np.where(h==1)[0]
        # adjust
        while cur_ones > target_ones:
            i = rng.choice(idx)
            h[i] = 0
            idx = np.where(h==1)[0]
            cur_ones -= 1
        idx0 = np.where(h==0)[0]
        while cur_ones < target_ones:
            i = rng.choice(idx0)
            h[i] = 1
            idx0 = np.where(h==0)[0]
            cur_ones += 1
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5

    start = time.time()
    T0 = 0.5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # temperature decreasing
        T = max(T0 * (1 - elapsed / time_budget), 1e-4)
        # propose swap
        # choose random i in ones, j in zeros
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5
```

import time, math, numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def _binary_from_initial(arr, rng):
    """Convert any initial array to a feasible binary vector of the same length."""
    n = arr.shape[0]
    bin_arr = (arr >= 0.5).astype(float)
    target_ones = n // 2
    cur_ones = int(bin_arr.sum())
    if cur_ones > target_ones:
        ones_idx = np.where(bin_arr == 1)[0]
        flip = rng.choice(ones_idx, cur_ones - target_ones, replace=False)
        bin_arr[flip] = 0.0
    elif cur_ones < target_ones:
        zeros_idx = np.where(bin_arr == 0)[0]
        flip = rng.choice(zeros_idx, target_ones - cur_ones, replace=False)
        bin_arr[flip] = 1.0
    return bin_arr

def _upscale_binary(bin_arr, target_n, rng):
    """Repeat pattern to reach target_n points and repair weight."""
    cur_n = bin_arr.shape[0]
    if target_n == cur_n:
        return bin_arr.copy()
    repeats = target_n // cur_n
    remainder = target_n % cur_n
    up = np.concatenate([np.repeat(bin_arr, repeats), bin_arr[:remainder]])
    target_ones = target_n // 2
    cur_ones = int(up.sum())
    if cur_ones != target_ones:
        if cur_ones > target_ones:
            ones_idx = np.where(up == 1)[0]
            flip = rng.choice(ones_idx, cur_ones - target_ones, replace=False)
            up[flip] = 0.0
        else:
            zeros_idx = np.where(up == 0)[0]
            flip = rng.choice(zeros_idx, target_ones - cur_ones, replace=False)
            up[flip] = 1.0
    return up

def _solve_milp(n):
    """Solve binary ILP: maximize the minimal intersection t across all shifts."""
    import cvxpy as cp
    x = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [
        cp.sum(x) == n // 2,
        t >= 0,
        t <= n // 2,
    ]
    for k in range(-n + 1, n):
        i_start = max(0, -k)
        i_end = min(n, n - k)
        L = i_end - i_start
        if L <= 0:
            continue
        y = cp.Variable(L, boolean=True)
        idx = 0
        for i in range(i_start, i_end):
            j = i + k
            constraints.append(y[idx] <= x[i])
            constraints.append(y[idx] <= x[j])
            constraints.append(y[idx] >= x[i] + x[j] - 1)
            idx += 1
        constraints.append(cp.sum(y) >= t)
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
    except Exception:
        return None, None, n
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None, n
    xv = np.rint(x.value).astype(float)
    tval = int(round(t.value))
    max_mismatch = (n // 2) - tval
    dx = 2.0 / n
    c5 = max_mismatch * dx
    return xv, c5, n

def _local_search_binary(n, h_init, time_budget, rng):
    """Simple simulated annealing on binary vectors (swap 1/0)."""
    h = h_init.copy()
    # ensure binary feasibility
    h = (h > 0.5).astype(float)
    target_ones = n // 2
    cur = int(h.sum())
    if cur != target_ones:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if cur > target_ones:
            flip = rng.choice(ones, cur - target_ones, replace=False)
            h[flip] = 0.0
        else:
            flip = rng.choice(zeros, target_ones - cur, replace=False)
            h[flip] = 1.0
    best_h, best_c5 = h.copy(), _compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T0 = 0.6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / time_budget), 1e-4)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i], h[j] = 0.0, 1.0
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine base size
    if "initial_h_values" in globals():
        base_arr = np.asarray(initial_h_values, dtype=float)
        n0 = base_arr.shape[0]
    else:
        n0 = 42
        base_arr = None
    # Build an initial binary candidate
    if base_arr is not None:
        h0 = _binary_from_initial(base_arr, rng)
    else:
        h0 = np.zeros(n0, dtype=float)

        h0[ones] = 1.0
    start_total = time.time()
    best_h, best_c5, best_n = h0.copy(), _compute_c5(h0), n0
    TARGET = 0.3808
    # candidate sizes (even, modest, up to 100)
    cand_sizes = [n0]
    inc = 6
    while cand_sizes[-1] + inc <= 100:
        cand_sizes.append(cand_sizes[-1] + inc)
    # ensure we also try the double size if reasonable
    if n0 * 2 <= 100 and n0 * 2 not in cand_sizes:
        cand_sizes.append(n0 * 2)
    cand_sizes = sorted(set(cand_sizes))
    # main loop: try ILP on each size, stop when good enough or out of time
    for n in cand_sizes:
        if time.time() - start_total > budget_s * 0.85:
            break
        # obtain a binary start for this size
        if n == n0:
            start_bin = h0.copy()
        else:
            start_bin = _upscale_binary(h0, n, rng)
        # attempt MILP

        if sol_h is not None and sol_c5 < best_c5:
            best_h, best_c5, best_n = sol_h, sol_c5, sol_n
            if best_c5 <= TARGET:
                break
        # if MILP failed or not good, fall back to local search on this size
        # allocate a modest slice of remaining time
        remaining = budget_s - (time.time() - start_total) - 1.0
        if remaining > 4.0:
            ls_h, ls_c5 = _local_search_binary(n, start_bin, remaining * 0.5, rng)

    # final safety: if still not good enough, run a short local search on the current best
    if best_c5 > TARGET:
        remaining = budget_s - (time.time() - start_total) - 0.5
        if remaining > 2.0:
            best_h, best_c5 = _local_search_binary(best_n, best_h, remaining, rng)
    return best_h, best_c5, best_n
