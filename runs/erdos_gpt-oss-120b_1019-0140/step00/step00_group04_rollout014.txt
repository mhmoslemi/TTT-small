def propose_move(h, s, max_delta=0.05):
    # find index i giving max contribution. Use vectorized ops.
    if s >= 0:
        i_range = np.arange(0, n - s)
        j = i_range + s
    else:
        i_range = np.arange(-s, n)  # s negative, -s positive offset?
        j = i_range + s # j < i
    contrib = h[i_range] * (1 - h[j])
    i_best = i_range[np.argmax(contrib)]
    j_best = i_best + s
    # compute max allowed delta:
    max_move = min(h[i_best], 1 - h[j_best])
    if max_move <= 1e-12:
        return None
    delta = min(max_move, max_delta)
    # optionally randomize delta maybe scaled to max_move
    return i_best, j_best, delta

def rebalance_sum(h, target_sum):
    diff = target_sum - h.sum()
    # if diff small, return
    if abs(diff) < 1e-12:
        return h
    # apply diff to elements with capacity
    if diff > 0:
        # increase entries not at 1
        # compute capacities:
        caps = 1 - h
        total_cap = caps.sum()
        if total_cap < diff - 1e-12:
            # if not enough capacity, clamp and remainder will be zero
            diff = total_cap
        # allocate proportionally:
        h += caps / total_cap * diff
    else:
        # decrease entries not at 0
        caps = h
        total_cap = caps.sum()
        need = -diff
        if total_cap < need - 1e-12:
            need = total_cap
        h -= caps / total_cap * need
    # Ensure bounds due to floating errors:
    np.clip(h, 0, 1, out=h)
    return h

import time, numpy as np, random, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def rebalance_sum(h, target_sum):
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        caps = 1 - h
        total = caps.sum()
        if total < diff:
            diff = total
        if total > 0:
            h += caps / total * diff
    else:
        caps = h
        total = caps.sum()
        need = -diff
        if total < need:
            need = total
        if total > 0:
            h -= caps / total * need
    np.clip(h, 0, 1, out=h)
    return h

def propose_move_shift(h, shift, max_delta):
    n = h.shape[0]
    if shift > 0:
        i_range = np.arange(0, n - shift)
        j_range = i_range + shift
    elif shift < 0:
        i_range = np.arange(-shift, n)
        j_range = i_range + shift
    else:
        return None
    contrib = h[i_range] * (1 - h[j_range])
    # Get order of indices descending by contribution
    order = np.argsort(-contrib)
    for idx in order:
        i = i_range[idx]
        j = j_range[idx]
        max_move = min(h[i], 1 - h[j])
        if max_move > 1e-12:
            delta = min(max_move, max_delta)
            return int(i), int(j), float(delta)
    return None

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, numpy as np, math
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)
    # Determine target number of points
    target_n = 256
    # Build initial h
    if 'initial_h_values' in globals():
        old_h = np.array(initial_h_values, dtype=float)
        n_old = old_h.shape[0]
        xs_old = np.linspace(0, 2, n_old, endpoint=False)
        xs_new = np.linspace(0, 2, target_n, endpoint=False)
        h = np.interp(xs_new, xs_old, old_h)
    else:
        h = np.full(target_n, 0.5, dtype=float)
    n = target_n
    target_sum = n / 2.0
    h = rebalance_sum(h, target_sum)
    # Parameters for simulated annealing
    temp = 0.05
    temp_decay = 0.9999
    max_delta0 = 0.1
    max_delta_decay = 0.9999
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # Main loop
    iter_count = 0
    no_improve = 0
    while True:
        now = time.time()
        if now - start > budget_s * 0.98:
            break
        # compute current score
        cur_score = compute_c5(h)
        # update best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # find worst shift
        corr_full = np.correlate(h, 1 - h, mode='full')
        idx_max = int(np.argmax(corr_full))
        shift = idx_max - (n - 1)
        move = None
        if shift != 0:
            move = propose_move_shift(h, shift, max_delta0)
        if move is None:
            # random move
            i, j = random.sample(range(n), 2)
            # ensure we can transfer from i to j
            if h[i] > 0 and h[j] < 1:
                max_move = min(h[i], 1 - h[j])
                delta = min(max_move, max_delta0)
                move = (i, j, delta)
            else:

                if h[j] > 0 and h[i] < 1:
                    max_move = min(h[j], 1 - h[i])
                    delta = min(max_move, max_delta0)
                    move = (j, i, delta)
        if move is None:
            # nothing feasible, break
            break
        i, j, delta = move
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # numeric safety clip
        np.clip(new_h, 0.0, 1.0, out=new_h)
        # adjust sum if drift
        new_h = rebalance_sum(new_h, target_sum)
        new_score = compute_c5(new_h)
        # acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            h = new_h
            cur_score = new_score
        # annealing schedule
        temp *= temp_decay
        max_delta0 *= max_delta_decay
        iter_count += 1
        # periodic repair
        if iter_count % 5000 == 0:

            h = rebalance_sum(h, target_sum)
            # random perturbation if stuck

                # apply few random moves

                        h[j] += delta
                h = rebalance_sum(h, target_sum)
                no_improve = 0
    # final best
    final_score = compute_c5(best_h)
    return (best_h, float(final_score), int(n))

import time
import random
import math
import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for discretized h."""
    # np.correlate gives the sum without the dx factor
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def _rebalance_sum(h, target_sum):
    """Adjust h in-place so that its sum equals target_sum while staying in [0,1]."""
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        caps = 1.0 - h
        total = caps.sum()
        if total < diff:
            diff = total
        if total > 0:
            h += caps / total * diff
    else:
        caps = h
        total = caps.sum()
        need = -diff
        if total < need:
            need = total
        if total > 0:
            h -= caps / total * need
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _propose_move_shift(h, shift, max_delta):
    """Find a pair (i,j=i+shift) with large contribution h[i]*(1-h[j]) and
    return a feasible move (i,j,delta) limited by max_delta."""
    n = h.shape[0]
    if shift > 0:
        i_range = np.arange(0, n - shift)
        j_range = i_range + shift
    elif shift < 0:
        i_range = np.arange(-shift, n)
        j_range = i_range + shift
    else:
        return None
    contrib = h[i_range] * (1.0 - h[j_range])
    order = np.argsort(-contrib)  # descending
    for idx in order:
        i = int(i_range[idx])
        j = int(j_range[idx])
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 1e-12:
            delta = min(max_move, max_delta)
            return i, j, float(delta)
    return None

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # target resolution (kept modest for speed)
    n = 256
    dx = 2.0 / n
    target_sum = n / 2.0

    # initialise h from the provided construction if available
    if "initial_h_values" in globals():
        old = np.array(initial_h_values, dtype=float)
        n_old = old.shape[0]
        xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xs_new, xs_old, old)
    else:
        h = np.full(n, 0.5, dtype=float)

    h = _rebalance_sum(h, target_sum)

    # simulated‑annealing parameters
    temp = 0.05            # initial temperature
    temp_decay = 0.9999
    max_delta = 0.1        # initial step size
    delta_decay = 0.9999

    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    iter_cnt = 0
    stagnant = 0

    while True:
        # time check
        if time.time() - start > budget_s * 0.98:
            break

        # current correlation and score
        corr_full = np.correlate(h, 1 - h, mode="full")
        cur_score = np.max(corr_full) * dx

        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        # locate the shift giving the worst overlap
        idx_max = int(np.argmax(corr_full))
        shift = idx_max - (n - 1)

        move = None
        if shift != 0:
            move = _propose_move_shift(h, shift, max_delta)

        # fall back to a random feasible transfer if needed
        if move is None:
            i, j = random.sample(range(n), 2)
            if h[i] > 0.0 and h[j] < 1.0:
                allow = min(h[i], 1.0 - h[j])
                delta = min(allow, max_delta)
                move = (i, j, delta)
            elif h[j] > 0.0 and h[i] < 1.0:
                allow = min(h[j], 1.0 - h[i])
                delta = min(allow, max_delta)
                move = (j, i, delta)

        if move is None:
            # no feasible move found – terminate
            break

        i, j, delta = move
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        np.clip(new_h, 0.0, 1.0, out=new_h)
        new_h = _rebalance_sum(new_h, target_sum)

        new_score = _compute_c5(new_h, dx)

        # Metropolis acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            h = new_h
            cur_score = new_score

        # annealing schedule
        temp *= temp_decay
        max_delta *= delta_decay
        iter_cnt += 1

        # occasional re‑balancing and shake‑up if stuck
        if iter_cnt % 5000 == 0:
            h = _rebalance_sum(h, target_sum)
            if stagnant > 20000:
                # restart from the best found with a few random perturbations
                h = best_h.copy()
                for _ in range(10):
                    ii, jj = random.sample(range(n), 2)
                    if h[ii] > 0.0 and h[jj] < 1.0:
                        allow = min(h[ii], 1.0 - h[jj])
                        d = allow * random.random()
                        h[ii] -= d
                        h[jj] += d
                h = _rebalance_sum(h, target_sum)
                stagnant = 0

    final_score = _compute_c5(best_h, dx)
    return best_h, float(final_score), int(n)
