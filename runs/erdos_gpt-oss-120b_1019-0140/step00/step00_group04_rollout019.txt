    return dx * np.max(np.correlate(h, 1 - h, mode='full'))

def repair(vec):
    # ensure within [0,1], sum = target
    n = vec.shape[0]
    target = n / 2.0
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    # simple scaling to target while within bounds
    # We'll iterate a few times.
    for _ in range(10):
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # Determine free indices not at bounds
        free = (v > 0) & (v < 1)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            # all at bounds, uniform adjust all
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
        total = v.sum()
    # final scaling if needed
    total = v.sum()
    if abs(total - target) > 1e-8:
        v *= target / total
        v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np

def _repair(vec):
    """Project vec onto [0,1]^n with sum = n/2."""
    n = vec.shape[0]
    target = n / 2.0
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(10):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            # all at bounds, distribute uniformly
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) > 1e-8:
        # small scaling fix
        scale = target / total if total != 0 else 0.0
        v *= scale
        v = np.clip(v, 0.0, 1.0)
    return v

def _score(vec, dx):
    """Return C5 bound for a given vector."""
