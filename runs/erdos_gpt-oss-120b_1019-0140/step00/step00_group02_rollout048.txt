import time
import random
import math
import numpy as np

def compute_c5(vec, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for discrete vector."""
    # np.correlate gives sum_i h[i] * (1-h[i+shift])
    return float(np.max(np.correlate(vec, 1.0 - vec, mode="full")) * dx)

def repair_sum(vec, target_sum):
    """Project vec onto the hyperplane sum(vec)=target_sum respecting 0≤vec≤1."""
    vec = np.clip(vec, 0.0, 1.0)
    diff = target_sum - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # Distribute the difference proportionally to the available capacity.
    if diff > 0:
        cap = 1.0 - vec
    else:
        cap = vec
    cap_sum = cap.sum()
    if cap_sum > 0:
        vec = vec + diff * cap / cap_sum
    vec = np.clip(vec, 0.0, 1.0)
    # Small residual correction (biases may appear after clipping)
    residual = target_sum - vec.sum()
    if abs(residual) > 1e-12:
        # distribute uniformly over non‑bound entries
        if residual > 0:
            cap = 1.0 - vec
        else:
            cap = vec
        cap_sum = cap.sum()
        if cap_sum > 0:
            vec = vec + residual * cap / cap_sum
        else:
            vec = vec + residual / len(vec)
        vec = np.clip(vec, 0.0, 1.0)
    # Final exact fix by adjusting a single index (if needed)
    final_diff = target_sum - vec.sum()
    if abs(final_diff) > 1e-12:
        for i in range(len(vec)):
            if final_diff > 0 and vec[i] < 1.0:
                add = min(final_diff, 1.0 - vec[i])
                vec[i] += add
                final_diff -= add
            elif final_diff < 0 and vec[i] > 0.0:
                sub = min(-final_diff, vec[i])
                vec[i] -= sub
                final_diff += sub
            if abs(final_diff) < 1e-12:
                break
    return vec

def propose_pair_move(vec):
    """Return a new vector where mass epsilon is transferred between two indices,
    preserving the total sum and staying in [0,1]. Returns None if no feasible move."""
    n = vec.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = vec[i], vec[j]
    eps_low = max(-hi, hj - 1.0)
    eps_up = min(1.0 - hi, hj)
    if eps_low >= eps_up:
        return None
    eps = np.random.uniform(eps_low, eps_up)
    new = vec.copy()
    new[i] = hi + eps
    new[j] = hj - eps
    return new
