import time
import numpy as np

def _repair(vec, target_ones, rng):
    """Flip bits to enforce exactly target_ones ones."""
    s = int(vec.sum())
    if s == target_ones:
        return vec
    diff = s - target_ones
    if diff > 0:
        # too many ones → turn some to zero
        ones_idx = np.where(vec == 1)[0]
        to_zero = rng.choice(ones_idx, size=diff, replace=False)
        vec[to_zero] = 0
    else:
        # too few ones → turn some zeros to one
        zeros_idx = np.where(vec == 0)[0]
        to_one = rng.choice(zeros_idx, size=-diff, replace=False)
        vec[to_one] = 1
    return vec

def _crossover(p1, p2, rng):
    """One‑point crossover preserving length."""
    n = p1.size
    cp = rng.integers(0, n + 1)  # cut point can be 0..n
    child = np.concatenate((p1[:cp], p2[cp:])).astype(np.uint8)
    return child

def _mutate_swap(vec, rng):
    """Swap a random 1 with a random 0 (keeps sum unchanged)."""
    ones = np.where(vec == 1)[0]
    zeros = np.where(vec == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return vec
    i1 = rng.choice(ones)
    i0 = rng.choice(zeros)
    vec[i1] = 0
    vec[i0] = 1
    return vec

def _tournament_select(pop, fitnesses, tournament_k, rng):
    """Return the best individual among k randomly chosen competitors."""
    idxs = rng.choice(len(pop), size=tournament_k, replace=False)
    best_i = idxs[0]
    best_f = fitnesses[best_i]
    for i in idxs[1:]:
        if fitnesses[i] < best_f:      # lower C5 is better
            best_f = fitnesses[i]
            best_i = i
    return pop[best_i]

def _evaluate(vec):
    """Compute the Erdős overlap bound for a candidate vector."""
    return evaluate_erdos_solution(vec.astype(float))
