def propose_move(h):
    n = len(h)
    # find candidate i with h[i] > 0
    # find candidate j with h[j] < 1
    # choose randomly
    # we can precompute lists every iteration but small n.

    # Let's build candidate i indices list

    idx_j = np.where(h < 1 - 1e-12)[0]
    if idx_i.size == 0 or idx_j.size == 0:
        return h, 0.0, 0, 0
    i = np.random.choice(idx_i)
    j = np.random.choice(idx_j)
    # Ensure i != j
    while j == i:
        j = np.random.choice(idx_j)
    # max delta we can move from i to j
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        return h, 0.0, i, j
    # choose delta random (uniform or scaled)
    delta = np.random.rand() * max_delta
    # make copy
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    # maybe enforce small epsilon
    return new_h, delta, i, j

def run(seed=42, budget_s=1100, **kwargs):

    import numpy as np

    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Helper functions
    # ----------------------------------------------------------------------
    def compute_c5_and_corr(arr):
        """Return (C5 value, full correlation array) for a given h."""
        corr = np.correlate(arr, 1.0 - arr, mode='full')
        c5 = float(corr.max()) * (2.0 / arr.shape[0])
        return c5, corr

    def repair_sum(vec, target):
        """Adjust a single entry to enforce exact sum while staying in [0,1]."""
        diff = target - vec.sum()
        if abs(diff) < 1e-12:
            return vec
        # try to fix a non‑boundary entry
        if diff > 0:
            idx = np.where(vec < 1.0 - 1e-12)[0]
        else:
            idx = np.where(vec > 1e-12)[0]
        if idx.size:
            i = idx[0]
            vec[i] = min(max(vec[i] + diff, 0.0), 1.0)
        else:
            # fallback: spread uniformly (tiny error remains)
            vec += diff / vec.size
            vec = np.clip(vec, 0.0, 1.0)
        return vec

    # ----------------------------------------------------------------------

    try:
