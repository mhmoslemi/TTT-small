                cur_sum = h.sum()
                if cur_sum > 0:
                    h *= target_sum / cur_sum
                else:
                    h = np.full(n, target_sum / n)

    h = h.astype(float)
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Optimization loop (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    max_step_initial = 0.5
    min_step = 0.001
    target_c5 = 0.3808

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.1:
            break
        # cooling schedules
        frac = elapsed / budget_s
        step = max(min_step, max_step_initial * (1.0 - frac))
        T = max(1e-6, 0.01 * (1.0 - frac))

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # feasible delta range for transferring mass from j to i
        lo = max(-step, -cur_h[i], cur_h[j] - 1.0)
        hi = min(step, 1.0 - cur_h[i], cur_h[j])
        if lo >= hi:
            continue
        delta = rng.uniform(lo, hi)

        # apply move
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c5 = _compute_c5(cur_h, dx)

        # acceptance test
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 <= target_c5:
                    break
        else:
            # revert move
            cur_h[i] -= delta
            cur_h[j] += delta

    final_c5 = _compute_c5(best_h, dx)
    return best_h.tolist(), final_c5, n
```
