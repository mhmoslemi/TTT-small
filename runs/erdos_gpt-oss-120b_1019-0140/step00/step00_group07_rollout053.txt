def _shift_zero(arr, shift):
    n = arr.shape[0]
    if shift >= n or shift <= -n:
        return np.zeros_like(arr)
    if shift > 0:
        return np.concatenate((np.zeros(shift), arr[:n - shift]))
    if shift < 0:
        s = -shift
        return np.concatenate((arr[s:], np.zeros(s)))
    return arr.copy()

def _project_box_sum(x, target):
    # find λ such that sum(clipped(x+λ)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

            h = _project_box_sum(h, target_sum)
            last_improve = it

    final_c5 = _c5_value(best_h, dx)
    return best_h.tolist(), float(final_c5), N
