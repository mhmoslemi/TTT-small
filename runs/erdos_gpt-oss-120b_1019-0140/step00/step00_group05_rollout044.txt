import time, math
import numpy as np

def _compute_c5(h, n):
    """Return max correlation C5 for a given step function."""
    dx = 2.0 / n
    # np.correlate with mode='full' gives linear correlation (no wrap)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _repair_h(h, target):
    """Project h onto box [0,1] with exact sum=target using water-filling."""
    # Clip initial out‑of‑bounds values
    np.clip(h, 0.0, 1.0, out=h)
    # Simple water‑filling loop
    for _ in range(100):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            free = h < 1.0
        else:
            free = h > 0.0
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        step = diff / idx.size
        if diff > 0.0:
            # limited by upper bound
            max_inc = np.min(1.0 - h[idx])
            step = min(step, max_inc)
        else:
            # limited by lower bound
            max_dec = np.min(h[idx])
            step = max(step, -max_dec)
        h[idx] += step
        np.clip(h, 0.0, 1.0, out=h)
    # Final tiny correction if needed
    total = h.sum()
    if abs(total - target) > 1e-8:
        scale = target / total
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _mutate(h, step):
    """Randomly transfer a small amount between two positions, preserving sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = np.random.uniform(-step, step)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    return new
