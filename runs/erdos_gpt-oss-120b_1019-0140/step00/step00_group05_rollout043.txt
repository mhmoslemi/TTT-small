            if T > 0 and rnd.rand() < math.exp(-(new_score - best_score) / T):
                best_score = float(new_score)
                best_h = h.copy()
                best_n = N
            else:
                # revert the move
                h[i] += delta
                h[j] -= delta

        iteration += 1
        if iteration - last_improve > max_no_improve:
            # stagnation – random binary restart
            h = np.zeros(N, dtype=np.float64)
            ones = rnd.choice(N, N // 2, replace=False)
            h[ones] = 1.0
            last_improve = iteration

    # final evaluation using the provided helper (guarantees the exact contract)
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, float(final_c5), int(best_h.shape[0]))
