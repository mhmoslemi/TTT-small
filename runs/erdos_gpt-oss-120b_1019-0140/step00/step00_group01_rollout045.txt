import numpy as np
import time

def project_vector(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1] with specified sum via bisection."""
    # Initial bounds for the Lagrange multiplier lambda
    low = -np.max(v) - 10.0
    high = 1.0 - np.min(v) + 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        total = w.sum()
        if total > target_sum:
            high = mid
        else:
            low = mid
    w = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    return w

def compute_c5(h: np.ndarray) -> float:
    """Compute the C5 value for a candidate h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def propose_transfer(h: np.ndarray, rng: np.random.Generator, max_step: float):
    """Return a new vector after transferring mass between two coordinates."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    sign = rng.choice([-1, 1])
    if sign > 0:
        delta_max = min(1.0 - h[i], h[j])
    else:
        delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0.0:
        return None
    limit = min(delta_max, max_step)
    delta = rng.random() * limit
    delta *= sign
    cand = h.copy()
    cand[i] += delta
    cand[j] -= delta
    cand[i] = np.clip(cand[i], 0.0, 1.0)
    cand[j] = np.clip(cand[j], 0.0, 1.0)
    return cand
