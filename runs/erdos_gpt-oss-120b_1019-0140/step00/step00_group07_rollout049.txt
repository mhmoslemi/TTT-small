import numpy as np
import time
import math

def compute_c5(arr):
    """Return max correlation integral for a step function defined by `arr`."""
    n = arr.size
    dx = 2.0 / n
    corr = np.correlate(arr, 1 - arr, mode="full") * dx
    return float(corr.max())

def thue_morse_seq(m):
    """Binary Thue–Morse sequence of length m (0/1)."""
    seq = np.empty(m, dtype=int)
    for i in range(m):
        seq[i] = bin(i).count("1") & 1
    return seq

def anneal(start_h, rng, start_time, budget_s):
    """Simulated‑annealing search starting from `start_h` within remaining budget."""
    curr_h = start_h.copy()
    curr_score = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_score = curr_score
    n = curr_h.size
    T0 = 0.5
    T_min = 1e-5
    no_improve = 0

    while True:
        now = time.time()
        if now - start_time >= budget_s * 0.99:
            break

        # linear temperature schedule
        progress = (now - start_time) / budget_s
        T = T0 * (1 - progress) + T_min * progress

        # pick a random 1 and a random 0 to swap
        ones_idx = np.flatnonzero(curr_h == 1)
        zeros_idx = np.flatnonzero(curr_h == 0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        new_h = curr_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_score = compute_c5(new_h)
        delta = new_score - curr_score

        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random perturbation to escape stagnation
        if no_improve > 5000:
            frac = 0.02
            k = max(1, int(frac * n))
            sel_one = rng.choice(np.flatnonzero(curr_h == 1), size=k, replace=False)
            sel_zero = rng.choice(np.flatnonzero(curr_h == 0), size=k, replace=False)
            curr_h[sel_one] = 0
            curr_h[sel_zero] = 1
            curr_score = compute_c5(curr_h)
            no_improve = 0

    return best_h, best_score
