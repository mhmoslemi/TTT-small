def simulated_annealing(init_h, budget_s, seed):
    rng = np.random.default_rng(seed)
    n = len(init_h)
    target_sum = n / 2.0
    cur_h = init_h.copy()
    cur_c5 = evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    iter = 0
    T0 = 0.1
    while True:
        t = time.time()
        elapsed = t - start
        if elapsed >= budget_s * 0.985: break
        iter += 1
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        low = max(-cur_h[i], -(1 - cur_h[j]))
        high = min(1 - cur_h[i], cur_h[j])
        if low >= high:
            continue
        # propose delta with magnitude limited to step_size
        step = 0.1 # can tune decreasing with iteration
        # choose d uniformly within [low, high] but limited by step
        max_delta = min(step, high)
        min_delta = max(-step, low)
        if min_delta >= max_delta:
            continue
        d = rng.uniform(min_delta, max_delta)
        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d
        new_c5 = evaluate_c5(new_h)
        delta_obj = new_c5 - cur_c5
        # annealing temperature schedule
        T = T0 / (1 + math.log(1 + iter))
        if delta_obj <= 0 or rng.random() < math.exp(-delta_obj / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    return best_h, best_c5

import time
import math
import numpy as np

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x) (1 - h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h):
    """Clip h to [0,1] and adjust to satisfy sum(h)=n/2 exactly (within tolerance)."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Simple correction using the element with most capacity.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - h
            i = np.argmax(caps)
            inc = min(diff, caps[i])
            h[i] += inc
            diff -= inc
        else:
            caps = h
            i = np.argmax(caps)
            dec = min(-diff, caps[i])
            h[i] -= dec
            diff += dec
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    target = n / 2.0
    # start from uniform random numbers and scale to target sum
    x = rng.random(n)
    if x.sum() == 0:
        x = np.ones(n) * (target / n)
    else:
        x = x * (target / x.sum())
    x = np.clip(x, 0.0, 1.0)
    return project_feasible(x)

def simulated_annealing(start_h, budget_s, rng):
    """Simple simulated annealing keeping the sum constraint exactly."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_c5 = evaluate_c5(best_h)
    cur_h = start_h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    it = 0
    no_improve = 0
    max_step_initial = 0.2
    T0 = 0.1

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        it += 1

        # temperature and step size decay
        T = T0 / (1.0 + math.log(1.0 + it))
        step = max_step_initial * (1.0 - elapsed / budget_s)

        # propose exchange move between two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        low = max(-cur_h[i], -(1.0 - cur_h[j]))
        high = min(1.0 - cur_h[i], cur_h[j])
        if low >= high:
            continue
        lo = max(low, -step)
        hi = min(high, step)
        if lo >= hi:
            continue
        delta = rng.uniform(lo, hi)

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c5 = evaluate_c5(new_h)
        delta_obj = new_c5 - cur_c5

        if delta_obj <= 0.0 or rng.random() < math.exp(-delta_obj / T):
            cur_h = new_h
            cur_c5 = new_c5
            no_improve = 0 if cur_c5 < best_c5 else no_improve + 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            no_improve += 1

        if no_improve >= 200_000:
            cur_h = random_feasible_vector(n, rng)
            cur_c5 = evaluate_c5(cur_h)
            no_improve = 0

    except Exception:
        final_c5 = evaluate_c5(best_h)

    return best_h.tolist(), float(final_c5), int(n)
