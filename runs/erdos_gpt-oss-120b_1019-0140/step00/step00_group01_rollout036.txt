    min_lr = 0.001
    init_temp = 0.01
    # Main loop
    while time.time() - start < budget_s * 0.97:

        lr = max(min_lr, init_lr * (1 - frac))

        # Gradient step
        v_new = h - lr * g
        v_new = _project_to_feasible(v_new, target_sum)
        new_c5 = _compute_c5(v_new)
        # Acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            h = v_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = v_new.copy()

        if rng.random() < 0.01:
            # random donor and receiver
            donor = rng.integers(n)
            receiver = rng.integers(n)

import numpy as np
import time
import math

def _project_to_feasible(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of v onto the box [0,1] with sum == target."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + lam, 0.0, 1.0)

def _compute_c5(v: np.ndarray) -> float:
    """Return C₅ = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = v.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(v, 1.0 - v, mode='full')) * dx

def _gradient_worst(v: np.ndarray):
    """
    Sub‑gradient of the worst‑case overlap (the lag attaining the maximum).
    Returns gradient array, lag, and the raw correlation maximum (without dx).
    """
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    g = np.zeros_like(v)
    if lag >= 0:
        g[: n - lag] += 1.0 - v[lag:]
        g[lag:] -= v[: n - lag]
    else:
        k = -lag
        g[k:] += 1.0 - v[: n - k]
        g[: n - k] -= v[k:]
    return g, lag, corr[idx]

def _random_binary_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a 0/1 vector of length n with exactly n/2 ones (fractional if n odd)."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

    h = np.clip(h, 0.0, 1.0)
    h = _project_to_feasible(h, target_sum)

    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ---------- optimisation parameters ----------
    init_lr = 0.5      # initial learning rate
    min_lr = 0.001     # lower bound for learning rate
