    max_iter=1000

    h = h_init.copy()
    n = len(h)
    best_val = compute_c5(h)
    while time.time() - start_time < budget:
        # pick two indices i, j
        i, j = np.random.choice(n, 2, replace=False)
        if h[i] == h[j]: continue
        # propose moving epsilon from one to other
        eps = np.random.uniform(0.0, min(0.01, (1 - h[i]) if h[i] < h[j] else h[i]))
        # Let's move from higher to lower
        if h[i] > h[j]:
            delta = min(eps, h[i] - 0.0, 1.0 - h[j])
            hi, hj = i, j
        else:
            delta = min(eps, h[j] - 0.0, 1.0 - h[i])
            hi, hj = j, i
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[hi] -= delta
        h_new[hj] += delta
        # enforce bounds (should be inside)
        val = compute_c5(h_new)
        if val < best_val:
            h = h_new
            best_val = val
    return h, best_val

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def enforce_sum_and_bounds(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = np.sum(v)
    # If nearly target, just return
    if abs(cur - target) < 1e-12:
        return v
    # Scale to meet target roughly
    if cur > 0:
        factor = target / cur
        v = v * factor
    else:
        # all zeros, distribute uniformly
        v = np.full_like(v, target / v.size)
    v = np.clip(v, 0.0, 1.0)
    # Fix any residual difference
    for _ in range(10):
        cur = np.sum(v)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0
        else:
            free = v > 0.0
        count = np.count_nonzero(free)
        if count == 0:
            break
        adj = diff / count
        v[free] = np.clip(v[free] + adj, 0.0, 1.0)
