def _score(vec, dx):
    """Compute C5 for a binary vector `vec`."""
    # correlation of h and (1-h)
    corr = np.correlate(vec, 1 - vec, mode="full") * dx
    return float(corr.max())

        n_points (int): number of discretization points used.
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)
    start_time = time.time()
    TARGET = 0.3808  # early stop if achieved

    # ---- Initialize h -------------------------------------------------------
    init_arr = globals().get("initial_h_values", None)
    if init_arr is not None:
        vec = np.asarray(init_arr, dtype=float)
        n_points = vec.shape[0]
        half = n_points // 2
        # Check if already binary with correct sum
        binary_ok = np.all(np.logical_or(np.isclose(vec, 0), np.isclose(vec, 1)))
        sum_ok = int(round(vec.sum())) == half
        if binary_ok and sum_ok:
            h = vec.astype(int)
        else:
            # Convert to binary by taking the largest half entries as 1
            idx = np.argsort(-vec)[:half]
            h = np.zeros(n_points, dtype=int)
            h[idx] = 1
    else:
        # Default size if no parent construction is provided
        n_points = 70
        half = n_points // 2
        perm = rng.permutation(n_points)
        h = np.zeros(n_points, dtype=int)
        h[perm[:half]] = 1

    # Ensure exact sum constraint (may be needed after conversion)
    half = n_points // 2
    cur_sum = h.sum()
    if cur_sum != half:
        diff = cur_sum - half
        if diff > 0:
            # Too many ones -> turn some to zero
            idx_one = np.where(h == 1)[0]
            turn_off = rng.choice(idx_one, size=diff, replace=False)
            h[turn_off] = 0
        else:
            # Too few ones -> turn some zeros to one
            idx_zero = np.where(h == 0)[0]
            turn_on = rng.choice(idx_zero, size=-diff, replace=False)
            h[turn_on] = 1

    dx = 2.0 / n_points

    # ---- Simulated annealing ------------------------------------------------
    best_h = h.copy()
    best_score = _score(best_h, dx)
    current_h = best_h.copy()
    current_score = best_score

    # Index sets for quick swaps
    ones = np.where(current_h == 1)[0]
    zeros = np.where(current_h == 0)[0]

    T_initial = 2.0
    T = T_initial
    decay = 0.9995
    iter_cnt = 0

    while True:
        # Time check
        if time.time() - start_time >= budget_s - 0.1:
            break

        # Ensure we have both one and zero positions
        if len(ones) == 0 or len(zeros) == 0:
            # Reinitialize randomly if degeneration occurs
            perm = rng.permutation(n_points)
            current_h[:] = 0
            current_h[perm[:half]] = 1
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            current_score = _score(current_h, dx)
            continue

        # Propose a swap between a random 1 and a random 0
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)

        # Apply swap
        current_h[i_one] = 0
        current_h[i_zero] = 1

        new_score = _score(current_h, dx)
        delta = new_score - current_score

        # Metropolis acceptance
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # Accept move
            current_score = new_score
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]

            if current_score < best_score:
                best_score = current_score
                best_h = current_h.copy()
                if best_score <= TARGET:
                    # Early exit: target achieved
                    break
        else:
            # Reject: revert swap
            current_h[i_one] = 1
            current_h[i_zero] = 0
            # ones/zeros unchanged

        # Temperature update
        T *= decay
        if T < 1e-8:
            T = 1e-8

        iter_cnt += 1

        # Periodic random restart to avoid stagnation
        if iter_cnt % 5000 == 0:
            perm = rng.permutation(n_points)
            current_h[:] = 0
            current_h[perm[:half]] = 1
            current_score = _score(current_h, dx)
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            T = T_initial

    # Final best score (recompute for safety)
    final_score = _score(best_h, dx)
    return best_h.astype(float), final_score, n_points
