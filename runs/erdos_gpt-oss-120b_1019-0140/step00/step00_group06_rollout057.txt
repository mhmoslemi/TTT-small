import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_grad(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    grad = np.zeros_like(h)
    i = np.arange(n)
    j = i + s
    mask = (j >= 0) & (j < n)
    ii = i[mask]
    jj = j[mask]
    grad[ii] += 1.0 - h[jj]
    grad[jj] -= h[ii]
    grad *= dx
    return grad

def fix_sum_and_bounds(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Iterative projection onto hyperplane sum=target
    for _ in range(20):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            # All at bounds, cannot adjust further
            break
        # Distribute diff equally among free variables
        x[free] += diff / cnt
        # Reclip
        x = np.clip(x, 0.0, 1.0)
    return x

def penalized_objective(x, target, mu):
    return compute_c5(x) + mu * (np.sum(x) - target) ** 2

def penalized_grad(x, target, mu):
    return compute_c5_grad(x) + 2.0 * mu * (np.sum(x) - target) * np.ones_like(x)

def generate_initial_vector(n, rng):
    try:
        init = initial_h_values
    except NameError:
        init = None
    if init is not None and len(init) == n:
        return init.copy()
    if init is not None:
        # Linear interpolation
        old_x = np.linspace(0.0, 2.0, len(init))
        new_x = np.linspace(0.0, 2.0, n)
        vec = np.interp(new_x, old_x, init)
        vec = fix_sum_and_bounds(vec, n / 2.0)
        return vec
    # Random start with uniform baseline
    target = n / 2.0
    vec = np.full(n, target / n)
    vec += rng.normal(scale=0.1, size=n)
    vec = fix_sum_and_bounds(vec, target)
    return vec

def local_optimize(h0, rng, time_limit):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    mu = 1e3  # penalty weight (adjustable)
    # Use L-BFGS-B with gradient and bounds
    options = {'maxiter': 500, 'ftol': 1e-12, 'disp': False}
    # we may try multiple attempts if time permits
    best_h = h0
    best_val = compute_c5(h0)
    start = time.time()
    attempt = 0
    while time.time() - start < time_limit and attempt < 5:
        # random perturbation around current best
        pert = best_h + rng.normal(scale=0.05, size=n)
        pert = fix_sum_and_bounds(pert, target)
        res = opt.minimize(penalized_objective, pert,
                           args=(target, mu),
                           method='L-BFGS-B',
                           jac=penalized_grad,
                           bounds=bounds,
                           options=options)
        h_opt = fix_sum_and_bounds(res.x, target)
        val_opt = compute_c5(h_opt)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
        attempt += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine base resolution
    try:
        base_n = len(initial_h_values)
    except NameError:
        base_n = 100
    # List of sample sizes to try
    n_list = [base_n, base_n * 2, base_n * 4]
    best_h = None
    best_val = np.inf
    for n in n_list:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # allocate a slice of remaining time
        remaining = budget_s - elapsed
        # use at most 300 seconds for each resolution
        sub_budget = min(300.0, remaining - 5.0)  # leave a little margin
        if sub_budget <= 0:
            break
        init_vec = generate_initial_vector(n, rng)
        # local optimization with time limit
        h_opt, val_opt = local_optimize(init_vec, rng, sub_budget)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
        # early stop if we already beat target
        if best_val < 0.3808:
            break
    if best_h is None:
        # fallback to initial
        try:
            best_h = initial_h_values.copy()
        except NameError:
            best_h = np.full(base_n, 0.5)
    # Final evaluation using provided helper if available
    try:
        c5 = evaluate_erdos_solution(best_h)
    except NameError:
        c5 = compute_c5(best_h)
    n_points = best_h.shape[0]
    # final repair
    best_h = fix_sum_and_bounds(best_h, n_points / 2.0)
    return best_h.astype(np.float64), float(c5), int(n_points)

import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap integral for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_grad(h):
    """Gradient of the max overlap integral (with respect to h)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)          # shift corresponding to the maximum
    grad = np.zeros_like(h)

    i = np.arange(n)
    j = i + s
    mask = (j >= 0) & (j < n)
    ii = i[mask]
    jj = j[mask]

    grad[ii] += 1.0 - h[jj]       # derivative from h_i * (1 - h_{i+s})
    grad[jj] -= h[ii]            # derivative from h_{i} * (1 - h_{i+s}) where i=j-s
    grad *= dx
    return grad

def fix_sum_and_bounds(x, target):
    """Project x onto the box [0,1]^n while enforcing sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        x[free] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def penalized_objective(x, target, mu):
    """Objective with quadratic penalty for the integral constraint."""
    return compute_c5(x) + mu * (np.sum(x) - target) ** 2

def penalized_grad(x, target, mu):
    """Gradient of the penalized objective."""
    return compute_c5_grad(x) + 2.0 * mu * (np.sum(x) - target) * np.ones_like(x)

def generate_initial_vector(n, rng):
    """Construct an initial feasible vector for a given resolution."""
    try:
        init = initial_h_values
    except NameError:
        init = None
    if init is not None and len(init) == n:

    if init is not None:
        old_x = np.linspace(0.0, 2.0, len(init))
        new_x = np.linspace(0.0, 2.0, n)
        vec = np.interp(new_x, old_x, init)
        return fix_sum_and_bounds(vec, n / 2.0)
    # Random start: uniform baseline with small noise
    target = n / 2.0
    vec = np.full(n, target / n)
    vec += rng.normal(scale=0.1, size=n)
    return fix_sum_and_bounds(vec, target)

def local_optimize(h0, rng, time_limit):
    """Run a few L‑BFGS‑B attempts with random perturbations."""
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    mu = 1000.0                     # penalty weight
    opts = {'maxiter': 500, 'ftol': 1e-12, 'disp': False}
    best_h = h0
    best_val = compute_c5(h0)
    start = time.time()
    attempts = 0
    while time.time() - start < time_limit and attempts < 5:
        pert = best_h + rng.normal(scale=0.05, size=n)
        pert = fix_sum_and_bounds(pert, target)
        res = opt.minimize(penalized_objective, pert,
                           args=(target, mu),
                           method='L-BFGS-B',

        val_opt = compute_c5(h_opt)
        if val_opt < best_val:

            best_h = h_opt
