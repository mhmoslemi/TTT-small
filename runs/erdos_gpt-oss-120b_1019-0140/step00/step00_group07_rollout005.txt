import numpy as np
import time
import random

def compute_c5(h):
    """Return the overlap bound C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h and (1‑h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_swap(h, shift):
    """
    Propose a pair of indices (i, j) to swap.
    The proposal is biased by the supplied shift: we look for indices where
    h[i] is large while its partner h[i+shift] is small, and vice‑versa.
    """
    n = h.shape[0]
    # overlapping region for the given shift
    i_start = max(0, -shift)
    i_end = min(n, n - shift)
    if i_end <= i_start:
        # shift has no overlap – fall back to a random swap
        i, j = np.random.choice(n, 2, replace=False)
        return int(i), int(j)

    idx = np.arange(i_start, i_end)
    idx_shift = idx + shift

    # candidates that would reduce the current worst overlap if swapped
    cond_i = (h[idx] > 0.5) & (h[idx_shift] < 0.5)   # high now, low at shift
    cond_j = (h[idx] < 0.5) & (h[idx_shift] > 0.5)   # low now, high at shift
    cand_i = idx[cond_i]
    cand_j = idx[cond_j]

    if cand_i.size and cand_j.size:
        i = int(np.random.choice(cand_i))
        j = int(np.random.choice(cand_j))
        if i == j:
            i, j = np.random.choice(n, 2, replace=False)
    else:
        i, j = np.random.choice(n, 2, replace=False)

    return int(i), int(j)
