def _evaluate_h(h):
    """Return the C5 value for a candidate h."""
    try:
        return evaluate_erdos_solution(h)  # pre‑imported helper
    except NameError:
        # Fallback: compute directly
        n = len(h)
        dx = 2.0 / n
        return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def _adjust_to_target(h, target):
    """Clip to {0,1} and force the sum to equal target."""
    h = np.where(h >= 0.5, 1.0, 0.0)
    cur = int(h.sum())
    if cur == target:
        return h
    if cur > target:
        ones = np.where(h == 1)[0]
        off = np.random.choice(ones, cur - target, replace=False)
        h[off] = 0.0
    else:
        zeros = np.where(h == 0)[0]
        on = np.random.choice(zeros, target - cur, replace=False)
        h[on] = 1.0
    return h
