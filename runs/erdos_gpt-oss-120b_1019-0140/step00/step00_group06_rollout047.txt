import numpy as np
import time

def _repair_sum_and_bounds(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h) = len(h)/2 exactly."""
    np.clip(h, 0.0, 1.0, out=h)
    n = h.shape[0]
    target = n * 0.5
    total = h.sum()
    diff = target - total
    # Adjust a single coordinate (or a few) to fix the sum.
    if abs(diff) > 1e-12:
        if diff > 0:
            # need to add diff
            idxs = np.where(h < 1.0)[0]
            for i in idxs:
                cap = 1.0 - h[i]
                add = diff if diff <= cap else cap
                if add > 0:
                    h[i] += add
                    diff -= add
                    if diff <= 1e-12:
                        break
        else:
            # need to subtract -diff
            idxs = np.where(h > 0.0)[0]
            for i in idxs:
                cap = h[i]
                sub = -diff if -diff <= cap else cap
                if sub > 0:
                    h[i] -= sub
                    diff += sub
                    if diff >= -1e-12:
                        break
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _compute_c5(h: np.ndarray) -> float:
    """Compute the overlap constant for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _hill_climb(init_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Greedy/random hill‑climbing with occasional uphill moves."""
    h_cur = init_h.copy()
    cur_c5 = _compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    n = h_cur.shape[0]
    start = time.time()
    no_improve = 0
    while True:
        if time.time() - start >= time_budget:
            break
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1  # ensure i != j
        # move mass from i to j
        max_transfer = min(h_cur[i], 1.0 - h_cur[j])
        if max_transfer <= 0.0:
            # try opposite direction
            max_transfer = min(h_cur[j], 1.0 - h_cur[i])
            if max_transfer <= 0.0:
                continue
            i, j = j, i
        eps = rng.random() * max_transfer
        if eps <= 0.0:
            continue
        h_new = h_cur.copy()
        h_new[j] += eps
        h_new[i] -= eps
        new_c5 = _compute_c5(h_new)
        # update global best
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        # accept if improvement or tiny random chance
        if new_c5 < cur_c5:
            h_cur = h_new
            cur_c5 = new_c5
            no_improve = 0
        elif rng.random() < 0.001:
            h_cur = h_new
            cur_c5 = new_c5
        # give up after many useless attempts
        if no_improve > 20000:
            break
    best_h = _repair_sum_and_bounds(best_h)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5

    """Return an improved step function for the Erdős C₅ problem.

    # ------------------------------------------------------------------
