import numpy as np, time

def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # Determine n and m
    init = globals().get('initial_h_values', None)
    if init is not None:
        init = np.asarray(init, dtype=np.float64)
        n = int(init.shape[0])
        # Ensure even
        if n % 2 != 0:
            # Pad with zero to make even
            init = np.concatenate([init, np.zeros(1, dtype=init.dtype)])
            n += 1
        m = n // 2
        # Convert to binary by selecting top m entries
        sorted_idx = np.argsort(-init)  # descending
        h = np.zeros(n, dtype=np.float64)
        h[sorted_idx[:m]] = 1.0
    else:
        n = 200
        m = n // 2
        h = np.zeros(n, dtype=np.float64)
        ones_idx = np.random.choice(n, m, replace=False)
        h[ones_idx] = 1.0

    # Create ones and zeros index lists for fast swapping
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    # Evaluate
    best_h = h.copy()
    best_val = evaluate(best_h)
    curr_val = best_val

    # SA parameters
    init_temp = 0.05  # starting temperature
    min_temp = 1e-6
    # Stagnation restart
    stagnation_limit = 20000
    iter_since_improve = 0
    total_iters = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # temperature schedule
        progress = elapsed / (budget_s * 0.95)
        T = max(min_temp, init_temp * (1 - progress))
        # random swap
        if len(ones) == 0 or len(zeros) == 0:
            # nothing to swap; break
            break
        idx_one = np.random.randint(len(ones))
        idx_zero = np.random.randint(len(zeros))
        i1 = ones[idx_one]
        i0 = zeros[idx_zero]

        # perform swap
        h[i1] = 0.0
        h[i0] = 1.0
        # swap indices in lists
        ones[idx_one] = i0
        zeros[idx_zero] = i1

        new_val = evaluate(h)
        delta = new_val - curr_val
        accept = False
        if delta < 0:
            accept = True
        else:
            # probability
            if T > 0:
                if np.random.rand() < math.exp(-delta / T):
                    accept = True
        if accept:
            curr_val = new_val
            iter_since_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i1] = 1.0
            h[i0] = 0.0
            # revert indices
            ones[idx_one] = i1
            zeros[idx_zero] = i0
            iter_since_improve += 1

        total_iters += 1
        # restart if stagnated too long
        if iter_since_improve >= stagnation_limit:
            # random reinitialize
            h = np.zeros(n, dtype=np.float64)
            ones_idx = np.random.choice(n, m, replace=False)
            h[ones_idx] = 1.0
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            curr_val = evaluate(h)
            iter_since_improve = 0
            # accept this as current; best remains unchanged
    # Final evaluation may have slight difference due to float rounding; compute best bound again
    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n)

import time, math, random
import numpy as np

def _evaluate(vec):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by correlation."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives linear correlation with zero padding
    corr = np.correlate(vec, 1.0 - vec, mode='full') * dx
    return float(corr.max()), corr  # also return full array for internal use


def _project_to_constraints(h, target_sum):
    """Make sure 0≤h≤1 and sum(h)=target_sum (within tiny tolerance)."""
    # simple clipping + redistribution
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute only to interior points
        interior = (h > 0.0) & (h < 1.0)
        cnt = interior.sum()
        if cnt == 0:
            # fallback: distribute uniformly (will stay clipped)
            h += diff / h.size
        else:
            h[interior] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
