import time, math, numpy as np

def _c5_value(h):
    """Return the discretized C₅ for a binary/real vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.shape[0])

def _random_binary(n, weight, rng):
    """Return a random {0,1} vector of length n with exactly `weight` ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, weight, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # initialise a feasible vector (binary, sum = n/2)

    try:

        # force binary values
        h0 = (h0 > 0.5).astype(float)
        target_w = n // 2
        cur_w = int(h0.sum())
        if cur_w != target_w:
            if cur_w > target_w:
                excess = cur_w - target_w
                idx = np.where(h0 == 1)[0]
                drop = rng.choice(idx, excess, replace=False)
                h0[drop] = 0.0
            else:
                deficit = target_w - cur_w
                idx = np.where(h0 == 0)[0]
                add = rng.choice(idx, deficit, replace=False)
                h0[add] = 1.0
    except NameError:
        n = 70
        target_w = n // 2
        h0 = _random_binary(n, target_w, rng)

    # -----------------------------------------------------------------
    # local optimisation: simulated‑annealing style swaps
    # -----------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _c5_value(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    start = time.time()
    target = 0.3808          # desired bound
    init_temp = 0.05         # initial temperature for annealing
    no_imp = 0
    max_no_imp = 5000        # restart after this many non‑improving attempts

    while True:
        now = time.time()
        if now - start > budget_s - 0.1:
            break

        # linear cooling schedule
        elapsed = now - start
        T = init_temp * max(0.0, 1.0 - elapsed / budget_s)

        # pick a random 1–0 swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        new_c5 = _c5_value(new_h)
