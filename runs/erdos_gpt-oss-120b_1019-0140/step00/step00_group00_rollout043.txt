import time
import numpy as np

def _c5_value(h, dx):
    """Return the max overlap C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum(vec, target):
    """
    Project vec onto the box [0,1]^n while enforcing sum(vec) == target.
    Uses a simple binary search on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(vec + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # --- Determine resolution and initial vector -----------------
    try:
        # Use the provided parent construction if it exists.
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        h = np.clip(init, 0.0, 1.0)
        target_sum = n / 2.0
        h = _project_to_sum(h, target_sum)
    except NameError:
        # No parent; work with a modest resolution.
        n = 256                     # chosen for a good speed/accuracy trade‑off
        target_sum = n / 2.0
        dx = 2.0 / n
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        u = np.abs(xs - 1.0)        # distance from centre, lies in [0,1]

        # ---- Simple parametric family search -------------------
        best_h = None
        best_val = np.inf

        # parameters a (centre value) and b (linear coefficient);
        # the quadratic coefficient c is fixed by the integral constraint.
        a_vals = np.linspace(0.0, 0.5, 51)          # centre height
        b_vals = np.linspace(0.0, 2.0, 101)         # slope magnitude

        for a in a_vals:
            for b in b_vals:

                h_tmp = a + b * u + c * u * u
                h_tmp = np.clip(h_tmp, 0.0, 1.0)

                val = _c5_value(h_tmp, dx)
                if val < best_val:
                    best_val = val
                    best_h = h_tmp.copy()

        # fall back to a simple V‑shape if the search failed for any reason
        if best_h is None:

            b_opt = 1.0 - 2.0 * a_opt
            best_h = a_opt + b_opt * np.abs(xs - 1.0)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = _project_to_sum(best_h, target_sum)

        h = best_h.copy()
        best_val = _c5_value(h, dx)

    dx = 2.0 / n

    # --- Record the current best ------------------------------
    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    # --- Projected sub‑gradient descent ------------------------
    eta0 = 0.2                      # initial step size factor
    iteration = 0

    while time.time() - start_time < budget_s * 0.94:

        # Evaluate current C5 and identify the worst shift

        O = corr * dx
        k_idx = int(np.argmax(O))

            best_h = h.copy()

        eta = eta0 / np.sqrt(iteration)

        h_new = h - eta * grad

    final_val = _c5_value(best_h, dx)
