        init_h = random_binary()

    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Local search with multiple random restarts
    # ------------------------------------------------------------------
    start_time = time.time()
    # Heuristic number of swaps per restart (scale with N)
    swaps_per_restart = max(5000, N * 50)

    while time.time() - start_time < budget_s * 0.95:
        # Occasionally start from the current best to refine it
        if random.random() < 0.2:
            cur_h = best_h.copy()
        else:
            cur_h = random_binary()
        cur_c5 = compute_c5(cur_h)

        # Maintain index lists for fast swapping
        ones_idx = list(np.where(cur_h == 1)[0])
        zeros_idx = list(np.where(cur_h == 0)[0])

        for _ in range(swaps_per_restart):
            i_one = random.choice(ones_idx)
            i_zero = random.choice(zeros_idx)

            # Perform the swap
            cur_h[i_one] = 0.0
            cur_h[i_zero] = 1.0

            new_c5 = compute_c5(cur_h)

            if new_c5 < cur_c5:
                # Accept improvement
                cur_c5 = new_c5
                # Update index bookkeeping
                ones_idx.remove(i_one)
                zeros_idx.remove(i_zero)
                ones_idx.append(i_zero)
                zeros_idx.append(i_one)

                # Update global best if needed
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()
                    # Early exit if target is already reached
                    if best_c5 < 0.3808:
                        break
            else:
                # Revert swap
                cur_h[i_one] = 1.0
                cur_h[i_zero] = 0.0

        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Final accurate evaluation (if the helper is available)
    # ------------------------------------------------------------------
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = float(best_c5)

    return best_h, final_c5, N
