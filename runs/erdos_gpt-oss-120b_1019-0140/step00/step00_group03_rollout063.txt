import time, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine dimension
    try:
        init_h = np.asarray(initial_h_values, dtype=np.float64)
        n = init_h.size
        cur_h = init_h.copy()
    except Exception:
        n = 70
        cur_h = np.full(n, 0.5, dtype=np.float64)
    target_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        return corr.max() * dx

    def gradient(vec, shift):
        n = vec.shape[0]
        grad = np.empty(n, dtype=np.float64)
        if shift >= 0:
            grad[:shift] = 1.0
            if shift < n:
                grad[shift:] = 1.0 - vec[:n - shift]
        else:
            s = -shift
            if s < n:
                grad[:n - s] = 1.0 - vec[s:]
                grad[n - s:] = 1.0
            else:
                grad[:] = 1.0
        idx = np.arange(n) + shift
        mask = (idx >= 0) & (idx < n)
        grad[mask] -= vec[idx[mask]]
        return grad

    def project_sum(vec):
        diff = target_sum - vec.sum()
        # attempt to fix diff by adjusting random feasible entries
        n = vec.size
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 1000:
            i = np.random.randint(0, n)
            if diff > 0:
                capacity = 1.0 - vec[i]
                if capacity <= 0:
                    attempts += 1
                    continue
                delta = min(diff, capacity)
                vec[i] += delta
                diff -= delta
            else:
                capacity = vec[i]
                if capacity <= 0:
                    attempts += 1
                    continue
                delta = min(-diff, capacity)
                vec[i] -= delta
                diff += delta
            attempts += 1
        # final clip (safety)
        np.clip(vec, 0.0, 1.0, out=vec)
        return vec

import time
import numpy as np

def _compute_c5(vec: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(vec, 1 - vec, mode="full")
    return float(corr.max() * dx)


def _gradient(vec: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap integral with respect to vec for a fixed shift."""
    n = vec.size
    grad = np.empty(n, dtype=np.float64)

    if shift >= 0:
        grad[:shift] = 1.0
        if shift < n:
            grad[shift:] = 1.0 - vec[: n - shift]
    else:
        s = -shift
        if s < n:
            grad[: n - s] = 1.0 - vec[s:]
            grad[n - s :] = 1.0
        else:
            grad[:] = 1.0

    idx = np.arange(n) + shift
    mask = (idx >= 0) & (idx < n)
    grad[mask] -= vec[idx[mask]]
    return grad


def _project_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """Adjust vec (in‑place) so that its sum equals target while staying in [0,1]."""
