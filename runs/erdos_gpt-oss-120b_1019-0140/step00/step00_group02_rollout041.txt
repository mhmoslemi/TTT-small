def adjust_sum(h, target):
    # Assume all entries within [0,1]
    total = h.sum()
    if total == target:
        return h
    # Compute needed diff
    diff = target - total
    # Identify indices not at bounds (0< h <1)
    free = (h > 0) & (h < 1)
    # if no free, adjust by scaling perhaps
    if free.sum() > 0:
        # Distribute diff equally
        step = diff / free.sum()
        # Clip after addition
        h[free] += step
        h = np.clip(h, 0, 1)
        # Recurse if still diff not satisfied
        return adjust_sum(h, target)
    else:
        # all at bounds, cannot adjust; fallback scaling
        h = np.clip(h, 0, 1)  # ensure
        # rescale proportionally and clip again
        scaling = target / h.sum()
        h = np.clip(h * scaling, 0, 1)
        return adjust_sum(h, target)

import numpy as np

def _repair(vec, target):
    """Clip to [0,1] and adjust the sum to exactly target."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(20):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # capacity to add (if diff>0) or remove (if diff<0)
        if diff > 0:
            cap = 1.0 - v
        else:
            cap = v
        cap_sum = cap.sum()
        if cap_sum == 0:
            break
        v += cap * (diff / cap_sum)
        v = np.clip(v, 0.0, 1.0)
    # final scaling if needed
    total = v.sum()
    if abs(total - target) > 1e-9:
        v *= target / total
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(vec, dx):
    """Compute C5 = max_k ∫ h(x)(1−h(x+k)) dx using discrete correlation."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx
