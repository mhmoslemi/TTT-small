    dx = 2.0 / n
    target_sum = n/2.0
    cur_h = h.copy()

import numpy as np
import time
import math

def _compute_c5(h, dx=None):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.size
    if dx is None:
        dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _gradient_at_shift(h, shift, dx):
    """Gradient of the correlation term at a given integer shift (lag)."""
    n = h.size
    g = np.full(n, dx)                     # derivative of Σ h_i term
    if shift >= 0:
        s = shift
        if s < n:
            # subtract h_{i+shift}
            g[:n - s] -= dx * h[s:]
            # subtract h_{i-shift}
            g[s:] -= dx * h[:n - s]
    else:
        s = -shift
        if s < n:
            g[s:] -= dx * h[:n - s]
            g[:n - s] -= dx * h[s:]
    return g


def _project_sum_and_bounds(h, target_sum):
    """Simple projection onto [0,1] box with exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # distribute the difference over non‑boundary entries
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    else:
        # fallback uniform adjustment (should be rare)
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        # small uniform tweak
        h += (target_sum - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h


def _upscale(h, new_n):
    """Linear interpolation of a feasible h to a larger grid."""
    n = h.size
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    x_old = np.arange(n, dtype=np.float64) * dx_old
    x_new = np.arange(new_n, dtype=np.float64) * dx_new
    h_new = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    return _project_sum_and_bounds(h_new, target)


def _local_search(h0, time_limit, rnd):
    """Projected random‑pair + occasional gradient descent."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0
    cur_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()

    while time.time() - start < time_limit:
        # decide move type
        if rnd.rand() < 0.15:          # gradient step
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            g = _gradient_at_shift(h, shift, dx)
            # make step preserve the sum (zero‑mean direction)
            g -= g.mean()
            # step size decays with time
            elapsed = time.time() - start
            frac = elapsed / time_limit
            alpha = 0.05 * (1.0 - frac) + 1e-4
            new_h = h - alpha * g
            new_h = _project_sum_and_bounds(new_h, target_sum)
        else:                          # random pair transfer
            i, j = rnd.choice(n, size=2, replace=False)
            max_transfer = min(h[i], 1.0 - h[j])
            min_transfer = -min(h[j], 1.0 - h[i])
            if max_transfer <= min_transfer:
                continue
            delta = rnd.uniform(min_transfer, max_transfer)
            # occasional fine‑scale move
            if rnd.rand() < 0.3:
                delta *= 0.2
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            # bounds already respected by construction
        new_c5 = _compute_c5(new_h, dx)

        # acceptance rule (simulated annealing)
        if new_c5 < cur_c5 - 1e-12:
            h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            # temperature schedule
            elapsed = time.time() - start
            T = 0.01 * max(0.0, 1.0 - elapsed / time_limit)
            if T > 0 and rnd.rand() < math.exp(-(new_c5 - cur_c5) / T):
                h, cur_c5 = new_h, new_c5
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible step function, its C5 bound, and number of points."""
    rnd = np.random.RandomState(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # 1. Obtain initial feasible construction
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n0 = h0.size
    except Exception:                     # no parent supplied
        n0 = 45
        h0 = np.full(n0, 0.5, dtype=np.float64)   # constant feasible function

    # enforce constraints (tiny rounding errors may exist)
    target0 = n0 / 2.0
    h0 = _project_sum_and_bounds(h0, target0)

    # ------------------------------------------------------------------
    # 2. First optimisation stage on the original resolution
    # ------------------------------------------------------------------
    time_spent = time.time() - start_total
    remaining = budget_s - time_spent
    stage1_time = min(remaining * 0.4, 250.0)   # keep a safe cap
    best_h, best_c5 = _local_search(h0, stage1_time, rnd)

    # ------------------------------------------------------------------
    # 3. Optional upscale and refinement when time permits
    # ------------------------------------------------------------------
    time_spent = time.time() - start_total
    remaining = budget_s - time_spent
    if remaining > 100.0:                     # allow a second stage
        # try a higher resolution (up to ~180 points to stay cheap)
        n_up = min(180, best_h.size * 2)
        if n_up > best_h.size:
            h_up = _upscale(best_h, n_up)
            # split remaining time between two passes
            t_up = remaining * 0.7
            best_h_up, best_c5_up = _local_search(h_up, t_up, rnd)
            if best_c5_up < best_c5:
                best_h, best_c5 = best_h_up, best_c5_up

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    n_points = best_h.size
    return (best_h, float(best_c5), n_points)
