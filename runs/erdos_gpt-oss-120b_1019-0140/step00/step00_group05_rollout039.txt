import numpy as np, time, math, random

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, copy=True)
        n = h0.shape[0]
    else:
        n = 70
        # generate binary vector with sum = n//2 (assuming n even)
        half = n // 2
        h0 = np.zeros(n)
        h0[:half] = 1.0
        rng.shuffle(h0)
    # Ensure values within [0,1] and sum constraint
    # Might be non-integer values; if non-binary, round to nearest 0/1 while preserving sum
    # Let's enforce binary:
    h0 = np.array(h0) # floats
    # If sum not exactly n/2, adjust:
    total = h0.sum()
    target = n/2
    if not np.isclose(total, target):
        # Convert to binary by thresholding top target values
        idx = np.argsort(h0)[::-1]
        binary = np.zeros(n)
        binary[idx[:int(target)]] = 1.0
        h0 = binary
    # h0 now binary.

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    max_time = budget_s * 0.95
    # annealing schedule parameters
    T0 = 0.1
    T_end = 1e-4
    # number of steps per temperature? We'll compute temperature based on elapsed fraction.
    iteration = 0
    no_improve_iters = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # temperature decreasing linearly
        T = T0 * max(0.0, (max_time - elapsed) / max_time) + T_end
        # propose a swap
        # randomly select two distinct indices i and j (binary mismatch)
        # efficient: choose a 1-index and a 0-index
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                no_improve_iters = 0
            else:
                no_improve_iters += 1
        else:
            no_improve_iters += 1

        # occasionally random restart if stuck
        if no_improve_iters >= 5000:
            # restart from best with random perturbations
            cur_h = best_h.copy()
            # apply some random swaps
            num_swaps = max(1, int(0.05 * n))
            for _ in range(num_swaps):
                # random swap
                ones_cur = np.where(cur_h == 1)[0]
                zeros_cur = np.where(cur_h == 0)[0]
                i2 = rng.choice(ones_cur)
                j2 = rng.choice(zeros_cur)
                cur_h[i2] = 0.0
                cur_h[j2] = 1.0
            cur_c5 = compute_c5(cur_h)
            no_improve_iters = 0

        iteration += 1
    # final result
    return best_h, float(best_c5), n

def adjust_sum(h, diff):
    # diff may be positive (increase sum) or negative
    # We'll distribute uniformly among entries not at bound
    # Perform iterative loop up to few times
    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        # find indices where we can adjust in needed direction
        if diff > 0:
            idx = np.where(h < 1)[0]
            if len(idx) == 0:
                break
        else:
            idx = np.where(h > 0)[0]
            if len(idx) == 0:
                break
        # compute possible change per index
        delta = diff / len(idx)
        # apply but clip
        if diff > 0:
            inc = np.minimum(delta, 1 - h[idx])
            h[idx] += inc
            actual = inc.sum()
        else:
            dec = np.maximum(delta, -h[idx])
            h[idx] += dec # note dec negative
            actual = dec.sum()
        diff -= actual
    return h
```

    # Iteratively distribute diff among entries not at bound.

import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """C5 value for a candidate h using the prescribed correlation."""
    n = h.shape[0]
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _compute_overlap(h: np.ndarray) -> np.ndarray:
    """Auto‑correlation (overlap) of h with itself, scaled by dx."""
    n = h.shape[0]
    return np.correlate(h, h, mode="full") * (2.0 / n)

def _adjust_sum(vec: np.ndarray, diff: float) -> np.ndarray:
    """
    Adjust `vec` so that its sum changes by `diff` while staying in [0,1].
    The routine distributes the needed amount uniformly among free entries.
    """
    eps = 1e-12
    for _ in range(10):
        if abs(diff) < eps:
            break
        if diff > 0.0:
            idx = np.where(vec < 1.0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            inc = np.minimum(delta, 1.0 - vec[idx])
            vec[idx] += inc
            actual = inc.sum()
        else:
            idx = np.where(vec > 0.0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            dec = np.maximum(delta, -vec[idx])   # both delta and -vec are ≤0
            vec[idx] += dec
            actual = dec.sum()
        diff -= actual
    if abs(diff) > eps:
        free = np.where((vec > 0.0) & (vec < 1.0))[0]
        if free.size:
            i = free[0]
            vec[i] = np.clip(vec[i] + diff, 0.0, 1.0)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
        target_sum = n / 2.0
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            h = _adjust_sum(h, target_sum - cur_sum)
    else:
        n = 200                     # a modest size that fits easily in time budget
        target_sum = n / 2.0
        h = np.zeros(n, dtype=float)
        h[:int(target_sum)] = 1.0   # exactly half ones
        rng.shuffle(h)
        # add a little noise to break symmetry
        h += rng.random(n) * 0.1
        h = np.clip(h, 0.0, 1.0)
        h = _adjust_sum(h, target_sum - h.sum())

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    cur_c5 = best_c5

    start_time = time.time()
    max_time = budget_s * 0.94   # a safety margin

    step_start = 0.20
    step_end = 0.001
    it = 0

    while time.time() - start_time < max_time:
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        step = step_start * (1 - frac) + step_end * frac

        # current overlap (auto‑correlation)
        overlap = _compute_overlap(h)
        center = overlap.size // 2
        # ignore zero lag – it never determines the bound
        overlap_no_center = overlap.copy()
        overlap_no_center[center] = np.inf
        min_idx = int(np.argmin(overlap_no_center))
        lag = min_idx - (n - 1)

        # gradient that increases the overlap for this worst lag
        grad = np.zeros_like(h)
        if lag >= 0:
            grad[: n - lag] += h[lag:]
            grad[lag:] += h[: n - lag]
        else:
            l = -lag
            grad[l:] += h[: n - l]
            grad[: n - l] += h[l:]

        # ascent step + projection
        h_new = h + step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            h_new = _adjust_sum(h_new, diff)

        new_c5 = _compute_c5(h_new)

        # Metropolis‑style acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / max(step, 1e-6)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:

    # ------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------
    return best_h, float(best_c5), n
