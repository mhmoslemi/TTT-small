def project_to_constraints(v, target_sum):
    # v = np.clip(v, 0, 1)
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target_sum) < 1e-10:
        return v
    # Use iterative water-filling: adjusting interior points.
    # We'll implement algorithm similar to projecting onto simplex with box:
    # We can solve as: find λ such that x_i = min(max(v_i + λ,0),1) and sum x_i = target.
    # This is a root-finding problem for λ.
    # Since objective convex, we can do bisection on λ.
    # λ in range [-max(v_i), 1 - min(v_i)], but we can use large range.
    # We'll implement simple bisection.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # After bisection may still be slightly off due to numerical; adjust by adding small delta to interior points.
    # Adjust residual:
    residual = target_sum - x.sum()
    if abs(residual) > 1e-8:
        # Add residual uniformly to points not at bounds (0 or 1)
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # if no free, adjust by scaling
            x = np.clip(x * (target_sum / x.sum()), 0.0, 1.0)
    return x

    h = best_h.copy()
    n = len(h)

    T_decay = (T_end / T_start) ** (1.0 / max_iters)

def project_onto_box_and_sum(x, target_sum):
    # clip to [0,1]
    x_clipped = np.clip(x, 0.0, 1.0)
    # if sum close to target, return
    s = x_clipped.sum()
    if abs(s - target_sum) <= 1e-12:
        return x_clipped
    # Bisection to find λ such that sum(clip(x + λ,0,1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x_clipped + mid, 0.0, 1.0)
        ss = y.sum()
        if ss > target_sum:
            hi = mid
        else:
            lo = mid
    y = np.clip(x_clipped + (lo + hi) / 2.0, 0.0, 1.0)
    # adjust tiny residual
    residual = target_sum - y.sum()
    if abs(residual) > 1e-9:
        # distribute residual among free variables (not at bounds)
        free = (y > 0) & (y < 1)
        if free.any():
            y[free] += residual / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

import time
import numpy as np

def _dx(n):
    return 2.0 / n

def _target_sum(n):
    return n / 2.0  # because sum(h)*dx = 1  -> sum(h) = n/2

def _project_to_constraints(v, target):
    """Project v onto the box [0,1]^n and the hyperplane sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    # Find lambda so that sum(clip(v+lam,0,1)) = target  (monotone in lam)
    lo, hi = -1.0, 1.0
    # expand interval if needed
    while np.clip(v + lo, 0, 1).sum() > target:
        lo *= 2.0
    while np.clip(v + hi, 0, 1).sum() < target:
        hi *= 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction
    resid = target - w.sum()
    if abs(resid) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += resid / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w *= target / w.sum()
    return w

def _c5_value(h):
    """Return the C5 bound for a step vector h."""
    n = h.shape[0]
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient_step(h, shift, eta):
    """Perform one gradient descent step for the worst shift."""
    n = h.shape[0]
    # forward = h_{i+shift}, backward = h_{i-shift}
    fwd = np.zeros_like(h)
    bwd = np.zeros_like(h)
    if shift >= 0:
        fwd[: n - shift] = h[shift:]
        bwd[shift:] = h[: n - shift]
    else:
        k = -shift
        fwd[k:] = h[: n - k]
        bwd[: n - k] = h[k:]
    grad = 1.0 - (fwd + bwd)
    return h - eta * grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # obtain initial vector if provided
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.asarray(init, dtype=float).copy()
        n = h.shape[0]
        target = _target_sum(n)
        h = _project_to_constraints(h, target)
    else:
        n = 70  # modest size, fast to evaluate
        target = _target_sum(n)
        h = np.full(n, 0.5, dtype=float)
        h = _project_to_constraints(h, target)

    cur_score = _c5_value(h)
    best_h = h.copy()
    best_score = cur_score

    # annealing parameters
    T = 0.12
    cooling = 0.99995

    while time.time() - start < budget_s - 0.5:
        # evaluate worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_idx = int(np.argmax(corr))
        shift = worst_idx - (n - 1)

        # gradient descent step
        eta = T * 0.8  # step size proportional to temperature
        h_new = _gradient_step(h, shift, eta)
        h_new = _project_to_constraints(h_new, target)

        new_score = _c5_value(h_new)

        # acceptance rule
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                h = h_new
                cur_score = new_score

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.04, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = _project_to_constraints(h, target)
            cur_score = _c5_value(h)

        T *= cooling

        # safety break if temperature too low
        if T < 1e-6:
            T = 1e-6

    # final best evaluation
    final_score = _c5_value(best_h)
    return best_h, final_score, n
