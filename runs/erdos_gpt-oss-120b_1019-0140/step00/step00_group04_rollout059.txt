def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(conv) * dx
    return c5

def hill_climb(initial_h, n_iter, step_max):
    h = initial_h.copy()
    best = h.copy()
    best_score = compute_c5(h)

    for it in range(n_iter):
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # propose transfer
        # direction: from i to j or from j to i randomly?
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        if h[src] <= 0:
            continue
        if h[dst] >= 1:
            continue
        # max transferrable
        max_delta = min(step_max, h[src], 1 - h[dst])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best = h_new
            best_score = score
            h = h_new
        else:
            # maybe accept with some probability decreasing
            if np.random.rand() < 0.001: # small chance
                h = h_new
    return best, best_score

def anneal(h, time_limit):
    n = len(h)
    S = n/2
    dx = 2/n
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_h = h.copy()
    cur_score = best_score
    T0 = 0.5
    T_end = 1e-4
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit) + T_end * (elapsed / time_limit)  # linear cooling
        # propose a move:
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # direction: transfer delta from i to j if possible else opposite
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        if cur_h[src] <= 0 or cur_h[dst] >= 1:
            continue
        max_delta = min(0.1, cur_h[src], 1 - cur_h[dst])
        delta = np.random.rand() * max_delta
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        # ensure sum unchanged (by construction)
        new_score = compute_c5(new_h)
        # acceptance:
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / (T + 1e-9)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
    return best_h, best_score

import time
import numpy as np
import random

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _anneal_improve(h0, time_budget):
    """Simulated‑annealing style improvement of h on a fixed time budget."""
    n = h0.shape[0]
    target_sum = n / 2.0
    cur_h = h0.copy()
    cur_score = _c5_score(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    # Temperature schedule
    T_start, T_end = 0.12, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T_start * (1 - elapsed / time_budget) + T_end * (elapsed / time_budget)

        # worst shift (the one attaining the max overlap)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)            # shift ∈ [-(n-1), n-1]

        # weight for each coordinate = h[i+shift] + h[i-shift] (where defined)
        weight = np.zeros(n)
        if shift >= 0:
            weight[: n - shift] += cur_h[shift:]
            weight[shift:] += cur_h[: n - shift]
        else:
            s = -shift
            weight[s:] += cur_h[: n - s]
            weight[: n - s] += cur_h[s:]

        # bias source (low weight, positive value) and destination (high weight, not full)
        median_w = np.median(weight)
        low_candidates = np.where((weight <= median_w) & (cur_h > 1e-12))[0]
        high_candidates = np.where((weight >= median_w) & (cur_h < 1 - 1e-12))[0]

        if low_candidates.size == 0 or high_candidates.size == 0:
            # fallback: pick two distinct random indices
            i, j = np.random.randint(n, size=2)
            src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
        else:
            src = int(np.random.choice(low_candidates))
            dst = int(np.random.choice(high_candidates))
            if src == dst:
                continue

        # transfer a small amount from src to dst, respecting bounds
        max_delta = min(0.2, cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta

        # correct any tiny sum drift
        diff = target_sum - new_h.sum()
        if abs(diff) > 1e-10:
            # try to absorb diff in src or dst without breaking bounds
            if 0 <= new_h[src] + diff <= 1:
                new_h[src] += diff
            elif 0 <= new_h[dst] + diff <= 1:
                new_h[dst] += diff
            else:
                # distribute to the first feasible coordinate
                for k in range(n):
                    if 0 <= new_h[k] + diff <= 1:
                        new_h[k] += diff
                        break

        new_score = _c5_score(new_h)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / (T + 1e-15)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
    return best_h, best_score
