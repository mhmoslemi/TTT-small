def project_to_simplex_box(v, t):
    # project onto [0,1] box and sum constraint.
    # Use dual: find λ such that sum_i clip(v_i + λ, 0, 1) = t.
    # λ could be negative or positive. Because clip is monotonic increasing in λ.
    # We can find lower bound λ_low = -np.max(v) (so v+λ ≥0) but maybe more.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < t:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    return w

def project_to_simplex_box(v, target_sum):
    # v: np.ndarray
    # Returns w such that w in [0,1], sum(w)=target_sum, and w is closest to v in L2 sense.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # After bisection, within tolerance may not sum exactly; adjust small residual by adding to first element not at bound:
    residual = target_sum - w.sum()
    if abs(residual) > 1e-12:
        # distribute residual to some element in interior
        # Find indices where w not at bound (0 or 1)
        interior = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if len(interior) > 0:
            w[interior[0]] += residual
        else:
            # all at bounds, adjust first element arbitrarily within bounds
            w[0] = np.clip(w[0] + residual, 0.0, 1.0)
    return w
```

import numpy as np, time, math, random

def compute_c5(h, dx):
    """Maximum overlap integral for the sampled step function."""
    # full linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def gradient_for_shift(h, k, dx):
    """Sub‑gradient of the active shift term (the one attaining the max)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        end = n - k
        # contribution of (1 - h_{i+k})
        g[:end] += (1.0 - h[k:]) * dx
        if k > 0:
            # tail where shifted index is out of range
            g[end:] += 1.0 * dx
        # derivative of the (1 - h_{i+k}) factor
        g[k:] -= h[:end] * dx
    else:
        k2 = -k
        end = n - k2
        g[k2:] += (1.0 - h[:end]) * dx
        if k2 > 0:
            g[:k2] += 1.0 * dx
        g[:end] -= h[k2:] * dx
    return g
