import numpy as np, time

def correlation(h):
    return np.correlate(h, h, mode='full')

def c5_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h is sum_h - correlation_of_h with h
    # compute correlation of h with h
    corr = correlation(h)
    min_corr = np.min(corr)
    # total sum_h = n/2 exactly for feasible h
    C = ((n/2.0) - min_corr) * dx
    return C, min_corr, corr

def gradient_from_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        g[:n-s] += h[s:]
        g[s:] += h[:n-s]
    else:
        s = -shift
        g[s:] += h[:n-s]
        g[:n-s] += h[s:]
    return g

def project_bounds_and_sum(h, total):
    # ensure 0 <= h <= 1 and sum = total (within small tolerance)
    # Simple projection using clipping then uniform redistribution of excess/deficit
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    current = h.sum()
    diff = total - current
    if abs(diff) > 1e-12:
        # distribute diff to indices not at bounds
        free = (h > 0) & (h < 1)
        if free.any():
            # spread linearly
            h[free] += diff / free.sum()
            # clip again (rare)
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds, adjust smallest violation
            # fallback: add diff to first element within bounds
            for i in range(len(h)):

import time

import numpy as np

def _project_to_simplex(x, total):
    """
    Clip x to [0,1] and then adjust to have exact sum == total.
    Simple iterative projection, sufficient for modest dimensions.
    """
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = total - cur
    if abs(diff) < 1e-12:
        return x
    # Distribute the deficit/excess uniformly over free coordinates
    free = (x > 0.0) & (x < 1.0)
    if free.any():
        x[free] += diff / free.sum()
        # Re‑clip in case of overshoot
        x = np.clip(x, 0.0, 1.0)
    else:
        # All entries at a bound – adjust the first non‑saturated entry
        for i in range(len(x)):
            if (diff > 0 and x[i] < 1.0) or (diff < 0 and x[i] > 0.0):
                step = min(abs(diff), 1.0 - x[i]) if diff > 0 else min(abs(diff), x[i])
                x[i] += math.copysign(step, diff)
                diff -= math.copysign(step, diff)
                if abs(diff) < 1e-12:
                    break
    # Final tiny correction
    if abs(x.sum() - total) > 1e-9:
        x[0] += total - x.sum()
        x[0] = min(max(x[0], 0.0), 1.0)
    return x


def _circular_min_corr(h):
    """
    Minimal circular autocorrelation of h.
    Returns (min_corr, shift_index) where shift_index is the integer offset
    giving the minimal sum of products (0 <= shift < n).
    """
    n = h.shape[0]
    # autocorrelation via FFT (real‑valued, circular)
    fft_h = np.fft.rfft(h, n=n)
    corr = np.fft.irfft(fft_h * np.conj(fft_h), n=n)
    idx = int(np.argmin(corr))
    return float(corr[idx]), idx


def _circular_c5(h):
    """
    Compute C5 assuming periodic boundary (the intended definition).
    """
    n = h.shape[0]
    dx = 2.0 / n
    min_corr, _ = _circular_min_corr(h)
    return 1.0 - min_corr * dx


def _evaluate(h):
    """
    Wrapper that uses the pre‑imported evaluator if present,
    otherwise falls back to the periodic implementation above.
    """
    try:
        # The harness provides this helper; it returns the same C5 value.
        return evaluate_erdos_solution(h)
    except NameError:
        return _circular_c5(h)


def _initial_vector(n):
    """
    Produce a feasible starting vector.
    Preference is given to the global `initial_h_values` when it exists.
    """
    try:
        init = np.array(initial_h_values, dtype=np.float64).copy()
        if init.size != n:
            # interpolate to required size
            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
    except NameError:
        # no parent construction – start from the constant 0.5 function
        init = np.full(n, 0.5, dtype=np.float64)

    # enforce strict feasibility
    total = n / 2.0
    return _project_to_simplex(init, total)


def _random_feasible(n, rng):
    """
    Random feasible vector: sample uniform numbers, scale, then project.
    """
    x = rng.random(n)
    total = n / 2.0
    return _project_to_simplex(x, total)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized discretised step function (values in [0,1]).
    c5_bound : float
        Corresponding upper bound on C5.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)

    # Choose a modest resolution (the record uses 42 points)
    n = 42
    h = _initial_vector(n)

    # bookkeeping
    best_h = h.copy()
    best_score = _evaluate(h)
    start_time = time.time()
    target = 0.3808

    # annealing parameters
    T0 = 0.5
    T = T0
    decay = 0.9995
    step = 0.1
    min_step = 0.001
    max_iter = 300000
    no_improve = 0

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.98:
            break

        # current score
        cur_score = _evaluate(h)

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            no_improve = 0
            if best_score <= target:
                break
        else:
            no_improve += 1

        # ---------- directed move based on worst correlation ----------
        min_corr, shift = _circular_min_corr(h)
        # gradient of the worst correlation (circular rolls)
        grad = np.roll(h, shift) + np.roll(h, -shift)

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        # propose moving a little mass from i_min to i_max
        delta_max = min(step, 1.0 - h[i_max], h[i_min])
        if delta_max > 1e-12:
            delta = delta_max * rng.random()
            h_new = h.copy()
            h_new[i_max] += delta
            h_new[i_min] -= delta
        else:
            # fallback to a random pair perturbation
            i, j = rng.choice(n, size=2, replace=False)
            # amount that can be transferred without breaking bounds
            delta_max = min(step, 1.0 - h[i], h[j])
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random()
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

        new_score = _evaluate(h_new)

        # accept if improvement or via Metropolis criterion
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(step, 1e-12)):
            h = h_new

        # schedule updates
        T *= decay
        step = max(step * decay, min_step)

        # occasional random re‑distribution to escape plateaus
        if no_improve >= 5000:
            idx = rng.choice(n, size=4, replace=False)
            total = h[idx].sum()
            rnd = rng.random(4)
            rnd = rnd / rnd.sum() * total
            h[idx] = np.clip(rnd, 0.0, 1.0)
            h = _project_to_simplex(h, n / 2.0)
            no_improve = 0

    # final short refinement with a very small step
    step = min_step
    for _ in range(20000):
        if time.time() - start_time > budget_s * 0.99:
            break
        cur_score = _evaluate(h)
        min_corr, shift = _circular_min_corr(h)
        grad = np.roll(h, shift) + np.roll(h, -shift)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        delta_max = min(step, 1.0 - h[i_max], h[i_min])
        if delta_max <= 0:
            break
        delta = delta_max * rng.random()
        h_new = h.copy()
        h_new[i_max] += delta
        h_new[i_min] -= delta
        new_score = _evaluate(h_new)
        if new_score < cur_score:
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

    return best_h, best_score, n
