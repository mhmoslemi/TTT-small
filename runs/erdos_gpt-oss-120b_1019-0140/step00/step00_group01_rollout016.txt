def _project_onto_capped_simplex(v, target):
    """Project vector v onto the set {x | 0<=x_i<=1, sum(x)=target}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # Quick check
    if v.min() >= 0 and v.max() <= 1 and abs(v.sum() - target) <= 1e-12:
        return v.copy()
    # Binary search for lambda such that sum(clip(v - lam,0,1)) = target
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Tiny residual correction to satisfy the sum exactly (within floating eps)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
            if idx.size:
                i = idx[0]
                inc = min(diff, 1.0 - x[i])
                x[i] += inc

            idx = np.where(x > 1e-12)[0]
            if idx.size:
                i = idx[0]
                dec = min(-diff, x[i])
                x[i] -= dec
    return x

def _compute_c5(h):
    """Return C5 = max correlation * dx for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
