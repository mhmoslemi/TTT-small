import numpy as np, time, math

def project_onto_bound_simplex(v, s):
    # bisection
    n = v.shape[0]
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        cur = proj.sum()
        if cur > s:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # final correction for tiny error:
    diff = s - proj.sum()
    if abs(diff) > 1e-10:
        # distribute diff to free entries
        free = (proj > 0.0 + 1e-9) & (proj < 1.0 - 1e-9)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def gradient_of_worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # term1
        grad[:n - shift] = 1.0 - h[shift:]
        # term2
        if shift > 0:
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        # term1 for indices where i+shift >=0 => i >= s
        grad[s:] = 1.0 - h[:n - s]
        # term2 for i < n - s
        grad[:n - s] -= h[s:]
    return grad, shift

def pair_move(h, rng, max_frac=0.5):
    n = h.shape[0]
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:

    i = rng.choice(pos)
    j = rng.choice(neg)

        donor = i
        recv = j
    else:
        donor = j
        recv = i
    max_amount = min(h[donor], 1.0 - h[recv])
    if max_amount <= 0:
        return h.copy()
    delta = max_amount * rng.random() * max_frac
    new_h = h.copy()
    new_h[donor] -= delta
    new_h[recv] += delta
    return new_h

    scale = step_scale * rng.random()
    cand = h - grad * scale
    # project onto feasible set
    cand = project_onto_bound_simplex(cand, h.shape[0] / 2)
    return cand

import numpy as np
import time
import math

def project_onto_bound_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum(x)=s}."""
    n = v.shape[0]
    # Quick path
    if np.all(v >= 0) and np.all(v <= 1) and abs(v.sum() - s) < 1e-12:
        return v.copy()
    lo = np.min(v) - 1.0          # at lo all entries are clipped to 1 -> sum=n
    hi = np.max(v)                # at hi all entries are clipped to 0 -> sum=0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        cur = proj.sum()
        if cur > s:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    proj = np.clip(v - lam, 0.0, 1.0)
    # Small redistribution for numeric drift
    diff = s - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 0) & (proj < 1)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on a grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def gradient_of_worst_shift(h: np.ndarray):
    """Gradient of the worst (max) overlap term w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[: n - shift] = 1.0 - h[shift:]
        if shift > 0:
            grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] = 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]
    return grad, shift

def pair_move(h: np.ndarray, rng: np.random.Generator, max_frac: float = 0.5) -> np.ndarray:
    """Transfer a random amount between two coordinates while preserving sum."""
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    if rng.random() < 0.5:
        donor, recv = i, j
    else:
        donor, recv = j, i
    max_amount = min(h[donor], 1.0 - h[recv])
    if max_amount <= 0:
        return h.copy()
    delta = max_amount * rng.random() * max_frac
    new_h = h.copy()
    new_h[donor] -= delta
    new_h[recv] += delta
    return new_h

def gradient_move(h: np.ndarray, rng: np.random.Generator, step_scale: float = 0.2) -> np.ndarray:
    """Small step opposite to gradient of the worst shift, then project."""
    grad, _ = gradient_of_worst_shift(h)
    scale = step_scale * rng.random()
    cand = h - grad * scale
    cand = project_onto_bound_simplex(cand, h.shape[0] / 2)
    return cand
