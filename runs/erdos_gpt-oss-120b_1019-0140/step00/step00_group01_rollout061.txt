    n = len(v)
    lo = -1.0  # maybe lower bound as -1
    hi = 1.0   # upper bound as 2? Since we subtract λ, need hi such that all are zero. Actually choose hi = max(v) + 1.
    lo = -max(v) - 1  # ensure v - lo >=1
    hi = max(v) + 1   # ensure v - hi <=0.
    for _ in range(50):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, 0, 1)
    # due to precision may not achieve exact sum; adjust small difference by distributing among free entries not at bounds.
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # Add diff uniformly to entries not at bounds
        free = (x>0) & (x<1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
        else:
            # no free entries: adjust all entries proportionally (rare)
            x += diff / n
            x = np.clip(x, 0, 1)
    return x

Implementation details:

def project_to_capped_simplex(v, target):
    # proj onto [0,1]^n intersect sum = target.
    # Use binary search on lambda.

    # define lambda bounds:
    # For lam very negative, after subtracting lam, all v_i - lam >= 1 => clipped to 1 -> sum=n
    # For lam very positive, all v_i - lam <=0 => clipped to 0 -> sum=0
    # So initial lo=-max(v)-1, hi=max(v)+1 (ensuring these regimes).
    lo = -np.max(v) - 1.0  # ensure v_i - lo >= 1
    hi = np.max(v) + 1.0   # ensure v_i - hi <= 0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    # adjust small numeric error
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # add diff proportionally to free variables (not at bounds)
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
        else:
            # fallback: uniformly distribute diff
            x += diff / len(v)
            x = np.clip(x, 0, 1)
    return x

def compute_c5_and_conv(h):
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = conv.max() * (2.0 / len(h))
    return c5, conv

def compute_gradient(h, shift):
    n = len(h)
    idx_minus = np.arange(n) - shift
    idx_plus = np.arange(n) + shift
    h_minus = np.where((idx_minus >= 0) & (idx_minus < n), h[idx_minus], 0.0)
    h_plus = np.where((idx_plus >= 0) & (idx_plus < n), h[idx_plus], 0.0)
    return 1.0 - (h_minus + h_plus)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    end = start + budget_s - 0.5

    # Determine n and initial h
    if 'initial_h_values' in globals():
        init0 = np.array(initial_h_values, dtype=float)
        n0 = init0.shape[0]
        # Choose target n: maybe 200 or 2*n0
        n = max(200, n0 * 4)  # perhaps 200 is fine
        if n != n0:
            # upsample using linear interpolation
            x0 = np.linspace(0, 1, n0, endpoint=False)
            x = np.linspace(0, 1, n, endpoint=False)
            init_h = np.interp(x, x0, init0)
        else:
            init_h = init0.copy()
    else:
        n = 200
        init_h = np.random.rand(n)

    target_sum = n / 2.0
    init_h = project_to_capped_simplex(init_h, target_sum)

    current_h = init_h
    current_c5, _ = compute_c5_and_conv(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # SA parameters
    T0 = 0.02
    min_T = 1e-6
    step0 = 0.5   # initial perturbation magnitude maybe 0.5 * (range)
    min_step = 0.005
    iter = 0

    while time.time() < end:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.5)
        T = T0 * (1 - frac) + min_T * frac
        step = step0 * (1 - frac) + min_step * frac

        # random perturbation
        delta = np.random.normal(scale=step, size=n)
        delta -= delta.mean()
        cand_h = current_h + delta
        cand_h = project_to_capped_simplex(cand_h, target_sum)

        cand_c5, _ = compute_c5_and_conv(cand_h)

        # accept
        if cand_c5 < current_c5 or np.random.rand() < math.exp(-(cand_c5 - current_c5) / (T + 1e-12)):
            current_h = cand_h
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand_h.copy()

        # local gradient descent attempts
        # Use a few steps of deterministic descent
        for _ in range(3):
            # compute conv for current
            _, conv = compute_c5_and_conv(current_h)
            s_max = np.argmax(conv) - (n - 1)
            grad = compute_gradient(current_h, s_max)
            # deterministic move opposite to gradient
            lr = step * 0.5
            new_h = current_h - lr * grad
            new_h = project_to_capped_simplex(new_h, target_sum)
            new_c5, _ = compute_c5_and_conv(new_h)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
            else:
                break

        # occasional restart
        if iter % 5000 == 0 and iter > 0:
            # small jitter around best
            jitter = np.random.normal(scale=step, size=n)
            jitter -= jitter.mean()
            current_h = best_h + jitter
            current_h = project_to_capped_simplex(current_h, target_sum)
            current_c5, _ = compute_c5_and_conv(current_h)

        iter += 1

    # final refinement: simple gradient descent until time end
    while time.time() < end - 0.1:
        _, conv = compute_c5_and_conv(best_h)
        s_max = np.argmax(conv) - (n - 1)
        grad = compute_gradient(best_h, s_max)
        lr = min_step * 0.5
        new_h = best_h - lr * grad
        new_h = project_to_capped_simplex(new_h, target_sum)
        new_c5, _ = compute_c5_and_conv(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
        else:
            break

    return (best_h.tolist(), float(best_c5), n)

import time
import math
import random

import numpy as np

def _project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum x_i = target}."""
    n = v.size
    lo = -np.max(v) - 1.0          # ensures v - lo >= 1 ⇒ clipped sum = n
    hi = np.max(v) + 1.0           # ensures v - hi <= 0 ⇒ clipped sum = 0
    for _ in range( fifty := 50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_and_corr(h: np.ndarray):
    """Return (C5, full correlation array)."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode='full')
    c5 = conv.max() * (2.0 / n)
    return float(c5), conv


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst shifted correlation w.r.t. h."""
    n = h.size
    idx = np.arange(n) - shift
    idxp = np.arange(n) + shift
    h_minus = np.where((idx >= 0) & (idx < n), h[idx], 0.0)
    h_plus = np.where((idxp >= 0) & (idxp < n), h[idxp], 0.0)
    return 1.0 - (h_minus + h_plus)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) solving the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    end = start + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 200
        h = np.random.rand(n)

    target_sum = n / 2.0
    h = _project_to_capped_simplex(h, target_sum)

    # ------------------------------------------------------------------
    # core optimisation loop (simulated annealing + occasional gradient steps)
    best_h = h.copy()
    best_c5, _ = _c5_and_corr(best_h)

    curr_h = h.copy()
    curr_c5, _ = _c5_and_corr(curr_h)

    # annealing schedule parameters
    T_initial = 0.05
    T_final = 1e-6
    step_initial = 0.3
    step_final = 0.005

    iteration = 0
    while time.time() < end:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.5)

        # temperature & step size decay (linear)
        T = T_initial * (1.0 - frac) + T_final * frac
        step = step_initial * (1.0 - frac) + step_final * frac

        # ---------- random move ----------
        delta = np.random.normal(scale=step, size=n)
        delta -= delta.mean()                     # keep sum zero
        cand = curr_h + delta
        cand = _project_to_capped_simplex(cand, target_sum)

        cand_c5, _ = _c5_and_corr(cand)

        # acceptance (Metropolis)
        if cand_c5 < curr_c5 or np.random.rand() < math.exp(-(cand_c5 - curr_c5) / (T + 1e-12)):
            curr_h, curr_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()

        # ---------- short deterministic refinement ----------
        for _ in range(3):

            shift = np.argmax(conv) - (n - 1)
            grad = _gradient(curr_h, shift)
            lr = step * 0.5
            new_h = curr_h - lr * grad
            new_h = _project_to_capped_simplex(new_h, target_sum)
            new_c5, _ = _c5_and_corr(new_h)
            if new_c5 < curr_c5:
                curr_h, curr_c5 = new_h, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, new_h.copy()
            else:

            jitter -= jitter.mean()
