    low = -1.0
    high = 1.0

    # Let's set low = -1.0, high = 2.0, adapt if needed.
    # Actually lam can be negative large causing all to be >1 then clipped to 1. sum = n.

    # So binary search between a low where sum >= S and high where sum <= S.
    # We can find bounds by expanding.
    # Start low bound as -max(y) - 1 ??? But easier: start low = -2.0, high = 2.0.
    # Ensure sum at low >= S, sum at high <= S.
    low = -2.0
    high = 2.0
    # expand if needed
    def total(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    # adjust low, high
    while total(low) < S:
        low -= 2.0
    while total(high) > S:
        high += 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        s = total(mid)
        if abs(s - S) < tol:
            break
        if s > S:
            low = mid
        else:
            high = mid
    x = np.clip(y - mid, 0, 1)
    # Adjust for any small rounding errors to ensure sum exactly S
    diff = S - np.sum(x)
    if abs(diff) > tol:
        # distribute diff across entries not at bounds
        free_idx = (x > 0) & (x < 1)
        if np.any(free_idx):
            # Add diff uniformly to free variables within bounds
            add_per = diff / np.sum(free_idx)
            # Clip again
            x[free_idx] = np.clip(x[free_idx] + add_per, 0, 1)
            # Possibly iterate again, but fine
    return x

def project_onto_box_sum(y, S, tol=1e-9, max_iter=200):
    n = y.size
    low = -1e5
    high = 1e5
    # but we can compute low as min(y) - 1 maybe.
    low = min(y) - 1.0
    high = max(y) + 1.0
    # enlarge to ensure sum extremes
    def sum_at(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    # Bridge to ensure low gives sum >= S
    while sum_at(low) < S:
        low -= 1.0
    while sum_at(high) > S:
        high += 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        s = sum_at(mid)
        if abs(s - S) < tol:
            break
        if s > S:
            low = mid
        else:
            high = mid
    x = np.clip(y - mid, 0.0, 1.0)
    # adjust small rounding diff
    diff = S - np.sum(x)
    if abs(diff) > tol:
        # distribute diff over free variables
        free = (x > 0) & (x < 1)
        if np.any(free):
            # adjust equally
            adjust = diff / np.sum(free)
            x[free] = np.clip(x[free] + adjust, 0.0, 1.0)
    return x
```

def local_search(h, steps, start_step, end_step, start_temp, end_temp):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    n = len(h)
    S = n/2.0
    for it in range(steps):
        # adjust step
        t = it / steps
        step = start_step * (end_step/start_step) ** t
        temp = start_temp * (end_temp/start_temp) ** t
        z = cur_h + step * np.random.randn(n)
        z = project_onto_box_sum(z, S)
        c5z = compute_c5(z)
        # acceptance
        if c5z < cur_c5 or np.random.rand() < math.exp((cur_c5 - c5z) / max(temp, 1e-12)):
            cur_h = z
            cur_c5 = c5z
            if c5z < best_c5:
                best_c5 = c5z
                best_h = z.copy()
    return best_h, best_c5

def interpolate_h(old_h, new_n):
    old_n = len(old_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)

def compute_gradient(h, k):
    n = len(h)
    g = np.ones(n)
    # term h_{i - k}
    if k >= 0:
        # h_{j - k} exists for j >= k
        g[k:] -= h[:-k]  # subtract h[j-k] from g[j] (j >= k)
        # h_{j + k} exists for j <= n-k-1
        g[:n - k] -= h[k:]
    else:
        # k negative: let kk = -k
        kk = -k
        # h_{j - (-kk)} = h_{j + kk}
        # So we can reuse logic swapping
        # For j such that j + kk < n
        g[:n - kk] -= h[kk:]   # subtract h[j+kk] from g[j]
        # and h_{j + (-kk)} = h_{j - kk}
        g[kk:] -= h[:n - kk]
    return g
