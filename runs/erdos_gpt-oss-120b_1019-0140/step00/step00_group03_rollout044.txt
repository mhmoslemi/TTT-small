import time
import math
import numpy as np

def _eval_c5(h):
    """Return the C5 value for a step function h."""
    try:
        # May be provided by the harness.
        return evaluate_erdos_solution(h)
    except NameError:
        # Fallback to the definition from the statement.
        return np.correlate(h, 1 - h, mode='full').max() * (2.0 / h.size)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound.

    Returns
    -------
    h_values : list of float
        The sampled values of the step function on [0,2].
    c5_bound : float
        The achieved upper bound for C₅.
    n_points : int
        Number of samples (length of h_values).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        n = 70
        h0 = np.full(n, 0.5, dtype=np.float64)
    else:
        n = h0.size

    target_sum = n / 2.0
    s = h0.sum()
    if abs(s - target_sum) > 1e-12:
        h0 = h0 * (target_sum / s)

    h0 = np.clip(h0, 0.0, 1.0)
    # Re‑balance any tiny drift caused by clipping.
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        h0 += diff / n
        h0 = np.clip(h0, 0.0, 1.0)

    current = h0.copy()
    current_score = _eval_c5(current)

    best = current.copy()
    best_score = current_score

    # ------------------------------------------------------------------
    # Simulated‑annealing style hill climbing on continuous values
    # ------------------------------------------------------------------
    temp_start = 5e-3
    temp_end = 1e-7
    step_frac_max = 0.5
    no_improve = 0
    max_no_improve = 200_000

    while True:
        if time.time() - start >= budget_s - 0.2:
            break

        # Linear temperature schedule.
        elapsed = time.time() - start
        t_frac = elapsed / budget_s
        temp = temp_start + (temp_end - temp_start) * t_frac

        # Propose a mass transfer between two distinct indices.
        i, j = rng.choice(n, size=2, replace=False)
        d_max = min(current[i], 1.0 - current[j])
        if d_max <= 0.0:
            continue
        d = d_max * (rng.random() * step_frac_max + 1e-6)
        if d <= 0.0:
            continue

        new = current.copy()
        new[i] -= d
        new[j] += d
        new_score = _eval_c5(new)

        # Acceptance rule.
        if new_score < current_score or rng.random() < math.exp(
            -(new_score - current_score) / max(temp, 1e-12)
        ):
            current = new
            current_score = new_score
            if new_score < best_score - 1e-12:
                best = new.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Occasionally inject a larger random redistribution to escape plateaus.
        if no_improve > max_no_improve:
            high = np.where(current > 0.1)[0]
            low = np.where(current < 0.9)[0]
            if high.size and low.size:
                a = rng.choice(high)
                b = rng.choice(low)
                transfer = min(current[a] * 0.3, 1.0 - current[b])
                current[a] -= transfer
                current[b] += transfer
                current_score = _eval_c5(current)
                if current_score < best_score:
                    best = current.copy()
                    best_score = current_score
            no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    idx_desc = np.argsort(-best)
    binary = np.zeros(n, dtype=np.float64)

    binary_score = _eval_c5(binary)
    if binary_score < best_score:

        best_score = binary_score

    bin_score = best_score

    # ------------------------------------------------------------------
    sum_best = best.sum()
    if abs(sum_best - target_sum) > 1e-12:

        # Adjust the first element (it stays within [0,1] because correction is tiny).
        best[0] = np.clip(best[0] + correction, 0.0, 1.0)

    return best.tolist(), float(best_score), n
