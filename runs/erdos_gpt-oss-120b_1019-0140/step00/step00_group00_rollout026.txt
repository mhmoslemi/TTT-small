def adjust_sum(h, target):
    # Simple iterative projection onto [0,1] and sum using water-filling.
    # Start with h clipped.
    h = np.clip(h, 0.0, 1.0)
    # Compute sum difference
    delta = target - h.sum()
    # While delta not near zero:
    # We'll compute free indices where 0 < h < 1.
    # Distribute delta equally among free.
    # If after distribution some exceed bounds, clip them and recompute free.
    # This converges.
    max_iter = 100
    for _ in range(max_iter):
        if abs(delta) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, cannot adjust further; break
            break
        adjustment = delta / n_free
        h[free] += adjustment
        # Clip again
        np.clip(h, 0.0, 1.0, out=h)
        delta = target - h.sum()
    return h

import time
import numpy as np

def _evaluate_c5(h):
    """Return C5 = max_k ∫ h(x)(1−h(x+k)) dx for the sampled vector."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

def _project_sum_bounds(v, target):
    """
    Euclidean projection of a vector onto the box [0,1] and the affine
    hyperplane sum(v) = target.
    Uses a simple bisection on the Lagrange multiplier λ for the
    formulation x_i = clip(v_i + λ, 0, 1).
    """
    v = np.asarray(v, dtype=float)
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v + lam, 0.0, 1.0)
    return x

def _gradient_max_lag(h, lag, dx):
    """Gradient of the lag‑specific correlation term with respect to h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + lag
    mask = (plus >= 0) & (plus < n)
    grad[idx[mask]] += (1.0 - h[plus[mask]])

    minus = idx - lag
    mask = (minus >= 0) & (minus < n)
    grad[idx[mask]] -= h[minus[mask]]

    grad *= dx
    return grad

def _local_optimize(start_h, time_budget, global_start):
    """Gradient‑descent local search for a fixed time budget."""
    n = start_h.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    h = _project_sum_bounds(start_h, target)
    cur_val = _evaluate_c5(h)
    best_h, best_val = h.copy(), cur_val
    stagnant = 0

    # step magnitudes tried each iteration
    steps = [0.2, 0.5, 1.0, 2.0, 4.0]

    while time.time() - global_start < time_budget:
        # compute full correlation and the worst lag
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(conv.argmax())

        grad = _gradient_max_lag(h, lag, dx)

        cand_h, cand_val = h, cur_val

            start_vec = best_h.copy()
        else:
            start_vec = np.random.rand(n)
        start_vec = _project_sum_bounds(start_vec, target_sum)

        h_opt, val_opt = _local_optimize(start_vec, remaining, start_time)

        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

        # quick exit if we have already beaten the target bound
        if best_val <= 0.3808:
            break

    return best_h, best_val, n
