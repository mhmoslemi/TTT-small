import time
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def shift_contributions(h, shift):
    n = h.shape[0]
    idx = np.arange(n) - shift
    # mask valid
    valid = (idx >= 0) & (idx < n)
    partner = np.zeros_like(h)
    partner[valid] = h[idx[valid]]
    return h * (1.0 - partner)

def repair_sum_and_bounds(h, target):
    # Clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # set uniform
        h[:] = target / len(h)
        return h
    # Scale to meet target sum
    h = h * (target / s)
    # Clip again to enforce bounds, and renormalize any remaining error
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to exact target (minor) by small uniform shift among non-bound entries
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Find indices not at bounds
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            inc = diff / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
    # final scaling if still off
    s2 = h.sum()
    if abs(s2 - target) > 1e-10:
        h = h * (target / s2)
        h = np.clip(h, 0.0, 1.0)
    return h

    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Contribution of each sample to the correlation at a given shift."""
    n = h.shape[0]
    idx = np.arange(n) - shift
    valid = (idx >= 0) & (idx < n)
    partner = np.zeros_like(h)
    partner[valid] = h[idx[valid]]
    return h * (1.0 - partner)

        return h
