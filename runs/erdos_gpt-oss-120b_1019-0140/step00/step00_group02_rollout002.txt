def propose_move(h, step):
    i, j = np.random.choice(len(h), size=2, replace=False)
    # choose direction randomly: either move from i to j or j to i?
    # Running both possibilities:
    # Move delta from i to j
    max_delta = min(h[i], 1 - h[j])  # can move at most h[i] from i to j
    if max_delta <= 0:
        return None, None
    delta = np.random.rand() * max_delta * step
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new, delta

def repair(h):
    n = h.shape[0]
    target = n/2
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    # Now adjust sum via water-filling.
    # If sum > target, need to reduce total; subtract from entries >0 uniformly but respect lower bound 0.
    # If sum < target, need to increase total; add to entries <1 uniformly.
    # Use iterative approach:
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices where we can move in direction
        if diff > 0:
            # need to increase; free indices are those <1
            free = h < 1 - 1e-12
        else:
            free = h > 1e-12
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        # Add delta but clip
        h[free_idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # After loop, ensure sum close enough
    # If still off due to rounding, adjust by small distribution.
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        # add small amount to a randomly selected free entry
        # For diff positive: need increase; pick entry <1
        # For diff negative: need decrease; pick entry >0
        for _ in range(100):
            if diff > 0:
                cand = np.where(h < 1 - 1e-12)[0]
            else:
                cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = np.random.choice(cand)
            # adjust by min(diff, allowed)
            if diff > 0:
                amt = min(diff, 1 - h[i])
            else:
                amt = max(diff, -h[i])
            h[i] += amt
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _repair(h):
    """Project h onto [0,1] and ∑h = n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple water-filling style projection
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        # keep within bounds
        delta = np.clip(delta,
                        -np.min(h[idx] + 1e-15),
                        np.min(1.0 - h[idx] + 1e-15))
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        for _ in range(1000):
            if abs(diff) <= 1e-12:
                break
            if diff > 0:
                cand = np.where(h < 1.0 - 1e-12)[0]
                if cand.size == 0:
                    break
                i = np.random.choice(cand)
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                cand = np.where(h > 1e-12)[0]
                if cand.size == 0:
                    break
                i = np.random.choice(cand)
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
    return h

def _bound(h):
    """Return the C5 bound for a given h."""
    try:
        # pre‑imported helper, may exist
        return evaluate_erdos_solution(h)
    except Exception:
        # fallback: explicit computation
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

def _propose_move(h, step):
    """Generate a feasible candidate by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # Random direction: move from i→j or j→i
    if np.random.rand() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0.0:
        return None
    delta = np.random.rand() * max_delta * step
    if delta <= 0.0:
        return None
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

def _upscale(h):
    """Double the resolution by linear interpolation and repair."""
    n = h.shape[0]
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # Initial construction
    # -----------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: a random feasible vector of length 45
        n0 = kwargs.get('n_points', 45)
        h = np.random.rand(n0)
    h = _repair(h)

    best_h = h.copy()
    best_score = _bound(best_h)

    # -----------------------------------------------------------------
    # Search parameters
    # -----------------------------------------------------------------
    step = 0.5                     # magnitude scale for moves
    max_n = 512                    # do not exceed this resolution
    no_improve = 0
    improvement_patience = 2000    # steps without improvement before reducing step
    last_upscale = 0.0
    upscale_interval = 5.0         # seconds between upscale attempts

    # -----------------------------------------------------------------
    # Main stochastic local search loop
    # -----------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        cand = _propose_move(best_h, step)
        if cand is None:
            continue

        cand_score = _bound(cand)

        if cand_score < best_score - 1e-12:
            best_h = cand
            best_score = cand_score
            step = max(step * 0.99, 0.001)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= improvement_patience:
                step = max(step * 0.5, 0.001)
                no_improve = 0

        # Periodically try to increase resolution
        if (time.time() - start_time - last_upscale) > upscale_interval:
            if len(best_h) * 2 <= max_n:
                up_h = _upscale(best_h)
                up_score = _bound(up_h)
                if up_score <= best_score:
                    best_h = up_h
                    best_score = up_score
                    step = 0.5   # reset step after a successful upscale
                else:
                    # even if worse, keep the finer grid as new search space
                    best_h = up_h
                    best_score = up_score
                last_upscale = time.time()
            else:
                # No further upscaling possible; just continue local search
                pass

    # Final repair (ensures exact feasibility) and final bound recomputation
    best_h = _repair(best_h)
    best_score = _bound(best_h)

    return best_h, float(best_score), best_h.shape[0]
