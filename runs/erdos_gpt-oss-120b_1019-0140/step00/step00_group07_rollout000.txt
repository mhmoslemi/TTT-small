import numpy as np, time, random, math

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def random_binary(n, half):
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    return h

def fix_binary(h):
    # ensure binary and proper sum
    n = h.shape[0]
    target = n // 2
    h = np.clip(h, 0, 1)
    h = (h >= 0.5).astype(np.float64)
    cur = int(h.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zero_idx = np.where(h < 0.5)[0]
            add = np.random.choice(zero_idx, diff, replace=False)
            h[add] = 1.0
        elif diff < 0:
            one_idx = np.where(h > 0.5)[0]
            rem = np.random.choice(one_idx, -diff, replace=False)
            h[rem] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # Determine n and initial vector
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
    except NameError:
        n = 80
        init = None
    target = n // 2
    if init is not None:
        cur_h = fix_binary(init)
    else:
        cur_h = random_binary(n, target)
    best_h = cur_h.copy()
    best_score = evaluate_c5(best_h)
    cur_score = best_score
    # Setup variables for restarts
    last_improve = start
    iter_no_imp = 0
    # Main loop
    while True:
        if time.time() - start > budget_s - 0.2:
            break
        # Temperature schedule
        elapsed = time.time() - start
        t_frac = max(0.0, 1.0 - elapsed / budget_s)
        T = 0.01 + 0.1 * t_frac
        # propose swap
        one_idxs = np.where(cur_h == 1.0)[0]
        zero_idxs = np.where(cur_h == 0.0)[0]
        if one_idxs.size == 0 or zero_idxs.size == 0:
            break
        i = np.random.choice(one_idxs)
        j = np.random.choice(zero_idxs)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = evaluate_c5(new_h)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > random.random():
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                last_improve = time.time()
                iter_no_imp = 0
            else:
                iter_no_imp += 1
        else:
            iter_no_imp += 1
        # restart if stuck too long
        if iter_no_imp > 5000:
            # random restart
            cur_h = random_binary(n, target)
            cur_score = evaluate_c5(cur_h)
            iter_no_imp = 0
    # recompute score of best to be safe
    final_score = evaluate_c5(best_h)
    return (best_h.tolist(), float(final_score), n)

import numpy as np, time, random, math

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def random_binary(n, half):
    h = np.zeros(n, dtype=np.float64)
    h[np.random.choice(n, half, replace=False)] = 1.0
    return h

def fix_binary(h):
    n = h.shape[0]
    target = n // 2
    h = np.clip(h, 0, 1)
    h = (h >= 0.5).astype(np.float64)
    cur = int(h.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(h < 0.5)[0]
            add = np.random.choice(zeros, diff, replace=False)
            h[add] = 1.0
        else:
            ones = np.where(h > 0.5)[0]
            rem = np.random.choice(ones, -diff, replace=False)
            h[rem] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # Determine problem size and starting vector
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
    except NameError:
        n = 80
        init = None
    target = n // 2
    if init is not None:
        cur_h = fix_binary(init)
    else:
        cur_h = random_binary(n, target)
        # quick random improvements
        best_tmp = evaluate_c5(cur_h)
        for _ in range(5):
            h = random_binary(n, target)
            s = evaluate_c5(h)
            if s < best_tmp:
                cur_h = h
                best_tmp = s
    best_h = cur_h.copy()
    best_score = evaluate_c5(best_h)
    cur_score = best_score
    last_imp = start
    no_imp_iters = 0
    # Main stochastic search
    while True:
        if time.time() - start > budget_s - 0.2:
            break
        elapsed = time.time() - start
        t_frac = max(0.0, 1.0 - elapsed / budget_s)
        T = 0.01 + 0.1 * t_frac
        one_idx = np.where(cur_h == 1.0)[0]
        zero_idx = np.where(cur_h == 0.0)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break
        i = np.random.choice(one_idx)
        j = np.random.choice(zero_idx)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = evaluate_c5(new_h)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > random.random():
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                last_imp = time.time()
                no_imp_iters = 0
            else:
                no_imp_iters += 1
        else:
            no_imp_iters += 1
        if no_imp_iters > 5000:
            cur_h = random_binary(n, target)
            cur_score = evaluate_c5(cur_h)
            no_imp_iters = 0
    final_score = evaluate_c5(best_h)
    return (best_h.tolist(), float(final_score), n)
