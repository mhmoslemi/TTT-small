import time, random, math
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(cor.max())

def compute_min_corr_shift_and_gradient(h):
    n = len(h)
    self_corr = np.correlate(h, h, mode='full')
    k_idx = int(np.argmin(self_corr))
    k_shift = k_idx - (n - 1)
    # gradient vector
    g = np.zeros(n, dtype=np.float64)
    if k_shift >= 0:
        # forward
        if k_shift < n:
            g[:n - k_shift] += h[k_shift:]
            g[k_shift:] += h[:n - k_shift]
    else:
        ks = -k_shift
        if ks < n:
            g[ks:] += h[:n - ks]
            g[:n - ks] += h[ks:]
    return k_shift, g

def propose_transfer(h, max_step=0.05):
    n = len(h)
    i_source, i_target = random.sample(range(n), 2)
    # ensure transferring from source (with higher value) to target (with lower?), not necessary

    if h[i_source] > 0 and h[i_target] < 1:

            delta = random.random() * max_transfer

            new_h[i_target] += delta
            return new_h
    # else try opposite direction
    if h[i_target] > 0 and h[i_source] < 1:
        max_transfer = min(max_step, h[i_target], 1.0 - h[i_source])
        if max_transfer > 0:
            delta = random.random() * max_transfer
            new_h = h.copy()
            new_h[i_target] -= delta
            new_h[i_source] += delta
            return new_h
    # no feasible transfer
    return None

def random_perturb(h, max_step=0.2, steps=10):
    n = len(h)
    new_h = h.copy()
    for _ in range(steps):
        i, j = random.sample(range(n), 2)
        max_transfer = min(max_step, new_h[i], 1.0 - new_h[j])
        if max_transfer <= 0:
            continue
        delta = random.random() * max_transfer
        new_h[i] -= delta
        new_h[j] += delta
    return new_h

def enforce_sum_and_bounds(h, target_sum):
    # Clip values first
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return h
    # Distribute diff among free entries (not at bounds)
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    if free.size > 0:
        h[free] += diff / free.size
    else:
        h += diff / len(h)
    # final clip and adjust
    h = np.clip(h, 0.0, 1.0)
    # recompute sum, adjust if still off
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # Add diff to first entry within bounds
        for i in range(len(h)):
            if diff > 0 and h[i] < 1:
                inc = min(diff, 1 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h

import time
import random
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(cor.max())

def propose_transfer(h: np.ndarray, max_step: float = 0.05) -> np.ndarray | None:
    """Randomly move a small amount from one index to another (preserving sum)."""
    n = h.size
    i, j = random.sample(range(n), 2)
    # try i -> j
    if h[i] > 0 and h[j] < 1:
        max_transfer = min(max_step, h[i], 1.0 - h[j])
        if max_transfer > 0:
            delta = random.random() * max_transfer
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
    # try j -> i
    if h[j] > 0 and h[i] < 1:
        max_transfer = min(max_step, h[j], 1.0 - h[i])
        if max_transfer > 0:
            delta = random.random() * max_transfer
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
            return new_h
    return None

def propose_gradient_transfer(h: np.ndarray, max_step: float = 0.05) -> np.ndarray | None:
    """Transfer mass from a low‑gradient index to a high‑gradient index,
    where the gradient is derived from the shift with minimal autocorrelation."""
    n = h.size
    # autocorrelation of h with itself
    ac = np.correlate(h, h, mode='full')
    k_idx = int(np.argmin(ac))
    k_shift = k_idx - (n - 1)

    # gradient of the minimal‑overlap shift
    g = np.zeros(n, dtype=np.float64)
    if k_shift >= 0:
        if k_shift < n:
            g[:n - k_shift] += h[k_shift:]
            g[k_shift:] += h[:n - k_shift]
    else:
        ks = -k_shift
        if ks < n:
            g[ks:] += h[:n - ks]
            g[:n - ks] += h[ks:]

    # admissible source / target sets
    mask_src = h > 1e-12
    mask_tgt = h < 1.0 - 1e-12
    if not np.any(mask_src) or not np.any(mask_tgt):
        return None

    src_candidates = np.where(mask_src)[0]
    tgt_candidates = np.where(mask_tgt)[0]

    s = src_candidates[np.argmin(g[src_candidates])]
    t = tgt_candidates[np.argmax(g[tgt_candidates])]
    if s == t:
        return None

    max_transfer = min(max_step, h[s], 1.0 - h[t])
    if max_transfer <= 0:
        return None
    delta = random.random() * max_transfer
    new_h = h.copy()
    new_h[s] -= delta
    new_h[t] += delta
    return new_h

def random_perturb(h: np.ndarray, max_step: float = 0.2, steps: int = 30) -> np.ndarray:
    """Apply several random mass transfers to explore new regions."""
    n = h.size
    new_h = h.copy()
    for _ in range(steps):
        i, j = random.sample(range(n), 2)
        if new_h[i] > 0 and new_h[j] < 1:
            max_transfer = min(max_step, new_h[i], 1.0 - new_h[j])
        elif new_h[j] > 0 and new_h[i] < 1:
            max_transfer = min(max_step, new_h[j], 1.0 - new_h[i])
            i, j = j, i  # swap direction
        else:
            continue
        if max_transfer <= 0:
            continue
        delta = random.random() * max_transfer
        new_h[i] -= delta
        new_h[j] += delta
    return new_h

def enforce_sum_and_bounds(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h

    free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
    if free.size:
        h[free] += diff / free.size
    else:
        h += diff / h.size

    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # tiny fix by scanning entries
        for i in range(h.size):
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] that yields a low C₅ value."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # type: ignore
        n_points = h0.size
    except Exception:
        n_points = 70
        h0 = np.full(n_points, 0.5, dtype=np.float64)

    target_sum = n_points / 2.0
    h0 = enforce_sum_and_bounds(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start = time.time()
    end = start + budget_s

    init_temp = 0.008
    min_temp = 1e-6
    max_step = 0.05
    no_improve = 0
    max_no_improve = 4000

    while True:
        now = time.time()
        if now >= end - 0.2:
            break

        # linear temperature schedule
        frac = (now - start) / budget_s
        temp = init_temp * (1.0 - frac) + min_temp * frac

        # pick proposal method
        if random.random() < 0.65:
            cand = propose_gradient_transfer(cur_h, max_step=max_step)
            if cand is None:
                cand = propose_transfer(cur_h, max_step=max_step)
        else:
            cand = propose_transfer(cur_h, max_step=max_step)
        if cand is None:
            continue

        cand_score = compute_c5(cand)

        # acceptance
        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if random.random() < prob:
                cur_h, cur_score = cand, cand_score
                no_improve = 0
            else:
                no_improve += 1

        if cur_score < best_score:
            best_h, best_score = cur_h.copy(), cur_score

        # restart when stuck
        if no_improve >= max_no_improve:
            cur_h = random_perturb(best_h, max_step=0.2, steps=30)
            cur_h = enforce_sum_and_bounds(cur_h, target_sum)
            cur_score = compute_c5(cur_h)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = enforce_sum_and_bounds(best_h, target_sum)
    final_score = compute_c5(best_h)
    return (best_h, final_score, n_points)
