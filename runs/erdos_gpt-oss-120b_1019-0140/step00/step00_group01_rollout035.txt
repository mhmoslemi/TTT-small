def project_sum(h, target_sum):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # compute current sum
    s = np.sum(h)
    # if s close enough
    if abs(s - target_sum) < 1e-12:
        return h
    # adjust via scaling and clipping iteratively
    # initially scale if s !=0
    if s >0:
        scale = target_sum / s
        h *= scale
    else:
        h = np.full_like(h, target_sum / len(h))
    # clip again
    h = np.clip(h, 0.0, 1.0)
    # iterative fix
    for _ in range(10):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distribute diff among those not at bounds
        free = (h > 0) & (h < 1)
        free_idx = np.where(free)[0]
        cnt = free_idx.size
        if cnt == 0:
            break
        delta = diff / cnt
        h[free_idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # final fix if any residual diff due to rounding but very small.
    return h

import numpy as np
import time
import math
import random

def compute_c5(h: np.ndarray) -> float:
    """Return the maximum overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector h onto the feasible set:
        0 <= h_i <= 1  and  sum(h) = target
    Simple iterative scaling+clipping suffices for modest dimensions.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # redistribute uniformly (tiny correction)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def transfer(h: np.ndarray, i: int, j: int, eps: float) -> None:
    """Move eps from entry i to entry j (preserving total sum)."""
    h[i] -= eps
    h[j] += eps

def worst_shift(h: np.ndarray):
    """Return (k_star, worst_value) where k_star is the shift producing
    the maximal overlap and worst_value is that overlap (already multiplied by dx)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    worst = corr[idx] * (2.0 / n)
    return k, worst

    h = np.random.rand(n)
