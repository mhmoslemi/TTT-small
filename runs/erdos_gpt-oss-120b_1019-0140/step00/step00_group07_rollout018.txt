def _compute_c5(h):
    """Return the overlap integral C5 for a binary step function h."""
    N = h.shape[0]

    # round to nearest integer to avoid floating noise
    conv = np.rint(conv)
    # maximum overlap (the worst shift)
    return np.max(conv) * dx
