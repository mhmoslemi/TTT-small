**Repair projection onto hypercube [0,1] plus sum constraint S = N/2**.

We'll implement function project_to_feasible(x, target_sum) that ensures x ∈ [0,1] and sum = target.

Method: Use simple approach: Clip x to [0,1], then compute sum s. If s != target, compute diff = target - s. Then while |diff| > 1e-12:

- free_indices = where 0 < x < 1 (i.e., not at bounds) vector.

- If free_indices empty, break.

- delta_per = diff / len(free_indices)

- For each idx in free_indices: x[idx] += delta_per

- Clip again (to within [0,1]), compute new sum and new diff.

This algorithm converges because after each iteration at least one index hits a bound and number of free indices reduces. Complexity O(N^2) worst-case but fine.

def mutate(h, step=0.1):
    n = len(h)
    # choose indices i and j distinct
    i, j = np.random.choice(n, 2, replace=False)
    # choose delta uniformly in [-step, step]
    delta = np.random.uniform(-step, step)
    # adjust copy
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # No direct bound clipping: project after
    new_h = project_to_feasible(new_h, target_sum= n/2)
    return new_h

import time
import numpy as np

def _project_to_feasible(x, target):
    """Project x onto the box [0,1] and enforce sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    # simple iterative water-filling
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        x[free_idx] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _max_corr_unscaled(h):
    """Maximum correlation without the dx factor."""
    return np.max(np.correlate(h, 1 - h, mode="full"))

def _gradient_descent(start_h, target, deadline, max_iter=5000, alpha=30.0, step0=0.5):
    """Projected gradient descent using a smooth max (log‑sum‑exp) surrogate."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_val = _max_corr_unscaled(h)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        a = 1.0 - h
        corr = np.correlate(h, a, mode="full")
        cur_max = corr.max()

        if cur_max < best_val:
            best_val = cur_max
            best_h = h.copy()

        # smooth weights for the max
        exp_term = np.exp(alpha * (corr - cur_max))          # subtract max for stability
        w = exp_term / exp_term.sum()

        # gradient of the smooth surrogate
        grad = np.zeros(n)
        # offsets run from -(n-1) .. (n-1)
        for offset, weight in enumerate(w):
            k = offset - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[: n - k] += weight * a[k:]
                    grad[k:]      -= weight * h[: n - k]
            else:
                kk = -k
                if n - kk > 0:
                    grad[kk:]     += weight * a[: n - kk]
                    grad[: n - kk] -= weight * h[kk:]

        # diminishing step size
        step = step0 * (0.99 ** it)
        h = h - step * grad
        h = _project_to_feasible(h, target)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97   # safety margin

    # ------------------------------------------------------------------
    # Determine the problem size and an initial feasible vector
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.shape[0]
    except Exception:
        init_vec = None
        n = 70                     # fallback size if no parent is provided

    target_sum = n / 2.0                     # ∑h = n/2
    dx = 2.0 / n

    if init_vec is not None:
        h0 = _project_to_feasible(init_vec, target_sum)
    else:
        h0 = _project_to_feasible(np.random.rand(n), target_sum)

    best_h = h0
    best_val = _max_corr_unscaled(best_h)    # unscaled max correlation

    # ------------------------------------------------------------------
    # First descent from the parent (if any) or the random start
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining > 0:
        h_local, val_local = _gradient_descent(
            best_h, target_sum, deadline, max_iter=4000
        )
        if val_local < best_val:
            best_h, best_val = h_local, val_local

    # ------------------------------------------------------------------
    # Random restarts with shorter descents while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # generate a fresh random feasible point
        h_start = _project_to_feasible(np.random.rand(n), target_sum)

        # allocate a modest number of iterations for this restart
        iter_cap = 2000
        h_candidate, val_candidate = _gradient_descent(
            h_start, target_sum, deadline, max_iter=iter_cap
        )
        if val_candidate < best_val:
            best_h, best_val = h_candidate, val_candidate

    # ------------------------------------------------------------------
    # Compute the final scaled bound and return
    # ------------------------------------------------------------------
    c5_bound = best_val * dx
    return best_h, float(c5_bound), n
