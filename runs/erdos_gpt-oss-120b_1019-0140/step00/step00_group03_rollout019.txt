import numpy as np
import time

# -------------------------------------------------------------
# Helper functions (top level, no closures)
# -------------------------------------------------------------
def compute_c5(vec):
    """Return the Erdős overlap constant for a candidate vector."""
    # evaluate_erdos_solution is provided by the harness.
    return evaluate_erdos_solution(vec)


def random_binary_vector(length, ones):
    """Create a binary (0/1) vector of given length with exactly `ones` ones."""
    vec = np.zeros(length, dtype=float)
    if ones > 0:
        idx = np.random.choice(length, ones, replace=False)
        vec[idx] = 1.0
    return vec


def hill_climb(start_vec, required_ones, deadline, max_no_improve=2000):
    """
    Perform a simple swap-based hill climbing from `start_vec`.
    Returns the best vector found and its C5 value.
    """
    cur = start_vec.copy()
    cur_score = compute_c5(cur)
    no_improve = 0
    n = cur.shape[0]

    while no_improve < max_no_improve and time.time() < deadline:
        # indices of 1s and 0s
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        i1 = np.random.choice(ones_idx)
        i0 = np.random.choice(zeros_idx)

        # propose swap
        new = cur.copy()
        new[i1] = 0.0
        new[i0] = 1.0

        new_score = compute_c5(new)

        if new_score < cur_score - 1e-12:
            cur = new
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1

    return cur, cur_score


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h over [0,2] using a simple stochastic search.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # small safety margin

    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        try:
            n_points = int(initial_h_values.shape[0])
        except Exception:
            n_points = 70
    else:
        n_points = 70

    # Required number of ones (integral constraint)
    required_ones = n_points // 2  # n_points is expected to be even

    # -----------------------------------------------------------------
    # Initialize best solution (use parent if it satisfies constraints)
    # -----------------------------------------------------------------
    best_vec = None
    best_score = None

    if 'initial_h_values' in globals():
        try:
            hv = np.array(initial_h_values, dtype=float)
            if hv.shape[0] == n_points and np.all((hv >= 0) & (hv <= 1)):
                total = hv.sum()
                if abs(total - required_ones) < 1e-9:
                    score = compute_c5(hv)
                    best_vec = hv.copy()
                    best_score = score
        except Exception:
            pass

    # Fallback: random binary vector
    if best_vec is None:
        best_vec = random_binary_vector(n_points, required_ones)
        best_score = compute_c5(best_vec)

    # -----------------------------------------------------------------
    # Main stochastic search: repeated hill climbs with random restarts
    # -----------------------------------------------------------------
    max_no_improve = 2000  # per hill climb
    target_bound = 0.3808   # early exit threshold

    while time.time() < deadline:
        # Choose starting point: occasionally continue from global best
        if np.random.rand() < 0.2:
            start = best_vec.copy()
        else:
            start = random_binary_vector(n_points, required_ones)

        # Perform hill climbing
        vec, score = hill_climb(start, required_ones, deadline, max_no_improve)

        # Update global best if improved
        if score < best_score - 1e-12:
            best_vec = vec
            best_score = score
            # Early termination if we reach the target bound
            if best_score <= target_bound:
                break

    # Return the best found construction
    return (best_vec, float(best_score), n_points)
