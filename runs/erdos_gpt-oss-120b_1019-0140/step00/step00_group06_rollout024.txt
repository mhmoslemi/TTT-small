def local_search(h0, budget):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_C5(h)
    n = len(h)
    max_step = 0.5
    while time.time() - start < budget:
        elapsed = time.time() - start
        step = max_step * (1 - elapsed / budget)  # linear decay
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        lower = max(-step, -h[i], h[j] - 1)
        upper = min(step, 1 - h[i], h[j])
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute C5
        val = compute_C5(h_new)
        if val < best_val - 1e-12 or (np.random.rand() < np.exp(-(val - best_val) / (0.001 + 1e-12))):
            h = h_new
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
    return best_h, best_val

import time
import numpy as np

def compute_C5(h):
    """Return the overlap integral C5 for a step function sampled in h."""
    n = h.size
    if n == 0:
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n
    return max_corr * dx

def project_to_sum_and_bounds(v, target_sum):
    """
    Euclidean projection of vector v onto the box [0,1]^n with the additional
    linear equality sum(v) = target_sum.  The solution has the form
    clip(v + λ, 0, 1) for a suitable λ found by binary search.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, 0.0, 1.0)

def _worst_shift_and_gradient(h):
    """Find the shift giving the maximal correlation and the sub‑gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)            # shift ∈ [-(n-1), n-1]
    # gradient of the worst‑shift correlation w.r.t. h
    g = np.ones(n)
    if shift > 0:
        g[:n - shift] -= h[shift:]
        g[shift:]   -= h[:n - shift]
    elif shift < 0:
        s = -shift
        g[s:]       -= h[:n - s]
        g[:n - s]   -= h[s:]
    else:  # shift == 0
        g -= 2.0 * h
    return shift, g, corr, max_idx

def _subgradient_optimize(h0, time_budget):
    """Sub‑gradient descent to minimise C5, returning best vector seen."""
    start = time.time()
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    # initial evaluation
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_val = np.max(corr) * dx
    best_h = h.copy()
    best_val = cur_val
    it = 0
    α0 = 0.5  # initial step size

    while time.time() - start < time_budget:
        it += 1
        shift, g, corr, max_idx = _worst_shift_and_gradient(h)
        # diminishing step size
        step = α0 / np.sqrt(it)
        h_candidate = h - step * g
        # enforce box and sum constraints
        h_candidate = project_to_sum_and_bounds(h_candidate, target_sum=n / 2.0)
        # evaluate candidate
        corr_new = np.correlate(h_candidate, 1.0 - h_candidate, mode='full')
        val_new = np.max(corr_new) * dx

        # accept if not worse
        if val_new <= cur_val:
            h = h_candidate
            corr = corr_new
            cur_val = val_new

        # record the best ever seen
        if val_new < best_val:
            best_val = val_new
            best_h = h_candidate.copy()

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size: keep the size of the provided initial vector if any
    try:
        init_vals = np.asarray(initial_h_values, dtype=float)
        n = init_vals.size
        h0 = init_vals.copy()
    except Exception:
        # fallback: even size 100 for simplicity
        n = 100
        h0 = np.random.rand(n)

    # Ensure feasibility (sum = n/2, values in [0,1])
    h0 = project_to_sum_and_bounds(h0, target_sum=n / 2.0)

    # Allocate time for the optimizer, leaving a small margin for finalisation
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed - 0.2)
    if remaining <= 0.0:
        best_h, best_val = h0, compute_C5(h0)
    else:
        best_h, best_val = _subgradient_optimize(h0, remaining)

    # Final bound computation (precise)
    c5_bound = compute_C5(best_h)

    return best_h, c5_bound, n
