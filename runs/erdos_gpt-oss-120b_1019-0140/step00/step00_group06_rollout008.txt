def hill_climb(h, max_iter, patience):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    no_improve = 0
    for it in range(max_iter):
        # select random swap
        idx_one = np.where(best_h > 0.5)[0]
        idx_zero = np.where(best_h < 0.5)[0]
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        # swap
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        # evaluate
        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-12:
            best_h = new_h
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            break
    return best_h, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n (size)
    try:
        init = initial_h_values
        has_init = isinstance(init, np.ndarray)
    except NameError:
        has_init = False

    if has_init:
        n0 = len(init)
        n = min(380, max(100, n0 * 2))
    else:
        n = 200
    dx = 2.0 / n
    # initial binary vector
    if has_init:
        # Interpolate init to size n
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_cont = np.interp(x_new, x0, init)
        # assign top n/2 indices to 1
        idx = np.argpartition(h_cont, -n//2)[-n//2:]
        h0 = np.zeros(n, dtype=np.float64)
        h0[idx] = 1.0
    else:
        h0 = np.zeros(n, dtype=np.float64)
        h0[rng.choice(n, size=n//2, replace=False)] = 1.0

    def c5_of(vec):
        # compute max correlation of vec and (1 - vec)
        corr = np.convolve(vec, 1 - vec, mode='full')
        return np.max(corr) * dx

    best_h = h0.copy()
    best_score = c5_of(best_h)
    target_score = 0.3808
    # if initial meets target, maybe done
    if best_score <= target_score:
        return best_h, best_score, n

    # Outer loop with restarts
    # Let's allocate max 10 restarts or until time.
    restart = 0
    max_restarts = 30

            max_val = corr.max()
            max_idx = corr.argmax()
            # stop if max_val*dx already <= target, we have solution.
            if max_val * dx <= target_score:
                best_h = h_cur.copy()
                best_score = max_val * dx
                return best_h, best_score, n
            shift = max_idx - (n - 1) # shift range -n+1 .. n-1
            # find contributing positions
            if shift >= 0:
                # positions i where h[i]==1 and h[i+shift]==0
                i_candidates = np.where((h_cur[:n-shift] == 1) & (h_cur[shift:] == 0))[0]
            else:
                s = -shift
                i_candidates = np.where((h_cur[s:] == 1) & (h_cur[:n-s] == 0))[0] + s
            # choose a move
            made_move = False
            if i_candidates.size > 0:
                # pick a random i from candidates
                i = rng.choice(i_candidates)
                # pick a random zero not equal to i
                zero_positions = np.where(h_cur == 0)[0]
                # ensure at least one zero outside i (i is one so not in zeros)
                if zero_positions.size > 0:
                    j = rng.choice(zero_positions)
                    # propose swap
                    h_new = h_cur.copy()
                    h_new[i] = 0.0
                    h_new[j] = 1.0
                    new_score = c5_of(h_new)
                    if new_score < cur_score - 1e-12:
                        h_cur = h_new
                        cur_score = new_score
                        stagnation = 0
                        made_move = True
                        if new_score < best_score:
                            best_score = new_score
                            best_h = h_cur.copy()
                        continue  # continue to next iteration
            # if not made_move or not improve, try random swap few attempts
            attempts = 0
            while attempts < 10:
                ones = np.where(h_cur == 1)[0]
                zeros = np.where(h_cur == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:

                j = rng.choice(zeros)
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = c5_of(h_new)

                    cur_score = new_score
                    stagnation = 0

                    if new_score < best_score:
                        best_score = new_score
                        best_h = h_cur.copy()
                    break
                attempts += 1
            if not made_move:
                stagnation += 1
                if stagnation >= max_stagnation:
                    break
        restart += 1

    # after loops, ensure best_h sum equals n/2
    # may need to correct rounding if floating errors
    # but binary thus fine.

    return best_h, best_score, n

import numpy as np
import time

def compute_c5(h, n):
    """Return C5 = max_corr * dx for binary (or any) vector h."""
    dx = 2.0 / n
    corr = np.convolve(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    """Full correlation array of h with (1-h)."""
    return np.convolve(h, 1 - h, mode='full')

def worst_shift(corr, n):
    """Return shift (int), its max value, and index in correlation array."""
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)
    return shift, max_val, idx

def swap_positions(h, i, j):
    """Swap a 1 at i with a 0 at j (binary vectors)."""
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

def targeted_swap(h, shift, n, rng):
    """Swap a "bad" 1 (contributing to the worst shift) with a random 0."""
    if shift >= 0:
        L = n - shift
        i_cand = np.where((h[:L] == 1) & (h[shift:] == 0))[0]
    else:
        s = -shift
        L = n - s
        i_cand = np.where((h[s:] == 1) & (h[:L] == 0))[0] + s
    if i_cand.size == 0:
        return None
    i = int(rng.choice(i_cand))
    zeros = np.where(h == 0)[0]
    if zeros.size == 0:
        return None
    j = int(rng.choice(zeros))
    return swap_positions(h, i, j)

def random_swap(h, rng):
    """Swap a random 1 with a random 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return None
    i = int(rng.choice(ones))
    j = int(rng.choice(zeros))
    return swap_positions(h, i, j)

def hill_climb(h_start, n, rng, target_c5, deadline):
    """Local improvement starting from h_start."""
    best_h = h_start.copy()
    best_score = compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_score = best_score
    max_stagnation = 2000
    stagnation = 0
    while time.time() < deadline:
        corr = compute_corr(cur_h)
        shift, max_val, _ = worst_shift(corr, n)
        # early exit if target reached
        if max_val * (2.0 / n) <= target_c5:
            return cur_h, max_val * (2.0 / n)
        # attempt targeted improvement
        new_h = targeted_swap(cur_h, shift, n, rng)
        improved = False
        if new_h is not None:
            new_score = compute_c5(new_h, n)
            if new_score < cur_score - 1e-12:
                cur_h, cur_score = new_h, new_score
                stagnation = 0
                improved = True
                if new_score < best_score:
                    best_h, best_score = new_h, new_score
        # fall back to random swaps
        if not improved:
            for _ in range(5):
                new_h = random_swap(cur_h, rng)
                if new_h is None:
                    break
                new_score = compute_c5(new_h, n)
                if new_score < cur_score - 1e-12:

                    stagnation = 0
                    improved = True
                    if new_score < best_score:
                        best_h, best_score = new_h, new_score
                    break
        if not improved:
            stagnation += 1
            if stagnation >= max_stagnation:
                break
    return best_h, best_score
