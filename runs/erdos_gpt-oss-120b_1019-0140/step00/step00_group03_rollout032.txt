import numpy as np, time, math, random

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # Determine problem size and starting vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        # Try to interpret as binary
        if np.all(np.logical_or(np.isclose(init, 0, atol=1e-9), np.isclose(init, 1, atol=1e-9))):
            h0 = (init > 0.5).astype(float)
            target = n // 2
            diff = int(h0.sum()) - target
            if diff != 0:
                if diff > 0:
                    ones = np.where(h0 == 1)[0]
                    np.random.shuffle(ones)
                    h0[ones[:diff]] = 0
                else:
                    zeros = np.where(h0 == 0)[0]
                    np.random.shuffle(zeros)
                    h0[zeros[: -diff]] = 1
        else:
            # fallback to random binary of length n (or 70 if n not even)
            n = max(70, n)
            if n % 2 != 0:
                n += 1
            target = n // 2
            h0 = np.zeros(n, dtype=float)
            h0[np.random.choice(n, target, replace=False)] = 1.0
    else:
        n = 70
        target = n // 2
        h0 = np.zeros(n, dtype=float)
        h0[np.random.choice(n, target, replace=False)] = 1.0

    # Ensure even size and correct sum
    if n % 2 != 0:
        n += 1
        target = n // 2
        h0 = np.zeros(n, dtype=float)
        h0[np.random.choice(n, target, replace=False)] = 1.0
    else:
        target = n // 2
    if int(h0.sum()) != target:
        # repair
        cur = int(h0.sum())
        diff = cur - target
        if diff > 0:
            ones = np.where(h0 == 1)[0]
            np.random.shuffle(ones)
            h0[ones[:diff]] = 0
        elif diff < 0:
            zeros = np.where(h0 == 0)[0]
            np.random.shuffle(zeros)
            h0[zeros[: -diff]] = 1

    dx = 2.0 / n

    # helper
    def score(v):
        return np.max(np.correlate(v, 1 - v, mode='full')) * dx

    current = h0.copy()
    cur_score = score(current)
    best = current.copy()
    best_score = cur_score

    # Simulated annealing parameters
    temperature = max(0.001, cur_score * 0.5)
    decay = 0.9995

    while time.time() - start < budget_s * 0.93:
        # random swap proposal
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if current[i] == current[j]:
            continue
        new = current.copy()
        new[i], new[j] = current[j], current[i]
        new_score = score(new)
        delta = new_score - cur_score
        if delta <= 0 or math.exp(-delta / max(temperature, 1e-12)) > np.random.rand():
            current = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        temperature *= decay
        if temperature < 1e-6:
            temperature = max(0.001, cur_score * 0.5)

    # Greedy local improvement from best solution
    # Run only if time permits
    while time.time() - start < budget_s * 0.99:
        improved = False
        # collect indices of ones and zeros
        ones_idx = np.where(best == 1)[0]
        zeros_idx = np.where(best == 0)[0]
        # Iterate over all possible swaps, track best improvement
        best_delta = 0.0
        best_pair = None
        for i in ones_idx:
            for j in zeros_idx:
                # compute score after swapping i and j
                cand = best.copy()
                cand[i], cand[j] = 0, 1
                cand_score = score(cand)
                delta = cand_score - best_score
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j, cand_score, cand)
        if best_pair is not None:
            i, j, cand_score, cand = best_pair
            best = cand
            best_score = cand_score
            improved = True
        if not improved:
            break

    return (best, float(best_score), n)

import numpy as np
import time
import math
import random

def _compute_c5_corr(h, dx):
    """Correlation‑based C5 evaluation."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem.

    The routine performs a simulated‑annealing search on binary constant‑weight
    vectors, followed by a greedy local improvement phase.
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise a binary vector of even length with exactly half ones.
    # Preference is given to a provided `initial_h_values` if it is binary.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        # Accept only truly binary (within tolerance) initial data
        if np.all(np.logical_or(np.isclose(init_arr, 0.0, atol=1e-9),
                                np.isclose(init_arr, 1.0, atol=1e-9))):
            h0 = (init_arr > 0.5).astype(float)
            # enforce exact half‑weight
            target = n // 2
            diff = int(h0.sum()) - target
            if diff != 0:
                if diff > 0:
                    ones = np.where(h0 == 1)[0]
                    np.random.shuffle(ones)
                    h0[ones[:diff]] = 0
                else:
                    zeros = np.where(h0 == 0)[0]
                    np.random.shuffle(zeros)
                    h0[zeros[: -diff]] = 1
        else:
            # fall back to a fresh 70‑point construction
            n = 70
            target = n // 2
            h0 = np.zeros(n, dtype=float)
            h0[np.random.choice(n, target, replace=False)] = 1.0
    else:
        n = 70
        target = n // 2
        h0 = np.zeros(n, dtype=float)
        h0[np.random.choice(n, target, replace=False)] = 1.0

    # Ensure even length and exact target weight
    if n % 2 != 0:
        n += 1
    target = n // 2
    if int(h0.sum()) != target:
        cur = int(h0.sum())
        diff = cur - target
        if diff > 0:
            ones = np.where(h0 == 1)[0]
            np.random.shuffle(ones)
            h0[ones[:diff]] = 0
        elif diff < 0:
            zeros = np.where(h0 == 0)[0]
            np.random.shuffle(zeros)
            h0[zeros[: -diff]] = 1

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Scoring function – use the pre‑imported helper if present.
    # ------------------------------------------------------------------
    eval_fn = globals().get('evaluate_erdos_solution')
    if eval_fn is not None:
        def score(v):
            return float(eval_fn(v))
    else:
        def score(v):
            return _compute_c5_corr(v, dx)

    # ------------------------------------------------------------------
    # Simulated annealing core
    # ------------------------------------------------------------------
    current = h0.copy()
    cur_score = score(current)
    best = current.copy()
    best_score = cur_score

    temperature = max(0.001, cur_score * 0.5)
    decay = 0.9995

    # Allocate a small buffer for the random index generation
    while time.time() - start_time < budget_s * 0.93:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # Swapping equal bits yields no change – skip
        if current[i] == current[j]:
            continue
        new = current.copy()
        new[i], new[j] = current[j], current[i]
        new_score = score(new)
        delta = new_score - cur_score
        if delta <= 0 or math.exp(-delta / max(temperature, 1e-12)) > random.random():
            current = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        temperature *= decay
        if temperature < 1e-6:
            temperature = max(0.001, cur_score * 0.5)

    # ------------------------------------------------------------------
    # Greedy local improvement from the best point found
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.99:
        improved = False
        ones_idx = np.where(best == 1)[0]
        zeros_idx = np.where(best == 0)[0]
        best_delta = 0.0
        best_swap = None
        for i in ones_idx:
            for j in zeros_idx:
                cand = best.copy()
                cand[i], cand[j] = 0.0, 1.0
                cand_score = score(cand)
                delta = cand_score - best_score
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j, cand, cand_score)
        if best_swap is not None:
            _, _, cand_vec, cand_sc = best_swap
            best = cand_vec
            best_score = cand_sc
            improved = True
        if not improved:
            break

    # Return final result
    return (best, float(best_score), n)
