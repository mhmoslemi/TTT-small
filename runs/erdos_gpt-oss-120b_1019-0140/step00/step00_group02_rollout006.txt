    target_sum = n / 2

    # step parameters

    # schedule: step_size decays every certain iterations
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # select two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi = current_h[i]
        hj = current_h[j]
        # feasible epsilon bounds
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        eps = (np.random.rand() - 0.5) * 2 * step_size
        eps = np.clip(eps, low, high)
        if np.abs(eps) < 1e-12:
            continue
        # propose new state
        current_h[i] = hi + eps
        current_h[j] = hj - eps
        # compute objective (maybe compute only occasional for speed? but compute each)

def repair_h(h):
    n = len(h)
    target = n/2.0
    h = np.clip(h,0,1)
    # Adjust sum by applying scaling factor and clipping iteratively
    for _ in range(5):
        total = h.sum()
        if abs(total - target) < 1e-8:
            break
        # compute scaling factor: want to bring sum to target while respecting bounds
        # scaling towards target: adjust each __free__ element proportionally.
        # Use simple additive correction.
        diff = target - total
        # Determine adjustable indices where h not saturating bounds:
        free = (h > 0) & (h < 1)
        # If no free, break
        if not free.any():
            break
        # distribute diff equally among free entries
        h[free] += diff / free.sum()
        # Clip again
        h = np.clip(h,0,1)
    # final adjust potential small rounding error
    # compute final diff; if not zero, distribute to free entries if possible
    total = h.sum()
    if abs(total - target) > 1e-6:
        # small redistribution
        diff = target - total
        # add to any entry not at bound
        for idx in range(n):
            if diff == 0:
                break
            if diff > 0 and h[idx] < 1:
                inc = min(diff, 1 - h[idx])
                h[idx] += inc
                diff -= inc
            elif diff < 0 and h[idx] > 0:
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_h(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative projection onto capped simplex
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # free indices where adjustment possible
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # distribute diff greedily
            for i in range(n):
                if diff == 0.0:
                    break
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
            # after greedy, break
            break
    # final tiny correction via scaling if still off
    total = h.sum()
    if abs(total - target) > 1e-8:
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)
        # possible slight sum error after clipping, final tweak
        total = h.sum()
        if abs(total - target) > 1e-8:
            # adjust smallest needed

                if diff == 0.0:
                    break
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
    return h

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_h(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            for i in range(n):
                if diff == 0.0:
                    break
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
            break
    total = h.sum()
    if abs(total - target) > 1e-8:
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-12:
            for i in range(n):
                if diff == 0.0:
                    break
                if diff > 0 and h[i] < 1.0:
                    inc = min(diff, 1.0 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0.0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
    return h

def hill_climb(h, max_iter, start_time, budget_s):
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    step = 0.2
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        eps = (np.random.rand() - 0.5) * 2.0 * step
        eps = np.clip(eps, low, high)
        if abs(eps) < 1e-12:
            continue
        cur_h[i] = hi + eps
        cur_h[j] = hj - eps
        new_score = compute_c5(cur_h)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = hi, hj
            no_improve += 1
        if it % max(500, max_iter // 20) == 0 and it > 0:
            if no_improve > max(500, max_iter // 20):
                step = max(step * 0.7, 1e-4)
                no_improve = 0
    return best_h, best_score

def upscale_h(h, new_n):
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return repair_h(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 45
        h = np.random.rand(n0)
        h = repair_h(h)
    h = repair_h(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    current_n = best_h.shape[0]
    max_n = 256
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        new_n = min(current_n * 2, max_n) if current_n < max_n else current_n
        if new_n != current_n:
            best_h = upscale_h(best_h, new_n)
            best_score = compute_c5(best_h)
            current_n = new_n
        est_iter = int(min(50000, max(2000, remaining * 5000 / (current_n / max_n + 1))))
        best_h, best_score = hill_climb(best_h, est_iter, start_time, budget_s)
        if current_n == max_n and remaining < 10.0:
            break
    final_score = compute_c5(best_h)
    return best_h, final_score, int(best_h.shape[0])
