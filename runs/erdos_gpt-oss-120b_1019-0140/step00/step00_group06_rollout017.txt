import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for step values h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_h(h: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the feasible set:
    - entries in [0,1]
    - sum equals n/2 (so that ∫h = 1)
    """
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    target = n / 2.0
    cur = h.sum()
    diff = target - cur

    # Adjust iteratively using any entry that can move in the needed direction
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1.0 - 1e-12)[0]
            if idxs.size == 0:
                break
        else:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
        i = idxs[0]
        if diff > 0:
            delta = min(diff, 1.0 - h[i])
        else:
            delta = max(diff, -h[i])
        h[i] += delta
        cur = h.sum()
        diff = target - cur

    return np.clip(h, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the provided parent solution if available.
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=np.float64)
        n = init_vec.shape[0]
    except NameError:
        # no parent provided – fall back to a uniform 0.5 function
        n = 95
        init_vec = np.full(n, 0.5, dtype=np.float64)

    init_vec = repair_h(init_vec)
    best_h = init_vec.copy()
    best_score = compute_c5(best_h)

    start = time.time()
    no_improve = 0
    restart_threshold = 2000  # after many failures, restart from random

    # ------------------------------------------------------------------
    # Stochastic local search within the allotted budget.
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        # leave a small safety margin before the time limit
        if elapsed >= budget_s - 1.0:
            break

        if no_improve >= restart_threshold:
            # fresh random feasible point
            cand = repair_h(rng.random(n))
            no_improve = 0
        else:
            # perturb the current best solution
            cand = best_h.copy()
            k = rng.integers(1, min(10, n) + 1)          # how many coordinates to touch
            idx = rng.choice(n, size=k, replace=False)   # indices to modify
            delta = rng.normal(scale=0.2, size=k)        # random changes
            delta -= delta.mean()                        # enforce zero sum (preserves integral roughly)
            cand[idx] += delta
            cand = repair_h(cand)

        score = compute_c5(cand)

        if score < best_score - 1e-9:
            best_h = cand
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

    # ------------------------------------------------------------------
    # Return the best feasible construction found.
    # ------------------------------------------------------------------
    return best_h.astype(np.float64), float(best_score), int(n)
