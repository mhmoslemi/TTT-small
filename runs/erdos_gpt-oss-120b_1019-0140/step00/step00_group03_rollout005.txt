    # function is monotonic increasing in lam.
    lo = low - np.max(v) - 1.0
    hi = high - np.min(v) + 1.0
    # ensure that at lo, sum <= target, and at hi sum >= target
    # adjust lo and hi if necessary via while loops up to 50 iterations

            lo -= 1.0

import time
import numpy as np

def project_box_sum(v, target, low=0.0, high=1.0, tol=1e-12, max_iter=100):
    """Project vector v onto the box [low,high]^n while enforcing sum(v)=target."""
    # Ensure target is feasible
    n = v.size
    lo = low - np.max(v) - 1.0
    hi = high - np.min(v) + 1.0
    # Expand bounds if needed to bracket the solution
    for _ in range(50):
        s_lo = np.clip(v + lo, low, high).sum()
        s_hi = np.clip(v + hi, low, high).sum()
        if s_lo > target:
            lo -= 1.0
        elif s_hi < target:
            hi += 1.0
        else:
            break
    # Bisection for lambda
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, low, high)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v + (lo + hi) * 0.5, low, high)

def evaluate_h(h, dx):
    """Return C5 for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def shift_and_gradient(h):
    """Return the shift achieving the maximal overlap and its subgradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                # integer shift (can be negative)
    grad = np.zeros_like(h)

    if shift >= 0:
        s = shift
        # contribution from (1 - h[i+s])
        grad[:n - s] += 1.0 - h[s:]
        # contribution from -h[i-s]
        grad[s:] -= h[:n - s]
    else:
        s = -shift
        # (1 - h[i-s]) term (i >= s)
        grad[s:] += 1.0 - h[:n - s]
        # -h[i+s] term (i < n-s)
        grad[:n - s] -= h[s:]
    return shift, grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = rng.random(n)
    target_sum = n / 2.0
    dx = 2.0 / n

    # Project initial guess onto feasible set
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, target_sum)

    # Initialize state
    current = h0.copy()
    cur_score = evaluate_h(current, dx)
    best = current.copy()
    best_score = cur_score

    # Parameters for the descent / annealing
    max_frac = 0.95  # use most of the allotted time
    time_limit = start_time + budget_s * max_frac
    iteration = 0

    # Main loop
    while time.time() < time_limit:
        # Compute worst shift and subgradient
        shift, grad = shift_and_gradient(current)

        # Schedule step size (larger early, smaller later)
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * max_frac)
        step = (0.12 * (1.0 - frac) + 0.004 * frac)  # heuristic schedule

        # Gradient descent step (projected)
        proposal = current - step * grad
        proposal = np.clip(proposal, 0.0, 1.0)
        proposal = project_box_sum(proposal, target_sum)

        prop_score = evaluate_h(proposal, dx)

        # Accept if improvement (simple descent)
        if prop_score <= cur_score:
            current = proposal
            cur_score = prop_score

        # Record best solution found
        if cur_score < best_score:
            best = current.copy()
            best_score = cur_score
            # Early exit if we already beat the target bound
            if best_score <= 0.3808:
                break

        # Every few hundred iterations, inject a small random perturbation around the best
        if iteration % 500 == 0 and iteration > 0:
            noise = rng.normal(scale=step * 0.6, size=n)
            perturbed = best + noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = project_box_sum(perturbed, target_sum)
            pert_score = evaluate_h(perturbed, dx)
            if pert_score < best_score:
                best = perturbed.copy()
                best_score = pert_score
                current = perturbed.copy()
                cur_score = pert_score

        iteration += 1

    return best.tolist(), float(best_score), n
