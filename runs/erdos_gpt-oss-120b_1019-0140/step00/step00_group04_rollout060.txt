def project_onto_box_sum(x, target, eps=1e-12, max_iter=100):
    lo = -1.0
    hi = 1.0
    # expand bound to ensure coverage
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)/2.0, 0, 1)
    # adjust exact sum due to numeric errors
    diff = target - y.sum()
    if abs(diff) > eps:
        # distribute diff to first component not at boundary
        idxs = np.where((y > 0 + eps) & (y < 1 - eps))[0]
        if len(idxs) > 0:
            y[idxs[0]] += diff
            # clip to be safe
            y = np.clip(y, 0, 1)
        else:
            # fallback: adjust first element within bounds
            y[0] = np.clip(y[0] + diff, 0, 1)
    return y
```
