def _compute_c5(x, n=None):
    """Return the C5 value for a step vector x (len n)."""
    if n is None:
        n = x.shape[0]
    # correlation of x with (1-x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    max_val = np.max(corr)
    return max_val * (2.0 / n)

def _repair(vec, required_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    x = np.clip(vec, 0.0, 1.0)
    # Simple proportional scaling, then iterative redistribution
    total = np.sum(x)
    if total == 0.0:
        # Fill with 1's up to required sum
        k = int(math.floor(required_sum))
        x[:k] = 1.0
        if k < len(x):
            x[k] = required_sum - k
        return np.clip(x, 0.0, 1.0)

    # First scaling
    scale = required_sum / total
    x *= scale
    x = np.clip(x, 0.0, 1.0)

    # Iterative redistribution to fix any residual error
    for _ in range(10):
        cur = np.sum(x)
        diff = required_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        x[free] = np.clip(x[free] + delta, 0.0, 1.0)
    # Final tiny adjustment if needed
    cur = np.sum(x)
    diff = required_sum - cur
    if abs(diff) > 1e-10:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:

                x[i] += inc

            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])

                diff += dec

        if elapsed >= time_limit:

        i, j = np.random.choice(n, 2, replace=False)
        xi, xj = cur[i], cur[j]
