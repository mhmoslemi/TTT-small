def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    # Set seeds
    np.random.seed(seed)
    random.seed(seed)

    # Get initial if provided
    initial = kwargs.get('initial_h_values')
    if initial is None:
        # If there is a global variable named initial_h_values in environment?
        try:
            initial = globals()['initial_h_values']
        except KeyError:
            initial = None

    if initial is not None:
        h = np.array(initial, dtype=float)
    else:
        # set default n_points = 60 maybe
        n_points = 60
        # create random feasible h
        # generate random values uniform (0,1)
        v = np.random.rand(n_points)
        target_sum = n_points/2.0
        h = project_feasible(v, target_sum)

    n = h.shape[0]
    target_sum = n/2.0
    dx = 2.0/n

    # Ensure feasible
    h = project_feasible(h, target_sum)

    def compute_score(vec):
        # compute max correlation * dx
        c = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(c) * dx

    best_h = h.copy()
    best_score = compute_score(best_h)
    current_h = h.copy()
    current_score = best_score

    start_time = time.time()
    elapsed = 0.0

    max_temp = 0.1
    min_temp = 0.0001
    total_budget = budget_s
    # initial sigma
    max_sigma = 0.5

    # counters for no improvement
    no_improve = 0
    # choose target iteration count maybe 200k but limited by time
    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_budget * 0.99:
            break
        # temperature schedule linear cooling
        t = 1.0 - elapsed / total_budget
        temp = max(min_temp, max_temp * t)
        sigma = max_sigma * t

        # generate neighbor
        # small perturbation
        new_h = current_h + sigma * np.random.randn(n)
        # project
        new_h = project_feasible(new_h, target_sum)

        new_score = compute_score(new_h)

        # acceptance
        if new_score < current_score or random.random() < math.exp((current_score - new_score) / (temp + 1e-12)):
            current_h = new_h
            current_score = new_score
            # record if new best
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random restart if stagnated
        if no_improve > 5000:
            # new random feasible start
            v = np.random.rand(n)
            current_h = project_feasible(v, target_sum)
            current_score = compute_score(current_h)
            no_improve = 0

    # Return best
    # Convert to list or numpy
    return best_h.tolist(), float(best_score), int(n)

import numpy as np
import random
import time
import math

def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum(v)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0  # bounds wide enough for any feasible shift
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _c5_score(vec: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on a uniform grid.
    """
    n = vec.shape[0]
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(np.max(corr) * dx)

def _random_feasible(n: int) -> np.ndarray:
    """Create a random feasible vector of length n."""
    v = np.random.rand(n)
    target = n / 2.0
    return _project_to_feasible(v, target)
