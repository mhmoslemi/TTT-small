def binary_hill_climb(n, init_bin=None, time_limit):
    # produce initial binary if not given
    if init_bin is None:
        arr = np.zeros(n, dtype=int)
        arr[:n//2] = 1
        np.random.shuffle(arr)
        h = arr
    else:
        # convert init_h to binary by threshold 0.5
        h = (init_h >= 0.5).astype(int)
        # ensure sum matches n//2: adjust if needed
        diff = h.sum() - n//2
        if diff > 0:
            # too many ones; turn some ones to zero
            idx_one = np.where(h==1)[0]
            np.random.shuffle(idx_one)
            h[idx_one[:diff]] = 0
        elif diff < 0:
            idx_zero = np.where(h==0)[0]
            np.random.shuffle(idx_zero)
            h[idx_zero[: -diff]] = 1
    best_h = h.copy()
    best_bound = compute_c5(best_h.astype(float))  # synergy
    # record indices sets
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    while time.time() - start < time_limit:
        # propose swap
        i = random.choice(ones)
        j = random.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        # recompute bound
        bound = compute_c5(h.astype(float))
        # if improvement
        if bound < best_bound - 1e-9:
            best_bound = bound
            best_h = h.copy()
            # update ones/zeros lists accordingly
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # optional: break if best_bound <= target
        if best_bound <= target_bound:
            break
    return best_h, best_bound
```

def hill_climb_binary(n, weight, init_bin, time_limit, target_max_mismatch=None):
    import time, random, numpy as np, math
    start = time.time()
    if init_bin is not None:
        h = init_bin.copy()
        # ensure sum weight
        cur_sum = int(h.sum())
        if cur_sum != weight:
            # adjust using set conversion
            delta = cur_sum - weight
            if delta > 0:
                # too many ones
                ones_idx = np.where(h==1)[0]
                np.random.shuffle(ones_idx)
                h[ones_idx[:delta]] = 0
            elif delta < 0:
                zeros_idx = np.where(h==0)[0]
                np.random.shuffle(zeros_idx)
                h[zeros_idx[: -delta]] = 1
    else:
        h = np.zeros(n, dtype=int)
        h[:weight] = 1
        np.random.shuffle(h)
    # sets
    ones = set(int(i) for i in np.where(h==1)[0])
    zeros = set(int(i) for i in np.where(h==0)[0])
    cur_max = max_mismatch(h)
    best_h = h.copy()
    best_max = cur_max
    # temperature schedule
    start_T = max(1.0, cur_max)
    # main loop
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = start_T * max(0.0, 1.0 - elapsed / time_limit)
        # random swap
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # perform swap
        h[i] = 0
        h[j] = 1
        new_max = max_mismatch(h)
        diff = new_max - cur_max
        accept = False
        if diff <= 0:
            accept = True
        else:
            if T > 0 and random.random() < math.exp(-diff / T):
                accept = True
        if accept:
            cur_max = new_max
            # update sets
            ones.remove(i); zeros.add(i)
            zeros.remove(j); ones.add(j)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                if target_max_mismatch is not None and best_max <= target_max_mismatch:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, best_max
```

import time, random, math
import numpy as np

def _max_mismatch(h):
    """Maximum of h(x)*(1-h(x+k)) over all integer shifts k (full correlation)."""
    # h is int or float array with values in [0,1]
    return int(np.max(np.correlate(h, 1 - h, mode='full')))

def _adjust_weight(arr, target):
    """Flip entries to obtain exactly target ones (arr is binary)."""
    cur = int(arr.sum())
    if cur > target:
        # turn some ones into zeros
        ones = np.where(arr == 1)[0]
        np.random.shuffle(ones)
        arr[ones[:cur - target]] = 0
    elif cur < target:
        zeros = np.where(arr == 0)[0]
        np.random.shuffle(zeros)
        arr[zeros[:target - cur]] = 1
    return arr

def _initial_binary(n, weight, init_h=None):
    """Return a binary vector of length n with exactly `weight` ones."""
    if init_h is not None:
        bin_vec = (init_h >= 0.5).astype(int)
        bin_vec = _adjust_weight(bin_vec, weight)
        return bin_vec
    # random start
    v = np.zeros(n, dtype=int)
    v[:weight] = 1
    np.random.shuffle(v)
    return v

def _hill_climb_binary(n, weight, start_vec, time_limit, target_max=None):
    """Simulated‑annealing on binary vectors to minimise the max mismatch."""
    start = time.time()
    h = start_vec.copy()
    ones = set(int(i) for i in np.where(h == 1)[0])
    zeros = set(int(i) for i in np.where(h == 0)[0])
    cur_max = _max_mismatch(h)
    best_h = h.copy()
    best_max = cur_max
    # temperature schedule (roughly weight)
    start_T = max(1.0, cur_max)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = start_T * max(0.0, 1.0 - elapsed / time_limit)
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # perform swap
        h[i] = 0
        h[j] = 1
        new_max = _max_mismatch(h)
        diff = new_max - cur_max
        accept = diff <= 0 or (T > 0 and random.random() < math.exp(-diff / T))
        if accept:
            cur_max = new_max
            # update sets
            ones.remove(i); zeros.add(i)
            zeros.remove(j); ones.add(j)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                if target_max is not None and best_max <= target_max:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n = int(init_h.shape[0])
    else:
        init_h = None
        n = 70                     # default size (even, under 1k)
    if n % 2 == 1:               # enforce even length for exact half‑weight binary solution
        n += 1
    weight = n // 2               # required number of 1‑entries
    dx = 2.0 / n
    # Target bound (converted to maximal integer mismatch)
    target_bound = 0.3808
    target_max = int(math.floor(target_bound / dx + 1e-12))

    # ------------------------------------------------------------
    # Generate a few deterministic candidates (irrational rotations)
    # ------------------------------------------------------------
    candidates = []

    # use the provided construction if any
    if init_h is not None:
        cand = _initial_binary(n, weight, init_h)
        candidates.append(cand)

    # irrational‑slope sequences
    alphas = [np.sqrt(2), np.sqrt(3), (np.sqrt(5) - 1) / 2, np.pi - 3, np.e - 2, 0.123456789, 0.375]
    idx = np.arange(n, dtype=float)
    for a in alphas:
        cand = ((idx * a) % 1) < 0.5
        cand = cand.astype(int)
        cand = _adjust_weight(cand, weight)
        candidates.append(cand)

    # a few random starts
    for _ in range(5):
        cand = np.zeros(n, dtype=int)
        cand[:weight] = 1
        np.random.shuffle(cand)
        candidates.append(cand)

    # ------------------------------------------------------------
    # Evaluate candidates, keep the best
    # ------------------------------------------------------------
    best_vec = None
    best_max = n + 1
    for cand in candidates:
        mm = _max_mismatch(cand)
        if mm < best_max:
            best_max = mm
            best_vec = cand.copy()
        if best_max <= target_max:
            break

    # ------------------------------------------------------------
    # If necessary, launch a stochastic optimisation phase
    # ------------------------------------------------------------
    start_time = time.time()
    time_budget = budget_s * 0.97   # small safety margin
    while time.time() - start_time < time_budget:
        remaining = time_budget - (time.time() - start_time)
        # allocate a slice (max 30 s) for a hill‑climb run
        slice_time = min(30.0, remaining)
        if slice_time <= 0.5:
            break
        # start from current best (or a fresh random binary)
        start_vec = best_vec if best_vec is not None else _initial_binary(n, weight)
        vec, mm = _hill_climb_binary(
            n, weight, start_vec, slice_time, target_max_mismatch=target_max
        )
        if mm < best_max:
            best_max = mm
            best_vec = vec.copy()
        if best_max <= target_max:
            break
        # next iteration start from a new random seed to diversify
        best_vec = None

    # ------------------------------------------------------------
    # Final result preparation
    # ------------------------------------------------------------
    if best_vec is None:                     # fallback (should not happen)
        best_vec = np.zeros(n, dtype=int)
        best_vec[:weight] = 1
    h_vals = best_vec.astype(float)
    c5_bound = best_max * dx                  # convert integer mismatch to integral value
    return h_vals, c5_bound, n
