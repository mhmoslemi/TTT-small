def local_search(init_h, time_limit):
    n = init_h.size
    cur_h = init_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h_local = cur_h.copy()
    best_c5_local = cur_c5
    start_local = time.time()
    # We'll schedule T0 based on time_left maybe
    T0 = 0.05
    while time.time() - start_local < time_limit:
        # propose swap
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h <= 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        cand_c5 = compute_c5(cur_h)
        # compute acceptance
        elapsed = time.time() - start_local
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        if cand_c5 < cur_c5:
            cur_c5 = cand_c5
        else:
            # revert with probability
            if T > 0 and rng.random() < math.exp(-(cand_c5 - cur_c5) / T):
                cur_c5 = cand_c5
            else:
                # revert swap
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                continue
        if cur_c5 < best_c5_local:
            best_c5_local = cur_c5
            best_h_local = cur_h.copy()
            if best_c5_local <= target_c5:   # target variable defined
                break
    return best_h_local, best_c5_local

def local_search_binary(h0, time_limit, rng, target):
    h = h0.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_limit:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # determine shift with max corr
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        # compute shifted for this shift
        if k >= 0:
            shifted = np.concatenate((np.zeros(k), h[:n - k]))
        else:
            kk = -k
            shifted = np.concatenate((h[kk:], np.zeros(kk)))
        mask_i = (h == 1) & (shifted == 0)
        mask_j = (h == 0) & (shifted == 1)
        if mask_i.any() and mask_j.any():
            i = rng.choice(np.flatnonzero(mask_i))
            j = rng.choice(np.flatnonzero(mask_j))
        else:
            # random swap
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # propose swap
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h)
        elapsed = time.time() - start
        remaining = time_limit - elapsed
        # temperature schedule
        T0 = 0.02
        T = T0 * max(0.0, remaining / time_limit)
        if new_val < cur_val or (T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= target:
                    break
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        # if time left small break
        if time.time() - start >= time_limit:
            break
    return best_h, best_val

import numpy as np, time, math

def _c5_value(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on samples."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _make_binary(vec, rng):
    """Round to {0,1} and enforce exactly half ones."""
    h = np.rint(vec).clip(0, 1).astype(np.float64)
    n = h.shape[0]
    target = n // 2
    cur = int(h.sum())
    if cur < target:
        zeros = np.flatnonzero(h == 0)
        if zeros.size:
            idx = rng.choice(zeros, size=target - cur, replace=False)
            h[idx] = 1.0
    elif cur > target:
        ones = np.flatnonzero(h == 1)
        if ones.size:
            idx = rng.choice(ones, size=cur - target, replace=False)
            h[idx] = 0.0
    return h

def _local_search_binary(h0, time_limit, rng, target):
    """Hill‑climb / simulated‑annealing on binary vectors."""
    h = h0.copy()
    n = h.shape[0]
    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.05  # initial temperature for SA
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)  # actual shift
        # shifted version of h for this lag
        if k >= 0:
            shifted = np.concatenate((np.zeros(k, dtype=h.dtype), h[: n - k]))
        else:
            kk = -k
            shifted = np.concatenate((h[kk:], np.zeros(kk, dtype=h.dtype)))
        mask_i = (h == 1) & (shifted == 0)   # 1 mismatched to 0
        mask_j = (h == 0) & (shifted == 1)   # 0 mismatched to 1
        if mask_i.any() and mask_j.any():
            i = rng.choice(np.flatnonzero(mask_i))
            j = rng.choice(np.flatnonzero(mask_j))
        else:
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # propose swap
        h[i], h[j] = h[j], h[i]
        new_val = _c5_value(h)
        # temperature schedule (linear cooling)
        remain = time_limit - elapsed
        T = T0 * max(0.0, remain / time_limit)
        if new_val < cur_val or (T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= target:
                    break
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_val

def _legendre_seq(p):
    """Binary Legendre sequence of length p‑1 (p odd prime)."""
    n = p - 1
    seq = np.empty(n, dtype=np.float64)
    for i in range(1, p):
        # Legendre symbol: 1 for quadratic residue, -1 otherwise (mod p)
        ls = pow(i, (p - 1) // 2, p)
        seq[i - 1] = 1.0 if ls == 1 else 0.0
    return seq

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_val = float('inf')
    best_n = None
    target_c5 = 0.3808

    # gather starting constructions
    candidates = []

    # use provided initial construction if present
    if 'initial_h_values' in globals():
        cand = _make_binary(np.array(initial_h_values, dtype=np.float64), rng)
        candidates.append(cand)

    # Legendre sequences for several primes
    for p in (79, 83, 101, 151, 179, 199):
        n = p - 1
        if n > 1000:
            continue
        cand = _legendre_seq(p)  # already binary with half ones
        candidates.append(cand)

    # a few random binary vectors of various lengths
    for n in (80, 100, 120, 150, 200):
        if n % 2:  # need even length for half ones
            continue
        for _ in range(2):
            h = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=n // 2, replace=False)
            h[ones] = 1.0
            candidates.append(h)

    # random order to avoid bias
    rng.shuffle(candidates)

    # iterate over candidates, allocating time dynamically
    for idx, cand in enumerate(candidates):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 4.0:
            break
        # allocate a slice of the remaining budget (up to 30 s)
        slice_time = min(remaining * 0.4, 30.0)
        # local optimisation on this candidate
        opt_h, opt_val = _local_search_binary(cand, slice_time, rng, target_c5)
        if opt_val < best_val:
            best_val = opt_val
            best_h = opt_h.copy()
            best_n = opt_h.shape[0]
            if best_val <= target_c5:
                break

    # fallback – simple balanced alternating pattern
    if best_h is None:
        n = 80
        h = np.zeros(n, dtype=np.float64)
        # ensure exactly n/2 ones
        ones_idx = rng.choice(n, size=n // 2, replace=False)
        h[ones_idx] = 1.0
        best_h = h
        best_val = _c5_value(best_h)
        best_n = n

    return (best_h, best_val, best_n)
