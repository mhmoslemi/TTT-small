def project_to_feasible(y):
    # Clip to [0,1] and adjust sum.
    N = len(y)
    target = N / 2.0
    # Clip initially
    y = np.clip(y, 0.0, 1.0)

    # Function f(lam)
    def f(lam):
        return np.sum(np.clip(y + lam, 0.0, 1.0))

    # Compute current sum
    cur_sum = np.sum(y)
    if abs(cur_sum - target) < 1e-10:
        return y
    # If cur_sum < target: need positive lam; else negative lam.
    low = -1.0
    high = 1.0
    # Ensure low and high bracket target.
    # Expand if needed.
    while f(low) > target:
        low *= 2
    while f(high) < target:
        high *= 2
    for _ in range(50):
        mid = (low + high) / 2.0
        val = f(mid)
        if val < target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    y_proj = np.clip(y + lam, 0.0, 1.0)
    # Might still have slight deviation because numeric; apply small correction:
    diff = target - np.sum(y_proj)
    if abs(diff) > 1e-12:
        # Distribute diff uniformly among interior points (not at bounds)
        interior = (y_proj > 1e-12) & (y_proj < 1-1e-12)
        count = np.sum(interior)
        if count > 0:
            y_proj[interior] += diff / count
            # Clip again
            y_proj = np.clip(y_proj, 0.0, 1.0)
    return y_proj

def subgradient_step(h, step):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (N - 1)
    grad = np.zeros_like(h)
    if lag >= 0:
        # grad[i] += (1 - h[i + lag]) for i < N - lag
        grad[:N - lag] += 1 - h[lag:]
        # grad[i] -= h[i - lag] for i >= lag
        grad[lag:] -= h[:N - lag]
    else:
        lag_abs = -lag
        # i+lag = i - lag_abs >=0 => i>=lag_abs
        grad[lag_abs:] += 1 - h[:N - lag_abs]
        # -h[i - lag] = -h[i + lag_abs] for i < N - lag_abs
        grad[:N - lag_abs] -= h[lag_abs:]
    # step descent
    new_h = h - step * grad
    # project to feasible
    return project_to_feasible(new_h)

def project_to_feasible(y):

    target = N / 2.0
    # Clip to bounds
    y = np.clip(y, 0.0, 1.0)
    # Determine sum
    total = np.sum(y)
    if abs(total - target) < 1e-12:
        return y
    # Use binary search for lambda
    # Because we want sum(clip(y + lambda, 0,1)) = target
    low = -1.0
    high = 1.0
    # adjust bounds to bracket target
    def f(lam):
        return np.sum(np.clip(y + lam, 0.0, 1.0))
    # Expand bounds if needed
    while f(low) > target:
        low *= 2
        if low < -1e6:
            break
    while f(high) < target:
        high *= 2
        if high > 1e6:
            break
    # binary search
    for _ in range(50):
        mid = (low + high) / 2.0
        val = f(mid)
        if val < target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0

import time
import math
import random
import numpy as np

def _compute_c5(h):
    """Return the max overlap C5 for a step function defined by samples h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_to_feasible(y):
    """Project vector y onto the feasible set:
       0 <= y_i <= 1 and sum(y) = N/2.
    """
    N = y.shape[0]
    target = N / 2.0
    y = np.clip(y, 0.0, 1.0)

    total = np.sum(y)
    if abs(total - target) < 1e-12:
        return y

    # function of lambda: sum clip(y + lam, 0, 1)
    def f(lam):
        return np.sum(np.clip(y + lam, 0.0, 1.0))

    low, high = -1.0, 1.0
    # Expand interval until it brackets target
    while f(low) > target:
        low *= 2.0
        if low < -1e6:
            break
    while f(high) < target:
        high *= 2.0
        if high > 1e6:
            break

    for _ in range(50):
        mid = (low + high) / 2.0
        val = f(mid)
        if val < target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    y = np.clip(y + lam, 0.0, 1.0)

    # Minor residual correction (rare)
    diff = target - np.sum(y)
    if abs(diff) > 1e-10:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = np.count_nonzero(interior)
        if cnt > 0:
            y[interior] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

def _subgradient_step(h, step):
    """One sub‑gradient descent step aimed at reducing the maximal overlap."""
    N = h.shape[0]
    # correlation vector
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    lag = max_idx - (N - 1)          # lag ∈ [-(N-1), N-1]

    grad = np.zeros_like(h)

    if lag >= 0:
        # (1 - h[i+lag]) term
        grad[: N - lag] += 1.0 - h[lag:]
        # -h[i-lag] term
        grad[lag:] -= h[: N - lag]
    else:
        lag_abs = -lag
        # (1 - h[i+lag]) term where i+lag = i - lag_abs >= 0
        grad[lag_abs:] += 1.0 - h[: N - lag_abs]
        # -h[i-lag] term where i-lag = i + lag_abs
        grad[: N - lag_abs] -= h[lag_abs:]

    new_h = h - step * grad
    return _project_to_feasible(new_h)

def _random_proposal(h, step):
    """Random perturbation with subsequent projection."""
    perturb = (np.random.rand(h.shape[0]) - 0.5) * 2.0 * step
    return _project_to_feasible(h + perturb)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h (as a list), its C5 bound, and the number of points.

    The algorithm mixes sub‑gradient descent with occasional random moves,
    respecting the constraints:
        0 ≤ h[i] ≤ 1,
        sum(h) = n_points/2   (integral = 1 over [0,2]).
    """
    np.random.seed(seed)
    random.seed(seed)

    # Starting vector: use the provided construction if present,
    # otherwise a simple uniform one.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 70
        h0 = np.full(n, 0.5, dtype=np.float64)

    N = h0.shape[0]
    h0 = _project_to_feasible(h0)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    start = time.time()
    deadline = start + budget_s * 0.95  # keep margin

    # Scheduling parameters
    init_step = 0.30
    final_step = 0.001
    init_temp = 0.05
    final_temp = 0.001

    # Main loop
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)

        step = init_step * (1.0 - frac) + final_step * frac
        temp = init_temp * (1.0 - frac) + final_temp * frac

        # 80% sub‑gradient, 20% random mutation
        if random.random() < 0.8:
            new_h = _subgradient_step(cur_h, step)
        else:
            new_h = _random_proposal(cur_h, step)

        new_c5 = _compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5

        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Return as list for compatibility
    return (best_h.tolist(), float(best_c5), N)
