def compute_c5(h):
    n = h.shape[0]
    # compute cross-correlation with complement using numpy's correlate
    # but careful: np.correlate returns 1-D array of size n + n - 1
    # Using mode='full' same as convolution that slides over shift from -(n-1) to (n-1)
    # Actually np.correlate uses sum_i a[i] * b[i+lag], but for mode='full' it's same.
    # We can compute conv = np.correlate(h, 1 - h, mode='full')
    # Then multiply by dx = 2.0 / n
    return np.max(conv) * (2.0 / n)

import time
import math
import numpy as np

def compute_c5(h):
    """Return the C5 value for a binary (or [0,1]) step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h with (1‑h), zero‑padded outside the interval
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def _initial_binary(rng, n, init_arr=None):
    """Produce a binary vector of length n with exactly n//2 ones.
    If init_arr is supplied and already binary, keep its pattern (balancing if needed)."""
    target = n // 2
    if init_arr is not None:
        arr = np.clip(np.asarray(init_arr, dtype=float), 0.0, 1.0)
        # already binary?
        if np.all((arr == 0) | (arr == 1)):
            binary = arr.copy()
            cur = int(binary.sum())
            if cur != target:
                # balance by flipping random bits
                if cur > target:
                    # turn some ones to zero
                    ones = np.flatnonzero(binary == 1)
                    flip = rng.choice(ones, cur - target, replace=False)
                    binary[flip] = 0.0
                else:
                    zeros = np.flatnonzero(binary == 0)
                    flip = rng.choice(zeros, target - cur, replace=False)
                    binary[flip] = 1.0
            return binary
        # not binary – take the largest entries as ones
        idx = np.argpartition(-arr, target)[:target]
        binary = np.zeros(n, dtype=float)
        binary[idx] = 1.0
        return binary
    # no supplied vector – random balanced binary
    binary = np.zeros(n, dtype=float)
    binary[rng.choice(n, target, replace=False)] = 1.0
    return binary

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that lowers the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points)."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    init_global = globals().get("initial_h_values", None)
    if init_global is not None:
        n = int(np.asarray(init_global).size)
    else:
        n = 80
    h_curr = _initial_binary(rng, n, init_global)
    cur_score = compute_c5(h_curr)

    best_h = h_curr.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Parameters for the stochastic search
    # ------------------------------------------------------------------
    # Temperature schedule (linear cooling)
    T_initial = 0.006
    # after this many non‑improving steps we perform a lightweight restart
    max_no_improve = 3000
    no_improve_cnt = 0
    # time allowance: keep a small margin to avoid hard termination
    time_limit = budget_s * 0.95

    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = max(1e-5, T_initial * (1.0 - frac))

        # choose a random 1‑index and a random 0‑index to keep the sum constant
        ones = np.flatnonzero(h_curr == 1)
        zeros = np.flatnonzero(h_curr == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        h_new = h_curr.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        new_score = compute_c5(h_new)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h_curr = h_new
            cur_score = new_score
            no_improve_cnt = 0
            # record best
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
                # early exit if we already beat the target record
                if best_score <= 0.3808:
                    break
        else:
            no_improve_cnt += 1

        # occasional restart to escape deep local minima
        if no_improve_cnt >= max_no_improve:

            h_curr = best_h.copy()
            cur_score = best_score
            # with some probability replace by a fresh random vector
            if rng.random() < 0.15:
                h_curr = _initial_binary(rng, n)
                cur_score = compute_c5(h_curr)
            no_improve_cnt = 0

    # ------------------------------------------------------------------
    # Return the best result discovered
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_score), int(n))
