import time
import numpy as np

# Helper: evaluate_erdos_solution is pre‑imported by the harness.

def _initial_binary_vector(n, init=None, rng=None):
    """Return a binary (0/1) vector of length n with exactly n/2 ones."""
    half = n // 2
    if init is not None and len(init) == n:
        v = np.where(np.asarray(init) >= 0.5, 1.0, 0.0)
        cur = int(v.sum())
        diff = half - cur
        if diff > 0:
            zeros = np.where(v == 0)[0]
            if len(zeros) >= diff:
                v[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(v == 1)[0]
            if len(ones) >= -diff:
                v[ones[: -diff]] = 0.0
        return v.astype(np.float64)
    # fallback random binary vector
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 1.0
    rng.shuffle(v)
    return v

def _score(h):
    """Compute the C5 bound for a candidate vector h."""
    return evaluate_erdos_solution(h)

def _annealing_search(h, time_limit, rng):
    """Simulated annealing on a binary vector h respecting sum constraint."""
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_score = _score(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # temperature schedule
    T = 1.0
    decay = 0.9995
    stagnation = 0
    max_stagnation = 5000

    while time.time() - start < time_limit:
        # pick a random 1 and a random 0 to swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = _score(cur_h)
        delta = new_score - cur_score

        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = cur_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            stagnation += 1

        # temperature decay
        T *= decay

        # occasional random restart on stagnation
        if stagnation > max_stagnation:
            cur_h = best_h.copy()
            # perturb a few random positions
            for _ in range(5):
                a, b = rng.choice(n, 2, replace=False)
                cur_h[a], cur_h[b] = cur_h[b], cur_h[a]
            cur_score = _score(cur_h)
            stagnation = 0

    return best_h, best_score

def _local_optimize(h, time_limit):
    """Greedy improvement by trying all single swaps until no gain."""
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_score = _score(best_h)

    while time.time() - start < time_limit:
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        cur_score = best_score
        for i in ones:
            for j in zeros:
                best_h[i], best_h[j] = best_h[j], best_h[i]
                new_score = _score(best_h)
                if new_score < cur_score - 1e-12:
                    best_score = new_score
                    cur_score = new_score
                    improved = True
                    break
                else:
                    # revert
                    best_h[i], best_h[j] = best_h[j], best_h[i]
            if improved:
                break
        if not improved:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 constant for a step function.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # Determine problem size
    try:
        init_vec = initial_h_values
    except NameError:
        init_vec = None

    n = None
    if init_vec is not None:
        n = len(init_vec)
    if n is None:
        n = int(kwargs.get("n_points", 70))
    if n % 2 != 0:
        n += 1  # enforce even length for exact half‑weight

    # Build initial binary vector
    h0 = _initial_binary_vector(n, init=init_vec, rng=rng)

    # Allocate time: 90% for annealing, remaining 10% for local improvement
    start_total = time.time()
    anneal_time = budget_s * 0.9
    local_time = budget_s * 0.1

    # Main annealing loop (single run, but may restart internally)
    best_h, best_score = _annealing_search(h0, anneal_time, rng)

    # Final greedy refinement if time permits
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 5.0:  # need a few seconds for local search
        best_h, best_score = _local_optimize(best_h, min(local_time, remaining))

    # Ensure the returned vector meets the exact sum constraint
    # Minor numeric adjustments (should already be satisfied)
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # fix by flipping smallest magnitude entries
        diff = int(round(target_sum - cur_sum))
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            best_h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(best_h > 0.5)[0]
            best_h[ones[: -diff]] = 0.0

    # Final score (re‑evaluate to be safe)
    final_score = _score(best_h)

    return best_h, float(final_score), n
