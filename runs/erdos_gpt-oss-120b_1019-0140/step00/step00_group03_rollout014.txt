import numpy as np
import random
import time

def compute_c5(h, dx):
    """Maximum overlap C5 for binary step function h (0/1 values)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * dx

def local_search(h, dx):
    """Greedy descent by swapping one 1 and one 0 until no improvement."""
    cur = h.copy()
    cur_val = compute_c5(cur, dx)
    tol = 1e-12
    while True:
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur < 0.5)[0])
        random.shuffle(ones)
        random.shuffle(zeros)
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                val = compute_c5(cur, dx)
                if val < cur_val - tol:
                    cur_val = val
                    improved = True
                    break
                # revert
                cur[i] = 1.0
                cur[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return cur, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a binary step function h, its C5 bound, and number of points."""
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size and target sum of h.
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)

    else:

        dx = 2.0 / n
