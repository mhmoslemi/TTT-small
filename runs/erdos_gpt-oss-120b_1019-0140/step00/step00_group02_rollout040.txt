def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    # Returns projection onto sum constraint and box constraints.
    # We'll find lambda such that sum_i clamp(v_i - lambda, lo, hi) = s.
    # Use bisection; lambda lower bound for f(lambda)=sum_i clamp(v_i - lambda, lo, hi).
    # Monotone decreasing.
    # Lower lambda: very low => v_i - lambda huge => clamp to hi => sum = n*hi
    # Upper lambda: very high => clamp to lo => sum = n*lo
    # So search between lowl = min(v_i) - hi, highl = max(v_i) - lo, maybe extended.
    n = v.size
    lo_lambda = min(v) - hi
    hi_lambda = max(v) - lo
    # but we need to ensure function values bracket s.
    # Compute f at lo_lambda and hi_lambda.
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # expand bounds if needed
    f_lo = f(lo_lambda)
    f_hi = f(hi_lambda)
    # if s outside [f_hi, f_lo], assign all lo or hi
    if s >= f_lo:
        return np.full_like(v, hi)
    if s <= f_hi:
        return np.full_like(v, lo)
    # else binary search
    for _ in range(max_iter):
        mid = (lo_lambda + hi_lambda) / 2
        fm = f(mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lo_lambda = mid
        else:
            hi_lambda = mid
    return np.clip(v - mid, lo, hi)
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=100):
    n = v.shape[0]
    # early return if sum already close
    # Use lambda function f
    lo_lambda = np.min(v) - hi
    hi_lambda = np.max(v) - lo
    # define function f
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds to bracket s
    f_lo = f(lo_lambda)
    f_hi = f(hi_lambda)
    if f_lo < s:
        # need to increase sum (the lambda is too high?), maybe set all to lo?
        # Actually f_lo is sum at low lambda => largest values (hi). If f_lo < s (shouldn't happen because sum at low lambda is max possible = n*hi)
        # In case s > n*hi, return hi
        return np.full_like(v, hi)
    if f_hi > s:
        return np.full_like(v, lo)
    # binary search
    for _ in range(max_iter):
        mid = (lo_lambda + hi_lambda) / 2.0
        fm = f(mid)
        if np.abs(fm - s) < 1e-12:
            break
        if fm > s:
            lo_lambda = mid
        else:
            hi_lambda = mid
    return np.clip(v - mid, lo, hi)

def hill_climbing_descent(h_initial, n_iter, step0):
    h = h_initial.copy()
    n = len(h)
    s = n/2
    dx = 2.0/n
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(n_iter):
        # compute shift and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        # choose i_max and i_min
        i_max = np.argmax(grad)  # largest positive gradient (decrease)
        i_min = np.argmin(grad)  # most negative (increase)
        # if no improvement possible continue random move
        if grad[i_max] <= 0 or grad[i_min] >= 0:
            # random transfer move
            i, j = rng.integers(0, n, size=2)
            # ensure distinct
            while j == i:
                j = rng.integers(0, n)
            # compute feasible delta
            delta_low = max(h[i] - 1.0, -h[j])  # lower bound for delta (negative)
            delta_high = min(h[i], 1.0 - h[j])  # upper bound for delta (positive)
            if delta_low >= delta_high:
                continue
            delta = rng.uniform(delta_low, delta_high) * (0.5)  # scaling
        else:
            # transfer from i_max (decrease) to i_min (increase)
            delta_max = min(h[i_max], 1.0 - h[i_min])
            if delta_max <= 0:
                continue
            # step size schedule
            max_step = step0 / np.sqrt(it + 1)
            delta = min(delta_max, max_step) * rng.random()
        # apply move
        h[i_max] -= delta
        h[i_min] += delta
        # compute score
        cur_score = compute_c5(h)
        # if not improvement, revert
        if cur_score <= best_score:
            best_score = cur_score
            best_h = h.copy()
        else:
            # revert
            h[i_max] += delta
            h[i_min] -= delta
    return best_h, best_score
```

import time, math
import numpy as np

def upsample(arr, new_n):
    """Linearly interpolate arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, m, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    # Clip to bounds for extreme cases
    if s <= lo * v.size:
        return np.full_like(v, lo)
    if s >= hi * v.size:
        return np.full_like(v, hi)

    # monotone decreasing f(lambda) = sum clip(v - lambda, lo, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_gradient(h, shift):
    """Gradient of the max-correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[:n - shift] = h[shift:]
        term2[shift:] = h[:n - shift]
    else:
        s = -shift
        term1[s:] = h[:n - s]
        term2[:n - s] = h[s:]
    return 1.0 - term1 - term2

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def projected_gradient_descent(h0, n_iter=2000, step0=0.5):
    """Simple projected gradient descent on the max-correlation objective."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, n_iter + 1):
        # correlation and shift of the active constraint
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def thue_morse(n):
    """Return the Thue–Morse binary sequence of length n (0/1 values)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Random binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_score = float('inf')
    target_n = 180  # primary resolution for intensive search

    # ------------------------------------------------------------------
    # 1. Base candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) Upsampled initial construction if present
    if 'initial_h_values' in globals():
        h0 = upsample(np.asarray(initial_h_values, dtype=float), target_n)
        h0 = project_onto_box_sum(h0, target_n / 2.0)
        candidates.append(h0)

    # b) Thue–Morse sequences (powers of two with exact half‑ones)
    for n in (64, 128, 256):
        if n % 2 == 0:
            tm = thue_morse(n)
            # sum is exactly n/2 for powers of two
            candidates.append(tm)

    # c) A few random binary and fractional starts
    for _ in range(4):
        n = rng.choice([64, 128, 256])
        candidates.append(random_binary_balanced(n, rng))
    for _ in range(4):
        n = rng.choice([64, 128, 256])
        v = rng.random(n)
        candidates.append(project_onto_box_sum(v, n / 2.0))

    # ------------------------------------------------------------------
    # 2. Optimize each candidate (more iterations for the primary size)
    # ------------------------------------------------------------------
    for base_h in candidates:
        if time.time() - start_time > budget_s - 0.5:
            break
        n = base_h.size
        # allocate iterations proportionally to size (more for larger n)
        if n == target_n:
            iters = 6000
        else:
            iters = 3000
        h_opt, score_opt = projected_gradient_descent(base_h, n_iter=iters, step0=0.5)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # ------------------------------------------------------------------
    # 3. Random restarts with short refinement until time expires
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.5:
        # pick a size at random (favor the primary size)
        n = rng.choice([target_n, target_n, 128, 256, 64])
        v = rng.random(n)
        h_start = project_onto_box_sum(v, n / 2.0)
        # short run
        h_opt, score_opt = projected_gradient_descent(h_start, n_iter=1500, step0=0.5)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

    # ------------------------------------------------------------------
    # 4. Final sanity projection (ensure feasibility) and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback uniform construction
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_n / 2.0)
        best_score = compute_c5(best_h)
    else:
        # final projection to guarantee constraints (tiny numerical drift possible)
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)

    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
