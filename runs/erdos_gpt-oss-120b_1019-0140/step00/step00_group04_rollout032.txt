    base_n = len(initial_h_values) if 'initial_h_values' in globals() else 42
    n_target = base_n * 2
    if n_target % 2 != 0:
        n_target += 1

    if 'initial_h_values' in globals():
        h = init_h_from_initial(n_target)
    else:
        y = np.random.rand(n_target)
        h = project_onto_capped_simplex(y, n_target / 2.0)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    time_limit = budget_s * 0.95

    T0 = 0.02
    max_step = 0.2
    no_improve = 0
    max_no_improve = 20000

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        T = T0 * (1 - fraction)

        i = random.randrange(n_target)
        if cur_h[i] < 1e-8:
            continue
        j = random.randrange(n_target)
        if i == j or cur_h[j] > 1.0 - 1e-8:
            continue

        max_delta = min(cur_h[i], 1.0 - cur_h[j], max_step)
        if max_delta <= 1e-12:
            continue
        delta = random.random() * max_delta

        cur_h[i] -= delta
        cur_h[j] += delta

        new_c5 = compute_c5(cur_h)
        delta_c5 = new_c5 - cur_c5

        accept = False
        if delta_c5 <= 0:
            accept = True
        else:
            if T > 1e-12 and random.random() < math.exp(-delta_c5 / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve = 0
                if best_c5 <= 0.3808:
                    break
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
            no_improve += 1

        if fraction > 0.5:
            max_step = max(0.01, max_step * 0.9999)

        if no_improve >= max_no_improve:
            y = np.random.rand(n_target)
            cur_h = project_onto_capped_simplex(y, n_target / 2.0)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    return (best_h, best_c5, n_target)
