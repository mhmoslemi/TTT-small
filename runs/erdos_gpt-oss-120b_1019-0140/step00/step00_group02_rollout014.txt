def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math

    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    remaining = lambda: budget_s - (time.time() - start)

    best_h = None
    best_val = float('inf')
    n_values = [45, 60, 90] # list of n points sizes to try
    for n in n_values:
        target_sum = n / 2.0
        dx = 2.0 / n
        # Initialize h
        if n == len(initial_h_values):
            h = np.array(initial_h_values, dtype=float)
            # ensure sum correct; maybe correct slight deviations by projection
            h = project_onto_box_and_sum(h, target_sum)
        else:
            # random feasible: sample uniform in [0,1], then project
            v = np.random.rand(n)
            h = project_onto_box_and_sum(v, target_sum)
        # evaluate
        cur_val = compute_c5(h, dx)
        # SA parameters
        temp = 0.2
        temp_min = 1e-4
        alpha = 0.9995  # cooling
        # neighbor sigma start
        sigma = 0.2
        sigma_min = 0.01
        # iteration loop
        steps = 0
        while remaining() > 0.0:
            # adapt sigma maybe decreasing
            # generate neighbor
            v = h + np.random.randn(n) * sigma
            h_new = project_onto_box_and_sum(v, target_sum)
            new_val = compute_c5(h_new, dx)
            # acceptance
            if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
                h = h_new
                cur_val = new_val
            # track best
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                best_n = n
                # optionally break early if improvement less than tolerance?
            # cooling schedule
            temp = max(temp * alpha, temp_min)
            sigma = max(sigma * alpha, sigma_min)
            steps += 1
            # optional early break if improvement plateau and time left > enough to move to next n?
            # but keep simple.
        # after phase time check:
        if remaining() <= 0:
            break
    # final return
    # compute final best val with dx appropriate for its n_points
    dx_best = 2.0 / best_n
    final_val = compute_c5(best_h, dx_best)
    # ensure sum constraint satisfied (maybe rounding errors)
    # Return as list for h_values
    return best_h.tolist(), float(final_val), int(best_n)

import time, math, random
import numpy as np

def project_to_feasible(v, target_sum):
    """Project vector v onto [0,1]^n with sum = target_sum using bisection."""
    lo, hi = -1.0, 1.0
    # Ensure that hi yields sum >= target_sum; at hi all entries are 1.
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)

def compute_c5(h, dx):
    """Return max_{k} integral h(x)*(1-h(x+k)) dx for discrete h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_neighbor(h, sigma, target_sum):
    """Add Gaussian noise and project back to feasible set."""
    v = h + np.random.randn(h.shape[0]) * sigma
    return project_to_feasible(v, target_sum)

def gradient_neighbor(h, sigma, target_sum):
    """Step opposite to the gradient of the worst-shift correlation."""
    n = h.shape[0]
    # find shift giving the maximum correlation
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)  # integer shift, can be negative
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        j = i + shift
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    else:
        t = -shift
        i = np.arange(t, n)
        j = i - t
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    norm = np.linalg.norm(g)
    if norm > 0:
        step = -sigma * g / norm
    else:
        step = np.zeros_like(h)
    v = h + step
    return project_to_feasible(v, target_sum)

def generate_neighbor(h, sigma, target_sum):
    """Mix random and gradient-driven perturbations."""
    if random.random() < 0.7:
        return random_neighbor(h, sigma, target_sum)
    else:
        return gradient_neighbor(h, sigma, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()

        init_vec = np.array(initial_h_values, dtype=float)

    best_val = float('inf')
    best_n = None

    # candidate point counts – start with the known size and grow
    sizes = [45, 60, 90, 120, 180]
    for n in sizes:
        if time.time() - start >= budget_s:
            break
        target_sum = n / 2.0
        dx = 2.0 / n

        if init_vec is not None and init_vec.shape[0] == n:
            h = project_to_feasible(init_vec.copy(), target_sum)
        else:
            # random feasible start
            h = project_to_feasible(np.random.rand(n), target_sum)

        cur_val = compute_c5(h, dx)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            best_n = n

        # Simulated‑annealing style search
        temp = 0.5
        temp_min = 1e-5
        sigma = 0.2
        sigma_min = 0.01
        cooling = 0.9995
        no_improve = 0
        max_no_improve = 2000

        while time.time() - start < budget_s:

            cand = generate_neighbor(h, sigma, target_sum)
            cand_val = compute_c5(cand, dx)

            # acceptance

                h = cand
                cur_val = cand_val
                no_improve = 0
            else:
                no_improve += 1

            # keep global best
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                best_n = n
                no_improve = 0

            # cooling
            temp = max(temp * cooling, temp_min)
            sigma = max(sigma * cooling, sigma_min)

            # restart if stuck
            if no_improve >= max_no_improve:
                h = project_to_feasible(np.random.rand(n), target_sum)
                cur_val = compute_c5(h, dx)
                no_improve = 0

            # optional early exit for this size
            if time.time() - start > budget_s * 0.95:
                break

    # if nothing found (unlikely) fall back to uniform
    if best_h is None:
        n = 45
        best_n = n
        best_h = np.full(n, 0.5)
        best_val = compute_c5(best_h, 2.0 / n)

    # final projection to be safe and recompute bound
    best_h = project_to_feasible(best_h, best_n / 2.0)
    final_val = compute_c5(best_h, 2.0 / best_n)
