import numpy as np
import time

def _c5_score(h):
    """Return C5 for a step function given by array h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _grad_for_max_shift(h):
    """
    Compute gradient of the overlap term that attains the current maximum.
    Returns indices where increasing (negative gradient) or decreasing
    (positive gradient) h would reduce that maximum.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)            # shift causing the max
    if s >= 0:
        # forward[i] = h[i+s] if i+s in range else 0
        forward = np.empty_like(h)
        forward[: n - s] = h[s:]
        forward[n - s :] = 0.0
        # backward[i] = h[i-s] if i-s in range else 0
        backward = np.empty_like(h)
        backward[s:] = h[: n - s]
        backward[: s] = 0.0
    else:
        sh = -s
        forward = np.empty_like(h)
        forward[sh:] = h[: n - sh]
        forward[: sh] = 0.0
        backward = np.empty_like(h)
        backward[: n - sh] = h[sh:]
        backward[n - sh :] = 0.0
    grad = 1.0 - (forward + backward)          # derivative of the max term
    # indices where we would like to decrease/increase h
    inc = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]   # increase h
    dec = np.where((grad > 1e-12) & (h > 1e-12))[0]         # decrease h
    return inc, dec

def _hill_climb(h0, max_iter, rng, budget_s, start_time):
    """Simple directed hill‑climbing with sum‑preserving swaps."""
    n = h0.shape[0]
    h = h0.copy()
    cur_score = _c5_score(h)
    best_h = h.copy()
    best_score = cur_score

    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break

        # try a gradient‑guided move
        inc_idx, dec_idx = _grad_for_max_shift(h)
        if inc_idx.size and dec_idx.size:
            i = rng.choice(dec_idx)   # will be decreased
            j = rng.choice(inc_idx)   # will be increased
        else:
            # fallback: random admissible pair
            i = rng.randint(n)
            if h[i] <= 0.0:
                continue
            j = rng.randint(n)
            if h[j] >= 1.0 or i == j:
                continue

        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max

        # apply move
        h[i] -= delta
        h[j] += delta

        new_score = _c5_score(h)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

    return best_h, best_score

        best_h = init_h
        best_score = _c5_score(best_h)
        best_n = best_h.shape[0]

    return best_h, float(best_score), int(best_n)
