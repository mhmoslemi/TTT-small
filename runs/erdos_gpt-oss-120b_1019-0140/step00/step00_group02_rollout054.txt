def repair_h(h, target_sum):
    # clamp
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    diff = target_sum - current_sum
    # if diff > 0, add to entries <1
    # if diff < 0, subtract from entries >0
    # distribution: proportional to distance to bound
    # For simplicity, use small loops adding small epsilon to random indices until diff satisfied.
    # But diff magnitude is small, we can adjust via scaling.
    if abs(diff) < 1e-9:
        return h
    if diff > 0:
        # need to add mass
        # find indices where h < 1
        idx = np.where(h < 1)[0]
        if len(idx) == 0:
            return h  # cannot add, but rare
        # distribute diff uniformly across idx
        add = diff / len(idx)
        h[idx] += add
        # clip again
        h = np.clip(h, 0.0, 1.0)
    else:
        # need to subtract
        idx = np.where(h > 0)[0]
        if len(idx) == 0:
            return h
        sub = (-diff) / len(idx)
        h[idx] -= sub
        h = np.clip(h, 0.0, 1.0)
    # final scaling to remove residual error if any
    # iterate a couple times
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-8:
        # final linear scaling, but ensure not violating bounds
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0.0, 1.0)
        # recursion: if still off, just accept small error; but we want exact sum in integer condition? The constraints: sum(h)*dx should be exactly 1.

    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def neighbor_swap(h, max_step):
    """Return a new feasible array by moving mass between two indices."""
    n = h.shape[0]
    # try a few random pairs until a non‑zero move is possible
    for _ in range(5):
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        # allowed delta range (i loses, j gains)
        delta_max = min(hi, 1.0 - hj)          # positive delta
        delta_min = -min(1.0 - hi, hj)        # negative delta
        lower = max(delta_min, -max_step)
        upper = min(delta_max, max_step)
        if lower < upper:
            delta = random.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
    # fallback: return unchanged copy
    return h.copy()

def cosine_start(n):
    """Construct the cosine step function (1+cos(pi x/2))/2."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 0.5 * (1.0 + np.cos(np.pi * x / 2.0))
    target = n / 2.0
    return repair_h(h, target)

def get_initial_h(n):
    """Obtain an initial feasible vector, preferring the provided parent."""
    target = n / 2.0
    hv = globals().get('initial_h_values')
    if hv is not None:
        hv_arr = np.array(hv, dtype=float)
        if hv_arr.shape[0] != n:
            # interpolate to required size
            old_x = np.linspace(0.0, 2.0, hv_arr.shape[0], endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            hv_arr = np.interp(new_x, old_x, hv_arr)
        hv_arr = repair_h(hv_arr, target)
        return hv_arr
    # no parent – use the analytic cosine start
    return cosine_start(n)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # initial resolution
    n = 45
    target_sum = n / 2.0

    # two possible starts: parent (if any) and analytic cosine
    h_parent = None
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        h_parent = repair_h(h_parent, target_sum)
    h_cos = cosine_start(n)

    # pick the better of the two as the initial current solution
    if h_parent is not None:
        score_parent = compute_c5(h_parent)
        score_cos = compute_c5(h_cos)
        if score_cos < score_parent:
            h_current = h_cos.copy()
            best_h = h_cos.copy()
            best_score = score_cos
        else:
            h_current = h_parent.copy()
            best_h = h_parent.copy()
            best_score = score_parent
    else:
        h_current = h_cos.copy()
        best_h = h_cos.copy()
        best_score = compute_c5(h_cos)

    # Simulated annealing parameters
    t0 = 0.02
    T = t0
    cooling = 0.9995
    max_step_initial = 0.15
    no_improve = 0
    # allocate ~90% of the budget to the main SA loop
    sa_time_limit = budget_s * 0.90

    while time.time() - start_time < sa_time_limit:
        # adapt step size with temperature
        step = max(1e-4, max_step_initial * (T / t0))
        h_new = neighbor_swap(h_current, step)
        score_new = compute_c5(h_new)

        # SA acceptance
        if score_new < compute_c5(h_current) or random.random() < math.exp(-(score_new - compute_c5(h_current)) / (T + 1e-12)):
            h_current = h_new
        # record best
        if score_new < best_score:
            best_h = h_new.copy()
            best_score = score_new
            no_improve = 0
        else:
            no_improve += 1

        # occasional large random jump to escape stagnation
        if random.random() < 0.002:
            h_current = neighbor_swap(best_h, max_step=0.5)

        # cooling schedule
        T *= cooling

        # restart if stuck for a while
        if no_improve > 5000:
            h_current = neighbor_swap(best_h, max_step=0.3)
            no_improve = 0

    # Local greedy refinement (use remaining time)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        # try many small perturbations
        refinements = int(min(50000, remaining * 20000))  # heuristic
        for _ in range(refinements):
            h_new = neighbor_swap(best_h, max_step=0.02)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h = h_new
                best_score = sc

    # Final feasibility clean‑up
    best_h = repair_h(best_h, target_sum)
    final_score = compute_c5(best_h)

    return best_h.tolist(), final_score, n
