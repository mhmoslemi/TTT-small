def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n_points = len(init)
        # make binary start
        target = n_points // 2
        idx_sorted = np.argsort(-init)  # descending values
        h_start = np.zeros(n_points, dtype=float)
        h_start[idx_sorted[:target]] = 1.0
    else:
        n_points = 200
        target = n_points // 2
        # random binary with exact target ones
        h_start = np.zeros(n_points, dtype=float)
        ones_indices = rng.choice(n_points, size=target, replace=False)
        h_start[ones_indices] = 1.0

    dx = 2.0 / n_points

    def compute_c5_vec(vec):
        # correlation full
        # use np.correlate for simplicity
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return float(np.max(corr) * dx)

    # initialize current and best
    h = h_start.copy()
    current_score = compute_c5_vec(h)
    best_h = h.copy()
    best_score = current_score

    # Simulated annealing
    T_initial = 0.5
    T_final = 1e-6

    # approximate max iteration count based on time, we will loop while time left
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.95:
            break
        # cooling schedule
        frac = elapsed / (budget_s * 0.95)
        T = T_initial * (1 - frac) + T_final * frac  # linear

        # propose swap
        i = rng.integers(0, n_points)
        j = rng.integers(0, n_points - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]

        new_score = compute_c5_vec(h)
        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # avoid overflow
            prob = math.exp(-delta / (T + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    # return best solution found
    # ensure integer sum? best_h is binary.

    # Convert to list (or keep as numpy)
    h_values = best_h.tolist()
    c5_bound = best_score
    return (h_values, c5_bound, n_points)

import numpy as np, time, math

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    import time as _time # to avoid conflict of time variable imported earlier? Not needed.
    rng = np.random.default_rng(seed)
    start = _time.time()
    # Determine n_points
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        n_points = max(200, n0 * 2)
        if n_points % 2:
            n_points += 1
        # Interpolate initial to new grid
        old_grid = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_grid = np.linspace(0.0, 2.0, n_points, endpoint=False)
        interp = np.interp(new_grid, old_grid, initial_h_values)
        target = n_points // 2
        idx_desc = np.argsort(-interp)
        h_start = np.zeros(n_points, dtype=float)
        h_start[idx_desc[:target]] = 1.0
    else:
        n_points = 200
        target = n_points // 2
        h_start = np.zeros(n_points, dtype=float)
        ones_idx = rng.choice(n_points, size=target, replace=False)
        h_start[ones_idx] = 1.0

    dx = 2.0 / n_points

    h = h_start.copy()
    current_score = compute_c5(h, dx)
    best_h = h.copy()
    best_score = current_score

    T_initial = 0.5
    T_min = 1e-6

    # Annealing loop
    while True:
        now = _time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.95:
            break

import numpy as np, time, math

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def hill_climb(h, dx, rng, start_time, time_limit):
    n = len(h)
    score = compute_c5(h, dx)
    no_improve = 0
    # maximum consecutive fails before stopping
    max_no_improve = 2000
    while True:
        if time.time() - start_time >= time_limit:
            break
        if no_improve >= max_no_improve:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # try swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h, dx)
        if new_score < score:
            score = new_score
            no_improve = 0
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
    return h, score

import numpy as np
import time
import math

def compute_c5(h, dx):
    """Return max overlap integral for step function defined by h."""
    # Full linear cross‑correlation of h and (1‑h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def hill_climb(h, dx, rng, start_time, time_limit):
    """
    Greedy local search on binary vector h.
    Swaps two entries of opposite value and keeps the change if the
    overlap bound improves. Stops after a time limit or after a number of
    consecutive unsuccessful attempts.
    """
    n = len(h)
    score = compute_c5(h, dx)
    no_improve = 0
    max_no_improve = 2000

    while True:
        if time.time() - start_time >= time_limit:
            break
        if no_improve >= max_no_improve:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue

        # try swapping a 1 and a 0
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h, dx)
        if new_score < score:
            score = new_score
            no_improve = 0
        else:
            # revert the swap
            h[i], h[j] = h[j], h[i]
            no_improve += 1

    return h, score


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 that minimizes
    max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------
    # Choose discretisation size, preferring an even number.

        n_points = max(200, n0 * 2)
        if n_points % 2:
            n_points += 1
        # Interpolate the provided values to the new grid and take the
        # positions with highest values as the initial binary placement.
        old_grid = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_grid = np.linspace(0.0, 2.0, n_points, endpoint=False)
        interp = np.interp(new_grid, old_grid, initial_h_values)
        target = n_points // 2
        idx_desc = np.argsort(-interp)                # indices of large values
        h_start = np.zeros(n_points, dtype=float)
        h_start[idx_desc[:target]] = 1.0
    else:
        n_points = 200
        target = n_points // 2
        h_start = np.zeros(n_points, dtype=float)
        ones_idx = rng.choice(n_points, size=target, replace=False)
        h_start[ones_idx] = 1.0

    dx = 2.0 / n_points
