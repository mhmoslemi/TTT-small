    dx = 2.0 / n

    best_h = h.copy()
    best_val = cur_val
    start_time = time.time()
    # loops
    iter_count = 0
    no_improve = 0
    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        # schedule step and temp
        step = 0.2 * max(0.01, 1 - elapsed / time_budget)  # gradually shrink
        temp = 0.01 * max(0.01, 1 - elapsed / time_budget)
        # choose move type
        if rng.random() < 0.8:
            # targeted move
            # compute correlation and lag of max
            conv = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(conv)
            lag = max_idx - (n - 1)
            # compute overlapping indices
            i_start = max(0, -lag)
            i_end = min(n - 1, n - 1 - lag)
            if i_end >= i_start:

                # contributions
                contrib = h[i_range] * (1 - h[i_range + lag])
                i_max_local = i_range[np.argmax(contrib)]
                j = i_max_local + lag

                    candidate = h.copy()
                    candidate[i_max_local] -= eps
                    candidate[j] += eps
                else:
                    # fallback to random pair
                    candidate = None
            else:
                candidate = None
            if candidate is None:

                i, j = rng.choice(n, 2, replace=False)

                if rng.random() < 0.5:

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n is assumed even)."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def optimize_h(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):

    Local optimisation starting from `start_h` for at most `time_budget` seconds.

    """

        temp = 0.01 * max(0.01, 1.0 - elapsed / time_budget)
