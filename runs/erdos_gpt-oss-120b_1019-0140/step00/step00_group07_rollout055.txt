def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    # ---------- helpers ----------
    def compute_c5(vec, dx):
        # overlap integral max_k ∫ h(x)(1‑h(x+k))dx
        return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * dx

    # ---------- initialization ----------
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        init = np.clip(init, 0.0, 1.0)
        n = init.size
        target = n / 2.0
        cur_sum = init.sum()
        # project onto the hyperplane sum = target while staying in [0,1]
        diff = target - cur_sum
        while abs(diff) > 1e-12:
            idx = rng.integers(0, n)
            if diff > 0:
                add = min(1.0 - init[idx], diff)
                init[idx] += add
                diff -= add
            else:
                sub = min(init[idx], -diff)
                init[idx] -= sub
                diff += sub
        h = init.copy()
    else:
        n = 80
        target = n / 2.0
        h = np.zeros(n, dtype=np.float64)
        h[:int(target)] = 1.0
        rng.shuffle(h)

    dx = 2.0 / n
    # ---------- simulated annealing ----------
    current_h = h.copy()
    current_c5 = compute_c5(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5

    start = time.time()
    total = max(0.0, budget_s - 0.5)   # small reserve for final steps
    T0, T_end = 5e-2, 1e-6            # temperature schedule

    while True:
        elapsed = time.time() - start
        if elapsed >= total:
            break
        frac = elapsed / total
        T = T0 * (1 - frac) + T_end * frac

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # try to move mass from j to i (increase i, decrease j)
        max_delta = min(1.0 - current_h[i], current_h[j])
        if max_delta <= 0.0:
            # opposite direction
            max_delta = min(1.0 - current_h[j], current_h[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i          # swap roles

        delta = max_delta * rng.random()
        # apply perturbation
        current_h[i] += delta
        current_h[j] -= delta

        new_c5 = compute_c5(current_h, dx)
        improve = new_c5 < current_c5
        if improve or rng.random() < math.exp(-(new_c5 - current_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # revert
            current_h[i] -= delta
            current_h[j] += delta

    # ---------- optional local refinement ----------
    try:
        from scipy.optimize import minimize

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0})
        bounds = [(0.0, 1.0) for _ in range(n)]

        res = minimize(
            lambda x: compute_c5(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[cons],
            options={'maxiter': 250, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            refined = np.clip(res.x, 0.0, 1.0)
            # re‑enforce exact sum (tiny adjustment on first element)
            diff = n / 2.0 - refined.sum()
            if abs(diff) > 1e-12:
                refined[0] = min(max(refined[0] + diff, 0.0), 1.0)
            refined_c5 = compute_c5(refined, dx)
            if refined_c5 < best_c5:
                best_h = refined
                best_c5 = refined_c5
    except Exception:
        # scipy may be missing; ignore refinement
        pass

    best_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), float(best_c5), int(n))
