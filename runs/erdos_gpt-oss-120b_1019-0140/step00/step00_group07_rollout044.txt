def get_initial_h(n):
    try:
        init = initial_h_values
        if isinstance(init, np.ndarray) and init.ndim == 1 and init.size > 0:
            # upsample to n if possible by integer repeat
            repeat = n // init.size
            rem = n % init.size
            if repeat > 0:
                h = np.repeat(init, repeat)
                if rem > 0:
                    h = np.concatenate([h, init[:rem]])
                # adjust sum to n/2 by flipping if needed
                diff = int(np.round(h.sum() - n/2))
                if diff != 0:
                    # adjust diff by flipping some bits
                    if diff > 0:
                        # too many ones, turn some ones to zero
                        ones_idx = np.where(h == 1)[0]
                        np.random.shuffle(ones_idx)
                        h[ones_idx[:diff]] = 0
                    else:
                        zeros_idx = np.where(h == 0)[0]
                        np.random.shuffle(zeros_idx)
                        h[zeros_idx[: -diff]] = 1
                return h.astype(int)
    except NameError:
        pass
    # fallback
    h = np.zeros(n, dtype=int)
    ones = n // 2
    h[:ones] = 1
    np.random.shuffle(h)
    return h

def simulated_annealing(h, time_budget):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    start = time.time()
    T0 = 0.01  # starting temperature
    # we may use decreasing temperature schedule
    # We'll loop until time exhausted
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature decay linearly
        T = T0 * (1 - elapsed / time_budget)
        # pick random indices to swap
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta <= 0 or (T > 0 and np.random.rand() < math.exp(-delta / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # optionally random restart after some stagnation
        # we can periodically check and restart
        # but not needed maybe
    return best_h, best_score

import numpy as np
import time
import random
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a discrete step function h."""
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / h.size
    return float(np.max(corr) * dx)

def _initial_h(n: int) -> np.ndarray:
    """Create a binary starting vector of length n with exactly n/2 ones.
    If a global `initial_h_values` exists, upsample it; otherwise use a random
    balanced vector."""
    # try to use the provided construction
    try:
        init = initial_h_values  # type: ignore
        if isinstance(init, np.ndarray) and init.ndim == 1 and init.size > 0:
            # up‑sample by integer repetition
            repeat = n // init.size
            remainder = n % init.size
            if repeat > 0:
                h = np.repeat(init, repeat)
                if remainder:
                    h = np.concatenate([h, init[:remainder]])
            else:
                # n smaller than the provided size – just truncate
                h = init[:n].copy()
            # ensure binary values and correct sum
            h = np.where(h > 0.5, 1, 0).astype(int)
            diff = int(h.sum() - n // 2)
            if diff != 0:
                if diff > 0:
                    # too many ones – turn some into zeros
                    ones = np.where(h == 1)[0]
                    np.random.shuffle(ones)
                    h[ones[:diff]] = 0
                else:
                    # too few ones – turn some zeros into ones
                    zeros = np.where(h == 0)[0]
                    np.random.shuffle(zeros)
                    h[zeros[: -diff]] = 1
            return h
    except NameError:
        pass

    # fallback: a random balanced binary vector
    h = np.zeros(n, dtype=int)
    h[: n // 2] = 1
    np.random.shuffle(h)
    return h

def _simulated_annealing(start_h: np.ndarray, time_budget: float) -> tuple:
    """Improve a binary sequence by swapping bits under a time limit."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    stagnation = 0
    # basic temperature schedule: linear decay from T0 to 0
    T0 = 0.015

    # pre‑allocate index arrays to avoid repeated allocation in the hot loop
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # linear temperature
        T = T0 * (1.0 - elapsed / time_budget)

        # choose a 1‑index and a 0‑index to swap
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = _compute_c5(cur_h)
        delta = new_score - cur_score

        if delta <= 0.0 or (T > 0.0 and random.random() < math.exp(-delta / T)):
            # accept move
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # reject – revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            stagnation += 1

        # random restart if stuck too long
        if stagnation >= 5000:
            cur_h = np.concatenate([
                np.ones(n // 2, dtype=int),
                np.zeros(n // 2, dtype=int)
            ])
            np.random.shuffle(cur_h)
            cur_score = _compute_c5(cur_h)
            stagnation = 0

    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of sample points.

    The function respects the time budget, using a simple simulated‑annealing
    search over binary vectors of length n=160 (twice the default construction).
    """
    np.random.seed(seed)
    random.seed(seed)

    # Choose number of samples – 160 keeps an exact half‑ones upsample from 80.
    n_points = 160

    # Initialise h (binary) – prefer an up‑sample of any supplied construction.
    h0 = _initial_h(n_points)

    # Allocate most of the budget to optimisation, leaving a tiny margin.
    remaining = max(0.0, budget_s - 0.2)

    best_h, best_score = _simulated_annealing(h0, remaining)

    # Return values as float array, the bound, and n_points.
    return best_h.astype(float), float(best_score), n_points
