import time
import numpy as np
from scipy.optimize import minimize


def compute_c5(h):
    """Return C5 = max convolution (h,1-h) * dx."""
    n = h.size
    dx = 2.0 / n
    # full linear correlation
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def project_box_simplex(v, target):
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane
    sum(x) = target (0 <= target <= n). Uses a simple binary search on the
    Lagrange multiplier λ for the projection x_i = clamp(v_i - λ, 0, 1).
    """
    n = v.shape[0]
    lo, hi = -1e6, 1e6
    for _ in range(80):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def objective(h):
    """Objective for SLSQP: max correlation times dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def interpolate_to(src, n_target):
    """Linearly interpolate src (numpy array) to length n_target and project."""
    n_src = src.size
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    interp = np.interp(x_tgt, x_src, src)
    return project_box_simplex(interp, n_target / 2.0)


def local_search(h, start_time, budget):
    """
    Simple hill‑climbing: move a small amount of mass from a point with h>0
    to a point with h<1 while improving C5. Stops when allocated time expires.
    """
    n = h.size
    best = h.copy()
    best_val = compute_c5(best)
    max_step = 0.07  # maximal transferred mass in one move

    while time.time() - start_time < budget:
        inc_candidates = np.where(best < 1.0 - 1e-12)[0]
        dec_candidates = np.where(best > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break

        i_inc = np.random.choice(inc_candidates)
        i_dec = np.random.choice(dec_candidates)

        max_transfer = min(1.0 - best[i_inc], best[i_dec], max_step)
        if max_transfer <= 0:
            continue

        delta = np.random.uniform(0.0, max_transfer)
        new_h = best.copy()
        new_h[i_inc] += delta
        new_h[i_dec] -= delta

        val = compute_c5(new_h)
        if val < best_val - 1e-12:
            best = new_h
            best_val = val
    return best, best_val
