def projected_gradient_descent(v, max_iter, time_budget):
    start = time.time()
    n = len(v)
    best_v = v.copy()
    best_obj = compute_objective(v)
    # Steps list
    step_sizes = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    v_cur = v.copy()
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
        grad = compute_gradient(v_cur)
        improved = False
        for step in step_sizes:
            v_new = project_to_constraints(v_cur - step * grad)
            obj_new = compute_objective(v_new)
            if obj_new < best_obj - 1e-12:
                v_cur = v_new
                best_v = v_new
                best_obj = obj_new
                improved = True
                break
        if not improved:
            # random perturbation fallback
            # small random noise
            noise = np.random.randn(n) * 0.005
            v_new = project_to_constraints(v_cur - 0.1 * grad + noise)
            obj_new = compute_objective(v_new)
            if obj_new < best_obj:
                v_cur = v_new
                best_v = v_new
                best_obj = obj_new
    return best_v, best_obj

    if abs(diff) < 1e-12:
        return v
    for idx in range(n):
        if 0.0 <= v[idx] + diff <= 1.0:
            v[idx] += diff
            return v
    # fallback scaling
    v = v * (target / v.sum())
    return v

def project_to_constraints(v):
    # Clip to [0,1] and adjust sum to target using water-filling
    n = v.shape[0]
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        adjustment = diff / cnt
        v[mask] += adjustment
        v = np.clip(v, 0.0, 1.0)
    v = enforce_sum(v)
    return v

def compute_objective(v):
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return dx * np.max(corr)

def compute_gradient(v):
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    idx = int(np.argmax(corr))
    s = idx - (n - 1)  # shift offset
    grad = np.zeros_like(v)
    if s >= 0:
        # indices where j <= n-1-s
        grad[:n - s] += dx * (1.0 - v[s:])
        # indices where j >= s
        grad[s:] -= dx * v[:n - s]
    else:
        t = -s
        grad[t:] += dx * (1.0 - v[:n - t])
        grad[:n - t] -= dx * v[t:]
    return grad

def projected_gradient_descent(v, max_iter, time_budget):
    start = time.time()
    best_v = v.copy()
    best_obj = compute_objective(v)
    step_sizes = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    cur_v = v.copy()
    for _ in range(max_iter):
        if time.time() - start > time_budget:
            break
        grad = compute_gradient(cur_v)
        improved = False
        for step in step_sizes:
            cand = project_to_constraints(cur_v - step * grad)
            obj = compute_objective(cand)
            if obj < best_obj - 1e-12:
                cur_v = cand
                best_v = cand
                best_obj = obj
                improved = True
                break
        if not improved:
            # small random perturbation
            noise = np.random.randn(v.shape[0]) * 0.005
            cand = project_to_constraints(cur_v - 0.1 * grad + noise)
            obj = compute_objective(cand)
            if obj < best_obj:
                cur_v = cand
                best_v = cand
                best_obj = obj
    return best_v, best_obj

def upscale(v, new_n):
    n_old = v.shape[0]
    dx_old = 2.0 / n_old
    x_old = np.arange(n_old) * dx_old + dx_old / 2.0
    dx_new = 2.0 / new_n
    x_new = np.arange(new_n) * dx_new + dx_new / 2.0
    v_new = np.interp(x_new, x_old, v)
    return v_new

def optimize(v, start_time, budget_s):
    n = v.shape[0]
    if n <= 100:
        max_iter = 3000
    elif n <= 200:
        max_iter = 5000
    else:
        max_iter = 8000
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining <= 0:
        return v, compute_objective(v)
    max_iter = min(max_iter, int(remaining * 2000))
    if max_iter < 1:
        max_iter = 1
    v_opt, obj_opt = projected_gradient_descent(v, max_iter, remaining)
    return v_opt, obj_opt

def hill_climb(v, start_time, budget_s):
    n = v.shape[0]
    best_v = v.copy()
    best_obj = compute_objective(best_v)
    while time.time() - start_time < budget_s * 0.99:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        max_removable = best_v[i]
        max_addable = 1.0 - best_v[j]
        max_delta = min(max_removable, max_addable, 0.05)
        if max_delta <= 1e-12:
            continue
        delta = np.random.rand() * max_delta
        v_new = best_v.copy()
        v_new[i] -= delta
        v_new[j] += delta
        obj_new = compute_objective(v_new)
        if obj_new < best_obj:
            best_v = v_new
            best_obj = obj_new
    return best_v, best_obj

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # obtain initial vector
    try:
        init_vals = np.array(initial_h_values, dtype=np.float64)
        n0 = init_vals.shape[0]
        v = project_to_constraints(init_vals)
    except NameError:
        n0 = 95
        v = np.full(n0, 0.5, dtype=np.float64)
        v = project_to_constraints(v)
    best_v = v.copy()
    best_obj = compute_objective(best_v)
    best_n = n0
    # random restarts at base size (limited time)
    for _ in range(3):
        if time.time() - start_time > budget_s * 0.3:
            break
        rand_v = np.random.rand(n0)
        rand_v = project_to_constraints(rand_v)
        v_opt, obj_opt = projected_gradient_descent(rand_v, max_iter=2000,
                                                    time_budget=budget_s - (time.time() - start_time))
        if obj_opt < best_obj:
            best_v, best_obj, best_n = v_opt, obj_opt, n0
    # hierarchical refinements
    for target_n in (190, 380):
        if time.time() - start_time > budget_s * 0.9:
            break
        v_up = upscale(best_v, target_n)
        v_up = project_to_constraints(v_up)
        v_opt, obj_opt = optimize(v_up, start_time, budget_s)
        if obj_opt < best_obj:
            best_v, best_obj, best_n = v_opt, obj_opt, target_n
    # final hill climbing if time permits
    if time.time() - start_time < budget_s * 0.95:
        v_opt, obj_opt = hill_climb(best_v, start_time, budget_s)
        if obj_opt < best_obj:
            best_v, best_obj = v_opt, obj_opt
            best_n = len(best_v)
    best_v = enforce_sum(project_to_constraints(best_v))
    final_obj = compute_objective(best_v)
    return best_v.astype(np.float64), float(final_obj), int(len(best_v))
