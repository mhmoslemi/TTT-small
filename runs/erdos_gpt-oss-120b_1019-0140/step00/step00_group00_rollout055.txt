def project_onto_box_and_sum(v, lower, upper, s, max_iter=100, eps=1e-9):
    # Solve sum clip(v - lam, lower, upper) = s for lam
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        sum_x = x.sum()
        if sum_x > s:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    return np.clip(v - lam, lower, upper)

import time, math
import numpy as np

def project_onto_box_and_sum(v, lower, upper, target, max_iter=100, eps=1e-9):
    """Euclidean projection of v onto {x: lower<=x<=upper, sum(x)=target}."""
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    return np.clip(v - lam, lower, upper)

def compute_c5(h):
    """Return max overlap integral for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search(start_h, time_budget, rng):
    """Simple simulated-annealing / gradient descent on a fixed-size vector."""
    n = len(start_h)
    h = project_onto_box_and_sum(start_h.astype(np.float64), 0.0, 1.0, n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    t0 = time.time()
    it = 0
    while time.time() - t0 < time_budget:
        elapsed = time.time() - t0
        prog = elapsed / time_budget
        temp = max(0.001, 1.0 - prog)          # linear cooling
        step_std = 0.5 * temp                  # magnitude of perturbation

        if rng.random() < 0.2:
            # random Gaussian perturbation
            cand = cur_h + rng.normal(scale=step_std, size=n)
        else:
            # gradient step aimed at reducing the worst lag
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            lag = int(np.argmax(corr) - (n - 1))
            g = np.ones_like(cur_h)
            if lag > 0:
                g[:n - lag] -= cur_h[lag:]
                g[lag:] -= cur_h[:n - lag]
            elif lag < 0:
                L = -lag
                g[:n - L] -= cur_h[L:]
                g[L:] -= cur_h[:n - L]
            else:
                g = 1.0 - 2.0 * cur_h
            cand = cur_h - step_std * g

        cand = project_onto_box_and_sum(cand, 0.0, 1.0, n / 2.0)
        cand_c5 = compute_c5(cand)

        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cand_c5 - cur_c5) / (temp + 1e-12)):
                cur_h, cur_c5 = cand, cand_c5

        it += 1
        # occasional restart to avoid stagnation
        if it % 20000 == 0:
            if rng.random() < 0.5:
                cur_h, cur_c5 = best_h.copy(), best_c5
            else:
                cur_h = project_onto_box_and_sum(rng.random(n), 0.0, 1.0, n / 2.0)
                cur_c5 = compute_c5(cur_h)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- initialise h -------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        n0 = init.shape[0]
        h0 = init.copy()
    except NameError:
        n0 = 45

        half = n0 // 2
        h0[:half] = 1.0
        if n0 % 2:
            h0[half] = 0.5

    # allocate time: 60% for the base resolution, remaining for refined one
    alloc1 = budget_s * 0.6
    alloc2 = budget_s - alloc1

    # --- first stage --------------------------------------------------
    best_h, best_c5 = local_search(h0, alloc1, rng)

    n1 = best_h.shape[0]

        h_up = np.repeat(best_h, 2)
        best_h, best_c5 = local_search(h_up, alloc2, rng)

    # final return
    return best_h, best_c5, best_h.shape[0]
