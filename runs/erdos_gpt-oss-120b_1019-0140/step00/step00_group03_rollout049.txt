def project_onto_box_and_sum(v, target_sum, lower=0., upper=1.):
    # project v onto [lower, upper] and ensure sum = target_sum
    # Use algorithm similar to water-filling
    n = len(v)
    # Start with v clamped to bounds
    v = np.clip(v, lower, upper)
    # Compute sum
    s = v.sum()
    # If sum is target, done
    # Else we need to adjust: if sum > target, we need to reduce some values (those not at lower bound)
    # if sum < target, we need to increase some values (those not at upper bound)
    # Use iterative water-filling: compute amount per free variable
    # The set of free indices: those not at bound in direction of adjustment
    free = np.arange(n)
    while True:
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0: # need to increase sum, raise values under upper
            mask = v < upper - 1e-12
        else: # need to decrease sum, lower values above lower
            mask = v > lower + 1e-12
        free = np.where(mask)[0]
        if free.size == 0:
            # cannot adjust further, break
            break
        # Distribute diff equally among free
        delta = diff / free.size
        # update
        v[free] += delta
        # clip again
        v = np.clip(v, lower, upper)
        # repeat because clipping may saturate some indices
    return v

def gradient_of_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute grad vector
    grad = np.empty_like(h)
    if shift >= 0:
        # forward shift: h[i+shift] inside
        # compute (1 - h[i+shift]) part for i in [0, n-shift)
        grad[:n-shift] = (1 - h[shift:])  # (1 - h[i+shift])
        grad[n-shift:] = 0.0  # no forward part for last shift entries
        # subtract h[i-shift] part for i in [shift, n)
        grad[shift:] -= h[:n-shift]
        # i < shift: subtraction term not present (i-shift negative)
    else:
        s = -shift
        # shift negative: i+shift means i-s; forward part for i in [s, n)
        grad[s:] = (1 - h[:n-s])
        # last s entries have no forward part
        grad[:n-s] -= h[s:]  # subtract h[i-shift], i-shift = i + s for i < n-s
        # others zero
    return grad, shift, idx, corr[idx] * (2.0 / n)  # optional return C5
```

import numpy as np, time, random, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_box_and_sum(v, target, lower=0.0, upper=1.0):
    # Clip
    v = np.clip(v, lower, upper)
    # iterative water-filling
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < upper - 1e-12
        else:
            mask = v > lower + 1e-12
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, lower, upper)
    # final tiny correction
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        # add diff to the element with most slack
        if diff > 0:
            slack = upper - v
            idx = np.argmax(slack)
            v[idx] = min(upper, v[idx] + diff)
        else:
            slack = v - lower
            idx = np.argmax(slack)
            v[idx] = max(lower, v[idx] + diff)
    return v

def grad_max_shift(h):
    n = h.shape[0]
    # compute correlation to find max shift
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute gradient
    grad = np.zeros_like(h)
    if shift >= 0:
        # forward term
        grad[:n - shift] = 1 - h[shift:]
        # backward term
        grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        grad[s:] = 1 - h[:n - s]
        grad[:n - s] -= h[s:]
    return grad, shift, idx, corr[idx]

    random.seed(seed)

    best_c5 = compute_c5(h)

    min_temp = 1e-6

    iteration = 0
    c5_current = best_c5
    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.2:
            break
        # temperature schedule
        temp = max(min_temp, max_temp * (1.0 - elapsed / budget_s))
        # random move or gradient step
        if np.random.rand() < 0.3:
            # random epsilon transfer between two indices
            i, j = np.random.choice(n_points, 2, replace=False)
            # decide direction: transfer from i to j
            max_pos = min(h[i], 1.0 - h[j])
            max_neg = -min(h[j], 1.0 - h[i])  # negative eps means transfer j->i

            h_candidate = h.copy()
            h_candidate[i] -= eps
            h_candidate[j] += eps

        diff = target - s
        if abs(diff) < 1e-12:
