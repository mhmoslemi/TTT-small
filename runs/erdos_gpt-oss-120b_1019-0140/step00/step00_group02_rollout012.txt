def propose_move(h, step):
    n = len(h)
    # choose i with h[i] < 1, j with h[j] > 0
    # Use numpy random choice with condition.
    # get indices arrays.
    # If none found (should not happen unless all 0/1 and extreme), return copy.
    idx_up = np.where(h < 1 - 1e-12)[0]
    idx_down = np.where(h > 1e-12)[0]
    if len(idx_up) == 0 or len(idx_down) == 0:
        return h.copy()
    i = np.random.choice(idx_up)
    j = np.random.choice(idx_down)
    # max possible delta
    max_inc = 1 - h[i]
    max_dec = h[j]
    max_delta = min(max_inc, max_dec)
    # random delta bounded by step and max_delta
    delta = (np.random.rand() - 0.5) * 2 * step  # range [-step, step]
    delta = np.clip(delta, -max_delta, max_delta)
    # ensure not zero
    if abs(delta) < 1e-12:
        # try small delta
        delta = np.sign(delta) * max_delta if max_delta > 1e-12 else 0
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # enforce bounds just in case
    h_new[i] = min(max(h_new[i], 0.0), 1.0)
    h_new[j] = min(max(h_new[j], 0.0), 1.0)
    return h_new

        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def propose_move(h, step):

    idx_inc = np.where(h < 1 - 1e-12)[0]
    idx_dec = np.where(h > 1e-12)[0]

    j = np.random.choice(idx_dec)

    max_delta = min(max_inc, max_dec)
    delta = (np.random.rand() - 0.5) * 2 * step

    if abs(delta) < 1e-14:
        return h.copy()
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # clamp just in case
    h_new[i] = min(max(h_new[i], 0.0), 1.0)
    h_new[j] = min(max(h_new[j], 0.0), 1.0)
    return h_new

import numpy as np, time, math

def compute_max_overlap(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    N = 1 << ((2 * n - 1).bit_length())   # next power of two
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)[: 2 * n - 1]
    return float(np.max(corr) * dx)


def repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet exact sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    diff = target_sum - cur
    # distribute diff among interior points if possible
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        add = diff / free.sum()
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
    # fallback scaling
    if abs(diff) > 1e-12:
        scale = target_sum / (cur + 1e-12)
        h = np.clip(h * scale, 0.0, 1.0)
    # final tiny correction on a non‑boundary element
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size == 0:
            idx = [np.argmin(np.abs(h - 0.5))]
        i = idx[0]
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def propose_move(h: np.ndarray, step: float) -> np.ndarray:
    """Generate a neighbour by moving mass between two coordinates."""
    n = h.shape[0]
    inc_idx = np.where(h < 1.0 - 1e-12)[0]
    dec_idx = np.where(h > 1e-12)[0]
    if inc_idx.size == 0 or dec_idx.size == 0:
        return h.copy()
    i = np.random.choice(inc_idx)
    j = np.random.choice(dec_idx)
    max_inc = 1.0 - h[i]
    max_dec = h[j]
    max_delta = min(max_inc, max_dec)
    delta = (np.random.rand() - 0.5) * 2.0 * step
    delta = np.clip(delta, -max_delta, max_delta)
    if abs(delta) < 1e-15:
        return h.copy()
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # safety clip (should already be within bounds)
    h_new[i] = min(max(h_new[i], 0.0), 1.0)
    h_new[j] = min(max(h_new[j], 0.0), 1.0)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function minimizing the Erdős C5 overlap constant.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – numpy array of the step heights,
        c5_bound – the attained maximal overlap,
        n_points – number of discretization points.
    """
    start = time.time()
    np.random.seed(seed)

    # ----- initialise -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: 64 points, random feasible vector
        n0 = 64
        h0 = np.random.rand(n0)
        h0 = repair(h0, n0 / 2.0)

    n = h0.shape[0]
    target_sum = n / 2.0
    h = repair(h0, target_sum)

    cur_score = compute_max_overlap(h)
    best_h = h.copy()
    best_score = cur_score

    # ----- SA parameters ---------------------------------------------
    max_iters = 250000
    T0 = best_score * 0.1 + 1e-3
    Tf = best_score * 1e-5 + 1e-6
    step0 = 0.2
    step_min = 1e-4
    no_improve = 0
    improve_patience = 20000

    # ----- main loop -------------------------------------------------
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.97:
            break
        # schedules
        t_frac = it / max_iters
        T = T0 * (Tf / T0) ** t_frac
        step = step0 * (step_min / step0) ** t_frac

        # propose neighbour
        cand = propose_move(h, step)
        cand_score = compute_max_overlap(cand)

        # acceptance test (Simulated Annealing)
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - cand_score) / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            h = cand
            cur_score = cand_score

        # update best
        if cand_score < best_score:
            best_h = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # occasional up‑sampling fallback (optional, rarely triggered)
        if no_improve >= improve_patience and n < 512:
            # linearly interpolate to double resolution
            old_x = np.linspace(0, 2, n, endpoint=False)
            new_n = n * 2
            new_x = np.linspace(0, 2, new_n, endpoint=False)
            h_interp = np.interp(new_x, old_x, best_h)
            target_sum = new_n / 2.0
            h = repair(h_interp, target_sum)
            n = new_n
            cur_score = compute_max_overlap(h)
            best_h = h.copy()
            best_score = cur_score
            no_improve = 0

    # final repair and evaluation
    best_h = repair(best_h, best_h.shape[0] / 2.0)
    final_score = compute_max_overlap(best_h)

    return best_h, final_score, best_h.shape[0]
