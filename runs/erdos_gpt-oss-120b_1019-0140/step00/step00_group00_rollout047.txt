def enforce_sum_and_bounds(h):
    n = len(h)
    target = n/2
    # Clamp to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Simple water-filling: adjust to reach target sum
    # Use while loop for small n
    max_iter = 1000
    for _ in range(max_iter):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # Determine free indices where we can add (if diff>0) or subtract (if diff<0)
        if diff > 0:
            free = np.where(h < 1 - 1e-12)[0]
            if len(free) == 0:
                break
            inc = diff / len(free)
            # cap inc to remain within bounds
            inc = min(inc, np.min(1 - h[free]))
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if len(free) == 0:
                break
            dec = (-diff) / len(free)
            dec = min(dec, np.min(h[free]))
            h[free] -= dec

def compute_gradient(h, shift):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        s = shift
        # part1: (1 - h[t+shift]) for t in 0..n-1-shift
        g[:n-s] += 1 - h[s:]
        # part2: -h[t-shift] for t in shift..n-1
        g[s:] -= h[:n-s]
    else:
        s = -shift
        # part1: (1 - h[t - s]) for t in s..n-1
        g[s:] += 1 - h[:n-s]
        # part2: -h[t + s] for t in 0..n-1-s
        g[:n-s] -= h[s:]
    return g

def enforce_sum_and_bounds(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative adjusting
    for _ in range(100):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some entries
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # ensure not exceeding bounds
            excess = np.maximum(h[free] + inc - 1.0, 0.0)
            if np.any(excess):
                # adjust inc so no exceed
                inc = min(inc, np.min(1.0 - h[free]))
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = -diff / free.size
            if np.any(h[free] - dec < 0):
                dec = np.min(h[free])
            h[free] -= dec
    # final clip ensures bounds
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h):
    """Return max overlap integral C5 for step function samples h."""
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_shift(h):
    """Return integer shift k (relative to zero) where correlation attains its maximum."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1)

def compute_gradient(h, shift):
    """Gradient of the max-correlation term (at given shift) w.r.t. h."""
    n = len(h)
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        s = shift
        if s < n:
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g

def enforce_sum_and_bounds(h):
    """Project h onto [0,1]^n with exact sum n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            inc = min(inc, np.min(1.0 - h[free]))
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            dec = -diff / free.size
            dec = min(dec, np.min(h[free]))
            h[free] -= dec
    return np.clip(h, 0.0, 1.0)

def optimize_h(h_init, time_budget):
    """Local descent on C5 using gradient‑guided two‑index swaps."""
    start = time.time()
    h = enforce_sum_and_bounds(h_init.copy())
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    cur_shift = compute_shift(h)

    step = 0.2
    min_step = 1e-5
    it = 0
    no_improve = 0

    while time.time() - start < time_budget:
        g = compute_gradient(h, cur_shift)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        eps = min(step, h[i], 1.0 - h[j])
        if eps <= 0:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j:
                continue
            eps = min(step, h[i], 1.0 - h[j])
            if eps <= 0:
                continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        new_val = compute_c5(h_new)

        if new_val < cur_val - 1e-12:
            h = h_new
            cur_val = new_val
            cur_shift = compute_shift(h)
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move to escape plateaus
            if np.random.rand() < 0.001:
                h = h_new
                cur_val = new_val
                cur_shift = compute_shift(h)
                no_improve = 0

        it += 1
        if it % 500 == 0:
            step = max(min_step, step * 0.9)

        # stochastic jitter when stuck

            for _ in range(10):

            cur_val = compute_c5(h)

            no_improve = 0

    return best_h, best_val
