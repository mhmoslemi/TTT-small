```python
import numpy as np, time, math, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # corr length = 2*n - 1; need max
    return dx * np.max(corr)

def random_move(h):
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # compute allowed range for epsilon:
    # new hi = hi + eps
    # new hj = hj - eps
    # constraints:
    # 0 <= hi+eps <= 1 => eps ∈ [-hi, 1-hi]
    # 0 <= hj-eps <= 1 => eps ∈ [hj-1, hj]
    lo = max(-h[i], h[j] - 1)
    hi = min(1 - h[i], h[j])
    if lo >= hi:
        return None, None
    # choose epsilon as random uniform in small step?; use uniform range
    eps = np.random.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h, (i, j, eps)
```

```
import cvxpy as cp

def project_to_feasible(v):
    n = v.shape[0]
    h_var = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(h_var - v))
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return h_var.value
```

import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 bound for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def random_move(h: np.ndarray):
    """Return a new feasible vector obtained by transferring a random epsilon
    from index i to index j (i != j).  If no feasible epsilon exists, return
    (None, None)."""
    n = h.shape[0]
    # pick two distinct indices
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    # Determine feasible range for epsilon so that 0<=h[i]+eps<=1 and 0<=h[j]-eps<=1
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return None, None
    eps = np.random.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h, (i, j, eps)

def feasible_binary(n: int) -> np.ndarray:
    """Return a binary feasible vector of length n (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # n is assumed even (the problem's integral constraint forces it)
    return h
