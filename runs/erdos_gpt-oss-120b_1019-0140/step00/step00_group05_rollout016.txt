def improve_h(h, time_budget, start_time):
    n=len(h)
    ones=np.where(h>0.5)[0]
    zeros=np.where(h<0.5)[0]
    best_h=h.copy()
    best_score=compute_c5(best_h)
    iter_count=0
    while True:
        elapsed=time.time()-start_time
        if elapsed>time_budget:
            break
        # temperature schedule:
        T = max(0.001, (time_budget - elapsed)/time_budget)
        # propose swap
        i = random.choice(ones)

import time
import math
import random
import numpy as np

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_to_simplex(x, target_sum):
    """
    Project a vector onto the intersection of the box [0,1] and the hyperplane sum(x)=target_sum.
    Simple algorithm: clip to [0,1], then adjust the free entries proportionally.
    """
    x = np.clip(x, 0.0, 1.0)
    cur_sum = x.sum()
    if abs(cur_sum - target_sum) <= 1e-12:
        return x
    # Identify free (non‑bound) indices
    free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
    if free.size == 0:
        # All entries are at a bound; distribute difference to the nearest bound
        # (should not happen for our usage)
        return x
    diff = target_sum - cur_sum
    adj = diff / free.size
    x[free] += adj
    # Clip again in case of tiny overflow
    x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimizes the
    Erdős overlap constant C5.

    Returns:
        h_values : np.ndarray of shape (n_points,) with values in [0,1]
        c5_bound : float, the achieved max overlap
        n_points : int, the number of discretisation points
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    init = globals().get('initial_h_values')
    if init is not None:
        h = np.array(init, dtype=float)
        n = h.shape[0]
        target_sum = n / 2.0
        # Enforce the integral constraint (may already hold)
        h = _project_to_simplex(h, target_sum)
    else:
        # Default resolution when no parent is supplied
        n = 70
        target_sum = n / 2.0
        h = np.zeros(n, dtype=float)
        ones_idx = np.random.choice(n, int(target_sum), replace=False)
        h[ones_idx] = 1.0

    # ------------------------------------------------------------------
    # Local search (simulated annealing + swaps)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_score(best_h)

    start_time = time.time()
    time_budget = budget_s * 0.95  # keep a safety margin

    # Parameters for the annealing schedule
    max_no_improve = 4000
    no_improve = 0

    # Pre‑compute indices sets once to avoid re‑allocation when possible
    # (they will be refreshed after each accepted move)
    while True:
        elapsed = time.time() - start_time
        if elapsed > time_budget:
            break

        # Linear cooling: start at 0.1, decay to 0.001
        T = max(0.001, (time_budget - elapsed) / time_budget * 0.1)

        # Choose a donor (h_i > 0) and a receiver (h_j < 1)
        donors = np.where(best_h > 1e-12)[0]
        receivers = np.where(best_h < 1.0 - 1e-12)[0]

        if donors.size == 0 or receivers.size == 0:
            break  # cannot move any mass

        i = np.random.choice(donors)
        j = np.random.choice(receivers)

        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue

        # Random amount to shift; occasionally take the whole mass (swap)
        if np.random.rand() < 0.2:
            delta = max_delta
        else:
            delta = np.random.rand() * max_delta

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # No need to re‑project: we transferred mass exactly, keeping the sum.
        # Clip just in case of numerical noise.
        np.clip(new_h, 0.0, 1.0, out=new_h)

        new_score = _c5_score(new_h)

        # Acceptance rule
        if new_score < best_score or np.random.rand() < math.exp(-(new_score - best_score) / T):
            best_h = new_h
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        # Random restart when stagnating
        if no_improve >= max_no_improve:

            rand_h = np.zeros(n, dtype=float)
            ones_idx = np.random.choice(n, int(target_sum), replace=False)
            rand_h[ones_idx] = 1.0
            rand_score = _c5_score(rand_h)
            if rand_score < best_score:

                best_score = rand_score
                no_improve = 0
            else:
                no_improve = 0  # give the current walk another chance

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    target_sum = n / 2.0

        best_h = _project_to_simplex(best_h, target_sum)

    final_score = _c5_score(best_h)

    return (best_h, float(final_score), int(n))
