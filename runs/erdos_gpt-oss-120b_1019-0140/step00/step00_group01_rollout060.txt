        for i in range(n):
            j = i + L
            if 0 <= j < n:

                constraints.append(w[i, offset] <= h[j])
                constraints.append(w[i, offset] >= h[i] + h[j] - 1)
                constraints.append(w[i, offset] >= 0)
            else:
                constraints.append(w[i, offset] == 0)

    prob = cp.Problem(cp.Minimize(t), constraints)

    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            prob.solve()  # default

        return None

    h_sol = np.clip(h_sol, 0.0, 1.0)
    # adjust sum via projection
    target = n/2.0

    return h_sol

import time
import math
import numpy as np
import cvxpy as cp

def compute_c5(h: np.ndarray) -> float:
    """Compute the C5 value for a step function h on [0,2] using full correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the box [0,1]^n while enforcing sum(x)=target.
    Uses bisection on a uniform shift lambda: sum(clamp(x+lambda,0,1)) = target.
    """
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def solve_lp(n: int) -> np.ndarray | None:
    """
    Solve the linear programming relaxation using McCormick envelopes.
    Returns a feasible h vector or None on failure.
    """
    target = n / 2.0
    h = cp.Variable(n)
    t = cp.Variable()
    w = cp.Variable((n, 2 * n - 1))
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target,
        t >= 0,
    ]

    for offset in range(2 * n - 1):
        L = offset - (n - 1)
        for i in range(n):
            j = i + L
            if 0 <= j < n:
                constraints.append(w[i, offset] <= h[i])
                constraints.append(w[i, offset] <= h[j])
                constraints.append(w[i, offset] >= h[i] + h[j] - 1)
                constraints.append(w[i, offset] >= 0)
            else:
                constraints.append(w[i, offset] == 0)
        constraints.append(target - cp.sum(w[:, offset]) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=20000,
                    abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000)
        except Exception:
            try:
                prob.solve()
            except Exception:
                return None

    if h.value is None:
        return None
    h_sol = np.array(h.value, dtype=float)
    h_sol = np.clip(h_sol, 0.0, 1.0)
    h_sol = project_box_sum(h_sol, target)
    return h_sol

                 budget: float,
                 start_time: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a guided simulated annealing / weighted hill‑climbing search.
    Returns the best h found and its C5 value.
    """
    n = h_init.size
    target = n / 2.0
    cur_h = np.copy(h_init)
    cur_val = compute_c5(cur_h)
    best_h = np.copy(cur_h)
    best_val = cur_val

    max_iter = 800_000
    T0 = 0.06
    max_step = 0.4

    for it in range(1, max_iter + 1):
        if time.time() - start_time > budget - 0.2:
            break

        # Linear cooling schedule
        T = T0 * (1.0 - it / max_iter)
        if T < 1e-7:
            T = 1e-7

        # Identify the worst shift (largest overlap)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = int(np.argmax(corr))
        L = idx - (n - 1)

        # Contribution of each index to the worst overlap
        if L >= 0:
            term = np.concatenate([1.0 - cur_h[L:], np.full(L, 1.0)])
        else:
            Lneg = -L
            term = np.concatenate([np.full(Lneg, 1.0), 1.0 - cur_h[:n - Lneg]])

        # Choose i with high contribution, j with low contribution
        prob_i = term / term.sum()
        i = rng.choice(n, p=prob_i)

        inv_term = term.max() - term + 1e-12
        prob_j = inv_term / inv_term.sum()
        j = rng.choice(n, p=prob_j)

        if i == j:
            continue

        # Adaptive step size respecting bounds and sum preservation
        step = max_step * (1.0 - it / max_iter) + 1e-5
        d_min = max(-cur_h[i], cur_h[j] - 1.0)
        d_max = min(1.0 - cur_h[i], cur_h[j])
        if d_min >= d_max:
            continue
        delta = rng.uniform(d_min, d_max)

        new_h = np.copy(cur_h)
        new_h[i] += delta
        new_h[j] -= delta

        new_val = compute_c5(new_h)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = np.copy(new_h)

    best_h = project_box_sum(best_h, target)
    best_val = compute_c5(best_h)
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Entry point required by the harness.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimension and initial vector
    if 'initial_h_values' in globals():
        init_h = np.copy(initial_h_values).astype(float)
        n = init_h.size
    else:
        n = 47
        target = n / 2.0
        init_h = rng.random(n)
        init_h = init_h / init_h.sum() * target
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = project_box_sum(init_h, target)

    # Try the LP relaxation for a strong starting point
    lp_h = solve_lp(n)
    if lp_h is not None:
        start_h = lp_h
    else:
        start_h = init_h

    # Allocate remaining budget to local refinement
    remaining = budget_s - (time.time() - start)
    if remaining > 6.0:
        best_h, best_val = local_search(start_h, remaining, start, rng)
    else:
        best_h = start_h
        best_val = compute_c5(best_h)

    # Final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    best_val = compute_c5(best_h)

    return best_h, float(best_val), n
