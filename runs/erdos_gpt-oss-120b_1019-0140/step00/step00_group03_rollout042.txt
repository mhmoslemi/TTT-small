def _normalize_h(vec: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(vec) == len(vec)/2 exactly."""
    vec = np.clip(vec, 0.0, 1.0)
    n = vec.size
    target = n * 0.5
    # simple iterative redistribution of the remaining error
    for _ in range(10):
        diff = target - vec.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = vec < 1.0
        else:
            mask = vec > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        add = diff / cnt
        vec[mask] = np.clip(vec[mask] + add, 0.0, 1.0)
    return np.clip(vec, 0.0, 1.0)


def _c5_score(h: np.ndarray) -> float:
    """Maximum overlap integral for shifts up to 2/5 of the interval."""
    n = h.size
    dmax = n // 5                     # shift bound corresponding to 2/5 of length 2
    start = n - 1                     # index of zero shift in full correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    # consider both positive and negative shifts within the bound
    window = corr[start - dmax : start + dmax + 1]
    return np.max(window) * (2.0 / n)
