def gradient_move(h, max_delta):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    lag = np.argmax(conv) - (n - 1)
    # compute gradient vector as earlier
    grad = np.zeros_like(h)
    if lag >= 0:
        # forward
        idx = np.arange(n - lag)
        grad[idx] += (1 - h[idx + lag])
        # backward
        idx_b = np.arange(lag, n)
        grad[idx_b] -= h[idx_b - lag]
    else:
        L = -lag
        idx = np.arange(L, n)
        grad[idx] += (1 - h[idx - L])
        idx_b = np.arange(0, n - L)
        grad[idx_b] -= h[idx_b + L]
    # direction: move mass from positive grad to negative grad
    pos = np.argmax(grad)  # most positive
    neg = np.argmin(grad)  # most negative
    # compute feasible change
    max_allowed = min(max_delta, h[pos], 1.0 - h[neg])
    if max_allowed <= 1e-12:
        return None, None, None
    return pos, neg, max_allowed

import numpy as np
import random
import time
import math

def evaluate_h(h, dx):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    # full correlation of h with 1-h, then scale by dx
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def gradient_move_info(h, lag):
    """Compute gradient for the given lag and return (i_pos, i_neg, max_delta)."""
    n = h.shape[0]
    grad = np.zeros_like(h)

    if lag >= 0:
        # contribution from h[i] * (1 - h[i+lag])
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        # contribution from -h[i-lag] in the second factor
        idx2 = np.arange(lag, n)
        grad[idx2] -= h[idx2 - lag]
    else:
        L = -lag
        idx = np.arange(L, n)
        grad[idx] += (1.0 - h[idx - L])
        idx2 = np.arange(0, n - L)
        grad[idx2] -= h[idx2 + L]

    # indices with largest positive and largest negative gradient
    i_pos = int(np.argmax(grad))
    i_neg = int(np.argmin(grad))

    # feasible amount that can be transferred from i_pos to i_neg
    max_delta = min(h[i_pos], 1.0 - h[i_neg])
    if max_delta <= 0.0:
        return None, None, None
    return i_pos, i_neg, max_delta

def propose_random_move(h, max_delta):
    """Generate a new h by moving a random amount between two positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)

    # delta added to i, subtracted from j
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi <= lo:
        return h.copy()
    # restrict magnitude
    lo = max(lo, -max_delta)
    hi = min(hi, max_delta)
    if hi <= lo:
        delta = (lo + hi) / 2.0
    else:
        delta = np.random.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def random_binary_feasible(n):
    """Create a binary feasible vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = n // 2
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h
