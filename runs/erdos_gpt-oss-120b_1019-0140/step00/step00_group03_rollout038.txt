    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    h0 = globals().get('initial_h_values', None)
    if h0 is not None:
        h = np.copy(h0).astype(float)
        n = h.shape[0]
    else:
        n = 200
        # create binary vector with n/2 ones
        h = np.zeros(n, dtype=float)
        indices = rng.choice(n, size=n//2, replace=False)
        h[indices] = 1.0

    # ensure sum constraint
    target_sum = n / 2
    cur_sum = h.sum()
    if np.abs(cur_sum - target_sum) > 1e-8:
        # adjust by setting some entries partially
        diff = target_sum - cur_sum
        # we may add diff to a random index
        # but diff should be integer or half integer? We'll just adjust one entry.
        # If diff >0, increase some zero to diff; else decrease some one.
        # Ensure bounds.
        if diff > 0:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = diff
        else:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 1 + diff
        # Clip ensure within [0,1]
        h = np.clip(h, 0, 1)

    # compute current best score
    dx = 2.0 / n
    def compute_c5(x):
        # full correlation
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    best_h = np.copy(h)
    best_score = compute_c5(best_h)

    target_score = 0.3808

    # parameters
    T = 0.1
    cooling = 0.9995

def step_reduce_max(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)  # shift could be negative
    if shift == 0:
        return h, max_val  # nothing
    # Determine indices where mismatch occurs
    if shift > 0:
        i_range = np.arange(0, n - shift)
        cond = (h[i_range] > 0) & (h[i_range + shift] < 1)
        mism_i = i_range[cond]
        if mism_i.size == 0:
            return h, max_val
        i = rng.choice(mism_i)
        # Move delta from i to i+shift
        max_delta = min(h[i], 1.0 - h[i + shift])
    else:
        # shift negative
        s = -shift
        i_range = np.arange(s, n)
        # h[i] is element for which shifted index i-shift = i-s
        # mismatches when h[i] >0 and h[i-shift] <1
        cond = (h[i_range] > 0) & (h[i_range - s] < 1)
        mism_i = i_range[cond]
        if mism_i.size == 0:
            return h, max_val
        i = rng.choice(mism_i)
        max_delta = min(h[i], 1.0 - h[i - s])

        return h, max_val

def _corr_max(h: np.ndarray) -> float:
    """Maximum of the correlation h * (1-h) (no dx scaling)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def _random_swap_step(h: np.ndarray, step_scale: float, rng: np.random.Generator):
    """Move a small amount of mass from a positive entry to a non‑full entry."""
    n = h.shape[0]
    pos = np.where(h > 0.0)[0]
    if pos.size == 0:
        return h, _corr_max(h)
    i = rng.choice(pos)

    free = np.where(h < 1.0)[0]
    if free.size == 0:
        return h, _corr_max(h)
    j = rng.choice(free)

    if i == j:
        return h, _corr_max(h)

    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return h, _corr_max(h)

    delta = rng.random() * max_delta * step_scale
    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand, _corr_max(cand)


def _directed_step(h: np.ndarray, step_scale: float, rng: np.random.Generator):
    """Reduce the current worst shift by moving mass from a mismatching 1 to a 0."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)   # shift can be negative

    # No improvement possible if shift is zero
    if shift == 0:
        return h, float(corr[max_idx])

    # Find a mismatching pair (h[i]=1, h[i+shift]=0)
    if shift > 0:
        rng_range = np.arange(0, n - shift)
        mask = (h[rng_range] > 0.0) & (h[rng_range + shift] < 1.0)
        mism = rng_range[mask]
        if mism.size == 0:
            return h, float(corr[max_idx])
        i = rng.choice(mism)
        max_delta = min(h[i], 1.0 - h[i + shift])
        if max_delta <= 0.0:
            return h, float(corr[max_idx])
        delta = rng.random() * max_delta * step_scale
        cand = h.copy()
        cand[i] -= delta
        cand[i + shift] += delta
    else:  # shift < 0
        s = -shift
        rng_range = np.arange(s, n)
        mask = (h[rng_range] > 0.0) & (h[rng_range - s] < 1.0)
        mism = rng_range[mask]
        if mism.size == 0:
            return h, float(corr[max_idx])
        i = rng.choice(mism)
        max_delta = min(h[i], 1.0 - h[i - s])
        if max_delta <= 0.0:
            return h, float(corr[max_idx])
        delta = rng.random() * max_delta * step_scale
        cand = h.copy()
        cand[i] -= delta
        cand[i - s] += delta

    return cand, _corr_max(cand)

        # adjust a random entry that can absorb the error
        if err > 0:
            cand = np.where(best_h > 0.0)[0]
        else:
            cand = np.where(best_h < 1.0)[0]
        if cand.size:
            idx = rng.choice(cand)
            best_h[idx] = np.clip(best_h[idx] - err, 0.0, 1.0)

    # Final C5 value with proper dx scaling
    c5_bound = best_corr * (2.0 / n)

    return best_h, c5_bound, n
