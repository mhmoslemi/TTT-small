    # Clip
    h = np.clip(h, 0, 1)
    # Compute current sum
    cur = np.sum(h)
    if np.isclose(cur, target_sum):
        return h
    # Compute needed adjustment
    diff = target_sum - cur
    # Identify indices not at bounds
    free = np.where((h > 0) & (h < 1))[0]
    if free.size == 0:
        # all at bounds, can't adjust; just return clipped (maybe infeasible)
        return h

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_to_feasible(h, target_sum):
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    # iterative adjustment to meet sum
    # We'll do a simple projection onto the hyperplane while respecting bounds:
    # Use algorithm: while error > tol, compute diff and distribute among non-bound values.
    tol = 1e-8
    max_iter = 10
    for _ in range(max_iter):
        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) <= tol:
            break
        # indices where h not at bounds
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # cannot adjust, break
            break
        # distribute diff equally
        delta = diff / free.size
        h[free] += delta
        # clip again
        h = np.clip(h, 0, 1)
    # final correction: adjust one element if still off by tolerance
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) > 1e-6:
        i = np.argmin(h)  # choose smallest (most ability to increase)
        adjust = target_sum - cur_sum
        h[i] = np.clip(h[i] + adjust, 0, 1)
    return h

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return the overlap bound C5 for a sampled step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h, target_sum, tol=1e-8, max_iter=10):
    """
    Project a vector onto the intersection of the box [0,1]^n and the hyperplane
    sum(h)=target_sum. Simple iterative scheme that distributes the deficit
    among interior entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h)
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = np.sum(h)
    if abs(cur - target_sum) > 1e-6:
        i = np.argmin(h)  # most room to increase
        h[i] = np.clip(h[i] + (target_sum - cur), 0.0, 1.0)
    return h


def resample_h(base_h, n_target):
    """Interpolate base_h (array) to length n_target and enforce feasibility."""
    base_h = np.asarray(base_h, dtype=float)
    n_base = base_h.shape[0]
    if n_target == n_base:
        h_new = base_h.copy()
    else:
        x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_base, base_h)
    h_new = project_to_feasible(h_new, n_target / 2.0)
    return h_new


def _objective(h, n):
    """Objective for optimizer: compute C5 for given n."""
    return compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Retrieve the provided parent construction if it exists.
    base_h = globals().get("initial_h_values")
    if base_h is not None:
        base_h = np.asarray(base_h, dtype=float)

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # Candidate sample counts (small enough for fast evaluation).
    candidate_ns = [95, 190, 380]

    for n in candidate_ns:
        # Number of restarts chosen heuristically (more for smaller n).
        max_restarts = 30 if n <= 95 else (10 if n <= 190 else 5)
        for restart in range(max_restarts):
            # Respect the overall time budget.
            if time.time() - start_time > budget_s * 0.97:
                break

            # Initialise h0.
            if base_h is not None and restart == 0:
                h0 = resample_h(base_h, n)
            else:
                h0 = rng.random(n)
                h0 = project_to_feasible(h0, n / 2.0)

            # Small random perturbation for diversity (skip on first restart).
            if restart > 0:
                noise = rng.normal(scale=0.07, size=n)
                h0 = np.clip(h0 + noise, 0.0, 1.0)
                h0 = project_to_feasible(h0, n / 2.0)

            # Define constraint and bounds for SLSQP.
            cons = {"type": "eq", "fun": lambda h, nd=n: np.sum(h) - nd / 2.0}
            bnds = [(0.0, 1.0)] * n

            # Run local optimizer.
            try:
                res = opt.minimize(
                    fun=lambda h: _objective(h, n),
                    x0=h0,
                    method="SLSQP",
                    bounds=bnds,
                    constraints=cons,
                    options={"maxiter": 800, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                continue

            # Extract solution, enforce feasibility explicitly.
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_to_feasible(h_opt, n / 2.0)

            c5_val = compute_c5(h_opt)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

            # Early exit if an excellent bound is already achieved.
            if best_c5 <= 0.3808:
                # continue a few more iterations to possibly improve further
                pass

    # Fallback in the unlikely event no solution was found.
    if best_h is None:
        if base_h is not None:
            best_h = base_h.copy()
            best_n = best_h.shape[0]
        else:
            best_n = 95
            best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Final safety projection.
    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
