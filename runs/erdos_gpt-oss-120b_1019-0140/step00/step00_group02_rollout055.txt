    # Determine direction: we can transfer delta from i to j

import numpy as np, random, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift

def propose_move(h, shift):
    n = len(h)
    # define overlapping region indices for given shift
    if shift >= 0:
        overlap = np.arange(n - shift)
    else:
        overlap = np.arange(-shift, n)
    # Choose i from overlap, j from complement (or random if complement empty)
    if overlap.size > 0:
        i = int(random.choice(overlap))
        # complement indices
        complement = np.setdiff1d(np.arange(n), overlap, assume_unique=True)
        if complement.size > 0:
            j = int(random.choice(complement))
        else:
            j = int(random.randrange(n))
    else:
        # fallback to any pair
        i, j = random.sample(range(n), 2)
    # Transfer mass from j to i if possible, else opposite direction
    max_delta = min(h[j], 1.0 - h[i])
    if max_delta > 0:
        delta = random.random() * max_delta
        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new
    else:
        # try opposite direction
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta > 0:
            delta = random.random() * max_delta
            new = h.copy()
            new[i] -= delta
            new[j] += delta
            return new
        else:
            # no feasible move, return unchanged copy
            return h.copy()

import numpy as np
import random
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def worst_shift(h: np.ndarray) -> int:
    """Index of the shift producing the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return idx - (n - 1)          # shift ∈ [-(n‑1), …, n‑1]

def propose_move(h: np.ndarray, shift: int) -> np.ndarray:
    """Create a feasible neighbour by moving mass from one index to another."""
    n = len(h)
    # indices that belong to the overlapping region for the given shift
    if shift >= 0:
        overlap = np.arange(n - shift)
    else:
        overlap = np.arange(-shift, n)

    # try to pick i from the overlap and j from the complement
    if overlap.size:
        i = int(random.choice(overlap))
        complement = np.setdiff1d(np.arange(n), overlap, assume_unique=True)
        if complement.size:
            j = int(random.choice(complement))
        else:
            j = int(random.randrange(n))
    else:
        i, j = random.sample(range(n), 2)

    # move mass from j to i if possible, otherwise the opposite direction
    max_delta = min(h[j], 1.0 - h[i])
    if max_delta > 0.0:
        delta = random.random() * max_delta
        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta > 0.0:
        delta = random.random() * max_delta
        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new
    # nothing feasible – return a copy unchanged
    return h.copy()
