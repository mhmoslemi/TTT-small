def compute_c5(h, dx):
    # Compute correlation of h with (1 - h) for all integer shifts in [-n+1, n-1] (full)
    # Using FFT-based cross-correlation length 2n-1.
    n = len(h)
    size = 2*n - 1
    # Next power of two for speed
    m = 1 << (size - 1).bit_length()
    # Zero pad to m
    H = np.fft.rfft(h, n=m)  # Real FFT
    G = np.fft.rfft(1 - h, n=m)
    # Cross-correlation: ifft( H * conj(G) )
    corr = np.fft.irfft(H * np.conj(G), n=m)
    # Extract the relevant part: length 2n-1, index 0..2n-2 => correlate length.
    # For np.correlate full, the zero lag is at index n-1 (center).
    corr = corr[:size]  # first 2n-1 elements
    # Might have small negative rounding errors; keep as float.
    return np.max(corr) * dx

def compute_c5_fast(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft((1 - h)[::-1], n=m)
    conv = np.fft.irfft(H * G, n=m)
    conv = conv[:2*n-1]
    return np.max(conv)

def compute_c5_true(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Use np.correlate (full)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_fast(h):
    n = h.shape[0]
    H = np.fft.rfft(h, n=1 << ((2*n - 1).bit_length()))
    G = np.fft.rfft((1 - h)[::-1], n=1 << ((2*n - 1).bit_length()))
    conv = np.fft.irfft(H * G, n=1 << ((2*n - 1).bit_length()))
    conv = conv[:2*n - 1]
    return np.max(conv)

def anneal_search(rng, n, init_h, time_budget):
    start = time.time()
    dx = 2.0 / n
    if init_h is not None:
        cur_h = init_h.copy()
    else:
        cur_h = np.zeros(n, dtype=np.float64)
        cur_h[:n//2] = 1.0
        rng.shuffle(cur_h)
    cur_c5 = compute_c5_fast(cur_h) * dx
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = 0.05
    cooling = 0.9995
    while time.time() - start < time_budget * 0.95:
        # Choose move type: swap or fractional
        # We'll just swap for simplicity (binary)
        # find indices where cur_h > 0.5, etc.
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i1] = 0.0
        new_h[i0] = 1.0
        new_c5 = compute_c5_fast(new_h) * dx
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        T *= cooling
        if T < 1e-6:
            T = 0.05
        # occasional random restart
        if rng.random() < 0.001:
            if rng.random() < 0.5:
                cur_h = best_h.copy()

            else:
                cur_h = np.zeros(n, dtype=np.float64)
                cur_h[:n//2] = 1.0
                rng.shuffle(cur_h)
                cur_c5 = compute_c5_fast(cur_h) * dx

import numpy as np, math, time

def _c5_fast(h, m):
    """Fast max of correlation (without multiplying by dx)."""
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft((1 - h)[::-1], n=m)
    conv = np.fft.irfft(H * G, n=m)
    L = 2 * h.shape[0] - 1
    return np.max(conv[:L])
