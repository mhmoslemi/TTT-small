import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def propose_move(h, step_limit=0.05):
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    # compute feasible delta range
    max_move_i_to_j = min(h[i], 1.0 - h[j])
    max_move_j_to_i = min(h[j], 1.0 - h[i])
    low = -max_move_j_to_i
    high = max_move_i_to_j
    # limit step magnitude
    low = max(low, -step_limit)
    high = min(high, step_limit)
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def simulated_annealing(initial_h, time_limit):
    start = time.time()
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    n = len(initial_h)
    target_sum = n / 2.0
    # Ensure sum correct within floating tolerance
    # Should be already satisfied.
    # Temperature schedule
    T0 = 0.01 # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        # temperature decays linearly
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        nxt = propose_move(current_h, step_limit=0.05)
        if nxt is None:
            continue
        # due to floating point, enforce bounds clipping
        # though propose_move ensures feasibility
        nxt_c5 = compute_c5(nxt)
        delta = nxt_c5 - current_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            current_h = nxt
            current_c5 = nxt_c5
            if nxt_c5 < best_c5:
                best_h = nxt.copy()
                best_c5 = nxt_c5
    return best_h, best_c5
```

def random_feasible_h(n):
    # generate random Dirichlet and adjust any >1
    # start with exponential draws
    x = np.random.exponential(scale=1.0, size=n)
    x = x / np.sum(x) * (n/2.0)
    # If any exceeds 1, clip and redistribute
    while np.any(x > 1):
        excess = x - 1
        excess[excess < 0] = 0
        total_excess = np.sum(excess)
        x = np.where(x > 1, 1, x)
        # distribute excess to those below 1 proportionally
        below = x < 1
        if not np.any(below):
            break
        # give each below element an amount proportionate to (1-x)
        capacity = 1 - x[below]
        # Normalize capacity
        cap_sum = np.sum(capacity)
        if cap_sum == 0:
            break
        add = np.minimum(capacity, excess.sum() * (capacity / cap_sum))
        x[below] += add
    # Now adjust sum precisely:
    # compute diff and adjust by adding small epsilon to random indices not at boundaries
    diff = (n/2.0) - np.sum(x)

import numpy as np
import time
import math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def propose_move(h, step_limit=0.05):
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    # feasible delta from i to j
    max_i_to_j = min(h[i], 1.0 - h[j])
    max_j_to_i = min(h[j], 1.0 - h[i])
    low = -max_j_to_i
    high = max_i_to_j
    # clamp by step limit
    low = max(low, -step_limit)
    high = min(high, step_limit)
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def random_jitter(h, steps=5, step_limit=0.05):
    new_h = h.copy()
    for _ in range(steps):
        cand = propose_move(new_h, step_limit)
        if cand is not None:
            new_h = cand
    return new_h

    n = start_h.shape[0]
    best_h = start_h.copy()

import numpy as np
import time
import math

def compute_c5(h):
    """Maximum overlap integral for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def propose_move(h, step_limit=0.05):
    """Generate a feasible move that conserves the total sum."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    # feasible range for delta transferred from i to j
    max_i_to_j = min(h[i], 1.0 - h[j])
    max_j_to_i = min(h[j], 1.0 - h[i])
    low = -max_j_to_i
    high = max_i_to_j
    low = max(low, -step_limit)
    high = min(high, step_limit)
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def random_jitter(h, steps=5, step_limit=0.05):
    """Apply a few random feasible moves without any acceptance test."""
    new_h = h.copy()
    for _ in range(steps):
        cand = propose_move(new_h, step_limit)
        if cand is not None:
            new_h = cand
    return new_h

def simulated_annealing(start_h, duration):
    """
    Simple SA that preserves the sum constraint.
    Returns the best vector seen and its C5 value.
    """
    start = time.time()
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.02  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= duration:
            break
        T = T0 * max(0.0, 1.0 - elapsed / duration)  # linear cooling
        cand = propose_move(cur_h, step_limit=0.05)
        if cand is None:
            continue
        cand_val = compute_c5(cand)
        delta = cand_val - cur_val
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_h = cand.copy()
                best_val = cand_val
    return best_h, best_val

    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
