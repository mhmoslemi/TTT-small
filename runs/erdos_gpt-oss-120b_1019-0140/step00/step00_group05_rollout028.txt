def project_to_sum_and_bounds(v, target):
    # solves min ||x - v||^2 s.t. sum x = target, 0 ≤ x ≤ 1
    # using dual variable lambda search.
    # The problem is convex; the solution is x_i = clip(v_i - λ, 0, 1).
    # We need find λ such that sum(x_i) = target.
    lo = -np.max(v) - 1.0  # ensures v_i - lo large => x_i=1
    hi = np.max(v) + 1.0   # ensures v_i - hi small => x_i=0
    # binary search
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2.0), 0.0, 1.0)

def solve_lp(n):
    import cvxpy as cp, numpy as np
    dx = 2.0 / n
    target = n / 2.0
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    # For each lag L
    for L in range(-(n-1), n):
        size = n - abs(L)
        w = cp.Variable(size)
        if L >= 0:
            i_idx = np.arange(size)
            j_idx = i_idx + L
        else:
            i_idx = np.arange(size) + (-L)
            j_idx = np.arange(size)
        # build constraints
        constraints.append(w >= 0)
        constraints.append(w >= h[i_idx] + h[j_idx] - 1)
        constraints.append(w <= h[i_idx])
        constraints.append(w <= h[j_idx])
        constraints.append(t >= dx * (target - cp.sum(w)))
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver='ECOS', warm_start=False, max_iters=5000) # optional parameters; may need to set verbose False.
    except Exception:
        return None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    h_val = h.value
    if h_val is None:
        return None
    # clip
    h_val = np.clip(h_val, 0.0, 1.0)
    # adjust sum to target (due to tolerance)
    h_val = project_to_sum_and_bounds(h_val, target)  # use earlier function
    return h_val

def random_transfer(h, rng, max_delta=0.2):

    if max_transfer <= 0:
        # maybe try opposite direction
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:

        # swap i and j
        i, j = j, i

# Helper: projection onto {0 ≤ x ≤ 1, sum(x)=target}
def _project_sum_bounds(v, target):
    """Project vector v onto the box [0,1] and hyperplane sum=v_target."""
    # dual variable (lambda) search: x_i = clip(v_i - lam, 0, 1)
    lo = -np.max(v) - 1.0   # enough negative -> all ones
    hi = np.max(v) + 1.0    # enough positive -> all zeros
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

# Helper: evaluate C5 (use pre‑imported evaluator if available)
def _eval_c5(vec):
    try:
        return evaluate_erdos_solution(vec)   # provided by the harness
    except Exception:
        # fallback: same definition as the harness
        n = vec.shape[0]
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return corr.max() * (2.0 / n)

# Helper: random mass transfer between two positions (keeps sum fixed)
def _random_transfer(h, rng, max_delta=0.2):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # try moving from i to j
    avail = min(h[i], 1.0 - h[j])
    if avail <= 0:
        # try opposite direction
        avail = min(h[j], 1.0 - h[i])
        if avail <= 0:
            return h.copy()
        i, j = j, i
    delta = max_delta * rng.rand()
    delta = min(delta, avail)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

# Helper: one gradient step for the currently active lag
def _gradient_step(h, dx, target_sum):
    n = h.shape[0]
    # correlation to locate worst lag
    corr = np.correlate(h, 1.0 - h, mode="full")
    lag = corr.argmax() - (n - 1)          # integer lag with max overlap
    grad = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            hi = h[: n - lag]
            hj = h[lag:]
            grad[: n - lag] += 1.0 - hj
            grad[lag:]      += -hi
    else:
        L = -lag
        hi = h[L:]
        hj = h[: n - L]
        grad[L:]      += 1.0 - hj
        grad[: n - L] += -hi
    return grad, lag

# Linear‑programming relaxation using McCormick envelopes
def _solve_lp(n, target_sum, dx):
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
    ]

    # product approximations for every lag (including negatives)
    for L in range(-(n - 1), n):
        size = n - abs(L)
        w = cp.Variable(size)               # w approximates h_i * h_{i+L}
        if L >= 0:
            i_idx = np.arange(size)
            j_idx = i_idx + L
        else:
            off = -L
            i_idx = np.arange(size) + off
            j_idx = np.arange(size)
        # McCormick envelope constraints
        constraints.append(w >= 0)
        constraints.append(w >= h[i_idx] + h[j_idx] - 1)
        constraints.append(w <= h[i_idx])
        constraints.append(w <= h[j_idx])
        # bound for the objective (t >= dx * (target - sum w))
        constraints.append(t >= dx * (target_sum - cp.sum(w)))

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver="ECOS", verbose=False, max_iters=5000)
    except Exception:
        return None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    h_val = np.asarray(h.value).reshape(-1)
    if h_val is None:
        return None
    h_val = np.clip(h_val, 0.0, 1.0)
    h_val = _project_sum_bounds(h_val, target_sum)
    return h_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)

    # --- initialise problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
    else:
        n = 70
        h0 = np.full(n, 0.5)                     # a neutral start
    target_sum = n / 2.0
    dx = 2.0 / n

    # enforce constraints on the start vector
    h0 = _project_sum_bounds(h0, target_sum)

    best_h = h0.copy()
    best_score = _eval_c5(best_h)

    # ------------------------------------------------------------
    # first attempt: linear‑programming relaxation (fast, often good)
    lp_h = _solve_lp(n, target_sum, dx)
    if lp_h is not None:
        lp_score = _eval_c5(lp_h)
        if lp_score < best_score:
            best_h, best_score = lp_h, lp_score

    # ------------------------------------------------------------
    # continue with stochastic local improvement
    current_h = best_h.copy()
    current_score = best_score

    start_time = time.time()
    iteration = 0
    max_iter = 200000                # generous upper bound; time controls exit

    while True:
        # time guard
        if time.time() - start_time > budget_s - 0.2:
            break

        # --- propose a random mass transfer ------------------------------------
        cand = _random_transfer(current_h, rng, max_delta=0.25)
        cand_score = _eval_c5(cand)

        # accept if improvement, otherwise occasional uphill move (simulated annealing)
        if cand_score < current_score:
            current_h, current_score = cand, cand_score
        else:
            # simple temperature schedule
            elapsed = time.time() - start_time
            temp = max(0.01, 0.4 * (1.0 - elapsed / budget_s))
            if rng.rand() < math.exp((current_score - cand_score) / temp):
                current_h, current_score = cand, cand_score

        # keep global best
        if current_score < best_score:
            best_h, best_score = current_h.copy(), current_score

        # --- every few hundred iterations perform a tiny gradient descent --------
        if iteration % 300 == 0:
            # short gradient refinement (3 sub‑steps)
            for _ in range(3):
                grad, active_lag = _gradient_step(current_h, dx, target_sum)
                norm = np.linalg.norm(grad)
                if norm == 0.0:
                    break
                # step size decays with time
                elapsed = time.time() - start_time
                step = 0.07 * (1.0 - elapsed / budget_s)
                step = max(step, 1e-5)
                new_h = current_h - step * grad
                new_h = _project_sum_bounds(new_h, target_sum)
                new_score = _eval_c5(new_h)
                if new_score < current_score:
                    current_h, current_score = new_h, new_score
                    if new_score < best_score:
                        best_h, best_score = new_h.copy(), new_score

        iteration += 1

        # ---- occasional random restart to escape possible plateaus -------------
        if iteration % 5000 == 0 and iteration > 0:
            # keep the best found, but explore from a noisy perturbation
            pert = best_h + 0.05 * (rng.rand(n) - 0.5)
            pert = _project_sum_bounds(pert, target_sum)
            pert_score = _eval_c5(pert)
            if pert_score < best_score:
                best_h, best_score = pert.copy(), pert_score
                current_h, current_score = pert.copy(), pert_score
            else:
                # occasionally jump to a fresh random start
                if rng.rand() < 0.15:
                    fresh = rng.rand(n)
                    fresh = _project_sum_bounds(fresh, target_sum)
                    fresh_score = _eval_c5(fresh)
                    if fresh_score < best_score:
                        best_h, best_score = fresh.copy(), fresh_score
                    current_h, current_score = fresh.copy(), fresh_score

    # final evaluation (in case rounding drifted)
    final_score = _eval_c5(best_h)
    return (best_h, final_score, n)
