import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_feasible(n):
    # produce binary vector with exactly n/2 ones (assuming n even)
    # else use fractional values to achieve sum.
    ones = n // 2
    arr = np.zeros(n)
    arr[:ones] = 1.0
    np.random.shuffle(arr)
    if n % 2 != 0:
        # need extra 0.5 to make sum = n/2
        # set one random entry to 0.5
        idx = np.random.randint(n)
        arr[idx] = 0.5
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # get initial solution if exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 60  # choose default moderate n if not provided
        h = random_feasible(n)
    
    # ensure sum constraint
    # compute dx
    dx = 2.0 / n
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # project onto sum constraint by scaling all values and clipping
        # simpler: adjust one element
        diff = target_sum - cur_sum
        # add diff to a random element within [0,1] bounds
        # find element with enough room
        for _ in range(100):
            idx = rng.integers(n)
            new_val = np.clip(h[idx] + diff, 0.0, 1.0)
            if abs(new_val - h[idx] - diff) < 1e-12:  # diff fits
                h[idx] = new_val
                break
        # if not fixed, just rescale all values to meet sum (and clip)
        # but this scenario unlikely
        h = np.clip(h, 0, 1)
        # final scaling
        sum_h = h.sum()
        if sum_h != 0:
            h *= target_sum / sum_h
        h = np.clip(h, 0, 1)
    
    # compute initial c5
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    
    # Simulated annealing parameters
    # We'll schedule temperature decreasing from initial_temp to near 0
    init_temp = 0.02  # small step
    final_temp = 0.0001
    # total iterations approximated
    iters = 0
    # Use while time left > small
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:  # leave some margin
            break
        # compute temperature decreasing linearly with elapsed fraction
        frac = elapsed / budget_s
        temp = init_temp * (1 - frac) + final_temp * frac
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # moving delta from i to j
        max_delta = min(current_h[i], 1.0 - current_h[j])
        if max_delta <= 1e-12:
            continue
        # draw delta uniform up to max_delta
        delta = rng.random() * max_delta
        # optionally scale delta by step size factor controlled by temp (larger delta early)
        # maybe we limit delta to e.g., max 0.1
        max_step = 0.2
        if delta > max_step:
            delta = max_step
        # apply move
        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # compute c5
        new_c5 = compute_c5(new_h)
        # accept if improvement or according to SA probability
        accept = False
        if new_c5 < current_c5:
            accept = True
        else:
            # probability
            prob = math.exp((current_c5 - new_c5) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        iters += 1
        # maybe periodically restart random perturbation
        if iters % 5000 == 0:
            # random restart with some probability
            if rng.random() < 0.1:
                # random feasible vector
                h_rand = random_feasible(n)
                # optionally small random noise within bounds
                # ensure sum constraint
                # we already guarantee sum via random_feasible
                current_h = h_rand
                current_c5 = compute_c5(h_rand)
    # final result
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), n

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----- local refinement with SLSQP ---------------------------------
    bounds = [(0.0, 1.0)] * n
    lin_cons = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])
    try:
        res = minimize(
            compute_c5, best_h, method='SLSQP',
            bounds=bounds, constraints=[lin_cons],
            options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            x_opt = np.clip(res.x, 0.0, 1.0)
            # enforce exact sum (tiny correction)
            diff = target_sum - x_opt.sum()
            if abs(diff) > 1e-12:
                for idx in range(n):
                    cap = (1.0 - x_opt[idx]) if diff > 0 else -x_opt[idx]
                    delta = max(min(diff, cap), -cap)
                    x_opt[idx] += delta
                    diff -= delta
                    if abs(diff) < 1e-12:
                        break
            c5_opt = compute_c5(x_opt)
            if c5_opt < best_c5:
                best_h = x_opt
                best_c5 = c5_opt
    except Exception:
        pass

    # ----- simulated annealing / hill climbing -------------------------
    current_h = best_h.copy()
    current_c5 = best_c5

    init_temp = 0.02
    final_temp = 0.0002
    max_step = 0.3

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97:
            break

        frac = elapsed / budget_s
        temp = init_temp * (1 - frac) + final_temp * frac

        # propose a transfer of mass from i to j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_delta = min(current_h[i], 1.0 - current_h[j])
        if max_delta <= 1e-12:
            continue
        scale = temp / init_temp + 0.01
        delta = rng.random() * max_delta * scale
        delta = min(delta, max_step)

        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < current_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - current_c5) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

        # occasional random restart (preserving feasibility)
        if elapsed > budget_s * 0.5 and rng.random() < 0.001:
            cand = random_feasible(n, rng)
            current_h = cand
            current_c5 = compute_c5(current_h)

    final_c5 = compute_c5(best_h)
    # Use provided evaluator if present
    if 'evaluate_erdos_solution' in globals():
        try:
            final_c5 = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h.tolist(), float(final_c5), n
