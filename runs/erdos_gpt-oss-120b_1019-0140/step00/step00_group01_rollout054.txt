import time, numpy as np

def _c5_of(h):
    # evaluate solution using pre-imported evaluate_erdos_solution
    return evaluate_erdos_solution(h)

def _feasible_random(n, rng):
    # start from 0.5 baseline, apply random pairwise moves many times
    h = np.full(n, 0.5, dtype=float)
    target = n/2.0
    # perform some random moves to randomize
    for _ in range(5 * n):
        i, j = rng.choice(n, 2, replace=False)
        # step
        lo = max(-h[i], -(1 - h[j]))
        hi = min(1 - h[i], h[j])
        if lo >= hi:
            continue
        delta = rng.uniform(lo, hi)
        h[i] += delta
        h[j] -= delta
    # ensure sum exactly target if small drift
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # adjust first index
        for idx in range(n):
            if 0 <= h[idx] + diff <= 1:
                h[idx] += diff
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine dimensionality
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 47
        h0 = np.full(n, 0.5, dtype=float)

    # Ensure feasibility
    target_sum = n / 2.0
    s = h0.sum()
    if abs(s - target_sum) > 1e-12:
        diff = target_sum - s
        # try to distribute diff to a coordinate within bounds
        for idx in range(n):
            if 0 <= h0[idx] + diff <= 1:
                h0[idx] += diff
                break
    # Clean any numerical out-of-bounds
    np.clip(h0, 0.0, 1.0, out=h0)

    best_h = h0.copy()
    best_c5 = _c5_of(best_h)

    curr_h = best_h.copy()
    curr_c5 = best_c5

    # SA parameters
    T0 = max(0.01, best_c5 * 0.5)
    Tmin = 1e-6
    step_init = 0.2

    # iteration loop
    it = 0
    while True:
        # Time check
        if time.time() - start > budget_s * 0.99:
            break
        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = T0 * (1 - frac) + Tmin * frac
        step = step_init * (1 - frac) + 1e-4 * frac

        # propose move
        i, j = rng.choice(n, 2, replace=False)
        hi = curr_h[i]
        hj = curr_h[j]
        lo_delta = max(-hi, -(1 - hj))
        hi_delta = min(1 - hi, hj)
        # Respect step bound
        lo_delta = max(lo_delta, -step)
        hi_delta = min(hi_delta, step)
        if lo_delta >= hi_delta:
            continue
        delta = rng.uniform(lo_delta, hi_delta)
        new_h = curr_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        # evaluate
        new_c5 = _c5_of(new_h)

        # acceptance
        if new_c5 < curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-12)):
            curr_h = new_h
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

        # occasional random perturbation to escape stagnation
        if it % 5000 == 0 and it > 0:
            # small random shake
            for _ in range(3):
                i2, j2 = rng.choice(n, 2, replace=False)
                lo2 = max(-curr_h[i2], -(1 - curr_h[j2]))
                hi2 = min(1 - curr_h[i2], curr_h[j2])
                if lo2 >= hi2:
                    continue
                delta2 = rng.uniform(lo2, hi2) * 0.5
                curr_h[i2] += delta2
                curr_h[j2] -= delta2
            # recompute cost after shake
            curr_c5 = _c5_of(curr_h)
            if curr_c5 < best_c5:
                best_h = curr_h.copy()
                best_c5 = curr_c5

        it += 1

    # final check: ensure feasible
    np.clip(best_h, 0.0, 1.0, out=best_h)
    s = best_h.sum()
    if abs(s - target_sum) > 1e-9:
        # small correction by scaling
        best_h = best_h * (target_sum / s)
        np.clip(best_h, 0.0, 1.0, out=best_h)

    return best_h, float(best_c5), n

import time
import numpy as np

def _c5_and_shift(h):
    """Return (C5 value, shift index) for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h (full mode)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx] * dx
    shift = idx - (n - 1)  # integer shift causing the maximum
    return float(max_val), shift

def _make_feasible(h, target):
    """Clamp to [0,1] and correct sum to target if needed."""
    np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    if abs(s - target) > 1e-12:
        # distribute the tiny difference to a non‑boundary entry
        diff = target - s
        for i in range(h.size):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):

    best_c5, best_shift = _c5_and_shift(best_h)

    while True:

        # schedule temperature and step size
