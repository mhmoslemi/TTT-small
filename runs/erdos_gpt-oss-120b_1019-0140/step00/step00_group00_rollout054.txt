import time, numpy as np

def evaluate(h):
    corr = np.correlate(h, 1 - h, mode='full')
    C5 = corr.max() * (2.0 / len(h))
    return C5

def hill_climb(initial_h, time_budget, start_time):
    h = initial_h.copy()
    n = len(h)
    best_h = h.copy()
    best_score = evaluate(h)
    # Determine step size schedule
    step = 0.5
    iteration = 0
    while time.time() - start_time < time_budget:
        # random i,j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # ensure we can transfer some amount
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            continue
        # choose delta
        delta = np.random.rand() * max_transfer * step # maybe step factor
        # small updates: take small delta
        h[i] -= delta
        h[j] += delta
        # evaluate
        score = evaluate(h)
        # accept if improved
        if score < best_score - 1e-12: # tolerance
            best_score = score
            best_h = h.copy()
            # optionally increase step or accept larger changes
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        # optionally adjust step size based on iteration
        iteration += 1
        # maybe reduce step after many iter
        if iteration % 10000 == 0:
            step *= 0.9
            # reinitialize from best, ensure small random jitter?
            h = best_h.copy()
    return best_h, best_score

def targeted_move(h, shift):
    n = len(h)
    if shift >= 0:
        # compute contributions for indices where i+shift < n
        contrib = np.zeros(n)
        valid = np.arange(n-shift)
        contrib[valid] = h[valid] * (1 - h[valid + shift])
    else:
        s = -shift
        # shift negative: correlation uses i+s vs i; for shift negative, pairs (i,s+i) where i+shift >= 0
        contrib = np.zeros(n)
        valid = np.arange(s, n)

def correct_sum(vec, target):
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # Try to adjust single entry that can accommodate diff
    if diff > 0:
        # Find index with value < 1 - diff (or at least capacity)
        idxs = np.where(vec < 1)[0]
        # If any can accommodate entirely, adjust first
        for idx in idxs:
            capacity = 1 - vec[idx]
            if capacity >= diff - 1e-12:
                vec[idx] += diff
                return vec
        # else distribute equally
        inc = diff / len(idxs)
        vec[idxs] += inc
        # may overshoot some >1 due to rounding; clip then re-balance
        vec = np.clip(vec, 0, 1)
        # recursively adjust remaining diff (approx)
        return correct_sum(vec, target)
    else:
        diff = -diff
        idxs = np.where(vec > 0)[0]
        for idx in idxs:
            capacity = vec[idx]
            if capacity >= diff - 1e-12:
                vec[idx] -= diff
                return vec
        dec = diff / len(idxs)
        vec[idxs] -= dec
        vec = np.clip(vec, 0, 1)
        return correct_sum(vec, target)
