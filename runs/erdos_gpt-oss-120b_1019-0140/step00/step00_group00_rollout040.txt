def init_h_from_initial(initial, n_target):
    n0 = len(initial)
    # x positions at centers
    x0 = np.linspace(0, 2, n0, endpoint=False) + (2/n0)/2
    x_target = np.linspace(0, 2, n_target, endpoint=False) + (2/n_target)/2
    h_target = np.interp(x_target, x0, initial)
    # Clip to [0,1]
    h_target = np.clip(h_target, 0, 1)
    # adjust sum to target
    target_sum = n_target / 2
    current_sum = h_target.sum()
    diff = target_sum - current_sum
    h_target += diff / n_target
    # now project to feasible set (box + sum)
    h_target = project_to_feasible(h_target, target_sum)
    return h_target

def project_to_feasible(y, target_sum, tol=1e-12, max_iter=100):
    # find lambda s.t sum(clip(y + lambda, 0, 1)) = target_sum
    lo = -np.max(y) - 1.0  # ensure lower bound yields all <=0
    hi = 1 - np.min(y) + 1.0  # ensure upper bound yields all >=1
    # but easier: we can bsearch between -1 and 1
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = y + mid
        np.clip(x, 0, 1, out=x)
        s = x.sum()
        if np.abs(s - target_sum) < tol:

            hi = mid
        else:
            lo = mid

def gradient_for_shift(h, shift):
    n = len(h)
    # compute term1 = (1 - h shifted left by shift). For i where i+shift >= n, term = 1.
    term1 = np.empty_like(h)
    if shift >= 0:
        term1[:n-shift] = 1 - h[shift:]  # i from 0 to n-1-shift
        term1[n-shift:] = 1.0           # beyond, pad with 1
    else:
        # shift negative: i+shift means i - |shift|

        term1[shift_abs:] = 1 - h[:n-shift_abs] # i >= shift_abs

import time, math
import numpy as np

def project_to_feasible(y: np.ndarray, target_sum: float, tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Project y onto the box [0,1] while enforcing sum(y)=target_sum via a scalar shift."""
    lo = -np.max(y) - 1.0          # guarantees all entries <=0 after shift
    hi = 1.0 - np.min(y) + 1.0      # guarantees all entries >=1 after shift
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    # final clamp
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    return x

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def gradient_for_shift(h: np.ndarray, shift: int, dx: float) -> np.ndarray:
    """Sub‑gradient of the overlap at the given integer shift (maximal shift)."""
    n = h.shape[0]
    if shift >= 0:
        term1 = np.empty(n)
        term1[: n - shift] = 1.0 - h[shift:]
        term1[n - shift :] = 1.0
        term2 = np.empty(n)
        term2[shift:] = h[: n - shift]
        term2[:shift] = 0.0
    else:
        s = -shift
        term1 = np.empty(n)
        term1[s:] = 1.0 - h[: n - s]
        term1[:s] = 1.0
        term2 = np.empty(n)
        term2[: n - s] = h[s:]
        term2[n - s :] = 0.0
    return (term1 - term2) * dx

def upscale_initial(init: np.ndarray, n_target: int) -> np.ndarray:
    """Interpolate a smaller initial vector to a larger grid preserving the integral."""
    n0 = init.shape[0]
    dx0 = 2.0 / n0
    dx1 = 2.0 / n_target
    # centers of the intervals
    x0 = (np.arange(n0) + 0.5) * dx0
    x1 = (np.arange(n_target) + 0.5) * dx1
    h1 = np.interp(x1, x0, init, left=init[0], right=init[-1])
    h1 = np.clip(h1, 0.0, 1.0)
    target = n_target / 2.0
    h1 = project_to_feasible(h1, target)
    return h1

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # resolution (feel free to adjust; 180 gives a good trade‑off)
    n = 180
    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- initial construction ------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        h = upscale_initial(init, n)
    except NameError:
        # fallback: uniform feasible start
        h = np.full(n, 0.5, dtype=float)

    # ensure feasibility
    h = project_to_feasible(h, target_sum)

    cur_cost = compute_c5(h)
    best_h = h.copy()
    best_cost = cur_cost

    # ----- optimization loop ---------------------------------------------------
    max_no_improve = 5000
    no_improve = 0
    step0 = 5.0          # initial step multiplier (effective update ≈ step*dx)
    min_step = 0.001

    while True:
        # time guard
        if time.time() - start >= budget_s - 0.3:
            break

        # evaluate current overlaps
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        idx_max = int(np.argmax(overlaps))
        cost = overlaps[idx_max]

        if cost < best_cost:
            best_cost = cost
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient w.r.t. the worst shift
        shift = idx_max - (n - 1)
        grad = gradient_for_shift(h, shift, dx)

        # step size decreasing with elapsed time
        elapsed = time.time() - start
        t_frac = elapsed / budget_s
        step = step0 * (1.0 - t_frac)
        step = max(step, min_step)

        # propose update
        h_candidate = h - step * grad
        h_candidate = project_to_feasible(h_candidate, target_sum)
        new_cost = compute_c5(h_candidate)

        # accept only improving moves (simple descent)
        if new_cost < cost - 1e-12:
            h = h_candidate
            cur_cost = new_cost
            no_improve = 0
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
        else:
            # occasional random jitter to escape plateaus
            if no_improve >= max_no_improve:
                jitter = (np.random.rand(n) - 0.5) * 0.02
                h = project_to_feasible(h + jitter, target_sum)
                cur_cost = compute_c5(h)
                no_improve = 0

    final_bound = compute_c5(best_h)
    return (best_h.tolist(), float(final_bound), n)
