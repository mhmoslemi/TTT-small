import numpy as np, random, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # Determine n
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = len(init_h)
        # Convert to binary
        h = np.rint(init_h).astype(int)
        target_sum = n // 2
        cur_sum = h.sum()
        if cur_sum > target_sum:
            # turn some 1's to 0
            excess = cur_sum - target_sum
            idx = np.where(h == 1)[0]
            flip = np.random.choice(idx, size=excess, replace=False)
            h[flip] = 0
        elif cur_sum < target_sum:
            deficit = target_sum - cur_sum
            idx = np.where(h == 0)[0]
            flip = np.random.choice(idx, size=deficit, replace=False)
            h[flip] = 1
        # optionally upsample
        factor = 2  # increase resolution
        if factor > 1:
            h = np.repeat(h, factor)
            n = len(h)
    else:
        n = 200
        target_sum = n // 2
        h = np.zeros(n, dtype=int)
        h[:target_sum] = 1
        np.random.shuffle(h)

    # Ensure sum constraint
    assert h.sum() == n // 2

    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    # Simulated annealing parameters
    max_time = budget_s * 0.95
    T0 = 0.005
    iteration = 0
    last_improve_iter = 0
    # main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # Temperature schedule linear decay
        T = T0 * max(0.0, 1.0 - elapsed / max_time)
        # choose random 1 and 0 positions
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        h[i], h[j] = 0, 1
        new_score = compute_c5(h)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and random.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                last_improve_iter = iteration
        else:
            # revert
            h[i], h[j] = 1, 0
        iteration += 1

        # occasional random restart if stuck for long time
        if iteration - last_improve_iter > 20000:
            # random reinitialize preserving constraints
            h = np.zeros(n, dtype=int)
            h[:n//2] = 1
            np.random.shuffle(h)
            cur_score = compute_c5(h)
            last_improve_iter = iteration

    # final best solution
    h_values = best_h.astype(float).tolist()
    c5_bound = best_score
    n_points = n
    return (h_values, c5_bound, n_points)
