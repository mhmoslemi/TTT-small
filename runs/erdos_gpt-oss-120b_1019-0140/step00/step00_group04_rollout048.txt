def project_to_feasible(v, target, max_iter=100, tol=1e-9):
    lo = -np.max(v) - 1  # ensure low enough
    hi = 1 - np.min(v) + 1  # ensure high enough
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2, 0, 1)
```

def project_to_feasible(v, target, tol=1e-9, max_iter=100):
    # v is numpy array
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

import time
import math
import numpy as np

def project_to_feasible(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Project vector v onto the box [0,1]^n with sum(v)=target (Euclidean projection)."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def compute_cost(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def subgradient_optimize(h0: np.ndarray, budget: float, rng: np.random.Generator) -> tuple:
    n = h0.shape[0]
    target = n * 0.5
    best_h = h0.copy()
    best_cost = compute_cost(best_h)
    cur_h = h0.copy()
    cur_cost = best_cost
    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget * 0.99:
            break
        # correlation and lag giving the max
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        # gradient of the active lag (subgradient of max)
        grad = np.zeros_like(cur_h)
        if lag >= 0:
            grad[: n - lag] += 1.0 - cur_h[lag:]
            grad[lag:] -= cur_h[: n - lag]
        else:
            l = -lag
            grad[l:] += 1.0 - cur_h[: n - l]
            grad[: n - l] -= cur_h[l:]
        # step size diminishing
        step0 = 0.02
        step = step0 / math.sqrt(it + 1.0)
        new_h = cur_h - step * grad
        new_h = project_to_feasible(new_h, target)
        new_cost = compute_cost(new_h)
        if new_cost <= cur_cost:
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost:
                best_h = new_h.copy()
                best_cost = new_cost
                if best_cost < 0.3808:
                    break
        it += 1
    return best_h, best_cost

def simulated_annealing_optimize(h0: np.ndarray, budget: float, rng: np.random.Generator) -> tuple:
    n = h0.shape[0]
    target = n * 0.5
    best_h = h0.copy()
    cur_h = h0.copy()
    best_cost = compute_cost(best_h)
    cur_cost = best_cost
    start = time.time()
    max_iter_est = 200000
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget * 0.99:
            break
        frac = min(it / max_iter_est, 1.0)
        T = 0.01 * (1.0 - frac) + 1e-8 * frac
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # random direction
        if rng.random() < 0.5:
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                it += 1
                continue
            delta = rng.random() * max_move
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_move = min(cur_h[j], 1.0 - cur_h[i])
            if max_move <= 0.0:
                it += 1
                continue
            delta = rng.random() * max_move
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        new_cost = compute_cost(new_h)
        dcost = new_cost - cur_cost
        if dcost < 0.0 or rng.random() < math.exp(-dcost / (T + 1e-12)):
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost:
                best_h = new_h.copy()
                best_cost = new_cost
                if best_cost < 0.3808:
                    break
        it += 1
    return best_h, best_cost

def optimize_h(initial_h: np.ndarray, budget: float, rng: np.random.Generator) -> tuple:
    """Run a short subgradient phase followed by simulated annealing."""
    # split budget roughly 40% subgradient, 60% SA
    sub_budget = budget * 0.4
    sa_budget = budget - sub_budget
    h_sub, cost_sub = subgradient_optimize(initial_h, sub_budget, rng)
    h_sa, cost_sa = simulated_annealing_optimize(h_sub, sa_budget, rng)
    if cost_sa < cost_sub:
        return h_sa, cost_sa
    return h_sub, cost_sub
