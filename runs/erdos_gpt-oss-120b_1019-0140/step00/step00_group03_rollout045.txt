import numpy as np, time, random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def simulated_annealing(n, seed, time_budget):
    rng = np.random.default_rng(seed)
    # initial binary vector
    h = np.zeros(n, dtype=np.float64)
    # maybe start from initial_h_values if provided
    if 'initial_h_values' in globals():
        init = initial_h_values
        # project: round to binary preserve sum
        # initial may not have exact half sum; we can order by value and set top half to 1.
        order = np.argsort(-init)  # descending
        h[:] = 0
        h[order[:n//2]] = 1.0
    else:
        # random
        indices = rng.choice(n, size=n//2, replace=False)
        h[indices] = 1.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    elapsed = 0

    # SA parameters
    max_iter = 10**9  # large; will break on time
    T0 = 0.1  # initial temperature (scale of objective diff)
    T_min = 1e-4
    # schedule: exponential cooling
    # but we can approximate T = T0 * (1 - elapsed / time_budget)
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.1:  # leave margin
            break
        # compute temperature based on elapsed proportion
        T = max(T_min, T0 * (1 - elapsed / time_budget))
        # propose move: swap a random 1 and 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # redeploy random restart if degenerate
            h[:] = 0
            indices = rng.choice(n, size=n//2, replace=False)
            h[indices] = 1
            cur_score = compute_c5(h)
            continue
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # reject, revert swap
            h[i], h[j] = h[j], h[i]
        # maybe occasional random restart if stagnant
        # If no improvement within N steps, reinitialize
        # We can track progress.

    return best_h, best_score

    dx = 2.0 / n

    # initialize h binary

    best_h = h.copy()

    cur_score = best_score

    start_time = time.time()

    steps_without_improve = 0
    restart_threshold = 20000

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_budget - 0.05:
            break
        # compute temperature
        T = max(0.001, 0.1 * (1 - elapsed / time_budget))
        # randomly pick swap indices
        # get positions of ones and zeros
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

            idx = rng.choice(n, size=n//2, replace=False)
            h[idx] = 1.0
            cur_score = compute_c5(h)
            continue
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)

        # acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):

import time
import math
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer lags
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initialize_h(n, rng):
    """Create a binary start vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _binary_from_initial(init, n):
    """Convert an arbitrary initial vector to a binary one with n/2 ones."""
    # take the indices of the largest n/2 entries
    order = np.argsort(-np.asarray(init, dtype=np.float64))
    h = np.zeros(n, dtype=np.float64)
    h[order[: n // 2]] = 1.0
    return h

def _simulated_annealing(n, seed, time_budget):
    rng = np.random.default_rng(seed)

    # ----- initial vector -----
    try:
        init = initial_h_values  # pre‑imported if present
        h = _binary_from_initial(init, n)
    except NameError:
        h = _initialize_h(n, rng)

    # maintain index lists for fast swaps
    ones_idx = [i for i, v in enumerate(h) if v > 0.5]
    zeros_idx = [i for i, v in enumerate(h) if v <= 0.5]

    best_h = h.copy()
    best_score = _compute_c5(best_h)
    cur_score = best_score

    start = time.time()
    no_improve = 0
    RESTART_AFTER = 30000          # steps without improvement before a restart
    T0 = 0.5                        # initial temperature
    T_min = 1e-4

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget - 0.05:  # small safety margin
            break

        # temperature schedule (linear decay)
        T = max(T_min, T0 * (1.0 - elapsed / time_budget))

        # ----- propose a swap -----
        i_pos = rng.integers(len(ones_idx))
        j_pos = rng.integers(len(zeros_idx))
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]

        # perform the swap (binary)
        h[i] = 0.0
        h[j] = 1.0
        ones_idx[i_pos] = j
        zeros_idx[j_pos] = i

        new_score = _compute_c5(h)

        # ----- acceptance test -----
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones_idx[i_pos] = i
            zeros_idx[j_pos] = j
            no_improve += 1

        # ----- occasional restart to escape stagnation -----
        if no_improve >= RESTART_AFTER:
            h = _initialize_h(n, rng)
            ones_idx = [i for i, v in enumerate(h) if v > 0.5]
            zeros_idx = [i for i, v in enumerate(h) if v <= 0.5]
            cur_score = _compute_c5(h)
            no_improve = 0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős minimum overlap problem.

    Returns
    -------
    h_values : list of float
        Optimised discretised values (length n_points, each in [0,1]).

    # Run the optimiser within the allotted time budget

    # Final bound (use the pre‑imported evaluator if it exists)
    try:
