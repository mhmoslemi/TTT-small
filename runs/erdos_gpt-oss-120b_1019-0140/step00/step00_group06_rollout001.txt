            while j == i:
                j = rng.integers(0, n)
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0:
                iter_count += 1
                continue
            delta = rng.random() * max_delta
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        np.clip(new_h, 0.0, 1.0, out=new_h)
        new_c5, _ = compute_c5_and_corr(new_h)

        # accept if not worse, otherwise Metropolis step
        if new_c5 <= c5_val + 1e-12:
            h = new_h
            c5_val = new_c5
            if new_c5 < best_val - 1e-12:
                best_val = new_c5
                best_h = new_h.copy()
        else:
            T = 0.01 * (1.0 - iter_count / max_iter)
            if T > 0 and rng.random() < math.exp(-(new_c5 - c5_val) / T):
                h = new_h
                c5_val = new_c5

        # occasional stochastic perturbation of the current best
        if iter_count % 20000 == 0 and iter_count > 0:
            perturb = rng.normal(scale=0.02, size=n)
            candidate = best_h + perturb
            candidate = np.clip(candidate, 0.0, 1.0)
            candidate = project_to_constraints(candidate, target)
            h = candidate

        iter_count += 1

    # final projection of the best vector
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_constraints(best_h, target)
    best_val, _ = compute_c5_and_corr(best_h)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a list
    of samples of the step function on [0,2].
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # obtain a starting construction if available
    try:
        base_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        # fallback: simple binary half‑ones vector
        n0 = 120
        base_h = np.zeros(n0, dtype=float)
        base_h[: n0 // 2] = 1.0
        rng.shuffle(base_h)

    n0 = base_h.size
    target0 = n0 / 2.0
    # ensure feasibility of the base vector
    base_h = np.clip(base_h, 0.0, 1.0)
    base_h = project_to_constraints(base_h, target0)

    # We'll try two resolutions: original and doubled size.
    candidates = []
    # 1. Original size
    candidates.append((base_h, n0, budget_s * 0.55))

    # 2. Doubled size (interpolated)
    n1 = n0 * 2
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n1, endpoint=False)
    h_up = np.interp(x_new, x_old, base_h)
    h_up = project_to_constraints(h_up, n1 / 2.0)
    candidates.append((h_up, n1, budget_s * 0.45))

    best_overall_val = float('inf')
    best_overall_h = None
    best_overall_n = None

    # run optimisation on each candidate within its allotted time
    for h0, n_pts, alloc in candidates:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        remaining = max(0.0, budget_s - elapsed)
        use_budget = min(alloc, remaining)
        h_opt, val = optimize_h(h0, use_budget, rng)
        if val < best_overall_val:
            best_overall_val = val
            best_overall_h = h_opt.copy()
            best_overall_n = n_pts

    # final safety: ensure returned vector satisfies constraints exactly
    if best_overall_h is None:
        # fallback to base vector
        best_overall_h = base_h.copy()
        best_overall_n = n0
        best_overall_val, _ = compute_c5_and_corr(best_overall_h)

    best_overall_h = np.clip(best_overall_h, 0.0, 1.0)
    best_overall_h = project_to_constraints(best_overall_h, best_overall_n / 2.0)
    best_overall_val, _ = compute_c5_and_corr(best_overall_h)

    return best_overall_h.tolist(), float(best_overall_val), int(best_overall_n)
