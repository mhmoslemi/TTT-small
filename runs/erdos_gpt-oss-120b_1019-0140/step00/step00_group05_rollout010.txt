import numpy as np, time, math

def compute_c5(h):
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

def normalize_h(h, target_sum):
    # ensures sum(h) == target_sum, clip [0,1]
    total = np.sum(h)
    if total == 0:
        return h
    # scale to meet target sum while preserving clip? Usually we need binary, but do scaling if needed
    # We'll just scale: h = h * (target_sum/total)
    h = h * (target_sum / total)
    np.clip(h, 0, 1, out=h)
    # recompute sum, may not be exact; adjust by adding small amounts to first elements
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-8:
        # distribute diff across first entries
        for i in range(len(h)):
            if h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-9:
                    break
        # if still diff>0 (unlikely), add to last elements
        for i in range(len(h)-1, -1, -1):
            if h[i] > 0:
                sub = min(diff, h[i])
                h[i] -= sub
                diff -= sub
                if abs(diff) < 1e-9:
                    break
    return h

def random_binary_vector(n, rng):
    # exactly n/2 ones
    ones = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h

def neighbor_swap_binary(h, rng):
    # h binary (0 or 1)
    n = len(h)
    # get indices of 1 and 0
    ones = np.flatnonzero(h > 0.5)   # treat as 1
    zeros = np.flatnonzero(h < 0.5)
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h

def neighbor_swap_continuous(h, rng, step=0.1):
    # small perturbation: choose index i to increase, index j to decrease
    n = len(h)
    # pick random i, j
    i, j = rng.choice(n, size=2, replace=False)
    # adjust amount to keep bounds
    amount = rng.uniform(0, step)
    hi = h[i]
    hj = h[j]
    # Ensure we can move amount from i to j (or reverse)
    if hi + amount > 1 or hj - amount < 0:
        # try opposite direction
        amount = min(hi, 1 - hj)
    new_h = h.copy()
    new_h[i] = hi - amount
    new_h[j] = hj + amount
    # Clip for safety
    np.clip(new_h, 0, 1, out=new_h)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.RandomState(seed)
    # Determine n_points
    try:
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
    except NameError:
        n = 100  # default
        init = None
    target_sum = n / 2.0  # sum of h_i needed
    # Initialize h
    if init is not None and len(init) == n:
        h_current = init.copy()
        # Ensure sum constraint
        current_sum = np.sum(h_current)
        if abs(current_sum - target_sum) > 1e-8:
            # Normalize to meet sum
            h_current = normalize_h(h_current, target_sum)
        # Clip to [0,1]
        np.clip(h_current, 0, 1, out=h_current)
    else:
        h_current = random_binary_vector(n, rng)  # binary start
    # Compute initial score
    best_h = h_current.copy()
    best_score = compute_c5(best_h)
    curr_h = h_current.copy()
    curr_score = best_score
    start_time = time.time()
    # Simulated annealing loop
    # Parameter settings
    max_time = budget_s - 0.5  # leave small slack
    # Temperature schedule: start high, end low
    T_start = 0.1
    T_end = 0.001
    # Use integer iteration limit to avoid infinite loop
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        iteration += 1
        # Determine temperature linearly decreasing
        t_frac = elapsed / max_time
        T = T_start * (1 - t_frac) + T_end * t_frac
        # propose neighbor
        # decide if binary or continuous based on h values
        if np.all((curr_h == 0) | (curr_h == 1)):
            # binary
            new_h = neighbor_swap_binary(curr_h, rng)
        else:
            # continuous small swap
            new_h = neighbor_swap_continuous(curr_h, rng, step=0.2)
        # Ensure sum constraint (swap maintains sum)
        # Evaluate
        new_score = compute_c5(new_h)
        # Acceptance
        if new_score < curr_score or rng.rand() < math.exp(-(new_score - curr_score) / max(T, 1e-12)):
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score:

                best_score = new_score
        # Occasionally restart from best to escape stagnation
        if iteration % 2000 == 0:
            # random perturbation around best
            if np.all((best_h == 0) | (best_h == 1)):
                # do a random swap chain of 2-5 swaps
                chance = rng.rand()
                if chance < 0.5:
                    # simple random swap
                    curr_h = neighbor_swap_binary(best_h, rng)
                else:
                    # multiple swaps
                    temp_h = best_h.copy()
                    for _ in range(rng.randint(1, 4)):
                        temp_h = neighbor_swap_binary(temp_h, rng)
                    curr_h = temp_h
            else:
                # continuous perturb
                curr_h = best_h.copy()
                for _ in range(rng.randint(1, 3)):
                    curr_h = neighbor_swap_continuous(curr_h, rng, step=0.2)
            curr_score = compute_c5(curr_h)
    # Final best result
    # Ensure final vector meets constraints exactly: clip, adjust sum
    np.clip(best_h, 0, 1, out=best_h)
    best_h = normalize_h(best_h, target_sum)
    # Compute final score
    best_score = compute_c5(best_h)
    return (best_h, float(best_score), int(n))
