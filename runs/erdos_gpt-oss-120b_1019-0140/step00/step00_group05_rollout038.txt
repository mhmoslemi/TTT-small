    import numpy as np, time, math, random
    from math import exp
    # Use global initial_h_values if exists.
    try:
        init = initial_h_values
    except NameError:
        init = None
    rng = np.random.default_rng(seed)
    # Determine n_points
    if init is not None:
        N = len(init)
    else:
        N = 70  # default
    M = N // 2
    dx = 2.0 / N
    # Define function to convert array to binary sum M
    def to_binary(arr):
        idx = np.argpartition(arr, -M)[-M:]  # indices of top M entries
        bin_vec = np.zeros(N, dtype=float)
        bin_vec[idx] = 1.0
        return bin_vec
    # Starting candidate
    if init is not None:
        h_curr = to_binary(init)
    else:
        # random binary with M ones
        h_curr = np.zeros(N, dtype=float)
        ones = rng.choice(N, size=M, replace=False)
        h_curr[ones] = 1.0
    best_h = h_curr.copy()
    # evaluate function
    def eval_h(h):
        # ensure constraints length sum: but we maintain sum exactly M.
        # Use evaluate_erdos_solution if available
        try:
            return evaluate_erdos_solution(h)
        except NameError:
            # fallback manual compute
            # correlation = np.correlate(h, 1-h, mode='full') * dx
            # return np.max(correlation)
            corr = np.correlate(h, 1 - h, mode='full') * dx
            return float(np.max(corr))
    best_score = eval_h(best_h)
    # current score
    curr_score = best_score
    # time
    start = time.time()
    # Temperature schedule:
    T0 = 0.5
    T1 = 0.001
    # loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # linear temperature
        frac = elapsed / (budget_s * 0.95)
        T = T0 * (1 - frac) + T1 * frac
        # random swap: choose a random 1 and 0 positions
        ones_idx = np.where(h_curr == 1)[0]
        zeros_idx = np.where(h_curr == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            # random reinitialize
            h_curr = np.zeros(N, dtype=float)
            h_curr[rng.choice(N, size=M, replace=False)] = 1.0
            curr_score = eval_h(h_curr)
            continue
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_new = h_curr.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = eval_h(h_new)
        # lower is better
        delta = new_score - curr_score
        accept = False

            if T > 0:

        if accept:

import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that gives a low
    Erdős overlap constant C5 = max_k ∫ h(x)*(1-h(x+k)) dx.
    The function returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----- Determine problem size -----
    try:
        init_vec = initial_h_values  # global provided by the harness
    except NameError:
        init_vec = None

    if init_vec is not None:
        n_points = int(len(init_vec))
    else:
        n_points = 70                     # default size if no parent is given
    half = n_points // 2                 # sum(h) must equal half (since dx = 2/n)
    dx = 2.0 / n_points

    # ----- Helper to evaluate C5 -----
    if "evaluate_erdos_solution" in globals():
        def c5_of(h):
            # evaluate_erdos_solution is expected to return the C5 value
            return float(evaluate_erdos_solution(h))
    else:
        # fallback definition matching the problem statement
        def c5_of(h):
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return float(np.max(corr))

    # ----- Build a binary start vector (respecting the sum constraint) -----
    def binary_from_values(vals):
        """Return a binary vector with exactly `half` ones (largest entries set to 1)."""
        idx = np.argpartition(vals, -half)[-half:]
        h = np.zeros(n_points, dtype=float)
        h[idx] = 1.0
        return h

    if init_vec is not None:
        cur_h = binary_from_values(np.asarray(init_vec))
    else:
        cur_h = np.zeros(n_points, dtype=float)
        cur_h[rng.choice(n_points, size=half, replace=False)] = 1.0

    # ----- Initial evaluation -----
    cur_score = c5_of(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    TARGET = 0.3808  # desired bound; stop early if reached

    # ----- Simulated annealing / hill‑climbing loop -----
    start_time = time.time()
    time_limit = budget_s * 0.95   # keep a safety margin

    # temperature schedule (linear cooling)
    T_start = 0.12
    T_end = 1e-4
    iter_cnt = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # update temperature
        frac = elapsed / time_limit
        T = T_start * (1 - frac) + T_end * frac

        # pick a random 1↔0 swap preserving the cardinality
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # degenerate situation – re‑initialise randomly
            cur_h = np.zeros(n_points, dtype=float)
            cur_h[rng.choice(n_points, size=half, replace=False)] = 1.0
            cur_score = c5_of(cur_h)
            continue

        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand_h = cur_h.copy()
        cand_h[i] = 0.0
        cand_h[j] = 1.0

        cand_score = c5_of(cand_h)
        delta = cand_score - cur_score

        if delta < 0:                     # improvement
            cur_h, cur_score = cand_h, cand_score
            if cand_score < best_score:
                best_h, best_score = cand_h.copy(), cand_score
                if best_score <= TARGET:
                    break
        else:
            # occasional uphill move (Metropolis criterion)
            if T > 0 and rng.random() < math.exp(-delta / T):
                cur_h, cur_score = cand_h, cand_score

        iter_cnt += 1

        # every few thousand iterations try a small “hyper‑mutation”
        if iter_cnt % 5000 == 0:
            if rng.random() < 0.03:   # 3 % chance of a broader jump
                swaps = max(1, int(0.07 * half))
                for _ in range(swaps):
                    ones = np.where(cur_h == 1)[0]
                    zeros = np.where(cur_h == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    cur_h[i] = 0.0
                    cur_h[j] = 1.0
                cur_score = c5_of(cur_h)
                if cur_score < best_score:
                    best_h, best_score = cur_h.copy(), cur_score
                    if best_score <= TARGET:
                        break

        # occasional full random restart to escape deep local minima
        if iter_cnt % 20000 == 0:

                cur_h[rng.choice(n_points, size=half, replace=False)] = 1.0
                cur_score = c5_of(cur_h)

    # ----- Final sanity check and return -----
    # Ensure binary and exact sum constraint (tiny floating noise may appear)
    best_h = np.where(best_h >= 0.5, 1.0, 0.0)
    # Re‑evaluate to guarantee consistency
    final_score = c5_of(best_h)

    return (best_h.astype(float), float(final_score), n_points)
