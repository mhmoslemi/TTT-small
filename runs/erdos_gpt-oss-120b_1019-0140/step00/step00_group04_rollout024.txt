def correlation_full(a, b):
    n = a.shape[0]
    # length of full convolution: n + n - 1 = 2n-1
    L = 1 << ((2*n-1).bit_length())  # next power of 2 for FFT speed
    # compute FFT of a_rev and b with zero padding length L
    a_rev = a[::-1]
    A = np.fft.rfft(np.concatenate([a_rev, np.zeros(L - n)]))
    B = np.fft.rfft(np.concatenate([b, np.zeros(L - n)]))
    conv = np.fft.irfft(A * B, n=L)
    # conv has length L, first (2n-1) entries are the full conv result
    result = conv[:2*n-1]
    return result

def compute_c5(h):
    # h is np.ndarray float; we assume values in [0,1]; we compute correlation.
    # Use FFT.
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    # compute H_rev * (1-H)
    a_rev = h[::-1]
    b = 1 - h
    A = np.fft.rfft(np.concatenate([a_rev, np.zeros(L - n)]))
    B = np.fft.rfft(np.concatenate([b, np.zeros(L - n)]))
    conv = np.fft.irfft(A*B, n=L)
    corr = conv[:2*n-1]
    c5 = np.max(corr) * (2.0 / n)
    return c5

Function compute_c5(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    a_rev = h[::-1]
    b = 1.0 - h
    # combine real arrays; rfft expects real input.
    A = np.fft.rfft(np.concatenate((a_rev, np.zeros(L - n))))
    B = np.fft.rfft(np.concatenate((b, np.zeros(L - n))))
    conv = np.fft.irfft(A * B, n=L)
    # conv is length L, we need first 2n-1 entries
    corr = conv[:2*n - 1]
    return np.max(corr) * (2.0 / n)

import time, math, random
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx using FFT‑based correlation."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2n‑1
    a_rev = h[::-1]
    b = 1.0 - h
    pad = L - n
    A = np.fft.rfft(np.concatenate((a_rev, np.zeros(pad, dtype=np.float64))))
    B = np.fft.rfft(np.concatenate((b,   np.zeros(pad, dtype=np.float64))))
    conv = np.fft.irfft(A * B, n=L)
    corr = conv[: 2 * n - 1]                  # full linear correlation
    return np.max(corr) * (2.0 / n)           # dx = 2/n

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
        # enforce sum = n/2 (tiny correction if needed)
        target = n / 2.0
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            # add/subtract diff on a free coordinate
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-12)[0]
                if idx.size:
                    i = idx[0]
                    h[i] = min(1.0, h[i] + diff)
            else:
                idx = np.where(h > 1e-12)[0]
                if idx.size:
                    i = idx[0]
                    h[i] = max(0.0, h[i] + diff)
    else:
        n = 256
        h = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    # ensure hard bounds
    np.clip(h, 0.0, 1.0, out=h)

    # ------------------------------------------------------------------
    # simulated annealing on two‑coordinate moves preserving the total mass
    cur_score = _max_overlap(h)
    best_h = h.copy()
    best_score = cur_score

    T = 0.5                     # initial temperature
    decay = 0.99994             # exponential cooling factor
    it = 0

    while True:
        if time.time() - start_time > budget_s * 0.94:

        # pick two distinct indices

        delta_max =  min(1.0 - hi, hj)
        if delta_max <= delta_min:

        delta = delta_min + np.random.rand() * (delta_max - delta_min)

            h[j] = hj

        it += 1

    while time.time() - start_time < budget_s * 0.98:

                    if new_score < best_score:
                        best_score = new_score
