            continue

        max_inc = 1.0 - current_h[i_up]

        # Choose random delta up to delta_bound

    return (best_h, best_score, n)

import time, math
import numpy as np

def _c5_score(h):
    """Maximum overlap integral for a given h (discretized)."""
    dx = 2.0 / h.size
    # correlation with zero padding (mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n = 70
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    n = h.size
    target_sum = n / 2.0

    # ensure exact feasibility (clip, then fix any tiny drift)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the small difference to a non‑boundary entry
        for i in range(n):
            if 0.0 < h[i] < 1.0:
                h[i] += diff
                break

    cur = h
    cur_score = _c5_score(cur)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # simulated‑annealing parameters
    # ------------------------------------------------------------------
    temperature = 1.0          # initial temperature
    decay = 0.9995             # geometric cooling
    min_temp = 1e-6
    max_delta = 0.5            # ceiling on transferred mass per move
    last_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while True:
        if time.time() - start > budget_s * 0.98:
            break
        it += 1

        # pick two distinct indices: one to increase, one to decrease
        i_up, i_down = rng.integers(0, n, size=2)
        if i_up == i_down:
            continue
        if cur[i_up] >= 1.0 or cur[i_down] <= 0.0:
            # try to find a feasible pair quickly
            for _ in range(5):
                i_up, i_down = rng.integers(0, n, size=2)
                if i_up != i_down and cur[i_up] < 1.0 and cur[i_down] > 0.0:
                    break
            else:
                continue

        # bound on how much we may move without breaking constraints
        max_inc = 1.0 - cur[i_up]
        max_dec = cur[i_down]
        bound = min(max_inc, max_dec, max_delta)
        if bound <= 0.0:
            continue

        # propose a transfer delta (scaled with temperature)
        delta = rng.random() * bound * (temperature / (temperature + 1.0))

        new = cur.copy()
        new[i_up] += delta
        new[i_down] -= delta
        new_score = _c5_score(new)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / temperature)
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                last_improve = it
                # early exit if already better than target bound
                if best_score <= 0.3808:
                    break

        # cool down
        temperature = max(temperature * decay, min_temp)

        # occasional restart if stagnating for a long time
        if it - last_improve > 50000 and temperature < 1e-3:
            restart = np.zeros(n, dtype=np.float64)
            idx = rng.choice(n, size=n // 2, replace=False)
            restart[idx] = 1.0
            cur = restart
            cur_score = _c5_score(cur)
            temperature = 1.0
            last_improve = it

    return (best, best_score, n)
