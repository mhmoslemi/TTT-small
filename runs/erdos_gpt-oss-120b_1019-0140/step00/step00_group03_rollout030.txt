import time
import numpy as np
import cvxpy as cp

def _c5_of(h):
    """Compute C5 for a binary vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    init = globals().get("initial_h_values", None)
    N = int(len(init)) if init is not None else 70
    half = N // 2  # target number of ones (integral = 1)

    # ------------------------------------------------------------------
    # Try a MILP formulation (binary variables, exact optimality)
    # ------------------------------------------------------------------
    try:
        h = cp.Variable(N, boolean=True)
        P = cp.Variable((N, N), boolean=True)       # P[i,j] = h[i] * h[j]
        s = cp.Variable(integer=True)               # max overlap count (not scaled)

        cons = [cp.sum(h) == half, s >= 0]

        # Linearization of the product h[i]*h[j] via standard McCormick constraints
        cons += [P <= cp.reshape(h, (N, 1))]
        cons += [P <= cp.reshape(h, (1, N))]
        cons += [P >= cp.reshape(h, (N, 1)) + cp.reshape(h, (1, N)) - 1]

        # Constraints that bound each shift's overlap count by s
        for k in range(-(N - 1), N):
            if k == 0:
                continue
            i0 = max(0, -k)
            i1 = min(N, N - k)
            if i0 >= i1:
                continue
            rows = slice(i0, i1)
            cols = slice(i0 + k, i1 + k)
            term = cp.sum(h[rows]) - cp.sum(P[rows, cols])
            cons.append(term <= s)

        prob = cp.Problem(cp.Minimize(s), cons)

        # Solve with CBC; if unavailable, fall back to GLPK_MI
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)

        # Extract solution if successful
        if h.value is not None:
            h_vec = np.rint(h.value).astype(int)
            # Repair possible rounding drift
            cur_sum = int(h_vec.sum())
            if cur_sum != half:
                diff = half - cur_sum
                if diff > 0:
                    zeros = np.where(h_vec == 0)[0]
                    h_vec[zeros[:diff]] = 1
                else:
                    ones = np.where(h_vec == 1)[0]
                    h_vec[ones[: -diff]] = 0
            c5 = _c5_of(h_vec)
            # Return immediately if within time
            if time.time() - start_time < budget_s:
                return (h_vec, c5, N)
    except Exception:
        # If any error occurs, silently fall back to heuristic search
        pass

    # ------------------------------------------------------------------
    # Heuristic fallback: Simulated annealing / hill climbing on binary vectors
    # ------------------------------------------------------------------
    # Initialise a feasible binary vector (half ones)
    if init is not None:
        flat = np.asarray(init).flatten()
        # Select the half largest entries to be 1
        idx = np.argpartition(flat, -half)[-half:]
        cur = np.zeros(N, dtype=int)
        cur[idx] = 1
    else:
        cur = np.zeros(N, dtype=int)
        cur[:half] = 1
        rng.shuffle(cur)

    best = cur.copy()
    best_score = _c5_of(best)
    cur_score = best_score

    # SA parameters
    T = 0.08
    decay = 0.9995
    no_improve = 0
    max_no_improve = 30000

    while time.time() - start_time < budget_s - 0.5:
        # propose a swap of a 1 and a 0
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        new = cur.copy()
        new[i] = 0
        new[j] = 1
        new_score = _c5_of(new)

        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= decay

        # Random restart if stagnating
        if no_improve > max_no_improve:
            cur = np.zeros(N, dtype=int)
            cur[:half] = 1
            rng.shuffle(cur)
            cur_score = _c5_of(cur)
            no_improve = 0
            T = 0.08

    return (best, best_score, N)
