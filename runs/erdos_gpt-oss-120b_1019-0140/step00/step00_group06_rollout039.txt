def project_to_constraints(v):
    n = len(v)
    h = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h - v))
    constraints = [cp.sum(h) == n/2, h >= 0, h <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True) # set appropriate solver
    return h.value

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / len(h)
    return max_corr * dx

import time
import math
import numpy as np

# Optional cvxpy for projection onto the feasible set
try:
    import cvxpy as cp
except Exception:
    cp = None


def _project_feasible(v, n):
    """
    Project vector v onto the set {h | 0 <= h_i <= 1, sum(h) = n/2 }.
    Uses cvxpy if available, otherwise a simple iterative clipping.
    """
    target = n / 2.0
    if cp is not None:
        h = cp.Variable(n)
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h - v)),
                          [cp.sum(h) == target,
                           h >= 0,
                           h <= 1])
        prob.solve(solver=cp.ECOS, warm_start=True)
        if h.value is not None:
            return np.clip(np.asarray(h.value).astype(np.float64), 0.0, 1.0)
    # Simple fallback: clip and rebalance linearly
    h = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    # Final tiny fix on first element if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def _c5_value(h):
    """
    Compute the Erdős C5 bound for a step function represented by vector h.
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _active_shifts(corr, n, atol=1e-12):
    """
    Return the list of integer shifts (relative to zero) where the correlation
    attains its maximum, within absolute tolerance atol.
    """
    max_val = np.max(corr)
    idxs = np.where(corr >= max_val - atol)[0]
    # shift = index - (n-1)
    return idxs - (n - 1)


def _neighbor_sum(h, shifts):
    """
    For each index i, compute the sum of h[i+s] over all active shifts s,
    treating out‑of‑range accesses as zero.
    """
    n = len(h)
    w = np.zeros(n, dtype=np.float64)
    for s in shifts:
        if s == 0:
            w += 2.0 * h
        elif s > 0:
            w[: n - s] += h[s:]
            w[s:] += h[: n - s]
        else:  # s < 0
            sp = -s
            w[sp:] += h[: n - sp]
            w[: n - sp] += h[sp:]
    return w


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.

    Returns
    -------
    h_values : np.ndarray
        Optimised values on [0,2].
    c5_bound : float
        Corresponding C5 bound.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    try:
        # Use the pre‑imported parent construction if it exists
        init_h = initial_h_values.copy()
        n = len(init_h)
    except NameError:
        # No parent provided – start from a feasible random vector
        n = 96                     # even size makes the sum integer
        rand = rng.random(n)
        init_h = _project_feasible(rand, n)

    # Ensure feasibility (tiny numerical drifts may appear)
    init_h = _project_feasible(init_h, n)

    best_h = init_h.astype(np.float64)
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # Parameters for the stochastic search
    # ------------------------------------------------------------------
    max_delta = 0.25                     # maximum mass transfer per move
    start_temp = 0.02                    # initial temperature (scaled)
    total_time = budget_s * 0.95         # keep a safety margin
    start_clock = time.time()
    iteration = 0
    no_improve = 0

    while True:
        elapsed = time.time() - start_clock
        if elapsed >= total_time:
            break

        # Cooling schedule (linear)
        temp = max(start_temp * (1.0 - elapsed / total_time), 1e-6)

        # Current correlation and worst shift(s)
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        active = _active_shifts(corr, n)

        # Gradient‑like neighbor‑sum
        w = _neighbor_sum(best_h, active)

        # Candidate indices for increase / decrease
        inc_candidates = np.where(best_h < 1.0 - 1e-12)[0]
        dec_candidates = np.where(best_h > 1e-12)[0]

        if inc_candidates.size == 0 or dec_candidates.size == 0:
            # Fallback: random transfer between any two indices
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(max_delta, best_h[i], 1.0 - best_h[j])
            if delta > 0:
                new_h = best_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                new_h = best_h.copy()
        else:
            # Choose the most favorable up‑ and down‑ directions
            inc_idx = inc_candidates[np.argmax(w[inc_candidates])]
            dec_idx = dec_candidates[np.argmin(w[dec_candidates])]

            # In the rare case they coincide, pick random distinct indices
            if inc_idx == dec_idx:
                inc_idx = inc_candidates[rng.integers(len(inc_candidates))]
                dec_idx = dec_candidates[rng.integers(len(dec_candidates))]
                if inc_idx == dec_idx:
                    i, j = rng.choice(n, size=2, replace=False)
                    delta = min(max_delta, best_h[i], 1.0 - best_h[j])

                    if delta > 0:

                        new_h[j] += delta
                    else:
                        new_h = best_h.copy()
                else:

                    new_h = best_h.copy()
                    new_h[dec_idx] -= delta
                    new_h[inc_idx] += delta

                        new_h[j] += delta
                    else:
                        new_h = best_h.copy()
                else:

                    new_h = best_h.copy()
                    new_h[dec_idx] -= delta
                    new_h[inc_idx] += delta

        # Tiny random perturbation now and then to avoid stagnation
        if rng.random() < 0.01:
            pert_idxs = rng.choice(n, size=min(3, n), replace=False)
            noise = (rng.random(len(pert_idxs)) - 0.5) * 0.04
            new_h[pert_idxs] += noise
            np.clip(new_h, 0.0, 1.0, out=new_h)
            # Repair the sum constraint by adjusting the first element
            diff = (n / 2.0) - new_h.sum()
            new_h[0] = np.clip(new_h[0] + diff, 0.0, 1.0)

        # Evaluate candidate

            best_score = cand_score
            no_improve = 0

        iteration += 1

        # Periodic full re‑projection to eliminate drift
        if iteration % 200 == 0:

    # Final safeguard projection
    best_h = _project_feasible(best_h, n)
    final_score = _c5_value(best_h)

    return best_h, float(final_score), int(n)
