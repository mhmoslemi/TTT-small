import time
import numpy as np

def compute_c5(vec, dx):
    """Maximum overlap integral for a sampled step function."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def random_binary(n, rng):
    """Uniform random 0/1 vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    arr[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return arr

def project_to_sum_bounds(vec, target):
    """
    Euclidean projection of vec onto the box [0,1]^len(vec) intersect
    {x : sum(x) = target}. Uses a simple water‑filling bisection.
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(vec - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)
