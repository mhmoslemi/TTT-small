import numpy as np, time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def compute_J(h):
    N = len(h)
    dx = 2.0 / N
    conv = np.correlate(h, h, mode='full') * dx
    return conv

def project_sum_bounds(v, target):
    # clip to bounds and project to target sum using water filling
    v = np.clip(v, 0.0, 1.0)
    # tolerance
    tol = 1e-12
    for _ in range(100):
        s = v.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        # free indices not at bound
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds -> adjust some belonging to bound to meet target using tolerance
            # distribute uniformly among all entries
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            break
        delta = diff / n_free
        # apply delta only to free entries
        v[free] += delta
        # clip again, may move some to bounds
        v = np.clip(v, 0.0, 1.0)

def gradient_improve(h_initial, time_limit):
    N = len(h_initial)
    target_sum = N / 2.0
    h = h_initial.copy()
    dx = 2.0 / N
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    iter = 0
    # parameters
    max_step = 0.02
    # main loop
    while True:
        now = time.time()
        if now - start > time_limit:
            break
        # compute auto-correlation J_k
        conv = np.correlate(h, h, mode='full') * dx
        min_J = conv.min()
        current_score = 1 - min_J
        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
        # find shift with min J
        idx = np.argmin(conv)
        shift = idx - (N - 1)
        shift_abs = abs(shift)
        if shift_abs == 0:
            # cannot improve further via gradient?
            # break maybe after some iterations
            # attempt random perturbation to maybe move off plateau
            # but we can break
            # Let's perform random perturbation and continue
            # Not break because we may still get improvement for other shifts?
            # Continue with random perturb
            # We'll do a small random mutation occasionally
            # We'll skip gradient update in this iteration
            pert = (rng.uniform(-0.05, 0.05, size=N))
            h_new = h + pert
            h_new = project_sum_bounds(h_new, target_sum)
            h = h_new
            continue
        # compute gradient
        g = np.zeros_like(h)
        # vectorized using slicing
        if shift_abs < N:
            # contribution from forward
            g[:N - shift_abs] += h[shift_abs:]
            # contribution from backward
            g[shift_abs:] += h[:N - shift_abs]
        # else shift_abs >= N => no overlap, gradient zero
        # Determine step size
        max_g = np.max(g)
        if max_g == 0:
            # cannot improve
            # random perturbation
            pert = (rng.uniform(-0.05, 0.05, size=N))
            h_new = h + pert
            h_new = project_sum_bounds(h_new, target_sum)
            h = h_new
            continue
        step = max_step / (max_g + 1e-12)
        # Update
        h_new = h + step * g
        # Clip and project
        h_new = project_sum_bounds(h_new, target_sum)
        h = h_new
        iter += 1
        # occasional random mutation
        if iter % 200 == 0:
            # add small random noise
            noise = rng.normal(scale=0.01, size=N)
            h_tmp = h + noise
            h_tmp = project_sum_bounds(h_tmp, target_sum)
            # evaluate and pick better
            if compute_c5(h_tmp) < compute_c5(h):
                h = h_tmp
    return best_h, best_score

def project_sum_bounds(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    # compute and adjust iteratively
    for _ in range(100):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= 1e-12:
            break
        # find free indices
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # adjust all equally (will clip afterward)
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / n_free
        v[free] += delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
    # final adjust
    v = np.clip(v, 0.0, 1.0)
    # Ensure sum exactly target within tolerance via redistribution of small diff to first free entry
    s = v.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # distribute diff to a non-bound element if possible else adjust first element
        # try to adjust element not at bound in direction of diff
        candidates = np.where((diff > 0) & (v < 1.0) | (diff < 0) & (v > 0.0))[0]
        if candidates.size > 0:
            i = candidates[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def gradient_improve(h_init, time_limit, rng):
    N = len(h_init)
    target = N / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    iteration = 0
    max_step_initial = 0.02
    while True:
        now = time.time()
        if now - start > time_limit:
            break
        # compute autocorrelation J
        conv = np.correlate(h, h, mode='full') * (2.0 / N)
        min_J = conv.min()
        cur_score = 1.0 - min_J
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
        # find shift with min J
        idx = np.argmin(conv)
        shift = idx - (N - 1)
        shift_abs = abs(shift)
        if shift_abs == 0:
            # possible plateau; add random perturb
            if iteration % 50 == 0:
                # random perturb
                h = project_sum_bounds(h + rng.normal(scale=0.01, size=N), target)
            else:
                # small step to random direction
                h = project_sum_bounds(h + rng.normal(scale=0.001, size=N), target)
            iteration += 1
            continue
        # compute gradient
        g = np.zeros_like(h)
        if shift_abs < N:
            g[:N - shift_abs] += h[shift_abs:]
            g[shift_abs:] += h[:N - shift_abs]
        # step size schedule
        # decay over iterations
        step_factor = max_step_initial / (1.0 + iteration * 0.001)  # decays slowly
        max_g = np.max(g)
        if max_g > 0:
            step = step_factor / max_g
        else:
            step = 0.0
        h_new = h + step * g
        # ensure bounds and sum
        h_new = project_sum_bounds(h_new, target)
        h = h_new
        iteration += 1
        # occasionally random perturbation
        if iteration % 200 == 0:
            if rng.random() < 0.3:
                # random new candidate: blend with best
                pert = rng.normal(scale=0.005, size=N)
                h_tmp = h + pert
                h_tmp = project_sum_bounds(h_tmp, target)
                # accept if better
                tmp_score = compute_c5(h_tmp)
                if tmp_score < cur_score:
                    h = h_tmp
    return best_h, best_score

import numpy as np
import time

def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 value using the prescribed correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to obtain exact sum == target."""
    v = np.clip(v, 0.0, 1.0)
    tol = 1e-12
    for _ in range(100):
        s = v.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (v > 0.0) & (v < 1.0)
        n_free = int(free.sum())
        if n_free == 0:
            # all at a bound – distribute uniformly and re‑clip
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny residual correction
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(v < 1.0)[0]
        else:
            cand = np.where(v > 0.0)[0]
        if cand.size:
            i = int(cand[0])
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum == n/2."""
    v = rng.random(n)
    return _project_sum_bounds(v, n / 2.0)

def _gradient_improve(h0: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Iteratively increase the minimal auto‑correlation (i.e. lower C5)."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(h)
    start = time.time()
    it = 0
    max_step_initial = 0.02

    while True:
        if time.time() - start > time_limit:
            break

        # auto‑correlation J_k = sum h_i h_{i+k}
        dx = 2.0 / n
        j = np.correlate(h, h, mode="full") * dx
        min_j = float(j.min())
        cur_score = 1.0 - min_j
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_h = h.copy()

        # shift giving the smallest J_k
        idx = int(np.argmin(j))
        shift = idx - (n - 1)
        sh = abs(shift)

        if sh == 0:
            # occasional jitter to escape flat regions
            if it % 30 == 0:
                h = _project_sum_bounds(h + rng.normal(scale=0.01, size=n), target_sum)
            it += 1
            continue

        # gradient of J_sh w.r.t. h
        g = np.zeros_like(h)
        if sh < n:
            g[: n - sh] += h[sh:]
            g[sh:] += h[: n - sh]

        # step size decays slowly
        scale = max_step_initial / (1.0 + it * 0.001)
        max_g = float(g.max())
        if max_g > 0.0:
            step = scale / max_g
        else:
            step = 0.0

        h_new = h + step * g
        h = _project_sum_bounds(h_new, target_sum)

        # occasional small random perturbation and hill‑climb acceptance
        if it % 200 == 0 and rng.random() < 0.3:
            cand = h + rng.normal(scale=0.005, size=n)
            cand = _project_sum_bounds(cand, target_sum)
            cand_score = _compute_c5(cand)
            if cand_score < cur_score:
                h = cand

        it += 1

    return best_h, best_score

    if "initial_h_values" in globals():
        h_start = np.array(initial_h_values, dtype=float)
        n_pts = int(h_start.size)
    else:
        n_pts = 70
        h_start = _random_feasible(n_pts, rng)

    # enforce feasibility exactly
    h_start = _project_sum_bounds(h_start, n_pts / 2.0)

    # best known solution so far
    best_h = h_start.copy()
    best_score = _compute_c5(best_h)

    # ------------------- main optimisation -------------------
    start_time = time.time()
    remaining = max(0.0, budget_s - (time.time() - start_time) - 0.05)

    if remaining > 0.0:
        # primary run starting from the supplied construction
        h_opt, sc_opt = _gradient_improve(best_h, remaining, rng)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

        # use any leftover time for a few short random restarts
        while True:
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed - 0.05
            if remaining < 1.0:
                break
            # allocate a short budget (max 8 s) to each restart
            sub_budget = min(remaining / 2.0, 8.0)
            h_rand = _random_feasible(n_pts, rng)
            h_loc, sc_loc = _gradient_improve(h_rand, sub_budget, rng)
            if sc_loc < best_score:
                best_h, best_score = h_loc, sc_loc

    # final safety projection and exact bound recomputation
    best_h = _project_sum_bounds(best_h, n_pts / 2.0)
    final_score = _compute_c5(best_h)

    return best_h.tolist(), final_score, n_pts
