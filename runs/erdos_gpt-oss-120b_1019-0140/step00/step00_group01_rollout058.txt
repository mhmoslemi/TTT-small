def adjust_sum(h, target_sum):
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return
    if diff > 0:
        # need to add diff, find indices where h < 1
        indices = np.where(h < 1)[0]
        for i in indices:
            cap = 1 - h[i]
            add = min(cap, diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # diff < 0, need to subtract
        diff_abs = -diff
        indices = np.where(h > 0)[0]
        for i in indices:
            cap = h[i]
            sub = min(cap, diff_abs)
            h[i] -= sub
            diff_abs -= sub
            if diff_abs <= 1e-12:
                break

def enforce_sum_exact(h, target_sum, rng):
    diff = target_sum - h.sum()
    # tolerance maybe 1e-12; if near zero do nothing.
    if abs(diff) < 1e-12:
        return

        idxs = np.where(h < 1)[0]

import time, math, numpy as np

def _compute_c5(h):
    """Return C5 for a step function h sampled on a uniform grid."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _max_corr_and_lag(h):
    """Return (max_corr, lag) where lag = index - (n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1), corr

def _enforce_sum_exact(h, target, rng):
    """Adjust h in-place so that sum(h) == target while keeping 0<=h<=1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return
    # increase sum
    if diff > 0:
        cand = np.where(h < 1.0)[0]
        order = rng.permutation(cand)
        for i in order:
            cap = 1.0 - h[i]
            add = min(cap, diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # decrease sum
        cand = np.where(h > 0.0)[0]
        order = rng.permutation(cand)
        for i in order:
            cap = h[i]
            sub = min(cap, -diff)
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    # final tiny correction (if any) by spreading uniformly
    if abs(target - h.sum()) > 1e-12:
        correction = (target - h.sum()) / h.size
        h += correction
        np.clip(h, 0.0, 1.0, out=h)

def _propose_random_move(h, rng, step):
    """Move a small amount from one index to another (preserves sum)."""
    n = h.size
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    # ensure j != i
    candidates = neg[neg != i]
    if candidates.size == 0:
        return h.copy()
    j = rng.choice(candidates)
    max_delta = min(step, h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = max_delta * rng.random()
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _propose_targeted_move(h, rng, step, lag):
    """Move mass from i to i+lag (or opposite) to reduce the worst overlap."""
    n = h.size
    if lag == 0:
        return None
    if lag > 0:
        i_start, i_end = 0, n - lag - 1
    else:
        i_start, i_end = -lag, n - 1
    if i_end < i_start:
        return None
    idx = np.arange(i_start, i_end + 1)
    mask = (h[idx] > 1e-12) & (h[idx + lag] < 1.0 - 1e-12)
    candidates = idx[mask]
    if candidates.size == 0:
        return None
    i = rng.choice(candidates)
    max_delta = min(step, h[i], 1.0 - h[i + lag])
    if max_delta <= 0:
        return None
    delta = max_delta * rng.random()
    h_new = h.copy()
    h_new[i] -= delta
    h_new[i + lag] += delta
    return h_new

def _optimize(initial_h, budget, rng):
    """Core simulated‑annealing optimiser."""
    n = initial_h.size
    target_sum = n / 2.0
    h = initial_h.copy()
    _enforce_sum_exact(h, target_sum, rng)
    cur_val = _compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    max_step = 0.2
    min_step = 0.001
    T0 = 0.015  # initial temperature
    iter_cnt = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget * 0.97:
            break
        frac = elapsed / budget
        step = max_step * (1.0 - frac) + min_step * frac
        temp = T0 * max(0.0, 1.0 - frac)

        cur_max, lag_max, _ = _max_corr_and_lag(h)

        # try a targeted move most of the time
        if lag_max != 0 and rng.random() < 0.75:
            h_new = _propose_targeted_move(h, rng, step, lag_max)
            if h_new is None:
                h_new = _propose_random_move(h, rng, step)
        else:
            h_new = _propose_random_move(h, rng, step)

        new_val = _compute_c5(h_new)

        # acceptance test
        if new_val <= cur_max:
            accept = True
        else:
            if temp > 0:
                accept = rng.random() < math.exp(-(new_val - cur_max) / temp)
            else:
                accept = False

        if accept:
            h = h_new
            cur_max = new_val
            # occasional sum correction
            if iter_cnt % 500 == 0:
                _enforce_sum_exact(h, target_sum, rng)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        iter_cnt += 1

    # final refinement: greedy hill climbing with tiny steps
    for _ in range(2000):
        if time.time() - start >= budget * 0.99:
            break
        h_candidate = _propose_random_move(best_h, rng, 5e-4)
        cand_val = _compute_c5(h_candidate)
        if cand_val < best_val:
            best_h = h_candidate
            best_val = cand_val
    _enforce_sum_exact(best_h, target_sum, rng)
    best_val = _compute_c5(best_h)
    return best_h, best_val

def _upsample(h, new_n, rng):
    """Interpolate h onto a finer grid of size new_n."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    target = new_n / 2.0
    _enforce_sum_exact(h_new, target, rng)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- Stage 1: start from the provided construction (if any) ---
    try:

    # final sanity check (clip just in case)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    _enforce_sum_exact(best_h, best_h.size / 2.0, rng)

    # compute final bound
    final_c5 = _compute_c5(best_h)
