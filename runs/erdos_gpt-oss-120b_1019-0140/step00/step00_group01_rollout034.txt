def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    start = time.time()
    np.random.seed(seed)
    # Choose n
    n = 48
    target = n // 2

    # try MILP
    try:
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        constraints = [cp.sum(h) == target,
                       t >= 0]
        for s in range(-n+1, n):
            i_start = max(0, -s)
            i_end = min(n, n - s)
            length = i_end - i_start
            if length <= 0:
                continue
            w = cp.Variable(length, boolean=True)
            # slices
            h_i = h[i_start:i_end]
            h_j = h[i_start + s:i_end + s]  # because i + s = j
            constraints += [w <= h_i,
                            w <= h_j,
                            w >= h_i + h_j - 1]
            constraints += [target - cp.sum(w) <= t]

        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.CBC, verbose=False, warm_start=True)
        # If solve fails result may be None
        if h.value is not None:
            h_vals = np.array(h.value).astype(float)
            # Ensure binary rounding due to numeric tolerance
            h_vals = np.rint(h_vals)   # round to 0 or 1
            # ensure sum equals target
            # if slight difference, adjust small
            if int(round(np.sum(h_vals))) != target:
                # reproject using simple adjustment
                diff = target - int(round(np.sum(h_vals)))
                # adjust first diff entries
                # ensure 0/1 bounds
                indices = np.arange(n)
                for idx in indices:
                    if diff == 0:
                        break
                    if diff > 0 and h_vals[idx] < 1:
                        h_vals[idx] += 1
                        diff -= 1
                    elif diff < 0 and h_vals[idx] > 0:
                        h_vals[idx] -= 1
                        diff += 1
            c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * (2.0 / n)
            return (h_vals, float(c5), n)
    except Exception as e:
        # fallback
        pass
    # Fallback heuristic
    # if initial_h_values available use as start else random
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
    else:
        n = 48
        target = n // 2
        # random binary with sum target
        init = np.zeros(n)
        ones = np.random.choice(n, target, replace=False)
        init[ones] = 1.0
    # ensure sum = target
    # do simple local search
    best_h = init.copy()
    best_score = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n)
    # Project function
    def project(v):
        # Clip [0,1]
        v = np.clip(v, 0.0, 1.0)
        # adjust sum to target via water-filling (lambda)
        lo = v.min() - 2.0
        hi = v.max() + 2.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            h_proj = np.clip(v - mid, 0.0, 1.0)
            s = h_proj.sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # gradient descent parameters
    step = 0.5
    max_iters = 20000
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break
        # compute correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # find shift with max value
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        # gradient: 1 - h[i+k] - h[i-k] with zero padding
        grad = np.ones_like(best_h)
        # shift plus
        if k >= 0:
            grad[:n - k] -= best_h[k:]
            grad[k:] -= best_h[:n - k]
        else:
            kk = -k
            grad[kk:] -= best_h[:n - kk]
            grad[:n - kk] -= best_h[kk:]
        # update
        new_h = best_h - step * grad
        new_h = project(new_h)
        new_score = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * (2.0 / n)
        if new_score < best_score - 1e-8:
            best_h, best_score = new_h, new_score
            step = min(step * 1.1, 2.0)
        else:
            step *= 0.5
    return (best_h, float(best_score), n)

def _project_to_feasible(v, target):
    """Clamp to [0,1] and shift so that sum equals target."""
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    """Return the C₅ value for a step function h."""
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

    h_cur = _project_to_feasible(h_cur, target)
    best_h = h_cur.copy()
    best_score = _c5_value(best_h)

    step = 0.5
    max_iters = 20000

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break

        # find shift giving the current maximal overlap
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        k_idx = int(np.argmax(corr))
        shift = k_idx - (n - 1)

        # gradient of the active term: 1 - h[i+shift] - h[i-shift]
        grad = np.ones_like(best_h)
        if shift >= 0:
            grad[: n - shift] -= best_h[shift:]
            grad[shift:]      -= best_h[: n - shift]
        else:
            kk = -shift
            grad[kk:]        -= best_h[: n - kk]
            grad[: n - kk]   -= best_h[kk:]

        # take a step and re‑project
        new_h = best_h - step * grad
        new_h = _project_to_feasible(new_h, target)
        new_score = _c5_value(new_h)

        if new_score < best_score - 1e-9:
            best_h, best_score = new_h, new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5

    return (best_h, float(best_score), n)
