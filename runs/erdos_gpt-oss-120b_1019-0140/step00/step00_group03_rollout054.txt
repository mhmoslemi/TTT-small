def random_feasible_h(n, target_sum, rng):

    remaining = target_sum
    for i in range(n - 1):
        # Minimum we can allocate to remaining slots: (remaining_min) = 0 * (n-i-1) = 0
        # Maximum we can allocate to remaining slots: (n-i-1) * 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def random_feasible_h(n, target, rng):
    h = np.empty(n)
    remaining = target
    for i in range(n - 1):
        low = max(0.0, remaining - (n - i - 1) * 1.0)

        val = rng.uniform(low, high)

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on a uniform grid."""
    n = h.size
    dx = 2.0 / n
    # np.correlate gives Σ h[i]·(1−h[i+k]) for each shift k (zero‑padded)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def random_feasible_h(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random h with 0≤h_i≤1 and Σh_i = target.
    Uses a simple sequential sampling that respects the box constraints.
    """
    h = np.empty(n, dtype=float)
    remaining = target
    for i in range(n - 1):
        low = max(0.0, remaining - (n - i - 1) * 1.0)
        high = min(1.0, remaining)
        val = rng.uniform(low, high)
        h[i] = val
        remaining -= val
    h[-1] = max(0.0, min(1.0, remaining))
    return h
