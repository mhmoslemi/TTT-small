def compute_c5(h):
    n = len(h)

    # Compute cross-correlation via np.correlate

    corr = np.correlate(h, 1 - h, mode='full')
    # The integral approximated by sum * dx
    # The maximum correlation across all lags: max(corr) * dx
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    rng = np.random.default_rng(seed)
    # Determine n_points, maybe use min(200, ...). Could also try multiple n's for exploration.
    n = 200
    # If initial_h_values exists and is of length >0:
    if 'initial_h_values' in globals():
         init = initial_h_values
         if init.shape[0] != n:
              h = upsample_initial(init, n)
         else:
              h = init.copy()
    else:
         h = np.full(n, 0.5)
    # project onto constraints
    h = project_onto_box_sum(h, 0.0, 1.0, n/2)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # define parameters for gradient descent
    max_iter = 20000   # but will break early based on time.

    for iter in range(max_iter):
        if time.perf_counter() - start_time > budget_s - 0.5:
            break
        # compute correlation per shift (positive)
        corr = compute_corrs(h)
        s_star = np.argmax(corr)
        current_c5 = corr.max() * (2.0/n)
        if current_c5 < best_c5 - 1e-12:
            best_c5 = current_c5
            best_h = h.copy()
        # compute gradient
        grad = gradient_worst_shift(h, s_star)
        # step size schedule: base = 0.1 / sqrt(iter+1) maybe
        step = 0.5 / np.sqrt(iter + 1)   # step decreasing
        # update
        h_new = h - step * grad
        # project
        h_new = project_onto_box_sum(h_new, 0.0, 1.0, n/2)
        # compute new c5 to see if improvement
        new_c5 = compute_c5(h_new)
        # accept if better else maybe accept with probability
        if new_c5 <= current_c5 or rng.random() < 0.01:
            h = h_new
        # else ignore (stay at current)

        # occasionally random restart: random vector near best
        if iter % 500 == 0 and iter > 0:
            # random perturbation around best
            perturb = rng.normal(scale=0.1, size=n)
            h_candidate = best_h + perturb
            h_candidate = project_onto_box_sum(h_candidate, 0.0, 1.0, n/2)
            cand_c5 = compute_c5(h_candidate)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h_candidate.copy()
            # Also maybe reset h to candidate to explore
            h = h_candidate

    # after loop, final best
    return (best_h, best_c5, n)

def gradient_of_shift(h, s):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if s == 0:
        grad[:] = 1 - 2*h
        return grad
    if s > 0:
        # first contribution: i from 0 to n-s-1
        grad[:n-s] += 1 - h[s:]
        # second contribution: i from s to n-1
        grad[s:] += -h[:n-s]
    else:
        t = -s
        # first contribution: i from t to n-1
        grad[t:] += 1 - h[:n - t]
        # second: i from 0 to n - t -1
        grad[:n - t] += -h[t:]
    return grad

def compute_c5_full(h):
    n = len(h)
    corr = np.convolve(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

import time
import numpy as np

def _dx(n):
    return 2.0 / n

def _compute_c5(h):
    """Full correlation C5 = max_k ∫ h(x)(1-h(x+k))dx."""
    n = h.shape[0]
    corr = np.convolve(h, 1.0 - h, mode="full")
    return corr.max() * _dx(n)

def _gradient_of_shift(h, shift):
    """Gradient of the worst‑shift correlation w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if shift > 0:
        # first part: d/dh_i of (1 - h_{i+shift})
        g[: n - shift] += 1.0 - h[shift:]
        # second part: d/dh_{i+shift} of -h_i
        g[shift:] += -h[: n - shift]
    else:  # shift < 0
        s = -shift
        g[s:] += 1.0 - h[: n - s]
        g[: n - s] += -h[s:]
    return g

def _project_box_sum(v, low, high, target):
    """Euclidean projection of v onto {x∈[low,high]^n, sum(x)=target}."""
    # binary search for the Lagrange multiplier λ
    lo, hi = -5.0, 5.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, low, high)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, low, high)

def _upsample(initial, n):
    """Linear interpolation of an existing vector to length n."""
    m = initial.shape[0]
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, initial)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.perf_counter()
    rng = np.random.default_rng(seed)

    # --- problem size -------------------------------------------------
    n = 80  # even number; small enough for fast eval, large enough for quality
    target_sum = n / 2.0
    low, high = 0.0, 1.0

    # --- initial constructions -----------------------------------------
    candidates = []

    # uniform 0.5
    candidates.append(np.full(n, 0.5))

    # linear ramp (0 → 1)
    x = np.linspace(0.0, 2.0, n, endpoint=False) + _dx(n) * 0.5
    linear = x / 2.0
    candidates.append(_project_box_sum(linear, low, high, target_sum))

    # cosine bump (peak 1, trough 0)
    cos = 0.5 + 0.5 * np.cos(np.pi * x)
    candidates.append(_project_box_sum(cos, low, high, target_sum))

    # upsample provided parent if available
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] != n:
            init = _upsample(init, n)
        candidates.append(_project_box_sum(init, low, high, target_sum))

    best_h = None
    best_c5 = np.inf

    def _update_best(vec):
        nonlocal best_h, best_c5
        val = _compute_c5(vec)
        if val < best_c5:
            best_c5 = val
            best_h = vec.copy()

    # evaluate deterministic starts
    for vec in candidates:
        _update_best(vec)

    # --- random search ------------------------------------------------
    # draw vectors from a beta distribution and scale to satisfy the sum
    max_random = 200_000
    for i in range(max_random):
        if time.perf_counter() - start > budget_s * 0.55:
            break
        v = rng.beta(2.0, 2.0, size=n)
        s = v.sum()
        if s < target_sum:          # scaling would exceed 1, reject
            continue
        vec = v * (target_sum / s)  # all entries ≤1 after scaling
        _update_best(vec)

    # --- subgradient refinement ----------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _compute_c5(best_h)

    h = best_h.copy()
    it = 0
    step0 = 0.8  # initial step magnitude
    while time.perf_counter() - start < budget_s * 0.85:
        corr = np.convolve(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_c5 = corr[idx] * _dx(n)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        shift = idx - (n - 1)
        grad = _gradient_of_shift(h, shift)

        step = step0 / np.sqrt(it + 1.0)
        h_new = h - step * grad
        h_new = _project_box_sum(h_new, low, high, target_sum)

        new_c5 = _compute_c5(h_new)
        # accept only if we do not worsen the worst shift
        if new_c5 <= cur_c5:
            h = h_new
        else:
            step0 *= 0.9  # gradually tighten step size if stuck
        it += 1
        if it > 20_000:
            break

    # --- local pairwise swaps -----------------------------------------
    cur_c5 = _compute_c5(best_h)
    for _ in range(30_000):
        if time.perf_counter() - start > budget_s * 0.975:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a balanced move: +d on i, -d on j
        max_inc_i = high - best_h[i]
        max_dec_i = best_h[i] - low
        max_inc_j = high - best_h[j]
        max_dec_j = best_h[j] - low
        d_min = max(-max_dec_i, -max_inc_j)
        d_max = min(max_inc_i, max_dec_j)
        if d_max <= d_min:
            continue
        d = rng.uniform(d_min, d_max)
        trial = best_h.copy()
        trial[i] += d
        trial[j] -= d
        val = _compute_c5(trial)
        if val < cur_c5:
            best_h = trial
            cur_c5 = val
            best_c5 = val

    return best_h, best_c5, n
