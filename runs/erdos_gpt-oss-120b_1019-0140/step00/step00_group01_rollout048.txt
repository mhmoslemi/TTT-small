def project_to_constraints(h):
    n = len(h)
    target = n / 2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # If sum == target within tolerance, done
    if abs(s - target) < 1e-12:
        return h
    # Use water-filling
    # We'll adjust using iterative approach.
    # Determine whether need to increase or decrease.
    if s > target:
        # need to reduce s
        excess = s - target
        # while excess > tolerance:
        # compute indices where h > 0
        while excess > 1e-12:
            idx = np.where(h > 0)[0]
            if idx.size == 0: break
            # delta to subtract per index
            dec = excess / idx.size
            # apply but not go below 0
            new_vals = h[idx] - dec
            # find those that would become negative
            neg = new_vals < 0
            if np.any(neg):
                # amount subtracted for those is h[idx[neg]]
                removed = h[idx[neg]].sum()
                h[idx[neg]] = 0.0
                excess -= removed
                # Continue loop with new excess
            else:
                h[idx] = new_vals
                excess = 0.0
    else:
        # s < target, need to add
        deficit = target - s
        while deficit > 1e-12:
            idx = np.where(h < 1)[0]
            if idx.size == 0: break
            inc = deficit / idx.size
            new_vals = h[idx] + inc
            over = new_vals > 1
            if np.any(over):
                added = (1 - h[idx[over]]).sum()
                h[idx[over]] = 1.0
                deficit -= added
            else:
                h[idx] = new_vals
                deficit = 0.0
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # Final check
    # Might still have small numeric mismatch; adjust by scaling the unconstrained entries.
    # Compute residual.
    s = h.sum()
    if abs(s - target) > 1e-8:
        # adjust a random non-bound index
        diff = target - s
        # find index not at bound (or any)
        idx = np.arange(len(h))
        # If diff >0, need to increase, pick index where h < 1
        # else diff<0, pick index where h >0
        if diff > 0:
            candidates = idx[h < 1 - 1e-12]
        else:
            candidates = idx[h > 1e-12]
        if candidates.size > 0:
            k = np.random.choice(candidates)
            h[k] = np.clip(h[k] + diff, 0.0, 1.0)
    return h

import numpy as np
import time
import random

def _project_to_constraints(h):
    """
    Clip h to [0,1] and adjust to satisfy sum(h) = n/2 exactly.
    Uses a water-filling style iterative algorithm.
    """
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) <= 1e-12:
        return h

    # Reduce excess mass
    if s > target:
        excess = s - target
        while excess > 1e-12:
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            dec = excess / idx.size
            new_vals = h[idx] - dec
            below = new_vals < 0
            if np.any(below):
                # take whole remaining values of those indices
                removed = h[idx[below]].sum()
                h[idx[below]] = 0.0
                excess -= removed
                # continue with remaining indices
            else:
                h[idx] = new_vals
                excess = 0.0
    # Add deficit mass
    else:
        deficit = target - s
        while deficit > 1e-12:
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            inc = deficit / idx.size
            new_vals = h[idx] + inc
            over = new_vals > 1
            if np.any(over):
                added = (1 - h[idx[over]]).sum()
                h[idx[over]] = 1.0
                deficit -= added
            else:
                h[idx] = new_vals
                deficit = 0.0

    # Small residual correction (if any)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        if diff > 0:
            candidates = np.where(h < 1 - 1e-12)[0]
        else:
            candidates = np.where(h > 1e-12)[0]
        if candidates.size:
            k = random.choice(candidates)
            h[k] = np.clip(h[k] + diff, 0.0, 1.0)

    return h

def _c5_from_h(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _random_feasible_h(n):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.random.rand(n)
    return _project_to_constraints(h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where h_values is a numpy
    array representing the step function, c5_bound is the achieved bound,
    and n_points is the number of discretisation points.
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        h_cur = _project_to_constraints(h0.copy())
    else:
        n = 47  # default size when no parent is provided
        h_cur = _random_feasible_h(n)

    # Initial evaluation
    best_h = h_cur.copy()
    best_c5 = _c5_from_h(best_h)

    cur_c5 = best_c5

    # Parameters for simulated annealing
    T0 = 0.02                      # initial temperature
    reinit_gap = 30.0              # seconds without improvement before restart
    last_improve = start_time

    # Main annealing loop
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.99:
            break

        # Propose a mass-transfer move that preserves the sum exactly
        for _ in range(5):
            i, j = np.random.randint(n), np.random.randint(n)
            if i == j:
                continue
            # Move mass from i to j (i -> j)
            max_delta = min(h_cur[i], 1.0 - h_cur[j])
            if max_delta <= 1e-12:
                # Try opposite direction
                max_delta = min(h_cur[j], 1.0 - h_cur[i])
                if max_delta <= 1e-12:
                    continue
                i, j = j, i
            # Scale step size with temperature (larger steps early)
            elapsed = now - start_time
            T = max(1e-8, T0 * (1.0 - elapsed / budget_s))
            # delta magnitude adapts to temperature
            scale = max(0.01, T / T0)
            delta = max_delta * np.random.rand() * scale
            if delta <= 1e-14:
                continue
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta
            break
        else:
            # No feasible move found; abort
            break

        new_c5 = _c5_from_h(h_new)

        # Acceptance rule
        if new_c5 < cur_c5:
            accept = True
        else:
            accept_prob = np.exp(-(new_c5 - cur_c5) / T)
            accept = np.random.rand() < accept_prob

        if accept:

            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h_new.copy()

            cur_c5 = _c5_from_h(h_cur)
            last_improve = now

    best_h = _project_to_constraints(best_h)
    best_c5 = _c5_from_h(best_h)

    return best_h, best_c5, n
