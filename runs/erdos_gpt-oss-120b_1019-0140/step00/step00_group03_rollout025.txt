import numpy as np
import random
import time

# ---------- Helper functions (top level) ----------

def random_binary_vector(n_points, n_ones):
    """Return a random binary vector of length n_points with exactly n_ones ones."""
    vec = np.zeros(n_points, dtype=np.int8)
    idx = np.random.choice(n_points, n_ones, replace=False)
    vec[idx] = 1
    return vec


def repair_sum(vec, target_ones):
    """Flip bits to enforce exactly target_ones ones."""
    s = int(vec.sum())
    if s == target_ones:
        return
    if s > target_ones:
        # need to turn (s-target) ones into zeros
        ones = np.where(vec == 1)[0]
        to_flip = np.random.choice(ones, s - target_ones, replace=False)
        vec[to_flip] = 0
    else:  # s < target_ones
        zeros = np.where(vec == 0)[0]
        to_flip = np.random.choice(zeros, target_ones - s, replace=False)
        vec[to_flip] = 1


def crossover(parent1, parent2):
    """One-point crossover."""
    n = parent1.shape[0]
    point = np.random.randint(0, n + 1)
    child = np.concatenate((parent1[:point], parent2[point:]))
    return child


def mutate_swap(vec):
    """Swap a random 1 with a random 0 (preserves sum)."""
    ones = np.where(vec == 1)[0]
    zeros = np.where(vec == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    vec[i] = 0
    vec[j] = 1


def tournament_select(pop, scores, tournament_size=3):
    """Select one individual via tournament (lower score = better)."""
    idxs = np.random.choice(len(pop), tournament_size, replace=False)
    best = idxs[0]
    best_score = scores[best]
    for idx in idxs[1:]:
        if scores[idx] < best_score:
            best = idx
            best_score = scores[idx]
    return pop[best]


def compute_score(candidate):
    """Wrap evaluate_erdos_solution for a candidate vector."""
    # Ensure float dtype for evaluation
    return evaluate_erdos_solution(candidate.astype(float))
