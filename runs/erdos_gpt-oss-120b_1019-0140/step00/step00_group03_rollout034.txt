def compute_c5(h):
    # Ensure np.asarray float
    h_arr = np.asarray(h, dtype=float)
    corr = np.correlate(h_arr, 1 - h_arr, mode='full')
    maxval = np.max(corr)
    return maxval * (2.0 / len(h_arr))

def hill_climb(h):
    current = h.copy()
    current_score = compute_c5(current)
    while True:
        improved = False
        # list of indices of ones and zeros
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        best_local_score = current_score
        best_swap = None
        # iterate over swaps
        for i in ones:
            # optional break early if time?
            for j in zeros:
                # try swap
                candidate = current.copy()
                candidate[i] = 0
                candidate[j] = 1
                score = compute_c5(candidate)
                if score < best_local_score - 1e-12:
                    best_local_score = score
                    best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            current[i] = 0
            current[j] = 1
            current_score = best_local_score
            improved = True
        if not improved:
            break
    return current, current_score

    import time, random, math, numpy as np
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    # Determine problem size
    try:
        N = len(initial_h_values)
    except NameError:
        N = 70
    n_ones = N // 2
    dx = 2.0 / N

    # define evaluation functions
    cache = {}
    def compute_c5(arr):
        # arr: binary/int array
        corr = np.correlate(arr.astype(float), 1 - arr.astype(float), mode='full')
        return float(np.max(corr) * dx)

    def evaluate(arr):
        key = arr.tobytes()
        v = cache.get(key)
        if v is None:
            v = compute_c5(arr)
            cache[key] = v
        return v

    # hill climbing function
    def hill_climb(arr):
        arr = arr.copy()
        score = evaluate(arr)
        # precompute indices each loop
        while True:
            ones = np.where(arr == 1)[0]
            zeros = np.where(arr == 0)[0]
            best_swap = None
            best_score = score
            # try all swaps
            for i in ones:
                for j in zeros:
                    # avoid swapping same index (can't happen)
                    cand = arr.copy()
                    cand[i] = 0
                    cand[j] = 1
                    cand_score = evaluate(cand)
                    if cand_score < best_score - 1e-12:
                        best_score = cand_score
                        best_swap = (i, j)
            if best_swap is None:
                break
            i, j = best_swap
            arr[i] = 0
            arr[j] = 1
            score = best_score
        return arr, score

    def make_balanced(v):
        cur = int(v.sum())
        if cur == n_ones:
            return v
        if cur < n_ones:
            zeros = np.where(v == 0)[0]
            need = n_ones - cur
            add = np.random.choice(zeros, need, replace=False)
            v[add] = 1
        else:
            ones = np.where(v == 1)[0]
            need = cur - n_ones
            rem = np.random.choice(ones, need, replace=False)
            v[rem] = 0

            init_bin = make_balanced(init_bin)
            h, s = hill_climb(init_bin)

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, numpy as np
    # Set reproducibility
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # Problem size
    try:
        N = len(initial_h_values)
    except NameError:
        N = 70
    n_ones = N // 2
    dx = 2.0 / N

    # Cache for evaluation results
    _cache = {}

    def _c5(arr):
        # arr: binary integer array
        corr = np.correlate(arr.astype(float), 1 - arr.astype(float), mode='full')
        return float(np.max(corr) * dx)

    def _evaluate(arr):
        key = arr.tobytes()
        val = _cache.get(key)
        if val is None:
            val = _c5(arr)
            _cache[key] = val
        return val

    def _make_balanced(v):
        cur = int(v.sum())
        if cur < n_ones:
            zeros = np.where(v == 0)[0]
            add = np.random.choice(zeros, n_ones - cur, replace=False)
            v[add] = 1
        elif cur > n_ones:
            ones = np.where(v == 1)[0]
            rem = np.random.choice(ones, cur - n_ones, replace=False)
            v[rem] = 0
        return v

    def _hill_climb(v):
        v = v.copy()
        best = _evaluate(v)
        while True:
            ones = np.where(v == 1)[0]
            zeros = np.where(v == 0)[0]
            improve = None
            improve_score = best
            # exhaustive search of all 1↔0 swaps
            for i in ones:
                for j in zeros:
                    cand = v.copy()
                    cand[i] = 0
                    cand[j] = 1
                    sc = _evaluate(cand)
                    if sc < improve_score - 1e-12:
                        improve_score = sc
                        improve = (i, j, cand)
            if improve is None:
                break
            i, j, cand = improve
            v[i] = 0
            v[j] = 1
            best = improve_score
        return v, best

    best_h = None
    best_score = float('inf')
    target = 0.3808

    # Start from provided construction if available
    try:
        init = np.array(initial_h_values)
        if init.shape[0] == N:
            init_bin = (init >= 0.5).astype(int)
            init_bin = _make_balanced(init_bin)
            h, sc = _hill_climb(init_bin)
            if sc < best_score:
                best_h, best_score = h, sc
    except Exception:
        pass

    # Random restart local search
    while True:
        if time.time() - start_time > budget_s - 5:
            break
        perm = np.random.permutation(N)
        v = np.zeros(N, dtype=int)
        v[perm[:n_ones]] = 1
        h, sc = _hill_climb(v)
        if sc < best_score:

            if best_score <= target:

        best_h[:n_ones] = 1
        np.random.shuffle(best_h)

    except Exception:

    return (best_h.astype(float), final_c5, N)
