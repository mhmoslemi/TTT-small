def project_sum_box(x, target_sum, lo=0., hi=1., tol=1e-9, max_iter=100):
    lo_lambda = -np.max(x) - 1.0 # ensures x+lambda <=0
    hi_lambda = 1.0 - np.min(x) + 1.0 # ensures x+lambda >=1
    for _ in range(max_iter):
        lam = (lo_lambda + hi_lambda) / 2.0
        proj = np.clip(x + lam, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) <= tol:
            return proj
        if s > target_sum:
            hi_lambda = lam
        else:
            lo_lambda = lam
    return proj

import numpy as np
import time
import math

def _compute_c5(h):
    """Return C5 = max_{shift} ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    # correlation without dx, then multiply by dx = 2/n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Simulated‑annealing search for a step function on [0,2] with
    sum(h)=1 (i.e. sum(h)=n/2) and 0≤h≤1 minimizing the Erdős C5 constant.

    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and an initial binary configuration.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        init = None

    if init is not None:
        # Use the parent's length (it is already a 0‑1/continuous vector of length 70)
        n = init.shape[0]
        if n % 2 == 1:                     # enforce even length for a strict half‑sum
            n -= 1
            init = init[:n]
        target_ones = n // 2                # exact number of ones required
        # Choose the largest entries of the parent as ones (greedy rounding)
        order = np.argsort(-init)           # descending
        h_cur = np.zeros(n, dtype=np.float64)
        h_cur[order[:target_ones]] = 1.0
    else:
        # No parent – work with a moderate size that balances quality and speed.
        n = 200
        if n % 2:
            n += 1
        target_ones = n // 2
        h_cur = np.zeros(n, dtype=np.float64)
        h_cur[np.random.choice(n, size=target_ones, replace=False)] = 1.0

    # ------------------------------------------------------------------
    # Simulated annealing on binary vectors (swap a 0 ↔ 1 at each step).
    # ------------------------------------------------------------------
    best_h = h_cur.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = h_cur.copy()
    cur_c5 = best_c5

    # Annealing schedule parameters
    T_initial = 0.06                # starting temperature
    total_time = budget_s
    deadline = start + total_time * 0.95   # leave a small safety margin

    iteration = 0
    while time.time() < deadline:
        # Linear cooling: temperature drops to zero at the end of the budget.
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = T_initial * max(0.0, 1.0 - fraction)

        # Propose a swap of a randomly chosen 0 and 1.
        i, j = np.random.randint(n, size=2)
        if cur_h[i] == cur_h[j]:
            # no change – try another pair next loop
            continue

        # Perform the swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]

        # Evaluate the new configuration
        new_c5 = _compute_c5(new_h)

        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            # Accept the move
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                # Early exit if we have already beaten the target record
                if best_c5 <= 0.3808:
                    return (best_h, best_c5, n)

        iteration += 1

        # Occasionally inject a small random perturbation to escape plateaus
        if iteration % 50000 == 0:
            # Apply a few random swaps based on the current best solution
            cur_h = best_h.copy()
            cur_c5 = best_c5

                if cur_h[a] != cur_h[b]:
                    cur_h[a], cur_h[b] = cur_h[b], cur_h[a]
    # ------------------------------------------------------------------
    # Return the best solution found within the time budget.
    # ------------------------------------------------------------------
    best_c5 = _compute_c5(best_h)   # final accurate evaluation
    return (best_h, best_c5, n)
