def gradient_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    # find index of max, pick first
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)  # shift amount (could be negative)
    # compute gradient
    grad = np.zeros_like(h)
    if s >= 0:
        # positive shift

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(x, target):
    lo = -np.max(x) - 5.0
    hi = np.max(x) + 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust small diff
    diff = np.sum(y) - target
    if abs(diff) > 1e-9:
        # find indices not at bounds
        mask = (y > 0) & (y < 1)
        if np.any(mask):
            # distribute diff equally among mask
            y[mask] -= diff / np.sum(mask)
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    grad = np.zeros_like(h)
    if s >= 0:

        i = np.arange(n - s)

        j = np.arange(s, n)

    else:
        t = -s
        i = np.arange(t, n)
        j = i - t
        grad[i] += 1 - h[j]
        grad[j] -= h[i]
    return grad

import time
import numpy as np

def compute_c5(h):
    """Return C5 for a discretized step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(x, target):
    """Projection of x onto the hypercube [0,1]^n with exact sum=target."""
    lo = -np.max(x) - 5.0
    hi = np.max(x) + 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = np.sum(y) - target
    if abs(diff) > 1e-9:
        mask = (y > 0.0) & (y < 1.0)
        cnt = np.count_nonzero(mask)
        if cnt:
            y[mask] -= diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y


def gradient_max_shift(h):
    """Gradient of the currently maximal overlap term w.r.t. h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)  # shift amount (can be negative)
    grad = np.zeros_like(h)
    if s >= 0:
        i = np.arange(n - s)
        grad[i] += 1.0 - h[i + s]
        j = np.arange(s, n)
        grad[j] -= h[j - s]
    else:
        t = -s
        i = np.arange(t, n)
        j = i - t
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    return grad


def _optimize_factor(factor, init_vec, rng, time_budget):
    """Upsample `init_vec` by `factor` and improve it within `time_budget` seconds."""
    h = np.repeat(init_vec, factor)
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-12:
        h = project_box_sum(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    max_iters = 8000
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        grad = gradient_max_shift(cur_h)
        t = it / max_iters
        step = (1 - t) * 0.30 + t * 0.005
        cand = cur_h - step * grad
        cand = project_box_sum(cand, target)
        cand_c5 = compute_c5(cand)

        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
        else:
            # occasional random restart to escape plateaus
            if rng.random() < 0.001:
                cur_h = project_box_sum(rng.random(n), target)
                cur_c5 = compute_c5(cur_h)

    # Light random perturbation phase until the allocated budget is consumed
    while time.time() - start < time_budget:
        delta = rng.normal(scale=0.05, size=n)
        cand = best_h + delta
        cand = project_box_sum(cand, target)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points used."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # Obtain the parent construction if supplied
    try:

    # Try several up‑sampling factors; each gets a share of the remaining budget

        elapsed = time.time() - start

        if init_vec is not None:
            h_opt, c5_opt = _optimize_factor(factor, init_vec, rng, time_left)
        else:

            rand = rng.random(n)

            c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Fallback – simple half‑ones, half‑zeros (unlikely to be needed)
