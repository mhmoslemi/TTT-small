def project_sum(vec, target):
    # Clip first
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    if s < target:
        # Need to increase some elements (those <1)
        cap = 1.0 - v
        total_cap = cap.sum()
        if total_cap <= 0:
            return v
        needed = target - s
        add = needed * cap / total_cap
        v = v + add
    else:
        # Need to decrease
        cap = v
        total_cap = cap.sum()
        if total_cap <= 0:
            return v
        excess = s - target
        sub = excess * cap / total_cap
        v = v - sub
    # Ensure numerical bounds after operation
    return np.clip(v, 0.0, 1.0)

def gradient_shift(vec, shift):
    n = vec.size
    g = np.zeros_like(vec)
    if shift >= 0:
        idx = np.arange(n - shift)
        g[idx] += 1.0 - vec[idx + shift]
        g[idx + shift] -= vec[idx]
    else:
        s = -shift
        idx = np.arange(s, n)
        j = idx - s
        g[idx] += 1.0 - vec[j]
        g[j] -= vec[idx]
    return g
```

import numpy as np
import time
import math

def project_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """Clamp vec to [0,1] and adjust to have exact sum = target."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    if s < target:
        # need to increase some entries
        inc_cap = 1.0 - v
        total_cap = inc_cap.sum()
        if total_cap <= 0:
            return v
        needed = target - s
        v += needed * inc_cap / total_cap
    else:
        # need to decrease some entries
        dec_cap = v
        total_cap = dec_cap.sum()
        if total_cap <= 0:
            return v
        excess = s - target
        v -= excess * dec_cap / total_cap
    return np.clip(v, 0.0, 1.0)


def compute_c5(vec: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(np.max(corr) * dx)


def gradient_shift(vec: np.ndarray, shift: int) -> np.ndarray:
    """
    Subgradient of the overlap term attaining the current maximum.
    d/dh_i (sum_j h_j (1-h_{j+shift})) for the shift with max value.
    """
    n = vec.size
    g = np.zeros_like(vec)
    if shift >= 0:
        idx = np.arange(n - shift)
        g[idx] += 1.0 - vec[idx + shift]
        g[idx + shift] -= vec[idx]
    else:
        s = -shift
        idx = np.arange(s, n)
        j = idx - s
        g[idx] += 1.0 - vec[j]
        g[j] -= vec[idx]
    return g

    best_h = project_sum(best_h, final_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), int(final_n)
