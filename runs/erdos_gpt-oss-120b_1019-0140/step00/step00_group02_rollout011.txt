def compute_c5(h):
    n = h.shape[0]
    a = 1.0 - h
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(np.pad(h, (0, m-n)))
    A = np.fft.rfft(np.pad(a, (0, m-n)))
    corr = np.fft.irfft(H * np.conj(A), n=m)[:2*n - 1]
    max_corr = np.max(corr)
    dx = 2.0 / n
    return max_corr * dx
