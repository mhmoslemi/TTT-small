def tweak(h, step_scale=0.1):
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    h_new = h.copy()
    # propose delta
    delta = np.random.uniform(-step_scale, step_scale)
    # ensure bounds after adjusting:
    # h[i] + delta in [0,1]
    # h[j] - delta in [0,1]
    # Clip delta accordingly
    max_delta_i = 1 - h[i]
    min_delta_i = -h[i]
    max_delta_j = h[j]  # because we will subtract delta: new h[j] = h[j] - delta; we need within [0,1], so delta <= h[j]; also >= h[j] -1 (so delta >= h[j] - 1)
    min_delta_j = h[j] - 1
    # combine constraints: delta <= max_delta_i and delta <= max_delta_j; delta >= min_delta_i and delta >= min_delta_j
    delta_low = max(min_delta_i, min_delta_j)
    delta_high = min(max_delta_i, max_delta_j)
    if delta_low > delta_high:
        return h  # cannot adjust
    # rescale random delta to this interval
    delta = np.random.uniform(delta_low, delta_high)
    # apply
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

import numpy as np
import time

def compute_c5(h):
    """Return max overlap integral C5 for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(v, desired_sum, max_iter=20):
    """Project onto [0,1] box and equality sum constraint."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = np.sum(v)
        diff = desired_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final scaling fallback
    cur_sum = np.sum(v)
    if abs(cur_sum - desired_sum) > 1e-8:
        v = v * (desired_sum / cur_sum)
        v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible_vector(n):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = np.random.rand(n)
    desired = n / 2.0
    return project_feasible(v, desired)

def tweak(h, step_scale=0.2):
    """Make a random two‑index adjustment that preserves the total sum."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # feasible delta range respecting box constraints:
    delta_low = max(-hi, hj - 1.0)
    delta_high = min(1.0 - hi, hj)
    if delta_low > delta_high:
        return h.copy()
    # pick a random delta within the allowed interval
    delta = np.random.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def simulated_annealing(start_h, start_score, max_seconds):
    """Simple SA that keeps the sum fixed via paired tweaks."""
    h = start_h.copy()
    score = start_score
    best_h = h.copy()
    best_score = score
    start = time.time()
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        frac = elapsed / max_seconds
        # temperature linearly decays from 0.01 to 1e-6
        T = max(1e-6, 0.01 * (1.0 - frac))
        h_new = tweak(h, step_scale=0.2)
        new_score = compute_c5(h_new)
        delta = new_score - score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
    return best_h, best_score

    start_time = time.time()
    time_limit = budget_s * 0.95  # leave small safety margin

    init = globals().get('initial_h_values')

        h0 = np.full(95, 0.5, dtype=np.float64)
    n = h0.size
    desired_sum = n / 2.0
    h0 = project_feasible(h0, desired_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Number of independent SA restarts (including the initial one)
    max_restarts = 5
    for r in range(max_restarts):
        elapsed = time.time() - start_time

        if r == 0:

            cur_score = best_score
        else:

    best_score = compute_c5(best_h)

    return best_h.astype(np.float64), float(best_score), int(n)
