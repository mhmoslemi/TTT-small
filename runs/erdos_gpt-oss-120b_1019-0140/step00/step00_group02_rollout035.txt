                avail = (1.0 - h[idx]) if diff > 0 else h[idx]
                if avail <= 0:

                add = min(avail, diff) if diff > 0 else max(-avail, diff)
                h[idx] += add

        if current_sum > 0:

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial solution -------------------------------------------------
    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.array(init, dtype=float)
        n0 = h0.size
        # upscale once (max 256 points) to give more freedom
        n = min(256, n0 * 2)
        if n != n0:
            dx0 = 2.0 / n0
            dx = 2.0 / n
            x0 = np.arange(n0) * dx0 + dx0 / 2.0
            x = np.arange(n) * dx + dx / 2.0
            h = np.interp(x, x0, h0)
            target_sum = n / 2.0
            s = h.sum()
            if s > 0:
                h *= target_sum / s
            h = np.clip(h, 0.0, 1.0)
        else:
            h = h0.copy()
    else:
        # triangular shape on [0,2] (integrates to 1)
        n = 128
        dx = 2.0 / n
        x = np.arange(n) * dx + dx / 2.0
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        target_sum = n / 2.0
        s = h.sum()
        if s > 0:
            h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)

    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(vec):
        return dx * np.max(np.correlate(vec, 1.0 - vec, mode='full'))

    # ----- repair to exact sum ---------------------------------------------
    def repair_sum(vec):
        vec = np.clip(vec, 0.0, 1.0)
        diff = target_sum - vec.sum()
        if abs(diff) < 1e-12:
            return vec
        # distribute diff greedily
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                idxs = np.where(vec < 1.0)[0]
                if idxs.size == 0:
                    break
                cap = 1.0 - vec[idxs]
                add = np.minimum(cap, diff / idxs.size)
                vec[idxs] += add
                diff = target_sum - vec.sum()
            else:
                idxs = np.where(vec > 0.0)[0]
                if idxs.size == 0:
                    break
                cap = vec[idxs]
                sub = np.minimum(cap, -diff / idxs.size)
                vec[idxs] -= sub
                diff = target_sum - vec.sum()
        return vec

    h = repair_sum(h)
    curr_h = h.copy()
    curr_score = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_score = curr_score

    # ----- simulated annealing parameters ----------------------------------
    init_temp = 0.02
    temp = init_temp
    decay = 0.9999
    step_init = 0.1
    min_step = 1e-5

    # ----- main loop --------------------------------------------------------
    while True:
