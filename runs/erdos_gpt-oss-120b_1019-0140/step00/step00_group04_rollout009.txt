def binary_anneal(n, time_budget):
    import numpy as np, time, random, math
    start = time.time()
    # initial h: random binary with exactly n/2 ones
    h = np.zeros(n, dtype=np.int8)
    ones_idx = random.sample(range(n), n//2)
    h[ones_idx] = 1
    best_h = h.copy()
    best_score = compute_c5(h.astype(float))
    cur_score = best_score
    # Precompute indices of ones and zeros for fast random selection
    # We'll maintain lists of indices for ones and zeros.
    ones = set(ones_idx)
    zeros = set(range(n)) - ones
    # Parameters
    init_temp = 0.01 * best_score  # maybe relative
    temp = init_temp
    # define schedule
    # we loop while time left>
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # update temperature
        # exponential cooling: temp = init_temp * 0.995**(it)
        temp = init_temp * (0.995 ** (it / 1000.0))  # adjust factor slower
        # select random i in ones, j in zeros
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # propose swap: just exchange values at i and j
        # compute new vector quickly? Instead compute new score via full correlation each time.
        # We'll compute new_h = h copy and swap bits. Score = compute_c5.
        h[i], h[j] = h[j], h[i]  # swap, now i becomes 0, j becomes 1
        new_score = compute_c5(h.astype(float))
        # decide accept
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            cur_score = new_score
            # update sets
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # optional early exit if best_score is already low (< target)
        if best_score < 0.38:
            break
    return best_h.astype(float), best_score

def binary_anneal(n, time_budget):
    import numpy as np, random, math, time
    h = np.zeros(n, dtype=np.int8)
    ones_idx = random.sample(range(n), n // 2)
    h[ones_idx] = 1
    b = 1 - h
    corr = np.correlate(h.astype(np.float64), b.astype(np.float64), mode='full')
    dx = 2.0 / n
    cur_score = np.max(corr) * dx
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    init_temp = cur_score * 0.1 if cur_score>0 else 0.001
    # maintain sets of indices for ones and zeros for selection
    ones_set = set(ones_idx)
    zeros_set = set(range(n)) - ones_set
    while time.time() - start < time_budget:
        it += 1
        # compute temperature
        temp = init_temp * (0.995 ** (it / 1000.0))
        # pick random i from ones, j from zeros
        i = random.choice(tuple(ones_set))
        j = random.choice(tuple(zeros_set))
        # delta values
        delta_h_i = -1
        delta_h_j = +1
        # apply corr update using old h and b
        # first factor contributions
        for s in range(-i, n - i):
            p = s + n - 1
            corr[p] += delta_h_i * b[i + s]
        for s in range(-j, n - j):
            p = s + n - 1
            corr[p] += delta_h_j * b[j + s]
        # second factor contributions
        for s in range(i - (n - 1), i + 1):
            p = s + n - 1
            t = i - s
            corr[p] += h[t] * (-delta_h_i)  # delta_b_i = -delta_h_i
        for s in range(j - (n - 1), j + 1):
            p = s + n - 1
            t = j - s
            corr[p] += h[t] * (-delta_h_j)
        # actually perform swap in h and b
        h[i] = 0
        h[j] = 1
        b[i] = 1
        b[j] = 0

        new_score = np.max(corr) * dx
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability
            if temp > 0:
                accept = random.random() < math.exp(-(new_score - cur_score) / temp)
        if accept:
            cur_score = new_score
            # update sets
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)

                best_h = h.copy()

            h[j] = 0

import time
import random
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    # use float for correlation to avoid integer overflow (although unlikely)
    corr = np.correlate(h.astype(np.float64), (1.0 - h).astype(np.float64), mode='full')
    return float(np.max(corr) * dx)

def _apply_swap_corr(corr: np.ndarray, h: np.ndarray, b: np.ndarray,
                     i: int, j: int, dh_i: int, dh_j: int):
    """Update full correlation array after swapping values at i and j.
    dh_i = new_h_i - old_h_i, dh_j = new_h_j - old_h_j (both +/-1 for binary swap).
    b = 1 - h (old values are used inside the update).
    """
    n = h.shape[0]
    # first factor contributions (h * b shifted)
    for s in range(-i, n - i):
        p = s + n - 1
        corr[p] += dh_i * b[i + s]
    for s in range(-j, n - j):
        p = s + n - 1
        corr[p] += dh_j * b[j + s]
    # second factor contributions (h * b where b is at i or j)
    # delta_b = -dh
    db_i = -dh_i
    db_j = -dh_j
    for s in range(i - (n - 1), i + 1):
        p = s + n - 1
        t = i - s
        corr[p] += h[t] * db_i
    for s in range(j - (n - 1), j + 1):
        p = s + n - 1
        t = j - s
        corr[p] += h[t] * db_j

def binary_anneal(n: int, time_budget: float):
    """Simulated annealing on binary vectors of length n (n even, sum = n/2)."""
    start = time.time()
    # initial random binary vector with exactly n/2 ones
    h = np.zeros(n, dtype=np.int8)
    ones = random.sample(range(n), n // 2)
    h[ones] = 1
    b = 1 - h
    corr = np.correlate(h.astype(np.int64), b.astype(np.int64), mode='full')
    dx = 2.0 / n
    cur_score = float(np.max(corr) * dx)
    best_h = h.copy()
    best_score = cur_score

    # maintain sets for fast random selection
    ones_set = set(ones)
    zeros_set = set(range(n)) - ones_set

    # temperature schedule parameters
    init_temp = max(cur_score * 0.1, 1e-4)

    it = 0
    while time.time() - start < time_budget:
        it += 1
        # exponential cooling
        temp = init_temp * (0.995 ** (it / 1000.0))

        i = random.choice(tuple(ones_set))
        j = random.choice(tuple(zeros_set))

        # proposed swap: i (1->0), j (0->1)
        dh_i = -1
        dh_j = +1

        # update correlation using old h,b
        _apply_swap_corr(corr, h, b, i, j, dh_i, dh_j)

        # perform the swap in h and b
        h[i] = 0
        h[j] = 1
        b[i] = 1
        b[j] = 0

        new_score = float(np.max(corr) * dx)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > 0.0:
                accept = random.random() < math.exp(-(new_score - cur_score) / temp)

        if accept:
            cur_score = new_score
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score < 0.38:   # early stopping threshold
                    break
        else:
            # revert swap
            # swap back values
            h[i] = 1
            h[j] = 0
            b[i] = 0
            b[j] = 1
            # revert correlation using opposite deltas
            _apply_swap_corr(corr, h, b, i, j, -dh_i, -dh_j)

        # optional early exit if time is almost out
        if time.time() - start > time_budget * 0.99:
            break

    return best_h.astype(float), best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    start_total = time.time()
    best_h = None
    best_score = float('inf')

    # Check for a parent construction

    seen = set()

        if best_score < 0.38:
