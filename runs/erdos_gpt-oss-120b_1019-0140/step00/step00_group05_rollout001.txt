import time
import math
import numpy as np

# Fast computation of the C5 bound for a binary (or real) step function.
def compute_c5(vec: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    corr = np.correlate(vec, 1 - vec, mode="full")
    return float(np.max(corr) * dx)


def generate_initial_binary(n_points: int, rng: np.random.RandomState) -> np.ndarray:
    """
    Produce an initial binary vector with exactly n_points//2 ones.
    If a global `initial_h_values` exists, interpolate it to the desired size.
    """
    half = n_points // 2
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        # linear interpolation onto the finer grid
        x_base = np.linspace(0.0, 2.0, len(base), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        interp = np.interp(x_new, x_base, base)
        # break ties slightly
        interp += 1e-9 * rng.rand(n_points)
        order = np.argsort(-interp)  # descending
        bin_vec = np.zeros(n_points, dtype=int)
        bin_vec[order[:half]] = 1
        return bin_vec
    # no parent: purely random binary vector
    bin_vec = np.zeros(n_points, dtype=int)
    bin_vec[:half] = 1
    rng.shuffle(bin_vec)
    return bin_vec


def random_binary_vector(n_points: int, rng: np.random.RandomState) -> np.ndarray:
    """Return a fresh random binary vector with half ones."""
    half = n_points // 2
    v = np.zeros(n_points, dtype=int)
    v[:half] = 1
    rng.shuffle(v)
    return v


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) with a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # Parameters
    # -------------------------------------------------------------
    n_points = 200                     # grid size (must be even)
    dx = 2.0 / n_points
    target_bound = 0.3808              # early‑stop threshold
    max_no_improve = 2000              # stagnation limit before restart
    time_margin = 0.2                  # keep a small safety margin

    # -------------------------------------------------------------
    # Initial solution
    # -------------------------------------------------------------
    current_vec = generate_initial_binary(n_points, rng)
    best_vec = current_vec.copy()
    best_score = compute_c5(best_vec, dx)

    # -------------------------------------------------------------
    # Main optimization loop (multiple random restarts)
    # -------------------------------------------------------------
    while time.time() - start_time < budget_s - time_margin:
        # First restart uses the best found so far; later restarts are random.
        if best_score <= target_bound:
            break

        if np.all(best_vec == current_vec):
            # after the first iteration we intentionally randomise
            cur = random_binary_vector(n_points, rng)
        else:
            cur = best_vec.copy()

        stagnation = 0
        while time.time() - start_time < budget_s - time_margin:
            # correlation and worst shift
            corr = np.correlate(cur, 1 - cur, mode="full")
            max_val = np.max(corr)
            cur_score = float(max_val * dx)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()
                if best_score <= target_bound:
                    break

            # index of the worst shift
            s_idx = int(np.argmax(corr))
            shift = s_idx - (n_points - 1)

            # collect candidate indices for a swap that directly reduces the
            # contribution at the worst shift
            i_cand = []
            j_cand = []
            if shift >= 0:
                limit = n_points - shift
                for i in range(limit):
                    if cur[i] == 1 and cur[i + shift] == 0:
                        i_cand.append(i)
                    if cur[i] == 0 and cur[i + shift] == 1:
                        j_cand.append(i)
            else:
                s = -shift
                start_i = s
                for i in range(start_i, n_points):
                    if cur[i] == 1 and cur[i - s] == 0:
                        i_cand.append(i)
                    if cur[i] == 0 and cur[i - s] == 1:
                        j_cand.append(i)

            improved = False
            if i_cand and j_cand:
                i_swap = int(rng.choice(i_cand))
                j_swap = int(rng.choice(j_cand))
                new_vec = cur.copy()
                new_vec[i_swap] = 0
                new_vec[j_swap] = 1
                new_score = compute_c5(new_vec, dx)
                if new_score < cur_score:
                    cur = new_vec
                    stagnation = 0
                    improved = True
                # else we keep the current vector

            if not improved:
                # fallback: random 1‑0 swap
                ones = np.flatnonzero(cur == 1)
                zeros = np.flatnonzero(cur == 0)
                if ones.size == 0 or zeros.size == 0:
                    break
                i_swap = int(rng.choice(ones))
                j_swap = int(rng.choice(zeros))
                new_vec = cur.copy()
                new_vec[i_swap] = 0
                new_vec[j_swap] = 1
                new_score = compute_c5(new_vec, dx)
                if new_score < cur_score:
                    cur = new_vec
                    stagnation = 0
                else:
                    stagnation += 1
                    if stagnation >= max_no_improve:
                        break

        if best_score <= target_bound:
            break

    # -------------------------------------------------------------
    # Final bound (use the provided evaluator if it exists)
    # -------------------------------------------------------------

        final_bound = float(evaluate_erdos_solution(best_vec.astype(float)))
    except Exception:
        pass

    return best_vec.astype(float), final_bound, n_points
