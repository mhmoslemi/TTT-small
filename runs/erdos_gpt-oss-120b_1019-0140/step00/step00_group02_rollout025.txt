            if temp > 0:
                prob = math.exp(-(new_c - cur_c) / temp)
                if random.random() < prob:
                    accept = True

        if accept:
            cur_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = h_cur.copy()
                # early exit if we beat the target record
                if best_c <= 0.3808:
                    break
        else:
            # revert the move
            if move_type == 'targeted':
                # we moved from i_pos -> i_neg
                h_cur[i_pos] += delta
                h_cur[i_neg] -= delta
            else:  # swap
                # reverse the last swap (delta may be either direction)
                # recompute which direction happened by checking which indices changed
                # simple approach: undo by swapping again the same delta (since we stored delta locally)
                # but delta is not stored for swap; recompute using saved values
                # Instead, redo the inverse move using same logic:
                # We know we attempted either i->j or j->i direction; we can just undo by swapping the same delta
                # To be safe, recompute from the current state by moving same amount back
                # We'll simply swap the same two indices by the same delta sign
                # Since we may not have delta value here, we recalculate from the difference:
                # Use original values before move (we didn't store), but because move kept total sum,
                # we can just swap back the same indices with a tiny step that restores previous sum? 
                # Simpler: recompute cur_h as best_h (since we reject the move).
                h_cur = best_h.copy()
                cur_c = best_c
            # rejected move, keep previous state
        no_improve += 1

        # occasional random perturbation if stuck
        if no_improve > 150000:
            # jump to a random feasible point near current best
            pert = np.random.randn(N) * 0.02
            h_alt = best_h + pert
            h_alt = np.clip(h_alt, 0.0, 1.0)
            h_alt = _project_feasible(h_alt, target_sum)
            cur_c = _evaluate_c5(h_alt)
            h_cur = h_alt.copy()
            no_improve = 0

    # final projection of the best candidate to ensure exact feasibility
    best_h = _project_feasible(best_h, target_sum)
    best_c = _evaluate_c5(best_h)

    return (best_h, best_c, N)
