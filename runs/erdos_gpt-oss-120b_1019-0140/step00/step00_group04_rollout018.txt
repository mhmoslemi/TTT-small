def propose_move(h):
    n = len(h)
    i,j = np.random.choice(n, size=2, replace=False)
    # max delta possible to keep within [0,1]
    max_plus = 1 - h[i]
    max_minus = h[j]

    # So delta <= min(max_plus, max_minus)

    # choose delta uniformly in [-min(h[i], 1-h[j]), min(1-h[i], h[j])]
    max_delta = min(max_plus, max_minus)

    new_h[i] += delta
    new_h[j] -= delta
    # Now sum unchanged
    return new_h

def project_feasible(x0, total):
    import cvxpy as cp
    n = len(x0)
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - x0))
    constraints = [cp.sum(x) == total, x >= 0, x <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if x.value is not None:
        return np.clip(x.value, 0, 1)  # ensure minor errors
    else:
        # fallback: simple clipping then adjust sum by adding delta to first not at bound
        x = np.clip(x0, 0, 1)
        current_sum = np.sum(x)
        diff = total - current_sum
        # distribute diff equally among indices not at bound
        # simplistic: add diff/n to all and clip again iteratively
        # but here diff small so we just adjust first element
        for i in range(n):
            if (diff > 0 and x[i] < 1) or (diff < 0 and x[i] > 0):
                adjust = min(diff, 1 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += adjust
                diff -= adjust
                if abs(diff) < 1e-12:
                    break
        return x

def proj_bounded_sum(x, total):
    n = len(x)
    # Use bisection on lambda
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo+hi)/2
        y = np.clip(x + lam, 0, 1)
        s = y.sum()
        if s > total:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo+hi)/2, 0, 1)
    # adjust small discrepancy by adding to first non-bound
    diff = total - y.sum()
    if abs(diff) > 1e-9:
        # distribute diff onto entries not at bounds
        free = np.where((y > 1e-12) & (y < 1 - 1e-12))[0]
        if free.size > 0:
            y[free[0]] += diff
            y = np.clip(y, 0, 1)
        else:
            # adjust arbitrary element within bounds
            if diff > 0:
                idx = np.argmax(y < 1)
                if y[idx] < 1:
                    y[idx] = min(1, y[idx] + diff)
            else:
                idx = np.argmin(y > 0)
                if y[idx] > 0:
                    y[idx] = max(0, y[idx] + diff)
    return y

def project_sum_bounds(x, total):
    # bisection for lambda
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo+hi)/2
        y = np.clip(x + lam, 0, 1)
        s = y.sum()
        if s > total:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo+hi)/2, 0, 1)
    # small discrepancy fix
    diff = total - y.sum()
    if abs(diff) > 1e-9:
        # distribute to any non-boundary element
        # find idx not at bound
        idx = np.where((y > 1e-12) & (y < 1 - 1e-12))[0]
        if idx.size > 0:
            y[idx[0]] += diff
            y = np.clip(y, 0, 1)
        else:
            # just adjust first element within range
            for i in range(len(y)):
                if diff > 0 and y[i] < 1:
                    add = min(diff, 1 - y[i])
                    y[i] += add
                    diff -= add
                elif diff < 0 and y[i] > 0:
                    sub = min(-diff, y[i])
                    y[i] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return y

import time
import numpy as np
import random

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def project_sum_bounds(x, total):
    """
    Project vector x onto the hypercube [0,1]^n while enforcing sum(x)=total.
    Uses a simple bisection on an additive offset followed by a small
    redistribution of the remaining error.
    """
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > total:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # fix any tiny residual error by moving mass to/from non‑boundary entries
    diff = total - y.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - y)          # most room to increase
            add = min(diff, 1.0 - y[idx])
            y[idx] += add
            diff -= add
        else:
            idx = np.argmin(y)                # most room to decrease
            sub = min(-diff, y[idx])
            y[idx] -= sub
            diff += sub
    return y
