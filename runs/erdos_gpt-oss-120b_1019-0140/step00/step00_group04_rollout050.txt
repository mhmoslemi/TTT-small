def targeted_move(h):
    n = len(h)
    # compute correlation array once
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)  # can be negative
    if shift == 0:
        return False # can't improve
    # Determine i candidates where i and i+shift are in bounds
    if shift > 0:
        idx_i = np.arange(n - shift)
        idx_j = idx_i + shift
    else:
        shift_abs = -shift
        idx_i = np.arange(shift_abs, n)
        idx_j = idx_i - shift_abs
    # filter where hi>0 and hj<1 (so we can move from i to j)
    cond = (h[idx_i] > 1e-8) & (h[idx_j] < 1 - 1e-8)
    if not np.any(cond):
        return False
    # randomly select a pair
    candidates = np.where(cond)[0]
    choice = np.random.choice(candidates)
    i = idx_i[choice]
    j = idx_j[choice]
    # Transfer some delta
    max_transfer = min(h[i], 1 - h[j])
    if max_transfer <= 0:
        return False
    delta = max_transfer * np.random.rand() * 0.5
    h[i] -= delta
    h[j] += delta
    return True
```

            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_score = _c5_of(best_h)
    return best_h, final_score, n
