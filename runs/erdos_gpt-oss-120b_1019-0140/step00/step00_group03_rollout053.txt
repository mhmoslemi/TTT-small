def hill_climb(h, dx, budget_s, start_time):
    n = len(h)
    w = n // 2
    best_h = h.copy()
    best_score = evaluate(best_h, dx)
    cur_h = best_h.copy()
    cur_score = best_score
    # precompute list of ones and zeros indices
    ones = np.where(cur_h == 1)[0].tolist()

import numpy as np, time, random, math

def _evaluate_max_corr(h):
    # h is 1D array of 0/1 or floats
    # compute correlation with 1-h and return max * dx
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    # Determine n_points
    try:
        n_points = len(initial_h_values)
        init = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        n_points = 70
        init = None
    # dx
    dx = 2.0 / n_points
    # target bound
    target = 0.3808
    # required sum
    required_sum = n_points / 2.0
    # build initial binary vector
    if init is not None:
        h = (init >= 0.5).astype(np.int8)
        cur_sum = h.sum()
        if cur_sum != required_sum:
            diff = int(round(required_sum - cur_sum))
            if diff > 0:
                zeros = np.where(h == 0)[0]
                if len(zeros) >= diff:
                    flip = np.random.choice(zeros, diff, replace=False)
                    h[flip] = 1
            elif diff < 0:
                ones = np.where(h == 1)[0]
                flip = np.random.choice(ones, -diff, replace=False)
                h[flip] = 0
    else:
        # random binary with exactly weight
        w = int(required_sum)
        h = np.zeros(n_points, dtype=np.int8)
        h[:w] = 1
        np.random.shuffle(h)
    # ensure valid
    # evaluate initial bound
    best_h = h.copy()
    best_score = _evaluate_max_corr(best_h) * dx
    cur_h = best_h.copy()
    cur_score = best_score
    start = time.time()
    total = float(budget_s)
    # annealing parameters
    init_temp = 1.0
    no_improve = 0
    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= total - 0.05:
            break
        # temperature cooling
        T = max(1e-4, init_temp * (1.0 - elapsed / total))
        # propose swap
        i = random.randrange(n_points)
        j = random.randrange(n_points)
        if cur_h[i] == cur_h[j]:
            continue
        # apply swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = _evaluate_max_corr(cur_h) * dx
        # acceptance
        if new_score < cur_score or random.random() < math.exp((cur_score - new_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                no_improve = 0
                if best_score <= target:
                    break
            else:
                no_improve += 1
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            no_improve += 1
        # random restart if stuck
        if no_improve > 20000:
            # reinitialize cur_h randomly
            w = int(required_sum)
            cur_h = np.zeros(n_points, dtype=np.int8)
            cur_h[:w] = 1
            np.random.shuffle(cur_h)
            cur_score = _evaluate_max_corr(cur_h) * dx
            no_improve = 0
    # Convert to float array for output
    h_values = best_h.astype(np.float64)
    c5_bound = best_score
    return (h_values, c5_bound, n_points)

import numpy as np, time, random, math

def max_corr_val(h):
    return float(np.max(np.correlate(h, 1 - h, mode='full')))

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    init_vals = globals().get('initial_h_values')
    if init_vals is not None:
        init_vals = np.array(init_vals, dtype=np.float64)
        n = int(init_vals.shape[0])
    else:
        n = 70
        init_vals = None
    dx = 2.0 / n
    required_sum = n / 2.0
    # Build initial binary vector
    if init_vals is not None:
        h = (init_vals >= 0.5).astype(np.int8)
        cur_sum = float(h.sum())
        # Adjust weight
        diff = int(round(required_sum - cur_sum))
        if diff != 0:
            if diff > 0:
                zero_idx = np.where(h == 0)[0]
                if len(zero_idx) >= diff:
                    flip = np.random.choice(zero_idx, diff, replace=False)
                    h[flip] = 1
            else:
                one_idx = np.where(h == 1)[0]
                flip = np.random.choice(one_idx, -diff, replace=False)
                h[flip] = 0
    else:
        # Random binary with exact weight
        w = int(required_sum)
        h = np.zeros(n, dtype=np.int8)
        h[:w] = 1
        np.random.shuffle(h)
    best_h = h.copy()
    best_score = max_corr_val(best_h) * dx
    cur_h = best_h.copy()
    cur_score = best_score
    start = time.time()
    total = float(budget_s)
    # parameters
    start_temp = 0.5
    no_improve = 0
    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= total - 0.05:
            break
        # temperature schedule
        T = max(1e-6, start_temp * (1.0 - elapsed / total))
        # propose random swap of differing entries
        i = random.randrange(n)
        j = random.randrange(n)
        if cur_h[i] == cur_h[j]:
            continue
        # apply swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = max_corr_val(cur_h) * dx
        # acceptance test
        if new_score < cur_score or random.random() < math.exp((cur_score - new_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                no_improve = 0
                # early exit if good enough
                if best_score <= 0.3808:
                    break
            else:
                no_improve += 1
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            no_improve += 1
        # random restart if stagnant
        if no_improve > 20000:
            # new random binary
            w = int(required_sum)
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[:w] = 1
            np.random.shuffle(cur_h)
            cur_score = max_corr_val(cur_h) * dx
            no_improve = 0
    # final conversion to float array
    h_values = best_h.astype(np.float64)
    return (h_values, best_score, n)

import numpy as np
import time
import random
import math

def _max_correlation(h):
    """Return the maximum correlation value of h with 1‑h."""
    # np.correlate implements the required overlap without padding
    return float(np.max(np.correlate(h, 1 - h, mode='full')))


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with small C₅ bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with values in [0,1]
        c5_bound : float, the achieved C₅ bound (max overlap integral)
        n_points : int, the number of discretisation points used
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector (if provided)
    # ------------------------------------------------------------------
    init_raw = globals().get('initial_h_values')
    if init_raw is not None:
        init_arr = np.array(init_raw, dtype=np.float64)
        n = int(init_arr.shape[0])
    else:
        n = 70
        init_arr = None

    dx = 2.0 / n
    required_sum = n / 2.0           # exact integral condition

    # ------------------------------------------------------------------
    # Build a binary starting point respecting the sum constraint
    # ------------------------------------------------------------------
    if init_arr is not None:
        # coarse binarisation, then fix weight
        h = (init_arr >= 0.5).astype(np.int8)
        cur = int(h.sum())
        diff = int(round(required_sum - cur))
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) >= diff:
                pick = np.random.choice(zeros, diff, replace=False)
                h[pick] = 1
        elif diff < 0:
            ones = np.where(h == 1)[0]
            pick = np.random.choice(ones, -diff, replace=False)
            h[pick] = 0
    else:
        # random balanced binary vector
        w = int(required_sum)
        h = np.zeros(n, dtype=np.int8)
        h[:w] = 1
        np.random.shuffle(h)

    # ------------------------------------------------------------------
    # Initialise bookkeeping
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _max_correlation(best_h) * dx
    cur_h = best_h.copy()
    cur_score = best_score

    start_time = time.time()
    total_time = float(budget_s)

    start_temp = 0.5          # starting temperature for simulated annealing
    no_improve = 0
    target_bound = 0.3808

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_time - 0.05:
            break

        # linear cooling schedule
        T = max(1e-6, start_temp * (1.0 - elapsed / total_time))

        # propose a swap of distinct values
        i = random.randrange(n)
        j = random.randrange(n)
        if cur_h[i] == cur_h[j]:
            continue

        # execute swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # evaluate new bound
        new_score = _max_correlation(cur_h) * dx

        # Metropolis acceptance
        if new_score < cur_score or random.random() < math.exp((cur_score - new_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                no_improve = 0
                if best_score <= target_bound:
                    break
            else:
                no_improve += 1
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            no_improve += 1

        # occasional restart to escape stagnation
        if no_improve > 20000:
            w = int(required_sum)
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[:w] = 1
            np.random.shuffle(cur_h)
            cur_score = _max_correlation(cur_h) * dx
            no_improve = 0

    # ------------------------------------------------------------------
    # Prepare return values
    # ------------------------------------------------------------------
    h_values = best_h.astype(np.float64)      # ensure type is float for the harness
    c5_bound = best_score
    return (h_values, c5_bound, n)
