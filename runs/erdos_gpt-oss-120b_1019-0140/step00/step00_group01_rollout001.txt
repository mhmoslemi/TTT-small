import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimization for the Erdős minimum overlap problem.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Setup problem dimensions and initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        # fallback: moderate size if no parent is provided
        n = 200
        # start with half 1, half 0 and a possible 0.5 for odd n
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        if n % 2 == 1:
            h[half] = 0.5
        np.random.shuffle(h)
    target_sum = n / 2.0
    # ensure exact sum while staying in [0,1]
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple proportional scaling with clipping
        h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        # final small correction if needed
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            for idx in range(n):
                if diff > 0 and h[idx] < 1.0:
                    inc = min(diff, 1.0 - h[idx])
                    h[idx] += inc
                    diff -= inc
                elif diff < 0 and h[idx] > 0.0:
                    dec = min(-diff, h[idx])
                    h[idx] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break

    dx = 2.0 / n

    def compute_c5(arr):
        # correlation of arr with (1-arr); max multiplied by dx
        corr = np.correlate(arr, 1.0 - arr, mode='full')
        return float(corr.max() * dx)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Helper: generate a random feasible vector (used for occasional jumps)
    # ------------------------------------------------------------------
    def random_feasible():
        v = np.random.rand(n)
        v *= target_sum / v.sum()
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - v.sum()
        if abs(diff) > 1e-12:

                    diff -= inc

                    v[i] -= dec
