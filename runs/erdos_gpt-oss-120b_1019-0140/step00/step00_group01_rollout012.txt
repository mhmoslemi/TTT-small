def repair_h(x):
    x = np.clip(x, 0, 1)
    target = n/2
    s = np.sum(x)
    eps = 1e-12
    max_iter = 10 * n
    iter = 0
    while abs(s - target) > 1e-9 and iter < max_iter:
        if s < target:
            deficit = target - s
            idx = np.where(x < 1 - eps)[0]
            if idx.size == 0:
                break
            inc = deficit / idx.size
            # ensure inc doesn't exceed capacity
            inc_cap = np.minimum(1 - x[idx], inc)
            # For variable inc, might need to distribute leftover if some reach capacity.
            # Simple approach: for each i in idx, add min(inc, capacity)
            # Compute total added = sum(inc_cap)

            x[idx] += inc_cap
            s += total_added

            excess = s - target
            idx = np.where(x > eps)[0]
            if idx.size == 0:

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution

    start = time.time()
    np.random.seed(seed)

    # Determine n
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.size
    except NameError:
        # no initial_h_values, choose n=80 as default (multiple of 2)
        n = 80
        init_vec = None

    target_sum = n / 2
    dx = 2.0 / n

    # Define repair function using closure over n, target_sum
    def repair_h(x):
        # Clip x to [0,1]
        h = np.clip(x, 0.0, 1.0)
        s = np.sum(h)
        eps = 1e-12
        max_iter = 10 * n
        it = 0
        # iterative adjust
        while abs(s - target_sum) > 1e-9 and it < max_iter:
            if s < target_sum:
                deficit = target_sum - s
                idx = np.where(h < 1 - eps)[0]
                if idx.size == 0:
                    break
                inc = deficit / idx.size
                # compute possible increase for each idx without exceeding 1
                inc_cap = np.minimum(1.0 - h[idx], inc)
                total_added = np.sum(inc_cap)
                h[idx] += inc_cap
                s += total_added
            else:  # s > target_sum
                excess = s - target_sum
                idx = np.where(h > eps)[0]
                if idx.size == 0:
                    break
                dec = excess / idx.size
                dec_cap = np.minimum(h[idx], dec)
                total_sub = np.sum(dec_cap)
                h[idx] -= dec_cap
                s -= total_sub
            it += 1
        # final clip to assure bounds
        h = np.clip(h, 0.0, 1.0)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        # Convert to feasible h
        h = repair_h(x)
        # Compute C5
        return compute_c5(h)

    # DE settings
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Determine popsize
    popsize = max(15, int(5 + n / 10))
    # Create initial population
    init_pop = np.random.rand(popsize, n)
    if init_vec is not None:
        # Ensure init_vec feasible (repair)
        init_pop[0] = repair_h(init_vec)
    # Callback to enforce time budget
    def de_callback(xk, convergence):
        if time.time() - start > budget_s * 0.95:
            return True  # stop
        return False

    # Run DE
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=popsize,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=False,
        init=init_pop,
        callback=de_callback,
        updating='deferred',  # less overhead
        disp=False,
        # set workers=1 (default)
    )
    # final solution
    best_h = repair_h(result.x)
    best_c5 = compute_c5(best_h)

    # ensure integral constraint satisfied; adjust if needed with final repair
    # Confirm sum approx target
    # Possibly tiny violations exist; we could renormalize
    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-6:
        # simple scaling with re-clip: distribute proportionally without exceeding 1
        # use iterative scaling
        best_h = repair_h(best_h)
        best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), n

import numpy as np
import time
from scipy.optimize import differential_evolution

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_h(x, target_sum, eps=1e-12, max_iter_factor=10):
    h = np.clip(x, 0.0, 1.0)
    s = np.sum(h)
    max_iter = int(max_iter_factor * len(x))
    it = 0
    while abs(s - target_sum) > 1e-9 and it < max_iter:
        if s < target_sum:
            deficit = target_sum - s
            idx = np.where(h < 1.0 - eps)[0]
            if idx.size == 0:
                break
            inc = deficit / idx.size
            inc_cap = np.minimum(1.0 - h[idx], inc)
            total_added = np.sum(inc_cap)
            h[idx] += inc_cap
            s += total_added
        else:
            excess = s - target_sum
            idx = np.where(h > eps)[0]
            if idx.size == 0:
                break
            dec = excess / idx.size
            dec_cap = np.minimum(h[idx], dec)
            total_sub = np.sum(dec_cap)
            h[idx] -= dec_cap
            s -= total_sub
        it += 1
    # final clipping, may be slight off target but okay.
    h = np.clip(h, 0.0, 1.0)
    return h

class Objective:
    def __init__(self, target_sum, dx):
        self.target_sum = target_sum
        self.dx = dx
    def __call__(self, x):

def local_search(h, current_score, target_sum, dx, start_time, budget_s):
    # perform simple random moves until time runs out

    eps = 1e-12
    # Precompute indices where h is not at bounds to speed up

        # we can vectorize random pick via np.random.choice with condition

        idx_neg = np.where(h < 1.0 - eps)[0]
        if idx_pos.size == 0 or idx_neg.size == 0:

import numpy as np
import time

def compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def repair_h(x, target_sum, eps=1e-12, max_iter_factor=10):
    """Project x onto the feasible region: entries in [0,1] and sum = target_sum."""
    h = np.clip(x, 0.0, 1.0)
    s = np.sum(h)
    max_iter = int(max_iter_factor * len(x))
    it = 0
    while abs(s - target_sum) > 1e-9 and it < max_iter:
        if s < target_sum:
            deficit = target_sum - s
            idx = np.where(h < 1.0 - eps)[0]
            if idx.size == 0:
                break
            inc = deficit / idx.size
            inc_cap = np.minimum(1.0 - h[idx], inc)
            added = np.sum(inc_cap)
            h[idx] += inc_cap
            s += added
        else:
            excess = s - target_sum
            idx = np.where(h > eps)[0]
            if idx.size == 0:
                break
            dec = excess / idx.size
            dec_cap = np.minimum(h[idx], dec)
            sub = np.sum(dec_cap)
            h[idx] -= dec_cap
            s -= sub
        it += 1
    return np.clip(h, 0.0, 1.0)


class Objective:
    """Callable used by differential evolution."""

    def __init__(self, target_sum, dx):
        self.target_sum = target_sum
        self.dx = dx

    def __call__(self, x):
        h = repair_h(x, self.target_sum)
        return compute_c5(h, self.dx)


class DECallback:
    """Stops DE when the allotted budget is almost exhausted."""

    def __init__(self, start_time, budget_s):
        self.start_time = start_time
        self.budget_s = budget_s

    def __call__(self, xk, convergence):
        if time.time() - self.start_time > self.budget_s * 0.95:
            return True
        return False


def local_search(h, current_score, target_sum, dx, start_time, budget_s):
    """Simple random transfer hill‑climbing within the remaining time."""
    n = len(h)
    eps = 1e-12
    while time.time() - start_time < budget_s * 0.95:
        idx_pos = np.where(h > eps)[0]
        idx_neg = np.where(h < 1.0 - eps)[0]
        if idx_pos.size == 0 or idx_neg.size == 0:
            break
        i = np.random.choice(idx_pos)
        j = np.random.choice(idx_neg)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        transfer = max_transfer * np.random.rand()
        if transfer < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= transfer
        h_new[j] += transfer
        c5 = compute_c5(h_new, dx)
        if c5 < current_score:
            h = h_new
            current_score = c5
    return h, current_score
