import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C₅ for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of y onto the set:
        0 <= x_i <= 1  and  sum(x) = n/2
    using a binary search for the Lagrange multiplier.
    """
    n = y.shape[0]
    target = n / 2.0
    lo, hi = -2.0, 2.0  # bounds guarantee sum at lo = n, at hi = 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction to hit the sum exactly (numerical safety)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x


def initial_solution(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible starting vector.
    Preference: use the provided global `initial_h_values` if present,
    otherwise try a modest sinusoidal sweep and a few random tries.
    """
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0.shape[0] != n:
            # resize by interpolation if needed
            x_old = np.linspace(0.0, 2.0, h0.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        return project_feasible(h0)
    except NameError:
        # sinusoid grid search (coarse)
        best_h = np.full(n, 0.5)
        best_val = compute_c5(best_h)
        xs = np.arange(n)
        for a in np.linspace(0.0, 0.5, 21):
            for phi in np.linspace(0.0, 2 * np.pi, 36, endpoint=False):
                h = 0.5 + a * np.sin(2 * np.pi * xs / n + phi)
                h = project_feasible(h)
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
        # a handful of random starts
        for _ in range(500):
            h = rng.uniform(0.0, 1.0, size=n)
            h = project_feasible(h)
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing C₅.
    Returns (list_of_h, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    try:
        n = len(initial_h_values)  # may raise NameError
    except NameError:
        n = 45

    # Initial feasible vector
    h = initial_solution(n, rng)
    cur_c5 = compute_c5(h)

    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated annealing parameters
    T_start = max(0.01, cur_c5)
    T_end = 1e-5
    max_iter = 200_000  # generous upper bound
    sigma = 0.20
    iter_count = 0
    no_improve = 0

    while True:
        # Time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        iter_count += 1
        # Temperature schedule (exponential cooling)
        T = T_start * ((T_end / T_start) ** (iter_count / max_iter))
        # Propose a Gaussian perturbation
        proposal = h + rng.normal(scale=sigma, size=n)
        proposal = project_feasible(proposal)
        prop_c5 = compute_c5(proposal)

        # Metropolis acceptance
        if prop_c5 < cur_c5 or rng.random() < math.exp(-(prop_c5 - cur_c5) / max(T, 1e-12)):
            h = proposal
            cur_c5 = prop_c5
            if prop_c5 < best_c5:
                best_c5 = prop_c5
                best_h = proposal.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive step size
        if iter_count % 5000 == 0:
            sigma = max(sigma * 0.9, 1e-4)

        # Early exit if target achieved
        if best_c5 <= 0.3808:
            break

        # Occasional restart if stagnating for a long stretch
        if no_improve > 20000:
            # restart from uniform 0.5 (feasible) to keep exploration alive
            h = np.full(n, 0.5)
            cur_c5 = compute_c5(h)
            no_improve = 0

    # Return solution
    return best_h.tolist(), float(best_c5), int(n)
