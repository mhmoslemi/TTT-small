def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")

    if abs(diff) < 1e-12:
        return h
    # try to fix by modifying one coordinate that stays in [0,1]
    for i in range(h.size):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h
    # fallback: scale and then adjust a single entry
    h *= target / cur
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # smallest index adjustment (diff is tiny, should stay inside bounds)
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (as list), its C5 bound, and number of points."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose number of points and initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float, copy=True)
        n = h.size
        target_sum = n / 2.0
        # ensure the sum constraint holds exactly
        h = _project_sum(h, target_sum)
    else:
        # default grid size (even for easy binary construction)
        n = 200
        target_sum = n / 2.0
        # binary start: exactly n/2 ones (or half+0.5 for odd)
        ones = int(math.floor(target_sum))
        frac = target_sum - ones
        h = np.zeros(n, dtype=float)
        idx = rng.permutation(n)
        h[idx[:ones]] = 1.0
        if frac > 1e-12:
            h[idx[ones]] = frac

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic local search (swap moves, greedy acceptance,
    # occasional random restart)
    # ------------------------------------------------------------------
    current_h = h.copy()
    current_score = best_score

    # parameters
    max_no_improve = 50000          # steps without improvement before restart
    restart_interval = 200000      # hard cap on steps per restart
    no_improve = 0
    steps_since_restart = 0

    # main loop respecting the time budget
    while time.time() - start < budget_s * 0.94:
        # propose a swap of two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        while i == j:
            j = rng.integers(0, n, endpoint=False)

        # if values are identical, the swap does nothing – skip
        if np.isclose(current_h[i], current_h[j]):
            continue

        # perform in‑place swap
        current_h[i], current_h[j] = current_h[j], current_h[i]

        new_score = _c5_score(current_h)

        if new_score < current_score:
            # accept improvement
            current_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # revert swap (greedy strategy)
            current_h[i], current_h[j] = current_h[j], current_h[i]
            no_improve += 1

        steps_since_restart += 1

        # restart if stagnated or after many steps
        if no_improve >= max_no_improve or steps_since_restart >= restart_interval:
            # re‑initialise a fresh binary configuration preserving the best found
            half = int(math.floor(target_sum))
            frac = target_sum - half
            current_h = np.zeros(n, dtype=float)
            perm = rng.permutation(n)
            current_h[perm[:half]] = 1.0
            if frac > 1e-12:
                current_h[perm[half]] = frac
            current_score = _c5_score(current_h)
            no_improve = 0
            steps_since_restart = 0

    # final verification (re‑compute to avoid accumulated floating error)
    best_score = _c5_score(best_h)

    return (best_h.tolist(), float(best_score), n)
