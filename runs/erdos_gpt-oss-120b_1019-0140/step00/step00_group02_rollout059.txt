def propose_gradient_move(h):
    # compute worst shift
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)
    # compute derivative
    der = np.zeros(n)
    if k >= 0:
        if k < n:
            der[:n - k] += (1 - h[k:])
            der[k:] -= h[:n - k]
    else:
        kk = -k
        der[kk:] += (1 - h[:n - kk])
        der[:n - kk] -= h[kk:]
    # find positive and negative indices
    pos_idx = np.where(der > 1e-12)[0]
    neg_idx = np.where(der < -1e-12)[0]
    if len(pos_idx) == 0 or len(neg_idx) == 0:
         return None
    i = np.random.choice(pos_idx)
    j = np.random.choice(neg_idx)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
         return None
    delta = random.uniform(0, max_delta)  # maybe scale to 0.5 * max?
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 overlap value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlate h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def propose_random_move(h: np.ndarray) -> np.ndarray | None:
    """Randomly transfer a small amount of mass between two indices while keeping constraints."""
    n = h.shape[0]
    # try a few attempts to find a feasible pair
    for _ in range(5):
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        # allowed delta so that new values stay inside [0,1]
        lower = max(-hi, -(1.0 - hj))
        upper = min(1.0 - hi, hj)
        if lower < upper:
            delta = random.uniform(lower, upper)
            new_h = h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            return new_h
    return None


def propose_gradient_move(h: np.ndarray) -> np.ndarray | None:
    """Move mass from a positive-derivative index to a negative-derivative index for the worst shift."""
    n = h.shape[0]
    # locate the shift where the overlap is maximal (worst shift)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # signed shift
    # derivative of the worst-shift overlap with respect to each entry
    der = np.zeros(n, dtype=np.float64)
    if k >= 0:
        if k < n:
            der[: n - k] += (1.0 - h[k:])
            der[k:] -= h[: n - k]
    else:
        kk = -k
        if kk < n:
            der[kk:] += (1.0 - h[: n - kk])
            der[: n - kk] -= h[kk:]
    pos = np.where(der > 1e-12)[0]
    neg = np.where(der < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i = np.random.choice(pos)
    j = np.random.choice(neg)
    max_delta = min(h[i], 1.0 - h[j])   # amount we can move from i to j
    if max_delta <= 0.0:
        return None
    delta = random.uniform(0.0, max_delta)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def random_feasible_vector(n: int) -> np.ndarray:
    """Create a simple feasible vector with entries in {0,1, fractional} and sum = n/2."""
    target = n / 2.0
    h = np.zeros(n, dtype=np.float64)
    full = int(target)
    if full > 0:
        h[:full] = 1.0
    frac = target - full
    if frac > 1e-12:
        h[full] = frac
    np.random.shuffle(h)
    return h


def simulated_annealing(h_init: np.ndarray, end_time: float, start_T: float = 0.2) -> tuple[np.ndarray, float]:
    """Run a simple SA on h_init until the supplied wall‑clock time is reached."""
    cur = h_init.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T = start_T
    cooling = 0.9997
    T_min = 1e-7
    not_improved = 0
    max_not_improved = 5000

    while time.time() < end_time:
        # choose move type
        if random.random() < 0.7:
            new = propose_gradient_move(cur) or propose_random_move(cur)
        else:
            new = propose_random_move(cur) or propose_gradient_move(cur)

        if new is None:
            # fallback: random restart
            cur = random_feasible_vector(n)
            cur_score = compute_c5(cur)
            continue

        new_score = compute_c5(new)
        delta = new_score - cur_score
        if delta < 0 or (T > 0 and random.random() < math.exp(-delta / T)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                not_improved = 0
            else:
                not_improved += 1
        else:
            not_improved += 1

        T = max(T_min, T * cooling)

        if not_improved >= max_not_improved:
            # perturb the current best a little to escape flat regions
            cur = best.copy()
            cur = propose_random_move(cur) or cur
            cur_score = compute_c5(cur)
            not_improved = 0

    return best, best_score

    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    end = start + budget_s

    # --- initialise starting vector ---------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    except NameError:
        n = 45
        h0 = random_feasible_vector(n)

    # enforce exact sum = n/2 (tiny numeric adjustments)
    target = n / 2.0
    cur_sum = h0.sum()
    if abs(cur_sum - target) > 1e-12 and cur_sum > 0:
        h0 = np.clip(h0 * (target / cur_sum), 0.0, 1.0)
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)
            h0 *= target / h0.sum()

    # --- Phase 1: optimise at original resolution --------------------
    phase1_end = start + budget_s * 0.5
    best_h, best_score = simulated_annealing(h0, phase1_end, start_T=0.2)

    # --- Phase 2: optionally increase resolution and continue -------
    remaining = end - time.time()
    if remaining > budget_s * 0.1 and n * 2 <= 500:
        # upscale by simple repetition (keeps feasibility)
        up_h = np.repeat(best_h, 2)
        up_end = end
        best_h, best_score = simulated_annealing(up_h, up_end, start_T=0.2)
        n = best_h.shape[0]

    # final sanity check (enforce constraints one last time)
    best_h = np.clip(best_h, 0.0, 1.0)
    # tiny correction to sum if needed
    diff = (n / 2.0) - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    h_vals = best_h.tolist()
    c5_bound = float(best_score)
    return h_vals, c5_bound, n
