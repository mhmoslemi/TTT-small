import time, math, random
import numpy as np

def _c5_value(vec, dx):
    """fallback C5 evaluator using numpy correlate."""
    return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------------------------------------------------
    # Setup RNG
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------
    # Initial vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        # default size (even, ≥80) – 200 gives a decent trade‑off
        n = 200
        h = np.full(n, 0.5, dtype=float)

    N = len(h)
    dx = 2.0 / N

    # ------------------------------------------------------------
    # C5 evaluator (use the provided helper if present)
    try:
        _eval = evaluate_erdos_solution   # type: ignore
    except Exception:   # pragma: no cover
        def _eval(v):
            return _c5_value(v, dx)

    # ------------------------------------------------------------
    # Initial score
    best_h = h.copy()
    best_score = _eval(best_h)

    # ------------------------------------------------------------
    # Simple simulated‑annealing / hill‑climbing loop
    start = time.time()
    # temperature linearly decreasing to ~0.001
    while time.time() - start < budget_s * 0.95:
        # propose a move: transfer a delta between two coordinates
        i, j = random.sample(range(N), 2)
        # compute admissible delta (keep both in [0,1])
        max_inc_i = 1.0 - h[i]
        max_dec_i = h[i]
        max_inc_j = 1.0 - h[j]
        max_dec_j = h[j]

        # random signed delta limited by both sides
        delta = random.uniform(-max_dec_i, max_inc_i)
        # adjust opposite side to keep total sum unchanged
        if delta > 0:
            delta = min(delta, max_dec_j)   # can only take from j
        else:
            delta = -min(-delta, max_inc_j) # can only give to j
        if abs(delta) < 1e-12:
            continue

        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        # evaluate
        new_score = _eval(new_h)

        # acceptance rule
        if new_score < best_score:
            best_score = new_score
            best_h = new_h.copy()
            h = new_h
        else:
            # simulated annealing probability
            elapsed = time.time() - start
            temp = max(0.001, 1.0 - elapsed / budget_s)
            # scale chosen so that typical differences (~0.01) are accepted early
            prob = math.exp(-(new_score - best_score) / (temp * 0.01 + 1e-12))
            if random.random() < prob:
                h = new_h

    # ------------------------------------------------------------
    # Final evaluation (ensure the best one is reported)
    final_score = _eval(best_h)

    # Return as required: (h_values, c5_bound, n_points)
    return best_h, final_score, N
