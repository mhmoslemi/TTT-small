import time
import numpy as np

def _compute_c5(h):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx), corr


def _max_shift_and_gradient(h):
    """Find shift giving maximal correlation and its sub‑gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)               # shift ∈ [-(n-1), n-1]
    grad = np.zeros_like(h)

    if shift >= 0:
        # term1 = 1 - h[i+shift] for i+shift < n
        grad[: n - shift] += 1 - h[shift:]
        # term2 = -h[i-shift] for i-shift >= 0
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] += 1 - h[: n - s]       # i-s ≥ 0
        grad[: n - s] -= h[s:]           # i+s < n
    return shift, float(corr[idx] * (2.0 / n)), grad


def run(seed=42, budget_s=1100, **kwargs):
