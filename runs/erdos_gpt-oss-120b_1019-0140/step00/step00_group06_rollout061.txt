import time
import numpy as np
from scipy.optimize import minimize

# --- constants ---------------------------------------------------------------
ALPHA = 15.0          # smoothing for max approximation
PENALTY_WEIGHT = 1e4 # penalty for violating the integral constraint

# --- helper functions --------------------------------------------------------
def project_onto_constraints(v):
    """Clip v to [0,1] and adjust uniformly so that sum(v)=len(v)/2."""
    n = v.shape[0]
    target = n * 0.5
    v = np.clip(v, 0.0, 1.0)

    # binary search on a uniform shift λ
    lo, hi = -1.0, 1.0
    # expand bounds until they definitely bracket the solution
    while np.sum(np.clip(v + lo, 0.0, 1.0)) > target:
        lo *= 2.0
    while np.sum(np.clip(v + hi, 0.0, 1.0)) < target:
        hi *= 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        s = np.sum(np.clip(v + lam, 0.0, 1.0))
        if s > target:
            hi = lam
        else:
            lo = lam
    v = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    # tiny correction for rounding errors
    diff = target - np.sum(v)
    if abs(diff) > 1e-12:
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        if np.any(free):
            v[free] += diff / np.sum(free)
            v = np.clip(v, 0.0, 1.0)
        else:
            v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v

def smooth_max_corr(h, alpha=ALPHA):
    """Smooth approximation of max_{k} ∫ h(x)(1-h(x+k)) dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    smooth = (a_max + np.log(np.exp(a - a_max).sum())) / alpha
    return smooth * dx

def compute_c5(h):
    """Exact C5 value using the max correlation (no smoothing)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def penalized_obj(h):
    """Objective for the optimizer: smooth max + penalty for integral."""
    smooth = smooth_max_corr(h, alpha=ALPHA)
    target = h.shape[0] * 0.5
    penalty = PENALTY_WEIGHT * (np.sum(h) - target) ** 2
    return smooth + penalty

def optimize_h(start_vec, max_iter=250):
    """Run L‑BFGS‑B from start_vec and project the result."""
    n = start_vec.shape[0]
    bounds = [(0.0, 1.0)] * n
    res = minimize(penalized_obj, start_vec, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'disp': False})
    return project_onto_constraints(res.x)

def interpolate_vector(old_h, new_n):
    """Linear interpolation of old_h (length m) to length new_n."""
    m = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h, left=old_h[0], right=old_h[-1])
    return project_onto_constraints(new_h)

def local_refine(best_h, time_budget, rng):
    """Simple stochastic hill‑climbing using remaining time."""

    n = best_h.shape[0]
    best_val = compute_c5(best_h)

        val = compute_c5(cand)
        if val < best_val:
