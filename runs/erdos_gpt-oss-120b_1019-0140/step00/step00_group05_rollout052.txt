def mutate_swap(h, rng, max_step=0.1):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # compute bounds for delta:
    delta_low = max(-max_step, -h[i], h[j] - 1)   # lower bound ensures h[i]+delta >=0, h[j]-delta <=1
    delta_high = min(max_step, 1 - h[i], h[j])   # upper bound ensures h[i]+delta <=1, h[j]-delta >=0
    if delta_low > delta_high:
        return h.copy()
    delta = rng.uniform(delta_low, delta_high)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    # no need to repair because delta respects bounds, sum unchanged.
    return new

def repair(v, target):
    v = np.clip(v, 0.0, 1.0)
    # adjust sum to target
    diff = target - v.sum()
    # Perform iterative distribution
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            add = min(diff, (1.0 - v[idx]).sum())
            share = add / idx.size
            v[idx] += share
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            sub = min(-diff, v[idx].sum())
            share = sub / idx.size
            v[idx] -= share
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    # final tiny adjustment
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        # adjust one suitable entry
        for i in range(len(v)):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break
    return v

def mutate_swap(h, rng, max_step=0.2):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # compute allowed delta range
    # delta added to h[i], sub from h[j]
    # Let's allow delta to be both positive or negative; for positive delta, we increase h[i] and decrease h[j]; keep sum unchanged
    # allowed range:
    # lower bound: -min(max_step, h[i], 1 - h[j])? Actually negative delta means decreasing h[i] and increasing h[j];
    delta_low = max(-max_step, -h[i], h[j] - 1.0)
    delta_high = min(max_step, 1.0 - h[i], h[j])
    if delta_low > delta_high:
        # cannot move, return copy
        return h.copy()
    delta = rng.uniform(delta_low, delta_high)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    return new

    best_score, _ = evaluate_score(best_h)

import numpy as np
import time
import math

def _repair(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            add = min(diff, (1.0 - v[idx]).sum())
            share = add / idx.size
            v[idx] += share
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            sub = min(-diff, v[idx].sum())
            share = sub / idx.size
            v[idx] -= share
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        for i in range(v.size):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break
    return v

def _evaluate_score(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def _mutate_swap(h, rng, max_step=0.2):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    lower = max(-max_step, -h[i], h[j] - 1.0)
    upper = min(max_step, 1.0 - h[i], h[j])
    if lower > upper:
        return h.copy()
    delta = rng.uniform(lower, upper)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    return new

def _targeted_swap(h, rng, target, corr):
    n = h.size
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    if lag == 0:
        return _mutate_swap(h, rng, max_step=0.2)
    if lag > 0:
        i = rng.integers(0, n - lag)
        j = i + lag
    else:
        i = rng.integers(-lag, n)
        j = i + lag
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta > 1e-12:
        delta = rng.uniform(0.0, max_delta * 0.5)
        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new
    max_delta_rev = min(h[j], 1.0 - h[i])
    if max_delta_rev > 1e-12:
        delta = rng.uniform(0.0, max_delta_rev * 0.5)
        new = h.copy()
        new[j] -= delta
        new[i] += delta
        return new
    return _mutate_swap(h, rng, max_step=0.2)
