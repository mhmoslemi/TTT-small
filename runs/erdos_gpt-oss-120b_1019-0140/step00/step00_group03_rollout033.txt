def compute_c5(arr, dx):
    return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

def evaluate(arr, dx):
    return compute_c5(arr.astype(float), dx)

def repair(arr, target_sum):
    # arr is int 0/1
    current = arr.sum()
    diff = int(target_sum - current)
    if diff == 0:
        return arr
    # if diff > 0, need add ones: flip diff zeros to ones
    # if diff < 0, need remove ones: flip -diff ones to zero
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if len(zeros) < diff:
            # Should not happen
            diff = len(zeros)
        flip = np.random.choice(zeros, size=diff, replace=False)
        arr[flip] = 1
    else:
        ones = np.where(arr == 1)[0]
        if len(ones) < -diff:
            diff = -len(ones)
        flip = np.random.choice(ones, size=-diff, replace=False)
        arr[flip] = 0
    return arr

def crossover(parent1, parent2):
    point = random.randint(1, len(parent1)-1)  # choose a cut point not at ends
    child = np.concatenate([parent1[:point], parent2[point:]]).copy()
    return child

def mutate(arr):
    # swap a random 1 with a random 0
    ones = np.where(arr == 1)[0]
    zeros = np.where(arr == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return arr
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    arr[i], arr[j] = arr[j], arr[i]
    return arr
```

def hill_climb(h, dx, target_sum, max_iter=10000):
    # try random swaps that improve C5; stop after max no-improve steps
    best = h.copy()
    best_c5 = evaluate(best, dx)
    no_improve = 0
    for _ in range(max_iter):
        # propose a swap: randomly choose a 1 and a 0 to swap
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = evaluate(cand, dx)
        if cand_c5 < best_c5 - 1e-12:  # improvement
            best = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 2000:  # stop if stuck
            break
    return best, best_c5
