def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # Get initial h if available
    global initial_h_values
    try:
        init = initial_h_values
    except NameError:
        init = None
    if init is not None:
        h = np.array(init, dtype=float)
        N = h.shape[0]
        # Ensure sum constraint, maybe adjust due to floating errors
        target_sum = N / 2
        diff = target_sum - h.sum()
        if abs(diff) > 1e-9:
            # adjust first element to fix sum within bounds
            h[0] += diff
    else:
        N = 80
        # random binary with N/2 ones
        h = np.zeros(N, dtype=float)
        h[:N//2] = 1.0
        rng.shuffle(h)
    # compute C5 function
    dx = 2.0 / N
    def compute_c5(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        return np.max(corr) * dx
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # Precompute indices of ones and zeros for swapping
    # We'll do random swaps
    # We'll maintain list of positions with ones and zeros for convenience, update after each swap
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    # Define schedule
    max_iter = 10**9  # effectively infinite, break by time
    # simple annealing schedule: temp starts at some value, decays log linearly
    # We'll compute temp = max(1e-4, init_temp * (budget_remaining / total_budget))
    init_temp = 0.1
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s:
            break
        # Choose random i from ones, j from zeros
        if not ones or not zeros:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # Swap
        # We'll create a copy maybe only modify and revert if not accepted
        # We'll compute new vector by swapping values at i, j
        # Since values are binary, swap 1 and 0.
        h[i], h[j] = h[j], h[i]
        # update sets
        ones.discard(i)
        zeros.add(i)
        zeros.discard(j)
        ones.add(j)
        # evaluate
        new_c5 = compute_c5(h)
        # acceptance
        temp = max(1e-6, init_temp * (budget_s - elapsed) / budget_s)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            cur_c5 = new_c5
            # maybe update best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
            ones.discard(j)
        # small occasional re-calc of ones/zeros after many swaps to avoid drift due to floating errors
        # Not needed for binary.
    # after loop, return best solution
    # Ensure best_h satisfies constraints (sum approx N/2). if necessary adjust small rounding errors
    # Recompute bound
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, N
