import numpy as np
import math
import time

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given discrete h."""
    n = h.size
    dx = 2.0 / n
    # linear correlation (no wrap)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_gradient(h: np.ndarray) -> np.ndarray:
    """Subgradient of C5: use the shift achieving the max correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr) - (n - 1))  # shift index, can be negative
    grad = np.zeros_like(h)

    if k >= 0:
        # term from h[i] * (1 - h[i+k])
        grad[: n - k] += dx * (1.0 - h[k:])
        # term from -(h[i-k] * h[i])
        grad[k:] -= dx * h[: n - k]
    else:
        kk = -k
        grad[kk:] += dx * (1.0 - h[: n - kk])
        grad[: n - kk] -= dx * h[kk:]

    return grad


def project_onto_feasible(x: np.ndarray, n: int) -> np.ndarray:
    """
    Project vector x onto the feasibility set:
      0 <= x_i <= 1 and sum(x) = n/2.
    Implements a simple water‑filling algorithm.
    """
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)

    for _ in range(10):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        free_mask = (x > 0.0) & (x < 1.0)
        free_cnt = int(free_mask.sum())
        if free_cnt == 0:
            # All variables are at a bound; distribute uniformly and re‑clip.
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / free_cnt
        x[free_mask] += delta
        x = np.clip(x, 0.0, 1.0)

    # Minor correction for any residual error
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # Adjust the first non‑bound element if possible
        for i in range(n):
            if 0.0 < x[i] < 1.0:
                x[i] += residual
                x[i] = min(max(x[i], 0.0), 1.0)
                break
    return x


def upscale_h(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly interpolate h onto a finer grid of size n_target."""
    n0 = h.size
    if n0 == n_target:
        return h.copy()
    # Mid‑point sampling positions
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False) + (2.0 / n0) / 2.0
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False) + (2.0 / n_target) / 2.0
    return np.interp(xt, x0, h)


def random_binary_feasible(n: int) -> np.ndarray:
    """Create a binary vector with exactly n/2 entries equal to 1."""
    half = n // 2
    arr = np.zeros(n, dtype=float)
    arr[:half] = 1.0
    np.random.shuffle(arr)
    return arr


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Base solution (use the provided one if available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = project_onto_feasible(h0, n0)
    else:
        n0 = 42
        h0 = random_binary_feasible(n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Grid sizes to explore (increase resolution gradually)
    # ------------------------------------------------------------------
    # Ensure we only try even sizes (required for exact sum).
    size_candidates = [n0, 84, 126, 168]
    size_candidates = [s for s in size_candidates if s >= n0 and s % 2 == 0]

    for n_target in size_candidates:
        # Prepare starting vector for this resolution
        if n_target == n0:
            h = h0.copy()
        else:
            # Upscale from the best solution found so far
            h = upscale_h(best_h, n_target)
        h = project_onto_feasible(h, n_target)

        # Track current objective
        cur_c5 = compute_c5(h)

        # ------------------------------------------------------------------
        # 3. Local optimisation loop (gradient + occasional random jitter)
        # ------------------------------------------------------------------
        lr = 1.0                     # initial learning rate
        noise_amp = 0.02             # magnitude of random perturbations
        max_iter = 20000
        it = 0
        while it < max_iter:
            elapsed = time.time() - start_time
            if elapsed >= budget_s * 0.97:
                break

            # Gradient step
            grad = compute_gradient(h)
            h_candidate = h - lr * grad

            # Occasionally add a zero‑mean random jitter to escape plateaus
            if np.random.rand() < 0.05:
                eps = np.random.randn(n_target) * noise_amp
                eps -= eps.mean()               # keep zero sum
                h_candidate += eps

            # Projection back onto feasibility region
            h_candidate = project_onto_feasible(h_candidate, n_target)
            cand_c5 = compute_c5(h_candidate)

            # Acceptance rule (deterministic improvement or Metropolis)
            if cand_c5 < cur_c5:
                accept = True
            else:
                # Linear temperature schedule: cools from 1 to ~0
                T = max(1e-4, (max_iter - it) / max_iter)
                accept = np.random.rand() < math.exp(-(cand_c5 - cur_c5) / T)

            if accept:
                h = h_candidate
                cur_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = h_candidate.copy()

            # Decay schedules
            lr *= 0.9995
            noise_amp *= 0.9999
            it += 1

        # Break early if time is exhausted
        if time.time() - start_time >= budget_s * 0.97:
            break

    # Final evaluation (safety)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, int(best_h.size))
