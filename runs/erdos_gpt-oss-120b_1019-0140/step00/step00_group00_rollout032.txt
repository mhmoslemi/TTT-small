def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using the discretized correlation."""
    n = h.shape[0]
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw.max() * (2.0 / n)

    """
    Project v onto the feasible set {x : 0<=x<=1, sum(x)=target}
    using a simple bisection on the Lagrange multiplier.

    for _ in range(max_iter):

        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
