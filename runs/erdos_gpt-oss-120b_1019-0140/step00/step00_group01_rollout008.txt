import time
import random
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h) over full range
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _initialize_h() -> np.ndarray:
    """Create the starting vector.
    Uses the pre‑imported `initial_h_values` if present,
    otherwise builds a simple half‑ones pattern.
    Ensures the sum constraint ∑h = n/2 exactly.
    """
    try:
        # initial_h_values is provided by the harness
        h = np.array(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: 0/1 pattern of length 47
        n = 47
        h = np.zeros(n, dtype=np.float64)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    n = h.shape[0]
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:
        # distribute the deficit/surplus over non‑boundary entries
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            inc = (target - s) / free.size
            h[free] = np.clip(h[free] + inc, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a small C₅ value.

    Returns (h_values, c5_bound, n_points).
    """
    # reproducibility
    random.seed(seed)
    np.random.seed(seed)

    h = _initialize_h()
    n = h.shape[0]
    target_sum = n / 2.0

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    start = time.time()
    T0 = 0.05                  # initial temperature
    step_max_initial = 0.5     # maximal transfer amount between two entries

    while True:
        if time.time() - start > budget_s:
            break

        elapsed = time.time() - start
        # linear cooling schedule
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # step size shrinks with temperature, but keeps a modest floor
        step_max = step_max_initial * max(0.1, T / T0)
        # occasional occasional larger jumps
        if random.random() < 0.02:          # 2 % chance
            step_max = step_max_initial

        # ---------- propose a feasible 2‑index transfer ----------
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        # feasible interval for amount a transferred from j to i
        a_low = max(-h[i], h[j] - 1.0)          # keep both within [0,1]
        a_high = min(1.0 - h[i], h[j])

        # restrict to current step size
        lo = max(a_low, -step_max)
        hi = min(a_high, step_max)
        if lo >= hi:                            # fallback to the full interval
            lo, hi = a_low, a_high
            if lo >= hi:
                continue                         # no feasible move

        a = random.random() * (hi - lo) + lo
        if abs(a) < 1e-12:                       # negligible change
            continue

        h_new = h.copy()
        h_new[i] += a
        h_new[j] -= a

        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5

        # ---------- Metropolis acceptance ----------
        if delta < 0.0 or random.random() < math.exp(-delta / (T + 1e-12)):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

    # ----- final feasibility repair (tiny drift may occur) -----
    s = best_h.sum()
    if abs(s - target_sum) > 1e-9:
        diff = target_sum - s
        free = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if free.size > 0:
            inc = diff / free.size
            best_h[free] = np.clip(best_h[free] + inc, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
