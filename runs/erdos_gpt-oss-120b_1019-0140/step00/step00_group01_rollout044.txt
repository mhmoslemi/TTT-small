def repair_sum(h, target):
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-10:
        return h
    # find index not at bound or at bound opposite direction
    # if diff>0 need to increase some element: find index where h[i] < 1
    # else need to decrease: find index where h[i] > 0
    idx = None
    if diff > 0:
        candidates = np.where(h < 1 - 1e-12)[0]
        if candidates.size > 0:
            idx = candidates[0]
    else:
        candidates = np.where(h > 1e-12)[0]
        if candidates.size > 0:
            idx = candidates[0]
    if idx is None:
        # all at bound, cannot adjust; just return after maybe scaling all
        return h
    # adjust element
    adjustment = diff
    # ensure within bound
    new_val = np.clip(h[idx] + adjustment, 0, 1)
    actual_adj = new_val - h[idx]
    h[idx] = new_val
    # if actual_adj != adjustment due to clipping, we need to adjust diff again recursively
    remaining = diff - actual_adj
    if abs(remaining) > 1e-12:
        # recursion (or loop)
        return repair_sum(h, target)
    return h
```

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' matches the provided evaluator
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single free entry to enforce the exact sum constraint."""
    cur = float(h.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Prefer an entry that can move in the needed direction
    if diff > 0:
        # need to increase some entry that is not already at 1
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        # need to decrease some entry that is not already at 0
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback: distribute evenly (should rarely happen)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        return h
    i = idx[0]
    new_val = float(np.clip(h[i] + diff, 0.0, 1.0))
    actual = new_val - h[i]
    h[i] = new_val
    # If we couldn't fix the whole difference because of clipping, recurse
    if abs(actual - diff) > 1e-12:
        return _repair_sum(h, target)
    return h
