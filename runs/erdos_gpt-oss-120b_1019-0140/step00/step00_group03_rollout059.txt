import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0/n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_binary(n):
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

    n = None

    best_score = compute_c5(best_h)
    # also compute current_score
    current_score = best_score

    # Annealing parameters
    T0 = 0.01
    temperature = T0
    decay = 0.9995  # per iteration
    iteration = 0
    # Restart parameters
    restart_interval = 5000  # iterations
    last_restart_iter = 0

    # while time left
    while True:
        # check time
        if time.time() - start > budget_s * 0.95:
            break

        # propose swap
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        if h_current[i] == h_current[j]:
            continue  # no change
        # swap
        h_new = h_current.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = compute_c5(h_new)

        # accept if improve or probabilistically
        if new_score < current_score:
            h_current = h_new
            current_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # Metropolis acceptance
            delta = new_score - current_score
            if temperature > 0 and np.random.rand() < math.exp(-delta / temperature):
                h_current = h_new
                current_score = new_score

        iteration += 1
        temperature = max(1e-8, temperature * decay)

        # restart after certain iterations
        if iteration - last_restart_iter >= restart_interval:
            # sometimes random restart to escape dead ends
            if np.random.rand() < 0.3:
                # generate new random binary
                h_current = random_binary(n)
                current_score = compute_c5(h_current)
                # maybe keep best
            last_restart_iter = iteration

    # after loop, compute best_score again (ensuring)
    best_score = compute_c5(best_h)
    # return as list for safety
    return best_h.tolist(), float(best_score), n

import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # maximal overlap integral = max correlation * dx
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _initial_binary(n: int, seed: int) -> np.ndarray:
    """Create a random binary vector with exactly n/2 ones."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1],
    integral 1, minimizing the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector (if supplied)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        init = np.asarray(init, dtype=np.float64)
        n = init.shape[0]
        # Use the supplied vector as the starting point.
        # Ensure feasibility (clip then fix the sum if needed).
        h_current = np.clip(init, 0.0, 1.0)
        target_sum = n / 2.0
        cur_sum = h_current.sum()
        if not np.isclose(cur_sum, target_sum, atol=1e-9):
            # Simple correction: add/subtract a tiny amount from a non‑boundary entry.
            diff = target_sum - cur_sum
            # Find an index that can absorb the difference without leaving [0,1].
            cand = np.where((h_current + diff >= 0.0) & (h_current + diff <= 1.0))[0]
            if cand.size == 0:
                # fallback – redistribute uniformly and clip again
                h_current = np.clip(h_current - (cur_sum - target_sum) / n, 0.0, 1.0)
            else:
                h_current[cand[0]] += diff
    else:
        # No parent construction – fall back to a random binary start.
        n = 70
        h_current = _initial_binary(n, seed)

    # ------------------------------------------------------------------
    # Helper for proposing a feasible move preserving the sum.
    # ------------------------------------------------------------------
    def propose_move(vec: np.ndarray, max_step: float):
        """Return a new vector after moving a small amount between two entries."""
        nn = vec.shape[0]
        i = np.random.randint(nn)
        j = np.random.randint(nn - 1)
        if j >= i:
            j += 1
        # direction: increase i, decrease j
        inc_cap = min(max_step, 1.0 - vec[i])
        dec_cap = min(max_step, vec[j])
        if inc_cap > 0.0 and dec_cap > 0.0:
            delta = np.random.rand() * min(inc_cap, dec_cap)
            new_vec = vec.copy()
            new_vec[i] += delta
            new_vec[j] -= delta
            return new_vec
        # try the opposite direction
        inc_cap = min(max_step, 1.0 - vec[j])
        dec_cap = min(max_step, vec[i])
        if inc_cap > 0.0 and dec_cap > 0.0:
            delta = np.random.rand() * min(inc_cap, dec_cap)
            new_vec = vec.copy()
            new_vec[j] += delta
            new_vec[i] -= delta
            return new_vec
        # move not possible – return original (will be rejected)
        return vec

    # ------------------------------------------------------------------
    # Simulated annealing / hill‑climbing parameters
    # ------------------------------------------------------------------
    best_h = h_current.copy()
    best_score = _c5_value(best_h)

    cur_h = h_current.copy()
    cur_score = best_score

    max_step = 0.12                     # maximum transferred mass per move
    temperature = 0.008                 # initial temperature for uphill steps
    decay = 0.9995                      # exponential cooling
    iter_cnt = 0
    restart_interval = 8000
    last_restart = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:    # leave a safety margin
            break

        # Propose a new candidate
        cand_h = propose_move(cur_h, max_step)
        cand_score = _c5_value(cand_h)

        # Acceptance test
        if cand_score < cur_score:
            cur_h, cur_score = cand_h, cand_score
            if cand_score < best_score:
                best_h, best_score = cand_h.copy(), cand_score
        else:
            # Metropolis criterion
            delta = cand_score - cur_score
            if temperature > 0 and np.random.rand() < math.exp(-delta / temperature):
                cur_h, cur_score = cand_h, cand_score

        # Cooling schedule
        temperature = max(1e-8, temperature * decay)
        iter_cnt += 1

        # Periodic random restart to escape plateaus
        if iter_cnt - last_restart >= restart_interval:
            if np.random.rand() < 0.25:
                # fresh random binary start
                cur_h = _initial_binary(n, seed + iter_cnt)
                cur_score = _c5_value(cur_h)
            last_restart = iter_cnt

        # Early exit if a provably good bound is found
        if best_score * n / 2.0 <= 13.0:   # equivalently C5 ≤ 13*dx = 0.3714...
            break

    # Final evaluation (ensure consistency)
    best_score = _c5_value(best_h)

    return best_h.tolist(), float(best_score), n
