    # Binary search for lambda
    # Function to compute sum after clipping
    n = len(v)
    # define lambda bounds
    l_low = np.min(v - upper) - 1.0  # ensure low enough
    l_high = np.max(v - lower) + 1.0 # ensure high enough
    for _ in range(100):
        lam = (l_low + l_high) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()

            return w

    diff = target - w.sum()

import time
import numpy as np

def compute_c5(h):
    """Compute the overlap bound C5 for a step function h on [0,2]."""
    try:
        # Prefer the provided evaluator if available.
        return float(evaluate_erdos_solution(h))
    except Exception:
        N = len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * (2.0 / N))

def project_to_box_simplex(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """
    Euclidean projection of v onto the set {x | lower <= x_i <= upper, sum x_i = target}.
    Uses a simple bisection on the dual variable lambda.
    """
    v = np.asarray(v, dtype=float)
    # Bounds for lambda
    lam_low = np.min(v) - upper - 1.0
    lam_high = np.max(v) - lower + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lam_low = lam  # need larger lambda to decrease sum
        else:
            lam_high = lam  # need smaller lambda to increase sum
    # Final clipping
    w = np.clip(v - (lam_low + lam_high) / 2.0, lower, upper)
    # Tiny residual fix
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # Adjust a non‑bound element if possible
        idx = np.where((w > lower + 1e-12) & (w < upper - 1e-12))[0]
        if idx.size == 0:
            idx = np.where((w < upper - 1e-12) if diff > 0 else (w > lower + 1e-12))[0]
        if idx.size:
            i = idx[0]
            w[i] = np.clip(w[i] + diff, lower, upper)
    return w

def mutate(parent, rng, target, flip_frac=0.10, noise_scale=0.02):
    """Generate a mutated child from parent."""
    N = len(parent)
    child = parent.copy()
    n_flip = max(1, int(flip_frac * N))
    idx = rng.choice(N, size=n_flip, replace=False)
    child[idx] = rng.random(n_flip)
    child += rng.normal(scale=noise_scale, size=N)
    child = project_to_box_simplex(child, target)
    return child

    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector

    except Exception:

        N = 70
        init = rng.random(N)
    else:
        N = init.size

    target_sum = N / 2.0
    # Ensure feasibility of initial vector
    init = project_to_box_simplex(init, target_sum)

    # Build initial population
    pop = [init]
    scores = [compute_c5(init)]
    while len(pop) < 10:
        cand = rng.random(N)
        cand = project_to_box_simplex(cand, target_sum)
        pop.append(cand)
        scores.append(compute_c5(cand))

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    # Evolutionary loop

    while time.time() - start_time < budget_s * 0.95:
