import time

def _c5_score(h):
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear correlation of h with 1‑h
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _initial_bits_from_h(h):
    """Convert a (possibly non‑binary) initial vector to a binary pattern
    with exactly n/2 ones, preserving the most “massive” positions."""
    n = h.shape[0]
    m = n // 2
    # indices of the m largest entries
    idx = np.argpartition(-h, m)[:m]
    bits = 0
    for i in idx:
        bits |= 1 << int(i)
    return bits


def _bits_to_array(bits, n):
    """Transform integer bitmask to a float array of 0/1 values."""
    return np.array(
        [(bits >> i) & 1 for i in range(n)], dtype=np.float64
    )


def _max_mismatch_from_bits(bits, n):
    """Fast C5 evaluation for binary h given as integer bits."""
    m = n // 2
    dx = 2.0 / n
    worst = 0
    # iterate over positive shifts; negative shifts give the same count
    for k in range(1, n):
        overlap = (bits & (bits >> k)).bit_count()
        mism = m - overlap
        if mism > worst:
            worst = mism
            # early pruning if already worse than known best
    return worst * dx

            use_bits = False  # fractional move destroys pure‑binary representation

        # evaluate
        new_score = _c5_score(h_new) if not use_bits else _max_mismatch_from_bits(new_bits, n)

        # Metropolis acceptance
        delta_score = new_score - cur_score
        if delta_score < 0.0 or random.random() < np.exp(-delta_score / max(T, 1e-12)):
            cur_h = h_new
            cur_score = new_score
            if use_bits:
                bits = new_bits
            # record best
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                if use_bits:
                    best_bits = bits

    # ----------------------------------------------------------------------
    #   Finalise result
    # ----------------------------------------------------------------------
    # Ensure the returned vector obeys the constraints within floating tolerance
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # small correction distributed uniformly
        correction = (target_sum - cur_sum) / n
        best_h += correction
        best_h = np.clip(best_h, 0.0, 1.0)

    # compute final bound (use the provided helper if available)
    try:
        final_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_bound = _c5_score(best_h)

    return best_h, final_bound, n
