    target = N/2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # compute sum diff
    s = h.sum()
    # If sum matches, return
    if abs(s - target) < 1e-12:

    # Compute amount needed
    # We'll use simple algorithm: keep a mask of free variables (not at bounds) to adjust.
    # Allocate diff equally among free variables; then repeat clip.

    free = np.full(N, True)  # all free initially

            break
        free_idx = np.where(free)[0]

            # we cannot adjust further, break
            break
        # distribute equally
        delta = diff / free_idx.size
        # apply delta
        h[free_idx] += delta
        # clip newly hitting bounds

        over = h > 1.0

import numpy as np, time, math

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h):
    N = h.shape[0]
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative projection onto bounded simplex
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (h > 0.0) & (h < 1.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # cannot adjust further
            # fallback: distribute uniformly
            # add diff / N to all entries then clip
            h += diff / N
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free_idx.size
        h[free_idx] += delta
        # clip overflow/underflow
        h = np.clip(h, 0.0, 1.0)
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def mutate(h, mutation_rate=0.2, sigma=0.1):
    N = h.shape[0]
    new = h.copy()
    num = max(1, int(mutation_rate * N))
    idx = np.random.choice(N, size=num, replace=False)
    new[idx] += np.random.randn(num) * sigma
    # Ensure values within bounds and sum constraint
    new = repair_h(new)
    return new

def random_feasible(N):
    # uniform random, then repair
    h = np.random.rand(N)
    h = repair_h(h)
    return h

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Compute C5 for a candidate h using the prescribed correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_h(v: np.ndarray) -> np.ndarray:
    """Project v onto the bounded simplex {0<=x_i<=1, sum x_i = n/2}."""
    n = v.shape[0]
    target = n / 2.0
    x = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            # nothing free – distribute uniformly then clip
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / idx.size
        x[idx] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny adjustment to hit the sum exactly (within numerical noise)
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            x[idx[0]] = np.clip(x[idx[0]] + diff, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)


def _random_feasible(n: int) -> np.ndarray:
    """Create a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_h(v)


def _mutate(parent: np.ndarray, rate: float, sigma: float) -> np.ndarray:
    """Gaussian mutation of a subset of entries followed by projection."""
    n = parent.shape[0]
    child = parent.copy()
    k = max(1, int(rate * n))
    idx = np.random.choice(n, size=k, replace=False)
    child[idx] += np.random.randn(k) * sigma
    return _repair_h(child)
