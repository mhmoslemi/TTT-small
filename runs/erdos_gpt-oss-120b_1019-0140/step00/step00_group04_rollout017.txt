import numpy as np
import time
import math

def compute_cost(h: np.ndarray) -> float:
    """Return C5 bound for a given step function values."""
    # full linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.shape[0]))

def repair_h(h: np.ndarray, target_sum: float) -> None:
    """Clip to [0,1] and adjust to satisfy exact sum target."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = float(np.sum(h))
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return
    # indices that can still move (strict interior)
    interior = np.where((h > 0.0) & (h < 1.0))[0]
    if interior.size == 0:
        # no interior point: adjust the most flexible endpoint
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        return
    # distribute the remaining mass uniformly over interior points
    add = diff / interior.size
    h[interior] = np.clip(h[interior] + add, 0.0, 1.0)
    # final tiny correction (in case of rounding)
    cur = float(np.sum(h))
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # apply to a single interior element
        idx = interior[0]
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
