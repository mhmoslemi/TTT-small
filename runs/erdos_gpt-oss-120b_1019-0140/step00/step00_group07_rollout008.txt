def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Helper to produce a binary vector with exactly n/2 ones
    # ------------------------------------------------------------------
    def make_binary_vec(n):
        """Return a random binary vector of length n with exactly n/2 ones."""
        m = n // 2
        vec = np.zeros(n, dtype=float)
        vec[:m] = 1.0
        np.random.shuffle(vec)
        return vec

    # ------------------------------------------------------------------
    # Obtain an initial binary vector (use the provided one if possible)
    # ------------------------------------------------------------------
    try:
        # initial_h_values is supplied by the harness (may be None)
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        m = n // 2
        # If already binary with correct weight, keep it
        if ((init == 0) | (init == 1)).all() and int(round(init.sum())) == m:
            cur_h = init.copy()
        else:
            # Convert to a binary vector: keep the m largest entries as 1
            idx = np.argsort(init)[::-1]
            cur_h = np.zeros(n, dtype=float)
            cur_h[idx[:m]] = 1.0
    except Exception:
        # No supplied vector – fall back to a standard size 80 binary start
        n = 80
        cur_h = make_binary_vec(n)

    # Ensure exact weight (might be off by a rounding issue)
    m = n // 2
    cur_h = (cur_h > 0.5).astype(float)  # make strictly binary
    current_sum = int(round(cur_h.sum()))
    if current_sum != m:
        # Fix by toggling the required number of entries
        diff = m - current_sum
        if diff > 0:
            zeros = np.where(cur_h == 0)[0]
            cur_h[np.random.choice(zeros, diff, replace=False)] = 1.0
        elif diff < 0:
            ones = np.where(cur_h == 1)[0]
            cur_h[np.random.choice(ones, -diff, replace=False)] = 0.0

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Function to compute the C5 value for a given vector
    # ------------------------------------------------------------------
    def compute_c5(vec):
        # vec is a 1‑D float array of 0/1 values
        corr = np.correlate(vec, 1.0 - vec, mode='full') * dx
        return float(corr.max())

    # Keep the best solution found
    best_h = cur_h.copy()
    best_c5 = compute_c5(best_h)

    # Current state for the simulated‑annealing walk
    cur_c5 = best_c5

    start_time = time.time()
    target_c5 = 0.3808

    # Parameters for the stochastic search
    start_temp = 0.08          # initial temperature (absolute scale)
    max_no_improve = 20000    # steps without improvement before a restart
    no_improve_cnt = 0

    # Main loop – runs until the allotted time expires
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # ----- propose a single 1↔0 swap -----
        ones_idx = np.flatnonzero(cur_h == 1)
        zeros_idx = np.flatnonzero(cur_h == 0)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # Perform the swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        new_c5 = compute_c5(cur_h)

        # Temperature decays linearly with elapsed time
        temp = start_temp * (1.0 - elapsed / budget_s)

        # Acceptance rule (Metropolis)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            if temp > 0.0:
                prob = np.exp(-(new_c5 - cur_c5) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_c5 = new_c5
            # Record a new global best if improved
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve_cnt = 0
                if best_c5 <= target_c5:
                    # excellent – stop early
                    break
            else:
                no_improve_cnt += 1
        else:
            # Revert the swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve_cnt += 1

        # ----- restart when stuck for a while -----
        if no_improve_cnt >= max_no_improve:
            cur_h = make_binary_vec(n)
            cur_c5 = compute_c5(cur_h)
            no_improve_cnt = 0

    # Return the best-found construction
    return (best_h.tolist(), best_c5, n)
