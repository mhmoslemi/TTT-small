import numpy as np, time, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # The convolution returns as float; return max * dx.
    return np.max(corr) * dx

def random_pair_move(h):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # decide direction based on random.
    # transfer amount delta.
    # maximum delta ensures values within [0,1].
    hi, hj = h[i], h[j]
    if np.random.rand() < 0.5:
        # move from i to j
        max_delta = min(hi, 1 - hj)
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    else:
        max_delta = min(hj, 1 - hi)
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        return h_new

def directed_move(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    offset = max_idx - (n - 1)
    # find overlapping indices i
    valid_i = np.arange(max(0, -offset), min(n, n - offset))
    # compute contributions
    contributions = h[valid_i] * (1 - h[valid_i + offset])
    # choose index where contribution > 0
    pos_i = valid_i[contributions > 1e-8]
    if pos_i.size == 0:
        return None
    i = np.random.choice(pos_i)
    j = i + offset
    # move mass from i to j
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        return None
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

import numpy as np
import time

def compute_c5(h):
    """Return the maximal overlap integral for the discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h, full mode gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_initial(n):
    """Create a feasible start: exactly half the mass (average 0.5) using
    0/1 values and at most one fractional entry."""
    h = np.zeros(n, dtype=float)
    full_ones = int(np.floor(n / 2))
    h[:full_ones] = 1.0
    remainder = n / 2.0 - full_ones  # 0 if n even, 0.5 if n odd
    if remainder > 1e-12:
        h[full_ones] = remainder
    np.random.shuffle(h)
    return h

def random_pair_move(h):
    """Select two distinct indices and transfer a feasible amount of mass."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    # random direction
    if np.random.rand() < 0.5:
        # move from i to j
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0.0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # move from j to i
        max_delta = min(hj, 1.0 - hi)
        if max_delta <= 0.0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    return h_new

def directed_move(h):
    """Perform a move aimed at reducing the current worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    offset = max_idx - (n - 1)                     # shift index (can be negative)
    # overlapping indices where both h[i] and h[i+offset] exist
    i_start = max(0, -offset)
    i_end = min(n, n - offset)
    if i_start >= i_end:
        return None
    idxs = np.arange(i_start, i_end)
    contrib = h[idxs] * (1.0 - h[idxs + offset])
    # pick a location with non‑zero contribution
    viable = idxs[contrib > 1e-12]
    if viable.size == 0:
        return None
    i = np.random.choice(viable)
    j = i + offset
    # move mass from i to j (this reduces the product term)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _adjust_to_target_sum(h, target):
    """Scale and then fine‑tune to meet the exact target sum while staying in [0,1]."""
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # simple scaling (may violate bounds)
    if cur > 0:
        h = np.clip(h * (target / cur), 0.0, 1.0)
    else:
        h = np.clip(h, 0.0, 1.0)
    # correct any remaining discrepancy via a single feasible adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find an index that can absorb the diff without breaking bounds
        if diff > 0:
            candidates = np.where(h < 1.0 - 1e-12)[0]
        else:
            candidates = np.where(h > 1e-12)[0]
        if candidates.size > 0:
            idx = np.random.choice(candidates)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    # final clipping (should already be feasible)
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    Heuristic search for a step function giving a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # final safeguard: recompute best bound in case of floating‑point drift
    best_val = compute_c5(best_h)

    return (best_h, best_val, int(best_h.shape[0]))
