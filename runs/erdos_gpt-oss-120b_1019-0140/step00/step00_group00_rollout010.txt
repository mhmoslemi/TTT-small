import numpy as np, time

def project_box_sum(y, target, lo=0.0, hi=1.0, max_iter=60):
    low = -1e6
    high = 1e6
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, lo, hi)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    # final
    return np.clip(y - (low + high) / 2.0, lo, hi)

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def compute_gradient(h, p, dx):
    # compute correlation of h with itself
    corr = np.correlate(h, h, mode='full') * dx
    overlap_vec = 1.0 - corr
    # avoid negative due to numerical error
    overlap_vec = np.maximum(overlap_vec, 0.0)
    w = p * np.power(overlap_vec, p - 1)  # length 2N-1
    N = h.size
    grad = np.zeros_like(h)
    # loop over lags
    for lag in range(-N+1, N):
        w_val = w[lag + N - 1]
        if w_val == 0.0:
            continue
        # vectorized update using slices might lighten loops:
        # For each lag, shift h accordingly.
        # We'll use python loops for simplicity.
        for i in range(N):
            j_plus = i + lag
            if 0 <= j_plus < N:
                grad[i] -= dx * w_val * h[j_plus]
            j_minus = i - lag
            if 0 <= j_minus < N:
                grad[i] -= dx * w_val * h[j_minus]
    return grad

import numpy as np
import time

def _project_box_sum(y, target, lo=0.0, hi=1.0, max_iter=60):
    """Project y onto {x: lo<=x<=hi, sum(x)=target} using water-filling."""
    low = -1e6
    high = 1e6
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(y - lam, lo, hi)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(y - (low + high) * 0.5, lo, hi)

def _c5_value(h, dx):
    """Return max overlap integral for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _gradient(h, p, dx):
    """Gradient of sum_k (overlap_k)^p w.r.t h."""
    N = h.size
    # autocorrelation of h (linear, not circular)
    corr = np.correlate(h, h, mode="full") * dx
    overlap = 1.0 - corr
    overlap = np.maximum(overlap, 0.0)          # protect against tiny negatives
    w = p * np.power(overlap, p - 1)           # length 2N-1
    grad = np.zeros_like(h)

    # lag loop (explicit; O(N^2) but cheap for N<200)
    for lag in range(-N + 1, N):
        w_val = w[lag + N - 1]
        if w_val == 0.0:
            continue
        # vectorized contribution for this lag
        if lag >= 0:
            # indices where i+lag is in range: i = 0..N-1-lag
            i = np.arange(N - lag)
            grad[i] -= dx * w_val * h[i + lag]
            # i-lag term: i = lag..N-1
            i2 = np.arange(lag, N)
            grad[i2] -= dx * w_val * h[i2 - lag]
        else:
            # lag < 0, treat symmetry
            pos = -lag
            i = np.arange(N - pos)
            grad[i + pos] -= dx * w_val * h[i]
            i2 = np.arange(pos, N)
            grad[i2 - pos] -= dx * w_val * h[i2]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- initial vector ----------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    base_n = init_h.size if init_h is not None else 45
    n_points = max(80, base_n * 2)        # modest up‑sampling
    n_points = int(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if init_h is not None:
        # interpolate to new grid (mid‑point sampling)
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False) + (2.0 / base_n) * 0.5
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False) + dx * 0.5
        h = np.interp(x_new, x_old, init_h)
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target_sum, 0.0, 1.0)
    else:
        h = np.full(n_points, 0.5)
        h = _project_box_sum(h, target_sum, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = _c5_value(h, dx)
    current_c5 = best_c5

    # ---------- optimisation parameters ----------
    p = 4                     # smooth power for the max‑approximation
    lr = 0.015                # base learning rate
    iter_cnt = 0

    while time.time() - start < budget_s * 0.97:
        # gradient descent step
        grad = _gradient(h, p, dx)
        h_candidate = _project_box_sum(h - lr * grad, target_sum, 0.0, 1.0)
        c5_candidate = _c5_value(h_candidate, dx)

        if c5_candidate < current_c5:
            h = h_candidate
            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        else:
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.02:
                noise = rng.normal(scale=0.02, size=n_points)
                h_pert = _project_box_sum(h + noise, target_sum, 0.0, 1.0)
                c5_pert = _c5_value(h_pert, dx)
                if c5_pert < current_c5:
                    h = h_pert
                    current_c5 = c5_pert
                    if c5_pert < best_c5:
                        best_c5 = c5_pert
                        best_h = h_pert.copy()

        # simple learning‑rate decay
        if iter_cnt % 500 == 0 and iter_cnt > 0:
            lr *= 0.90
        iter_cnt += 1

    return best_h, best_c5, n_points
