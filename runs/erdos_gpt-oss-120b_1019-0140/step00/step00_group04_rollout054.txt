            # need to reduce
            excess = cur - target
            # eligible indices where new_h > 0
            elig = new_h > 0
            if not np.any(elig):

            reduction = excess / np.count_nonzero(elig)
            # reduce each eligible element by min reduction and its value
            # avoid negative

def project_to_constraints(h):
    n = h.shape[0]
    target = n / 2.0
    # clip
    np.clip(h, 0.0, 1.0, out=h)
    # adjust sum:
    cur = h.sum()
    # water-filling iterative method
    eps = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        diff = target - cur
        if abs(diff) < eps:
            break
        if diff > 0:
            # need to increase sum
            elig = h < 1.0 - eps
            cnt = np.count_nonzero(elig)
            if cnt == 0:
                break
            inc = diff / cnt
            # add inc but not exceed 1
            add = np.minimum(1.0 - h[elig], inc)
            h[elig] += add
            cur = h.sum()
        else:
            # need to decrease sum
            elig = h > eps
            cnt = np.count_nonzero(elig)
            if cnt == 0:
                break
            dec = (-diff) / cnt
            sub = np.minimum(h[elig], dec)
            h[elig] -= sub
            cur = h.sum()
    # final clip
    np.clip(h, 0.0, 1.0, out=h)
    return h
