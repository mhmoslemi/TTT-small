def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_constraints(h, target_sum):
    # H clipping and sum adjustment to exactly target_sum
    # clip values
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    # adjust uniformly
    diff = target_sum - cur
    # distribute diff over entries not at bounds (or all)
    # naive: add diff/n to all then clip again until sum close
    # We'll do iterative simple approach
    if abs(diff) > 1e-12:
        # compute mask of interior entries not saturated
        mask = (h > 0.0) & (h < 1.0)
        # if none interior, fallback to entire vector
        idx = np.where(mask)[0]
        if len(idx) == 0:
            idx = np.arange(h.size)
        # distribute diff uniformly across idx
        inc = diff / len(idx)
        h[idx] += inc
        # Clip again
        h = np.clip(h, 0.0, 1.0)
        # Might still be off; simple iteration
        for _ in range(10):
            cur = h.sum()
            diff = target_sum - cur
            if abs(diff) < 1e-12:
                break
            mask = (h > 0.0) & (h < 1.0)
            idx = np.where(mask)[0]
            if len(idx) == 0:
                idx = np.arange(h.size)
            inc = diff / len(idx)
            h[idx] += inc
            h = np.clip(h, 0.0, 1.0)
    return h

def propose_swap(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # choose direction random
    if rng.random() < 0.5:
        # move mass from i to j
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0:
            return None
        d = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] -= d
        new_h[j] += d
    else:
        max_delta = min(hj, 1.0 - hi)
        if max_delta <= 0:
            return None
        d = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] += d
        new_h[j] -= d

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    total_time = budget_s * 0.95   # keep margin
    rng = np.random.default_rng(seed)
    # Determine n_points
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.size
    except NameError:
        n = 70
        init = None
    target_sum = n / 2.0
    # initial h
    if init is not None:
        h = init.copy()
        # ensure constraints
        cur_sum = h.sum()
        if not np.isclose(cur_sum, target_sum):
            h = project_to_constraints(h, target_sum)
        # ensure bounds
        h = np.clip(h, 0.0, 1.0)
    else:
        # random initial continuous vector with sum constraint
        # start with uniform random and project
        h = rng.random(n)
        h = project_to_constraints(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h
    current_c5 = best_c5
    # set temperature schedule
    max_temp = 0.02  # adjust
    # iteration loop
    iter_count = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= total_time:
            break
        # temperature decreasing linearly
        temp = max_temp * (1.0 - elapsed / total_time)
        # propose neighbor
        new_h = propose_swap(current_h, rng)
        if new_h is None:
            continue
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        iter_count += 1
        # occasional larger perturbation to escape local minima
        if iter_count % 5000 == 0:
            # random recombination: random convex combination with best
            mix = rng.random()
            candidate = mix * best_h + (1 - mix) * rng.random(n)
            candidate = project_to_constraints(candidate, target_sum)
            candidate_c5 = compute_c5(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = candidate_c5
                current_h = candidate
                current_c5 = candidate_c5
    # final verification and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_constraints(best_h, target_sum)  # ensure sum
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n

import numpy as np
import time
import math

def _compute_c5(h):
    """Return the overlap constant C5 for a given h (numpy 1‑D)."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _project_to_sum_and_bounds(h, target_sum, max_iter=10):
    """
    Clip to [0,1] and adjust so that sum(h) == target_sum.
    Simple iterative redistribution on non‑saturated entries.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # indices that can still move (strictly inside bounds)
        free = (h > 0.0) & (h < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:          # everything at a bound, spread over all
            idx = np.arange(h.size)
        inc = diff / idx.size
        h[idx] += inc
        h = np.clip(h, 0.0, 1.0)
    return h

def _propose_swap(h, rng):
    """Transfer a random amount of mass between two different positions."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # random direction
    if rng.random() < 0.5:
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0.0:
            return None
        d = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] -= d
        new_h[j] += d
    else:
        max_delta = min(hj, 1.0 - hi)
        if max_delta <= 0.0:
            return None
        d = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] += d
        new_h[j] -= d
    return new_h

def _propose_targeted(h, rng):
    """
    Reduce the current maximal overlap by moving mass from a
    position that contributes positively to that lag towards the
    partner position that would increase the autocorrelation.
    """
    n = h.shape[0]
    # full correlation (without the dx factor)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)          # signed lag
    if lag == 0:
        return None  # nothing to fix at lag zero for binary‑type vectors

    if lag > 0:
        i_vals = np.arange(0, n - lag)
        j_vals = i_vals + lag
    else:
        i_vals = np.arange(-lag, n)
        j_vals = i_vals + lag    # lag negative

    # contribution h_i * (1 - h_j); we want hi > hj
    diff = h[i_vals] - h[j_vals]
    cand = np.where(diff > 1e-12)[0]
    if cand.size == 0:
        return None
    k = rng.choice(cand)
    i = i_vals[k]
    j = j_vals[k]
    hi, hj = h[i], h[j]
    max_delta = min(hi, 1.0 - hj)
    if max_delta <= 0.0:
        return None
    d = rng.random() * max_delta
    new_h = h.copy()
    new_h[i] -= d
    new_h[j] += d
    return new_h

def _propose_move(h, rng, prob_targeted=0.7):
    """Choose a move: targeted with prob_targeted, otherwise random swap."""
    if rng.random() < prob_targeted:
        nh = _propose_targeted(h, rng)
        if nh is not None:
            return nh
    return _propose_swap(h, rng)
