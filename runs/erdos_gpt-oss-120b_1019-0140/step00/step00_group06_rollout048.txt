    best_h = None
    best_score = float("inf")

    for n in n_vals:
        # ------------------------------------------------------------------
        # Build starting vector for current resolution.
        if n == n0:
            h_start = init_vec.copy()
        else:
            src = best_h if best_h is not None else init_vec
            n_src = src.size
            x_src = np.arange(n_src) * (2.0 / n_src)
            x_new = np.arange(n) * (2.0 / n)
            h_start = np.interp(x_new, x_src, src)
        target_sum = n / 2.0
        h_start = _project_to_constraints(h_start, target_sum)

        # ------------------------------------------------------------------
        # Optimization loop with multiple random restarts.
        max_restarts = 30
        maxiter = 150  # SLSQP internal iterations
        restart = 0
        while restart < max_restarts and (time.time() - start_time) < budget_s * 0.95:
            restart += 1
            # Random perturbation of the start point.
            perturb = np.random.randn(n) * 0.02
            x0 = h_start + perturb
            x0 = _project_to_constraints(x0, target_sum)

            # Equality constraint: sum(x) = target_sum
            cons = {"type": "eq", "fun": lambda h, tgt=target_sum: np.sum(h) - tgt}
            bounds = [(0.0, 1.0)] * n

            try:
                res = minimize(
                    _compute_c5,
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[cons],
                    options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
                )
            except Exception:
                continue

            h_opt = _project_to_constraints(res.x, target_sum)
            score = _compute_c5(h_opt)
            if score < best_score:
                best_score = float(score)
                best_h = h_opt.copy()
                # Very early exit if we beat the target bound.
                if best_score < 0.3808:
                    break

        # ------------------------------------------------------------------
        # If we already achieved the target, stop trying larger resolutions.
        if best_score < 0.3808:
            break

    # ----------------------------------------------------------------------
    # Fallback if nothing succeeded (unlikely).
    if best_h is None:
        best_h = h_start
        best_score = _compute_c5(best_h)

    n_final = int(best_h.size)

    # Use the provided evaluator if present for the final reported bound.
    if "evaluate_erdos_solution" in globals():
        try:
            final_score = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_score = float(best_score)
    else:
        final_score = float(best_score)

    return best_h.astype(np.float64), final_score, n_final
