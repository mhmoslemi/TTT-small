def local_search(h, max_iter, init_T, final_T):
    n = len(h)
    best_score = evaluate_c5(h)
    best_h = h.copy()
    T = init_T
    for it in range(max_iter):
        # update temperature schedule
        T = init_T * (final_T/init_T) ** (it/max_iter) # geometric cooling
        # propose move
        # select random i where h_i not at bound to increase, and j where opposite.
        i = np.random.randint(n)
        j = np.random.randint(n)
        while i == j:
            j = np.random.randint(n)
        # Determine max allowable delta
        # We will attempt to increase h_i, decrease h_j (or opposite)
        # Choose direction randomly
        direction = np.random.choice([1, -1]) # 1 means increase i, decrease j
        if direction == 1:
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            max_delta = min(max_inc, max_dec)
            # if max_delta very small, skip move
            if max_delta <= 0:
                continue
            delta = np.random.uniform(0, max_delta)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_dec = h[i]
            max_inc = 1.0 - h[j]
            max_delta = min(max_dec, max_inc)
            if max_delta <= 0:
                continue
            delta = np.random.uniform(0, max_delta)
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        # evaluate
        new_score = evaluate_c5(new_h)
        # acceptance
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / max(T, 1e-12)):
            h = new_h
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
    return best_h, best_score

def upsample_initial_to_n(init_vec, n):
    n0 = init_vec.shape[0]
    x0 = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    h_new = np.interp(xn, x0, init_vec)
    # ensure within [0,1]
    h_new = np.clip(h_new, 0.0, 1.0)
    # adjust sum
    target = n / 2.0
    s = np.sum(h_new)
    diff = target - s
    # distribute diff onto a random index not at bound
    if abs(diff) > 1e-12:
        # find safe indices to adjust (bounded)
        candidates = np.where((diff > 0) & (h_new < 1-1e-12) | (diff < 0) & (h_new > 1e-12))[0]
        if len(candidates) > 0:
            idx = np.random.choice(candidates)
            h_new[idx] = np.clip(h_new[idx] + diff, 0.0, 1.0)
        else:
            # distribute uniformly small
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
    # final adjust to ensure sum close
    # If still off, adjust first element
    s = np.sum(h_new)
    diff = target - s
    if abs(diff) > 1e-12:
        h_new[0] = np.clip(h_new[0] + diff, 0.0, 1.0)
    return h_new

def slsq_optimize(h_initial):
    import numpy as np
    from scipy.optimize import minimize

    n = h_initial.shape[0]
    dx = 2.0 / n
    target_sum = n/2.0

    # initial t (bound)
    t0 = evaluate_c5(h_initial)  # compute max correlation*dx

    x0 = np.concatenate([h_initial, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = []
    cons.append({'type':'eq', 'fun': lambda x, target=target_sum: np.sum(x[:n]) - target})
    # inequality t - corr*dx >= 0 for all shifts
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full')
        return t - corr * dx
    cons.append({'type':'ineq', 'fun': ineq_fun})

    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        # return original
        return h_initial, t0
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum exactly by small tweak
    s = np.sum(h_opt)
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # adjust any component with capacity
        if diff > 0:
            idx = np.where(h_opt < 1 - 1e-12)[0][0]
            h_opt[idx] = np.clip(h_opt[idx] + diff, 0.0, 1.0)
        else:
            idx = np.where(h_opt > 1e-12)[0][0]
            h_opt[idx] = np.clip(h_opt[idx] + diff, 0.0, 1.0)
    t_opt = res.x[-1]
    return h_opt, t_opt

def hill_climb(h_initial, max_iter, time_limit):
    import numpy as np, math, time
    n = len(h_initial)
    dx = 2.0 / n
    h = h_initial.copy()
    best_h = h.copy()
    # compute initial
    corr = np.correlate(h, 1 - h, mode='full')
    best_score = np.max(corr) * dx
    curr_score = best_score
    # temperature schedule
    init_T = max(0.1, best_score)  # initial temperature as proportion
    final_T = 1e-4
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # linear or geometric schedule
        T = init_T * ((final_T / init_T) ** (it / max_iter))
        # propose move:
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # direction
        if np.random.rand() < 0.5:
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            max_delta = min(max_inc, max_dec)
            if max_delta <= 0:
                continue
            delta = np.random.uniform(0, max_delta)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            max_dec = h[i]
            max_inc = 1.0 - h[j]
            max_delta = min(max_dec, max_inc)
            if max_delta <= 0:
                continue
            delta = np.random.uniform(0, max_delta)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        # evaluate
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_score = np.max(corr_new) * dx
        # acceptance
        if new_score < curr_score or np.random.rand() < math.exp((curr_score - new_score) / max(T, 1e-12)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

import numpy as np
import time
import math
from scipy.optimize import minimize

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by discrete steps."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _adjust_sum_to_target(vec, target):
    """Clip to [0,1] and adjust a single entry to hit the required sum."""
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - np.sum(vec)
    if abs(diff) > 1e-12:
        # Find an index that can absorb the difference
        if diff > 0:
            cand = np.where(vec < 1.0 - 1e-12)[0]
        else:
            cand = np.where(vec > 1e-12)[0]
        if cand.size == 0:
            # fallback uniform tweak
            vec += diff / vec.size
            vec = np.clip(vec, 0.0, 1.0)
        else:
            i = np.random.choice(cand)
            vec[i] = np.clip(vec[i] + diff, 0.0, 1.0)
    return vec

def upsample_initial_to_n(init_vec, n):
    """Linear interpolation of a parent construction to a new resolution."""
    n0 = init_vec.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    up = np.interp(xn, x0, init_vec)
    up = np.clip(up, 0.0, 1.0)
    target = n / 2.0
    up = _adjust_sum_to_target(up, target)
    return up

def hill_climb(h0, max_iter, time_limit):
    """Simple simulated‑annealing style improvement preserving sum and bounds."""
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = np.max(corr) * dx
    best_h = h.copy()
    curr_score = best_score
    init_T = max(0.1, best_score)
    final_T = 1e-4
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        T = init_T * ((final_T / init_T) ** (it / max_iter))
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if np.random.rand() < 0.5:
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            dmax = min(max_inc, max_dec)
            if dmax <= 0.0:
                continue
            delta = np.random.uniform(0.0, dmax)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            max_dec = h[i]
            max_inc = 1.0 - h[j]
            dmax = min(max_dec, max_inc)
            if dmax <= 0.0:
                continue
            delta = np.random.uniform(0.0, dmax)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_score = np.max(corr_new) * dx
        if new_score < curr_score or np.random.rand() < math.exp((curr_score - new_score) / max(T, 1e-12)):
            h = h_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
    return best_h, best_score

def slsq_optimize(h_start):
    """Refine a candidate with SLSQP (minimize the maximal overlap)."""
    n = h_start.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    t0 = evaluate_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {'type': 'eq',
               'fun': lambda x, target=target_sum: np.sum(x[:n]) - target}
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * dx
    ineq_cons = {'type': 'ineq', 'fun': ineq_fun}
    cons = [eq_cons, ineq_cons]

    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})

    if not res.success:
        return h_start, t0
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # Tiny sum correction
    diff = target_sum - np.sum(h_opt)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h_opt < 1.0 - 1e-12)[0][0]
            h_opt[idx] = np.clip(h_opt[idx] + diff, 0.0, 1.0)
        else:
            idx = np.where(h_opt > 1e-12)[0][0]
            h_opt[idx] = np.clip(h_opt[idx] + diff, 0.0, 1.0)
    t_opt = res.x[-1]
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C5."""
    np.random.seed(seed)
    start_time = time.time()

    # Base resolution – aim for an even number of points.
    init_parent = globals().get('initial_h_values', None)
    n = 200
    if init_parent is not None:
        n0 = init_parent.shape[0]
        n = max(2 * n0, 200)
        if n % 2 == 1:
            n += 1

    # Build a small pool of starters.
    starters = []

    if init_parent is not None:
        starters.append(upsample_initial_to_n(init_parent, n))

    # Sinusoidal (guaranteed average 0.5)
    idx = np.arange(n)
    h_sin = 0.5 + 0.5 * np.cos(2.0 * np.pi * idx / n)
    h_sin = np.clip(h_sin, 0.0, 1.0)
    h_sin = _adjust_sum_to_target(h_sin, n / 2.0)
    starters.append(h_sin)

    # A few random balanced binary vectors.
    for _ in range(3):
        h = np.zeros(n)
        h[:n // 2] = 1.0
        np.random.shuffle(h)
        starters.append(h)

    best_h = None
    best_score = float('inf')

    # Main loop over initial guesses.
    for h0 in starters:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Hill‑climbing phase (fast, many iterations).
        remaining = budget_s - (time.time() - start_time)
        max_iter = 5000 if remaining > 10 else int(remaining * 500)
        h_hc, score_hc = hill_climb(h0, max_iter, start_time + budget_s * 0.9)
        if score_hc < best_score:
            best_score = score_hc
            best_h = h_hc

        # SLSQP refinement (more expensive, limited by remaining budget).
        if time.time() - start_time > budget_s * 0.9:
            continue
        h_ref, t_ref = slsq_optimize(h_hc)
        final_score = evaluate_c5(h_ref)
        if final_score < best_score:
            best_score = final_score
            best_h = h_ref

    # Fallback in the unlikely event that nothing was generated.
    if best_h is None:
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        np.random.shuffle(best_h)

    # Final repair to satisfy the integral exactly.
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = (n / 2.0) - np.sum(best_h)
    if abs(diff) > 1e-12:
        # Adjust the first non‑saturated entry.
        if diff > 0:
            idx = np.where(best_h < 1.0 - 1e-12)[0][0]
        else:
            idx = np.where(best_h > 1e-12)[0][0]
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    c5_bound = evaluate_c5(best_h)
    return best_h.astype(np.float64), float(c5_bound), n
