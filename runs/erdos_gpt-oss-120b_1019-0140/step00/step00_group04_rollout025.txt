import time, numpy as np, cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max()) * dx

def solve_milp(n, time_limit):
    # Build variables and constraints
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(nonneg=True) # ensure T>=0
    constraints = [cp.sum(h) == n // 2]

    for k in range(1, n):
        length = n - k
        w = cp.Variable(length, boolean=True)
        constraints.append(w <= h[:length])
        constraints.append(w <= h[k:])
        constraints.append(w >= h[:length] + h[k:] - 1)
        overlap = cp.sum(w)
        constraints.append(overlap >= n // 2 - T)
    constraints.append(T >= 0)

    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        # set a time limit using solver options if possible
        prob.solve(solver=cp.GLPK_MI, verbose=False, glpk={'msg_lev': 'GLP_MSG_OFF'})
        # GLPK does not accept time limit directly; we rely on budget.
    except cp.SolverError:
        try:
            prob.solve(solver=cp.ECOS_BB, verbose=False)
        except cp.SolverError:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except cp.SolverError:
                return None, None
    if h.value is None:
        return None, None
    h_sol = np.array(h.value, dtype=float)
    # round to 0/1
    h_bin = (h_sol > 0.5).astype(float)
    # enforce exact sum
    target = n // 2
    current = int(h_bin.sum())
    if current != target:
        # adjust
        diff = target - current
        if diff > 0:
            # set diff zeros to 1
            zero_idx = np.where(h_bin == 0)[0]
            if len(zero_idx) >= diff:
                h_bin[zero_idx[:diff]] = 1.0
        elif diff < 0:
            one_idx = np.where(h_bin == 1)[0]
            if len(one_idx) >= -diff:
                h_bin[one_idx[: -diff]] = 0.0
    t_val = float(T.value) if T.value is not None else None
    return h_bin, t_val

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)
    n = 86  # even
    # try MILP
    h, t_val = solve_milp(n, budget_s - (time.time() - start))

        h = np.zeros(n)
        h[:n//2] = 1.0
        rng.shuffle(h)
        # simple hill climb
        best = h.copy()
        best_c5 = compute_c5(best)
        # local search
        # simple random swaps
        while time.time() - start < budget_s - 0.5:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if best[i] == best[j]:
                continue
            new = best.copy()
            new[i], new[j] = new[j], new[i]
            c5_new = compute_c5(new)
            if c5_new < best_c5:
                best = new
                best_c5 = c5_new
        h = best
    # compute final bound
    c5 = compute_c5(h)
    return h, float(c5), n

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Return the overlap integral C5 for a given step function vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * dx

def _solve_milp(n, time_budget):
    """Solve the mixed‑integer program for a given n.
    Returns a binary numpy array and the optimal T (real), or (None, None) on failure."""
    # Decision variables
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(nonneg=True)  # the maximal mismatches we are minimising
    constraints = [cp.sum(h) == n // 2]

    # For each positive shift, enforce overlap >= n/2 - T
    for k in range(1, n):
        length = n - k
        w = cp.Variable(length, boolean=True)          # w[i] = h[i] * h[i+k]
        constraints += [
            w <= h[:length],
            w <= h[k:],
            w >= h[:length] + h[k:] - 1,
            cp.sum(w) >= n // 2 - T
        ]

    prob = cp.Problem(cp.Minimize(T), constraints)

    # Try a few solvers; most installations have at least one of them.
    for solver in (cp.GLPK_MI, cp.ECOS_BB, cp.CBC):
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except (cp.SolverError, Exception):
            continue
    else:
        return None, None

    if h.value is None:
        return None, None

    # Extract a binary solution
    h_val = np.array(h.value, dtype=float)
    h_bin = (h_val > 0.5).astype(float)

    # Enforce exact sum = n/2 (tiny rounding may upset it)
    target = n // 2
    cur = int(h_bin.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zero_idx = np.where(h_bin == 0)[0]
            if len(zero_idx) >= diff:
                h_bin[zero_idx[:diff]] = 1.0
        elif diff < 0:
            one_idx = np.where(h_bin == 1)[0]
            if len(one_idx) >= -diff:
                h_bin[one_idx[: -diff]] = 0.0

    T_val = None if T.value is None else float(T.value)
    return h_bin, T_val

def _hill_climb_binary(n, rng, time_limit):
    """Simple stochastic hill‑climbing for a binary vector with exactly n/2 ones."""
    # start from a random feasible binary vector
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    while time.time() - start < time_limit:
        # propose a swap of a 1 and a 0
        i, j = rng.integers(0, n, size=2)
        if best_h[i] == best_h[j]:
            continue
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h that (hopefully) gives C5 <= 0.3808.
    The function returns (h_values, c5_bound, n_points)."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose an even n large enough that T=16 would already beat the target.
    # n = 86 gives dx = 2/86 and C5 = 16*dx ≈ 0.372 < 0.3808.
    n = 86

    # Try the exact MILP formulation first.
    remaining = max(0.0, budget_s - (time.time() - start_time) - 5.0)  # leave a few seconds margin
    h_sol, T_opt = _solve_milp(n, remaining)

    if h_sol is None:
        # Fall back to a cheap stochastic search if the MILP did not finish.
        time_left = max(0.0, budget_s - (time.time() - start_time))
        h_sol = _hill_climb_binary(n, rng, time_left)

    # Compute the final bound.
    c5_val = compute_c5(h_sol)

    # Ensure the returned array respects the required constraints (clip and correct sum).
    h_sol = np.clip(h_sol, 0.0, 1.0)
    target_sum = n // 2
    cur_sum = int(round(h_sol.sum()))
    if cur_sum != target_sum:
        # Adjust by flipping the smallest number of entries.
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.where(h_sol < 0.5)[0]
            if len(zeros) >= diff:
                h_sol[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(h_sol > 0.5)[0]
            if len(ones) >= -diff:
                h_sol[ones[: -diff]] = 0.0

    # Final recompute of C5 after any tiny adjustments.
    c5_val = compute_c5(h_sol)

    return h_sol, float(c5_val), n
