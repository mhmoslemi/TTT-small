import time, math
import numpy as np


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a small C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with values in [0,1]
        c5_bound : float, the achieved C5 bound
        n_points : int, the discretisation size used
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine problem size and initialise a binary candidate
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.size
    else:
        n = 80
        init_arr = None

    dx = 2.0 / n
    target_c5 = 0.3808
    target_corr = target_c5 / dx  # integer bound we try to stay under

    ones_needed = n // 2
    # start from a deterministic binary rounding of initial_h_values if possible
    if init_arr is not None:
        idx_desc = np.argsort(-init_arr, kind="stable")
        h = np.zeros(n, dtype=float)
        h[idx_desc[:ones_needed]] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=ones_needed, replace=False)] = 1.0

    # ------------------------------------------------------------------
    # Helper to compute C5 bound for a candidate vector
    # ------------------------------------------------------------------
    def compute_c5(vec):
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return corr.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_corr = best_c5 / dx

    # The current state used for the Metropolis walk
    cur_h = best_h.copy()
    cur_c5 = best_c5
    cur_corr = best_corr

    start_time = time.time()
    # parameters for the stochastic walk
    max_no_improve = 6000  # restart after this many iterations without improvement
    iters_without_best = 0
    total_iters = 0

    while time.time() - start_time < budget_s * 0.95:
        total_iters += 1
        # ------------------------------------------------------------------
        # Identify the shift giving the worst (largest) correlation
        # ------------------------------------------------------------------
        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(full_corr.argmax())
        max_corr_val = float(full_corr[max_idx])
        shift = max_idx - (n - 1)          # integer shift corresponding to max correlation

        # ------------------------------------------------------------------
        # Find indices that contribute to this worst correlation.
        # These are positions i where cur_h[i]==1 and cur_h[i+shift]==0
        # (taking care of sign of shift)
        # ------------------------------------------------------------------
        if shift >= 0:
            length = n - shift
            h_i = cur_h[:length]
            h_j = cur_h[shift:shift + length]
            mism = np.where((h_i == 1.0) & (h_j == 0.0))[0]
        else:
            s = -shift
            length = n - s
            h_i = cur_h[s:]
            h_j = cur_h[:length]
            mism = np.where((h_i == 1.0) & (h_j == 0.0))[0] + s

        # ------------------------------------------------------------------
        # Propose a swap: turn a mismatching 1 into 0 and turn a 0 elsewhere into 1
        # ------------------------------------------------------------------
        if mism.size == 0:
            # no direct mismatches (rare); pick random 1 and random 0
            ones = np.where(cur_h == 1.0)[0]
            zeros = np.where(cur_h == 0.0)[0]
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        else:
            i = int(rng.choice(mism))
            zeros = np.where(cur_h == 0.0)[0]
            j = int(rng.choice(zeros))

        # create neighbour
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0

        new_c5 = compute_c5(new_h)
        new_corr = new_c5 / dx

        # ------------------------------------------------------------------
        # Acceptance rule: always take improvement; otherwise Metropolis step
        # ------------------------------------------------------------------
        if new_corr < cur_corr:
            cur_h, cur_c5, cur_corr = new_h, new_c5, new_corr
            # record best if improved
            if new_corr < best_corr - 1e-12:
                best_h, best_c5, best_corr = new_h, new_c5, new_corr
                iters_without_best = 0
            else:
                iters_without_best += 1
        else:
            # simulated‑annealing style acceptance
            elapsed = time.time() - start_time
            frac = elapsed / budget_s
            temp = max(1e-6, 0.4 * (1.0 - frac))          # temperature linearly cools
            prob = math.exp(-(new_corr - cur_corr) / temp)
            if rng.random() < prob:
                cur_h, cur_c5, cur_corr = new_h, new_c5, new_corr
            iters_without_best += 1

        # ------------------------------------------------------------------
        # Periodic restart to escape plateaus
        # ------------------------------------------------------------------
        if iters_without_best >= max_no_improve:
            # start from the best found so far but perturb it a little
            cur_h = best_h.copy()
            # perform a few random swaps to shake the state
            for _ in range(4):
                ones = np.where(cur_h == 1.0)[0]
                zeros = np.where(cur_h == 0.0)[0]
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                cur_h[i] = 0.0
                cur_h[j] = 1.0
            cur_c5 = compute_c5(cur_h)
            cur_corr = cur_c5 / dx
            iters_without_best = 0

        # early exit if we already meet the target
        if best_corr <= target_c5:
            break

    # ------------------------------------------------------------------
    # Return the best solution found
    # ------------------------------------------------------------------
    return best_h, best_c5, n
