import time, math, numpy as np

def _initial_binary():
    """Create a binary start vector using the global initial_h_values if present."""
    # Determine length
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        # Select the top n//2 entries as 1's (preserves any structure in init)
        idx = np.argsort(-init)[: n // 2]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    except Exception:
        # Fallback: 80 points, half ones
        n = 80
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    return h

def _score(h, dx):
    """Return max overlap count and the bound (max*dx)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_cnt = float(corr.max())
    return max_cnt, max_cnt * dx

def _worst_shift(corr, n):
    """Return a shift (integer) where the correlation attains its maximum."""
    max_val = corr.max()
    idxs = np.flatnonzero(corr == max_val)
    idx = np.random.choice(idxs)
    return idx - (n - 1)  # zero shift is at n-1

def _contributing_ones(h, shift):
    """Indices i where h[i]=1 and h[i+shift]=0 (valid i)."""
    n = h.size
    if shift >= 0:
        i = np.arange(0, n - shift)
        cond = (h[i] == 1) & (h[i + shift] == 0)
    else:
        i = np.arange(-shift, n)
        cond = (h[i] == 1) & (h[i + shift] == 0)
    return i[cond]

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a binary step function h on [0,2] with ∫h=1 that minimizes C5."""
    np.random.seed(seed)
    start = time.time()
    h = _initial_binary()
    n = h.size
    dx = 2.0 / n
    # initial evaluation
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_cnt, best_bound = _score(h, dx)
    best_h = h.copy()
    best_cnt = max_cnt

    # maintain lists of indices for fast sampling
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)

    # search parameters
    target_bound = 0.3808
    no_improve = 0
    focus_prob = 0.8      # prob. to use a focused move
    T0 = 0.01             # initial temperature for occasional uphill moves

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break

        # recompute correlation for the current state
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_cnt, cur_bound = _score(h, dx)

        # decide move type
        if np.random.rand() < focus_prob:
            # focused move: target a worst shift
            shift = _worst_shift(corr, n)
            contributors = _contributing_ones(h, shift)
            if contributors.size == 0:
                # fallback to random 1‑index
                i = np.random.choice(ones_idx)
            else:
                i = np.random.choice(contributors)
        else:
            # plain random 1‑index
            i = np.random.choice(ones_idx)

        # pick a zero position to swap with (avoid i)
        j = np.random.choice(zeros_idx)

        # propose swap

        h_new[i] = 0.0

        new_max_cnt, new_bound = _score(h_new, dx)

        # acceptance test (simulated annealing)
        accept = False
        if new_max_cnt < max_cnt:
            accept = True
        else:

            zeros_idx = np.flatnonzero(h == 0)

                best_bound = new_bound

        # random restart if stuck

            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)

    return (best_h.tolist(), final_bound, n)
