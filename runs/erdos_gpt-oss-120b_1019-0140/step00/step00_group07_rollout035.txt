def compute_c5(h):
    import numpy as np
    N = h.shape[0]
    dx = 2.0 / N
    # correlation using np.correlate
    # Using double float ensures sum within double precision.
    corr = np.correlate(h, 1.0 - h, mode='full')
    # convert to float
    max_corr = np.max(corr)
    return max_corr * dx

    h = h0.astype(float).copy()
    N = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    T0 = 0.02  # initial temperature
    # threshold for stagnation to optionally restart: may not be needed.
    # We'll just run until time out.
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:

        # propose swap
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i:
            j += 1

        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        # temperature
        frac = elapsed / max_seconds
        T = T0 * (1.0 - frac)

        else:

            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

            h[i], h[j] = h[j], h[i]

def replicate_pattern(pattern, target_n):
    rep_factor = target_n // len(pattern)
    remainder = target_n % len(pattern)
    new = np.tile(pattern, rep_factor)

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h sampled on [0,2]."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with (1-h) using zero padding (np.correlate mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def replicate_pattern(pattern: np.ndarray, target_n: int, rng: np.random.Generator) -> np.ndarray:
    """Repeat a binary pattern to a longer length and adjust to exact half density."""
    N = pattern.shape[0]
    rep = target_n // N
    rem = target_n % N
    # repeat whole pattern
    if rep > 0:
        new = np.tile(pattern, rep)
    else:
        new = np.empty(0, dtype=pattern.dtype)
    # add a prefix of the pattern for the remainder
    if rem > 0:
        new = np.concatenate([new, pattern[:rem]])
    # enforce exact half‑ones
    target_sum = target_n // 2
    cur_sum = int(new.sum())
    diff = target_sum - cur_sum
    if diff > 0:
        zero_idx = np.where(new == 0)[0]
        if diff > len(zero_idx):
            new[zero_idx] = 1.0
        else:
            choose = rng.choice(zero_idx, size=diff, replace=False)
            new[choose] = 1.0
    elif diff < 0:
        one_idx = np.where(new == 1)[0]
        diff = -diff
        if diff > len(one_idx):
            new[one_idx] = 0.0
        else:
            choose = rng.choice(one_idx, size=diff, replace=False)
            new[choose] = 0.0
    return new.astype(float)

def local_search(h0: np.ndarray, rng: np.random.Generator, max_seconds: float):
    """Simulated‑annealing style hill climbing on a binary vector."""
    h = h0.astype(float).copy()
    N = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    T0 = 0.07            # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:
            break
        # propose a swap of distinct values
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # perform swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h)
        # linear cooling
        frac = elapsed / max_seconds
        T = T0 * (1.0 - frac)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        # occasional diversification
        if rng.random() < 5e-4:
            h = best_h.copy()
            cur_score = best_score
            swaps = rng.integers(1, 5) + 1
            for _ in range(swaps):
                a = rng.integers(N)
                b = rng.integers(N - 1)
                if b >= a:
                    b += 1
                if h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]
            cur_score = compute_c5(h)
    return best_h, best_score
