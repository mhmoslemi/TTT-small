def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx

def project_to_feasible(v, target_sum):
    # Clip to [0,1] and adjust sum using lambda.
    # Use binary search for lambda.
    lo = -1.0
    hi = np.max(v) + 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Adjust any small residual
    diff = target_sum - x.sum()

        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust smallest entry
            idx = np.argmin(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

import numpy as np, time, math

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def project_to_feasible(v, target):
    # project onto [0,1] and sum constraint using water-filling.
    lo = -1.0
    hi = np.max(v) + 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust a single element safely
            idx = np.argmin(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def gradient_for_lag(h, lag, dx):

        i = np.arange(d, n)
        j = i - d
        g[i] += (1.0 - h[j]) * dx

def compute_c5(h, dx):
    """Maximum overlap integral for current step function."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_to_feasible(v, target):
    """Project vector onto [0,1]^n and sum(v)=target via water‑filling."""
    lo, hi = -1.0, np.max(v) + 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x
