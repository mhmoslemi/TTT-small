def project_to_feasible(x, desired_sum):
    N = len(x)
    h = cp.Variable(N)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h - x)),
                      [cp.sum(h) == desired_sum,
                       h >= 0,
                       h <= 1])
    prob.solve(solver=cp.ECOS, warm_start=True)
    if h.value is None:
        # fallback: simple scaling
        y = np.clip(x, 0, 1)
        s = np.sum(y)
        if s == 0:
            y[0] = desired_sum
        else:
            y = y * (desired_sum / s)
        y = np.clip(y, 0, 1)
        # adjust again maybe
        return y
    return np.array(h.value).flatten()
```

def compute_c5(h):
    """Return the C5 value for a step function h."""
    # Use the pre‑imported evaluator if available, otherwise fall back.
    if 'evaluate_erdos_solution' in globals():
        return evaluate_erdos_solution(h)
    # Fallback (periodic correlation) – should match the evaluator.
    N = len(h)
    dx = 2.0 / N
    # circular correlation via FFT for speed
    fft_h = np.fft.rfft(h)
    fft_one_minus_h = np.fft.rfft(1.0 - h)
    corr = np.fft.irfft(fft_h * np.conj(fft_one_minus_h), n=N)
    # include all circular shifts
    return np.max(corr) * dx

    rng.shuffle(h)
    return h
