def propose_move(h):
    N = len(h)
    i, j = np.random.choice(N, 2, replace=False)
    # direction: transfer amount delta from i to j (increase j)
    # compute epsilon_max = min(h[i], 1-h[j])
    eps_max = min(h[i], 1.0 - h[j])
    if eps_max <= 0: # then transfer opposite direction
        eps_max = min(h[j], 1.0 - h[i])
        if eps_max <= 0:
            # cannot transfer, return None
            return None, None
        else:
            # transfer from j to i
            delta = np.random.rand() * eps_max
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
            return new_h, (i,j,delta,'j->i')
    else:
        # random direction either i->j or j->i
        if np.random.rand() < 0.5:

            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta

def repair_h(h):
    N = len(h)
    # Clip
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    target_sum = N/2.0
    diff = target_sum - current_sum
    # While diff > some small tolerance, distribute proportionally.
    # We can add diff equally to all entries with capacity left.
    if abs(diff) < 1e-12:
        return h
    # To adjust, we can add diff/N to each entry, then clip and iterate; similar to projection onto box with equality.
    # Use simple iterative approach:
    # We'll treat diff >0: need to increase sum.
    # We'll increase values that are <1.
    # We'll do for i in order: add amount = min(diff, 1 - h[i]) to each entry, update diff.
    # Similarly for diff <0: decrease values >0.
    # That yields exact sum.
    if diff > 0:
        # need to increase
        idx = np.where(h < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease
        idx = np.where(h > 0.0)[0]
        for i in idx:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    # After loop diff should be near zero.
    return h

import time
import math
import numpy as np

def evaluate_c5(h):
    """Compute the C5 overlap value for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def repair_h(h):
    """Clip to [0,1] and adjust to satisfy sum(h)=n/2 exactly."""
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    target = n / 2.0
    current = float(np.sum(h))
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase some entries.
        for i in range(n):
            if diff <= 0:
                break
            inc = min(diff, 1.0 - h[i])
            if inc > 0:
                h[i] += inc
                diff -= inc
    else:
        # Need to decrease some entries.
        diff = -diff
        for i in range(n):
            if diff <= 0:
                break
            dec = min(diff, h[i])
            if dec > 0:
                h[i] -= dec
                diff -= dec
    return h

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    start_time = time.time()

    # --- initialise h -------------------------------------------------------

        h0 = np.array(initial_h_values, dtype=np.float64)

    except NameError:

        n = 70
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    h0 = repair_h(h0)
    cur_h = h0.copy()
    cur_score = evaluate_c5(cur_h)

    best_h = cur_h.copy()
    best_score = cur_score

    # --- simulated annealing parameters ------------------------------------
    T0 = 0.02                     # initial temperature
    no_improve = 0                # counter for stagnation detection
    stagnation_limit = 80000      # restart after this many non‑improving steps

    # --- main optimisation loop --------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:
            break

        # Linear temperature schedule
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # Propose a simple mass‑transfer move between two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if np.random.rand() < 0.5:
            # transfer from i to j
            eps_max = min(cur_h[i], 1.0 - cur_h[j])
            if eps_max <= 0:

            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:

            eps_max = min(cur_h[j], 1.0 - cur_h[i])
            if eps_max <= 0:
