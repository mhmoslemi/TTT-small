def compute_c5(h):
    # h: 1-D numpy array length n
    n = h.size
    # correlation h with (1-h)
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / n)

def repair_to_target_ones(arr, target_ones):
    # arr is binary float array (0.0/1.0)
    cur = int(arr.sum())
    if cur == target_ones:
        return arr
    if cur > target_ones:
        # Too many ones: turn some to 0
        idx_one = np.where(arr == 1)[0]
        to_zero = cur - target_ones
        if to_zero > 0:
            flip = np.random.choice(idx_one, to_zero, replace=False)
            arr[flip] = 0.0
    else:
        # Too few ones: turn some zeros to 1
        idx_zero = np.where(arr == 0)[0]
        to_one = target_ones - cur
        flip = np.random.choice(idx_zero, to_one, replace=False)
        arr[flip] = 1.0
    return arr

def random_binary_vector(n, target_ones):
    vec = np.zeros(n, dtype=float)
    vec[np.random.choice(n, target_ones, replace=False)] = 1.0
    return vec

import numpy as np
import time
import random
import math

def compute_c5(h):
    """Return the overlap integral C5 for a binary step function."""
    n = h.size
    # correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * (2.0 / n)

def repair_to_target_ones(arr, target):
    """Flip bits so that exactly target ones remain (arr is float 0/1)."""
    cur = int(arr.sum())
    if cur > target:
        # too many ones → turn some to zero
        idx_one = np.where(arr == 1)[0]
        to_zero = cur - target
        flip = np.random.choice(idx_one, to_zero, replace=False)
        arr[flip] = 0.0
    elif cur < target:
        # too few ones → turn some zeros to one
        idx_zero = np.where(arr == 0)[0]
        to_one = target - cur
        flip = np.random.choice(idx_zero, to_one, replace=False)
        arr[flip] = 1.0
    return arr

def random_binary_vector(n, target):
    """Generate a random binary (0/1) vector of length n with target ones."""
    vec = np.zeros(n, dtype=float)
    vec[np.random.choice(n, target, replace=False)] = 1.0
    return vec

def hillclimb_sa(initial, target, budget_s, start_time):
    """Simulated annealing with single-bit swaps."""
    n = initial.size
    best = initial.copy()
    best_score = compute_c5(best)
    best_global = best.copy()
    best_global_score = best_score

    no_improve = 0
    target_score = 0.3808  # early stop if reached

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s - 0.1:
            break
        if best_global_score <= target_score:
            break

        # temperature decreases linearly
        progress = elapsed / budget_s
        T = max(0.001, (1.0 - progress) * 0.1)

        # select a 1 and a 0 to swap
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_score = compute_c5(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            if cand_score < best_global_score:
                best_global = cand.copy()
                best_global_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = cand_score - best_score
            if random.random() < math.exp(-delta / T):
                best = cand
                best_score = cand_score
                no_improve += 1
            else:
                no_improve += 1

        # random restart if stagnated
        if no_improve >= 20000:
            best = random_binary_vector(n, target)
            best_score = compute_c5(best)
            no_improve = 0

    return best_global, best_global_score

def local_refine_swaps(arr, target, max_evals):
    """Exhaustive single-swap descent limited by max_evals."""
    n = arr.size
    best = arr.copy()
    best_score = compute_c5(best)
    evals = 0
    while evals < max_evals:
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        improved = False
        for i in ones:
            for j in zeros:
                evals += 1
                cand = best.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                cand_score = compute_c5(cand)
                if cand_score < best_score - 1e-12:
                    best = cand
                    best_score = cand_score
                    improved = True
                    break
                if evals >= max_evals:
                    break
            if improved or evals >= max_evals:
                break
        if not improved:
            break
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = None

    target_ones = int(round(n / 2))

    if h0 is not None:
        init_bin = (h0 >= 0.5).astype(float)
        init_bin = repair_to_target_ones(init_bin, target_ones)
    else:
        init_bin = random_binary_vector(n, target_ones)

    # Simulated annealing / hill climbing
    best_vec, best_score = hillclimb_sa(init_bin, target_ones, budget_s, start_time)

    # Small local refinement if time permits
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2.0:
        max_evals = int(min(20000, max(0, time_left * 5000)))
        best_vec, best_score = local_refine_swaps(best_vec, target_ones, max_evals)

    # Ensure constraints exactly met
    best_vec = repair_to_target_ones(best_vec, target_ones)

    final_score = compute_c5(best_vec)
    return (best_vec, final_score, n)
