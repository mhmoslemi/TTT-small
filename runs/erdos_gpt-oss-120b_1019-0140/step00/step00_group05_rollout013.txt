def run(seed=42, budget_s=1100, **kwargs):
    import time, random, numpy as np

    # Seed RNGs
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Obtain initial construction
    # --------------------------------------------------------------
    try:
        # provided by the harness when a parent exists
        h = np.array(initial_h_values, copy=True, dtype=float)
        N = h.size
    except Exception:
        # no parent – create a fresh binary start
        N = 200                     # modest size (<1000) for fast search
        h = np.zeros(N, dtype=float)
        ones = np.random.choice(N, N // 2, replace=False)
        h[ones] = 1.0

    # Ensure feasibility: values in [0,1] and sum = N/2
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    target_sum = N / 2.0
    if not np.isclose(current_sum, target_sum):
        # force binary and correct count
        h = (h > 0.5).astype(float)
        ones = int(h.sum())
        if ones > N // 2:
            idx = np.where(h == 1)[0]
            drop = np.random.choice(idx, ones - N // 2, replace=False)
            h[drop] = 0.0
        elif ones < N // 2:
            idx = np.where(h == 0)[0]
            add = np.random.choice(idx, N // 2 - ones, replace=False)
            h[add] = 1.0

    dx = 2.0 / N

    # --------------------------------------------------------------
    # Helper to evaluate C5
    # --------------------------------------------------------------
    def compute_c5(arr):
        """Fast C5 computation using full correlation."""
        return np.max(np.correlate(arr, 1.0 - arr, mode='full')) * dx

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    try:
        best_score = evaluate_erdos_solution(h)  # official evaluation if present
    except Exception:
        best_score = compute_c5(h)
    best_h = h.copy()

    # --------------------------------------------------------------
    # Simulated‑annealing / stochastic search
    # --------------------------------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.95          # keep margin for finalisation
    max_temp = 0.010                     # start temperature
    min_temp = 1e-6                      # final temperature
    prob_big = 0.002                     # chance of a large random jump
    eps_base = 0.12                      # maximal mass transferred in a step

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break

        # linear temperature schedule
        t_frac = elapsed / max_time
        temp = max_temp * (1.0 - t_frac) + min_temp * t_frac

        # ----------------------------------------------------------
        # Large random restart move (occasionally)
        # ----------------------------------------------------------
        if random.random() < prob_big:
            cand = np.zeros(N, dtype=float)
            cand[np.random.choice(N, N // 2, replace=False)] = 1.0
            try:
                cand_score = evaluate_erdos_solution(cand)
            except Exception:
                cand_score = compute_c5(cand)

            # Metropolis acceptance
            if cand_score < best_score or np.exp(-(cand_score - best_score) / max(temp, 1e-12)) > random.random():
                h = cand
                best_h = cand.copy()
                best_score = cand_score
            continue

        # ----------------------------------------------------------
        # Continuous epsilon transfer move
        # ----------------------------------------------------------
        # pick two distinct indices i (to increase) and j (to decrease)
        i, j = random.sample(range(N), 2)

        # Determine feasible epsilon (must keep values in [0,1])
        eps_up = 1.0 - h[i]
        eps_down = h[j]
        eps_max = min(eps_up, eps_down, eps_base)

        if eps_max <= 0:
            continue  # cannot transfer any mass between these two

        # random epsilon in (0, eps_max]
        epsilon = random.random() * eps_max

        # apply transfer
        h[i] += epsilon
        h[j] -= epsilon

        try:
            new_score = evaluate_erdos_solution(h)
        except Exception:
            new_score = compute_c5(h)

        # Metropolis rule
        if new_score < best_score or np.exp(-(new_score - best_score) / max(temp, 1e-12)) > random.random():
            # accept move
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

            # revert move
            h[i] -= epsilon
            h[j] += epsilon

    # Final sanity check and return
    # --------------------------------------------------------------

    best_h = np.clip(best_h, 0.0, 1.0)

    try:
        final_score = evaluate_erdos_solution(best_h)
    except Exception:
        final_score = compute_c5(best_h)
