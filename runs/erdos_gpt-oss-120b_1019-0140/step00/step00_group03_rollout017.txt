def make_binary_initial(arr):
    n = len(arr)
    w = n // 2
    # binarize
    bin_arr = (arr >= 0.5).astype(float)
    current_w = int(bin_arr.sum())
    # adjust
    if current_w > w:
        # too many ones: turn some ones to zero
        ones_idx = np.where(bin_arr == 1)[0]
        to_zero = np.random.choice(ones_idx, current_w - w, replace=False)
        bin_arr[to_zero] = 0
    elif current_w < w:
        zeros_idx = np.where(bin_arr == 0)[0]
        to_one = np.random.choice(zeros_idx, w - current_w, replace=False)
        bin_arr[to_one] = 1
    return bin_arr
```

import numpy as np
import time
import random

def compute_c5(arr, dx):
    # correlation in 'full' mode yields length 2n-1
    # compute max of correlation of arr with (1 - arr)
    # (arr is 1D array)
    return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

def binarize_to_weight(arr, target_weight, rng):
    # arr is a 1-D array of floats
    bin_arr = (arr >= 0.5).astype(float)
    cur = int(bin_arr.sum())
    if cur > target_weight:
        idx = np.where(bin_arr == 1)[0]
        to_zero = rng.choice(idx, cur - target_weight, replace=False)
        bin_arr[to_zero] = 0.0
    elif cur < target_weight:
        idx = np.where(bin_arr == 0)[0]
        to_one = rng.choice(idx, target_weight - cur, replace=False)
        bin_arr[to_one] = 1.0
    return bin_arr

def hill_climb(start_vec, weight, dx, time_limit, rng):
    # Returns best vector and its C5 value found within time limit.
    best_local = start_vec.copy()
    best_score = compute_c5(best_local, dx)
    current = start_vec.copy()
    # counts attempts without improvement
    no_improve = 0
    max_no_improve = 5000
    while time.time() < time_limit:
        # choose random positions to swap
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        current[i] = 0.0
        current[j] = 1.0
        new_score = compute_c5(current, dx)
        if new_score < best_score - 1e-14:
            best_local = current.copy()
            best_score = new_score
            no_improve = 0
        else:
            # revert
            current[i] = 1.0
            current[j] = 0.0
            no_improve += 1
        if no_improve >= max_no_improve:
            break
    return best_local, best_score

import numpy as np
import time
import random

def compute_c5(arr, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for a discrete step function."""
    # full linear correlation of h with (1-h)
    return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

def binarize_to_weight(arr, target_weight, rng):
    """Convert any array to a binary vector with exactly target_weight ones."""
    bin_arr = (arr >= 0.5).astype(float)
    cur = int(bin_arr.sum())
    if cur > target_weight:
        idx = np.where(bin_arr == 1)[0]
        to_zero = rng.choice(idx, cur - target_weight, replace=False)
        bin_arr[to_zero] = 0.0
    elif cur < target_weight:
        idx = np.where(bin_arr == 0)[0]
        to_one = rng.choice(idx, target_weight - cur, replace=False)
        bin_arr[to_one] = 1.0
    return bin_arr

def hill_climb(start_vec, weight, dx, time_limit, rng):
    """
    Greedy hill climbing: repeatedly swap a 1 with a 0 if it strictly reduces C5.
    Stops after time limit or after a streak of non‑improving swaps.
    Returns the best vector encountered and its C5 value.
    """
    best_local = start_vec.copy()
    best_score = compute_c5(best_local, dx)
    current = start_vec.copy()
    no_improve = 0
    max_no_improve = 5000

    while time.time() < time_limit:
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        current[i] = 0.0
        current[j] = 1.0
        new_score = compute_c5(current, dx)

        if new_score < best_score - 1e-14:
            best_local = current.copy()
            best_score = new_score
            no_improve = 0
        else:
            # revert swap
            current[i] = 1.0
            current[j] = 0.0
            no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_local, best_score
