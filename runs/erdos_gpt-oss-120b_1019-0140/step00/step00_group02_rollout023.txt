def project_onto_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # Ensures after projection sum(v) == target_sum and lower <= v_i <= upper
    # Use bisection on lambda: x = clip(v + lam, lower, upper); find lam such that sum(x) = target.
    # If target_sum < lower*n or > upper*n: impossible, but we assume feasible.
    # Determine lambda bounds:
    lam_low = -np.max(v - lower) - 1.0  # ensure v+lam <= lower.
    lam_high = upper - np.min(v) + 1.0  # ensure >= upper.
    # But we can start with lam_low = -np.max(v) - 1, lam_high = 1 - np.min(v) + 1.
    for _ in range(max_iter):

        x = np.clip(v + lam_mid, lower, upper)
        s = np.sum(x)

            return x
        if s < target_sum:
            # need larger lam to increase sum
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    # final
    return np.clip(v + lam_mid, lower, upper)

import time
import numpy as np

TARGET_BOUND = 0.3808  # desirable upper bound for C5


def compute_c5(h):
    """Return C5 for a given step function h (numpy array)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)


def random_binary_best(n, time_budget, best_sofar=np.inf):
    """
    Random search for binary vectors of length n with exactly n/2 ones.
    Returns the best vector found within time_budget seconds.
    """
    target_sum = n // 2
    dx = 2.0 / n
    best_h = None
    best_c5 = best_sofar
    start = time.time()

    # Pre‑allocate a template array for speed
    template = np.zeros(n, dtype=np.float64)
    template[:target_sum] = 1.0

    while time.time() - start < time_budget:
        # shuffle a copy of the template
        h = template.copy()
        np.random.shuffle(h)

        # compute C5
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = dx * np.max(corr)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 <= TARGET_BOUND:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Try a modest even grid first (n = 64). 64 points give dx = 0.03125,
    # so a count of 12 corresponds to C5 = 0.375 < TARGET_BOUND.
    n_candidates = [64, 128, 256]

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in n_candidates:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # allocate a slice of the remaining budget (up to 300 s per size)
        remaining = budget_s - elapsed
        time_for_n = min(remaining * 0.5, 300.0)
        if time_for_n < 0.1:  # too little time left
            break

        # Random binary search for this grid size
        h, c5 = random_binary_best(n, time_for_n, best_c5)

        if h is not None and c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n
            if best_c5 <= TARGET_BOUND:
                # sufficient bound, stop early
                break

    # If nothing was found (unlikely), fall back to any supplied construction
    if best_h is None:
        init = globals().get("initial_h_values")
        if init is not None:
            best_h = np.array(init, dtype=np.float64)
            best_n = best_h.size
            best_c5 = compute_c5(best_h)
        else:
            # trivial fallback: constant 0.5 on two points
            best_n = 2
            best_h = np.full(best_n, 0.5, dtype=np.float64)
            best_c5 = compute_c5(best_h)

    # Ensure the returned vector satisfies the exact integral constraint
    # (minor floating‑point adjustments may be needed)
    target_sum = best_n / 2.0
    current_sum = best_h.sum()
    if abs(current_sum - target_sum) > 1e-9:
        # simple redistribution to fix the sum while keeping bounds
        diff = target_sum - current_sum
        # distribute the difference uniformly and clip
        best_h += diff / best_n
        np.clip(best_h, 0.0, 1.0, out=best_h)

    # Final C5 recomputation for safety
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), best_n
