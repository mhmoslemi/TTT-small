import numpy as np
import time

def project_onto_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the set {x: 0<=x<=1, sum(x)=target_sum} using binary search."""
    # Clip to box first
    x = np.clip(v, 0.0, 1.0)
    # Simple binary search for λ in x_i = clip(v_i - λ, 0, 1)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(30):
        lam = (lo + hi) / 2.0
        x = v - lam
        np.clip(x, 0.0, 1.0, out=x)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = v - (lo + hi) / 2.0
    np.clip(x, 0.0, 1.0, out=x)
    # Final tiny correction to hit the sum exactly (within numerical tolerance)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # adjust one non‑boundary element if possible
        mask = (x > 0.0) & (x < 1.0)
        if np.any(mask):
            idx = np.where(mask)[0][0]
            x[idx] = min(1.0, max(0.0, x[idx] + diff))
        else:
            # all at bounds – distribute evenly (very rare)
            x += diff / len(x)
            np.clip(x, 0.0, 1.0, out=x)
    return x

def compute_c5(h: np.ndarray) -> float:
    """Return C5 for a given step function h (dx = 2/len(h))."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_shift_and_c5(h: np.ndarray):
    """Return (shift, C5 value, full correlation array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)          # integer shift in sample units
    c5_val = corr[max_idx] * (2.0 / n)
    return shift, c5_val, corr

def gradient_of_corr_at_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation at a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    i = np.arange(n)
    j = i + shift
    mask = (j >= 0) & (j < n)
    i_valid = i[mask]
    j_valid = j[mask]
    # contribution of each pair (i, j)
    g[i_valid] += 1.0 - h[j_valid]
    g[j_valid] -= h[i_valid]
    return g

def init_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Construct an initial feasible h for given resolution n."""
    if 'initial_h_values' in globals():
        init = globals()['initial_h_values']
        if len(init) == n:
            h = np.array(init, dtype=float).copy()
        else:
            xs = np.linspace(0.0, 2.0, len(init), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(xt, xs, init)
    else:
        h = np.full(n, 0.5, dtype=float)
    target = n / 2.0
    h = project_onto_box_sum(h, target)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf

    # Resolutions to explore (starting from any provided initial size)
    resolutions = []
    if 'initial_h_values' in globals():
        resolutions.append(len(initial_h_values))
    resolutions.extend([45, 90, 180, 360])
    # keep unique and sorted
    resolutions = sorted(set(resolutions))

    for n in resolutions:
        # time guard
        if time.time() - start >= budget_s * 0.95:
            break

        # initialise h: interpolate previous best if size differs, else use init_h
        if best_h is not None and len(best_h) != n:
            xs = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(xt, xs, best_h)
            h = project_onto_box_sum(h, n / 2.0)
        else:
            h = init_h(n, rng)

        target_sum = n / 2.0

        # iteration budget per resolution (coarser grids get more steps)
        if n <= 45:
            max_iters = 20000
        elif n <= 90:
            max_iters = 12000
        elif n <= 180:
            max_iters = 8000
        else:
            max_iters = 4000

        lr_initial = 0.8
        it = 0
        while it < max_iters:
            # time guard inside loop
            if time.time() - start >= budget_s * 0.95:
                break

            shift, c5_val, _ = max_shift_and_c5(h)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()

            # Gradient descent step on the worst shift
            g = gradient_of_corr_at_shift(h, shift)
            lr = lr_initial * (1.0 - it / max_iters)
            if lr < 1e-5:
                lr = 1e-5
            y = h - lr * g
            h = project_onto_box_sum(y, target_sum)

            # occasional small random perturbation to escape plateaus
            if (it + 1) % 2000 == 0:
                noise = rng.normal(scale=0.02, size=n)
                h = project_onto_box_sum(h + noise, target_sum)

            it += 1

    # Fallback if no improvement was ever recorded
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
    else:
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
