import time
import random
import math
import numpy as np

def _project_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector onto the box [0,1]^n with given L1 sum = target_sum."""
    low, high = -1.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        tmp = np.clip(v - mid, 0.0, 1.0)
        s = tmp.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    proj = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # Tiny residual adjustment (optional)
    residual = target_sum - proj.sum()
    if abs(residual) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            add = residual / free.sum()
            proj[free] += add
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Compute the overlap bound C5 for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimizer for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Desired discretisation size (trade‑off between resolution and speed)
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------ #
    # Initialise from parent if provided, otherwise random feasible start.
    # ------------------------------------------------------------------ #
    init = kwargs.get('initial_h_values')
    if init is not None:
        init = np.asarray(init, dtype=np.float64)
        # Upsample / downsample to required size using linear interpolation
        if init.size != n_points:
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
    else:
        # pure random start, then project onto feasible set
        h = np.random.rand(n_points)
    h = _project_feasible(h, target_sum)

    # ------------------------------------------------------------------ #
    # Core optimisation loop (simulated annealing + gradient steps)
    # ------------------------------------------------------------------ #
    cur_h = h.copy()
    cur_score = _compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_score = cur_score

    # Simulated‑annealing schedule
    T = 1.0
    T_end = 1e-4
    alpha = 0.9995
    step0 = 0.2     # base step size for gradient moves
    sigma0 = 0.15   # base noise magnitude for random moves

    start_time = time.time()
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break

        # ----- decide on move type -------------------------------------------------
        if random.random() < 0.2:  # random perturbation
            sigma = sigma0 * T
            cand = cur_h + sigma * np.random.randn(n_points)
            cand = _project_feasible(cand, target_sum)
        else:                     # gradient move on the worst lag
            full_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            idx_max = int(np.argmax(full_corr))
            shift = idx_max - (n_points - 1)

            # gradient = (A_shift + A_-shift) h   with zero padding
            grad = np.zeros_like(cur_h)
            if shift > 0:
                grad[:-shift] += cur_h[shift:]
                grad[shift:] += cur_h[:-shift]
            elif shift < 0:
                s = -shift
                grad[s:] += cur_h[:-s]
                grad[:-s] += cur_h[s:]
            else:
                grad = 2.0 * cur_h

            step = step0 * T
            cand = cur_h + step * grad
            cand = _project_feasible(cand, target_sum)

        cand_score = _compute_c5(cand, dx)

        # ----- keep global best ----------------------------------------------------
        if cand_score < best_score:
            best_h = cand.copy()
            best_score = cand_score

        # ----- SA acceptance -------------------------------------------------------
        delta = cand_score - cur_score
        if delta < 0.0 or random.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h = cand
            cur_score = cand_score

        # ----- cooling -------------------------------------------------------------
        T = max(T_end, T * alpha)
        it += 1

    # -------------------------------------------------------------------------- #
    # Return solution as plain Python list (as required by the harness)
    # -------------------------------------------------------------------------- #
    return best_h.tolist(), float(best_score), n_points
