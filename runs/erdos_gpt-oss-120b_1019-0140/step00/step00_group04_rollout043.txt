    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:

        h = h * (target_sum / cur_sum)
        # clip to [0,1]
        np.clip(h, 0.0, 1.0, out=h)

        cur_sum = h.sum()

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    
    # try initial values
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)
        n = len(init_arr)
        h = init_arr.copy()

        n = 84  # choose an even number > 42
        w = n // 2
        # start with 0/1 vector
        h = np.zeros(n, dtype=np.float64)
        h[:w] = 1.0
        rng.shuffle(h)
    
    target_sum = n / 2.0
    cur_sum = h.sum()
    # adjust sum if needed
    if abs(cur_sum - target_sum) > 1e-12:
        scaling = target_sum / cur_sum
        h *= scaling
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # adjust residual
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # distribute to a non-bound element
            for i in range(n):
                if (diff > 0 and h[i] < 1.0) or (diff < 0 and h[i] > 0.0):
                    adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-12:
                        break
    dx = 2.0 / n
    # compute objective
    def compute_c5(arr):
        corr = np.correlate(arr, 1 - arr, mode='full')
        return corr.max() * dx
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # annealing parameters
    T0 = 0.02
    # allow many iterations
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.1:
            break
        # temperature linearly decreasing
        T = T0 * (1 - elapsed / budget_s)
        # pick two indices
        i, j = rng.integers(0, n, size=2)

import numpy as np
import time
import math

def compute_c5(arr):
    """Return the C5 bound for a step function represented by `arr`."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using linear (zero‑padded) convolution
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return corr.max() * dx

def _fix_sum(vec, target):
    """Adjust `vec` (in‑place) so that its sum equals `target` without leaving [0,1]."""
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # distribute the residual to any element that can move
    for i in range(vec.size):
        if diff > 0 and vec[i] < 1.0:
            adj = min(diff, 1.0 - vec[i])
            vec[i] += adj
            diff -= adj
        elif diff < 0 and vec[i] > 0.0:
            adj = max(diff, -vec[i])
            vec[i] += adj
            diff -= adj
        if abs(diff) < 1e-12:
            break
