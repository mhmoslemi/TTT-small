import time
import math
import random
import numpy as np

def shift_vec(h: np.ndarray, shift: int) -> np.ndarray:
    """Return vector g where g[i] = h[i+shift] if in range else 0."""
    n = h.shape[0]
    if shift > 0:
        g = np.empty_like(h)
        g[: n - shift] = h[shift:]
        g[n - shift :] = 0.0
    elif shift < 0:
        s = -shift
        g = np.empty_like(h)
        g[s:] = h[: n - s]
        g[:s] = 0.0
    else:
        g = h.copy()
    return g


def compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
