def project_onto_bounds_and_sum(x, target_sum, low=0.0, high=1.0, max_iter=50):
    # find lambda such that sum(clip(x + lam, low, high)) = target_sum.
    lo = -np.max(x) - 1.0  # large negative
    hi = high - np.min(x) + 1.0  # large positive
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0

def project_onto_bounds_and_sum(x, target, low=0.0, high=1.0, tol=1e-12, max_iter=60):
    # solve sum(clip(x+lam, low, high)) = target via binary search on lam
    # lam lower bound
    lo = -np.max(x) - 1.0  # ensure all x+lam <= low
    hi = high - np.min(x) + 1.0  # ensure all x+lam >= high
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, low, high)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            hi = lam
        else:
            lo = lam
    # final clip
    return np.clip(x + (lo + hi) / 2.0, low, high)

def upsample_initial(initial, n):
    # use linear interpolation

import numpy as np, time

def project_onto_bounds_and_sum(x, target, low=0.0, high=1.0, tol=1e-12, max_iter=60):
    lo = -np.max(x) - 1.0
    hi = high - np.min(x) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, low, high)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) / 2.0, low, high)

def upsample_initial(initial, n):
    m = len(initial)
    old_x = np.linspace(0, 2, m, endpoint=False) + (2.0 / m) / 2.0
    new_x = np.linspace(0, 2, n, endpoint=False) + (2.0 / n) / 2.0
    new_h = np.interp(new_x, old_x, initial)
    target = n / 2.0
    new_h = project_onto_bounds_and_sum(new_h, target, 0.0, 1.0)
    return new_h

def compute_c5_from_corr(corr, n):
    dx = 2.0 / n
    return 1.0 - np.min(corr) * dx

def gradient_for_lag(h, lag):
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            g[:n-lag] += h[lag:]
            g[lag:] += h[:n-lag]
    else:
        L = -lag
        if L < n:
            g[L:] += h[:n-L]
            g[:n-L] += h[L:]
    return g

    best_h = h.copy()
    current_c5 = best_c5
    start = time.time()
    elapsed = 0.0

    stagnant = 0
