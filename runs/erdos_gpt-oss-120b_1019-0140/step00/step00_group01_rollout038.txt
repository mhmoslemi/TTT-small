```
def project_onto_bounds_and_sum(v, S, lower=0., upper=1., max_iter=100, tol=1e-9):
    n = v.shape[0]
    # initial low, high
    low = np.min(v) - upper
    high = np.max(v) - lower
    # ensure low < high
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - S) < tol:
            return x
        if s > S:
            low = mid
        else:
            high = mid
    # final return
    return np.clip(v - (low+high)/2.0, lower, upper)
```

```
def project_to_simplex_box(v, S, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # Ensure lower <= upper.
    low = np.min(v) - upper
    high = np.max(v) - lower
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - S) <= tol:
            return x
        if s > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, lower, upper)
```

import numpy as np, time, math

def _project_to_bounds_sum(v, S, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    # Project vector v onto the box [lower,upper] with equality sum(v)=S.
    low = np.min(v) - upper
    high = np.max(v) - lower
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - S) <= tol:
            return x
        if s > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, lower, upper)

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _gradient_max_lag(h, lag):
    n = h.shape[0]
    g = np.ones_like(h)
    if lag >= 0:
        l = lag
        if l < n:
            g[:n - l] -= h[l:]
            g[l:] -= h[:n - l]
    else:
        l = -lag
        g[l:] -= h[:n - l]
        g[:n - l] -= h[l:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine number of points
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=np.float64)
        n = h_init.size
        target_sum = n / 2.0
        h_init = _project_to_bounds_sum(h_init, target_sum)
        start_vectors = [h_init]
    else:
        n = 200
        target_sum = n / 2.0
        start_vectors = []
    # Add a deterministic triangular start (always feasible)
    xs = np.linspace(0.0, 2.0, n, endpoint=False) + 2.0 / n / 2.0  # sample at cell centers
    tri = 1.0 - np.abs(xs - 1.0)
    tri = _project_to_bounds_sum(tri, target_sum)
    start_vectors.append(tri)

    best_h = None
    best_score = np.inf

    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # leave margin

    # Main loop with multiple restarts
    while time.time() < end_time:
        # Choose a start vector
        if start_vectors and rng.random() < 0.5:
            h = start_vectors[int(rng.integers(0, len(start_vectors)))].copy()
        else:
            # random feasible vector
            v = rng.random(n)
            h = _project_to_bounds_sum(v, target_sum)

        cur_h = h
        cur_score = _c5_value(cur_h)

        # Local optimisation
        lr0 = 0.6
        max_local_iters = 4000
        for it in range(max_local_iters):
            if time.time() >= end_time:
                break
            # compute correlation and locate worst lag
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx = int(np.argmax(corr))
            lag = idx - (n - 1)

            # gradient step targeting the worst lag
            g = _gradient_max_lag(cur_h, lag)
            lr = lr0 * (0.9995 ** it)
            cand = cur_h - lr * g
            cand = _project_to_bounds_sum(cand, target_sum)
            cand_score = _c5_value(cand)

            # accept if improvement or with small probability (simulated annealing)
            if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / (lr + 1e-12)):
                cur_h = cand
                cur_score = cand_score
                if cand_score < best_score:

        best_score = _c5_value(best_h)

import numpy as np
import time
import math

def _project_to_bounds_sum(v, S, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """Project v onto {x | lower≤x≤upper, sum(x)=S} using a simple bisection."""
    low = np.min(v) - upper
    high = np.max(v) - lower
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - S) <= tol:
            return x
        if s > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, lower, upper)

def _c5_value(h):
    """Compute C5 for a discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _gradient_max_lag(h, lag):
    """Subgradient of the worst‑lag term with respect to h."""
    n = h.shape[0]
    g = np.ones_like(h)
    if lag >= 0:
        l = lag
        if l < n:
            g[: n - l] -= h[l:]
            g[l:] -= h[: n - l]
    else:
        l = -lag
        g[l:] -= h[: n - l]
        g[: n - l] -= h[l:]
    return g
