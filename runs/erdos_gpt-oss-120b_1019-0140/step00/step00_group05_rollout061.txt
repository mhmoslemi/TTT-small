def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

import numpy as np
import random
import time
import math

def _compute_c5(h):
    """Return the C5 value for a vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' includes all lags
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def _project_to_constraints(z, target_sum):
    """
    Project vector z onto the box [0,1]^n and the affine hyperplane sum(z)=target_sum.
    Simple iterative Dykstra-like scheme; works well for modest dimensions.
    """
    y = np.clip(z, 0.0, 1.0)
    diff = target_sum - y.sum()
    # iterate a few times to satisfy both constraints
    for _ in range(15):
        if abs(diff) < 1e-12:
            break
        free = (y > 0.0) & (y < 1.0)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds; distribute remaining error uniformly (will be tiny)
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
            diff = target_sum - y.sum()
            break
        delta = diff / nfree
        y[free] += delta
        y = np.clip(y, 0.0, 1.0)
        diff = target_sum - y.sum()
    # final tiny correction if needed
    if abs(diff) > 1e-9:
        # adjust a random non‑boundary entry
        idxs = np.where((y > 0.0) & (y < 1.0))[0]
        if idxs.size == 0:
            idx = np.random.randint(len(y))
        else:
            idx = np.random.choice(idxs)
        y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
    return y

def _greedy_move(h, offset):
    """
    Perform a small mass transfer that directly reduces the contribution
    for the given offset (the worst shift). Returns a new vector.

        i_start, i_end = d, n
    if i_start >= i_end:
        return h  # nothing to do
    # contributions for this offset
    src_idxs = np.arange(i_start, i_end)
    dst_idxs = src_idxs + offset
    contrib = h[src_idxs] * (1.0 - h[dst_idxs])
    # choose index with largest contribution
    i_max = src_idxs[np.argmax(contrib)]
    src = i_max
    dst = i_max + offset
    # amount that can be moved from src to dst without breaking bounds
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0.0:
        return h
    # move a modest fraction
    epsilon = max_transfer * random.random() * 0.5  # up to half of what is feasible
    new_h = h.copy()
    new_h[src] -= epsilon
    new_h[dst] += epsilon
    return new_h

def _random_transfer(h):
    """Randomly pick two indices and transfer a random admissible mass."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    # randomly decide direction
    if random.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0.0:
        return h
    epsilon = max_transfer * random.random() * 0.5
    new_h = h.copy()
    new_h[src] -= epsilon
    new_h[dst] += epsilon
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    random.seed(seed)
    np.random.seed(seed)

    # ----- initialise -------------------------------------------------
    try:
        # use the provided parent construction if it exists
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    except NameError:
        # fallback: uniform 0.5 on a modest grid
        n = 70
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0           # because ∑h = n/2  <=> integral = 1
    # enforce constraints exactly on the start vector
    h0 = _project_to_constraints(h0, target_sum)

    cur_h = h0.copy()
    cur_c5 = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    time_limit = budget_s * 0.96   # a safety margin

    # ----- main optimisation loop ------------------------------------
    # parameters
    prob_greedy = 0.7            # probability to try a greedy reduction step
    stale_counter = 0
    stale_limit = int(0.02 * budget_s * 100)  # rough iteration count

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break

        # decide move type
        if random.random() < prob_greedy:
            # compute the worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_idx = int(np.argmax(corr))
            offset = max_idx - (n - 1)   # shift amount (may be negative)
            new_h = _greedy_move(cur_h, offset)
        else:
            new_h = _random_transfer(cur_h)

        # project back onto feasible set (the moves already keep sum, but clipping may be needed)
        new_h = _project_to_constraints(new_h, target_sum)

        new_c5 = _compute_c5(new_h)

        # accept if improvement, otherwise keep the current state
        if new_c5 < cur_c5 - 1e-12:
            cur_h = new_h
            cur_c5 = new_c5
            stale_counter = 0
        else:
            stale_counter += 1
            # occasional escape: if many non‑improving steps, try a larger random perturbation
            if stale_counter > 5000:
                # make a bigger random jump while staying feasible
                perturb = np.random.rand(n)
                perturb = perturb * target_sum / perturb.sum()
                cur_h = _project_to_constraints(perturb, target_sum)
                cur_c5 = _compute_c5(cur_h)
                stale_counter = 0

        # keep the best ever seen
        if cur_c5 < best_c5 - 1e-12:
            best_h = cur_h.copy()
            best_c5 = cur_c5

    # final sanity projection (ensuring exact constraints)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    # return plain Python list for compatibility
    return best_h.tolist(), best_c5, n
