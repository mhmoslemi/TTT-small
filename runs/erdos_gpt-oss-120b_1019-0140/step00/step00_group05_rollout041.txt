import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

def project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum(v)=target.
    Uses a simple binary search on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    # expand bounds if needed
    for _ in range(30):
        s_lo = np.clip(v - lo, 0.0, 1.0).sum()
        s_hi = np.clip(v - hi, 0.0, 1.0).sum()
        if s_lo < target:
            lo -= 10.0 * (abs(lo) + 1.0)
        elif s_hi > target:
            hi += 10.0 * (abs(hi) + 1.0)
        else:
            break
    # binary search for multiplier
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_mid = np.clip(v - mid, 0.0, 1.0).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)

    # tiny residual correction to hit the sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
            if idx.size:
                i = idx[0]
                x[i] = min(1.0, x[i] + diff)
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size:
                i = idx[0]
                x[i] = max(0.0, x[i] + diff)
    return x

def generate_h(a: float, b: float, n: int) -> np.ndarray:
    """Create a three‑plateau vector from parameters a (length of 1‑plateau) and b (length of middle plateau)."""
    dx = 2.0 / n
    t = 0.0
    if b > 0.0:

    t = np.clip(t, 0.0, 1.0)

    h = np.where(x < a, 1.0, np.where(x < a + b, t, 0.0))
    target = n / 2.0
    h = project_onto_bounds_sum(h, target)
    return h

def gradient_step(h: np.ndarray, step: float) -> np.ndarray:
    """One gradient‑descent step aimed at reducing the worst shift."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx_max = conv.argmax()
    shift = idx_max - (n - 1)           # integer shift (can be negative)

    i = np.arange(n)
    pos = i + shift
    term1 = np.where((pos >= 0) & (pos < n), 1.0 - h[pos], 1.0)

    neg = i - shift
    term2 = np.where((neg >= 0) & (neg < n), h[neg], 0.0)

    grad = term1 - term2               # ∂f_max/∂h_i  (up to a constant factor)

    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    h_new = project_onto_bounds_sum(h_new, target)
    return h_new
