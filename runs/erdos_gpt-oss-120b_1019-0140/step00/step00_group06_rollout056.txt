def repair_vector(v, n):
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    target = n / 2.0
    if s == 0:
        # assign equal values
        v = np.full_like(v, target / n)
        return np.clip(v,0,1)
    factor = target / s
    v = v * factor
    # clip again
    v = np.clip(v, 0, 1)
    # after clipping could break sum again, re-scale remaining difference proportionally to non-clipped elements
    # use iterative approach:
    for _ in range(5):
        diff = target - np.sum(v)
        if abs(diff) < 1e-9:
            break
        # compute mask of entries not at bounds
        mask = (v > 0) & (v < 1)
        if not np.any(mask):
            # cannot distribute, break
            break
        # distribute diff proportionally
        v[mask] += diff / np.sum(mask)
        v = np.clip(v, 0, 1)
    # final adjustment: ensure sum close to target
    # small numerical error okay; we could adjust first element
    s = np.sum(v)
    if abs(s - target) > 1e-6:
        v[0] = np.clip(v[0] + (target - s), 0, 1)
    return v

def hill_climb_n(h0, rng, time_budget):
    import time, numpy as np
    def compute_c5(h):
        dx = 2.0 / h.shape[0]
        # correlation of h and (1-h)
        # np.correlate returns array of length 2n-1
        # we need max of correlation * dx
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start = time.time()
    curr_h = h0.copy()
    curr_score = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_score = curr_score
    n = len(h0)
    # parameters
    temp = 0.001
    cooling = 0.9999
    # iterate
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # propose move
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        # compute feasible delta
        # lower bound: -curr_h[i] to keep >=0; also curr_h[j] - delta <=1 => delta >= curr_h[j] -1
        lower = max(-curr_h[i], curr_h[j] - 1.0)
        # upper bound: 1 - curr_h[i] to keep <=1; also delta <= curr_h[j] to keep j >=0
        upper = min(1.0 - curr_h[i], curr_h[j])
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        # apply move
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_score = compute_c5(new_h)
        # accept if better or by probability
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # metropolis probability
            prob = np.exp(-(new_score - curr_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score:
                best_h = new_h
                best_score = new_score
        # temperature decay
        temp = max(temp * cooling, 1e-8)
        # occasional random restart if stagnation? optional
        # we could track iteration count without improvement
        # but skip for simplicity
    return best_h, best_score

def hill_climb_continuous(h_start, rng, time_budget):
    start = time.time()
    n = len(h_start)
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_score = best_score
    temp = 0.01
    cooling = 0.9999
    while time.time() - start < time_budget:
        # pick two indices i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute feasible delta bounds:
        lower = max(-curr_h[i], curr_h[j] - 1.0)
        upper = min(1.0 - curr_h[i], curr_h[j])
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_score = compute_c5(new_h)
        if new_score < curr_score or rng.random() < np.exp(-(new_score - curr_score) / (temp + 1e-12)):
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score:
                best_h = new_h
                best_score = new_score
        temp = max(temp * cooling, 1e-8)
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return max overlap C5 for step function values h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def repair_vector(v, n):
    """Clip to [0,1] and enforce sum(v) = n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(v)
    if s == 0:
        v[:] = target / n
        return v
    factor = target / s
    v *= factor
    v = np.clip(v, 0.0, 1.0)
    # iterative re‑balance for any clipping loss
    for _ in range(5):
        diff = target - np.sum(v)
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        if not np.any(mask):
            break
        v[mask] += diff / np.count_nonzero(mask)
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction on first entry
    diff = target - np.sum(v)
    if abs(diff) > 1e-9:
        v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v

def sum_constraint(h, n):
    """Equality constraint: integral (= sum * 2/n) equals 1."""
    return np.sum(h) - n * 0.5

def p_norm_objective(h, n, p):
    """Smooth approximation of max overlap using L^p norm."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # all entries are non‑negative, so plain Lp norm works
    return np.linalg.norm(corr, ord=p)

def hill_climb_continuous(h_start, rng, time_budget):
    """Simple SA‑style improvement preserving box and sum constraints."""
    start = time.time()
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_score = best_score
    temp = 1e-2
    cooling = 0.9999
    while time.time() - start < time_budget:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # feasible delta keeping both entries in [0,1]
        lo = max(-curr_h[i], curr_h[j] - 1.0)
        hi = min(1.0 - curr_h[i], curr_h[j])
        if lo >= hi:
            continue
        delta = rng.uniform(lo, hi)
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_score = compute_c5(new_h)
        if new_score < curr_score or rng.random() < np.exp(-(new_score - curr_score) / (temp + 1e-12)):
            curr_h, curr_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h, new_score
        temp = max(temp * cooling, 1e-8)
    return best_h, best_score

def optimize_slsqp(h0, n, time_budget, rng):
    """Run a few SLSQP minimizations with an L^p smooth objective."""
    # allocate a modest portion for the deterministic optimizer
    sub_budget = min(time_budget * 0.6, 30.0)  # cap to keep total time reasonable
    start = time.time()
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    # try a sequence of p‑values
    for p in (4, 6, 8):
        if time.time() - start >= sub_budget:
            break
        cons = {'type': 'eq', 'fun': sum_constraint, 'args': (n,)}
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            p_norm_objective,
            best_h,
            args=(n, p),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        cand = repair_vector(res.x, n)
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
    # optional refinement with stochastic hill climbing
    remaining = time_budget - (time.time() - start)
    if remaining > 5.0:
        best_h, best_score = hill_climb_continuous(best_h, rng, remaining)
    return best_h, best_score

def random_binary_vector(n, rng):
    """Return a binary vector with exactly n/2 ones."""
    half = n // 2
    idx = rng.permutation(n)[:half]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

    """Hill‑climbing on binary vectors (swap a 1 and a 0)."""
    start = time.time()
