import time, random, math, numpy as np

def project_to_bounds_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    # v: np.ndarray
    n = v.shape[0]
    min_sum = n * lower
    max_sum = n * upper
    if target_sum <= min_sum:
        return np.full_like(v, lower)
    if target_sum >= max_sum:
        return np.full_like(v, upper)
    lo = lower - np.max(v)
    hi = upper - np.min(v)
    # Ensure lo <= hi; if not, adjust:
    if lo > hi:
        lo, hi = hi, lo
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        clipped = np.clip(v + mid, lower, upper)
        s = clipped.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v + (lo + hi) / 2.0, lower, upper)

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_neighbor(h):
    n = h.shape[0]
    # copy
    new = h.copy()
    i, j = random.sample(range(n), 2)
    # random direction
    if random.random() < 0.5:
        max_transfer = min(new[i], 1.0 - new[j])
        if max_transfer > 0:
            delta = random.random() * max_transfer
            new[i] -= delta
            new[j] += delta
        # else do nothing
    else:
        max_transfer = min(new[j], 1.0 - new[i])
        if max_transfer > 0:
            delta = random.random() * max_transfer
            new[j] -= delta
            new[i] += delta
    return new

import time

def project_to_bounds_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """Project vector v onto the box [lower,upper] while enforcing sum(v)=target_sum."""
    n = v.shape[0]
    min_sum = n * lower
    max_sum = n * upper
    if target_sum <= min_sum:
        return np.full_like(v, lower)
    if target_sum >= max_sum:
        return np.full_like(v, upper)
    # binary search for shift t such that sum(clipped(v+t)) = target_sum
    lo = lower - np.max(v)
    hi = upper - np.min(v)
    # Ensure bounds are ordered
    if lo > hi:
        lo, hi = hi, lo
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        clipped = np.clip(v + mid, lower, upper)
        s = clipped.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v + (lo + hi) / 2.0, lower, upper)

def evaluate_c5(h):
    """Return the C5 value for a step function defined by discretization h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_neighbor(h):
    """Generate a feasible neighbor of h by swapping or a small mass transfer."""
    n = h.shape[0]
    new = h.copy()
    if random.random() < 0.7:
        # swap two entries (keeps sum and bounds)
        i, j = random.sample(range(n), 2)
        new[i], new[j] = new[j], new[i]
    else:
        i, j = random.sample(range(n), 2)
        if random.random() < 0.5:
            max_transfer = min(new[i], 1.0 - new[j])
            if max_transfer > 0.0:
                delta = random.random() * max_transfer
                new[i] -= delta
                new[j] += delta
        else:
            max_transfer = min(new[j], 1.0 - new[i])
            if max_transfer > 0.0:
                delta = random.random() * max_transfer
                new[j] -= delta
                new[i] += delta
    return new

def random_binary_feasible(n):
    """Return a binary‑like feasible vector (half 1, half 0, one 0.5 if odd)."""
    arr = np.zeros(n, dtype=float)
    half = n // 2
    arr[:half] = 1.0
    if n % 2 == 1:
        arr[half] = 0.5
    np.random.shuffle(arr)
    return arr

def local_sa(initial_h, remaining_time, global_start):
    """Simulated‑annealing run limited by remaining_time."""
    h = initial_h.copy()
    n = len(h)
    target_sum = n / 2.0
    h = project_to_bounds_sum(h, target_sum)
    cur_score = evaluate_c5(h)
    best_h = h.copy()
    best_score = cur_score

    # Temperature schedule
    T = max(cur_score * 0.5, 0.01)
    cooling = 0.9997
    no_improve = 0
    max_no_improve = 15000

    while True:
        now = time.time()
        if now - global_start >= remaining_time:
            break
        cand = propose_neighbor(h)
        # cand is already feasible (swap or transfer respects bounds and sum)
        cand_score = evaluate_c5(cand)
        delta = cand_score - cur_score
        if delta < 0 or random.random() < math.exp(-delta / T):
            h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T = max(T * cooling, 1e-9)
        if no_improve >= max_no_improve:
            # restart from a fresh binary‑like feasible point
            h = random_binary_feasible(n)
            cur_score = evaluate_c5(h)
            no_improve = 0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    time_limit = budget_s - 0.01  # tiny safety margin

    # Determine initial discretization
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.shape[0]
    else:
        n_points = 45
        h0 = random_binary_feasible(n_points)

    target_sum = n_points / 2.0
    h0 = project_to_bounds_sum(h0, target_sum)

    # First SA run at the original resolution
    remaining = time_limit - (time.time() - start_time)
    best_h, best_score = local_sa(h0, remaining, start_time)
    best_n = n_points

    # If time permits, try a higher resolution (doubling)
    while True:
        now = time.time()
        elapsed = now - start_time
        remaining = time_limit - elapsed
        if remaining < 15.0:  # need a few seconds to be worthwhile
            break
        if best_n >= 180:  # cap resolution for speed
            break
        # upscale by repeating each value twice
        new_n = best_n * 2
        up_h = np.repeat(best_h, 2)
        # ensure correct length (should be exactly new_n)
        if up_h.shape[0] != new_n:
            up_h = np.interp(
                np.linspace(0, 1, new_n, endpoint=False),
                np.linspace(0, 1, best_h.shape[0], endpoint=False),
                best_h
            )
        up_h = project_to_bounds_sum(up_h, new_n / 2.0)
        # run SA at the higher resolution
        new_best_h, new_best_score = local_sa(up_h, remaining, start_time)
        if new_best_score < best_score:
            best_h = new_best_h
            best_score = new_best_score
            best_n = new_n
        else:
            # No improvement, stop trying larger sizes
            break

    # Return result
    return best_h.tolist(), float(best_score), best_n
