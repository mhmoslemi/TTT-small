def project_onto_box_and_sum(h, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # ensure h is numpy array
    h = np.asarray(h)
    # initial lambda bounds
    lam_low = -np.max(h) - 1.0  # ensure low enough to push all to lower bound
    lam_high = upper - np.min(h) + 1.0  # ensure high enough to push all to upper bound
    # binary search
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(h + lam, lower, upper)
        s = x.sum()
        if np.abs(s - target) < tol:
            return x
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
    # after loop, return clipped with final lam
    return np.clip(h + lam, lower, upper)

def optimize_slsqp(h0, target, n, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target}
    res = scipy.optimize.minimize(
        lambda h: evaluate_c5(h, n),
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    )
    if not res.success:
        # in case of failure, still return result
        pass
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, scipy.optimize, time, random
    # set seed
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    # Determine if initial_h_values variable exists in globals
    if 'initial_h_values' in globals():
        init_h_parent = np.array(initial_h_values, dtype=float)
        parent_n = init_h_parent.shape[0]
    else:
        init_h_parent = None
        parent_n = None

    # candidate n list
    n_list = [95, 190, 380]
    # Use n from parent if present
    if init_h_parent is not None and parent_n not in n_list:
        n_list.append(parent_n)
    # dedup and sort
    n_list = sorted(set(n_list))

    for n in n_list:
        # check time left
        if time.time() - start > budget_s * 0.95:
            break
        # initial guess
        if init_h_parent is not None and parent_n == n:
            h0 = init_h_parent.copy()
        else:
            # Interpolate if parent available with different n

import numpy as np
import time
import random
from scipy.optimize import minimize

def evaluate_c5(h, n):
    """Return the C5 value for a given step function h sampled at n points."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_and_sum(h, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """Project vector h onto the intersection of [lower,upper]^n and the hyperplane sum=h_target."""
    h = np.asarray(h, dtype=float)
    h = np.clip(h, lower, upper)
    lam_low = -np.max(h) - 1.0
    lam_high = upper - np.min(h) + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(h + lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
    lam = (lam_low + lam_high) / 2.0
    return np.clip(h + lam, lower, upper)

def optimize_slsqp(h0, target, n, max_iter=200):
    """Run a single SLSQP optimization from start h0."""
    bounds = [(0.0, 1.0)] * n

        best_h = np.full(best_n, 0.5)
        best_c5 = evaluate_c5(best_h, best_n)

    # final safety projection
    best_h = project_onto_box_and_sum(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h, best_n)

    return best_h.tolist(), float(best_c5), int(best_n)
