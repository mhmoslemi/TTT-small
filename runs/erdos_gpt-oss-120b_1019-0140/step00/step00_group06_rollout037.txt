def adjust_sum(h):
    target = len(h) / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Try to adjust by moving diff to/from indices
    # If diff >0, need to increase total sum by diff.
    # Find indices where h < 1
    can_inc = np.where(h < 1)[0]
    if diff > 0 and can_inc.size > 0:
        # increase at random indices proportionally.
        # We'll fill diff by adding to randomly selected indices up to capacity.
        # Use loop
        while diff > 1e-12:
            idx = np.random.choice(can_inc)
            cap = 1 - h[idx]
            add = min(diff, cap)
            h[idx] += add
            diff -= add
            if h[idx] >= 1 - 1e-12:
                can_inc = np.where(h < 1)[0]
                if can_inc.size == 0:
                    break
    elif diff < 0:
        # need to decrease sum
        can_dec = np.where(h > 0)[0]
        while diff < -1e-12:
            idx = np.random.choice(can_dec)
            cap = h[idx]
            sub = min(-diff, cap)
            h[idx] -= sub
            diff += sub
            if h[idx] <= 1e-12:
                can_dec = np.where(h > 0)[0]
                if can_dec.size == 0:
                    break

def max_shift_and_grad(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    l = idx - (n - 1)
    # compute gradient vectorized:
    grad = np.zeros(n, dtype=np.float64)
    # term (1 - h[i+l])
    if l >= 0:
        # i from 0 to n-l-1 have i+l in [l,n-1]
        grad[:n-l] += (1.0 - h[l:])
        # term -h[i-l] for i from l to n-1, i-l in [0,n-l-1]
        grad[l:] -= h[:n-l]
    else:
        # l negative, let k = -l >0
        k = -l
        # i+l = i-k, valid for i>=k
        grad[k:] += (1.0 - h[:n-k])
        # -h[i-l] where i-l = i + k, valid for i <= n-k-1
        grad[:n-k] -= h[k:]
    return l, grad
```

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def max_shift_and_grad(h: np.ndarray):
    """
    Return the integer shift l attaining the maximum correlation
    and the gradient of that correlation w.r.t. h.
    Gradient for shift l: grad[i] = (1 - h[i+l]) - h[i-l],
    treating out‑of‑range indices as 0.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    l = idx - (n - 1)          # shift index
    grad = np.zeros(n, dtype=np.float64)

    if l >= 0:
        # (1 - h[i+l]) term for i = 0 .. n-l-1
        grad[:n - l] += (1.0 - h[l:])
        # -h[i-l] term for i = l .. n-1
        grad[l:] -= h[:n - l]
    else:
        k = -l
        # (1 - h[i+l]) = (1 - h[i-k]) for i = k .. n-1
        grad[k:] += (1.0 - h[:n - k])
        # -h[i-l] = -h[i+k] for i = 0 .. n-k-1
        grad[:n - k] -= h[k:]

    return l, grad

def upsample_h(h_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of h_old to n_new points, then enforce constraints."""
    n_old = h_old.shape[0]
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # sample positions at midpoints of each interval
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = np.clip(h_new, 0.0, 1.0)

    # adjust to exact integral constraint
    target = n_new / 2.0
    cur = h_new.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # simple redistribution by random transfers
        rng = np.random.default_rng()
        if diff > 0:
            # need to increase total mass
            free = np.where(h_new < 1.0)[0]
            while diff > 1e-12 and free.size:
                idx = rng.choice(free)
                cap = 1.0 - h_new[idx]
                add = min(diff, cap)
                h_new[idx] += add
                diff -= add
                if h_new[idx] >= 1.0 - 1e-12:
                    free = np.where(h_new < 1.0)[0]
        else:
            # need to decrease total mass
            free = np.where(h_new > 0.0)[0]
            while diff < -1e-12 and free.size:
                idx = rng.choice(free)
                cap = h_new[idx]
                sub = min(-diff, cap)
                h_new[idx] -= sub
                diff += sub
                if h_new[idx] <= 1e-12:
                    free = np.where(h_new > 0.0)[0]
    return h_new
