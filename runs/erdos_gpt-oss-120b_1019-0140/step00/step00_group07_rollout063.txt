    if best_p is not None:
        p_refine = np.linspace(max(0.5, best_p - 0.05), min(2.0, best_p + 0.05), 81)
        lam_refine = np.linspace(max(0.0, best_lam - 0.05), min(1.0, best_lam + 0.05), 81)
        for p in p_refine:
            base = np.sin(np.pi * xs / 2) ** p
            s = (n / 2) / base.sum()
            shape = base * s
            if shape.max() > 1.0 + 1e-12:
                continue
            for lam in lam_refine:
                h = (1 - lam) * 0.5 + lam * shape
                c_val = c5_of(h)
                if c_val < best_c:
                    best_c = c_val
                    best_h = h.copy()
                    best_p, best_lam = p, lam
                if time.time() - start_time > budget_s * 0.98:
                    break
            if time.time() - start_time > budget_s * 0.98:
                break

    # ------------------------------------------------------------
    # Final sanity adjustments (ensure exact sum constraint)
    # ------------------------------------------------------------
    if best_h is not None:
        target_sum = n / 2.0
        diff = target_sum - best_h.sum()
        if abs(diff) > 1e-12:
            # adjust the first element that can absorb the difference without leaving [0,1]
            idx = np.argmax(best_h) if diff > 0 else np.argmin(best_h)
            new_val = np.clip(best_h[idx] + diff, 0.0, 1.0)
            best_h[idx] = new_val
            # re‑check residual
            diff2 = target_sum - best_h.sum()
            if abs(diff2) > 1e-12:
                # tiny leftover – spread it over a second index
                idx2 = (idx + 1) % n
                best_h[idx2] = np.clip(best_h[idx2] + diff2, 0.0, 1.0)

        # recompute final C5 after the tiny tweak
        best_c = c5_of(best_h)

    # ------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------
    return (best_h, float(best_c), int(n))
