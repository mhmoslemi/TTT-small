def project_to_constraints(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative projection onto hyperplane with bounds
    # Using Dykstra's algorithm? We'll opt for repeated scaling and clipping
    # Approach: adjust using water-filling
    # Compute deficit/excess:
    diff = target_sum - h.sum()
    # Iterate up to some loops
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        # Identify indices that are not at bounds in direction of adjustment
        if diff > 0:
            # need to add mass, only to those <1
            free = (h < 1 - 1e-12)
        else:
            # need to subtract mass, only from those >0
            free = (h > 1e-12)
        n_free = free.sum()
        if n_free == 0:
            break
        # distribute diff equally among free indices
        adjustment = diff / n_free
        # Limit adjustment to stay within bounds
        # For each free index, compute max possible addition/subtraction
        delta = np.zeros_like(h)
        if diff > 0:
            # max add per index = 1 - h[i]
            max_add = 1 - h[free]
            # use min(adjustment, max_add)
            add = np.minimum(adjustment, max_add)
            delta[free] = add
        else:
            # diff < 0: need to subtract, max subtract = h[i]
            max_sub = h[free]
            sub = np.maximum(adjustment, -max_sub)  # note adjustment negative
            delta[free] = sub
        h += delta
        diff = target_sum - h.sum()
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project h onto box [0,1] and linear equality sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # iterative water-filling style adjustment
    for _ in range(10):
        total = h.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            movable = h < 1.0
        else:
            movable = h > 0.0
        n_mov = np.count_nonzero(movable)
        if n_mov == 0:
            break
        adj = diff / n_mov
        if diff > 0:
            cap = 1.0 - h[movable]
            adj_vec = np.minimum(adj, cap)
        else:
            cap = -h[movable]
            adj_vec = np.maximum(adj, cap)
        h[movable] += adj_vec
        h = np.clip(h, 0.0, 1.0)
    # final tiny residual fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-9:

                h[i] -= sub
                residual += sub

    return h

            it_since_best = 0

    # Final verification of constraints (safety)
    best_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
