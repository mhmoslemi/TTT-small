import time
import math
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for a given h."""
    # correlation of h with (1-h), zero-padded (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_bounds(v, target_sum):
    """
    Euclidean projection of vector v onto the hypercube [0,1]^n
    with an additional linear equality sum(v)=target_sum.
    Uses a binary search for the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0   # guarantee all v+lo <= 0  -> sum = 0
    hi = 1.0 - np.min(v) + 1.0  # guarantee all v+hi >= 1 -> sum = len(v)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    return w

def random_feasible_vector(n, target_sum, rng):
    """Draw a random vector in [0,1]^n and project onto the sum constraint."""
    v = rng.random(n)
    return project_onto_bounds(v, target_sum)

def simulated_annealing(start_h, dx, target_sum, time_limit, rng):
    """
    Simple simulated-annealing optimizer.
    Keeps the sum exactly target_sum by moving mass between two coordinates.
    """
    n = start_h.shape[0]
    h = start_h.copy()
    cur_score = compute_c5(h, dx)
    best_h = h.copy()
    best_score = cur_score
    t0 = time.time()
    T_start = 0.02
    T_end = 1e-5

    while True:
        if time.time() - t0 > time_limit:
            break

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # random direction of mass transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        available = h[src]
        capacity = 1.0 - h[dst]
        max_delta = min(available, capacity)
        if max_delta <= 0.0:
            continue

        delta = rng.random() * max_delta
        h[src] -= delta
        h[dst] += delta

        new_score = compute_c5(h, dx)

        # temperature schedule (linear)
        elapsed_frac = (time.time() - t0) / time_limit
        T = T_start * (1.0 - elapsed_frac) + T_end * elapsed_frac

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert move
            h[src] += delta
            h[dst] -= delta

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimizes a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
        have_init = True
    except Exception:
        have_init = False
        n = 70  # fallback size
    target_sum = n / 2.0
    dx = 2.0 / n

    # Prepare an initial feasible solution
    if have_init:
        init_h = project_onto_bounds(init_h, target_sum)
        best_h = init_h.copy()
        best_score = compute_c5(best_h, dx)
    else:
        best_h = random_feasible_vector(n, target_sum, rng)
        best_score = compute_c5(best_h, dx)

    used_initial = False
    max_run_time = 8.0  # seconds per SA segment

    # Main optimization loop respecting the global time budget
    while True:
