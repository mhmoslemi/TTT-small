def project_h(h, n, tol=1e-9, max_iter=10):
    target = n/2
    h = np.clip(h, 0, 1)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target) <= tol:
            break
        # adjust proportionally where not at bounds
        # Determine which entries not at bounds (0 < h < 1)
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            # all at bounds, can't adjust further; break
            break
        # Compute additive delta to apply equally to non-bound entries
        delta = (target - s) / mask.sum()
        h[mask] += delta
        # after addition, clip again
        h = np.clip(h, 0, 1)
    return h
```

def project_h(h, n, max_iter=100):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine indices that are not at bound.
        mask = (h > 0) & (h < 1)
        free = mask.sum()
        if free == 0:
            # All at bounds, cannot adjust further
            break
        delta = diff / free
        # Update free entries
        h[mask] += delta
        # Clip again to bounds
        h = np.clip(h, 0.0, 1.0)
    # ensure final sum close to target
    # final adjustment: scale proportionally if sum not exact (rare)
    s = h.sum()
    if abs(s - target) > 1e-8:
        # scale uniformly if within bounds
        scale = target / s
        h = np.clip(h * scale, 0.0, 1.0)
        # final correction to sum, adjust a random free element
        s = h.sum()
        diff = target - s
        if abs(diff) > 1e-8:
            # adjust a random index not at bound
            idxs = np.where((h > 0) & (h < 1))[0]
            if idxs.size > 0:
                i = idxs[0]
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def local_optimize(h0, time_limit, start_time, rng):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    # parameters
    max_iters = 2000000  # just large; loop will break by time
    delta_initial = 0.1
    it = 0
    while time.time() - start_time < time_limit and it < max_iters:
        # exponential decay for delta
        delta = delta_initial * (0.5 ** (it / 2000))
        if delta < 1e-6:
            delta = 1e-6
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose direction: transfer delta from i to j or j to i
        if rng.random() < 0.5:
            src, tgt = i, j
        else:
            src, tgt = j, i
        max_transfer = min(h[src], 1.0 - h[tgt])
        if max_transfer <= 0:
            continue
        # pick transfer amount
        d = rng.random() * min(delta, max_transfer)
        if d <= 0:
            continue
        # apply
        h[src] -= d
        h[tgt] += d
        c5_new = evaluate_c5(h)
        if c5_new <= best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # revert
            h[src] += d
            h[tgt] -= d
        it += 1
    return best_h, best_c5

def project_h(h, n, max_iter=100):
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        free = mask.sum()
        if free == 0:
            break
        delta = diff / free
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tweak if needed
    s = h.sum()
    if abs(s - target) > 1e-8:
        scale = target / s
        h = np.clip(h * scale, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) > 1e-10:
            idxs = np.where((h > 0) & (h < 1))[0]
            if idxs.size:
                i = idxs[0]
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

import numpy as np, time

def project_h(h, n, max_iter=100):
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        free = mask.sum()
        if free == 0:
            break
        delta = diff / free
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-8:
        scale = target / s
        h = np.clip(h * scale, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) > 1e-10:
            idxs = np.where((h > 0) & (h < 1))[0]
            if idxs.size:
                i = idxs[0]
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def evaluate_c5(h):
