def improve_one_step(h):
    n = len(h)
    # compute overlap via correlation (full)
    overlap = np.correlate(h, h, mode='full')
    # convert to actual values (dx factor optional; but comparison)
    # find lag index with minimal overlap (center at n-1 corresponds to lag0)
    # convert index to lag in terms of shift steps
    # shift = idx - (n-1)
    # we need to ensure overlapping portion >0, i.e., shift >= -(n-1) etc.
    idx_min = overlap.argmin()
    lag = idx_min - (n - 1)
    # compute product contributions for i where i+lag within range
    # if lag >=0: i from 0 to n - 1 - lag else i from -lag to n-1
    if lag >= 0:
        i_vals = np.arange(0, n - lag)
        j_vals = i_vals + lag
    else:
        i_vals = np.arange(-lag, n)
        j_vals = i_vals + lag
    # contributions:
    contrib = h[i_vals] * h[j_vals]
    # find index i with smallest contribution
    min_idx = np.argmin(contrib)
    i = i_vals[min_idx]

import numpy as np

def _project_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    # binary search for offset t such that sum(clip(x+t,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding missed the target
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
        else:
            y += diff / y.size
        y = np.clip(y, 0.0, 1.0)
    return y

def _c5_value(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _worst_lag(h: np.ndarray) -> int:

    min_idx = int(ov.argmin())

    new_h[d] -= 2.0 * delta
    return _project_to_sum(new_h, target)

def _perturb(h: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    n = h.size
    m = max(1, int(0.2 * n))
    idx = rng.choice(n, size=m, replace=False)
    sigma = 0.1
    new_h = h.copy()
    new_h[idx] += rng.normal(scale=sigma, size=m)
    return _project_to_sum(new_h, target)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----- initialise ---------------------------------
    try:
        base = np.array(initial_h_values, dtype=float)  # type: ignore
        n0 = base.shape[0]
        # aim for a modestly finer grid (up to ~2×)
        target_n = min(200, max(70, n0 * 2))
        if target_n != n0:
            rep = target_n // n0
            rem = target_n - rep * n0
            h = np.repeat(base, rep)
            if rem:
                h = np.concatenate([h, base[:rem]])
        else:
            h = base.copy()
    except Exception:
        target_n = 140
        h = np.full(target_n, 0.5, dtype=float)

    target = target_n / 2.0
    h = _project_to_sum(h, target)

    best_h = h.copy()
    best_score = _c5_value(best_h)
    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    stagnant = 0

    while time.time() - start < budget_s * 0.95:
        # choose move
        if rng.random() < 0.7:
            cand = _improve_worst(cur_h, rng, target)
        else:
            cand = _perturb(cur_h, rng, target)

        cand_score = _c5_value(cand)

        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            stagnant = 0
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            # simulated-annealing style acceptance
            elapsed = time.time() - start
            remaining = max(0.0, budget_s - elapsed)
            T = remaining / budget_s
            prob = np.exp((cur_score - cand_score) / (T + 1e-12))
            if rng.random() < prob:
                cur_h, cur_score = cand, cand_score
            stagnant += 1

        if stagnant > 5000:
            # shake out of a plateau
            cur_h = _perturb(best_h, rng, target)
            cur_score = _c5_value(cur_h)
            stagnant = 0

    best_h = _project_to_sum(best_h, target)
    final_score = _c5_value(best_h)
    return best_h, float(final_score), target_n
