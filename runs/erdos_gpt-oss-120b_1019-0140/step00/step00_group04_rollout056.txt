import time
import math
import numpy as np

# Helper: compute C5 upper bound (max overlap integral) for a binary step function
def _c5_from_h(h, dx):
    # correlation with (1-h) gives number of mismatched ones for each shift
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    cmax = float(corr[idx]) * dx
    shift = idx - (len(h) - 1)
    return cmax, shift, corr

# Convert a possibly non‑binary initial vector to a binary one with exactly n/2 ones
def _binary_from_initial(arr):
    n = len(arr)
    h = np.zeros(n, dtype=np.int8)
    # pick the n//2 largest entries to be 1
    if n // 2 > 0:
        idx = np.argpartition(arr, -n // 2)[-n // 2 :]
        h[idx] = 1
    return h

# Create a random binary vector with exactly n/2 ones
def _random_binary(n, rng):
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    return h

# Return mismatched ones for a given shift and candidate zeros that would align with a one
def _mismatch_and_candidates(h, shift):
    n = len(h)
    mismatched = []
    candidates = []
    for i in range(n):
        j = i + shift
        if h[i] == 1:
            if j < 0 or j >= n or h[j] == 0:
                mismatched.append(i)
        else:  # h[i] == 0
            if 0 <= j < n and h[j] == 1:
                candidates.append(i)
    return mismatched, candidates

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.

    Returns:
        h_values (np.ndarray) : array of length n_points containing values in [0,1]
        c5_bound (float)       : the achieved upper bound C5
        n_points (int)         : number of discretisation points used
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size
    # ----------------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        n = init_arr.shape[0]
        have_initial = True
    except Exception:
        # fallback size (even and <=200 for speed)
        n = 84
        have_initial = False

    # Ensure even n (required for exact sum constraint)
    if n % 2 != 0:
        n += 1
    if n > 200:               # keep the vector modest
        n = 200
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Build the first candidate
    # ----------------------------------------------------------------------
    if have_initial:
        if init_arr.shape[0] != n:
            # Resample/interpolate to required length
            x_old = np.linspace(0, 2, init_arr.shape[0], endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            from scipy.interpolate import interp1d
            f = interp1d(x_old, init_arr, kind="linear", fill_value="extrapolate")
            init_arr = f(x_new)
        h_start = _binary_from_initial(init_arr)
    else:

    best_h = h_start.copy()
    best_c, _, _ = _c5_from_h(best_h, dx)
