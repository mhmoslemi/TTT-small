def project_onto_bounds_and_sum(x, lower=0, upper=1, target_sum=None):
    if target_sum is None:
        target_sum = np.sum(x)  # not needed.

    # Clip to bounds
    x = np.clip(x, lower, upper)

    # Compute total mass
    total = np.sum(x)
    if np.isclose(total, target_sum):
        return x

    # use water-filling: need to increase or decrease mass uniformly on non-bound elements.
    # We'll iterate:
    for _ in range(10):
        diff = target_sum - total
        # compute free indices where x not at bound (or can move in diff direction)
        if diff > 0:
            free = x < upper - 1e-12
        else:
            free = x > lower + 1e-12
        if not np.any(free):
            break
        # distribute diff equally among free elements
        delta = diff / np.sum(free)
        x[free] += delta
        # clip again
        x = np.clip(x, lower, upper)
        total = np.sum(x)
        if np.abs(total - target_sum) < 1e-9:
            break
    return x

    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # iterative scaling:
    for _ in range(10):
        total = np.sum(x)
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        # Determine free indices that can be moved in direction of diff
        if diff > 0:
            free = x < 1.0 - 1e-12  # can increase
        else:
            free = x > 1e-12       # can decrease
        if not np.any(free):
            # no free capacity, break
            break
        delta = diff / np.sum(free)
        # apply delta
        x[free] += delta
        # clip newly changed
        x = np.clip(x, 0.0, 1.0)
    # final clip for safety
    x = np.clip(x, 0.0, 1.0)
    return x

    for _ in range(15):
        total = np.sum(x)
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0 - 1e-12
        else:
            free = x > 1e-12
        if not np.any(free):
            break
        delta = diff / np.sum(free)
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final safety clip
    x = np.clip(x, 0.0, 1.0)
    return x

def _exchange_step(h):
    """
    Perform a deterministic exchange move aiming to reduce the worst shift.
    Returns (new_h, changed_flag).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)  # integer shift index

    # cost_i = (1 - h[i+shift]) for valid i+shift, otherwise 1
    cost = np.empty(n, dtype=float)
    if shift >= 0:
        cost[: n - shift] = 1.0 - h[shift:]
        cost[n - shift :] = 1.0
    else:
        s = -shift
        cost[s:] = 1.0 - h[: n - s]
        cost[:s] = 1.0

    # candidates that can decrease / increase
    dec_mask = h > 1e-12
    inc_mask = h < 1.0 - 1e-12
    if not np.any(dec_mask) or not np.any(inc_mask):
        return h, False

    dec_idx = np.where(dec_mask)[0]
    inc_idx = np.where(inc_mask)[0]

    # pick highest‑cost decrease and lowest‑cost increase
    i_dec = dec_idx[np.argmax(cost[dec_idx])]
    i_inc = inc_idx[np.argmin(cost[inc_idx])]

    if cost[i_dec] <= cost[i_inc] + 1e-12:
        return h, False

    delta = min(h[i_dec], 1.0 - h[i_inc])
    if delta < 1e-12:
        return h, False

    h_new = h.copy()
    h_new[i_dec] -= delta
    h_new[i_inc] += delta
    return h_new, True

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise the vector h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 128                     # fallback size
        h0 = rng.rand(n)

    dx = 2.0 / n
    target_sum = n / 2.0

    # project the start to the feasible polytope
    cur_h = _project_to_feasible(h0, target_sum)
    cur_c = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_c = cur_c

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    stagnation = 0
    max_no_improve = 5000

    while time.time() - start_time < budget_s * 0.97:
        # Choose move type
        if rng.rand() < 0.65:
            # Deterministic exchange move
            cand_h, moved = _exchange_step(cur_h)
            if not moved:
                # fallback: small Gaussian perturbation
                cand_h = cur_h + rng.randn(n) * 0.02
        else:
            # Random perturbation (different amplitudes)
            scale = 0.04 if rng.rand() < 0.7 else 0.12
            cand_h = cur_h + rng.randn(n) * scale

        # Enforce feasibility
        cand_h = _project_to_feasible(cand_h, target_sum)

        cand_c = _compute_c5(cand_h, dx)

        # Accept if improvement
        if cand_c < cur_c - 1e-9:
            cur_h, cur_c = cand_h, cand_c
            stagnation = 0
            if cand_c < best_c - 1e-12:
                best_h, best_c = cand_h.copy(), cand_c
        else:
            stagnation += 1

        # Restart logic when stuck
        if stagnation > max_no_improve:
            # go back to the global best
            cur_h, cur_c = best_h.copy(), best_c
            stagnation = 0
            # occasional random restart
            if rng.rand() < 0.25:
                cur_h = rng.rand(n)
                cur_h = _project_to_feasible(cur_h, target_sum)
                cur_c = _compute_c5(cur_h, dx)

    # ------------------------------------------------------------------
    # Optional refinement on a finer grid if time permits
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30 and n < 256:          # give a modest extra budget
        new_n = min(256, n * 2)
        # linearly interpolate to the finer grid
        xs_old = np.linspace(0, 2, n, endpoint=False)
        xs_new = np.linspace(0, 2, new_n, endpoint=False)
        fine_h = np.interp(xs_new, xs_old, best_h)
        dx_fine = 2.0 / new_n
        target_fine = new_n / 2.0
        fine_h = _project_to_feasible(fine_h, target_fine)

        cur_h = fine_h
        cur_c = _compute_c5(cur_h, dx_fine)
        best_h, best_c, n = cur_h.copy(), cur_c, new_n

        stagnation = 0
        while time.time() - start_time < budget_s * 0.995:
            if rng.rand() < 0.6:
                cand_h, moved = _exchange_step(cur_h)
                if not moved:
                    cand_h = cur_h + rng.randn(new_n) * 0.015
            else:
                scale = 0.03 if rng.rand() < 0.8 else 0.09
                cand_h = cur_h + rng.randn(new_n) * scale

            cand_h = _project_to_feasible(cand_h, target_fine)
            cand_c = _compute_c5(cand_h, dx_fine)

            if cand_c < cur_c - 1e-9:
                cur_h, cur_c = cand_h, cand_c
                stagnation = 0
                if cand_c < best_c - 1e-12:
                    best_h, best_c = cand_h.copy(), cand_c
            else:
                stagnation += 1

            if stagnation > max_no_improve:
                cur_h, cur_c = best_h.copy(), best_c
                stagnation = 0
                if rng.rand() < 0.2:
                    cur_h = rng.rand(new_n)
                    cur_h = _project_to_feasible(cur_h, target_fine)
                    cur_c = _compute_c5(cur_h, dx_fine)

    # Final safety clip and exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_feasible(best_h, best_h.shape[0] / 2.0)
    best_c = _compute_c5(best_h, 2.0 / best_h.shape[0])

    return best_h, float(best_c), best_h.shape[0]
