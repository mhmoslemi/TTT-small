def simulated_annealing(h, best_val, budget_s, start_time):
    n = h.shape[0]
    target_sum = n/2.0
    current = h.copy()
    curr_val = best_val
    best_h = current.copy()
    best_val = curr_val
    rng = np.random.default_rng()
    iter = 0
    while time.time() - start_time < budget_s * 0.95:
        iter += 1
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # compute max transfer
        cap_i = current[i]  # can take at most this
        cap_j = 1.0 - current[j]  # can add at most this
        max_transfer = min(cap_i, cap_j)
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        # small delta threshold maybe skip too tiny
        if delta < 1e-8:
            continue
        new = current.copy()
        new[i] -= delta
        new[j] += delta
        # Clip just in case (should not exceed)
        # new already respects bounds due to delta.
        new_val = compute_c5(new)

                best_h = new.copy()

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a given discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that its sum equals target while respecting [0,1] bounds."""
    diff = target - h.sum()
    # tolerance for floating point
    if abs(diff) < 1e-12:
        return h
    # Simple random redistribution
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            # need to remove mass
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

    if scale != 1.0:
        h *= scale
    np.clip(h, 0.0, 1.0, out=h)
    # Fix any remaining sum error
    h = adjust_sum_to_target(h, target, rng)
    return h


def simulated_annealing(initial_h: np.ndarray, init_val: float,
                        budget_s: float, start_time: float,
                        rng: np.random.Generator) -> tuple:
    """Refine a solution using a simple simulated‑annealing scheme."""
    n = initial_h.shape[0]
    target = n / 2.0
    current = initial_h.copy()
    curr_val = init_val
    best_h = current.copy()
    best_val = init_val

    # annealing parameters
    T0 = 0.08
    decay = 0.9995
    it = 0
    while time.time() - start_time < budget_s * 0.90:
        it += 1
        # temperature schedule
        T = max(1e-7, T0 * (decay ** it))

        # propose a transfer move that keeps the sum invariant
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cap_i = current[i]          # how much we can take from i
        cap_j = 1.0 - current[j]    # how much we can add to j
        max_transfer = min(cap_i, cap_j)
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random()
        if delta <= 1e-12:
            continue

        new_h = current.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = compute_c5(new_h)

        if new_val < curr_val:
            accept = True
        else:
            prob = math.exp(-(new_val - curr_val) / T)
            accept = rng.random() < prob

        if accept:
            current = new_h
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                # early exit if we beat the target bound
                if best_val <= 0.3808:
                    break
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine candidate sample sizes
    candidates = []
    init_vec = None
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        candidates.append(init_vec.shape[0])
    except Exception:
        init_vec = None

    # Add a few modest even sizes (smaller is faster)
    candidates.extend([48, 60, 80, 100, 120])
    # Ensure uniqueness while preserving order
    seen = set()
    candidates = [c for c in candidates if not (c in seen or seen.add(c))]

    best_h = None
    best_val = float("inf")

    # Evaluate each candidate shape (cosine) and the optional initial vector
    for n in candidates:
        if init_vec is not None and n == init_vec.shape[0]:
            # Use the provided initial construction after ensuring feasibility
            h = init_vec.copy()
            np.clip(h, 0.0, 1.0, out=h)
            target = n / 2.0
            if abs(h.sum() - target) > 1e-12:
                # scale then clip then adjust sum
                scale = target / h.sum() if h.sum() != 0 else 0.0
                h *= scale
                np.clip(h, 0.0, 1.0, out=h)
                h = adjust_sum_to_target(h, target, rng)
        else:
            h = cosine_shape(n, rng)

        c5 = compute_c5(h)
        if c5 < best_val:
            best_val = c5
            best_h = h
        # If we already beat the required bound, we can stop early
        if best_val <= 0.3808:
            break

    # If the best we have already satisfies the target, return it
    if best_val <= 0.3808:
        return best_h, best_val, best_h.shape[0]

    # Otherwise refine the best candidate with simulated annealing
    best_h, best_val = simulated_annealing(best_h, best_val,
                                           budget_s, start, rng)

    return best_h, best_val, best_h.shape[0]
