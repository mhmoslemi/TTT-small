def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    # using np.correlate
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h):
    # ensure within bounds and sum=target
    n = len(h)
    target = n / 2.0
    # clip
    h = np.clip(h, 0.0, 1.0)
    # compute current sum
    cur = h.sum()
    # if cur close to target, done
    if abs(cur - target) < 1e-12:
        return h
    # we need to adjust by diff = target - cur
    # We'll do simple algorithm: while diff != 0:
    #   if diff > 0: increase some element <1 by min(diff, 1 - val)
    #   else: decrease element >0 by min(-diff, val)
    # We'll modify in vectorized way
    diff = target - cur
    if diff > 0:
        # Need to add total diff
        # Find indices with h < 1
        idx = np.where(h < 1 - 1e-12)[0]
        if len(idx) == 0:
            return h
        # distribute equally (or one element)
        for i in idx:
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:

import time

import numpy as np

def _evaluate_h(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target by simple redistribution."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    diff = target_sum - cur
    if diff > 0:
        # need to add mass
        idx = np.where(h < 1.0 - 1e-12)[0]
        for i in idx:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to remove mass
        diff = -diff
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    return h


def _local_search(init_h: np.ndarray, budget: float, start_time: float):
    """Simple simulated‑annealing hill‑climber within a time slice."""
    h = init_h.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    cur_score = _evaluate_h(h)
    best_h = h.copy()
    best_score = cur_score
    T0 = 0.01  # initial temperature

    while True:
        now = time.time()
        if now - start_time >= budget - 0.01:
            break

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            continue

        # occasional full transfer to preserve binary nature
        if np.random.rand() < 0.1:
            eps = max_eps
        else:
            eps = max_eps * np.random.rand()

        # propose move
        h[i] -= eps
        h[j] += eps

                best_score = new_score
                best_h = h.copy()
        else:
            # revert

            h[j] -= eps

    while True:
        now = time.time()
        remaining = budget_s - (now - start_total)
        if remaining < 5.0:
            break

        # Choose start vector for this restart
        if restart == 0 and have_initial:
            start_vec = best_h.copy()
        else:
            # fresh random binary (or fractional) vector
            start_vec = np.zeros(n, dtype=np.float64)
            if target_sum.is_integer():
                k = int(target_sum)
                idx = np.random.choice(n, k, replace=False)
                start_vec[idx] = 1.0
            else:
                k = int(math.floor(target_sum))
                idx = np.random.choice(n, k, replace=False)
                start_vec[idx] = 1.0
                frac = target_sum - k
                remaining_idx = [i for i in range(n) if i not in idx]
                if remaining_idx:
                    start_vec[np.random.choice(remaining_idx)] = frac

        # Allocate a slice of the remaining budget
        slice_budget = min(remaining * 0.6, max(5.0, remaining - 5.0))
        local_best_h, local_best_score = _local_search(start_vec, slice_budget, time.time())

        if local_best_score < best_score - 1e-12:
            best_score = local_best_score
            best_h = local_best_h.copy()

        if best_score < 0.3808:
            break

        restart += 1

    # Final polishing with whatever time is left
    remaining = budget_s - (time.time() - start_total)
    if remaining > 2.0:
        refined_h, refined_score = _local_search(best_h, remaining, time.time())
        if refined_score < best_score:
            best_h = refined_h
            best_score = refined_score

    # Ensure feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(best_h < 1.0 - 1e-12)[0]
            if cand.size:
                i = np.random.choice(cand)
                best_h[i] += min(diff, 1.0 - best_h[i])
        else:
            cand = np.where(best_h > 1e-12)[0]
            if cand.size:
                i = np.random.choice(cand)
                best_h[i] -= min(-diff, best_h[i])

    final_score = _evaluate_h(best_h)

    return best_h.tolist(), float(final_score), n
