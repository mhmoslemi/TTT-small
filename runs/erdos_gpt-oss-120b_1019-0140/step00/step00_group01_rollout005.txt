import numpy as np
import time
from scipy.optimize import differential_evolution

def _project_bounded_simplex(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=20):
    """
    Project vector v onto the set {w : lower <= w_i <= upper, sum w_i = target}.
    Simple iterative water-filling implementation.
    """
    w = np.clip(v, lower, upper)
    for _ in range(max_iter):
        s = w.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        # indices that can still move (not at a bound)
        free = (w > lower + tol) & (w < upper - tol)
        if not np.any(free):
            break
        adjust = diff / free.sum()
        w[free] += adjust
        w = np.clip(w, lower, upper)
    # final tiny correction if needed
    diff = target - w.sum()
    if abs(diff) > tol:
        # distribute remaining diff greedily
        for i in range(len(w)):
            if diff > 0 and w[i] < upper:
                inc = min(diff, upper - w[i])
                w[i] += inc
                diff -= inc
            elif diff < 0 and w[i] > lower:
                dec = min(-diff, w[i] - lower)
                w[i] -= dec
                diff += dec
            if abs(diff) <= tol:
                break
    return w

def _c5_of_h(h, dx):
    """compute C5 = max correlation * dx for a given h"""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
