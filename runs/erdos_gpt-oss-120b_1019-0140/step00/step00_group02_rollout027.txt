def propose_move(h, step):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # decide direction sign
    direction = np.random.choice([-1, 1])
    if direction == 1:
        # increase i, decrease j
        max_delta = min(step, 1.0 - h[i], h[j])
        if max_delta <= 0: return None
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0: return None
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    return new_h

def project_onto_bounds_and_sum(v, target_sum):
    # v is real array; we want find h = clip(v - lam, 0, 1) s.t. sum(h) = target_sum
    # lam is scalar.
    lo = np.min(v) - 1.0  # ensures all clipped to 1 when lam=lo => v - lo >= 1 => clip to 1 => sum = n
    hi = np.max(v)        # ensures all clipped to 0 when lam=hi => v - hi <= 0 => sum = 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final projection
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # due to numeric issues, adjust minor diff to sum target via small redistribution if needed
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        # adjust by adding diff to entries not at bound, spread equally
        # find indices where 0 < h < 1
        mask = (h > 1e-12) & (h < 1 - 1e-12)
        if np.any(mask):
            # distribute diff equally among these indices
            h[mask] += diff / mask.sum()
            # ensure stays in [0,1] after adjustment (should be small)
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: normalize to sum using scaling of all entries (may violate bounds)
            h = h * (target_sum / h.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

    # Use bisection on lambda
    lo = np.min(v) - 1.0
    hi = np.max(v)

        h = np.clip(v - lam, 0.0, 1.0)

        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust tiny residual sum diff
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff to non-bound entries if any
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():

            # clip again to ensure bounds
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback scaling
            h = h * (target / h.sum())
            h = np.clip(h, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the step function defined by samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_onto_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {h | 0 <= h_i <= 1, sum(h)=target}.
    Uses bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0   # at lo all entries are clipped to 1 -> sum=n
    hi = np.max(v)         # at hi all entries are clipped to 0 -> sum=0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback (should rarely happen)
            h = h * (target / h.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb(start_h: np.ndarray, target_sum: float,
               time_limit: float, start_step: float = 0.2,
               max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic pairwise hill‑climbing that respects the sum constraint.
    Returns the best vector seen and its C5 value.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    step = start_step
    no_improve = 0
    it = 0

    while it < max_iters and time.time() < time_limit:
        # choose two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # random direction: increase i / decrease j or opposite
        if np.random.rand() < 0.5:
            max_delta = min(step, 1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0.0:
                it += 1
                continue
            delta = np.random.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                it += 1
                continue
            delta = np.random.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = compute_c5(new_h)

        if new_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
                no_improve = 0
        else:
            # occasional uphill acceptance to escape plateaus
            if np.random.rand() < 0.001:
                cur_h, cur_c5 = new_h, new_c5

        it += 1
        no_improve += 1

        # step decay
        if it % 2000 == 0:
            step = max(step * 0.9, 1e-5)

        # restart after stagnation
        if no_improve >= 5000:
            perturb = np.random.normal(scale=step, size=n)
            new_h = best_h + perturb
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_bounds_and_sum(new_h, target_sum)
            cur_h = new_h
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    return best_h, best_c5
