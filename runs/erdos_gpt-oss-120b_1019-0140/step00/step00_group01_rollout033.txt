def project_box_simplex(v, S):
    # v: np.ndarray
    # Solve min ||x - v||^2 subject to 0 <= x <= 1, sum(x) = S
    # Use bisection for lambda.
    # find lambda such that sum(clip(v - lam, 0, 1)) = S
    # lam range could be [-c, c] large enough.
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def compute_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    k_idx = np.argmax(corr)
    lag = k_idx - (n - 1)
    grad = np.zeros_like(h)
    # first part
    if lag >= 0:
        # i+lag within bounds for i from 0 to n-lag-1
        grad[:n-lag] += (1 - h[lag:])    # h[i+lag] = h[lag:][i]
        grad[lag:] -= h[:n-lag]         # h[i-lag] = h[:n-lag] maps to i in lag..n-1
    else:
        # lag negative: shift opposite
        d = -lag
        # i+lag = i-d, valid for i>=d
        grad[d:] += (1 - h[:n-d])
        grad[:n-d] -= h[d:]
    return grad * (2.0 / n)  # dx factor

import numpy as np
import time

def project_box_simplex(v, S):
    """Project v onto the set {x: 0<=x_i<=1, sum x_i = S}."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        if proj.sum() > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def generate_random_feasible(n, rng):
    """Uniform random vector on [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_simplex(v, n / 2.0)

def compute_c5(h):
    """Return the overlap constant for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h):
    """Gradient of max correlation *dx w.r.t. h for the worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argmax(corr)
    lag = k_idx - (n - 1)
    grad = np.zeros_like(h)
    if lag >= 0:
        end = n - lag
        grad[:end] += (1.0 - h[lag:])
        grad[lag:] -= h[:end]
    else:
        d = -lag
        end = n - d
        grad[d:] += (1.0 - h[:end])
        grad[:end] -= h[d:]
    return grad * (2.0 / n)

def gradient_step(h, step):
    """Take a projected gradient step."""
    new = h - step * compute_gradient(h)
    return project_box_simplex(new, len(h) / 2.0)
