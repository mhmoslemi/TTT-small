def project_sum_bounds(v, lo, hi, total):
    # v: np.ndarray of floats.
    # returns projection onto [lo,hi] intersection hyperplane sum=total.
    # Clip initial feasible bounds? We'll find lambda.
    # Use binary search for lambda.
    left = -1e6
    right = 1e6
    for _ in range(60):
        mid = (left + right) / 2
        x = np.clip(v - mid, lo, hi)
        s = x.sum()
        if s > total:
            # need larger lambda (would reduce sum)
            left = mid
        else:
            right = mid
    # final lambda
    x = np.clip(v - (left + right) / 2, lo, hi)
    # To ensure sum exactly, we can adjust small remaining diff by distributing to entries not at bound.
    diff = total - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff over non-clipped entries (i.e., those strictly between lo and hi)
        mask = (x > lo + 1e-12) & (x < hi - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            # Add equally
            x[mask] += diff / cnt
            # Ensure numeric within bounds; clip again
            x = np.clip(x, lo, hi)
    return x

import numpy as np
import random
import time
import math

def project_sum_bounds(v, lo, hi, total, iters=60):
    # Project v onto the hyperplane sum(x)=total with box constraints lo <= x <= hi
    left = -1e6
    right = 1e6
    for _ in range(iters):
        mid = (left + right) / 2.0
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if s > total:
            left = mid
        else:
            right = mid
    x = np.clip(v - (left + right) / 2.0, lo, hi)
    # Minor correction to hit total exactly (distribute residual on free variables)
    diff = total - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > lo + 1e-12) & (x < hi - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            np.clip(x, lo, hi, out=x)
    return x

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def neighbor_random(h, sigma, target_sum):
    v = h + np.random.normal(scale=sigma, size=h.shape)
    return project_sum_bounds(v, 0.0, 1.0, target_sum)

def neighbor_targeted(h, step, target_sum):
    n = h.shape[0]
    # Identify shift with largest overlap
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    # Overlap interval for i where both i and i+k are valid
    i_start = max(0, -k)
    i_end = min(n - 1, n - 1 - k)
    if i_end <= i_start:
        # fallback to random move
        return neighbor_random(h, sigma=step, target_sum=target_sum)
    i = random.randint(i_start, i_end)
    j = random.randint(0, n - 1)

    v[j] = np.clip(v[j] + delta, 0.0, 1.0)
