```
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = h.shape[0] / 2.0

    h = h * (target / total)

    for _ in range(10):
```

```
import cvxpy as cp

def solve_milp_binary(n):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2]
    # total sum of h = n/2
    total_sum = n/2
    # create list of y variables
    y_vars = {}
    for lag in range(-(n-1), n):
        # list of overlapping i indices
        i_start = max(0, -lag)
        i_end = min(n, n - lag)  # exclusive
        length = i_end - i_start
        if length <= 0:
            continue
        # create vector of y for this lag
        y = cp.Variable(length, boolean=True)
        y_vars[lag] = (y, i_start) # store variable and start index
        # linking constraints
        for idx in range(length):
            i = i_start + idx
            j = i + lag
            # constraints linking y[idx] with h[i] and h[j]
            constraints += [ y[idx] <= h[i] ]
            constraints += [ y[idx] <= h[j] ]
            constraints += [ y[idx] >= h[i] + h[j] - 1 ]
        # compute M_l
        M_l = total_sum - cp.sum(y)
        constraints.append( M_l <= t )
    # t variable must be >=0
    constraints.append( t >= 0 )
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    return h.value, t.value
```

import numpy as np
import time
import math

def _eval_c5(h):
    """return C5 value for a step function h (array of length n)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _worst_lag_info(h):
    """return (lag, max_corr, indices, contributions) for the lag
    that gives the current maximal correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = float(corr[max_idx])
    lag = max_idx - (n - 1)            # lag can be negative
    if lag >= 0:
        idx = np.arange(0, n - lag)
        contrib = h[idx] * (1.0 - h[idx + lag])
    else:
        idx = np.arange(-lag, n)
        contrib = h[idx] * (1.0 - h[idx + lag])
    return lag, max_val, idx, contrib

def _propose_swap(h, lag):
    """propose a new vector by swapping one entry that contributes a lot
    to the worst lag with a random other entry."""
    n = h.shape[0]
    # indices that actually contribute for this lag
    if lag >= 0:
        idx = np.arange(0, n - lag)
        contrib = h[idx] * (1.0 - h[idx + lag])
    else:
        idx = np.arange(-lag, n)
        contrib = h[idx] * (1.0 - h[idx + lag])
    if contrib.sum() > 0:
        probs = contrib / contrib.sum()
        i = np.random.choice(idx, p=probs)
    else:
        i = np.random.randint(n)
    # pick a different position uniformly
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    h_new = h.copy()
    h_new[i], h_new[j] = h[j], h[i]
    return h_new

def _perturb_pair(h):
    """small continuous perturbation of two entries preserving sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    if lo >= hi_delta:
        # fall back to swap if no feasible delta
        h_new = h.copy()
        h_new[i], h_new[j] = h[j], h[i]
        return h_new
    delta = np.random.uniform(lo, hi_delta)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    TARGET = 0.3808  # target bound

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # may be supplied
        n = init.shape[0]
        h = init.copy()
    except NameError:
        # no parent construction – start with a modest even grid
        n = 48                     # even length makes a binary start easy
        half = n // 2
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, half, replace=False)
        h[ones] = 1.0

    # project onto the feasible region: 0<=h<=1 and sum(h)=n/2
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    for _ in range(12):
        s = h.sum()
        if s == 0:
            break
        factor = target_sum / s
        h *= factor
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target_sum) < 1e-12:
            break

    cur = h.copy()
    cur_score = _eval_c5(cur)
    best = cur.copy()
    best_score = cur_score

    # early exit if already good enough
    if best_score <= TARGET:
        return (best, best_score, n)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    T = 0.5                # initial temperature
    Tmin = 1e-6
    decay = 0.9995
    stagnant = 0           # iterations without improvement

        lag, _, _, _ = _worst_lag_info(cur)

        if np.random.rand() < 0.15:
            cand = _perturb_pair(cur)
        else:
            cand = _propose_swap(cur, lag)

        cand_score = _eval_c5(cand)
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):

    return (best, final_score, n)
