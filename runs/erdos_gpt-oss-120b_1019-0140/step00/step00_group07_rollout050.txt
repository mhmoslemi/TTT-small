        init = initial_h_values

    except NameError:

        target_sum = N // 2
        # start with random balanced binary
        current_h = np.zeros(N, dtype=float)
        ones = rng.choice(N, size=target_sum, replace=False)
        current_h[ones] = 1.0
    # ensure binary
    current_h = np.clip(current_h, 0, 1)
    # maybe make integer
    # Search
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # Precompute positions:
    # positions of ones and zeros for swapping
    # We'll use while loop for annealing
    max_iter = 10**7, but break by time.

    # temperature schedule: linear decay from t0 to tf = 0
    t0 = 0.01
    tf = 0.0001
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.96:   # stop a bit early
            break
        t = t0 * (1 - elapsed / (budget_s * 0.96)) + tf * (elapsed / (budget_s * 0.96))
        # choose two indices with different values
        i = rng.integers(N)
        j = rng.integers(N-1)
        if j >= i: j += 1
        if current_h[i] == current_h[j]:
            continue
        # swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        new_c5 = compute_c5(current_h)
        delta = new_c5 - current_c5
        accept = False
        if delta < -1e-12:  # improvement
            accept = True
        else:
            # acceptance probability
            prob = math.exp(-delta / t) if t > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = current_h.copy()

            # revert swap
            current_h[i], current_h[j] = current_h[j], current_h[i]

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def solve_milp(N, time_limit):
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N//2, t >= 0]
    for d in range(1, N):
        hi = h[:N-d]
        hj = h[d:]
        w = cp.Variable(N-d, boolean=True)
        constraints.append(w <= hi)
        constraints.append(w <= hj)
        constraints.append(w >= hi + hj - 1)
        Ld = cp.sum(w)
        Md = cp.sum(hi)
        constraints.append(Md - Ld <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, time_limit=time_limit)
    except Exception as e:
        # try without time limit
        prob.solve(solver=cp.CBC, warm_start=True)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
        h_val = np.round(h.value).astype(float)
        # fix any rounding errors making sum exactly N/2
        diff = int(N//2 - h_val.sum())

            # adjust by flipping some bits

            # choose diff indices
            if len(indices) >= abs(diff):
                h_val[indices[:abs(diff)]] = 1.0 if diff > 0 else 0.0
        return h_val, float(t.value)
    else:
        return None, None

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a binary (or 0/1) vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def solve_milp(N, time_limit):
    """Solve the integer program that minimizes the maximal overlap."""
    import cvxpy as cp

    # binary decision variables for h
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()                     # upper bound on all C_d
    constraints = [cp.sum(h) == N // 2, t >= 0]

    # for each positive shift d
    for d in range(1, N):
        hi = h[: N - d]                    # h[i]
        hj = h[d:]                         # h[i+d]

        # auxiliary variables w_{i,d} = hi * hj
        w = cp.Variable(N - d, boolean=True)
        constraints.append(w <= hi)
        constraints.append(w <= hj)
        constraints.append(w >= hi + hj - 1)

        Ld = cp.sum(w)                     # number of 1–1 pairs at shift d
        Md = cp.sum(hi)                    # number of 1's that appear in the overlap region
        constraints.append(Md - Ld <= t)   # C_d <= t

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try to respect the time budget (CBC accepts max_seconds)
    try:
        prob.solve(solver=cp.CBC, warm_start=True, max_seconds=time_limit)
    except TypeError:
        # fallback if the solver does not accept the argument
        prob.solve(solver=cp.CBC, warm_start=True)

    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
        h_val = np.round(h.value).astype(float)

        # enforce exact half‑density in case of rounding issues
        cur = int(round(h_val.sum()))
        target = N // 2
        if cur != target:
            diff = target - cur
            if diff > 0:
                zeros = np.where(h_val == 0)[0]
                if zeros.size >= diff:
                    h_val[zeros[:diff]] = 1.0
            elif diff < 0:
                ones = np.where(h_val == 1)[0]
                if ones.size >= -diff:
                    h_val[ones[: -diff]] = 0.0
        return h_val, float(t.value)
    return None, None

    """Return a step function h, its C5 bound, and the number of samples."""
    start = time.time()
    # Determine dimension and possibly an initial vector
    try:
        init = initial_h_values
        N = int(init.shape[0])
        init_vec = np.clip(init.astype(float), 0.0, 1.0)
    except NameError:
        init = None
        N = 80
        init_vec = None

    # ------------------------------------------------------------------ #
    # 1. Try the exact MILP model (may succeed quickly for N=80)
    h_best = None
    try:
        elapsed = time.time() - start
        remaining = max(0.0, budget_s - elapsed - 5.0)  # keep a tiny safety margin
        if remaining > 15.0:                           # give it a chance only if enough time
            h_best, _ = solve_milp(N, remaining)
    except Exception:
        h_best = None

    if h_best is not None:
        return h_best, compute_c5(h_best), N

    # ------------------------------------------------------------------ #
    # 2. Heuristic local search (hill climbing with random swaps)
    rng = np.random.default_rng(seed)

    # initialise a feasible binary vector
    if init_vec is not None and np.isclose(init_vec.sum(), N / 2):
        # make sure we start with a strict 0/1 vector
        cur_h = np.where(init_vec >= 0.5, 1.0, 0.0)
