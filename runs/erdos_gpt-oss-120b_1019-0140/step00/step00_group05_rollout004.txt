import time
import numpy as np

def _compute_c5(vec):
    """Return the overlap bound C5 for a step function vector."""
    try:
        # Use the provided evaluator if it exists.
        return evaluate_erdos_solution(vec)
    except NameError:
        dx = 2.0 / len(vec)
        return (np.correlate(vec, 1 - vec, mode="full") * dx).max()

def _fix_sum(vec):
    """Adjust the first entry to enforce exact sum = n/2 (preserving bounds)."""
    n = len(vec)
    target = n / 2.0
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # Apply the correction to the first element, clipping to [0,1].
    vec[0] = np.clip(vec[0] + diff, 0.0, 1.0)
    # If clipping broke the sum, distribute the remaining error uniformly.
    rem = target - vec.sum()
    if abs(rem) > 1e-12:
        # Very small remaining error; spread over all entries proportionally.
        vec += rem / n
        vec = np.clip(vec, 0.0, 1.0)
        # Final small correction on the first element.
        vec[0] = np.clip(vec[0] + (target - vec.sum()), 0.0, 1.0)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that lowers the Erdős C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values : list of floats, length n_points, each in [0,1] and sum = n/2.
        c5_bound : float, the achieved max overlap integral.
        n_points : int, length of the discretization.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Initialization
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():

        n = len(h)

    h = _fix_sum(h)

    # Initial evaluation.
    current_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # Annealing schedule parameters.
    T_initial = 0.02   # start temperature
    T_final = 0.0005   # final temperature
    scale_initial = 0.5
    scale_final = 0.01

    # Counters for stagnation handling.
    no_improve = 0
    iter_count = 0

    # ------------------------------------------------------------------
    # Main optimization loop (simulated annealing with pairwise moves)
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start_time
        # Leave a small safety margin before the hard timeout.
        if elapsed >= budget_s * 0.96:
            break

        # Linear schedule for temperature and step scale.
        progress = elapsed / (budget_s * 0.96)
        T = T_initial * (1.0 - progress) + T_final * progress
        step_scale = scale_initial * (1.0 - progress) + scale_final * progress

        # Propose a pairwise adjustment that preserves the total sum.
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Randomly decide which index will increase and which will decrease.
        if rng.random() < 0.5:
            inc_idx, dec_idx = i, j
        else:
            inc_idx, dec_idx = j, i

        delta_max = min(1.0 - h[inc_idx], h[dec_idx])
        if delta_max <= 0.0:
            continue

        delta = delta_max * rng.random() * step_scale
        new_h = h.copy()
        new_h[inc_idx] += delta
        new_h[dec_idx] -= delta

        new_c5 = _compute_c5(new_h)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                no_improve = 0
            else:

            current_c5 = best_c5

    # ------------------------------------------------------------------
