        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        src, dst = i, j
        max_delta = min(eps_max, h[src], 1.0 - h[dst])
        if max_delta <= 0.0:
            src, dst = j, i
            max_delta = min(eps_max, h[src], 1.0 - h[dst])
            if max_delta <= 0.0:
                continue

        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        new_score = _c5_value(h_new)

        # temperature that linearly cools to zero
        T = max(1e-12, T0 * (1.0 - elapsed / float(budget_s)))
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()

        it += 1
        if it % 2000 == 0:
            eps_max = max(0.001, eps_max * 0.99)

    return best_h.tolist(), float(best_score), n
