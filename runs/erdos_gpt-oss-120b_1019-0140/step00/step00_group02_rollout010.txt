def repair(h, total_needed):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # adjust by scaling then clip iteratively
    if s == 0:
        h[:] = total_needed / h.size
        return h
    # Simple scaling then clipping
    h = h * (total_needed / s)
    # after scaling some may exceed bounds -> clip and renormalize again
    for _ in range(10):
        over = h > 1.0
        under = h < 0.0
        if not over.any() and not under.any():
            break
        # Clip
        h[over] = 1.0
        h[under] = 0.0
        # compute sum of fixed values
        fixed = h.copy()
        fixed[~(over|under)] = 0
        # compute remaining sum needed
        remaining = total_needed - fixed.sum()
        # number of free idx
        free_idx = ~(over|under)
        if free_idx.sum() == 0:
            break
        # distribute remaining equally among free entries, proportionally?
        h[free_idx] = np.maximum(0, np.minimum(1, remaining / free_idx.sum()))
    return h

def project_to_feasible(h, target_sum):
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-9:
        return h
    # Simple iterative scaling with clipping
    for _ in range(10):

            break
        adjust = diff / free.sum()
