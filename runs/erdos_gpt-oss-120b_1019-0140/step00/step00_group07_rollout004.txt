import time, math, random
import numpy as np

def _compute_c5(vec):
    """Return C5 for a given vector."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded mode
    corr = np.correlate(vec, 1.0 - vec, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function lowering the Erdős C5 bound.

    Returns:
        (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------
    # Determine problem size and a feasible starting point
    # ------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
        n = init_h.shape[0]
        target_sum = n / 2.0
        cur_sum = init_h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            # simple rescaling + clipping to satisfy constraints
            init_h = init_h * (target_sum / cur_sum)
            np.clip(init_h, 0.0, 1.0, out=init_h)
            diff = target_sum - init_h.sum()
            init_h[0] = np.clip(init_h[0] + diff, 0.0, 1.0)
    except Exception:
        init_h = None
        n = 200                     # default size when no parent is given
        if n % 2:
            n += 1
        target_sum = n / 2.0

    # ------------------------------------------------------------
    # Helper to create a random feasible vector (continuous)
    # ------------------------------------------------------------
    def _random_feasible():
        # Dirichlet‑like random point on the simplex, then clip & renorm
        x = np.random.rand(n)
        x = x / x.sum() * target_sum
        np.clip(x, 0.0, 1.0, out=x)
        diff = target_sum - x.sum()
        x[0] = np.clip(x[0] + diff, 0.0, 1.0)
        return x

    # ------------------------------------------------------------
    # Initial best solution (fallback)
    # ------------------------------------------------------------
    if init_h is not None:
        best_vec = init_h.copy()
    else:
        best_vec = _random_feasible()
    best_score = _compute_c5(best_vec)

    # ------------------------------------------------------------
    # Core optimisation: simulated annealing with mass‑transfer moves
    # ------------------------------------------------------------
    start_time = time.time()
    max_temp = 0.02          # initial temperature
    min_temp = 1e-6          # final temperature
    max_frac = 0.5           # maximal fraction of transferable mass per move
    global_iter = 0

    while True:
        now = time.time()
        if now - start_time >= budget_s * 0.99:
            break

        # initialise a new candidate (often from the parent, else random)
        if init_h is not None and random.random() < 0.15:
            vec = init_h.copy()
        else:
            vec = _random_feasible()
        cur_score = _compute_c5(vec)

        # number of inner moves for this run (adapt to remaining time)
        remaining = budget_s - (now - start_time)
        inner_moves = int(min(20000, remaining * 30))   # roughly 30 moves per second

        for _ in range(inner_moves):
            now = time.time()
            if now - start_time >= budget_s * 0.99:
                break

            # linear temperature schedule
            frac_time = (now - start_time) / budget_s
            temp = max_temp * (1.0 - frac_time) + min_temp * frac_time

            # ---- propose a mass‑transfer move ---------------------------------
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

            # try to move mass from j -> i; if impossible, reverse the direction
            inc = 1.0 - vec[i]          # how much i can increase
            dec = vec[j]                # how much j can decrease
            if inc <= 0.0 or dec <= 0.0:
                inc = 1.0 - vec[j]
                dec = vec[i]
                if inc <= 0.0 or dec <= 0.0:
                    continue           # both positions stuck at a bound
                i, j = j, i            # swap indices so the move direction stays i<-j

            max_delta = min(inc, dec)
            # stochastic step size
            delta = max_delta * np.random.rand() * max_frac
            if delta <= 0.0:
                continue

            vec[i] += delta
            vec[j] -= delta

            new_score = _compute_c5(vec)

            # ---- acceptance test ---------------------------------------------
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # Metropolis acceptance
                prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
                if random.random() < prob:
                    accept = True

            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_vec = vec.copy()
            else:
                # revert
                vec[i] -= delta
                vec[j] += delta

            global_iter += 1

    # ------------------------------------------------------------
    # Final sanity check (ensure constraints)
    # ------------------------------------------------------------
    # Clip numeric noise
    np.clip(best_vec, 0.0, 1.0, out=best_vec)
    diff = target_sum - best_vec.sum()
    best_vec[0] = np.clip(best_vec[0] + diff, 0.0, 1.0)

    return (best_vec.tolist(), float(best_score), n)
