import numpy as np
import time
import random
import math
from scipy.optimize import minimize

def c5_score(h):
    """Compute the C5 overlap bound for a step function h."""
    try:
        # Pre‑imported evaluator, if available
        return evaluate_erdos_solution(h)
    except NameError:
        dx = 2.0 / len(h)
        c = np.correlate(h, 1 - h, mode="full") * dx
        return c.max()

def smooth_obj_and_grad(x, target_sum, beta, lam):
    """Smooth approximation of max C5 with gradient for L‑BFGS‑B."""
    n = x.shape[0]
    sum_h = np.sum(x)
    # autocorrelation of x with itself (zero‑padded)
    corr = np.correlate(x, x, mode="full")
    corr = corr[n - 1 :]  # length n, corr[k] = sum_i x_i * x_{i+k}
    c = sum_h - corr  # C_k without the dx factor (constant scaling irrelevant)
    max_bc = np.max(beta * c)
    expb = np.exp(beta * c - max_bc)          # stabilized exponentials
    sumexp = np.sum(expb)
    smooth_val = (max_bc + np.log(sumexp)) / beta
    penalty = lam * (sum_h - target_sum) ** 2
    val = smooth_val + penalty
    # softmax weights for each shift
    soft = expb / sumexp
    # gradient of Σ soft_k * corr[k] w.r.t. x
    grad_corr = np.zeros_like(x)
    # k = 0 term (contributes 2*x)
    grad_corr += 2.0 * soft[0] * x
    # k > 0 terms
    for k in range(1, n):
        wk = soft[k]
        if wk == 0.0:
            continue
        # forward contribution (x_{j+k})
        grad_corr[: n - k] += wk * x[k:]
        # backward contribution (x_{j-k})
        grad_corr[k:] += wk * x[: n - k]
    # gradient of smooth max part: 1 - grad_corr
    grad = np.ones_like(x) - grad_corr
    # gradient of penalty term
    grad += 2.0 * lam * (sum_h - target_sum)
    return float(val), grad

def optimize_h(initial_h, target_sum, beta_seq, lam):
    """Run a sequence of smooth L‑BFGS‑B optimisations."""
    n = len(initial_h)
    bounds = [(0.0, 1.0)] * n
    h = np.clip(initial_h, 0.0, 1.0)
    cur_sum = h.sum()
    if cur_sum > 0.0:
        h = h * (target_sum / cur_sum)
    h = np.clip(h, 0.0, 1.0)
    for beta in beta_seq:
        res = minimize(
            smooth_obj_and_grad,
            h,
            args=(target_sum, beta, lam),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={"maxiter": 300, "ftol": 1e-12},
        )
        h = res.x
        # gentle re‑projection onto the sum hyperplane
        cur_sum = h.sum()
        diff = cur_sum - target_sum
        if abs(diff) > 1e-8:
            h = h - diff / n
            np.clip(h, 0.0, 1.0, out=h)
    return h

def random_binary(n, target_sum):
    """Generate a binary vector with exactly target_sum ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, int(target_sum), replace=False)
    h[idx] = 1.0
    return h

def local_search(start_h, target_sum, budget_s, start_time):
    """Simulated‑annealing style refinement to reduce C5."""
    n = len(start_h)
    best_h = start_h.copy()
    best_score = c5_score(best_h)
    cur_h = start_h.copy()
    cur_score = best_score
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # Choose move type
        if random.random() < 0.2:
            # Fractional delta move preserving sum
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]
            lower = max(-hi, hj - 1.0)
            upper = min(1.0 - hi, hj)
            if lower >= upper:
                continue
            delta = lower + (upper - lower) * random.random()
            new_h = cur_h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
        else:

            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
        new_score = c5_score(new_h)
        # Temperature schedule (linear decay)
        elapsed = time.time() - start_time

            cur_score = new_score

        cur_sum = init_h.sum()
        if abs(cur_sum - target_sum) > 1e-12 and cur_sum != 0.0:
            init_h = init_h * (target_sum / cur_sum)
        np.clip(init_h, 0.0, 1.0, out=init_h)
        base_start = init_h
    else:
        n = 80
        target_sum = n / 2.0
        base_start = None
    start_time = time.time()
    best_h = None
    best_score = float("inf")
    restart = 0
    # Parameters for the smooth optimiser
    beta_seq = [5.0, 20.0, 80.0]
    lam = 1000.0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        if restart == 0 and base_start is not None:
            h0 = base_start.copy()
        else:
            h0 = random_binary(n, target_sum)
        # Phase 1: smooth L‑BFGS‑B optimisation
        h_opt = optimize_h(h0, target_sum, beta_seq, lam)
        # Phase 2: local refinement
        h_loc, score_loc = local_search(h_opt, target_sum, budget_s, start_time)
        if score_loc < best_score:
            best_score = score_loc
            best_h = h_loc.copy()
        restart += 1
        if restart >= 10:
            # allow continued local refinement only; keep looping until time out
            continue
    if best_h is None:
        best_h = random_binary(n, target_sum)
    # Final sum correction (tiny numerical drifts)
    cur_sum = best_h.sum()
    diff = cur_sum - target_sum
    if abs(diff) > 1e-8:
        best_h = best_h - diff / n
        np.clip(best_h, 0.0, 1.0, out=best_h)
    final_score = c5_score(best_h)
    return best_h, float(final_score), n
