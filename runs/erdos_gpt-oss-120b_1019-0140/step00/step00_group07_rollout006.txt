import math

def compute_c5(vec: np.ndarray) -> float:
    """Return the overlap integral C5 for a discretised step function."""
    n = vec.shape[0]
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(np.max(corr) * dx)


def worst_shift(vec: np.ndarray):
    """Return (shift, full_correlation_array) for the shift attaining maximal overlap."""
    n = vec.shape[0]
    full = np.correlate(vec, 1.0 - vec, mode="full")
    idx = int(np.argmax(full))
    shift = idx - (n - 1)
    return shift, full
