def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    N = len(initial_h_values) if 'initial_h_values' in globals() else 80
    # build initial binary vector:
    if 'initial_h_values' in globals():
        # convert to binary with exactly N/2 ones based on values
        v = np.array(initial_h_values, dtype=float)
        # choose top N/2 entries as 1
        idx = np.argsort(-v)[:N//2]
        h = np.zeros(N, dtype=float)
        h[idx] = 1.0
    else:
        # random
        h = np.zeros(N, dtype=float)
        h[rng.choice(N, N//2, replace=False)] = 1.0

    def compute_c5(arr):
        dx = 2.0 / len(arr)
        # correlation as numpy correlate: returns array length 2N-1
        corr = np.correlate(arr, 1.0 - arr, mode='full') * dx
        return float(corr.max())
    
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # parameters
    start = time.time()
    # Simulated annealing settings
    T_init = 0.1
    T_final = 1e-4
    max_iters = 10_000_000  # large, but will break by time.
    # we may adjust iteration count based on time.

    # keep current solution
    cur_h = best_h.copy()
    cur_score = best_score

    # precompute indices of ones and zeros for current
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break
        # cooling schedule linear in log time
        frac = elapsed / budget_s
        T = T_init * (1 - frac) + T_final * frac
        # propose swap
        # choose random one and zero
        if len(ones) == 0 or len(zeros) == 0:
            # reinitialize?
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # swap values
        # compute new score
        new_score = compute_c5(cur_h)
        # acceptance
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / T):
            # accept
            cur_score = new_score
            # update ones/zeros sets
            # swap indices in sets
            # Remove i from ones, add to zeros; similar for j.
            # easier: recompute sets
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # reject: revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            # no change to ones/zeros
        it += 1

    # final evaluation using evaluate_erdos_solution if available
    try:
        c5_bound = evaluate_erdos_solution(best_h)
    except Exception:
        c5_bound = best_score

    return (best_h.tolist(), c5_bound, N)

import time
import math
import numpy as np

def _compute_c5(arr):
    """Return C5 for a binary array arr."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation of arr with (1-arr)
    corr = np.correlate(arr, 1.0 - arr, mode='full') * dx
    return float(corr.max())


def _random_binary(rng, n):
    """Random binary vector of length n with exactly n//2 ones."""
    vec = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    vec[ones] = 1.0
    return vec


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h over [0,2] (discretized) that yields a small
    C5 value. Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        # Binarize: take the n//2 largest entries as 1, rest 0
        order = np.argsort(-init)
        h_cur = np.zeros(n, dtype=float)
        h_cur[order[: n // 2]] = 1.0
    else:
        n = 80  # default size if no parent provided
        h_cur = _random_binary(rng, n)

    best_h = h_cur.copy()
    best_score = _compute_c5(best_h)

    # Simulated annealing parameters
    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # leave a margin
    T_start = 0.12
    T_end = 1e-5

    # Current state
    cur_h = h_cur.copy()
    cur_score = best_score

    # Main loop
    while time.time() < end_time:
        # Linear temperature schedule based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.95)
        T = T_start * (1.0 - frac) + T_end * frac

        # Propose a swap of a 1 and a 0
        ones_idx = np.where(cur_h == 1.0)[0]
        zeros_idx = np.where(cur_h == 0.0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break  # degenerate, should not happen

        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # Apply swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_score = _compute_c5(cur_h)

        # Acceptance test
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(T, 1e-12)):
            cur_score = new_score
            # Update global best if improved
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # Revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # Occasionally perform a random restart to escape stuck regions
        if rng.random() < 0.0004:
            cur_h = _random_binary(rng, n)
            cur_score = _compute_c5(cur_h)

    # Final evaluation using the provided helper if available
    try:
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_bound = best_score

    # Return as plain Python list for compatibility
    return (best_h.tolist(), c5_bound, n)
