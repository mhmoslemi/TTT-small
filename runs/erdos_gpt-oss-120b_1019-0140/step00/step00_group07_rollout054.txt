        if n0 == n:

        else:

            xn = np.linspace(0,2,n,endpoint=False)
            h = np.interp(xn, x0, h0)
    else:

        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.5 * np.sin(np.pi * (x - 1) / 2)  # monotonic from 0 to 1

    target_sum = n/2
    for _ in range(10):
        s = h.sum()
        delta = (target_sum - s) / n
        h += delta
        h = np.clip(h, 0, 1)
    # Final fix: adjust by distributing remainder if needed.
    # Compute residual error after clipping.
    s = h.sum()

            h[free_idx] += (target_sum - s) / len(free_idx)
            h = np.clip(h, 0, 1)
    # Ensure sum close.
    return h

import numpy as np

def _c5_of_h(h: np.ndarray) -> float:
    """Return the C₅ value for a discretised step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1‑h); zero‑padding outside the interval.
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Construct a step function h on [0,2] that satisfies
        0 ≤ h ≤ 1 ,   sum(h) * (2/n) = 1
    and returns a good upper bound for the Erdős constant C₅.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – list of function samples on a uniform grid,
        c5_bound – the evaluated C₅ value,
        n_points – number of samples used.
    """
    # Determine the resolution.  If a parent construction is supplied,
    # adopt its length (still even to keep the symmetry property).
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 200                       # default modest size (< 1 k)

    # Ensure an even number of points – needed for the exact sinusoidal
    # complement property h(x+1) = 1‑h(x) in the discretisation.
    if n % 2:
        n += 1

    # Sinusoidal construction: h_i = sin²(π·i / n).
    # This satisfies the integral constraint exactly (average ½) and yields
    # C₅ = 3/8 = 0.375 analytically.
    i = np.arange(n, dtype=float)
    h = np.sin(np.pi * i / n) ** 2

    # Clamp to the feasible box and enforce the sum constraint
    # (necessary only for floating‑point round‑off).
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    for _ in range(5):
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) < 1e-12:
            break
        delta = (target_sum - cur_sum) / n
        h += delta
        h = np.clip(h, 0.0, 1.0)

    # Evaluate the bound.
    c5 = _c5_of_h(h)

    return h.tolist(), c5, n
