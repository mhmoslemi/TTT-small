        # choose swap indices
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h <= 0.5)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):

            if new_c5 < best_c5:

                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        T *= 0.9995
        iteration += 1
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral C5 for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def binary_from_continuous(v: np.ndarray) -> np.ndarray:
    """Convert a real‑valued vector to a binary one with exactly half the entries =1."""
    n = v.shape[0]
    m = n // 2
    idx = np.argsort(-v)[:m]          # positions of the largest values
    binv = np.zeros(n, dtype=float)
    binv[idx] = 1.0
    return binv

def simulated_annealing_binary(start_h: np.ndarray, start_time: float,
                               budget_s: float, target: float = 0.3808) -> tuple:
    """
    Simple simulated annealing on a binary vector that keeps the sum fixed.
    Returns the best vector found and its C5 value.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # temperature schedule
    T = 0.1
    min_T = 1e-12
    decay = 0.9995
    # pre‑compute index pools
    while True:
        if time.time() - start_time > budget_s * 0.94:
            break
        if best_c5 <= target:
            break
        # select a random 1 and a random 0 to swap
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h <= 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            # avoid division by zero when T is extremely small
            if T < min_T:
                prob = 0.0
            else:
                prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        T = max(min_T, T * decay)
    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of sample points."""
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # Determine number of points and an initial binary vector
    # -------------------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness
        n0 = int(init.shape[0])
        # we aim for a finer discretisation; at least 80 points and even.
        n = max(80, n0 * 2)
        n += n % 2                     # force even
        # interpolate the supplied values onto the new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x, x0, init)
        start_bin = binary_from_continuous(interp)
    except NameError:
        # no parent construction – use a fresh binary vector
        n = 120
        n += n % 2
        start_bin = np.zeros(n, dtype=float)
        start_bin[np.random.choice(n, n // 2, replace=False)] = 1.0

    best_h = None
    best_c5 = np.inf

    # -------------------------------------------------------------
    # Main optimisation loop – repeated SA restarts
    # -------------------------------------------------------------
    trial = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break
        # first trial uses the deterministic start, later trials are random
        if trial == 0:
            h0 = start_bin.copy()
        else:
            h0 = np.zeros(n, dtype=float)
            h0[np.random.choice(n, n // 2, replace=False)] = 1.0
        # run simulated annealing on this binary start
        h_opt, c5_opt = simulated_annealing_binary(
            h0, start_time, budget_s, target=0.3808
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            # early exit if we beat the target bound
            if best_c5 <= 0.3808:
                break
        trial += 1

    # -------------------------------------------------------------
    # Fallback – should never trigger, but ensures a valid return.
    # -------------------------------------------------------------
    if best_h is None:
        n = 100
        best_h = np.zeros(n, dtype=float)
        best_h[np.random.choice(n, n // 2, replace=False)] = 1.0
        best_c5 = compute_c5(best_h)

    # Explicitly enforce the exact integral constraint (tiny numerical drift)
    cur_sum = best_h.sum()
    target_sum = n / 2.0
    diff = target_sum - cur_sum
    if abs(diff) > 1e-10:
        interior = np.where((best_h > 1e-8) & (best_h < 1 - 1e-8))[0]
        if interior.size:
            best_h[interior] += diff / interior.size
            best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, int(best_h.shape[0])
