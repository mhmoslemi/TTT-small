    for _ in range(10):

def project_to_bounds(x, target, lower=0.0, upper=1.0, max_iter=100):
    # Clip initial
    x = np.clip(x, lower, upper)
    for _ in range(max_iter):
        total = x.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # Determine indices that can move
        if diff > 0:
            # need to increase sum: can increase indices where x < upper
            idx = np.where(x < upper)[0]
        else:
            # need to decrease sum: can decrease where x > lower
            idx = np.where(x > lower)[0]
        if idx.size == 0:
            # no free variables, break
            break
        # compute distribution amount per idx
        inc = diff / idx.size
        # apply increment respecting bounds
        x[idx] += inc
        # Clip again to bounds
        x = np.clip(x, lower, upper)
    # for safety, adjust tiny rounding
    # final adjust using scaling if still off by jitter: 
    total = x.sum()
    diff = target - total
    if abs(diff) > 1e-10:
        # fallback: distribute diff across all entries proportionally
        x += diff / len(x)
        x = np.clip(x, lower, upper)
    # final normalization to ensure sum exactly target: 
    # (should be close)
    total = x.sum()
    if abs(total - target) > 1e-8:
        # adjust minor error by adding diff to some entry not at bound
        for i in range(len(x)):
            if lower < x[i] < upper:
                x[i] += (target - total)
                x[i] = min(max(x[i], lower), upper)
                break
    return x

import time
import numpy as np

def project_to_bounds(x, target, lower=0.0, upper=1.0, max_iter=100):
    """Project vector x onto the hyperplane sum(x)=target with box bounds."""
    x = np.clip(np.asarray(x, dtype=float), lower, upper)
    for _ in range(max_iter):
        total = x.sum()
        diff = target - total
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            idx = np.where(x < upper)[0]
        else:
            idx = np.where(x > lower)[0]
        if idx.size == 0:
            break
        inc = diff / idx.size
        x[idx] += inc
        x = np.clip(x, lower, upper)
    # tiny residual correction
    total = x.sum()
    diff = target - total
    if abs(diff) > 1e-10:
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if free.size:
            x[free[0]] += diff
            x = np.clip(x, lower, upper)
        else:
            # fallback uniform adjustment
            x += diff / x.size
            x = np.clip(x, lower, upper)
    return x

def compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def generate_basis(N):
    """Create a small set of zero‑mean trigonometric basis vectors."""
    idx = np.arange(N)
    basis = []
    for freq in (1, 2):
        basis.append(np.cos(2.0 * np.pi * freq * idx / N))
        basis.append(np.sin(2.0 * np.pi * freq * idx / N))
    return np.vstack(basis)   # shape (M, N)

def random_candidate(N, target, basis):
    """Generate a feasible random step function using the basis."""
    base = np.full(N, 0.5)
    coeffs = np.random.uniform(-0.5, 0.5, size=basis.shape[0])
    h = base + coeffs @ basis
    h = project_to_bounds(h, target)
    return h

def two_index_move(h):
    """Perform a zero‑sum move between two indices preserving bounds."""
    N = h.shape[0]
    i, j = np.random.choice(N, size=2, replace=False)
    if np.random.rand() < 0.5:
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return h.copy()
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return h.copy()
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C5 bound."""
    np.random.seed(seed)

    # Determine problem size and starting vector
    try:
        init = np.array(initial_h_values, dtype=float).flatten()
        N = init.size
        h0 = init.copy()
    except NameError:
        N = 70
        h0 = random_candidate(N, N / 2.0, generate_basis(N))

    target_sum = N / 2.0
    h0 = project_to_bounds(h0, target_sum)
    dx = 2.0 / N

    # Pre‑compute basis for structured random restarts
    basis = generate_basis(N)

    # Current and best solutions
    curr = h0
    curr_val = compute_c5(curr, dx)
    best = curr.copy()
    best_val = curr_val

    start_time = time.time()
    T0 = 0.05            # initial temperature
    p_reset = 0.001      # probability of a full random restart per iteration

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed > budget_s * 0.99:
            break

        # Temperature schedule (linear cooling)
        T = T0 * max(0.01, 1.0 - elapsed / budget_s)

        # Occasionally try a fresh random candidate from the basis
        if np.random.rand() < p_reset:
            cand = random_candidate(N, target_sum, basis)
        else:
            cand = two_index_move(curr)

        cand_val = compute_c5(cand, dx)

        # Metropolis acceptance
        if cand_val < curr_val or np.random.rand() < np.exp((curr_val - cand_val) / T):
            curr = cand
            curr_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val

    return best, best_val, N
