import time
import math
import numpy as np

def _compute_c5(vec):
    """Compute the overlap constant C5 for a discrete step function."""
    dx = 2.0 / vec.size
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def _project_bounds_sum(v, target):
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v)=target.
    Simple water-filling iteration (fast for small n).
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            break
        v[free] += diff / n_free
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n, rng):
    """Generate a random feasible vector of length n (average 0.5, bounds [0,1])."""
    # start from uniform random and project
    v = rng.random(n)
    target = n / 2.0
    v = _project_bounds_sum(v, target)
    return v

def _local_optimize(h_start, time_budget, rng):
    """
    Simple simulated‑annealing style optimizer.
    Returns the best vector found within time_budget seconds.
    """
    h_cur = np.array(h_start, dtype=float)
    n = h_cur.size
    target_sum = n / 2.0

    best_h = h_cur.copy()
    best_val = _compute_c5(best_h)
    cur_val = best_val

    start = time.time()
    # initial temperature – a modest fraction of current value
    init_temp = max(0.005, best_val * 0.05)

    # stagnation counter for occasional re‑randomisation
    stagnation = 0
    while True:
        if time.time() - start > time_budget * 0.98:
            break

        # propose a two‑coordinate move keeping the sum fixed
        i, j = rng.choice(n, size=2, replace=False)
        max_up = min(1.0 - h_cur[i], h_cur[j])          # delta > 0
        max_down = min(h_cur[i], 1.0 - h_cur[j])        # delta < 0
        if max_up + max_down <= 0:
            continue
        delta = rng.uniform(-max_down, max_up)

        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_val = _compute_c5(h_new)

        # acceptance rule
        if new_val < cur_val:
            h_cur = h_new
            cur_val = new_val
            stagnation = 0
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            # temperature linearly decays
            elapsed = time.time() - start
            temp = init_temp * (1.0 - elapsed / time_budget)
            if temp > 0 and rng.random() < math.exp(-(new_val - cur_val) / temp):
                h_cur = h_new
                cur_val = new_val
            stagnation += 1

        if stagnation % 5000 == 0:

        if stagnation >= 200000:
            h_cur = best_h.copy()
            cur_val = best_val
            stagnation = 0
