def perturb(h, rng, step):
    # step: max magnitude delta.
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Determine allowable delta range:
    max_delta = min(step, 1 - h[i], h[j])  # because we will add delta to i and subtract from j.
    min_delta = max(-step, -h[i], h[j] - 1) # ensure h[i] stays >=0, h[j]<=1
    if max_delta < min_delta:
        return h  # no feasible move
    delta = rng.uniform(min_delta, max_delta)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

import time, math, numpy as np

def score_of(h):
    return float(evaluate_erdos_solution(h))

def perturb_vector(vec, rng, max_delta):
    n = vec.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi = vec[i]; hj = vec[j]
    delta_min = max(-max_delta, -hi, hj - 1.0)
    delta_max = min(max_delta, 1.0 - hi, hj)
    if delta_min >= delta_max:
        return vec
    delta = rng.uniform(delta_min, delta_max)
    new_vec = vec.copy()
    new_vec[i] = hi + delta
    new_vec[j] = hj - delta
    return new_vec

def local_search(start_vec, rng, time_budget):
    h = start_vec.copy()
    best_h = h.copy()
    best_score = score_of(best_h)
    start_time = time.time()
    it = 0
    max_delta = 0.2
    # schedule: gradually reduce max_delta
    while time.time() - start_time < time_budget:
        it += 1
        # Temperature schedule based on time fraction
        elapsed = time.time() - start_time
        # initial temp 0.02 perhaps.
        temp = max(1e-6, 0.02 * (1 - elapsed / time_budget))
        cand = perturb_vector(h, rng, max_delta)
        cand_score = score_of(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()
        # acceptance for current state (hillclimbing)
        if cand_score < best_score or rng.rand() < math.exp(-(cand_score - best_score) / temp):
            h = cand
        # reduce max_delta occasionally
        if it % 500 == 0:
            max_delta *= 0.9
            if max_delta < 0.001:
                max_delta = 0.001
    # final: ensure sum constraint
    n = len(best_h)
    target_sum = n/2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-8:
        # Adjust uniformly
        best_h = best_h * (target_sum / cur_sum)
    # Clip bounds
    np.clip(best_h, 0.0, 1.0, out=best_h)
    final_score = score_of(best_h)
    return best_h, final_score

def _perturb(vec, rng, max_delta):
    n = vec.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # Choose move type: swap (10% chance) else transfer
    if rng.rand() < 0.1:
        # swap values
        new_vec = vec.copy()
        new_vec[i], new_vec[j] = vec[j], vec[i]

    new_vec[source] -= delta
    new_vec[target] += delta
    return new_vec

    h = start_vec.copy()
    best_h = h.copy()
    best_score = _score(best_h)
    start = time.time()

        init = np.array(initial_h_values, dtype=float)
        factor = 2                     # up‑sample by simple repetition
        n = len(init) * factor
        h0 = np.repeat(init, factor)
    else:
        n = 140                        # modest size for speedy search
        h0 = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h0[ones] = 1.0

    # enforce the integral constraint exactly
    target_sum = n / 2.0
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h0 = h0 * (target_sum / cur_sum)
    np.clip(h0, 0.0, 1.0, out=h0)

    best_h = h0.copy()
    best_score = _score(best_h)

    # main optimisation loop with multiple restarts
    while time.time() - start_time < budget_s - 1.0:
        time_left = budget_s - (time.time() - start_time)
        local_time = min(30.0, time_left)

        # occasionally restart from random balanced vector, otherwise refine current best
        if rng.rand() < 0.3:
            start_vec = best_h.copy()
        else:
            start_vec = np.zeros(n, dtype=float)
            ones = rng.choice(n, size=n // 2, replace=False)
            start_vec[ones] = 1.0

        cand_h, cand_score = _local_search(start_vec, rng, local_time)

        if cand_score < best_score:
            best_h = cand_h.copy()
            best_score = cand_score

        if best_score <= 0.3808:
            break

    # final post‑processing to satisfy constraints exactly
    np.clip(best_h, 0.0, 1.0, out=best_h)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h * (target_sum / cur_sum)
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_score = _score(best_h)
    return (best_h.tolist(), float(final_score), int(n))
