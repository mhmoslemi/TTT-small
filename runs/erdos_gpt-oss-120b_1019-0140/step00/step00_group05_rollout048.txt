    diff = target - w.sum()

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # Determine n_points
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        use_init = True
    except NameError:
        n = 70
        init = None
        use_init = False

    target_sum = n / 2.0
    dx = 2.0 / n

    # Helper functions inside run for clarity.
    def project(v):
        lo = -1.0
        hi = 1.0
        for _ in range(50):
            lam = (lo + hi) / 2.0
            w = np.clip(v + lam, 0.0, 1.0)
            s = w.sum()
            if s < target_sum:
                lo = lam
            else:
                hi = lam
        w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
        diff = target_sum - w.sum()
        if abs(diff) > 1e-12:
            # adjust an interior element
            mask = (w > 0) & (w < 1)
            if mask.any():
                i = np.flatnonzero(mask)[0]
                w[i] = np.clip(w[i] + diff, 0.0, 1.0)
            else:
                # all at bounds, just adjust first element respecting bounds
                i = 0
                w[i] = np.clip(w[i] + diff, 0.0, 1.0)
        return w

    def compute_c5(h):
        # correlation
        c = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(c) * dx)

    # initialize population
    pop_size = 7
    population = []
    scores = []
    if use_init:
        h0 = np.array(initial_h_values, dtype=float)
        # ensure projection (if needed)
        h0 = project(h0)
        c0 = compute_c5(h0)
        population.append(h0)
        scores.append(c0)
    # fill rest random
    while len(population) < pop_size:
        v = rng.random(n)  # uniform [0,1)
        h = project(v)
        population.append(h)
        scores.append(compute_c5(h))

    # Main loop
    start = time.time()
    time_limit = budget_s * 0.95
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]
    it = 0

    # parameters for mutation and annealing
    sigma0 = 0.5
    sigma_min = 0.001

    while time.time() - start < time_limit:
        it += 1
        # choose parent: best or random chance for diversity
        if rng.random() < 0.8:
            parent = best_h
        else:
            parent = population[rng.integers(pop_size)]

        # adaptive sigma: decay exponentially
        fraction = (time.time() - start) / time_limit
        sigma = sigma0 * (1 - fraction) + sigma_min * fraction

        # mutate: Gaussian noise
        cand = parent + rng.normal(0, sigma, size=n)
        cand = project(cand)

        cand_score = compute_c5(cand)

        # Insert into population if better than worst
        worst_idx = int(np.argmax(scores))
        if cand_score < scores[worst_idx]:
            population[worst_idx] = cand
            scores[worst_idx] = cand_score

        # Update best
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()

        # Occasionally generate random new individual to escape
        if it % 200 == 0:
            rnd = project(rng.random(n))
            rnd_score = compute_c5(rnd)
            worst_idx = int(np.argmax(scores))
            if rnd_score < scores[worst_idx]:
                population[worst_idx] = rnd
                scores[worst_idx] = rnd_score
            if rnd_score < best_score:
                best_score = rnd_score
                best_h = rnd.copy()
    # Return solution
    return best_h.tolist(), best_score, n

import numpy as np

def project_vector(v, target_sum):
    # binary-like search for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0) & (w < 1)
        if mask.any():
            i = np.flatnonzero(mask)[0]
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
        else:
            # adjust first element staying in bounds
            i = 0
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w

import time
import numpy as np

def project_vector(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the box [0,1]^n and the affine hyperplane sum(v)=target_sum."""
    lo, hi = -2.0, 2.0  # wide enough to cover any shifted values
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0) & (w < 1)
        if mask.any():
            i = np.flatnonzero(mask)[0]
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
        else:
            # All entries are at a bound; adjust the first element safely
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def mutate_candidate(parent: np.ndarray, sigma: float, rng: np.random.Generator,
                    target_sum: float) -> np.ndarray:
    cand = parent + rng.normal(0.0, sigma, size=parent.shape)
    return project_vector(cand, target_sum)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        init_arr = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n = init_arr.size
        use_init = True
    except Exception:
        n = 70
        init_arr = None
        use_init = False

    target_sum = n / 2.0
    dx = 2.0 / n

    pop_size = 7
    population = []
    scores = []

    # Seed population
    if use_init:
        h0 = project_vector(init_arr, target_sum)
        population.append(h0)
        scores.append(compute_c5(h0, dx))

    while len(population) < pop_size:
        v = rng.random(n)
        h = project_vector(v, target_sum)
        population.append(h)
        scores.append(compute_c5(h, dx))

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    time_limit = budget_s * 0.95  # leave a safety margin
    sigma_start = 0.5
    sigma_end = 0.001
    iteration = 0

    while time.time() - start < time_limit:
        iteration += 1

        # Choose a parent (mostly the current best)
        if rng.random() < 0.8:
            parent = best_h
        else:
            parent = population[rng.integers(pop_size)]

        # Linear schedule for mutation strength
        frac = (time.time() - start) / time_limit
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac

        # Generate a mutant
        mutant = mutate_candidate(parent, sigma, rng, target_sum)
        mutant_score = compute_c5(mutant, dx)

        # Replace the worst individual if improvement
        worst_idx = int(np.argmax(scores))
        if mutant_score < scores[worst_idx]:
            population[worst_idx] = mutant
            scores[worst_idx] = mutant_score

        if mutant_score < best_score:
            best_score = mutant_score

                best_score = rnd_score
