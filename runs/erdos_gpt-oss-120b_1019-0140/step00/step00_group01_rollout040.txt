import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Construct a simple linear ramp h(x) = x/2 on [0,2] sampled uniformly.

    # Choose a modest number of points (well below 1000 for speed)
    n_points = 200

    h = (np.arange(n_points, dtype=float) + 0.5) / n_points   # h_i = (i+0.5)/n
