def compute_subgradient(h, lags):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    for lag in lags:
        if lag >= 0:
            # first term for i = 0..n-1-lag
            grad[:n - lag] += 1 - h[lag:]
            # second term for i = lag..n-1
            grad[lag:] -= h[:n - lag]

            grad[l:] += 1 - h[:n - l]
            # second term for i = 0..n-1-l
            grad[:n - l] -= h[l:]
    grad *= dx / len(lags)
    return grad

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n intersect {sum = target_sum}."""
    # bounds for lambda such that sum_i clip(v_i - lam,0,1) covers the target
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        if w.sum() > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if numerical error remains
    diff = w.sum() - target_sum
    if abs(diff) > 1e-9:
        free = np.where((w > 0) & (w < 1))[0]
        if free.size:
            w[free] -= diff / free.size
            w = np.clip(w, 0.0, 1.0)
    return w


def _c5_and_active_lags(h: np.ndarray):
    """Return (C5 value, array of integer lags achieving the maximum)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    # treat near‑ties as equal
    tol = 1e-12
    active_idx = np.where(corr >= max_val - tol)[0]
    lags = active_idx - (n - 1)
    dx = 2.0 / n
    return max_val * dx, lags


def _subgradient(h: np.ndarray, lags: np.ndarray):
    """Subgradient of the max‑overlap objective at h (averaged over active lags)."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    for lag in lags:
        if lag >= 0:
            # first term: (1 - h[i+lag]) for i=0..n-1-lag
            grad[: n - lag] += 1.0 - h[lag:]
            # second term: -h[i-lag] for i=lag..n-1
            grad[lag:] -= h[: n - lag]
        else:
            l = -lag
            # first term for i=l..n-1
            grad[l:] += 1.0 - h[: n - l]
            # second term for i=0..n-1-l
            grad[: n - l] -= h[l:]
    grad *= dx / float(len(lags))
    return grad

            cur = best_h + pert
            cur = np.clip(cur, 0.0, 1.0)
            cur = _project_onto_constraints(cur, target_sum)

        # gradient‑descent inner loop
        for _ in range(block_iters):
            c5_val, active = _c5_and_active_lags(cur)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = cur.copy()
                if best_c5 <= 0.3808:
                    # reached the target bound – we can quit early
                    break

            # compute a subgradient and take a projected step
            g = _subgradient(cur, active)
            nxt = cur - lr * g
            cur = _project_onto_constraints(nxt, target_sum)

            # decay learning rate modestly
            lr = max(min_lr, lr * decay)

            iteration += 1
            if iteration % 1000 == 0 and time.time() - start > budget_s - 0.5:
                break
        else:
            # normal exit of inner loop (no break on target)
            continue
        # break outer while if target achieved
        if best_c5 <= 0.3808:
            break

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_constraints(best_h, target_sum)
    final_c5, _ = _c5_and_active_lags(best_h)

    return (best_h, final_c5, n)
