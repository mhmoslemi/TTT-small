import numpy as np

def _c5_score(h):
    """Compute C5 for a binary step function h using the provided evaluator."""
    # Use the harness's evaluator if available; otherwise compute directly.
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        # fallback computation
        N = h.shape[0]
        dx = 2.0 / N
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] minimizing the Erdős C5 bound.

    Returns:
        (h_values, c5_bound, n_points) where
        - h_values is a 1‑D NumPy array with values in [0,1],
        - c5_bound is the achieved C5 value (float),
        - n_points is the number of discretisation points (int).
    """
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # --------------------------------------------------------------------
    # Initialise a feasible binary vector.
    # --------------------------------------------------------------------
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.array(init, dtype=float)
        N = h0.size
        # Ensure binary values and exact half‑sum.
        h_bin = (h0 > 0.5).astype(float)
        target_sum = N // 2
        cur_sum = int(h_bin.sum())
        if cur_sum != target_sum:
            # Adjust the smallest deviations to meet the sum constraint.
            diff = target_sum - cur_sum
            if diff > 0:
                zero_idx = np.where(h_bin == 0)[0]
                chosen = rng.choice(zero_idx, diff, replace=False)
                h_bin[chosen] = 1.0
            else:
                one_idx = np.where(h_bin == 1)[0]
                chosen = rng.choice(one_idx, -diff, replace=False)

        current_h = h_bin
    else:

        N = 70
        target_sum = N // 2
        current_h = np.zeros(N, dtype=float)
        current_h[rng.choice(N, target_sum, replace=False)] = 1.0

    # --------------------------------------------------------------------
    # Evaluate the initial configuration.
    # --------------------------------------------------------------------
    best_h = current_h.copy()
    best_score = _c5_score(best_h)
    current_score = best_score

    # --------------------------------------------------------------------
    # Simulated annealing parameters.
    # --------------------------------------------------------------------
    T = 1.0                     # initial temperature
    min_T = 1e-5                # floor temperature
    decay = 0.9995              # exponential cooling factor
    stagnation_limit = 5000     # steps without improvement before reset
    stagnation = 0

    # To avoid re‑scanning indices each iteration, maintain lists of positions.
    ones_pos = set(np.where(current_h == 1)[0].tolist())
    zeros_pos = set(np.where(current_h == 0)[0].tolist())

    # --------------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------------
    while True:
        # Time check – leave a safety margin.
        if time.time() - start_time > budget_s * 0.95:

        # Randomly pick a 1‑position and a 0‑position to swap.
        i = rng.choice(list(ones_pos))
        j = rng.choice(list(zeros_pos))

        # Perform the swap.
        current_h[i] = 0.0
        current_h[j] = 1.0
        # Update the position sets.
        ones_pos.remove(i)
        zeros_pos.remove(j)
        ones_pos.add(j)
        zeros_pos.add(i)

        # Evaluate new candidate.
        new_score = _c5_score(current_h)

        # Acceptance test.
        delta = new_score - current_score
        if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-12)):

            if new_score < best_score:

                best_h = current_h.copy()
                stagnation = 0

            current_h[j] = 0.0
            ones_pos.add(i)
            zeros_pos.add(j)
            ones_pos.remove(j)
            zeros_pos.remove(i)

            zeros_pos = set(np.where(best_h == 0)[0].tolist())
            current_score = best_score
            stagnation = 0
            T = 1.0

    # --------------------------------------------------------------------
    # Return the best solution discovered.
    # --------------------------------------------------------------------
    return (best_h, float(best_score), int(N))
