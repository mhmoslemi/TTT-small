def project(v, total):
    x = np.clip(v, 0, 1)
    diff = total - x.sum()
    # While |diff| > 1e-12
    while abs(diff) > 1e-9:
        # Get free indices where x > 0 and x <1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if len(free) == 0:
            # no free variables: distribution can't be fixed, adjust by distributing diff across all (maybe rounding errors). Break.
            break
        delta = diff / len(free)
        # Apply delta, but clip again
        x[free] += delta
        # Clip overflow and recompute diff
        x = np.clip(x, 0, 1)
        diff = total - x.sum()
        # If diff is small, break
    return x

import time
import numpy as np


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 bound for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _gradient(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the worst overlap term (without dx) w.r.t. h."""
    n = h.shape[0]
    idx1 = np.arange(n) - lag
    term1 = np.zeros_like(h)
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1[valid1] = 1.0 - h[idx1[valid1]]

    idx2 = np.arange(n) + lag
    term2 = np.zeros_like(h)
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2[valid2] = h[idx2[valid2]]

    return term1 - term2


def _project_feasible(v: np.ndarray, total: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set:
        {x | 0 <= x_i <= 1, sum(x) = total}
    Uses a simple bisection water‑filling.
    """
    lo = -np.max(v) - 1.0      # forces all entries to 0 after clipping
    hi = 1.0 - np.min(v) + 1.0  # forces all entries to 1 after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        if x.sum() > total:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction to enforce the sum exactly (helps later)
    diff = total - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
    return x


def _random_feasible(rng: np.random.RandomState, n: int, total: float) -> np.ndarray:
    """Generate a random vector satisfying the constraints."""
    v = rng.random(n)
    v = v * (total / v.sum())
    return _project_feasible(v, total)

    Search for a step function h on [0,2] with ∫h = 1 that minimizes
    the maximal overlap integral C5. Returns (h, C5, n_points).
    """

    # Determine dimension and initial vector
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
    except NameError:

        h0 = np.full(n, 0.5, dtype=float)

    total = n / 2.0                # sum(h) must equal total

    # Ensure the starting point respects the constraints
    if abs(h0.sum() - total) > 1e-12 or (h0 < 0).any() or (h0 > 1).any():
        h0 = _project_feasible(h0, total)

    best_h = h0.copy()
    best_score = _c5_value(best_h)

    start_time = time.time()
    max_iters = 200                # inner gradient steps per start
    step0 = 0.3                    # initial step size (decays)

    # Main stochastic search
    while time.time() - start_time < budget_s - 0.5:

        h = _random_feasible(rng, n, total)
        cur_score = _c5_value(h)
