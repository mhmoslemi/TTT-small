def adjust_initial(h, target):
    diff = target - h.sum()

                        iter_since_improve += 1
                    move_done = True
                break
        if not move_done:
            iter_since_improve += 1

        # ------------------------------------------------------------------
        # Restart when stagnating
        # ------------------------------------------------------------------
        if iter_since_improve >= 250000 or (now - last_improve) > 40.0:
            cur_h = best_h.copy()
            cur_val = best_val
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                hi = cur_h[i]
                hj = cur_h[j]
                dmin = max(-hi, hj - 1.0)
                dmax = min(1.0 - hi, hj)
                if dmin < dmax:
                    delta = rng.uniform(dmin, dmax) * 0.2
                    cur_h[i] = hi + delta
                    cur_h[j] = hj - delta
            cur_val = compute_c5(cur_h, dx)
            iter_since_improve = 0
            last_improve = now

    # ------------------------------------------------------------------
    # Final correction of the integral constraint
    # ------------------------------------------------------------------
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    return best_h.tolist(), float(best_val), int(n)
