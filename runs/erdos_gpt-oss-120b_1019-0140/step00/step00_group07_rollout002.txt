def project_onto_box_simplex(v, target_sum, bound=1.0):
    # Starting with v, project onto feasible set { y: sum(y)=target_sum, 0<=y_i<=bound}
    # Use bisection for tau.
    # Provide tolerance maybe 1e-10
    lo = np.min(v) - bound
    hi = np.max(v)
    for _ in range(50):
        tau = (lo + hi) / 2.0
        y = np.clip(v - tau, 0, bound)
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            return y
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(v - tau, 0, bound)

import time
import math
import numpy as np

def _project_to_box_simplex(v: np.ndarray, target_sum: float, bound: float = 1.0) -> np.ndarray:
    """Project vector v onto {0 ≤ x_i ≤ bound, sum x_i = target_sum}."""
    # Bisection on tau such that sum(clip(v - tau, 0, bound)) = target_sum
    lo = np.min(v) - bound
    hi = np.max(v)
    for _ in range(60):
        tau = (lo + hi) * 0.5
        y = np.clip(v - tau, 0.0, bound)
        s = y.sum()
        if abs(s - target_sum) <= 1e-12:
            return y
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(v - tau, 0.0, bound)

def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return max overlap integral for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine dimension and starting vector
    if 'initial_h_values' in globals():

        h0 = _project_to_box_simplex(h0, target_sum, bound=1.0)
    else:
        n = 80
        target_sum = n / 2.0
        # Random feasible start
        h0 = rng.random(n)
        h0 = _project_to_box_simplex(h0, target_sum, bound=1.0)

    dx = 2.0 / n
    cur_h = h0.copy()
    cur_obj = _c5_value(cur_h, dx)
    best_h = cur_h.copy()
    best_obj = cur_obj

    # Simulated annealing parameters
    init_temp = 0.12
    final_temp = 0.001
    max_delta_initial = 0.2   # max absolute change at start
    iter_cnt = 0
    # Main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.1:
            break

        # Linear cooling schedule
        frac = 1.0 - elapsed / budget_s
        temp = final_temp + (init_temp - final_temp) * frac
        max_delta = max_delta_initial * frac + 0.001

        # Propose a move that preserves sum (via projection)
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)

        delta = (rng.random() - 0.5) * 2.0 * max_delta
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = _project_to_box_simplex(new_h, target_sum, bound=1.0)

        new_obj = _c5_value(new_h, dx)

        # Acceptance rule
