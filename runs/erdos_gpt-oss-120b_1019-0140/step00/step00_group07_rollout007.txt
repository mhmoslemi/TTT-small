    constraints = []
    # link constraints Y[i,j] <= H[i]
    # Y[i,j] <= 1 - H[j]
    # Y[i,j] >= H[i] - H[j]
    # Actually also Y[i,j] >= 0 (auto from bool) and <= 1 (auto).
    # Use cp constraints.
    for i in range(n):
        for j in range(n):
            constraints.append(Y[i, j] <= H[i])
            constraints.append(Y[i, j] <= 1 - H[j])
            constraints.append(Y[i, j] >= H[i] - H[j])
    # weight constraint
    constraints.append(cp.sum(H) == m)
    # shift constraints
    t = cp.Variable()
    # for each shift s
    for s in range(-(n-1), n):
        # compute sum over valid i
        # i index i such that j = i - s in range 0..n-1
        # i ranges max(0,s) to min(n-1, n-1+s)
        if s >= 0:

            i_end = n - 1

            i_start = 0
            i_end = n - 1 + s
        # Actually i_end inclusive. We'll sum for i in range(i_start, i_end+1)

def _solve_erdos_milp(n):
    import cvxpy as cp, numpy as np
    m = n // 2
    H = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    constraints = []
    # sum constraint
    constraints.append(cp.sum(H) == m)
    # link constraints
    for i in range(n):
        for j in range(n):
            constraints.append(Y[i, j] <= H[i])
            constraints.append(Y[i, j] <= 1 - H[j])
            constraints.append(Y[i, j] >= H[i] - H[j])
    # t variable
    t = cp.Variable()
    # shift constraints
    for s in range(-(n-1), n):
        if s >= 0:
            i_start = s
            i_end = n - 1
        else:
            i_start = 0
            i_end = n - 1 + s
        # gather Y[i, i-s] terms
        terms = []
        for i in range(i_start, i_end + 1):
            terms.append(Y[i, i - s])
        overlap = cp.sum(cp.hstack(terms))  # cp.sum returns scalar; for empty? no empties.
        constraints.append(overlap <= t)
    # objective minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    # solve using available solver(s)
    sol_status = None
    sol_h = None
    for solver in [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]:
        try:
            prob.solve(solver=solver, verbose=False)
            sol_status = prob.status
            if sol_status in ["optimal", "optimal_inaccurate", "optimal_exact"]:
                break
        except Exception:
            continue
    if sol_status not in ["optimal", "optimal_inaccurate", "optimal_exact"]:
        return None, None
    # retrieve solution
    if H.value is None:
        return None, None
    h_vals = np.rint(H.value).astype(int)
    # ensure exactly half ones, maybe adjust due to rounding
    # If not exact, adjust using top values:
    current_sum = int(h_vals.sum())
    target = m
    if current_sum != target:
        # adjust by flipping some values based on H.value
        # sort indices by H.value descending for ones, ascending for zeros.
        # Set half largest to 1, rest to 0.
        idx = np.argsort(H.value)[::-1]
        h_vals[:] = 0
        h_vals[idx[:target]] = 1
    return h_vals, None

def _solve_erdos_milp(n):
    import cvxpy as cp
    import numpy as np
    m = n // 2
    H = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    constraints = []
    constraints.append(cp.sum(H) == m)
    # link constraints
    for i in range(n):
        for j in range(n):
            constraints.append(Y[i, j] <= H[i])
            constraints.append(Y[i, j] <= 1 - H[j])
            constraints.append(Y[i, j] >= H[i] - H[j])
    t = cp.Variable()
    for s in range(-(n - 1), n):
        if s >= 0:
            i_start = s
            i_end = n - 1
        else:

            i_end = n - 1 + s

except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper: compute the Erdős overlap bound for a given step function h
# ----------------------------------------------------------------------
def _compute_bound(h):
    """Return C5 bound for binary/real vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# ----------------------------------------------------------------------
# Exact MILP formulation (binary step function) for a modest n
# ----------------------------------------------------------------------
def _solve_erdos_milp(n):
    """
    Solve the binary Erdős minimum‑overlap problem for a given n
    (must be even) via MILP.  Returns a numpy array of 0/1 values or
    None if the solve fails.
    """
    m = n // 2  # required number of ones

    # Decision variables
    H = cp.Variable(n, boolean=True)                # h_i ∈ {0,1}
    Y = cp.Variable((n, n), boolean=True)          # y_ij = h_i·(1‑h_j)
    t = cp.Variable()                               # maximal overlap (continuous)

    constraints = []

    # exactly half the entries are 1
    constraints.append(cp.sum(H) == m)

    # linearisation of y_ij = h_i·(1‑h_j)
    for i in range(n):
        for j in range(n):
            constraints.append(Y[i, j] <= H[i])
            constraints.append(Y[i, j] <= 1 - H[j])
            constraints.append(Y[i, j] >= H[i] - H[j])

    # for every integer shift s, the overlapping count must not exceed t
    for s in range(-(n - 1), n):
        if s >= 0:
            i_start = s
            i_end = n - 1
        else:
            i_start = 0
            i_end = n - 1 + s
        terms = [Y[i, i - s] for i in range(i_start, i_end + 1)]
        overlap = cp.sum(cp.hstack(terms))
        constraints.append(overlap <= t)

    # objective: minimise the maximal overlap
    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few common open‑source solvers
    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, verbose=False)  # type: ignore[arg-type]
            if prob.status in ("optimal", "optimal_inaccurate", "optimal_exact"):
                break
        except Exception:
            continue

    if prob.status not in ("optimal", "optimal_inaccurate", "optimal_exact"):
        return None

    # extract a binary solution for h
    h_val = np.rint(H.value).astype(int)  # rounding to {0,1}
    # safeguard against possible rounding artefacts
    if h_val.sum() != m:
        idx = np.argsort(H.value)[::-1]
        h_val[:] = 0
        h_val[idx[:m]] = 1
    return h_val
