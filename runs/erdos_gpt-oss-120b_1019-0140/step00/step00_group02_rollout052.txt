    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    """Generate a feasible neighbor by moving mass between two positions."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]

    delta = random.uniform(low, high)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

        diff = target_sum - cur_sum
        h0[0] = min(max(h0[0] + diff, 0.0), 1.0)
    np.clip(h0, 0.0, 1.0, out=h0)

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    start_temp = 0.02
    temp = start_temp
    cooling = 0.9995
    last_improve = 0
    it = 0

    curr_h = h0.copy()
    curr_val = compute_overlap(curr_h, dx)
    best_h = curr_h.copy()
    best_val = curr_val

    start_time = time.time()
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.99:
            break

        new_h = propose_neighbor(curr_h)
        if new_h is None:
            continue

        new_val = compute_overlap(new_h, dx)

        # Acceptance test
        if new_val < curr_val:
            accept = True
        else:
            prob = math.exp(-(new_val - curr_val) / max(temp, 1e-12))
            accept = random.random() < prob

        if accept:
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
                last_improve = it

        # Cooling
        temp *= cooling

        # Reheat if stuck for a long time
        if it - last_improve > 200_000 and temp < start_temp * 0.1:
            temp = start_temp

        it += 1

    # ------------------------------------------------------------------
    # Final feasibility repair (tiny numerical drift)
    final_sum = best_h.sum()
    diff = target_sum - final_sum
    if abs(diff) > 1e-12:
        best_h[0] = min(max(best_h[0] + diff, 0.0), 1.0)
    np.clip(best_h, 0.0, 1.0, out=best_h)

    final_bound = compute_overlap(best_h, dx)
    return best_h.tolist(), float(final_bound), n
