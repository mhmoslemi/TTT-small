import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the box [0,1]^n intersected with a given sum.
# ----------------------------------------------------------------------
def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x in [0,1]^n : sum(x)=target} using bisection on λ."""
    # λ such that w = clip(v - λ,0,1) satisfies sum(w)=target.
    low, high = -1.0, 2.0  # wide enough for any feasible target
    for _ in range(50):
        lam = (low + high) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    w = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # tiny adjustment for rounding errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

# Helper: evaluate the Erdős overlap constant for a candidate vector.
# ----------------------------------------------------------------------
def _evaluate(h: np.ndarray) -> float:
    try:
        # Helper provided by the harness (may be more accurate).
        return evaluate_erdos_solution(h)  # type: ignore[name-defined]
    except Exception:
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(corr.max() * dx)


# ----------------------------------------------------------------------
# Helper: gradient of the correlation term that attains the current maximum.
# ----------------------------------------------------------------------
def _gradient_max_corr(h: np.ndarray):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)  # signed shift giving the max value

    if lag >= 0:
        hf = np.concatenate([h[lag:], np.zeros(lag, dtype=h.dtype)])
        hb = np.concatenate([np.zeros(lag, dtype=h.dtype), h[: n - lag]])
    else:
        k = -lag
        hf = np.concatenate([np.zeros(k, dtype=h.dtype), h[: n - k]])
        hb = np.concatenate([h[k:], np.zeros(k, dtype=h.dtype)])

    grad = 1.0 - hf - hb  # d/dh of the (unnormalized) correlation
    max_corr = float(corr[idx] * (2.0 / n))
    return grad, max_corr, lag

# Core optimizer for a fixed resolution.

def _optimize_at_resolution(init_h: np.ndarray, n_target: int,
                            time_limit: float, rng: np.random.Generator):
    h = _project_to_sum_box(init_h, n_target / 2.0)

    cur_h = h.copy()
    cur_score = _evaluate(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    it = 0
    while True:
        if time.time() - start >= time_limit:
            break

        grad, _, _ = _gradient_max_corr(cur_h)

        # Step size decays linearly from ~0.5 to ~0.01 over the allotted time.
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.5 * (1.0 - frac) + 0.01

        new_h = cur_h - step * grad
        new_h = _project_to_sum_box(new_h, n_target / 2.0)

        new_score = _evaluate(new_h)

        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
        else:
            # Simulated‑annealing style acceptance
            T = max(1e-6, 0.01 * (1.0 - frac))
            if np.exp(-(new_score - cur_score) / T) > rng.random():
                cur_h, cur_score = new_h, new_score

        if cur_score < best_score:
            best_h, best_score = cur_h.copy(), cur_score

        # Occasional random swap to escape flat regions.
        it += 1

                pert[i], pert[j] = pert[j], pert[i]

                if pert_score < cur_score:
                    cur_h, cur_score = pert, pert_score
                    if cur_score < best_score:
                        best_h, best_score = cur_h.copy(), cur_score

    return best_h, best_score
