def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def contributions_for_shift(h, k):
    n = h.shape[0]
    contr = np.empty(n)
    if k >= 0:
        overlap_len = n - k
        if overlap_len > 0:
            contr[:overlap_len] = h[:overlap_len] * (1 - h[k:])
        if overlap_len < n:
            contr[overlap_len:] = h[overlap_len:]   # (1 - 0) =1
    else:
        k_abs = -k
        # i < k_abs: h[i] * 1
        contr[:k_abs] = h[:k_abs]
        # overlapping region
        if k_abs < n:
            contr[k_abs:] = h[k_abs:] * (1 - h[:n - k_abs])
        else:
            contr[:] = h[:]
    return contr

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by full correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def contributions_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Contribution of each index i to the overlap value for a given integer shift k."""
    n = h.shape[0]
    contr = np.empty(n, dtype=float)
    if k >= 0:
        overlap = n - k
        if overlap > 0:
            contr[:overlap] = h[:overlap] * (1.0 - h[k:])
        if overlap < n:
            contr[overlap:] = h[overlap:]  # (1 - 0) = 1
    else:
        kabs = -k
        # tail where shifted index is out of range
        contr[:kabs] = h[:kabs]
        if kabs < n:
            contr[kabs:] = h[kabs:] * (1.0 - h[: n - kabs])
        else:
            contr[:] = h[:]
    return contr
