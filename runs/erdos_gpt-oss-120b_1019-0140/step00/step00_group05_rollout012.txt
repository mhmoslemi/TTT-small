    diff = target - total

    # While diff not zero:
    # gather indices where can add (if diff>0) or subtract (if diff<0).
    # We'll add/sub equally up to capacity.
    if diff > 0:

    else:
        idx = np.where(vec > 1e-12)[0]
    if len(idx) == 0:
        # cannot adjust, just return as is
        return vec
    # distribute evenly: add amount = diff / len(idx) limited by capacity

    while abs(diff) > 1e-12 and len(idx) > 0:

    # choose mutation type:
    # 0: pairwise exchange delta.
    # 1: random reset of some entries
    # 2: Gaussian perturbation followed by projection.
    t = rng.random()
    if t < 0.6:
        # pairwise delta moves many times
        num_moves = rng.integers(1, min(5, N//2)+1)
        for _ in range(num_moves):
            i, j = rng.choice(N, size=2, replace=False)
            lower = max(-child[i], child[j] - 1.0)
            upper = min(1.0 - child[i], child[j])
            if lower >= upper:

            delta = rng.uniform(lower, upper)
            child[i] += delta
            child[j] -= delta
    elif t < 0.8:

        fraction = rng.uniform(0.05, 0.2)  # 5%-20% entries
        k = max(1, int(fraction * N))
        indices = rng.choice(N, size=k, replace=False)

    # enforce sum constraint
    child = enforce_sum(child)
    # ensure bounds
    np.clip(child, 0.0, 1.0, out=child)
    return child

import numpy as np
import time

def compute_c5(h):
    """Return the max overlap integral for a discretized step function."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def enforce_sum(vec, target):
    """Clip to [0,1] and adjust to make sum(vec) == target."""
    vec = np.clip(vec, 0.0, 1.0)
    total = np.sum(vec)
    diff = target - total
    if abs(diff) < 1e-12:
        return vec
    N = len(vec)
    # Try to fix by moving a single coordinate
    for i in range(N):
        lo = -vec[i]          # min change to keep >=0
        hi = 1.0 - vec[i]     # max change to keep <=1
        if lo <= diff <= hi:
            vec[i] += diff
            return vec
    # Fallback: scale globally and re‑clip, then adjust one entry again
    if total != 0:
        scale = target / total
        vec *= scale
        np.clip(vec, 0.0, 1.0, out=vec)
        total2 = np.sum(vec)
        diff2 = target - total2
        if abs(diff2) >= 1e-12:
            for i in range(N):
                lo = -vec[i]
                hi = 1.0 - vec[i]
                if lo <= diff2 <= hi:
                    vec[i] += diff2
                    break
    return vec

def mutate_candidate(parent, rng, target):
    """Produce a mutated offspring that respects the constraints."""
