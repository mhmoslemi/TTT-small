def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def optimize_h(init_h, maxiter=200, time_limit=None):
    n = init_h.shape[0]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(lambda x: compute_c5(x), init_h,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        return res.x, compute_c5(res.x)
    else:
        return init_h, compute_c5(init_h)
