def propose_neighbor(h, rng):
    n = len(h)
    # randomly pick two distinct indices i, j
    i, j = rng.choice(n, size=2, replace=False)
    # Determine direction: transfer from i to j
    # Max transfer amount that keeps h[i]>=0 and h[j]<=1
    max_transfer = min(h[i], 1.0 - h[j])
    # Also maybe we allow direction reversed: randomly choose to transfer from i to j or from j to i.
    if rng.random() < 0.5:
        # transfer from i to j as above
        source, target = i, j
    else:
        # transfer from j to i
        source, target = j, i
        max_transfer = min(h[source], 1.0 - h[target])
    if max_transfer <= 0.0:
        # cannot transfer; return original (or try again)
        return h.copy()
    # Choose delta as random fraction of max_transfer
    delta = rng.random() * max_transfer
    new_h = h.copy()
    new_h[source] -= delta
    new_h[target] += delta
    # Ensure within bounds (should be)
    return new_h

def project_bounds_sum(h, target):
    # Ensure 0 <= h <= 1 and sum = target.
    # Use simple iterative scheme:
    # first clip.
    h = np.clip(h, 0.0, 1.0)
    # then adjust sum by scaling the non-bound elements.
    # While sum not close to target:
    for _ in range(10):
        s = h.sum()
        if abs(s - target) < 1e-9:
            break
        diff = target - s
        # find indices not at bound (0< h <1)
        free = (h > 0.0) & (h < 1.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:  # No free var, adjust arbitrarily by small epsilon to satisfy sum (unlikely)
            # Distribute diff uniformly by relaxing bounds a bit
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            # distribute diff equally among free entries
            inc = diff / free_idx.size
            h[free_idx] += inc
            # clip again
            h = np.clip(h, 0.0, 1.0)
    # final clip
    return h

import time
import math
import numpy as np

def _project_bounds_sum(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum == target_sum."""
    x = np.clip(x, 0.0, 1.0)
    # iterative redistribution of the excess/deficit among non‑boundary entries
    for _ in range(20):
        s = x.sum()
        if abs(s - target_sum) <= 1e-12:
            break
        diff = target_sum - s
        free = (x > 0.0) & (x < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            # No free variable – spread uniformly (may violate bounds, will be clipped next)
            inc = diff / x.size
            x += inc
        else:
            inc = diff / idx.size
            x[idx] += inc
        x = np.clip(x, 0.0, 1.0)
    return x

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap value C5 for a given vector h."""
    n = h.size
    # If the harness provides evaluate_erdos_solution, use it for consistency.
    try:
        return float(evaluate_erdos_solution(h))
    except NameError:
        # fallback to the direct definition
        return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that yields a low C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initialise h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 200  # fallback size when no parent is supplied
        h = rng.random(n)
        target = n / 2.0
        h = h / h.sum() * target
    target_sum = n / 2.0
    h = _project_bounds_sum(h, target_sum)

    # ----------------------------------------------------------------------
    # Evaluation helpers
    # ----------------------------------------------------------------------
    def _corr_vector(vec):
        return np.correlate(vec, 1 - vec, mode='full')

    # Initial best
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Search parameters
    # ----------------------------------------------------------------------
    start_time = time.time()
    lr = 0.12                     # initial learning rate
    lr_decay = 0.99995
    no_improve = 0
    max_no_improve = 5000        # trigger jitter
    iteration = 0

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break

        # --------------------------------------------------------------
        # Identify worst shift (maximal overlap)
        # --------------------------------------------------------------
        corr = _corr_vector(h)
        idx_max = int(np.argmax(corr))
        cur_val = corr[idx_max] * (2.0 / n)

        # Track best solution
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Gradient step for the worst shift
        # --------------------------------------------------------------
        shift = idx_max - (n - 1)          # shift can be negative
        grad = np.zeros_like(h)

        if shift >= 0:
            # term1: (1 - h[i+shift]) for i where i+shift < n
            i1 = np.arange(0, n - shift)
            grad[i1] += 1.0 - h[i1 + shift]
            # term2: -h[i-shift] for i where i-shift >= 0
            i2 = np.arange(shift, n)
            grad[i2] -= h[i2 - shift]
        else:
            s = -shift
            # term1 for negative shift: (1 - h[i-s]) where i >= s
            i1 = np.arange(s, n)
            grad[i1] += 1.0 - h[i1 - s]
            # term2: -h[i+s] where i+s < n
            i2 = np.arange(0, n - s)
            grad[i2] -= h[i2 + s]

        # Gradient descent step
        h_new = h - lr * grad
        h_new = _project_bounds_sum(np.clip(h_new, 0.0, 1.0), target_sum)

        # --------------------------------------------------------------
        # Acceptance (deterministic for improvement, otherwise keep current)
        # --------------------------------------------------------------
        # Since we are minimizing max overlap, we simply move to the new point.
        # The projection already respects constraints, and exploration is
        # provided by occasional jitter.
        h = h_new

        # --------------------------------------------------------------
        # Decay learning rate
        # --------------------------------------------------------------
        lr *= lr_decay

        # --------------------------------------------------------------
        # Stagnation handling: occasional random jitter
        # --------------------------------------------------------------
        if no_improve > max_no_improve:
            # Apply a small random perturbation while preserving the sum.
            k = max(1, n // 20)               # number of coordinates to touch
            idxs = rng.choice(n, size=k, replace=False)
            delta = (rng.random(k) - 0.5) * 0.2   # [-0.1, 0.1]
            h_temp = h.copy()
            h_temp[idxs] += delta
            h_temp = _project_bounds_sum(np.clip(h_temp, 0.0, 1.0), target_sum)
            h = h_temp
            no_improve = 0
            # Reset learning rate modestly to keep movement alive
            lr = max(lr, 0.02)

        iteration += 1

    # ----------------------------------------------------------------------
    # Final tidy‑up of the best solution
    # ----------------------------------------------------------------------
    best_h = _project_bounds_sum(best_h, target_sum)

    # Ensure the integral constraint holds to machine precision
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # Repair by distributing the tiny residual to the first non‑boundary entry
        for i in range(best_h.size):
            cap = (1.0 - best_h[i]) if diff > 0 else best_h[i]
            if cap >= abs(diff) - 1e-12:
                best_h[i] += diff
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
