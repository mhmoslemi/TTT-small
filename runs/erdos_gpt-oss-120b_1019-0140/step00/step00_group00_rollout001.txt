def project_to_bounds_and_sum(v, target_sum, max_iter=100):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # Determine direction: need to increase or decrease.
        if diff > 0:
            # increase values not at 1
            free = v < 1 - 1e-12
        else:
            free = v > 1e-12
        free_count = np.count_nonzero(free)
        if free_count == 0:
            # cannot adjust further; break
            break
        delta = diff / free_count
        # limit delta to keep within bounds
        # compute new v for free indices: add delta, then clip again.
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final correction by scaling if still off (rare)
    s = v.sum()
    if abs(s - target_sum) > 1e-8:
        # scaling factor
        factor = target_sum / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)

    return v

def project_to_bounds_and_sum(v, target_sum, max_iter=100):
    # Ensure bounds
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_to_bounds_and_sum(v: np.ndarray, target_sum: float, max_iter: int = 100) -> np.ndarray:
    """Project v onto the box [0,1] while forcing sum(v)=target_sum (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        v[mask] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target_sum) > 1e-8 and s > 0:
        v = np.clip(v * (target_sum / s), 0.0, 1.0)
    return v

def local_search(h_init: np.ndarray, rng: np.random.Generator, time_deadline: float):
    """Simple hill‑climbing / simulated‑annealing on the max‑overlap objective."""
    n = h_init.shape[0]
    cur_h = h_init.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    total_budget = max(time_deadline - start, 1e-12)

    while True:
        now = time.time()
        if now >= time_deadline:
            break

        # propose a mass transfer between two coordinates
        i, j = rng.choice(n, 2, replace=False)
        if rng.random() < 0.5:
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps
            new_h = cur_h.copy()
            new_h[i] += eps
            new_h[j] -= eps

        new_c5 = compute_c5(new_h)

        if new_c5 <= cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            # occasional acceptance according to a linear cooling schedule
            elapsed = now - start
            frac = elapsed / total_budget
            temp = max(1e-4, (0.2 * best_c5) * (1.0 - frac))
            if rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
                cur_h = new_h
                cur_c5 = new_c5

    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s

    # ------------------------------------------------------------------
    # 1. Obtain an initial feasible vector (use supplied parent if present)
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = project_to_bounds_and_sum(h0, n0 / 2.0)
    except NameError:
        n0 = 45
        h0 = np.full(n0, 0.5)
        h0 = project_to_bounds_and_sum(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Multi‑resolution refinement (45 → 90 → 180 points)
    # ------------------------------------------------------------------
    resolutions = [n0, 90, 180]
    time_per_stage = budget_s / len(resolutions)

    for n in resolutions:
        if time.time() >= end:
            break
        stage_deadline = min(end, time.time() + time_per_stage)

        if best_h.shape[0] != n:
            old_n = best_h.shape[0]
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h_start = np.interp(new_x, old_x, best_h)
            h_start = project_to_bounds_and_sum(h_start, n / 2.0)
        else:
            h_start = best_h.copy()

        stage_h, stage_c5 = local_search(h_start, rng, stage_deadline)

        if stage_c5 < best_c5:
            best_c5 = stage_c5
            best_h = stage_h.copy()

    # Final sanity projection and evaluation
    best_h = project_to_bounds_and_sum(best_h, best_h.shape[0] / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(best_h.shape[0])
