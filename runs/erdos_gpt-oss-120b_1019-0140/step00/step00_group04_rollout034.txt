import numpy as np
import time
import math

def _project_to_sum_bounds(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=1000):
    """Euclidean projection of v onto {x: sum(x)=target, lower<=x<=upper}."""
    x = np.clip(v, lower, upper)
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = np.where((x > lower + tol) & (x < upper - tol))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, lower, upper)
    # final tiny correction if needed
    residual = target - x.sum()
    if abs(residual) > tol:
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if free.size:
            x[free[0]] += residual
            x = np.clip(x, lower, upper)
    return x

def _random_feasible(n, rng):
    """Generate a random feasible vector of length n (values in [0,1], sum = n/2)."""
    v = rng.random(n)
    return _project_to_sum_bounds(v, n / 2.0)

def _compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx  (discretized)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_time = budget_s * 0.9  # reserve a safety margin

    # --- choose discretisation size ---
    n = 300                     # even number < 1000; gives reasonable resolution
    target_sum = n / 2.0

    # --- initialise h, using provided construction if available ---
    try:
        init_vals = np.asarray(initial_h_values, dtype=float)
        have_initial = True
    except NameError:
        have_initial = False

    if have_initial:
        m = init_vals.size
        if m == n:
            h = init_vals.copy()
        else:
            # linear up‑sampling / interpolation
            x_old = np.linspace(0.0, 2.0, m, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init_vals)
    else:
        h = rng.random(n)

    h = _project_to_sum_bounds(h, target_sum)

    # best solution so far
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # current working solution
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # annealing / hill‑climbing parameters
    base_max_step = 0.08   # maximal mass moved in a single swap
    T0 = 0.06              # initial temperature
    no_improve = 0
    restart_interval = 6000  # steps without improvement before a restart

    it = 0
    while time.time() - start_time < total_time:
        # linearly decreasing step size and temperature
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        max_step = base_max_step * (1.0 - frac)
        T = T0 * (1.0 - frac)

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # permissible delta (move mass from i to j)
        # positive delta means decreasing i, increasing j
        max_delta_pos = min(cur_h[i], 1.0 - cur_h[j], max_step)
        max_delta_neg = min(1.0 - cur_h[i], cur_h[j], max_step)
        if max_delta_pos + max_delta_neg <= 0.0:

        delta = rng.uniform(-max_delta_neg, max_delta_pos)

        new_h[i] -= delta
        new_h[j] += delta

        new_h[i] = max(0.0, min(1.0, new_h[i]))
        new_h[j] = max(0.0, min(1.0, new_h[j]))

            if T > 1e-12:
