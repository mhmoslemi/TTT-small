import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_move(h, step_scale):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # direction: from i to j
    bound = min(h[i], 1 - h[j])
    if bound <= 0:
        # try opposite direction?
        bound = min(h[j], 1 - h[i])
        if bound <= 0:
            return None, None, 0
        # swap i and j
        i, j = j, i
    eps = bound * step_scale * np.random.rand()
    # If eps is zero, no move
    if eps <= 0:
        return None, None, 0
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h, (i, j, eps)

def worst_shift(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # compute max index
    max_idx = np.argmax(corr)
    # shift of that index
    shift = max_idx - (n - 1)
    return shift, corr[max_idx] * dx

            forbidden = {src, paired[chosen]}
            donor_pool = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
            if donor_pool.size == 0:
                continue
            donor = np.random.choice(donor_pool)
            bound = min(cur_h[donor], 1.0 - cur_h[src])
            if bound <= 0:
                continue
            eps = bound * step_scale * np.random.rand()
            if eps <= 0:
                continue
            new_h = cur_h.copy()
            new_h[donor] -= eps
            new_h[src] += eps

        # ------------------------------------------------------------------
        # evaluate and possibly accept
        new_score = _compute_c5(new_h)
        delta = new_score - cur_score
        if delta <= 0 or np.random.rand() < math.exp(-delta / temperature):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

    # ------------------------------------------------------------------
    # final correction of the integral constraint (tiny numerical drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # try to fix a single coordinate without breaking the box constraints
        for i in range(n):
            if 0.0 <= best_h[i] + diff <= 1.0:
                best_h[i] += diff
                break
        else:
            # fall back to a uniform correction
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)

    final_score = _compute_c5(best_h)
    return best_h, final_score, n
