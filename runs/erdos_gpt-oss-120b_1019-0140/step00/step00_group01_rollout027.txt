def hill_climb_binary(n, time_limit):
    target_ones = n // 2
    # initial random binary vector
    h = np.zeros(n, dtype=np.int8)
    h[:target_ones] = 1
    np.random.shuffle(h)
    best_h = h.copy()
    best_score = compute_c5_int(best_h)
    start = time.time()
    iteration = 0
    while time.time() - start < time_limit:
        # random swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_int(h)
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            # optional early stop if score below target
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        iteration += 1
        # optionally schedule restarts based on time or no improvement.
    return best_h.astype(float), best_score

def simulated_annealing(n, time_limit):
    # start random binary
    h = np.zeros(n, dtype=np.int8)
    h[:n//2] = 1
    np.random.shuffle(h)
    best_h = h.copy()
    best_score = compute_c5_int(best_h)
    current_score = best_score
    T0 = 1.0
    # decaying temperature based on iteration count
    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5_int(h)
        delta = new_score - current_score
        # accept if improvement or probability
        # compute temperature schedule: T = T0 * (1 - iter / max_iter)
        # we don't have max_iter predetermined; use exp decay: T = T0 * exp(-iter / 10000)
        T = T0 * np.exp(-iter / 10000)
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-8)):
            # accept
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                # optional early stop if below threshold
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        iter += 1
        # optional restart if temperature low and no improvement for long
    return best_h.astype(float), best_score

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C₅ = max_k ∫ h(x)(1‑h(x+k)) dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation; np.correlate works with float inputs
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def generate_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary feasible vector (plus a 0.5 entry when n is odd)."""
    target_ones = n // 2
    h = np.zeros(n, dtype=np.float64)
    perm = rng.permutation(n)
    h[perm[:target_ones]] = 1.0
    if n % 2 == 1:
        half_idx = perm[target_ones]
        h[half_idx] = 0.5
    return h

def make_initial_feasible(init: np.ndarray) -> np.ndarray:
    """Clip and adjust a provided vector to satisfy ∑h = n/2."""
    h = np.clip(init.astype(np.float64), 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # try to fix by distributing the discrepancy over a non‑boundary entry
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            i = free[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            # fallback: adjust the first element (tiny numerical error tolerated)
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def simulated_annealing(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simple SA that swaps two entries while keeping the sum invariant."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_score = compute_c5(best_h)

    cur_h = start_h.copy()
    cur_score = best_score

    start = time.time()
    it = 0
    stagnation = 0
    max_stagnation = 20000
    target_bound = 0.3808

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget or best_score <= target_bound:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:

        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score

        if accept:

            if new_score + 1e-12 < best_score:

                best_h = cur_h.copy()
                stagnation = 0

            cur_h = generate_feasible_vector(n, rng)
            cur_score = compute_c5(cur_h)
            stagnation = 0

    rng = np.random.default_rng(seed)

    # reference the optional parent construction
    has_initial = False
    try:
        _ = initial_h_values
        has_initial = True
    except NameError:
        has_initial = False

    # decide problem size
    if has_initial:
        n = len(initial_h_values)
        # if the provided length is odd we keep it (Legendre‑type construction is typical)
        start_h = make_initial_feasible(initial_h_values)
    else:
        n = 200                     # even, easy to satisfy the sum constraint
        start_h = generate_feasible_vector(n, rng)

    # allocate time for the optimizer (leave a tiny safety margin)
    start_all = time.time()
    remaining = budget_s - (time.time() - start_all) - 0.05
    if remaining < 0.0:
        remaining = 0.0

    best_h, best_score = simulated_annealing(start_h, remaining, rng)

    # final safety checks: clipping and exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        diff = target - cur_sum
        free = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if free.size:
            best_h[free[0]] = np.clip(best_h[free[0]] + diff, 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    c5_bound = compute_c5(best_h)
    return best_h, c5_bound, n
