def _project_capsimplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=target_sum}."""
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        interior = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if interior.size:
            x[interior] += diff / interior.size
            x = np.clip(x, 0.0, 1.0)
    return x
