def anneal_h(initial_h, max_iter, time_limit, start_temp, end_temp, step_scale=0.01):

   best_h = initial_h.copy()
   best_val = compute_c5(best_h)
   cur_h = best_h.copy()
   cur_val = best_val

   n = len(initial_h)
   # Determine list of indices where h near 0 or 1 to avoid delta beyond bounds.
   for it in range(max_iter):
       # time check break if time exceeded

       # Random propose:
       i, j = rng.choice(n, 2, replace=False)
       # Determine max delta possible moving from i to j
       max_delta = min(cur_h[i], 1 - cur_h[j])
       # if max_delta is very small, skip maybe choose other direction
       if max_delta < 1e-12:
           # try opposite direction
           max_delta = min(cur_h[j], 1 - cur_h[i])
           if max_delta < 1e-12:
               continue
           # swap roles
           delta = step_scale * max_delta * rng.random()
           new_h = cur_h.copy()
           new_h[j] -= delta
           new_h[i] += delta
       else:
           delta = step_scale * max_delta * rng.random()
           new_h = cur_h.copy()
           new_h[i] -= delta
           new_h[j] += delta

       # compute new value
       new_val = compute_c5(new_h)

       # accept ? temperature schedule
       t = start_temp + (end_temp - start_temp) * (it / max_iter)
       # decreasing linear from high to low: start_temp > end_temp
       # accept if new_val < cur_val or random < exp((cur_val - new_val)/t)
       if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(t, 1e-12)):
           cur_h = new_h
           cur_val = new_val
           if new_val < best_val:
               best_h = new_h
               best_val = new_val

   return best_h, best_val

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = len(h) / 2.0
    if total != 0:
        factor = target / total
        h *= factor
    else:
        h.fill(target/len(h))
    # clip again after scaling
    h = np.clip(h, 0.0, 1.0)
    # ensure sum is target within tolerance; if not, adjust by distributing diff across indices where not at limit
    diff = h.sum() - target
    if abs(diff) > 1e-10:
        # adjust uniformly small amount
        # find indices not at bound
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if len(free) > 0:
            h[free] -= diff / len(free)
            h = np.clip(h, 0, 1)
    return h

import numpy as np, time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, n=2*n-1)
    OH = np.fft.rfft(1.0 - h, n=2*n-1)
    corr = np.fft.irfft(H * np.conj(OH), n=2*n-1)
    return np.max(corr) * dx

def repair(h):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    if s == 0:
        return np.full_like(h, target / len(h))
    factor = target / s
    h = h * factor
    h = np.clip(h, 0.0, 1.0)
    # final adjust diff if any
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if len(free) > 0:
            h[free] -= diff / len(free)
        else:
            # fallback: scale again
            h = h - diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_initial(h):
    n = len(h)
    k = n // 2
    # get indices of top k values
    idx = np.argpartition(-h, k-1)[:k]
    binary = np.zeros(n, dtype=np.float64)
    binary[idx] = 1.0
    return binary

def anneal_binary(initial_h, rng, time_limit):
    # returns best_h, best_val
    n = len(initial_h)
    cur_h = initial_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # parameters
    max_temp = 0.02
    min_temp = 0.001
    # compute total steps as function of time
    # We'll run as many steps as possible until time expires
    it = 0
    # We'll compute temperature schedule based on elapsed_time fraction
    while time.time() - start < time_limit:
        it += 1
        # temperature linear schedule
        fraction = (time.time() - start) / time_limit
        temp = max_temp * (1 - fraction) + min_temp * fraction
        # pick random i where h=1
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = compute_c5(new_h)
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(temp, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
    return best_h, best_val

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using FFT correlation."""
    n = h.size
    dx = 2.0 / n
    N = 2 * n - 1                     # length of linear correlation
    H = np.fft.rfft(h, n=N)
    OH = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(OH), n=N)
    return float(np.max(corr) * dx)


def repair(vec: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(vec)=len(vec)/2."""
    v = np.clip(vec, 0.0, 1.0)
    target = v.size / 2.0
    s = v.sum()
    if s > 0:
        factor = target / s
        v *= factor
    else:
        v.fill(target / v.size)
    v = np.clip(v, 0.0, 1.0)

    # Small residual correction if needed
    diff = v.sum() - target
    if abs(diff) > 1e-12:
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size:
            v[free] -= diff / free.size
            v = np.clip(v, 0.0, 1.0)
        else:
            # fallback uniform shift
            v -= diff / v.size
            v = np.clip(v, 0.0, 1.0)
    return v


def interpolate_to(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of arr (defined on [0,2]) to new_n points."""
    old_n = arr.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def anneal_continuous(init_h: np.ndarray,
                      rng: np.random.Generator,
                      time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on a continuous step function.
    Moves are transfers of a random amount between two coordinates,
    preserving the total integral.
    """

    cur = init_h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    max_temp = 0.02
    min_temp = 0.001
    step_scale = 0.5

    while time.time() - start < time_limit:
        # temperature schedule (linear)
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # propose a transfer between two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # direction: move from i to j if possible, else opposite
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0.0:
            max_delta = min(cur[j], 1.0 - cur[i])
            if max_delta <= 0.0:
                continue
            # opposite direction
            delta = step_scale * max_delta * rng.random()
            new = cur.copy()
            new[j] -= delta
            new[i] += delta
        else:
            delta = step_scale * max_delta * rng.random()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        new_val = compute_c5(new)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(temp, 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val

    # final repair to enforce exact sum
    best = repair(best)
    best_val = compute_c5(best)
    return best, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # -----------------------------------------------------------------
    # Base vector (provided by the harness if available)
    # -----------------------------------------------------------------
    try:
        base_vec = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: uniform 0.5 on a modest size
        base_vec = np.full(100, 0.5, dtype=np.float64)

    base_vec = repair(base_vec)
    n0 = base_vec.size

    # -----------------------------------------------------------------
    # Resolutions to explore (coarse -> finer)
    # -----------------------------------------------------------------
    resolutions = [n0]
    if budget_s > 300:
        resolutions.append(min(2000, n0 * 2))
    if budget_s > 800:
        resolutions.append(min(2000, n0 * 4))

    best_overall_h = None
    best_overall_val = float("inf")
    remaining_budget = budget_s

    for idx, n in enumerate(resolutions):
        # allocate a share of the remaining time for this resolution
        remaining_res = len(resolutions) - idx
        time_for_this = remaining_budget / remaining_res * 0.95  # small safety margin

        # generate initial candidates for this resolution
        candidates = []

        if n == n0:
            candidates.append(base_vec.copy())
        else:
            interp = interpolate_to(base_vec, n)
            candidates.append(repair(interp))

        # also try a fresh random start
        random_vec = rng.random(n, dtype=np.float64)
        candidates.append(repair(random_vec))

        # run annealing on each candidate (splitting time evenly)
        per_run_time = time_for_this / len(candidates)
        for cand in candidates:
            h_opt, val_opt = anneal_continuous(cand, rng, per_run_time)
            if val_opt < best_overall_val:
                best_overall_val = val_opt
                best_overall_h = h_opt.copy()

        elapsed = time.time() - start_total
        remaining_budget = budget_s - elapsed
        if remaining_budget <= 0.0:
            break

    # fallback if nothing succeeded
    if best_overall_h is None:
        best_overall_h = base_vec.copy()
        best_overall_val = compute_c5(best_overall_h)

    # final repair and bound computation
    best_overall_h = repair(best_overall_h)
    final_c5 = compute_c5(best_overall_h)

    return best_overall_h.astype(np.float64), float(final_c5), best_overall_h.size
