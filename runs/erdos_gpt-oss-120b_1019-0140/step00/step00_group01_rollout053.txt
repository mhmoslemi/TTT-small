import numpy as np
import time
import math

def compute_c5(h):
    """Return the overlap bound C5 for a step function described by vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full auto‑correlation (linear, zero‑padded)
    auto = np.correlate(h, h, mode='full')
    min_auto = auto.min()
    # target integral of h is exactly n/2 because ∫h = 1 and dx = 2/n
    return ((n / 2.0) - min_auto) * dx

def _ensure_feasible(vec, target):
    """Scale/clip a vector so that the sum equals target and values stay in [0,1]."""
    cur = vec.sum()
    if abs(cur - target) > 1e-12:
        # simple rescaling
        vec = vec * (target / cur)
        vec = np.clip(vec, 0.0, 1.0)
        cur = vec.sum()
        if abs(cur - target) > 1e-12:
            diff = target - cur
            # adjust a random position that can absorb the difference
            idx = np.random.randint(0, vec.size)
            vec[idx] = np.clip(vec[idx] + diff, 0.0, 1.0)
    return vec

        h0[:n_one] = 1.0
        if target_sum - n_one > 1e-12:
            h0[n_one] = target_sum - n_one
        rng.shuffle(h0)

    n = h0.shape[0]
    target_sum = n / 2.0
    cur_h = _ensure_feasible(h0.copy(), target_sum)

    # ------------------------------------------------------------------
    # Simulated annealing
    # ------------------------------------------------------------------
    best_h = cur_h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5

    T0 = 0.05                     # initial temperature
    start = time.time()
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        # occasional re‑projection to avoid floating‑point drift
        if it % 5000 == 0:
            cur_h = _ensure_feasible(cur_h, target_sum)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:

        # choose direction of transfer
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        donor_val = cur_h[donor]
        receiver_val = cur_h[receiver]

        # maximum admissible transfer while staying inside [0,1]
        delta_max = min(donor_val, 1.0 - receiver_val)
        if delta_max <= 0.0:

        scale = max(0.01, T / T0)      # shrink steps as temperature falls
