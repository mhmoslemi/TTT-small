def propose_neighbor(h):
    h_new = h.copy()
    i, j = rng.choice(len(h), size=2, replace=False)
    # compute max amount we can move from i to j while staying in bounds:
    # if we want to move from i (decrease) to j (increase)
    # we can randomly decide direction: either shift from i to j or j to i.
    # Choose direction random: sign = 1 for i->j else -1.
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    # max remo from src: h_new[src] (cannot become negative)
    max_rem = h_new[src]
    # max add to dst: 1 - h_new[dst]
    max_add = 1.0 - h_new[dst]
    max_move = min(max_rem, max_add)
    if max_move <= 0:
        return h_new # no feasible move
    epsilon = rng.random() * max_move  # uniform in [0, max_move)
    h_new[src] -= epsilon
    h_new[dst] += epsilon
    return h_new
```

def project_box_sum(h, lower=0.0, upper=1.0, target_sum):
    h = np.clip(h, lower, upper)
    cur = h.sum()
    diff = target_sum - cur
    if np.abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to add mass: distribute proportionally to slack (upper - h_i)
        slack = upper - h
        total_slack = slack.sum()
        if total_slack < 1e-12:
            return h  # can't adjust
        factor = diff / total_slack
        h += slack * factor
    else:
        # need to remove mass: distribute proportionally to current values
        down = h - lower
        total_down = down.sum()
        if total_down < 1e-12:
            return h
        factor = -diff / total_down
        h -= down * factor

    h = np.clip(h, lower, upper)
    # Might need small second adjustment if sum off: recursion or simple adjust small diff
    # We'll ignore small error as it's within tolerance.
    return h

def project_feasible(h, target):
    # hide such that sum is target and within [0,1]
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Use slack or down adjustments
    if diff > 0:
        slack = 1.0 - h
        total_slack = slack.sum()
        if total_slack == 0:
            return h
        factor = diff / total_slack
        h += slack * factor
    else:
        down = h
        total_down = down.sum()
        if total_down == 0:
            return h
        factor = -diff / total_down
        h -= down * factor
    # Clip again to correct numerical overshoot
    h = np.clip(h, 0.0, 1.0)
    # Recompute sum; if still off due to numeric, adjust small difference by adding to first element within bounds.
    cur2 = h.sum()
    diff2 = target - cur2
    if abs(diff2) > 1e-12:
        # adjust first element (or any) within bound
        for i in range(len(h)):
            avail = 1.0 - h[i] if diff2 > 0 else h[i]
            adjust = np.sign(diff2) * min(abs(diff2), avail)
            h[i] += adjust
            diff2 -= adjust
            if abs(diff2) < 1e-12:
                break
    return h
```

def neighbor_targeted(h, rng):
    n = len(h)
    # compute product correlation
    corr = np.correlate(h, h, mode='full')
    w = np.argmin(corr)
    shift = w - (n - 1)
    # Choose valid i range
    if shift >= 0:
        i_min = 0
        i_max = n - shift - 1
    else:
        i_min = -shift
        i_max = n - 1
    if i_max < i_min:
        return h
    i = rng.integers(i_min, i_max + 1)
    j = i + shift
    # find two donor indices not i,j
    available = np.where((np.arange(n) != i) & (np.arange(n) != j))[0]
    if len(available) < 2:
        return h
    donor1, donor2 = rng.choice(available, size=2, replace=False)
    # compute possible delta
    max_inc_i = 1.0 - h[i]
    max_inc_j = 1.0 - h[j]
    max_dec1 = h[donor1]
    max_dec2 = h[donor2]
    max_delta = min(max_inc_i, max_inc_j, max_dec1, max_dec2)
    if max_delta <= 0:
        return h
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] += delta
    h_new[donor1] -= delta
    h_new[donor2] -= delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    except NameError:
        n = 45
        init_h = None
    target_sum = n / 2.0
    dx = 2.0 / n
    if init_h is not None:
        # Ensure feasibility
        current = project_feasible(init_h.copy(), target_sum)
    else:
        current = random_feasible(n, rng)
    cur_cost = compute_c5(current, dx)
    best = current.copy()
    best_cost = cur_cost
    # initial parameters
    temp = 0.1
    cooling = 0.9995
    no_improve = 0
    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        # generate neighbor
        if rng.random() < 0.7:
            new_h = neighbor_move_random(current, rng)
        else:
            new_h = neighbor_move_targeted(current, rng)
        # enforce feasibility
        new_h = project_feasible(new_h, target_sum)
        new_cost = compute_c5(new_h, dx)
        # acceptance
        if new_cost < cur_cost or rng.random() < math.exp(-(new_cost - cur_cost) / (temp + 1e-12)):
            current = new_h
            cur_cost = new_cost
        # update best
        if new_cost < best_cost:
            best = new_h.copy()
            best_cost = new_cost
            no_improve = 0
        else:
            no_improve += 1
        # cool down temperature
        temp = max(temp * cooling, 1e-8)
        # restart if stuck
        if no_improve > 5000:
            current = random_feasible(n, rng)
            cur_cost = compute_c5(current, dx)
            no_improve = 0
    # final projection to ensure constraints
    best = project_feasible(best, target_sum)
    best_cost = compute_c5(best, dx)
    return best.tolist(), best_cost, n

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            total_slack = slack.sum()
            if total_slack > 0:
                factor = diff / total_slack
                h += slack * factor
        else:
            down = h
            total_down = down.sum()
            if total_down > 0:
                factor = -diff / total_down
                h -= down * factor
        h = np.clip(h, 0.0, 1.0)
        # final correction for tiny diff
        cur2 = h.sum()
        diff2 = target - cur2
        if abs(diff2) > 1e-12:
            # adjust first entry (or many) within bounds
            # find index where we can add or subtract diff2
            for idx in range(len(h)):
                if diff2 > 0:
                    add = min(diff2, 1.0 - h[idx])
                else:
                    add = -min(-diff2, h[idx])
                if add != 0.0:
                    h[idx] += add
                    diff2 -= add
                    if abs(diff2) < 1e-12:
                        break
    return h

import numpy as np
import time
import math

def compute_c5(h, dx):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_feasible(h, target):
    """Clamp to [0,1] and adjust to meet exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            tot = slack.sum()
            if tot > 0:
                factor = diff / tot
                h += slack * factor
        else:
            down = h
            tot = down.sum()
            if tot > 0:
                factor = -diff / tot
                h -= down * factor
        h = np.clip(h, 0.0, 1.0)
        # tiny residual correction
        cur2 = h.sum()
        diff2 = target - cur2
        if abs(diff2) > 1e-12:
            for i in range(len(h)):
                if diff2 > 0:
                    add = min(diff2, 1.0 - h[i])
                else:
                    add = -min(-diff2, h[i])
                if add != 0.0:
                    h[i] += add
                    diff2 -= add
                    if abs(diff2) < 1e-12:
                        break
    return h

def neighbor_random(h, rng):
    """Move a random amount of mass from one index to another."""
    n = h.shape[0]
    a, b = rng.choice(n, size=2, replace=False)
    if rng.random() < 0.5:
        src, dst = a, b
    else:
        src, dst = b, a
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 0.0:
        return h.copy()
    delta = rng.random() * max_move
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

def neighbor_targeted(h, rng):
    """Increase overlap for the shift with the smallest correlation."""
    n = h.shape[0]
    corr = np.correlate(h, h, mode="full")
    w = np.argmin(corr)
    shift = w - (n - 1)  # signed shift
    if shift >= 0:
        i_min, i_max = 0, n - shift - 1
    else:
        i_min, i_max = -shift, n - 1
    if i_max < i_min:
        return h.copy()
    i = rng.integers(i_min, i_max + 1)
    j = i + shift
    # choose two donors distinct from i and j
    all_idx = np.arange(n)
    mask = (all_idx != i) & (all_idx != j)
    cand = all_idx[mask]
    if cand.size < 2:
        return h.copy()
    donor1, donor2 = rng.choice(cand, size=2, replace=False)
    max_inc_i = 1.0 - h[i]
    max_inc_j = 1.0 - h[j]
    max_dec1 = h[donor1]
    max_dec2 = h[donor2]
    max_delta = min(max_inc_i, max_inc_j, max_dec1, max_dec2)
    if max_delta <= 0.0:
        return h.copy()
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] += delta
    h_new[donor1] -= delta
    h_new[donor2] -= delta
    return h_new

def random_feasible(n, rng):
    """Generate a random feasible vector (sum=n/2, entries in [0,1])."""
    if rng.random() < 0.5:
        # binary‑like construction
        ones = int(np.floor(n / 2))
        frac = (n / 2.0) - ones

            new_h = neighbor_targeted(current, rng)

        new_h = project_feasible(new_h, target_sum)
        new_cost = compute_c5(new_h, dx)

        # acceptance step (simulated annealing)
        if new_cost < cur_cost or rng.random() < math.exp(-(new_cost - cur_cost) / (temp + 1e-12)):
            current = new_h
            cur_cost = new_cost

        # update global best
        if new_cost < best_cost:
            best = new_h.copy()
            best_cost = new_cost
            no_improve = 0
            if best_cost <= target_bound:
                break
        else:
            no_improve += 1

        # temperature decay
        temp = max(temp * cooling, 1e-12)

        # restart if stuck for a long time
        if no_improve >= 20000:
            current = random_feasible(n, rng)
            cur_cost = compute_c5(current, dx)
            no_improve = 0

    # final projection and evaluation
    best = project_feasible(best, target_sum)
    best_cost = compute_c5(best, dx)
    return best.tolist(), float(best_cost), n
