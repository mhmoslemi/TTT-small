def gradient_max_corr(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        # for indices j where j+shift < n
        grad[:n-shift] += (1 - h[shift:])   # (1 - h[j+shift]) for j from 0 to n-shift-1
        # for indices j where j-shift >=0 => j>=shift
        grad[shift:] -= h[:n-shift]        # -h[j-shift] for j from shift to n-1
    else:
        s = -shift
        # shift negative: shift = -s -> -s
        # j+shift = j - s
        # For indices j where j+shift >=0 => j >= s
        grad[s:] += (1 - h[:n-s])   # (1 - h[j+shift]) = (1 - h[j-s]) for j >= s
        # For indices j where j - shift = j + s < n => j < n - s
        grad[:n-s] -= h[s:]         # -h[j - shift] => -h[j + s] for j < n-s
    # multiply by dx
    grad *= (2.0 / len(h))
    return grad

def max_corr_and_shift(h):
    n = len(h)
    # compute cross-correlation full
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)
    return max_val, shift, corr

def project_onto_constraints(v, target):
    # v shape (n,)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # while hi - lo > tolerance
    for _ in range(60):  # enough for double precision
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = tau
        else:
            hi = tau
    # final projection
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # if due to rounding sum might be slightly off; adjust by small iteration (maybe not needed)
    return x

def compute_full_corr(h):
    n = h.shape[0]
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def max_corr_and_shift(h):
    corr = compute_full_corr(h)
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = max_idx - (h.shape[0] - 1)
    return max_val, shift, corr

def gradient_max_corr(h, shift):
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # (1 - h[j + s]) term for j=0..n-s-1
        grad[:n - s] += (1.0 - h[s:])  # shape (n-s)
        # -h[j - s] term for j=s..n-1
        grad[s:] -= h[:n - s]
    else:
        s = -shift
        # (1 - h[j - s]) term for j=s..n-1
        grad[s:] += (1.0 - h[:n - s])
        # -h[j + s] term for j=0..n-s-1
        grad[:n - s] -= h[s:]
    grad *= dx
    return grad

def project_onto_constraints(v, target):
    # v is numpy array
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # bisection
    for _ in range(60):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # ensure sum close to target (within epsilon). If not adjust small shift:
    # Weighted adjust using bounds not needed.
    return x

import numpy as np
import time
import random

def compute_full_corr(h):
    """Full cross-correlation C(k) = ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def max_corr_and_shift(h):
    """Return (max_corr, shift_of_max, full_corr_array)."""
    corr = compute_full_corr(h)
    max_idx = int(np.argmax(corr))
    max_val = float(corr[max_idx])
    shift = max_idx - (h.shape[0] - 1)  # zero shift corresponds to centre index
    return max_val, shift, corr

def gradient_max_corr(h, shift):
    """Gradient of the correlation at the given shift (including dx factor)."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # (1 - h[j+s]) term for j in [0, n-s)
        grad[:n - s] += (1.0 - h[s:])
        # -h[j-s] term for j in [s, n)
        grad[s:] -= h[:n - s]
    else:
        s = -shift
        # (1 - h[j-s]) term for j in [s, n)
        grad[s:] += (1.0 - h[:n - s])
        # -h[j+s] term for j in [0, n-s)
        grad[:n - s] -= h[s:]
    grad *= dx
    return grad

def project_onto_constraints(v, target_sum):
    """
    Project vector v onto the set {x | 0 ≤ x_i ≤ 1, sum x_i = target_sum}.
    Uses a simple scalar bisection on the Lagrange multiplier tau with
    x_i = clip(v_i - tau, 0, 1).
    """
    lo = np.min(v) - 1.0   # ensures all entries are clipped to 1 at lo
    hi = np.max(v) + 1.0   # ensures all entries are clipped to 0 at hi
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    x = np.clip(v - tau, 0.0, 1.0)
    return x

def random_feasible(n, target_sum):
    """Generate a random vector in [0,1]^n with the required sum."""
    v = np.random.rand(n)
    return project_onto_constraints(v, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to lower the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # Acquire initial construction if provided
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = int(init_h.shape[0])
        target = n / 2.0
        h0 = project_onto_constraints(init_h, target)
    else:
        n = 60  # fallback size
        target = n / 2.0
        h0 = random_feasible(n, target)

    best_h = h0.copy()
    best_val, _, _ = max_corr_and_shift(best_h)

    # Optimization parameters
    step0 = 1.0
    max_inner_iters = 30000   # per restart
    restart = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # Choose start point for this restart
        if restart == 0:
            h = h0.copy()
        else:
            h = random_feasible(n, target)

        # Inner gradient descent
        for it in range(max_inner_iters):
            if time.time() - start_time > budget_s * 0.95:
                break

            cur_val, shift, _ = max_corr_and_shift(h)

            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()

            # Gradient of the currently maximal correlation
            grad = gradient_max_corr(h, shift)

            # Simple decreasing step schedule
            alpha = step0 * (0.9995 ** it)

            # Gradient step + projection onto constraints
            h_new = h - alpha * grad
            h_new = project_onto_constraints(h_new, target)

            # Occasional random jitter to escape flats
            if it % 5000 == 0 and it > 0:
                h_new += np.random.randn(n) * 0.01
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_onto_constraints(h_new, target)

            h = h_new

        restart += 1

    # Final evaluation of the best found solution
    best_val, _, _ = max_corr_and_shift(best_h)
    return best_h, float(best_val), n
