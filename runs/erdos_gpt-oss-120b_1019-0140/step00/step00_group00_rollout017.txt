def project_onto_box_sum(v, target, tol=1e-9, max_iter=100):
    # Solve for lambda such that sum(clip(v + lam, 0, 1)) = target
    low = -np.max(v) - 1.0  # ensures v+low <=0
    high = 1.0 - np.min(v) + 1.0  # ensures v+high >=1
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        projected = np.clip(v + lam, 0.0, 1.0)
        s = projected.sum()
        if abs(s - target) <= tol:
            return projected
        if s < target:
            low = lam
        else:
            high = lam
    # after max_iter, return best approx
    return np.clip(v + (low + high) / 2.0, 0.0, 1.0)

import numpy as np
import time
import random

# ----- Helper Functions -----------------------------------------------------

def compute_c5(h, dx):
    """Return the maximal overlap integral C5 for a given h."""
    # Full linear correlation of h with (1-h), then scale by dx.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def autocorr_full(h):
    """Linear autocorrelation of h (full mode) using FFT."""
    n = h.shape[0]
    size = 2 * n
    # zero‑pad to size 2n
    padded = np.concatenate([h, np.zeros(n, dtype=h.dtype)])
    H = np.fft.fft(padded)
    S = H * np.conj(H)
    corr = np.fft.ifft(S).real
    return corr[: 2 * n - 1]


def project_onto_box_sum(v, target, tol=1e-9, max_iter=60):
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum = target.
    Uses bisection on the scalar offset lambda.
    """
    lo = -np.max(v) - 2.0          # guarantees all entries <= 0 after shift
    hi = 1.0 - np.min(v) + 2.0     # guarantees all entries >= 1 after shift
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            lo = lam
        else:
            hi = lam
    # final approximation
    lam = (lo + hi) / 2.0
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n, target):
    """Generate a random feasible h with entries in [0,1] and given sum."""
    v = np.random.rand(n)
    return project_onto_box_sum(v, target)


# ----- Main Optimization ---------------------------------------------------

    start = time.time()

    # ---- Determine problem size and initial h ---------------------------

        h = np.interp(xn, x0, init_h.astype(float))
    except NameError:
        # No parent construction – start from a random feasible vector.
        n = 200

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility
    h = project_onto_box_sum(h, target_sum)

    # Track the best solution seen
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # ----- Parameters for the smoothed max‑overlap descent ---------------
    alpha = 10.0          # smoothing parameter for the soft‑max
    lr = 50.0             # initial step size (will be decayed)
    it = 0
    no_improve = 0

    # Main loop – stop a little before the budget expires
    while time.time() - start < budget_s * 0.94:

        # Autocorrelation of h (linear, full) scaled by dx
        s = autocorr_full(h) * dx                     # shape 2n-1
