```python
def compute_c5_fft(h, dx):
    n = h.shape[0]
    # compute correlation of h with 1 - h
    b = 1.0 - h
    # Use FFT with length power of 2 >= 2*n - 1
    nfft = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(b[::-1], n=nfft)
    conv = np.fft.irfft(H * B, n=nfft)[:2*n-1]
    # So conv[k] corresponds to correlation for shift s = k - (n-1)
    # Weighted by dx
    c5 = np.max(conv) * dx
    return float(c5)
```

import time
import numpy as np
from scipy.optimize import minimize_scalar

def compute_c5_fft(h, dx):
    """Return max_{shift} ∫ h(x)*(1-h(x+shift)) dx approximated by FFT."""
    n = h.shape[0]
    b = 1.0 - h
    # length for convolution (next power of two >= 2*n-1)
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    B_rev = np.fft.rfft(b[::-1], n=nfft)
    conv = np.fft.irfft(H * B_rev, n=nfft)[: 2 * n - 1]
    return float(np.max(conv) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        use_initial = True
    except NameError:
        init = None
        n = 512          # default size if no parent is given
        use_initial = False

    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx  # mid‑point sample positions

    best_h = None
    best_c5 = np.inf

    # Helper to evaluate a candidate
    def evaluate_candidate(h_vec):
        nonlocal best_h, best_c5
        # enforce sum constraint (exactly n/2)
        total = h_vec.sum()
        target = n / 2.0
        if not np.isclose(total, target, atol=1e-12):
            h_vec = h_vec * (target / total)
        np.clip(h_vec, 0.0, 1.0, out=h_vec)
        val = compute_c5_fft(h_vec, dx)
        if val < best_c5:
            best_c5 = val
            best_h = h_vec.copy()
        return val

    # 1. start from the provided construction if any
    if use_initial:
        evaluate_candidate(init)

    # 2. linear families h(x)=a + (0.5-a)*x
    def c5_from_a(a):
        s = 0.5 - a
        h = a + s * x
        return compute_c5_fft(h, dx)

    res = minimize_scalar(c5_from_a, bounds=(0.0, 0.5), method='bounded',
                          options={'xatol': 1e-5, 'maxiter': 100})
    if res.success:

        h_opt = a_opt + (0.5 - a_opt) * x

    # 3. sinusoidal families h(x) = (1 - cos(p * pi * (x + phi))) / 2

    # Determine whether current best is essentially binary
