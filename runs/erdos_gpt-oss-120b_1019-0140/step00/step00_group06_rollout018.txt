def compute_c5(h):
   n = len(h)
   b = 1 - h
   # reverse b
   b_rev = b[::-1]
   # pad to size 2*n for linear convolution
   L = 2 * n
   H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
   B_rev = np.fft.rfft(np.concatenate([b_rev, np.zeros(n)]))
   conv = np.fft.irfft(H * B_rev, n=2*n)
   # extract correlation for shifts k = -(n-1)...(n-1)
   # conv index l = 0..2n-2 => shift k = n-1 - l
   # So max correlation value = max(conv) (since mapping is just reordering)
   max_corr = conv.max()
   dx = 2.0 / n
   return max_corr * dx

import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    b = 1.0 - h
    # reverse b
    b_rev = b[::-1]
    # zero pad both to length 2n
    L = 2 * n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    B = np.fft.rfft(np.concatenate([b_rev, np.zeros(n)]))
    conv = np.fft.irfft(H * B, n = L)
    # conv length = 2n
    max_corr = np.max(conv)
    dx = 2.0 / n
    return max_corr * dx

def project_to_constraints(h, target_sum):
    # Clip into [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Iteratively adjust sum via scaling factor but keep within bounds
    # We'll use simple scaling and re-clip iteratively.
    for _ in range(10):
        current_sum = h.sum()
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            break
        # Determine indices that can be increased or decreased.
        if diff > 0:
            # need to increase sum: increase values where h < 1
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            # distribute diff equally
            inc = diff / free.size
            # but ensure not exceed bound
            h[free] += inc
            # clip again

        h = h * factor
        h = np.clip(h, 0.0, 1.0)

    return h

import numpy as np
import time
import scipy.optimize as _opt

# Global variable used by the constraint function
_GLOBAL_TARGET_SUM = 0.0


def _constraint_sum_eq(x):
    """Equality constraint: sum(x) == target_sum."""
    return np.sum(x) - _GLOBAL_TARGET_SUM


def _enforce_sum(v, target):
    """
    Project vector v onto the feasible set:
    - clip to [0,1]
    - adjust entries so that sum(v) == target while staying in [0,1]
    """
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    # Simple iterative redistribution of the residual
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
        else:
            idx = np.where(v > 0.0)[0]
        if idx.size == 0:
            break
        # distribute equally among available indices
        delta = diff / idx.size
        # limit delta so we stay inside bounds
        if diff > 0:
            delta = np.minimum(delta, 1.0 - v[idx])
        else:
            delta = np.maximum(delta, -v[idx])
        v[idx] += delta
        diff = target - np.sum(v)
    # final tiny correction on a single free entry if needed
    diff = target - np.sum(v)
    if abs(diff) > 1e-12:
        if diff > 0:
            i = np.where(v < 1.0)[0]
        else:
            i = np.where(v > 0.0)[0]
        if i.size:
            i = i[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v


def _compute_c5(h):
    """
    Compute C5 for a step function represented by vector h.
    Uses FFT to obtain the linear correlation of h with (1-h).
    """
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]                 # reversed (1-h)
    # zero‑pad to length 2n for linear convolution
    padded_h = np.concatenate([h, np.zeros(n)])
    padded_b = np.concatenate([b_rev, np.zeros(n)])
    H = np.fft.rfft(padded_h)
    B = np.fft.rfft(padded_b)
    conv = np.fft.irfft(H * B, n=2 * n)
    max_corr = np.max(conv)                # max over all shifts
    dx = 2.0 / n
    return max_corr * dx


def _hill_climb_binary(start_h, target_sum, rng, time_budget):
    """
    Simple hill climbing for binary vectors:
    repeatedly swap a 1 and a 0 if it reduces C5.
    """
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # maintain mutable lists of indices of ones and zeros
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])

    start = time.time()
    while time.time() - start < time_budget:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        val = _compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

        if val < best_val or rng.random() < 0.0:  # accept only downhill moves
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_val

    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    # we will perform a few SLSQP runs (including the start vector)
    max_slsqp_runs = 4
    for run_idx in range(max_slsqp_runs):
        elapsed = time.time() - start_total
        if elapsed >= budget_s * 0.85:
            break

        if run_idx == 0:
            x0 = best_h
        else:
            # random feasible start
            x0 = rng.random(n)
            x0 = _enforce_sum(x0, target_sum)

        _GLOBAL_TARGET_SUM = target_sum
        try:
            res = _opt.minimize(
                _compute_c5,
                x0,
                method="SLSQP",
                bounds=[(0.0, 1.0)] * n,
                constraints={"type": "eq", "fun": _constraint_sum_eq},
                options={"maxiter": 800, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        if res.success:
            cand = _enforce_sum(np.clip(res.x, 0.0, 1.0), target_sum)
            cand_val = _compute_c5(cand)
            if cand_val < best_val:
                best_h, best_val = cand, cand_val

    # ------------------------------------------------------------------
    # Second stage – binary hill climbing (swap 1↔0)
    # ------------------------------------------------------------------
    # Convert the current best (or a fresh) vector to a binary one
    binary_start = np.zeros(n, dtype=float)
    # choose the largest target_sum entries
    idx = np.argpartition(best_h, -target_sum)[-target_sum:]
    binary_start[idx] = 1.0

    # allocate remaining time for hill climbing
    remaining = budget_s - (time.time() - start_total)
    if remaining > 2.0:
        # a few random‑restart hill‑climbs
        n_restarts = 6
        time_per = max(0.5, remaining / (n_restarts + 1))
        # first climb from the binary version derived above
        bc, bv = _hill_climb_binary(binary_start, target_sum, rng, time_per)
        if bv < best_val:
            best_h, best_val = bc, bv

        for _ in range(n_restarts):
            elapsed = time.time() - start_total
            if elapsed >= budget_s * 0.95:
                break
            # random binary feasible vector
            rnd_idx = rng.choice(n, size=target_sum, replace=False)
            rnd_h = np.zeros(n, dtype=float)
            rnd_h[rnd_idx] = 1.0
            bc, bv = _hill_climb_binary(rnd_h, target_sum, rng, time_per)
            if bv < best_val:
                best_h, best_val = bc, bv

    # final safety projection (should already be binary and feasible)
    best_h = _enforce_sum(np.clip(best_h, 0.0, 1.0), target_sum)

    return best_h, best_val, n
