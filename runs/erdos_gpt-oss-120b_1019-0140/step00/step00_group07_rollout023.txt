import time

def _compute_c5(arr: np.ndarray) -> float:
    """Return the overlap bound C5 for a step function sampled in arr."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def _initialize_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible start vector with exactly n/2 mass."""
    # Try the provided global if it exists and looks feasible.
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64, copy=True)
        if h0.shape[0] == n and np.all((0.0 <= h0) & (h0 <= 1.0)):
            # enforce the exact sum if necessary
            target_sum = n / 2
            cur_sum = h0.sum()
            if abs(cur_sum - target_sum) <= 1e-9:
                return h0
    # Fall back to a balanced Thue‑Morse / random binary pattern.
    # Thue‑Morse bits
    bits = np.fromiter((bin(i).count("1") % 2 for i in range(n)), dtype=np.int8, count=n)
    cur_sum = bits.sum()
    target_sum = n // 2
    if cur_sum > target_sum:
        idx = np.where(bits == 1)[0]
        to_flip = rng.choice(idx, size=cur_sum - target_sum, replace=False)
        bits[to_flip] = 0
    elif cur_sum < target_sum:
        idx = np.where(bits == 0)[0]
        to_flip = rng.choice(idx, size=target_sum - cur_sum, replace=False)
        bits[to_flip] = 1
    h0 = bits.astype(np.float64)
    # If somehow still off, finish with a random binary fill.
    if h0.sum() != target_sum:
        h0[:] = 0.0
        ones = rng.choice(n, size=target_sum, replace=False)
        h0[ones] = 1.0
    return h0

                j += 1
            if rng.random() < 0.5:
                # i -> j
                max_transfer = min(h_cur[i], 1.0 - h_cur[j])
                if max_transfer <= 0:
                    continue
                eps = rng.random() * max_transfer * 0.5
                new_h = h_cur.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                # j -> i
                max_transfer = min(h_cur[j], 1.0 - h_cur[i])
                if max_transfer <= 0:
                    continue
                eps = rng.random() * max_transfer * 0.5
                new_h = h_cur.copy()
                new_h[i] += eps
                new_h[j] -= eps

        elif move_type < 0.9:
            # Random split of the mass on a pair.
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            total = h_cur[i] + h_cur[j]
            lo = max(0.0, total - 1.0)
            hi = min(total, 1.0)
            if hi - lo <= 1e-12:
                continue
            new_i = rng.random() * (hi - lo) + lo
            new_j = total - new_i
            new_h = h_cur.copy()
            new_h[i] = new_i
            new_h[j] = new_j

        else:
            # Rare random restart to a fresh balanced binary vector.
            cand = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=n // 2, replace=False)
            cand[ones] = 1.0
            new_h = cand

        # Evaluate candidate.
        cand_val = _compute_c5(new_h)

        # Metropolis acceptance.
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur = new_h
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_h = new_h.copy()
                if best_val <= target:
                    break

    # Final repair: enforce the exact integral constraint.
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # Find a safely adjustable entry.
        for idx in range(n):
            new_val = best_h[idx] + diff
            if 0.0 <= new_val <= 1.0:
                best_h[idx] = new_val
                break

    final_c5 = _compute_c5(best_h)

    return (best_h, final_c5, n)
