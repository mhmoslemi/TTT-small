import numpy as np, time.

def compute_c5(h):
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def repair_h(h, target):
    # clip
    h = np.clip(h, 0, 1)
    # ensure sum = target via scaling and clipping iteratively
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Add diff equally to all entries, but after addition clip again.
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    # If sum not exact due to clipping, adjust using proportion of free entries
    # For simplicity, after loop, enforce sum by adding diff to random non-bound entries
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        # find indices not at bounds
        free_idx = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if len(free_idx) == 0:
            # fallback: distribute to all indices respecting bounds
            # but diff should be negligible
            pass
        else:
            # add diff uniformly across free indices
            add = diff / len(free_idx)
            h[free_idx] += add
            h = np.clip(h, 0, 1)
    # final correction to enforce sum exactly by scaling entire vector (if still off)
    s = h.sum()
    if abs(s - target) > 1e-9:
        # scaling factor may push out of bounds, but we'll accept minor violation
        h *= target / s
        h = np.clip(h, 0, 1)
        # adjust sum again
        s = h.sum()
        diff = target - s
        if abs(diff) > 1e-9:
            # distribute diff to arbitrary index
            idx = np.argmax(h) if diff < 0 else np.argmin(h)
            h[idx] += diff
            h = np.clip(h, 0, 1)
    return h

def anneal(h0, time_budget):
    start = time.time()
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = 0.1 # initial temp
    T_min = 1e-5
    # schedule: exponential decay based on elapsed fraction
    # We'll assume we can make ~200k iterations at most.
    iter = 0
    max_iter = 200000
    while time.time() - start < time_budget * 0.9 and iter < max_iter:
        # propose move: randomly pick i != j, modify values with sum preservation
        # For continuous: pick two indices and transfer delta
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # choose a delta that does not push values out of bounds
        # delta max such that h[i] + delta <= 1 and h[j] - delta >= 0
        max_delta = min(1.0 - h[i], h[j])
        min_delta = max(-h[i], - (1.0 - h[j]))
        # random delta uniform
        delta = (np.random.rand() - 0.5) * 0.2 # max +/-0.1, but adjust later if out of bounds
        # clip delta to feasible range
        delta = np.clip(delta, -max_delta, max_delta)
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # sum unchanged, no need repair except clipping (already ensured)
        # compute c5
        c_new = compute_c5(h_new)
        # accept if better or with probability
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # acceptance criterion
        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (time_budget * 0.9)
        T = T0 * (0.99995 ** iter) # not using frac
        # but more controlled: T = max(T0 * (1 - frac), T_min)
        T = max(T0 * (1 - frac), T_min)
        if c_new < compute_c5(h) or np.random.rand() < np.exp(-(c_new - compute_c5(h)) / T):
            h = h_new
        iter += 1
    return best_h, best_c

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.fft(h, n=L)
    fg = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(fh * np.conj(fg)).real[:2*n-1]
    return np.max(corr) * dx

def repair_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling
    for _ in range(5):
        s = h.sum()
        if s == 0:
            break
        h = np.clip(h * (target / s), 0.0, 1.0)
    # final adjustment
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        # adjust a random index to fix sum
        idx = np.random.randint(len(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        # maybe need to correct slight over/under
    h = np.clip(h, 0.0, 1.0)
    return h

def anneal(h0, budget):
    start = time.time()
    n = len(h0)
    target = n / 2.0
    L = 1 << ((2*n - 1).bit_length())
    h = h0.copy()
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    # SA parameters
    T0 = 0.2
    T_end = 1e-5
    total_iters = 0
    # approximate iteration count allowed based on time
    # set max_iter = 500000
    max_iter = 500000
    while time.time() - start < budget * 0.99 and total_iters < max_iter:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        if max_inc <= 0 and max_dec <= 0:
            continue
        # set step scale: allow up to 0.1 or max feasible
        step_limit = min(0.1, max_inc, max_dec)
        if step_limit <= 0:
            continue
        # propose delta in [-step_limit, step_limit]
        delta = (np.random.rand() * 2 - 1) * step_limit
        # ensure feasibility
        delta = np.clip(delta, -max_dec, max_inc)
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute new C5
        c_new = compute_c5(h_new)
        # temperature schedule linear from T0 to T_end over iterations
        frac = total_iters / max_iter
        T = T0 * (1 - frac) + T_end * frac
        accept = False
        if c_new < current_c:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(c_new - current_c) / T)
                accept = np.random.rand() < prob
        if accept:
            h = h_new
            current_c = c_new
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
            # if good enough, break early
            if best_c <= 0.3808:
                break
        total_iters += 1
    # final repair to ensure sum
    best_h = repair_h(best_h, target)
    best_c = compute_c5(best_h)
    return best_h, best_c

def random_binary_h(n):
    h = np.zeros(n)
    half = n // 2
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        # assign .5 to one extra entry
        remaining = list(set(range(n)) - set(idx))
        h[remaining[0]] = 0.5
    return h

import numpy as np
import math
import time

# ---------- Helper Functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using FFT-based correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fh = np.fft.fft(h, n=L)
    fg = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(fh * np.conj(fg)).real[: 2 * n - 1]
    return np.max(corr) * dx


def repair_h(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto [0,1] box and enforce sum == target (up to tiny tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling to approach the target sum
    for _ in range(5):
        s = h.sum()
        if s == 0:
            break
        factor = target / s
        h = np.clip(h * factor, 0.0, 1.0)
    # final tiny correction if needed
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # spread the residual over unfixed entries
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            delta = diff / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust a random coordinate safely
            idx = np.random.randint(h.size)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    # final clamp
    return np.clip(h, 0.0, 1.0)


def random_binary_h(n: int) -> np.ndarray:
    """Generate a binary (0/1) vector with exactly n/2 ones (half‑integer handled)."""
    h = np.zeros(n)
    half = n // 2
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    if n % 2:  # odd length – add a single 0.5 entry to hit the exact sum
        remaining = list(set(range(n)) - set(idx))
        h[remaining[0]] = 0.5
    return h


def anneal(h0: np.ndarray, budget: float) -> tuple:
    """Simulated‑annealing style improvement of h0 within the given time budget."""
    start = time.time()
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c

    # temperature schedule
    T0, T_end = 0.20, 1e-5
    max_iters = 400_000  # safety cap
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget * 0.98 or it >= max_iters:
            break

        # pick two distinct indices and propose a mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_inc = 1.0 - h[i]   # how much we can raise i
        max_dec = h[j]         # how much we can lower j
        if max_inc <= 0 and max_dec <= 0:
            continue
        step_limit = min(0.10, max_inc, max_dec)
        if step_limit <= 0:
            continue
        delta = (np.random.rand() * 2.0 - 1.0) * step_limit
        delta = np.clip(delta, -max_dec, max_inc)
        if abs(delta) < 1e-12:
            continue

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        c_new = compute_c5(h_new)

        # linear cooling
        frac = elapsed / budget
        T = T0 * (1.0 - frac) + T_end * frac

        if c_new < current_c or (T > 0 and np.random.rand() < math.exp(-(c_new - current_c) / T)):
            h = h_new
            current_c = c_new

        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
            if best_c <= 0.3808:
                break

        it += 1

    best_h = repair_h(best_h, target)
    best_c = compute_c5(best_h)
    return best_h, best_c
