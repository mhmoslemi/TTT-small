def hill_climbing(h):
    N = len(h)
    O_raw = np.correlate(h, 1 - h, mode='full')
    O_raw = O_raw.astype(np.int32)
    current_max = O_raw.max()
    best_h = h.copy()
    best_max = current_max
    # maintain list of indices
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    import random
    start_time = time.time()
    # parameters
    max_no_improve = 20000
    no_improve = 0
    while time.time() - start_time < budget_s * 0.95:
        # pick random swap
        i1 = random.choice(ones)
        i2 = random.choice(zeros)
        # compute delta O and new max in one pass
        delta = np.empty(2*N-1, dtype=np.int32)
        new_max = -1
        for k in range(- (N-1), N):
            idx = k + N - 1
            if k >= 0:
                limit = N - 1 - k
                deltaS = (i2 <= limit) - (i1 <= limit)
            else:
                bound = -k
                deltaS = (i2 >= bound) - (i1 >= bound)
            if k == 0:
                deltaA = 0
            else:
                inc = 0
                pos = i2 + k
                if 0 <= pos < N and pos != i1:
                    inc += h[pos]
                pos = i2 - k
                if 0 <= pos < N and pos != i1:
                    inc += h[pos]
                dec = 0
                pos = i1 + k
                if 0 <= pos < N and pos != i2:
                    dec += h[pos]
                pos = i1 - k
                if 0 <= pos < N and pos != i2:
                    dec += h[pos]
                deltaA = inc - dec
            deltaO = deltaS - deltaA
            delta[idx] = deltaO
            val = O_raw[idx] + deltaO
            if val > new_max:
                new_max = val
        # acceptance
        accept = False
        if new_max <= current_max:
            accept = True
        else:
            # allow occasional uphill moves
            # temperature schedule: linear decrease over time
            elapsed = time.time() - start_time
            frac = elapsed / (budget_s * 0.95)
            # simple schedule: T = 0.1 * (1 - frac)
            T = max(0.001, 0.1 * (1 - frac))
            if random.random() < np.exp(-(new_max - current_max) / T):
                accept = True
        if accept:
            # apply swap
            h[i1], h[i2] = 0, 1  # h[i1] currently 1, h[i2] currently 0
            # update O_raw
            O_raw += delta
            current_max = new_max
            # update indices lists
            # move i1 from ones to zeros, i2 from zeros to ones
            ones.remove(i1)
            zeros.remove(i2)
            # add swapped:
            zeros.append(i1)
            ones.append(i2)
            # check best
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if stuck
        if no_improve >= max_no_improve:
            # random restart
            # generate new random binary with half ones
            perm = np.random.permutation(N)
            h = np.zeros(N, dtype=np.int8)
            h[perm[:N//2]] = 1
            O_raw = np.correlate(h, 1 - h, mode='full').astype(np.int32)
            current_max = O_raw.max()
            ones = list(np.where(h==1)[0])
            zeros = list(np.where(h==0)[0])
            no_improve = 0

    return best_h, best_max

import numpy as np, random, time, math

def compute_initial_h(N):
    try:
        init = np.array(initial_h_values)
        # if lengths differ, fallback to N
        if init.shape[0] != N:
            # set N = init length
            N = init.shape[0]
        # binarize via rounding
        h = (init >= 0.5).astype(np.int8)
        target = N // 2
        cur = int(h.sum())
        if cur < target:
            zeros = np.where(h == 0)[0]
            np.random.shuffle(zeros)
            h[zeros[:target - cur]] = 1
        elif cur > target:
            ones = np.where(h == 1)[0]
            np.random.shuffle(ones)
            h[ones[:cur - target]] = 0
        return h
    except NameError:
        # generate random binary vector
        h = np.zeros(N, dtype=np.int8)
        h[np.random.choice(N, N//2, replace=False)] = 1
        return h

def optimize_h(h, budget_s, start_time):
    N = len(h)
    O_raw = np.correlate(h, 1 - h, mode='full').astype(np.int32)
    current_max = int(O_raw.max())
    best_h = h.copy()
    best_max = current_max
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    ks = list(range(-(N-1), N))
    offset = N - 1
    max_no_improve = 5000
    no_improve = 0
    target_max = int(math.floor(0.3808 * N / 2))  # maybe floor
    while time.time() - start_time < budget_s * 0.95:
        i1 = random.choice(ones)
        i2 = random.choice(zeros)
        delta = [0]*(2*N-1)
        new_max = -1
        for idx, k in enumerate(ks):
            # deltaS
            if k >= 0:
                limit = N - 1 - k
                deltaS = (i2 <= limit) - (i1 <= limit)
            else:
                bound = -k
                deltaS = (i2 >= bound) - (i1 >= bound)
            if k == 0:
                deltaA = 0
            else:
                inc = 0
                pos = i2 + k
                if 0 <= pos < N and pos != i1:
                    inc += h[pos]
                pos = i2 - k
                if 0 <= pos < N and pos != i1:
                    inc += h[pos]
                dec = 0
                pos = i1 + k
                if 0 <= pos < N and pos != i2:
                    dec += h[pos]
                pos = i1 - k
                if 0 <= pos < N and pos != i2:
                    dec += h[pos]
                deltaA = inc - dec
            deltaO = deltaS - deltaA
            delta[idx] = deltaO
            val = O_raw[idx] + deltaO
            if val > new_max:
                new_max = val
        # decide acceptance
        accept = False
        if new_max <= current_max:
            accept = True
        else:
            # simulated annealing probability
            elapsed = time.time() - start_time
            frac = elapsed / (budget_s * 0.95)
            T = max(1e-4, 0.1 * (1 - frac))
            if random.random() < math.exp(-(new_max - current_max) / T):
                accept = True
        if accept:
            # apply swap
            h[i1] = 0
            h[i2] = 1
            # update O_raw
            O_raw += delta
            current_max = new_max
            # update indices
            ones.remove(i1)
            zeros.remove(i2)
            zeros.append(i1)
            ones.append(i2)
            # update best
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                no_improve = 0
                if best_max <= target_max:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # random restart
            h = np.zeros(N, dtype=np.int8)
            h[np.random.choice(N, N//2, replace=False)] = 1
            O_raw = np.correlate(h, 1 - h, mode='full').astype(np.int32)
            current_max = int(O_raw.max())
            ones = list(np.where(h==1)[0])
            zeros = list(np.where(h==0)[0])
            no_improve = 0
    return best_h, best_max

import numpy as np, random, time, math

def _initial_h(N):
    try:
        init_arr = np.array(initial_h_values)
        # use size of provided array if different
        if init_arr.size != N:
            N = int(init_arr.size)
        # binary rounding
        h = (init_arr >= 0.5).astype(np.int8)
        target = N // 2
        cur = int(h.sum())
        diff = target - cur
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) >= diff:
                chosen = np.random.choice(zeros, diff, replace=False)
                h[chosen] = 1
        elif diff < 0:
            ones = np.where(h == 1)[0]
            chosen = np.random.choice(ones, -diff, replace=False)
            h[chosen] = 0
        return h
    except NameError:
        h = np.zeros(N, dtype=np.int8)
        h[np.random.choice(N, N // 2, replace=False)] = 1
        return h
