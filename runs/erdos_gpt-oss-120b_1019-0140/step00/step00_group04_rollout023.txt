                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = cur_h.copy()
                            if best_c5 <= target_bound:
                                break
                    else:
                        # revert move
                        cur_h[i] = hi
                        cur_h[j] = hj
        else:
            # fallback: random paired swap respecting bounds
            i, j = rng.integers(n, size=2)
            while i == j:
                j = rng.integers(n)
            hi, hj = cur_h[i], cur_h[j]
            delta_max = min(eps, hi, 1.0 - hj)
            delta_min = -min(eps, hj, 1.0 - hi)  # move opposite direction if needed
            if delta_max > delta_min:
                delta = rng.uniform(delta_min, delta_max)
                cur_h[i] = hi + delta
                cur_h[j] = hj - delta

                new_c5 = _compute_c5(cur_h)

                if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = cur_h.copy()
                        if best_c5 <= target_bound:
                            break
                else:
                    cur_h[i] = hi
                    cur_h[j] = hj

    # ------------------------------------------------------------------
    # 3. Final post‑processing to guarantee feasibility
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # correct any tiny sum deviation
    sum_best = best_h.sum()
    diff = target_sum - sum_best
    if abs(diff) > 1e-9:
        # adjust the first non‑saturated element
        idx = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(n)
        best_h[idx[0]] = np.clip(best_h[idx[0]] + diff, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, n
