import numpy as np
import time
import math

def max_overlap(h, dx):
    """Maximum overlap integral C5 for a given step vector."""
    # correlation of h with 1-h, zero‑padded, multiplied by dx
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def project_onto_box_sum(v, target):
    """
    Euclidean projection of v onto the hyperplane sum(u)=target
    with box constraints 0 <= u_i <= 1.
    The solution is u_i = clip(v_i - λ, 0, 1) for a suitable λ.
    """
    low = -np.max(v) - 10.0      # λ low enough → all entries clip to 1
    high = np.max(v) + 10.0      # λ high enough → all entries clip to 0
    for _ in range(60):          # enough for double precision
        mid = (low + high) * 0.5
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def random_feasible(n, target):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return project_onto_box_sum(v, target)

def smooth_weights(overlaps, alpha):
    """Soft‑max weights approximating the max of overlaps."""
    max_o = overlaps.max()
    exps = np.exp(alpha * (overlaps - max_o))
    return exps / exps.sum()

def gradient_phi(h, w, dx):
    """
    Gradient of the smooth objective φ = (1/α) log Σ exp(α·overlap_k)
    with respect to h, given the weight vector w.
    """
    q = 1.0 - 2.0 * h                     # = 1 - 2h
    r = np.correlate(q, w, mode='full')
    center = len(w) - 1
    grad = dx * r[center - np.arange(len(h))]
    return grad

def local_swap_refine(h, dx, max_swaps, time_limit, start_time):
    """
    Simple coordinate‑wise descent for binary‑like vectors.
    Swaps a 1 with a 0 (or fractional counterpart) whenever it improves the score.
    """
    n = len(h)
    best = h.copy()
    best_score = max_overlap(best, dx)
    i = 0
    while i < max_swaps:
        if time.time() - start_time > time_limit:
            break
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2 or best[i1] == best[i2]:
            continue
        new = best.copy()
        new[i1], new[i2] = best[i2], best[i1]   # swap preserves sum and bounds
        sc = max_overlap(new, dx)
        if sc < best_score:
            best = new
            best_score = sc
        i += 1
    return best, best_score
