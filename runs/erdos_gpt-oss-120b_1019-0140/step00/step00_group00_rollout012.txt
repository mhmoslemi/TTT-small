def project_onto_hyperplane_box(x, target):
    # x initial may be out of bounds.
    # Approach: clip to [0,1] first.
    x = np.clip(x, 0.0, 1.0)
    # Compute excess = sum(x) - target.
    excess = x.sum() - target
    if abs(excess) < 1e-12:
        return x
    # We'll distribute adjustment to free variables.
    # Iterate.
    max_iter = 100
    for _ in range(max_iter):
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further; break.
            break
        # Compute adjustment per free variable.
        delta = excess / n_free
        x[free] -= delta
        # Clip again.
        x = np.clip(x, 0.0, 1.0)
        excess = x.sum() - target
        if abs(excess) < 1e-12:
            break
    # If still off due to rounding, adjust via small adjustments.
    # final clip.
    return x

def gradient_of_max_corr(h):
    n = len(h)
    # compute full correlation
    corr_full = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr_full)
    shift = idx - (n - 1)
    # Compute gradient as described
    g = np.ones_like(h)
    idxs = np.arange(n)
    # plus indices
    plus = idxs + shift
    mask_plus = (plus >= 0) & (plus < n)
    g[mask_plus] -= h[plus[mask_plus]]
    # minus indices
    minus = idxs - shift
    mask_minus = (minus >= 0) & (minus < n)
    g[mask_minus] -= h[minus[mask_minus]]
    return g, idx, shift

def project_onto_hyperplane_box(x, target):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # adjust sum to target using iterative water-filling
    # tolerance
    tol = 1e-10
    max_iter = 100
    for _ in range(max_iter):
        excess = x.sum() - target
        if abs(excess) <= tol:
            break
        # free variables are those not at bounds
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further; break
            break
        # adjust free variables equally
        delta = excess / n_free
        x[free] -= delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
    # final correction if still off due to rounding
    excess = x.sum() - target
    if abs(excess) > 1e-9:
        # distribute to any variable within bounds
        # find indices where we can add/subtract small amount
        if excess > 0:
            # need to decrease sum
            # find indices > 0
            idx = np.where(x > 0.0)[0]
            # distribute equally among them
            if len(idx) > 0:
                delta = excess / len(idx)
                x[idx] -= delta
        else:
            # need to increase sum
            idx = np.where(x < 1.0)[0]
            if len(idx) > 0:
                delta = (-excess) / len(idx)
                x[idx] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    n = len(h)

    return np.max(corr)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def gradient_of_max_corr(h: np.ndarray):
    """Gradient of the maximal overlap term w.r.t. h."""
    n = h.size
    # full correlation (without scaling)
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr_full))
    shift = idx - (n - 1)  # integer offset
    # gradient: g_j = 1 - h_{j+shift} - h_{j-shift}
    g = np.ones_like(h)
    idxs = np.arange(n)
    plus = idxs + shift
    mask_plus = (plus >= 0) & (plus < n)
    g[mask_plus] -= h[plus[mask_plus]]
    minus = idxs - shift
    mask_minus = (minus >= 0) & (minus < n)
    g[mask_minus] -= h[minus[mask_minus]]
    return g, idx, shift

def project_onto_hyperplane_box(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while enforcing sum(x)=target_sum."""
    x = np.clip(x, 0.0, 1.0)
    tol = 1e-12
    for _ in range(100):
        excess = x.sum() - target_sum
        if abs(excess) <= tol:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = int(free.sum())
        if n_free == 0:
            break
        delta = excess / n_free
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction
    excess = x.sum() - target_sum
    if abs(excess) > 1e-9:
        if excess > 0:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                delta = excess / idx.size
                x[idx] -= delta
        else:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                delta = (-excess) / idx.size
                x[idx] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def local_optimize(h_init: np.ndarray, target_sum: float, deadline: float,
                   max_iter: int = 500):
    """Simple projected gradient descent on the maximal correlation."""
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = 0.2
    n = h.size
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        g, _, _ = gradient_of_max_corr(h)
        new_h = h - lr * g
        new_h = project_onto_hyperplane_box(new_h, target_sum)
        val = compute_c5(new_h)
        if val < best_val - 1e-9:
            best_h, best_val = new_h.copy(), val
            lr = min(lr * 1.1, 0.5)
        else:
            lr = lr * 0.7
        h = new_h
        if lr < 1e-5:
            # random jitter to escape stagnation
            jitter = (np.random.rand(n) - 0.5) * 0.05
            h = project_onto_hyperplane_box(best_h + jitter, target_sum)
            lr = 0.2
    return best_h, best_val

def optimize_n(n: int, base_h: np.ndarray, deadline: float):
    """Run a few restarts of local optimization for a given size n."""
    target_sum = n / 2.0
    # Prepare an initial vector of length n
    if base_h.size == n:
        init_h = base_h.copy()
    else:
        factor = n // base_h.size
        if factor * base_h.size == n:
            init_h = np.repeat(base_h, factor)
        else:
            # Linear interpolation fallback
            x_old = np.linspace(0.0, 2.0, base_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init_h = np.interp(x_new, x_old, base_h)
    init_h = project_onto_hyperplane_box(init_h, target_sum)

    best_h = init_h
    best_val = compute_c5(best_h)

    # a few jittered restarts
    for _ in range(5):
        if time.time() >= deadline:
            break
        jitter = (np.random.rand(n) - 0.5) * 0.1
        start_h = project_onto_hyperplane_box(best_h + jitter, target_sum)
        opt_h, opt_val = local_optimize(start_h, target_sum, deadline)
        if opt_val < best_val:
            best_h, best_val = opt_h, opt_val

    # final refinement from the current best
    if time.time() < deadline:
        opt_h, opt_val = local_optimize(best_h, target_sum, deadline)
        if opt_val < best_val:
            best_h, best_val = opt_h, opt_val

    return best_h, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # Load parent construction if it exists

    except NameError:
        base_h = None

    # Candidate sizes (prefer multiples of the base size)
    candidates = []
    if base_h is not None:
        n0 = base_h.size
        candidates.append(n0)
        for factor in (2, 4):
            n_candidate = n0 * factor
            if n_candidate <= 200:
                candidates.append(n_candidate)
    else:
        candidates = [60, 80, 100]

    best_h = None
    best_val = float('inf')
    best_n = None

    for n in candidates:
        if time.time() >= deadline:
            break
        # Determine the starting vector for this size
        if base_h is not None:
            start_vec = base_h
        else:
            start_vec = np.full(60, 0.5, dtype=np.float64)  # placeholder
        h_opt, val_opt = optimize_n(n, start_vec, deadline)
        if val_opt < best_val:
            best_h, best_val, best_n = h_opt, val_opt, n
        if best_val <= 0.3808:
            break

    # Fallback if nothing succeeded
    if best_h is None:
        best_n = 60
        best_h = np.full(best_n, 0.5, dtype=np.float64)
        best_val = compute_c5(best_h)

    # Final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_hyperplane_box(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), int(best_n))
