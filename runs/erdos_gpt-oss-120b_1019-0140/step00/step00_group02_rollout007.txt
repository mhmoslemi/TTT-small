def propose_move(h, rng, max_delta):
    i = rng.integers(len(h))
    # ensure we can move from i to some j
    # find index j != i where h[j] < 1
    # we can just while loop to find suitable j
    # choose j
    j = rng.integers(len(h))
    # ensure i != j
    while j == i:
        j = rng.integers(len(h))
    # compute max possible delta: min(max_delta, h[i], 1 - h[j])
    delta = rng.random() * max_delta
    max_possible = min(delta, h[i], 1.0 - h[j])
    # Actually we want delta <= h[i] and delta <= (1 - h[j])

    if limit <= 0:
        return None, None, None

    # propose new h copy
    new_h = h.copy()

import numpy as np
import time

# ----- Helper utilities -------------------------------------------------

def _project_to_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to have the given sum."""
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if cur_sum == 0.0:
        # evenly distribute the mass
        h[:] = target_sum / len(h)
        return h
    # scale to match target sum, then clip again
    scale = target_sum / cur_sum
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute remaining diff over entries that are not at a bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust a single element
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def _c5_value(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a discretised h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate old_h to a new grid of size new_n."""
    old_n = len(old_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return _project_to_feasible(new_h, new_n / 2.0)


# ----- Local search (simulated annealing) ------------------------------

def _local_search(initial_h: np.ndarray, n: int, time_limit: float, rng: np.random.Generator):
    """Hill‑climbing / simulated‑annealing on the feasible set."""
    # Ensure feasibility
    h_cur = _project_to_feasible(initial_h.copy(), n / 2.0)
    cur_score = _c5_value(h_cur)

    best_h = h_cur.copy()
    best_score = cur_score

    start = time.time()
    max_delta = 0.25          # maximal transferable mass per move
    T0 = 0.015                # initial temperature
    Tf = 1e-6                 # final temperature
    no_improve = 0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        # temperature and step size shrink linearly
        T = T0 * (1.0 - prog) + Tf
        delta_limit = max_delta * (1.0 - prog) + 1e-5

        # choose donor i (h[i] > 0) and receiver j (h[j] < 1)
        i = rng.integers(n)
        if h_cur[i] <= 1e-12:
            donors = np.where(h_cur > 1e-12)[0]
            if donors.size == 0:
                break
            i = int(rng.choice(donors))
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        if h_cur[j] >= 1.0 - 1e-12:
            receivers = np.where(h_cur < 1.0 - 1e-12)[0]
            receivers = receivers[receivers != i]
            if receivers.size == 0:
                break
            j = int(rng.choice(receivers))

        limit = min(delta_limit, h_cur[i], 1.0 - h_cur[j])
        if limit <= 0.0:
            continue
        delta = rng.random() * limit

        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = _c5_value(h_new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart from the current best if stuck
        if no_improve >= 3000:
            h_cur = best_h.copy()
            cur_score = best_score
            no_improve = 0

    return best_h, best_score


# ----- Multi‑stage optimization -----------------------------------------

def _optimize_multi_stage(initial_h: np.ndarray, total_budget: float, rng: np.random.Generator):
    """Progressively increase resolution and apply local search."""
    start = time.time()
    best_h = None
    best_score = np.inf

    # Determine increasing grid sizes (start, then double up to 256)
    grids = [len(initial_h)]
    while grids[-1] < 256:
        nxt = min(grids[-1] * 2, 256)
        if nxt == grids[-1]:
            break
        grids.append(nxt)
    if grids[-1] != 256:
        grids.append(256)

    for idx, n in enumerate(grids):
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        if remaining <= 0.0:
            break
        # allocate equal share of the remaining time for each upcoming stage
        time_for_stage = remaining / (len(grids) - idx)

        # initialise h at this resolution
        if best_h is None:
            h0 = np.array(initial_h, copy=True)
            if len(h0) != n:
                h0 = _interpolate_h(h0, n)
        else:
            h0 = _interpolate_h(best_h, n)

        # run local search
        h_opt, score_opt = _local_search(h0, n, time_for_stage, rng)

        if score_opt < best_score:
            best_h, best_score = h_opt, score_opt

        # early exit if we reached a target bound comfortably
        if best_score <= 0.3808:
            break

    return best_h, best_score


# ----- Main entry point -------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """

    # Obtain initial construction

        h_start = np.array(initial_h_values, dtype=float)  # pre‑imported if available

        h_start = rng.random(n0)
        h_start = _project_to_feasible(h_start, n0 / 2.0)

    # Core optimisation
    best_h, best_score = _optimize_multi_stage(h_start, budget_s, rng)

    # Safety fallback
    if best_h is None:
        best_h = h_start
        best_score = _c5_value(best_h)

    # Final check using the provided evaluator if present
    try:
        bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        bound = best_score
