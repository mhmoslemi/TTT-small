import time
import random
import math
import numpy as np

def _binarize_and_balance(arr):
    """Return a binary (0/1) array with exactly half ones."""
    n = len(arr)
    target = n // 2
    # initial binarization by threshold
    bin_arr = (arr > 0.5).astype(float)
    cur = int(bin_arr.sum())
    if cur > target:
        ones_idx = np.where(bin_arr == 1)[0]
        # turn off the ones with smallest original values
        sorted_idx = ones_idx[np.argsort(arr[ones_idx])]
        bin_arr[sorted_idx[:cur - target]] = 0.0
    elif cur < target:
        zeros_idx = np.where(bin_arr == 0)[0]
        # turn on the zeros with largest original values
        sorted_idx = zeros_idx[np.argsort(-arr[zeros_idx])]
        bin_arr[sorted_idx[:target - cur]] = 1.0
    return bin_arr

def _random_binary_vector(n):
    """Generate a binary vector of length n with exactly n/2 ones."""
    vec = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    vec[ones] = 1.0
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after a heuristic search."""
    np.random.seed(seed)
    random.seed(seed)

    # Determine starting vector
    if 'initial_h_values' in globals():

        n = len(h0)

        h_current = _binarize_and_balance(h0)
        # safety: if still not balanced, replace
        if int(h_current.sum()) != n // 2:
            h_current = _random_binary_vector(n)
    else:
        n = 200  # even length for a fresh search
        h_current = _random_binary_vector(n)

    # Initial evaluation
    cur_score = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    best_score = cur_score

    # Annealing parameters
    total_budget = budget_s - 0.2  # small safety margin
    start_time = time.time()
    T0 = 0.02
    Tmin = 1e-6
    cooling = 0.9995
    T = T0
    stagnation = 0
    stagnation_limit = 20000

    # Main optimization loop
    while time.time() - start_time < total_budget:
        # pick a 1-index and a 0-index
        while True:
