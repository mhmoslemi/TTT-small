import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr), corr

def random_move(h, rng):
    n = h.shape[0]
    src = np.where(h > 1e-12)[0]
    tgt = np.where(h < 1 - 1e-12)[0]
    if src.size == 0 or tgt.size == 0:
        return None
    i = rng.choice(src)
    # target cannot be same as source
    possible_tgt = tgt[tgt != i]
    if possible_tgt.size == 0:
        return None
    j = rng.choice(possible_tgt)
    delta_max = min(h[i], 1 - h[j])
    if delta_max <= 0:
        return None
    delta = delta_max * rng.random()
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] = h[i] - delta
    h_new[j] = h[j] + delta
    return h_new

def guided_move(h, corr, rng):
    n = h.shape[0]
    # locate shift with maximal overlap cost
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)  # shift index (positive means h shifted right)
    # overlapping index interval
    i_start = max(0, -k)
    i_end = min(n, n - k)
    if i_start >= i_end:
        return None
    i_range = np.arange(i_start, i_end)
    # contributions to the overlap at this shift
    contrib = h[i_range] * (1 - h[i_range + k])
    # pick index with largest contribution
    local_max = int(np.argmax(contrib))
    i_max = i_range[local_max]
    # pick target index with smallest h (and not i_max)
    sorted_idx = np.argsort(h)
    j = None
    for idx in sorted_idx:
        if idx != i_max and h[idx] < 1 - 1e-12:
            j = idx
            break
    if j is None:
        return None
    # try moving mass from i_max (high) to j (low)
    hi = h[i_max]; hj = h[j]
    if hi > 0 and hj < 1:
        delta_max = min(hi, 1 - hj)
        if delta_max <= 0:
            return None
        delta = delta_max * rng.random()
        if delta <= 0:
            return None
        h_new = h.copy()
        h_new[i_max] = hi - delta
        h_new[j] = hj + delta
        return h_new
    # else try opposite direction
    if hj > 0 and hi < 1:
        delta_max = min(hj, 1 - hi)
        if delta_max <= 0:
            return None
        delta = delta_max * rng.random()
        if delta <= 0:
            return None
        h_new = h.copy()
        h_new[i_max] = hi + delta
        h_new[j] = hj - delta
        return h_new
    return None

def repair_sum(h, target_sum, rng):
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    # find an index that can absorb diff without violating bounds
    if diff > 0:
        cand = np.where(h < 1 - 1e-12)[0]
    else:
        cand = np.where(h > 1e-12)[0]
    if cand.size == 0:
        # no room, simply return h (tiny violation)
        return h
    idx = rng.choice(cand)
    h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialization
    init = globals().get('initial_h_values', None)
    if init is not None:
        h = np.array(init, dtype=np.float64)
        n = h.shape[0]
    else:
        n = 84  # fallback size (even)
        h = np.full(n, 0.5, dtype=np.float64)
    # enforce feasibility
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / h.sum())
    # initial score
    best_h = h.copy()
    best_score, cur_corr = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    # annealing parameters
    temp_start = 0.01
    temp_end = 1e-5
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.99:
            break
        # temperature schedule based on elapsed fraction
        frac = elapsed / budget_s
        temp = temp_start * (1 - frac) + temp_end * frac
        # decide move type
        if rng.random() < 0.6:
            cand = guided_move(cur_h, cur_corr, rng)
        else:
            cand = random_move(cur_h, rng)
        if cand is None:
            continue
        # repair any tiny sum deviation
        cand = repair_sum(cand, target_sum, rng)
        # compute score for candidate
        cand_score, cand_corr = compute_c5(cand)
        # acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            delta = cand_score - cur_score
            prob = math.exp(-delta / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h = cand
            cur_score = cand_score
            cur_corr = cand_corr
        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()
        iteration += 1
    # final repair for best solution
    best_h = repair_sum(best_h, target_sum, rng)
    # ensure constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    # final score recomputed
    final_score, _ = compute_c5(best_h)
    return (best_h.tolist(), float(final_score), n)

        delta = delta_max * rng.random()
        if delta <= 0:
            return None
        h_new = h.copy()
        h_new[i_max] = hi + delta
        h_new[j] = hj - delta
        return h_new
    return None


def repair_sum(h, target_sum, rng):
    """Adjust a single free entry to enforce the exact sum constraint."""
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cand = np.where(h < 1 - 1e-12)[0]
    else:
        cand = np.where(h > 1e-12)[0]
    if cand.size == 0:
        return h
    idx = rng.choice(cand)
    h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C₅ constant."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.array(init, dtype=np.float64)
        n = h.shape[0]
    else:
        n = 84                     # a modest even size when no parent is given
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / h.sum())          # enforce exact integral

    # ------------------------------------------------------------------
    # evaluate initial state
    best_h = h.copy()
    best_score, cur_corr = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    # annealing schedule
    temp_start = 0.01
    temp_end = 1e-5
    iteration = 0

    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.99:
            break
        frac = elapsed / budget_s
        temp = temp_start * (1 - frac) + temp_end * frac

        # decide move type (guided preferred)
        if rng.random() < 0.6:
            cand = guided_move(cur_h, cur_corr, rng)
        else:
            cand = random_move(cur_h, rng)
        if cand is None:
            continue

        cand = repair_sum(cand, target_sum, rng)
        cand_score, cand_corr = compute_c5(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            delta = cand_score - cur_score
            prob = math.exp(-delta / (temp + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h, cur_score, cur_corr = cand, cand_score, cand_corr

        if cand_score < best_score:
            best_h, best_score = cand.copy(), cand_score

        iteration += 1

    # final clean‑up
    best_h = repair_sum(best_h, target_sum, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_score, _ = compute_c5(best_h)

    return best_h.tolist(), float(final_score), n
