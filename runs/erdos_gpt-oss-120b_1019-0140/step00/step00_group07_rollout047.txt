def _compute_c5(h):
    """Return the C5 value for a step function given by sample array h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))

def _worst_shift_gradient(h):
    """Gradient of the worst (max) overlap term with respect to h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)          # shift = -(n-1)...+(n-1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # h[i] * (1 - h[i+shift])  for i=0..n-1-shift
        grad[: n - shift] += 1.0 - h[shift:]
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        # h[i] * (1 - h[i-s]) for i=s..n-1
        grad[s:] += 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]
    return grad, conv[idx] * (2.0 / n), shift

def _project_feasible(x, target_sum):
    """Clamp to [0,1] and then adjust to meet exactly target_sum."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(12):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly over free variables
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        # apply, respecting bounds
        upd = np.where(free, np.clip(x[free] + delta, 0.0, 1.0) - x[free], 0.0)
        x[free] += upd
    # final tiny correction if needed
    s = x.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        for i in range(len(x)):
            if diff == 0.0:
                break
            if diff > 0 and x[i] < 1.0:
                take = min(diff, 1.0 - x[i])
                x[i] += take
                diff -= take
            elif diff < 0 and x[i] > 0.0:
                take = min(-diff, x[i])
                x[i] -= take
                diff += take
    return x
