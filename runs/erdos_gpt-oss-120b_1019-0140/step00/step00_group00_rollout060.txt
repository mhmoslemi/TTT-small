def project_cap_simplex(y, target):
    n = len(y)
    low = -np.max(y) - 1  # ensures y+low <= -something <0 -> clipped to 0
    high = 1 - np.min(y) + 1 # ensures clipped to 1
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(y + (low+high)/2.0, 0.0, 1.0)
    # due to rounding might be off; adjust by small normalization
    diff = np.sum(x) - target
    if abs(diff) > 1e-8:
        # adjust interior points not at bounds
        interior = (x > 1e-8) & (x < 1-1e-8)
        if np.any(interior):
            x[interior] -= diff / np.count_nonzero(interior)
            x = np.clip(x, 0.0, 1.0)

import numpy as np, time, math, random

def _project_cap_simplex(v, target):
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    diff = x.sum() - target
    if abs(diff) > 1e-12:

        if free.any():
            x[free] -= diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    idx1 = np.arange(n) - lag
    term1 = np.ones(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1[mask1] = 1.0 - h[idx1[mask1]]
    idx2 = np.arange(n) + lag
    term2 = np.zeros(n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2[mask2] = h[idx2[mask2]]
    g = term1 - term2
    return h - step * g

import numpy as np, time, math, random

def _project_cap_simplex(v, target):
    """Projection of v onto the hyperplane sum(x)=target with 0<=x<=1."""
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    x = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    diff = x.sum() - target
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] -= diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_candidate(h, lag, step):
    """Return h moved a small step against the gradient of the maximal overlap."""
    n = len(h)
    idx1 = np.arange(n) - lag
    term1 = np.ones(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1[mask1] = 1.0 - h[idx1[mask1]]
    idx2 = np.arange(n) + lag
    term2 = np.zeros(n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2[mask2] = h[idx2[mask2]]
    g = term1 - term2
    return h - step * g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # ----- initial construction ------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
    else:
        n = 200
        xs = np.linspace(0, 2, n, endpoint=False) + (2.0 / n) / 2.0
        h0 = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))

    target = n / 2.0
    cur_h = _project_cap_simplex(h0, target)
    cur_val = evaluate_erdos_solution(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    if best_val <= 0.3808:
        return (best_h.tolist(), float(best_val), n)

    # ----- simulated annealing -------------------------------------------------
    T = 0.5
    Tmin = 1e-7
    cooling = 0.9995
    perturb_frac = 0.2

    while True:
        if time.time() - start > budget_s * 0.96:
            break

        # gradient‑based candidate (using the current worst lag)
        corr_full = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx_max = int(np.argmax(corr_full))
        lag = idx_max - (n - 1)
        cand_grad = _project_cap_simplex(_gradient_candidate(cur_h, lag, 0.02 * T + 1e-12), target)
        val_grad = evaluate_erdos_solution(cand_grad)

        # random perturbation candidate
        cand_rand = cur_h.copy()
        k = max(1, int(perturb_frac * n))
        idx = np.random.choice(n, k, replace=False)
        cand_rand[idx] += np.random.normal(scale=0.2 * T + 1e-12, size=k)
        cand_rand = _project_cap_simplex(cand_rand, target)
        val_rand = evaluate_erdos_solution(cand_rand)

        # choose the better of the two proposals
        if val_grad < val_rand:
            cand, cand_val = cand_grad, val_grad
        else:
            cand, cand_val = cand_rand, val_rand

        # SA acceptance
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_h = cand_val, cand.copy()
                if best_val <= 0.3808:
                    break

        T = max(Tmin, T * cooling)

    # ----- final deterministic refinement --------------------------------------
    for _ in range(30):
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        idx_max = int(np.argmax(corr_full))
        lag = idx_max - (n - 1)
        cand = _project_cap_simplex(_gradient_candidate(best_h, lag, 0.005), target)
        val = evaluate_erdos_solution(cand)
        if val < best_val:
            best_h, best_val = cand.copy(), val
        else:
            break

    return (best_h.tolist(), float(best_val), n)
