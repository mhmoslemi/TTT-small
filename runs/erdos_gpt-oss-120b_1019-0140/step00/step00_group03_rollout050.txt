import time
import numpy as np

def _project_onto_simplex_box(z, target_sum, max_iter=60, tol=1e-10):
    """Project vector z onto the set {x: 0<=x<=1, sum(x)=target_sum}."""
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(z - mid, 0.0, 1.0)
        total = proj.sum()
        if total > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    # Adjust tiny discrepancy in the sum
    diff = proj.sum() - target_sum
    if abs(diff) > tol:
        active = (proj > 1e-12) & (proj < 1 - 1e-12)
        if active.any():
            proj[active] -= diff / active.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj -= diff / len(z)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_bound(h, dx):
    """Return max correlation * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _gradient_max_corr(h):
    """Gradient of the max correlation term (subgradient)."""
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # integer shift corresponding to max
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(0, n - shift, dtype=int)
        j = i + shift
    else:
        i = np.arange(-shift, n, dtype=int)
        j = i + shift
    g[i] += b[j]
    g[j] -= h[i]
    return g, shift

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Obtain initial vector

        h0 = np.array(initial_h_values, dtype=float)

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility
    h = _project_onto_simplex_box(h0, target_sum)

    cur_h = h.copy()

    best_h = cur_h.copy()

    start_time = time.time()

        # Decaying learning rate
        lr = lr0 * (0.995 ** (iteration // 500))
