import numpy as np, time, math, random

def compute_c5(h):
    N = h.shape[0]
    autocorr = np.correlate(h, h, mode='full')
    mid = N - 1
    # exclude zero shift at autocorr[mid]
    # consider positive shifts 1..N-1
    min_A = np.min(autocorr[mid+1:mid+N])
    # compute bound
    c5 = (2.0/N) * (N/2 - min_A)
    return c5, min_A

import numpy as np
import time
import math

def _compute_c5_and_corr(h: np.ndarray):
    """Return C5 value and the full correlation array."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    return c5, corr

def _adjust_sum_to_target(h: np.ndarray, target: float):
    """Adjust the sum of h to be exactly target by moving mass between entries."""
    total = float(np.sum(h))
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # perform a few greedy adjustments
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase sum: pick a slot <1

            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:

            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            idx = np.random.choice(cand)
