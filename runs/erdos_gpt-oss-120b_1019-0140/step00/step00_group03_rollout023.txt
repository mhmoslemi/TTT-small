def gradient_descent_update(h, dx, target_sum, step_size):
    # compute correlation values for each shift
    # compute full correlation: corr = np.correlate(h, 1-h, mode='full')
    # Determine shift index with max; correlation array length = 2n-1; index offset = n-1.
    # s = idx - (n-1) # shift (positive means shift to right?)
    # compute gradient g_i = 1 - (h[i+s] + h[i-s]) where indices in range
    n = len(h)
    # full out of range variety
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)  # shift index
    # compute gradient
    g = np.ones(n)
    if s != 0:
        # add terms for i+s and i-s if within range
        # g_i -= h[i+s] for i where i+s valid
        # but we need to subtract both terms. Use vectorized.
        # For i in [0, n-s-1] if s>0: h[i+s] is defined.
        # For i in [0, n-s-1] (i+s valid): g[i] -= h[i+s]
        # For i in [s, n-1] (i-s valid): g[i] -= h[i-s]
        # If s negative, the ranges swap.
        if s > 0:
            g[:n-s] -= h[s:]
            g[s:] -= h[:n-s]
        else: # s < 0
            s_abs = -s
            g[s_abs:] -= h[:n-s_abs]
            g[:n-s_abs] -= h[s_abs:]
    # gradient descent step (we want minimize C5, which is max correlation)
    # Since C5 is max over all s, we are descending the active one.
    # Use step = step_size * dx maybe?
    # The objective C5 includes factor dx * correlation values. So gradient derived for correlation not multiplied by dx.
    # Actually f_s = sum_i hi - sum_i hi*hi+s => C5 = max_s f_s * dx
    # gradient of f_s: as we derived (1 - (h_{i+s}+h_{i-s}))
    # So step update: h_new = h - lr * g
    lr = step_size
    h_new = h - lr * g
    # project onto feasible set
    h_proj = project_onto_bounds_and_sum(h_new, target_sum)
    return h_proj

    i = np.random.randint(0, n)
    j = np.random.randint(0, n-1)

        j += 1  # ensure j != i

    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import time
import math
import numpy as np

def evaluate_c5(h, dx):
    """Return C5 = max convolution of h with (1-h) times dx."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_onto_bounds_and_sum(v, target):
    """
    Euclidean projection of v onto the set {0 <= x <= 1, sum(x)=target}.
    Finds lambda such that sum(clip(v - lambda, 0, 1)) = target.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def propose_move(h, max_step):
    """Return a new feasible h by transferring a delta between two coordinates."""
    n = h.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    # delta so that h[i] - delta and h[j] + delta stay within [0,1]
    low = max(h[i] - 1.0, -h[j], -max_step)
    high = min(h[i], 1.0 - h[j], max_step)
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def random_binary_vector(n):
    """Binary vector of length n with exactly n/2 ones (feasible for the problem)."""
    vec = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    vec[ones] = 1.0
    return vec

def gradient_descent_update(h, dx, target, step):
    """
    One projected gradient descent step on the active shift that attains C5.
    Gradient of the active term f_s = sum h - sum h_i h_{i+s}.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = np.argmax(corr) - (n - 1)   # shift that achieves the maximum
    if s == 0:
        g = 1.0 - 2.0 * h
    else:
        g = np.ones(n)
        if s > 0:
            g[:n - s] -= h[s:]
            g[s:]     -= h[:n - s]
        else:
            s_abs = -s
            g[s_abs:] -= h[:n - s_abs]
            g[:n - s_abs] -= h[s_abs:]
    h_new = h - step * g
    return project_onto_bounds_and_sum(h_new, target)

def local_search(h_start, dx, target, time_limit, start_temp=0.1):
    """
    Simulated‑annealing style search limited by time_limit seconds.
    Returns the best h found and its C5 value.
    """
    cur = h_start.copy()
    cur_score = evaluate_c5(cur, dx)
    best_h = cur.copy()
    best_score = cur_score
    step_max = 0.2
    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        cand = propose_move(cur, step_max)
        if cand is None:
            continue
        cand_score = evaluate_c5(cand, dx)
        # linear cooling schedule
        T = max(1e-4, start_temp * (1.0 - elapsed / time_limit))
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            accept = (np.random.rand() <
                      math.exp(-(cand_score - cur_score) / T))
        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
        # slowly shrink the maximal step size
        if it % 2000 == 0 and it > 0:
            step_max = max(step_max * 0.99, 1e-4)
        it += 1
    return best_h, best_score
