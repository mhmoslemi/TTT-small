        else:
            right = mid
    return np.clip(v + (left + right) * 0.5, lo, hi)

def _c5_value(h):
    """Return the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _initial_h(n, base_vec):
    """Create an initial vector of length n from base_vec, repeating if possible."""
    n0 = base_vec.shape[0]
    if n % n0 == 0:
        factor = n // n0
        return np.repeat(base_vec, factor)
    # otherwise linear interpolation on the interval [0,2)
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, base_vec)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # Base vector – either the provided one or a uniform fallback.
    if 'initial_h_values' in globals():
        base0 = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: a simple feasible step (half ones, half zeros) at n=200
        base0 = np.concatenate([np.ones(100), np.zeros(100)], dtype=np.float64)

    n0 = base0.shape[0]

    # Scaling factors – keep total points ≤ 1000 to stay fast.
    scaling_factors = [1, 2, 4, 5, 8]  # 8*95=760 <1000
    for factor in scaling_factors:
        n = n0 * factor
        if n > 1000:
            continue

        # Build initial candidate and enforce constraints.
        h = _initial_h(n, base0).astype(np.float64)
        target_sum = n / 2.0
        h = _project_to_bounds_and_sum(h, target_sum)

        # Evaluate initial solution.
        cur_c5 = _c5_value(h)
        if cur_c5 < best_c5:
            best_c5, best_h, best_n = cur_c5, h.copy(), n

        # Determine number of iterations based on remaining budget.
        time_left = budget_s - (time.time() - start_time)
        if time_left <= 0:
            break
        # Rough estimate: ~0.001 s per iteration at n≈500, scale conservatively.
        max_iters = int(min(6000, max(500, time_left * 8)))

        for it in range(max_iters):
            # Global timeout guard.
            if time.time() - start_time > budget_s * 0.99:
                break

            total = h.sum()
            auto = np.correlate(h, h, mode='full')          # length 2n‑1
            C_vec = total - auto                           # overlap before scaling

            # Smooth max via soft‑max – slowly raise temperature.
            # α grows from ~1 to ~15 throughout the run.
            if it < max_iters // 2:
                alpha = 1.0 + 4.0 * it / (max_iters // 2)
            else:
                alpha = 5.0 + 10.0 * (it - max_iters // 2) / (max_iters - max_iters // 2)

            maxC = np.max(C_vec)
            w = np.exp(alpha * (C_vec - maxC))
            w /= w.sum()                                   # probabilities, sum=1

            # Gradient of the smooth objective.
            conv1 = np.convolve(w, h, mode='full')
            conv2 = np.convolve(w[::-1], h, mode='full')
            offset = len(w) // 2                           # = n‑1
            grad = 1.0 - conv1[offset:offset + n] - conv2[offset:offset + n]

            # Simple step‑size schedule.
            lr = 0.5 * (1.0 - it / max_iters) + 0.01
            h = h - lr * grad

            # Project back to feasible region.
            h = _project_to_bounds_and_sum(h, target_sum)

            # Occasionally jitter a little to escape plateaus.
            if it % 250 == 0 and it > 0:
                i, j = np.random.choice(n, 2, replace=False)
                max_transfer = min(h[i], 1.0 - h[j])
                if max_transfer > 1e-12:
                    delta = max_transfer * np.random.rand() * 0.05
                    h[i] -= delta
                    h[j] += delta
                    h = _project_to_bounds_and_sum(h, target_sum)

            # Check improvement.
            cur_c5 = _c5_value(h)
            if cur_c5 < best_c5:
                best_c5, best_h, best_n = cur_c5, h.copy(), n

                # Early stop if record goal is already reached.
                if best_c5 <= 0.3808:
                    # keep searching a bit for possible further improvement,
                    # but we already have a qualifying bound.
                    pass

        if time.time() - start_time > budget_s * 0.99:
            break

    # Fallback guarantee.
    if best_h is None:
        n = 200
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_c5 = _c5_value(best_h)
        best_n = n

    return best_h, float(best_c5), int(best_n)
