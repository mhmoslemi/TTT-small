def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95  # safety margin

    # ---------- helper functions ----------
    def compute_c5(h):
        """Return (c5_bound, max_corr) for a given h vector."""
        n = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * 2.0 / n
        return float(c5), float(max_corr)

    def is_binary(arr):
        """Check if array consists only of 0/1 (within tolerance)."""
        return np.all(np.logical_or(np.isclose(arr, 0.0), np.isclose(arr, 1.0)))

    # ---------- keep track of the best solution ----------
    best_h = None
    best_c5 = np.inf

    def maybe_update_best(h):
        """Update global best if h yields a lower C5."""
        nonlocal best_h, best_c5
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # ---------- start from supplied initial solution ----------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        maybe_update_best(h0)
    except NameError:
        h0 = None

    # ---------- candidate dimensions ----------
    n_candidates = []
    if h0 is not None:
        n_candidates.append(h0.shape[0])
    for n in (60, 80, 100, 120, 150, 180, 200):
        if n % 2 == 0:
            n_candidates.append(n)

    # ---------- local optimisation for binary patterns ----------
    for n in n_candidates:
        if time.time() > deadline:
            break

        # initialise a starting vector of length n
        if h0 is not None and h0.shape[0] == n:
            cur_h = h0.copy()
        else:
            cur_h = np.zeros(n, dtype=np.float64)
            ones = n // 2
            idx = rng.choice(n, size=ones, replace=False)
            cur_h[idx] = 1.0

        maybe_update_best(cur_h)

        # attempt hill‑climbing only if the vector is binary
        if not is_binary(cur_h):
            continue

        # hill climbing parameters
        max_iter = 20000
        cur_max_corr = np.max(np.correlate(cur_h, 1.0 - cur_h, mode="full"))
        best_local_h = cur_h.copy()
        best_local_corr = cur_max_corr
        it = 0

        while it < max_iter and time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_h[i] == cur_h[j]:

            # perform swap

            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")

            if new_max_corr <= cur_max_corr:
                cur_max_corr = new_max_corr
                if new_max_corr < best_local_corr:
                    best_local_corr = new_max_corr
                    best_local_h = cur_h.copy()
                    # update global best immediately
                    maybe_update_best(cur_h)
            else:
                # revert swap
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

            it += 1

        # evaluate the best local variant we saw
        maybe_update_best(best_local_h)

    # ---------- fallback in the unlikely event no solution was found ----------
