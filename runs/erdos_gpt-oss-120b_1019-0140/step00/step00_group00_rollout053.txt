def project_bounds_sum(x, S):
    # clip to [0,1]
    x = np.clip(x, 0, 1)
    # iterative adjust
    for _ in range(10): # loop few times
        total = x.sum()
        if abs(total - S) <= 1e-12:
            break
        diff = S - total
        # Determine free indices where we can adjust: if diff>0 => not at upper bound, else not at lower bound.
        if diff > 0:
            free = (x < 1 - 1e-12)
        else:
            free = (x > 1e-12)
        free_idx = np.where(free)[0]
        if len(free_idx) == 0:
            # can't adjust further, break
            break
        # amount per free variable
        adjust = diff / len(free_idx)
        # apply but limit to bounds
        # for each free index, compute allowed: for diff>0, capacity = 1 - x[i]; adjust per is min(adjust, capacity)
        # However using vectorized: new_x = x.copy(); compute delta = np.clip(adjust, -x[free_idx], 1 - x[free_idx]); then add
        deltas = np.full_like(x[free_idx], adjust)
        if diff > 0:
            # cannot exceed capacity
            deltas = np.minimum(deltas, 1 - x[free_idx])
        else:
            deltas = np.maximum(deltas, -x[free_idx])
        x[free_idx] += deltas
    # final clip
    x = np.clip(x, 0, 1)
    return x

def project_bounds_sum(x, S):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # iterative adjustment
    for _ in range(100):
        total = x.sum()
        diff = S - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase x where x < 1
            mask = x < 1 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            # distribute equally
            inc = diff / cnt
            # limit by capacity
            delta = np.minimum(inc, 1 - x[mask])
            x[mask] += delta
        else:
            # Need to decrease where x > 0
            mask = x > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            dec = diff / cnt  # diff negative
            delta = np.maximum(dec, -x[mask])
            x[mask] += delta
    # final clip
    return np.clip(x, 0.0, 1.0)

def project_bounds_sum(v, S):
    # first clip to bounds [0,1]
    v_clipped = np.clip(v, 0, 1)
    # compute sum of clipped
    total = v_clipped.sum()
    if abs(total - S) < 1e-12:
        return v_clipped
    # If total > S: need to subtract some mass from elements >0
    # If total < S: need to add mass to elements <1
    # Use while loop:
    # maintain set of free variables not at bound after each iteration.
    # At each step compute delta = (S - total) / len(free) and adjust each free variable.
    # Clip adjustments to capacity.
    v = v_clipped.copy()
    # set mask where variables can be modified: initially all
    free = np.ones_like(v, dtype=bool) # free variables (not at bound)
    for _ in range(10):
        total = v.sum()
        diff = S - total
        if abs(diff) < 1e-12:
            break
        # Determine free indices where we can adjust in needed direction
        if diff > 0:
            # need to increase; free indices where v < 1
            free = (v < 1 - 1e-12)
        else:
            free = (v > 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        # compute adjustment amount per free variable
        adj = diff / nfree
        # apply but clip
        if diff > 0:
            caps = 1 - v[free]
            adj_vec = np.minimum(adj, caps)
        else:
            caps = -v[free]
            adj_vec = np.maximum(adj, caps)
        v[free] += adj_vec
    # final clip
    return np.clip(v, 0, 1)
```

import time
import numpy as np

# ---------- Helper Functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def gradient_corr(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the correlation term for a fixed shift k."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        # +1 - h[i+k] part
        i1 = np.arange(0, n - k)
        grad[i1] += 1.0 - h[i1 + k]
        # -h[i-k] part
        i2 = np.arange(k, n)
        grad[i2] -= h[i2 - k]
    else:
        kp = -k
        # +1 - h[i-kp] part (i >= kp)
        i1 = np.arange(kp, n)
        grad[i1] += 1.0 - h[i1 - kp]
        # -h[i+kp] part (i <= n-1-kp)
        i2 = np.arange(0, n - kp)
        grad[i2] -= h[i2 + kp]
    return grad


def project_bounds_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | 0≤x≤1, sum(x)=target_sum}.
    Simple water-filling style iteration.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0 - 1e-12
        else:
            free = x > 1e-12
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        if diff > 0:
            cap = 1.0 - x[free]
            adj = np.minimum(delta, cap)
        else:
            cap = -x[free]
            adj = np.maximum(delta, cap)
        x[free] += adj
    return np.clip(x, 0.0, 1.0)


def random_h(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Generate a random feasible vector."""
    v = rng.rand(n)
    return project_bounds_sum(v, n / 2.0)


def interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate old_h to length new_n, then enforce constraints."""
    old_n = old_h.shape[0]
    old_dx = 2.0 / old_n
    new_dx = 2.0 / new_n
    old_x = (np.arange(old_n) + 0.5) * old_dx
    new_x = (np.arange(new_n) + 0.5) * new_dx
    new_h = np.interp(new_x, old_x, old_h)
    new_h = project_bounds_sum(new_h, new_n / 2.0)
    return new_h


def triangular_h(n: int) -> np.ndarray:
    """Triangular function on [0,2] with apex 1 at x=1."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    h = project_bounds_sum(h, n / 2.0)
    return h


def local_optimize(
    h0: np.ndarray,
    start_time: float,
    budget_s: float,
    rng: np.random.RandomState,
    target_val: float = 0.3808,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the max‑correlation objective."""
    h = project_bounds_sum(h0, h0.shape[0] / 2.0)
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)

    max_iters = 5000
    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.96:
            break

        # evaluate current objective
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_val = float(corr.max() * (2.0 / n))

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < target_val:
                break

        # sub‑gradient step on the worst shift
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = gradient_corr(h, shift)

        # decreasing step size
        step = 0.5 / (1.0 + it * 0.01)
        h_new = h - step * grad
        h = project_bounds_sum(h_new, target_sum)

        # occasional random perturbation to escape plateaus
        if (it + 1) % 200 == 0:
            pert = rng.normal(scale=0.01, size=n)
            h = np.clip(h + pert, 0.0, 1.0)
            h = project_bounds_sum(h, target_sum)

    return best_h, best_val

                h0 = triangular_h(n)
            else:
                h0 = random_h(n, rng)

            # Optimize locally
            h_opt, val_opt = local_optimize(h0, start, budget_s, rng)

            # Keep the best overall
            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n
                if best_val < 0.3808:
                    # Sufficiently good; we can stop early
                    break
        if best_val < 0.3808:
            break

    # Fallback to parent if no improvement was found
    if best_h is None:
        if init_present:
            best_h = np.array(initial_h_values, copy=True)
            best_n = base_n
        else:
            best_n = 45
            best_h = np.full(best_n, 0.5)
        best_h = project_bounds_sum(best_h, best_n / 2.0)
        best_val = compute_c5(best_h)

    # Final projection to guarantee feasibility
    best_h = project_bounds_sum(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)

    return (best_h, float(best_val), int(best_n))
