def objective(h, n, penalty_factor=1e3):
    # compute max correlation
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n
    val = max_corr * dx
    # penalty for sum deviation
    sum_dev = np.sum(h) - n/2
    val += penalty_factor * (sum_dev**2)
    # may also add penalty for out of bound but we set bounds.
    return val

def optimize_h(initial_h, n, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - n/2}
    res = minimize(lambda x: np.max(np.correlate(x, 1-x, mode='full')) * (2.0/n), 
                   initial_h, 
                   method='SLSQP', 
                   bounds=bounds, 
                   constraints=cons,
                   options={'maxiter':max_iter, 'ftol':1e-9, 'disp':False})
    # After optimization, project to constraints to correct any slight violation:
    h_opt = project_to_constraints(res.x, n/2)
    # recompute bound
    corr = np.correlate(h_opt, 1-h_opt, mode='full')
    bound = np.max(corr)*(2.0/n)
    return h_opt, bound, res

def project_to_constraints(v, target):
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo+hi)/2.
        clipped = np.clip(v + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + mid, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

# ---------------------------------------------------------
# Helper: project a vector onto the box [0,1] with a given sum.
def _project_to_sum(v, target_sum, tol=1e-10, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        clipped = np.clip(v + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + mid, 0.0, 1.0)

# ---------------------------------------------------------
# Build the one‑parameter family: plateau of 1s, linear ramp to 0.
def _h_from_L(L, n):
    # Avoid division by zero; L very small creates a nearly‑rectangular shape.
    eps = 1e-12
    L = max(L, eps)
    a = 1.0 - L / 2.0            # length of the constant‑1 part
    b = a + L                    # end of the ramp
    x = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n)  # mid‑point sample
    h = np.zeros_like(x)
    # region with value 1
    mask1 = x < a
    h[mask1] = 1.0
    # linear ramp from 1 down to 0
    mask2 = (x >= a) & (x < b)
    h[mask2] = 1.0 - (x[mask2] - a) / L
    # remaining part already zero
    return h

    except Exception:
        # fallback (should resemble the helper used by the harness)
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / h.size
        return np.max(corr) * dx

# Local refinement using SLSQP (bounds + equality sum constraint).
def _refine_h(h0, max_iter=300):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target},)

    def obj(x):
        # max correlation * dx
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * (2.0 / n)

    # SLSQP may wander slightly outside the sum hyperplane; we keep it cheap.
    res = minimize(
        obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
    )
    # final projection to meet the sum exactly (tiny numerical drift)
    h_opt = _project_to_sum(res.x, target)
    return h_opt
