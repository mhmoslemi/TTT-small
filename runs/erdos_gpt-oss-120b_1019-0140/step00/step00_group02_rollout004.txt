    # Choose direction: move from i to j or vice versa randomly.
    if np.random.rand() < 0.5:
        donor, receiver = i, j
    else:
        donor, receiver = j, i
    max_transfer = min(h[donor], 1 - h[receiver])
    if max_transfer <= 0:
        return h  # no change
    eps = max_transfer * np.random.rand() * 0.5  # at most half of max
    new_h = h.copy()
    new_h[donor] -= eps
    new_h[receiver] += eps
    return new_h

import numpy as np
import time
import math

def _c5_value(h):
    """Return C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _repair(v, target_sum):
    """Clip to [0,1] and adjust linearly to hit target_sum."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    # Simple water‑filling: distribute the needed correction equally among non‑boundary entries.
    for _ in range(10):
        diff = target_sum - s
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        v[free] += diff / nfree
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős minimum overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    start = time.time()
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Initialise h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fall back: 100 points, half ones, half zeros (exact integer sum)
        n = kwargs.get("n_points", 100)
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _repair(h, target_sum)

    # ----------------------------------------------------------------------
    # Simulated annealing / stochastic local search

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # Annealing schedule parameters
    T0 = 0.025
    T = T0
    it = 0
    # main loop
    while True:
        # budget check
        elapsed = time.time() - start
        if elapsed > budget_s * 0.99:
            break

        # current correlation and the shift giving the maximum
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(corr.argmax())
        cur_c5 = corr[max_idx] * (2.0 / n)
        if cur_c5 < best_c5:

        # shift corresponding to the current worst overlap
        shift = max_idx - (n - 1)   # shift may be negative
