import numpy as np
import time
import math
import random

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a discrete step function."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))

def adjust_sum_to_target(h: np.ndarray, target: float) -> np.ndarray:
    """Scale/shift vector h so that sum(h) == target while keeping 0≤h[i]≤1."""
    # Simple deterministic fix: try to adjust a single entry.
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try a single element
    for idx in range(h.shape[0]):
        new_val = h[idx] + diff
        if 0.0 <= new_val <= 1.0:
            h[idx] = new_val
            return h
    # otherwise distribute over two indices
    if diff > 0:
        # need to increase total sum: raise one element, lower another
        inc_idx = np.argmax(1.0 - h)          # most space to increase
        dec_idx = np.argmin(h)                # smallest value to decrease
        inc_cap = 1.0 - h[inc_idx]
        dec_cap = h[dec_idx]
        inc_delta = min(inc_cap, diff + dec_cap)
        dec_delta = inc_delta - diff
        h[inc_idx] += inc_delta
        h[dec_idx] -= dec_delta
    else:
        # need to decrease total sum
        diff = -diff
        dec_idx = np.argmax(h)                # most to decrease
        inc_idx = np.argmin(1.0 - h)          # most space to increase
        dec_cap = h[dec_idx]
        inc_cap = 1.0 - h[inc_idx]
        dec_delta = min(dec_cap, diff + inc_cap)
        inc_delta = dec_delta - diff
        h[dec_idx] -= dec_delta
        h[inc_idx] += inc_delta
    # Recursively clean any remaining tiny discrepancy
    return adjust_sum_to_target(h, target)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] with small overlap C5.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise the candidate vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        # default: even length so that exact half‑unit sum is possible
        n = 200
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    target_sum = n / 2.0
    h0 = adjust_sum_to_target(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    start = time.time()
    # Annealing parameters
    T0 = 0.20
    T_end = 1e-6
    step_max_initial = 0.20

    iteration = 0
    # ------------------------------------------------------------------
    # Main simulated‑annealing loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.95:
            break

        # Linear temperature and step‑size schedules
        t_ratio = elapsed / (budget_s * 0.95)
        T = T0 * (1.0 - t_ratio) + T_end
        step_max = step_max_initial * (1.0 - t_ratio) + 0.001

        # Propose a feasible mass transfer between two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        lo = max(-cur_h[i], cur_h[j] - 1.0)
        hi = min(1.0 - cur_h[i], cur_h[j])
        if lo >= hi:
            continue

        delta = np.random.uniform(lo, hi)
        # Enforce a maximum step size
        if delta > step_max:
            delta = step_max
        elif delta < -step_max:
            delta = -step_max
        # Clamp again to stay feasible
        delta = max(lo, min(hi, delta))

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c5 = compute_c5(new_h)
