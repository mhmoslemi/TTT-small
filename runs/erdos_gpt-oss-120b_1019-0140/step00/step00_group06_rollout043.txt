def enforce_constraints(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    # compute sum
    current_sum = h.sum()
    diff = target_sum - current_sum
    # adjust using indices not at bounds
    if abs(diff) < 1e-12:
        return h
    # find indices where we can adjust in needed direction
    # iterate until diff small or cannot adjust
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to increase sum, choose indices with h < 1
            idx = np.where(h < 1.0 - 1e-12)[0]

            delta = min(-diff, h[i])
            h[i] -= delta
        current_sum = h.sum()
        diff = target_sum - current_sum

import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def enforce_constraints(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    max_iter = 10000
    itr = 0
    while abs(diff) > 1e-12 and itr < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            delta = min(diff, 1.0 - h[i])
            h[i] += delta
            diff -= delta
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            delta = min(-diff, h[i])
            h[i] -= delta
            diff += delta
        itr += 1
    return h

    else:

        best_score = compute_c5(best_h)

    # Stage 2: double resolution if time permits
    elapsed = time.time() - start
    time_remaining = budget_s - elapsed
    if time_remaining > 30:
        N2 = best_h.shape[0] * 2
        x_old = np.linspace(0.0, 2.0, best_h.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        target_sum2 = N2 / 2.0
        h2 = np.clip(h2, 0.0, 1.0)
        cur_sum2 = h2.sum()
        if cur_sum2 > 0:
            h2 = h2 * (target_sum2 / cur_sum2)
        h2 = enforce_constraints(h2, target_sum2)

        elapsed = time.time() - start
        time_remaining = budget_s - elapsed
        stage2_time = max(5.0, time_remaining * 0.9)
        h_opt2, score2 = optimize_h(h2, stage2_time)
        if score2 < best_score:
            best_h = h_opt2
            best_score = score2

    # final sanity check
    N_final = best_h.shape[0]
    best_h = enforce_constraints(best_h, N_final / 2.0)
    final_score = compute_c5(best_h)

    return best_h.astype(np.float64), float(final_score), int(N_final)
