import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = target / h.shape[0]
        return h
    scale = target / s
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        adjust = np.clip(diff, -h[idx], 1.0 - h[idx])
        h[idx] += adjust
    return h

def upsample_vector(v, new_n):
    old_n = v.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, v)

def propose_move(h, step, rng):
    n = h.shape[0]
    for _ in range(10):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        # adjust hi + delta, hj - delta
        delta_min = max(-step, -hi, hj - 1.0)
        delta_max = min(step, 1.0 - hi, hj)
        if delta_min < delta_max:
            delta = rng.random() * (delta_max - delta_min) + delta_min
            new_h = h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            return new_h
    return None

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine starting vector and dimension
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        base_n = base.shape[0]

        if target_n % 2 != 0:
            target_n += 1
        h = upsample_vector(base, target_n)

        target_n = 256
        h = rng.random(target_n)
    target_sum = target_n / 2.0
    h = repair_sum(h, target_sum)
    # Ensure sum constraint
    current_h = h.copy()
    current_val = compute_c5(current_h)
    best_h = current_h.copy()
    best_val = current_val
    # Annealing parameters
    T0 = 0.02
    T = T0
    cooling = 0.9995
    step0 = 0.2
    step = step0
    # Main loop
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        if best_val < 0.3808:
            break
        new_h = propose_move(current_h, step, rng)
        if new_h is None:
            continue
        # Compute objective
        new_val = compute_c5(new_h)
        # Accept criteria
        if new_val < current_val or rng.random() < np.exp(-(new_val - current_val) / (T + 1e-16)):
            current_h = new_h
            current_val = new_val

import numpy as np
import time

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_sum(h, target):
    """Clip to [0,1] and adjust sum to target (approximate projection)."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target / h.shape[0]
        return h
    scale = target / s
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.argmax(1.0 - h)          # most room to increase
        else:
            idx = np.argmax(h)                # largest value to decrease
        adjust = np.clip(diff, -h[idx], 1.0 - h[idx])
        h[idx] += adjust
    return h

def upsample_vector(v, new_n):
    """Linear interpolation of a periodic sample vector v onto new_n points."""
    old_n = v.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, v)

def propose_random_swap(h, step, rng):
    """Swap mass between two coordinates while preserving total sum."""
    n = h.shape[0]
    for _ in range(10):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        delta_min = max(-step, -hi, hj - 1.0)
        delta_max = min(step, 1.0 - hi, hj)
        if delta_min < delta_max:
            delta = rng.random() * (delta_max - delta_min) + delta_min
            nh = h.copy()
            nh[i] = hi + delta
            nh[j] = hj - delta
            return nh
    return None

def propose_directed_transfer(h, step, shift, rng):
    """Transfer a small amount from a high‑value site to a low‑value site
    along the worst‑shift direction (shift may be positive or negative)."""
    n = h.shape[0]
    if shift >= 0:
        max_start = n - shift
        if max_start <= 0:
            return None
        for _ in range(10):
            i = rng.integers(0, max_start)
            if h[i] > 0.0 and h[i + shift] < 1.0:
                delta_max = min(step, h[i], 1.0 - h[i + shift])
                if delta_max <= 0.0:
                    continue
                delta = rng.random() * delta_max
                nh = h.copy()
                nh[i] -= delta
                nh[i + shift] += delta
                return nh
    else:  # shift < 0
        shift_abs = -shift
        min_start = shift_abs
        if min_start >= n:
            return None
        for _ in range(10):
            i = rng.integers(min_start, n)
            j = i - shift_abs
            if h[i] > 0.0 and h[j] < 1.0:
                delta_max = min(step, h[i], 1.0 - h[j])
                if delta_max <= 0.0:
                    continue
                delta = rng.random() * delta_max
                nh = h.copy()
                nh[i] -= delta
                nh[j] += delta
                return nh
    return None

            current_h = rng.random(target_n)
            current_h = repair_sum(current_h, target_sum)
            current_val = compute_c5(current_h)
            stagnation = 0

        it += 1

    # final clean‑up
    best_h = repair_sum(best_h, target_sum)
    final_val = compute_c5(best_h)

    return best_h.tolist(), float(final_val), int(target_n)
