def compute_c5_score(h, dx):

    idx = np.argsort(vals)[::-1]
    binary = np.zeros(n, dtype=np.float64)
    binary[idx[:half]] = 1.0

def local_improve(h, dx):

    h = h.copy()
    n = len(h)
    current_score = compute_c5_score(h, dx)

    while improved:

            h[j] = 1.0
            current_score = new_score

import time
import math
import numpy as np

def compute_c5_score(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def binary_from_values(vals):
    """Convert a real-valued array to a binary one with exactly half ones."""
    n = vals.shape[0]
    half = n // 2
    idx = np.argsort(vals)[::-1]
    bin_vec = np.zeros(n, dtype=np.float64)
    bin_vec[idx[:half]] = 1.0
    return bin_vec

def local_improve(h, dx):
    """
    Greedy hill‑climbing: repeatedly perform the best improving swap
    of a 1 and a 0 until no improvement is possible.
    Returns a new binary vector and its score.
    """
    h = h.copy()
    cur_score = compute_c5_score(h, dx)
    while True:
        best_delta = 0.0
        best_swap = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                # try swapping
                h[i] = 0.0
                h[j] = 1.0
                new_score = compute_c5_score(h, dx)
                delta = new_score - cur_score
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j, new_score)
                # revert
                h[i] = 1.0
                h[j] = 0.0
        if best_swap is None:
            break
        i, j, new_score = best_swap
        h[i] = 0.0
        h[j] = 1.0
        cur_score = new_score
    return h, cur_score

    while time.time() - start_time < time_limit:
        # occasional restart from best, otherwise random binary vector
        if rng.random() < 0.2:

        else:

            cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
        cur_score = compute_c5_score(cur_h, dx)

        # inner annealing loop

            if time.time() - start_time >= time_limit:
