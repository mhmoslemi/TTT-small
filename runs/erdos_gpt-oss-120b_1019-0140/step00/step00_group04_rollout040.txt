def project_to_feasible(h_guess):
    n = len(h_guess)
    h = cvxpy.Variable(n)
    constraints = [h >= 0, h <= 1, cvxpy.sum(h) == n/2]
    prob = cvxpy.Problem(cvxpy.Minimize(cvxpy.sum_squares(h - h_guess)), constraints)
    prob.solve(solver=cvxpy.ECOS, warm_start=True, verbose=False)
    return np.array(h.value, dtype=float)
```

def compute_bound(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    # use rfft
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    return float(np.max(corr) * dx)
```

def solve_lp(n, h0=None):
    import cvxpy as cp
    h = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n/2,
        M >= 0,
    ]
    # optional warm start
    if h0 is not None:
        h.value = h0
    for s in range(1, n):
        ns = n - s
        t = cp.Variable(ns)
        constraints += [
            t >= 0,
            t <= h[:ns],
            t <= h[s:],
            t >= h[:ns] + h[s:] - 1,
            cp.sum(t) + cp.sum(h[n-s:]) >= (n/2) * (1 - M)
        ]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    return h.value, M.value

import time, random, math
import numpy as np

# Compute the C5 bound for a given step function h.
def _c5_bound(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    m = 2 * n - 1
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(G), n=m)
    return float(np.max(corr) * dx)

# Project a vector onto the feasible set {0≤h≤1, sum(h)=n/2}.
def _project_feasible(v: np.ndarray) -> np.ndarray:
    try:
        import cvxpy as cp
        n = v.size
        h = cp.Variable(n)
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h - v)),
                          [h >= 0, h <= 1, cp.sum(h) == n / 2.0])
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=20000)
        if h.value is not None:
            return np.array(h.value).astype(np.float64)
    except Exception:
        pass
    # Simple fallback: clip then adjust by a few sweeps.
    h = np.clip(v, 0.0, 1.0)
    target = h.size / 2.0
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if not free.any():
            break
        delta = diff / free.sum()
        delta = np.clip(delta, -0.1, 0.1)
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction.
    diff = target - h.sum()
    while abs(diff) > 1e-9:
        idx = random.randrange(h.size)
        if diff > 0 and h[idx] < 1.0:
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        elif diff < 0 and h[idx] > 0.0:
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Determine grid size.
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        n0 = init.size
        # Use at least 128 points for a decent resolution.
        n = max(128, 2 ** int(math.ceil(math.log2(n0))))  # power of two >= n0 & >=128
    except NameError:
        n = 256
        init = None

    # Build an initial feasible vector.
    if init is not None:
        if init.size != n:
            # Upsample linearly to the target size.
            old_x = np.linspace(0.0, 2.0, init.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(new_x, old_x, init)
        else:
            h0 = init.copy()
    else:
        # Random binary start (exactly n/2 ones).
        h0 = np.zeros(n, dtype=np.float64)
        ones = n // 2
        idx = np.random.choice(n, ones, replace=False)
        h0[idx] = 1.0

    # Project onto the feasible polytope.
    h0 = _project_feasible(h0)

    # ------------------------------------------------------------------
    # Linear programming formulation (McCormick envelopes) for the
    # minimax overlap problem.
    # ------------------------------------------------------------------
    start = time.time()
    try:
        import cvxpy as cp

        h = cp.Variable(n)
        M = cp.Variable()
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n / 2.0,
            M >= 0
        ]

        # Warm start with the feasible vector we already have.
        h.value = h0

        # Build constraints for each shift s = 1 .. n-1.
        for s in range(1, n):
            ns = n - s
            t = cp.Variable(ns)                     # t_i = h_i * h_{i+s}
            # McCormick relaxation for a product of two variables in [0,1].
            constraints += [
                t >= 0,
                t <= h[:ns],
                t <= h[s:],
                t >= h[:ns] + h[s:] - 1,
                # f_s = (sum_{i=0}^{ns-1} h_i - sum t_i) * (2/n) ≤ M
                cp.sum(t) + cp.sum(h[n - s:]) >= (n / 2.0) * (1 - M)
            ]

        prob = cp.Problem(cp.Minimize(M), constraints)
        # Solver parameters tuned for speed; ECOS handles large linear problems well.
        prob.solve(solver=cp.ECOS, max_iters=50000, abstol=1e-8,
                   reltol=1e-8, feastol=1e-8, verbose=False)
        if h.value is not None and M.value is not None:
            h_opt = np.array(h.value).astype(np.float64)
            # Ensure feasibility after the solver (tiny numerical drift may appear).
            h_opt = _project_feasible(h_opt)
            bound = _c5_bound(h_opt)
            # If time remains, try a quick local refinement.
            elapsed = time.time() - start
            if elapsed < budget_s * 0.9:
                # Simple greedy improvement: swap a 0↔1 pair if it reduces the bound.
                best_h = h_opt.copy()
                best_val = bound
                # Pre‑compute indices of zeros and ones.
                zeros = np.where(best_h < 0.5)[0]
                ones = np.where(best_h > 0.5)[0]
                for _ in range(2000):
                    if len(zeros) == 0 or len(ones) == 0:
                        break
                    i = random.choice(ones)
                    j = random.choice(zeros)
                    # trial swap
                    best_h[i] = 0.0
                    best_h[j] = 1.0
                    val = _c5_bound(best_h)
                    if val < best_val:
                        best_val = val
                        # keep the swap
                    else:
                        # revert
                        best_h[i] = 1.0
                        best_h[j] = 0.0
                h_opt = best_h
                bound = best_val
            return (h_opt, bound, n)
    except Exception:
        # If cvxpy fails for any reason, fall back to a simple simulated annealing.
        pass

    # ------------------------------------------------------------------
    # Simulated annealing fallback (continuous moves preserving sum).
    # ------------------------------------------------------------------
    h = h0.copy()
    cur_bound = _c5_bound(h)
    best_h = h.copy()
    best_bound = cur_bound
    T0 = 0.02
    Tmin = 1e-6
    decay = 0.9995
    max_step = 0.05
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        T = max(Tmin, T0 * (decay ** it))
        step = max_step * (T / T0)
        i, j = random.sample(range(n), 2)
        low = max(-h[i], -(1.0 - h[j]))
        high = min(1.0 - h[i], h[j])
        if high <= low:
            continue
        delta = random.uniform(low, high)
        delta = max(min(delta, step), -step)
        if abs(delta) < 1e-12:
            continue
        h[i] += delta
        h[j] -= delta
        new_bound = _c5_bound(h)
        if new_bound < cur_bound or random.random() < math.exp(-(new_bound - cur_bound) / T):
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        it += 1

    # Final projection and bound computation.
    best_h = _project_feasible(best_h)
    best_bound = _c5_bound(best_h)
    return (best_h, best_bound, n)
