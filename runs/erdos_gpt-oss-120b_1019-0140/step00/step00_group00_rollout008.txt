        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Minor adjustment: if sum slightly deviates due to rounding, adjust by distributing slight diff across non-bound entries
    diff = target - x.sum()

import numpy as np
import time
import random

def _project_onto_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum=x."""
    lo = -np.max(v) - 1.0          # makes all v+lo ≤ 0  → sum=0
    hi = 1.0 - np.min(v)           # makes all v+hi ≥ 1  → sum=n
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # Tiny adjustment if numerical error remains
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt:
            x[mask] += diff / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 overlap value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _active_shifts(h: np.ndarray):
    """Return the shifts (lags) attaining the maximal correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    tol = max_corr * 1e-12 + 1e-12
    idxs = np.where(corr >= max_corr - tol)[0]
    return idxs - (n - 1), max_corr

def _gradient(h: np.ndarray, shifts):
    """Gradient of the maximal correlation w.r.t. h (scaled by dx)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            g[: n - k] += (1.0 - h[k:])
            g[k:] -= h[: n - k]
        else:
            d = -k
            g[d:] += (1.0 - h[: n - d])
            g[: n - d] -= h[d:]
    g *= (2.0 / n)               # multiply by dx = 2/n
    return g

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 45
        h0 = np.full(n, 0.5, dtype=np.float64)   # uniform feasible start

    target_sum = n * 0.5                       # ∑h = n/2
    h = _project_onto_box_and_sum(h0, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # tiny safety margin
    step = 0.15                                 # initial step size

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # evaluate current state
        shifts, max_corr = _active_shifts(h)
        cur_val = max_corr * (2.0 / n)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()

        # ------------------------------------------------------------------
        # occasionally try a random pairwise mass transfer
        if random.random() < 0.04:
            pos_idx = np.where(h > 1e-8)[0]
            neg_idx = np.where(h < 1.0 - 1e-8)[0]
            if pos_idx.size and neg_idx.size:
                i = int(np.random.choice(pos_idx))
                j = int(np.random.choice(neg_idx))
                eps_max = min(h[i], 1.0 - h[j])
                eps = np.random.uniform(0.0, min(0.04, eps_max))
                if eps > 1e-12:
                    h_trial = h.copy()
                    h_trial[i] -= eps
                    h_trial[j] += eps
                    val_trial = _c5_value(h_trial)
                    if val_trial < cur_val:
                        h = h_trial
                        step = max(step, 0.01)   # reinvigorate step size
                        continue

        # ------------------------------------------------------------------
        # gradient descent on the active lag(s)
        g = _gradient(h, shifts)
        improved = False
        # simple back‑tracking line search
        for _ in range(5):
            h_new = _project_onto_box_and_sum(h - step * g, target_sum)
            new_val = _c5_value(h_new)
            if new_val < cur_val - 1e-12:
                h = h_new
                step *= 1.07
                improved = True
                break
            step *= 0.55
        if not improved:
            # if step became too tiny, reset it
            if step < 1e-7:
                step = 0.05

    # ----------------------------------------------------------------------
    # final fine‑tuning with very small steps (optional)
    for _ in range(30):
        shifts, max_corr = _active_shifts(best_h)
        g = _gradient(best_h, shifts)

        val_try = _c5_value(h_try)
        if val_try < best_val - 1e-13:
            best_val = val_try
            best_h = h_try
        else:
            break

    return best_h, best_val, n
