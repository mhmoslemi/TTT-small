import time
import random
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Compute C₅ = max_k ∫ h(x) (1‑h(x+k)) dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero‑padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _make_random_binary(n: int, rng) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones (n must be even)."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    rng.shuffle(h)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] that lowers the Erdős C₅ bound.

        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the achieved C₅ value
        n_points : int, the number of samples used (len(h_values))
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # -------------- main optimization parameters ------------------------

    half = n_points // 2

    cur_val = _c5_value(cur_h)

    # keep indices of ones and zeros for O(1) swaps
    ones = [i for i, v in enumerate(cur_h) if v == 1.0]

    # simple simulated‑annealing schedule
    T_initial = 0.02

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:          # leave a safety margin
            break
        # early exit if we already beat the target
        if best_val <= target:
            break

        # propose a swap of a random 1 with a random 0
        i_idx = random.randrange(len(ones))
        j_idx = random.randrange(len(zeros))
        i_pos = ones[i_idx]

        # perform the swap
        cur_h[i_pos] = 0.0
