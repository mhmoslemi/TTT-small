def compute_c5(h):

import time, random, math
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a given h (numpy 1‑D array)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using full mode (zero padding)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_to_sum_one_half(v, total):
    """Clip v to [0,1] and then adjust to have sum == total."""
    v = np.clip(v, 0.0, 1.0)
    # iterative redistribution of the remaining deficit/surplus
    for _ in range(10):
        s = v.sum()
        diff = total - s
        if abs(diff) < 1e-12:
            break
        interior = (v > 0.0) & (v < 1.0)
        cnt = interior.sum()
        if cnt == 0:
            break
        v[interior] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction on a random interior element if needed
    if abs(v.sum() - total) > 1e-9:
        interior_idx = [i for i in range(v.size) if 0.0 < v[i] < 1.0]
        if interior_idx:
            i = random.choice(interior_idx)
            v[i] = max(0.0, min(1.0, v[i] + (total - v.sum())))
    return v


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (sampled uniformly) that minimizes
    max_k ∫ h(x)(1−h(x+k))dx.
    Returns (list_of_h, c5_value, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # --- Determine problem size and initial vector -----------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)

    total_needed = n / 2.0  # because ∫h = 1 ⇒ sum(h)=n/2

    # If the supplied initial vector does not satisfy the constraints,
    # build a feasible one (linear decreasing shape then project).
    if (
        h0.shape[0] != n
        or np.any(h0 < 0.0)
        or np.any(h0 > 1.0)
        or abs(h0.sum() - total_needed) > 1e-8
    ):
        # linear decreasing from 1 to 0 (or a cosine shape) and scale
        lin = 1.0 - np.arange(n, dtype=float) / float(n - 1)
        h = _project_to_sum_one_half(lin, total_needed)
    else:
        # Use the provided construction after clipping to the box.
        h = np.clip(h0, 0.0, 1.0)
        h = _project_to_sum_one_half(h, total_needed)

    # --- Optimization by simulated annealing ---------------------
    cur_h = h.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    start_time = time.time()
    time_limit = budget_s * 0.95  # leave a safety margin

    # Temperature schedule parameters
    T0 = best_score * 0.5 if best_score > 0 else 0.1
    it = 0
    # number of iterations without improvement before a gentle restart
    stagnation_counter = 0
    max_stagnation = 50000

    while time.time() - start_time < time_limit:
        # propose a move that preserves the sum exactly
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        if hi <= 0.0 or hj >= 1.0:
            continue
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0.0:
            continue
        delta = random.random() * max_delta
        new_h = cur_h.copy()
        new_h[i] = hi - delta
        new_h[j] = hj + delta

        new_score = _compute_c5(new_h)
        delta_score = new_score - cur_score

        # linear cooling
        elapsed = time.time() - start_time
        T = max(1e-8, T0 * (1.0 - elapsed / time_limit))

        if delta_score < 0.0 or random.random() < math.exp(-delta_score / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
                stagnation_counter = 0
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1

        # occasional gentle restart to escape long plateaus
        if stagnation_counter >= max_stagnation:
            # perturb the best solution slightly and re‑project
            perturb = np.random.uniform(-0.03, 0.03, size=n)
            cur_h = np.clip(best_h + perturb, 0.0, 1.0)
            cur_h = _project_to_sum_one_half(cur_h, total_needed)
            cur_score = _compute_c5(cur_h)
            stagnation_counter = 0

        # early exit if we have already reached the target bound
        if best_score * (2.0 / n) <= 0.3808:   # multiply by dx to compare to constant
            break

        it += 1

    # Return the best found solution
    c5_bound = best_score
    return (best_h.tolist(), float(c5_bound), n)
