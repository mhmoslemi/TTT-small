def subgradient_opt(h0, time_left):
    h = h0.copy()
    n = len(h)
    target = n/2
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    step0 = 0.2
    iteration = 0
    while time_left() > 0 and iteration < 20000:
        c = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(c)
        shift = idx - (n - 1)
        # compute subgradient
        g = np.ones_like(h)
        # shift ==0 case:
        if shift == 0:
            g -= 2 * h
        else:
            if shift > 0:
                # vectors shifted forward
                g[:n-shift] -= h[shift:]
                g[shift:] -= h[:n-shift]
            else:
                s = -shift
                g[s:] -= h[:n-s]
                g[:n-s] -= h[s:]
        # step size schedule: step = step0 / sqrt(iteration+1)
        step = step0 / math.sqrt(iteration + 1)
        y = h - step * g
        # project onto capped simplex sum target
        h = project_onto_capped_simplex(y, target)
        # compute new C5
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        # optional: if new_c5 not improving, maybe increase step? we just continue
        iteration += 1
    return best_h, best_c5

def project_onto_capped_simplex(v, target):
    # solves min ||x-v||^2 subject to sum(x)=target, 0<=x<=1
    # using bisection on lambda
    n = len(v)
    # initial lower bound: we need sum_{i} clip(v_i + lam, 0, 1) >= target
    # As lam-> -inf, clip->0, sum=0 <= target. As lam-> +inf, clip->1, sum=n >= target.
    lam_low = -np.max(v) - 1.0
    lam_high = 1 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v + (lam_low + lam_high) / 2.0, 0, 1)
```

def greedy_binary(n):
    h = np.zeros(n, dtype=float)
    target = n//2
    for _ in range(target):
        best_pos = None
        best_val = float('inf')
        zeros = np.where(h == 0)[0]
        # evaluate each candidate
        for p in zeros:
            h[p] = 1
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_pos = p
            h[p] = 0
        # set best position
        h[best_pos] = 1
    return h, compute_c5(h)
```

import time

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(dx * np.max(corr))

def project_onto_capped_simplex(v, target):
    # Euclidean projection onto {x: sum(x)=target, 0<=x<=1}
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def greedy_binary(n):
    h = np.zeros(n, dtype=float)
    target = n // 2
    for _ in range(target):
        zeros = np.where(h == 0)[0]
        best_val = float('inf')
        best_p = -1
        for p in zeros:
            h[p] = 1.0
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_p = p
            h[p] = 0.0
        h[best_p] = 1.0
    return h, compute_c5(h)

def anneal_binary(h0, time_remaining_func):

    n = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    for it in range(max_iter):

        j = np.random.choice(zeros_idx)
        h[i], h[j] = h[j], h[i]

def subgradient_opt(h0, time_remaining_func):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step0 = 0.2
    it = 0
    while time_remaining_func() > 0 and it < 20000:
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        g = np.ones_like(h)
        if shift == 0:
            g -= 2.0 * h
        elif shift > 0:
            g[:n - shift] -= h[shift:]
            g[shift:] -= h[:n - shift]
        else:  # shift < 0
            s = -shift
            g[s:] -= h[:n - s]
            g[:n - s] -= h[s:]
        step = step0 / math.sqrt(it + 1)
        y = h - step * g

        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5

        it += 1
