                donor_candidates = np.where(h > 1e-12)[0]
                recip_candidates = np.where(h < 1.0 - 1e-12)[0]
                if donor_candidates.size == 0 or recip_candidates.size == 0:
                    continue
                donor = int(rng.choice(donor_candidates))
                recipient = int(rng.choice(recip_candidates))

        max_delta = min(step, h[donor], 1.0 - h[recipient])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        if delta <= 0:
            continue

        new_h = h.copy()
        new_h[donor] -= delta
        new_h[recipient] += delta
        new_h = np.clip(new_h, 0.0, 1.0)

        # re‑balance the total sum (tiny drift may appear)
        sum_diff = new_h.sum() - target_sum
        if abs(sum_diff) > 1e-12:
            free = np.where((new_h > 1e-12) & (new_h < 1.0 - 1e-12))[0]
            if free.size:
                corr = sum_diff / free.size
                new_h[free] -= corr
                new_h = np.clip(new_h, 0.0, 1.0)
            else:
                # fallback: adjust donor directly
                new_h[donor] -= sum_diff
                new_h = np.clip(new_h, 0.0, 1.0)

        new_cost = _evaluate_c5(new_h)

        elapsed = time.time() - start
        temperature = temp_init * max(0.0, 1.0 - elapsed / budget_s)

        accept = False
        if new_cost < cur_cost - 1e-12:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_cost - cur_cost) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            h = new_h
            cur_cost = new_cost
            if new_cost < best_cost - 1e-12:
                best_cost = new_cost
                best_h = new_h.copy()

        step = max(step * decay, min_step)

    # ----- final projection to exact feasibility -----------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    while abs(diff) > 1e-12:
        idx = rng.integers(0, n)
        if diff > 0 and best_h[idx] < 1.0:
            add = min(diff, 1.0 - best_h[idx])
            best_h[idx] += add
            diff -= add
        elif diff < 0 and best_h[idx] > 0.0:
            sub = min(-diff, best_h[idx])
            best_h[idx] -= sub
            diff += sub

    best_cost = _evaluate_c5(best_h)
    return (best_h.tolist(), float(best_cost), n)
