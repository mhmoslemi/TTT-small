def project_onto_box_and_sum(y, target, max_iter=50, tol=1e-9):
    n = len(y)
    # function that returns sum_i clip(y_i - lam, 0, 1)
    low = -np.max(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = np.sum(x)
        if np.abs(s - target) <= tol:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    # after loop return last x
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

def worst_shift_and_subgradient(h):
    n = len(h)
    # correlation without dx scaling
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    grad = np.zeros_like(h)
    if lag >= 0:
        grad[:n-lag] += (1 - h[lag:])
        grad[lag:] -= h[:n-lag]
    else:
        d = -lag
        grad[d:] += (1 - h[:n-d])
        grad[:n-d] -= h[d:]
    # return corr value and gradient scaled later
    return lag, corr[idx], grad

import numpy as np
import time

def project_to_sum_and_box(y, target):
    # bisection on lambda
    low = -np.max(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(50):
        lam = (low + high) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(y - (low + high) * 0.5, 0.0, 1.0)

def evaluate_c5_h(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def worst_shift_and_subgradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    grad = np.zeros_like(h)
    if lag >= 0:
        # first factor contributions
        grad[:n-lag] += (1.0 - h[lag:])
        # second factor contributions
        grad[lag:] -= h[:n-lag]
    else:
        d = -lag
        grad[d:] += (1.0 - h[:n-d])
        grad[:n-d] -= h[d:]
    # return lag, corr value, gradient (without dx)
    return lag, corr[idx], grad

def random_feasible(n, target):
    y = np.random.rand(n)
    return project_to_sum_and_box(y, target)

import time
import numpy as np

def _project_to_simplex_box(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the set {x | sum x = target, 0 <= x <= 1}.
    Uses bisection on a Lagrange multiplier lambda.
    """
    # lambda bounds that guarantee sum >= target and sum <= target
    low = -np.max(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(50):
        lam = (low + high) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def _evaluate_c5(h: np.ndarray, dx: float) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def _worst_shift_and_subgrad(h: np.ndarray):
    """
    Compute the lag that attains the maximal overlap integral and the
    gradient of that integral w.r.t. h (without the dx factor).
    Returns (lag, max_corr_value, gradient_array).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    grad = np.zeros_like(h)

    if lag >= 0:
        # first factor contribution
        grad[: n - lag] += (1.0 - h[lag:])
        # second factor contribution
        grad[lag:] -= h[: n - lag]
    else:
        d = -lag
        grad[d:] += (1.0 - h[: n - d])
        grad[: n - d] -= h[d:]

    return lag, corr[idx], grad


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n and project onto the feasible set."""
    y = np.random.rand(n)
    return _project_to_simplex_box(y, target)
