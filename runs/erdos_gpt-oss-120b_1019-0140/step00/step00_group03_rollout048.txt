import numpy as np, time, math, random

def compute_c5_fft(h):
    # returns C5 via FFT
    n = h.shape[0]
    N = 2 * n
    h_fft = np.fft.rfft(h, n=N)
    comp_fft = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(h_fft * comp_fft, n=N)
    conv = conv[:2 * n - 1]
    max_val = conv.max()
    if max_val < 0:
        max_val = 0.0
    return max_val * (2.0 / n)

def compute_c5(h):
    # simple check using np.correlate for accuracy
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

import numpy as np
import time
import math
import random

def correlation_fft(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using zero‑padded FFT."""
    n = h.shape[0]
    N = 2 * n                     # zero‑pad length ≥ 2·n‑1
    h_fft = np.fft.rfft(h, n=N)
    comp_fft = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(h_fft * comp_fft, n=N)[: 2 * n - 1]
    return conv

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the discretised step function."""
    n = h.shape[0]
    corr = correlation_fft(h)
    return float(corr.max()) * (2.0 / n)

def propose_targeted_swap(h: np.ndarray, lag: int):
    """Find a 1↔0 swap that directly reduces the mismatch count for `lag`.
    Returns a tuple (i_one, i_zero) or None if no such complementary pair exists."""
    n = h.shape[0]
    if lag >= 0:
        rng = np.arange(0, n - lag)
        mask_one = (h[rng] == 1) & (h[rng + lag] == 0)   # 1 followed by 0
        mask_zero = (h[rng] == 0) & (h[rng + lag] == 1)  # 0 followed by 1
        idx_one = rng[mask_one]
        idx_zero = rng[mask_zero]
    else:
        p = -lag
        rng = np.arange(p, n)
        mask_one = (h[rng] == 1) & (h[rng - p] == 0)
        mask_zero = (h[rng] == 0) & (h[rng - p] == 1)
        idx_one = rng[mask_one]
        idx_zero = rng[mask_zero]
    if idx_one.size and idx_zero.size:
        i_one = np.random.choice(idx_one)
        i_zero = np.random.choice(idx_zero)
        return int(i_one), int(i_zero)
    return None

def _adjust_to_binary(v: np.ndarray, target_ones: int) -> np.ndarray:
    """Convert any vector to a binary vector with exactly `target_ones` ones."""
    h = (v > 0.5).astype(np.float64)
    cur = int(h.sum())
    if cur > target_ones:
        excess = cur - target_ones
        ones = np.where(h == 1)[0]
        drop = np.random.choice(ones, size=excess, replace=False)
        h[drop] = 0.0
    elif cur < target_ones:
        deficit = target_ones - cur
        zeros = np.where(h == 0)[0]
        add = np.random.choice(zeros, size=deficit, replace=False)
        h[add] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step function for the Erdős C₅ bound.

    Returns
    -------
    h_values : np.ndarray
        Optimised values on the uniform grid.
    c5_bound : float
        Achieved C₅ value (max overlap integral).
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and initialise a binary candidate
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.copy(initial_h_values).astype(np.float64)
        n = init.shape[0]
        if n % 2 == 1:                     # enforce even length for binary split
            init = np.concatenate([init, np.zeros(1, dtype=np.float64)])
            n += 1
    else:
        n = 120                               # a modest even size when no parent is given
        init = None

    if n % 2 == 1:                           # safety net
        n += 1
        if init is not None:
            init = np.concatenate([init, np.zeros(1, dtype=np.float64)])

    target_ones = n // 2

    if init is not None:
        h = _adjust_to_binary(init, target_ones)
    else:
        h = np.zeros(n, dtype=np.float64)
        h[np.random.choice(n, size=target_ones, replace=False)] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # ----------------------------------------------------------------------
    # Simulated‑annealing loop with targeted swaps
    # ----------------------------------------------------------------------
    start = time.time()
    time_limit = float(budget_s) - 0.2   # small safety margin

    no_improve = 0
    T_start = 0.08
    T_end = 1e-4

    while True:
        now = time.time()
        if now - start > time_limit:
            break

        # Correlation of current state – gives current C₅ and the worst lag
        corr = correlation_fft(cur_h)
        cur_c5 = float(corr.max()) * (2.0 / n)
        lag = int(corr.argmax() - (n - 1))

        # Try a targeted swap that reduces mismatches for the worst lag
        swap = propose_targeted_swap(cur_h, lag)
        if swap is None:
            # fall back to a completely random 1↔0 swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i_one = np.random.choice(ones)
            i_zero = np.random.choice(zeros)
        else:
            i_one, i_zero = swap

        # Perform the swap
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0

        # Evaluate the new configuration
        new_corr = correlation_fft(cur_h)
        new_c5 = float(new_corr.max()) * (2.0 / n)

        # Temperature schedule (linear decay)
        elapsed = now - start
        frac = elapsed / time_limit
        T = T_start * (1.0 - frac) + T_end * frac

        # Metropolis acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert the swap
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
            no_improve += 1

        # Periodic random restart if stagnating
        if no_improve > 3000 and (now - start) < time_limit - 2.0:
            cur_h = np.zeros(n, dtype=np.float64)
            cur_h[np.random.choice(n, size=target_ones, replace=False)] = 1.0
            cur_c5 = compute_c5(cur_h)
            no_improve = 0

    # ----------------------------------------------------------------------
    # Final evaluation using the straightforward np.correlate for consistency
    # ----------------------------------------------------------------------
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
