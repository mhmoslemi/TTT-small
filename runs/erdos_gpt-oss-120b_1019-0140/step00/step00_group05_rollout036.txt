import time
import numpy as np

def _project_bounds_sum(v: np.ndarray, total: float,
                        lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """Projection of v onto the set {x: lower<=x<=upper, sum(x)=total}."""
    # binary search for lambda such that sum(clip(v - lam, lower, upper)) == total
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, lower, upper)
    # tiny correction for numerical drift
    diff = total - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > lower + 1e-12) & (x < upper - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, lower, upper)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    np.random.seed(seed)

    # ----- initial construction -----

        n0 = base.size
        # upsample to gain flexibility (factor chosen so n <= 500)
        max_n = 500
        factor = max(2, min(6, max_n // n0))
        h = np.repeat(base, factor)
    except NameError:
        # no parent construction – start from a random feasible vector
        n0 = 200
        factor = 1
        h = np.random.rand(n0)

    n = h.size
    total = n / 2.0          # required sum of h_i
    h = _project_bounds_sum(h, total)

    dx = 2.0 / n

    # evaluate initial bound
    cross = np.correlate(h, h, mode='full') * dx
    best_c5 = 1.0 - np.min(cross)
    best_h = h.copy()

    start = time.time()
    lr = 0.5                # learning rate
    decay = 0.9995
    it = 0

    while time.time() - start < budget_s - 0.2:
        # current correlation and bound
        cross = np.correlate(h, h, mode='full') * dx
        c5 = 1.0 - np.min(cross)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # worst shift (minimum autocorrelation)
        idx_min = np.argmin(cross)
        shift = idx_min - (n - 1)

        # gradient that raises the autocorrelation at this shift
        grad = np.zeros_like(h)
        if shift == 0:

        elif shift > 0:

        else:  # shift < 0
