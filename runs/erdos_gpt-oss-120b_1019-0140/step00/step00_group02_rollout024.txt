import numpy as np
import time, math

def compute_c5(h):
    n = h.shape[0]
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / n)

def project_to_box_sum(v, target):
    # Euclidean projection onto [0,1]^n intersect sum=v target
    x = np.clip(v, 0.0, 1.0)
    # iterative water-filling
    for _ in range(20):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # free indices
        free = (x > 1e-12) & (x < 1 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust, break
            break
        delta = diff / n_free
        x[free] += delta
        # clip again in case overshoot
        x = np.clip(x, 0.0, 1.0)
    # final correction if still off due to rounding
    # distribute remaining diff to any index that can move
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        # adjust the first element within bounds if possible
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return x

def gradient_for_shift(h, k):
    n = h.shape[0]
    # term1 = 1 - h[i+k] (or 1 if out of bounds)
    # term2 = h[i-k] (or 0 if out of bounds)
    if k >= 0:
        term1 = np.empty_like(h)
        term1[:n - k] = 1.0 - h[k:]
        term1[n - k:] = 1.0
        term2 = np.empty_like(h)
        term2[:k] = 0.0
        term2[k:] = h[:n - k]
    else:
        kp = -k
        term1 = np.empty_like(h)
        term1[:kp] = 1.0
        term1[kp:] = 1.0 - h[:n - kp]
        term2 = np.empty_like(h)
        term2[:n - kp] = h[kp:]

    return term1 - term2

def upscale(arr, new_n):
    old_n = len(arr)
    if old_n == new_n:
        return np.asarray(arr, dtype=float).copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, arr)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rnd = np.random.RandomState(seed)
    start_time = time.time()
    # Determine number of points
    # Prefer a power of two >= 128
    base_n = 256
    # Use initial_h_values if present and large enough
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = max(base_n, len(init_arr))
    else:
        n = base_n
    # Ensure even
    if n % 2 != 0:
        n += 1
    # Build initial vector
    if 'initial_h_values' in globals():
        h = upscale(initial_h_values, n)
    else:
        h = rnd.rand(n)
    # Project to feasible set (box [0,1] and sum = n/2)
    target_sum = n / 2.0
    h = project_to_box_sum(h, target_sum)
    # Initialize best
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5
    # Parameters for stochastic gradient descent
    eta0 = 0.5          # initial step size
    decay = 1e-5        # step size decay per iteration
    temp0 = 0.1         # initial temperature for Metropolis acceptance
    temp_decay = 1e-5
    iter_count = 0
    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.97:
            break
        # full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        max_idxs = np.where(corr == max_val)[0]
        # compute gradient averaged over worst shifts
        grad = np.zeros_like(h)
        for idx in max_idxs:
            k = idx - (n - 1)
            grad += gradient_for_shift(h, k)
        grad /= len(max_idxs)
        # step size schedule
        eta = eta0 / (1.0 + decay * iter_count)
        # temperature schedule
        temp = max(temp0 * math.exp(-temp_decay * iter_count), 1e-12)
        # propose move

        cand_c5 = compute_c5(h_candidate)

        accept = False

            accept = True
        else:
            # probability
            prob = math.exp(-(cand_c5 - current_c5) / temp)
            if rnd.rand() < prob:

        if accept:
            h = h_candidate

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for discretized h."""
    n = h.shape[0]
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c) * (2.0 / n))

def project_to_box_sum(v, target):
    """
    Euclidean projection of v onto the set {x | 0≤x_i≤1 , sum x = target}.
    Simple water‑filling iteration.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        n_free = int(free.sum())
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:

                diff -= add

                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:

def gradient_for_shift(h, k):
    """
    Gradient of the overlap for a single shift k:
    g_i = (1‑h_{i+k}) (if i+k in range) – h_{i‑k} (if i‑k in range)

    term1 = np.zeros_like(h)

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)

    return best_h.tolist(), float(best_c5), int(n)
