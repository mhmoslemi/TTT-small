import time, math
import numpy as np

def _compute_c5(h):
    """Return (raw_max, C5) for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h (full mode)
    corr = np.correlate(h, 1 - h, mode='full')
    raw = float(corr.max())
    return raw, raw * dx

def _hill_climb_once(n, rng, deadline, max_swaps=2000):
    """Start from a random balanced binary vector of length n and perform greedy swaps."""
    h = np.zeros(n, dtype=int)
    h[: n // 2] = 1
    rng.shuffle(h)
    raw, _ = _compute_c5(h)

    swaps = 0
    while swaps < max_swaps and time.time() < deadline:
        # pick a random 1 and a random 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i] = 0
        h[j] = 1
        new_raw, _ = _compute_c5(h)

        # keep if not worse, otherwise revert
        if new_raw <= raw:
            raw = new_raw
        else:
            h[i] = 1
            h[j] = 0

        swaps += 1

    return h.copy(), raw

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s * 0.95  # safety margin

    best_h = None
    best_raw = float('inf')
    best_n = None

    # --- use provided initial solution if it exists ---

        n0 = init_h.shape[0]
        # enforce sum constraint (tiny numeric adjustments)

        cur_sum = init_h.sum()
        if abs(cur_sum - target_sum) > 1e-8:
            init_h = init_h * (target_sum / cur_sum)
            init_h = np.clip(init_h, 0.0, 1.0)
            # re‑normalize exactly (distribute remaining mass uniformly)
            diff = target_sum - init_h.sum()
            if abs(diff) > 1e-12:
                # add/subtract a tiny amount from a random position that stays feasible
                idx = rng.integers(0, n0)
                adj = np.clip(diff, -init_h[idx], 1.0 - init_h[idx])
                init_h[idx] += adj
        raw0, c5_0 = _compute_c5(init_h)
        best_h = init_h.copy()

    except NameError:
