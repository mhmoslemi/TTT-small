    f = np.correlate(h, 1 - h, mode='full') * dx
    # compute softmax weights

def project_capped_simplex(v, S):
    # Clip to [0,1]
    x = np.clip(v, 0, 1)
    # Compute current sum
    current = x.sum()
    if abs(current - S) < 1e-9:
        return x
    # Use water-filling:
    # We'll adjust values within bounds to meet sum.
    # If current < S, need to increase some values (those not at 1)
    # If current > S, need to decrease some values (those not at 0)
    # In each iteration, distribute diff equally among free indices.
    # Use loop limit.
    max_iter = 10
    for _ in range(max_iter):
        diff = S - x.sum()
        if abs(diff) < 1e-9:
            break
        if diff > 0:
            # increase
            free = (x < 1)
            n_free = free.sum()
            if n_free == 0:
                # cannot increase, break
                break
            inc = diff / n_free
            # Add inc but ensuring not exceed caps:
            x[free] += inc
            x = np.clip(x, 0, 1)
        else:
            free = (x > 0)
            n_free = free.sum()
            if n_free == 0:
                break
            dec = diff / n_free
            x[free] += dec  # diff negative, so dec negative
            x = np.clip(x, 0, 1)
    return x

import time
import numpy as np

def _project_capped_simplex(v, target_sum):
    """Project v onto the set {x| 0<=x_i<=1, sum x_i = target_sum}."""
    x = np.clip(v, 0.0, 1.0)
    diff = target_sum - x.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0
            n_free = free.sum()
            if n_free == 0:
                break
            inc = diff / n_free
            x[free] += inc
            np.clip(x, 0.0, 1.0, out=x)
        else:
            free = x > 0.0
            n_free = free.sum()
            if n_free == 0:
                break
            dec = diff / n_free
            x[free] += dec
            np.clip(x, 0.0, 1.0, out=x)
        diff = target_sum - x.sum()
    np.clip(x, 0.0, 1.0, out=x)
    return x

def _random_feasible_h(n, rng):
    """Generate a random feasible h (0<=h_i<=1, sum h_i = n/2)."""
    target = n / 2.0
    h = rng.random(n)
    if h.sum() == 0:
        h[:] = 0.5
    else:
        h = h * (target / h.sum())
    h = _project_capped_simplex(h, target)
    return h

def _c5_of(h, dx):
    """Compute the C5 value for a given h."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _gradient(h, dx, alpha):
    """
    Gradient of a smooth approximation of max_k ∫ h (1-h shifted) dx.
    Returns (grad, max_f) where max_f is the actual (unsmoothed) maximum.
    """
    n = h.shape[0]
    f = np.correlate(h, 1 - h, mode='full') * dx
    max_f = np.max(f)
    # smooth max via softmax
    exp_vals = np.exp(alpha * (f - max_f))
    w = exp_vals / exp_vals.sum()
    offset = n - 1
    sum_fwd = np.zeros(n)
    sum_bwd = np.zeros(n)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - offset
        if s >= 0:
            # shift left by s (forward)
            sum_fwd[:n - s] += weight * h[s:]
            # shift right by s (backward)
            sum_bwd[s:] += weight * h[:n - s]
        else:
            s_abs = -s
            # shift right by s_abs (forward)
            sum_fwd[s_abs:] += weight * h[:n - s_abs]
            # shift left by s_abs (backward)
            sum_bwd[:n - s_abs] += weight * h[s_abs:]
    grad = dx * (1.0 - sum_fwd - sum_bwd)
    return grad, max_f

    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise h
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        h = init_arr.copy()
    except Exception:
        n = 70
        h = _random_feasible_h(n, rng)

    target_sum = n / 2.0
    h = _project_capped_simplex(h, target_sum)
    dx = 2.0 / n

    best_h = h.copy()
    best_score = _c5_of(best_h, dx)

    lr = 1.0                # learning rate
    alpha = 40.0            # smoothing parameter for max
    stagn = 0
    max_stagn = 3000
    it = 0

    while time.time() - start < budget_s:

        new_score = _c5_of(h_new, dx)

            stagn = 0
            lr = max(lr * 0.5, 1e-3)

        # early stop if we meet the target bound
        if best_score <= 0.3808:
            break
