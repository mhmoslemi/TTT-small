import numpy as np
import time

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete step h."""
    n = len(h)
    dx = 2.0 / n
    # cross-correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_h(h):
    """Clip to [0,1] and enforce exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-12:
        if diff > 0:
            # need to increase sum: add to the element with most headroom
            cap = 1.0 - h
            idx = np.argmax(cap)
            inc = min(diff, cap[idx])
            h[idx] += inc
        else:
            # need to decrease sum: subtract from the largest element
            cap = h
            idx = np.argmax(cap)
            dec = min(-diff, cap[idx])
            h[idx] -= dec
    # tiny residual fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # distribute to first element safely
        h[0] = np.clip(h[0] + residual, 0.0, 1.0)
    return h

def _propose_move(h, max_step, rng):
    """Return a new vector by moving a small amount of mass between two indices."""
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # feasible delta range respecting [0,1] and max_step
    lo = max(-h[i], -(1.0 - h[j]), -max_step)
    hi = min(1.0 - h[i], h[j], max_step)
    if lo >= hi:
        return None
    delta = rng.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on a uniform grid.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretization points.
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------------
    # Initial vector
    init_vec = globals().get('initial_h_values', None)
    if init_vec is not None:
        h = np.asarray(init_vec, dtype=np.float64).copy()
        h = _repair_h(h)
    else:
        n0 = 200
        h = np.full(n0, 0.5, dtype=np.float64)
    n = len(h)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.99  # slight safety margin

    # Step size schedule (large to fine)
    step_schedule = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for max_step in step_schedule:
        # Perform as many moves as we can before time runs out
        while True:
            if time.time() - start_time > time_limit:
                break
            cand = _propose_move(best_h, max_step, rng)
            if cand is None:
                continue
            cand_score = _compute_c5(cand)
            if cand_score + 1e-12 < best_score:

            # occasional re‑repair to counteract rounding drift
