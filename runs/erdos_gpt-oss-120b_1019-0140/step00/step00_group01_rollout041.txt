def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine n points
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n_init = init_h.shape[0]
        dx_init = 2.0 / n_init
        c5_init = np.max(np.correlate(init_h, 1 - init_h, mode='full')) * dx_init
        best_h = init_h.copy()
        best_c5 = float(c5_init)
        best_n = n_init
        # optionally treat initial as candidate for other n.
    else:
        best_h = None
        best_c5 = float('inf')
        best_n = None

    # Choose n for optimization
    n = 200  # ensure even
    if n % 2 != 0:
        n += 1

    dx = 2.0 / n

    # Initialize binary vector
    # Use simple method: choose random positions for ones
    h = np.zeros(n, dtype=float)
    # Balanced: ensure exactly n/2 ones
    ones_positions = rng.choice(n, size=n//2, replace=False)
    h[ones_positions] = 1.0

    # initial c5
    corr = np.correlate(h, 1 - h, mode='full')
    cur_c5 = np.max(corr) * dx
    cur_h = h.copy()
    # update best if better
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = cur_h.copy()
        best_n = n

    # Precompute ones_set (list of indices)
    # We will update sets after each accepted swap
    # Use Python sets for O(1) membership
    ones_set = set(ones_positions.tolist())
    zeros_set = set(range(n)) - ones_set

    # Simulated annealing parameters
    T0 = 0.5
    Tf = 0.001
    # approximate number of iterations we aim: maybe 20000
    # schedule: T = max(Tf, T0 * (0.9995 ** iteration))

    iteration = 0
    while time.time() - start_time < budget_s * 0.97:
        iteration += 1
        # Choose random swap
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        # Swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        # Compute new C5
        corr_new = np.correlate(cur_h, 1 - cur_h, mode='full')
        new_c5 = np.max(corr_new) * dx

        # Temperature schedule
        T = max(Tf, T0 * (0.9995 ** iteration))
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # accept with prob
            if rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True

        if accept:
            # commit
            cur_c5 = new_c5
            # update sets
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
            # update best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                best_n = n
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        # optionally, after some iterations, try a random restart if no improvement for many steps
        # but given time may not needed.

    # Return best solution found (prefer smallest n if multiple best?) But choose best_h with best_n.

    # If best_h is None (no solution), fallback to initial or uniform 0.5 fractions.
    if best_h is None:
        # fallback: uniform h = 0.5 on all points size n_init if there is initial n else 2 points.
        best_n = n if n is not None else 2
        best_h = np.full(best_n, 0.5)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / best_n)

    # Return result as numpy array, best_c5, best_n
    return (best_h, float(best_c5), int(best_n))

        else:
            # probabilistic uphill move
            prob = math.exp(-(new_c5 - cur_c5) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur_h.copy(), cur_c5, n
                if best_c5 <= target:
                    break
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    # ---------- fallback in the unlikely event nothing was found ----------
    if best_h is None:
        # uniform half function as a safe fallback (h = 0.5 everywhere)
        best_n = 2
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, float(best_c5), int(best_n))
```
