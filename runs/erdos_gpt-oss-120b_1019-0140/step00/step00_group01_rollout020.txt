import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discrete h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / n))

def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary (plus a 0.5 if n is odd) vector with sum = n/2."""
    h = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        ones = n // 2
        idx = rng.choice(n, size=ones, replace=False)
        h[idx] = 1.0
    else:
        m = n // 2
        idx_one = rng.choice(n, size=m, replace=False)
        h[idx_one] = 1.0
        remaining = np.setdiff1d(np.arange(n), idx_one)
        h[rng.choice(remaining, size=1)] = 0.5
    return h

def _project_to_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum = len(h)/2."""
    target = h.size / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute the residual greedily
    if diff > 0:
        # need to add mass
        while diff > 1e-12:
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = cand[0]
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
    else:
        # need to remove mass
        diff = -diff
        while diff > 1e-12:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            idx = cand[0]
            sub = min(diff, h[idx])
            h[idx] -= sub
            diff -= sub
    return h

    start = time.time()
