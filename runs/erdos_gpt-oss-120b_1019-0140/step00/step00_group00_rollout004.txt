def compute_g_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    # compute g
    g = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    # compute weights
    max_g = np.max(g)  # for numerical stability in exp
    # optionally subtract max_g
    exp_term = np.exp(beta * (g - max_g))
    w = exp_term / np.sum(exp_term)
    # gradient:
    grad = np.zeros_like(h)
    # for each shift index s (lag from -(n-1) to n-1)
    for idx_shift, weight in enumerate(w):
        k = idx_shift - (n - 1)  # shift
        if weight == 0:
            continue
        if k >= 0:
            # a: (1 - h[j+k]) for j=0..n-1-k
            grad[:n - k] += weight * (1 - h[k:])
            # b: h[j-k] for j=k..n-1
            grad[k:] -= weight * h[:n - k]
        else:
            # k negative: shift left
            kp = -k
            # For a: (1 - h[j - kp]) when j - kp >=0 => j>=kp. So for j=kp..n-1 => a = 1 - h[j - kp]
            grad[kp:] += weight * (1 - h[:n - kp])
            # For b: h[j + kp] for j + kp < n => j <= n - kp -1. So j=0..n - kp -1 => subtract weight * h[j + kp]
            grad[:n - kp] -= weight * h[kp:]
    grad *= dx
    return g, grad

import time
import numpy as np

def project_to_box_simplex(v, target_sum):
    """Project vector v onto the set {x | 0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    return x

def compute_g_and_grad(h, beta=30.0):
    """Return correlation values g and gradient of smooth max of g."""
    n = h.shape[0]
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode="full") * dx          # length 2n-1
    # smooth‑max weights
    m = np.max(g)
    exp_term = np.exp(beta * (g - m))
    w = exp_term / np.sum(exp_term)                         # sum to 1
    # gradient (vectorized over shifts)
    grad = np.zeros_like(h)
    shift_center = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        k = idx - shift_center                               # lag
        if k >= 0:
            # contribution from (1‑h_{i+k})
            grad[: n - k] += weight * (1.0 - h[k:])
            # contribution from -h_i·h_{i+k}
            grad[k:]      -= weight * h[: n - k]
        else:
            kp = -k
            grad[kp:]     += weight * (1.0 - h[: n - kp])
            grad[: n - kp] -= weight * h[kp:]
    grad *= dx
    return g, grad

def interpolate_h(old_h, old_n, new_n):
    """Linear interpolation of a step function to a finer grid."""
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def binary_candidate(h, target_sum):
    """Produce a binary vector with exactly target_sum ones (target_sum integer)."""
    n = h.shape[0]
    k = int(target_sum)
    idx = np.argsort(-h)                # descending order
    bin_vec = np.zeros(n, dtype=np.float64)
    bin_vec[idx[:k]] = 1.0
    return bin_vec

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- initial construction -------------------------------------------------
    init_arr = globals().get("initial_h_values", None)
    if init_arr is not None:
        h0 = np.array(init_arr, dtype=np.float64)
        n0 = h0.shape[0]
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_box_simplex(h0, n0 / 2.0)
    else:
        n0 = 45
        h0 = project_to_box_simplex(rng.random(n0), n0 / 2.0)

    # --- resolution schedule --------------------------------------------------
    max_n = 256
    resolutions = [n0]
    while resolutions[-1] * 2 <= max_n:
        resolutions.append(resolutions[-1] * 2)

    best_h = None
    best_n = None
    best_c5 = np.inf

    for n in resolutions:
        # time budget for this resolution
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:
            break
        time_left = budget_s - elapsed
        # allocate iterations (roughly 15 000 per level, limited by remaining time)
        max_iters = int(min(15000, time_left * 20))
        max_iters = max(2000, max_iters)

        # start point for this grid
        if best_h is not None:
            h_start = interpolate_h(best_h, best_n, n)
        else:
            if n == n0:
                h_start = h0.copy()
            else:
                h_start = project_to_box_simplex(rng.random(n), n / 2.0)

        target_sum = n / 2.0
        base_step = 0.5 * (2.0 / n)          # proportional to dx

        # --- multiple random restarts ---------------------------------------
        for restart in range(3):
            if restart == 0:
                h = h_start.copy()
            else:
                h = project_to_box_simplex(rng.random(n), target_sum)

            for it in range(max_iters):
                if time.time() - start_time > budget_s - 0.5:
                    break
                beta = 2.0 + 48.0 * it / max_iters            # schedule from 2 → 50
                g, grad = compute_g_and_grad(h, beta=beta)
                c5_val = np.max(g)
                if c5_val < best_c5:
                    best_c5 = float(c5_val)
                    best_h = h.copy()
                    best_n = n
                step = base_step * (1.0 - it / max_iters)
                h = h - step * grad
                h = project_to_box_simplex(h, target_sum)

            if time.time() - start_time > budget_s - 0.5:
                break

        # try a binary version when n is even (exact half‑integral sum)
        if n % 2 == 0:
            bin_h = binary_candidate(best_h, target_sum)
            g_bin = np.correlate(bin_h, 1.0 - bin_h, mode="full") * (2.0 / n)
            c5_bin = np.max(g_bin)
            if c5_bin < best_c5:
                best_c5 = float(c5_bin)
                best_h = bin_h
                best_n = n

    # fallback if nothing succeeded
    if best_h is None:
        best_h = h0.copy()
        best_n = n0
        best_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full") * (2.0 / best_n)))

    # final feasibility projection
    best_h = project_to_box_simplex(best_h, best_n / 2.0)
    final_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full") * (2.0 / best_n)))

    return (best_h, final_c5, int(best_n))
