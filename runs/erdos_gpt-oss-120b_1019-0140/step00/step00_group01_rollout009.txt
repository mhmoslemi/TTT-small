import numpy as np, time, scipy.optimize as opt

def compute_c5(h, dx):
    # ensure h within [0,1], hopefully.
    # compute correlation
    conv = np.correlate(h, 1 - h, mode='full')
    # max
    max_val = np.max(conv) * dx
    return max_val

def projection_to_bounds_and_sum(v, target):
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return w

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c5 = compute_c5(h, dx)
    # If sum not close to n/2, add penalty
    target_sum = n / 2.0
    violation = abs(h.sum() - target_sum)
    penalty = 1e4 * violation
    return c5 + penalty

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    from scipy.optimize import differential_evolution

    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Determine n_points
    try:
        n = len(initial_h_values)
        use_initial = True

        n = 100  # default
        use_initial = False

    dx = 2.0 / n
    target_sum = n / 2.0

    # Initialize population for DE
    bounds = [(0.0, 1.0)] * n

    # Helper to generate initial population
    def init_population():
        popsize = 15  # default from DE
        # generate random population
        pop = np.random.rand(popsize, n)
        # project each to satisfy sum constraint (approx)
        for i in range(popsize):
            pop[i] = projection_to_bounds_and_sum(pop[i], target_sum)
        # replace first individual with provided initial solution if available
        if use_initial:
            init_vec = np.array(initial_h_values, dtype=float)
            # Ensure shape matches n
            if init_vec.shape[0] != n:
                # fallback: truncate or pad
                init_vec = init_vec[:n]
                if init_vec.shape[0] < n:
                    # pad with zeros
                    init_vec = np.pad(init_vec, (0, n - init_vec.shape[0]), constant_values=0)
            # Scale to target sum if needed
            sum_init = init_vec.sum()
            if abs(sum_init - target_sum) > 1e-9:

            pop[0] = init_vec
        return pop

    # Callback for time limit
    def callback(xk, convergence):

    # Run DE with time limit
    # Determine maxiter maybe large
    maxiter = 2000

    result = differential_evolution(

import numpy as np, time
from scipy.optimize import differential_evolution

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_sum_bounds(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return w

def evaluate_h(v, dx, target):
    # ensure bounds
    h = np.clip(v, 0.0, 1.0)
    c5 = compute_c5(h, dx)
    # sum penalty
    deviation = abs(h.sum() - target)
    penalty = 1e5 * deviation
    return c5 + penalty

def local_refine(best_h, dx, target, max_seconds):
    best = best_h.copy()
    best_score = compute_c5(best, dx)
    start = time.time()
    while time.time() - start < max_seconds:
        # propose perturbation
        new = best + np.random.normal(scale=0.05, size=best.shape)
        new = np.clip(new, 0.0, 1.0)
        new = project_to_sum_bounds(new, target)
        score = compute_c5(new, dx)
        if score < best_score:
            best = new
            best_score = score
    return best, best_score

import time
import numpy as np
from scipy.optimize import differential_evolution


def _c5_value(h, dx):
    """Return the C5 overlap value for a given step function h."""
    # correlation of h with (1-h) over all integer shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_to_sum_bounds(v, target):
    """
    Project a vector v onto the hypercube [0,1]^n while enforcing sum(v) = target.
    Implements a bisection on the Lagrange multiplier lambda for the formula
        w_i = clip(v_i - lambda, 0, 1)
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _objective(x, dx, target):
    """
    DE objective: C5 value plus a heavy penalty for violating the integral constraint.
    """
    h = np.clip(x, 0.0, 1.0)
    c5 = _c5_value(h, dx)
    deviation = abs(h.sum() - target)
    # penalty factor large enough to keep the sum essentially exact
    return c5 + 1e5 * deviation


def _local_refine(best_h, dx, target, budget):
    """
    Simple stochastic hill climbing using Gaussian perturbations.
    Runs for at most ``budget`` seconds.
    """
    best = best_h.copy()
    best_score = _c5_value(best, dx)
    start = time.time()
    while time.time() - start < budget:
        # propose a new point by small random moves
        cand = best + np.random.normal(scale=0.05, size=best.shape)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_to_sum_bounds(cand, target)
        score = _c5_value(cand, dx)
        if score < best_score:
            best, best_score = cand, score
    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function for the Erdős C5 constant.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values : np.ndarray of length n_points with entries in [0,1]
        c5_bound : float, the achieved C5 value
        n_points : int, the number of discretisation points used
    """
    np.random.seed(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and optionally a starting vector
    # ------------------------------------------------------------------
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)  # provided by harness
        n = init_vec.size
        use_initial = True
    except NameError:
        # No parent construction; fall back to a modest size
        n = 100
        init_vec = None
        use_initial = False

    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Build initial population for Differential Evolution
    # ------------------------------------------------------------------
    popsize = 30
    population = np.random.rand(popsize, n)
    for i in range(popsize):
        population[i] = _project_to_sum_bounds(population[i], target_sum)

    if use_initial:
        # replace the first individual with the supplied construction (projected if needed)
        proj = init_vec.copy()
        if abs(proj.sum() - target_sum) > 1e-12:
            proj = _project_to_sum_bounds(proj, target_sum)
        population[0] = np.clip(proj, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Objective wrapper for DE
    # ------------------------------------------------------------------
    def obj(x):
        return _objective(x, dx, target_sum)

    # ------------------------------------------------------------------
    # Callback to respect the overall time budget
    # ------------------------------------------------------------------
    def de_callback(xk, convergence):
        return (time.time() - start_total) > (budget_s - 5.0)

    # ------------------------------------------------------------------
    # Run Differential Evolution
    # ------------------------------------------------------------------
    result = differential_evolution(
        func=obj,
        bounds=[(0.0, 1.0)] * n,
        init=population,
        maxiter=10000,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=de_callback,
        polish=False,
        disp=False,
    )

    best = _project_to_sum_bounds(np.clip(result.x, 0.0, 1.0), target_sum)

    # ------------------------------------------------------------------
    # Use any remaining time for a quick local refinement
    # ------------------------------------------------------------------
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed - 0.5
    if remaining > 0.5:
        best, _ = _local_refine(best, dx, target_sum, remaining)

    final_c5 = _c5_value(best, dx)

    return best, final_c5, n
