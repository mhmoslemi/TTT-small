def propose_random_transfer(h):
    n = len(h)
    # choose two different indices i, j
    i = np.random.randint(n)
    j = np.random.randint(n-1)
    if j >= i:
        j += 1
    # decide direction: either from i -> j or from j -> i
    # want to move from location with higher h to lower maybe.
    # compute feasible eps in both directions and choose randomly among them.
    # We'll compute max_eps_pos = min(h[i], 1 - h[j]) to move from i to j
    max_eps_pos = min(h[i], 1 - h[j])
    # also direction reverse:
    max_eps_rev = min(h[j], 1 - h[i])
    possible = []
    if max_eps_pos > 0:
        possible.append((i, j, max_eps_pos))
    if max_eps_rev > 0:
        possible.append((j, i, max_eps_rev))
    if not possible:
        return None
    i_move, j_move, max_eps = random.choice(possible)
    eps = np.random.rand() * max_eps
    return i_move, j_move, eps

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _random_transfer(h: np.ndarray):
    """Propose a feasible epsilon transfer between two indices."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    # two possible directions
    max1 = min(h[i], 1.0 - h[j])          # i -> j
    max2 = min(h[j], 1.0 - h[i])          # j -> i
    candidates = []
    if max1 > 0:
        candidates.append((i, j, max1))
    if max2 > 0:
        candidates.append((j, i, max2))
    if not candidates:
        return None
    i_move, j_move, max_eps = random.choice(candidates)
    eps = np.random.rand() * max_eps
    return i_move, j_move, eps
