    # Solve min ||x - v||^2 s.t. sum(x) = target, lower <= x <= upper
    n = v.shape[0]
    # Ensure feasible: lower*n <= target <= upper*n
    # Use bisection on lambda
    # We find lambda such that sum(clip(v - lambda, lower, upper)) = target
    # Initially lambda_low = -np.max(v) - 1 (ensures clip to upper)
    # lambda_high = np.max(v) - lower + 1 (ensures clip to lower)
    # Actually need bracket: for lambda large positive, clip to lower (0); sum will be low (<target). For lambda large negative, clip to upper (1); sum high.
    lamb_low = -np.max(v) - 2.0
    lamb_high = np.max(v) + 2.0
    for _ in range(50):

        x = np.clip(v - lamb_mid, lower, upper)
        s = x.sum()

            lamb_low = lamb_mid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

import numpy as np
import time
import random

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary_vector(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n//2 ones."""
    ones = n // 2
    v = np.zeros(n, dtype=float)
    v[:ones] = 1.0
    np.random.shuffle(v)
    return v

def greedy_hill_climb(h: np.ndarray, max_iters: int, target_c5: float = None):
    """Greedy swap hill‑climb; only accepts improvements."""
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stagnation = 0
    for _ in range(max_iters):
        i, j = np.random.randint(n, size=2)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]                     # swap a 1 with a 0
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            stagnation = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if target_c5 is not None and best_c5 <= target_c5:
                    break
        else:
            h[i], h[j] = h[j], h[i]                 # revert
            stagnation += 1
        if stagnation > 2000:
            break
    return best_h, best_c5

def simulated_annealing(h: np.ndarray, time_limit: float, target_c5: float = None):
    """Temperature–guided swap search for a given time budget."""
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    base_T = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        frac = elapsed / time_limit
        T = max(1e-6, base_T * (1.0 - frac))
        if new_c5 < cur_c5 or np.exp(-(new_c5 - cur_c5) / T) > np.random.rand():
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if target_c5 is not None and best_c5 <= target_c5:
                    break
        else:
            h[i], h[j] = h[j], h[i]                 # revert
    return best_h, best_c5
