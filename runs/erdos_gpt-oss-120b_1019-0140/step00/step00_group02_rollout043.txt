def solve_lp_circular(n):
    import numpy as np, scipy.sparse as sp, scipy.optimize as opt
    total_vars = n + n*n + 1
    t_idx = total_vars - 1
    h_start = 0
    z_start = n
    # Build lists to fill data.
    rows = []
    cols = []
    data = []
    b_ub = []

    # constraint: z_{i,k} <= h_i
    for i in range(n):
        for k in range(n):
            row_idx = len(b_ub)
            # z variable index
            z_idx = z_start + i*n + k
            rows.append(row_idx)
            cols.append(z_idx)
            data.append(1.0)
            rows.append(row_idx)
            cols.append(i)
            data.append(-1.0)
            b_ub.append(0.0)
    # constraint: z_{i,k} <= h_{j}
    for i in range(n):
        for k in range(n):
            row_idx = len(b_ub)  # new row
            z_idx = z_start + i*n + k
            j = (i + k) % n
            rows.append(row_idx)
            cols.append(z_idx)
            data.append(1.0)
            rows.append(row_idx)
            cols.append(j)
            data.append(-1.0)
            b_ub.append(0.0)
    # shift constraints: - sum_i z_{i,k} + u <= 0
    for k in range(n):
        row_idx = len(b_ub)
        # for each i
        for i in range(n):
            z_idx = z_start + i*n + k
            rows.append(row_idx)
            cols.append(z_idx)
            data.append(-1.0)
        rows.append(row_idx)
        cols.append(t_idx)
        data.append(1.0)
        b_ub.append(0.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(b_ub), total_vars))
    # Equality: sum_i h_i = n/2
    A_eq = sp.lil_matrix((1, total_vars))
    for i in range(n):
        A_eq[0, i] = 1.0
    b_eq = np.array([n/2.0])

    # bounds
    bounds = [(0,1)] * n + [(0,1)] * (n*n) + [(0, None)]

    # objective: minimize -u
    c = np.zeros(total_vars)
    c[t_idx] = -1.0

    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq.tocsr(), b_eq=b_eq,
                       bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")

    h = res.x[h_start:h_start+n]
    u = res.x[t_idx]
    c5 = 1 - (2*u)/n
    return h, c5

def solve_mip_binary(n):
    import cvxpy as cp
    import numpy as np
    # Variables
    x = cp.Variable(n, integer=True)
    y = cp.Variable((n, n), integer=True)  # y[i,k] corresponds to product x[i] * x[(i+k)%n]
    u = cp.Variable(integer=True)
    constraints = []
    # x bounds
    constraints += [x >= 0, x <= 1]
    # sum x = n/2
    constraints.append(cp.sum(x) == n/2)
    # constraints linking y to x
    for i in range(n):
        for k in range(n):
            j = (i + k) % n
            constraints.append(y[i, k] <= x[i])
            constraints.append(y[i, k] <= x[j])
            constraints.append(y[i, k] >= x[i] + x[j] - 1)
    # shift sum constraints
    for k in range(n):
        constraints.append(cp.sum(y[:, k]) >= u)
    # u bounds
    constraints.append(u >= 0)
    # objective
    obj = cp.Maximize(u)
    prob = cp.Problem(obj, constraints)
    result = prob.solve(solver='ECOS_BB', verbose=False, warm_start=True, max_iters=500000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise RuntimeError("MIP did not find optimal solution, status: " + prob.status)
    h = np.array(x.value).flatten()
    u_val = float(u.value)
    c5 = 1 - (2 * u_val) / n
    return h, c5

def solve_lp_circular(n):
    import numpy as np
    import scipy.sparse as sp
    import scipy.optimize as opt

    total_vars = n + n*n + 1
    idx_u = total_vars - 1
    # Compute number of constraints:
    # 2 constraints per pair for z <= hi and z <= h_j: 2*n*n
    # shift constraints: n
    # total inequalities = 2*n*n + n
    # We'll allocate as lil and fill row by row.
    rows = 0
    A_ub = sp.lil_matrix((2*n*n + n, total_vars))
    b_ub = np.zeros(2*n*n + n)

    # First set of constraints: z_{i,k} - h_i <= 0
    for i in range(n):
        for k in range(n):
            idx_z = n + i*n + k
            A_ub[rows, idx_z] = 1.0
            A_ub[rows, i] = -1.0
            b_ub[rows] = 0.0
            rows += 1

    # Second set: z_{i,k} - h_j <= 0
    for i in range(n):
        for k in range(n):
            idx_z = n + i*n + k
            j = (i + k) % n
            A_ub[rows, idx_z] = 1.0
            A_ub[rows, j] = -1.0
            b_ub[rows] = 0.0
            rows += 1

    # Shift constraints: - sum_i z_{i,k} + u <= 0
    for k in range(n):
        for i in range(n):
            idx_z = n + i*n + k
            A_ub[rows, idx_z] = -1.0
        A_ub[rows, idx_u] = 1.0
        b_ub[rows] = 0.0
        rows += 1

    # Equality constraint: sum_i h_i = n/2
    A_eq = sp.lil_matrix((1, total_vars))
    for i in range(n):
        A_eq[0, i] = 1.0
    b_eq = np.array([n/2.0])

    # Variable bounds
    bounds = [(0,1)] * n + [(0,1)] * (n*n) + [(0, None)]

    c = np.zeros(total_vars)
    c[idx_u] = -1.0

    res = opt.linprog(c, A_ub=A_ub.tocsr(), b_ub=b_ub,
                      A_eq=A_eq.tocsr(), b_eq=b_eq,
                      bounds=bounds, method='highs')
    if res.status != 0:
        raise RuntimeError(f"LP failed status {res.status}")
    h = res.x[:n]
    return h

def solve_binary_mip(n):
    import cvxpy as cp, numpy as np
    total_vars = n + n*n + 1  # but we will define separate
    x = cp.Variable(n, integer=True)
    y = cp.Variable(n*n, integer=True)
    u = cp.Variable(integer=True)

    # y bounds
    constraints += [y >= 0, y <= 1]

    # product constraints
    for i in range(n):
        for k in range(n):

            idx = i*n + k
            constraints.append(y[idx] <= x[i])
            constraints.append(y[idx] <= x[j])

    for k in range(n):

        indices = [i*n + k for i in range(n)]

    obj = cp.Maximize(u)

    for solver in ['CBC', 'SCIP', 'ECOS_BB']:

    u_val = int(round(u.value))
    h = x_val.astype(float)  # convert to float for correlation (0/1).

import numpy as np, time, cvxpy as cp

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def solve_binary_mip(n, time_limit):
    # MIP to maximize minimal autocorrelation sum for binary h
    # n must be even to have integer half-sum
    if n % 2 != 0:
        raise ValueError("n must be even")
    half = n // 2
    # variables
    x = cp.Variable(n, integer=True)
    y = cp.Variable(n * n, integer=True)  # y[i*n + k] corresponds to x_i * x_{i+k mod n}
    u = cp.Variable(integer=True)

    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == half,
        u >= 0,
        y >= 0,
        y <= 1,
    ]

    # product constraints
    for i in range(n):
        for k in range(n):
            j = (i + k) % n
            idx = i * n + k
            constraints.append(y[idx] <= x[i])
            constraints.append(y[idx] <= x[j])
            constraints.append(y[idx] >= x[i] + x[j] - 1)

    # shift constraints: sum_i y[i*n + k] >= u for each shift k
    for k in range(n):
        constraints.append(cp.sum(y[k::n]) >= u)

    # objective: maximize u
    objective = cp.Maximize(u)

    prob = cp.Problem(objective, constraints)

    # try several solvers, stop when solved
    for solver in ('CBC', 'SCIP', 'ECOS_BB'):
        try:
            prob.solve(solver=solver, verbose=False, max_iters=200000, time_limit=time_limit)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        raise RuntimeError(f"MIP not solved, status={prob.status}")

    x_val = np.rint(x.value).astype(float)  # 0/1 values
    return x_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # evaluate provided initial_h_values if any

        init_h = np.asarray(initial_h_values, dtype=float)
        init_c5 = compute_c5(init_h)
        best_h = init_h.copy()
        best_c5 = float(init_c5)
        best_n = init_h.shape[0]
    except Exception:
        pass

    # Candidate grid sizes (even numbers)
    candidates = [44, 46, 48, 50, 52, 56]
    for n in candidates:
        if time.time() - start > budget_s * 0.9:
            break
        # allocate a small portion of the remaining budget to each MIP
        remaining = budget_s - (time.time() - start)
        # reserve at most 200 seconds per solve

        try:
            h = solve_binary_mip(n, time_limit=t_limit)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = float(c5)
            best_h = h.copy()
            best_n = n
        if best_c5 <= 0.3808:

        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

import time
import numpy as np
import cvxpy as cp

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap constant for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def solve_binary_mip(n: int, time_limit: float = 120.0) -> np.ndarray:
    """
    Solve the binary max‑min overlap problem for an even length n.
    Returns a 0‑1 array h of length n.
    """
    if n % 2:
        raise ValueError("n must be even for binary formulation")
    half = n // 2

    # decision variables
    x = cp.Variable(n, integer=True)                     # h values (0/1)
    y = cp.Variable(n * n, integer=True)                # products x_i * x_{i+k mod n}
    u = cp.Variable(integer=True)                        # lower bound on each shift sum

    cons = [
        x >= 0,
        x <= 1,
        cp.sum(x) == half,
        u >= 0,
        y >= 0,
        y <= 1,
    ]

    # linearisation of the product y_{i,k}=x_i*x_{i+k}
    for i in range(n):
        for k in range(n):
            j = (i + k) % n
            idx = i * n + k
            cons.append(y[idx] <= x[i])
            cons.append(y[idx] <= x[j])
            cons.append(y[idx] >= x[i] + x[j] - 1)

    # each cyclic shift must have total product at least u
    for k in range(n):
        cons.append(cp.sum(y[k::n]) >= u)

    prob = cp.Problem(cp.Maximize(u), cons)

    # try a few solvers; ECOS_BB is always available, others may be faster if installed
    for s in ("CBC", "SCIP", "ECOS_BB"):
        try:
            prob.solve(solver=s, verbose=False, max_iters=200_000, time_limit=time_limit)
            if prob.status in ("optimal", "optimal_inaccurate"):
                break
        except Exception:
            continue

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"MIP not solved – status {prob.status}")

    # retrieve a 0‑1 solution
    h = np.rint(x.value).astype(float)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function whose max overlap is ≤ 0.3808."""
    np.random.seed(seed)
    start = time.time()

    best_h = None

    best_n = None

    # use the supplied construction if present

        init_c5 = compute_c5(init_h)
        best_h, best_c5, best_n = init_h.copy(), float(init_c5), init_h.size
    except Exception:
        pass

    # try several even grid sizes (binary formulation)
    candidates = [44, 46, 48, 50, 52, 56]
    for n in candidates:
        if time.time() - start > budget_s * 0.9:
            break
        # give each MILP at most a few minutes of CPU time
        remaining = budget_s - (time.time() - start)

    if best_h is None:
