    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    # compute list of zero and one indices
    zeros = np.where(h == 0)[0]

    # Temperature schedule maybe simple.
    while time.time() - start < time_budget:

        i = rng.choice(zeros)

        # swap bits
        h[i], h[j] = 1, 0
        # recompute score
        score = compute_c5(h)
        if score < best_score - 1e-12:  # improvement

            best_h = h.copy()
            # update zero/one lists
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
        else:
            # revert
            h[i], h[j] = 0, 1
        # optionally random restart/perturbation: after certain iterations, try random swap

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def greedy_optimize(h, time_budget):
    start = time.time()

    best_h = h.copy()
    best_score = compute_c5(h)

        ones = np.where(best_h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = np.random.randint(len(zeros))
        j = np.random.randint(len(ones))
        # swap in copy
        new_h = best_h.copy()
        new_h[zeros[i]] = 1
        new_h[ones[j]] = 0
        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-12:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

def binary_hill_climb(h, time_budget):
    import time, numpy as np, random, math
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    # precompute indices:
    while time.time() - start < time_budget:
        # choose random indices
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]

def continuous_annealing(h, time_budget):
    import time, numpy as np, math, random
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    # Temperature schedule
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(0.001, 0.1 * (1 - frac) + 0.001 * frac)   # maybe
        # pick two indices random
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Determine delta bound
        delta_max_pos = min(1 - cur_h[i], cur_h[j])
        delta_max_neg = min(cur_h[i], 1 - cur_h[j])
        # combine both: pick delta uniform in [-delta_max_neg, delta_max_pos]
        if delta_max_pos + delta_max_neg <= 0:
            continue
        delta = (np.random.rand() * (delta_max_pos + delta_max_neg)) - delta_max_neg
        if abs(delta) < 1e-12:
            continue
        cur_h[i] += delta
        cur_h[j] -= delta
        # ensure within bounds
        # compute new score
        new_score = compute_c5(cur_h)
        # acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta
    return best_h, best_score

def solve_binary_mip(n, time_limit):
    import cvxpy as cp, numpy as np, time
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    half = n // 2
    constraints = [cp.sum(h) == half, t >= 0]
    for k in range(-n+1, n):
        if k >= 0:
            i_idx = np.arange(0, n - k)
        else:
            i_idx = np.arange(-k, n)
        if i_idx.size == 0:
            continue
        j_idx = i_idx + k
        w = cp.Variable(i_idx.size, boolean=True)
        constraints.append(w <= h[i_idx])
        constraints.append(w <= h[j_idx])
        constraints.append(w >= h[i_idx] + h[j_idx] - 1)
        s = cp.sum(w)
        constraints.append(t >= half - s)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Provide time limit if solver supports
    params = {}
    # GLPK time limit: 'tmlim' in seconds as an integer.
    # In cvxpy, solver options passed as dictionary.
    # We'll attempt to set "glpk" options if supported.
    try:
        prob.solve(solver=cp.GLPK_MI, glpk_opts={'tmlim': int(time_limit)})
    except Exception:
        prob.solve(solver=cp.GLPK_MI)
    if h.value is None:
        return None, None
    h_sol = np.rint(np.array(h.value)).astype(float)
    # adjust sum if needed via rounding positions
    diff = int(h_sol.sum() - half)
    if diff != 0:
        # flip diff bits
        ones_idx = np.where(h_sol == 1)[0]
        zeros_idx = np.where(h_sol == 0)[0]
        if diff > 0:
            h_sol[ones_idx[:diff]] = 0
        else:
            h_sol[zeros_idx[: -diff]] = 1
    t_val = float(t.value)
    return h_sol, t_val

import time, math
import numpy as np

# ----- helper functions -----------------------------------------------------

def compute_c5(h):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def binary_hill_climb(start_h, time_budget, rng):
    """Swap 0/1 entries while the maximum overlap improves."""
    start = time.time()
    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]

    while time.time() - start < time_budget:
        ones = np.where(cur_h == 1.0)[0]
        zeros = np.where(cur_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_c5(cur_h)

        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0

        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
    return best_h, best_score


def continuous_annealing(start_h, time_budget, rng):
    """Move mass between two coordinates while respecting [0,1] and sum constraint."""
    start = time.time()
    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.shape[0]

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(1e-4, 0.1 * (1 - frac) + 0.001 * frac)  # linear cooling

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        hi, hj = cur_h[i], cur_h[j]
        # feasible delta keeping both in [0,1]
        max_pos = min(1.0 - hi, hj)
        max_neg = min(hi, 1.0 - hj)
        if max_pos + max_neg <= 0.0:
            continue
        delta = rng.random() * (max_pos + max_neg) - max_neg

        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_score = compute_c5(cur_h)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = hi
            cur_h[j] = hj
    return best_h, best_score


def solve_binary_mip(n, time_limit):
    """Attempt a MILP for binary variables using cvxpy. Returns (h, t) or (None, None)."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    half = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == half, t >= 0]

    for k in range(-n + 1, n):
        if k >= 0:
            i_idx = np.arange(0, n - k)
        else:
            i_idx = np.arange(-k, n)
        if i_idx.size == 0:
            continue
        j_idx = i_idx + k
        w = cp.Variable(i_idx.size, boolean=True)
        constraints.append(w <= h[i_idx])
        constraints.append(w <= h[j_idx])
        constraints.append(w >= h[i_idx] + h[j_idx] - 1)
        constraints.append(t >= half - cp.sum(w))

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
        except Exception:
            return None, None

    if h.value is None or t.value is None:
        return None, None

    h_sol = np.rint(np.array(h.value)).astype(float)
    # fix possible rounding drift
    diff = int(h_sol.sum() - half)
    if diff != 0:
        ones = np.where(h_sol == 1)[0]
        zeros = np.where(h_sol == 0)[0]
        if diff > 0:
            h_sol[ones[:diff]] = 0.0
        else:
            h_sol[zeros[: -diff]] = 1.0
    return h_sol, float(t.value)

    """Return (h_values, c5_bound, n_points) within the given time budget."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- try a binary MILP (takes only a fraction of the budget) ----
    elapsed = time.time() - start_time

    if remaining > 30 and n <= 120:  # keep problem size modest
        h_bin, t_val = solve_binary_mip(n, remaining * 0.5)
        if h_bin is not None:
            score_bin = compute_c5(h_bin)
            if score_bin < best_score:
                best_h, best_score = h_bin, score_bin

    # ---- binary hill climbing (if any time left) ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5:
        t_bin = remaining * 0.3
        h_opt, s_opt = binary_hill_climb(best_h, t_bin, rng)
        if s_opt < best_score:
            best_h, best_score = h_opt, s_opt
