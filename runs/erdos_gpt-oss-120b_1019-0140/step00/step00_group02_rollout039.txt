def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        factor = target_sum / s
        h = np.clip(h * factor, 0.0, 1.0)
    # final adjustment for any leftover difference, distribute uniformly among interior points
    # after loop ensure sum approx
    s = np.sum(h)
    if abs(s - target_sum) > 1e-8:
        # distribute diff among coordinates not at bounds
        free = (h > 0) & (h < 1)
        if np.any(free):
            diff = target_sum - s
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n0 = len(init_h)
    else:
        n0 = 45
        init_h = rng.random(n0)
    # choose target n between 45 and maybe 360
    target_n = 180
    if n0 != target_n:
        # upscale via linear interpolation
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h = np.interp(x_new, x_old, init_h)
    else:
        h = init_h.copy()
    # ensure feasible
    target_sum = target_n / 2
    h = project_to_feasible(h, target_sum)
    # define evaluate
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    best_h = h.copy()
    best_score = evaluate(best_h)
    current_h = best_h.copy()
    current_score = best_score
    # parameters
    T0 = 0.5
    Tmin = 1e-5
    max_iter = 200000  # generous; will break according to time
    iteration = 0
    while time.time() - start < budget_s * 0.95 and iteration < max_iter:
        # compute temperature schedule linearly decreasing
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)
        T = T0 * (1 - frac) + Tmin * frac
        sigma = 0.2 * T + 0.01  # minimal sigma resolution
        # propose candidate
        candidate = current_h + rng.normal(scale=sigma, size=target_n)
        # project to feasible
        candidate = project_to_feasible(candidate, target_sum)
        cand_score = evaluate(candidate)
        # acceptance
        if cand_score < current_score or rng.random() < math.exp(-(cand_score - current_score)/max(T,1e-12)):
            current_h = candidate
            current_score = cand_score
            if cand_score < best_score:
                best_h = candidate.copy()
                best_score = cand_score
        iteration += 1
    # final return
    # Ensure final h is feasible
    best_h = project_to_feasible(best_h, target_sum)
    final_score = evaluate(best_h)
    return best_h, final_score, target_n

    if shift >= 0:

import numpy as np
import time, math

def project_to_feasible(h, target_sum):
    # Clip and scale iteratively to satisfy sum.
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) <= 1e-12:
            break
        factor = target_sum / s if s != 0 else 0.0
        h = np.clip(h * factor, 0.0, 1.0)
    # final adjustment if small residual
    s = h.sum()
    if abs(s - target_sum) > 1e-8:
        free = (h > 0) & (h < 1)
        if np.any(free):
            diff = target_sum - s
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_corr_max(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx] * (2.0 / n)
    shift = max_idx - (n - 1)
    return max_val, shift

def compute_gradient(h, shift):
    n = len(h)
    g = np.ones_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            # subtract h[i+s]
            g[:n-s] -= h[s:]
            # subtract h[i-s]
            g[s:] -= h[:n-s]
    else:
        s = -shift
        if s < n:
            # shift negative: treat s positive
            g[s:] -= h[:n-s]  # h[i-s] term where i>=s
            g[:n-s] -= h[s:]  # h[i+s] term where i < n-s
    return g

def upscale_initial(h_initial, target_n):

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine target n
    target_n = 180  # prime? Might be 180.
    # Try to get initial_h
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is not None:
        h0 = upscale_initial(init_h, target_n)
    else:
        # random feasible
        h0 = rng.random(target_n)
        h0 = project_to_feasible(h0, target_n/2.0)

    best_h = h0.copy()
    best_score, _ = evaluate_corr_max(best_h)

    # parameters for inner loop
    # Max iterations per attempt determined by time
    # We'll run iterative subgradient descent
    # Setting step schedule
    base_step = 0.05
    min_step = 0.001
    # Outer restart
    restart_limit = 10
    restart_count = 0

    while True:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 0.5:
            break
        # start a new restart if needed
        if restart_count == 0 and init_h is not None:
            h = best_h.copy()
        else:
            h = rng.random(target_n)
            h = project_to_feasible(h, target_n/2.0)
        # inner optimization: gradient descent
        # We may limit inner steps based on remaining time and some max iteration
        max_inner = 5000
        for it in range(max_inner):
            now = time.time()
            if now - start >= budget_s * 0.95:
                break
            # evaluate
            score, shift = evaluate_corr_max(h)
            # compute gradient
            g = compute_gradient(h, shift)
            # step size schedule decays with overall time progress
            progress = (now - start) / (budget_s * 0.95)
            step = base_step * (1 - progress) + min_step * progress
            # update
            h = h - step * g
            # optional noise term to escape local minima
            if rng.random() < 0.01:
                noise = rng.normal(scale=step*0.5, size=target_n)
                h += noise
            # project
            h = project_to_feasible(h, target_n/2.0)
            # evaluate again for best update
            new_score, _ = evaluate_corr_max(h)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # early exit if we have a sufficiently good bound maybe < 0.38?
            if best_score < 0.3805:
                # Good enough; break loops
                break
        restart_count += 1
        # stop if many restarts surpassed
        if restart_count >= restart_limit:
            # we could also increase target_n after restarts but for now break
            break

    # final projection
    best_h = project_to_feasible(best_h, target_n/2.0)
    final_score, _ = evaluate_corr_max(best_h)
    # ensure final_score not worse than best_score
    if final_score < best_score:
        best_score = final_score
    return best_h, best_score, target_n

import numpy as np
import time
import math

def _project_to_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and scale to achieve the required sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) <= 1e-12:
            break
        if s == 0:
            factor = 0.0
        else:
            factor = target_sum / s
        h = np.clip(h * factor, 0.0, 1.0)
    # final tiny residual correction
    s = h.sum()
    if abs(s - target_sum) > 1e-8:
        free = (h > 0) & (h < 1)
        if free.any():
            diff = target_sum - s
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _max_corr_and_shift(h: np.ndarray):
    """Return (max C5 value, shift index) for a given h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx] * (2.0 / n)          # C5 = max * dx, dx = 2/n
    shift = idx - (n - 1)
    return max_val, shift


def _binary_from_continuous(h_cont: np.ndarray) -> np.ndarray:
    """Turn a continuous vector into a binary one with exactly half 1s."""
    n = len(h_cont)
    k = n // 2
    # indices of the k largest entries
    idx = np.argpartition(h_cont, -k)[-k:]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    return h_bin
