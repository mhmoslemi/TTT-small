import time
import numpy as np
from scipy.optimize import minimize

def _repair_vector(v, n, max_iter=15):
    """Project v onto [0,1]^n with sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v > 0.0
        else:
            mask = v < 1.0
        n_free = mask.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        v[mask] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final correction if still slightly off
    s = v.sum()
    if abs(s - target) > 1e-9:
        factor = target / s if s > 0 else 0.0
        v = np.clip(v * factor, 0.0, 1.0)
        # tiny residual fix
        s = v.sum()
        if abs(s - target) > 1e-8:
            # distribute the remaining difference uniformly within bounds
            diff = target - s
            inc = diff / n
            v += inc
            v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c5(v):
    """Return the overlap bound for vector v."""
    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return np.max(corr) * dx

def _upsample_vector(v, n_target):
    """Linear interpolation of v to length n_target, then repair."""
    n_src = len(v)
    src_x = np.arange(n_src) * (2.0 / n_src)
    tgt_x = np.arange(n_target) * (2.0 / n_target)
    new_v = np.interp(tgt_x, src_x, v, left=v[0], right=v[-1])
    return _repair_vector(new_v, n_target)

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    v = np.random.rand(n)
    return _repair_vector(v, n)

def _optimize_at_resolution(n, init_v, time_limit):
    """Run SLSQP from init_v (and a few random restarts) within time_limit."""
    start = time.time()
    best_v = init_v.copy()
    best_score = _compute_c5(best_v)

    def obj(x):
        return _compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n

    # primary run from init_v
    try:
        res = minimize(
            obj,
            init_v,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            v = _repair_vector(res.x, n)
            score = _compute_c5(v)
            if score < best_score:
                best_score, best_v = score, v
    except Exception:
        pass

    # random restarts
    while time.time() - start < time_limit:
        rnd = _random_feasible(n)
        try:
            res = minimize(
                obj,
                rnd,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                v = _repair_vector(res.x, n)
                score = _compute_c5(v)
                if score < best_score:
                    best_score, best_v = score, v
        except Exception:
            pass
        # limit restarts to keep within time
        if time.time() - start > time_limit * 0.9:
            break
    return best_v, best_score

def _local_search(v, n, time_limit):
    """Simple hill‑climbing: exchange small amounts between two coordinates."""
    start = time.time()
    best_v = v.copy()
    best_score = _compute_c5(best_v)
    step = 0.05  # max magnitude of a single exchange
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2.0 * step
        cand = best_v.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = _repair_vector(cand, n)
        score = _compute_c5(cand)
        if score < best_score:
            best_score, best_v = score, cand
    return best_v, best_score
