def project_onto_constraints(h, target_sum):
    n = len(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Now adjust to meet sum constraint.
    # Use simple water-filling:
    # compute current sum
    current = h.sum()
    diff = target_sum - current
    # if diff close to 0 -> done
    # iterate at most few times:
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # identify free indices (not at bound)
        if diff > 0:
            mask = h < 1 - 1e-12
        else:
            mask = h > 1e-12
        free = np.where(mask)[0]
        if free.size == 0:
            # cannot adjust further; break
            break
        delta = diff / free.size
        # add delta to free entries
        h[free] += delta
        # clip again to [0,1]
        h = np.clip(h, 0, 1)
        new_sum = h.sum()
        new_diff = target_sum - new_sum
        # if diff changes sign or small, then break
        if np.sign(new_diff) != np.sign(diff):
            # might need to adjust again due to bounds crossing
            diff = new_diff
            continue
        diff = new_diff
    return h
```

def random_move(h, eps, rng):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h  # no move
    # choose amount to transfer from i to j
    max_transfer = min(eps, h[i])  # cannot go below zero
    # also ensure h[j]+transfer <= 1
    max_transfer = min(max_transfer, 1 - h[j])
    if max_transfer <= 0:
        return h
    delta = rng.uniform(0, max_transfer)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
```

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target - current
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # Clip delta to avoid overshooting bounds
        # For each free idx, compute max possible adjustment
        # but we will just apply and clip
        h[free] += delta
        # Clip again to ensure bounds
        h = np.clip(h, 0.0, 1.0)
        new = h.sum()
        diff = target - new
    return h

def _c5_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _smooth_c5_and_grad(h, alpha=5.0):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # stable softmax
    exp_vals = np.exp(alpha * (corr - max_corr))
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    # loss (smooth max)
    loss = (max_corr + np.log(Z)) / alpha * dx
    # gradient
    grad = np.full(n, dx)  # start with dx * 1 term for each i
    # subtract weighted contributions
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # i where i+k in bounds
            grad[:n - k] -= w * dx * h[k:]
            # i where i-k in bounds
            grad[k:] -= w * dx * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= w * dx * h[:n - kk]
            grad[:n - kk] -= w * dx * h[kk:]
    return loss, grad

import time
import math
import numpy as np

def _project(vec, target_sum, max_iter=20):
    """Project vec onto the box [0,1]^n and the hyperplane sum(vec)=target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    cur = v.sum()
    diff = target_sum - cur
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target_sum - cur
    # final tiny adjustment to hit the sum exactly (within floating error)
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        idx = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if idx.size > 0:
            v[idx[0]] += diff
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h):
    """Exact C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _smooth_c5_and_grad(h, alpha=5.0):
    """Smooth approximation of C5 (log‑sum‑exp) and its gradient."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # stable softmax
    exp_vals = np.exp(alpha * (corr - max_corr))
    Z = exp_vals.sum()
    weights = exp_vals / Z
    # smooth loss = (log-sum-exp)/alpha * dx
    loss = (max_corr + np.log(Z)) / alpha * dx

    # gradient of the smooth loss
    grad = np.full(n, dx)   # start with dx * 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        k = idx - (n - 1)   # shift value
        if k >= 0:
            grad[:n - k] -= w * dx * h[k:]
            grad[k:]     -= w * dx * h[:n - k]
        else:
            kk = -k
            grad[kk:]     -= w * dx * h[:n - kk]
            grad[:n - kk] -= w * dx * h[kk:]
    return loss, grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # number of sample points (the benchmark uses 70)
    n = 70
    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise h
    try:
        init = initial_h_values
    except NameError:
        init = None
    if init is not None and hasattr(init, '__len__') and len(init) == n:
        h = np.asarray(init, dtype=float)
        h = _project(h, target_sum)
    else:
        h = rng.random(n)
        h = _project(h, target_sum)

    best_h = h.copy()
    best_score = _c5_score(h)
    cur_score = best_score

    lr = 0.5          # learning rate
    alpha = 5.0       # smoothing parameter
    it = 0
    no_improve = 0

    while time.time() - start_time < budget_s - 0.2:
        # smooth objective and gradient
        loss, grad = _smooth_c5_and_grad(h, alpha=alpha)

        # gradient step
        cand = h - lr * grad
        cand = _project(cand, target_sum)

        cand_score = _c5_score(cand)

        # deterministic improvement acceptance
        if cand_score < cur_score:
            h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # occasional uphill moves (simulated‑annealing style)
            remain = budget_s - (time.time() - start_time)
            T = max(1e-4, 0.03 * remain / budget_s)
            if rng.random() < math.exp(-(cand_score - cur_score) / T):
                h = cand
                cur_score = cand_score
            no_improve += 1

        it += 1

        # learning‑rate and smoothing schedules
        if it % 250 == 0:
            lr = max(0.02, lr * 0.93)
            alpha = min(80.0, alpha * 1.07)

        # periodic random perturbation to escape plateaus
        if it % 800 == 0:
            noise = rng.normal(scale=0.015, size=n)
            h = _project(np.clip(h + noise, 0.0, 1.0), target_sum)
            cur_score = _c5_score(h)

        # hard restart if stuck for a long time
        if no_improve > 5000:
            h = rng.random(n)
            h = _project(h, target_sum)
            cur_score = _c5_score(h)
            no_improve = 0

    final_score = _c5_score(best_h)
    return best_h.tolist(), float(final_score), n
