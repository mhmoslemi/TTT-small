    hi_lambda = hi - np.min(v) + 1.0

            hi_lambda = mid
        else:
            lo_lambda = mid
    return np.clip(v + (lo_lambda + hi_lambda) / 2.0, lo, hi)

    if d >= 0:

        g[p] += 1.0 - h[p + d]

        s = proj.sum()
        if s > target:
            hi_lambda = mid
        else:
            lo_lambda = mid
    lam = (lo_lambda + hi_lambda) * 0.5
    return np.clip(v + lam, lo, hi)

def _grad_worst_shift(h, d):
    """Gradient of the overlap for the shift d (index offset)."""
    n = h.size
    dx = 2.0 / n
    g = np.zeros_like(h)
    if d >= 0:
        i = np.arange(n - d)
        g[i] += 1.0 - h[i + d]
        i = np.arange(d, n)
        g[i] -= h[i - d]
    else:
        d_abs = -d
        i = np.arange(d_abs, n)
        g[i] += 1.0 - h[i - d_abs]
        i = np.arange(0, n - d_abs)
        g[i] -= h[i + d_abs]
    return g * dx

def _gd_optimize(h0, max_iters=2000, lr=0.5, decay=0.995):
    """Simple gradient descent on the worst shift."""
    h = h0.copy()
    n = h.size
    target = n * 0.5
    best_h, best_val = h.copy(), _c5_and_corr(h)[0]

    for _ in range(max_iters):
        val, corr = _c5_and_corr(h)
        d = corr.argmax() - (n - 1)
        g = _grad_worst_shift(h, d)
        h = h - lr * g
        h = _proj_box_sum(h, target)
        new_val = _c5_and_corr(h)[0]
        if new_val < best_val:
            best_val, best_h = new_val, h.copy()
        lr *= decay
        if lr < 1e-6:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # obtain a starting vector (if the harness provides one)
    try:
        init = np.array(initial_h_values, dtype=float)  # noqa: F821
        n0 = init.size
        init = _proj_box_sum(init, n0 * 0.5)
        have_init = True
    except Exception:
        have_init = False

    if have_init:
        base_vec = init
    else:
        n0 = 90
        base_vec = np.random.rand(n0)
        base_vec = _proj_box_sum(base_vec, n0 * 0.5)

    best_h, best_val, best_n = None, np.inf, n0

    # ------------------------------------------------------------------
    # resolutions to explore (start, maybe double)
    resolutions = [n0]
    if budget_s > 300:
        resolutions.append(n0 * 2)

    for n in resolutions:
        if time.time() - start > budget_s * 0.95:
            break

        # up‑sample if needed
        if n == n0:
            h_start = base_vec.copy()
        else:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_start = np.interp(x_new, x_old, base_vec)
            h_start = _proj_box_sum(h_start, n * 0.5)

        # several random restarts at this resolution
        for restart in range(5):
            if time.time() - start > budget_s * 0.95:
                break

            if restart == 0:
                h0 = h_start.copy()
            else:
                # perturb and re‑project
                noise = (np.random.rand(n) - 0.5) * 0.5
                h0 = h_start + noise
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = _proj_box_sum(h0, n * 0.5)

            # gradient descent
            iters = 3000 if n > 120 else 2500
            h_opt, val_opt = _gd_optimize(h0, max_iters=iters, lr=0.5, decay=0.996)

            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n
                if best_val <= 0.3808:
                    break
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # final fine‑tuning if time permits
    if best_h is not None and time.time() - start < budget_s * 0.90:
        h_fine, val_fine = _gd_optimize(best_h, max_iters=4000, lr=0.1, decay=0.999)
        if val_fine < best_val:
            best_h, best_val = h_fine, val_fine
            best_n = best_h.size

    # Ensure the returned bound matches the returned vector
    final_val = _c5_and_corr(best_h)[0]

    return best_h, float(final_val), int(best_n)
