        cur = h.sum()
        diff = target - cur
        # Adjust the first entry if possible, otherwise scale uniformly.
        if 0.0 <= h[0] + diff <= 1.0:
            h[0] += diff
        else:
            # Uniform scaling while staying inside bounds.
            scaling = target / cur if cur != 0 else 0.0
            h = np.clip(h * scaling, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    target_c5 = 0.3808

    # ------------------------------------------------------------------
    # Helper to evaluate C5 – use the pre‑imported helper if present.
    # ------------------------------------------------------------------
    def compute_c5(arr):
        try:
            return evaluate_erdos_solution(arr)
        except Exception:
            # Fallback direct computation.
            return np.max(np.correlate(arr, 1.0 - arr, mode="full") * (2.0 / len(arr)))

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Early exit if the starting point already beats the target.
    if best_c5 <= target_c5:
        return best_h, best_c5, n

    # ------------------------------------------------------------------
    # Optimisation parameters.
    # ------------------------------------------------------------------
    step = 0.2               # initial mass transfer magnitude
    iter_count = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while True:
        # Time check.
        if time.time() - start_time >= budget_s * 0.99:
            break

        # Correlation for the current best candidate.
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)  # shift index (can be negative)

        moved = False

        # ------------------------------------------------------------------
        # If the worst shift is non‑zero, try a directed transfer.
        # ------------------------------------------------------------------
        if shift != 0:
            # Indices that participate in this shift.
            if shift > 0:
                i_vals = np.arange(0, n - shift)
            else:
                i_vals = np.arange(-shift, n)
            j_vals = i_vals + shift

            # Product contribution for each pair (h[i] * (1 - h[j])).
            prod = best_h[i_vals] * (1.0 - best_h[j_vals])
            # Indices ordered by descending contribution – try the biggest few.
            order = np.argsort(prod)[::-1]

            # Try to move mass from i → j (reducing the product).
            for idx in order[:3]:
                i = i_vals[idx]
                j = j_vals[idx]
                delta_max = min(step, best_h[i], 1.0 - best_h[j])
                if delta_max <= 0:
                    continue
                delta = rng.random() * delta_max
                if delta == 0.0:
                    continue
                trial = best_h.copy()
                trial[i] -= delta
                trial[j] += delta
                c5 = compute_c5(trial)
                if c5 < best_c5:
                    best_h = trial
                    best_c5 = c5
                    moved = True
                    break

            # If that failed, try the opposite direction j → i.
            if not moved:
                for idx in order[:3]:
                    i = i_vals[idx]
                    j = j_vals[idx]
                    delta_max = min(step, best_h[j], 1.0 - best_h[i])
                    if delta_max <= 0:
                        continue
                    delta = rng.random() * delta_max
                    if delta == 0.0:
                        continue
                    trial = best_h.copy()
                    trial[j] -= delta
                    trial[i] += delta
                    c5 = compute_c5(trial)
                    if c5 < best_c5:
                        best_h = trial
                        best_c5 = c5
                        moved = True
                        break

        # ------------------------------------------------------------------
        # If no directed move succeeded (or shift was zero), fall back to a
        # random pair transfer.
        # ------------------------------------------------------------------
        if not moved:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta_max = min(step, best_h[i], 1.0 - best_h[j])
                if delta_max > 0:
                    delta = rng.random() * delta_max
                    if delta > 0.0:
                        trial = best_h.copy()
                        trial[i] -= delta
                        trial[j] += delta
                        c5 = compute_c5(trial)
                        if c5 < best_c5:
                            best_h = trial
                            best_c5 = c5
                            moved = True

        # ------------------------------------------------------------------
        # Book‑keeping.
        # ------------------------------------------------------------------
        if moved:
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        # Gradually shrink the step size for finer tuning.
        if iter_count % 2000 == 0:
            step = max(step * 0.9, 1e-5)

        # Occasionally inject a random perturbation to escape stagnation.
        if no_improve > 5000:
            i1, i2 = rng.integers(0, n, size=2)
            if i1 != i2:
                delta_max = min(step, best_h[i1], 1.0 - best_h[i2])
                if delta_max > 0:
                    delta = rng.random() * delta_max
                    best_h[i1] -= delta
                    best_h[i2] += delta
                    best_c5 = compute_c5(best_h)
            no_improve = 0

        # Early termination if we already beat the target.
        if best_c5 <= target_c5:
            break

    # Final evaluation (in case the loop exited without a fresh check).
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
