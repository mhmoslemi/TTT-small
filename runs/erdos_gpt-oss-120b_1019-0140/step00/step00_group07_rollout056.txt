def hill_climb_binary(b, best_c5, start_time, time_limit):
    n = len(b)
    dx = 2.0/n
    best = b.copy()
    best_val = best_c5
    # precompute indices of ones and zeros
    ones = np.where(b==1)[0]
    zeros = np.where(b==0)[0]
    while time.time() - start_time < time_limit:
        # pick random ones and zeros
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # create candidate
        cand = b.copy()
        cand[i] = 0
        cand[j] = 1
        # compute C5
        val = np.max(np.correlate(cand, 1-cand, mode='full')) * dx
        if val < best_val - 1e-9:
            best = cand
            best_val = val
            # update ones and zeros arrays
            ones = np.where(best==1)[0]
            zeros = np.where(best==0)[0]
        # optionally accept worse moves with probability
    return best, best_val

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Helper classes and functions ----------
class SumConstraint:
    """Equality: sum(h) == n/2."""
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

class OverlapConstraint:
    """Inequality: t >= overlap(k) * dx  (i.e. t - overlap*dx >= 0)."""
    def __init__(self, k, n):
        self.k = k
        self.n = n
        self.dx = 2.0 / n
    def __call__(self, x):
        h = x[:self.n]
        k = self.k
        if k >= 0:
            overl = np.dot(h[:self.n - k], 1.0 - h[k:])
        else:
            kk = -k
            overl = np.dot(h[kk:], 1.0 - h[:self.n - kk])
        return x[-1] - overl * self.dx

def compute_c5(h):
    """Exact overlap value for a given (continuous) h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_t(x):
    """Objective for SLSQP: minimize the auxiliary variable t (last entry)."""
    return x[-1]

def hill_climb_binary(b_start, start_c5, start_time, budget_s):
    """Simple swap hill‑climbing on a binary weight‑half vector."""
    n = len(b_start)
    dx = 2.0 / n
    best_h = b_start.copy()
    best_val = start_c5
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.1:
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        val = np.max(np.correlate(cand, 1 - cand, mode='full')) * dx
        if val + 1e-12 < best_val:      # accept only strict improvements
            best_h = cand
            best_val = val
    return best_h, best_val
