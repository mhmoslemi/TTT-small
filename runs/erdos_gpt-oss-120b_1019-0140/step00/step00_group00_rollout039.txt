def project_to_sum_and_bounds(v, target, lower=0.0, upper=1.0, max_iter=50):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def improve_binary(h, max_iter, time_limit, start_time):
    n = len(h)
    best_h = h.copy()
    best_val = int(np.correlate(h, 1 - h, mode='full').max())
    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        conv = np.correlate(h, 1 - h, mode='full')
        max_idx = int(conv.argmax())
        max_val = int(conv[max_idx])
        if max_val <= best_val:
            best_val = max_val
            best_h = h.copy()
        if max_val == 0:
            break
        shift = max_idx - (n - 1)
        # get mismatched indices
        if shift >= 0:
            ones_i = np.where((h[:n - shift] == 1) & (h[shift:] == 0))[0]
            zeros_i = np.where((h[:n - shift] == 0) & (h[shift:] == 1))[0]
        else:
            s = -shift
            ones_i = np.where((h[s:] == 1) & (h[:n - s] == 0))[0] + s
            zeros_i = np.where((h[s:] == 0) & (h[:n - s] == 1))[0] + s
        if ones_i.size == 0 or zeros_i.size == 0:
            # cannot improve this shift; break out of loop for this h
            break
        i = int(np.random.choice(ones_i))
        j = int(np.random.choice(zeros_i))
        # swap
        h[i], h[j] = h[j], h[i]
        # continue loop
    return best_h, best_val

import numpy as np
import time
import random

def _max_overlap_count(h):
    """Return the maximal overlap count (without dx scaling) for binary h."""
    # h should be integer 0/1
    conv = np.correlate(h, 1 - h, mode="full")
    return int(conv.max())

def _improve_binary(h, max_iter, deadline):
    """Greedy improvement: reduce the maximal overlap count by swapping.
    Returns the best binary vector seen and its overlap count."""
    n = h.size
    best_h = h.copy()
    best_cnt = _max_overlap_count(h)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = int(conv.argmax())
        max_cnt = int(conv[max_idx])
        if max_cnt < best_cnt:
            best_cnt = max_cnt
            best_h = h.copy()
        if max_cnt == 0:
            break  # perfect solution

        shift = max_idx - (n - 1)  # k such that overlap uses h[i] & (1-h[i+shift])

        # find positions where h[i]=1, h[i+shift]=0  (call them ones)
        # and positions where h[i]=0, h[i+shift]=1 (call them zeros)
        if shift >= 0:
            part_len = n - shift
            ones = np.where((h[:part_len] == 1) & (h[shift:] == 0))[0]
            zeros = np.where((h[:part_len] == 0) & (h[shift:] == 1))[0]
        else:
            s = -shift
            part_len = n - s
            ones = np.where((h[s:] == 1) & (h[:part_len] == 0))[0] + s
            zeros = np.where((h[s:] == 0) & (h[:part_len] == 1))[0] + s

        if ones.size == 0 or zeros.size == 0:
            break  # cannot improve this shift further

        i = int(random.choice(ones))
        j = int(random.choice(zeros))
        # swap a 1 and a 0, preserving the total number of ones
        h[i], h[j] = h[j], h[i]

    return best_h, best_cnt
