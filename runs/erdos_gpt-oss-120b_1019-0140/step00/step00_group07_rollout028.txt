import time, math, numpy as np

def _c5_of(h, dx):
    """Return C5 = max correlation * dx for binary/float vector h."""
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)


def _anneal(N, dx, budget, rng, start_vec=None):
    """Simulated‑annealing search for a low‑C5 binary vector of length N."""
    best_h = None
    best_c5 = np.inf

    # initialise current state
    if start_vec is not None and np.isclose(start_vec.sum(), N / 2):
        cur = np.where(start_vec > 0.5, 1.0, 0.0).copy()
    else:
        cur = np.zeros(N, dtype=np.float64)
        cur[rng.choice(N, N // 2, replace=False)] = 1.0
    cur_c5 = _c5_of(cur, dx)
    best_h, best_c5 = cur.copy(), cur_c5

    start = time.time()
    no_improve = 0

    while time.time() - start < budget:
        elapsed = time.time() - start

        ones = np.where(cur > 0.5)[0]

        if ones.size == 0 or zeros.size == 0:

        j = zeros[rng.integers(zeros.size)]

            diff = target - cur_sum
            if diff > 0:
                zeros = np.where(bin_vec == 0)[0]
                pick = rng.choice(zeros, diff, replace=False)
                bin_vec[pick] = 1.0
            else:
                ones = np.where(bin_vec == 1)[0]
                pick = rng.choice(ones, -diff, replace=False)
                bin_vec[pick] = 0.0
        val = _c5_of(bin_vec, dx)
        best_h, best_c5 = bin_vec.copy(), val

    # Time left for the stochastic optimisation phase
    elapsed = time.time() - start_all
    remaining = budget_s - elapsed
    if remaining > 5.0:
        # leave a small safety margin for finalisation
        anneal_budget = remaining * 0.95
        h_opt, c5_opt = _anneal(n, dx, anneal_budget, rng, start_vec=init)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Fallback – should never happen, but guarantees a return value
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_c5 = _c5_of(best_h, dx)

    # Final bound using the provided evaluator if available
    try:
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_bound = float(best_c5)

    return best_h, c5_bound, n
