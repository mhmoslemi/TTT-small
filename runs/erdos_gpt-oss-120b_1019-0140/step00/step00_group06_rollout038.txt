import numpy as np
import time
import math

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _random_feasible_pair(h: np.ndarray, rng: np.random.Generator, max_step: float):
    """Pick two distinct indices and a feasible delta respecting bounds and sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]

    # feasible delta interval so that 0 <= hi+δ <= 1 and 0 <= hj-δ <= 1
    lower = max(-hi, -(1.0 - hj))
    upper = min(1.0 - hi, hj)

    if lower >= upper:
        return None  # no feasible move for this pair

    # restrict step size for stability
    lim = min(max_step, upper - lower)
    if lim <= 0:
        return None
    delta = rng.uniform(-lim, lim)
    # clip to feasible interval
    delta = max(lower, min(upper, delta))
    return i, j, delta

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function vector h, its C5 bound, and the number of points."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Obtain initial vector (if provided) or fall back to a simple construction
    try:
        h_current = np.array(initial_h_values, dtype=np.float64, copy=True)
    except NameError:
        # fallback: n = 100, half ones, half zeros (satisfies sum = n/2)
        n = 100
        h_current = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(np.float64)

    n = h_current.size
    # Ensure basic feasibility (clipping and sum correction)
    h_current = np.clip(h_current, 0.0, 1.0)
    current_sum = h_current.sum()
    target_sum = n / 2.0
    if not np.isclose(current_sum, target_sum, atol=1e-9):
        # simple scaling preserving bounds (iterative projection)
        scale = target_sum / current_sum if current_sum > 0 else 0.0
        h_current *= scale
        h_current = np.clip(h_current, 0.0, 1.0)
        # if still off due to clipping, distribute the residue uniformly
        resid = target_sum - h_current.sum()
        if abs(resid) > 1e-9:
            # add/subtract small amount equally to all non‑boundary entries
            mask = (h_current > 0.0) & (h_current < 1.0)
            if mask.any():
                inc = resid / mask.sum()
                h_current[mask] += inc
                h_current = np.clip(h_current, 0.0, 1.0)

    # best solution tracking
    best_h = h_current.copy()
    best_val = _c5_value(best_h)

    # SA parameters
    max_iter = 500_000
    max_step = 0.05           # maximal change per move
    init_temp = 0.02
    decay_rate = 0.9995

    cur_h = best_h.copy()
    cur_val = best_val
    temp = init_temp

    it = 0
    while it < max_iter and (time.time() - start_time) < budget_s * 0.95:
        pair = _random_feasible_pair(cur_h, rng, max_step)
        if pair is None:
            it += 1
            continue
        i, j, delta = pair
        # apply move
        hi_old, hj_old = cur_h[i], cur_h[j]
        cur_h[i] = hi_old + delta
        cur_h[j] = hj_old - delta

        new_val = _c5_value(cur_h)

        # acceptance test
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_val = new_val
            # update global best if improved
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert move
            cur_h[i] = hi_old
            cur_h[j] = hj_old

        # temperature cooling
        temp *= decay_rate
        it += 1

    # Final evaluation on the best found vector
    best_val = _c5_value(best_h)

    return best_h.astype(np.float64), best_val, n
