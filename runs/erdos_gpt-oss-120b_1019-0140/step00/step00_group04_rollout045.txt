def project_box_sum(v, target, lower=0., upper=1., tol=1e-9, max_iter=100):
    # ensure target between lower*len(v) and upper*len(v)
    n = v.shape[0]
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + mid, lower, upper)
```

import numpy as np
import time
import math

def _project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto the intersection of [lower,upper]^n and the hyperplane sum=w."""
    lo, hi = -1e5, 1e5
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, lower, upper)

def _compute_c5(h):
    """Return the discrete C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    # Seed RNG
    np.random.seed(seed)

    # --- initialise h -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    except NameError:
        n = 84
        h0 = np.random.rand(n)

    target_sum = 0.5 * n
    h0 = _project_box_sum(h0, target_sum)

    cur_h = h0.copy()
    cur_cost = _compute_c5(cur_h)

    best_h = cur_h.copy()
    best_cost = cur_cost

    start_time = time.time()
    # Simulated‑annealing schedule
    T0 = 0.02
    step0 = 0.15

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.98:
            break
        if best_cost <= 0.3808:
            break

        frac = elapsed / budget_s
        T = T0 * (1.0 - frac)
        step = step0 * (1.0 - frac)

        # propose a new candidate
        cand = cur_h + step * np.random.randn(n)
        cand = _project_box_sum(cand, target_sum)
        cand_cost = _compute_c5(cand)

        # update global best
        if cand_cost < best_cost:
            best_cost = cand_cost
            best_h = cand.copy()

        # accept/reject
        if cand_cost < cur_cost or np.random.rand() < math.exp(-(cand_cost - cur_cost) / max(T, 1e-12)):
            cur_h = cand
            cur_cost = cand_cost

    # final safety projection
    best_h = _project_box_sum(best_h, target_sum)

    return best_h, float(best_cost), int(n)
