def compute_delta(h, i, j):
    n = len(h)
    delta = np.zeros(n, dtype=int)
    # i->0:
    # left part indices s=0..n-1-i
    # delta[:n-i] -= (1 - h[i:]) # shape (n-i)
    # right part s=0..i
    # delta[:i+1] += h[i::-1]
    # j->1:
    # left: delta[:n-j] += (1 - h[j:])
    # right: delta[:j+1] -= h[j::-1]
    # However careful: contributions overlapping where i and j both affect same shift; they sum.
    # We'll implement using slices.

    left_len_i = n - i
    delta[:left_len_i] -= (1 - h[i:])
    # right

    left_len_j = n - j
    delta[:left_len_j] += (1 - h[j:])
    delta[:j+1] -= h[j::-1]

    return delta

def improve_h(h, time_limit):
    n = len(h)
    corr = compute_corr_int(h)  # array length n ints
    best_h = h.copy()
    best_val = corr.max()
    start = time.time()
    # Precompute indices of ones and zeros lists
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    # We'll use for loop random sampling pairs
    while time.time() - start < time_limit:
        # choose random swap
        i = random.choice(ones)
        j = random.choice(zeros)
        delta = compute_delta(h, i, j)
        new_corr = corr + delta
        new_max = new_corr.max()
        # Accept if new_max <= best_val (or <= current max)
        if new_max <= corr.max():
            # Accept
            h[i] = 0
            h[j] = 1
            # Update ones, zeros lists
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            # Update corr and max
            corr = new_corr
            # update best_global
            if new_max < best_val:

                best_h = h.copy()

import time, random, math, numpy as np

def compute_corr_int(h):
    n = h.shape[0]
    corr_full = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full')
    return corr_full[n-1:]

def compute_delta_inplace(h, i, j, delta):
    n = h.shape[0]
    delta.fill(0)
    # i from 1 to 0
    delta[:n - i] -= (1 - h[i:])
    delta[:i + 1] += h[i::-1]
    # j from 0 to 1
    delta[:n - j] += (1 - h[j:])
    delta[:j + 1] -= h[j::-1]

def improve_h(h0, time_limit):
    start = time.time()
    n = h0.shape[0]
    h = h0.copy()
    corr = compute_corr_int(h)
    cur_max = int(corr.max())
    best_h = h.copy()
    best_val = cur_max
    ones = set(np.nonzero(h)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    delta = np.empty(n, dtype=np.int32)
    T0 = 10.0
    while time.time() - start < time_limit:
        # Choose random pair
        i = random.sample(ones, 1)[0]
        j = random.sample(zeros, 1)[0]
        compute_delta_inplace(h, i, j, delta)
        new_corr = corr + delta
        new_max = int(new_corr.max())
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            elapsed = time.time() - start
            frac = elapsed / time_limit
            T = T0 * (1.0 - frac) + 1e-8
            # probability
            if random.random() < math.exp(-(new_max - cur_max) / T):
                accept = True
        if accept:
            # apply swap
            h[i] = 0
            h[j] = 1
            # update index sets
            ones.remove(i); zeros.add(i)
            zeros.remove(j); ones.add(j)
            # update corr and cur_max
            corr = new_corr
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_h = h.copy()
    return best_h, best_val

def binarize_initial(arr):
    n = arr.shape[0]
    # Get indices of top n//2 values
    order = np.argsort(-arr)  # descending
    h_bin = np.zeros(n, dtype=np.int8)
    half = n // 2
    h_bin[order[:half]] = 1
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, math, numpy as np
    random.seed(seed)
    np.random.seed(seed)
    start_total = time.time()
    best_h = None
    best_bound = float('inf')
    best_n = None

    # Determine candidate n values
    candidate_ns = []
    init_n = None
    if 'initial_h_values' in globals():
        init_n = len(initial_h_values)
        if init_n % 2 == 0:
            candidate_ns.append(init_n)
        else:
            # If odd, adjust by adding one dummy zero at end
            # but we will treat later.
            pass

    # Add multiples of init_n if init_n present
    if init_n:
        for mult in [2, 3, 4, 5, 6]:
            n = init_n * mult
            if n % 2 == 0 and n <= 800:
                candidate_ns.append(n)

    # Add some fixed even sizes
    for n in [80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 450, 500]:
        if n % 2 == 0:
            candidate_ns.append(n)

    candidate_ns = sorted(set(candidate_ns))

    # Main loop over candidates, using remaining time splits
    # We'll run each candidate for a portion of remaining budget (max 30 sec)
    for n in candidate_ns:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 5:
            break
        # allocate time for this n
        time_for_n = min(30.0, remaining * 0.25)
        # build initial vector
        if init_n and n == init_n:
            # use given initial_h_values, ensure binary
            h0 = np.array(initial_h_values, copy=False)
            if h0.dtype != np.int8 or not np.all(np.logical_or(h0 == 0, h0 == 1)):
                h0 = binarize_initial(h0)
        elif init_n and n % init_n == 0:
            base = np.array(initial_h_values, copy=False)
            if base.dtype != np.int8 or not np.all(np.logical_or(base == 0, base == 1)):
                base = binarize_initial(base)
            repeats = n // init_n
            h0 = np.tile(base, repeats)
        else:
            # random binary vector
            h0 = np.zeros(n, dtype=np.int8)
            h0[:n//2] = 1
            np.random.shuffle(h0)

        # run improvement
        h_opt, max_corr = improve_h(h0, time_for_n)
        bound = (max_corr * 2.0) / n
        if bound < best_bound:
            best_bound = bound
            best_h = h_opt.copy()
            best_n = n
            # early exit if good enough
            if best_bound < 0.3808:
                break

    # Fallback if never found any solution
    if best_h is None:
        n = init_n if init_n else 100
        h = np.zeros(n, dtype=np.int8)
        h[:n//2] = 1
        np.random.shuffle(h)
        best_h = h
        best_n = n
        best_bound = (np.max(compute_corr_int(best_h)) * 2.0) / n

    # Convert to float array output
    h_values = best_h.astype(np.float64)
    c5_bound = float(best_bound)
    n_points = int(best_n)

    return (h_values, c5_bound, n_points)

import time
import random
import math
import numpy as np

# ---------- utility functions ----------
def compute_c5_int(h: np.ndarray) -> int:
    """Return the integer max correlation sum for binary vector h."""
    n = h.shape[0]
    # correlation h with (1-h); mode='full' returns length 2n-1
    corr = np.correlate(h, 1 - h, mode="full")
    # keep only non‑negative shifts (indices n-1 .. 2n-2)
    max_corr = int(np.max(corr[n - 1 :]))
    return max_corr


def greedy_improve(h_start: np.ndarray, time_limit: float):
    """
    Simple greedy hill‑climb: repeatedly try swapping a 1 with a 0.
    Accept a swap only if it does not increase the max correlation.
    Return the best vector found and its max‑correlation value.
    """
    n = h_start.shape[0]
    h = h_start.copy()
    cur_max = compute_c5_int(h)
    best_h = h.copy()
    best_val = cur_max

    # sets of positions for quick random sampling
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    start = time.time()
    while time.time() - start < time_limit:
        # pick a random 1 and a random 0
        i = random.sample(ones, 1)[0]
        j = random.sample(zeros, 1)[0]

        # perform the swap
        h[i], h[j] = 0, 1
        new_max = compute_c5_int(h)

        if new_max <= cur_max:
            # accept the move
            cur_max = new_max
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)

            if cur_max < best_val:
                best_val = cur_max
                best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = 1, 0

    return best_h, best_val


def random_binary_vector(n: int) -> np.ndarray:
    """Create a random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    h[: n // 2] = 1
    np.random.shuffle(h)
    return h


def binarize_initial(arr: np.ndarray) -> np.ndarray:
    """Project any initial array onto a binary vector with half ones."""
    n = arr.shape[0]
    half = n // 2
    order = np.argsort(-arr)  # descending
    h = np.zeros(n, dtype=np.int8)
    h[order[:half]] = 1
    return h


def expand_initial(h0: np.ndarray, target_n: int) -> np.ndarray:
    """Tile the initial pattern to reach target_n (must be multiple of len(h0))."""
    repeats = target_n // len(h0)
    return np.tile(h0, repeats)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    total_start = time.time()
    best_h = None
    best_bound = float("inf")
    best_n = None

    # Determine candidate problem sizes
    candidate_ns = []
    init_n = None
    if "initial_h_values" in globals():
        init_n = len(initial_h_values)
        if init_n % 2 == 0:
            candidate_ns.append(init_n)

    # multiples of the parent size (if any)
    if init_n:
        for mul in (2, 3, 4, 5, 6):
            n = init_n * mul
            if n % 2 == 0 and n <= 800:
                candidate_ns.append(n)

    # some standard even sizes
    for n in (80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 450, 500, 600):
        if n % 2 == 0:
            candidate_ns.append(n)

    candidate_ns = sorted(set(candidate_ns))

    # iterate over candidate sizes, allocating time slices
    for n in candidate_ns:
        elapsed = time.time() - total_start
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # allocate up to 30 s (or a fraction of the remaining budget) for this size
        time_for_n = min(30.0, remaining * 0.25)

        # ----- prepare an initial vector for this n -----
        if init_n and n == init_n:
            h0 = np.array(initial_h_values, copy=False)
            if h0.dtype != np.int8 or not np.all(np.logical_or(h0 == 0, h0 == 1)):
                h0 = binarize_initial(h0)
        elif init_n and n % init_n == 0:
            base = np.array(initial_h_values, copy=False)
            if base.dtype != np.int8 or not np.all(np.logical_or(base == 0, base == 1)):
                base = binarize_initial(base)
            h0 = expand_initial(base, n)
        else:
            h0 = random_binary_vector(n)

        # ----- perform several random restarts within the allocated budget -----
        n_start = time.time()
        while time.time() - n_start < time_for_n:
            # short budget for each restart
            per_restart = min(5.0, time_for_n - (time.time() - n_start))
            if per_restart <= 0.1:
                break

            # start from a fresh random (or tiled) vector
            if random.random() < 0.5:
                start_vec = random_binary_vector(n)
            else:
                start_vec = h0.copy()
                # optionally shuffle a little to escape local symmetries
                if n > 100:
                    idx = np.random.choice(n, size=4, replace=False)
                    start_vec[idx] = 1 - start_vec[idx]

            h_opt, max_corr = greedy_improve(start_vec, per_restart)
            bound = (max_corr * 2.0) / n  # C5 = max_corr * dx,  dx = 2/n

            if bound < best_bound:
                best_bound = bound
                best_h = h_opt.astype(float)
                best_n = n
                if best_bound < 0.3808:
                    # good enough, exit early
                    return (best_h, best_bound, best_n)

    # fallback: use whatever was best (or a simple construction)
    if best_h is None:
        n = init_n if init_n else 100
        best_h = random_binary_vector(n).astype(float)
        best_n = n
        best_bound = (compute_c5_int(best_h.astype(np.int8)) * 2.0) / n

    return (best_h, float(best_bound), int(best_n))
