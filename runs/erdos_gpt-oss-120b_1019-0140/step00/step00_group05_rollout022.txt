def project_to_box_simplex(v, target_sum):
    n = v.shape[0]
    # clamp to [0,1]
    v = np.clip(v, 0, 1)
    # need to adjust sum to target_sum
    # If sum already equal target, done.
    # else if sum too small, distribute additional mass to entries <1 evenly.
    # else if sum too large, subtract from entries >0 evenly.
    # Use simple iterative water-filling: while diff !=0 and there are eligible entries.
    diff = target_sum - v.sum()
    # Continue while absolute diff > 1e-12 and there are eligible indices.
    # For simplicity, we can adjust diff by distributing equally among eligible indices.
    # Because diff could be small but we can approximate.
    # We'll just compute:
    # If diff > 0: increase entries proportionally up to 1.
    # If diff < 0: decrease entries proportionally down to 0.
    # This is linear redistribution.
    if abs(diff) < 1e-12:
        return v
    # Indices eligible:
    if diff > 0:
        eligible = (v < 1)
        if not np.any(eligible):
            return v
        # compute how much we can add to each eligible index: up to 1 - v[i]
        capacity = 1 - v[eligible]
        total_capacity = capacity.sum()
        if diff > total_capacity:
            # Not possible, just set all to 1
            v[eligible] = 1
            # diff left not resolvable; but target sum may be infeasible (should not happen)
            return v

        v[eligible] += diff * (capacity / total_capacity)

    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx
