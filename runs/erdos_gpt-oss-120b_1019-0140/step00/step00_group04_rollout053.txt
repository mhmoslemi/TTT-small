def evaluate_h(h):

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    # shift = idx - (len(h) - 1) (since zero shift at position N-1)
    # Not needed but for gradient shift index.
    shift = idx - (len(h) - 1)

def gradient_of_shift(h, shift):
    N = len(h)
    g = np.empty(N, dtype=np.float64)
    # compute for each i:
    # term1: (1 - h[i+shift]) if i+shift in range else 1
    # term2: -h[i-shift] if i-shift in range else 0
    # so g[i] = (1 - h[i+shift]) if i+shift < N else 1
    #        - (h[i-shift] if i-shift >=0 else 0)
    # Multiply by dx later
    # Use vectorized approach
    # For term1:
    term1 = np.ones(N)
    idx1 = np.arange(N) + shift
    mask1 = (idx1 >= 0) & (idx1 < N)
    term1[mask1] = 1.0 - h[idx1[mask1]]
    # term2:
    term2 = np.zeros(N)
    idx2 = np.arange(N) - shift
    mask2 = (idx2 >= 0) & (idx2 < N)
    term2[mask2] = h[idx2[mask2]]
    g = term1 - term2
    return g

def project_feasible(x, desired_sum):

def evaluate_h(h):
    """Return C5 for a step function h on [0,2] represented by samples."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def worst_shift(h):
    """Return integer shift (relative to centre) that gives maximal overlap."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return idx - (h.size - 1)

    mask2 = (idx2 >= 0) & (idx2 < n)
    if np.any(mask2):
        g[mask2] -= h[idx2[mask2]]
    return g

def project_feasible(x, desired_sum):
    """
    Project x onto the set {0 <= x_i <= 1, sum(x) = desired_sum}.
    Uses iterative water-filling.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(200):
        cur_sum = np.sum(x)
        diff = desired_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0 - 1e-12
        else:
            free = x > 1e-12
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        adj = diff / n_free
        x[free] += adj
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # --- initialise h -------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
    except NameError:
        n = 84  # fallback size
        h0 = np.full(n, 0.5, dtype=np.float64)

    desired_sum = n / 2.0
    h = project_feasible(h0, desired_sum)

    cur_h = h.copy()
    cur_score = evaluate_h(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # --- parameters ---------------------------------------------------------
    step = 0.5                # initial step length for gradient moves
    decay = 0.9995            # exponential decay of step
    prob_grad = 0.7           # probability of using a gradient step
    iter_count = 0

    # --- optimisation loop --------------------------------------------------
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        iter_count += 1

        # decide move type
        if random.random() < prob_grad:
            # gradient based move
            shift = worst_shift(cur_h)
            g = gradient_of_shift(cur_h, shift)
            norm = np.linalg.norm(g)
            if norm > 1e-12:
                direction = -g / norm
                new_h = cur_h + step * direction
                new_h = project_feasible(new_h, desired_sum)
            else:
                # fall back to random move if gradient is zero
                continue
        else:
            # random pair move preserving sum exactly
            i, j = random.sample(range(n), 2)
            lo = max(-cur_h[i], cur_h[j] - 1.0)
            hi = min(1.0 - cur_h[i], cur_h[j])
            if lo >= hi:
                continue
            delta = random.uniform(lo, hi)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_h = project_feasible(new_h, desired_sum)

        new_score = evaluate_h(new_h)

        # simulated annealing acceptance
        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
        else:
            temp = max(step * 0.1, 1e-6)
            if random.random() < math.exp(-(new_score - cur_score) / temp):
                cur_h, cur_score = new_h, new_score

        # record best
        if cur_score < best_score:
            best_h, best_score = cur_h.copy(), cur_score

        # decay step size
        step *= decay

        # occasional diversification around the current best
        if iter_count % 5000 == 0:
            cur_h = best_h.copy()
            cur_score = best_score
            cur_h += np.random.uniform(-0.01, 0.01, size=n)
            cur_h = project_feasible(cur_h, desired_sum)

    final_h = best_h
    final_score = evaluate_h(final_h)
    return final_h.tolist(), float(final_score), int(n)
