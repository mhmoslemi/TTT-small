def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c1 = np.convolve(h, np.ones_like(h), mode='full')
    corr = np.convolve(h, h[::-1], mode='full')
    # Note that both arrays length 2n-1, index alignment matches
    diff = c1 - corr
    return dx * diff.max(), diff.max() # Return both for debugging maybe

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a binary step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # sum of h over the overlapping region (linear convolution with ones)
    c1 = np.convolve(h, np.ones_like(h), mode='full')
    # autocorrelation of h (linear convolution with reversed h)
    corr = np.convolve(h, h[::-1], mode='full')
    diff = c1 - corr  # corresponds to sum_i h_i (1 - h_{i+k})
    return dx * diff.max()

def _init_binary(n: int, rng: np.random.RandomState, base: np.ndarray = None) -> np.ndarray:
    """Create a binary vector of length n with exactly n/2 ones."""
    if base is not None and base.size == n:
        return base.copy()
    # replicate a smaller base pattern if it tiles evenly
    if base is not None and n % base.size == 0:
        factor = n // base.size
        return np.tile(base, factor).astype(np.float64)
    # otherwise random balanced binary vector
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h
