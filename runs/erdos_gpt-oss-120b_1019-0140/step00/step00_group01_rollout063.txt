import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_corr_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    maxc = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)
    return maxc, shift, corr

def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        end = n - shift
        g[:end] += 1
        g[:end] -= h[shift:]
        g[shift:] -= h[:end]
    else:
        p = -shift
        start = p
        g[start:] += 1
        g[start:] -= h[:n - p]
        g[:n - p] -= h[p:]
    return g
