def compute_c5_fft(h, dx):
    n = len(h)
    # Pad to length 2n for convolution
    size = 1 << ((2*n-1).bit_length())
    # Compute FFT of h and of 1-h using zero-padding
    fh = np.fft.rfft(h, n=size)
    fg = np.fft.rfft(1 - h, n=size)
    prod = fh * fg
    conv = np.fft.irfft(prod, n=size)
    # conv length = size; the linear convolution result is first 2n-1 entries (starting at 0)
    conv = conv[:2*n-1]
    return np.max(conv) * dx

def project_to_sum(h, target):
    # assume h within [0,1]
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    # Simple method: distribute diff proportionally among positions that can absorb.
    # We'll iterate and adjust.
    # If diff > 0: need to increase some entries
    while abs(diff) > 1e-12:
        if diff > 0:
            # find index with max headroom
            idx = np.argmax(1.0 - h)
            cap = 1.0 - h[idx]
            add = min(cap, diff)
            h[idx] += add
            diff -= add

            idx = np.argmin(h)  # smallest value
            cap = h[idx]  # can reduce up to this amount
            sub = min(cap, -diff)
            h[idx] -= sub
            diff += sub
    return h

import time
import math
import numpy as np
from scipy.signal import fftconvolve

def _project_to_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h (in [0,1]) so that its sum equals target, keeping bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # simple water-filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to increase sum: find element with most headroom
            room = 1.0 - h
            idx = int(np.argmax(room))
            cap = room[idx]
            if cap <= 0:
                break
            add = min(cap, diff)
            h[idx] += add
            diff -= add
        else:
            # need to decrease sum: find element with largest value
            idx = int(np.argmax(h))
            cap = h[idx]
            if cap <= 0:
                break
            sub = min(cap, -diff)
            h[idx] -= sub
            diff += sub
    return h

def _initialize_h(n: int) -> np.ndarray:
    """Return a feasible sinusoidal start vector of length n."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    # h(x) = 0.5 + 0.5 * sin(pi * x)   (values already in [0,1])
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    return _project_to_sum(h, n / 2.0)

def _upsample_initial(init_h: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a given initial vector to length n, then project."""
    n0 = init_h.shape[0]
    # sample points are taken at mid‑interval positions
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    xt = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(xt, x0, init_h)
    h = np.clip(h, 0.0, 1.0)
    return _project_to_sum(h, n / 2.0)

def _compute_c5(h: np.ndarray) -> float:
    """C5 value for the current step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation via convolution (np.correlate(h, 1-h) == fftconvolve(h, (1-h)[::-1]))
    conv = fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return float(np.max(conv) * dx)

def _anneal(start_h: np.ndarray, max_time: float) -> tuple[np.ndarray, float]:
    """Simple simulated annealing that preserves sum and bounds."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = start_h.copy()
    cur_c5 = _compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5

    # temperature schedule (linear in elapsed fraction)
    init_temp = 0.02 * cur_c5
    final_temp = 0.001 * cur_c5

    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # propose a transfer of mass between two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random direction: move from src to dst
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_sub = h[src]              # how much we can take from src
        max_add = 1.0 - h[dst]        # how much we can give to dst
        max_delta = min(max_sub, max_add)
        if max_delta <= 1e-14:
            continue
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _compute_c5(h_new)

        # temperature linearly decreasing
        frac = elapsed / max_time
        temp = (1.0 - frac) * init_temp + frac * final_temp
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / (temp + 1e-15))
            if np.random.rand() < prob:
                accept = True
        if accept:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        it += 1
    return best_h, best_c5
