import time
import math
import numpy as np

def evaluate_h(h):
    """Return the C5 value and the shift attaining it for a step vector h."""
    arr = np.asarray(h, dtype=np.float64)
    n = arr.shape[0]
    corr = np.correlate(arr, 1 - arr, mode='full')
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)          # integer shift index
    c5 = max_val * (2.0 / n)          # multiply by dx = 2/n
    return c5, shift

def compute_c5(h):
    """Plain C5 computation (max correlation * dx)."""
    arr = np.asarray(h, dtype=np.float64)
    n = arr.shape[0]
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 200
        h0 = np.full(n, 0.5, dtype=np.float64)

    # enforce feasibility (clip + exact sum)
    h0 = np.clip(h0, 0.0, 1.0)
    target_sum = n / 2.0
    if abs(h0.sum() - target_sum) > 1e-12:
        h0[:] = 0.5                     # uniform vector satisfies the sum exactly

    cur_h = h0.copy()
    cur_c5, cur_shift = evaluate_h(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    T0 = 0.05            # initial temperature
    Tmin = 1e-4          # final temperature
    total_time = budget_s * 0.95   # leave a safety margin
    p_target = 0.3       # probability to try a targeted move

    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_time:
            break

        # linear cooling schedule
        t_frac = elapsed / total_time
        T = max(Tmin, T0 * (1.0 - t_frac))

        # --------------------------------------------------------------
        # Targeted move (reduce the worst overlap)
        # --------------------------------------------------------------
        if rng.random() < p_target and cur_shift != 0:
            k = cur_shift
            if k > 0:
                i_start = 0
                i_end = n - k - 1
            else:
                i_start = -k
                i_end = n - 1
            i = rng.integers(i_start, i_end + 1)
            j = i + k
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max > 0.0:
                delta = rng.random() * delta_max
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                new_c5, new_shift = evaluate_h(new_h)
                if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                    cur_h = new_h
                    cur_c5 = new_c5
                    cur_shift = new_shift
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = cur_h.copy()
                continue   # go to next iteration

        # --------------------------------------------------------------
        # Random pairwise move (preserves the integral)
        # --------------------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        low = -min(cur_h[i], 1.0 - cur_h[j])
        high = min(1.0 - cur_h[i], cur_h[j])
        if low >= high:
            continue
        delta = rng.uniform(low, high)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c5, new_shift = evaluate_h(new_h)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h = new_h
            cur_c5 = new_c5
            cur_shift = new_shift
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()

    # ------------------------------------------------------------------
    # Final feasibility fix (exact integral)
    # ------------------------------------------------------------------
    best_h = np.asarray(best_h, dtype=np.float64)
    sum_h = best_h.sum()
    diff = target_sum - sum_h
    if abs(diff) > 1e-12:
        best_h += diff / n
        np.clip(best_h, 0.0, 1.0, out=best_h)
        residual = target_sum - best_h.sum()
        if abs(residual) > 1e-12:
            free = np.where((best_h > 0) & (best_h < 1))[0]
            if free.size == 0:
                idx = rng.integers(0, n)
                best_h[idx] = np.clip(best_h[idx] + residual, 0.0, 1.0)
            else:
                idx = free[0]
                best_h[idx] = np.clip(best_h[idx] + residual, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h.tolist(), final_c5, n
