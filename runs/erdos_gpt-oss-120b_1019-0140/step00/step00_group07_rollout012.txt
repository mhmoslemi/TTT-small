import time
import numpy as np

def _compute_c5(arr):
    """Return the C5 value for a binary step function given as a 1‑D array."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement (full mode)
    return np.max(np.correlate(arr, 1.0 - arr, mode='full')) * dx

def _enforce_half(arr, rng):
    """Force a binary array to have exactly half ones (rounding if needed)."""
    n = arr.shape[0]
    target = n // 2
    cur = int(arr.sum())
    diff = cur - target
    if diff > 0:
        ones = np.where(arr == 1)[0]
        flip = rng.choice(ones, size=diff, replace=False)
        arr[flip] = 0.0
    elif diff < 0:
        zeros = np.where(arr == 0)[0]
        flip = rng.choice(zeros, size=-diff, replace=False)
        arr[flip] = 1.0
    return arr

def _golden_pattern(n, rng):
    """Sturmian (Beatty) word based on the golden ratio, forced to half ones."""
    phi = (1 + 5**0.5) / 2
    idx = np.arange(n)
    frac = (idx * (phi % 1)) % 1
    arr = (frac < 0.5).astype(float)
    return _enforce_half(arr, rng)
