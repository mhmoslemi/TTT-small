        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:

def local_optimize(h0, schedule, time_limit, start_time):
    h = h0.copy()
    N = h.shape[0]
    dx = 2.0 / N
    best_h = h.copy()
    best_score = compute_c5(h)
    for beta, lr, max_iter in schedule:
        for _ in range(max_iter):
            if time.time() - start_time > time_limit:
                break
            # compute overlaps
            overlaps = np.correlate(h, h, mode='full')
            # compute p
            min_ov = overlaps.min()
            exp_vals = np.exp(-beta * (overlaps - min_ov))
            p = exp_vals / exp_vals.sum()
            # gradient via convolution
            conv = np.convolve(p, h, mode='full')
            grad = -2.0 * conv[N-1:N-1+N]
            # update
            h = h - lr * grad
            # projection
            h = project_box_sum(h, target_sum=N/2)
            # periodical scoring
            # we can evaluate after each iteration, but cost low; we can evaluate each iteration.
            score = compute_c5(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
    return best_h, best_score

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n intersect {sum = target}."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_score(h: np.ndarray) -> float:
    """Compute the Erdős overlap bound C5 for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_step(h: np.ndarray, beta: float, lr: float) -> np.ndarray:
    """Perform one gradient step for the smoothed min-overlap objective."""
    n = h.shape[0]
    # self‑correlation (overlap values for all shifts)
    ov = np.correlate(h, h, mode="full")
    # soft‑minimum weights (numerically stable)
    m = ov.min()
    w = np.exp(-beta * (ov - m))
    p = w / w.sum()
    # gradient via convolution: grad = -2 * (p * h) convolved
    conv = np.convolve(p, h, mode="full")
    grad = -2.0 * conv[n - 1 : n - 1 + n]
    # update and return
    return h - lr * grad


def _local_optimize(start_h: np.ndarray, schedule, deadline, start_time):
    """Gradient‑descent optimizer with continuation over β."""
    h = start_h.copy()
    best_h = h.copy()
    best_score = _c5_score(h)

    for beta, lr, max_iter in schedule:
        for _ in range(max_iter):
            if time.time() - start_time > deadline:
                return best_h, best_score
            h = _gradient_step(h, beta, lr)
            h = _project_box_sum(h, h.shape[0] / 2.0)
            sc = _c5_score(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score


def _random_refine(h: np.ndarray, steps: int, deadline, start_time, rng):
    """Simple greedy random refinement preserving the sum constraint."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_sc = _c5_score(best_h)

    for _ in range(steps):
        if time.time() - start_time > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # maximum feasible transfer from i to j
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            continue
        delta = (rng.random() - 0.5) * 2.0 * max_transfer * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = _project_box_sum(h_new, target)
        sc = _c5_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new
    return best_h, best_sc
