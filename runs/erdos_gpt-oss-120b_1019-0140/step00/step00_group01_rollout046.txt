def project_feasible(h, target_sum, rng):
    # Clip
    np.clip(h, 0.0, 1.0, out=h)
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    # Use up to 10 iterations
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        # choose direction
        if diff > 0:
            # need increase some h < 1
            idx_candidates = np.where(h < 1 - 1e-12)[0]
            if len(idx_candidates) == 0:
                break
            idx = rng.choice(idx_candidates)
            max_inc = 1 - h[idx]
            delta = min(diff, max_inc)
            h[idx] += delta
        else:
            idx_candidates = np.where(h > 1e-12)[0]
            if len(idx_candidates) == 0:
                break
            idx = rng.choice(idx_candidates)
            max_dec = h[idx]
            delta = min(-diff, max_dec)
            h[idx] -= delta
        cur_sum = h.sum()
        diff = target_sum - cur_sum
    return h

def perturb(h, target_sum, rng, max_change=0.3):

import numpy as np
import time
import math
import sys

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max overlap integral for discretized h."""
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)

def project_to_sum_and_bounds(h: np.ndarray, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to hit exact sum while staying feasible."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    # at most a few adjustments are needed; diff is tiny after each move
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            delta = min(diff, 1.0 - h[idx])
            h[idx] += delta
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            idx = rng.choice(cand)
            delta = -min(-diff, h[idx])
            h[idx] += delta
        diff = target_sum - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def perturb_move(h: np.ndarray, rng: np.random.Generator, max_change: float = 0.4) -> np.ndarray | None:
    """Return a new h after moving mass between two indices, preserving sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # maximum delta moving from j to i (positive) and i to j (negative)
    max_pos = min(1.0 - h[i], h[j])
    max_neg = min(h[i], 1.0 - h[j])
    max_pos = min(max_pos, max_change)
    max_neg = min(max_neg, max_change)
    if max_pos <= 0 and max_neg <= 0:
        return None
    delta = rng.uniform(-max_neg, max_pos)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h
