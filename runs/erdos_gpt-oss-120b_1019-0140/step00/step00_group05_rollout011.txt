import numpy as np
import time
import math

def compute_corr(arr):
    """Full cross‑correlation of h with (1‑h)."""
    return np.correlate(arr, 1.0 - arr, mode='full')

def compute_c5(arr, dx):
    """Maximum overlap integral C5 for a given discretisation."""
    return np.max(compute_corr(arr)) * dx

def max_corr_shift(arr):
    """Return the shift (relative index) where the correlation attains its maximum."""
    corr = compute_corr(arr)
    idx = int(np.argmax(corr))
    shift = idx - (len(arr) - 1)
    return shift, corr

def neighbor_sum(arr, shift):
    """For each index i compute h[i+shift] + h[i-shift] (missing terms are omitted)."""
    n = len(arr)
    out = np.zeros(n, dtype=float)
    if shift >= 0:
        if shift < n:
            out[:n - shift] += arr[shift:]
            out[shift:] += arr[:n - shift]
    else:
        s = -shift
        if s < n:
            out[s:] += arr[:n - s]
            out[:n - s] += arr[s:]
    return out

def _adjust_sum_to_target(vec, target):

    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    if cur > 0:

    vec = np.clip(vec, 0.0, 1.0)

                sub = min(-diff, vec[i])
