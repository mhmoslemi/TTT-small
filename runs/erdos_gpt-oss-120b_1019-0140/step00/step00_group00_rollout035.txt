def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    from scipy.signal import fftconvolve # but we can use np.fft as well

    rng = np.random.default_rng(seed)

    start_time = time.time()

    # desired n_points modest for speed, maybe 200
    n_points = 200

    # Build initial h:
    # If initial_h_values provided: get n0 and upsample to n_points
    # else: generate random initial vector meeting constraints.

    # ensure sum(h) = n_points/2.

    # define evaluation function
    def max_C(h):
        # compute correlation via fftconvolve for speed
        # h and 1-h arrays length n_points.
        conv = np.convolve(h, (1 - h)[::-1])  # shape len 2n-1
        return np.max(conv) * (2.0 / n_points)

    # initial solution: if initial available, upsample via simple repeat
    # else start with random feasible: random in [0,1] then project onto constraints.

    # projection function onto [0,1] and sum constraint.
    def project_sum_bound(v, target_sum):
        # Use simple algorithm: clip to [0,1] and adjust by distributing deficit.
        # iterative: while sum != target_sum: compute delta = target_sum - sum(v)
        # adjust unfixed entries proportionally.
        v = np.clip(v, 0.0, 1.0)
        # Compute sum now.
        s = v.sum()
        if abs(s - target_sum) < 1e-12:
            return v
        # iterative water-filling: treat indices where v is not at bounds.
        # We'll use simple method: sort indices by v, but for speed we can use bisection to find lambda.
        # Solve for lambda such that sum_i clip(v_i + lambda, 0,1) = target.
        lo = -1.0
        hi = 1.0
        for _ in range(50):
            lam = (lo + hi) / 2.0
            w = np.clip(v + lam, 0.0, 1.0)
            s = w.sum()
            if s < target_sum:
                lo = lam
            else:
                hi = lam
        return w

    # Build initial h
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        # upsample to n_points by linear interpolation
        n0 = base.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xv = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(xv, x0, base)
    else:
        # random vector
        h = rng.random(n_points)
        target_sum = n_points / 2.0
        h = project_sum_bound(h, target_sum)

    # compute best
    best_h = h.copy()
    best_val = max_C(best_h)

    # local search parameters
    max_iter = 2000000  # big but will break with time
    iter = 0
    # Pre-calc target sum
    target_sum = n_points / 2.0
    # Use simple simulated annealing schedule
    T0 = 0.01
    while True:
        # time check
        if time.time() - start_time > budget_s - 0.1:
            break
        if iter >= max_iter:
            break
        # propose a move: choose two indices i,j with i != j
        i, j = rng.integers(0, n_points, size=2)
        # ensure i!=j
        if i == j:
            continue
        # maximum amount we can transfer from i to j while keeping bounds
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # choose delta as random fraction
        delta = rng.random() * max_transfer
        # produce candidate
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # no need to reproject sum: sum unchanged.
        # compute new val
        val_new = max_C(h_new)
        # accept if better or with probability
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h_new.copy()
            h = h_new
        else:
            # temperature schedule: decrease with iteration
            T = T0 * (1 - iter / max_iter)
            if T < 1e-6:
                # no acceptance of worse moves
                pass
            else:
                # Metropolis
                if rng.random() < math.exp(-(val_new - max_C(h)) / T):
                    h = h_new
        iter += 1

    # At end, return best solution
    return (best_h.tolist(), best_val, n_points)

import time
import math
import numpy as np

def _max_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discrete step function."""
    # np.correlate implements the linear cross‑correlation needed.
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / h.size))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------------
    # Initialise h
    # --------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        target_sum = n / 2.0
        # tiny correction if the imported vector is off by rounding
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # adjust a non‑boundary entry
            for i in range(n):
                if 0.0 < h[i] < 1.0:
                    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
                    break
    else:
        # fall back to a modest size if no parent is supplied
        n = 60
        target_sum = n / 2.0
        base = rng.random(n)
        # projection onto the box [0,1] and the hyperplane sum = target_sum
        lo, hi = -1.0, 1.0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            proj = np.clip(base + lam, 0.0, 1.0)
            if proj.sum() < target_sum:
                lo = lam
            else:
                hi = lam
        h = np.clip(base + hi, 0.0, 1.0)

    # --------------------------------------------------------------------
    # Track the best solution seen
    # --------------------------------------------------------------------
    best_h = h.copy()
    best_val = _max_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # --------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # --------------------------------------------------------------------
    max_iters = 10_000_000
    T0 = 0.02
    it = 0
    while True:
        # respect the time budget
        if time.time() - start > budget_s - 0.1:
            break
        if it >= max_iters:
            break

        # ------------------------------------------------------------
        if rng.random() < 0.5:
            # random mass transfer between two positions
            i, j = rng.integers(0, n, size=2)
            if i != j:
                mx = min(cur_h[i], 1.0 - cur_h[j])
                if mx > 0.0:
                    delta = rng.random() * mx
                    new_h = cur_h.copy()
                    new_h[i] -= delta
                    new_h[j] += delta
                else:
                    it += 1
                    continue
            else:
                it += 1
                continue
        else:
            # targeted move that tries to improve the worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx = int(np.argmax(corr))
            k = idx - (n - 1)                     # shift corresponding to the max C5
            # pick a pair (i, i+k) inside the overlapping region

                    it += 1

                j = i + k

            new_h[i] += delta
