            - recompute diff.

def project_to_constraints(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # get mask of free entries
    while abs(diff) > 1e-12:
        if diff > 0:
            free = (h < 1.0)
        else:
            free = (h > 0.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = diff / free_idx.size
        # apply
        h[free_idx] += delta
        # clip again
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    return h

import time, random, math
import numpy as np

def _project_to_constraints(arr, target):
    """Project arr onto box [0,1] and enforce sum == target using water-filling."""
    # Ensure float array
    arr = np.clip(arr, 0.0, 1.0)
    diff = target - arr.sum()
    # iterative water-filling
    # tolerance
    eps = 1e-12
    while abs(diff) > eps:
        if diff > 0:
            free = arr < 1.0 - eps
        else:
            free = arr > eps
        idx = np.where(free)[0]
        if idx.size == 0:
            # cannot adjust further
            break
        delta = diff / idx.size
        arr[idx] += delta
        np.clip(arr, 0.0, 1.0, out=arr)
        diff = target - arr.sum()
    return arr

def _upsample_h(old_h, new_n):
    """Linear interpolation of old_h to length new_n on interval [0,2]."""
    old_n = len(old_h)
    # midpoints of each subinterval for interpolation
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + (1.0 / old_n)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + (1.0 / new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h

def _compute_c5(h):
    """Return max_k ∫ h(x) (1-h(x+k)) dx discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = h_init.shape[0]
    best_h = h_init.copy()

    cur_h = best_h.copy()
    cur_val = best_val

    # SA schedule
    T = 0.02                # initial temperature
    cooling = 0.9997
    min_T = 1e-6

    start = time.time()
    no_improve = 0
    # thresholds
    restart_limit = 5000
    while time.time() - start < budget:
        # mutation
        if random.random() < 0.03:
            # global Gaussian perturbation
            sigma = 0.15 * (T / 0.02 + 0.1)
            cand = cur_h + sigma * np.random.randn(n)
            cand = _project_to_constraints(cand, target_sum)
        else:
            # pairwise mass transfer
            max_step = 0.3 * (T / 0.02 + 0.01)
            i, j = random.sample(range(n), 2)
            hi = cur_h[i]
            hj = cur_h[j]
            lo = -max_step
            hi_step = max_step
            lo = max(lo, -hi, hj - 1.0)
            hi_step = min(hi_step, 1.0 - hi, hj)
            if lo >= hi_step:
                # fallback to copy (no change)
                cand = cur_h
            else:
                delta = random.uniform(lo, hi_step)
                cand = cur_h.copy()
                cand[i] = hi + delta
                cand[j] = hj - delta
                # sum preserved automatically
        cand_val = _compute_c5(cand)

        # acceptance
        if cand_val < cur_val or random.random() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_h = cand.copy()
                best_val = cand_val
                no_improve = 0
            else:
                no_improve += 1
