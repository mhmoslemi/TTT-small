def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time
    # set seeds
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Determine n_points, target_ones
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, copy=False)
        n_points = init_arr.shape[0]
    else:
        n_points = 70
    if n_points % 2 != 0:
        n_points += 1  # make even? but we also need to adjust? We'll keep even.

    target_ones = n_points // 2

    # Helper functions
    def random_binary():
        v = np.zeros(n_points, dtype=np.int8)
        idx = rng.choice(n_points, target_ones, replace=False)
        v[idx] = 1
        return v

    def repair(v):
        # v is np.ndarray dtype int8
        current = int(v.sum())
        if current == target_ones:
            return v
        # In case diff > 0 or <0
        if current > target_ones:
            # Need to turn some 1s into 0s
            excess = current - target_ones
            ones_idx = np.where(v == 1)[0]
            # choose excess indices to flip
            to_flip = rng.choice(ones_idx, excess, replace=False)
            v[to_flip] = 0
        else:
            deficit = target_ones - current
            zero_idx = np.where(v == 0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            v[to_flip] = 1
        return v

    def evaluate(v):
        # ensure proper type
        # maybe need to make it float for evaluate_erdos_solution
        return float(evaluate_erdos_solution(v.astype(float)))

    def tournament(pop, fitness, size=3):
        idxs = rng.choice(len(pop), size, replace=False)
        best_idx = idxs[0]
        best_fit = fitness[best_idx]
        for i in idxs[1:]:
            if fitness[i] > best_fit:
                best_fit = fitness[i]
                best_idx = i
        return pop[best_idx]

    def crossover(p1, p2):
        if rng.random() < CROSS_RATE:
            point = rng.integers(0, n_points+1)
            child = np.concatenate([p1[:point], p2[point:]]).astype(np.int8)
        else:
            child = p1.copy()
        return child

    def mutate(v):
        # swap mutation: pick a 1 and 0 and swap with prob MUT_RATE each iteration
        if rng.random() < MUT_RATE:
            ones = np.where(v == 1)[0]
            zeros = np.where(v == 0)[0]
            if ones.size > 0 and zeros.size > 0:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                v[i] = 0
                v[j] = 1
        return v

def repair_vector(v, target_ones, rng):
    # as described
    curr = int(v.sum())
    if curr == target_ones:
        return v
    if curr > target_ones:
        excess = curr - target_ones
        ones_idx = np.where(v == 1)[0]
        # ensure enough ones:
        if excess > len(ones_idx): excess = len(ones_idx)
        flip_idx = rng.choice(ones_idx, excess, replace=False)
        v[flip_idx] = 0
    else:
        deficit = target_ones - curr
        zeros_idx = np.where(v == 0)[0]
        if deficit > len(zeros_idx): deficit = len(zeros_idx)
        flip_idx = rng.choice(zeros_idx, deficit, replace=False)
        v[flip_idx] = 1
    return v

def tournament_select(pop, scores, rng, tournament_size=3):
    idxs = rng.choice(len(pop), tournament_size, replace=False)
    best_idx = idxs[0]
    best_score = scores[best_idx]
    for i in idxs[1:]:
        if scores[i] < best_score:  # lower is better
            best_score = scores[i]
            best_idx = i
    return pop[best_idx]

def crossover(parent1, parent2, n_points, rng, cross_rate):
    if rng.random() < cross_rate:
        point = rng.integers(0, n_points+1)
        child = np.concatenate([parent1[:point], parent2[point:]]).astype(np.int8)
    else:
        child = parent1.copy()
    return child

def mutate_swap(v, rng, mut_rate):
    if rng.random() < mut_rate:
        ones = np.where(v == 1)[0]
        zeros = np.where(v == 0)[0]
        if ones.size and zeros.size:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            v[i] = 0
            v[j] = 1
    return v

def hill_climb(v, current_score, n_points, target_ones, rng, start_time, budget_s, max_steps=2000):
    best_v = v.copy()
    best_score = current_score
    for step in range(max_steps):
        # time check
        if time.time() - start_time >= budget_s - 0.1:
            break
        # propose swap
        ones = np.where(best_v == 1)[0]
        zeros = np.where(best_v == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_v.copy()
        cand[i] = 0
        cand[j] = 1
        cand_score = evaluate_candidate(cand)
        if cand_score < best_score:
            best_v = cand
            best_score = cand_score
            # possible early exit if good enough
            if best_score <= 0.3808:
                break
    return best_v, best_score

def generate_random_binary(n_points: int, target_ones: int, rng: np.random.Generator) -> np.ndarray:
    """Create a binary vector with exactly target_ones ones."""
    v = np.zeros(n_points, dtype=np.int8)
    idx = rng.choice(n_points, target_ones, replace=False)
    v[idx] = 1
    return v


def repair_vector(v: np.ndarray, target_ones: int, rng: np.random.Generator) -> np.ndarray:
    """Adjust a binary vector so that it has exactly target_ones ones."""
    cur = int(v.sum())
    if cur == target_ones:
        return v
    if cur > target_ones:
        excess = cur - target_ones
        ones_idx = np.where(v == 1)[0]
        flip = rng.choice(ones_idx, excess, replace=False)
        v[flip] = 0
    else:
        deficit = target_ones - cur
        zeros_idx = np.where(v == 0)[0]
        flip = rng.choice(zeros_idx, deficit, replace=False)
        v[flip] = 1
    return v


def evaluate_candidate(v: np.ndarray) -> float:
    """Return the C5 value for a candidate step function."""
    if 'evaluate_erdos_solution' in globals():
        return float(evaluate_erdos_solution(v.astype(float)))
    # Fallback manual computation
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v.astype(float), 1 - v.astype(float), mode='full') * dx
    return float(np.max(corr))


def tournament_select(pop: list, scores: list, rng: np.random.Generator, tournament_size: int = 3) -> np.ndarray:
    """Select one individual via tournament (lower score is better)."""
    idxs = rng.choice(len(pop), tournament_size, replace=False)
    best_idx = idxs[0]
    best_score = scores[best_idx]
    for i in idxs[1:]:
        if scores[i] < best_score:
            best_score = scores[i]
            best_idx = i
    return pop[best_idx]


def crossover(parent1: np.ndarray, parent2: np.ndarray, n_points: int,
              rng: np.random.Generator, cross_rate: float) -> np.ndarray:
    """One‑point crossover with given probability."""
    if rng.random() < cross_rate:
        pt = rng.integers(0, n_points + 1)
        child = np.concatenate([parent1[:pt], parent2[pt:]]).astype(np.int8)
    else:
        child = parent1.copy()
    return child


def mutate_swap(v: np.ndarray, rng: np.random.Generator, mut_rate: float) -> np.ndarray:
    """Swap a randomly chosen 1 with a 0 (keeps the sum unchanged)."""
    if rng.random() < mut_rate:
        ones = np.where(v == 1)[0]
        zeros = np.where(v == 0)[0]
        if ones.size and zeros.size:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            v[i] = 0
            v[j] = 1
    return v


def hill_climb(v: np.ndarray, cur_score: float, n_points: int, target_ones: int,
               rng: np.random.Generator, start_time: float, budget_s: float,
               max_steps: int = 2000) -> tuple:
    """Greedy local improvement by swapping a 1 and a 0."""
    best_v = v.copy()
    best_score = cur_score
    for _ in range(max_steps):
        if time.time() - start_time >= budget_s - 0.1:
            break
        ones = np.where(best_v == 1)[0]
        zeros = np.where(best_v == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_v.copy()
        cand[i] = 0
        cand[j] = 1
        cand_score = evaluate_candidate(cand)
        if cand_score < best_score:
            best_v = cand
            best_score = cand_score
            if best_score <= 0.3808:
                break
    return best_v, best_score


# ---------- Main entry point ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)

    # Determine discretization size
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, copy=False)
        n_points = int(init_arr.shape[0])
    else:
        n_points = 70
    # Ensure even number of points (required for exact half‑sum)
    if n_points % 2:
        n_points += 1
    target_ones = n_points // 2

    # GA parameters
    POP_SIZE = 80
    CROSS_RATE = 0.7
    MUT_RATE = 0.05
    TARGET_BOUND = 0.3808

    # ----- Initial population -----
    population = []
    scores = []

    if 'initial_h_values' in globals():
        # Binary version of the parent, repaired to satisfy the sum constraint
        parent_bin = (init_arr >= 0.5).astype(np.int8)
        parent_bin = repair_vector(parent_bin, target_ones, rng)
        population.append(parent_bin)
        scores.append(evaluate_candidate(parent_bin))

    while len(population) < POP_SIZE:
        cand = generate_random_binary(n_points, target_ones, rng)
        population.append(cand)
        scores.append(evaluate_candidate(cand))

    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()

    # ----- Genetic algorithm -----
    while time.time() - start_time < budget_s - 0.5:
        if best_score <= TARGET_BOUND:
            break

        new_pop = [best]          # elitism
        new_scores = [best_score]

        while len(new_pop) < POP_SIZE:
            p1 = tournament_select(population, scores, rng)
            p2 = tournament_select(population, scores, rng)
            child = crossover(p1, p2, n_points, rng, CROSS_RATE)
            child = mutate_swap(child, rng, MUT_RATE)
            child = repair_vector(child, target_ones, rng)
            child_score = evaluate_candidate(child)

            new_pop.append(child)
            new_scores.append(child_score)

            if child_score < best_score:
                best = child
                best_score = child_score

        population = new_pop
        scores = new_scores

    # ----- Local refinement -----
    best, best_score = hill_climb(best, best_score, n_points, target_ones,
                                 rng, start_time, budget_s, max_steps=2000)

    # Return the result
    h_values = best.astype(float)
    c5_bound = float(best_score)
    return (h_values, c5_bound, n_points)
