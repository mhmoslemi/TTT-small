def run(seed=42, budget_s=1100, **kwargs):
    import time, math
    import numpy as np

    np.random.seed(seed)

        has_init = False

    # use the size of the supplied construction if it exists, else a modest size
    n_points = init_arr.shape[0] if has_init else 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx must equal 1

    best_h = None
    best_c5 = math.inf

    # ----------------------------------------------------------------------
    if has_init:
        if init_arr.shape[0] != n_points:
            # resample to the chosen resolution
            src_x = (np.arange(init_arr.shape[0]) + 0.5) * (2.0 / init_arr.shape[0])
