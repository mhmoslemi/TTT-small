def project_onto_capped_simplex(v, target, low, high):
    # project onto [low, high] with sum=target
    # Clip
    v = np.clip(v, low, high)
    # compute sum
    s = v.sum()
    if abs(s - target) < 1e-9:
        return v
    # Determine if we need to increase or decrease
    if s > target:
        # need to reduce excess = s-target
        excess = s - target
        # only reduce from entries > low
        # We'll iterate: while excess > 0:
        # find indices where v > low
        idx = np.where(v > low)[0]
        if len(idx) == 0:
            return v  # cannot adjust
        # compute total reducible = sum(v[idx] - low)
        total_reducible = (v[idx] - low).sum()
        if excess > total_reducible:
            excess = total_reducible  # set to max reducible
        # proportional reduction
        reduction = np.minimum(v[idx] - low, excess * (v[idx] - low) / total_reducible)
        v[idx] -= reduction
    else:
        # need to increase deficit = target - s
        deficit = target - s
        idx = np.where(v < high)[0]
        if len(idx) == 0:
            return v
        total_increasable = (high - v[idx]).sum()
        if deficit > total_increasable:
            deficit = total_increasable
        increase = np.minimum(high - v[idx], deficit * (high - v[idx]) / total_increasable)
        v[idx] += increase
    # Might need to iterate due to clipping after proportional distribution? but hopefully works.
    # Ensure sum approx target
    return v

def project_onto_capped_simplex(v, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    # Clip initial to bounds
    v = np.clip(v, low, high)
    # If sum already target, return
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    # lambda search
    # Determine lower and upper bounds for lambda
    # For lambda = low_bound, all v_i - lambda > high => after clip, u_i = high
    # For lambda = high_bound, all v_i - lambda < low => u_i = low
    # But we can start lo = -max(high, np.max(v)) and hi = np.max(v) maybe.
    lo = -max(high, np.max(v)) - 1.0
    hi = np.max(v) - low + 1.0
    # Ensure the sum at lo is >= target and at hi <= target
    def sum_with_lambda(lam):
        return np.clip(v - lam, low, high).sum()
    # Expand bounds if necessary
    while sum_with_lambda(lo) < target:
        lo -= 2.0
    while sum_with_lambda(hi) > target:
        hi += 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        s_mid = sum_with_lambda(mid)
        if s_mid > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    lam = (lo + hi) / 2.0
    u = np.clip(v - lam, low, high)
    # Due to rounding, adjust small residual
    res = target - u.sum()
    if abs(res) > 1e-12:
        # Distribute residual to some component not at bounds
        for idx in range(len(u)):
            if (res > 0 and u[idx] < high) or (res < 0 and u[idx] > low):
                adj = min(high - u[idx], res) if res > 0 else max(low - u[idx], res)
                u[idx] += adj
                res -= adj
                if abs(res) < 1e-12:
                    break
    return u

import time
import numpy as np

def project_onto_capped_simplex(v, target, low=0.0, high=1.0, tol=1e-12, max_iter=100):
    """Euclidean projection of v onto the set {x | low <= x_i <= high, sum(x)=target}."""
    v = np.clip(v, low, high)
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    # Helper to compute sum after shifting by lambda and clipping
    def sum_shift(lam):
        return np.clip(v - lam, low, high).sum()
    # Find bounds for lambda such that sum_shift(lo) >= target >= sum_shift(hi)
    lo = -max(high, np.max(v)) - 1.0
    hi = np.max(v) - low + 1.0
    while sum_shift(lo) < target:
        lo -= 2.0
    while sum_shift(hi) > target:
        hi += 2.0
    # Binary search
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        sm = sum_shift(mid)
        if sm > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    lam = (lo + hi) / 2.0
    proj = np.clip(v - lam, low, high)
    # Adjust tiny residual if any
    residual = target - proj.sum()
    if abs(residual) > tol:
        # Distribute residual to the first non‑bound element(s)
        for idx in range(len(proj)):
            if (residual > 0 and proj[idx] < high) or (residual < 0 and proj[idx] > low):
                delta = min(high - proj[idx], residual) if residual > 0 else max(low - proj[idx], residual)
                proj[idx] += delta
                residual -= delta
                if abs(residual) <= tol:
                    break
    return proj

def random_feasible_vector(n, target, rng):

    v = rng.random(n)
