        # pick source (h>0) and destination (h<1)
        src_idx = rng.choice(np.where(cur_h > 1e-12)[0])
        dst_idx = rng.choice(np.where(cur_h < 1.0 - 1e-12)[0])
        if src_idx == dst_idx:
            continue

        delta_max = min(cur_h[src_idx], 1.0 - cur_h[dst_idx])
        if delta_max <= 1e-14:
            continue

        step = min(delta_max, max_delta)
        delta = rng.random() * step
        if delta <= 0.0:
            continue

        # propose move
        cur_h[src_idx] -= delta
        cur_h[dst_idx] += delta
        new_score = compute_c5(cur_h, dx)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # reject – revert
            cur_h[dst_idx] -= delta
            cur_h[src_idx] += delta

    # ------------------------------------------------------------------
    # Small greedy refinement
    # ------------------------------------------------------------------
    for _ in range(N * 10):
        src_idx = rng.choice(np.where(best_h > 1e-12)[0])
        dst_idx = rng.choice(np.where(best_h < 1.0 - 1e-12)[0])
        if src_idx == dst_idx:
            continue
        delta_max = min(best_h[src_idx], 1.0 - best_h[dst_idx])
        if delta_max <= 1e-14:
            continue
        delta = delta_max * rng.random() * 0.02  # tiny tweak
        if delta <= 0.0:
            continue
        best_h[src_idx] -= delta
        best_h[dst_idx] += delta
        new_score = compute_c5(best_h, dx)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best_h[dst_idx] -= delta
            best_h[src_idx] += delta

    # ------------------------------------------------------------------
    # Final projection and output
    # ------------------------------------------------------------------
    best_h = enforce_sum(best_h)
    best_score = compute_c5(best_h, dx)
    return best_h.tolist(), float(best_score), N
