def project(u, target):
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo+hi)/2
        v = np.clip(u + mid, 0, 1)
        s = v.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(u + (lo+hi)/2, 0, 1)
    return v

def local_search(h, max_iter, T0, time_left):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    current_h = h.copy()

    start = time.time()

        # time check

def compute_c5_from_h(h):
    n = h.shape[0]
    # compute self_corr
    sc = np.correlate(h, h, mode='full')
    min_sc = np.min(sc)
    c5 = 1 - (2.0 / n) * min_sc
    return c5

def gradient_of_shift(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        # term from i-s
        grad[shift:] += h[:n-shift]
        # term from i+s
        grad[:n-shift] += h[shift:]
    else:
        s = -shift
        grad[:n-s] += h[s:]   # h[i+s] term when shift negative; i+s = i - (-s)
        grad[s:] += h[:n-s]   # h[i-s] term
    return grad

def project_to_constraints(u, target_sum):
    lo = -np.max(u) - 1.0
    hi = 1.0 - np.min(u) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        v = np.clip(u + mid, 0.0, 1.0)
        s = v.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    v = np.clip(u + (lo + hi) / 2.0, 0.0, 1.0)
    return v

import numpy as np
import time

def _project_to_sum_and_bounds(v, target):
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target."""
    lo = -np.max(v) - 1.0       # ensures v+lo <= 0 → sum=0 after clipping
    hi = 1.0 - np.min(v) + 1.0  # ensures v+hi >= 1 → sum=n after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vv = np.clip(v + mid, 0.0, 1.0)
        s = vv.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5_direct(h):
    """Exact C5 according to the problem definition."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _self_corr(h):
    """Correlation of h with itself (full mode)."""
    return np.correlate(h, h, mode='full')

    return grad
