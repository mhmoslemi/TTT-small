import numpy as np, random, time, math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    # Determine n_points and initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 45
        # start with random feasible
        v = np.random.rand(n)
        h0 = project_onto_simplex(v, n/2)  # function defined later
    # Ensure feasibility via projection
    h0 = project_onto_simplex(h0, n/2)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    current_h = best_h.copy()
    current_c5 = best_c5

    target_sum = n/2
    start = time.time()
    allowed = budget_s - 0.2
    temperature = 1.0
    cooling = 0.9995
    sigma0 = 0.2
    iteration = 0
    no_improve = 0
    while time.time() - start < allowed:
        sigma = sigma0 * temperature
        # generate neighbor
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.uniform(-sigma, sigma)
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # clamp and project
        new_h = np.clip(new_h, 0, 1)
        new_h = project_onto_simplex(new_h, target_sum)
        new_c5 = compute_c5(new_h, n)
        # acceptance
        if new_c5 < current_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - current_c5) / max(temperature, 1e-8))
            accept = random.random() < prob
        if accept:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-9:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        temperature = max(temperature * cooling, 1e-6)
        iteration += 1
        # occasional random restart if stuck
        if no_improve > 5000 and time.time() - start < allowed * 0.9:
            # restart from best with random perturbation
            # pick random i,j again
            pert_h = best_h.copy()
            for _ in range(5):
                i2, j2 = np.random.choice(n, 2, replace=False)
                d2 = np.random.uniform(-sigma0, sigma0)
                pert_h[i2] += d2
                pert_h[j2] -= d2
            pert_h = np.clip(pert_h, 0, 1)
            pert_h = project_onto_simplex(pert_h, target_sum)
            current_h = pert_h
            current_c5 = compute_c5(current_h, n)
            temperature = 1.0
            no_improve = 0

    # final local refinement: greedy small moves
    for _ in range(2000):

        new_c5 = compute_c5(new_h, n)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    return best_h.tolist(), float(best_c5), n

def project_onto_simplex(v, target):
    # v: np.ndarray shape (n,)
    lo = -np.max(v)
    hi = 1 - np.min(v)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # correct small residual
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute to non-bound elements
        mask = (w > 0) & (w < 1)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            w[mask] += diff / cnt
            w = np.clip(w, 0, 1)
    return w

import numpy as np
import random
import time
import math

def _project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with sum equal to target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(40):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        mask = (w > 0.0) & (w < 1.0)
        cnt = int(mask.sum())
        if cnt:
            w[mask] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w

def _compute_c5(h: np.ndarray, n: int) -> float:
    """Return C5 = max convolution * dx for a given h."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def _sa_optimize(init_h: np.ndarray, n: int, time_limit: float):
    """Simulated annealing on a fixed size."""
    target = n / 2.0
    best_h = init_h.copy()
    best_score = _compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    temp = 1.0
    cooling = 0.9997
    sigma0 = 0.2
    no_imp = 0

    while time.time() - start < time_limit:
        sigma = sigma0 * temp
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1  # ensure i != j
        delta = np.random.uniform(-sigma, sigma)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _project_onto_simplex(new_h, target)
        new_score = _compute_c5(new_h, n)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = random.random() < prob

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h.copy()
                best_score = new_score
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        temp = max(temp * cooling, 1e-6)

        if no_imp > 5000:
            # restart from best with a few random tweaks
            tmp = best_h.copy()
            for _ in range(5):
                a = np.random.randint(n)
                b = np.random.randint(n - 1)
                if b >= a:
                    b += 1
                d = np.random.uniform(-sigma0, sigma0)
                tmp[a] += d
                tmp[b] -= d
            tmp = np.clip(tmp, 0.0, 1.0)
            cur_h = _project_onto_simplex(tmp, target)
            cur_score = _compute_c5(cur_h, n)
            temp = 1.0
            no_imp = 0

    return best_h, best_score

def _local_refine(h: np.ndarray, n: int, steps: int, step_size: float):
    """Simple greedy local search."""
    target = n / 2.0
    best_h = h.copy()
    best_score = _compute_c5(best_h, n)
    for _ in range(steps):
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.uniform(-step_size, step_size)
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _project_onto_simplex(new_h, target)
        new_score = _compute_c5(new_h, n)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------

    return best_h.tolist(), float(best_score), int(n_pts)
