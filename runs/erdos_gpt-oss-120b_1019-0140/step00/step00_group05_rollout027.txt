    start = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
        # Ensure sum correct. Compute sum and scale to enforce exactly n_points/2
        target_sum = n_points / 2.0
        current_sum = h0.sum()
        if abs(current_sum - target_sum) > 1e-9:
            # adjust by scaling within bounds
            scale = target_sum / current_sum
            h0 = np.clip(h0 * scale, 0, 1)
            # Renormalize precisely: adjust one element
            diff = target_sum - h0.sum()
            if abs(diff) > 1e-12:
                # add diff to a random index that can accommodate
                idx = prng.randint(n_points)
                # adjust within [0,1] bound
                avail = 1 - h0[idx] if diff>0 else h0[idx]
                adjust = min(abs(diff), avail)
                h0[idx] += np.sign(diff) * adjust
        # If we still don't match exactly, we can accept slight error.
        h = h0.copy()
    else:
        n_points = 200  # default
        # generate random initial satisfying sum constraint
        # random values in [0,1], then scale and adjust.
        h = prng.rand(n_points)
        # enforce sum
        target_sum = n_points/2.0
        # scale to target sum
        h = h / h.sum() * target_sum
        # Ensure within bounds (since scaling may cause >1). Clip and fix iteratively
        # iterate few times
        for _ in range(10):
            over = h > 1
            if not over.any():
                break
            excess = h[over] - 1
            total_excess = excess.sum()
            h[over] = 1
            # distribute excess reduction proportionally to those below 1
            below = h < 1
            if below.any():
                h[below] -= total_excess * (1 - h[below]) / (1 - h[below]).sum()
        # ensure sum close
        current_sum = h.sum()
        h = h * (target_sum / current_sum)
    # compute initial best
    best_h = h.copy()
    bestC = evaluate_erdos_solution(best_h)
    current_h = h.copy()
    currentC = bestC
    # Simulated annealing parameters
    max_time = budget_s - 0.5
    # constants for schedule
    T0 = 1e-3  # temperature scale
    step0 = 0.2
    # optionally set a schedule for step size
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        # apply random move
        i, j = prng.randint(n_points, size=2)
        while j == i:
            j = prng.randint(n_points)
        hi = current_h[i]
        hj = current_h[j]
        # compute allowed delta range for moving from j to i
        low = max(-hi, hj - 1)
        high = min(1 - hi, hj)
        if low >= high:
            continue  # cannot move
        # update step size based on elapsed time
        progress = (now - start) / max_time
        step = step0 * (1 - progress) + 0.001 * progress

        delta = (prng.rand() * 2 - 1) * step
        if delta < low:
            delta = low
        elif delta > high:
            delta = high

        new_h[j] = hj - delta
        # evaluate

        # acceptance

            accept = True
        else:

            T = T0 * (1 - progress) + 1e-6 * progress
            # accept with probability exp(-(newC - currentC) / T)
            # Actually we want acceptance if newC lower (good) else maybe uphill
            accept_prob = np.exp(-(newC - currentC) / T) if T > 0 else 0.0
            accept = prng.rand() < accept_prob
        if accept:
            current_h = new_h
            currentC = newC
            if newC < bestC:
                bestC = newC
                best_h = new_h.copy()
    # final best result
    # final check adjust sum exactly
    target_sum = n_points/2.0
    # adjust using scaling small

    if abs(total - target_sum) > 1e-9:

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); np.correlate implements sum_i a[i]*v[i+lag]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def adjust_sum_to_target(h: np.ndarray, target: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Adjust array h (in-place) so that its sum equals target while respecting 0≤h_i≤1.
    Small residuals are distributed uniformly and clipped.
    """
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Try to fix the whole deficit/surplus using a single index repeatedly
    attempts = 0
    max_attempts = 500
    while attempts < max_attempts and abs(diff) > 1e-12:
        idx = rng.randint(h.size)
        if diff > 0.0:
            # we need to increase this entry, but not beyond 1
            inc = min(diff, 1.0 - h[idx])
            if inc > 0.0:
                h[idx] += inc
                diff -= inc
        else:
            # need to decrease this entry, but not below 0
            dec = min(-diff, h[idx])
            if dec > 0.0:
                h[idx] -= dec
                diff += dec
        attempts += 1
    if abs(diff) > 1e-12:
        # distribute any remaining tiny error uniformly (may slightly violate bounds)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the step function h to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start = time.time()
    max_time = budget_s - 0.5  # safety margin

    # ---------- initialise ----------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
        target_sum = n_points / 2.0
        h = adjust_sum_to_target(init_h.copy(), target_sum, rng)
    except Exception:
        # no parent solution available
        n_points = 200
        target_sum = n_points / 2.0
        h = rng.rand(n_points)
        h = h / h.sum() * target_sum
        h = adjust_sum_to_target(h, target_sum, rng)

    # best-so-far record
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # annealing schedules
    T0 = 1e-3          # initial temperature
    step0 = 0.2        # initial step magnitude

    while True:
        now = time.time()
        if now - start >= max_time:
            break

        progress = (now - start) / max_time
        # linearly decreasing step and temperature
        step = step0 * (1.0 - progress) + 1e-4 * progress
        T = T0 * (1.0 - progress) + 1e-6 * progress

        # 30% chance to try a gradient-based move, otherwise pairwise exchange
        if rng.rand() < 0.3:
            # ---- gradient move ----
            # Identify shift that yields the current worst overlap
            corr_full = np.correlate(current_h, 1.0 - current_h, mode="full")
            k_idx = int(np.argmax(corr_full))
            k_shift = k_idx - (n_points - 1)        # integer lag (can be negative)

            # compute gradient of the maximal term w.r.t. h
            g = np.zeros_like(current_h)
            idx = np.arange(n_points)

            plus = idx + k_shift
            valid_plus = (plus >= 0) & (plus < n_points)
            g[valid_plus] += (1.0 - current_h[plus[valid_plus]])

            minus = idx - k_shift
            valid_minus = (minus >= 0) & (minus < n_points)
            g[valid_minus] -= current_h[minus[valid_minus]]

            # Project gradient to the hyperplane sum(g)=0 to keep total mass unchanged
            g -= g.mean()

            candidate = current_h - step * g
            np.clip(candidate, 0.0, 1.0, out=candidate)
            candidate = adjust_sum_to_target(candidate, target_sum, rng)
        else:
            # ---- pairwise exchange move ----
            i, j = rng.randint(n_points, size=2)
            if i == j:

            hi, hj = current_h[i], current_h[j]
            # feasible delta so that both entries stay in [0,1]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:

            delta = (rng.rand() * 2.0 - 1.0) * step
            if delta < low:
                delta = low
            elif delta > high:
                delta = high
            candidate = current_h.copy()
            candidate[i] = hi + delta
            candidate[j] = hj - delta

        cand_c5 = compute_c5(candidate)

        # decide acceptance
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - current_c5) / T) if T > 0.0 else 0.0
            accept = rng.rand() < prob

        if accept:
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()

    # Final sanity: ensure exact mass
    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n_points)
