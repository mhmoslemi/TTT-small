        # adjust to have exactly n/2 ones
        current_ones = int(np.sum(h))
        target_ones = n // 2
        if current_ones > target_ones:
            # reduce ones: find indices of ones and set some to 0
            idx_ones = np.flatnonzero(h == 1)
            to_zero = np.random.choice(idx_ones, size=current_ones - target_ones, replace=False)
            h[to_zero] = 0.0
        elif current_ones < target_ones:
            idx_zeros = np.flatnonzero(h == 0)
            to_one = np.random.choice(idx_zeros, size=target_ones - current_ones, replace=False)
            h[to_one] = 1.0

            break

            break
        # pick random swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:

            # revert

            h[j] = 0.0
        # optionally random restart after many failures
        # not needed
    return best_h, best_c5

def project_onto_box_and_sum(v, target_sum):
    # water-filling algorithm via binary search
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # correct any residual
    residual = target_sum - proj.sum()
    if abs(residual) > 1e-9:
        # distribute residual to non-bound entries
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            adjustment = residual / free.sum()
            proj[free] += adjustment
            # clip again
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import numpy as np, time, random, math

def _c5_of(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _project_box_sum(v, target):
    lo, hi = -np.max(v) - 1.0, np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - proj.sum()
    if abs(residual) > 1e-9:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += residual / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _grad_max_corr(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[:n - shift] += 1.0 - h[shift:]
        g[shift:]   -= h[:n - shift]
    else:
        s = -shift
        g[s:]      += 1.0 - h[:n - s]
        g[:n - s]  -= h[s:]
    return g

def _opt_continuous(h0, budget, target_c5=0.3808):
    n = h0.shape[0]
    target_sum = n / 2.0
    h = _project_box_sum(h0.copy(), target_sum)
    best_h, best_val = h.copy(), _c5_of(h)
    lr = 0.5
    start = time.time()
    while True:
        now = time.time()
        if now - start >= budget * 0.99:
            break
        conv = np.correlate(h, 1.0 - h, mode='full')
        val = np.max(conv) * (2.0 / n)
        if val < best_val - 1e-12:
            best_val, best_h = val, h.copy()
            if best_val <= target_c5:
                break
        lag = int(np.argmax(conv))
        shift = lag - (n - 1)
        g = _grad_max_corr(h, shift)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h = _project_box_sum(h_new, target_sum)
        lr = max(lr * 0.9995, 1e-5)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_total = time.time()

    # ---------- initial vector ----------
    try:
        parent = initial_h_values
        use_parent = True
    except NameError:
        use_parent = False

    if use_parent:
        n = len(parent)
        h_start = np.array(parent, dtype=np.float64)
    else:
        n = 200                     # even size, modest resolution
        h_start = np.random.rand(n)

    # enforce box and integral constraints
    h_start = np.clip(h_start, 0.0, 1.0)
    h_start = _project_box_sum(h_start, n / 2.0)

    # ---------- optimisation ----------
    remaining = budget_s - (time.time() - start_total)
    best_h, best_c5 = _opt_continuous(h_start, remaining)

    # optional random restarts if time permits
    while time.time() - start_total < budget_s * 0.95:
        budget_left = budget_s - (time.time() - start_total)
        if budget_left <= 1.0:
            break
        # fresh random start
        h_rand = np.random.rand(n)
        h_rand = _project_box_sum(h_rand, n / 2.0)
        h_opt, c5_opt = _opt_continuous(h_rand, min(5.0, budget_left))
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        if best_c5 <= 0.3808:
            break

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = _c5_of(best_h)

    return best_h, float(best_c5), len(best_h)
