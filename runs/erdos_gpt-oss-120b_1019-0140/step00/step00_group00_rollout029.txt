def binary_local_search(h, max_iters, time_limit):
    n = len(h)
    # Ensure binary integer (0 or 1) and sum = n//2
    # We'll operate on integer array (np.int8)
    hbin = np.round(h).astype(int)
    # adjust to correct sum: compute current sum; if less, set some 0s to 1; if more, set some 1s to 0.
    target_sum = n // 2
    cur_sum = hbin.sum()
    if cur_sum < target_sum:
        zeros = np.where(hbin == 0)[0]
        np.random.shuffle(zeros)
        hbin[zeros[:target_sum - cur_sum]] = 1
    elif cur_sum > target_sum:
        ones = np.where(hbin == 1)[0]
        np.random.shuffle(ones)
        hbin[ones[:cur_sum - target_sum]] = 0
    best_h = hbin.copy()
    best_c5 = evaluate_c5(best_h)
    start = time.time()
    iter = 0
    while iter < max_iters and (time.time() - start) < time_limit:
        # propose swap
        # select random 1 and 0 positions
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = evaluate_c5(cand)
        if cand_c5 < best_c5 - 1e-9:
            best_h = cand
            best_c5 = cand_c5
        else:

            pass
        iter += 1
    return best_h, best_c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Use convolution as float
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def hill_climb_binary(h, max_iter, time_limit):
    best_h = h.copy()
    best_val = evaluate_c5(best_h)
    n = len(h)
    iteration = 0
    while iteration < max_iter:
        # propose swap
        i = np.random.randint(n)
        # ensure we propose swapping opposite values
        if best_h[i]==1:
            # choose random zero
            zero_idxs = np.where(best_h==0)[0]
            if len(zero_idxs)==0:
                break
            j = zero_idxs[np.random.randint(len(zero_idxs))]
        else:
            # start with zero, swap with random one
            one_idxs = np.where(best_h==1)[0]
            if len(one_idxs)==0:
                break
            j = one_idxs[np.random.randint(len(one_idxs))]
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_val = evaluate_c5(cand)
        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
        iteration += 1
        if time.time() - start_time > time_limit:
            break
    return best_h, best_val
```

import time
import numpy as np

def evaluate_c5(h):
    """Return C5 for a sample vector h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    hf = h.astype(float, copy=False)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return float(np.max(corr) * dx)

def binarize_to_half(vec):
    """Return a binary (0/1) vector with exactly half the entries 1,
    choosing the largest entries of vec."""
    n = len(vec)
    target = n // 2
    idx = np.argsort(-vec)  # descending
    b = np.zeros(n, dtype=int)
    b[idx[:target]] = 1
    return b

def random_binary(n, rng):
    """Generate a random binary vector with half the entries 1."""
    perm = rng.permutation(n)
    b = np.zeros(n, dtype=int)
    b[perm[:n // 2]] = 1
    return b

def hill_climb_binary(start_h, rng, time_limit):
    """Simple hill‑climbing on binary vectors.
    Swaps a 1 with a 0 if the move lowers C5."""
    start = time.time()
    best_h = start_h.copy()
    best_val = evaluate_c5(best_h)
    n = best_h.size
    # maintain index lists for fast swaps
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]

    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_one = rng.integers(len(ones))
        i_zero = rng.integers(len(zeros))
        i = ones[i_one]
        j = zeros[i_zero]

        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = evaluate_c5(cand)

        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
            # update index arrays in O(1)
            ones[i_one] = j
            zeros[i_zero] = i

        if best_val <= 0.3808:  # target reached – can stop early
            break
    return best_h, best_val

def greedy_binary(n, rng):
    """Greedy construction of a binary vector.
    Repeatedly inserts a 1 at a position that most improves C5."""
    h = np.zeros(n, dtype=int)
    target = n // 2
    free = list(range(n))
    for _ in range(target):
        sample = rng.choice(free, size=min(12, len(free)), replace=False)
        best_pos = None
        best_val = np.inf
        for pos in sample:
            h[pos] = 1
            val = evaluate_c5(h)
            if val < best_val:
                best_val = val
                best_pos = pos
            h[pos] = 0
        if best_pos is None:
            best_pos = rng.choice(free)
        h[best_pos] = 1
        free.remove(best_pos)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Obtain a base vector (make length even for binary representation)
    # ------------------------------------------------------------------
    raw = globals().get('initial_h_values')
    if raw is not None:
        base = np.array(raw, dtype=float)
        if base.size % 2 == 1:               # repeat to obtain even length
            base = np.repeat(base, 2)
    else:
        base = None

    # Choose an even length not exceeding 256
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 64
    if n0 > 256:
        n0 = 256 - (256 % 2)          # force even

    # ------------------------------------------------------------------
    # 2. Build a collection of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    if base is not None:
        candidates.append(binarize_to_half(base))
    candidates.append(random_binary(n0, rng))
    candidates.append(greedy_binary(n0, rng))

    # ------------------------------------------------------------------
    # 3. Initial hill‑climbing phase (≈40% of total budget)
    # ------------------------------------------------------------------
    total_budget = float(budget_s)
    time_for_initial = min(total_budget * 0.4, total_budget - 5.0)
    per_candidate = time_for_initial / len(candidates)

    best_h = None
    best_val = np.inf

    for cand in candidates:
        h_opt, v_opt = hill_climb_binary(cand, rng, per_candidate)
        if v_opt < best_val:
            best_h, best_val = h_opt, v_opt

    # ------------------------------------------------------------------
    # 4. Successive refinement by doubling the resolution
    # ------------------------------------------------------------------
    max_n = 256
    while True:
        elapsed = time.time() - start_time
        remaining = total_budget - elapsed
        if remaining < 5.0 or best_h.shape[0] >= max_n:
            break

        # upsample current best (simple repeat keeps integrality)
        up_h = np.repeat(best_h, 2)
        new_n = up_h.shape[0]

        # allocate a slice of the remaining budget to this level
        phase_time = min(remaining * 0.3, 30.0)
        sub_time = phase_time / 2.0

        # try the upsampled version
        h1, v1 = hill_climb_binary(up_h, rng, sub_time)
        if v1 < best_val:
            best_h, best_val = h1, v1

        # also try a fresh random start at this resolution
        rand_start = random_binary(new_n, rng)
        h2, v2 = hill_climb_binary(rand_start, rng, sub_time)
        if v2 < best_val:
            best_h, best_val = h2, v2

        # if we already beat the target, stop early
        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    h_out = best_h.astype(float)
    n_out = int(h_out.shape[0])
    c5_out = float(best_val)
    return (h_out, c5_out, n_out)
