def propose(h):
    n = len(h)
    # choose two distinct indices i, j.
    i, j = np.random.choice(n, 2, replace=False)
    # compute max allowable delta such that h_i + delta ≤1 and h_j - delta ≥0
    # Also we might want to allow negative delta? We'll set delta positive in direction of moving some mass.
    # Let's generate delta randomly between -max_negative ... max_positive? Actually we can decide to move mass from i to j.
    # Option: choose sign randomly; we treat i as donor (decrease) and j as receiver (increase), or vice versa.
    # For generality: let donor = i, receiver = j.
    # We'll generate delta positive up to min(h[donor], 1 - h[receiver]).
    # Additionally to allow both directions, we can randomly swap donor/receiver.
    if random.random() < 0.5:
        donor, receiver = i, j
    else:
        donor, receiver = j, i
    max_delta = min(h[donor], 1.0 - h[receiver])
    if max_delta <= 0:
        # can't move; return unchanged
        return h
    # Choose delta uniformly between 0 and max_delta * factor
    # Factor maybe 0.5 for moderate moves.
    delta = max_delta * random.random()
    # Apply
    new_h = h.copy()
    new_h[donor] -= delta
    new_h[receiver] += delta
    return new_h

def binary_from_continuous(arr, target_ones):
    # arr: continuous values in [0,1].
    # produce binary vector of same length with exactly target_ones ones.
    order = np.argsort(-arr)  # descending
    bin_vec = np.zeros_like(arr, dtype=int)

    return bin_vec

import numpy as np
import random
import math
import time

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def propose_move(h: np.ndarray, step_scale: float) -> np.ndarray:
    """Generate a new feasible h by moving a small amount from one index to another."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # random direction: i receives, j gives
    if random.random() < 0.5:
        inc, dec = i, j
    else:
        inc, dec = j, i
    max_inc = 1.0 - h[inc]
    max_dec = h[dec]
    max_delta = min(max_inc, max_dec)
    if max_delta <= 0.0:
        return h
    delta = max_delta * step_scale * np.random.rand()
    new_h = h.copy()
    new_h[inc] += delta
    new_h[dec] -= delta
    return new_h


def initialize_h(n: int) -> np.ndarray:
    """Construct an initial feasible vector of length n."""
    # try to use the pre‑imported initial_h_values if present
    try:
        init = initial_h_values  # type: ignore
    except NameError:
        init = None

    if init is not None:
        init_arr = np.asarray(init, dtype=float)
        n0 = init_arr.shape[0]
        xs = np.linspace(0.0, 2.0, n0, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, init_arr)
        # binary projection: keep exactly half the entries equal to 1
        target_ones = n // 2
        idx = np.argsort(-interp)[:target_ones]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        return h
    else:
        # cosine seed – integrates to 1 analytically
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = (1.0 + np.cos(np.pi * x)) / 2.0
        target_sum = n / 2.0
        s = h.sum()
        if s != 0.0:
            h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            idx = np.random.choice(n)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        return h

    """
