    # We find lambda such that sum(min(max(h_i + lambda, 0),1)) = target_sum

        p = np.clip(h + mid, 0.0, 1.0)
        s = p.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    p = np.clip(h + (low+high)/2, 0.0, 1.0)

import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def projection_feasible(h):
    n = len(h)
    target = n / 2.0
    # using binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        p = np.clip(h + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            low = mid
        else:
            high = mid
    p = np.clip(h + (low + high) / 2.0, 0.0, 1.0)
    # correct any remainder
    diff = target - p.sum()
    if abs(diff) > 1e-9:
        # adjust free entries
        free = (p > 0) & (p < 1)
        if free.any():
            p[free] += diff / free.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def objective(v):
    h = np.clip(v, 0.0, 1.0)
    n = len(h)
    target = n/2.0
    diff = h.sum() - target
    penalty = 1e5 * (diff**2)  # high penalty
    C5 = evaluate_c5(h)
    return C5 + penalty

def project_sum(h, target):
    low, high = -1.0, 1.0
    # binary search 50 iterations
    for _ in range(60):
        mid = (low+high)/2
        p = np.clip(h + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            low = mid
        else:
            high = mid
    # final p
    p = np.clip(h + (low+high)/2, 0.0, 1.0)
    # adjust any small diff
    diff = target - p.sum()
    # Spread diff over non-bound entries if any remain
    if abs(diff) > 1e-12:
        free = (p > 0) & (p < 1)
        if free.any():
            p[free] += diff / free.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

import time
from scipy.optimize import differential_evolution

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    # binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            low = mid
        else:
            high = mid
    p = np.clip(h + (low + high) * 0.5, 0.0, 1.0)
    # adjust final tiny diff
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        free = (p > 0) & (p < 1)
        if free.any():
            p[free] += diff / free.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def objective(v, target):
    h = project_to_sum(v, target)
    return compute_c5(h)

def hill_climb(h, score, time_deadline):
    n = h.shape[0]
    rng = np.random.default_rng()
    best_score = score
    best_h = h.copy()
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # max feasible delta moving from j to i (adding delta to i, subtracting delta from j)
        max_inc_i = 1.0 - best_h[i]
        max_dec_j = best_h[j]
        max_delta = min(max_inc_i, max_dec_j)
        if max_delta == 0.0:
            continue
        # also allow moving opposite direction
        max_inc_j = 1.0 - best_h[j]
        max_dec_i = best_h[i]
        # pick a direction randomly
        if rng.random() < 0.5:
            # move mass from j to i
            delta = rng.uniform(0, max_delta)
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # move mass from i to j
            max_delta2 = min(max_inc_j, max_dec_i)
            if max_delta2 == 0.0:
                continue
            delta = rng.uniform(0, max_delta2)
            new_h = best_h.copy()
            new_h[j] += delta
            new_h[i] -= delta
        # evaluate
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

    start = time.time()

    # Determine n_points and initial vector

    best_h = init.copy()
    best_score = compute_c5(best_h)
    # DE optimization
    max_iter = 500  # will be cut by time
    pop_factor = 8

    def de_callback(xk, convergence=None):

        # xk is the best candidate seen this iteration
        h = project_to_sum(xk, target_sum)
        score = compute_c5(h)
        if score < best_score:
            best_h = h.copy()
            best_score = score
        if time.time() - start > budget_s * 0.95:
            return True  # stop diffusion
        return False

    # Run DE
    result = differential_evolution(
        func=lambda v: objective(v, target_sum),
        bounds=bounds,

    best_h = project_to_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

        best_h, best_score = hill_climb(best_h, best_score, deadline)

    best_h = project_to_sum(best_h, target_sum)
    final_score = compute_c5(best_h)
    n_points = best_h.shape[0]

import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while keeping 0≤h≤1."""
    low, high = -1.0, 1.0
    # binary search for the shift λ such that sum(clamp(h+λ,0,1))=target
    for _ in range(30):
        mid = (low + high) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() < target:
            low = mid
        else:
            high = mid
    p = np.clip(h + (low + high) * 0.5, 0.0, 1.0)
    # tiny adjustment if needed
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        free = (p > 0.0) & (p < 1.0)
        if free.any():
            p[free] += diff / free.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def objective(v: np.ndarray) -> float:
    """Objective for DE: C5 after projecting v onto the feasible set."""
    n = v.shape[0]
    target = n / 2.0
    h = project_to_sum(v, target)
    return compute_c5(h)

def hill_climb(start_h: np.ndarray, start_score: float, deadline: float):
    """Simple pairwise mass‑transfer hill climbing."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_score = start_score
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            # move mass from j to i
            max_delta = min(1.0 - best_h[i], best_h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.uniform(0.0, max_delta)
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # move mass from i to j
            max_delta = min(1.0 - best_h[j], best_h[i])
            if max_delta <= 0.0:
                continue
            delta = rng.uniform(0.0, max_delta)
            new_h = best_h.copy()
            new_h[j] += delta
            new_h[i] -= delta
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

    bounds = [(0.0, 1.0)] * n

    pop_factor = 10          # actual population size = pop_factor * n
    max_iter = 400           # enough for good exploration (early stop if time runs out)

    # Run DE
    result = differential_evolution(

    best_h = project_to_sum(result.x, target)

    best_h = project_to_sum(best_h, target)
