def subgradient_max(h):
    n = h.size
    v = 1 - h
    corr = np.correlate(h, v, mode='full')
    max_val = corr.max()
    # indices with max_val (floating errors)
    # Use tolerance = 1e-12 maybe but near >? Might have exact equality.
    # We'll use np.flatnonzero(np.isclose(corr, max_val, atol=1e-12)).
    max_idxs = np.flatnonzero(np.isclose(corr, max_val, atol=1e-12))
    # choose random shift among them
    idx = np.random.choice(max_idxs)
    s = idx - (n - 1)
    g = np.zeros_like(h)
    if s >= 0:
        g[:n-s] += v[s:]
        g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] += v[:n-t]
        g[:n-t] -= h[t:]
    return g

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _subgradient_max(h: np.ndarray) -> np.ndarray:
    """Subgradient of the max overlap functional at h."""
    n = h.size
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    max_val = corr.max()
    # choose a random shift attaining the maximum (break ties randomly)
    idxs = np.flatnonzero(np.isclose(corr, max_val, atol=1e-12))
    idx = np.random.choice(idxs)
    s = idx - (n - 1)  # shift amount (in samples)
    g = np.zeros_like(h)
    if s >= 0:
        # overlap of h[i] with v[i+s]
        g[: n - s] += v[s:]
        g[s:] -= h[: n - s]
    else:
        t = -s
        g[t:] += v[: n - t]
        g[: n - t] -= h[t:]
    return g

def _project_box_sum(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project onto the feasible set:
        0 ≤ h_i ≤ 1,  sum_i h_i = target_sum
    Using a simple binary‑search on the Lagrange multiplier λ solving
        sum_i clip(x_i - λ, 0, 1) = target_sum .
    """
    lo = np.min(x) - 1.0      # forces all entries to 1 after clipping
    hi = np.max(x)            # forces all entries to 0 after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a plain list), its C5 bound, and the number of points.
    The routine performs a projected stochastic gradient descent with simulated‑annealing
    acceptance, starting from the supplied `initial_h_values` if present.
    """
    np.random.seed(seed)
    start = time.time()
    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        n_target = 256               # work at a richer resolution
        # linear interpolation to the new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(x, x0, init)
    else:
        n_target = 256
        h = np.full(n_target, 0.5, dtype=float)

    target_sum = n_target / 2.0
    h = _project_box_sum(h, target_sum)

    cur_h = h.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # ----- optimisation parameters -----------------------------------------
    base_step = 0.20                     # initial step magnitude
    min_step = 1e-4
    max_iter_without_impr = 0

    # ----- main loop -------------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s - 0.05:               # small safety margin
            break

        # stop early if we already beat the target bound
        if best_score <= 0.3808:
            break

        # stochastic projected (sub)gradient step
        grad = _subgradient_max(cur_h)
        # linear annealing schedule
        frac = max(0.0, 1.0 - elapsed / budget_s)
        step = max(min_step, base_step * frac)

        proposed = _project_box_sum(cur_h - step * grad, target_sum)
        prop_score = _compute_c5(proposed)

        delta = prop_score - cur_score
        # Metropolis‑type acceptance
        if delta < 0.0 or (frac > 0.0 and np.exp(-delta / (frac + 1e-12)) > np.random.rand()):
            cur_h = proposed
            cur_score = prop_score
            if prop_score < best_score:
                best_score = prop_score
                best_h = proposed.copy()
                max_iter_without_impr = 0
            else:
                max_iter_without_impr += 1
        else:
            max_iter_without_impr += 1

        # occasional random mass transfer to improve mixing
        if np.random.rand() < 0.012:
            i, j = np.random.randint(0, n_target, size=2)
            if i != j:
                # move a small amount δ from i to j (preserving the sum)
                max_up = min(cur_h[i], 1.0 - cur_h[j])
                max_down = min(cur_h[j], 1.0 - cur_h[i])
                delta_rand = (np.random.rand() * 2.0 - 1.0) * 0.08
                if delta_rand > max_up:
                    delta_rand = max_up
                if delta_rand < -max_down:
                    delta_rand = -max_down
                if abs(delta_rand) > 1e-12:
                    cur_h[i] -= delta_rand
                    cur_h[j] += delta_rand
                    cur_h = _project_box_sum(cur_h, target_sum)
                    cur_score = _compute_c5(cur_h)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = cur_h.copy()
                        max_iter_without_impr = 0

        # optional early exit if stuck for many iterations
        if max_iter_without_impr > 50000:
            break

    return best_h.tolist(), float(best_score), n_target
