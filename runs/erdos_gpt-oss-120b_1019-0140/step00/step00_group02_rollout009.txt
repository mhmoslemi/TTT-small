import numpy as np, time, random, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def adjust_to_sum(h):
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # find index where adjustment feasible
    if diff > 0:
        # need to raise some element
        idx = np.where(h < 1 - 1e-12)[0]
        if idx.size == 0:
            return h
        i = idx[0]
        incr = min(diff, 1.0 - h[i])
        h[i] += incr
        # if still diff not zero due to clipping, adjust another
        if abs(diff - incr) > 1e-12:
            return adjust_to_sum(h)  # recursion (few steps)
    else:
        idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            return h
        i = idx[0]
        decr = min(-diff, h[i])
        h[i] -= decr
        if abs(diff + decr) > 1e-12:
            return adjust_to_sum(h)
    return h

import time, math, random, numpy as np

def _c5_score(vec):
    """Return the C5 value for a step function given as a 1‑D numpy array."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h)
    c = np.correlate(vec, 1.0 - vec, mode='full')
    return float(c.max() * dx)

def _repair(vec):
    """Clip into [0,1] and enforce the exact integral constraint."""
    n = vec.shape[0]
    target = n / 2.0
    np.clip(vec, 0.0, 1.0, out=vec)
    diff = target - vec.sum()
    if abs(diff) > 1e-12:
        # pick a component that can absorb the whole difference
        feas = np.where((vec + diff >= 0.0) & (vec + diff <= 1.0))[0]
        if feas.size:
            idx = np.random.choice(feas)
            vec[idx] += diff
        else:
            # fallback: disperse the correction uniformly (should be rare)
            vec += diff / n
            np.clip(vec, 0.0, 1.0, out=vec)
            # final tiny correction on a random feasible entry
            diff2 = target - vec.sum()
            if abs(diff2) > 1e-12:
                feas2 = np.where((vec + diff2 >= 0.0) & (vec + diff2 <= 1.0))[0]
                if feas2.size:
                    vec[np.random.choice(feas2)] += diff2
    return vec

def run(seed=42, budget_s=1100, **kwargs):

    The function respects the budget, keeps h in [0,1] and enforces

    else:
        n0 = 45
        n = n0 * 2
        h = np.zeros(n, dtype=float)
        half = n // 2

        np.random.shuffle(h)

    h = _repair(h.copy())
