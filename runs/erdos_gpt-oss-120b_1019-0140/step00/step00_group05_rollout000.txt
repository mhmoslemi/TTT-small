Projection algorithm:

Given x, we want y = min(1, max(0, x - lambda)), where lambda is chosen to satisfy sum(y) = target. Since the function sum(min(1, max(0, x - lambda))) is piecewise linear decreasing in lambda, we can find lambda by binary search.

def project_to_bounds_sum(x, target, eps=1e-9, max_iter=100):
    low = -np.max(x) - 1.0  # sufficiently low ensures all x - low > 1
    high = np.max(x) + 1.0  # sufficiently high ensures all x - high < 0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low+high)/2.0, 0.0, 1.0)
```

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # initialize h
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        n_points = 70
        h = np.zeros(n_points, dtype=float)
        # assign half entries to 1 in alternating pattern
        h[::2] = 1.0
    n_points = h.shape[0]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    # clip and project to satisfy bounds and sum
    np.clip(h, 0.0, 1.0, out=h)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        h = project_to_bounds_sum(h, target_sum)
    # compute correlation function
    def compute_c5(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return corr.max() * dx
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.perf_counter()
    max_step = 0.05  # max amount moved per iteration
    it = 0
    no_improve = 0
    while True:
        elapsed = time.perf_counter() - start
        if elapsed >= budget_s * 0.98:
            break
        # propose move
        i, j = rng.choice(n_points, size=2, replace=False)
        max_inc = 1.0 - cur_h[i]
        max_dec = cur_h[j]
        if max_inc <= 0 and max_dec <= 0:
            continue
        lo = -max_dec
        hi = max_inc
        if lo >= hi:
            continue
        # limit step magnitude
        if hi - lo > 2 * max_step:
            # restrict to a symmetric interval around 0 with width 2*max_step
            lo = max(lo, -max_step)
            hi = min(hi, max_step)
            if lo >= hi:
                continue
        d = rng.uniform(lo, hi)
        if abs(d) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d
        # no need to clip; ensured within bounds.
        new_score = compute_c5(new_h)
        # accept if better else maybe probabilistic acceptance
        if new_score < cur_score:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:

                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:

            T = 0.01 * t_frac + 0.0001
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                cur_h = new_h
                cur_score = new_score
            no_improve += 1
        it += 1

            no_improve = 0
    # final projection to guarantee feasibility
    best_h = project_to_bounds_sum(best_h, target_sum)

import numpy as np
import time
import math

def _project_to_bounds_sum(x, target, eps=1e-12, max_iter=100):
    """Project x onto [0,1]^n with sum equal to target (L2 minimal)."""
    lo = -np.max(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ---- initialization -------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        n0 = 70
        h = np.zeros(n0, dtype=float)
        h[::2] = 1.0  # simple alternating start
    n_points = h.shape[0]
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # enforce bounds and exact integral
    np.clip(h, 0.0, 1.0, out=h)
    if abs(h.sum() - target_sum) > 1e-12:
        h = _project_to_bounds_sum(h, target_sum)

    # ---- helper ---------------------------------------------------------
    def _c5_score(vec):
        """Compute C5 for a given discretised h."""
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return corr.max() * dx

    cur_h = h.copy()
    cur_score = _c5_score(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # ---- local search ---------------------------------------------------
    start = time.perf_counter()
    max_step = 0.05
    it = 0
    no_improve = 0

    while True:
        elapsed = time.perf_counter() - start
        if elapsed >= budget_s * 0.98:
            break

        i, j = rng.choice(n_points, size=2, replace=False)
        max_inc = 1.0 - cur_h[i]   # how much we can add to i
        max_dec = cur_h[j]         # how much we can take from j
        if max_inc <= 0 and max_dec <= 0:
            continue

        lo = -max_dec
        hi = max_inc
        if lo >= hi:
            continue

        # cap the move size to keep exploration smooth
        if hi - lo > 2 * max_step:
            lo = max(lo, -max_step)
            hi = min(hi, max_step)
            if lo >= hi:
                continue

        d = rng.uniform(lo, hi)
        if abs(d) < 1e-14:
            continue

        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d

        new_score = _c5_score(new_h)

        if new_score < cur_score:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Simulated‑annealing style acceptance
            t_frac = (budget_s - elapsed) / budget_s
            T = 0.01 * t_frac + 0.0001
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                cur_h = new_h
                cur_score = new_score
            no_improve += 1

        it += 1
        if it % 5000 == 0:
            max_step *= 0.99

        # restart with jitter if stuck too long
        if no_improve >= 20000:
            jitter = rng.normal(scale=0.02, size=n_points)
            cur_h = best_h + jitter
            np.clip(cur_h, 0.0, 1.0, out=cur_h)
            cur_h = _project_to_bounds_sum(cur_h, target_sum)
            cur_score = _c5_score(cur_h)
            no_improve = 0

    # final feasibility check
    best_h = _project_to_bounds_sum(best_h, target_sum)
    final_score = _c5_score(best_h)

    return best_h, final_score, n_points
