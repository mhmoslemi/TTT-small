def propose_move(h):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # direction: increase i, decrease j
    max_inc = 1.0 - h[i]
    max_dec = h[j]  # can transfer at most this from j to i
    max_transfer = min(max_inc, max_dec)
    # Also consider opposite direction; random turn
    if rng.random() < 0.5:
        # increase i, decrease j
        if max_transfer > 0:
            delta = max_transfer * rng.random() * step_scale
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            return h_new
    else:
        max_inc_j = 1.0 - h[j]
        max_dec_i = h[i]
        max_transfer2 = min(max_inc_j, max_dec_i)
        if max_transfer2 > 0:
            delta = max_transfer2 * rng.random() * step_scale
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            return h_new
    return None

import time
import numpy as np

# --- Helper functions ---------------------------------------------------------

def _project_h(h: np.ndarray, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Clip values to [0,1] and adjust to satisfy sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # try to fix the sum in at most two passes
    for _ in range(2):
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1 - 1e-12)[0]
            if idxs.size == 0:
                break
            idx = rng.choice(idxs)
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
        else:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
            idx = rng.choice(idxs)
            dec = min(-diff, h[idx])
            h[idx] -= dec
        diff = target_sum - h.sum()
    # final tiny correction if needed
    final_diff = target_sum - h.sum()
    if abs(final_diff) > 1e-12:
        if final_diff > 0:
            idxs = np.where(h < 1)[0]
        else:
            idxs = np.where(h > 0)[0]
        if idxs.size:
            idx = rng.choice(idxs)
            h[idx] = np.clip(h[idx] + final_diff, 0.0, 1.0)
    return h


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C5 for a discrete step function."""
    try:
        # the harness may provide a specialized evaluator
        return evaluate_erdos_solution(h)
    except Exception:
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return dx * np.max(corr)


def _transfer_worst_shift(h: np.ndarray, shift: int, rng: np.random.Generator) -> np.ndarray:
    """
    Perform a mass transfer from a larger entry to a smaller entry
    across the worst shift direction.
    """
    n = len(h)
    i_start = max(0, -shift)
    i_end = min(n, n - shift)
    if i_end <= i_start:
        return h.copy()
    for _ in range(10):
        i = rng.integers(i_start, i_end)
        j = i + shift
        # direction  i -> j
        if h[i] > 0 and h[j] < 1:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 0:
                delta = max_delta * rng.random() * 0.5
                nh = h.copy()
                nh[i] -= delta
                nh[j] += delta
                return nh
        # opposite direction  j -> i
        if h[j] > 0 and h[i] < 1:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta > 0:
                delta = max_delta * rng.random() * 0.5
                nh = h.copy()
                nh[j] -= delta
                nh[i] += delta
                return nh
    return h.copy()


def _random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.1) -> np.ndarray:
    """Add a zero‑sum random perturbation to a few coordinates."""
    n = len(h)
    k = max(2, n // 20)                     # number of perturbed entries
    idx = rng.choice(n, size=k, replace=False)
    changes = (rng.random(k) - 0.5) * scale
    changes -= changes.mean()                # enforce zero sum
    nh = h.copy()
    nh[idx] += changes
    return nh

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    the Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)

    # ----- initialise --------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 84                                   # a modest default size
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    target_sum = n / 2.0
    h = _project_h(h, target_sum, rng)

    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    start = time.time()
    it = 0
    no_improve = 0
    max_no_improve = 20000

    while time.time() - start < budget_s * 0.99:
        # find the shift producing the current maximum overlap
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        if shift != 0:
            new_h = _transfer_worst_shift(current_h, shift, rng)
        else:
            # shift 0 is never the worst (its contribution is zero);
            # we simply apply a small random tweak.
            new_h = _random_perturb(current_h, rng, scale=0.05)

        new_h = _project_h(new_h, target_sum, rng)
        new_c5 = _c5_from_h(new_h)

        # accept only improving moves (simple hill‑climb)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            no_improve += 1

        # occasional extra random shake to escape plateaus
        if it % 500 == 0:
            shake = _random_perturb(current_h, rng, scale=0.02)
            shake = _project_h(shake, target_sum, rng)
            shake_c5 = _c5_from_h(shake)
            if shake_c5 < best_c5:
                best_h = shake.copy()
                best_c5 = shake_c5
                current_h = shake.copy()
                current_c5 = shake_c5
                no_improve = 0

        # restart from the best solution if stuck for too long
        if no_improve >= max_no_improve:
            current_h = best_h.copy()
            # add a tiny random jitter before continuing
            jitter = (rng.random(n) - 0.5) * 0.02
            current_h = _project_h(current_h + jitter, target_sum, rng)
            current_c5 = _c5_from_h(current_h)
            no_improve = 0

        it += 1

    # final projection & evaluation
    best_h = _project_h(best_h, target_sum, rng)
    best_c5 = _c5_from_h(best_h)

    return (best_h, best_c5, n)
