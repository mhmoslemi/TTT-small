def proj_box_sum(x, target):
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

import numpy as np
import time

def _proj_box_sum(x, target):
    # Project onto [0,1] box and sum constraint sum(x)=target.
    # Solve by finding lambda such that sum(clip(x+lambda,0,1)) = target.
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode='full')
    return np.max(O) * dx

def _evaluate(h):
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        return _compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Choose number of sample points (multiple of 45 if possible)
    base_n = 45
    factor = 4   # 45*4 = 180 points, well under 1000
    n = base_n * factor
    target_sum = n / 2.0
    dx = 2.0 / n

    # Initialise h
    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xt, x0, init, left=init[0], right=init[-1])
        h = np.clip(h, 0.0, 1.0)
    except NameError:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0

    h = _proj_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    start_time = time.time()
    beta = 5.0          # smoothing for soft‑max
    step_factor = 0.5   # base step magnitude
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break

        # correlation vector (unscaled)
        O = np.correlate(h, 1.0 - h, mode='full')
        O_scaled = O * dx
        # soft‑max weights
        exp_arg = beta * O_scaled
        max_arg = exp_arg.max()
        w = np.exp(exp_arg - max_arg)
        w /= w.sum()

        # gradient of the smoothed max (scale by dx)
        w_rev = w[::-1]
        conv1 = np.convolve(w_rev, 1.0 - h, mode='full')
        conv2 = np.convolve(w, h, mode='full')
        start = n - 1
        grad = (conv1[start:start + n] - conv2[start:start + n]) * dx

        # step size inversely proportional to gradient norm
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = step_factor / norm
        else:
            step = 0.0

        h_new = h - step * grad
        h_new = _proj_box_sum(h_new, target_sum)

        # evaluate
        cur_c5 = _evaluate(h_new)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_new.copy()

        # simple acceptance: move to new point always
        h = h_new

        iteration += 1

        # schedule updates
        if iteration % 300 == 0:
            beta = min(beta * 1.4, 200.0)
            step_factor *= 0.85

                    h = _proj_box_sum(h, target_sum)
