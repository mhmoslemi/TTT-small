def max_overlap_fft(h):
    n = len(h)
    a = h
    b = 1 - h
    b_rev = b[::-1]
    N = 1
    while N < n + n - 1:
        N <<= 1
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b_rev, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)
    # conv length N; first (n + n -1) entries correspond to full correlation.
    corr = conv[:2*n-1]   # real values
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)    # dx = 2/n

def compute_score(h):
    n = len(h)
    a = h
    b_rev = (1 - h)[::-1]
    N = 1
    while N < 2*n -1:
        N <<= 1
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b_rev, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)
    # only first 2*n-1 entries
    corr = conv[:2*n-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def project_to_box_sum(v, target):
    lo = -1.0
    hi = 1.0
    # expand bounds
    # can enlarge lo, hi if needed
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # final small adjustment for sum errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly among interior points
        interior = (x > 1e-12) & (x < 1 - 1e-12)
        cnt = interior.sum()
        if cnt > 0:
            adjustment = diff / cnt
            x[interior] += adjustment
            x = np.clip(x, 0.0, 1.0)
    return x

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """next power of two >= x"""
    p = 1
    while p < x:
        p <<= 1
    return p

def _project_to_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto {0<=x_i<=1, sum x_i = target}."""
    # binary search for lambda such that sum(clamp(v - lam,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(interior.sum())
        if cnt:
            x[interior] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_fft(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h) using FFT (returns array, not scaled)."""
    n = h.shape[0]
    a = h
    b_rev = (1.0 - h)[::-1]
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b_rev, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)
    return conv[: 2 * n - 1]

def _score_fft(h: np.ndarray) -> float:
    """Maximum overlap value (C5) computed via FFT, returns scaled by dx."""
    n = h.shape[0]
    max_corr = _corr_fft(h).max()
    return max_corr * (2.0 / n)

def _score_exact(h: np.ndarray) -> float:
    """Maximum overlap using numpy.correlate (exact formula)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _grad_from_corr(h: np.ndarray, corr: np.ndarray, beta: float) -> np.ndarray:
    """Weighted gradient of smoothed max-overlap (beta-regularized)."""
    n = h.shape[0]
    # log-sum-exp weights
    m = corr.max()
    exp_vals = np.exp(beta * (corr - m))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros_like(h)
    shift = -(n - 1)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            shift += 1
            continue
        s = shift
        i0 = max(0, -s)
        i1 = min(n, n - s)
        if i0 >= i1:
            shift += 1
            continue
        ip = i0 + s
        length = i1 - i0
        grad[i0:i1] += weight * (1.0 - h[ip:ip + length])
        grad[ip:ip + length] += weight * (-h[i0:i1])
        shift += 1
    return grad * (2.0 / n)  # include dx factor
