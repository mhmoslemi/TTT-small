def perturb_pairwise(h, max_step):
    n = len(h)
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    ai = h[i]
    aj = h[j]
    # max delta to add to i and subtract from j
    max_pos = min(max_step, 1 - ai, aj)   # positive delta: increase i, decrease j
    max_neg = min(max_step, ai, 1 - aj)   # negative delta: decrease i, increase j
    # Choose direction
    if max_pos <= 0 and max_neg <= 0:
        return None
    if max_pos > 0 and max_neg > 0:
        # random direction
        if np.random.rand() < 0.5:
            # positive
            delta = np.random.uniform(0, max_pos)
        else:
            delta = -np.random.uniform(0, max_neg)
    elif max_pos > 0:
        delta = np.random.uniform(0, max_pos)
    else:
        delta = -np.random.uniform(0, max_neg)
    h_new = h.copy()
    h_new[i] = ai + delta

def project_to_capped_simplex(v, target, lower=0.0, upper=1.0, max_iter=100):
    v = np.clip(v, lower, upper)
    # If sum already target, return
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    # binary search for lambda
    lo = -1.0  # lower bound for lambda
    hi = 1.0   # upper bound; ensure that v - lambda after clipping yields sum monotonic decreasing with lambda
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, lower, upper)
        s = np.sum(proj)
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)/2, lower, upper)
    # final tweak if still not exactly target
    diff = target - np.sum(proj)
    if abs(diff) > 1e-9:
        # distribute diff equally among non-bound entries
        free = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if np.any(free):
            proj[free] += diff / np.sum(free)
            proj = np.clip(proj, lower, upper)
        else:
            # all at bounds; just distribute over all
            proj += diff / len(v)
            proj = np.clip(proj, lower, upper)
    return proj

def perturb_pairwise(h, max_step):
    n = len(h)
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    ai = h[i]
    aj = h[j]
    # allowed delta positive: increase i, decrease j
    max_pos = min(max_step, 1 - ai, aj)
    # allowed delta negative: decrease i, increase j
    max_neg = min(max_step, ai, 1 - aj)
    if max_pos <= 0 and max_neg <= 0:
        return None
    if max_pos > 0 and max_neg > 0:
        if np.random.rand() < 0.5:
            delta = np.random.uniform(0, max_pos)
        else:
            delta = -np.random.uniform(0, max_neg)
    elif max_pos > 0:
        delta = np.random.uniform(0, max_pos)
    else:
        delta = -np.random.uniform(0, max_neg)
    new_h = h.copy()
    new_h[i] = ai + delta
    new_h[j] = aj - delta
    # final clip not needed as we ensured within bounds
    return new_h

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_capped_simplex(v: np.ndarray, target: float,
                             lower: float = 0.0, upper: float = 1.0,
                             max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | sum(x)=target, lower<=x<=upper}.
    Uses binary search on the Lagrange multiplier.
    """
    v = np.clip(v, lower, upper).astype(np.float64)
    s = np.sum(v)
    if abs(s - target) <= 1e-12:
        return v

    lo, hi = -2.0, 2.0  # wide enough to bracket the solution
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, lower, upper)
        s = np.sum(proj)
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) / 2.0, lower, upper)

    # Small residual correction if needed
    diff = target - np.sum(proj)
    if abs(diff) > 1e-9:
        free = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if np.any(free):
            proj[free] += diff / np.sum(free)
        else:
            proj += diff / v.size
        proj = np.clip(proj, lower, upper)
    return proj


def perturb_pairwise(h: np.ndarray, max_step: float) -> np.ndarray:
    """Randomly transfer mass between two entries while preserving the sum."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    ai, aj = h[i], h[j]

    # feasible positive delta (increase i, decrease j)
    max_pos = min(max_step, 1.0 - ai, aj)
    # feasible negative delta (decrease i, increase j)
    max_neg = min(max_step, ai, 1.0 - aj)

    if max_pos <= 0 and max_neg <= 0:
        return None

    if max_pos > 0 and max_neg > 0:
        if np.random.rand() < 0.5:
            delta = np.random.uniform(0.0, max_pos)
        else:
            delta = -np.random.uniform(0.0, max_neg)
    elif max_pos > 0:
        delta = np.random.uniform(0.0, max_pos)
    else:
        delta = -np.random.uniform(0.0, max_neg)

    new_h = h.copy()
    new_h[i] = ai + delta
    new_h[j] = aj - delta
    return new_h


def gradient_step(h: np.ndarray, step: float, target: float) -> np.ndarray:
    """
    Take a step that increases the minimal self‑correlation of h.
    The step is projected onto the box and sum constraints.
    """
    n = h.size
    # autocorrelation (no scaling needed for argmin)
    autocorr = np.correlate(h, h, mode='full')
    idx = int(np.argmin(autocorr))
    shift = idx - (n - 1)

    if shift == 0:                     # degenerate case, pick a non‑zero shift
        shift = 1 if n > 1 else 0

    grad = np.zeros_like(h)

    if shift > 0:
        # forward shift
        grad[:n - shift] += h[shift:]
        grad[shift:] += h[:n - shift]
    else:
        s = -shift
        grad[s:] += h[:n - s]
        grad[:n - s] += h[s:]

    new_h = h + step * grad
    new_h = project_to_capped_simplex(new_h, target)
    return new_h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a candidate step function, its C5 bound, and the number of samples."""
    np.random.seed(seed)

    # Initialise h
    try:
        init = np.array(initial_h_values, dtype=np.float64)  # provided by the harness

    except NameError:

    best_h = h_current.copy()
    best_score = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    # schedule parameters
    step_start = 0.5
    step_end = 0.001
    prob_random = 0.25  # chance to do a simple random pairwise move

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        progress = elapsed / (budget_s * 0.95)
        # linear decay of step size
        step = step_end + (step_start - step_end) * (1.0 - progress)

        if np.random.rand() < prob_random:
            cand = perturb_pairwise(cur_h, step)
        else:
            cand = gradient_step(cur_h, step, target_sum)

        if cand is None:
            continue

        cand_score = compute_c5(cand)

        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
        else:
            # occasional uphill move (simulated annealing style)
            temp = max(step, 1e-6)
            accept_prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < accept_prob:
                cur_h, cur_score = cand, cand_score

    # Final projection & evaluation to guarantee feasibility
    best_h = project_to_capped_simplex(best_h, target_sum)
    final_score = compute_c5(best_h)

    return (best_h.tolist(), float(final_score), n)
