def run(seed=42, budget_s=1100, **kwargs):
    import time, math, random, numpy as np
    random.seed(seed)
    np.random.seed(seed)
    # Determine N, initial_h if exists
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = len(h0)
        # ensure sum constraint; adjust
        target_sum = N // 2
        # round to binary if needed
        # maybe keep as given if already binary.
        if np.all((h0 == 0) | (h0 == 1)):
            cur_h = h0.astype(np.int8)
        else:
            # convert to binary by selecting top target_sum values as ones.
            idx = np.argsort(-h0)[:target_sum]
            cur_h = np.zeros(N, dtype=np.int8)
            cur_h[idx] = 1
    else:
        N = 80
        target_sum = N // 2
        cur_h = np.zeros(N, dtype=np.int8)
        cur_h[:target_sum] = 1
        np.random.shuffle(cur_h)
    dx = 2.0 / N
    # compute objective function
    def compute_c5(vec):
        # vec is int/float in [0,1]
        c = np.correlate(vec, 1 - vec, mode='full')
        return np.max(c) * dx
    best_h = cur_h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    # Random sampling
    while time.time() - start < budget_s * 0.3:
        # generate random binary vector
        perm = np.random.permutation(N)
        cand = np.zeros(N, dtype=np.int8)
        cand[perm[:N//2]] = 1
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
    # Local search using simulated annealing starting from best_h
    cur_h = best_h.copy()
    cur_c5 = best_c5
    # keep indices sets
    ones = set(np.where(cur_h == 1)[0].tolist())
    zeros = set(np.where(cur_h == 0)[0].tolist())
    # annealing parameters
    T0 = 0.01
    T = T0
    iter_count = 0
    while time.time() - start < budget_s * 0.95:
        # propose swap
        i = random.sample(ones, 1)[0]
        j = random.sample(zeros, 1)[0]
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        dC = new_c5 - cur_c5
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c5 = new_c5
            # update sets
            ones.remove(i); ones.add(j)
            zeros.remove(j); zeros.add(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        iter_count += 1
        if iter_count % 1000 == 0:
            # cool down
            T *= 0.99
            # also maybe restart if stuck
    # return
    h_values = best_h.astype(float)  # maybe cast to float for evaluation.
    return (h_values, float(best_c5), N)
