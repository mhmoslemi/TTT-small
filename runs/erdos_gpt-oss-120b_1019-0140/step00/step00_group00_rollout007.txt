def project_onto_sum_and_bounds(x, target):
    # clip to [0,1]
    x_clipped = np.clip(x, 0, 1)
    # compute diff
    diff = target - np.sum(x_clipped)
    # If diff is near zero, return.
    # We need to distribute diff among variables that are not at bounds.
    # Get mask of free indices: (x_clipped > 0) & (x_clipped < 1)
    free = np.where((x_clipped > 1e-12) & (x_clipped < 1-1e-12))[0]
    # If free empty, cannot adjust; just return clipped.
    if free.size == 0:
        return x_clipped
    # Distribute diff equally among free
    # but may push some beyond bounds; we need to iterate.
    # We'll implement iterative water-filling.
    x = x_clipped.copy()
    while True:
        free = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if free.size == 0 or abs(diff) < 1e-12:
            break
        delta = diff / free.size
        x[free] += delta
        # Clip again and recompute diff
        x = np.clip(x, 0, 1)
        diff = target - np.sum(x)
    return x

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # Initialize h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        # fallback: use an even size for easier integral constraint
        n = 45
        h = np.full(n, 0.5, dtype=np.float64)   # satisfies sum = n/2
    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def compute_c5(vec):
        """Exact C5 value for a candidate vector."""
        corr = np.correlate(vec, 1.0 - vec, mode='full') * dx
        return np.max(corr)

    def project_onto_constraints(vec):
        """
        Project `vec` onto the box [0,1] and enforce the linear sum constraint.
        Simple water-filling iteration; converges quickly for small n.
        """
        v = np.clip(vec, 0.0, 1.0)
        diff = target_sum - v.sum()
        # iterative redistribution among non‑boundary entries
        for _ in range(20):
            if abs(diff) < 1e-12:
                break
            free_mask = (v > 1e-12) & (v < 1.0 - 1e-12)
            free_idx = np.where(free_mask)[0]
            if free_idx.size == 0:
                # all variables are at a bound; distribute remaining diff evenly
                v += diff / n
                v = np.clip(v, 0.0, 1.0)
                diff = target_sum - v.sum()
                continue
            delta = diff / free_idx.size
            v[free_idx] += delta
            v = np.clip(v, 0.0, 1.0)
            diff = target_sum - v.sum()
        return v

    def gradient_for_shift(vec, shift):
        """Subgradient of the worst‑shift correlation (max over k)."""
        n_local = vec.size
        if shift == 0:
            return 1.0 - 2.0 * vec
        s = abs(shift)
        grad = np.ones(n_local)
        left = np.arange(0, n_local - s)
        right = left + s
        grad[left] -= vec[right]
        grad[right] -= vec[left]
        return grad

    # Ensure starting point meets constraints exactly
    h = project_onto_constraints(h)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    total_budget = float(budget_s)

    # ------------------------------------------------------------------
    # Phase 1: subgradient descent (approx. 70 % of the budget)
    # ------------------------------------------------------------------
    subgrad_fraction = 0.70
    subgrad_deadline = start_time + total_budget * subgrad_fraction
    max_subgrad_iters = 20000          # generous upper bound
    lr_start = 0.5
    lr_end = 0.001

    it = 0
    while time.time() < subgrad_deadline and it < max_subgrad_iters:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)   # shift in range [-(n-1), n-1]

        # gradient of the active shift
        grad = gradient_for_shift(h, shift)

        # linearly decaying step size
        lr = lr_start + (lr_end - lr_start) * (it / max_subgrad_iters)

        # take a step and re‑project
        h_new = project_onto_constraints(h - lr * grad)

        # evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Always move to the new point (deterministic descent)
        h = h_new
        it += 1

    # ------------------------------------------------------------------
    # Phase 2: stochastic pairwise swaps (remaining budget)
    # ------------------------------------------------------------------
    h = best_h.copy()
    swap_deadline = start_time + total_budget - 0.1   # small safety margin

    while time.time() < swap_deadline:
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        # maximum feasible epsilon respecting bounds
        eps_max = min(1.0 - h[i], h[j])
        if eps_max <= 1e-14:
            continue
        eps = rng.rand() * eps_max   # uniform in [0, eps_max)
        new_h = h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        c5_new = compute_c5(new_h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = new_h.copy()
            h = new_h
        # otherwise keep the current h (no uphill moves)

    # ------------------------------------------------------------------
    # Final projection and output
    # ------------------------------------------------------------------
    best_h = project_onto_constraints(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), int(n)
