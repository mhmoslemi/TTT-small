def hill_climb_sa(start_h, rng, max_swaps, init_temp=0.05):
    h = start_h.copy()
    current_c5 = compute_c5_vals(h)
    n = len(h)
    temp = init_temp
    # schedule: linear decay to near zero
    for step in range(1, max_swaps+1):
        # compute temperature: linear decay
        t = init_temp * (1 - step / max_swaps)
        # random swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1

    n = h.shape[0]

        # temperature schedule
        temp = init_temp * (1.0 - step / max_swaps)
        # choose swap indices i, j with different values
        i = rng.integers(n)
        j = rng.integers(n-1)

            j += 1
        if h[i] == h[j]:

        h[i], h[j] = h[j], h[i]

            else:
                # revert
                h[i], h[j] = h[j], h[i]

import numpy as np
import time
import math

def _c5_of(h: np.ndarray) -> float:
    """Return the C₅ value for a binary (or [0,1]) step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # max_k ∫ h(x)(1−h(x+k))dx  →  max of discrete correlation * dx
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a length‑n binary vector with exactly n/2 ones (n must be even)."""
    arr = np.zeros(n, dtype=np.float64)
    arr[: n // 2] = 1.0
    rng.shuffle(arr)
    return arr


def _anneal(start_h: np.ndarray,
            rng: np.random.Generator,
            start_time: float,
            budget_s: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Swaps two positions of different values while keeping the total sum fixed.
    """
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_c5 = _c5_of(cur)
    best = cur.copy()
    best_c5 = cur_c5

    # main loop – stop a little before the official budget to avoid a kill
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.95:
            break

        # linear temperature schedule, start ~0.2, end →0
        frac = elapsed / (budget_s * 0.95)
        temp = 0.20 * (1.0 - frac)

        # propose a swap of a 0 and a 1
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue

        # perform swap
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _c5_of(cur)
        delta = new_c5 - cur_c5

        # Metropolis acceptance
        if delta < 0.0 or (temp > 0.0 and rng.random() < math.exp(-delta / temp)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

    return best, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Find a step function h that minimizes the Erdős C₅ constant.

    Returns
    -------
    h_values : np.ndarray
        The discretized step function (values in [0,1]).
    c5_bound : float
        The achieved upper bound for C₅.
    n_points : int
        Number of samples used (length of ``h_values``).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the number of points and an initial candidate.
    # ------------------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)  # provided by harness
        n = init_arr.shape[0]
    except NameError:
        n = 80
        init_arr = None

    # Ensure even length – the constraint sum(h)=n/2 is integral for even n.
    if n % 2 != 0:
        # Resort to the always‑feasible constant 0.5 function.
        h_best = np.full(n, 0.5, dtype=np.float64)
        return h_best, _c5_of(h_best), n

    target_ones = n // 2

    if init_arr is not None:
        # Convert the provided construction to a binary vector with exact weight.
        h0 = np.where(init_arr >= 0.5, 1.0, 0.0)
        cur_ones = int(h0.sum())
        diff = target_ones - cur_ones
        if diff > 0:
            zeros = np.where(h0 == 0)[0]
            flip = rng.choice(zeros, size=diff, replace=False)
            h0[flip] = 1.0
        elif diff < 0:
            ones = np.where(h0 == 1)[0]
            flip = rng.choice(ones, size=-diff, replace=False)
            h0[flip] = 0.0
        start_h = h0.astype(np.float64)
    else:
        start_h = _random_binary(n, rng)

    # ------------------------------------------------------------------
    # 2. Main annealing run – use the whole time budget.
    # ------------------------------------------------------------------
    start_time = time.time()
    best_h, best_c5 = _anneal(start_h, rng, start_time, budget_s)

    # ------------------------------------------------------------------
    # 3. Optional short restarts if time permits (helpful if the first
    #    run gets stuck in a poor local optimum).
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now - start_time >= budget_s * 0.95:
            break
        # a fresh random start followed by a shorter anneal
        h_rand = _random_binary(n, rng)
        h_candidate, c5_candidate = _anneal(h_rand, rng, start_time, budget_s)
        if c5_candidate < best_c5:
            best_h, best_c5 = h_candidate, c5_candidate
            if best_c5 <= 0.3808:
                break

    return best_h, float(best_c5), n
