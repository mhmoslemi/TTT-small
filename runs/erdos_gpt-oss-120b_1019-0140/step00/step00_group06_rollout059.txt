import numpy as np
import time

def _project_to_sum_and_bounds(v, target):
    # binary search lambda such that sum(clip(v+lambda,0,1)) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # correct residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_value(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _optimize_vector(h_start, max_iter, start_time, budget):
    n = h_start.shape[0]
    target_sum = n / 2.0
    h = _project_to_sum_and_bounds(h_start.copy(), target_sum)
    best_h = h.copy()
    best_val = _c5_value(best_h)
    step = 0.5
    # compute current value once
    cur_val = best_val
    for it in range(max_iter):
        if time.time() - start_time > budget * 0.95:
            break
        # correlation evaluation
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        # gradient of correlation w.r.t. h
        g = np.ones_like(h)
        if lag > 0:
            g[:n - lag] -= h[lag:]
            g[lag:]   -= h[:n - lag]
        elif lag < 0:
            L = -lag
            g[L:]     -= h[:n - L]
            g[:n - L] -= h[L:]
        else:
            g -= 2.0 * h
        # gradient descent step
        h_new = h - step * g
        h_new = _project_to_sum_and_bounds(h_new, target_sum)
        new_val = _c5_value(h_new)
        # update best
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        # accept improvement
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
        else:
            # occasional random move to escape plateaus
            if np.random.rand() < 0.02:
                delta = (np.random.rand() - 0.5) * 0.02
                i, j = np.random.choice(n, 2, replace=False)
                # adjust i by delta, j by -delta keeping bounds
                hi = min(delta, 1.0 - h[i])
                lo = max(delta, -h[i])
                delta_adj = hi if delta > 0 else lo
                # adjust j opposite
                delta_j = -delta_adj
                # ensure bounds for j
                delta_j = min(delta_j, 1.0 - h[j]) if delta_j > 0 else max(delta_j, -h[j])
                # recompute delta_i accordingly

import numpy as np
import time

def _project_to_sum_and_bounds(v, target):
    """Project vector v onto the set {x: 0<=x_i<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0          # guarantees all entries become 0 after clipping
    hi = 1.0 - np.min(v) + 1.0     # guarantees all entries become 1 after clipping
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny residual sum error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _c5_value(h):
    """Return the C5 bound for step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _optimize_vector(h_start, max_iter, start_time, budget):
    """Projected gradient descent on the worst lag correlation."""
    n = h_start.shape[0]
    target_sum = n / 2.0
    h = _project_to_sum_and_bounds(h_start.copy(), target_sum)
    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_val = best_val
    step = 0.5
    for it in range(max_iter):
        if time.time() - start_time > budget * 0.95:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)

        # gradient of the correlation w.r.t. h
        g = np.ones_like(h)
        if lag > 0:
            g[:n - lag] -= h[lag:]
            g[lag:]     -= h[:n - lag]
        elif lag < 0:
            L = -lag
            g[L:]       -= h[:n - L]
            g[:n - L]   -= h[L:]
        else:
            g -= 2.0 * h

        h_new = h - step * g
        h_new = _project_to_sum_and_bounds(h_new, target_sum)
        new_val = _c5_value(h_new)

        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        if new_val < cur_val:
            h = h_new
            cur_val = new_val
        else:
            if np.random.rand() < 0.02:
                delta = (np.random.rand() - 0.5) * 0.02
                i, j = np.random.choice(n, 2, replace=False)
                # adjust i by delta (clipped), j by opposite amount
                adj_i = np.clip(delta, -h[i], 1.0 - h[i])
                adj_j = -adj_i
                # ensure j stays feasible
                adj_j = np.clip(adj_j, -h[j], 1.0 - h[j])
                adj_i = -adj_j
                h_tmp = h.copy()
                h_tmp[i] += adj_i
                h_tmp[j] += adj_j
                h = _project_to_sum_and_bounds(h_tmp, target_sum)
                cur_val = _c5_value(h)

        if (it + 1) % 200 == 0:
            step = max(step * 0.7, 0.01)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # obtain initial vector (may be None)
    init = globals().get('initial_h_values')
    if init is None:
        n0 = 200
        h0 = np.random.rand(n0)
    else:
        h0 = np.array(init, dtype=np.float64)
        n0 = h0.shape[0]

    h0 = _project_to_sum_and_bounds(h0, n0 / 2.0)

    # primary optimization at the original resolution
    best_h, best_val = _optimize_vector(h0, max_iter=5000,
                                        start_time=start, budget=budget_s)

    # try higher resolutions if time permits
    for factor in (2, 4):
        if time.time() - start > budget_s * 0.9:

        n_high = n0 * factor

        x_new = np.linspace(0.0, 2.0, n_high)
        h_up = np.interp(x_new, x_old, best_h)

        h_opt, val_opt = _optimize_vector(h_up, max_iter=5000,
