def subgradient_descent(h, time_deadline):
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_score = compute_c5(best_h)
    lr = 0.5
    while time.time() < time_deadline:
        # compute correlation and find worst shift
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        # find max c
        max_idx = np.argmax(c)
        # shift s
        s = max_idx - (n - 1)
        # compute gradient for shift s
        grad = np.zeros_like(h)
        i = np.arange(n)
        j = i + s
        mask = (j >= 0) & (j < n)
        i_valid = i[mask]
        j_valid = j[mask]
        grad[i_valid] += (1 - h[j_valid])
        grad[j_valid] -= h[i_valid]
        grad *= (2.0 / n)  # dx factor
        # step
        h_new = h - lr * grad
        # project to feasible set
        h_new = project_box_sum(h_new, target_sum)
        # compute score
        score = compute_c5(h_new)
        # acceptance and learning rate update
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            # potentially increase lr
            lr = min(lr * 1.05, 3.0)
        else:
            # no improvement: possibly decrease lr
            lr = lr * 0.9
        h = h_new
        # early stop if lr becomes too small
        if lr < 1e-6:
            break
    return best_h, best_score

def smoothed_obj_grad(h, alpha=20.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # shift for numerical stability
    maxc = np.max(c)
    exp_term = np.exp(alpha * (c - maxc))
    Z = exp_term.sum()
    obj = maxc + np.log(Z) / alpha
    weights = exp_term / Z
    # compute gradient
    grad = np.zeros_like(h)
    s_range = np.arange(-(n-1), n)
    for idx, s in enumerate(s_range):
        w = weights[idx]
        i = np.arange(n)
        j = i + s
        mask = (j >= 0) & (j < n)
        i_valid = i[mask]
        j_valid = j[mask]
        grad[i_valid] += w * (1 - h[j_valid])
        grad[j_valid] -= w * h[i_valid]
    grad *= dx
    return obj, grad

import time
import numpy as np

def compute_c5(h):
    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(y, target):
    """
    Project y onto the box [0,1]^n while enforcing sum(y)=target.
    Uses a simple bisection on the shift lambda: x = clip(y+lambda,0,1).
    """
    lo, hi = -2.0, 2.0  # guaranteed bounds for lambda
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_h(n, target):
    """Generate a random feasible h (values in [0,1] with correct sum)."""
    y = np.random.rand(n)
    return project_box_sum(y, target)

def subgradient_descent(h0, deadline):
    """
    Subgradient descent on the max overlap C5.
    Stops when time runs out.
    """
    n = h0.shape[0]
    target = n * 0.5
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(best_h)
    lr = 0.5
    while time.time() < deadline:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)          # integer shift
        # gradient for this shift
        i = np.arange(n)
        j = i + s
        mask = (j >= 0) & (j < n)
        i_valid = i[mask]
        j_valid = j[mask]
        grad = np.zeros_like(h)
        grad[i_valid] += (1.0 - h[j_valid])
        grad[j_valid] -= h[i_valid]
        grad *= dx
        # step and projection
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        score = compute_c5(h_new)
        # simple adaptive step size
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            lr = min(lr * 1.05, 3.0)
        else:
            lr = max(lr * 0.90, 1e-6)
        h = h_new
        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.01:
            eps = np.random.rand() * 0.02
            i_rand, j_rand = np.random.choice(n, 2, replace=False)
            delta = min(eps, h[i_rand])
            h[i_rand] -= delta
            h[j_rand] = min(h[j_rand] + delta, 1.0)
            h = project_box_sum(h, target)
    return best_h, best_score

def smoothed_obj_grad(h, alpha=20.0):
    """Smooth approximation of max C5 and its gradient via log‑sum‑exp."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    maxc = np.max(corr)
    exp_term = np.exp(alpha * (corr - maxc))
    Z = exp_term.sum()
    obj = maxc + np.log(Z) / alpha
    weights = exp_term / Z
    grad = np.zeros_like(h)
    shifts = np.arange(-(n - 1), n)            # corresponds to indices of corr
    for idx, s in enumerate(shifts):
        w = weights[idx]
        i = np.arange(n)
        j = i + s
        mask = (j >= 0) & (j < n)
        i_valid = i[mask]
        j_valid = j[mask]
        grad[i_valid] += w * (1.0 - h[j_valid])
        grad[j_valid] -= w * h[i_valid]
    grad *= dx
    return obj, grad

def smoothed_descent(h, deadline, alpha=20.0):
    """A short refinement using the smooth objective."""
    n = h.shape[0]
    target = n * 0.5
    lr = 0.1
    while time.time() < deadline:
        _, grad = smoothed_obj_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        h = h_new
        lr *= 0.995  # gentle decay
    return h
