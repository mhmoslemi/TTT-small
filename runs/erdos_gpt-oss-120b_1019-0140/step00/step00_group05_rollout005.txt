import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Return the overlap constant C5 for a discrete h."""
    n = len(h)
    dx = 2.0 / n
    # full convolution yields length 2n-1, zero padding outside the interval
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def make_binary_vector(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    return h


def binary_from_initial(init_h, n, rng):
    """Convert any initial vector to a binary one respecting the cardinality."""
    # pick the n/2 largest entries as ones
    target_ones = n // 2
    idx = np.argsort(-init_h)[:target_ones]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def propose_swap(h, rng):
    """
    Propose a swap (i, j) that tries to reduce the current maximal overlap.
    Returns the indices (i, j) and the resulting vector after the tentative swap.
    """
    n = len(h)
    # correlation and shift where the maximum occurs
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)  # shift k such that h[i] aligns with (1-h)[i+k]

    ones = np.flatnonzero(h)          # positions where h == 1
    zeros = np.flatnonzero(h == 0)    # positions where h == 0

    # Positions that directly contribute to the worst shift
    contrib = []
    for i in ones:
        j = i + shift
        if 0 <= j < n and h[j] == 0:
            contrib.append(i)

    if contrib and zeros.size > 0:
        i_swap = rng.choice(contrib)
        # choose a zero that is *not* the counterpart of i_swap (if possible)
        # simply pick any zero different from i_swap
        cand = zeros[zeros != i_swap]
        if cand.size == 0:
            # fallback to any zero
            j_swap = rng.choice(zeros)
        else:
            j_swap = rng.choice(cand)
    else:
        # generic random swap
        i_swap = rng.choice(ones)
        j_swap = rng.choice(zeros)

    # Perform the tentative swap in-place
    h[i_swap] = 0.0
    h[j_swap] = 1.0
    return i_swap, j_swap


def revert_swap(h, i, j):
    """Undo a swap performed by propose_swap."""
    h[i] = 1.0
    h[j] = 0.0


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine problem size and initial vector
    # ----------------------------------------------------------------
    init_h = globals().get("initial_h_values", None)
    if init_h is not None:
        n = len(init_h)
    else:
        n = 70                     # default size (even)

    # enforce even length for binary representation
    if n % 2 == 1:
        n += 1

    # Build the starting vector
    if init_h is not None and len(init_h) == n:
        h_current = binary_from_initial(np.asarray(init_h, dtype=np.float64), n, rng)
    else:
        h_current = make_binary_vector(n, rng)

    current_score = compute_c5(h_current)
    best_h = h_current.copy()
    best_score = current_score

    # keep a reference binary initial vector for occasional restarts
    init_binary = None
    if init_h is not None and len(init_h) == n:
        init_binary = binary_from_initial(np.asarray(init_h, dtype=np.float64), n, rng)

    # ----------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ----------------------------------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.95            # leave a small safety margin
    stagnation = 0
    stagnation_limit = 3000               # restart after this many non‑improving swaps
    initial_temp = 0.05
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break

        # Temperature linearly decays to near zero
        T = max(1e-8, initial_temp * (1.0 - elapsed / max_time))

        # Propose a move
        i_swap, j_swap = propose_swap(h_current, rng)

        new_score = compute_c5(h_current)
        delta = new_score - current_score

        # Accept if improvement or with Metropolis probability
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            stagnation = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h_current.copy()
        else:
            # reject: undo the swap
            revert_swap(h_current, i_swap, j_swap)
            stagnation += 1

        # Periodic restart if we are stuck
        if stagnation >= stagnation_limit:
            if init_binary is not None and rng.random() < 0.15:
                h_current = init_binary.copy()
            else:
                h_current = make_binary_vector(n, rng)
            current_score = compute_c5(h_current)
            stagnation = 0

        iteration += 1

    # ----------------------------------------------------------------
    # Final evaluation (use the provided helper if present)
    # ----------------------------------------------------------------
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = float(compute_c5(best_h))

    return (best_h, final_c5, n)
