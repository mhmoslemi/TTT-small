import numpy as np, time, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(c)

def project_feasible(arr, target_sum):
    # clip and adjust sum iteratively
    arr = np.clip(arr, 0.0, 1.0)
    # try simple iterative redistribution
    for _ in range(5):
        s = arr.sum()
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        # indices where arr is not at bound: 0 < arr < 1
        mask = (arr > 0.0) & (arr < 1.0)
        if mask.sum() == 0:
            # all at bounds, just distribute to all (ignoring bounds)
            arr += diff / len(arr)
            arr = np.clip(arr, 0.0, 1.0)
        else:
            delta = diff / mask.sum()
            arr[mask] += delta
            arr = np.clip(arr, 0.0, 1.0)
    # final adjustment: if still off due to rounding, adjust smallest slack
    s = arr.sum()
    diff = target_sum - s
    if abs(diff) > 1e-8:
        # distribute remaining diff proportionally to slack
        slack_up = 1.0 - arr
        slack_down = arr
        total_slack = slack_up.sum() + slack_down.sum()
        if total_slack > 0:
            arr += diff * slack_up / total_slack
            arr = np.clip(arr, 0.0, 1.0)
    # final rounding
    return arr

def random_binary(n, target_sum):
    arr = np.zeros(n)
    idx = np.random.choice(n, target_sum, replace=False)
    arr[idx] = 1.0
    return arr

def hill_climb_binary(n, target_sum, time_limit, best_initial=None, target_c5=0.3808):
    start = time.time()
    best_h = np.full(n, 0.5)
    best_score = compute_c5(best_h)
    if best_initial is not None:
        arr = project_feasible(best_initial, target_sum)
        score = compute_c5(arr)
        if score < best_score:
            best_h, best_score = arr, score
    # allocate overhead
    while time.time() - start < time_limit:
        # generate random binary start
        h = random_binary(n, target_sum)
        score = compute_c5(h)
        # early track
        if score < best_score:
            best_h, best_score = h, score
            if best_score <= target_c5:
                return best_h, best_score
        # local hill climbing
        # define number of local steps based on remaining time
        # perhaps 1000 steps per start
        local_steps = 1500
        for _ in range(local_steps):
            # choose a 0 and 1
            zeros = np.flatnonzero(h == 0.0)
            ones = np.flatnonzero(h == 1.0)
            if zeros.size == 0 or ones.size == 0:
                break
            i = zeros[np.random.randint(zeros.size)]
            j = ones[np.random.randint(ones.size)]
            # swap
            new_h = h.copy()
            new_h[i] = 1.0
            new_h[j] = 0.0
            new_score = compute_c5(new_h)
            # accept if improvement
            if new_score < score:
                h, score = new_h, new_score
                if score < best_score:
                    best_h, best_score = h, score
                    if best_score <= target_c5:
                        return best_h, best_score
        # loop continues with new random start
    return best_h, best_score

def hill_climb_continuous(h, target_sum, time_limit, target_c5=0.3808):
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = len(h)
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # feasible delta range
        delta_low = max(-best_h[i], best_h[j] - 1.0)
        delta_high = min(1.0 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score
            if best_score <= target_c5:
                break
    # final projection just in case
    best_h = project_feasible(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score

import time
import random
import numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(c)

def _project_feasible(v, target):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(5):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        cnt = mask.sum()
        if cnt == 0:
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
        else:
            delta = diff / cnt
            v[mask] += delta
            v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-8:
        slack_up = 1.0 - v
        slack_down = v
        total = slack_up.sum() + slack_down.sum()
        if total > 0:
            v += diff * slack_up / total
            v = np.clip(v, 0.0, 1.0)
    return v

def _random_binary(n, target):
    a = np.zeros(n, dtype=float)
    a[np.random.choice(n, target, replace=False)] = 1.0
    return a

def _binary_hill_climb(n, target, timelimit, init=None, goal=0.3808):
    start = time.time()
    best_h = np.full(n, 0.5, dtype=float)
    best_score = _compute_c5(best_h)
    if init is not None:
        cand = _project_feasible(init, target)
        sc = _compute_c5(cand)
        if sc < best_score:
            best_h, best_score = cand, sc
    while time.time() - start < timelimit:
        h = _random_binary(n, target)
        score = _compute_c5(h)
        if score < best_score:
            best_h, best_score = h, score
            if best_score <= goal:
                return best_h, best_score
        # local improvement
        for _ in range(1500):
            zeros = np.flatnonzero(h == 0.0)
            ones = np.flatnonzero(h == 1.0)
            if zeros.size == 0 or ones.size == 0:
                break
            i = zeros[np.random.randint(zeros.size)]
            j = ones[np.random.randint(ones.size)]
            nh = h.copy()
            nh[i] = 1.0
            nh[j] = 0.0
            ns = _compute_c5(nh)
            if ns < score:
                h, score = nh, ns
                if ns < best_score:
                    best_h, best_score = nh, ns
                    if best_score <= goal:
                        return best_h, best_score
    return best_h, best_score

def _continuous_refine(h, target, timelimit, goal=0.3808):
    start = time.time()
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    n = h.size
    while time.time() - start < timelimit:
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        lo = max(-best_h[i], best_h[j] - 1.0)
        hi = min(1.0 - best_h[i], best_h[j])
        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        nh = best_h.copy()
        nh[i] += delta
        nh[j] -= delta
        ns = _compute_c5(nh)
        if ns < best_score:
            best_h, best_score = nh, ns
            if best_score <= goal:
                break
    best_h = _project_feasible(best_h, target)
    best_score = _compute_c5(best_h)
    return best_h, best_score
