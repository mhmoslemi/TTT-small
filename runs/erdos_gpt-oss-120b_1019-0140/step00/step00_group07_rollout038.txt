def solve_milp(N, time_limit=30):
    import cvxpy as cp
    h = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == N//2, t >= 0]
    for k in range(1, N):
        L = N - k
        y = cp.Variable(L, boolean=True)
        constraints += [
            y <= h[:L],
            y <= h[k:],
            y >= h[:L] + h[k:] - 1,
            cp.sum(y) >= t
        ]
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    # attempt solve with CBC first
    try:
        prob.solve(solver=cp.CBC, warm_start=True, verbose=False, 
                   max_seconds=time_limit)
    except Exception as e:
        # fallback ECOS_BB
        try:
            prob.solve(solver=cp.ECOS_BB, warm_start=True, verbose=False, 
                       max_iters=5000)  # maybe not set time.
        except Exception as e2:
            # no solve; return None
            return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # maybe unsolved: return None
        pass
    # obtain solution
    h_val = np.round(h.value).astype(float)
    t_val = int(round(t.value))
    return h_val, t_val

def solve_milp(N, time_limit=60):
    import cvxpy as cp
    h = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == N // 2, t >= 0]
    for k in range(1, N):
        L = N - k
        y = cp.Variable(L, boolean=True)
        constraints += [
            y <= h[:L],
            y <= h[k:],
            y >= h[:L] + h[k:] - 1,
            cp.sum(y) >= t
        ]
    prob = cp.Problem(cp.Maximize(t), constraints)
    # Try CBC, GLPK_MI, ECOS_BB
    # The solve method may have solver-specific parameter names for time limit
    # Use try-catch
    try:
        prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        solved = prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
    except Exception:
        solved = False
    if not solved:
        try:
            prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
            solved = prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
        except Exception:
            solved = False
    if not solved:
        try:
            prob.solve(solver=cp.ECOS_BB, warm_start=True, verbose=False)
            solved = prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
        except Exception:
            solved = False
    if not solved:
        return None, None
    # Extract values
    h_val = np.round(h.value).astype(float)
    t_val = int(round(t.value))
    # enforce exact sum, if rounding leads to discrepancy, adjust
    if abs(np.sum(h_val) - N/2) > 1e-6:
        # enforce by fixing sum using top entries
        idx = np.argsort(-h_val)  # descending
        h_val[:] = 0
        h_val[idx[:N//2]] = 1
    return h_val, t_val

def solve_milp(N, time_limit=None):
    import cvxpy as cp
    import numpy as np

    h = cp.Variable(N, boolean=True)
    M = cp.Variable(integer=True)

    constraints = [cp.sum(h) == N // 2]

    for k in range(1, N):
        L = N - k
        y = cp.Variable(L, boolean=True)
        # product constraints
        constraints += [
            y <= h[:L],
            y <= h[k:],
            y >= h[:L] + h[k:] - 1,
            cp.sum(h[:L]) - cp.sum(y) <= M
        ]
    # Additionally, M >= 0 maybe:
    constraints.append(M >= 0)

    prob = cp.Problem(cp.Minimize(M), constraints)
    # Solve
    try:
        prob.solve(solver=cp.CBC, warm_start=True, verbose=False, max_seconds=time_limit if time_limit else None)
    except Exception:
        # try other solver
        try:
            prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS_BB, warm_start=True, verbose=False)
            except Exception:
                return None, None

    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None

    h_val = np.round(h.value).astype(float)
    M_val = int(round(M.value))
    return h_val, M_val

import numpy as np
import math

# optional cvxpy import; may be unavailable in some environments
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def compute_c5_linear(h):
    """Maximum overlap integral for a binary (or [0,1]) vector h using linear correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    return max_val * (2.0 / h.shape[0])


def solve_milp(N, time_limit=None):
    """Mixed‑integer linear formulation for the linear (zero‑padded) overlap problem.
    Returns (h_vector, max_overlap) if solved, else (None, None)."""
    if cp is None:  # cvxpy not available
        return None, None

    h = cp.Variable(N, boolean=True)
    M = cp.Variable(integer=True)

    constraints = [cp.sum(h) == N // 2, M >= 0]

    # for each shift k = 1 .. N-1 (k=0 gives zero overlap automatically)
    for k in range(1, N):
        L = N - k
        y = cp.Variable(L, boolean=True)               # y[i] = h[i] * h[i+k]
        # McCormick linearisation (exact for binary)
        constraints += [
            y <= h[:L],
            y <= h[k:],
            y >= h[:L] + h[k:] - 1,
            cp.sum(h[:L]) - cp.sum(y) <= M               # overlap count for this shift
        ]

    prob = cp.Problem(cp.Minimize(M), constraints)

    # try a few solvers; CBC is preferred, fall back to GLPK_MI or ECOS_BB
    solve_kwargs = {"warm_start": True, "verbose": False}
    if time_limit is not None:
        solve_kwargs["max_seconds"] = time_limit

    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, **solve_kwargs)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                h_val = np.round(h.value).astype(float)
                M_val = int(round(M.value))
                # enforce exact sum in case of rounding drift

                    h_val[:] = 0
                    h_val[idx[: N // 2]] = 1.0
                return h_val, M_val
        except Exception:
            continue
    return None, None


def local_search(N, init_h, budget_s, rng, target_bound=0.3808):
    """Simple hill‑climbing with random restarts within a time budget."""
    start = time.time()
    best_h = init_h.copy()
    best_c5 = compute_c5_linear(best_h)

    # start from the supplied vector
    cur_h = best_h.copy()
    cur_c5 = best_c5
    no_improve = 0
    max_no_improve = 4000

    while time.time() - start < budget_s:
        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5_linear(cur_h)

        if new_c5 + 1e-12 < cur_c5:          # improvement
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                if best_c5 <= target_bound:
                    break
        else:
            # revert the swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1

        if no_improve >= max_no_improve:
            # random restart
            cur_h[:] = 0.0
            cur_h[rng.choice(N, size=N // 2, replace=False)] = 1.0
            cur_c5 = compute_c5_linear(cur_h)
            no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        N = len(init_vec)
    else:
        N = 80  # fallback size

    # Ensure even size (required for exact half‑sum)
    if N % 2:
        N += 1

    # ----- try exact MILP first -----
    time_elapsed = time.time() - start_time
    remaining = budget_s - time_elapsed
    milp_h, milp_M = None, None
    if remaining > 20 and cp is not None:
        # give the MILP about half the remaining time
        milp_time = max(5, int(remaining * 0.5))
        milp_h, milp_M = solve_milp(N, time_limit=milp_time)

    if milp_h is not None:
        # successful MILP solution
        c5_bound = milp_M * (2.0 / N)
        return (milp_h.tolist(), float(c5_bound), N)

    # ----- otherwise fall back to heuristic search -----
    # build an initial binary vector
    if init_vec is not None:
        init_arr = np.clip(np.asarray(init_vec, dtype=float), 0.0, 1.0)
        # keep the N//2 largest entries as ones
        idx = np.argsort(-init_arr)
        start_h = np.zeros(N, dtype=float)
        start_h[idx[: N // 2]] = 1.0
    else:
        start_h = np.zeros(N, dtype=float)
        start_h[rng.choice(N, size=N // 2, replace=False)] = 1.0

    # time budget for local search
    time_elapsed = time.time() - start_time
    remaining = budget_s - time_elapsed
    if remaining <= 1:  # nothing left
        final_h = start_h
        final_c5 = compute_c5_linear(final_h)
        return (final_h.tolist(), float(final_c5), N)

    # split remaining time between a few independent runs
    runs = 3
    per_run_budget = max(1.0, remaining / runs)
    best_h, best_c5 = start_h, compute_c5_linear(start_h)

    for _ in range(runs):
        # small random perturbation of the start vector for diversification
        pert_h = best_h.copy()
        # optionally shuffle a few bits
        for __ in range(10):
            ones = np.flatnonzero(pert_h == 1)
            zeros = np.flatnonzero(pert_h == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            pert_h[i] = 0.0
            pert_h[j] = 1.0

        h_sol, c5_sol = local_search(N, pert_h, per_run_budget, rng)
        if c5_sol < best_c5:
            best_h, best_c5 = h_sol, c5_sol
            if best_c5 <= 0.3808:
                break

    # final verification / rounding
    best_h = np.round(best_h).astype(float)
    if int(np.sum(best_h)) != N // 2:
        # enforce exact half‑sum
        idx = np.argsort(-best_h)
        best_h[:] = 0
        best_h[idx[: N // 2]] = 1.0

    final_c5 = compute_c5_linear(best_h)

    # If the harness provides a helper for official evaluation, use it for the reported bound
    eval_fn = globals().get("evaluate_erdos_solution")
    if callable(eval_fn):
        try:
            final_c5 = eval_fn(best_h)
        except Exception:
            pass

    return (best_h.tolist(), float(final_c5), N)
