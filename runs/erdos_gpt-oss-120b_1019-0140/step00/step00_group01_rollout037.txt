def compute_C5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def make_initial_h(n):

    L = 0.6                # length of central plateau (in [0,2] units)
    dx = 2.0 / n

    mask = np.abs(x - 1.0) <= L / 2
    R = np.count_nonzero(mask)
    target = n / 2.0
    if R >= n:
        return np.full(n, 0.5)
    v = (target - R) / (n - R)
    v = max(0.0, min(1.0, v))
    h = np.where(mask, 1.0, v)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h
