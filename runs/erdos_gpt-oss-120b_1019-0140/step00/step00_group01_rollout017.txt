def project_to_sum_and_bounds(u, lower=-0.5, upper=0.5):
    # want sum(u)=0, with bounds.
    # Use binary search for lambda, solving sum(clip(u - lam, lower, upper)) = 0.
    # lam in real.
    # Because clipping monotonic decreasing in lam.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2
        proj = np.clip(u - lam, lower, upper)
        s = proj.sum()
        if s > 0:
            lo = lam
        else:
            hi = lam
    return np.clip(u - (lo+hi)/2, lower, upper)

import numpy as np, time, random

def project_to_sum_and_bounds(u, lower=-0.5, upper=0.5):
    # binary search for lambda
    lam_lo = -2.0
    lam_hi = 2.0
    for _ in range(60):
        lam_mid = (lam_lo + lam_hi) / 2.0
        proj = np.clip(u - lam_mid, lower, upper)
        if proj.sum() > 0:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    proj = np.clip(u - (lam_lo + lam_hi) / 2.0, lower, upper)
    # adjust for tiny residual
    resid = -proj.sum()
    if abs(resid) > 1e-12:
        # find a free index not at bound
        free = np.where((proj > lower + 1e-12) & (proj < upper - 1e-12))[0]
        if free.size > 0:
            proj[free[0]] += resid
        else:
            proj += resid / len(proj)
            proj = np.clip(proj, lower, upper)
    return proj

def compute_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = int(np.argmax(corr))
    mid = n - 1
    s = max_idx - mid
    return s, corr[max_idx]

def gradient_worst_shift(h, shift):
    n = h.shape[0]
    grad = np.ones_like(h)
    s = shift
    if s >= 0:
        grad[:n - s] -= h[s:]
        grad[s:] -= h[:n - s]
    else:
        t = -s
        grad[:n - t] -= h[t:]
        grad[t:] -= h[:n - t]
    return grad

import numpy as np

def _project_to_sum_and_bounds(u, lower=-0.5, upper=0.5):
    """Project vector u onto the hyperplane sum(u)=0 with box bounds."""
    # binary search for lambda such that sum(clip(u - lam)) = 0
    lam_lo, lam_hi = -2.0, 2.0
    for _ in range(60):
        lam_mid = (lam_lo + lam_hi) * 0.5
        proj = np.clip(u - lam_mid, lower, upper)
        if proj.sum() > 0:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    proj = np.clip(u - (lam_lo + lam_hi) * 0.5, lower, upper)
    # tiny residual correction to hit the equality exactly
    resid = -proj.sum()
    if abs(resid) > 1e-12:
        free = np.where((proj > lower + 1e-12) & (proj < upper - 1e-12))[0]
        if free.size:
            proj[free[0]] += resid
        else:
            proj += resid / proj.size
            proj = np.clip(proj, lower, upper)
    return proj

def _score(h):
    """Maximum overlap integral C5 for the discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _worst_shift(h):
    """Return the integer shift (relative to the centre of the full correlation) that attains the maximum overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift can be negative
    return shift, corr[idx]

def _gradient_worst_shift(h, shift):
    """Sub‑gradient of the worst overlap w.r.t. h."""
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        grad[:n - shift] -= h[shift:]
        grad[shift:]     -= h[:n - shift]
    else:
        t = -shift
        grad[:n - t] -= h[t:]
        grad[t:]    -= h[:n - t]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with an improved upper bound for C5."""
    import time
    np.random.seed(seed)

    # ----- initialise -------------------------------------------------------

    n = h.shape[0]

    u = _project_to_sum_and_bounds(u)

    best_h = h.copy()
    best_val = _score(h)
