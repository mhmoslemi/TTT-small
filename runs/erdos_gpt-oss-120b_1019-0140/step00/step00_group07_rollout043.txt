def compute_max_corr(h):
    # assume integer 0/1
    return np.max(np.correlate(h, 1-h, mode='full'))

def evaluate(h):
    return compute_max_corr(h) * (2.0 / len(h))

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    rnd = np.random.RandomState(seed)
    start = time.time()
    # get initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n = h0.size
        # enforce binary and correct sum
        # Round each to 0 or 1
        # Maybe h0 contains float; we produce a binary with exactly n/2 ones.
        target_ones = n // 2
        # To respect initial structure we can sort indices by h0 descending and set top target_ones to 1.
        order = np.argsort(-h0)  # descending
        h = np.zeros(n, dtype=int)
        h[order[:target_ones]] = 1
    else:
        n = 80
        target_ones = n // 2
        h = rnd.choice(n, size=target_ones, replace=False)
        h_vec = np.zeros(n, dtype=int)
        h_vec[h] = 1
        h = h_vec

    best_h = h.copy()
    best_score = evaluate(best_h)

    # Precompute mask for index pairs for each lag? Not needed.

    # Settings
    it = 0
    no_improve = 0
    max_no_improve = 2000
    while time.time() - start < budget_s * 0.95:  # keep margin
        # propose a swap
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        # random swap indices
        i = rnd.choice(ones)
        j = rnd.choice(zeros)
        # create candidate
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        # evaluate
        cand_score = evaluate(cand)
        if cand_score < best_score - 1e-12:  # improvement
            best_h = cand
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # occasional random restart if stuck
        if no_improve >= max_no_improve:
            # generate random candidate
            cand = rnd.choice(n, size=target_ones, replace=False)
            cand_vec = np.zeros(n, dtype=int)
            cand_vec[cand] = 1
            cand_score = evaluate(cand_vec)
            if cand_score < best_score:
                best_h = cand_vec
                best_score = cand_score
            no_improve = 0

        it += 1

    c5_bound = best_score
    return (best_h.astype(float), c5_bound, n)

    # Determine max delta such that bounds satisfied
    max_delta_i = 1 - h[i]  # can increase up to 1
    max_delta_j = h[j]      # can decrease down to 0
    max_up = max_delta_i
    max_down = max_delta_j
    # we need delta <= max_up and delta <= max_down (since subtract from j)
    max_delta = min(max_up, max_down)
    # also we may allow negative delta (decrease i, increase j). So we can choose random sign.
    # Choose random delta uniformly in [-max_delta_possible, max_delta_possible] where both directions possible.
    # Compute max possible for decreasing i: h[i], and increasing j: 1 - h[j]
    max_down_i = h[i]         # can decrease
    max_up_j = 1 - h[j]       # can increase
    # So delta can be in [-min(max_down_i, max_up_j), min(max_up, max_down)]
    delta_low = -min(max_down_i, max_up_j)
    delta_high = min(max_up, max_down)
    if delta_low >= delta_high:
        return None # no move possible
    delta = rnd.uniform(delta_low, delta_high)
    return i, j, delta

### compute_c5

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
```

def build_h(c, r, n):

    if t0 < 0:

    left_const_end = t0
    left_ramp_end = t0 + r

    # left constant region

def search_best(n, time_limit):
    best_h = None
    best_score = np.inf
    best_params = None
    start = time.time()
    # coarse grid
    for c in np.linspace(0.0, 0.2, 21):  # 0 to 0.2
        denom = 1 - c
        max_r = max(0.0, 2.0 - 1.0 / denom)
        if max_r <= 0:
            continue
        for r in np.linspace(0.0, max_r, 41):  # 41 points
            h = build_h(c, r, n)
            if h is None:
                continue
            score = compute_c5(h)
            if score < best_score:
                best_score = score
                best_h = h
                best_params = (c, r)
        # optional time break
        if time.time() - start > time_limit * 0.5:
            break
    # Refine around best
    if best_params is not None:
        c0, r0 = best_params
        for _ in range(3):
            # narrow grid
            c_vals = np.linspace(max(0.0, c0 - 0.02), min(0.3, c0 + 0.02), 21)
            for c in c_vals:
                denom = 1 - c
                max_r = max(0.0, 2.0 - 1.0 / denom)
                if max_r <= 0:
                    continue
                r_vals = np.linspace(max(0.0, r0 - 0.02), min(max_r, r0 + 0.02), 21)
                for r in r_vals:
                    h = build_h(c, r, n)
                    if h is None:
                        continue
                    score = compute_c5(h)
                    if score < best_score:
                        best_score = score
                        best_h = h
                        best_params = (c, r)
            c0, r0 = best_params
    return best_h, best_score, best_params

import numpy as np
import time
import math

def _compute_c5(h):
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _build_h(c, r, n):
    """
    Construct a symmetric step function with:
      - edge value  c   (length t0 on each side)
      - linear ramp of length r from c to 1
      - flat top of height 1
    Parameters:
        c : float, edge value (0 <= c <= 0.5)
        r : float, ramp length (0 <= r <= 2 - 1/(1-c))
        n : number of samples
    Returns:
        h array of length n (or None if parameters infeasible)
    """
    if c < 0 or c > 0.5:
        return None
    denom = 1.0 - c
    if denom <= 0:
        return None
    # maximal ramp length allowed by the area constraint
    max_r = max(0.0, 2.0 - 1.0 / denom)
    if r < 0.0 or r > max_r:
        return None
    # edge constant length t0 = (1/(1-c) - r) / 2
    t0 = (1.0 / denom - r) / 2.0
    if t0 < -1e-12:
        return None
    t0 = max(t0, 0.0)

    # region boundaries
    left_const_end = t0
    left_ramp_end = t0 + r
    plateau_start = left_ramp_end
    plateau_end = 2.0 - left_ramp_end
    right_const_start = 2.0 - t0

    x = (np.arange(n) + 0.5) * (2.0 / n)   # mid‑point sampling
    h = np.empty_like(x)

    # left constant region
    mask = x < left_const_end
    h[mask] = c

    # left ramp
    if r > 1e-12:
        mask = (x >= left_const_end) & (x < left_ramp_end)
        h[mask] = c + (x[mask] - left_const_end) * ((1.0 - c) / r)

    # plateau (height 1)
    mask = (x >= left_ramp_end) & (x <= plateau_end)
    h[mask] = 1.0

    # right ramp (mirror of left)
    if r > 1e-12:
        mask = (x > plateau_end) & (x < right_const_start)
        h[mask] = c + (right_const_start - x[mask]) * ((1.0 - c) / r)

    # right constant region
    mask = x >= right_const_start
    h[mask] = c

    # Clip numerical noise
    h = np.clip(h, 0.0, 1.0)

    # Enforce exact sum constraint sum(h) = n/2
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Adjust a few free entries (those not at a bound)
        if diff > 0:   # need to increase sum
            idxs = np.where(h < 1.0 - 1e-12)[0]
            for idx in idxs:
                delta = min(diff, 1.0 - h[idx])
                h[idx] += delta
                diff -= delta
                if diff <= 1e-12:
                    break
        else:          # need to decrease sum
            idxs = np.where(h > 1e-12)[0]
            for idx in idxs:
                delta = max(diff, -h[idx])
                h[idx] += delta
                diff -= delta
                if diff >= -1e-12:
                    break
    return h

def _search_best(n, time_limit):
    """Coarse‑to‑fine grid search over (c,r) to minimise C5."""
    start = time.time()
    best_h = None
    best_score = np.inf
    best_params = None

    # coarse grid
    for c in np.linspace(0.0, 0.25, 26):
        denom = 1.0 - c
        max_r = max(0.0, 2.0 - 1.0 / denom)
        if max_r <= 0.0:
            continue
        for r in np.linspace(0.0, max_r, 41):
            h = _build_h(c, r, n)
            if h is None:
                continue
            score = _compute_c5(h)
            if score < best_score:
                best_score = score
                best_h = h
                best_params = (c, r)
        if time.time() - start > time_limit * 0.5:
            break

    # fine refinement around the current best
    if best_params is not None:
        c0, r0 = best_params
        for _ in range(3):
            c_vals = np.linspace(max(0.0, c0 - 0.02), min(0.25, c0 + 0.02), 21)
            for c in c_vals:
                denom = 1.0 - c
                max_r = max(0.0, 2.0 - 1.0 / denom)
                if max_r <= 0.0:
                    continue
                r_vals = np.linspace(max(0.0, r0 - 0.02), min(max_r, r0 + 0.02), 21)
                for r in r_vals:
                    h = _build_h(c, r, n)
                    if h is None:
                        continue
                    score = _compute_c5(h)
                    if score < best_score:
                        best_score = score
                        best_h = h
                        best_params = (c, r)
            c0, r0 = best_params
            if time.time() - start > time_limit * 0.9:
                break
    return best_h, best_score, best_params

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rnd = np.random.RandomState(seed)

    # Determine number of points – use the supplied one if present
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
    else:
        n = 200  # a modest resolution that stays well below the 1k limit

    # First, try a structured two‑parameter family (edge value c, ramp length r)
    time_limit = budget_s * 0.95
    best_h, best_score, _ = _search_best(n, time_limit)

    # If an initial construction exists, evaluate it as a fallback candidate
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        # ensure exact sum constraint
        target = n / 2.0
        diff = target - init_h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution to satisfy the sum (preserve bounds)
            if diff > 0:
                idxs = np.where(init_h < 1.0 - 1e-12)[0]
                for idx in idxs:
                    add = min(diff, 1.0 - init_h[idx])
                    init_h[idx] += add
                    diff -= add
                    if diff <= 1e-12:
                        break
            else:
                idxs = np.where(init_h > 1e-12)[0]
                for idx in idxs:
                    sub = max(diff, -init_h[idx])
                    init_h[idx] += sub
                    diff -= sub
                    if diff >= -1e-12:
                        break
        init_score = _compute_c5(init_h)
        if init_score < best_score:
            best_h = init_h
            best_score = init_score

    # Optional local stochastic improvement (simple hill‑climb)
    start = time.time()
    current_h = best_h.copy()
    current_score = best_score
    while time.time() - start < (budget_s - (time.time() - start) - 1):
        i, j = rnd.choice(n, size=2, replace=False)
        # compute how much we can move while staying in [0,1]
        max_inc_i = 1.0 - current_h[i]
        max_dec_i = current_h[i]
        max_inc_j = 1.0 - current_h[j]
        max_dec_j = current_h[j]
        delta_low = -min(max_dec_i, max_inc_j)
        delta_high = min(max_inc_i, max_dec_j)
        if delta_low >= delta_high:
            continue
        delta = rnd.uniform(delta_low, delta_high)
        cand = current_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_score = _compute_c5(cand)
        if cand_score < current_score:
            current_h = cand
            current_score = cand_score
            if cand_score < best_score:
                best_h = cand
                best_score = cand_score
                # early stop if we already beat the target record
                if best_score <= 0.3808:
                    break

    return (best_h.astype(float), float(best_score), int(n))
