import numpy as np
import random
import time

def compute_c5(h):
    """Return max overlap integral for binary step function h."""
    # correlation of h with its complement
    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max() * (2.0 / len(h)))


def make_binary_from_initial(arr):
    """Convert a float array to a binary vector with exactly half ones."""
    N = len(arr)
    ones_needed = N // 2
    # indices of the largest entries become 1
    idx = np.argpartition(arr, -ones_needed)[-ones_needed:]
    bin_vec = np.zeros(N, dtype=int)
    bin_vec[idx] = 1
    return bin_vec


def random_binary(N):
    """Random binary vector with exactly N/2 ones (N must be even)."""
    ones_needed = N // 2
    perm = np.random.permutation(N)
    vec = np.zeros(N, dtype=int)
    vec[perm[:ones_needed]] = 1
    return vec


def mutate_swap(vec, max_swaps):

            break
        i = random.choice(ones.tolist())
        j = random.choice(zeros.tolist())
        child[i] = 0
        child[j] = 1
    return child
