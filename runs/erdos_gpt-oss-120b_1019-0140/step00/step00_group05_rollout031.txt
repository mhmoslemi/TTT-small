def project_to_box_simplex(v, target):
    # if target out of feasible range, clamp.
    n = v.shape[0]
    lower = -np.max(v) - 1.0
    upper = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lower + upper) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lower = lam
        else:
            upper = lam
    x = np.clip(v + (lower + upper) / 2.0, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) > 1e-8:
        # distribute residual to non-bound entries
        residual = target - s
        # Find eligible indices
        idx = (x > 1e-12) & (x < 1 - 1e-12)
        if idx.any():
            # distribute equally
            inc = residual / idx.sum()
            x[idx] += inc
            # clip after distribution
            x = np.clip(x, 0.0, 1.0)
            # might still not sum exactly; not big deal
    return x

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the set {x | 0 <= x_i <= 1, sum(x) = target},
    using a simple bisection on the shift lambda.
    """
    n = v.shape[0]
    # bounds for lambda that guarantee all entries are clipped to 0 or 1
    lam_low = -np.max(v) - 1.0      # forces v+lam <= 0  => all zeros
    lam_high = 1.0 - np.min(v) + 1.0  # forces v+lam >= 1  => all ones

    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(v + (lam_low + lam_high) * 0.5, 0.0, 1.0)

    # tiny fixation of numerical drift
    s = x.sum()
    if abs(s - target) > 1e-8:
        diff = target - s
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that minimizes the
    Erdős overlap constant C5. Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initial construction
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=float)
        n = h.shape[0]
    else:
        n = 70
        h = np.full(n, 0.5, dtype=float)

    target_sum = n * 0.5  # because integral must be 1, dx = 2/n
    h = _project_to_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Search parameters
    sigma_initial = 0.25   # initial scale of random perturbations
    sigma_final = 0.001    # final scale
    prob_global = 0.12     # probability to draw a full‑vector proposal
    max_no_improve = 5000  # steps before optional restart (unused currently)

    # Main loop
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.95:
            break
        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

        # linearly anneal sigma
        frac = elapsed / budget_s
        sigma = sigma_initial * (1.0 - frac) + sigma_final * frac

        # ------------------------------------------------------------------
        # Propose a new candidate
        if rng.random() < prob_global:
            # Global random perturbation
            cand = best_h + sigma * rng.standard_normal(n)
            cand = _project_to_box_simplex(cand, target_sum)
        else:
            # Local two‑coordinate move that preserves the sum exactly
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

            # feasible interval for delta so that both entries stay in [0,1]
            low = max(-best_h[i], best_h[j] - 1.0)
            high = min(1.0 - best_h[i], best_h[j])
            if low >= high:
                continue
            # limit step size to sigma (optional scaling)
            max_step = min(high - low, sigma)
            if max_step <= 0:
                continue
            delta = rng.uniform(-max_step, max_step)
            # shift delta to fit inside feasible interval
            delta = np.clip(delta, low, high)
            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta

        cand_c5 = _compute_c5(cand)

        # ------------------------------------------------------------------
        # Acceptance: keep any improvement; otherwise keep current best
        if cand_c5 < best_c5:
            best_h = cand
            best_c5 = cand_c5

    # ------------------------------------------------------------------
    # Final sanity projection (ensure constraints)
    best_h = _project_to_box_simplex(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), int(n)
