def propose_move(h, step, rng):
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i: j += 1
    delta = (rng.random() - 0.5) * 2 * step  # uniform in [-step, step]

    if h[i] + delta < 0 or h[i] + delta > 1:
        return None
    if h[j] - delta < 0 or h[j] - delta > 1:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    # Determine n
    try:
        init = np.array(initial_h_values, copy=True)
        n = init.shape[0]
        use_initial = True
    except NameError:
        n = 60
        use_initial = False
    target = n / 2.0
    rng = np.random.default_rng(seed)
    # projection function defined locally
    def proj(v):
        # binary search for lambda to satisfy sum constraint
        lo = -1e5
        hi = 1e5
        for _ in range(60):
            lam = (lo + hi) / 2.0
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    if use_initial:
        h = proj(init)
    else:
        # random feasible
        v = rng.random(n)
        h = proj(v)
    # evaluate
    def score_vec(vec):
        arr = np.correlate(vec, 1 - vec, mode='full')
        return arr.max() * (2.0 / n)
    best_h = h.copy()
    best_score = score_vec(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    start_time = time.time()
    max_iter = 10**9  # effectively infinite, break on time
    iter_count = 0
    # Annealing schedule parameters
    T0 = best_score * 0.1 if best_score > 0 else 0.01
    if T0 < 1e-4:
        T0 = 1e-4
    T = T0
    # step size initial
    step_init = 0.2
    # track no-improvement counter
    no_improve = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # temperature decrease
        T *= 0.9995  # exponential decay
        # step size proportional to temperature
        step = step_init * (T / T0) if T0>0 else step_init
        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        delta = (rng.random() - 0.5) * 2 * step
        # Ensure feasible
        hi_i = cur_h[i] + delta
        lo_j = cur_h[j] - delta
        if hi_i < 0.0 or hi_i > 1.0 or lo_j < 0.0 or lo_j > 1.0:
            # reject move
            continue
        new_h = cur_h.copy()
        new_h[i] = hi_i
        new_h[j] = lo_j
        # Evaluate new
        new_score = score_vec(new_h)
        # Accept criteria
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # Random restart if stuck
        if no_improve > 5000:
            # restart from random feasible
            cur_h = proj(rng.random(n))
            cur_score = score_vec(cur_h)
            no_improve = 0
        iter_count += 1
    return (best_h, float(best_score), n)

import time
import math
import numpy as np

def _project_capped_simplex(v, target, upper=1.0):
    """Project v onto the capped simplex {x: 0<=x_i<=upper, sum x_i = target}."""
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, upper)

def _score(vec):
    """Return C5 value for a step function given by vec."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return corr.max() * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C5 bound."""
    # Determine problem size and initial vector
    try:
        init = np.array(initial_h_values, copy=True)
        n = init.shape[0]
        use_init = True
    except NameError:
        n = 60                     # fallback size if no parent provided
        use_init = False

    target = n / 2.0               # required sum of h values
    rng = np.random.default_rng(seed)

    # Prepare a feasible starting point
    if use_init:
        h = _project_capped_simplex(init, target)
    else:
        h = _project_capped_simplex(rng.random(n), target)

    best_h = h.copy()
    best_score = _score(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    start_time = time.time()
    # Simulated‑annealing parameters
    T0 = max(0.1, best_score * 0.5)   # initial temperature
    T = T0
    step_init = 0.5                    # max mass moved in one proposal
    no_improve = 0

    # Main optimisation loop
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Exponential cooling
        T *= 0.9995
        if T < 1e-7:
            T = 1e-7

        # Step size follows temperature
        step = step_init * (T / T0) if T0 > 0 else step_init

        # Propose a mass transfer between two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = (rng.random() - 0.5) * 2.0 * step
        hi_i = cur_h[i] + delta
        lo_j = cur_h[j] - delta
        if not (0.0 <= hi_i <= 1.0 and 0.0 <= lo_j <= 1.0):
            continue  # infeasible proposal

        new_h = cur_h.copy()
        new_h[i] = hi_i
        new_h[j] = lo_j
        new_score = _score(new_h)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
