def compute_c5(h):

    conv = np.correlate(h, 1 - h, mode='full')

def propose_move(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]; hj = h[j]
    # compute max delta such that hi+delta in [0,1] and hj-delta in [0,1]
    delta_max = min(1 - hi, hj)  # increase hi, decrease hj
    delta_min = max(-hi, -(1 - hj))  # decrease hi, increase hj
    if delta_max <= delta_min:
        return None, None, None
    # Choose delta uniform in [delta_min, delta_max]
    delta = rng.uniform(delta_min, delta_max)
    # Usually we want move that changes values: if delta is zero, skip.
    if abs(delta) < 1e-12:
        return None, None, None
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h, i, j
```

import numpy as np
import time
import math

def compute_c5(h):
    """Return C5 for a given vector h using the prescribed correlation formula."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); the max times dx is the bound
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0):
    """
    Project vector v onto the hyperplane sum(x)=target_sum while keeping each entry in [lower,upper].
    Uses a binary-search on a Lagrange multiplier (water-filling).
    """
    lo, hi = -1e6, 1e6
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # tiny residual correction (distribute to non‑bound entries)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, lower, upper)
    return x
