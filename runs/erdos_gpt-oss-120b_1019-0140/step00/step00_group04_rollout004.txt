def compute_c5(h):
    try:
        # use pre-defined helper if available
        return evaluate_erdos_solution(h)
    except NameError:

        dx = 2.0 / n
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_pair_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]; hj = h[j]
    # choose direction
    # Determine max feasible positive delta (increase hi, decrease hj)
    max_pos = min(1.0 - hi, hj)
    max_neg = min(hi, 1.0 - hj)  # negative delta magnitude
    if max_pos <= 0 and max_neg <= 0:
        return None, None, None
    # Randomly decide direction
    if max_pos > 0 and max_neg > 0:
        if rng.rand() < 0.5:
            delta = rng.uniform(0, max_pos)
        else:
            delta = -rng.uniform(0, max_neg)
    elif max_pos > 0:
        delta = rng.uniform(0, max_pos)
    else:
        delta = -rng.uniform(0, max_neg)
    return i, j, delta

def project_to_feasible(h):
    # Ensure bounds [0,1] and sum = n/2 by simple clipping and redistribution.
    n = len(h)
    target = n / 2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum by adding uniform delta and then clipping iteratively
    # Use simple algorithm: while sum not target, shift mass from elements >0 and <1.

    diff = target - h.sum()
    # distribute diff uniformly to all entries within bounds

import numpy as np
import random
import time

def compute_c5(h):
    """Return C5 = max correlation * dx for a step function h."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h), full mode
    c = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return float(c * dx)

def random_pair_move(h, rng):
    """Propose a feasible pairwise mass transfer that preserves the sum."""
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # Maximum feasible positive delta (increase hi, decrease hj)
    max_pos = min(1.0 - hi, hj)
    # Maximum feasible negative delta (decrease hi, increase hj) expressed as magnitude
    max_neg = min(hi, 1.0 - hj)

    if max_pos <= 0 and max_neg <= 0:
        return None, None, None

    # Choose direction uniformly
    if max_pos > 0 and max_neg > 0:
        if rng.random() < 0.5:
            delta = rng.uniform(0.0, max_pos)
        else:
            delta = -rng.uniform(0.0, max_neg)
    elif max_pos > 0:
        delta = rng.uniform(0.0, max_pos)
    else:
        delta = -rng.uniform(0.0, max_neg)

    return i, j, delta

def project_to_feasible(h):
    """
    Project vector h onto the box [0,1]^n while forcing sum(h) = n/2.
    Simple iterative redistribution; sufficient for modest n.
    """
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass
            free = h < 1.0
            if not np.any(free):
                break
            add = diff / np.count_nonzero(free)
            add = np.minimum(add, 1.0 - h[free])
            h[free] += add
        else:
            # need to remove mass
            free = h > 0.0
            if not np.any(free):
                break
            sub = (-diff) / np.count_nonzero(free)
            sub = np.minimum(sub, h[free])
            h[free] -= sub

    # Final tweak to enforce exact sum (tiny adjustment on first element)
    cur_sum = h.sum()
    if abs(cur_sum - target) > 1e-9:
        h[0] += target - cur_sum
        h[0] = min(max(h[0], 0.0), 1.0)

    return h
