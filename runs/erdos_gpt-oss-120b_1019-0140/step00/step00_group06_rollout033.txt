def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # use np.correlate
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def repair_h(h):
    # ensure within [0,1] and sum = n/2
    n = len(h)
    # clip
    h = np.clip(h, 0, 1)
    # compute sum
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) > 1e-12:
        # project onto box [0,1] with sum target using water-filling
        # solve for lambda as described
        lo = -1.0
        hi = 1.0
        # adjust bounds
        while np.sum(np.clip(h - lo, 0, 1)) < target:
            lo -= 1.0
        while np.sum(np.clip(h - hi, 0, 1)) > target:
            hi += 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0, 1))
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0, 1)
    return h

def hill_climb(initial_h, time_limit):
    # local search until time expires
    n = len(initial_h)
    cur_h = initial_h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    # Precompute indices where h not at bound for sampling
    # We'll sample random pairs i, j each iteration.
    # Temperature schedule maybe none.
    while True:
        now = time.time()
        if now - start > time_limit:
            break
        # pick random i,j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # try to move delta from j to i (increase i, decrease j)
        # compute allowed delta
        max_up = 1.0 - cur_h[i]
        max_down = cur_h[j]
        max_delta = min(max_up, max_down)
        if max_delta <= 1e-8:
            continue
        # choose delta uniformly in (0, max_delta]
        delta = max_delta * np.random.rand()
        # apply delta
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Evaluate step
        new_score = compute_c5(new_h)
        # Accept if improvement
        if new_score < cur_score - 1e-12:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        # else maybe accept with tiny probability
        # ignore uphill moves for simplicity
    return best_h, best_score

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=m)

    max_val = np.max(c)
    return max_val * dx

import time

def _c5_precise(h: np.ndarray) -> float:
    """Exact C5 as defined by the problem (uses np.correlate)."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def _c5_fast(h: np.ndarray) -> float:
    """Fast C5 via FFT (same result up to rounding)."""
    n = h.shape[0]
    dx = 2.0 / n
    m = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft((1.0 - h)[::-1], n=m)
    conv = np.fft.irfft(H * np.conj(B), n=m)
    corr = conv[: 2 * n - 1]
    return float(np.max(corr) * dx)


def _project_to_constraints(z: np.ndarray) -> np.ndarray:
    """
    Project vector z onto the box [0,1]^n while enforcing sum(z) = n/2.
    Uses a simple water‑filling (bisection) scheme.
    """
    n = z.shape[0]
    target = n / 2.0
    z = np.clip(z, 0.0, 1.0)

    # lambda bounds guaranteeing sum(z - λ) after clipping spans [0, n]
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)


def _hill_climb(start_h: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing with a tiny simulated‑annealing component.
    Keeps the exact sum constraint by moving a delta from one index to another.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_score = _c5_fast(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()

    while True:
        if time.time() - start > time_limit:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # we will add delta to i and remove the same delta from j
        max_up = 1.0 - cur_h[i]
        max_down = cur_h[j]
        max_delta = max_up if max_up < max_down else max_down
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand()

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_score = _c5_fast(new_h)

        # linear temperature schedule, max temperature = 0.01
        elapsed = time.time() - start
        T = 0.01 * max(0.0, 1.0 - elapsed / time_limit)

        if new_score < cur_score - 1e-12 or (
            T > 0.0 and np.random.rand() < math.exp(-(new_score - cur_score) / T)
        ):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

    return best_h, best_score

    best_h = base.copy()
    best_score = _c5_precise(best_h)
    best_n = n0

    # ------------------------------------------------------------------
    # 2. Stage 1 – improve at the original resolution
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0.0:
        stage1_time = remaining * 0.35
        h_opt, sc_opt = _hill_climb(best_h, stage1_time)
        sc_opt_precise = _c5_precise(h_opt)
        if sc_opt_precise < best_score:
            best_h, best_score, best_n = h_opt, sc_opt_precise, n0

    # ------------------------------------------------------------------
    # 3. Stage 2 – upscale to double resolution and optimise further
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0.0:
        n1 = n0 * 2
        # linear interpolation from the current best vector
        xs_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n1, endpoint=False)
        up = np.interp(xs_new, xs_old, best_h)
        up = _project_to_constraints(up)

        stage2_time = remaining * 0.65
        h_opt2, sc_opt2 = _hill_climb(up, stage2_time)
        sc_opt2_precise = _c5_precise(h_opt2)
        if sc_opt2_precise < best_score:
            best_h, best_score, best_n = h_opt2, sc_opt2_precise, n1

    # ------------------------------------------------------------------
    # 4. Final projection and exact scoring
    # ------------------------------------------------------------------
    best_h = _project_to_constraints(best_h)
    best_score = _c5_precise(best_h)

    return best_h.astype(np.float64), best_score, best_n
