def repair_h(h, target_sum):

    s = h.sum()
    if s == target_sum:
        return h
    # compute difference
    diff = target_sum - s
    # if diff > 0, need to increase some values
    # find indices where h < 1
    idx = np.where(h < 1)[0]
    # if idx empty, just fill a random element up to diff? But diff should be small.
    # allocate addition equally across idx but clip to not exceed 1.
    while abs(diff) > 1e-9 and idx.size > 0:
        inc = diff / idx.size
        # adjust
        add = np.minimum(1 - h[idx], np.maximum(inc, 0))
        # for diff negative (need decrease)
        if diff < 0:
            # decrease values
            add = np.maximum(-h[idx], np.minimum(inc, 0))
        # apply
        h[idx] += add
        # update diff
        diff = target_sum - h.sum()
        # update idx for next iteration
        idx = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]  # interior
    # after loop, maybe slight diff remains due numeric; adjust by adding to first element.

def propose_neighbor(h, step_max):
    # Choose two distinct indices
    i, j = np.random.choice(N, size=2, replace=False)
    # Determine feasible delta bounds:
    lower = max(-h[i], h[j] - 1)
    upper = min(1 - h[i], h[j])
    if lower >= upper:
        # cannot move, return same copy
        return h
    delta = (np.random.rand() * 2 - 1) * step_max  # in [-step_max, step_max]
    delta = np.clip(delta, lower, upper)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

import time

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)

    # --- initial solution -------------------------------------------------

        N = h0.size
    else:
        N = 70

        target_sum = N / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        s = h0.sum()
        if s > 0:
            h0 *= target_sum / s
        h0 = np.clip(h0, 0.0, 1.0)

    target_sum = N / 2.0

    # --- helper to enforce exact sum --------------------------------------
    def repair_exact(vec):
        v = np.clip(vec, 0.0, 1.0)
        diff = target_sum - v.sum()
        if abs(diff) > 1e-12:
            # distribute the deficit/excess among non‑boundary entries
            idx = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
            # if none are interior, fall back to any index
            if idx.size == 0:
                idx = np.arange(N)
            for i in rng.permutation(idx):
                if diff == 0:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - v[i])
                else:
                    add = max(diff, -v[i])
                v[i] += add
                diff = target_sum - v.sum()
        # final tiny correction (should stay in bounds)
        if abs(target_sum - v.sum()) > 1e-12:
            v[0] += target_sum - v.sum()
        return np.clip(v, 0.0, 1.0)

    h_current = repair_exact(h0.copy())

    # --- scoring -----------------------------------------------------------
    dx = 2.0 / N
    def score_of(vec):
        # correlation of h with (1‑h)
        return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * dx

    current_score = score_of(h_current)
    best_h = h_current.copy()
    best_score = current_score

    # --- simulated annealing / hill‑climbing -------------------------------
    start = time.time()
    timeout = budget_s * 0.95            # leave a safety margin
    init_temp = 0.02
    min_temp = 0.0005
    no_improve = 0

    while True:

        if now - start >= timeout:

        # temperature and step schedules
        elapsed = now - start
        temp = init_temp * (1.0 - elapsed / budget_s) + min_temp

        hi, hj = h_current[i], h_current[j]

        lower = max(-hi, hj - 1.0)   # ensure hi+δ ≥0 and hj‑δ ≤1
        upper = min(1.0 - hi, hj)    # ensure hi+δ ≤1 and hj‑δ ≥0

            no_improve = 0

    best_h = repair_exact(best_h)
    final_score = score_of(best_h)

    return best_h, final_score, N
