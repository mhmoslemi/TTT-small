    corr = np.correlate(h, 1 - h, mode='full')

import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def random_feasible(N):
    target = N / 2.0
    h = np.random.rand(N)
    delta = (target - h.sum()) / N
    h += delta
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    # adjust diff by adding/subtracting small amounts
    while abs(diff) > 1e-12:
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            i = np.random.choice(candidates)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            i = np.random.choice(candidates)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

def hill_climb(N, init_h=None, time_limit=1.0):
    start = time.time()
    total = time_limit
    target = N / 2.0

    if init_h is not None and len(init_h) == N:
        cur = np.array(init_h, dtype=float)
        np.clip(cur, 0.0, 1.0, out=cur)
        # fix sum if needed
        cur_sum = cur.sum()
        if abs(cur_sum - target) > 1e-9:
            cur = random_feasible(N)
    else:
        cur = random_feasible(N)

    best = cur.copy()
    best_score = compute_c5(best)
    cur_score = best_score

    Tmax = 0.05
    Tmin = 1e-7
    no_improve = 0
    restart_thr = 5000

    while time.time() - start < total:
        # propose neighbor
        i, j = np.random.choice(N, 2, replace=False)
        lo = max(cur[i] - 1.0, -cur[j])
        hi = min(cur[i], 1.0 - cur[j])
        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        # apply
        cur[i] -= delta
        cur[j] += delta

        cand_score = compute_c5(cur)

        elapsed = time.time() - start
        T = Tmax * (1 - elapsed / total) + Tmin

        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / T):
            cur_score = cand_score
            if cand_score < best_score:
                best = cur.copy()
                best_score = cand_score
                no_improve = 0

                no_improve += 1

            cur[i] += delta
            cur[j] -= delta
            no_improve += 1

            cur = best.copy()
            cur_score = best_score

            for _ in range(10):

                delta = np.random.uniform(lo, hi)
                cur[i] -= delta
                cur[j] += delta

            no_improve = 0

    # final repair to meet sum exactly (tiny numerical drift)
    diff = target - best.sum()
    if abs(diff) > 1e-12:
        # distribute diff greedily
        while abs(diff) > 1e-12:
            if diff > 0:
                candidates = np.where(best < 1.0)[0]
                if candidates.size == 0:

                i = np.random.choice(candidates)

                best[i] += inc
                diff -= inc

                if candidates.size == 0:
                    break
                i = np.random.choice(candidates)
                dec = min(-diff, best[i])
                best[i] -= dec
                diff += dec

    final_score = compute_c5(best)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def random_feasible(N):
    target = N / 2.0
    h = np.random.rand(N)
    h += (target - h.sum()) / N
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            i = np.random.choice(candidates)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            i = np.random.choice(candidates)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

def hill_climb(N, init_h=None, time_limit=1.0):

        # ensure sum
        if abs(cur.sum() - target) > 1e-9:
            cur = random_feasible(N)
    else:
        cur = random_feasible(N)

import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    target = n / 2.0
    h = np.random.rand(n)
    h += (target - h.sum()) / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    # tiny correction loop – fast because diff quickly becomes zero
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break

            diff -= inc

            if cand.size == 0:
                break

            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h
