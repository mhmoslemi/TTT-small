def propose_move(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    hi = h[i]; hj = h[j]
    # viable delta range
    delta_low = max(-hi, hj - 1.0)
    delta_high = min(1.0 - hi, hj)
    # also bound by max_step
    low = max(delta_low, -max_step)
    high = min(delta_high, max_step)
    if low >= high:
        return None, None # no move
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    # other entries unchanged
    return new_h, (i, j, delta)

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * dx

def propose_move(h, max_step):
    n = h.shape[0]
    for _ in range(10):  # attempts
        i, j = np.random.choice(n, size=2, replace=False)
        hi = h[i]; hj = h[j]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        # bound by max_step
        low = max(delta_low, -max_step)
        high = min(delta_high, max_step)
        if low < high:
            delta = np.random.uniform(low, high)
            new_h = h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            return new_h
    return None

def project_to_sum_and_bounds(x, target_sum):
    # Clip to [0,1]
    y = np.clip(x, 0.0, 1.0)
    # Compute sum
    s = y.sum()
    if abs(s - target_sum) < 1e-9:
        return y

    # We'll treat as projection onto lower and upper bound constraints.
    # Find lambda such that sum(clip(y_i + lambda, 0, 1)) = target_sum.

            lo = mid

            hi = mid

import time
import math
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def _project_to_sum_and_bounds(x, target):
    """
    Project x onto the box [0,1]^n while enforcing sum(x)=target.
    Uses a simple bisection on a global shift lambda.
    """
    y = np.clip(x, 0.0, 1.0)
    if abs(y.sum() - target) <= 1e-12:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        z = np.clip(y + mid, 0.0, 1.0)
        s = z.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def _propose_move(h, max_step):
    """
    Return a new feasible h created by moving a delta from index i to j.
    The move respects box constraints and preserves the total sum exactly.
    """
    n = h.shape[0]
    # try a few random index pairs to find a feasible interval
    for _ in range(10):
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]
        # feasible delta range from box constraints
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        # intersect with step magnitude limits
        low = max(delta_low, -max_step)
        high = min(delta_high, max_step)
        if low < high:
            delta = np.random.uniform(low, high)
            new_h = h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            return new_h
    return None  # no feasible move found this round

    """Search for a step function h on [0,2] giving a small C5 bound.

    Returns

        c5_bound : float, the achieved C5 bound
        n_points : int, the number of samples used
    """

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h

    except NameError:
