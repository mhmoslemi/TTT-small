import time
import math
import random
import numpy as np

def _enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h (in-place) so that sum(h) == target while keeping 0<=h<=1."""
    diff = target - h.sum()
    # Simple greedy correction; diff will be tiny after all moves.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - h
            idx = int(np.argmax(caps))
            delta = min(diff, caps[idx])
            h[idx] += delta
            diff -= delta
        else:
            caps = h
            idx = int(np.argmax(caps))
            delta = min(-diff, caps[idx])
            h[idx] -= delta
            diff += delta
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _transfer_move(h: np.ndarray, step: float) -> np.ndarray | None:
    """Transfer a small amount between two positions while preserving the sum."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    lo = max(-h[i], h[j] - 1.0)          # lower bound for delta
    hi = min(1.0 - h[i], h[j])           # upper bound for delta
    if lo >= hi:
        return None
    lo = max(lo, -step)
    hi = min(hi, step)
    if lo >= hi:
        return None
    delta = np.random.uniform(lo, hi)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    return new

def _bound_move(h: np.ndarray) -> np.ndarray | None:
    """Set a random entry to 0 or 1 and compensate by moving the same amount from another entry."""
    n = len(h)
    i = np.random.randint(n)
    bound = 1.0 if random.random() < 0.5 else 0.0
    delta = bound - h[i]                 # amount to add to index i
    if abs(delta) < 1e-12:
        return None
    if delta > 0:
        eligible = np.where(h >= delta)[0]
    else:
        eligible = np.where(h <= 1.0 + delta)[0]   # delta negative
    eligible = eligible[eligible != i]
    if eligible.size == 0:
        return None
    j = int(np.random.choice(eligible))
    new = h.copy()
    new[i] = bound
    new[j] -= delta
    new = np.clip(new, 0.0, 1.0)
    return new

def _targeted_move(h: np.ndarray, step: float) -> np.ndarray | None:
    """Move mass from a position contributing to the worst overlap toward its partner."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    shift = k - (n - 1)                 # signed lag producing the worst overlap
    start = max(0, -shift)
    end = min(n, n - shift)
    if end - start <= 0:
        return None
    i = np.random.randint(start, end)
    j = i + shift
    # decide direction: move from larger to smaller to reduce product
    if h[i] > h[j]:
        src, dst = i, j
    else:
        src, dst = j, i
    max_decrease = h[src]               # can lower src to 0
    max_increase = 1.0 - h[dst]         # can raise dst to 1
    allowed = min(max_decrease, max_increase)
    if allowed <= 0:
        return None
    delta = np.random.uniform(0, min(step, allowed))
    if delta <= 0:
        return None
    new = h.copy()
    new[src] -= delta
    new[dst] += delta
    return new

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) attempting to lower the Erdős C5 constant."""
    random.seed(seed)
    np.random.seed(seed)

    # ---- initial vector ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _enforce_sum(h0, target)
    else:
        n = 45
        target = n / 2.0
        h0 = np.full(n, 0.5, dtype=float)

    # tiny correction to guarantee exact sum (helps later arithmetic)
    diff = target - h0.sum()
    if abs(diff) > 1e-12:
        for idx in range(n):
            low = -h0[idx]
            high = 1.0 - h0[idx]
            if low <= diff <= high:
                h0[idx] += diff
                break

    current = h0.copy()
    cur_val = _c5_value(current)
    best = current.copy()
    best_val = cur_val

    # ---- simulated annealing parameters ------------------------------------
    start_time = time.time()
    temp = 0.2                     # initial temperature
    cooling = 0.99995              # exponential cooling factor
    max_step = 0.2                 # maximal transfer amount
    no_improve = 0
    target_bound = 0.3808          # stop early if we beat this

    while time.time() - start_time < budget_s * 0.99:
        # adapt step size with temperature
        step = max_step * (temp / 0.2)

        # choose move type
        r = random.random()
        if r < 0.2:
            cand = _bound_move(current)
        elif r < 0.4:
            cand = _targeted_move(current, step)
        else:
            cand = _transfer_move(current, step)

        if cand is None:
            continue

        cand_val = _c5_value(cand)

        # acceptance rule (Metropolis)
        if cand_val < cur_val or random.random() < math.exp((cur_val - cand_val) / max(temp, 1e-12)):
            current = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val
                no_improve = 0
                if best_val <= target_bound:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        temp *= cooling

        # occasional restart from best to avoid dead ends
        if no_improve >= 5000:
            current = best.copy()
            cur_val = best_val
            no_improve = 0

    return best, float(best_val), n
