def compute_c5(h, dx):
    conv = np.correlate(h, 1 - h, mode='full')
    return (conv * dx).max()

def project_to_simplex(v, target, max_iter=60):
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        u = np.clip(v + mid, 0, 1)
        s = u.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    u = np.clip(v + (lo+hi)/2, 0, 1)
    # residual fix
    residual = target - u.sum()
    if abs(residual) > 1e-9:
        free = (u > 1e-12) & (u < 1 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            u[free] += residual / cnt
            u = np.clip(u, 0, 1)
        else:
            # distribute to any index
            u[0] = np.clip(u[0] + residual, 0, 1)
    return u

import time
import numpy as np

def _c5_value(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by discrete correlation."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return (conv * dx).max()

def _project_to_capped_simplex(v, target, max_iter=60, tol=1e-12):
    """Project v onto the set {0≤x_i≤1, sum x_i = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        u = np.clip(v + mid, 0.0, 1.0)
        s = u.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    u = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # small residual correction
    residual = target - u.sum()
    if abs(residual) > tol:
        free = (u > tol) & (u < 1.0 - tol)
        cnt = free.sum()
        if cnt > 0:
            u[free] += residual / cnt
            u = np.clip(u, 0.0, 1.0)
        else:
            u[0] = np.clip(u[0] + residual, 0.0, 1.0)
    return u

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start = time.time()

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)   # sum = n/2 → integral = 1
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # project to feasible set
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    # ----- optimisation parameters ---------------------------------------
    lr = 0.7                 # initial learning rate
    it = 0
    no_improve = 0
    max_no_improve = 20000
    # main optimisation loop
    while True:
        if time.time() - start > budget_s - 0.2:
            break

        # current correlation and worst shift
        conv = np.correlate(h, 1.0 - h, mode="full")
        conv_scaled = conv * dx
        max_idx = int(conv_scaled.argmax())
        cur_val = float(conv_scaled[max_idx])
        shift = max_idx - (n - 1)

        # record best

            best_h = h.copy()

            h = h_new
        # learning‑rate schedule
        if it % 5000 == 0 and it > 0:
            lr *= 0.9

        # random restart if stuck
        if no_improve >= max_no_improve:
            # perturb the best known solution
            noise = (np.random.rand(n) - 0.5) * 0.2
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_capped_simplex(h, target_sum)
            no_improve = 0

        it += 1

    return best_h, best_val, n
