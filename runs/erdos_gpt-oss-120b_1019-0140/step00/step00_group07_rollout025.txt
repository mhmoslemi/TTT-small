def propose_move(h):
    n = len(h)
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    hi, hj = h[i], h[j]
    # Determine max allowed delta positive: inc hi (up) and dec hj
    max_inc = min(1 - hi, hj)   # δ >0
    max_dec = min(hi, 1 - hj)   # δ <0 (i.e., δ negative with magnitude up to max_dec)
    if max_inc <=0 and max_dec <=0:
        return None
    # Choose delta uniform within [-max_dec, max_inc]
    delta = np.random.uniform(-max_dec, max_inc)
    # Optionally scale by step_scale factor
    return i, j, delta

def compute_c5(h):
    n = len(h)

    # Max across all entries
    return np.max(corr) * dx

def random_feasible_h(rng, n, target_sum):
    # generate random numbers from Uniform(0,1)
    h = rng.random(n)
    # normalize sum to target_sum
    h = h / h.sum() * target_sum
    # clip to [0,1] and renormalize iteratively
    for _ in range(10):
        over = h > 1
        if not np.any(over):
            break
        excess = h[over] - 1
        total_excess = excess.sum()
        h[over] = 1
        # redistribute excess to underfull entries proportionally
        under = h < 1
        if not np.any(under):
            break
        h[under] += total_excess * (h[under] / h[under].sum())
    # final clip
    h = np.clip(h, 0, 1)
    # adjust sum precisely
    diff = target_sum - h.sum()
    # add diff uniformly to all entries (can cause slight violation) and clip again
    if abs(diff) > 1e-12:
        # distribute diff equally among those not at bounds
        # We'll just adjust first element for simplicity but maintain bounds
        i = np.argmin(h)  # smallest element
        h[i] = np.clip(h[i] + diff, 0, 1)
        # after adjust ensure sum maybe still off due to clipping; if still off recompute diff and adjust else
        # for simplicity, ignore small residual
    return h

def propose_move(h, rng, max_step=1.0):
    n = len(h)
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    hi = h[i]
    hj = h[j]
    max_inc = min(1 - hi, hj)
    max_dec = min(hi, 1 - hj)
    # delta positive up to max_inc, negative down to -max_dec
    if max_inc <= 0 and max_dec <= 0:
        return None
    # Choose random delta within allowed range, scaled by step factor
    delta = (rng.random() * (max_inc + max_dec) - max_dec) * max_step
    # But we need to ensure delta stays within allowed range (if max_step >1 there may be overflow)
    delta = max(-max_dec, min(delta, max_inc))
    return i, j, delta

import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # cross‑correlation of h with (1‑h); max over all shifts
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def random_binary_vector(n, total, rng):
    """Create a binary vector of length n with exactly total ones (as floats)."""
    vec = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=int(total), replace=False)
    vec[idx] = 1.0
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    init = globals().get('initial_h_values')
    if init is not None:
        n = int(len(init))
        h_cur = np.array(init, dtype=float).copy()
    else:
        n = 80                     # fallback size
        target_sum = n // 2
        h_cur = random_binary_vector(n, target_sum, rng)

    target_sum = n / 2.0
    # Ensure the sum constraint (rarely needed)
    cur_sum = h_cur.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h_cur = h_cur * (target_sum / cur_sum)
        np.clip(h_cur, 0.0, 1.0, out=h_cur)

    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5

    start = time.time()
    stop = start + budget_s * 0.98          # keep a safety margin
    target_bound = 0.3808
    no_improve = 0

    while time.time() < stop:
        # annealing temperature (decreases linearly)
        remaining = stop - time.time()
        progress = 1.0 - remaining / (budget_s * 0.98)
        T = max(1e-6, (1.0 - progress) * 0.1)   # start ~0.1, cool to ~0

        # propose a mass transfer between two positions
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        hi, hj = h_cur[i], h_cur[j]

        # allowed increase/decrease so that bounds [0,1] are kept
        max_inc = min(1.0 - hi, hj)      # positive delta
        max_dec = min(hi, 1.0 - hj)      # negative magnitude

        if max_inc <= 0.0 and max_dec <= 0.0:
            continue

        # uniform delta in the feasible interval
        delta = rng.random() * (max_inc + max_dec) - max_dec
        delta = max(-max_dec, min(delta, max_inc))

        # apply move
        new_h = h_cur.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_c5 = compute_c5(new_h)

        # acceptance test (simulated annealing)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            h_cur = new_h
            cur_c5 = new_c5

        # record best solution
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = new_h.copy()
            no_improve = 0
            if best_c5 <= target_bound:
                break
        else:
            no_improve += 1

        # occasional random restart if stuck
        if no_improve >= 8000:
            h_cur = random_binary_vector(n, target_sum, rng)
            cur_c5 = compute_c5(h_cur)
            no_improve = 0

    # final sanity: enforce exact sum (tiny numeric drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # adjust a non‑boundary entry
        for k in range(n):
            if 0.0 <= best_h[k] + diff <= 1.0:
                best_h[k] += diff
                break
    np.clip(best_h, 0.0, 1.0, out=best_h)

    return (best_h.tolist(), float(best_c5), n)
