    i = rng.integers(n)

def targeted_move(h, rng):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # shift: index shift = max_idx - (n - 1)
    shift = max_idx - (n - 1)
    # compute contributions for each i s.t. i+shift in range
    contrib = np.full(n, -np.inf)
    if shift >= 0:
        # i from 0 to n - shift -1
        valid_i = np.arange(0, n - shift)
    else:
        # shift negative, i from -shift to n-1
        valid_i = np.arange(-shift, n)
    # contributions
    contrib[valid_i] = h[valid_i] * (1 - h[valid_i + shift])
    # choose source as max, destination as min among valid_i (could be any)
    src = np.argmax(contrib)
    # find candidate dest indices where h is low or product low; we can pick the min contrib among valid_i (but might be -inf if not valid)
    # To ensure dest is not same as src and can receive mass:
    # compute dest list: indices with h < 1 - 1e-12 and not src.
    mask = (h < 1 - 1e-12) & (np.arange(n) != src)
    if not np.any(mask):
        # fallback random move
        return random_move(h, rng)
    dest_candidates = np.where(mask)[0]
    dest = rng.choice(dest_candidates)
    max_delta = min(h[src], 1.0 - h[dest])
    if max_delta <= 0:
        return random_move(h, rng)
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dest] += delta
    return new_h

import numpy as np, time, random, math

def compute_c5(h):
    overlap = np.correlate(h, 1 - h, mode='full')
    return overlap.max() * (2.0 / h.size)

def random_feasible(n, rng):
    h = np.full(n, 0.5)  # sum = n/2
    # make some random variation: random moves
    moves = n * 20
    for _ in range(moves):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
    return h

def random_move(h, rng):
    n = h.size
    # Choose source with >0
    src = rng.integers(n)
    attempts = 0
    while h[src] <= 1e-12 and attempts < 10:
        src = rng.integers(n)
        attempts += 1

def targeted_move(h, rng):
    n = h.size
    # compute correlation to find worst shift directly
    # Overlap values = np.correlate(h, 1 - h, mode='full')
    corr = np.correlate(h, 1 - h, mode='full')
    # max overlap shift
    max_idx = np.argmax(corr)
    # shift = max_idx - (n - 1)
    shift = max_idx - (n - 1)
    # compute contributions for each i where i+shift in bounds
    if shift >= 0:
        indices = np.arange(0, n - shift)
    else:
        indices = np.arange(-shift, n)
    # if no valid indices (should not), fallback
    if len(indices) == 0:
        return random_move(h, rng)
    # contributions = h[i] * (1 - h[i+shift]) for valid i
    contrib = h[indices] * (1.0 - h[indices + shift])
    src = indices[np.argmax(contrib)]
    # choose destination among indices where h<1 (or any)
    # We'll choose a random index that can receive mass
    # ensure dest not src and h[dest] < 1
    dest_candidates = np.where((h < 1.0) & (np.arange(n) != src))[0]
    if dest_candidates.size == 0:
        return random_move(h, rng)
    dst = rng.choice(dest_candidates)
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0:
        return random_move(h, rng)
    delta = rng.random() * max_delta
    new = h.copy()
    new[src] -= delta
    new[dst] += delta
    return new

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return C₅ for a given vector h using the provided evaluation rule."""
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / h.size))


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a feasible h (0≤h≤1, sum(h)=n/2) by starting from 0.5 and random mass transfers."""
    h = np.full(n, 0.5, dtype=float)          # sum = n/2, already feasible
    moves = n * 5
    for _ in range(moves):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
    return h


def random_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount of mass from a positive entry to a non‑full entry."""
    n = h.size
    src = rng.integers(n)
    attempts = 0
    while h[src] <= 1e-12 and attempts < 10:
        src = rng.integers(n)
        attempts += 1
    dst = rng.integers(n)
    attempts = 0
    while h[dst] >= 1.0 - 1e-12 and attempts < 10:
        dst = rng.integers(n)
        attempts += 1
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0.0:
        return h
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h


def targeted_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the shift giving the current maximal overlap and move mass from the
    most contributing index to a random receptive index.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)                    # integer shift that maximizes the overlap

    if shift >= 0:
        idxs = np.arange(0, n - shift)
    else:
        idxs = np.arange(-shift, n)
    if idxs.size == 0:
        return random_move(h, rng)

    contrib = h[idxs] * (1.0 - h[idxs + shift])
    src = idxs[int(np.argmax(contrib))]
    # Choose any index that can receive additional mass
    mask = (h < 1.0) & (np.arange(n) != src)
    if not np.any(mask):
        return random_move(h, rng)
    dst = int(rng.choice(np.where(mask)[0]))
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0.0:
        return random_move(h, rng)
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h
