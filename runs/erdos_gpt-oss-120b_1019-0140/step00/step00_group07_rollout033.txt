def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random
    from scipy.optimize import minimize

    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine n_points
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80

    target_sum = n / 2.0
    dx = 2.0 / n

    # Helper: project onto bounds, fix sum
    def project(h):
        # clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # adjust sum to target by scaling then clipping
        cur_sum = np.sum(h)
        if cur_sum == 0:
            # set uniform
            h = np.full_like(h, target_sum / n)
            cur_sum = target_sum
        # scaling factor
        factor = target_sum / cur_sum
        h = h * factor
        # after scaling may exceed 1; iterative clip
        # Simple iterative approach: clamp and distribute excess/deficit
        # We'll use simple iteration up to few times
        for _ in range(5):
            over = h - 1.0
            under = -h
            over_mask = over > 0
            under_mask = under > 0
            excess = np.sum(over[over_mask])
            deficit = np.sum(-under[under_mask])
            if excess > 1e-12:
                # reduce excess uniformly among over entries
                h[over_mask] = 1.0
                # distribute excess to under entries proportionally
                if deficit > 1e-12:
                    h[under_mask] += excess * (-under[under_mask]) / deficit
                else:
                    # no deficit entries, just reduce uniformly across all entries
                    h = np.clip(h, 0, 1)
                    break
            elif deficit > 1e-12:
                h[under_mask] = 0.0
                # distribute deficit uniformly among over entries (which are at 1)
                if excess > 1e-12:
                    h[over_mask] -= deficit * (h[over_mask] - 1.0) / excess
                else:
                    # no over entries
                    h = np.clip(h, 0, 1)
                    break
            else:
                # within bounds
                break
        # final clip
        h = np.clip(h, 0.0, 1.0)
        # correct small sum discrepancy by adding diff to first low-value entry
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            # adjust first element not at bounds
            for i in range(len(h)):
                if (diff > 0 and h[i] < 1.0) or (diff < 0 and h[i] > 0.0):
                    h[i] += diff
                    break
        # final clip again
        h = np.clip(h, 0.0, 1.0)
        return h

    # Initial point uniform 0.5 (feasible)
    init_h = np.full(n, 0.5)
    # but ensure sum exactly target
    init_h = project(init_h)

    # optionally use initial_h_values if available
    if 'initial_h_values' in globals():
        try:
            ih = np.array(initial_h_values, dtype=float)
            # Ensure shape
            if ih.shape[0] == n:
                # normalize to satisfy sum condition
                ih = project(ih)
                init_h = ih
        except Exception:
            pass

    best_h = init_h.copy()
    best_score = evaluate_erdos_solution(best_h)

    # Optimization parameters
    beta = 80.0  # for smooth max approximation
    max_iterations_per_run = 80  # for SLSQP
    # Define objective and gradient functions (closing over n,dx,beta)
    def objective(h):
        h = np.asarray(h)
        conv = np.correlate(h, 1 - h, mode='full')
        conv_scaled = conv * dx
        # approximate max via log-sum-exp
        max_val = np.max(conv_scaled)
        # stable lse
        exp_vals = np.exp(beta * (conv_scaled - max_val))
        approx_max = max_val + np.log(np.sum(exp_vals)) / beta
        return approx_max

    def grad_obj(h):
        h = np.asarray(h)
        conv = np.correlate(h, 1 - h, mode='full')
        conv_scaled = conv * dx
        max_val = np.max(conv_scaled)
        exp_vals = np.exp(beta * (conv_scaled - max_val))
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp  # weights, length 2n-1

        N = len(h)
        sum_hk = np.zeros(N)
        # iterate over shift indices
        for idx, wi in enumerate(w):
            if wi == 0:
                continue
            k = idx - (N - 1)
            # contribution from h[j + k]
            if k >= 0:
                # indices j from 0 to N-1-k
                sum_hk[:N - k] += wi * h[k:]
            else:
                # k < 0
                sum_hk[-k:] += wi * h[:N + k]
            # contribution from h[j - k]
            # This is same as shift -k
            k2 = -k
            if k2 >= 0:
                sum_hk[:N - k2] += wi * h[k2:]
            else:
                sum_hk[-k2:] += wi * h[:N + k2]
        # gradient
        grad = dx * (1.0 - sum_hk)
        return grad

    # Equality constraint function and Jacobian
    def eq_sum(h):
        return np.sum(h) - target_sum
    def jac_eq_sum(h):
        # derivative w.r.t each h_i is 1
        return np.ones_like(h)

    # Setup constraints
    constraints = {'type': 'eq', 'fun': eq_sum, 'jac': jac_eq_sum}

    # Begin search loop
    max_total_iters = 2000  # number of restarts or iterations across loop
    # We'll loop until time budget expires
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:  # leave a little margin
            break
        # Randomize initial point near current best or uniform
        # Choose random perturbation around best_h
        perturb = np.random.uniform(-0.2, 0.2, size=n)
        h0 = best_h + perturb
        h0 = project(h0)

        # Perform local optimization
        try:
            res = minimize(objective, h0, method='SLSQP', jac=grad_obj,
                           bounds=[(0.0, 1.0)] * n,
                           constraints=constraints,
                           options={'maxiter': max_iterations_per_run, 'ftol': 1e-9, 'disp': False})
        except Exception:
            # if optimizer fails, continue
            continue

        # Get solution, project just in case
        h_opt = project(res.x)

        # Evaluate actual C5
        try:
            score = evaluate_erdos_solution(h_opt)
        except Exception:
            # fallback to computing manually
            conv = np.correlate(h_opt, 1 - h_opt, mode='full')
            score = np.max(conv) * dx

        if score < best_score:
            best_score = score
            best_h = h_opt
            # If good enough, break early
            if best_score <= 0.3808:
                break

    # Final evaluation (in case best_h not evaluated)
    final_score = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), float(final_score), n)

import time
import random
import numpy as np
from scipy.optimize import minimize

# Global variable used by equality constraint (set inside run)
_TARGET_SUM = None


def _project_bounds_and_sum(x, target):
    """Clip to [0,1] and adjust to meet the target sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    # Distribute the deficit / surplus proportionally among free variables.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = x < 1.0
            if not np.any(mask):
                break
            cap = 1.0 - x[mask]
            total_cap = np.sum(cap)
            if total_cap <= 0.0:
                break
            inc = diff * (cap / total_cap)
            inc = np.minimum(inc, cap)
            x[mask] += inc
        else:
            mask = x > 0.0
            if not np.any(mask):
                break
            cap = x[mask]
            total_cap = np.sum(cap)
            if total_cap <= 0.0:
                break
            dec = (-diff) * (cap / total_cap)
            dec = np.minimum(dec, cap)
            x[mask] -= dec
        diff = target - np.sum(x)
    # Tiny correction if still off
    diff = target - np.sum(x)
    if abs(diff) > 1e-9:
        # adjust first non‑boundary element
        for i in range(len(x)):
            if (diff > 0 and x[i] < 1.0) or (diff < 0 and x[i] > 0.0):
                x[i] += diff
                break
    return np.clip(x, 0.0, 1.0)


def _objective(h, n, dx, beta):
    """Smooth approximation of max_{k} ∫ h(1−h) using log‑sum‑exp."""
    h = np.asarray(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    conv_scaled = conv * dx
    m = np.max(conv_scaled)
    w = np.exp(beta * (conv_scaled - m))
    approx_max = m + np.log(np.sum(w)) / beta
    return approx_max


def _gradient(h, n, dx, beta):
    """Gradient of the smooth max objective."""
    h = np.asarray(h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    conv_scaled = conv * dx
    m = np.max(conv_scaled)
    exp_vals = np.exp(beta * (conv_scaled - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp                     # weights over shifts, length 2n‑1
    N = len(h)
    sum_hk = np.zeros(N)

    # Accumulate contributions of h_{j+k} and h_{j‑k}
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        k = idx - (N - 1)

        # j + k term
        if k >= 0:
            sum_hk[:N - k] += wi * h[k:]
        else:
            sum_hk[-k:] += wi * h[:N + k]

        # j - k term  (equivalent to shift –k)
        k2 = -k
        if k2 >= 0:
            sum_hk[:N - k2] += wi * h[k2:]
        else:
            sum_hk[-k2:] += wi * h[:N + k2]

    grad = dx * (1.0 - sum_hk)
    return grad


def _eq_sum_constraint(x):
    """Equality constraint: sum(x) == target."""
    return np.sum(x) - _TARGET_SUM


def _eq_sum_jac(x):
    """Jacobian of the equality constraint."""
    return np.ones_like(x)


def _c5_from_h(h):
    """Exact C5 value using the provided evaluator (fallback provided)."""
    try:
        return evaluate_erdos_solution(h)
    except Exception:
        dx = 2.0 / len(h)
        conv = np.correlate(h, 1.0 - h, mode='full')
        return np.max(conv) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with integral 1 giving a low C5."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of points
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 80
    target_sum = n / 2.0
    dx = 2.0 / n
    beta = 80.0                         # smooth‑max sharpness

    global _TARGET_SUM
    _TARGET_SUM = target_sum           # used by constraint

    # Initial feasible vector (uniform 0.5)
    best_h = _project_bounds_and_sum(np.full(n, 0.5), target_sum)
    best_score = _c5_from_h(best_h)

    # If a parent construction exists, try it first
    if 'initial_h_values' in globals():
        try:
            cand = np.array(initial_h_values, dtype=float)
            if cand.shape[0] == n:
                cand = _project_bounds_and_sum(cand, target_sum)
                cand_score = _c5_from_h(cand)
                if cand_score < best_score:
                    best_h = cand
                    best_score = cand_score
        except Exception:
            pass

    # Optimization loop
    max_iter_per_opt = 80
    while True:
        # time guard
        if time.time() - start_time > budget_s - 0.7:
            break

        # Random perturbation around current best
        perturb = np.random.uniform(-0.2, 0.2, size=n)
        h0 = _project_bounds_and_sum(best_h + perturb, target_sum)

        # Local continuous optimization
        try:
            res = minimize(
                _objective,
                h0,
                args=(n, dx, beta),
                method='SLSQP',
                jac=_gradient,
                bounds=[(0.0, 1.0)] * n,
                constraints={'type': 'eq',
                             'fun': _eq_sum_constraint,
                             'jac': _eq_sum_jac},
                options={'maxiter': max_iter_per_opt, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            continue

        h_opt = _project_bounds_and_sum(res.x, target_sum)
        score = _c5_from_h(h_opt)

        if score < best_score:
            best_score = score
            best_h = h_opt
            # early exit if we beat the target bound
            if best_score <= 0.3808:
                break

        # occasional random restart without using previous best
        if random.random() < 0.05:
            h0 = _project_bounds_and_sum(np.random.rand(n), target_sum)

    final_score = _c5_from_h(best_h)
    return (best_h.tolist(), float(final_score), n)
