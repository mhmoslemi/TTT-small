def optimize_binary(h, time_budget, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    # we might set a limit for attempts without improvement to break
    attempts_without_improve = 0
    while time.time() - start < time_budget:
        # compute correlation and worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (n - 1)
        # try a targeted swap
        i, j = find_swap(cur_h, s)
        if i is None:
            # random swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # execute swap
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c5 = evaluate_c5(new_h)
        # accept if better or probability
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            # update best
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                attempts_without_improve = 0
            else:
                attempts_without_improve = 0
        else:
            # Simulated annealing acceptance
            # Temperature decays with time
            elapsed = time.time() - start
            progress = elapsed / time_budget
            T = 0.01 * (1 - progress) + 0.0001
            if rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                cur_h = new_h
                cur_c5 = new_c5
                attempts_without_improve = 0

            cur_c5 = evaluate_c5(cur_h)
            attempts_without_improve = 0

    n = len(h)
    target = n / 2.0
    # ensure exact sum
    cur_sum = h.sum()
    if abs(cur_sum - target) > 1e-9:
        # adjust uniformly (distribute diff)
        h = h + (target - cur_sum) / n
        # clip to [0,1] and then adjust again
        h = np.clip(h, 0, 1)
        cur_sum = h.sum()
        # apply small correction to achieve exact sum
        diff = target - cur_sum
        # distribute diff across all entries not at bound
        mask = (h > 0) & (h < 1)
        if mask.any():
            correction = diff / mask.sum()
            h[mask] += correction
            h = np.clip(h, 0, 1)
    # function evaluate_c5 defined
    cur_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    while time.time() - start < time_budget:
        # choose two distinct indices i (increase) and j (decrease)
        i, j = rng.choice(n, 2, replace=False)
        # compute allowed delta
        up_limit = 1.0 - h[i]
        down_limit = h[j]
        max_delta = min(up_limit, down_limit)
        if max_delta <= 1e-8:

        j_region = np.arange(0, n - s)

    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.array(init, dtype=float)
        n = h0.shape[0]
    else:
        n = 100
        h0 = np.zeros(n, dtype=float)
        h0[:n // 2] = 1.0
        rng.shuffle(h0)
    # build starting pool
    starts = [h0]
    for _ in range(3):
        if n % 2 == 0:
            v = np.zeros(n, dtype=float)
            v[:n // 2] = 1.0
            rng.shuffle(v)
        else:
            v = np.zeros(n, dtype=float)
            v[:n // 2] = 1.0
            v[n // 2] = 0.5
            rng.shuffle(v)
        starts.append(v)
    total_time = budget_s * 0.95
    per_start = total_time / len(starts)
    best_vec, best_val = None, np.inf
    overall_start = time.time()
    for vec in starts:
        if time.time() - overall_start > budget_s * 0.95:
            break
        h_opt, c5_opt = continuous_sa(vec, per_start, rng)
        if c5_opt < best_val:
            best_vec, best_val = h_opt, c5_opt
    # final exact sum projection
    target = n / 2.0
    cur_sum = best_vec.sum()
    if abs(cur_sum - target) > 1e-9:
        best_vec += (target - cur_sum) / n
        best_vec = np.clip(best_vec, 0.0, 1.0)
        diff = target - best_vec.sum()
        if abs(diff) > 1e-12:
            mask = (best_vec > 0) & (best_vec < 1)
            if mask.any():
                best_vec[mask] += diff / mask.sum()
                best_vec = np.clip(best_vec, 0.0, 1.0)
    return best_vec.tolist(), float(best_val), int(n)
