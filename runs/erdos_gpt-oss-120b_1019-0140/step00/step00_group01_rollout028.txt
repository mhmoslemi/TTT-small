import time, numpy as np

def project_onto_bounds_and_sum(v, target_sum):
    lo = -np.max(v) - 1.0
    hi = 1 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2.0, 0, 1)

def compute_c5(h):
    n = h.shape[0]
    conv = np.convolve(h, h, mode='full')
    min_corr = conv.min()
    return 1.0 - min_corr * (2.0 / n)

def compute_autocorr_and_shift(h):
    conv = np.convolve(h, h, mode='full')
    min_idx = conv.argmin()
    min_val = conv[min_idx]
    shift = min_idx - (h.shape[0] - 1)
    return conv, min_val, shift

def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[:n-shift] += h[shift:]
            g[shift:] += h[:n-shift]
    else:
        s = -shift
        if s < n:
            g[s:] += h[:n-s]
            g[:n-s] += h[s:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    try:
        init = initial_h_values
        n_points = int(init.shape[0])
        h0 = init.astype(float).copy()
    except NameError:
        n_points = 200
        h0 = np.full(n_points, 0.5)
    target_sum = n_points / 2.0
    # ensure feasibility
    h = project_onto_bounds_and_sum(h0, target_sum)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    start = time.time()
    it = 0
    lr = 0.3
    no_improve_counter = 0
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        # compute autocorr and worst shift
        conv, min_val, shift = compute_autocorr_and_shift(h)
        # gradient
        g = gradient_for_shift(h, shift)
        # step
        step = lr * g
        # maybe add jitter occasional
        if it % 200 == 0 and it > 0:
            # small random perturbation
            jitter = (rng.random(n_points) - 0.5) * 0.02
            step += jitter
        h_new = h + step
        # project onto constraints
        h = project_onto_bounds_and_sum(h_new, target_sum)
        # evaluate
        score = compute_c5(h)
        if score < best_score - 1e-9:
            best_score = score
            best_h = h.copy()
            no_improve_counter = 0
            # maybe reduce lr to fine-tune
            lr = max(lr * 0.98, 1e-4)
        else:
            no_improve_counter += 1
            # occasionally adjust learning rate
            if no_improve_counter > 500:
                # restart from random
                rand_vec = rng.random(n_points)
                h = project_onto_bounds_and_sum(rand_vec, target_sum)
                no_improve_counter = 0
                lr = 0.3
        it += 1
        if best_score <= 0.3808:
            break
    return best_h, best_score, n_points

import time
import numpy as np

def project_onto_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x: 0<=x<=1, sum(x)=target_sum}."""
    # binary search for lambda: sum(clip(v+lam,0,1)) = target_sum
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx for the discrete vector h."""
    n = h.shape[0]
    # autocorrelation of h with itself
    autocorr = np.convolve(h, h, mode="full")
    min_corr = autocorr.min()
    # dx = 2/n, integral of h is 1, so C5 = 1 - min_corr*dx
    return 1.0 - min_corr * (2.0 / n)


def autocorr_and_worst_shift(h: np.ndarray):
    """Return full autocorrelation and the shift (in index units) where it is minimal."""
    n = h.shape[0]
    full = np.convolve(h, h, mode="full")
    idx_min = int(full.argmin())
    shift = idx_min - (n - 1)   # shift == k in h[i] * h[i+k]
    return full, shift


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the autocorrelation at a given shift with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += h[shift:]
            g[shift:] += h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += h[: n - s]
            g[: n - s] += h[s:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initial vector ----------------------------------------------------
    try:
        init = initial_h_values
        n = int(init.shape[0])

    except NameError:

        n = 200
        h = np.full(n, 0.5, dtype=float)
    # enforce feasibility ----------------------------------------------------------------
    target_sum = n / 2.0
    h = project_onto_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    start = time.time()
    it = 0
    lr = 0.6                     # initial learning rate
    no_improve = 0

    while True:
        now = time.time()
        if now - start >= budget_s:
            break

        # ------------------------------------------------------------------
        # core step: improve the worst autocorrelation shift
        _, worst_shift = autocorr_and_worst_shift(h)
        grad = gradient_for_shift(h, worst_shift)

        # add a tiny decaying stochastic component to escape plateaus
        if it % 50 == 0:
            noise = (rng.random(n) - 0.5) * (0.02 / (1 + it * 0.005))
            grad = grad + noise

        # gradient ascent on the minimal autocorrelation (i.e. descent on C5)

        else:
            no_improve += 1
            # enlarge step size a bit if stuck for many iterations
            if no_improve > 400:
                lr = min(lr * 1.05, 0.7)
                # random restart to diversify search
                rand_vec = rng.random(n)
                h = project_onto_bounds_and_sum(rand_vec, target_sum)
                no_improve = 0

        if best_score <= 0.3808:
            # good enough – stop early
            break

        it += 1

    return best_h, best_score, n
