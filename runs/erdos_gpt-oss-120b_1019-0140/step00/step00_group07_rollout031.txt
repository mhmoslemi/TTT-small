    # correlation with complement
    return (np.correlate(h, 1 - h, mode='full') * dx).max()

import numpy as np, random, time, math

def compute_c5(h, dx):
    """Maximum overlap integral for a given step vector."""
    return (np.correlate(h, 1 - h, mode='full') * dx).max()

def random_binary_vector(n):
    """Generate a length‑n binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    h[np.random.choice(n, n // 2, replace=False)] = 1.0
    return h

def anneal(start_h, dx, time_limit):
    """Simulated‑annealing style local search on binary vectors."""
    n = len(start_h)
    h = start_h.copy()
    score = compute_c5(h, dx)
    best_h = h.copy()
    best_score = score
    start = time.time()
    T0 = 0.3                     # initial temperature
    stale = 0
    max_stale = 50000            # restart after this many non‑improving steps

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break

        # pick one 1 and one 0 to swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = random.choice(ones)
        j = random.choice(zeros)

        # propose swap
        h[i], h[j] = h[j], h[i]
        new_score = compute_c5(h, dx)
        delta = new_score - score

        # linear temperature decay
        frac = elapsed / time_limit
        T = T0 * (1 - frac)

        # Metropolis acceptance
        if delta <= 0 or random.random() < math.exp(-delta / (T + 1e-12)):
            score = new_score
            if new_score < best_score:

                best_h = h.copy()
                stale = 0
            else:
