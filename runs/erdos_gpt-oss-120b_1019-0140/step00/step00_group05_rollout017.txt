def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    # full linear correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_feasible(x):
    """
    Euclidean projection of x onto the box [0,1]^n intersected with
    the hyperplane sum(x) = n/2 .
    """
    n = x.shape[0]
    target = n / 2.0
    # start by clipping to the box
    y = np.clip(x, 0.0, 1.0)

    # iterative redistribution of the excess/deficit
    for _ in range(100):
        s = y.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound can absorb/emit the diff
        free = (y > 0.0) & (y < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # All variables are at a bound; adjust the one with largest slack
            # (this situation rarely occurs)
            slack_up = 1.0 - y
            slack_down = y
            if diff > 0:
                idx = np.argmax(slack_up)
                y[idx] = min(1.0, y[idx] + diff)
            else:
                idx = np.argmax(slack_down)
                y[idx] = max(0.0, y[idx] + diff)
            break
        inc = diff / cnt
        y[free] += inc
        y = np.clip(y, 0.0, 1.0)
    # final tiny correction if needed
    s = y.sum()
    if abs(s - target) > 1e-9:
        # distribute the remaining tiny difference uniformly over non‑fixed entries
        diff = target - s
        free = (y > 0.0) & (y < 1.0)
        cnt = free.sum()
        if cnt > 0:
            y[free] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

def _simulated_annealing(start_h, rng, time_budget):
    """
    Simple SA that conserves the total mass by transferring a small amount
    between two coordinates while respecting the box constraints.
    """
    n = start_h.shape[0]
    dx = 2.0 / n
    cur = _project_to_feasible(start_h)
    cur_score = _compute_c5(cur, dx)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    T0 = 0.5  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget * 0.95:
            break
        # linear cooling
        T = T0 * max(0.0, 1.0 - elapsed / (time_budget * 0.95))
        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # propose a transfer amount; magnitude scales with temperature
        max_delta = 0.2 * (T / T0 + 0.1)   # between 0.02 and 0.2
        delta = (rng.random() - 0.5) * 2.0 * max_delta
        # adjust delta to keep both entries inside [0,1]
        d = delta
        if cur[i] - d < 0.0:
            d = cur[i]
        if cur[i] - d > 1.0:
            d = cur[i] - 1.0
        if cur[j] + d < 0.0:
            d = -cur[j]
        if cur[j] + d > 1.0:
            d = 1.0 - cur[j]
        if abs(d) < 1e-12:
            continue
        new = cur.copy()
        new[i] -= d
        new[j] += d
        # sum is preserved exactly; only box constraints may be touched
        new_score = _compute_c5(new, dx)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < math.exp(-delta_score / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new
                best_score = new_score
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a feasible vector.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: half ones, half zeros (exactly n/2 ones)
        n = 70
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    # Ensure feasibility (box + sum constraint)
    h0 = _project_to_feasible(h0)

    # ------------------------------------------------------------------
    # Optimisation time allocation.
    # ------------------------------------------------------------------
    start_total = time.time()
    remaining = budget_s - (time.time() - start_total) - 2.0
    if remaining < 0.5:
        remaining = 0.5

    # Run a single SA pass; it is cheap enough for the given budget.
    best_h, best_score = _simulated_annealing(h0, rng, remaining)

    # Final projection & scoring to guarantee constraints.
    best_h = _project_to_feasible(best_h)
    n_points = best_h.shape[0]
    final_score = _compute_c5(best_h, 2.0 / n_points)

    return best_h.tolist(), float(final_score), n_points
