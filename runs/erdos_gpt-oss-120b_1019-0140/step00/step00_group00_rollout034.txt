import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate returns cross-correlation with reversed second arg; but using 1-h.
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def initial_shape(n):
    # U-shaped cosine
    x = np.arange(n) * (2.0 / n)  # values from 0 to 2-dx
    h = 0.5 + 0.5 * np.cos(np.pi * x)  # cos(pi*x) goes 1 to -1 to 1
    # Clip subtle floating truncation
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    target = n / 2
    cur_sum = h.sum()
    diff = target - cur_sum
    # distribute diff uniformly to all non-bound entries
    # but keep inside [0,1]
    # simple approach: add diff/n to all entries then clip and iterate few times
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # after clipping, recompute sum and correct linearly on free entries
        # iterative projection
        for _ in range(5):
            cur_sum = h.sum()
            diff = target - cur_sum
            if abs(diff) < 1e-12:
                break
            free = (h > 0) & (h < 1)
            cnt = free.sum()
            if cnt == 0:
                break
            adjust = diff / cnt
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n and initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        # enforce constraints via projection if needed
        target_sum = n / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            # simple scaling
            h0 = h0 * (target_sum / cur_sum)
            # clip afterwards
            h0 = np.clip(h0, 0.0, 1.0)
            # reproject if sum changed due to clipping
            cur_sum = h0.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                # distribute remaining diff uniformly among non-bound entries
                diff = target_sum - cur_sum
                free = (h0 > 0) & (h0 < 1)
                cnt = free.sum()
                if cnt > 0:
                    h0[free] += diff / cnt
                h0 = np.clip(h0, 0.0, 1.0)
        h_cur = h0.copy()
    else:
        # choose n moderate
        n = 100
        h_cur = initial_shape(n)
    # Ensure constraints
    target_sum = n / 2.0
    # Slight numerical clip to guarantee sum
    cur_sum = h_cur.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # adjust using scaling preserving bounds
        # simple scaling factor
        scale = target_sum / cur_sum
        h_cur *= scale
        h_cur = np.clip(h_cur, 0.0, 1.0)
    dx = 2.0 / n
    # compute best
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5

    # SA parameters

    # Temperature schedule: exponential cooling based on time fraction
    while True:
