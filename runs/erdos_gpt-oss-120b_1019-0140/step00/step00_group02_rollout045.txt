import time
import random
import math
import numpy as np

# -----------------------------------------------------------
# Helper utilities
# -----------------------------------------------------------
def _target_sum(n_points):
    """Desired sum of h values (since integral must be 1)."""
    return n_points / 2.0  # because dx = 2/n_points


def _repair_h(h, target):
    """
    Project a vector onto the feasible set:
    - clip to [0,1]
    - adjust to satisfy sum == target (approximately)
    """
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling+clipping
    for _ in range(5):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > 0:
            h = h * (target / s)
        else:
            h = np.full_like(h, target / len(h))
        h = np.clip(h, 0.0, 1.0)

    # final tiny redistribution for any residual error
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 0.0) & (h < 1.0)
        free = np.where(mask)[0]
        if free.size == 0:
            free = np.arange(len(h))  # fallback: all indices
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    return h


def _c5_value(h, dx):
    """Compute the overlap maximum C5 for a given step function h."""
    # Using the problem definition: max correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _neighbor(h, step_scale=0.1):
    """Create a neighbor by random perturbations and repair."""
    n = len(h)
    h_new = h.copy()
    # Apply 1‑3 small random updates
    for _ in range(random.randint(1, 3)):
        i = random.randrange(n)
        delta = (random.random() * 2.0 - 1.0) * step_scale
        h_new[i] += delta
    # Clip and repair sum
    target = _target_sum(n)
    h_new = _repair_h(h_new, target)
    return h_new


# -----------------------------------------------------------
# Main entry point
# -----------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing the Erdős overlap constant C5.

    Returns
    -------
    h_values : list of float
        The best step function sampled at n_points points.
    c5_bound : float
        The attained C5 value for h_values.
    n_points : int
        Number of discretisation points used.
    """
    # -------------------------------------------------------
    # Setup
    # -------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    # Initial construction: use the provided parent if available
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.array(init, dtype=float)
        n_points = len(h0)
    else:
        # fallback size – small enough for fast search yet expressive
        n_points = 60
        h0 = np.random.rand(n_points)

    # Ensure feasibility of the starting point
    h0 = _repair_h(h0, _target_sum(n_points))

    dx = 2.0 / n_points

    # -------------------------------------------------------
    # Simulated annealing loop
    # -------------------------------------------------------
    best_h = h0.copy()
    best_score = _c5_value(best_h, dx)

    current_h = h0.copy()
    current_score = best_score

    start_time = time.time()
    temperature = 1.0          # initial temperature
    cooling = 0.9997           # geometric cooling factor
    min_temp = 1e-6
    no_improve = 0
    RESTART_AFTER = 3000       # steps without improvement before random restart
    TARGET = 0.3808             # early‑stop threshold

    while True:
        # time check
        if time.time() - start_time > budget_s:
            break

        # generate neighbour
        step = max(0.05, temperature)  # adapt step size with temperature
        cand_h = _neighbor(current_h, step_scale=step)
        cand_score = _c5_value(cand_h, dx)

        # acceptance rule
        delta = cand_score - current_score
        if delta < 0 or random.random() < math.exp(-delta / max(temperature, 1e-12)):
            current_h = cand_h
            current_score = cand_score
            if cand_score < best_score:
                best_h = cand_h.copy()
                best_score = cand_score
                no_improve = 0
                # early exit if we beat the target bound
                if best_score <= TARGET:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # cooling
        temperature = max(temperature * cooling, min_temp)

        # occasional random restart to escape stagnation
        if no_improve >= RESTART_AFTER:
            current_h = np.random.rand(n_points)
            current_h = _repair_h(current_h, _target_sum(n_points))
            current_score = _c5_value(current_h, dx)
            no_improve = 0

    # Final evaluation (guard against tiny numerical drift)
    final_score = _c5_value(best_h, dx)

    return best_h.tolist(), final_score, n_points
