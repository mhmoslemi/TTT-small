    best_h = h0.copy()
    best_score = evaluate(best_h)
    
    # set up parameters
    temp0 = 0.1
    # more iterations:
    it = 0
    no_improve = 0
    # While time left...
    while time.time() - start < budget_s - 0.1:
        # Temperature schedule: exponential decay
        t = max(0.001, temp0 * (1 - (time.time() - start) / budget_s))
        # random move: select two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # compute max possible delta
        # delta positive: move from j to i
        delta_pos = min(1 - best_h[i], best_h[j])
        delta_neg = min(best_h[i], 1 - best_h[j])
        # If both zero, skip
        if delta_pos + delta_neg == 0:
            continue
        # choose delta uniformly in [-delta_neg, delta_pos]
        delta = (np.random.rand() * (delta_pos + delta_neg)) - delta_neg
        # apply move to copy
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # evaluate
        score_new = evaluate(h_new)
        # acceptance
        if score_new < best_score or np.random.rand() < math.exp(-(score_new - best_score) / t):
            best_h = h_new
            if score_new < best_score:
                best_score = score_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            
        # maybe occasionally restart from initial_h_values or random feasible
        if no_improve > 5000:
            # reinitialize random feasible from best known:
            h_rand = best_h.copy()
            # apply many random pairwise moves to diversify
            for _ in range(100):
                i2, j2 = np.random.choice(n, 2, replace=False)

    target_sum = n / 2.0
    popsize = 15
    # create initial population
    pop = np.empty((popsize, n), dtype=np.float64)
    scores = np.empty(popsize, dtype=np.float64)
    # first individual: use given initial_h (or repair if needed)
    pop[0] = project_onto_constraints(initial_h.copy(), target_sum)
    scores[0] = evaluate(pop[0])
    # rest: generate random feasible vectors
    for i in range(1, popsize):
        # start from random uniform or from best plus noise
        # generate random vector and project
        rnd = np.random.rand(n)
        # scale to meet sum while staying within [0,1] via water-filling
        pop[i] = project_onto_constraints(rnd, target_sum)
        scores[i] = evaluate(pop[i])
    # track best
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # main loop
    while time.time() - start_time < budget_s - 0.1:

        F = 0.8

        for i in range(popsize):

Thus we can compute lambda via simple root-finding: binary search on range [-1, 1] maybe extended.

def project_onto_constraints(x, target):

    lo = -1.0

import time
import numpy as np

def _project_constraints(x, target, eps=1e-12):
    # Find lambda so that sum(clip(x+lambda,0,1)) = target via binary search
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_feasible(n, target):
    x = np.random.rand(n) * 2.0 - 0.5  # wide range to increase diversity
    return _project_constraints(x, target)

def _upscale(h, n_new):
    n_old = h.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project_constraints(h_new, n_new / 2.0)

def _differential_evolution(init, budget_s, start_time):
    n = init.shape[0]
    target = n / 2.0
    popsize = 15
    F = 0.8
    Cr = 0.9
    # initialise population
    pop = np.empty((popsize, n), dtype=float)
    scores = np.empty(popsize, dtype=float)
    pop[0] = _project_constraints(init.copy(), target)
    scores[0] = _evaluate(pop[0])
    for i in range(1, popsize):
        pop[i] = _random_feasible(n, target)
        scores[i] = _evaluate(pop[i])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # evolutionary loop
    while True:
        if time.time() - start_time >= budget_s - 0.05:
            break
        for i in range(popsize):
            if time.time() - start_time >= budget_s - 0.05:
                break
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            cross = np.random.rand(n) < Cr
            if not np.any(cross):
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _project_constraints(trial, target)
            trial_score = _evaluate(trial)
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_h = trial.copy()
                    best_score = trial_score
        # early stop if good enough
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # obtain initial construction
    try:
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        init = np.full(45, 0.5, dtype=float)
    n0 = init.shape[0]
    init = _project_constraints(init, n0 / 2.0)

    best_h, best_score = _differential_evolution(init, budget_s, start)
