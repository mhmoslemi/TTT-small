def random_move(h, rng, max_delta=0.5):
    N = len(h)
    i, j = rng.integers(N, size=2)
    while i == j:
        j = rng.integers(N)
    delta = (rng.random() - 0.5) * 2 * max_delta
    # adjust to ensure bounds satisfied:
    # new_h_i = h_i - delta, new_h_j = h_j + delta
    # require new_h_i in [0,1], new_h_j in [0,1]
    # cur limits:
    max_delta_i = h[i]  # can decrease at most h[i]
    max_delta_j = 1 - h[j]  # can increase at most 1 - h[j]
    # we need delta <= max_delta_i (if delta positive decrease i)

import numpy as np, time, math

def _c5_of_h(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1−h(x+k))dx  using discrete correlation."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    limit = budget_s * 0.95          # keep a safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size and initial vector.
    # ------------------------------------------------------------------
    target_N = 200                   # larger grid gives finer search
    target_sum = target_N / 2.0

    if "initial_h_values" in globals():
        # Upsample the supplied construction to the target resolution.
        old_h = np.array(initial_h_values, dtype=float)
        old_N = old_h.size
        if old_N != target_N:
            old_x = np.linspace(0.0, 2.0, old_N, endpoint=False)
            new_x = np.linspace(0.0, 2.0, target_N, endpoint=False)
            h = np.interp(new_x, old_x, old_h)
        else:
            h = old_h.copy()
        h = np.clip(h, 0.0, 1.0)
    else:
        # Random continuous start, then force the sum constraint.
        h = rng.random(target_N)
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        h += diff / target_N
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            idx = rng.integers(target_N)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    # Ensure the linear constraint exactly (tiny numerical drift may exist).
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum):
        diff = target_sum - cur_sum
        # adjust the first element within bounds
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style local optimisation.
    # ------------------------------------------------------------------
    cur_h = h.copy()
    cur_c5 = _c5_of_h(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    T0 = 0.02               # initial temperature
    decay = 0.9997
    it = 0
    max_iter = 10_000_000

    N = target_N
    while time.time() - start < limit and it < max_iter:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1

        # direction: move mass from i to j if possible, otherwise opposite
        if cur_h[i] > 0.0 and cur_h[j] < 1.0:
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
        elif cur_h[j] > 0.0 and cur_h[i] < 1.0:
            i, j = j, i
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
        else:
            continue

        if max_delta <= 0.0:
            continue

        # random amount to transfer (full range encourages both fine and coarse steps)
        delta = rng.random() * max_delta
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = _c5_of_h(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

    # ------------------------------------------------------------------
    # 3. Final repair (tiny numerical errors) and return.
    # ------------------------------------------------------------------
    sum_diff = target_sum - best_h.sum()
    if abs(sum_diff) > 1e-12:
        best_h[0] = np.clip(best_h[0] + sum_diff, 0.0, 1.0)

    best_c5 = _c5_of_h(best_h)
    return (best_h.tolist(), best_c5, N)
