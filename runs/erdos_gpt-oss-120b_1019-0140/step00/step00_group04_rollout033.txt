import time
import random
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a given step function sampled as vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h) using full mode (zero padding)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a binary step function h on [0,2] that minimizes the
    Erdős overlap constant C5.  Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    best_h = None           # best vector found (numpy array)
    best_score = float("inf")
    best_n = 0

    # ------------------------------------------------------------------
    # 1.  Use the parent construction if it exists (initial_h_values)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        # enforce box constraints
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce exact integral sum = n/2 by a tiny correction on the first entry
        target = n0 / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            h0[0] = min(1.0, max(0.0, h0[0] + diff))
        score0 = _evaluate(h0)
        best_h, best_score, best_n = h0, score0, n0

    # ------------------------------------------------------------------
    # 2.  Core stochastic search (binary vectors, hill‑climbing)
    # ------------------------------------------------------------------
    # a list of problem sizes – larger n gives finer granularity
    n_list = [84, 126, 168, 200]          # all even, modest <1 k
    # parameters for the local search
    max_restarts = 6
    max_iters_per_restart = 20000
    target_bound = 0.3808                # we aim to get below this

    for n in n_list:
        if time.time() - start > budget_s * 0.90:
            break
        target_sum = n // 2
        dx = 2.0 / n

        rng = np.random.RandomState(seed + n)

        for _ in range(max_restarts):
            # ---- random feasible start (binary, exactly n/2 ones) ----
            h = np.zeros(n, dtype=float)
            ones_idx = rng.choice(n, target_sum, replace=False)
            h[ones_idx] = 1.0
            cur_score = _evaluate(h)

            # keep globally best if this start is already good
            if cur_score < best_score:
                best_h, best_score, best_n = h.copy(), cur_score, n

            # ---- maintain explicit index lists for fast swaps ----
            ones = list(np.flatnonzero(h == 1.0))
            zeros = list(np.flatnonzero(h == 0.0))

            # ---- simple simulated‑annealing hill climb ----
            it = 0
            while True:
                now = time.time()
                if now - start > budget_s * 0.95:
                    break
                if it >= max_iters_per_restart:
                    break

                # temperature decreases linearly to zero
                T = 0.2 * (1.0 - it / max_iters_per_restart)

                # propose swapping a random 1 with a random 0
                i = random.choice(ones)
                j = random.choice(zeros)

                # perform swap
                h[i], h[j] = 0.0, 1.0
                # update index containers
                ones.remove(i)
                zeros.append(i)
                zeros.remove(j)
                ones.append(j)

                new_score = _evaluate(h)
                delta = new_score - cur_score
                accept = False
                if delta < 0:
                    accept = True
                elif T > 1e-12:
                    # Metropolis criterion
                    if random.random() < np.exp(-delta / T):
                        accept = True

                if accept:
                    cur_score = new_score
                    if new_score < best_score:
                        best_h = h.copy()
                        best_score = new_score
                        best_n = n
                    # early exit if we beat the target bound
                    if best_score <= target_bound:
                        break
                else:
                    # revert swap
                    h[i], h[j] = 1.0, 0.0
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    ones.remove(j)

                it += 1

            if best_score <= target_bound:
                break
        if best_score <= target_bound:
            break

    # ------------------------------------------------------------------
    # 3.  Final sanity correction (ensure exact sum and box constraints)
    # ------------------------------------------------------------------
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        n = best_n
        target = n / 2.0
        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            # adjust the first element; diff is tiny so bounds stay satisfied
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        final_score = _evaluate(best_h)
    else:
        best_h = np.array([], dtype=float)
        final_score = float("inf")
        n = 0

    return (best_h.tolist(), final_score, n)
