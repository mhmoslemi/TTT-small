def optimize_h(initial_h, budget_s):
    n = len(initial_h)
    S = n / 2.0
    h = initial_h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    start_time = time.time()
    step = 0.5  # initial gradient step size
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:  # leave some buffer
            break
        # compute correlation and worst shift
        conv = np.correlate(h, 1.0 - h, mode='full')
        kstar = int(np.argmax(conv) - (n - 1))
        # compute gradient
        shift_plus = shift_array(h, kstar)
        shift_minus = shift_array(h, -kstar)
        grad = 1.0 - shift_plus - shift_minus
        # take gradient step
        v = h - step * grad
        # projection
        v = project_onto_box_simplex(v, S)
        # compute new score
        new_score = np.max(np.correlate(v, 1.0 - v, mode='full')) * (2.0 / n)
        # Accept if improvement else maybe reduce step
        if new_score < best_score - 1e-12:
            best_h = v.copy()
            best_score = new_score
            # optionally increase step
            step = min(step * 1.05, 1.0)
        else:
            # maybe reduce step if not improving
            step = max(step * 0.95, 1e-4)
        # set h for next iteration to v (even if not improved) to continue exploring.
        h = v
        iteration += 1
        # occasional random perturbation to escape local minima
        if iteration % 100 == 0:
            # apply random noise with small amplitude
            noise = np.random.uniform(-0.1, 0.1, size=n)
            v2 = h + noise
            v2 = np.clip(v2, 0.0, 1.0)
            # adjust to sum S via projection
            v2 = project_onto_box_simplex(v2, S)
            # compute and maybe accept if better
            score2 = np.max(np.correlate(v2, 1.0 - v2, mode='full')) * (2.0 / n)
            if score2 < best_score:
                best_h = v2.copy()
                best_score = score2
            h = v2
    return best_h, best_score

import numpy as np
import time

def _project_onto_box_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Projection of v onto the set {w | 0<=w_i<=1, sum w_i = S}."""
    # Initial bounds for lambda
    lo = np.min(v - 1.0)          # makes all entries clip to 1 (sum = n)
    hi = np.max(v)                # makes all entries clip to 0 (sum = 0)
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        total = w.sum()
        if total > S:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _shift_array(v: np.ndarray, shift: int) -> np.ndarray:
    """Zero‑padded shift: result[i] = v[i+shift] if index in range, else 0."""
    n = v.shape[0]
    if shift == 0:
        return v.copy()
    res = np.empty_like(v)
    if shift > 0:
        res[: n - shift] = v[shift:]
        res[n - shift :] = 0.0
    else:
        k = -shift
        res[k:] = v[: n - k]
        res[:k] = 0.0
    return res

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)

def _gradient_worst_shift(h: np.ndarray) -> tuple:
    """
    Return (k_star, gradient) where k_star is the shift attaining the
    maximal overlap and gradient is the sub‑gradient of that term.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(conv) - (n - 1))
    shift_plus = _shift_array(h, k_star)      # h[i+k_star] (zero‑padded)
    shift_minus = _shift_array(h, -k_star)    # h[i‑k_star] (zero‑padded)
    grad = 1.0 - shift_plus - shift_minus
    return k_star, grad

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a list), its C5 bound, and the number of points.
    The construction uses a simple linear ramp, which analytically gives C5≈1/3.
    A short projected‑gradient refinement is performed within the allotted time.
    """
    np.random.seed(seed)

    # Determine number of points (use provided initial_h_values if present)
    try:
        n = int(initial_h_values.shape[0])  # type: ignore[name-defined]
    except Exception:
        n = 80  # fallback size

    # Linear ramp: values from 0 to 1 inclusive
    h = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Ensure the exact sum constraint (sum h must be n/2)
    S = n / 2.0
    total = h.sum()
    if abs(total - S) > 1e-12:
        h = _project_onto_box_simplex(h, S)

    best_h = h.copy()
    best_score = _c5_value(best_h)

    # Simple projected‑gradient refinement (time‑bounded)
    start = time.time()
    step = 0.5
    dx = 2.0 / n
    while time.time() - start < budget_s * 0.9:
        # gradient of the worst‑shift term
        _, grad = _gradient_worst_shift(best_h)
        candidate = best_h - step * grad
        candidate = _project_onto_box_simplex(candidate, S)
        cand_score = _c5_value(candidate)

        if cand_score < best_score - 1e-12:
            best_h, best_score = candidate, cand_score
            step = min(step * 1.05, 1.0)   # accelerate when improving
        else:
            step = max(step * 0.95, 1e-5)   # shrink otherwise

            pert = best_h + noise
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_onto_box_simplex(pert, S)
            pert_score = _c5_value(pert)
            if pert_score < best_score:
                best_h, best_score = pert, pert_score

    # Return as plain Python list for compatibility
    return (best_h.tolist(), float(best_score), int(n))
