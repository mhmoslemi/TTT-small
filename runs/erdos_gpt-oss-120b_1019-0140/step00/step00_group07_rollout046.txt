import numpy as np
import time
import math

def _compute_c5(h):
    """Return the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # convolution h * (1-h) over all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def _to_binary(arr, target_sum):
    """Project an arbitrary float array to a binary (0/1) vector with exact sum."""
    n = arr.shape[0]
    # pick the indices of the largest `target_sum` entries
    idx = np.argpartition(arr, -target_sum)[-target_sum:]
    bin_arr = np.zeros(n, dtype=int)
    bin_arr[idx] = 1
    return bin_arr

def _random_binary(n, target_sum):
    """Generate a random binary vector of length n with exact target_sum ones."""
    arr = np.zeros(n, dtype=int)
    arr[:target_sum] = 1
    np.random.shuffle(arr)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős C5 bound. Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, copy=True)
        n = int(init.shape[0])
        h = init.astype(float)
    except Exception:
        # fallback: use 80 points (the size mentioned in the description)
        n = 80
        h = np.zeros(n, dtype=float)

    target_sum = n // 2  # because integral of h must be 1 and dx = 2/n

    # ensure binary feasibility and exact sum
    if not np.all((h == 0) | (h == 1)):
        h = _to_binary(h, target_sum).astype(float)
    else:
        current = int(np.rint(np.sum(h)))
        if current != target_sum:
            # adjust by random flips
            diff = current - target_sum
            if diff > 0:
                ones = np.flatnonzero(h == 1)
                drop = np.random.choice(ones, size=diff, replace=False)
                h[drop] = 0
            else:
                zeros = np.flatnonzero(h == 0)
                add = np.random.choice(zeros, size=-diff, replace=False)
                h[add] = 1

    # work with integer array for cheap swaps
    h_int = h.astype(int)

    cur_val = _compute_c5(h_int)
    best_h = h_int.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # simulated annealing / hill climbing
    # ------------------------------------------------------------------
    T0 = 0.01                  # initial temperature
    no_improve = 0
    RESTART_AFTER = 20000      # iterations without improvement → random restart
    max_iter = 10**9           # effectively unlimited; loop ends by time

    for _ in range(max_iter):
        # time check (break before hitting the hard limit)
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.2:
            break

        # pick a random 1 and a random 0 to swap
        ones_idx = np.flatnonzero(h_int == 1)
        zeros_idx = np.flatnonzero(h_int == 0)

        if ones_idx.size == 0 or zeros_idx.size == 0:
            break  # degenerate state, shouldn't happen

        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # perform swap
        h_int[i] = 0
        h_int[j] = 1

        new_val = _compute_c5(h_int)
        delta = new_val - cur_val

        # temperature linearly decays to 0
        remaining = budget_s - elapsed
        T = T0 * (remaining / budget_s)

        accept = False
        if delta < 0:
            accept = True
        else:
            # uphill move with Metropolis probability
            if T > 0:
                prob = math.exp(-delta / T)
                if np.random.rand() < prob:
                    accept = True

            h_int[j] = 0

    except Exception:
        final_c5 = _compute_c5(best_h_float)

    return (best_h_float, final_c5, n)
