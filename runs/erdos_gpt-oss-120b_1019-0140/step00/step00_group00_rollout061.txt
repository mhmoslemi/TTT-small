def compute_c5_fft(h):
    n = len(h)
    # note: correlation = np.convolve(h, np.flip(1-h), mode='full')
    # Use FFT with zero padding to length L = 2^ceil(log2(2*n-1))
    m = 2*n - 1
    L = 1 << (m-1).bit_length()
    # compute FFT
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(np.flip(1-h), n=L)
    conv = np.fft.irfft(H * G, n=L)
    # extract first m entries
    conv = conv[:m]
    c5 = (2.0 / n) * conv.max()
    return float(c5)

import numpy as np, time

def project_box_sum(v, target):
    # iterative projection onto [0,1] box and sum target
    y = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        total = y.sum()
        diff = total - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = y > 0
        else:
            free = y < 1
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        delta = diff / cnt
        y[free] -= delta
        y = np.clip(y, 0.0, 1.0)
    # final small correction
    total = y.sum()
    diff = total - target
    if abs(diff) > 1e-9:
        # adjust the first element not at bound
        flexible = (y > 0) & (y < 1)
        if np.any(flexible):
            idx = np.where(flexible)[0][0]
            y[idx] -= diff
        else:
            # all at bound, adjust first element within bounds
            y[0] = min(max(y[0] - diff, 0.0), 1.0)
    return y

import numpy as np
import time

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the intersection of the box [0,1]^n and the hyperplane sum=v."""
    y = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        total = y.sum()
        diff = total - target
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            free = y > 0.0
        else:
            free = y < 1.0
        cnt = int(free.sum())
        if cnt == 0:
            break
        delta = diff / cnt
        y[free] -= delta
        y = np.clip(y, 0.0, 1.0)
    # final tiny correction (should be extremely small)
    total = y.sum()
    diff = total - target
    if abs(diff) > 1e-9:
        flexible = (y > 0.0) & (y < 1.0)
        if flexible.any():
            idx = np.where(flexible)[0][0]
            y[idx] -= diff
        else:
            # all at bounds – adjust the first element safely
            y[0] = min(max(y[0] - diff, 0.0), 1.0)
    return y


def _corr_full_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1 - h) using FFT."""
    n = h.shape[0]
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(np.flip(1.0 - h), n=L)
    conv = np.fft.irfft(H * G, n=L)[:m]
    conv[conv < 0] = 0.0          # numerical safety
    return conv


def _c5_value(h: np.ndarray) -> float:
    """Return C₅ for a given step vector."""
    n = h.shape[0]
    conv = _corr_full_fft(h)
    return (2.0 / n) * float(conv.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation term that attains the maximal value."""
    n = h.shape[0]
    g = np.ones_like(h)
    if shift >= 0:
        if shift < n:
            g[:n - shift] -= h[shift:]
            g[shift:] -= h[:n - shift]
    else:
        t = -shift
        g[t:] -= h[:n - t]
        g[:n - t] -= h[t:]
    return g


def _upscale_initial(iv: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a parent construction to a new resolution."""
    n0 = iv.shape[0]
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, iv)


def _subgradient_descent(start_h: np.ndarray, n: int, target: float,
                         start_time: float, budget_s: float,
                         max_iters: int = 20000):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = _project_box_sum(start_h.copy(), target)
    best_h = h.copy()
    best_val = _c5_value(h)

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.97:
            break

        conv = _corr_full_fft(h)
        idx = int(np.argmax(conv))
        cur_val = (2.0 / n) * float(conv[idx])
        if cur_val < best_val:
            best_val, best_h = cur_val, h.copy()

        shift = idx - (n - 1)                     # map index to signed shift
        grad = _gradient_for_shift(h, shift)
        step = 0.5 / np.sqrt(it + 1.0)            # diminishing step size
        h = _project_box_sum(h - step * grad, target)

    return best_h, best_val
