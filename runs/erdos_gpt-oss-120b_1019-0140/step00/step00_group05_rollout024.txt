import time

def _repair_h(vec):
    """Project a vector onto the box [0,1] and the hyperplane sum=target."""
    v = np.clip(vec, 0.0, 1.0)
    N = v.size
    target = N / 2.0
    s = v.sum()
    # Simple redistribution of excess/deficit
    if abs(s - target) > 1e-12:
        diff = target - s
        # Indices that can move in the required direction
        if diff > 0:
            idx = np.where(v < 1.0 - 1e-12)[0]
        else:
            idx = np.where(v > 1e-12)[0]
        if idx.size == 0:
            # fallback uniform scaling
            v *= target / s
        else:
            # Distribute uniformly (might need a few passes)
            while abs(diff) > 1e-12 and idx.size > 0:
                inc = diff / idx.size
                # apply limited by bounds
                if diff > 0:
                    allowed = np.minimum(inc, 1.0 - v[idx])
                else:
                    allowed = np.maximum(inc, -v[idx])
                v[idx] += allowed
                diff = target - v.sum()
                idx = np.where(v < 1.0 - 1e-12)[0] if diff > 0 else np.where(v > 1e-12)[0]
    return np.clip(v, 0.0, 1.0)


def _random_feasible(N):

        high = min(1.0, remaining)
        val = np.random.rand() * (high - low) + low
        h[i] = val
        remaining -= val

def _targeted_move(h):
    """
    Increase the pair of points that belong to the shift
    with the smallest autocorrelation, taking mass from two donors.
    """
    N = h.size
    # autocorrelation of h with itself (zero‑padded)
    corr = np.correlate(h, h, mode='full')
    idx_min = int(np.argmin(corr))
    shift = idx_min - (N - 1)  # can be negative

    if shift >= 0:
        i_cands = np.arange(N - shift)
        j_cands = i_cands + shift
    else:
        i_cands = np.arange(-shift, N)
        j_cands = i_cands + shift

    # feasible increments (both places can accept more mass)
    inc_limits = np.minimum(1.0 - h[i_cands], 1.0 - h[j_cands])
    feasible = inc_limits > 1e-9
    if not np.any(feasible):
        return _random_move(h)

    # pick one feasible pair randomly
    feasible_idx = np.where(feasible)[0]
    pick = np.random.choice(feasible_idx)
    i = i_cands[pick]
    j = j_cands[pick]
    max_inc = inc_limits[pick]

    # choose two donor indices (different from i and j)
    donor_idx = np.where(h > 1e-9)[0]
    donor_idx = donor_idx[(donor_idx != i) & (donor_idx != j)]
    if donor_idx.size < 2:
        return _random_move(h)
    d1, d2 = np.random.choice(donor_idx, size=2, replace=False)

    # amount we can steal from donors
    take1 = h[d1]
    take2 = h[d2]
    inc = min(max_inc, take1, take2)
    if inc <= 0.0:
        return _random_move(h)

    cand = h.copy()
    cand[i] += inc
    cand[j] += inc
    cand[d1] -= inc
    cand[d2] -= inc
    return _repair_h(cand)

            candidate = _targeted_move(best_h)
        else:
            candidate = _random_move(best_h)

        cand_score = float(evaluate_erdos_solution(candidate))

        if cand_score < best_score - 1e-12:
            best_h = candidate
            best_score = cand_score
            iter_since_improve = 0
            if best_score <= target_bound:
                break
        else:
            # occasional uphill acceptance
            if np.random.rand() < accept_prob:
                best_h = candidate
                best_score = cand_score
            iter_since_improve += 1

        # Restart if stagnating
        if iter_since_improve >= max_iter_no_improve:
            # replace current best with a fresh random feasible vector
            best_h = _repair_h(_random_feasible(N))
            best_score = float(evaluate_erdos_solution(best_h))
            iter_since_improve = 0

    # Final safety repair
    best_h = _repair_h(best_h)
    final_score = float(evaluate_erdos_solution(best_h))
    return best_h.tolist(), final_score, N
