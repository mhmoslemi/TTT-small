def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search(h, rng, time_allow):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    # We'll attempt many moves.
    while time.time() - start < time_allow:
        # propose move
        # sample two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # Random direction: try moving mass from i to j or j to i
        # Determine max transfer possible both ways
        # Option 1: i->j
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        # Option 2: j->i
        max_transfer2 = min(best_h[j], 1.0 - best_h[i])
        # Choose direction with positive max_transfer; if both zero continue
        # Randomly pick direction weighted by max_transfer
        if max_transfer <= 0 and max_transfer2 <= 0:
            continue
        # Choose direction probabilistically
        direction = 0
        if max_transfer > 0 and max_transfer2 > 0:
            if rng.random() < max_transfer / (max_transfer + max_transfer2):
                direction = 0
            else:
                direction = 1
        elif max_transfer > 0:
            direction = 0
        else:
            direction = 1
        if direction == 0:
            delta = rng.random() * max_transfer
            if delta == 0:
                continue
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            delta = rng.random() * max_transfer2
            if delta == 0:
                continue
            new_h = best_h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        # Evaluate
        new_c5 = compute_c5(new_h)
        if new_c5 + 1e-12 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    return best_h, best_c5

import time
import numpy as np

def _compute_c5(h):
    """Return max shift overlap C5 for step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    # length for linear correlation without wrap‑around
    L = 1 << int(np.ceil(np.log2(2 * n)))
    # FFT based correlation: corr = ifft(fft(h) * conj(fft(1-h)))
    a = np.fft.rfft(h, n=L)
    b = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(a * np.conj(b), n=L)
    # linear correlation has length 2*n-1
    corr = conv[: 2 * n - 1]
    return np.max(corr) * dx

def _upscale(h_old, n_new):
    """Linear interpolation of a step function to a finer grid."""
    n_old = h_old.shape[0]
    pos_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    pos_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(pos_new, pos_old, h_old, left=h_old[0], right=h_old[-1])

def _repair(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur == 0:
        # fallback uniform fill
        h = np.full_like(h, target_sum / h.shape[0])
        return h
    factor = target_sum / cur
    h = np.clip(h * factor, 0.0, 1.0)
    # Minor correction if clipping disturbed the sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def _local_search(h_start, rng, time_allow):
    """Hill‑climbing with mass‑transfer moves, bounded by a time budget."""
    start = time.time()
    best_h = h_start.copy()
    best_c5 = _compute_c5(best_h)
    n = best_h.shape[0]

    while time.time() - start < time_allow:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # possible transfer directions
        max_ij = min(best_h[i], 1.0 - best_h[j])   # i -> j
        max_ji = min(best_h[j], 1.0 - best_h[i])   # j -> i

        if max_ij <= 0 and max_ji <= 0:
            continue

        # choose direction proportionally to feasible range
        if max_ij > 0 and max_ji > 0:
            direction = 0 if rng.random() < max_ij / (max_ij + max_ji) else 1
        elif max_ij > 0:
            direction = 0
        else:
            direction = 1

        if direction == 0:
            delta = rng.random() * max_ij
            if delta == 0.0:
                continue
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            delta = rng.random() * max_ji
            if delta == 0.0:
                continue
            new_h = best_h.copy()
            new_h[j] -= delta
            new_h[i] += delta

        new_c5 = _compute_c5(new_h)
        if new_c5 + 1e-12 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    return best_h, best_c5
