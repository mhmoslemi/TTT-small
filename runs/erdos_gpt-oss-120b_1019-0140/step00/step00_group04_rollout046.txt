import time
import math
import numpy as np

def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact target sum."""
    v = np.clip(v, 0.0, 1.0)
    # Simple water-filling loop
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # No free variables, push the smallest change onto a boundary variable
            if diff > 0:
                idx = np.argmax(v < 1.0)
                v[idx] = min(1.0, v[idx] + diff)
            else:
                idx = np.argmax(v > 0.0)
                v[idx] = max(0.0, v[idx] + diff)
            break
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point in the feasible polytope."""
    # start from a Dirichlet-like random vector
    a = rng.random(n)
    v = a / a.sum() * target
    v = _project_to_feasible(v, target)
    return v

def _c5_cost(h: np.ndarray, dx: float) -> float:
    """Compute the C5 value for a given discretised function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
