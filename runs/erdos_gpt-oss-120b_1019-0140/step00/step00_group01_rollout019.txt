def project_to_bounds_sum(x, target_sum):
    # find lambda such that sum(clipped(x + lambda)) = target_sum
    # using binary search.
    n = x.shape[0]
    # computing lower and upper bounds for lambda
    # lower: ensure all become 0. The sum at lower bound = 0, which is <= target
    # So lambda_low = -max(x) ensures x+lambda <= 0 for all -> sum 0.
    # Upper: all become 1. So lambda_up = 1 - min(x) ensures x+lambda >=1 for all => sum n.
    lo = -np.max(x)
    hi = 1 - np.min(x)
    # binary search for approx lambda to match target_sum
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    # final lambda:
    lam = (lo + hi) / 2.0
    y = np.clip(x + lam, 0.0, 1.0)
    # Might still be slight off due to numerical tolerance; distribute remaining diff.
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        # adjust only on free variables (not at 0 or 1)

        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

    # retrieve initial if exists
    try:
        init_h = initial_h_values
    except NameError:
        init_h = None

    # Choose number of points
    if init_h is not None:
        n_points = len(init_h)
        # maybe upsample if too coarse? For now keep original.
        # Optionally upsample to 200 if initial present to get more degrees.
        # We could check n_points < 200: upsample.
        if n_points < 150:
            target_n = 200
            # upscale using interpolation
            x_old = np.linspace(0, 2, n_points, endpoint=False)
            x_new = np.linspace(0, 2, target_n, endpoint=False)
            init_h_up = np.interp(x_new, x_old, init_h)
            init_h = init_h_up
            n_points = target_n
    else:
        n_points = 200
        init_h = None

    target_sum = n_points / 2.0

    # Define projection
    def project(x):
        # binary search lambda to satisfy sum constraint with 0-1 bounds
        lo = -np.max(x)
        hi = 1.0 - np.min(x)
        for _ in range(50):
            mid = (lo + hi) * 0.5
            y = np.clip(x + mid, 0.0, 1.0)
            s = y.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
        # adjust tiny residual
        diff = target_sum - y.sum()
        if abs(diff) > 1e-12:
            free = (y > 1e-12) & (y < 1.0 - 1e-12)
            if free.any():
                y[free] += diff / free.sum()
                y = np.clip(y, 0.0, 1.0)
        return y

    # objective function
    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = corr.max() * (2.0 / n_points)
        return val

    # Keep track of best
    best_h = None
    best_val = np.inf

    # wrapper to update best
    def obj_wrapped(x):
        nonlocal best_h, best_val
        h = project(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = corr.max() * (2.0 / n_points)
        if val < best_val:
            best_val = val
            best_h = h.copy()

        init_pop[0] = np.clip(init_h, 0.0, 1.0)
        init = init_pop
    else:
        init = 'random'

        obj_wrapped,
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        mutation=(0.5, 1.0),

    if best_h is None:
        best_h = project(result.x)

        val = corr.max() * (2.0 / n_points)

            best_val = val
            best_h = cand

    final_corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    final_c5 = final_corr.max() * (2.0 / n_points)

    return best_h.tolist(), float(final_c5), n_points
