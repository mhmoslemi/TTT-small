def project_sum_bounds(h, target_sum, max_iter=10):
    # iterative projection onto hyperplane and bound constraints
    for _ in range(max_iter):
        # enforce sum
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        # clip
        np.clip(h, 0.0, 1.0, out=h)
    # final final correction: compute exact residue and adjust within bounds if possible
    # compute residual after clipping
    current = h.sum()
    residual = target_sum - current
    if abs(residual) > 1e-12:
        # adjust free variables (not at bounds) equally
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free > 0:
            h[free] += residual / n_free

def adjust_sum(h, target):
    # ensure sum equals target within tolerance
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) < 1e-12:
        return h
    # compute free indices
    free = (h > 0) & (h < 1)
    n_free = free.sum()
    if n_free > 0:
        # distribute equally
        h[free] += diff / n_free
        # clip after distribution
        np.clip(h, 0, 1, out=h)
        # if still not enough due to clipping, repeat recursively
        return adjust_sum(h, target)
    else:
        # all at bounds: can't adjust; force small tolerance by distributing to all entries
        h += diff / len(h)
        np.clip(h, 0, 1, out=h)
        # re-evaluate, maybe not exact but close
        return h
```

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h and (1-h) over full overlap
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def propose_move(h: np.ndarray, rng: np.random.Generator, step: float) -> np.ndarray | None:
    """Return a new h where the sum is unchanged and bounds are respected.
    Two indices are selected, a delta is transferred from one to the other.
    The magnitude of delta is limited by `step`."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # allowable delta to stay in [0,1] for both entries
    lower = max(-hi, -(1.0 - hj))
    upper = min(1.0 - hi, hj)

    # respect step magnitude
    lower = max(lower, -step)
    upper = min(upper, step)

    if lower >= upper:
        return None
    delta = rng.uniform(lower, upper)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new


def _adjust_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Iteratively project onto the hyperplane sum(h)=target while staying in [0,1]."""
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute uniformly
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    # final clean‑up: distribute any remaining residue over free entries
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        n_free = int(free.sum())
        if n_free > 0:
            h[free] += diff / n_free
            np.clip(h, 0.0, 1.0, out=h)
    return h

    else:
