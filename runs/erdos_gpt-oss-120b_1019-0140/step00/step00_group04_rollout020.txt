import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """
    Compute the Erdős C5 constant for a given step function sampled vector h.
    The integral is approximated by discrete correlation with dx = 2/n.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Construct a simple monotone step function that yields a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    # Determine number of points: use the length of the provided initial solution
    # if it exists; otherwise fall back to a modest default.
    try:
        n = int(initial_h_values.shape[0])
        if n < 2:
            n = 200
    except Exception:
        n = 200

    # Build a linear ramp: h_i = (i + 0.5) / n  for i = 0 .. n-1
    # This satisfies 0 < h_i < 1 and sum(h) = n/2 exactly.
    h = (np.arange(n, dtype=np.float64) + 0.5) / n

    # Minor numerical correction to enforce the integral exactly (optional)
    # The exact target sum is n/2; if floating error occurs, rescale uniformly.
    target_sum = n / 2.0
    current_sum = h.sum()
    if not np.isclose(current_sum, target_sum, rtol=0, atol=1e-12):
        h *= target_sum / current_sum

    c5 = _compute_c5(h)

    return h, c5, n
