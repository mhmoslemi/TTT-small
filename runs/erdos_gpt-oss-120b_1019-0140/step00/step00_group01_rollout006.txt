def project_box_simplex(x, target, tol=1e-9, max_iter=100):
    low = np.min(x) - 1.0
    high = np.max(x)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low+high)/2.0, 0.0, 1.0)

            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)
