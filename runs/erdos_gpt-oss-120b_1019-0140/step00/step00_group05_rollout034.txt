    # want x = clip(v + lambda, 0,1) s.t sum x = target
    # binary search for lambda, tolerance 1e-9 maybe

def project_to_bounds_and_sum(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Solve for lambda using binary search
    # it's monotonic: sum(clip(v + lambda,0,1)) increasing in lambda.
    low = -np.max(v) - 1.0  # ensures all v+low <= 0 => sum 0
    high = 1.0 - np.min(v) + 1.0  # ensures all v+high >= 1 => sum = len(v)
    # limit to tolerance
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    # final projection
    x = np.clip(v + (low + high) / 2.0, 0.0, 1.0)
    # Due to numerical error, adjust small diff
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # adjust free entries within [0,1] proportionally
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)  # clip again
    return x

def _project_to_bounds_and_sum(v, target):
    # Clip initially
    v = np.clip(v, 0.0, 1.0)
    # Binary search for lambda
    low = -np.max(v) - 1.0   # ensures all v+low <=0 -> sum 0
    high = 1.0 - np.min(v) + 1.0   # ensures v+high >=1 -> sum = len(v)

        x = np.clip(v + mid, 0.0, 1.0)

        else:

    diff = target - x.sum()

            cur_h = _project_to_bounds_and_sum(np.clip(new_h, 0.0, 1.0), target_sum)
            best_h = cur_h.copy()
            best_score = _evaluate(best_h)
            cur_score = best_score
            iteration = 0
            upsampled = True

    # ----- final evaluation (use provided helper if available) -----
    try:
        final_score = evaluate_erdos_solution(best_h)
    except Exception:
        final_score = best_score

    return best_h.tolist(), float(final_score), best_h.shape[0]
