import numpy as np, time, math, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with 1-h
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def hill_climb(h, budget, start_time):
    n = h.shape[0]; dx = 2.0/n
    best_h = h.copy()
    best_score = compute_c5(h)
    current_h = h.copy()
    current_score = best_score
    # main iteration
    while time.time() - start_time < budget:
        # compute worst shift index
        corr = np.correlate(current_h, 1.0 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n - 1) # shift relative
        # attempt targeted swap for this shift
        # compute partner array
        indices = np.arange(n) + max_shift
        valid = (indices >= 0) & (indices < n)
        partner = np.zeros(n, dtype=current_h.dtype)
        partner[valid] = current_h[indices[valid]]
        # find candidates
        cand_i = np.where((current_h == 1) & (partner == 0))[0]
        cand_j = np.where((current_h == 0) & (partner == 1))[0]
        performed = False
        if cand_i.size > 0 and cand_j.size > 0:
            i = np.random.choice(cand_i)
            j = np.random.choice(cand_j)
            # swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
            new_score = compute_c5(current_h)
            if new_score <= current_score:
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = current_h.copy()
                performed = True
            else:
                # revert

import random

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a binary (or [0,1]) step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def hill_climb(initial_h: np.ndarray, time_budget: float, start_time: float):
    """Hill‑climb with random restarts to minimise C5."""
    h = initial_h.copy()
    n = h.shape[0]

    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score

    no_improve = 0
    RESTART_LIMIT = 5000

    while time.time() - start_time < time_budget - 0.01:
        # restart if stuck too long
        if no_improve >= RESTART_LIMIT:
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            np.random.shuffle(h)
            cur_score = compute_c5(h)
            no_improve = 0
            continue

        # worst shift for the current configuration
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr_full))
        shift = max_idx - (n - 1)  # shift can be negative

        # --- try a targeted swap that directly improves the worst shift ---
        idx = np.arange(n) + shift
        valid = (idx >= 0) & (idx < n)
        partner = np.zeros(n, dtype=float)
        partner[valid] = h[idx[valid]]

        cand_one = np.where((h == 1) & (partner == 0))[0]
        cand_zero = np.where((h == 0) & (partner == 1))[0]

        performed = False
        if cand_one.size and cand_zero.size:
            i = int(np.random.choice(cand_one))
            j = int(np.random.choice(cand_zero))
            h[i], h[j] = h[j], h[i]                     # swap
            new_score = compute_c5(h)
            if new_score <= cur_score:
                cur_score = new_score
                no_improve = 0
                performed = True
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]                 # revert swap

        # --- fallback: random opposite‑value swap ---
        if not performed:
            while True:
                i = int(np.random.randint(n))
                j = int(np.random.randint(n))
                if h[i] != h[j]:
                    break
            h[i], h[j] = h[j], h[i]
            new_score = compute_c5(h)
            if new_score <= cur_score:
                cur_score = new_score
                no_improve = 0
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]                 # revert
                no_improve += 1

    return best_h, best_score


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] that minimises the Erdős C₅ bound."""
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()

    # Determine problem size and an initial vector
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        # coerce to strict binary and fix the exact half‑sum if needed
        h0 = np.where(h0 >= 0.5, 1.0, 0.0)
        target = n // 2
        diff = int(target - h0.sum())
        if diff > 0:
            zeros = np.where(h0 == 0)[0]
            flip = np.random.choice(zeros, size=diff, replace=False)
            h0[flip] = 1.0
        elif diff < 0:
            ones = np.where(h0 == 1)[0]
            flip = np.random.choice(ones, size=-diff, replace=False)
            h0[flip] = 0.0
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    remaining = budget_s - (time.time() - start)
    if remaining <= 0:
        return h0, compute_c5(h0), n

    best_h, best_score = hill_climb(h0, remaining, start)
    return best_h, best_score, n
