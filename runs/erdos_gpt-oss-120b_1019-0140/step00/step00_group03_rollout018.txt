def compute_c5(h):
    # compute correlation: see np.correlate.
    # In evaluation, they use np.correlate(h, 1 - h, mode='full') * dx.
    # We'll compute using np.correlate. The result length 2n-1. The max is the max correlation.
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))

def random_solution(n):
    # generate valid h with sum n/2.
    # Could generate random numbers then project.
    # Use Dirichlet: generate random>0; then scale to sum target; but may exceed 1 per entry.
    # So use simple: start with zeros, then allocate mass uniformly random to indices until sum target, ensuring each <=1.
    target_sum = n / 2
    h = np.zeros(n, dtype=float)
    remaining = target_sum
    indices = np.arange(n)
    np.random.shuffle(indices)
    for idx in indices:
        max_val = min(1.0, remaining)
        # assign random value between 0 and max_val
        val = np.random.rand() * max_val
        h[idx] = val
        remaining -= val
        if remaining <= 0:
            break
    # If some leftover due to rounding, distribute small epsilon.
    # Ensure sum exactly using scaling small epsilon.
    if remaining > 1e-12:
        # distribute to random indices that are not at max.
        possible = [i for i in range(n) if h[i] < 1.0 - 1e-9]
        # If none, just scale down all (should not happen)
        if possible:
            inc = remaining / len(possible)
            for i in possible:
                h[i] = min(1.0, h[i] + inc)
            # recompute sum diff
        # else fallback: just scale entire vector
    # Ensure sum exactly target, adjust small epsilon globally.
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        # adjust by adding diff uniformly then clipping, iterate a few times
        h += diff / n
        # clip again
        np.clip(h, 0, 1, out=h)
        # final scaling
        if h.sum() != 0:
            h *= target_sum / h.sum()
    # final clip
    np.clip(h, 0, 1, out=h)
    # final ensure sum close
    return h

    # ensures 0 <= x_i <= 1 and sum x_i = target
    # Using KKT: find lambda s.t. sum_i clip(x_i - lambda, 0,1) = target.

import time
import numpy as np

def project_to_sum_and_bounds(v, target):
    """Project vector v onto the box [0,1]^n with sum equal to target.
    Uses bisection on lambda for the L2 projection."""
    # Ensure v is numpy array float
    v = np.asarray(v, dtype=float)
    # Clamp lambda search bounds
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def random_solution(n, rng, target):
    """Generate a random feasible h."""
    # start from uniform random and scale to target sum
    h = rng.random(n)
    h = h / h.sum() * target
    # project to meet bounds and exact sum
    h = project_to_sum_and_bounds(h, target)
    return h

def compute_c5(h, dx):
    """Compute C5 bound for discrete h."""
    # convolution h with (1-h)
    # np.correlate yields sum_i h[i] * (1-h[i+lag]) for each lag
    corr = np.correlate(h, 1.0 - h, mode='full')
    # maximum overlap
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of points
    try:
        n_points = len(initial_h_values)
        has_initial = True
    except NameError:
        n_points = 70
        has_initial = False

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialise h
    if has_initial:
        h_cur = np.array(initial_h_values, dtype=float)
        # Clip and project to satisfy constraints exactly
        np.clip(h_cur, 0.0, 1.0, out=h_cur)
        h_cur = project_to_sum_and_bounds(h_cur, target_sum)
    else:
        h_cur = random_solution(n_points, rng, target_sum)

    # Initial evaluation
    cur_score = compute_c5(h_cur, dx)
    best_h = h_cur.copy()
    best_score = cur_score

    # Optimization parameters
    no_improve = 0
    max_no_improve = 4000
    iter_count = 0

    # Main loop
    while True:
        # time check
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.05:
            break

        # annealing/step schedule
        progress = elapsed / budget_s  # 0..1
        step_factor = 0.5 * (1.0 - progress) + 0.02   # from ~0.52 to 0.02
        temperature = 0.001 * (1.0 - progress) + 1e-6

        # Propose a move: transfer mass between two coordinates
        i, j = rng.choice(n_points, size=2, replace=False)
        hi = h_cur[i]
        hj = h_cur[j]

        max_forward = min(hi, 1.0 - hj)   # amount we can move from i to j
        max_backward = min(hj, 1.0 - hi)  # amount we can move from j to i

        if rng.random() < 0.5:
            # move i -> j
            max_delta = max_forward
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * step_factor
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # move j -> i
            max_delta = max_backward
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * step_factor
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta

        # No need to project because move respects bounds,
        # but rounding may introduce tiny violations; enforce clipping
        np.clip(h_new, 0.0, 1.0, out=h_new)

        # Evaluate
        new_score = compute_c5(h_new, dx)

        # Acceptance rule
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probabilistic acceptance
            if temperature > 0 and rng.random() < np.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            h_cur = h_new
            cur_score = new_score
            # Record best
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iter_count += 1

        # Restart / jitter if stuck
        if no_improve >= max_no_improve:
            # jitter around best solution
            eps = 0.2 * (1.0 - progress) + 0.01   # magnitude of jitter

            h_cur = project_to_sum_and_bounds(h_cur, target_sum)
            cur_score = compute_c5(h_cur, dx)
            no_improve = 0

    # Final projection of best solution to be safe
    best_h = project_to_sum_and_bounds(best_h, target_sum)
    final_score = compute_c5(best_h, dx)

    return best_h.tolist(), float(final_score), n_points
