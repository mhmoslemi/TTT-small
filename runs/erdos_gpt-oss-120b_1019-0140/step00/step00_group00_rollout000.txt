# Helper: compute the C5 bound for a candidate h.
def _c5_of(h):
    """Return max_k ∫ h(x) (1 - h(x+k)) dx for discretized h."""
    N = h.shape[0]
    dx = 2.0 / N
    # linear correlation (zero‑padded) of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

    Returns
    -------
    h_values : list of float
        The optimized samples of h.
    c5_bound : float
        The achieved upper bound for C₅.
    n_points : int
        Number of discretisation points (len(h_values)).
    """

    # ------------------------------------------------------------------

    np.random.seed(seed)

    h = np.repeat(init, factor).astype(float)

    # Ensure feasibility (0 ≤ h ≤ 1 and sum(h)=N/2)
    h = np.clip(h, 0.0, 1.0)
    target_sum = N / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust uniformly, then re‑clip
        h += (target_sum - cur_sum) / N
        h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Optimization state
    # ------------------------------------------------------------------
    cur_h = h.copy()
    cur_c5 = _c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # Simulated‑annealing parameters
    T_initial = 0.02            # initial temperature (controls uphill moves)
    max_no_improve = 20000      # steps before a random restart
    no_improve_cnt = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        N = cur_h.shape[0]

        # ---- propose a mass‑transfer move that respects the sum constraint ----
        # pick donor (must have room to give) and receiver (must have room to take)
        donor = random.randrange(N)
        for _ in range(5):
            if cur_h[donor] > 1e-12:
                break
            donor = random.randrange(N)

        receiver = random.randrange(N)
        for _ in range(5):
            if cur_h[receiver] < 1.0 - 1e-12:
                break
            receiver = random.randrange(N)

        if donor == receiver:
            continue

        max_delta = min(cur_h[donor], 1.0 - cur_h[receiver])
        if max_delta <= 0.0:
            continue

        # random step size (up to the full feasible amount)
        delta = max_delta * random.random()

        # apply the move
        cur_h[donor] -= delta
        cur_h[receiver] += delta

        # evaluate
        new_c5 = _c5_of(cur_h)

        # temperature schedule (linear decay)
        elapsed = time.time() - start
        T = T_initial * (1.0 - elapsed / budget_s)
        if T < 0.0:
            T = 0.0

        # acceptance rule
        accept = False
        if new_c5 <= cur_c5:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_c5 - cur_c5) / T)
            if random.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            # record best solution
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1
        else:
            # revert
            cur_h[donor] += delta
            cur_h[receiver] -= delta
            no_improve_cnt += 1

        # ----- occasional random restart to escape stagnation -----
        if no_improve_cnt >= max_no_improve:
            # create a fresh random feasible vector (half‑ones, half‑zeros shuffled)
            half = N // 2
            cur_h = np.concatenate([np.ones(half), np.zeros(N - half)])
            np.random.shuffle(cur_h)
            cur_c5 = _c5_of(cur_h)
            no_improve_cnt = 0

    # ------------------------------------------------------------------
    # Finalise result
    # ------------------------------------------------------------------
    # Ensure the final vector satisfies the integral exactly (tiny tolerance)
    final_sum = best_h.sum()
    if abs(final_sum - target_sum) > 1e-8:
        correction = (target_sum - final_sum) / N
        best_h += correction
        best_h = np.clip(best_h, 0.0, 1.0)

    # Compute final C5 using the official evaluator if available
    try:
        c5_bound = evaluate_erdos_solution(best_h)  # provided by harness
        c5_bound = float(c5_bound)
    except Exception:
        c5_bound = _c5_of(best_h)

    return best_h.tolist(), c5_bound, N
