import time
import random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _improve_continuous(start_h: np.ndarray, time_budget: float, target: float = None):
    """
    Greedy / simulated‑annealing style improvement for a real‑valued step function.
    Moves a small amount of mass from one entry to another while preserving
    the integral constraint and bounds.  The move direction is guided by the
    shift that currently gives the smallest self‑correlation (i.e. the worst
    overlapping shift).
    Returns the best vector found and its C5 value.
    """
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = start_h.copy()
    cur_val = _c5_value(h)
    best_val = cur_val
    best_h = h.copy()

    start = time.time()
    no_improve = 0
    step_factor = 0.5  # fraction of available epsilon to transfer
    # main loop
    while time.time() - start < time_budget:
        # self‑correlation (h with itself) to find the worst shift
        self_corr = np.correlate(h, h, mode='full')
        k_min = int(np.argmin(self_corr) - (n - 1))

        # gradient for that shift: g_i = h_{i+k}+h_{i-k} (zero outside)
        grad = np.zeros(n, dtype=float)
        if k_min != 0:
            k = abs(k_min)
            if k_min > 0:
                grad[: n - k] += h[k:]          # h_{i+k}
                grad[k:] += h[: n - k]          # h_{i-k}
            else:
                grad[k:] += h[: n - k]          # h_{i+k} (k negative)
                grad[: n - k] += h[k:]          # h_{i-k}
        else:
            grad = 2.0 * h  # shift zero

        # pick indices with smallest / largest gradient that can move mass
        idx_sorted = np.argsort(grad)
        i_idx = idx_sorted[0]   # candidate for decreasing
        j_idx = idx_sorted[-1]  # candidate for increasing

        # ensure feasibility of the transfer
        if h[i_idx] <= 0.0:
            pos = np.where(h > 1e-12)[0]
            if pos.size == 0:
                break
            i_idx = int(np.random.choice(pos))
        if h[j_idx] >= 1.0:
            pos = np.where(h < 1.0 - 1e-12)[0]
            if pos.size == 0:
                break
            j_idx = int(np.random.choice(pos))

        # maximum transferable amount respecting the box constraints
        eps_max = min(1.0 - h[j_idx], h[i_idx])
        if eps_max <= 0.0:
            # pick random feasible pair as a fallback
            pos_i = np.where(h > 1e-12)[0]
            pos_j = np.where(h < 1.0 - 1e-12)[0]
            if pos_i.size == 0 or pos_j.size == 0:
                break
            i_idx = int(np.random.choice(pos_i))
            j_idx = int(np.random.choice(pos_j))
            eps_max = min(1.0 - h[j_idx], h[i_idx])
            if eps_max <= 0.0:
                break

        # random step (biased by step_factor)
        eps = eps_max * step_factor * np.random.rand()
        if eps <= 0.0:
            continue

        # perform the transfer
        h[i_idx] -= eps
        h[j_idx] += eps

        new_val = _c5_value(h)
        delta = new_val - cur_val

        # accept only improvements (greedy); can be relaxed with a temperature later
        if delta < 0.0:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if target is not None and best_val <= target:
                    break
        else:
            # revert
            h[i_idx] += eps
            h[j_idx] -= eps
            no_improve += 1

        # adapt step size if stuck
        if no_improve > 4000:
            step_factor *= 0.5
            if step_factor < 0.01:
                step_factor = 0.01
            no_improve = 0

    # final safeguard: enforce exact sum (tiny numerical drift)
    s = best_h.sum()
    if abs(s - target_sum) > 1e-9:
        # simple affine correction then clipping
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)
        # re‑balance any remaining error uniformly if possible
        s2 = best_h.sum()
        diff = target_sum - s2
        if abs(diff) > 1e-12:
            # distribute diff over entries that are not at the box limits
            free = np.where((best_h > 0.0 + 1e-12) & (best_h < 1.0 - 1e-12))[0]
            if free.size:
                add = diff / free.size
                best_h[free] += add
                best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_val
