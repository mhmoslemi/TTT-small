import time
import numpy as np

def _compute_c5(arr):
    """Return max overlap integral for binary (or real) step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)

def _to_binary(vec):
    """Convert a real-valued vector with sum = n/2 to a binary vector of length n."""
    n = vec.shape[0]
    target = n // 2
    # choose the largest entries to become 1
    idx = np.argsort(-vec)[:target]
    bin_vec = np.zeros(n, dtype=int)
    bin_vec[idx] = 1
    return bin_vec

def _local_optimize(h, compute_c5):
    """Greedy hill‑climb: swap a 1 with a 0 if it reduces the C5 bound."""
    n = h.shape[0]
    while True:
        cur_score = compute_c5(h)
        improved = False
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]          # swap
                new_score = compute_c5(h)
                if new_score < cur_score - 1e-12:
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]          # revert
            if improved:
                break
        if not improved:
            break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.shape[0]
        target_sum = n_points // 2
        # Ensure binary and correct sum
        if not np.all((init == 0) | (init == 1)):
            h = _to_binary(init)
        else:
            h = init.astype(int).copy()
            cur_sum = int(h.sum())
            if cur_sum != target_sum:
                # adjust by flipping smallest/large values (here simply random)
                diff = target_sum - cur_sum
                if diff > 0:
                    zeros = np.flatnonzero(h == 0)
                    flip = rng.choice(zeros, size=diff, replace=False)
                    h[flip] = 1
                elif diff < 0:
                    ones = np.flatnonzero(h == 1)
                    flip = rng.choice(ones, size=-diff, replace=False)
                    h[flip] = 0
    else:
        n_points = 70
        target_sum = n_points // 2
        h = np.zeros(n_points, dtype=int)
        idx = rng.choice(n_points, size=target_sum, replace=False)
        h[idx] = 1

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _compute_c5(best_h)
    # If an external evaluator exists, keep its result for final reporting
    if "evaluate_erdos_solution" in globals():
        best_score = float(evaluate_erdos_solution(best_h))

    TARGET = 0.3808  # stop if we beat this bound

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while True:
        if time.time() - start > budget_s * 0.98:
            break
        if best_score <= TARGET:
            break

        # Decide whether to start from scratch or perturb the best known solution
        if rng.random() < 0.15:
            # fresh random start
            h_candidate = np.zeros(n_points, dtype=int)
            idx = rng.choice(n_points, size=target_sum, replace=False)
            h_candidate[idx] = 1
        else:

            h_candidate = best_h.copy()
            # perform a small random number of swaps

    # Final evaluation & return

        final_c5 = float(evaluate_erdos_solution(best_h))
    else:
        final_c5 = _compute_c5(best_h)

    return best_h, final_c5, n_points
