def solve_milp(n, p, lower_overlap):
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    constraints = []
    constraints.append(cp.sum(h) == p)
    # linking constraints
    # w[i,j] <= h[i]
    constraints += [w[i,j] <= h[i] for i in range(n) for j in range(n)]
    constraints += [w[i,j] <= h[j] for i in range(n) for j in range(n)]
    # w[i,j] >= h[i] + h[j] - 1
    constraints += [w[i,j] >= h[i] + h[j] - 1 for i in range(n) for j in range(n)]
    # shift constraints

        i = np.random.choice(ones)

        h[i], h[j] = h[j], h[i]

            # revert

import time, numpy as np, cvxpy as cp, random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
```

def simulated_annealing(h0, time_budget, target_c5=None):
    n = len(h0)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    h = h0.copy()
    cur_val = best_val
    start = time.time()
    # initial temperature adapt: T0 based on best_val maybe
    T0 = best_val * 0.5  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed > time_budget:
            break
        # temperature schedule linearly decreasing
        T = T0 * (1 - elapsed / time_budget)
        # choose two distinct indices
        i, j = random.sample(range(n), 2)
        # move mass from i to j (or opposite)
        # determine direction: we will move from donor i to receiver j
        # choose direction randomly
        if random.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        # compute max delta we can transfer
        max_delta = min(h[donor], 1 - h[receiver])
        if max_delta <= 0:
            continue
        delta = random.random() * max_delta
        # apply move
        h_new = h.copy()
        h_new[donor] -= delta
        h_new[receiver] += delta
        new_val = compute_c5(h_new)
        # acceptance
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):

            cur_val = new_val

                best_h = h_new.copy()
                # early exit if meets target
                if target_c5 is not None and best_val <= target_c5:
                    break
    return best_h, best_val

import time
import random
import math
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def simulated_annealing(start_h, time_budget, target_c5=None):
    """Simple SA that conserves the total integral and respects bounds."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_val = compute_c5(best_h)
    cur_h = start_h.copy()
    cur_val = best_val
    start = time.time()
    # initial temperature scaled to the current value
    T0 = max(0.01, best_val * 0.5)

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # linear cooling
        T = T0 * (1.0 - elapsed / time_budget)
        # pick two distinct indices and a direction (donor → receiver)
        i, j = random.sample(range(n), 2)
        if random.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_delta = min(cur_h[donor], 1.0 - cur_h[receiver])
        if max_delta <= 0.0:
            continue
        delta = random.random() * max_delta
        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta
        new_val = compute_c5(new_h)
        # Metropolis acceptance
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h
                best_val = new_val
                if target_c5 is not None and best_val <= target_c5:
                    break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the sample count."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ----- initialise h -------------------------------------------------
    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.array(init, dtype=float)
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=float)
    n = h0.shape[0]
    p = n // 2                     # needed sum (since integral = 1)
    dx = 2.0 / n

    # Force exact sum while staying inside bounds
    cur_sum = np.sum(h0)
    if abs(cur_sum - p) > 1e-9:
        diff = p - cur_sum
        # adjust the first entry conservatively
        adj = max(-h0[0], min(1.0 - h0[0], diff))
        h0[0] += adj
    # Clip any stray numerical noise
    np.clip(h0, 0.0, 1.0, out=h0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- early exit if initial already good ----------------------------
    target_c5 = 0.3808
    if best_val <= target_c5:
        return (best_h.tolist(), best_val, n)

    # ----- simulated annealing -------------------------------------------
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining > 0:
        h_candidate, val_candidate = simulated_annealing(
            best_h, remaining, target_c5=target_c5
        )
        if val_candidate < best_val:
            best_h, best_val = h_candidate, val_candidate

    # ----- final correction of the integral -------------------------------
    total = np.sum(best_h)
    err = p - total
    if abs(err) > 1e-9:
        # distribute the small error to the first feasible element
        for idx in range(n):
            if err > 0 and best_h[idx] < 1.0:
                delta = min(err, 1.0 - best_h[idx])
                best_h[idx] += delta
                err -= delta
            elif err < 0 and best_h[idx] > 0.0:
                delta = min(-err, best_h[idx])
                best_h[idx] -= delta
                err += delta
            if abs(err) < 1e-12:
                break

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n)
