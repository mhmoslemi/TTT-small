import numpy as np
import random
import time

def compute_c5(h):
    """C5 = max_k dx * sum_i h[i] * (1 - h[i+k]) using linear correlation."""
    n = h.shape[0]
    # linear (zero‑padded) correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)

def project_to_feasible(v, target_sum):
    """
    Euclidean projection of v onto the hypercube [0,1]^n with the linear
    equality sum(v)=target_sum.
    Finds λ such that y_i = clip(v_i - λ, 0, 1) satisfies sum(y)=target_sum.
    """
    lo, hi = -1.0, 2.0  # bounds wide enough to bracket the solution
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return y

def evaluate_candidate(h):
    """Wrapper that prefers the supplied evaluator if available."""
    try:
        return float(evaluate_erdos_solution(h))  # pre‑imported by the harness
    except Exception:
        return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a problem size.  Use the parent construction when present.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 200          # any even n < 1000 works well
        h0 = np.full(n, 0.5, dtype=float)

    target = n / 2.0                     # ∑ h_i must equal n/2
    h0 = project_to_feasible(h0, target)

    best_h = h0.copy()
    best_c5 = evaluate_candidate(best_h)

    # --------------------------------------------------------------
    # One‑parameter family h_p(x) = ((p+1)/2) * |x-1|^p,   p∈[0,1]
    # --------------------------------------------------------------
    dx = 2.0 / n
    x = np.arange(n) * dx                # grid points, 0 ≤ x < 2

    for p in np.linspace(0.0, 1.0, 101):
        raw = ((p + 1.0) / 2.0) * np.abs(x - 1.0) ** p
        cand = project_to_feasible(raw, target)
        c5 = evaluate_candidate(cand)
