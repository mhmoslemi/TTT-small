def project_to_constraints(v, target):
    # v is array
    # Clip to [0,1]
    x = np.clip(v, 0.0, 1.0)
    # Compute diff
    diff = np.sum(x) - target
    # if diff is zero within tolerance, return
    # else need to move mass from some elements that are not at bound.
    # We'll use water-filling: adjust x_i = x_i - diff * w_i where w_i = 1 (equal) for all free indices.
    # but if after subtraction some go below 0, we need to clamp them and continue.
    # We'll iteratively adjust.
    max_iter = 10
    for _ in range(max_iter):
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        # compute amount to adjust per free element
        adjust = diff / n_free
        # update free entries
        x[free] -= adjust
        # clip again
        x = np.clip(x, 0.0, 1.0)
        new_diff = np.sum(x) - target
        if np.abs(new_diff) < 1e-9:
            break
        # update diff for next loop
        diff = new_diff
    # final correction: if still off due to rounding, distribute across free ignoring bounds minimally
    # but should be good enough
    return x
```

import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraints(v, target):
    x = np.clip(v, 0.0, 1.0)
    diff = np.sum(x) - target
    if abs(diff) < 1e-12:
        return x
    for _ in range(100):
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
        diff = np.sum(x) - target
        if abs(diff) < 1e-12:
            break
    return x

def improve_h(initial_h, budget_seconds, max_iter=200000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    start = time.time()
    n = len(initial_h)
    target = n / 2.0
    best_h = initial_h.copy()
    best_score = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_score = best_score
    alpha = 0.5
    T0 = 0.1
    iteration = 0
    while True:
        now = time.time()
        if now - start >= budget_seconds or iteration >= max_iter:
            break
        # compute correlation and shift giving max overlap
        corr = np.correlate(curr_h, 1 - curr_h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        # gradient of the maximal term
        grad = np.ones_like(curr_h)
        if k >= 0:
            grad[:n - k] -= curr_h[k:]
            grad[k:] -= curr_h[:n - k]
        else:

import time
import math
import numpy as np

def _compute_c5(h):
    """Return the max overlap integral for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_constraints(v, target):
    """
    Project vector v onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target.
    Uses an iterative water‑filling scheme.
    """
    x = np.clip(v, 0.0, 1.0)
    diff = np.sum(x) - target
    if abs(diff) < 1e-12:
        return x
    for _ in range(100):
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
        diff = np.sum(x) - target
        if abs(diff) < 1e-12:
            break
    return x


def _improve_h(initial_h, budget_seconds, max_iter=200000, rng=None):
    """
    Heuristic search for a better h.
    Uses a sub‑gradient step on the currently worst shift together with a
    simple simulated‑annealing acceptance rule and projection onto the
    feasible set.
    """
    if rng is None:
        rng = np.random.default_rng()
    start = time.time()
    n = len(initial_h)
    target = n / 2.0

    best_h = initial_h.copy()
    best_score = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    alpha = 0.5       # base step size
    T0 = 0.1          # initial temperature for SA

    it = 0
    while True:
        now = time.time()
        if now - start >= budget_seconds or it >= max_iter:
            break

        # find shift achieving the current maximum overlap
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)

        # gradient of the selected term f_k = sum_i h_i - sum_i h_i h_{i+k}
        grad = np.ones_like(cur_h)
        if k >= 0:
            grad[: n - k] -= cur_h[k:]
            grad[k:] -= cur_h[: n - k]
        else:
            kp = -k
            grad[kp:] -= cur_h[: n - kp]
            grad[: n - kp] -= cur_h[kp:]

        # descent direction with occasional noise
        step = -alpha * grad
        if it % 100 == 0:
            step += 0.05 * alpha * rng.standard_normal(n)

        cand_h = cur_h + step
        cand_h = _project_to_constraints(cand_h, target)
        cand_score = _compute_c5(cand_h)

        # deterministic improvement acceptance
        if cand_score < cur_score:
            cur_h = cand_h
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand_h.copy()
                best_score = cand_score
        else:
            # simulated‑annealing fallback
            elapsed = now - start
            frac = elapsed / budget_seconds
            T = T0 * (1.0 - frac)
            if T > 0 and rng.random() < math.exp(-(cand_score - cur_score) / T):
                cur_h = cand_h
                cur_score = cand_score

        # decay step size
        alpha *= 0.9995
        it += 1

        if best_score <= 0.3808:
            break

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):

    Returns

        # simple equal‐share of the remaining budget among still‑to‑process candidates
        time_for_this = remain / (total - idx)

        h_opt, sc_opt = _improve_h(h0, time_for_this, rng=rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt.copy()

        if best_score <= 0.3808:
            break

    # fallback – uniform 0.5 if everything failed
    if best_h is None:
        n = 2
        best_h = np.full(n, 0.5)
        best_score = _compute_c5(best_h)

    # final feasibility enforcement
    n_final = len(best_h)
    best_h = _project_to_constraints(best_h, n_final / 2.0)
    final_score = _compute_c5(best_h)

    return best_h.tolist(), float(final_score), n_final
