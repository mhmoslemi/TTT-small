def project_to_simplex_cap(v, target):
    # target within [0,n] and n is length.
    # Bisection for λ.
    n = v.size
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        # compute clipped
        w = v - mid
        w = np.clip(w, 0, 1)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
        if high - low < 1e-12:
            break
    return np.clip(v - ((low+high)/2.0), 0, 1)

import time
import numpy as np

# Project a vector onto the capped simplex: 0 <= x_i <= 1, sum(x)=target
def _proj_capped_simplex(v, target, tol=1e-12, max_iter=100):
    n = v.shape[0]
    # Quick check
    if np.all(v >= 0) and np.all(v <= 1) and abs(v.sum() - target) <= tol:
        return v.copy()
    # Bisection on lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

# Correlation (raw, without dx) and max shift
def _max_shift_raw(h):
    n = h.shape[0]
    raw = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(raw))
    L = max_idx - (n - 1)
    return L, raw[max_idx], raw

# C5 value with dx factor
def _c5_value(h):
    n = h.shape[0]
    _, raw_max, _ = _max_shift_raw(h)
    return raw_max * (2.0 / n)

# Gradient of the maximal shifted correlation w.r.t. h
def _grad_wrt_shift(h, L):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if L >= 0:
        # term (1 - h[i+L])
        idx = np.arange(0, n - L)
        grad[idx] += 1.0 - h[idx + L]
        # term -h[i-L]
        idx2 = np.arange(L, n)
        grad[idx2] -= h[idx2 - L]
    else:
        k = -L
        idx = np.arange(k, n)
        grad[idx] += 1.0 - h[idx + L]          # idx+L = idx-k
        idx2 = np.arange(0, n - k)
        grad[idx2] -= h[idx2 - L]              # idx2-L = idx2+k
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Desired number of points – use a moderately fine grid
    N = 200  # even, allows binary‑like patterns if needed
    target_sum = N / 2.0

    # Initialise from provided data if available
    if 'initial_h_values' in globals():
        hv = np.array(initial_h_values, dtype=float)
        n0 = hv.shape[0]
        # Resample to length N by linear interpolation
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h = np.interp(x_new, x_old, hv)
    else:
        h = np.random.rand(N)

    # Project onto feasible set
    h = _proj_capped_simplex(h, target_sum)

    # Evaluate initial solution
    best_h = h.copy()
    best_c5 = _c5_value(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    # Annealing / schedule parameters
    max_lr = 0.5
    min_lr = 0.001
    max_sigma = 0.3
    min_sigma = 0.001
    # main optimisation loop
    while True:
        now = time.time()
        if now - start >= budget_s * 0.95:
            break
        elapsed = now - start
        frac = elapsed / (budget_s * 0.95)

        # learning rates decay linearly
        lr = max_lr * (1.0 - frac) + min_lr * frac
        sigma = max_sigma * (1.0 - frac) + min_sigma * frac

        # --- gradient descent step on worst shift ---
        Lmax, _, _ = _max_shift_raw(cur_h)
        grad = _grad_wrt_shift(cur_h, Lmax)
        propose = cur_h - lr * grad
        propose = _proj_capped_simplex(propose, target_sum)

        new_c5 = _c5_value(propose)

        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (0.01 + 0.1 * (1 - frac))):
            cur_h, cur_c5 = propose, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, propose.copy()

        # --- occasional random perturbation to escape local minima ---
        if np.random.rand() < 0.02:
            rand_step = sigma * np.random.randn(N)
            cand = cur_h + rand_step
            cand = _proj_capped_simplex(cand, target_sum)
            cand_c5 = _c5_value(cand)
            if cand_c5 < cur_c5 or np.random.rand() < 0.1:
                cur_h, cur_c5 = cand, cand_c5
                if cand_c5 < best_c5:
                    best_c5, best_h = cand_c5, cand.copy()

    # Final bound computation – use the helper if present
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = _c5_value(best_h)

    return (best_h, final_c5, best_h.shape[0])
