def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random

    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine length n_points using initial_h if available and even.
    try:
        init_h = initial_h_values
    except NameError:
        init_h = None

    if init_h is not None:
        n_initial = len(init_h)
        if n_initial % 2 == 0:
            n_points = n_initial
        else:
            # fallback to default size
            init_h = None
            n_points = 70
    else:
        n_points = 70

    target_sum = n_points // 2

    # Helper to create binary initial solution from float vector
    def binary_from_cont(vals):
        idx = np.argsort(-vals)  # descending
        h = np.zeros(n_points, dtype=np.float64)
        h[idx[:target_sum]] = 1.0
        return h

    if init_h is not None:
        # Ensure we slice or pad to n_points length
        vals = np.asarray(init_h[:n_points], dtype=np.float64)
        h_current = binary_from_cont(vals)
    else:
        # random binary selection
        perm = rng.permutation(n_points)
        h_current = np.zeros(n_points, dtype=np.float64)
        h_current[perm[:target_sum]] = 1.0

    # Evaluate function
    def evaluate(h):
        n = h.shape[0]
        dx = 2.0 / n
        # correlation of h with 1-h
        corr = np.correlate(h, 1.0 - h, mode='full')
        # max value times dx
        return float(corr.max() * dx)

    best_h = h_current.copy()
    best_score = evaluate(best_h)
    current_h = h_current.copy()
    current_score = best_score

    # Precompute zero and one index sets for current
    # We'll compute each iteration.

    # Parameters for search
    # Maximum iterations: we can estimate based on time left.
    # We'll run loop until time exhausted.
    # Use counters for stagnation
    no_improve_iters = 0
    # Set thresholds
    restart_threshold = 20000  # if no improvement for this many iterations, restart
    # Temperature schedule (simulated annealing)
    start_temp = 0.01  # initial temperature
    end_temp = 0.0001

    # Kernel for random swap: function that returns neighbor
    n = n_points

    # Main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.5:  # leave a little margin
            break

        # Compute temperature relative to remaining budget
        t_frac = max(0.0, (budget_s - elapsed) / budget_s)
        temp = end_temp + (start_temp - end_temp) * t_frac

        # Choose random swap
        # Determine indices of ones and zeros
        ones = np.flatnonzero(current_h == 1)
        zeros = np.flatnonzero(current_h == 0)
        if ones.size == 0 or zeros.size == 0:
            # shouldn't happen, but break
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # Apply swap to get neighbor
        neighbor = current_h.copy()
        neighbor[i] = 0.0
        neighbor[j] = 1.0

        # Evaluate neighbor
        neighbor_score = evaluate(neighbor)

        # Determine acceptance
        accept = False
        if neighbor_score < current_score:
            accept = True
        else:
            # worse move acceptance with probability exp(-(Δ)/temp)
            if temp > 0:
                delta = neighbor_score - current_score
                prob = math.exp(-delta / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            current_h = neighbor
            current_score = neighbor_score
            # Reset no_improve counter when improvement in current solution (even if not best)
            if neighbor_score < best_score - 1e-12:
                best_h = neighbor.copy()
                best_score = neighbor_score
                no_improve_iters = 0
            else:
                no_improve_iters += 1
        else:
            no_improve_iters += 1

        # Restart if stagnation too long, but keep best overall
        if no_improve_iters >= restart_threshold:

import numpy as np
import time
import math

def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def binary_from_vals(vals):
    n = vals.shape[0]
    target = n // 2
    idx = np.argsort(-vals)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    return h

def max_shift_and_contributions(h):
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr_full)
    shift = max_idx - (n - 1)
    # Identify bad positions
    if shift >= 0:
        left = np.arange(0, n - shift)
        bad = left[(h[left] == 1) & (h[left + shift] == 0)]
    else:
        s = -shift
        right = np.arange(s, n)
        bad = right[(h[right] == 1) & (h[right - s] == 0)]
    return shift, bad, corr_full

import numpy as np
import time
import math

def evaluate_h(h):
    """Return C5 value for binary/float vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def binary_from_vals(vals):
    """Project a continuous vector to a binary one with exactly half ones."""
    n = vals.shape[0]
    target = n // 2
    idx = np.argsort(-vals)               # descending order
    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    return h

def max_shift_and_bad(h):
    """Find shift with maximal overlap and indices that contribute to it."""
    n = h.shape[0]

    shift = max_idx - (n - 1)              # integer shift (negative allowed)
    if shift >= 0:
        left = np.arange(0, n - shift)
        bad = left[(h[left] == 1) & (h[left + shift] == 0)]
    else:
        s = -shift
        right = np.arange(s, n)
        bad = right[(h[right] == 1) & (h[right - s] == 0)]
    return shift, bad, corr_full
