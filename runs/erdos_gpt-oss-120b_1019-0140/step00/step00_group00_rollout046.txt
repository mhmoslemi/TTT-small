    def compute_bound(h):
        n = h.shape[0]
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n)

    def project_onto_simplex_with_bounds(y, target):
        # Uses lambda search
        lo = -np.max(y) - 2.0
        hi = 1 - np.min(y) + 2.0
        # binary search for lambda
        for _ in range(60):
            mid = (lo + hi) / 2.0
            x = np.clip(y + mid, 0.0, 1.0)
            s = x.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        return np.clip(y + (lo+hi)/2.0, 0.0, 1.0)

    def gradient_max_corr(h):
        n = h.shape[0]
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        g = np.zeros_like(h)
        if shift >= 0:
            # part where i+shift valid
            g[:n-shift] = 1.0 - h[shift:]
            # part where i-shift valid give negative h[i-shift]
            g[shift:] -= h[:n-shift]
        else:
            s = -shift
            # i+shift = i - s valid for i >= s
            g[s:] = 1.0 - h[:n-s]
            # i-shift = i + s valid for i < n-s
            g[:n-s] -= h[s:]
        return g, shift, max_idx, corr[max_idx]

import time
import numpy as np

def _compute_bound(h: np.ndarray) -> float:
    """Return C5 bound for step function h."""
    n = h.shape[0]
    # full correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _project_to_bounds_and_sum(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of y onto the hyperplane sum(x)=target_sum
    intersected with the box [0,1]^n.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(y) - 2.0
    hi = 1.0 - np.min(y) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_of_worst_shift(h: np.ndarray):
    """
    Compute sub‑gradient of the shift that attains the maximal overlap.
    Returns gradient vector g, the shift k (integer), and the maximal raw
    correlation value (before scaling by dx).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    max_corr = float(corr[idx_max])
    shift = idx_max - (n - 1)          # signed shift where max occurs

    g = np.zeros_like(h)

    if shift >= 0:
        # term from h[i] * (1 - h[i+shift])
        g[: n - shift] = 1.0 - h[shift:]
        # term from - h[i] * (1 - h[i-shift]) appearing with i >= shift
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # i+shift = i - s  (valid for i >= s)
        g[s:] = 1.0 - h[: n - s]
        # i-shift = i + s (valid for i < n-s)
        g[: n - s] -= h[s:]

    return g, shift, max_corr


def _optimize_h(initial_h: np.ndarray, time_limit: float):
    """Local gradient‑descent / simulated‑annealing optimizer."""
    n = initial_h.shape[0]
    target_sum = n * 0.5

    # ensure feasibility
    h = np.clip(initial_h, 0.0, 1.0)
    h = _project_to_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_val = _compute_bound(h)

    start = time.time()
    it = 0
    init_step = 0.5

    while time.time() - start < time_limit:
        it += 1
        # gradient of the currently worst shift
        g, shift, max_corr = _gradient_of_worst_shift(h)
        cur_val = max_corr * (2.0 / n)

        # step size decreasing with sqrt(iteration)
        step = init_step / np.sqrt(it)

        # line‑search-like descent
        h_new = None
        new_val = None
        step_try = step
        for _ in range(8):
            y = h - step_try * g
            y = _project_to_bounds_and_sum(y, target_sum)
            val = _compute_bound(y)
            if val <= cur_val + 1e-12:
                h_new = y
                new_val = val
                break
            step_try *= 0.5

        if h_new is not None:
            h = h_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            continue

        # fallback: random exchange of mass between two coordinates
        i, j = np.random.choice(n, size=2, replace=False)
        # move delta from i to j (i loses, j gains) keeping sum unchanged
        delta_min = max(h[i] - 1.0, -h[j])
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max > delta_min:
            delta = np.random.uniform(delta_min, delta_max)
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta
            val = _compute_bound(cand)
            if val < best_val:
                best_h = cand.copy()
                best_val = val
            if val <= cur_val + 1e-12:
                h = cand
                continue

        # periodic random perturbation to escape stagnation
        if it % 5000 == 0:
            perturb = np.random.randn(n) * 0.05
            cand = best_h + perturb
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_bounds_and_sum(cand, target_sum)
            val = _compute_bound(cand)
            if val < best_val:
                h = cand
                best_h = cand.copy()
                best_val = val

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # Determine starting vector
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n0 = init_vec.shape[0]
    except Exception:
        init_vec = None
        n0 = None

    # Target resolution (chosen to stay well below 1000 samples)
    target_n = 200
    if init_vec is not None:
        if n0 == target_n:
            h_start = init_vec
        else:
            # Interpolate to desired resolution
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h_start = np.interp(x_new, x_old, init_vec)
    else:
        # Simple feasible start: uniform 0.5
        h_start = np.full(target_n, 0.5, dtype=float)

    # Reserve a small safety margin for finalisation
    start_time = time.time()
    remaining = max(0.1, budget_s - (time.time() - start_time) - 0.3)

    best_h, best_val = _optimize_h(h_start, remaining)

    n = best_h.shape[0]
    target_sum = n * 0.5
    # Final projection to guarantee feasibility
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    final_val = _compute_bound(best_h)

    return best_h, final_val, n
