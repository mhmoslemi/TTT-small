import numpy as np, time

def project_onto_capped_simplex(v, S, lo=0.0, hi=1.0, max_iter=50, tol=1e-9):
    v = np.asarray(v, dtype=float)
    n = v.size
    # constrain target sum
    lo_sum = lo * n
    hi_sum = hi * n
    if S < lo_sum:
        return np.full(n, lo)
    if S > hi_sum:
        return np.full(n, hi)
    # Bisection for lambda
    # compute lower bound lambda_low such that sum_i clip(v_i - lambda_low, lo, hi) >= S
    # We'll use lambda lower = -np.max(v) (??)
    # Actually if lambda -> -inf, v_i - lambda -> +inf, clip to hi = 1, sum -> hi_sum.
    # If lambda -> +inf, v_i - lambda -> -inf, clip to lo = 0, sum -> lo_sum.
    # So we can start with low = -max(v) - 1, high = max(v) + 1.
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, lo, hi)
        sum_proj = np.sum(proj)
        if np.abs(sum_proj - S) < tol:
            return proj
        if sum_proj > S:
            low = lam
        else:
            high = lam
    return np.clip(v - lam, lo, hi)

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


def project_onto_capped_simplex(v: np.ndarray, S: float,
                                lo: float = 0.0, hi: float = 1.0,
                                max_iter: int = 60, tol: float = 1e-9) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | lo <= x_i <= hi,
    sum_i x_i = S}.  Implements a simple bisection on the Lagrange multiplier.
    """
    v = np.asarray(v, dtype=float)
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if S <= lo_sum:
        return np.full(n, lo, dtype=float)
    if S >= hi_sum:
        return np.full(n, hi, dtype=float)

    # Bracket lambda: low -> sum > S, high -> sum < S
    low = -np.max(v) - 1.0   # very negative => all clipped to hi
    high = np.max(v) + 1.0   # very positive => all clipped to lo
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, lo, hi)
        s = np.sum(proj)
        if abs(s - S) < tol:
            return proj
        if s > S:
            low = lam
        else:
            high = lam
    return np.clip(v - lam, lo, hi)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)


def local_search(h_init: np.ndarray, dx: float,
                 time_deadline: float, rng: np.random.Generator) -> tuple:
    """
    Simple hill‑climbing with pairwise exchange moves.
    Returns (best_h, best_c5) found before time_deadline.
    """
    h = h_init.copy()
    cur_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    n = h.size
    stagnation = 0
    while time.time() < time_deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # feasible delta: increase i by δ, decrease j by δ
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)

        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        new_c5 = compute_c5(h_new, dx)

        if new_c5 < cur_c5 - 1e-12:          # strict improvement
            h = h_new
            cur_c5 = new_c5
            stagnation = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            stagnation += 1

        # occasional random restart to escape plateaus
        if stagnation > 5000:
            h = random_feasible_vector(n, rng)
            cur_c5 = compute_c5(h, dx)
            stagnation = 0
    return best_h, best_c5
