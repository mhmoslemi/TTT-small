def local_optimize(h, time_budget, start_time):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    S = n/2
    while time.time() - start_time < time_budget:
        # choose random move
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # ensure we can move mass from i to j
        # choose direction based on values: If h[i] > h[j], we can move delta from i to j
        # else swap
        if h[i] > h[j]:
            source = i; target = j
        else:
            source, target = j, i
        max_delta = min(h[source], 1 - h[target])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        # update
        h_new = h.copy()
        h_new[source] -= delta
        h_new[target] += delta
        # compute new objective
        val = compute_c5(h_new)
        # accept if better or with some probability
        if val < best_val:
            best = h_new
            best_val = val
            h = h_new
        else:
            # maybe accept with small probability decreasing over time
            # temperature schedule: T = initial_T * (1 - (elapsed/ time_budget))
            elapsed = time.time() - start_time
            T = max(1e-4, (time_budget - elapsed) / time_budget)  # decreasing
            if np.random.rand() < np.exp(-(val - best_val) / (T + 1e-8)):
                h = h_new
    return best, best_val

def project_to_bounds_sum(v, S):

    low = -np.max(v) - 1.0

    while sum_clip(low) < S:
        low -= 1.0
    # Expand high upwards until sum_clip(high) < S:
    while sum_clip(high) > S:
        high += 1.0

        else:
            high = mid

import time
import numpy as np

def _project_to_bounds_sum(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the set {x | sum(x)=S, 0<=x<=1}."""
    # binary search for tau such that sum(clip(v - tau, 0, 1)) = S
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0

    def _sum_clip(tau):
        return np.clip(v - tau, 0.0, 1.0).sum()

    # Ensure interval brackets the solution
    while _sum_clip(low) < S:
        low -= 1.0
    while _sum_clip(high) > S:
        high += 1.0

    for _ in range(60):
        mid = (low + high) * 0.5
        s = _sum_clip(mid)
        if s > S:
            low = mid
        else:
            high = mid
    tau = (low + high) * 0.5
    return np.clip(v - tau, 0.0, 1.0)


def _shift_fill(arr: np.ndarray, shift: int) -> np.ndarray:
    """Shift array with zero padding (no wrap). Positive shift moves elements to the right."""
    n = arr.shape[0]
    if shift > 0:
        return np.concatenate([arr[shift:], np.zeros(shift, dtype=arr.dtype)])
    if shift < 0:
        s = -shift
        return np.concatenate([np.zeros(s, dtype=arr.dtype), arr[:n - s]])
    return arr.copy()


def _neighbor_sum(arr: np.ndarray, lag: int) -> np.ndarray:
    """For each index i, sum of arr[i+lag] and arr[i-lag] (zero when out of range)."""
    return _shift_fill(arr, lag) + _shift_fill(arr, -lag)


def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a given step function sample vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _random_init(n: int) -> np.ndarray:
    """Generate a random feasible h (values in [0,1] summing to n/2)."""
    v = np.random.rand(n)
    return _project_to_bounds_sum(v, n / 2.0)


def _local_optimize(start_h: np.ndarray, max_seconds: float, start_time: float):
    """Simple projected-subgradient descent with occasional random moves."""
    n = start_h.shape[0]
    S = n / 2.0
    cur = start_h.copy()
    cur_val = _compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    eps = 1e-12

    while time.time() - start_time < max_seconds:
        # worst lag (max overlap)
        corr = np.correlate(cur, 1.0 - cur, mode="full") * (2.0 / n)
        k = np.argmax(corr) - (n - 1)

        # neighbor sums give a subgradient direction
        ns = _neighbor_sum(cur, k)

        # feasible indices
        mask_src = cur > eps
        mask_tgt = cur < 1.0 - eps
        if not np.any(mask_src) or not np.any(mask_tgt):
            break

        src_idx = np.where(mask_src)[0][np.argmin(ns[mask_src])]
        tgt_idx = np.where(mask_tgt)[0][np.argmax(ns[mask_tgt])]

        max_delta = min(cur[src_idx], 1.0 - cur[tgt_idx])
        if max_delta <= eps or ns[tgt_idx] <= ns[src_idx]:
            # fallback random move
            i, j = np.random.randint(n), np.random.randint(n)
            if i == j or cur[i] <= eps or cur[j] >= 1.0 - eps:
                continue
            max_delta = min(cur[i], 1.0 - cur[j])
            delta = max_delta * np.random.rand()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            delta = max_delta * 0.5 * np.random.rand()
            new = cur.copy()
            new[src_idx] -= delta
            new[tgt_idx] += delta

        new_val = _compute_c5(new)

        # accept if improvement (no uphill moves to keep monotonic)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val

    # final projection to guarantee exact constraints
    best = _project_to_bounds_sum(best, S)
    return best, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    start = time.time()

    # --- initial construction -------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n = init_h.shape[0]
        cur_h = _project_to_bounds_sum(init_h, n / 2.0)
    except Exception:
        # fallback: moderate size even integer for ease of binary sum
        n = 120
        cur_h = _random_init(n)

    # --- main optimization (single start) ------------------------------------
    remaining = max(0.0, budget_s - (time.time() - start))
    best_h, best_val = _local_optimize(cur_h, remaining, time.time())

    # --- additional short random restarts if time permits --------------------
    while True:
        elapsed = time.time() - start
        time_left = budget_s - elapsed
        if time_left < 2.0:
            break
        chunk = min(5.0, time_left)
        rnd_h = _random_init(n)
        cand_h, cand_val = _local_optimize(rnd_h, chunk, time.time())
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

    # final sanity projection and evaluation
    best_h = _project_to_bounds_sum(best_h, n / 2.0)
    final_val = _compute_c5(best_h)

    return best_h, float(final_val), n
