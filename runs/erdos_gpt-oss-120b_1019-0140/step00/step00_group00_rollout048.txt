    target_sum = n_points / 2.0
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # Adjust the first entry; the perturbation is tiny and remains in [0,1].
        h[0] += diff
        # Re‑clip just in case.
        h[0] = min(max(h[0], 0.0), 1.0)

    # Compute the overlap bound C5 using the same formula as the evaluator.
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5_bound = float(np.max(corr))

    # Ensure we respect the time budget (trivial here, but idiomatic).
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # In the unlikely event of a timeout, return the current solution.
        pass

    return h, c5_bound, n_points
