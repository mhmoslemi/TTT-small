def grad_softmin(h, alpha):
    n = len(h)
    corr = np.correlate(h, h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(alpha * (max_corr - corr))
    # zero center
    w[n-1] = 0.0
    grad = np.zeros_like(h)
    for s in range(1, n):
        weight = w[n-1 + s]  # corr symmetric weight same for -s
        if weight == 0:
            continue
        # contributions
        grad[:n-s] += weight * h[s:]
        grad[s:]   += weight * h[:n-s]
    return grad
```

```
def project_onto_box_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    # For lambda decreasing, x_i = clip(v_i - lam, lower, upper)
    lam_low = -1e6
    lam_high = 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high)/2
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - (lam_low + lam_high)/2, lower, upper)
```

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    lam_low = -1e6
    lam_high = 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - (lam_low + lam_high) / 2.0, lower, upper)

def grad_softmin_corr(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    # shift to avoid underflow, using max corr
    max_corr = np.max(corr)
    w = np.exp(alpha * (max_corr - corr))
    # zero out self-correlation
    w[n-1] = 0.0
    grad = np.zeros_like(h)
    # loop over positive lags
    for s in range(1, n):
        wt = w[n-1 + s]
        if wt == 0.0:
            continue
        grad[:n-s] += wt * h[s:]
        grad[s:]   += wt * h[:n-s]
    return grad

import time, math
import numpy as np

def compute_c5(h):
    """Integral approximation of max_k ∫ h(x)(1‑h(x+k)) dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    """
    Euclidean projection of v onto {x | lower ≤ x_i ≤ upper, Σ x_i = target}.
    Uses simple bisection on the Lagrange multiplier.
    """
    lam_low, lam_high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - (lam_low + lam_high) * 0.5, lower, upper)


def grad_softmin_corr(h, alpha):
    """
    Gradient of a soft‑minimum of the autocorrelation of h.
    Larger weight is given to smaller correlation values.
    """
    n = h.shape[0]
    corr = np.correlate(h, h, mode="full")
    max_corr = np.max(corr)
    # weights favour lower correlation (w = exp(alpha*(max‑corr)))
    w = np.exp(alpha * (max_corr - corr))
    w[n - 1] = 0.0                     # exclude lag 0
    grad = np.zeros_like(h)
    # accumulate contributions for each positive lag
    for s in range(1, n):
        wk = w[n - 1 + s]
        if wk == 0.0:
            continue
        grad[: n - s] += wk * h[s:]
        grad[s:]       += wk * h[: n - s]
    return grad
