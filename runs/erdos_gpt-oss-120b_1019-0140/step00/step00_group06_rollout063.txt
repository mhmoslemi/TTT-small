    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)

    sum_exp = np.sum(exp_term)

    F = np.zeros_like(h)
    B = np.zeros_like(h)

            B[s:] += weight * h[:n - s] # because backward for shift s is forward for -s? Let's compute.

import numpy as np
import time, math

def project_to_box_and_sum(v, target):
    low, high = -10.0, 10.0
    for _ in range(60):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smooth_max_and_grad(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_term = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    F = np.zeros_like(h)
    B = np.zeros_like(h)
    # iterate shifts
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            F[:n - s] += weight * h[s:]
            B[s:] += weight * h[:n - s]
        else:
            t = -s
            F[t:] += weight * h[:n - t]
            B[:n - t] += weight * h[t:]
    grad = (1.0 - F - B) * (2.0 / n)
    f = (max_corr + np.log(sum_exp) / alpha) * (2.0 / n)
    return f, grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_vec = np.asarray(initial_h_values, dtype=np.float64)
        base_n = init_vec.shape[0]
    else:
        init_vec = None
        base_n = 100
    best_h = None
    best_val = np.inf
    ns = [base_n]
    if base_n * 2 <= 500:
        ns.append(base_n * 2)
    if base_n * 4 <= 800:
        ns.append(base_n * 4)
    ns = sorted(set(ns))
    for n in ns:
        target = n / 2.0
        if init_vec is not None and n == base_n:
            h0 = init_vec.copy()
        elif init_vec is not None:
            old_x = np.linspace(0.0, 2.0, base_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(new_x, old_x, init_vec)
        else:
            h0 = np.random.rand(n)
        h0 = project_to_box_and_sum(h0, target)
        schedule = [(400, 5.0, 0.3), (600, 15.0, 0.15), (600, 50.0, 0.08)]
        restarts = 2
        for r in range(restarts):
            h = h0.copy()
            if r > 0:
                h = np.random.rand(n)
                h = project_to_box_and_sum(h, target)
            it = 0
            for its, alpha, step0 in schedule:
                for _ in range(its):
                    if time.time() - start_time >= budget_s:
                        break
                    f_val, grad = smooth_max_and_grad(h, alpha)
                    if f_val < best_val:
                        best_val = f_val
                        best_h = h.copy()
                    step = step0 / math.sqrt(it + 1)
                    y = h - step * grad
                    h = project_to_box_and_sum(y, target)
                    it += 1
                if time.time() - start_time >= budget_s:
                    break
            c5_exact = compute_c5(h)
            if c5_exact < best_val:
                best_val = c5_exact
                best_h = h.copy()
        init_vec = best_h.copy() if best_h is not None else None
        if time.time() - start_time >= budget_s:
            break
    if best_h is None:
        n = base_n

def _project(v, target):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _smooth_max_and_grad(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    expv = np.exp(alpha * (corr - m))
    s = np.sum(expv)
    w = expv / s
    F = np.zeros_like(h)
    B = np.zeros_like(h)
    for idx, wk in enumerate(w):

            continue

            F[: n - shift] += wk * h[shift:]
            B[shift:] += wk * h[: n - shift]

            B[: n - t] += wk * h[t:]
    grad = (1.0 - F - B) * (2.0 / n)
