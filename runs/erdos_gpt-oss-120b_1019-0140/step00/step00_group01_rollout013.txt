    rng.shuffle(base)

def local_search(init_h, init_c5, rng, target_sum, budget_s):

    no_improve = 0

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(v, target):
    lower = -1.0
    upper = 1.0
    for _ in range(60):
        lam = (lower + upper) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lower = lam
        else:
            upper = lam
    return np.clip(v - (lower + upper) / 2.0, 0.0, 1.0)

def random_feasible(rng, n):
    target = n / 2.0
    h = np.zeros(n)
    k = int(np.floor(target))
    h[:k] = 1.0
    if target - k > 1e-12:
        h[k] = target - k
    rng.shuffle(h)
    return h

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(v, target):
    """Project v onto the box [0,1] with sum equal to target."""
    lo, hi = -1.0, 1.0  # bounds for lambda
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def random_feasible(rng, n):
    """Return a feasible h with ∑h = n/2, values in [0,1] (binary + possible half)."""
    target = n / 2.0
    h = np.zeros(n)
    k = int(np.floor(target))
    if k > 0:
        h[:k] = 1.0
    if target - k > 1e-12:
        h[k] = target - k
    rng.shuffle(h)
    return h

def local_search(init_h, init_c5, rng, target, budget):
    """Simple simulated‑annealing style refinement."""
    n = len(init_h)
    best_h = init_h.copy()
    best_c5 = init_c5
    cur_h = init_h.copy()
    cur_c5 = init_c5
    start = time.time()
    temp = 0.01
    decay = 0.9995
    no_imp = 0
    max_no_imp = 5000
    while time.time() - start < budget:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_decr = cur_h[i]
        max_incr = 1.0 - cur_h[j]
        delta_max = min(max_decr, max_incr)
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c5 = compute_c5(cur_h)
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            cur_h[i] += delta
            cur_h[j] -= delta
            no_imp += 1
        temp *= decay
        if no_imp > max_no_imp:
            cur_h = random_feasible(rng, n)
            cur_h = project_to_simplex(cur_h, target)
            cur_c5 = compute_c5(cur_h)
            no_imp = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback size if no parent is supplied
        h0 = random_feasible(rng, 101)
    n = len(h0)
    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    if abs(h0.sum() - target) > 1e-12:
        h0 = project_to_simplex(h0, target)

    # --- global optimisation (DE) ----------------------------------
    bounds = [(0.0, 1.0)] * n

    def de_obj(x):
        y = np.clip(x, 0.0, 1.0)
        if abs(y.sum() - target) > 1e-9:
            y = project_to_simplex(y, target)
        corr = np.correlate(y, 1.0 - y, mode='full')
        return float(np.max(corr) * (2.0 / n))

    # modest iteration budget – cheap for n≈50
    maxiter = 200
    popsize = max(5, n // 5)
    result = differential_evolution(
        de_obj,
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        seed=seed,
        polish=False,
    )
    h_opt = project_to_simplex(result.x, target)
    c5_opt = compute_c5(h_opt)

    # --- refinement with local search --------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        h_opt, c5_opt = local_search(
            h_opt, c5_opt, rng, target, remaining * 0.9
        )

    # final safety projection and evaluation
    h_opt = project_to_simplex(h_opt, target)
    c5_opt = compute_c5(h_opt)

    return h_opt, c5_opt, n
