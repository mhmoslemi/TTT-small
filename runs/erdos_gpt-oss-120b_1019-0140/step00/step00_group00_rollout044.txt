import numpy as np
import time

def _project_to_feasible(v, target):
    """Project v onto {0 <= x_i <= 1, sum x_i = target}."""
    x = np.clip(v, 0.0, 1.0)
    tol = 1e-9
    for _ in range(100):
        s = x.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = x < 1 - tol
        else:
            free = x > tol
        if not np.any(free):
            break
        adj = diff / np.count_nonzero(free)
        x[free] += adj
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    s = x.sum()
    if abs(s - target) > 1e-6:
        free = (x > 0) & (x < 1)
        if np.any(free):
            adj = (target - s) / np.count_nonzero(free)
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
        else:
            # scale uniformly as last resort
            if s != 0:
                x *= target / s
                x = np.clip(x, 0.0, 1.0)
    return x

def _max_overlap(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx for the discrete sample."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _gradient(h, alpha):
    """Gradient of the smooth max objective (log‑sum‑exp) w.r.t. h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    p = exp_vals / np.sum(exp_vals)          # softmax weights, sum to 1

    offsets = np.arange(-(N - 1), N)         # length 2N‑1
    grad = np.ones(N)                        # the constant part (dx factor added later)

    for off, pt in zip(offsets, p):

            continue

            # forward contribution: h[i+off]
            if off < N:

                grad[s:] -= pt * h[:N - s]   # i+off = i‑s
                grad[:N - s] -= pt * h[s:]   # i‑off = i+s
    grad *= dx
