def project_onto_simplex_bounded(v, S):
    # v: array of shape (n,)
    # Guarantee each entry in [0,1] and sum = S
    # Use binary search for tau: find tau s.t sum_i clip(v_i - tau, 0, 1) = S.
    # Since sum_i clip(v_i - tau, 0, 1) is decreasing in tau.
    lo = -1.0  # v_i - lo = v_i + 1 >= 1 ensures clip=1 for all; sum = n
    hi = 2.0   # v_i - hi <= -? ensures clip=0; sum=0. We'll just set hi = np.max(v)+1.
    for _ in range(50):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0, 1)
        cur_sum = x.sum()
        if cur_sum > S:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    return x

def hill_climb(h, max_iter, time_limit, start_time):
    n = len(h)
    current = h.copy()
    cur_score = evaluate(current, n)
    best_h = current.copy()
    best_score = cur_score
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start_time > time_limit: break
        # propose move
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = current[i], current[j]
        # compute allowed transfer bounds
        max_pos = min(1 - hi, hj)
        max_neg = min(hi, 1 - hj)
        if max_pos == 0 and max_neg == 0:
            continue
        # choose t uniformly in [-max_neg, max_pos]
        t = np.random.uniform(-max_neg, max_pos)
        if abs(t) < 1e-12:
            continue
        new = current.copy()
        new[i] = hi + t
        new[j] = hj - t
        # evaluate
        new_score = evaluate(new, n)
        # Accept if better
        if new_score < cur_score - 1e-12:
            current = new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score - 1e-12:
                best_h = new.copy()
                best_score = new_score
        else:
            # optionally accept worse with small probability based on temperature
            # temperature schedule
            if temp_schedule is not None:
                T = temp_schedule(it)
                if T > 0 and np.random.rand() < np.exp(-(new_score - cur_score)/T):
                    current = new
                    cur_score = new_score
            no_improve += 1
        if no_improve >= max_no_improve:
            break
    return best_h, best_score

    for _ in range(60):

import numpy as np
import time

def _project_onto_simplex_bounded(v, S):
    """Project v onto the set {x: 0<=x_i<=1, sum x_i = S}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        cur = x.sum()
        if cur > S:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def _c5_bound(h):
    """Return the C5 bound for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _build_initial_h(n):
    """Construct an initial feasible vector of length n."""
    if 'initial_h_values' in globals():
        parent = initial_h_values.astype(np.float64)
        n0 = parent.shape[0]
        if n0 == n:
            h = parent.copy()
        else:
            x_parent = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_target = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_target, x_parent, parent)
    else:
        h = np.full(n, 0.5, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _project_onto_simplex_bounded(h, target)
    return h

def _hill_climb(start_h, max_iter, start_time, budget_s):
    """Simple hill‑climbing with pairwise transfers that preserve the sum."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_score = _c5_bound(cur)
    best = cur.copy()
    best_score = cur_score
    stagn = 0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        i, j = np.random.choice(n, size=2, replace=False)
        hi, hj = cur[i], cur[j]
        max_pos = min(1.0 - hi, hj)          # transfer from j to i
        max_neg = min(hi, 1.0 - hj)          # transfer from i to j
        if max_pos == 0.0 and max_neg == 0.0:
            continue
        t = np.random.uniform(-max_neg, max_pos)
        if abs(t) < 1e-14:
            continue
        new = cur.copy()
        new[i] = hi + t
        new[j] = hj - t
        new_score = _c5_bound(new)
        if new_score < cur_score - 1e-12:
            cur = new
            cur_score = new_score
            stagn = 0
            if new_score < best_score - 1e-12:
                best = new.copy()
                best_score = new_score
        else:
            stagn += 1
        if stagn >= 3000:
            break
    return best, best_score

    # Determine resolutions to explore
    if 'initial_h_values' in globals():
        base_n = int(initial_h_values.shape[0])
        ns = [base_n, 150, 240]
    else:
        base_n = None
        ns = [120, 200, 300]
    ns = [n for n in sorted(set(ns)) if n <= 500]
    for n in ns:
        init_vec = _build_initial_h(n)
        # Number of independent restarts for this size
        restarts = 5 if n <= 200 else 3
        for _ in range(restarts):
            # Random perturbation around the initializer
            pert = np.random.uniform(-0.05, 0.05, size=n)
            h0 = init_vec + pert
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = _project_onto_simplex_bounded(h0, n / 2.0)
            # Allocate iterations proportionally to remaining time
            elapsed = time.time() - start
            remaining = budget_s - elapsed
            if remaining <= 5.0:
                break
            max_iter = int(20000 * (remaining / budget_s))
            max_iter = max(2000, min(max_iter, 12000))
            local_h, local_score = _hill_climb(h0, max_iter, start, budget_s)
            if local_score < best_score:
                best_score = local_score
                best_h = local_h.copy()
            if time.time() - start > budget_s:
                break
        if time.time() - start > budget_s:
            break
    # Fallback to the parent if nothing better was found
    if best_h is None:
        if base_n is not None:
            best_h = _build_initial_h(base_n)
            best_score = _c5_bound(best_h)
        else:
            raise RuntimeError("Unable to produce a feasible solution")
    best_h = best_h.astype(np.float64)
    final_score = _c5_bound(best_h)
    return best_h, final_score, best_h.shape[0]
