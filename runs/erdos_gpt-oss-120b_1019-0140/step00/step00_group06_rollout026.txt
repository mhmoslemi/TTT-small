import cvxpy as cp
def project_feasible(v, target):
    n = len(v)
    w = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(w - v)),
                      [cp.sum(w) == target,
                       w >= 0,
                       w <= 1])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return w.value

def project_feasible(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:

    return w

import numpy as np
import time
import math

# ---------- Helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Correlation based C5 evaluation (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / h.size))


def evaluate_h(h: np.ndarray) -> float:
    """Wrapper that prefers the pre‑imported evaluator if present."""
    try:
        return float(evaluate_erdos_solution(h))  # type: ignore[name-defined]
    except Exception:
        return compute_c5(h)


def project_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector `v` onto the hyperplane sum(w)=target with box constraints 0≤w≤1.
    The solution has the form w_i = clip(v_i - λ, 0, 1) for a suitable λ.
    """
    lo, hi = -1.0, 2.0  # bounds that guarantee sum ranges from n to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:
            break
    return w


def upsample_vector(prev_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from a coarse vector to a finer grid, then projection."""
    old_n = prev_h.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, prev_h)
    return project_feasible(new_h, new_n / 2.0)


def optimize_h(initial_h: np.ndarray, n: int, time_limit: float, rng: np.random.Generator):
    """Hill‑climbing / simulated‑annealing optimizer for a fixed resolution."""
    # ensure feasibility
    target = n / 2.0
    h = np.clip(initial_h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-9:
        h = project_feasible(h, target)

    best_h = h.copy()
    best_score = evaluate_h(best_h)
    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    step_factor = 0.5
    it = 0

    while True:
        if time.time() - start >= time_limit:
            break

        # slowly shrink step size
        if it and it % 2000 == 0:
            step_factor = max(0.01, step_factor * 0.9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # mass transfer i -> j respecting bounds
        max_take = cur_h[i]
        max_give = 1.0 - cur_h[j]
        max_delta = max_take if max_take < max_give else max_give
        if max_delta <= 0:
            continue

        delta = rng.random() * max_delta * step_factor
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_score = evaluate_h(new_h)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start
            temp = max(1e-4, 1.0 - elapsed / time_limit)
            prob = math.exp((cur_score - new_score) / (temp * 0.01))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()

        # occasional random perturbation to diversify
        if it and it % 5000 == 0:
            pert = (rng.random(n) - 0.5) * 0.02  # roughly ±0.01
            cand = cur_h + pert
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_feasible(cand, target)
            cand_score = evaluate_h(cand)
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                cur_h = cand
                cur_score = cand_score

        it += 1

    return best_h, best_score

    else:
        n0 = 95
        init = np.full(n0, 0.5, dtype=np.float64)
        init = project_feasible(init, n0 / 2.0)

    # ----- Multi‑resolution schedule -----
    levels = [n0]
    max_n = 380
    while levels[-1] < max_n:
        nxt = min(levels[-1] * 2, max_n)
        if nxt == levels[-1]:
            break
        levels.append(int(nxt))

    # Allocate a fraction of the overall budget to each level
    if len(levels) == 1:
        fractions = [1.0]
    elif len(levels) == 2:
        fractions = [0.35, 0.65]
    else:
        fractions = [0.20, 0.30, 0.50]  # sums to 1

    best_h_global = None
    best_score_global = float('inf')
    prev_best = init

    for idx, n in enumerate(levels):
        elapsed = time.time() - start_total
        time_left = budget_s - elapsed
        if time_left <= 0:
            break

        # allocate time for this resolution
        alloc = fractions[idx] * budget_s
        time_limit = min(alloc, time_left * 0.95)

        if n != prev_best.shape[0]:
            h0 = upsample_vector(prev_best, n)
        else:
            h0 = prev_best.copy()

        h_opt, score_opt = optimize_h(h0, n, time_limit, rng)

        if score_opt < best_score_global:
            best_score_global = score_opt
            best_h_global = h_opt.copy()

        prev_best = h_opt

    # Fallback safety
    if best_h_global is None:
        best_h_global = init.copy()
        best_score_global = evaluate_h(best_h_global)

    n_out = best_h_global.shape[0]
    best_h_global = np.clip(best_h_global, 0.0, 1.0)
    best_h_global = project_feasible(best_h_global, n_out / 2.0)
    best_score_global = evaluate_h(best_h_global)

    return best_h_global, float(best_score_global), int(n_out)
