def normalize_h(h):
    n = len(h)
    target_sum = n / 2
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if current_sum == 0:
        # all zeros? set uniform
        h[:] = target_sum / n
        return h
    factor = target_sum / current_sum
    h = h * factor
    h = np.clip(h, 0, 1)
    # after clipping, may need small adjustment
    current_sum = h.sum()
    diff = target_sum - current_sum
    if abs(diff) > 1e-12:
        # distribute diff to free entries
        free = (h > 1e-12) & (h < 1-1e-12)
        nfree = free.sum()
        if nfree > 0:
            h[free] += diff / nfree
            # clip again
            h = np.clip(h, 0, 1)
        else:
            # fallback: adjust uniformly, may violate bounds slightly but fine.
            h += diff / n
            h = np.clip(h, 0, 1)
    # final check
    # enforce exact sum via scaling small amount
    # This may cause slight violation; but we trust this fine.
    return h

def propose_move(h, step_size, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Randomly decide direction: either increase i, decrease j, or opposite
    if rng.random() < 0.5:
        # inc i, dec j
        inc_idx, dec_idx = i, j
    else:
        inc_idx, dec_idx = j, i
    # Determine max delta
    max_inc = min(step_size, 1.0 - h[inc_idx])
    max_dec = min(step_size, h[dec_idx])
    max_delta = min(max_inc, max_dec)
    if max_delta <= 0:
        return None, None, None
    delta = rng.uniform(0, max_delta)
    return inc_idx, dec_idx, delta

def propose_move(h, step, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Random direction
    if rng.random() < 0.5:
        inc_idx, dec_idx = i, j
    else:
        inc_idx, dec_idx = j, i
    max_inc = min(step, 1.0 - h[inc_idx])
    max_dec = min(step, h[dec_idx])
    max_delta = min(max_inc, max_dec)

        return None, None, None
    delta = rng.uniform(0, max_delta)
    return inc_idx, dec_idx, delta

import numpy as np
import time
import math

def compute_c5(h):
    n = h.shape[0]
    N = 2 * n
    fh = np.fft.fft(h, n=N)
    fg = np.fft.fft(1.0 - h, n=N)
    corr = np.fft.ifft(fh * np.conj(fg)).real
    corr = corr[:2 * n - 1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def normalize_h(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target / n
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            i = free[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        # final tiny correction if needed
        s2 = h.sum()
        if abs(target - s2) > 1e-10:
            h += (target - s2) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def propose_move(h, step, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if rng.random() < 0.5:
        inc_idx, dec_idx = i, j
    else:
        inc_idx, dec_idx = j, i
    max_inc = min(step, 1.0 - h[inc_idx])
    max_dec = min(step, h[dec_idx])
    max_delta = min(max_inc, max_dec)
    if max_delta <= 0.0:
        return None, None, None
    delta = rng.uniform(0.0, max_delta)
    return inc_idx, dec_idx, delta

def local_optimize(h_initial, deadline, rng):
    cur_h = h_initial.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    iteration = 0
    init_temp = 0.005
    while time.time() < deadline:
        step = max(0.0005, 0.2 * (0.99 ** iteration))
        inc_idx, dec_idx, delta = propose_move(cur_h, step, rng)
        if delta is None:
            iteration += 1
            continue
        new_h = cur_h.copy()
        new_h[inc_idx] += delta
        new_h[dec_idx] -= delta
        new_score = compute_c5(new_h)
        if new_score < cur_score - 1e-12:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h.copy()
                best_score = new_score
        else:
            T = init_temp * (0.995 ** iteration)
            if T > 1e-6:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    cur_h = new_h
                    cur_score = new_score
        iteration += 1
    return best_h, best_score

def upsample_h(old_h, new_n):
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = normalize_h(new_h)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        h0 = normalize_h(h0)
    else:
        n0 = 100
        h0 = np.zeros(n0, dtype=np.float64)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        h0[idx] = 1.0
    # stage 1
    deadline1 = start + budget_s * 0.6
    best_h, best_score = local_optimize(h0, deadline1, rng)
    # stage 2 if time permits
    elapsed = time.time() - start
    if budget_s - elapsed > 10.0:
        new_n = min(len(best_h) * 2, 380)
        h_up = upsample_h(best_h, new_n)
        deadline2 = start + budget_s - 0.05
        up_h, up_score = local_optimize(h_up, deadline2, rng)
        if up_score < best_score:
            best_h, best_score = up_h, up_score
    best_h = normalize_h(best_h)
    final_score = compute_c5(best_h)
    return best_h.astype(np.float64), final_score, len(best_h)

        h0 = np.zeros(n0, dtype=np.float64)
        idx = rng.choice(n0, size=n0 // 2, replace=False)
        h0[idx] = 1.0
    deadline1 = start + budget_s * 0.6
    best_h, best_score = local_optimize(h0, deadline1, rng)
    elapsed = time.time() - start
    if budget_s - elapsed > 10.0:
        new_n = min(len(best_h) * 2, 380)
        h_up = upsample_h(best_h, new_n)
        deadline2 = start + budget_s - 0.05
        up_h, up_score = local_optimize(h_up, deadline2, rng)
        if up_score < best_score:
            best_h, best_score = up_h, up_score
    best_h = normalize_h(best_h)
    final_score = compute_c5(best_h)
    return best_h.astype(np.float64), final_score, len(best_h)
