import time
import numpy as np

def _shift(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift array by k with zero padding (positive k shifts right)."""
    n = arr.shape[0]
    if k > 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[: n - k]
        return out
    if k < 0:
        k = -k
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    return arr.copy()


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return C5 for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx
