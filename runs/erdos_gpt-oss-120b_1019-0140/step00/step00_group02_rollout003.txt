def directed_move(h):
    n = len(h)
    # compute full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # compute weight array
    if shift >= 0:
        # weight_i = (1 - h[i+shift]) for i in 0..n-1-shift, else weight_i = 0 for i > n-1-shift
        weight = np.zeros(n)
        weight[:n-shift] = 1 - h[shift:]  # because i+shift -> indices shift..n-1
    else:
        # shift negative, i+shift < i, valid indices i from -shift .. n-1
        weight = np.zeros(n)
        weight[-shift:] = 1 - h[:n+shift]  # shift negative so war
    # Now find i_max = argmax(weight) with h[i] > 0 (so we can decrease)
    # find i_min = argmin(weight) with h[i] < 1 (so we can increase)
    # Might have weight zeros for many indices.
    # Use masks:
    mask_decrease = h > 1e-12
    mask_increase = h < 1 - 1e-12
    if not np.any(mask_decrease) or not np.any(mask_increase):
        return h.copy()
    # Values:
    # weight for decrease candidates maybe high; choose index with max weight among mask_decrease
    weighted_decrease = weight * mask_decrease
    i_max = np.argmax(weighted_decrease)
    # For increase choose index with minimal weight among mask_increase.
    weighted_increase = weight.copy()
    # For indices not allowed to increase, set to large number
    weighted_increase[~mask_increase] = np.inf
    i_min = np.argmin(weighted_increase)
    # compute max delta possible: limited by h[i_max] (cannot go below 0)
    # So max decrease = h[i_max]; max increase = 1 - h[i_min]
    max_delta = min(h[i_max], 1 - h[i_min])
    if max_delta <= 1e-12:
        return h.copy()
    # Choose delta as random fraction of max_delta, e.g., uniform(0, max_delta)
    delta = rng.uniform(0, max_delta)
    h_new = h.copy()
    h_new[i_max] -= delta
    h_new[i_min] += delta
    return h_new

def random_pair_move(h):

    """Swap a random amount of mass between two coordinates, preserving sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lower = max(-hi, -(1 - hj))
    upper = min(1 - hi, hj)
    if upper <= lower:
        return h.copy()
    delta = rng.uniform(lower, upper)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def directed_move(h, rng):
    """Move mass from a high‑weight to a low‑weight position for the current worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)

    weight = np.zeros(n, dtype=float)
    if shift >= 0:
        weight[: n - shift] = 1 - h[shift:]
    else:
        weight[-shift:] = 1 - h[: n + shift]

    mask_decrease = h > 1e-12
    mask_increase = h < 1 - 1e-12
    if not np.any(mask_decrease) or not np.any(mask_increase):
        return h.copy()

    weighted_decrease = weight * mask_decrease
    i_max = int(np.argmax(weighted_decrease))

    weighted_increase = weight.copy()
    weighted_increase[~mask_increase] = np.inf
    i_min = int(np.argmin(weighted_increase))

    max_delta = min(h[i_max], 1 - h[i_min])
    if max_delta <= 1e-12:
        return h.copy()
    delta = rng.uniform(0, max_delta)

    h_new = h.copy()
    h_new[i_max] -= delta
    h_new[i_min] += delta
    return h_new

def repair_feasible(h, target_sum):
    """Clip to [0,1] and adjust the sum to the exact target."""
    np.clip(h, 0.0, 1.0, out=h)
    residual = target_sum - h.sum()
    if abs(residual) < 1e-12:
        return h

    free = np.where((h > 0) & (h < 1))[0]
    if free.size == 0:
        # all entries are 0 or 1; adjust a single index safely
        idx = np.argmax(h) if residual < 0 else np.argmin(h)
        h[idx] = min(max(h[idx] + residual, 0.0), 1.0)
        return h

    delta = residual / free.size
    for idx in free:
        val = h[idx] + delta
        if val < 0:
            val = 0.0
        elif val > 1:
            val = 1.0
        h[idx] = val

    # tiny correction if clipping disturbed the total
    new_res = target_sum - h.sum()
    if abs(new_res) > 1e-12:
        idx = free[0]
        h[idx] = min(max(h[idx] + new_res, 0.0), 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # --- initial feasible vector ------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = np.full(45, 0.5, dtype=float)

    n = int(init_h.shape[0])
    target_sum = n / 2.0
    init_h = repair_feasible(init_h, target_sum)

    dx = 2.0 / n

    best_h = init_h.copy()
    best_score = compute_c5(best_h, dx)
    current_h = best_h.copy()
    current_score = best_score

    # --- simulated‑annealing loop -----------------------------------------------
    start = time.time()
    temp = 1.0
    cooling = 0.9997
    p_direct = 0.3
    no_improve = 0
    target_bound = 0.3808

    while True:
        if time.time() - start >= budget_s - 0.1:
            break

        # propose a neighbour
        if rng.random() < p_direct:
            new_h = directed_move(current_h, rng)
        else:
            new_h = random_pair_move(current_h, rng)

        new_h = repair_feasible(new_h, target_sum)
        new_score = compute_c5(new_h, dx)

        # acceptance test (simulated annealing)
        accept = False
        if new_score < current_score:
            accept = True
        else:
            prob = math.exp((current_score - new_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            current_h = new_h
            current_score = new_score

        # record best solution
        if new_score < best_score - 1e-12:
            best_h = new_h
            best_score = new_score
            no_improve = 0
            if best_score <= target_bound:
                break
        else:
            no_improve += 1

        # restart if stuck for a long time
        if no_improve > 50000:
            current_h = best_h.copy()
            for _ in range(10):
                current_h = random_pair_move(current_h, rng)
            current_h = repair_feasible(current_h, target_sum)
            current_score = compute_c5(current_h, dx)
            no_improve = 0

        temp *= cooling

    # final safeguard and output
    best_h = repair_feasible(best_h, target_sum)
    final_score = compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_score), int(n))
