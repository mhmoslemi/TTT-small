        # cooling schedule

        frac = elapsed / budget_s
        T = max(1e-5, (0.2 * (1 - frac) + 1e-5)) # Some temperature
        # propose move

        while i == j:

        lo = max(-cur_h[i], cur_h[j] - 1.0)
        hi = min(1.0 - cur_h[i], cur_h[j])
        if lo >= hi:

        new_h = cur_h.copy()

        new_h[j] -= delta
        new_score = compute_c5(new_h)

        # occasional random restart if stuck
        if iter % 5000 == 0 and iter > 0:

            pass
        iter += 1
    return (best_h.tolist(), float(best_score), n)

    # returns vector x = clip(v + lambda, 0,1) such that sum(x) = target.
    lo = -np.max(v)
    hi = 1 - np.min(v)

            hi = lam

            lo = lam

def project_to_sum(v, target):
    lo = -np.max(v) - 1e-12
    hi = 1.0 - np.min(v) + 1e-12
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # adjust residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-saturated entries
        mask = (x > 0.0) & (x < 1.0)
        idxs = np.where(mask)[0]
        if idxs.size > 0:

        else:
            # all at bounds, adjust first index within tolerance
            i = 0
            x[i] = min(max(x[i] + diff, 0.0), 1.0)

import time
import numpy as np

def project_to_sum(v, target):
    """Project vector v onto the box [0,1] while fixing sum(v)=target."""
    # Binary search for lambda such that sum(clamp(v+lam)) = target
    lo = -np.max(v) - 1e-12
    hi = 1.0 - np.min(v) + 1e-12
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        idxs = np.nonzero(mask)[0]
        if idxs.size > 0:
            i = idxs[0]
            x[i] = min(max(x[i] + diff, 0.0), 1.0)
        else:
            x[0] = min(max(x[0] + diff, 0.0), 1.0)
    return x

def random_feasible_vector(n):
    """Generate a random feasible vector of length n (values in [0,1], sum=n/2)."""
    v = np.random.beta(0.5, 0.5, size=n)
    return project_to_sum(v, n / 2.0)

def compute_c5(h):
    """Return the overlap constant C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def descent_direction(h, shift):
    """
    Gradient (negated) of the worst‑shift correlation term.
    Returns a direction that reduces the chosen shift's overlap.
    """
    n = h.shape[0]
    d = -np.ones(n)               # baseline -1
    if shift == 0:
        d += 2.0 * h              # both neighbours are h[i]
        return d
    if shift > 0:
        s = shift
        # left neighbour (i-s) exists for i>=s
        d[s:] += h[:-s]
        # right neighbour (i+s) exists for i < n-s
        d[:n - s] += h[s:]
    else:
        s = -shift
        # left neighbour (i-s) = i+s exists for i < n-s
        d[:n - s] += h[s:]
        # right neighbour (i+s) = i-s exists for i >= s
        d[s:] += h[:n - s]
    return d

    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h (possibly at higher resolution)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h_small = np.asarray(initial_h_values, dtype=float)
        n_small = h_small.size
        # use double resolution for more flexibility
        n = max(20, n_small * 2)
        xs = np.linspace(0.0, 2.0, n_small, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xt, xs, h_small)
        h = project_to_sum(h, n / 2.0)
    else:
        n = 84
        h = random_feasible_vector(n)

    best_h = h.copy()
    best_score = compute_c5(h)

    # ------------------------------------------------------------------
    # Main optimisation loop (gradient descent + random swaps)
    # ------------------------------------------------------------------
    stagnation = 0
    max_stagnation = 5000
    iteration = 0

    while True:
        now = time.time()
        if now - start_time > budget_s - 0.1:
            break

        cur_score = compute_c5(h)

        # record best solution
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            stagnation = 0
        else:

        # early stop if we have reached the target bound

        # --------------------------------------------------------------

        h_candidate = project_to_sum(h + alpha * d, n / 2.0)
