def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_swap_move(h, max_step):
    n = len(h)
    # try random pairs up to e.g., 10 attempts
    for _ in range(10):
        i = np.random.randint(0, n)
        j = np.random.randint(0, n-1)
        if j >= i: j += 1
        # allowed delta
        hi = h[i]; hj = h[j]
        low = max(-max_step, -hi, hj - 1)
        high = min(max_step, 1 - hi, hj)

            new_h[j] -= delta

            return new_h, delta

    return None, None

import time
import numpy as np

def _compute_c5(h):
    """Return C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_to_constraints(h):
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple water‑filling."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h

    if diff > 0:                     # need to add mass
        idx = np.where(h < 1.0 - 1e-12)[0]
        while diff > 1e-12 and idx.size:
            inc = diff / idx.size
            for i in idx:
                add = min(inc, 1.0 - h[i])
                h[i] += add
                diff -= add
            idx = np.where(h < 1.0 - 1e-12)[0]
    else:                             # need to remove mass
        diff = -diff                  # positive amount to remove
        idx = np.where(h > 1e-12)[0]
        while diff > 1e-12 and idx.size:
            dec = diff / idx.size
            for i in idx:
                sub = min(dec, h[i])
                h[i] -= sub
                diff -= sub
            idx = np.where(h > 1e-12)[0]

    return np.clip(h, 0.0, 1.0)

def _random_swap_move(h, max_step):
    """Generate a new vector by moving a bounded amount between two coordinates."""
    n = h.shape[0]
    for _ in range(10):
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0, -max_step)
        high = min(1.0 - hi, hj, max_step)
        if low < high:
            delta = np.random.uniform(low, high)
            h_new = h.copy()
            h_new[i] = hi + delta
            h_new[j] = hj - delta
            return h_new
    return None

def _anneal(h0, time_budget):
    """Simple simulated‑annealing using two‑index swaps."""
    start = time.time()
    h_cur = _adjust_to_constraints(h0.copy())
    best_h = h_cur.copy()
    best_val = _compute_c5(best_h)
    cur_val = best_val

    n = h_cur.shape[0]
    max_step_start = 0.15
    max_step_end = 0.001

    while True:
        if time.time() - start > time_budget - 0.05:
            break
        # linear schedules
        elapsed = time.time() - start
        frac = elapsed / (time_budget - 0.05)
        temp = max(1e-8, 1.0 - frac)           # temperature for Metropolis
        max_step = max_step_start * (1 - frac) + max_step_end * frac

        h_new = _random_swap_move(h_cur, max_step)
        if h_new is None:
            continue
        new_val = _compute_c5(h_new)
        dv = new_val - cur_val
        if dv < 0 or np.random.rand() < np.exp(-dv / max(temp, 1e-12)):
            h_cur = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
    return best_h, best_val

def _resize_h(h, n_target):
    """Interpolate or downsample h to length n_target, then enforce constraints."""
    n0 = h.shape[0]
    if n0 == n_target:
        return _adjust_to_constraints(h.copy())

    if n0 < n_target:                     # up‑sample by linear interpolation
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_new = np.interp(x_new, x_old, h)
    else:                                 # down‑sample by block averaging
        factor = n0 / n_target
        h_new = np.empty(n_target)
        for i in range(n_target):
            start = int(round(i * factor))
            end = int(round((i + 1) * factor))
            if end <= start:
                end = start + 1
            h_new[i] = np.mean(h[start:end])
    return _adjust_to_constraints(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Base construction (either provided or simple binary fallback)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
    else:
        # simple fallback: half ones, half zeros
        n_fallback = 100
        base_h = np.concatenate([np.ones(n_fallback // 2),
                                 np.zeros(n_fallback - n_fallback // 2)])
        base_h = _adjust_to_constraints(base_h)

    best_h = None
    best_val = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 2. Try a few resolutions within the time budget
    # ------------------------------------------------------------------
    start_time = time.time()

    # a) Original resolution
    n0 = base_h.shape[0]
    time_left = max(0.0, budget_s - (time.time() - start_time))
    if time_left > 0.5:
        h_opt, val_opt = _anneal(base_h, time_left)
        if val_opt < best_val:
            best_h, best_val, best_n = h_opt, val_opt, n0

    # b) Double resolution (if time permits)
    n2 = n0 * 2
    if n2 <= 2000:                # protect against too large vectors
        time_elapsed = time.time() - start_time
        time_left = max(0.0, budget_s - time_elapsed)
        if time_left > 0.5:
            init2 = _resize_h(best_h if best_h is not None else base_h, n2)
            h_opt2, val_opt2 = _anneal(init2, time_left)
            if val_opt2 < best_val:
                best_h, best_val, best_n = h_opt2, val_opt2, n2

    # c) Optional random restart (short run) if still time remains
    time_elapsed = time.time() - start_time
    time_left = max(0.0, budget_s - time_elapsed)
    if time_left > 1.0:
        # generate a binary random vector with exact half‑mass

    # ------------------------------------------------------------------
    # 3. Final sanity check and return

    if best_h is None:

        best_val = _compute_c5(best_h)
        best_n = best_h.shape[0]

    best_h = _adjust_to_constraints(best_h)
    best_val = _compute_c5(best_h)

    return best_h, float(best_val), int(best_n)
