    def time_left():

        no_impr = 0

            ones_idx = np.flatnonzero(h)

            if ones_idx.size == 0 or zeros_idx.size == 0:

            i = rng.choice(ones_idx)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # Determine initial sequence and problem size
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        h0 = np.array(iv, dtype=np.float64)
        n = h0.size
        # enforce binary and exact half-density
        half = n // 2
        ones = int(round(h0.sum()))
        if ones != half:
            idx = np.argsort(-h0)
            h0[:] = 0.0
            h0[idx[:half]] = 1.0
    else:
        # default size (even) when no parent is supplied
        n = 120
        half = n // 2
        h0 = np.zeros(n, dtype=np.float64)
        h0[rng.choice(n, size=half, replace=False)] = 1.0
    half = n // 2
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper to evaluate the raw (unscaled) maximal overlap
    def raw_max_overlap(arr):
        return np.max(np.correlate(arr, 1.0 - arr, mode='full'))

    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_raw = raw_max_overlap(best_h)

    # ------------------------------------------------------------------
    # Main optimisation loop: random restarts + greedy swaps
    def time_left():
        return budget_s - (time.time() - start)

    # Parameters for local search
    max_no_improve = 3000   # stop local search after this many non‑improving tries
    while time_left() > 0.2:
        # start a new random candidate (occasionally reuse best)
        if rng.random() < 0.15:
            h = best_h.copy()
        else:
            h = np.zeros(n, dtype=np.float64)
            h[rng.choice(n, size=half, replace=False)] = 1.0

        cur_raw = raw_max_overlap(h)

            if len(ones_idx) == 0 or len(zeros_idx) == 0:
