def repair_h(h):
    n = len(h)
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) > 1e-12:
        # compute scaling factor then clip again
        factor = target / current if current != 0 else 0.0
        h = h * factor
        h = np.clip(h, 0.0, 1.0)
        # After scaling, may not hit target exactly due to clipping
        # Do simple redistribution: compute diff left and adjust some entries within bounds
        current = np.sum(h)
        diff = target - current
        # distribute diff uniformly across entries which aren't at bounds
        if abs(diff) > 1e-12:
            # find indices not at bound
            free = np.where((h > 0) & (h < 1))[0]
            if free.size > 0:
                # distribute equally
                delta = diff / free.size
                h[free] += delta
                # Clip again
                h = np.clip(h, 0.0, 1.0)
                # final adjust by scaling if still off slightly
                current = np.sum(h)
                if abs(current - target) > 1e-9:
                    # scale uniformly small
                    h = h * (target / current)
                    h = np.clip(h, 0.0, 1.0)
    # final adjust to exactly meet target by tiny adjustment on a free index
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-12:
        # find any index not at bound
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size > 0:
            i = idx[0]
            h[i] += diff
            h[i] = min(max(h[i], 0.0), 1.0)
        else:
            # all at bounds, fallback: adjust first entry within tolerance
            h[0] = min(max(h[0] + diff, 0.0), 1.0)
    return h

def enforce_sum_and_bounds(h):

    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-12:
        # find index where adjusting won't exceed bounds
        # If diff > 0, need to increase some entry not at 1
        # If diff < 0, need to decrease some entry not at 0
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            # Should not happen because target is within feasible region (n/2 average 0.5)
            pass

            # distribute diff uniformly across these indices
            # might exceed bounds; clamp after
            delta = diff / idx.size
            h[idx] += delta
            h = np.clip(h, 0.0, 1.0)

import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

# Global variable used by the equality constraint
global_target = None


def evaluate_c5(h):
    """Return the overlap integral C5 for a step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def objective_function(h):
    """Objective supplied to the optimizer (same as evaluate_c5)."""
    return evaluate_c5(h)


def eq_sum_constraint(h):
    """Equality constraint: sum(h) == n/2."""
    return np.sum(h) - global_target


def enforce_sum_and_bounds(h):
    """
    Project h onto the feasible set:
    - entries in [0,1]
    - sum(h) = n/2  (exactly, up to floating‑point tolerance)
    """
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    current = np.sum(h)
    diff = target - current

    if abs(diff) > 1e-12:
        # indices that can be increased (if diff>0) or decreased (if diff<0)
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]

        if idx.size > 0:
            delta = diff / idx.size
            h[idx] += delta
            h = np.clip(h, 0.0, 1.0)
            current = np.sum(h)
            diff = target - current

    # Final tiny correction on a free coordinate if needed
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            i = free[0]
            h[i] += diff
            h[i] = min(max(h[i], 0.0), 1.0)
        else:
            # All entries at a bound – adjust first entry within bounds
            h[0] = min(max(h[0] + diff, 0.0), 1.0)

    return np.clip(h, 0.0, 1.0)


def local_search(h, best_score, time_limit, rng):
    """
    Simple hill‑climbing that preserves the integral exactly by moving a
    random amount of mass from one index to another.
    """
    n = len(h)
    while time.time() < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Randomly choose direction of transfer
        if rng.integers(0, 2) == 0:
            src, dst = i, j
        else:
            src, dst = j, i

        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0.0:
            continue

        delta = max_delta * rng.random()
        if delta <= 0.0:
            continue

        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        score = evaluate_c5(h_new)
        if score + 1e-12 < best_score:
            h = h_new
            best_score = score

    return h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial construction ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform 0.5 on a modest grid
        h0 = np.full(100, 0.5, dtype=np.float64)

    best_h = enforce_sum_and_bounds(h0)
    best_score = evaluate_c5(best_h)

    # ----- iterative improvement ------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 1.0:
            break

        n = len(best_h)

        # Equality constraint target for the current resolution
        global global_target
        global_target = n / 2.0

        # --- SLSQP optimization ------------------------------------------------
        bounds = [(0.0, 1.0)] * n
        constraints = [{"type": "eq", "fun": eq_sum_constraint}]
        try:
            res = minimize(
                objective_function,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            res = None

        if res is not None and res.success:
            cand = enforce_sum_and_bounds(res.x)
            cand_score = evaluate_c5(cand)
            if cand_score + 1e-12 < best_score:
                best_h = cand
                best_score = cand_score

        # --- Local random search ------------------------------------------------
        time_limit = start + budget_s - 0.5
        best_h, best_score = local_search(best_h, best_score, time_limit, rng)

        # --- Optional resolution increase ---------------------------------------
        if n < 380 and (time.time() - start) < budget_s * 0.5:
            # Linear interpolation to double the number of samples
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, 2 * n, endpoint=False)
            interp = interp1d(x_old, best_h, kind="linear", fill_value="extrapolate")
            best_h = enforce_sum_and_bounds(interp(x_new))
            # continue loop to optimise the higher‑resolution version
            continue
        else:
            break

    final_score = evaluate_c5(best_h)
    return best_h.astype(np.float64), float(final_score), len(best_h)
