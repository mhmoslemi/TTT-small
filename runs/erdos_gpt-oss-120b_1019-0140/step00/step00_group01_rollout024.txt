def project_to_constraints(v, target):
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(60):
        lam = (lo + hi) / 2
        h = np.clip(v - lam, 0, 1)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # final

import numpy as np

def _c5_of_h(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx for a discretized h."""
    n = h.size
    dx = 2.0 / n

    return float(c.max())

    i = np.arange(n, dtype=float)
    # Linear ramp: h_i = (i+0.5)/n  -> sum(h)=n/2, 0<h_i<1
    h = (i + 0.5) / n
