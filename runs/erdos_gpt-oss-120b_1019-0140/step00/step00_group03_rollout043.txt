        i = rng.randint(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.RandomState(seed)
    # Determine n_points
    try:
        h0_arr = np.array(initial_h_values, copy=False)
        n = h0_arr.shape[0]
    except NameError:
        n = 70
    dx = 2.0 / n
    target_ones = n // 2

    # Initial vector binary
    # Use random if not binary from initial
    def is_binary(arr):
        return np.all((arr == 0) | (arr == 1))
    init_use = None
    try:
        init_arr = np.array(initial_h_values, copy=False)
        if init_arr.shape[0] == n and is_binary(init_arr):
            init_use = init_arr.astype(int)
    except Exception:
        pass
    if init_use is not None and init_use.sum() == target_ones:
        h = init_use.copy()
    else:
        # random balanced binary
        h = np.zeros(n, dtype=int)
        h[:target_ones] = 1
        rng.shuffle(h)

    # compute initial
    best_h = h.copy()
    best_score = compute_c5(best_h)
    current_score = best_score

    start = time.time()
    temp = 0.5
    decay = 0.999
    min_temp = 1e-5
    iteration = 0

    while time.time() - start < budget_s:
        # Propose swap
        zero_idxs = np.where(h == 0)[0]
        one_idxs = np.where(h == 1)[0]
        if zero_idxs.size == 0 or one_idxs.size == 0:
            break
        i = zero_idxs[rng.randint(zero_idxs.size)]
        j = one_idxs[rng.randint(one_idxs.size)]

        h_new = h.copy()
        h_new[i] = 1
        h_new[j] = 0

        new_score = compute_c5(h_new)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis
            if temp > 0:
                prob = math.exp(-delta / temp)
                if rng.rand() < prob:
                    accept = True
        if accept:
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                # early exit if we've reached target bound
                if best_score <= 0.3808:
                    # optional break early to return best
                    # but continue maybe improve
                    pass
        temp = max(min_temp, temp * decay)
        iteration += 1

    # finalize
    c5_bound = float(best_score)
    # Convert h values to float in [0,1]
    h_values = best_h.astype(float)
    return (h_values, c5_bound, n)

import numpy as np, time, math

def _c5_score(vec):
    """Return the C5 bound for a given h vector."""
    n = len(vec)
    dx = 2.0 / n
    # use the pre‑imported verifier if present, otherwise the direct formula
    try:
        return float(evaluate_erdos_solution(vec))
    except NameError:
        corr = np.correlate(vec, 1 - vec, mode="full")
        return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a feasible h
    # ------------------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, copy=False).astype(float)
        n = init_arr.shape[0]

        n = 70
        init_arr = None

    target_sum = n / 2          # because sum(h) * (2/n) = 1  =>  sum(h)=n/2
    dx = 2.0 / n

    if init_arr is not None and np.all((init_arr >= 0) & (init_arr <= 1)):
        # start from the provided construction, fixing the integral if needed
        h = init_arr.copy()
        cur = h.sum()
        if abs(cur - target_sum) > 1e-9:
            # simple linear scaling then a tiny projection to meet the exact sum
            h *= target_sum / cur
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            # distribute the remaining discrepancy greedily
            while abs(diff) > 1e-12:
                if diff > 0:
                    idx = np.where(h < 1.0)[0]
                    if idx.size == 0:
                        break
                    i = rng.choice(idx)
                    delta = min(diff, 1.0 - h[i])
                    h[i] += delta
                    diff -= delta
                else:
                    idx = np.where(h > 0.0)[0]
                    if idx.size == 0:

                    diff += delta
    else:

        h = np.zeros(n, dtype=float)
        h[:int(target_sum)] = 1.0
        rng.shuffle(h)

    # ------------------------------------------------------------------
    # Simulated annealing on the simplex (preserve sum, keep within [0,1])
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_score(best_h)
    cur_score = best_score

    start = time.time()

    min_temp = 1e-6

        i = rng.randint(n)

        if j >= i:
