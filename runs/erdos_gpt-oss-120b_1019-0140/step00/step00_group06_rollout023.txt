def random_move(h, total_sum, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # Determine max amount we can transfer from i to j within [0,1] bounds
    max_transfer = min(h[i], 1 - h[j])

    # transfer epsilon: random fraction of max_transfer
    epsilon = rng.random() * max_transfer
    delta = np.zeros_like(h)
    delta[i] = -epsilon
    delta[j] = epsilon
    return delta, epsilon

def local_search(h, max_iters, rng, time_budget, start_time):
    deadline = start_time + time_budget
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute correlation
        corr = correlation_full(h)
        idx = np.argmax(corr)
        max_val = corr[idx] * (2.0 / n)
        shift = idx - (n - 1)
        # determine i range
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        # find best i
        positions = np.arange(i_start, i_end)
        if positions.size == 0:
            continue
        prod = h[positions] * (1 - h[positions + shift])
        i_sel = positions[np.argmax(prod)]
        # compute transfer
        max_transfer = min(h[i_sel], 1 - h[i_sel + shift])
        if max_transfer > 1e-12:
            # epsilon fraction
            epsilon = max_transfer * (0.2 + 0.5 * rng.random())  # 0.2-0.7
            # apply move
            h[i_sel] -= epsilon
            h[i_sel + shift] += epsilon
        else:
            # fallback random pair
            i_rand = rng.integers(0, n)
            j_rand = rng.integers(0, n)
            if i_rand == j_rand:
                continue
            max_transfer = min(h[i_rand], 1 - h[j_rand])
            if max_transfer <= 1e-12:
                continue
            epsilon = max_transfer * rng.random()
            h[i_rand] -= epsilon
            h[j_rand] += epsilon
        # compute new C5:
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # maybe revert move with probability
            if rng.random() > 0.001:
                # revert
                if max_transfer > 1e-12:
                    h[i_sel] += epsilon
                    h[i_sel + shift] -= epsilon
                else:
                    h[i_rand] += epsilon
                    h[j_rand] -= epsilon
        # optional small perturbation to escape plateau: every 1000 iterations
        if (it + 1) % 1000 == 0:
            # small random perturbation preserving sum via random pair transfers
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)

                    continue

                    h[j] += eps

import numpy as np
import time

def project_onto_capped_simplex(v, total):
    # binary search lambda
    # ensure total between 0 and len(v) because each entry max 1
    n = len(v)
    # if total is 0 or n, trivial
    low = -1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            low = lam
        else:
            high = lam
    x = np.clip(v - high, 0.0, 1.0)
    # small adjustment
    diff = total - x.sum()
    if abs(diff) > 1e-12:
        # try to adjust a free variable
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            idxs = np.where(free)[0]
            # distribute diff equally
            add = diff / len(idxs)
            x[idxs] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust the element with max possible margin
            # find element not at bound from v
            # This rarely needed
            # fallback: add to first element, clip
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    # final clip
    return np.clip(x, 0.0, 1.0)

def correlation_full(h):
    n = len(h)
    nfft = 2 * n - 1
    A = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(A * np.conj(B), n=nfft)
    return corr

def compute_c5(h):
    n = len(h)
    corr = correlation_full(h)
    return np.max(corr) * (2.0 / n)

def upsample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def normalize_sum(h):
    n = len(h)
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # try adjust first non-boundary entry
    for i in range(n):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h
    # else project
    return project_onto_capped_simplex(h, target)

    n = len(h)
    dx = 2.0 / n
    deadline = start_time + time_budget
    # initial correlation and best
    corr = correlation_full(h)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    # iteration loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # find shift with max overlap
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        # define range of i for this shift
        if shift > 0:
            i_start = 0
            i_end = n - shift
        elif shift < 0:
            i_start = -shift
            i_end = n
        else:
            # shift 0 case: use random pair move
            i_start = i_end = 0
        move = None
        if i_end > i_start:
            # compute product h[i] * (1 - h[i+shift])
            i_range = np.arange(i_start, i_end)
            prod = h[i_range] * (1.0 - h[i_range + shift])
            i_sel = i_range[np.argmax(prod)]
            max_transfer = min(h[i_sel], 1.0 - h[i_sel + shift])
            if max_transfer > 1e-12:
                eps = max_transfer * (0.2 + 0.6 * rng.random())
                i_from = i_sel
                i_to = i_sel + shift
                move = (i_from, i_to, eps)
                h[i_from] -= eps
                h[i_to] += eps
        if move is None:

            i_from = rng.integers(0, n)
            i_to = rng.integers(0, n)
            if i_from == i_to:
                continue
            max_transfer = min(h[i_from], 1.0 - h[i_to])
            if max_transfer <= 1e-12:
                continue
            eps = max_transfer * rng.random()
            move = (i_from, i_to, eps)
            h[i_from] -= eps
            h[i_to] += eps
        # evaluate new correlation
        corr = correlation_full(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        else:

            if rng.random() > 0.001:
                # revert move
                i_from, i_to, eps = move
                h[i_from] += eps
                h[i_to] -= eps
                # recompute corr for next iteration as old state
                corr = correlation_full(h)
        # periodic small perturbation to escape stagnation

                    h[i_a] -= eps
                    h[i_b] += eps
                    # update correlation for next loop

                max_transfer = min(h[a], 1.0 - h[b])
                if max_transfer > 1e-12:
                    eps = max_transfer * 0.01 * rng.random()
                    h[a] -= eps
                    h[b] += eps
                    corr = correlation_full(h)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # Obtain or create an initial feasible vector
    try:
        init_vals = initial_h_values  # provided by the harness
        h = np.array(init_vals, dtype=float)
    except NameError:
        n0 = 150
        v = rng.random(n0)
        h = project_onto_capped_simplex(v, n0 / 2.0)

    # Ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_capped_simplex(h, len(h) / 2.0)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Define a hierarchy of resolutions (95 → 190 → 380 … up to ~500 points)
    sizes = [len(best_h)]
    while sizes[-1] * 2 <= 500:
        sizes.append(sizes[-1] * 2)

    for stage, n in enumerate(sizes):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        stages_left = len(sizes) - stage
        time_for_stage = remaining / stages_left

        # Prepare vector at current resolution
        if len(best_h) != n:
            h = upsample_h(best_h, n)
        else:
            h = best_h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_capped_simplex(h, n / 2.0)

        # Optimize locally
        h_opt, c5_opt = local_search(h, rng, time_for_stage, start)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final correction to meet the exact sum constraint
    best_h = normalize_sum(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), len(best_h)
