import numpy as np
import random
import time

def _c5_value(h, dx):
    """Compute C5 for a given h."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _random_feasible(n, steps=2000):
    """Generate a random feasible h with sum = n/2 and 0<=h<=1."""
    h = np.full(n, 0.5, dtype=np.float64)
    for _ in range(steps):
        i, j = random.sample(range(n), 2)
        up = min(1.0 - h[j], h[i])          # transfer from i to j
        down = min(1.0 - h[i], h[j])        # transfer from j to i
        if up + down == 0.0:
            continue
        delta = random.uniform(-down, up)   # may be positive or negative
        h[i] -= delta
        h[j] += delta
    return h

def _targeted_pair(h):
    """Propose a pair move that targets the dominant overlap contribution."""
    n = h.shape[0]
    # locate shift with maximal correlation
    corr = np.correlate(h, 1 - h, mode='full')
    ind = int(np.argmax(corr))
    shift = ind - (n - 1)          # actual integer shift
    contrib = np.zeros(n, dtype=np.float64)
    if shift >= 0:
        contrib[:n - shift] = h[:n - shift] * (1.0 - h[shift:])
    else:
        s = -shift
        contrib[s:] = h[s:] * (1.0 - h[:n - s])
    i = int(np.argmax(contrib))
    # pick a distinct partner at random
    j = random.choice([k for k in range(n) if k != i])
    up = min(1.0 - h[j], h[i])
    down = min(1.0 - h[i], h[j])
    if up + down == 0.0:
        return None, None
    delta = random.uniform(-down, up)
    return (i, j, delta)

        best_c5_overall = _c5_value(best_h_overall, 2.0 / best_n)

    target_sum = best_n / 2.0
    cur_sum = best_h_overall.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # adjust entries sequentially respecting bounds
        for idx in range(best_n):
            if diff > 0.0:
                cap = 1.0 - best_h_overall[idx]
                add = min(diff, cap)
                best_h_overall[idx] += add
                diff -= add
            else:
                cap = best_h_overall[idx]
                sub = min(-diff, cap)
                best_h_overall[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_c5 = _c5_value(best_h_overall, 2.0 / best_n)

    # return plain Python list for compatibility
    return best_h_overall.tolist(), float(final_c5), best_n
