def project_box_sum(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if np.isclose(total, target, atol=1e-10):
        return v
    if total > target:
        # need to reduce
        excess = total - target
        # Reduce uniformly on elements >0
        # We'll use while loop as described.
        while excess > 1e-12:
            pos_idx = v > 0
            npos = np.sum(pos_idx)
            if npos == 0:
                break
            # max reduction without hitting zero
            max_reduce = np.min(v[pos_idx])
            # reduction per element
            delta = min(excess / npos, max_reduce)
            v[pos_idx] -= delta
            excess -= delta * npos
    else:
        deficit = target - total
        while deficit > 1e-12:
            neg_idx = v < 1
            nneg = np.sum(neg_idx)
            if nneg == 0:
                break
            max_inc = np.min(1 - v[neg_idx])
            delta = min(deficit / nneg, max_inc)
            v[neg_idx] += delta
            deficit -= delta * nneg
    # Clip again just in case
    v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np

def _project_to_feasible(v, target, eps=1e-12):
    """
    Project vector v onto the box [0,1]^n intersected with sum(v)==target.
    Simple waterfilling algorithm: uniformly shrink/expand active entries.
    """
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= eps:
        return v
    if s > target:
        excess = s - target
        while excess > eps:
            pos = v > eps
            npos = pos.sum()
            if npos == 0:
                break
            max_reduce = v[pos].min()
            delta = min(excess / npos, max_reduce)
            v[pos] -= delta
            excess -= delta * npos
    else:
        deficit = target - s
        while deficit > eps:
            neg = v < 1.0 - eps
            nneg = neg.sum()
            if nneg == 0:
                break
            max_inc = (1.0 - v[neg]).min()
            delta = min(deficit / nneg, max_inc)
            v[neg] += delta
            deficit -= delta * nneg
    return v

def _c5_bound(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr

def _gradient_at_max_shift(h, max_idx):
    """Gradient of raw correlation at the shift achieving the maximum."""
    n = h.shape[0]
    dx = 2.0 / n
    shift = max_idx - (n - 1)      # integer lag
    grad = np.ones_like(h) * dx   # derivative of sum_i h_i term
    if shift >= 0:
        # subtract h_{i+shift}
        grad[: n - shift] -= dx * h[shift:]
        # subtract h_{i-shift}
        grad[shift:] -= dx * h[: n - shift]
    else:
        s = -shift
        grad[s:] -= dx * h[: n - s]
        grad[: n - s] -= dx * h[s:]
    return grad

    n = 200

    dx = 2.0 / n

    # ---- initial candidate -------------------------------------------------
    def _initial_h():
