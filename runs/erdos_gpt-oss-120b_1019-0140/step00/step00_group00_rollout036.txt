def gradient_optimize_continuous(h, steps, lr, alpha):
    n = len(h)
    offset = n - 1
    diff = np.arange(n)[np.newaxis, :] - np.arange(n)[:, np.newaxis]  # shape (n,n): j-i
    mask_upper = diff >= 0
    mask_lower = diff <= 0
    target = n / 2.0

    best_h = h.copy()
    best_val = evaluate_c5(h)
    for step in range(steps):
        # compute cross correlation (raw)
        cross = np.correlate(h, 1 - h, mode='full')
        # compute probs
        max_cross = np.max(cross)
        exp_vals = np.exp(alpha * (cross - max_cross))
        probs = exp_vals / np.sum(exp_vals)  # shape (2n-1,)
        prob_mat = probs[diff + offset]  # broadcast diff to index
        term1 = np.dot(prob_mat * mask_upper, (1 - h))
        term2 = np.dot(prob_mat * mask_lower, h)
        grad = term1 - term2  # shape (n,)
        # update
        h = h - lr * grad
        # project onto [0,1] and sum constraint
        h = project_box_sum(h, target)
        # evaluate
        val = evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # optionally decay lr
        lr = lr * 0.999
    return best_h, best_val

def gradient_optimize_continuous(h, steps, lr, alpha, target):
    n = len(h)
    offset = n - 1
    # diff matrix: shape (n, n) where entry is j - i
    diff = np.arange(n).reshape(1, -1) - np.arange(n).reshape(-1, 1)  # j-i
    mask_upper = diff >= 0
    mask_lower = diff <= 0
    best_h = h.copy()
    best_val = evaluate_c5(h)
    for s in range(steps):
        cross = np.correlate(h, 1 - h, mode='full')
        max_cross = np.max(cross)
        exp_vals = np.exp(alpha * (cross - max_cross))
        probs = exp_vals / np.sum(exp_vals)
        prob_mat = probs[diff + offset]  # shape (n,n)
        term1 = prob_mat * mask_upper
        term2 = prob_mat * mask_lower
        # compute gradient as term1 @ (1 - h) - term2 @ h
        grad = term1.dot(1 - h) - term2.dot(h)
        # update
        h = h - lr * grad
        # project onto constraints
        h = project_box_sum(h, target)
        # evaluate
        val = evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # maybe adapt lr: reduce gradually
        lr *= 0.9995
    return best_h, best_val

Note: dot product with a vector (1 - h) uses broadcasting. type: term1 is (n,n), (1 - h) is (n,). So term1.dot(1 - h) yields shape (n,). Works.

import time
import random
import numpy as np


def evaluate_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] and the hyperplane sum(x)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_h(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with exact sum = n/2."""
    x = np.random.rand(n)
    return _project_box_sum(x, n / 2.0)


def gradient_optimize_continuous(
    h: np.ndarray,
    steps: int,
    lr: float,
    alpha: float,
    target: float,
) -> tuple[np.ndarray, float]:
    """Smoothly minimise max correlation using a log‑sum‑exp surrogate."""
    n = h.shape[0]
    offset = n - 1
    # matrix of shifts j-i (shape n×n)
    diff = np.arange(n).reshape(1, -1) - np.arange(n).reshape(-1, 1)
    mask_up = diff >= 0
    mask_lo = diff <= 0
    best_h = h.copy()
    best_val = evaluate_c5(h)

    for _ in range(steps):
        # correlation of h with 1‑h
        corr = np.correlate(h, 1.0 - h, mode="full")
        # log‑sum‑exp weights (smooth max)
        maxc = np.max(corr)
        w = np.exp(alpha * (corr - maxc))
        probs = w / w.sum()
        # turn probs into an n×n matrix of weights for each pair (i,j)
        prob_mat = probs[diff + offset]
        # gradient of the smooth objective
        grad = (prob_mat * mask_up).dot(1.0 - h) - (prob_mat * mask_lo).dot(h)
        # gradient step and projection onto constraints
        h = h - lr * grad
        h = _project_box_sum(h, target)
        # evaluate exact C5
        val = evaluate_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
        lr *= 0.9995  # slow decay
    return best_h, best_val


def hill_climb_binary(h: np.ndarray, max_iters: int) -> tuple[np.ndarray, float]:
    """Simple greedy hill‑climbing for binary vectors (0/1) with sum = n/2."""
    n = h.shape[0]
    cur = h.copy()
    cur_val = evaluate_c5(cur)
    best = cur.copy()
    best_val = cur_val

    for _ in range(max_iters):
        i1 = random.randrange(n)
        i2 = random.randrange(n - 1)
        if i2 >= i1:
            i2 += 1
        if cur[i1] == cur[i2]:
            continue
        # swap a 1 with a 0
        cur[i1], cur[i2] = cur[i2], cur[i1]
        new_val = evaluate_c5(cur)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val, best = new_val, cur.copy()
        else:
            cur[i1], cur[i2] = cur[i2], cur[i1]  # revert
    return best, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_val = float("inf")
    best_n = None

    # candidate sizes (even to allow exact half‑ones)
    candidate_ns = [80, 100, 120, 140, 160, 200]

    for n in candidate_ns:
        if time.time() - start_time > budget_s * 0.95:
            break
        target_sum = n / 2.0

        # possible starting points
        starts = []

        # use the provided parent if its length matches n
        parent = globals().get("initial_h_values")
        if isinstance(parent, (list, tuple, np.ndarray)):
            arr = np.asarray(parent, dtype=float)
            if arr.size == n:
                starts.append(_project_box_sum(arr, target_sum))

        # a few random initial vectors
        for _ in range(3):
            starts.append(random_feasible_h(n))

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            # --- smooth gradient optimisation (continuous) ---
            steps = max(2000, int(5000 * (120 / n)))  # more steps for smaller n
            opt_h, opt_val = gradient_optimize_continuous(
                h0.copy(), steps=steps, lr=0.5, alpha=12.0, target=target_sum
            )
            if opt_val < best_val:
                best_val, best_h, best_n = opt_val, opt_h.copy(), n

            # --- turn continuous result into a binary candidate ---
            bin_h = np.zeros(n, dtype=float)
            idx = np.argsort(-opt_h)[: n // 2]
            bin_h[idx] = 1.0

            # local improvement on the binary vector
            local_h, local_val = hill_climb_binary(bin_h, max_iters=3000)
            if local_val < best_val:
                best_val, best_h, best_n = local_val, local_h.copy(), n

            # a few extra random binary samples as cheap baselines
            for _ in range(5):
                if time.time() - start_time > budget_s * 0.95:
                    break
                rnd_bin = np.zeros(n, dtype=float)
                rnd_bin[: n // 2] = 1.0
                np.random.shuffle(rnd_bin)
                val = evaluate_c5(rnd_bin)
                if val < best_val:
                    best_val, best_h, best_n = val, rnd_bin.copy(), n

    # fallback in the unlikely case no candidate was found
    if best_h is None:
        n = 80
        best_h = random_feasible_h(n)
        best_val = evaluate_c5(best_h)
        best_n = n

    return best_h, best_val, int(best_n)
