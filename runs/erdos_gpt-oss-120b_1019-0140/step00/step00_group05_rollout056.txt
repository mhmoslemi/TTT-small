def project_to_capped_simplex(v, target_sum, lower=0.0, upper=1.0, max_iter=100, tol=1e-9):
    # Works by binary search for tau.
    # Input v is np.ndarray of shape (N,)
    if target_sum < lower * len(v) - 1e-12 or target_sum > upper * len(v) + 1e-12:
        raise ValueError('target sum out of bounds')
    # Use binary search for tau.
    lo = -np.max(v) - 1.0  # sufficiently low
    hi = np.max(v) - lower + 1.0  # sufficiently high
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        # compute projection
        w = np.clip(v - tau, lower, upper)
        s = w.sum()
        if np.abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    # after loop, return w
    return np.clip(v - (lo+hi)/2.0, lower, upper)

import numpy as np, time, random, math

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_capped_simplex(v, target, lower=0.0, upper=1.0, tol=1e-10, max_iter=100):
    # binary search for tau s.t. sum clamp(v - tau) = target
    n = v.shape[0]
    lo = -np.max(v) - 10.0
    hi = np.max(v) - lower + 10.0
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        w = np.clip(v - tau, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = tau
        else:
            hi = tau
    # fallback
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def perturb_h(h, eps):
    new_h = h.copy()
    N = h.shape[0]
    n_swaps = np.random.randint(1, 4)  # 1-3 swaps per perturb
    for _ in range(n_swaps):
        i, j = np.random.choice(N, size=2, replace=False)

        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        new_h[i] += delta
        new_h[j] -= delta
    return new_h

import time
import random
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_capped_simplex(v: np.ndarray,
                              target: float,
                              lower: float = 0.0,
                              upper: float = 1.0,
                              tol: float = 1e-10,
                              max_iter: int = 100) -> np.ndarray:
    """Project v onto the set {x | lower ≤ x_i ≤ upper, sum(x)=target}."""
    # Binary search on the Lagrange multiplier τ such that sum(clamp(v-τ)) = target.
    lo = -np.max(v) - 10.0
    hi = np.max(v) - lower + 10.0
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        w = np.clip(v - tau, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = tau
        else:
            hi = tau
    # Final approximation
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, lower, upper)

def perturb_h(h: np.ndarray, eps: float) -> np.ndarray:
    """Return a new h obtained by moving a small amount of mass between a few indices."""
    new_h = h.copy()
    n = new_h.size
    swaps = np.random.randint(1, 4)  # 1–3 swaps per perturbation
    for _ in range(swaps):
        i, j = np.random.choice(n, size=2, replace=False)
        # Transfer δ from j to i (i gains, j loses) while respecting bounds and eps.
        lo = max(-new_h[i], -(1.0 - new_h[j]), -eps)
        hi = min(1.0 - new_h[i], new_h[j], eps)
        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        new_h[i] += delta
        new_h[j] -= delta
    return new_h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 and 0≤h≤1 minimizing C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of length n_points satisfying the constraints
        c5_bound : float, the achieved C5 value
        n_points : int, number of discretisation points used
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback: 70 points uniformly split (will be projected to satisfy sum)
        h0 = np.full(70, 0.5, dtype=np.float64)

    n = h0.size
    target_sum = n / 2.0

    # Ensure feasibility of the starting point
    h0 = np.clip(h0, 0.0, 1.0)
    if abs(h0.sum() - target_sum) > 1e-12:
        h0 = project_to_capped_simplex(h0, target_sum)

    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Simulated annealing / evolutionary loop
    # ------------------------------------------------------------------
    start_time = time.time()
    eps = 0.25                # initial maximal transferred mass per swap
    init_T = 1e-2
    final_T = 1e-6
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # Temperature schedule (linear decay in elapsed fraction)
        frac = elapsed / (budget_s * 0.95)
        T = final_T + (init_T - final_T) * (1.0 - frac)

        # Propose a new candidate
        cand_h = perturb_h(cur_h, eps)
        cand_score = compute_c5(cand_h)

        # Acceptance test (Metropolis)
        if cand_score < cur_score - 1e-12:
            cur_h, cur_score = cand_h, cand_score
            if cand_score < best_score - 1e-12:
                best_h, best_score = cand_h.copy(), cand_score
        else:
            delta = cand_score - cur_score
            if T > 0.0:
                prob = math.exp(-delta / T)
                if prob > random.random():
                    cur_h, cur_score = cand_h, cand_score

        # Gradually shrink the mutation amplitude
        if it % 5000 == 0 and it > 0:
            eps = max(eps * 0.95, 1e-4)

        it += 1

    # ------------------------------------------------------------------
    # Final projection & evaluation
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target_sum) > 1e-9:
        best_h = project_to_capped_simplex(best_h, target_sum)

    try:
        final_score = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_score = compute_c5(best_h)

    return best_h, final_score, n
