import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return C5 = dx * max correlation of h with (1-h)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation = convolution of h with reversed (1-h)
    corr = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return dx * np.max(corr)
