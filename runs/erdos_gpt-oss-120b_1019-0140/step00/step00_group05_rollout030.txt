            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            # fallback: scale globally (may slightly violate bounds)
            h *= target_sum / cur
            np.clip(h, 0.0, 1.0, out=h)
        else:
            i = np.random.choice(idx)
            adj = min(abs(diff), 1.0 - h[i]) if diff > 0 else min(abs(diff), h[i])
            h[i] += adj if diff > 0 else -adj
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _initial_candidates(n, init_h):
    """Generate a list of starting vectors."""
    candidates = []
    target_sum = n / 2.0
    if init_h is not None and len(init_h) == n:
        candidates.append(np.array(init_h, copy=True))
    # uniform 0.5 (always feasible)
    candidates.append(np.full(n, 0.5, dtype=float))
    # binary patterns if n is even
    if n % 2 == 0:
        for _ in range(5):
            h = np.zeros(n, dtype=float)
            ones = np.random.choice(n, n // 2, replace=False)
            h[ones] = 1.0
            candidates.append(h)
    # a few random fractional vectors respecting the sum
    for _ in range(3):
        h = np.random.rand(n)
        h = h / h.sum() * target_sum
        np.clip(h, 0.0, 1.0, out=h)
        candidates.append(h)
    return candidates

def _local_search(h0, dx, time_deadline):
    """Simple simulated‑annealing hill‑climb starting from h0."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    cur_val = _compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur_val

    max_iter = 20000
    init_temp = 0.05

    for it in range(max_iter):
        if time.time() > time_deadline:
            break
        # pick a source and a sink for mass transfer
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue

        # direction: try moving from i to j
        if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
            max_delta = min(h[i], 1.0 - h[j])
            direction = -1  # decreasing i, increasing j
        elif h[j] > 1e-12 and h[i] < 1.0 - 1e-12:
            max_delta = min(h[j], 1.0 - h[i])
            direction = 1   # decreasing j, increasing i
            i, j = j, i     # swap so we always add to j
        else:
            continue

        # propose a transfer amount (30‑70 % of max_delta)
        delta = max_delta * (0.3 + 0.4 * np.random.rand())
        if direction == -1:
            h[i] -= delta
            h[j] += delta
        else:
            # direction == 1 after swapping; already handled
            pass

        new_val = _compute_c5(h, dx)

        # simulated‑annealing acceptance
        if new_val < cur_val:
            accept = True
        else:
            temp = max(1e-4, init_temp * (1.0 - it / max_iter))
            prob = math.exp((cur_val - new_val) / temp)
            accept = np.random.rand() < prob

        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert
            if direction == -1:
                h[i] += delta
                h[j] -= delta
            else:
                # never reaches here due to swap handling
                pass
    # enforce exact sum before returning
    best_h = _enforce_sum_and_bounds(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allotted time."""
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    try:
        init_h = initial_h_values  # provided by the harness if any
        n = int(len(init_h))
    except NameError:
        init_h = None
        n = 70  # default size when no parent exists

    dx = 2.0 / n
    time_start = time.time()
    time_deadline = time_start + budget_s * 0.98  # leave a safety margin

    candidates = _initial_candidates(n, init_h)

    best_h = None
    best_c5 = float('inf')

    # Main multi‑start loop
    for h0 in candidates:
        if time.time() > time_deadline:
            break
        h_opt, c5_opt = _local_search(h0, dx, time_deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # If nothing found (unlikely), fall back to uniform
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = _compute_c5(best_h, dx)

    # Final safety enforcement
    best_h = _enforce_sum_and_bounds(best_h, n / 2.0)
    best_c5 = _compute_c5(best_h, dx)

    return best_h, float(best_c5), n
