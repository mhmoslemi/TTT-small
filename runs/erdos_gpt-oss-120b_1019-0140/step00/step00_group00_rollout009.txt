def compute_c5(h):

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # binary search lambda
    lo = -np.max(v) - 10.0
    hi = np.max(v) - lower + 10.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return x

import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} dx * sum_i h[i] * (1 - h[i+k]) using np.correlate."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_bounds_and_sum(v: np.ndarray, target: float,
                               lower: float = 0.0, upper: float = 1.0,
                               tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """
    Euclidean projection of v onto the box [lower,upper] with the linear
    equality sum(x)=target. Uses a simple binary search on the Lagrange multiplier.
    """
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, lower, upper)


def random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2 (exactly)."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    if n % 2 == 0:
        ones = int(target)
        idx = rng.choice(n, ones, replace=False)
        h[idx] = 1.0
    else:
        ones = int(np.floor(target))
        idx_ones = rng.choice(n, ones, replace=False)
        h[idx_ones] = 1.0
        remaining = np.setdiff1d(np.arange(n), idx_ones, assume_unique=True)
        half_idx = rng.choice(remaining, 1)[0]
        h[half_idx] = 0.5
    return h


def random_transfer_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Move a random amount from a positive entry to a non‑full entry."""
    eps = 1e-12
    pos = np.where(h > eps)[0]
    neg = np.where(h < 1.0 - eps)[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return None
    delta = rng.random() * max_eps
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def improve_h_local(h: np.ndarray, steps: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple local improvement by swaps or fractional transfers."""
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    n = cur_h.shape[0]
    for _ in range(steps):
        if rng.random() < 0.5:
            i, j = rng.choice(n, 2, replace=False)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
        else:
            cand = random_transfer_move(cur_h, rng)
            if cand is None:
                continue
            new_h = cand
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
    return cur_h, cur_c5


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate h onto a finer grid and re‑project onto feasible set."""
    n = h.shape[0]
    if new_n == n:
        return h.copy()
    old_dx = 2.0 / n
    new_dx = 2.0 / new_n
    old_x = (np.arange(n) + 0.5) * old_dx
    new_x = (np.arange(new_n) + 0.5) * new_dx
    h_new = np.interp(new_x, old_x, h, left=0.0, right=0.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = new_n / 2.0
    h_new = project_onto_bounds_and_sum(h_new, target)
    return h_new
