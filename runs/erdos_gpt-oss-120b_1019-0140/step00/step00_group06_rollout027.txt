def optimize_h(h0, max_iter, eta0):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    h = h0.copy()
    n = len(h)
    for it in range(max_iter):
        c = np.correlate(h, 1-h, mode='full')
        max_c = np.max(c)
        max_indices = np.flatnonzero(c == max_c)
        # compute subgradient, average over ties
        grad = np.zeros_like(h)
        for L in (max_indices - (n-1)): # shift values
            for i in range(n):
                j = i + L
                if 0 <= j < n:
                    grad[i] += (1 - h[j])
                k = i - L
                if 0 <= k < n:

import time
import math
import numpy as np

def compute_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * (2.0 / n)

def subgradient_max(h):
    """
    Subgradient of max_k ∫ h(x)(1−h(x+k))dx at the current h.
    Returns an n‑vector (average if multiple lags tie).
    """
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(c)
    # lags corresponding to indices where correlation attains its maximum
    lags = np.where(c == max_c)[0] - (n - 1)
    grad = np.zeros_like(h)
    for L in lags:
        if L >= 0:
            # grad[i] += (1 - h[i+L])  for i = 0..n-L-1
            grad[:n - L] += 1.0 - h[L:]
            # grad[i] -= h[i-L]       for i = L..n-1
            grad[L:] -= h[:n - L]
        else:
            p = -L
            # grad[i] += (1 - h[i-p])  for i = p..n-1
            grad[p:] += 1.0 - h[:n - p]
            # grad[i] -= h[i+p]       for i = 0..n-p-1
            grad[:n - p] -= h[p:]
    grad /= len(lags)
    return grad

def project_to_capped_simplex(v, target, lower=0.0, upper=1.0, max_iter=100, tol=1e-10):
    """
    Euclidean projection of v onto {x | sum(x)=target, lower ≤ x_i ≤ upper }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low, high = -1e5, 1e5
    for _ in range(max_iter):
        tau = (low + high) * 0.5
        x = np.clip(v - tau, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = tau
        else:
            high = tau
    return np.clip(v - tau, lower, upper)

def descent_until_budget(h, target_sum, start_time, budget_s, max_iters=20000):
    """
    Simple projected subgradient descent on C5.
    Stops when the overall budget is nearly exhausted or after max_iters.
    Returns (best_h, best_c5).
    """
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = h.shape[0]
    eta0 = 0.05
    it = 0
    while it < max_iters:
        if time.time() - start_time > budget_s - 2.0:
            break
        grad = subgradient_max(h)
        eta = eta0 / math.sqrt(it + 1.0)
        h = project_to_capped_simplex(h - eta * grad, target_sum)
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        # occasional stochastic kick to escape plateaus
        if (it + 1) % 500 == 0:
            scale = 0.02 * (0.5 ** ((it + 1) // 500))
            noise = np.random.randn(n) * scale
            h = project_to_capped_simplex(best_h + noise, target_sum)
        it += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Obtain initial construction (if supplied) -------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:

        init = project_to_capped_simplex(init, 95.0 / 2.0)

    # Keep track of the best solution found
    best_h = init.copy()
    best_c5 = compute_c5(best_h)

    # Hierarchical refinement: 95 → 190 → 380 points (if time permits)
    for target_n in (190, 380):
        if time.time() - start > budget_s - 2.0:
            break
        # Interpolate current best to the new resolution
        cur_n = best_h.shape[0]
        if target_n == cur_n:
            h0 = best_h.copy()
        else:
            old_x = np.linspace(0.0, 2.0, cur_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h0 = np.interp(new_x, old_x, best_h)
        # Project to satisfy constraints exactly at the new size
        target_sum = target_n / 2.0
        h0 = project_to_capped_simplex(h0, target_sum)
        # Optimize at this resolution
        best_h, best_c5 = descent_until_budget(h0, target_sum, start, budget_s)

    # Final safety projection
    final_n = best_h.shape[0]
    best_h = project_to_capped_simplex(best_h, final_n / 2.0)
    best_c5 = compute_c5(best_h)

    # Return the best found function, its bound, and the sample count
    return best_h.astype(np.float64), float(best_c5), final_n
