                    centre = 0.0
                    lo = max(delta_min, centre - max_step / 2)
                    hi_ = min(delta_max, centre + max_step / 2)
                    delta = rng.uniform(lo, hi_)
                else:
                    delta = rng.uniform(delta_min, delta_max)
                new_h = cur_h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                    cur_h = new_h
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
        # Occasionally restart from a fresh random feasible point
        if rng.random() < restart_prob:
            cur_h = random_feasible()
            cur_c5 = compute_c5(cur_h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    # ------------------------------------------------------------
    # Final repair to ensure strict feasibility
    # ------------------------------------------------------------
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # Try a simple redistribution
        for _ in range(500):
            if abs(diff) < 1e-12:
                break
            idx = rng.integers(0, n)
            if diff > 0:
                slack = 1.0 - best_h[idx]
                delta = min(diff, slack, 0.05)
                best_h[idx] += delta
                diff -= delta
            else:
                slack = best_h[idx]
                delta = max(diff, -slack, -0.05)
                best_h[idx] += delta
                diff -= delta
        # Final uniform correction (tiny)
        if abs(target_sum - best_h.sum()) > 1e-12:
            best_h += (target_sum - best_h.sum()) / n
            np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), n
