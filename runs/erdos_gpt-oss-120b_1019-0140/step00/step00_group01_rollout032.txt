import time
import numpy as np

# ----- Helper functions -----------------------------------------------------

def _c5_value(h):
    """Return the overlap integral C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _gradient_shift(h, shift):
    """Gradient of the worst-shift overlap functional w.r.t. h."""
    n = h.shape[0]
    i = np.arange(n)
    idx1 = i + shift
    term1 = np.where((idx1 >= 0) & (idx1 < n), 1.0 - h[idx1], 0.0)
    idx2 = i - shift
    term2 = np.where((idx2 >= 0) & (idx2 < n), h[idx2], 0.0)
    return term1 - term2


def _project_onto_sum_box(v, target, tol=1e-9, max_iter=100):
    """
    Euclidean projection of v onto the intersection of the box [0,1]^n
    and the hyperplane sum(x) = target.
    """
    lo = v.min() - 1.0          # ensures all entries become 1 after clipping
    hi = v.max()                # ensures all entries become 0 after clipping

    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    # final clipping if tolerance not met
    return np.clip(v - lam, 0.0, 1.0)


# ----- Main entry point ----------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initialise -------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n = init.size
        h = init.copy()
        # ensure feasibility (tiny numeric drift may exist)
        h = _project_onto_sum_box(h, n / 2.0)
    except Exception:
        # fall back to a fresh random start
        n = 200                     # moderate size, well under 1k
        v = rng.random(n)
        h = _project_onto_sum_box(v, n / 2.0)

    best_h = h.copy()
    best_val = _c5_value(best_h)
    target_sum = n / 2.0
    step = 0.12                     # initial step size
    improve_counter = 0

    # --- iterative improvement --------------------------------------------
    while time.time() - start < budget_s - 0.2:
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # gradient descent step (projected)
        grad = _gradient_shift(h, shift)
        cand = _project_onto_sum_box(h - step * grad, target_sum)

        cand_val = _c5_value(cand)
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_h = cand.copy()
            h = cand
            step = min(step * 1.08, 0.6)   # cautiously enlarge step
            improve_counter = 0
            continue

        # No improvement: reduce step and occasionally perturb
        step = max(step * 0.70, 1e-5)
        improve_counter += 1

        if improve_counter % 200 == 0:
            # small random perturbation around the current best
            pert = best_h + (rng.random(n) - 0.5) * 0.08
            cand = _project_onto_sum_box(pert, target_sum)
            cand_val = _c5_value(cand)
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()
                h = cand
                step = 0.08
                improve_counter = 0
                continue

        # occasional random swap (useful for binary‑like vectors)
        if rng.random() < 0.015:
            i, j = rng.choice(n, size=2, replace=False)
            swapped = h.copy()
            swapped[i], swapped[j] = swapped[j], swapped[i]
            swapped_val = _c5_value(swapped)
            if swapped_val < best_val:
                best_val = swapped_val
                best_h = swapped.copy()
                h = swapped
                step = 0.05
                improve_counter = 0

    # final evaluation (ensuring feasibility)
    best_h = _project_onto_sum_box(best_h, target_sum)
    final_val = _c5_value(best_h)

    return (best_h.tolist(), float(final_val), int(n))
