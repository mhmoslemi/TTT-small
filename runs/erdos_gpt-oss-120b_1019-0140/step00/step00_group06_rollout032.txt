    best_score = compute_c5(best_h)

    cur_score = best_score

    while time.time() - start < time_limit:
        # temperature linearly decays based on elapsed
        elapsed = time.time() - start
        t = start_temp + (end_temp - start_temp) * (elapsed / time_limit)
        # move: pick i,j; epsilon = (max_eps) * random * 0.1 maybe?

def hill_climb(h, time_limit):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    start = time.time()
    # parameters
    target_iter = 0  # we run until time limit
    # Keep track of last improvement time to trigger perturbation
    last_improve_time = start
    # We'll keep simple move acceptance criteria:
    while time.time() - start < time_limit:
        # Choose move size adaptively based on progress
        # We can choose epsilon scaling with remaining time or random
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 1e-12:
            continue
        # Choose epsilon as random fraction up to 0.5 of max_eps
        eps = max_eps * np.random.rand() * 0.5
        if eps < 1e-8:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_score = compute_c5(new_h)
        # accept if better, else maybe accept with probability if improvement small? Use simple SA.
        if new_score < cur_score:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h
                best_score = new_score
                last_improve_time = time.time()
        else:
            # occasional uphill acceptance, probability decreasing with time
            elapsed = time.time() - start
            # define temperature dropping linearly from t0 to t1
            t0, t1 = 0.0005, 1e-7
            temperature = t0 + (t1 - t0) * (elapsed / time_limit)
            if np.random.rand() < np.exp(-(new_score - cur_score) / temperature):
                cur_h = new_h
                cur_score = new_score
        # random perturbation if stuck

def fix_sum(h):
    n = h.shape[0]
    target = n / 2.0
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        # adjust using first element that can absorb diff
        for i in range(n):
            if diff > 0:
                avail = 1.0 - h[i]
                if avail > 0:
                    add = min(diff, avail)
                    h[i] += add
                    diff -= add
            else:
                avail = h[i]
                if avail > 0:
                    sub = min(-diff, avail)
                    h[i] -= sub
                    diff += sub
            if abs(diff) < 1e-12:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h
```

import numpy as np, time, random, math

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def fix_sum(h):
    n = h.shape[0]
    target = n / 2.0
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0:
                avail = 1.0 - h[i]
                if avail > 0:
                    add = min(diff, avail)
                    h[i] += add
                    diff -= add
            else:
                avail = h[i]
                if avail > 0:
                    sub = min(-diff, avail)
                    h[i] -= sub
                    diff += sub
            if abs(diff) < 1e-12:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h
