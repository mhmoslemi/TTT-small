def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n

    # We then multiply by dx.

def simulated_annealing(h, rng, time_limit):
    n = len(h)
    curr = h.copy()
    curr_score = compute_c5(curr)
    best = curr.copy()
    best_score = curr_score

    while time.time() < time_limit:

        # find random i where curr[i]==1, j where curr[j]==0
        ones = np.where(curr == 1)[0]

        curr[i] = 0
        curr[j] = 1
        new_score = compute_c5(curr)
        # acceptance

                best = curr.copy()
                best_score = new_score
                # optionally restart temperature?

            curr[i] = 1
            curr[j] = 0

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function `h`."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def _compute_c5_details(h: np.ndarray):
    """Return (C5, shift_of_max, full correlation array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = float(corr[max_idx]) * dx
    shift = max_idx - (n - 1)   # positive => right shift
    return max_val, shift, corr

    except Exception:  # initial_h_values not available
