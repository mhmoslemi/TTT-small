    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and initial feasible vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
        h_current = init.copy()
        # Basic feasibility check; if it fails fall back to the uniform vector.
        target_sum = n_points / 2.0
        if not (np.all((h_current >= 0.0) & (h_current <= 1.0))
                and np.isclose(np.sum(h_current), target_sum, atol=1e-9)):
            h_current = np.full(n_points, 0.5)
    else:
        n_points = 45  # default size when no parent is provided
        h_current = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure feasibility (box and sum) of the starting point.
    s = np.sum(h_current)
    if not np.isclose(s, target_sum, atol=1e-12):
        # Simple projection onto the equality hyperplane (rescale, then clip)
        h_current = h_current * (target_sum / s)
        h_current = np.clip(h_current, 0.0, 1.0)
        s = np.sum(h_current)
        if not np.isclose(s, target_sum, atol=1e-12):
            # Final uniform correction – deviation is tiny and acceptable.
            diff = target_sum - s
            h_current += diff / n_points
            h_current = np.clip(h_current, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Simulated annealing core
    # ----------------------------------------------------------------------
    def propose_move(h, sigma):
        """Generate a feasible neighbour by transferring mass between two coordinates."""
        n = h.shape[0]
        i, j = rng.choice(n, size=2, replace=False)
        # delta added to h[i] and subtracted from h[j]
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            return None  # No feasible move for this pair
        delta = rng.normal(0.0, sigma)
        delta = max(lo, min(hi, delta))
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        return h_new

    best_h = h_current.copy()
    best_score = _compute_c5(best_h, dx)

    current_h = h_current.copy()
    current_score = best_score

    start_time = time.time()
    # Linear cooling schedule
    T_start = 0.02
    T_end = 1e-5
    sigma_start = 0.03
    sigma_end = 1e-6
    # Restart control
    no_improve = 0
    max_no_improve = 50000

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        frac = elapsed / budget_s
        T = T_start * (1.0 - frac) + T_end * frac
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac

        h_new = propose_move(current_h, sigma)
        if h_new is None:
            continue

        new_score = _compute_c5(h_new, dx)

        if new_score < current_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - current_score) / T) if T > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            current_h = h_new
            current_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Random restart if stuck for a while
        if no_improve >= max_no_improve:
            # Simple random feasible vector (uniform 0–1 then rescale)
            for _ in range(10):
                rand_vec = rng.random(n_points)
                rand_vec = rand_vec / np.sum(rand_vec) * target_sum
                if np.all(rand_vec <= 1.0):
                    break
            else:
                rand_vec = np.full(n_points, 0.5)
            # Project onto constraints (already satisfied)
            current_h = rand_vec
            current_score = _compute_c5(current_h, dx)
            no_improve = 0

    # ----------------------------------------------------------------------
    # Final post‑processing to guarantee exact feasibility
    # ----------------------------------------------------------------------
    # Rescale to meet the integral exactly (tiny adjustments)
    s = np.sum(best_h)
    if not np.isclose(s, target_sum, atol=1e-12):
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)
        # If clipping altered the sum, correct uniformly
        s2 = np.sum(best_h)
        if not np.isclose(s2, target_sum, atol=1e-12):
            diff = target_sum - s2
            best_h += diff / n_points
            best_h = np.clip(best_h, 0.0, 1.0)

    final_score = _compute_c5(best_h, dx)

    return best_h, final_score, n_points
