    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo + hi)/2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if abs(s - target_sum) < tol:
            return w
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    # after loop, return clipped at mid
    return np.clip(v + mid, lower, upper)

import numpy as np, time, random

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max(), corr

def project(v, target_sum):
    # binary search lambda
    lower = -1e6
    upper = 1e6
    for _ in range(60):
        mid = (lower+upper)/2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            upper = mid
        else:
            lower = mid
    return np.clip(v + (lower+upper)/2.0, 0.0, 1.0)

def subgradient_step(h, target_sum, lr):
    N = len(h)
    dx = 2.0/N
    C, corr = compute_c5(h)
    idx = np.argmax(corr)  # index of max correlation
    shift = idx - (N-1)
    # compute gradient
    indices = np.arange(N)
    idx_plus = indices + shift
    idx_minus = indices - shift
    h_plus = np.where((idx_plus >=0) & (idx_plus < N), h[idx_plus], 0.0)
    h_minus = np.where((idx_minus >=0) & (idx_minus < N), h[idx_minus], 0.0)
    grad = dx * (1.0 - (h_plus + h_minus))
    # descent
    v = h - lr * grad
    # projection
    h_new = project(v, target_sum)
    return h_new, C

import time
import random
import numpy as np

# ----- core utilities ---------------------------------------------------------

def compute_c5(h):
    """Return (max_overlap, full_corr_array) for a step vector h."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max()), corr


def project_onto_sum_and_bounds(v, target_sum, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    """Project vector v onto {x: lo<=x<=hi, sum(x)=target_sum} by binary search on a shift."""
    # binary search for lambda such that sum(clip(v+lam, lo, hi)) == target_sum
    lam_low, lam_high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v + lam, lo, hi)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lam_high = lam
        else:
            lam_low = lam
    return np.clip(v + (lam_low + lam_high) * 0.5, lo, hi)


def subgradient_step(h, target_sum, lr):
    """One sub‑gradient descent step on the max‑overlap objective."""
    N = len(h)
    dx = 2.0 / N
    # correlation & worst shift
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)                     # integer lag
    i = np.arange(N)
    ip = i + shift
    im = i - shift
    h_ip = np.where((0 <= ip) & (ip < N), h[ip], 0.0)
    h_im = np.where((0 <= im) & (im < N), h[im], 0.0)
    grad = dx * (1.0 - (h_ip + h_im))        # dC/dh
    v = h - lr * grad
    h_new = project_onto_sum_and_bounds(v, target_sum)
    new_score, _ = compute_c5(h_new)
    return h_new, new_score

def solve_ilp(N, time_limit):
    """Attempt a MILP solution (binary h) to maximize the minimal overlap."""
    try:
        import cvxpy as cp
    except Exception:
        return None
    target = N // 2  # exact integer
    # variables
    h = cp.Variable(N, integer=True)
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]

    y_shifts = []          # list of y vectors, one per shift
    for j in range(N):
        length = N - j
        yj = cp.Variable(length, integer=True)
        constraints += [yj >= 0, yj <= 1]
        constraints += [yj <= h[:length]]
        constraints += [yj <= h[j:]]
        constraints += [yj >= h[:length] + h[j:] - 1]
        y_shifts.append(yj)

    t = cp.Variable(integer=True)
    constraints += [t >= 0]
    for yj in y_shifts:
        constraints.append(cp.sum(yj) >= t)

    prob = cp.Problem(cp.Maximize(t), constraints)

    # try GLPK_MI with a time limit; fall back to CBC if unavailable
    try:
        prob.solve(solver=cp.GLPK_MI,
                    glpk_opts={"tm_lim": int(time_limit * 1000)},
                    warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    h_val = np.rint(np.array(h.value)).astype(int)
    # ensure feasibility (project just in case)
    h_val = project_onto_sum_and_bounds(h_val.astype(float), target)
    c5_val, _ = compute_c5(h_val)
    return h_val, c5_val

                curr_h = cand

    # final projection and evaluation
    best_h = project_onto_sum_and_bounds(best_h, target_sum)
    final_score, _ = compute_c5(best_h)

    return (best_h.tolist(), float(final_score), N)
