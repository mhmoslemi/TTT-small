def repair_to_target(h):
    n = len(h)
    target = n / 2.0
    # Clip
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    # Scale globally (to get near target)
    if total > 0:
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)
    # Adjust remaining diff iteratively
    diff = target - h.sum()
    # tolerance
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to add mass
            idxs = np.where(h < 1.0)[0]
            if len(idxs) == 0:
                break
            # add proportionally maybe equal distribution
            # compute total capacity
            caps = 1.0 - h[idxs]
            cap_sum = caps.sum()
            if cap_sum < 1e-12:
                break
            # add proportionally up to diff
            add = np.minimum(caps, caps * diff / cap_sum)
            h[idxs] += add
        else:
            # need to remove mass
            idxs = np.where(h > 0.0)[0]
            if len(idxs) == 0:
                break
            caps = h[idxs]
            cap_sum = caps.sum()
            if cap_sum < 1e-12:
                break
            sub = np.minimum(caps, caps * (-diff) / cap_sum)
            h[idxs] -= sub
        diff = target - h.sum()
    # Clip final
    h = np.clip(h, 0.0, 1.0)
    return h

def repair_to_target(h):
    n = len(h)
    target = n / 2.0
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total > 0:
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)
    # Now adjust diff
    diff = target - h.sum()
    # tolerance
    eps = 1e-12
    # Use iterative proportional redistribution
    iter_cnt = 0
    while abs(diff) > eps and iter_cnt < 100:
        if diff > 0:
            # need to add mass
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            caps = 1.0 - h[idxs]
            cap_sum = caps.sum()
            if cap_sum < eps:
                break
            # distribute proportionally; we can add min(cap, caps * diff/cap_sum) to each
            add = np.minimum(caps, caps * (diff / cap_sum))
            h[idxs] += add
        else:
            # need to remove mass
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            caps = h[idxs]
            cap_sum = caps.sum()
            if cap_sum < eps:
                break
            sub = np.minimum(caps, caps * ((-diff) / cap_sum))
            h[idxs] -= sub
        diff = target - h.sum()
        iter_cnt += 1
    # final clip just to be safe
    h = np.clip(h, 0.0, 1.0)
    # enforce exact sum by adjusting a random index if still off small tolerance
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # pick an index that can accommodate diff
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
        else:
            idxs = np.where(h > 0.0)[0]
        if idxs.size > 0:
            idx = idxs[0]
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

        h = np.clip(h * scale, 0.0, 1.0)

    diff = target - h.sum()
    eps = 1e-12
    iter_cnt = 0
    while abs(diff) > eps and iter_cnt < 100:

                break

            h[idxs] += add

            cap_sum = caps.sum()
            if cap_sum < eps:
                break

            h[idxs] -= sub
        diff = target - h.sum()

    # final tiny correction if needed
    diff = target - h.sum()

                current_score = cand_score

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_to_target(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total > 0:
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)

            cap_sum = caps.sum()

def hill_climb(start_h: np.ndarray, end_time: float) -> tuple[np.ndarray, float]:
    """Simple pairwise‑exchange hill climbing with occasional random acceptance."""
    n = start_h.shape[0]
    best = start_h.copy()
    best_score = compute_c5(best)
    cur = best.copy()
    cur_score = best_score

    while time.time() < end_time:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        hi, hj = cur[i], cur[j]

        lo = max(-hi, hj - 1.0)
        hi_delta = min(1.0 - hi, hj)
        if hi_delta <= lo:

        cand[i] = hi + delta
        cand[j] = hj - delta
        cand_score = compute_c5(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:

                cur, cur_score = cand, cand_score
