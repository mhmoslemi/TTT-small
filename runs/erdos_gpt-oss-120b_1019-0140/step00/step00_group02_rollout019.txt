def project(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    # If total equals target within tol, return.
    if abs(total - target) < 1e-12:
        return v
    # Determine direction
    if total < target:
        # need to add
        deficit = target - total
        # indexes where v < 1
        mask = v < 1.0 - 1e-12
        # while deficit > 0
        while deficit > 1e-12 and np.any(mask):
            free = 1.0 - v[mask]
            # distribute equally, but not exceed capacity
            add = np.minimum(free, deficit / mask.sum())
            v[mask] += add
            deficit = target - v.sum()
            mask = v < 1.0 - 1e-12
    else:
        # total > target: need to remove
        excess = total - target
        mask = v > 1e-12
        while excess > 1e-12 and np.any(mask):
            avail = v[mask]
            sub = np.minimum(avail, excess / mask.sum())
            v[mask] -= sub
            excess = v.sum() - target
            mask = v > 1e-12
    # final clipping
    v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time
import math

def _evaluate(h: np.ndarray) -> float:
    """Return C5 bound for a given step function sampled on uniform grid."""
    n = h.shape[0]
    # correlation of h with (1-h); np.correlate uses zero padding outside range
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))


def _binary_from_continuous(v: np.ndarray) -> np.ndarray:
    """Return a binary vector with exactly half 1's (rounded down if odd)."""
    n = v.shape[0]
    m = n // 2
    idx = np.argpartition(-v, m)[:m]          # indices of the m largest entries
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def _hill_climb(h_start: np.ndarray, max_no_improve: int,
                start_time: float, time_limit: float,
                target_score: float = 0.3808) -> tuple:
    """Greedy hill climbing on binary vectors using 1↔0 swaps."""
    h = h_start.copy()
    cur_score = _evaluate(h)
    best_h = h.copy()
    best_score = cur_score
    no_improve = 0
    n = h.shape[0]

    while time.time() - start_time < time_limit and no_improve < max_no_improve:
        # pick a random 1 and a random 0
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # perform the swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = _evaluate(h)

        if new_score + 1e-12 < cur_score:          # improvement
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target_score:
                    return best_h, best_score
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

    return best_h, best_score


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize the Erdős C5 constant bound.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial continuous construction (if supplied)
    # ------------------------------------------------------------------
    init_vec = globals().get('initial_h_values')
    if init_vec is None:
        # fallback: random continuous sample of length 45
        base_n = 45
        cont = np.random.rand(base_n)
    else:
        cont = np.asarray(init_vec, dtype=np.float64)
        base_n = cont.shape[0]

    # ------------------------------------------------------------------
    # 2. Choose an even target grid size (larger grids give more flexibility)
    # ------------------------------------------------------------------
    target_n = base_n * 2               # double the resolution
    if target_n % 2 == 1:
        target_n += 1

    # Linear interpolation to the finer grid
    old_x = np.linspace(0.0, 2.0, base_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, target_n, endpoint=False)
    finer = np.interp(new_x, old_x, cont)

    # ------------------------------------------------------------------
    # 3. Initialise a binary candidate (exactly half 1's)
    # ------------------------------------------------------------------
    cur_h = _binary_from_continuous(finer)
    n = target_n

    # ------------------------------------------------------------------
    # 4. Run greedy hill‑climbing from the seeded candidate
    # ------------------------------------------------------------------
    best_h, best_score = _hill_climb(
        cur_h,
        max_no_improve=5000,
        start_time=start,
        time_limit=budget_s * 0.6,          # reserve part of budget for restarts
        target_score=0.3808
    )

    # ------------------------------------------------------------------
    # 5. Additional random restarts (if time permits)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # random binary start
        perm = np.random.permutation(n)
        rand_h = np.zeros(n, dtype=np.float64)
        rand_h[perm[:n // 2]] = 1.0

        # hill‑climb from this start
        h_loc, score_loc = _hill_climb(
            rand_h,
            max_no_improve=5000,
            start_time=start,
            time_limit=budget_s * 0.9,
            target_score=0.3808
        )

        if score_loc < best_score:
            best_score = score_loc
            best_h = h_loc.copy()
            if best_score <= 0.3808:
                break

    # ------------------------------------------------------------------
    # 6. Return the best solution found
    # ------------------------------------------------------------------
    return best_h, best_score, n
