def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time, math, random
    
    rng = np.random.default_rng(seed)
    start = time.time()
    
    # Determine n_points
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        n = len(h_init)
    else:
        n = 70
        h_init = None
    
    if h_init is None:
        # start from random binary vector with sum n/2
        target_sum = n // 2
        h = np.zeros(n, dtype=np.int8)
        # choose target_sum indices at random
        h[rng.choice(n, size=target_sum, replace=False)] = 1
    else:
        # convert initial to binary (preserve sum constraint)
        # Since values may be fractional, threshold at 0.5
        target_sum = n // 2
        # Sort indices by value descending and set top target_sum to 1
        idx = np.argsort(-h_init)
        h = np.zeros(n, dtype=np.int8)
        h[idx[:target_sum]] = 1
    # compute current score function
    def c5_score(vec):
        # compute max correlation * dx
        corr = np.correlate(vec, 1 - vec, mode='full')
        return np.max(corr) * (2.0 / n)
    
    current = h.copy()
    curr_score = c5_score(current)
    best = current.copy()
    best_score = curr_score
    
    # Simulated annealing parameters
    temperature = 0.5  # initial temperature
    # If initial solution already has low score, we can start lower
    # Temperature schedule
    decay = 0.9995
    # iteration counter
    it = 0
    # we shall also try random restarts after stagnation
    last_improve_iter = 0
    # loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s * 0.95:  # stop with margin
            break
        # propose a swap
        # pick two indices of opposite values
        ones_idx = np.where(current == 1)[0]
        zeros_idx = np.where(current == 0)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # create new vector
        cand = current.copy()
        cand[i] = 0
        cand[j] = 1
        cand_score = c5_score(cand)
        # acceptance
        if cand_score < curr_score:
            accept = True
        else:
            # Metropolis
            delta = cand_score - curr_score
            accept = math.exp(-delta / max(temperature, 1e-12)) > rng.random()
        if accept:
            current = cand
            curr_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                last_improve_iter = it
        # temperature decay

        if temperature < 1e-6:
            temperature = 1e-6
        it += 1

import numpy as np
import time
import math

def compute_c5(vec: np.ndarray) -> float:
    """Maximum overlap integral for a given step function vector."""
    n = vec.shape[0]
    dx = 2.0 / n
    corr = np.correlate(vec, 1 - vec, mode='full')
    return float(np.max(corr) * dx)

def make_binary_from_initial(arr: np.ndarray) -> np.ndarray:
    """Convert a possibly non‑binary initial array to a feasible binary vector."""
    n = arr.shape[0]
    target = n // 2
    idx = np.argsort(-arr)                # descending order
    h = np.zeros(n, dtype=np.float64)
    h[idx[:target]] = 1.0
    # If n is odd, give a single 0.5 entry to meet the exact integral constraint
    if n % 2 == 1:
        h[idx[target]] = 0.5
    return h

def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible binary (or half‑binary) vector."""
    target = n // 2
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, size=target, replace=False)] = 1.0
    if n % 2 == 1:
        # add a single 0.5 to satisfy sum = n/2
        zero_idx = np.where(h == 0)[0]
        h[rng.choice(zero_idx)] = 0.5
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimize the Erdős C5 bound and return (h, bound, n_points)."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------------
    # Determine problem size and initial feasible vector
    # --------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        n = init_arr.shape[0]
        # Preserve binary structure if already binary, otherwise binarize
        if np.all(np.isin(init_arr, [0.0, 1.0])):
            current = init_arr.copy()
        else:
            current = make_binary_from_initial(init_arr)
    else:
        n = 70
        current = random_binary_vector(n, rng)

    # Ensure the exact sum constraint (sum = n/2)
    target_sum = n / 2.0
    cur_sum = float(current.sum())
    if not np.isclose(cur_sum, target_sum):
        diff = int(round(target_sum - cur_sum))
        if diff > 0:
            zero_idx = np.where(current == 0)[0]
            sel = rng.choice(zero_idx, size=diff, replace=False)
            current[sel] = 1.0
        elif diff < 0:
            one_idx = np.where(current == 1)[0]
            sel = rng.choice(one_idx, size=-diff, replace=False)
            current[sel] = 0.0

    # --------------------------------------------------------------------
    # Simulated annealing core
    # --------------------------------------------------------------------
    best = current.copy()
    best_score = compute_c5(best)
    curr_score = best_score

    temperature = 1.0          # initial temperature
    decay = 0.9999             # per‑iteration decay factor
    it = 0
    last_improve = 0
    max_no_improve = 200_000   # restart after this many idle iterations

    # Helper to pick random opposite‑valued indices (0↔1)
    def pick_swap(vec):
        ones = np.where(vec == 1)[0]
        zeros = np.where(vec == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        return i, j

    while True:
        # Periodic time check (every ~1k iterations)
        if (it & 0x3FF) == 0:
            if time.time() - start_time > budget_s * 0.95:
                break

        # Propose a simple swap move (preserves sum)
        i, j = pick_swap(current)
        candidate = current.copy()
        candidate[i] = 0.0
        candidate[j] = 1.0

        cand_score = compute_c5(candidate)

        if cand_score < curr_score:

            curr_score = compute_c5(current)
            last_improve = it

    h_best = best.astype(np.float64)
    try:
        final_bound = evaluate_erdos_solution(h_best)
    except Exception:
        final_bound = compute_c5(h_best)
