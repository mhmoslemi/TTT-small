        g = np.ones(n)
        if lag > 0:

            g[:n-lag] -= h[lag:]
            g[lag:] -= h[:n-lag]
        elif lag < 0:
            L = -lag
            g[L:] -= h[:n-L]

        else:
            g = 1 - 2*h

            new_h[j] -= delta

import time
import math
import numpy as np

def compute_c5(h):
    """Return the overlap constant C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full mode correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)


def local_search(h0, max_iters, rng, time_limit):
    """Simple pairwise hill climbing that preserves the sum constraint."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.02  # initial step size

    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break

        # correlation and lag where the max occurs
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)

        # gradient of the max correlation entry
        g = np.ones(n)
        if lag > 0:
            g[: n - lag] -= h[lag:]
            g[lag:] -= h[: n - lag]
        elif lag < 0:
            L = -lag
            g[L:] -= h[: n - L]
            g[: n - L] -= h[L:]
        else:
            g = 1.0 - 2.0 * h

        # indices that would decrease / increase the objective
        inc_idxs = np.where(g < -1e-12)[0]
        dec_idxs = np.where(g > 1e-12)[0]

        if inc_idxs.size and dec_idxs.size:
            i = int(rng.choice(inc_idxs))
            j = int(rng.choice(dec_idxs))
        else:
            # fall back to random pair
            i, j = rng.integers(n, size=2)
            while i == j:
                j = rng.integers(n)

        # feasible delta respecting bounds
        delta = min(step, 1.0 - h[i], h[j])
        if delta <= 0:
            continue

        h[i] += delta
        h[j] -= delta

        cur_c5 = compute_c5(h)
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # slowly decay the step size
        step *= 0.9995

    return best_h, best_c5


def solve_milp(n, remaining_time):
    """Binary MILP formulation using cvxpy. Returns (h, t) where t is the max overlap."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # Variables
    x = cp.Variable(n, integer=True)
    t = cp.Variable()
    constraints = [
        cp.sum(x) == n // 2,
        t >= 0,
        x >= 0,
        x <= 1,
    ]

    # For each positive lag create product variables y_{i,k}
    for k in range(1, n):
        m = n - k
        y = cp.Variable(m, integer=True)
        constraints += [
            y >= 0,
            y <= 1,
            y <= x[:m],
            y <= x[k:],
            y >= x[:m] + x[k:] - 1,
            cp.sum(x) - cp.sum(y) <= t,
        ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    # attempt solve, give a small slack to avoid blowing the time budget
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False)
    except Exception:
        return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    x_val = np.rint(np.clip(x.value, 0, 1)).astype(float)
    t_val = float(t.value)
    return x_val, t_val
