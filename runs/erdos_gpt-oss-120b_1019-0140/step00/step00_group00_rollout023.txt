function hill_climb(h):
  best = h, best_val = compute_c5.
  while True:
    improved=False
    for each i where h[i]==1:
      for each j where h[j]==0:
        h_swapped = h.copy()
        h_swapped[i]=0; h_swapped[j]=1
        val = compute_c5(h_swapped)
        if val < best_val - 1e-9: # improvement
            best = h_swapped; best_val = val; improved=True; break loops
    if not improved: break
  return best, best_val

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def project_feasible(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection onto {0≤x_i≤1, sum x_i = target_sum} via water‑filling."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - hi, 0.0, 1.0)
    # tiny adjustment for numeric drift
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(mask)[0]
        if idx.size:
            delta = diff / idx.size
            x[idx] += delta
            np.clip(x, 0.0, 1.0, out=x)
    return x

def gradient_of_worst(h: np.ndarray):
    """Sub‑gradient of the correlation term attaining the current maximum."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                     # integer shift producing the max
    grad = np.zeros(n, dtype=float)

    if k >= 0:
        # contribution of ∑ h_i
        grad[:n - k] = dx
        # -h_i*h_{i+k} part
        grad[:n - k] -= dx * h[k:]       # subtract h_{i+k}
        grad[k:]   -= dx * h[:n - k]     # subtract h_{i-k}
    else:
        kk = -k
        grad[kk:] = dx
        grad[kk:] -= dx * h[:n - kk]    # subtract h_{j-kk}
        grad[:n - kk] -= dx * h[kk:]    # subtract h_{j+kk}
    return grad, k

def optimize_h(h0: np.ndarray, max_iters: int, time_limit: float):
    """Projected sub‑gradient descent with occasional random perturbations."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    step0 = 1.0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        grad, _ = gradient_of_worst(h)
        step = step0 / np.sqrt(it + 1.0)
        y = h - step * grad
        h = project_feasible(y, target)
        val = compute_c5(h)
        if val < best_val:

            best_h = h.copy()

        if (it + 1) % 200 == 0 and np.random.rand() < 0.3:
            noise = np.random.normal(scale=0.1, size=n)
            h = project_feasible(h + noise, target)
    return best_h, best_val

    # Determine problem size and a reasonable starting point
