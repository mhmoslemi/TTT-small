import numpy as np, time, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute correlation for shifts from -(n-1) to (n-1)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
```

                        new_h[i] += delta
                        new_h[j] += delta
                        new_h[d] -= 2.0 * delta
                        new_c5 = _c5_value(new_h)

                        # Metropolis acceptance
                        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
                            cur_h, cur_c5 = new_h, new_c5
                            if cur_c5 < best_c5:
                                best_h, best_c5 = cur_h.copy(), cur_c5
                                if best_c5 <= 0.3808:
                                    break
                        continue  # move on to next iteration

        # --------------------------------------------------------------
        # generic pair‑transfer move (random)
        # --------------------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 1e-12:
            continue
        delta = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = _c5_value(new_h)

        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
                if best_c5 <= 0.3808:
                    break

    # ------------------------------------------------------------------
    # final feasibility tweak (exact sum)
    # ------------------------------------------------------------------
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        # distribute the tiny remainder on the first free coordinate
        for idx in range(n):
            if diff > 0 and best_h[idx] < 1.0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            elif diff < 0 and best_h[idx] > 0.0:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    np.clip(best_h, 0.0, 1.0, out=best_h)

    # ------------------------------------------------------------------
    # final C5 evaluation (use provided helper if present)
    # ------------------------------------------------------------------
    try:
        c5_bound = evaluate_erdos_solution(best_h)
    except Exception:
        c5_bound = _c5_value(best_h)

    return best_h, float(c5_bound), n
