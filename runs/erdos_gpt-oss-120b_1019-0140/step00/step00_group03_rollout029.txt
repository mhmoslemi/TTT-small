import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """
    Compute the C5 bound for a step function represented by the array h.
    h should contain values in [0, 1]; for binary vectors it is 0/1.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)


def make_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a binary vector of length n containing exactly n//2 ones.
    (n is assumed even; if odd the function will put floor(n/2) ones.)
    """
    half = n // 2
    vec = np.zeros(n, dtype=int)
    ones_idx = rng.choice(n, size=half, replace=False)
    vec[ones_idx] = 1
    return vec


def prepare_initial_vector(seed: int, rng: np.random.Generator):
    """
    Return a binary start vector and its length.
    Uses the pre‑imported `initial_h_values` when available,
    otherwise generates a random binary vector.
    """
    try:
        # initial_h_values is provided by the harness (might be list or ndarray)
        init = np.asarray(initial_h_values).astype(float)
        n = init.shape[0]
        # binarise via simple threshold
        h = (init > 0.5).astype(int)
        half = n // 2
        cur = int(h.sum())
        if cur != half:
            # adjust to exact half using the provided rng
            if cur < half:
                zero_idx = np.where(h == 0)[0]
                add = rng.choice(zero_idx, size=half - cur, replace=False)
                h[add] = 1
            else:
                one_idx = np.where(h == 1)[0]
                rem = rng.choice(one_idx, size=cur - half, replace=False)
                h[rem] = 0
        return h, n
    except NameError:
        # No parent construction – fall back to a random binary vector of length 70
        n = 70
        h = make_binary_vector(n, rng)
        return h, n


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a binary step function h on [0,2] that minimizes C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.97  # a safety margin before the kill

    # initialise
    current, n = prepare_initial_vector(seed, rng)
    current_score = compute_c5(current)
    best_h = current.copy()
    best_score = current_score

    # simulated‑annealing parameters
    temp = 1.0
    min_temp = 1e-4
    cool_rate = 0.9995
    stagnation = 0
    max_stagnation = 20000  # restart after this many non‑improving steps
    target_score = 0.3808

    # main loop
    it = 0
    while True:
        # time check
        if time.time() - start_time > time_limit:
            break
        # early exit if we already beat the target
        if best_score <= target_score:
            break

        # choose a random 1‑index and a random 0‑index to swap
        ones_idx = np.where(current == 1)[0]
        zeros_idx = np.where(current == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break  # should never happen with a balanced vector
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # propose swap
        trial = current.copy()
        trial[i] = 0
        trial[j] = 1
        trial_score = compute_c5(trial)

        # acceptance test
        if trial_score <= current_score:
            accept = True
        else:
            # Metropolis criterion
            delta = trial_score - current_score
            prob = np.exp(-delta / max(temp, min_temp))
            accept = rng.random() < prob

        if accept:
            current = trial
            current_score = trial_score
            # update best if improved
            if trial_score < best_score:
                best_h = trial.copy()
                best_score = trial_score
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # temperature cooling
        temp = max(temp * cool_rate, min_temp)

        # restart heuristic
        if stagnation >= max_stagnation:
            # re‑initialise from a fresh random balanced vector
            current = make_binary_vector(n, rng)
            current_score = compute_c5(current)
            stagnation = 0

        it += 1

    # final check – compute bound one last time to avoid numerical drift
    final_score = compute_c5(best_h)

    return best_h, final_score, n
