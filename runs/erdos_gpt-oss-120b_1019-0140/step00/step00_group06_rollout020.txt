import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Construct a monotone linear ramp h on [0,2] that satisfies the integral
    constraint exactly and yields a low C5 bound.
    Returns:
        h_values (np.ndarray): sample values of h on a uniform grid.
        c5_bound (float): maximal overlap integral computed via correlation.
        n_points (int): number of samples used.
    """
    # Choose a modest resolution (< 1000) for speed and sufficient accuracy.
    n = 500  # can be any integer <= 1000; larger n gives C5 → 1/3.
    # Linear ramp: h_i = (i + 0.5) / n evaluated at midpoints of intervals.
    h = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Clip to [0, 1] for safety (should already hold).
    np.clip(h, 0.0, 1.0, out=h)

    # Correct any floating‑point drift so that sum(h) == n/2 exactly.
    target_sum = n / 2.0
    current_sum = h.sum()
    diff = target_sum - current_sum
    if abs(diff) > 1e-14:          # tiny adjustment needed
        # Adjust the first entry (still within bounds because diff is tiny).
        new_val = h[0] + diff
        new_val = max(0.0, min(1.0, new_val))
        h[0] = new_val

    dx = 2.0 / n
    # Correlation of h with (1 - h); mode='full' yields all integer shifts.
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(corr.max())

    return h, c5, n
