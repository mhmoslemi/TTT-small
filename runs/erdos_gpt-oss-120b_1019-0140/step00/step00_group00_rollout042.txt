import numpy as np
import time
import math
import random

def compute_cmax(h: np.ndarray) -> float:
    """Maximum overlap integral C₅ for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust the total sum to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # distribute the remaining mass uniformly, then re‑clip
    h += diff / h.size
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # fix any tiny residual on the first element that still respects bounds
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h with a low C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise discretisation
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = globals()["initial_h_values"]
        n0 = int(init.shape[0])
        n = max(100, n0 * 2)                      # a modest even size
        # linear interpolation of the parent construction
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xn = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xn, x0, init)
        h = np.clip(h, 0.0, 1.0)
    else:
        n = 200                                   # default resolution
        h = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0                          # ∑h = n/2  ⇔  ∫h = 1
    h = _project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_c = compute_cmax(best_h)

    cur_h = h.copy()
    cur_c = best_c

    start = time.time()
    max_time = budget_s * 0.95
    T0 = 0.08                                     # initial temperature for SA
    it_without_improve = 0

    while time.time() - start < max_time:
        # ---- propose a random mass transfer between two positions -------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # feasible delta range so that 0 ≤ h ≤ 1 and total sum unchanged
        delta_low = -min(cur_h[i], 1.0 - cur_h[j])
        delta_high = min(1.0 - cur_h[i], cur_h[j])
        if delta_low >= delta_high:
            continue

        delta = delta_low + (delta_high - delta_low) * np.random.rand()
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c = compute_cmax(new_h)

        # ---- Metropolis acceptance -----------------------------------------
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / max_time)
        accept = False
        if new_c < cur_c:
            accept = True
        elif T > 1e-12:
            prob = math.exp((cur_c - new_c) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_h = new_h
            cur_c = new_c
            it_without_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        else:
            it_without_improve += 1

        # ---- occasional restart if stuck ------------------------------------
        if it_without_improve > 10000:
            cur_h = np.random.rand(n)
            cur_h = _project_to_feasible(cur_h, target_sum)
            cur_c = compute_cmax(cur_h)
            it_without_improve = 0

    # ------------------------------------------------------------------
    # Final local polishing using the worst lag gradient
    # ------------------------------------------------------------------
    for _ in range(6):
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        lag_index = int(np.argmax(corr))
        L = lag_index - (n - 1)                    # signed shift producing max C₅

        # gradient of C_L = ∑ h_i (1 - h_{i-L})
        grad = np.ones(n)
        if L >= 0:
            grad[L:] -= best_h[:-L]
            grad[: n - L] -= best_h[L:]
        else:
            p = -L
            grad[: n - p] -= best_h[p:]
            grad[p:] -= best_h[: -p]

        step = 0.02
        new_h = best_h - step * grad
        new_h = _project_to_feasible(new_h, target_sum)
        new_c = compute_cmax(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h

    # ------------------------------------------------------------------
    # Compute the final bound with the official evaluator (more reliable)
    # ------------------------------------------------------------------
    final_c = evaluate_erdos_solution(best_h)

    # Return the step function, its bound and the discretisation size
    return best_h, float(final_c), int(n)
