def compute_c5(h):
    # fast compute using convolution
    n = h.size
    # correlation length 2*n-1
    corr = np.correlate(h, 1.0 - h, mode='full')

def hill_climb(initial_h, budget, start_time):
    h = initial_h.copy()
    cur_score = compute_c5(h)
    best_local_h = h.copy()
    best_local_score = cur_score
    n = h.size
    # get indices of ones, zeros each iteration inside loop
    while time.time() - start_time < budget:

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = prng.choice(ones)
        j = prng.choice(zeros)

        if new_score < cur_score:

            # keep if best overall

                best_local_h = h.copy()

            h[i], h[j] = h[j], h[i]

import time
import numpy as np

def compute_c5(h):
    """Return the overlap constant for a step function h."""
    # The harness provides evaluate_erdos_solution; use it if available.
    # Otherwise compute directly via correlation.
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / n)


def random_binary(prng, n):
    """Generate a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[prng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_from_continuous(v):
    """Turn a real‑valued vector into a binary one with the same integral."""
    n = v.shape[0]
    half = n // 2
    idx = np.argsort(-v)[:half]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def hill_climb(prng, start_h, target, budget_s, start_time):
    """
    Simple greedy local search that swaps a 1 with a 0.
    Stops after a timeout or after many unsuccessful attempts.
    """
    best_h = start_h.copy()
    best_score = compute_c5(best_h)
    cur_h = start_h.copy()
    cur_score = best_score
    n = cur_h.size
    no_improve = 0
    max_no_improve = 5000

    while time.time() - start_time < budget_s - 0.5:
        if no_improve >= max_no_improve:
            break
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = prng.choice(ones)
        j = prng.choice(zeros)

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)

        if new_score < cur_score:
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score <= target:
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            no_improve += 1

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1
    to minimize C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    prng = np.random.RandomState(seed)
    start_time = time.time()
    target_score = 0.3808  # early‑stop threshold

    # ------------------------------------------------------------------
    # Initialise the candidate vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        raw = np.array(initial_h_values, dtype=float)
        n = raw.size
        # Ensure correct integral (sum = n/2)
        half = n // 2
        # Original may already be binary and satisfy the sum.
        if np.all(np.logical_or(raw == 0, raw == 1)) and int(raw.sum()) == half:
            init_h = raw.copy()
        else:
            # Convert to a binary vector with the required number of ones.
            init_h = binary_from_continuous(raw)
        # Evaluate the raw (possibly fractional) version as an alternative.
        raw_score = compute_c5(raw)
        bin_score = compute_c5(init_h)
        if raw_score <= bin_score:
            init_h = raw.copy()
            best_score = raw_score
        else:
            best_score = bin_score
    else:
        n = 70
        init_h = random_binary(prng, n)
        best_score = compute_c5(init_h)

    best_h = init_h.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts + greedy hill climbing)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 1.0:
        # decide whether to start from the current best or from a fresh random point
        if prng.rand() < 0.15:
            start_vec = random_binary(prng, n)
        else:
            start_vec = best_h.copy()
            # add a tiny perturbation to diversify the search
            for _ in range(prng.randint(1, 4)):

                zeros = np.where(start_vec <= 0.5)[0]
                if ones.size == 0 or zeros.size == 0:
