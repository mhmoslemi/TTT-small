import numpy as np
import time

def project_box_simplex(x, lower, upper, target):
    """Project x onto the box [lower,upper] and the hyperplane sum = target."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) / 2.0
        y = np.clip(x - mid, lower, upper)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    return np.clip(x - lam, lower, upper)

def compute_c5(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discrete representation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    return (c * dx).max()

def gradient_max_shifts(h, dx):
    """Gradient of the current maximal correlation shift (subgradient)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = corr.max()
    mask = np.isclose(corr, max_val, atol=1e-12)
    idxs = np.flatnonzero(mask)
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)          # shift corresponding to this index
        i_start = max(0, -s)
        i_end = min(n - 1, n - 1 - s)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            grad[i] += 1.0 - h[i + s]
            grad[i + s] -= h[i]
    if idxs.size:
        grad /= idxs.size
    return grad, max_val

def optimize_h(h0, rng, max_iters, step0):
    """Local descent on a single resolution."""
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h_cur = project_box_simplex(np.clip(h0, 0.0, 1.0), 0.0, 1.0, target)

    best_h = h_cur.copy()
    best_val = compute_c5(h_cur, dx)

    for it in range(max_iters):
        grad, cur_val = gradient_max_shifts(h_cur, dx)

        step = step0 / np.sqrt(it + 1.0)
        h_new = h_cur - step * grad
        h_new = project_box_simplex(h_new, 0.0, 1.0, target)

        new_val = compute_c5(h_new, dx)

        if new_val <= cur_val or rng.random() < 0.001:
            h_cur, cur_val = h_new, new_val

        if cur_val < best_val:
            best_val, best_h = cur_val, h_cur.copy()

        if (it + 1) % 2000 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(h_cur[i], 1.0 - h_cur[j])
            if max_transfer > 1e-12:
                delta = rng.uniform(0.0, 0.5 * max_transfer)
                h_tmp = h_cur.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                val_tmp = compute_c5(h_tmp, dx)
                if val_tmp < best_val:
                    best_val, best_h = val_tmp, h_tmp.copy()
                    h_cur = h_tmp
                elif rng.random() < 0.001:
                    h_cur = h_tmp

    return best_h, best_val
