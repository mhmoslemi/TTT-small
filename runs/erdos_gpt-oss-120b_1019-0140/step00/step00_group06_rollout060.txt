    upper = 1.0
    # initial lambda bounds

        x = np.clip(v + lam, lower, upper)
        s = x.sum()
        if s > target:
            lam_high = lam
        else:
            lam_low = lam
    # final x
    x = np.clip(v + lam_low, lower, upper)
    # adjust for any tiny sum error due to precision
    # Distribute remaining difference uniformly to those not at bounds
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff evenly across non-bound indices
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, lower, upper)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def c5_score(h):
    """Return the C5 bound for a step function h (array of values)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)


def project_box_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-12, max_iter=100):
    """
    Project vector v onto the intersection of the box [lower, upper]^n
    and the hyperplane sum(x) = target_sum.
    Uses a simple binary search on the Lagrange multiplier.
    """
    # Find λ bounds such that clamped sum goes from 0 to n
    lam_low = -np.max(v) - 1.0   # makes v+λ <= 0 -> all zeros
    lam_high = upper - np.min(v) + 1.0  # makes v+λ >= upper -> all ones
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v + lam, lower, upper)
        s = x.sum()
        if s > target_sum:
            lam_high = lam
        else:
            lam_low = lam
        if abs(s - target_sum) <= tol:
            break
    x = np.clip(v + lam_low, lower, upper)
    # final tiny adjustment to hit target_sum exactly (if any free variables)
    diff = target_sum - x.sum()
    if abs(diff) > tol:
        free = (x > lower + tol) & (x < upper - tol)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, lower, upper)
    return x


def propose_move(h, max_delta=0.2):
    """Propose a feasible move that transfers mass between two indices."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # try moving from i to j
    max_transfer = min(h[i], 1.0 - h[j], max_delta)
    direction = 1
    if max_transfer <= 0.0:
        # try opposite direction
        max_transfer = min(h[j], 1.0 - h[i], max_delta)
        if max_transfer <= 0.0:
            return None  # no feasible move found
        i, j = j, i
        direction = -1
    delta = np.random.rand() * max_transfer
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def simulated_annealing(h0, max_seconds, target_sum):
    """Simple SA that respects the sum constraint via pairwise transfers."""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_score(h)
    n = h.size
    start = time.time()
    # temperature schedule
    T0 = 0.12
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:
            break
        T = T0 * max(0.0, 1.0 - elapsed / max_seconds)
        new_h = propose_move(h, max_delta=0.2)
        if new_h is None:
            continue
        # new_h already respects bounds and sum (pairwise transfer)
        val = c5_score(new_h)
        if val < best_val:
            best_val = val
            best_h = new_h.copy()
            h = new_h
        else:
            # Metropolis acceptance
            if T > 0 and np.random.rand() < np.exp(-(val - best_val) / T):
                h = new_h
    return best_h, best_val


def gradient_refine(h, steps, target_sum):
    """Few gradient-descent steps using the exact max-shift gradient."""
    n = h.size
    dx = 2.0 / n
    for t in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)  # lag corresponding to max correlation
        grad = np.zeros_like(h)
        if shift >= 0:
            k = shift
            grad[: n - k] += (1 - h[k:]) * dx
            grad[k:] -= h[: n - k] * dx
        else:
            k = -shift
            grad[k:] += (1 - h[: n - k]) * dx
            grad[: n - k] -= h[k:] * dx
        # step size shrinking
        step_size = 0.6 * (0.5 ** (t / steps))
        h_new = h - step_size * grad
        h = project_box_sum(h_new, target_sum)
    return h
