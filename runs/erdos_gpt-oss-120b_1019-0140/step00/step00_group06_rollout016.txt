import time
import numpy as np

def _project_onto_capped_simplex(v, target, max_iter=60, tol=1e-12):
    """Project v onto the set {0<=x_i<=1, sum x_i = target}."""
    lo = -np.max(v) - 1.0  # ensures all x_i = 1
    hi = np.max(v) + 1.0   # ensures all x_i = 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # final tiny correction to hit the sum more accurately
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # distribute the tiny diff over non‑saturated entries
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(mask):
            adjust = diff / np.count_nonzero(mask)
            x[mask] += adjust
            x = np.clip(x, 0.0, 1.0)
    return x
