def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx using discrete correlation."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / N
    return float(np.max(corr) * dx)
