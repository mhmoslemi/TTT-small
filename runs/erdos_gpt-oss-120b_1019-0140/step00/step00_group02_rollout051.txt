def enforce_sum(vec, target_sum):
    # Ensure sum = target_sum by scaling and clipping iteratively
    vec = np.clip(vec, 0.0, 1.0)
    total = np.sum(vec)
    if total == target_sum:
        return vec
    # Simple scaling, then clip and distribute leftover
    # Use iterative method to satisfy both bounds and sum exactly
    # start with scaling factor
    scale = target_sum / total if total > 0 else 0.0
    vec = vec * scale
    for _ in range(10):
        # correct any bound violations
        over = vec > 1.0
        under = vec < 0.0
        if not np.any(over) and not np.any(under):
            break
        # handle over
        if np.any(over):
            excess = np.sum(vec[over] - 1.0)
            vec[over] = 1.0
            # distribute excess to non-over elements
            mask = ~over
            count = np.sum(mask)
            if count > 0:

        # handle under
        if np.any(under):

            vec[under] = 0.0
            mask = ~under
            count = np.sum(mask)
            if count > 0:

import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def generate_neighbor(h: np.ndarray, step_scale: float) -> np.ndarray:
    """Swap a small amount of mass between two entries while preserving sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi = h[i]
    hj = h[j]
    low = max(-step_scale, -hi, hj - 1.0)
    high = min(step_scale, 1.0 - hi, hj)
    if low >= high:
        return h.copy()
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def binary_neighbor(h: np.ndarray) -> np.ndarray:
    """Force one entry to 0 or 1 and compensate another entry to keep the sum."""
    n = h.shape[0]
    i = np.random.randint(n)
    new_val = np.random.choice([0.0, 1.0])
    delta = new_val - h[i]  # amount to add to the total sum (positive if we increase)
    idx = np.arange(n)
    idx = idx[idx != i]
    if delta > 0:
        # need another entry to decrease by delta
        candidates = idx[h[idx] >= delta]
    else:
        # need another entry to increase by -delta
        candidates = idx[h[idx] <= 1.0 + delta]
    if candidates.size == 0:
        return h.copy()
    j = np.random.choice(candidates)
    new_h = h.copy()
    new_h[i] = new_val
    new_h[j] = h[j] - delta
    return new_h

def random_feasible_h(n: int) -> np.ndarray:

    total = arr.sum()
    if total > target:
        arr *= (target / total) * 0.999
    total = arr.sum()
    last = target - total
    if last < 0.0:
        arr = np.random.rand(n - 1)
        total = arr.sum()
        arr *= (target / total)
        last = 0.0
    elif last > 1.0:
        needed = target - 1.0
        if needed < 0:
            needed = 0.0
        arr = np.random.rand(n - 1)
        total = arr.sum()
        if total > 0:

        last = 1.0

def enforce_sum(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vec onto the simplex defined by 0≤vec≤1 and sum(vec)=target_sum."""
    vec = np.clip(vec, 0.0, 1.0)
    total = vec.sum()
    if abs(total - target_sum) < 1e-12:
        return vec
    # Scale to match sum, then iteratively fix bound violations
    if total > 0:
        vec *= (target_sum / total)
    else:
        vec[:] = target_sum / vec.size
    for _ in range(10):
        over = vec > 1.0
        under = vec < 0.0
        if not np.any(over) and not np.any(under):
            break
        if np.any(over):
            excess = np.sum(vec[over] - 1.0)
            vec[over] = 1.0
            mask = ~over
            cnt = np.count_nonzero(mask)
            if cnt > 0:

            vec[under] = 0.0
            mask = ~under

    vec = np.clip(vec, 0.0, 1.0)

        cnt = np.count_nonzero(free)
        if cnt > 0:
