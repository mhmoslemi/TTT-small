def project_to_simplex_box(v, target):
    v = np.clip(v, 0.0, 1.0)
    # repeated projection
    for _ in range(100):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-9:
            break
        mask = (v > 1e-12) & (v < 1-1e-12)  # interior
        if mask.sum() == 0:
            # all at bounds, can't adjust; break
            break
        # distribute diff equally to interior entries
        v[mask] += diff / mask.sum()
        # clip again
        v = np.clip(v, 0.0, 1.0)
    return v

def fix_sum(h, target):
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # find index where adjusting does not violate bounds
    for i in range(len(h)):
        new = h[i] + diff
        if 0 <= new <= 1:
            h[i] = new
            return h
    # fallback: distribute diff across two indices: one increase, one decrease
    # find i where h[i] < 1 and j where h[j] > 0
    inc_candidates = np.where(h < 1)[0]
    dec_candidates = np.where(h > 0)[0]
    if len(inc_candidates) > 0 and len(dec_candidates) > 0:
        i = inc_candidates[0]
        j = dec_candidates[0]
        # compute split: add up to 1-i capacity and subtract from j accordingly
        cap = 1 - h[i]
        if diff > 0:
            delta = min(diff, cap)
            h[i] += delta
            # subtract delta from j
            h[j] -= delta
        else:
            # diff <0
            cap_dec = h[j]
            delta = min(-diff, cap_dec)
            h[j] -= delta
            h[i] += delta
    # else return as is
    return h
```

    v = np.clip(v, 0.0, 1.0)
    # compute current sum
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # Iterative redistribution
    for _ in range(100):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        # identify free indices where 0 < v_i < 1
        free = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if free.size == 0:
            # cannot adjust, fallback add to first entry within bounds
            # distribute to any index not at bound

            for i in range(v.shape[0]):
                new = v[i] + diff
                if 0 <= new <= 1:
                    v[i] = new
                    break

        inc = diff / free.size
        v[free] += inc
        # clip again
        v = np.clip(v, 0.0, 1.0)
    # final clip
    return np.clip(v, 0.0, 1.0)

        if abs(diff) < 1e-12:
            break
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size == 0:
            # cannot spread evenly – try a single feasible adjust
            for i in range(v.size):
                if 0.0 <= v[i] + diff <= 1.0:
                    v[i] += diff
                    break
            break
        inc = diff / free.size
        v[free] += inc
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction (if any)
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        for i in range(v.size):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break
    return np.clip(v, 0.0, 1.0)

def upscale_vector(v, new_n):
    """Linearly interpolate vector v (defined on [0,2)) to length new_n."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, v)

def _initial_vector(n, rng, best_known=None):
    """Create a feasible starting vector of length n."""
    target = n / 2.0
    if best_known is not None:
        vec = upscale_vector(best_known, n)
    else:
        # random start then project
        vec = rng.random(n)
        vec = project_to_simplex_box(vec, target)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_score = math.inf
    best_h = None
    best_n = None

    # Determine which resolutions to try according to the time budget.
    if budget_s >= 800:
        n_list = [95, 190, 380]
    elif budget_s >= 400:
        n_list = [95, 190]
    else:
        n_list = [95]

    # Grab the parent solution if it exists.
    try:
        init_parent = initial_h_values.astype(np.float64)
        init_parent_n = init_parent.shape[0]
    except Exception:
        init_parent = None
        init_parent_n = None

    for n in n_list:
        # Stop early if time is exhausted.
        if time.time() - start_time >= budget_s - 0.5:
            break

        target_sum = n / 2.0

        # Choose an initial vector.
        if init_parent is not None and n == init_parent_n:
            h0 = init_parent.copy()
        else:
            # Prefer up‑scaling the best vector found so far.
            h0 = _initial_vector(n, rng, best_known=best_h)

        h0 = project_to_simplex_box(h0, target_sum)

        # ----- SLSQP refinement -----
        bounds = [(0.0, 1.0)] * n
        constraints = ({
            "type": "eq",
            "fun": lambda x, t=target_sum: np.sum(x) - t,
        },)

        try:
            res = minimize(
                compute_c5,
                h0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 250, "ftol": 1e-9, "disp": False},
            )
            h_opt = res.x
        except Exception:
            h_opt = h0

        h_opt = project_to_simplex_box(h_opt, target_sum)

        # ----- Local stochastic improvement -----
        best_local_h = h_opt.copy()
        best_local_score = compute_c5(best_local_h)
        max_local_iters = 50000

        for _ in range(max_local_iters):
            if time.time() - start_time >= budget_s - 0.2:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # amount we can move from j to i while staying in bounds
            max_inc = min(1.0 - best_local_h[i], best_local_h[j])
            if max_inc <= 0.0:
                continue
            delta = max_inc * rng.random()
            h_new = best_local_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            score_new = compute_c5(h_new)
            if score_new < best_local_score:
                best_local_h = h_new
                best_local_score = score_new

        # Ensure feasibility after local search.
        best_local_h = project_to_simplex_box(best_local_h, target_sum)
        best_local_score = compute_c5(best_local_h)

        # Keep the best across all resolutions.
        if best_local_score < best_score:
            best_score = best_local_score
            best_h = best_local_h
            best_n = n

    # Fallback if nothing succeeded.
    if best_h is None:
        best_n = 95
        best_h = np.full(best_n, 0.5, dtype=np.float64)
        best_score = compute_c5(best_h)

    # Final enforcement of the exact integral.
    best_h = best_h.astype(np.float64, copy=False)
    diff = (best_n / 2.0) - best_h.sum()
    if abs(diff) > 1e-12:
        for i in range(best_n):
            if 0.0 <= best_h[i] + diff <= 1.0:
                best_h[i] += diff
                break

    final_score = compute_c5(best_h)

    return best_h, float(final_score), int(best_n)
