def project_to_constraints(v, target_sum):
    n = v.shape[0]
    lo = -np.max(v) - 1e-12
    hi = 1 - np.min(v) + 1e-12
    for _ in range(40):

        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid

import time
import numpy as np

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n and the hyperplane sum=v_target."""
    lo = -np.max(v) - 1.0  # guarantee all v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0  # guarantee all v+hi >= 1
    for _ in range(45):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a periodic sample on [0,2)."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, v)


def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap for the shift that currently gives the
    maximal C5 (i.e., the shift with minimal product sum)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        k = shift
        grad[: n - k] += h[k:]
        grad[k:] += h[: n - k]
    else:
        k = -shift
        grad[k:] += h[: n - k]
        grad[: n - k] += h[k:]
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # choose a reasonable resolution (<1000 points)
    n = 360
    target_sum = n / 2.0

    # --- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_h = _upsample_vector(base_h, n)
    else:
        # uniform feasible start (sum = n/2)
        base_h = np.full(n, 0.5, dtype=float)

    h = _project_to_constraints(base_h, target_sum)
    c5 = _compute_c5(h)

    best_h = h.copy()
    best_c5 = c5

    # parameters for the search
    max_no_improve = 8000          # after this many non‑improving steps, restart
    jitter_interval = 2000         # how often to apply a small random jitter
    base_step = 0.12               # initial step size for gradient moves
    jitter_scale = 0.03            # magnitude of random jitter
    iter_cnt = 0
    no_improve = 0

    while time.time() - start_time < budget_s * 0.96:
        # find the shift that currently yields the worst (largest) C5
        raw_corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(raw_corr.argmax())
        k_star = max_idx - (n - 1)

        # gradient step aimed at increasing the overlap for that shift
        grad = _gradient_worst_shift(h, k_star)

        # linearly decay step size with elapsed time
        elapsed = time.time() - start_time
        t_frac = elapsed / (budget_s * 0.96)
        step = base_step * max(0.0, 1.0 - t_frac)

        h_candidate = _project_to_constraints(h + step * grad, target_sum)
        c5_candidate = _compute_c5(h_candidate)

        improved = False
        if c5_candidate < c5:
            h, c5 = h_candidate, c5_candidate
            improved = True
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            if rng.random() < 0.001:
                h, c5 = h_candidate, c5_candidate
                improved = True

        if improved:
            no_improve = 0
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            no_improve += 1

        # periodic small random jitter to escape flat regions
        if iter_cnt % jitter_interval == 0 and iter_cnt > 0:
            jitter = jitter_scale * (rng.random(n) - 0.5)
            h_jit = _project_to_constraints(best_h + jitter, target_sum)
            c5_jit = _compute_c5(h_jit)
            if c5_jit < c5:
                h, c5 = h_jit, c5_jit
                improved = True
                no_improve = 0
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5

        # if stuck for a while, restart around the best known point
        if no_improve >= max_no_improve:
            perturb = 0.1 * (rng.random(n) - 0.5)
            h = _project_to_constraints(best_h + perturb, target_sum)
            c5 = _compute_c5(h)
            no_improve = 0

        iter_cnt += 1

    # final refinement with a tiny step size
    for _ in range(2000):
        if time.time() - start_time >= budget_s * 0.99:
            break
        raw_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_star = int(raw_corr.argmax()) - (n - 1)
        grad = _gradient_worst_shift(best_h, k_star)
        h_refine = _project_to_constraints(best_h + 0.005 * grad, target_sum)
        c5_refine = _compute_c5(h_refine)
        if c5_refine < best_c5:
            best_h, best_c5 = h_refine, c5_refine

    return list(best_h), float(best_c5), int(n)
