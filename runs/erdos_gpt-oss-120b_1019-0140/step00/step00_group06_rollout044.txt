import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_overlap(h, dx):
    """Return max overlap integral for step values h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair_h(h, target_sum):
    """Clip to [0,1] and scale to satisfy sum constraint (approx)."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur == 0.0:
        h = np.full_like(h, target_sum / h.size)
        return h
    factor = target_sum / cur
    h = h * factor
    h = np.clip(h, 0.0, 1.0)
    # Small adjustment if rounding shifted sum away from target
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute diff uniformly to non‑boundary entries
        mask = (h > 0.0) & (h < 1.0)
        if np.any(mask):
            h[mask] += diff / mask.sum()
        else:
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate_h(old_h, new_n):
    """Linear interpolation of old_h (on [0,2)) to new_n points."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_score = np.inf
    best_n = None

    # Base resolution (provided by the parent if present)
    base_n = 95
    candidate_sizes = [base_n]

    # Allow higher resolutions if time budget is generous
    if budget_s > 400:
        candidate_sizes.append(base_n * 2)   # 190 points
    if budget_s > 800:
        candidate_sizes.append(base_n * 4)   # 380 points

    # Helper to fetch an initial vector for a given size
    def _initial_vector(n):
        if "initial_h_values" in globals() and n == initial_h_values.size:
            return initial_h_values.copy()
        if "initial_h_values" in globals():
            vec = _interpolate_h(initial_h_values, n)
        else:
            vec = np.random.rand(n)
        target = n / 2.0
        return _repair_h(vec, target)

    # Main optimisation loop over resolutions
    for n in candidate_sizes:
        dx = 2.0 / n
        target_sum = n / 2.0
        init_vec = _initial_vector(n)

        # Perform a few multi‑start attempts at this resolution
        for attempt in range(3):
            # Abort if budget nearly exhausted
            if time.time() - start_time > budget_s * 0.95:
                break

            # Perturb initial guess on later attempts
            if attempt == 0:
                start_vec = init_vec.copy()
            else:
                noise = 0.05 * (np.random.rand(n) - 0.5)
                start_vec = _repair_h(init_vec + noise, target_sum)

            bounds = [(0.0, 1.0)] * n
            constraints = ({
                "type": "eq",
                "fun": lambda h, targ=target_sum: np.sum(h) - targ,
            },)

            # SLSQP optimisation (finite‑difference gradients)
            res = minimize(
                fun=lambda h: _evaluate_overlap(h, dx),
                x0=start_vec,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )

            # Repair and evaluate the outcome
            h_opt = _repair_h(res.x, target_sum)
            score = _evaluate_overlap(h_opt, dx)

            if score < best_score:
                best_score = score
                best_h = h_opt.copy()
                best_n = n

        # Break early if time limit reached
        if time.time() - start_time > budget_s * 0.95:

    # Fallback in the unlikely case no solution was found
    if best_h is None:
        best_n = base_n
        best_h = np.full(best_n, 0.5)  # satisfies sum constraint exactly
        best_score = _evaluate_overlap(best_h, 2.0 / best_n)

    return best_h.astype(np.float64), float(best_score), int(best_n)
