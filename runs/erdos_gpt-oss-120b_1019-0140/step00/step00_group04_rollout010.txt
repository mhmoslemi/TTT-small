def _compute_c5(h: np.ndarray) -> float:
    """Return max correlation integral for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    """Project vector v onto [0,1]^n with sum(v)=target (L2 projection)."""
    lo = -np.max(v)            # makes v+lam <= 0 → all zeros
    hi = 1.0 - np.min(v)       # makes v+lam >= 1 → all ones
    for _ in range(max_iter):

        h = np.clip(v + lam, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = lam
        else:
            hi = lam

                continue
            delta = max_delta * rng.random()

        # propose new state
        h_new = cur_h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        # evaluate
        new_c5 = _compute_c5(h_new)
        delta_obj = new_c5 - cur_c5

        # Metropolis acceptance
        if delta_obj <= 0.0 or rng.random() < np.exp(-delta_obj / max(T, 1e-12)):
            cur_h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

        it += 1

    # final projection (tiny numerical drift possible)
    best_h = _project_to_sum_box(best_h, target)
    best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), n
