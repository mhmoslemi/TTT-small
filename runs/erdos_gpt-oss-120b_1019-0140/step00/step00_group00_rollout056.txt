def ensure_feasible(h):
    target = len(h) / 2.0
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - target) < 1e-9:
        return h
    # Use cvxpy to find nearest feasible h
    n = len(h)
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - h))
    constraints = [cp.sum(x) == target,
                   x >= 0,
                   x <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # fallback: simple scaling
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)
        # adjust any residual left: while sum not exact, adjust first feasible index.
        diff = target - np.sum(h)
        # add diff to an index not hitting bound
        for i in range(n):
            if 0 <= h[i]+diff <= 1:
                h[i] += diff
                break
    else:
        h = x.value
    return h
