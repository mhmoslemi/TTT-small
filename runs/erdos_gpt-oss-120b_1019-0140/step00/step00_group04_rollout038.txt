import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_feasible(h: np.ndarray, max_iter: int = 10) -> np.ndarray:
    """
    Project h onto the convex set:
        0 ≤ h_i ≤ 1  and  mean(h) = 0.5   (i.e. sum(h) = n/2)
    Uses alternating projections between the box and the hyperplane.
    """
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        mean = np.mean(h)
        if abs(mean - 0.5) < 1e-12:
            break
        h += 0.5 - mean
    return np.clip(h, 0.0, 1.0)

    best_h = None

    best_n = None

    # resolutions to explore (coarse → fine)
    n_list = [64, 128, 256]

    prev_h = None
    for n in n_list:
        # stop if we are out of time
        if time.time() - start_time > budget_s * 0.99:
            break

        dx = 2.0 / n

        # ---------- initial construction ----------
        if prev_h is not None:
            # upscale previous result by linear interpolation
            x_old = np.linspace(0, 2, prev_h.shape[0], endpoint=False) + (2.0 / prev_h.shape[0]) / 2
            x_new = np.linspace(0, 2, n, endpoint=False) + dx / 2
            h = np.interp(x_new, x_old, prev_h)
        else:

            if init is not None and len(init) == n:

            elif init is not None:
                init_arr = np.asarray(init, dtype=float)
                n0 = init_arr.shape[0]

        max_iters = {64: 50000, 128: 30000, 256: 20000}[n]
