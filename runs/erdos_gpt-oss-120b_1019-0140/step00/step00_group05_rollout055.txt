import numpy as np
import time, math

def project_to_capped_simplex(v, desired_sum, lower=0.0, upper=1.0):
    w = np.clip(v, lower, upper)
    sum_w = w.sum()
    if abs(sum_w - desired_sum) < 1e-12:
        return w
    if sum_w > desired_sum:
        excess = sum_w - desired_sum
        # reduce
        adjustable = np.where(w > lower)[0]
        while excess > 1e-12 and adjustable.size > 0:
            amounts = w[adjustable] - lower
            total = amounts.sum()
            if excess < total:
                w[adjustable] -= excess * (amounts / total)
                excess = 0.0
            else:
                w[adjustable] = lower
                excess -= total
                adjustable = np.where(w > lower)[0]
    else:
        deficit = desired_sum - sum_w
        adjustable = np.where(w < upper)[0]
        while deficit > 1e-12 and adjustable.size > 0:
            amounts = upper - w[adjustable]
            total = amounts.sum()
            if deficit < total:
                w[adjustable] += deficit * (amounts / total)
                deficit = 0.0
            else:
                w[adjustable] = upper
                deficit -= total
                adjustable = np.where(w < upper)[0]
    # final correction for floating error
    correction = desired_sum - w.sum()
    if abs(correction) > 1e-12:
        # distribute correction uniformly to non-boundary entries
        adjustable = np.where((w > lower) & (w < upper))[0]
        if adjustable.size > 0:
            w[adjustable] += correction / adjustable.size
            # clip again
            w = np.clip(w, lower, upper)
    return w

def compute_c5(h, dx):
    # correlation of h and (1-h)
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(cor.max())

    # project
    desired_sum = n / 2.0
    new = project_to_capped_simplex(new, desired_sum)
    return new

import numpy as np, time, math

def project_to_capped_simplex(v, desired_sum, lower=0.0, upper=1.0):
    w = np.clip(v, lower, upper)
    sum_w = w.sum()
    if abs(sum_w - desired_sum) < 1e-12:
        return w
    if sum_w > desired_sum:
        excess = sum_w - desired_sum
        adjustable = np.where(w > lower)[0]
        while excess > 1e-12 and adjustable.size > 0:
            amounts = w[adjustable] - lower
            total = amounts.sum()
            if excess < total:
                w[adjustable] -= excess * (amounts / total)
                excess = 0.0
            else:
                w[adjustable] = lower
                excess -= total
                adjustable = np.where(w > lower)[0]
    else:
        deficit = desired_sum - sum_w
        adjustable = np.where(w < upper)[0]
        while deficit > 1e-12 and adjustable.size > 0:
            amounts = upper - w[adjustable]
            total = amounts.sum()
            if deficit < total:
                w[adjustable] += deficit * (amounts / total)
                deficit = 0.0
            else:
                w[adjustable] = upper
                deficit -= total
                adjustable = np.where(w < upper)[0]
    # final tiny correction
    correction = desired_sum - w.sum()
    if abs(correction) > 1e-12:
        interior = np.where((w > lower) & (w < upper))[0]
        if interior.size > 0:
            idx = interior[0]
            w[idx] = np.clip(w[idx] + correction, lower, upper)
        else:
            # spread uniformly (may hit bounds)
            w += correction / w.size
            w = np.clip(w, lower, upper)
    return w

def compute_c5(h, dx):

    cor = np.correlate(h, 1.0 - h, mode='full') * dx

        j = i + k
        max_transfer = min(step, new[i], 1.0 - new[j])
        if max_transfer <= 0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        new[i] -= delta
        new[j] += delta
    return project_to_capped_simplex(new, n / 2.0)

def mutate(h, rng, step):
    if rng.random() < 0.5:
        return targeted_mutate(h, rng, step)
    else:
        return random_mutate(h, rng, step)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        h0 = project_to_capped_simplex(h0, n / 2.0)
    else:
        n = 70
        h0 = project_to_capped_simplex(rng.random(n), n / 2.0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    current_h = best_h.copy()
    current_c5 = best_c5
    start = time.time()
    limit = start + budget_s * 0.94
    while time.time() < limit:
        elapsed = time.time() - start
        frac = 1.0 - elapsed / budget_s
        if frac < 0.0:
            frac = 0.0
        temp = 0.02 * frac + 1e-6
        step = 0.4 * frac + 0.001
        neighbor_h = mutate(current_h, rng, step)
        neighbor_c5 = compute_c5(neighbor_h, dx)
        delta = neighbor_c5 - current_c5
        if delta < 0.0:
            accept = True
        else:
            prob = math.exp(-delta / temp) if temp > 0 else 0.0
            accept = rng.random() < prob
        if accept:
            current_h, current_c5 = neighbor_h, neighbor_c5
            if neighbor_c5 < best_c5:
                best_h, best_c5 = neighbor_h, neighbor_c5
    try:
        val = evaluate_erdos_solution(best_h)
        final_c5 = float(val[0]) if isinstance(val, (list, tuple, np.ndarray)) else float(val)
    except Exception:
        final_c5 = compute_c5(best_h, dx)
    best_h = project_to_capped_simplex(best_h, n / 2.0)
    return (best_h.tolist(), final_c5, n)
