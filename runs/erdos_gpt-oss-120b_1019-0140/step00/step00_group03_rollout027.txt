def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1.0
        rng.shuffle(h)

    n = len(h)
    # ensure weight
    target_sum = n//2
    cur_sum = int(round(np.sum(h)))
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.where(h < 0.5)[0]
            if len(zeros) >= diff:
                idx = rng.choice(zeros, diff, replace=False)
                h[idx] = 1.0
        elif diff < 0:
            ones = np.where(h > 0.5)[0]
            idx = rng.choice(ones, -diff, replace=False)
            h[idx] = 0.0

    best_h = h.copy()
    best_score = compute_c5(best_h)

    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    safety = 0.1
    T = 0.5
    decay = 0.9995
    stagnation = 0
    best_iter = 0

    while time.time() - start < budget_s - safety:
        # compute correlation
        cor = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(cor))
        max_corr = cor[max_idx]
        # compute lag
        lag = max_idx - (n - 1)

        # indices for this lag
        if lag >= 0:
            i_start = 0
            i_end = n - lag
        else:
            i_start = -lag
            i_end = n
        i_range = np.arange(i_start, i_end)
        j_range = i_range + lag

        hi = cur_h[i_range]
        hj = cur_h[j_range]
        bad_mask = (hi == 1) & (hj == 0)
        good_mask = (hi == 0) & (hj == 1)
        bad_indices = i_range[bad_mask]
        good_indices = i_range[good_mask]

        improved = False
        if bad_indices.size > 0 and good_indices.size > 0:
            # Sample few candidate swaps
            sample_swaps = min(5, len(bad_indices) * len(good_indices))
            # generate random candidate swaps
            for _ in range(sample_swaps):
                i = rng.choice(bad_indices)
                j = rng.choice(good_indices)
                # ensure i != j (they are distinct because they belong to different groups)
                new_h = cur_h.copy()
                new_h[i], new_h[j] = new_h[j], new_h[i]
                new_score = compute_c5(new_h)
                if new_score < cur_score:
                    cur_h, cur_score = new_h, new_score
                    improved = True
                    # update best if needed
                    if new_score < best_score:
                        best_score = new_score
                        best_h = new_h.copy()
                        best_iter = 0
                    break
                else:
                    # accept with probability
                    delta = new_score - cur_score
                    if T > 0 and rng.random() < math.exp(-delta / T):
                        cur_h, cur_score = new_h, new_score
                        # may not be best but accept
                        improved = True
                        break
        if not improved:
            # random swap
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_score = compute_c5(new_h)
            delta = new_score - cur_score
            if delta <= 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
                cur_h, cur_score = new_h, new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = new_h.copy()
                    best_iter = 0
        # update temperature
        T *= decay

        if best_iter > 20000:
            # restart from best
            cur_h = best_h.copy()
            cur_score = best_score
            best_iter = 0

    return (best_h, best_score, n)

import time
import math
import numpy as np

def compute_c5(h):
    """Return the overlap bound C5 for a binary (or [0,1]) vector h."""
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h that minimizes the Erdős overlap constant C5.

    Returns:
        h_values (np.ndarray): the best step vector found.
        c5_bound (float): the corresponding C5 value.
        n_points (int): length of the discretization.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h from a parent construction if present; otherwise random.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    n = len(h)
    target_sum = n // 2
    cur_sum = int(round(np.sum(h)))
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.where(h < 0.5)[0]
            if len(zeros) >= diff:
                idx = rng.choice(zeros, diff, replace=False)
                h[idx] = 1.0
        elif diff < 0:
            ones = np.where(h > 0.5)[0]
            idx = rng.choice(ones, -diff, replace=False)
            h[idx] = 0.0

    # ------------------------------------------------------------------
    # Local search with a simple simulated-annealing acceptance rule.
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = compute_c5(best_h)

    cur_h = h.copy()
    cur_score = best_score

    start_time = time.time()
    safety_margin = 0.1  # seconds left as buffer
    temperature = 0.5
    decay = 0.9995
    # counter for periods without improving the global best
    stagnation_counter = 0

    while time.time() - start_time < budget_s - safety_margin:
        # correlation of current state
        cor = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(cor))
        lag = max_idx - (n - 1)  # lag in [-n+1, n-1]

        # indices that participate in this lag
        if lag >= 0:
            i_start, i_end = 0, n - lag
        else:
            i_start, i_end = -lag, n
        i_range = np.arange(i_start, i_end)
        j_range = i_range + lag

        hi = cur_h[i_range]
        hj = cur_h[j_range]

        bad_mask = (hi == 1) & (hj == 0)   # 1 with 0 on the shifted side
        good_mask = (hi == 0) & (hj == 1)  # 0 with 1 on the shifted side
        bad_idxs = i_range[bad_mask]
        good_idxs = i_range[good_mask]

        improved = False

        # --------------------------------------------------------------
        # Targeted move: try a few swaps that directly reduce the worst lag.
        # --------------------------------------------------------------
        if bad_idxs.size and good_idxs.size:
            # try a handful of random candidate swaps
            trials = min(5, bad_idxs.size * good_idxs.size)
            for _ in range(trials):
                i = rng.choice(bad_idxs)
                j = rng.choice(good_idxs)
                candidate = cur_h.copy()
                candidate[i], candidate[j] = candidate[j], candidate[i]
                cand_score = compute_c5(candidate)
                if cand_score < cur_score:
                    cur_h, cur_score = candidate, cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_h = candidate.copy()
                        stagnation_counter = 0
                    break
                # occasional uphill step
                delta = cand_score - cur_score
                if temperature > 0 and rng.random() < math.exp(-delta / temperature):
                    cur_h, cur_score = candidate, cand_score
                    improved = True
                    break

        # ------------------------------------------------------------------
        # If no targeted improvement, perform a generic random swap.
        # ------------------------------------------------------------------
        if not improved:

            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            candidate = cur_h.copy()
            candidate[i], candidate[j] = candidate[j], candidate[i]
            cand_score = compute_c5(candidate)
            delta = cand_score - cur_score
            if delta <= 0 or (temperature > 0 and rng.random() < math.exp(-delta / temperature)):

                    best_h = candidate.copy()

            cur_score = best_score
            stagnation_counter = 0

    return best_h, best_score, n
