import time
import numpy as np

def compute_c5(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def improve_h(initial_h, dx, time_limit, target_c5, rng):
    """Greedy improvement: move mass from positions that increase the worst shift."""
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    eps = 1e-12
    start = time.time()
    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_idx = int(np.argmax(corr))
        shift = worst_idx - (n - 1)  # integer lag
        if shift == 0:
            break

        moved = False
        src = dst = None
        delta = 0.0

        if shift > 0:
            # try to move from left side to right side
            mask = (h[: n - shift] > eps) & (h[shift:] < 1.0 - eps)
            if np.any(mask):
                i = int(rng.choice(np.where(mask)[0]))
                src = i
                dst = i + shift
                max_delta = min(h[src], 1.0 - h[dst])
                delta = max_delta * rng.random()
                if delta > eps:
                    h[src] -= delta
                    h[dst] += delta
                    moved = True
        else:  # shift < 0
            lpos = -shift
            mask = (h[lpos:] > eps) & (h[: n - lpos] < 1.0 - eps)
            if np.any(mask):
                i = int(rng.choice(np.where(mask)[0]))
                src = i + lpos
                dst = i
                max_delta = min(h[src], 1.0 - h[dst])
                delta = max_delta * rng.random()
                if delta > eps:
                    h[src] -= delta
                    h[dst] += delta
                    moved = True

        # fallback: random transfer preserving bounds
        if not moved:
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > eps:
                delta = max_delta * rng.random()
                src, dst = i, j
                h[src] -= delta
                h[dst] += delta
                moved = True

        # evaluate new C5
        cur_c5 = compute_c5(h, dx)
        if cur_c5 + 1e-12 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 <= target_c5:
                break
        else:
            # revert move
            if moved and src is not None and dst is not None:
                h[src] += delta
                h[dst] -= delta
