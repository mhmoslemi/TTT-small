def project_to_bounds_sum(v, target, tol=1e-9, max_iter=100):
    v = np.asarray(v, dtype=float)
    n = v.size
    # quick check:
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0

            # need increase lambda to decrease sum

            high = mid

def propose_swap(h, rng, max_delta=0.2):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # compute feasible delta interval
    hi = h[i]
    hj = h[j]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    # intersect with [-max_delta, max_delta]
    lo = max(lo, -max_delta)
    hi_delta = min(hi_delta, max_delta)
    if lo >= hi_delta:
        return None
    delta = rng.uniform(lo, hi_delta)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

import time
import math
import numpy as np

def _project_to_bounds_sum(v, target, tol=1e-9, max_iter=100):
    """Project vector v onto the feasible set {0<=x<=1, sum(x)=target}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    # Quick checks for trivial cases
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)
    # Binary search for lambda such that sum(clamp(v - lam,0,1)) = target
    lo = -np.max(v) - 1.0  # guarantees all entries become 1 -> sum=n
    hi = np.max(v) + 1.0   # guarantees all become 0 -> sum=0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _propose_swap(h, rng, max_delta):
    """Generate a neighbour by moving delta from index j to i, preserving sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # feasible delta interval respecting bounds [0,1]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    # intersect with symmetric max_delta
    lo = max(lo, -max_delta)
    hi_delta = min(hi_delta, max_delta)
    if lo >= hi_delta:
        return None
    delta = rng.uniform(lo, hi_delta)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new


def _c5_score(h):
    """Compute the C5 value for a discretised step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function on [0,2] with low overlap integral C5.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values : list of floats, length n_points, each in [0,1]
        c5_bound : float, the achieved C5 value
        n_points : int, number of discretisation points
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise starting vector
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
        n = h0.size
    except NameError:
        n = 70
        h0 = np.zeros(n, dtype=float)
        ones = n // 2
        idx = rng.choice(n, size=ones, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    # Ensure feasibility
    h0 = _project_to_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = _c5_score(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Simulated‑annealing style search
    init_temp = 0.04
    min_temp = 1e-5
    max_delta_start = 0.5
    min_delta = 1e-4
    no_improve = 0
    restart_interval = 20000

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.15:
            break

        # temperature and delta schedule (linear decay)
        frac = elapsed / budget_s
        temp = max(min_temp, init_temp * (1.0 - frac))
        max_delta = max(min_delta, max_delta_start * (1.0 - frac))

        # neighbour generation
        if rng.rand() < 0.12:  # occasional global perturbation
            noise = rng.normal(scale=0.07 * (1.0 - frac), size=n)
            cand = cur_h + noise
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_bounds_sum(cand, target_sum)
        else:
            cand = _propose_swap(cur_h, rng, max_delta)
            if cand is None:
                continue

        cand_score = _c5_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.rand() < math.exp((cur_score - cand_score) / (temp + 1e-12)):
            cur_h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart from best with a fresh perturbation
        if no_improve >= restart_interval:
            noise = rng.normal(scale=0.15, size=n)
            cand = best_h + noise
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_bounds_sum(cand, target_sum)
            cand_score = _c5_score(cand)
            cur_h = cand
            cur_score = cand_score
            no_improve = 0

    # Final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_bounds_sum(best_h, target_sum)

    final_score = _c5_score(best_h)

    return best_h.tolist(), float(final_score), int(n)
