def upsample(initial, target_n):
    # repeat or interpolate to length target_n
    n0 = len(initial)
    if target_n % n0 == 0:
        factor = target_n // n0
        h = np.repeat(initial, factor)
    else:
        # linear interpolation to shape target_n
        # but we need binary values; we can use nearest neighbor: use np.interp on integer grid.
        xs = np.linspace(0, 2, n0, endpoint=False)  # not guaranteed
        t = np.linspace(0, 2, target_n, endpoint=False)
        # Use nearest neighbor interpolation rounding to 0/1
        h_interp = np.interp(t, xs, initial, left=initial[0], right=initial[-1])
        # round to 0 or 1 based on threshold 0.5
        h = (h_interp >= 0.5).astype(float)
        # adjust sum if not exactly half
        diff = h.sum() - target_n/2
        if diff > 0:
            # set diff ones to zero
            idx_one = np.where(h==1)[0]
            np.random.shuffle(idx_one)
            h[idx_one[:int(diff)]] = 0
        elif diff < 0:
            idx_zero = np.where(h==0)[0]
            np.random.shuffle(idx_zero)
            h[idx_zero[:int(-diff)]] = 1
    return h

def hill_climb(h, max_iters, temp0, cooling):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = len(h)
    indices_one = np.where(cur_h == 1)[0]
    indices_zero = np.where(cur_h == 0)[0]
    for it in range(max_iters):
        # propose swap
        i = np.random.choice(indices_one)
        j = np.random.choice(indices_zero)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # acceptance probability
            temp = max(temp0 * (cooling ** it), 1e-8)
            # delta positive, acceptance probability = exp(-delta / temp)
            if np.random.rand() < np.exp(-(new_score - cur_score)/temp):

            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            # indices unchanged
        # perhaps break if best_score < target threshold
    return best_h, best_score

def local_search(initial_h, max_iters, time_limit):
    cur_h = initial_h.copy()
    n = len(cur_h)
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    while it < max_iters:
        if time.time() - start > time_limit:
            break
        # propose swap
        i, j = np.random.randint(n, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        # accept if improvement or with probability
        # temperature schedule: start temp = 0.01 maybe.
        # gradually decrease.
        # We'll set temp = 0.1 * (1 - it / max_iters)
        temp = 0.1 * (1 - it / max_iters)
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (temp+1e-12)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        it += 1
    return best_h, best_score, it

def upsample_h(init_h, target_n):
    init_h = np.asarray(init_h, dtype=float)
    init_n = init_h.size
    if target_n == init_n:
        h = init_h.copy()
    elif target_n > init_n:
        # repeat each element ceil(target_n/init_n) times, then slice
        repeat = target_n // init_n
        remainder = target_n % init_n
        if repeat > 0:
            h = np.repeat(init_h, repeat)
        else:
            h = np.empty(0, dtype=float)
        if remainder > 0:
            h = np.concatenate([h, init_h[:remainder]])
    else:
        # target smaller: random sample of positions to maintain count half
        # We'll generate random binary with half ones.
        half = target_n // 2
        h = np.zeros(target_n, dtype=float)
        h[:half] = 1.0
        np.random.shuffle(h)
    # ensure integer count
    half = target_n // 2
    cur = int(np.round(h.sum()))
    if cur != half:
        diff = cur - half
        if diff > 0:
            # turn diff ones to zero
            idx = np.where(h > 0.5)[0]
            if len(idx) >= diff:
                np.random.shuffle(idx)
                h[idx[:diff]] = 0.0
        else:
            idx = np.where(h < 0.5)[0]
            if len(idx) >= -diff:
                np.random.shuffle(idx)
                h[idx[: -diff]] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Use np.correlate with float result
    # It returns array of shape (2*n -1) of dtype same as input.
    # Ensure h & (1 - h) are float.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search(h_init, max_iters, time_budget):
    import time, math, random, numpy as np
    cur_h = h_init.copy()
    n = cur_h.shape[0]
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    it = 0
    # temperature schedule: start with something like 0.05 maybe.
    for it in range(max_iters):
        # check time
        if time.time() - start > time_budget:
            break
        # propose swap
        i, j = np.random.randint(n, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        # acceptance probability: if better accept else if random
        if new_score < cur_score:
            accept = True
        else:
            # temperature gradually decreasing
            temp = 0.05 * (1 - it / max_iters) + 1e-9
            accept = np.random.rand() < math.exp(-(new_score - cur_score) / temp)
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
                # early exit if target reached
                if best_score < 0.3808:
                    break
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score, it

import numpy as np
import time
import random
import math

def compute_c5(h):
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample_h(init_h, target_n):
    """Resize a binary vector to target_n entries while preserving the total mass."""
    init_h = np.asarray(init_h, dtype=float)
    init_n = init_h.size
    if target_n == init_n:
        h = init_h.copy()
    elif target_n > init_n:
        # repeat each entry as many times as needed, then cut to exact length
        repeat = target_n // init_n
        remainder = target_n % init_n
        if repeat > 0:
            h = np.repeat(init_h, repeat)
        else:
            h = np.empty(0, dtype=float)
        if remainder:
            h = np.concatenate([h, init_h[:remainder]])
    else:
        # target smaller: generate fresh binary vector of correct size
        half = target_n // 2
        h = np.zeros(target_n, dtype=float)
        h[:half] = 1.0
        np.random.shuffle(h)
    # enforce exactly half ones (integral constraint)
    half = target_n // 2
    cur = int(round(h.sum()))
    if cur != half:
        diff = cur - half
        if diff > 0:
            idx = np.where(h > 0.5)[0]
            if len(idx) >= diff:
                np.random.shuffle(idx)
                h[idx[:diff]] = 0.0
        else:
            idx = np.where(h < 0.5)[0]
            if len(idx) >= -diff:
                np.random.shuffle(idx)
                h[idx[: -diff]] = 1.0
    return h

def local_search(h_init, max_iters, time_budget):
    """Hill‑climbing / simulated‑annealing on binary vectors with fixed mass."""
    cur_h = h_init.copy()
    n = cur_h.shape[0]
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    start_temp = 0.2
    end_temp = 1e-4

    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        i, j = np.random.randint(n, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)

        if new_score < cur_score:
            accept = True
        else:
            t = start_temp * (1 - it / max_iters) + end_temp * (it / max_iters)
            accept = np.random.rand() < math.exp(-(new_score - cur_score) / t)

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
                if best_score < 0.3808:
                    break
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_score, it
