def _evaluate_score(vec):
    """Return C5 score for a given h vector."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' yields length 2*n-1
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Obtain initial feasible vector
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        # default: 47 points uniformly 0.5 (satisfies sum constraint)
        n_default = 47
        h = np.full(n_default, 0.5, dtype=float)

    n = h.shape[0]
    target_sum = n / 2.0

    # Ensure exact sum constraint (minor adjustments)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # Distribute diff greedily while respecting bounds
        for idx in range(n):
            if diff == 0:
                break
            if diff > 0:
                room = 1.0 - h[idx]
                change = min(diff, room)
            else:
                room = h[idx]
                change = max(diff, -room)
            h[idx] += change
            diff -= change

    # Initial state
    cur_h = h.copy()
    cur_score = _evaluate_score(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    goal = 0.3808
    if best_score <= goal:

        now = time.time()
