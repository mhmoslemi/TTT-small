def upsample_h(h, target_n):
    n = h.shape[0]
    if target_n == n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # Clip to [0,1] and project sum exactly
    h_new = np.clip(h_new, 0.0, 1.0)
    # Adjust sum to target_n/2
    total = h_new.sum()
    target = target_n / 2.0
    diff = target - total
    # distribute diff uniformly across entries not at bounds
    # use projection: add diff / n to all entries, then clip,
    # then repeat until diff small.
    # Instead use simple iterative adjust:
    if abs(diff) > 1e-8:
        # distribute to entries with slack
        slack_up = (1.0 - h_new)
        slack_down = h_new
        # Weighted distribution proportional to slack (if diff>0, add)
        if diff > 0:
            # add proportionally to slack_up
            slack = slack_up
        else:
            slack = slack_down
        total_slack = slack.sum()
        if total_slack > 0:
            delta = diff * (slack / total_slack)
            h_new += delta
        h_new = np.clip(h_new, 0.0, 1.0)

            if diff > 0:
                slack = (1.0 - h_new)
            else:
                slack = h_new
            total_slack = slack.sum()
            if total_slack == 0:
                break
            delta = diff * (slack / total_slack)

def compute_c5(h, dx):
    """Maximum overlap integral for discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h):
    """Project array h onto the box [0,1] and the hyperplane sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Iteratively redistribute the deficit/excess to non‑saturated entries.
    for _ in range(12):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - h
        else:
            slack = h
        total_slack = slack.sum()
        if total_slack == 0.0:
            break
        delta = diff * (slack / total_slack)
        h += delta
        h = np.clip(h, 0.0, 1.0)
    return h
