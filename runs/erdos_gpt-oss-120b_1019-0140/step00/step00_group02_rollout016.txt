def neighbor_transfer(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # compute allowable range for epsilon to keep both in [0,1]
    if i == j:
        return h.copy()
    # Determine bounds
    hi = h[i]; hj = h[j]
    # we want to add epsilon to j and subtract from i: new_i=hi - eps; new_j=hj + eps
    # So eps must satisfy 0 <= hi - eps <=1 => eps ∈ [hi-1, hi]
    # and 0 <= hj + eps <=1 => eps ∈ [-hj, 1-hj]
    eps_low = max(hi - 1.0, -hj)
    eps_high = min(hi, 1.0 - hj)
    # also limit to max_step magnitude
    eps_low = max(eps_low, -max_step)
    eps_high = min(eps_high, max_step)
    if eps_low >= eps_high:
        return h.copy()
    eps = np.random.uniform(eps_low, eps_high)
    new_h = h.copy()
    new_h[i] = hi - eps
    new_h[j] = hj + eps
    return new_h
```

def repair_sum_and_bounds(h):
    # Ensure bounds and exact sum
    n = h.shape[0]
    target = n / 2.0

    np.clip(h, 0.0, 1.0, out=h)

    if abs(diff) < 1e-12:

    # adjust proportionally among non-clipped indices
    free = (h > 0) & (h < 1)

import numpy as np
import time
import random
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _repair(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the exact integral constraint."""
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute the remaining mass among non‑boundary entries
    free = (h > 0.0) & (h < 1.0)
    cnt = free.sum()
    if cnt == 0:
        # fallback to uniform 0.5
        h[:] = 0.5
        return h
    h[free] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    # final tiny scaling if needed
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        h *= target / cur
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _neighbor_transfer(h: np.ndarray, max_step: float) -> np.ndarray:
    """Generate a feasible neighbor by moving a small amount from i to j."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # epsilon such that new_i = hi - eps, new_j = hj + eps stay in [0,1]
    eps_low = max(-max_step, hi - 1.0, -hj)
    eps_high = min(max_step, hi, 1.0 - hj)
    if eps_low >= eps_high:
        return h  # nothing feasible
    eps = np.random.uniform(eps_low, eps_high)
    nh = h.copy()
    nh[i] = hi - eps
    nh[j] = hj + eps
    return nh


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a low C5 bound.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------ #
    # Acquire a starting feasible vector
    # ------------------------------------------------------------------ #
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
        h0 = _repair(init.copy())
    except Exception:
        # no supplied construction – start from a random feasible vector
        n = 120                     # modest size, still fast to evaluate
        h0 = np.random.rand(n)
        target = n / 2.0
        h0 = h0 / h0.sum() * target   # scale to correct mass
        h0 = _repair(h0)

    current = h0.copy()
    cur_val = _compute_c5(current)
    best = current.copy()
    best_val = cur_val

    start = time.time()

    # Simulated‑annealing parameters
    temp = 0.12                     # initial temperature
    cooling = 0.9996                # exponential decay factor
    min_temp = 1e-6
    max_step = 0.25                 # maximal transfer amount at start

    # ------------------------------------------------------------------ #
    # Main optimisation loop – stop a little before the budget expires
    # ------------------------------------------------------------------ #
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break

        # step size scales with temperature
        step = max_step * (temp / 0.12)

        cand = _neighbor_transfer(current, step)
        cand_val = _compute_c5(cand)

        # Metropolis acceptance
        if cand_val < cur_val or random.random() < math.exp((cur_val - cand_val) / max(temp, 1e-12)):
            current = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val

        # temperature schedule
        temp = max(temp * cooling, min_temp)

    # Final verification (ensures exact feasibility)
    best = _repair(best)
    best_val = _compute_c5(best)

    return best.tolist(), float(best_val), int(n)
