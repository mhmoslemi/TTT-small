def project_to_bounds_and_sum(v, target):
    # binary search for lambda
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) / 2.0, 0, 1)
    # adjust leftover diff if any and possible indices not at bounds
    s = w.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size > 0:
            w[free] += diff / free.size
            w = np.clip(w, 0, 1)
        # if still off due to rounding, distribute over all
        s2 = w.sum()
        if abs(s2 - target) > 1e-9:
            # final scaling (rare)
            w = w * (target / s2)
            w = np.clip(w, 0, 1)

import numpy as np
import time

def compute_c5(h):
    """Return the max overlap integral for a step function h."""
    N = h.shape[0]
    dx = 2.0 / N
    conv = np.correlate(h, 1 - h, mode="full")
    return conv.max() * dx

def project_to_bounds_and_sum(v, target):
    """
    Euclidean projection of v onto the set {w | 0 <= w_i <= 1, sum(w) = target}.
    Implements the classic water‑filling (binary‑search) method.
    """
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) / 2.0, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size:
            w[free] += diff / free.size
            w = np.clip(w, 0.0, 1.0)
        if abs(w.sum() - target) > 1e-9:
            w = w * (target / w.sum())
            w = np.clip(w, 0.0, 1.0)
    return w

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h with a low Erdős C5 overlap bound."""
    np.random.seed(seed)
    start = time.time()

    # -----------------------------------------------------------------
    # initial construction
    # -----------------------------------------------------------------
    N = None
    try:
        init = np.array(initial_h_values, dtype=float)
        N = init.shape[0]
        h = init.copy()
        target_sum = N / 2.0
        h = project_to_bounds_and_sum(h, target_sum)
    except Exception:
        pass

    if N is None:
        N = 200                     # a modest size that respects the <1k rule
        target_sum = N / 2.0
        h = np.zeros(N, dtype=float)
        h[np.random.choice(N, N // 2, replace=False)] = 1.0  # binary start
        h = project_to_bounds_and_sum(h, target_sum)

    dx = 2.0 / N
    target_sum = N / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    step_initial = 0.5
    it_without_improve = 0

    # -----------------------------------------------------------------
    # main optimisation loop (projected sub‑gradient descent + random restarts)
    # -----------------------------------------------------------------
    while time.time() - start < budget_s - 0.2:
        # correlation and identification of the worst shift(s)
        conv = np.correlate(h, 1 - h, mode="full")
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient of the max‑shift term
        g = np.zeros_like(h)
        for idx in max_idxs:
            s = idx - (N - 1)          # shift (negative … positive)
            if s >= 0:
                i = slice(0, N - s)
                j = slice(s, N)
                hi = h[i]
                hj = h[j]
                g[i] += (1.0 - hj) * dx
                g[j] += -hi * dx
            else:
                ss = -s
                i = slice(ss, N)
                j = slice(0, N - ss)
                hi = h[i]
                hj = h[j]
                g[i] += (1.0 - hj) * dx
                g[j] += -hi * dx

        g /= len(max_idxs)

        # diminishing step size
        elapsed = time.time() - start
        t_frac = elapsed / budget_s
        step = step_initial * (1.0 - t_frac)

        # gradient step + projection
        new_h = h - step * g
        new_h = project_to_bounds_and_sum(new_h, target_sum)

        c5 = compute_c5(new_h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
            it_without_improve = 0
        else:
            it_without_improve += 1

        h = new_h

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.005 or it_without_improve > 5000:
            noise = np.random.normal(scale=0.02, size=N)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h = project_to_bounds_and_sum(h, target_sum)
            it_without_improve = 0

        # rare full restart (binary random start)
        if np.random.rand() < 0.001:
            h = np.zeros(N, dtype=float)
            h[np.random.choice(N, N // 2, replace=False)] = 1.0
            h = project_to_bounds_and_sum(h, target_sum)

    # -----------------------------------------------------------------
    # final result
    # -----------------------------------------------------------------
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N
