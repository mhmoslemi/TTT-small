def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random, cvxpy as cp

    np.random.seed(seed)
    random.seed(seed)

    # Determine starting h and n
    # ... as above ...

    n = h.shape[0]
    dx = 2.0 / n

    # compute initial correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_score = cur_max

    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # leave some margin

    # Temperature schedule parameters
    T0 = cur_max * 0.5 + 1e-6
    # We'll compute linear decay to small value near 0
    # iteration count can be indefinite; compute temperature based on elapsed fraction.
    iteration = 0

    while time.time() < end_time:
        # Random indices
        i, j = random.sample(range(n), 2)
        # Compute feasible delta bounds
        hi = h[i]
        hj = h[j]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        # Limit max step size
        max_step = 0.2  # could be adaptively decreased later
        low = max(delta_min, -max_step)
        high = min(delta_max, max_step)
        if low >= high:
            continue
        delta = random.uniform(low, high)

        # Apply move
        apply_delta(h, corr, i, delta)
        apply_delta(h, corr, j, -delta)  # decreasing h[j] by delta

        new_max = corr.max()
        delta_obj = new_max - cur_max

        # Compute temperature based on progress
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s * 0.95)
        # Linear decay: T = T0 * (1 - progress)
        T = max(T0 * (1.0 - progress), 1e-8)

        accept = False
        if delta_obj <= 0:
            accept = True
        else:
            # Accept with probability exp(-delta_obj / T)
            prob = np.exp(-delta_obj / T) if T > 0 else 0.0

def apply_delta(h, corr, idx, delta):
    # h: mutable numpy array
    # corr: mutable numpy array, length 2*len(h)-1
    n = h.shape[0]
    # compute w before updating h
    w = 1.0 - h
    # first contribution: h[idx] factor
    start = n - 1 - idx
    # slice length n
    corr[start:start + n] += delta * w
    # update h
    h[idx] += delta
    # second contribution: w[idx] factor (after h updated)
    # also need reversed h
    corr[idx: idx + n] -= delta * h[::-1]

    while abs(diff) > 1e-12:

            # need to decrease sum
            # find indices where v > 0
            idxs = np.where(v > 0)[0]
            if len(idxs) == 0:
                break
            # compute per-index reduction
            dec = diff / len(idxs)
            # limit dec to each element's current value
            for i in idxs:
                if diff <= 0:
                    break

                v[i] -= allowable
                diff -= allowable

            for i in idxs:
                if diff >= 0:
                    break
                allowable = min(1 - v[i], inc)
                v[i] += allowable
                diff += allowable
    return v

import time
import random
import numpy as np

def _apply_delta(h: np.ndarray, corr: np.ndarray, idx: int, delta: float) -> None:
    """Apply a change of `delta` to h[idx] (h[idx] += delta) and update correlation."""
    n = h.shape[0]
    # w before h is changed
    w = 1.0 - h
    # first part: h[idx] factor
    start = n - 1 - idx                     # index in corr where shift = -idx starts
    corr[start:start + n] += delta * w      # vectorised update over all overlapping shifts
    # update h
    h[idx] += delta
    # second part: w[idx] factor (after h changed)
    # slice corresponds to shifts where w[idx] appears as second factor
    corr[idx:idx + n] -= delta * h[::-1]    # reversed h aligns with the order of shifts


def _make_feasible(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum(v) = len(v)/2."""
    n = v.shape[0]
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = total - target
    if abs(diff) < 1e-12:
        return v
    # Try a quick convex projection via cvxpy (optional)
    try:
        import cvxpy as cp
        x = cp.Variable(n)
        prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)),
                         [x >= 0, x <= 1, cp.sum(x) == target])
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        if prob.status in ("optimal", "optimal_inaccurate"):
            return np.array(x.value).flatten()
    except Exception:
        pass
    # Simple heuristic: distribute the excess uniformly over free entries
    if diff > 0:
        # need to decrease sum
        while diff > 1e-12:
            idxs = np.where(v > 0)[0]
            if len(idxs) == 0:
                break
            dec = diff / len(idxs)
            for i in idxs:
                step = min(v[i], dec)
                v[i] -= step
                diff -= step

                    break
