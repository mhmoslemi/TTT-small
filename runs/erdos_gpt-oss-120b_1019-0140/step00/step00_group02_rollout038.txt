def project_bound_simplex(v, target, max_iter=100, tol=1e-9):
    v = np.clip(v, 0.0, 1.0)
    lower = -1.0  # lam can be negative enough to push many to 1
    upper = 1.0   # lam can be positive enough to push many to 0
    for _ in range(max_iter):
        lam = (lower + upper) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(w)
        if abs(s - target) <= tol:
            return w
        if s > target:
            # need to increase lam to reduce sum
            lower = lam
        else:
            # need to decrease lam to increase sum
            upper = lam
    return np.clip(v - (lower+upper)/2.0, 0.0, 1.0)

def compute_gradient(h, lag):
    n = len(h)
    # Using vectorized: shift arrays with zeros.
    pos_shift = np.zeros_like(h)
    neg_shift = np.zeros_like(h)
    if lag >= 0:
        pos_shift[:n-lag] = h[lag:]  # h_{i+lag}
        # negative part: h_{i-lag} corresponds to shift in opposite direction.
        # Since lag>=0, i-lag >=0 for i>=lag: we can set neg_shift[lag:] = h[:n-lag]
        # Actually for i-lag >=0 -> i>=lag; we set neg_shift[lag:] = h[:n-lag]
        neg_shift[lag:] = h[:n-lag]
    else:  # lag negative
        l = -lag
        # pos_shift part: i+lag = i - l; need i-l >=0 => i>=l
        pos_shift[l:] = h[:n-l]  # because i+lag = i-l, and h_i+lag = h_{i-l}
        # neg_shift part: i-lag = i + l; i + l < n => i < n - l
        neg_shift[:n-l] = h[l:]  # because i-lag = i+l
    # gradient = 1 - pos_shift - neg_shift
    g = 1.0 - pos_shift - neg_shift
    return g

def project_bound_simplex(v, target, tol=1e-8, max_iter=100):
    # ensure vector within [0,1] then find lambda s.t. w = clip(v - lam, 0, 1) has sum target.
    v = np.clip(v, 0.0, 1.0)
    # If sum already equals target: return.
    s0 = np.sum(v)
    if abs(s0 - target) <= tol:
        return v
    # define bounds for lambda
    # lower bound (lam_low) when all w_i = 1, i.e., v_i - lam >=1 => lam <= v_i - 1 for all i => lam <= min(v_i - 1)
    lam_low = min(v) - 1.0  # because v_i - 1 <= lam_low such that w_i=1
    # upper bound (lam_up) when all w_i = 0, i.e., v_i - lam <=0 => lam >= max(v_i)
    lam_up = max(v)
    # However target may be out of possible sum [0, n]; but target = n/2, feasible.

def compute_gradient(h, lag):
    n = h.shape[0]
    # vector versions:
    pos_shift = np.zeros_like(h)
    neg_shift = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            pos_shift[:n-lag] = h[lag:]
            neg_shift[lag:] = h[:n-lag]
    else:
        l = -lag
        if l < n:
            pos_shift[l:] = h[:n-l]
            neg_shift[:n-l] = h[l:]
    g = 1.0 - pos_shift - neg_shift
    return g

import numpy as np
import time
import math

def project_bound_simplex(v, target, tol=1e-9, max_iter=100):
    """Project vector v onto the set {w: 0<=w_i<=1, sum(w)=target}."""
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) <= tol:
        return v
    # lambda bounds: when lambda very negative, all entries clip to 1
    lam_low = -1.0  # enough to push everything to 1
    lam_high = 1.0  # enough to push everything to 0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(w)
        if abs(s - target) <= tol:
            return w
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - (lam_low + lam_high) / 2.0, 0.0, 1.0)

def compute_c5(h):
    """Return the C5 value for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def evaluate(h):
    """Return (C5, lag) where lag is the integer shift giving the max."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return corr[idx] * dx, idx - (n - 1)

def worst_lag(h):
    """Lag (positive or negative) achieving the maximal correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return idx - (n - 1)

def gradient(h, lag):
    """Gradient of the worst lag term w.r.t. h."""
    n = h.shape[0]
    pos = np.zeros_like(h)
    neg = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            pos[:n - lag] = h[lag:]
            neg[lag:] = h[:n - lag]
    else:
        l = -lag
        if l < n:
            pos[l:] = h[:n - l]
            neg[:n - l] = h[l:]
    return 1.0 - pos - neg

def shape_h(a, n):
    """Construct the piecewise‑linear step function for a given a (0<a<0.5)."""
    b = 1.0 - a
    width = b - a
    if width <= 0:
        width = 1e-6
    dx = 2.0 / n
    x = np.linspace(0.0, 2.0, n, endpoint=False) + dx * 0.5
    h = np.zeros_like(x)
    # rising ramp
    mask = (x >= a) & (x < b)
    h[mask] = (x[mask] - a) / width
    # plateau
    mask = (x >= b) & (x < 2.0 - b)
    h[mask] = 1.0
    # falling ramp
    mask = (x >= 2.0 - b) & (x < 2.0 - a)
    h[mask] = (2.0 - a - x[mask]) / width
    # enforce bounds and exact integral
    target = n / 2.0
    return project_bound_simplex(h, target)

def upsample_h(h0, n):
    """Linearly interpolate a coarse vector to length n."""
    n0 = h0.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x, x0, h0)

def random_feasible(n, rng):
    """Generate a random vector satisfying the constraints."""
    v = rng.random(n)
    return project_bound_simplex(v, n / 2.0)

def perturb(h, sigma, rng):
    """Random small perturbation followed by projection."""
    n = h.shape[0]
    m = max(1, int(0.03 * n))
    idx = rng.choice(n, size=m, replace=False)
    delta = rng.normal(scale=sigma, size=m)
    h_new = h.copy()
    h_new[idx] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- discretisation ---
    n_points = 256                     # moderate size, good trade‑off
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # --- initial candidates ---
    best_h = None
    best_score = float('inf')

    # 1) use parent construction if available
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        h0 = upsample_h(parent, n_points)
        h0 = project_bound_simplex(h0, target_sum)
        s0 = compute_c5(h0)
        best_h, best_score = h0, s0

    # 2) random starts
    for _ in range(5):
        h = random_feasible(n_points, rng)
        s = compute_c5(h)
        if s < best_score:
            best_h, best_score = h, s

    # 3) one‑parameter shape sweep
    a_vals = np.linspace(0.02, 0.48, 120)
    for a in a_vals:
        h = shape_h(a, n_points)
        s = compute_c5(h)
        if s < best_score:
            best_h, best_score = h, s

    # ensure we have a current state
    current_h = best_h.copy()
    current_score = best_score

    # --- simulated annealing / local search ---
    it = 0
    max_iter = 1000000  # safety cap
    sigma0 = 0.15
    T0 = 0.02

    while True:
        now = time.time()
        if now - start_time > budget_s - 0.5 or it >= max_iter:
            break

        # cooling schedules
        sigma = sigma0 * (0.9995 ** it)
        T = T0 * (0.9995 ** it)

        # propose a new point
        cand = perturb(current_h, sigma, rng)
        cand = project_bound_simplex(cand, target_sum)
        cand_score = compute_c5(cand)

        # Metropolis acceptance
        if cand_score < current_score or rng.random() < math.exp(-(cand_score - current_score) / max(T, 1e-12)):
            current_h = cand
            current_score = cand_score

        # keep global best
        if cand_score < best_score:
            best_h = cand.copy()
            best_score = cand_score

        # occasional aggressive move to escape stagnation
        if it % 20000 == 0 and it > 0:
            # inject a larger random perturbation
            tmp = random_feasible(n_points, rng)
            current_h = tmp
            current_score = compute_c5(tmp)

        it += 1

    # --- final refinement using gradient steps ---
    for _ in range(2000):
        lag = worst_lag(best_h)
        g = gradient(best_h, lag)
        step = 0.005
        cand = best_h - step * g
        cand = project_bound_simplex(cand, target_sum)
        s = compute_c5(cand)
        if s < best_score:
            best_h = cand
            best_score = s

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_bound_simplex(best_h, target_sum)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_points
