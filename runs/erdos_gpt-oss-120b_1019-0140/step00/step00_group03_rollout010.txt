def random_binary(n, target_sum, rng):
    h = np.zeros(n, dtype=float)

    return h

def greedy_optimize(start_h, time_end):
    cur_h = start_h.copy()
    cur_score = evaluate_score(cur_h)
    # Greedy descent loops until no improvement or time out
    while True:

import numpy as np
import time
import math

def _evaluate_score(h):
    # evaluate_erdos_solution is provided in the execution environment
    return float(evaluate_erdos_solution(h))

def _random_binary(n, target_sum, rng):
    """Return a binary (0/1) vector of length n with exactly target_sum ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=target_sum, replace=False)
    h[idx] = 1.0
    return h

def _greedy_optimize(start_h, time_limit):
    """
    Repeatedly examine all possible 1↔0 swaps and perform the swap that gives the
    greatest reduction of the max‑overlap value. Stop when no improving swap exists
    or the allotted time expires.
    """
    cur_h = start_h.copy()
    cur_score = _evaluate_score(cur_h)

    while True:
        if time.time() > time_limit:
            break

        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)

        best_swap = None
        best_score = cur_score

        # Exhaustively test every feasible swap
        for i in ones:
            for j in zeros:
                cand = cur_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                cand_score = _evaluate_score(cand)
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_swap = (i, j)

        if best_swap is None:
            break  # local optimum reached

        i, j = best_swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cur_score = best_score

    return cur_h, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a binary step function h on [0,2] (represented by n points)
    that minimizes the Erdős overlap constant C₅.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine problem size and initialise a feasible vector
    init_vals = globals().get('initial_h_values')
    if init_vals is not None:
        h0 = np.array(init_vals, dtype=float)
        n = h0.size
        target_sum = n // 2
        h0 = np.clip(h0, 0, 1)
        h0 = (h0 > 0.5).astype(float)          # force binary
        if int(round(h0.sum())) != target_sum: # ensure correct cardinality
            h0 = _random_binary(n, target_sum, rng)
    else:
        n = 70
        target_sum = n // 2
        h0 = _random_binary(n, target_sum, rng)

    best_h = h0.copy()

    start_time = time.time()

    restart = 0

        restart += 1
        # First iteration starts from the current best; later ones are random
        if restart == 1:
