def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def improve_h(h, max_iter, rng):
    best = h.copy()
    best_score = compute_c5(best)
    n = len(h)
    idx_ones = np.where(best == 1)[0]
    idx_zeros = np.where(best == 0)[0]
    for _ in range(max_iter):
        # randomly select i from idx_ones, j from idx_zeros
        i = rng.choice(idx_ones)
        j = rng.choice(idx_zeros)
        # swap
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        # compute
        score = compute_c5(cand)
        if score < best_score - 1e-9:  # improvement
            best = cand
            best_score = score
            idx_ones = np.where(best == 1)[0]
            idx_zeros = np.where(best == 0)[0]
    return best, best_score

    # term1: indices i where h_i appears in first factor

import numpy as np
import time
import math

def _c5_of(h):
    """Compute C5 = max correlation of h with (1-h) times dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _simulated_annealing(h0, time_limit, start, total_time, rng):
    """Simple SA that preserves the sum constraint by moving mass between two entries."""
    h = h0.copy()
    cur_score = _c5_of(h)
    best_h = h.copy()
    best_score = cur_score
    n = len(h)

    # temperature schedule parameters
    T_start = 0.1
    T_end = 1e-5
    stagnant = 0

    while True:
        now = time.time()
        if now >= time_limit:
            break
        # progress for temperature
        prog = (now - start) / total_time
        T = T_start * (1.0 - prog) + T_end * prog

        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue

        # transfer mass from i to j or opposite
        if rng.rand() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = rng.rand() * max_delta
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0:
                continue
            delta = rng.rand() * max_delta
            cand = h.copy()
            cand[j] -= delta
            cand[i] += delta

        cand_score = _c5_of(cand)
        dE = cand_score - cur_score
        if dE < 0 or rng.rand() < math.exp(-dE / (T + 1e-12)):
            h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        # occasional restart if stuck
        if stagnant > 5000:
            # random feasible vector (average 0.5)
            r = rng.rand(n)
            r = r / r.mean() * 0.5
            r = np.clip(r, 0.0, 1.0)
            # enforce exact sum by uniform adjustment
            diff = n / 2.0 - r.sum()
            r += diff / n
            r = np.clip(r, 0.0, 1.0)
            h = r
            cur_score = _c5_of(h)
            stagnant = 0

    return best_h, best_score
