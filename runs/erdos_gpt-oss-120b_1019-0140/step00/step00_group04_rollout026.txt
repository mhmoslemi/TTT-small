   best_h = h.copy()

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Using np.correlate; ensures float
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def to_binary(arr):
    n = arr.shape[0]
    k = n // 2
    # Use argpartition to get indices of top k elements
    idx = np.argpartition(-arr, k-1)[:k]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def hill_climb_binary(h_init, start_time, budget_s):
    h = h_init.copy()
    n = h.shape[0]
    rng = np.random.default_rng()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    # Lists for quick random selection
    idx_one = list(np.where(h == 1)[0])
    idx_zero = list(np.where(h == 0)[0])
    # Parameters
    steps_without = 0
    reset_after = max(2000, n)  # may be 2000 or n
    max_no_improve = reset_after
    # main loop
    while True:
        if time.time() - start_time >= budget_s * 0.95:
            break
        # propose swap
        i_pos = rng.integers(len(idx_one))
        j_pos = rng.integers(len(idx_zero))
        i = idx_one[i_pos]
        j = idx_zero[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # update index lists
        idx_one[i_pos] = j
        idx_zero[j_pos] = i
        # evaluate
        new_c5 = compute_c5(h)
        if new_c5 < current_c5:
            current_c5 = new_c5
            steps_without = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            idx_one[i_pos] = i
            idx_zero[j_pos] = j
            steps_without += 1
        # reset/perturb if stuck
        if steps_without >= reset_after:
            # randomize few swaps
            num_swaps = max(1, n // 20)
            for _ in range(num_swaps):
                i_pos = rng.integers(len(idx_one))
                j_pos = rng.integers(len(idx_zero))
                i = idx_one[i_pos]
                j = idx_zero[j_pos]
                h[i] = 0.0
                h[j] = 1.0
                idx_one[i_pos] = j
                idx_zero[j_pos] = i
            current_c5 = compute_c5(h)
            steps_without = 0
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def to_binary(arr):
    n = arr.shape[0]
    k = n // 2
    idx = np.argpartition(-arr, k - 1)[:k]

    h[idx] = 1.0
    return h
