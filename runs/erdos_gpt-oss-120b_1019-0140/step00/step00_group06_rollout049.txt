    n = len(h)
    current_h = h.copy()
    current_score = compute_c5(current_h)
    best_h = current_h.copy()
    best_score = current_score
    start = time.time()
    for it in range(n_iters):
        # check time
        if time_limit is not None and time.time() - start > time_limit:
            break
        # propose move
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        sum_ij = current_h[i] + current_h[j]
        lo = max(0.0, sum_ij - 1.0)
        hi = min(1.0, sum_ij)
        if hi - lo < 1e-12:
            continue
        new_i = np.random.uniform(lo, hi)
        new_j = sum_ij - new_i
        # Save old and set new
        old_i, old_j = current_h[i], current_h[j]
        current_h[i] = new_i
        current_h[j] = new_j
        new_score = compute_c5(current_h)
        # acceptance:
        if new_score < current_score:
            accept = True
        else:
            # simulated annealing acceptance
            if temperature_schedule is not None:
                T = temperature_schedule(it)
                accept = np.random.rand() < np.exp((current_score - new_score) / T)
            else:
                accept = False
        if accept:
            current_score = new_score
            # maybe update best
            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # revert
            current_h[i] = old_i
            current_h[j] = old_j
    return best_h, best_score

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
import time

def compute_c5(h):
    """Return max overlap integral C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraints(x, target):
    """Clip x to [0,1] and adjust to satisfy sum(x)=target (within tiny tolerance)."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if np.isclose(s, target, atol=1e-12):
        return x
    if s > 0.0:
        factor = target / s
        x = x * factor
        x = np.clip(x, 0.0, 1.0)
    # fix any tiny residual difference
    diff = target - x.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            cand = np.where(x < 1.0 - 1e-12)[0]
        else:
            cand = np.where(x > 1e-12)[0]
        if cand.size == 0:
            break
        idx = np.random.choice(cand)
        max_adj = (1.0 - x[idx]) if diff > 0.0 else x[idx]
        adj = min(abs(diff), max_adj)
        adj = adj if diff > 0.0 else -adj
        x[idx] += adj
        diff = target - x.sum()
    return x

def upscale_vector(h0, n_new):
    """Linearly interpolate a step vector to a finer resolution and re‑project."""
    n0 = h0.shape[0]

    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x0, h0)
    target = n_new / 2.0
    return project_to_constraints(h_new, target)

    """Simulated‑annealing style hill‑climb on the C5 objective."""

        j = np.random.randint(0, n - 1)

        new_j = sum_ij - new_i
