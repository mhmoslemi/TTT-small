    N = h.shape[0]

    rev = np.flip(1 - h)  # reversed 1-h for correlation (full)

    # Determine initial resolution

    # Ensure even base_n
    if base_n % 2 != 0:
        base_n += 1

    # Function to ensure binary vector of length N
    def init_binary(N):
        if init is not None and init.shape[0] == N:
            # threshold to binary
            h = (init >= 0.5).astype(float)
        elif init is not None:
            # upsample to N using repeat
            rep = N // init.shape[0]
            if rep > 0:
                h = np.repeat((init >= 0.5).astype(float), rep)
                # if N not multiple, fill rest with random
                remaining = N - h.shape[0]
                if remaining > 0:
                    extra = np.random.choice([0.,1.], size=remaining)
                    h = np.concatenate([h, extra])
            else:
                # N smaller than init size: just randomly sample
                indices = np.random.choice(init.shape[0], N, replace=False)
                h = (init[indices] >= 0.5).astype(float)
        else:

            h = np.zeros(N)
            half = N // 2
            h[:half] = 1

def simulated_annealing(N, init_h, start_temp, end_temp, max_iters, time_deadline):
    h = init_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # Precompute constant target sum and lists for sampling
    for it in range(max_iters):
        # time check
        if time.time() > time_deadline:
            break
        # temperature schedule: linear or exponential
        t = start_temp * ((end_temp / start_temp) ** (it / max_iters))
        # sample swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / max(t, 1e-12)):
            # accept
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def gradient_for_shift(h, k):
    N = h.shape[0]
    g = np.zeros(N)
    if k >= 0:
        # term1: 1 - h[i+k] for i=0..N-k-1
        g[:N-k] = 1 - h[k:]
        # term2: h[i - k] for i=k..N-1
        g[k:] -= h[:N-k]
    else:
        kp = -k
        g[kp:] = 1 - h[:N-kp]
        g[:N-kp] -= h[kp:]
    return g

def continuous_descent(h, target_sum, time_deadline):
    # Use adaptive step size and gradient descent for a fixed number of steps.
    # Determine number of steps based on time left.
    steps = 2000  # baseline
    # adjust step size schedule variable
    lr = 0.5
    decay = 0.995
    N = h.shape[0]
    for it in range(steps):
        if time.time() > time_deadline:
            break
        # compute conv and max index
        conv = full_overlap_counts(h)
        idx = conv.argmax()
        k = idx - (N - 1)
        g = gradient_for_shift(h, k)
        h = h - lr * g
        # projection
        h = project_onto_box_and_sum(h, target_sum)
        lr *= decay
    return h

def simulated_annealing_binary(h0, target_sum, time_deadline, max_iters=200000, start_temp_ratio=0.5, end_temp=1e-5):
    h = h0.copy()
    N = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start_temp = cur_c5 * start_temp_ratio + 1e-6
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # temperature schedule: exponential decreasing
        t = start_temp * ((end_temp / start_temp) ** (it / max_iters))
        # pick random ones and zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / max(t, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def binary_round_topk(h):
    N = h.shape[0]
    k = N // 2
    idx = np.argpartition(h, -k)[-k:]
    h_bin = np.zeros(N, dtype=float)
    h_bin[idx] = 1.0

def project_onto_box_and_sum(x, target_sum):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # Simple iterative adjustment
    # Use water-filling algorithm to meet sum while respecting bounds.
    # Compute deficit
    deficit = target_sum - x.sum()
    if abs(deficit) < 1e-12:
        return x
    # Identify free indices (not at bounds) to adjust
    # We'll use iterative approach
    max_iter = 10
    for _ in range(max_iter):
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size == 0:
            # adjust all equally (may violate bounds but small)
            x += deficit / len(x)
            x = np.clip(x, 0.0, 1.0)
            deficit = target_sum - x.sum()
            if abs(deficit) < 1e-12:
                break
            continue
        delta = deficit / free.size
        x[free] += delta
        # after addition, clip again to keep within bounds
        x = np.clip(x, 0.0, 1.0)
        deficit = target_sum - x.sum()
        if abs(deficit) < 1e-12:
            break
    # Final correction if still off by small epsilon
    if abs(deficit) > 1e-9:
        # distribute to all entries proportionally
        x += deficit / len(x)
        x = np.clip(x, 0.0, 1.0)
    # Ensure sum within tolerance
    return x

import numpy as np
import math
import time

def _full_overlap_counts(h: np.ndarray) -> np.ndarray:
    """Return the full overlap counts (not scaled) for h."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    rev = np.fft.rfft(np.flip(1.0 - h), n=L)
    conv = np.fft.irfft(H * rev, n=L)
    # eliminate tiny negative noise
    return np.maximum(conv, 0.0)


def compute_c5(h: np.ndarray) -> float:
    """C5 = max overlap * dx."""
    conv = _full_overlap_counts(h)
    max_val = conv.max()
    return max_val * (2.0 / h.shape[0])


def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of overlap_k w.r.t. h (for a given shift k)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        # (1 - h[i+k]) term
        g[: n - k] = 1.0 - h[k:]
        # -h[i-k] term
        g[k:] -= h[: n - k]
    else:
        kp = -k
        g[kp:] = 1.0 - h[: n - kp]
        g[: n - kp] -= h[kp:]
    return g


def _project_box_sum(x: np.ndarray, target_sum: float) -> np.ndarray:

    x = np.clip(x, 0.0, 1.0)

            # spread uniformly (will be re‑clipped)
            x += deficit / x.size
            x = np.clip(x, 0.0, 1.0)

                break
            continue
        delta = deficit / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        deficit = target_sum - x.sum()
        if abs(deficit) < 1e-12:
            break
    if abs(deficit) > 1e-9:
        x += deficit / x.size
        x = np.clip(x, 0.0, 1.0)
    return x


def _continuous_descent(h: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """Simple projected gradient descent on the max‑overlap objective."""
    n = h.shape[0]
    lr = 0.5
    decay = 0.995
    while time.time() < deadline:
        conv = _full_overlap_counts(h)
        idx = int(conv.argmax())
        k = idx - (n - 1)
        g = _gradient_for_shift(h, k)
        h = h - lr * g
        h = _project_box_sum(h, target_sum)
        lr *= decay
        if lr < 1e-6:
            break
    return h


def _binary_round_topk(h: np.ndarray) -> np.ndarray:
    """Return binary vector with exactly n/2 ones (largest entries become 1)."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out


def _simulated_annealing_binary(h0: np.ndarray, deadline: float,
                                max_iters: int = 200_000,
                                end_temp: float = 1e-5) -> tuple:
    """Binary SA that preserves the sum constraint."""
    h = h0.copy()
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    start_temp = max(cur, 0.1)
    n = h.shape[0]
    while time.time() < deadline and max_iters > 0:
        it = max_iters
        max_iters -= 1
        # temperature schedule (exponential)
        t = start_temp * ((end_temp / start_temp) ** (it / 200_000))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        delta = new - cur
        if delta < 0 or np.random.rand() < math.exp(-delta / max(t, 1e-12)):
            cur = new
            if cur < best:
                best, best_h = cur, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # construct a working length (even, at least twice the parent length)
    # ------------------------------------------------------------------
    try:
        base = np.array(initial_h_values, dtype=float)  # provided by harness
        base_len = base.shape[0]
    except NameError:
        base = None
        base_len = 0

    target_n = max(base_len * 2, 80)
    if target_n % 2:
        target_n += 1

    # ------------------------------------------------------------------
    # build several starting vectors
    # ------------------------------------------------------------------
    starters = []

    # 1) up‑sample the parent if available
    if base is not None:
        rep = max(1, target_n // base_len)
        up = np.repeat(base, rep)
        if up.shape[0] > target_n:
            up = up[:target_n]
        elif up.shape[0] < target_n:
            up = np.concatenate([up, np.zeros(target_n - up.shape[0])])
        starters.append(up)

    # 2) random balanced binary
    rand_bin = np.zeros(target_n)
    half = target_n // 2
    rand_bin[:half] = 1.0
    np.random.shuffle(rand_bin)
    starters.append(rand_bin)

    # 3) Beatty (Sturmian) sequence using golden‑ratio conjugate
    phi = (np.sqrt(5) - 1) / 2.0
    idxs = np.arange(target_n)
    beat = ((idxs * phi) % 1.0) < 0.5
    beat = beat.astype(float)
    # adjust exact half if off by one because of floating rounding
    cur_sum = beat.sum()
    need = int(target_n / 2 - cur_sum)
    if need > 0:
        zeros = np.where(beat == 0)[0]
        beat[zeros[:need]] = 1.0
    elif need < 0:
        ones = np.where(beat == 1)[0]
        beat[ones[: -need]] = 0.0
    starters.append(beat)

    # ------------------------------------------------------------------
    # optimisation loop – divide budget among starters
    # ------------------------------------------------------------------
    best_h, best_c5 = None, float("inf")
    time_budget = budget_s * 0.95  # keep a safety margin
    per_start = time_budget / len(starters)

    target_sum = target_n / 2.0

    for s0 in starters:
        t0 = time.time()
        deadline = t0 + per_start

        # ensure feasibility (box + sum)
        h = _project_box_sum(s0, target_sum)

        # --- continuous descent (quick) --------------------------------
        descent_deadline = min(deadline, t0 + per_start * 0.3)
        if time.time() < descent_deadline:
            h = _continuous_descent(h, target_sum, descent_deadline)

        # --- binary rounding -------------------------------------------
        h_bin = _binary_round_topk(h)
        c5 = compute_c5(h_bin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_bin.copy()

        # --- simulated annealing ----------------------------------------
        sa_deadline = deadline
        h_sa, c5_sa = _simulated_annealing_binary(h_bin, sa_deadline)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()

    # fallback – should never trigger but guarantees a return value
    if best_h is None:
        n = target_n
        bh = np.zeros(n)
        bh[: n // 2] = 1.0
        np.random.shuffle(bh)
        best_h, best_c5 = bh, compute_c5(bh)

    return (best_h, float(best_c5), int(target_n))
