import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max np.correlate(h, 1-h, mode='full') * dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _max_corr_index(h: np.ndarray):
    """Return (idx, value) where idx is the index in full correlation of the
    maximal entry and value is the (non‑scaled) correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx, float(corr[idx])

def _gradient_of_max(h: np.ndarray, idx: int) -> np.ndarray:
    """Gradient of the correlation entry at full‑index idx."""
    n = h.shape[0]
    dx = 2.0 / n
    k = idx - (n - 1)           # shift corresponding to idx
    grad = np.zeros_like(h)

    if k >= 0:                  # shift to the right
        # term hi * (1 - h_{i+k})
        i = np.arange(n - k)
        grad[i] += (1.0 - h[i + k]) * dx
        # term h_{i} appears as -(h_{i} * h_{i+k}) for i>=k
        i2 = np.arange(k, n)
        grad[i2] -= h[i2 - k] * dx

        grad[i] += (1.0 - h[i - L]) * dx
        i2 = np.arange(0, n - L)
        grad[i2] -= h[i2 + L] * dx
    return grad

                        h = hp

        # learning‑rate schedule
        it += 1
        if it % 800 == 0:
            lr *= 0.85

        # safety wall near the end of the budget
        if time.time() - start > budget * 0.97:
            break

    return best_h, best_score

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős C₅ problem.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of float
            The discretised step function on [0,2].
        c5_bound : float
            The achieved C₅ value.
        n_points : int
            Number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).ravel()
        n = h0.shape[0]
    else:
        n = 70
        h0 = np.zeros(n, dtype=np.float64)
        ones_idx = rng.choice(n, size=n // 2, replace=False)
        h0[ones_idx] = 1.0

    # ------------------------------------------------------------------
    # Optimization (use most of the time budget)
    # ------------------------------------------------------------------
    remaining = max(0.0, budget_s - 0.2)   # tiny safety margin
    best_h, best_score = _optimize(h0, remaining)

    # ------------------------------------------------------------------
    # Final evaluation using the provided helper if available
    # ------------------------------------------------------------------
    try:
        c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5 = _compute_c5(best_h)

    return best_h.tolist(), c5, int(n)
