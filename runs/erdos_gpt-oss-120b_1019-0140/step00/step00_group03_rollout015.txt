def milp_optimize(n, time_limit):
    import cvxpy as cp
    import numpy as np
    ones_needed = n // 2
    x = cp.Variable(n, boolean=True)
    # pairs
    pair_indices = []
    for i in range(n):
        for j in range(i):
            pair_indices.append((i,j))
    m = len(pair_indices)
    y = cp.Variable(m, boolean=True)
    # Build constraints list
    constraints = []
    constraints.append(cp.sum(x) == ones_needed)
    # linearizations
    for idx,(i,j) in enumerate(pair_indices):
        constraints.append(y[idx] <= x[i])
        constraints.append(y[idx] <= x[j])
        constraints.append(y[idx] >= x[i] + x[j] - 1)
    # t variable
    t = cp.Variable()
    constraints.append(t >= 0)
    # correlation constraints
    # Precompute mapping from pair to index
    pair_to_idx = { (i,j): idx for idx,(i,j) in enumerate(pair_indices) }
    for lag in range(-n+1, n):
        if lag == 0:
            continue
        pairs_for_lag = []
        if lag > 0:
            for i in range(lag, n):
                j = i - lag
                # i > j
                pairs_for_lag.append(pair_to_idx[(i,j)])
        else:
            d = -lag
            for i in range(0, n-d):
                j = i + d
                # j > i
                pairs_for_lag.append(pair_to_idx[(j,i)])
        if pairs_for_lag:
            sum_y = cp.sum(cp.hstack([y[idx] for idx in pairs_for_lag]))
            constraints.append(ones_needed - sum_y <= t)
    # objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    # Solve with time limit: set solver options if possible.
    # Choose solver. Try "CBC", "GLPK_MI", "SCIP".
    solvers = ["CBC", "GLPK_MI", "SCIP"]
    for solver in solvers:
        try:
            result = prob.solve(solver=solver, timeLimit=time_limit, verbose=False)
            # timeLimit parameter may not be accepted by all; catch TypeError.
            break
        except Exception as e:
            # try without timeLimit
            try:
                result = prob.solve(solver=solver, verbose=False)
                break
            except Exception as e2:
                continue
    # After solving, check status
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback
        return None, None
    x_val = np.round(x.value).astype(int) # ensure binary
    # compute bound
    bound = evaluate_erdos_solution(x_val)
    return x_val, bound

def simulated_annealing(h_initial, time_limit):
    import numpy as np, time, random, math
    n = len(h_initial)
    required_ones = n // 2
    # Ensure binary
    cur = np.array(h_initial, dtype=int)
    cur_sum = cur.sum()
    # adjust if not correct:
    if cur_sum != required_ones:
        # If too many ones, set some ones to zero; else set zeros to one.
        diff = cur_sum - required_ones
        if diff > 0:
            # Too many ones: randomly select diff ones and set to zero
            ones_idx = np.where(cur == 1)[0]
            np.random.shuffle(ones_idx)
            cur[ones_idx[:diff]] = 0
        elif diff < 0:
            zeros_idx = np.where(cur == 0)[0]
            np.random.shuffle(zeros_idx)
            cur[zeros_idx[: -diff]] = 1
    cur_score = evaluate_erdos_solution(cur)
    best = cur.copy()
    best_score = cur_score
    # Temperature schedule
    T = 1.0
    decay = 0.9995 # gradually reduce
    # accept threshold
    start = time.time()
    iterations = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit - 0.1:
            break
        # propose swap: pick one index i where cur[i]==1 and one index j where cur[j]==0
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_score = evaluate_erdos_solution(cur)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # acceptance probability
            if T > 0:
                prob = math.exp(-delta / T)
                if random.random() < prob:
                    accept = True
        if accept:
            cur_score = new_score
            # if new is better than best update best
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
        # decay temperature
        T *= decay
        # occasionally reset temperature if stagnated?
        iterations += 1
        # maybe restart if T too low? Not needed.
    return best, best_score
