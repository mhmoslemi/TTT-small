def neighbor(h, step):
    n = len(h)
    # pick two indices
    i, j = np.random.choice(n, size=2, replace=False)
    # compute max delta that can be moved from i to j while staying in [0,1]
    max_delta = min(h[i], 1 - h[j])

    n = len(h)
    eps = 1e-12
    if np.random.rand() < p_target:

        if ds >= 0:

            i_end = n - ds
            if i_end > i_start:
                idx_range_i = np.arange(i_start, i_end)
                mask_i = (h[idx_range_i] > eps) & (h[idx_range_i + ds] < 1 - eps)
                idx_i = idx_range_i[mask_i]
                # improvement candidates
                mask_j = (h[idx_range_i] < eps) & (h[idx_range_i + ds] > 1 - eps)

    n = h.shape[0]
    eps = 1e-12
    if shift >= 0:
        start, end = 0, n - shift
        if end <= start:
            return None
        idx = np.arange(start, end)
        mask_i = (h[idx] > eps) & (h[idx + shift] < 1 - eps)   # donors
        mask_j = (h[idx] < eps) & (h[idx + shift] > 1 - eps)   # recipients
        idx_i = idx[mask_i]
        idx_j = idx[mask_j]
    else:
        s = -shift
        start, end = s, n
        if end <= start:
            return None
        idx = np.arange(start, end)
        mask_i = (h[idx] > eps) & (h[idx - s] < 1 - eps)
        mask_j = (h[idx] < eps) & (h[idx - s] > 1 - eps)
        idx_i = idx[mask_i]
        idx_j = idx[mask_j]
    if idx_i.size == 0 or idx_j.size == 0:
        return None
    i = int(np.random.choice(idx_i))
    j = int(np.random.choice(idx_j))
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = min(step, max_delta) * np.random.rand()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def random_neighbor(h, step):
    """Simple mass‑transfer move that preserves the total sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0:
            return h.copy()
        i, j = j, i
    delta = min(step, max_delta) * np.random.rand()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes the
    Erdős overlap constant C5. Returns (h_values, c5_bound, n_points)."""
    start = time.time()
    np.random.seed(seed)

    # ----- initial feasible vector ---------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 200                           # fallback size
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        for idx in range(n):
            if (diff > 0 and h[idx] + diff <= 1.0) or (diff < 0 and h[idx] + diff >= 0.0):
                h[idx] += diff
                break

    cur_c, cur_conv = evaluate(h)
    best_h, best_c = h.copy(), cur_c

    # ----- annealing schedule ---------------------------------------
    T0 = 0.001          # initial temperature
    step = 0.3          # initial maximal transferred mass
    min_step = 1e-4

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break

        frac = elapsed / budget_s
        T = max(1e-12, T0 * (1.0 - frac))     # linear cooling to 0
        step = max(min_step, step * (1.0 - 5e-5))

        # worst shift for the current configuration
        worst_idx = int(np.argmax(cur_conv))
        shift = worst_idx - (n - 1)

        # propose a neighbour
        new_h = targeted_neighbor(h, shift, step)
        if new_h is None:
            new_h = random_neighbor(h, step)

        new_c, new_conv = evaluate(new_h)

        # Metropolis acceptance
        if new_c <= cur_c:
            accept = True
        else:
            prob = np.exp(-(new_c - cur_c) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            h, cur_c, cur_conv = new_h, new_c, new_conv
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
                if best_c <= 0.3808:
                    break

    return best_h.tolist(), float(best_c), int(n)
