def project_onto_bounds_and_sum(v, target):
    v = np.clip(v, 0, 1)
    # if sum close, return
    current = np.sum(v)
    if np.isclose(current, target):
        return v
    # If current > target, we need to reduce some entries >0.
    # If current < target, we need to increase some entries <1.
    # Classic algorithm: use water-filling, but for simplicity we can do:
    # Use scaling factor first
    if current > 0:
        v = v * (target / current)
    else:
        v = np.full_like(v, target / len(v)) # but sum will be target, each same <1 maybe.
    # after scaling, clip again and adjust using small offset.
    v = np.clip(v, 0, 1)
    # If sum still not target (due to clipping), adjust by adding/subtracting small delta to entries not at bound.
    # Simple iterative correction:
    for _ in range(10):
        cur = np.sum(v)
        diff = target - cur
        if abs(diff) < 1e-10:
            break
        # Identify indices not at bound for adjustment
        free_idx = np.where((v > 0) & (v < 1))[0]
        if len(free_idx) == 0:
            # No free entries; distribute diff uniformly to all (clipping)
            v = np.clip(v + diff / len(v), 0,1)
        else:
            # adjust uniformly among free entries
            v[free_idx] += diff / len(free_idx)
            v = np.clip(v, 0,1)
    return v

import numpy as np
import time
from scipy.optimize import minimize

def _project_to_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v) = target.
    Simple iterative water-filling approach.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        cur_sum = v.sum()
        diff = target - cur_sum
        if abs(diff) < 1e-12:
            break
        # free indices are those not at a bound
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size == 0:
            # all variables are at a bound; distribute diff uniformly and re‑clip
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
        else:
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
    # Final tiny adjustment to hit the target as closely as possible
    cur_sum = v.sum()
    if abs(cur_sum - target) > 1e-10:
        # Distribute remaining discrepancy over non‑saturated entries
        free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
        if free.size > 0:
            v[free] += (target - cur_sum) / free.size
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial guess

    try:
