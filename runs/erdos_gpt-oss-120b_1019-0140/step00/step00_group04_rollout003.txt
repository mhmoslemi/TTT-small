def upsample(old_h, new_n):

    if old_n == new_n:
        return old_h.copy()
    # normalized positions from 0 to 1 (excluded endpoint)

import time
import numpy as np

# Helper: compute C5 for a given h using the same definition as the evaluator
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

# Project a vector onto the feasible set {0<=h_i<=1, sum h_i = n/2}
def project_feasible(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # iterative water-filling
    tol = 1e-12
    while abs(diff) > tol:
        if diff > 0:  # need to add mass
            caps = 1.0 - h
            idx = caps > tol
            total = caps[idx].sum()
            if total < tol:
                break
            add = min(diff, total) * caps[idx] / total
            h[idx] += add
        else:          # need to remove mass
            caps = h
            idx = caps > tol
            total = caps[idx].sum()
            if total < tol:
                break
            sub = min(-diff, total) * caps[idx] / total
            h[idx] -= sub
        diff = target - h.sum()
    # tiny numerical overshoots
    h = np.clip(h, 0.0, 1.0)
    # final small adjustment if still off (rare)
    final_diff = target - h.sum()
    if abs(final_diff) > tol:
        h += final_diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

# Upsample a vector to a larger grid using linear interpolation
def upsample(old_h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

# Simple random transfer between two indices (keeps sum invariant)
def random_transfer(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # max amount we can move from i to j
    max_dec_i = h[i]
    max_inc_j = 1.0 - h[j]
    max_pos = min(max_dec_i, max_inc_j)
    # opposite direction
    max_dec_j = h[j]
    max_inc_i = 1.0 - h[i]
    max_neg = -min(max_dec_j, max_inc_i)
    if max_pos == 0.0 and max_neg == 0.0:
        return h.copy()
    delta = rng.uniform(max_neg, max_pos)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

# Gradient-based move aimed at reducing the current maximal overlap
def gradient_transfer(h: np.ndarray, lag: int, rng: np.random.Generator) -> np.ndarray | None:
    n = h.shape[0]
    idx = np.arange(n)
    j1 = idx + lag
    term1 = np.zeros(n)
    mask1 = (j1 >= 0) & (j1 < n)
    term1[mask1] = 1.0 - h[j1[mask1]]
    j2 = idx - lag
    term2 = np.zeros(n)
    mask2 = (j2 >= 0) & (j2 < n)
    term2[mask2] = -h[j2[mask2]]
    grad = term1 + term2
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i_from = rng.choice(pos)
    i_to = rng.choice(neg)
    max_transfer = min(h[i_from], 1.0 - h[i_to])
    if max_transfer <= 0.0:
        return None
    # transfer a random fraction (0 < f <= 1) of the available amount
    f = rng.random()
    delta = max_transfer * f
    h_new = h.copy()
    h_new[i_from] -= delta
    h_new[i_to] += delta
    return h_new

# Small perturbation of a solution (used when stuck)
def perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.05) -> np.ndarray:
    n = h.shape[0]
    k = max(1, n // 10)
    idx = rng.choice(n, size=k, replace=False)
    noise = rng.normal(0.0, scale, size=k)
    h_new = h.copy()
    h_new[idx] += noise
    return h_new
