    # Compute initial best
    best_h = h.copy()

    # Variables

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # dx = 2/n
    return float(np.max(corr) * (2.0 / n))

def repair(vec: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and enforce sum(vec) == n/2 exactly."""
    vec = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - vec.sum()
    # Greedy redistribution of the residual difference.
    if abs(diff) > 1e-12:
        # First try to add to entries that are not at the upper bound.
        if diff > 0:
            for i in range(n):
                cap = 1.0 - vec[i]
                if cap <= 0:
                    continue
                add = min(diff, cap)
                vec[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:  # diff < 0, need to subtract
            for i in range(n):
                cap = vec[i]  # amount we can subtract
                if cap <= 0:
                    continue
                sub = min(-diff, cap)
                vec[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    # Any remaining tiny discrepancy is spread over the first element safely.
    residual = target - vec.sum()
    if abs(residual) > 1e-12:
        # There must be capacity somewhere; adjust the first feasible index.
        for i in range(n):
            if residual > 0 and vec[i] < 1.0:
                add = min(residual, 1.0 - vec[i])
                vec[i] += add
                break
            if residual < 0 and vec[i] > 0.0:
                sub = min(-residual, vec[i])
                vec[i] -= sub
                break
    return vec

def local_search(h_start: np.ndarray, n: int, time_limit: float, rng: np.random.Generator):
    """Simple simulated‑annealing hill‑climbing on the feasible set."""
    # Ensure we start from a feasible point.
    h_cur = repair(h_start.copy(), n)
    cur_score = compute_c5(h_cur)

    best_h = h_cur.copy()
    best_score = cur_score

    # Parameters for the annealing schedule.
    step_max = 0.15        # maximum mass moved between two entries
    T = 0.01               # initial temperature
    cooling = 0.9995       # geometric cooling rate

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # Pick two distinct indices.
        i, j = rng.choice(n, size=2, replace=False)

        # Propose a transfer delta (positive means i gains, j loses).
        delta = rng.uniform(-step_max, step_max)
        if delta > 0:
            delta = min(delta, 1.0 - h_cur[i], h_cur[j])
        else:
            delta = max(delta, -h_cur[i], -(1.0 - h_cur[j]))
        if abs(delta) < 1e-15:
            continue

        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_score = compute_c5(h_new)

        # Metropolis acceptance.
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

        # Cooling schedule.
        T *= cooling
        it += 1
        if it % 1000 == 0:
            # slowly shrink the proposal size to allow fine tuning.
            step_max = max(step_max * 0.99, 1e-4)

    return best_h, best_score

    Returns a tuple (h_values, c5_bound, n_points).

    rng = np.random.default_rng(seed)
    start_total = time.time()
    margin = 0.5  # guard time before hard limit

    # ------------------------------------------------------------------

    best_h = repair(best_h, best_n)
    final_score = compute_c5(best_h)

    return best_h.astype(np.float64), float(final_score), int(best_n)
