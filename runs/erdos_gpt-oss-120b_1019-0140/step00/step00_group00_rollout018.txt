def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b = (1 - h)[::-1]
    # FFT convolution size:
    size = 1 << (2*n-2).bit_length()
    # Actually size should be at least 2*n-1.
    size = 1 << ((2*n - 1).bit_length())  # smallest power of 2 >= 2n-1.
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    # conv has length size, but only first 2*n-1 entries relevant.
    conv = conv[:2*n-1]
    return np.max(conv) * dx

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation via convolution with reversed complement
    conv = np.convolve(h, (1.0 - h)[::-1], mode="full")
    return float(np.max(conv) * dx)


def _adjust_sum(h: np.ndarray, target: float):
    """Adjust one entry (if possible) to make sum exactly target."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # find index that can absorb the whole diff without leaving [0,1]
    cand = np.where((h + diff >= 0.0) & (h + diff <= 1.0))[0]
    if cand.size:
        h[cand[0]] += diff
    else:
        # fallback: distribute diff uniformly and clip
        step = diff / h.size
        h += step
        np.clip(h, 0.0, 1.0, out=h)
        # re‑scale to target (tiny correction)
        scale = target / h.sum()
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
    return h
