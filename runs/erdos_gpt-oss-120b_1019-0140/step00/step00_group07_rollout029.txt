def _c5_of(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx via discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # linear correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

            scale = target_sum / cur if cur > 0 else 0.0
            h *= scale
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - float(np.sum(h))
            if abs(diff) > 1e-12:
                idx = np.where((h > 0.0) & (h < 1.0))[0]

                    idx = np.arange(n)
                h[idx[0]] += diff
