import numpy as np, time, math
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # use float dtype
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_binary_start(n, init=None):
    if init is not None:
        arr = np.asarray(init, dtype=float)
        # ensure length n
        if arr.shape[0] != n:
            # fallback to random
            ones = np.random.choice(n, n//2, replace=False)
            h = np.zeros(n, dtype=float)
            h[ones] = 1.0
            return h
        # sort by value descending, assign top n/2 to 1
        idx = np.argsort(-arr)
        h = np.zeros(n, dtype=float)
        h[idx[:n//2]] = 1.0
        return h
    else:
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1.0
        np.random.shuffle(h)
        return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    # Determine n from initial_h_values if present
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        # Ensure even length
        if n % 2 != 0:
            # make it even by dropping last element
            n = n - 1
            init_arr = init_arr[:n]
        h0 = get_binary_start(n, init_arr)
    else:
        n = 80
        h0 = get_binary_start(n, None)
    # Ensure sum constraint
    # (Binary guarantee)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_c5 = best_c5
    # parameters
    start_time = time.time()
    max_no_improve = 2000
    no_improve = 0
    # Simulated annealing parameters
    T0 = 0.005  # initial temperature for acceptance of worse moves
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # pick random swap
        ones_idx = np.where(curr_h > 0.5)[0]
        zeros_idx = np.where(curr_h <= 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        new_h = curr_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)
        if new_c5 < curr_c5:
            curr_h = new_h
            curr_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            # accept with probability exp(-(new - curr) / T)
            # temperature linearly decays
            T = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.9))
            if T > 0 and np.random.rand() < math.exp(-(new_c5 - curr_c5) / T):
                curr_h = new_h
                curr_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        # restart if stagnant
        if no_improve >= max_no_improve:
            # random restart preserving weight
            curr_h = np.zeros(n, dtype=float)
            curr_h[:n//2] = 1.0
            np.random.shuffle(curr_h)
            curr_c5 = compute_c5(curr_h)
            no_improve = 0
    # Final evaluation to ensure bound
    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), n)

def solve_milp(n):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n//2, t >= 0]
    # For each shift k
    for k in range(-(n-1), n):
        # Determine overlapping indices
        if k >= 0:
            idx_i = np.arange(0, n - k)
            idx_j = idx_i + k
        else:
            idx_j = np.arange(0, n + k)
            idx_i = idx_j - k
        m = len(idx_i)
        if m == 0: continue
        # Create y variables for this shift (binary)
        y = cp.Variable(m, boolean=True)
        # Link y with x
        constraints += [y <= x[idx_i], y <= x[idx_j],
                        y >= x[idx_i] + x[idx_j] - 1]
        # Overlap constraint
        constraints += [cp.sum(y) >= (n//2) - t]
    prob = cp.Problem(cp.Minimize(t), constraints)

import numpy as np
import time
import math

def _compute_c5(h):
    """Return the Erdős overlap constant for a binary vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _binary_start(n, init=None):
    """Return a binary (0/1) vector of length n with exactly n/2 ones."""
    if init is not None:
        arr = np.asarray(init, dtype=float)
        if arr.shape[0] != n:
            arr = None
    else:
        arr = None
    h = np.zeros(n, dtype=float)
    if arr is not None:
        # keep the n/2 largest entries as ones
        idx = np.argsort(-arr)[: n // 2]
        h[idx] = 1.0
    else:
        h[: n // 2] = 1.0
        np.random.shuffle(h)
    return h

def _hill_climb(initial_h, time_limit):
    """Simple simulated‑annealing hill‑climb on binary vectors."""
    n = initial_h.shape[0]
    best_h = initial_h.copy()
    best_c5 = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    start = time.time()
    stagnant = 0
    max_stagnant = 2000
    T0 = 5e-3                           # initial temperature

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # pick a random 1–0 swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0

        new_c5 = _compute_c5(new_h)

        # temperature decays linearly
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        elif T > 0.0 and np.random.rand() < math.exp(-(new_c5 - cur_c5) / T):
            accept = True

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            stagnant = 0
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            stagnant += 1

        # restart if stuck too long
        if stagnant >= max_stagnant:
            cur_h = np.zeros(n, dtype=float)
            cur_h[: n // 2] = 1.0
            np.random.shuffle(cur_h)
            cur_c5 = _compute_c5(cur_h)
            stagnant = 0

    return best_h, best_c5

        h0 = _binary_start(n, init_vec)
        h_opt, c5_opt = _hill_climb(h0, per_n)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

    # final polish on the best size with any remaining time
    time_left = budget_s * 0.98 - (time.time() - overall_start)
    if time_left > 1.0 and best_h is not None:
        h_opt, c5_opt = _hill_climb(best_h, time_left)
        best_h, best_c5 = h_opt, c5_opt

    # ensure the constraints are exactly satisfied (numeric safety)
    best_h = np.where(best_h > 0.5, 1.0, 0.0)
    # return the solution
    return (best_h, float(best_c5), int(best_n))
