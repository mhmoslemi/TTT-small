def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math

    # ------------------------------------------------------------
    # Setup reproducibility
    # ------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------
    # Determine problem size and initial feasible vector
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
    else:
        n = 90
        h0 = np.random.rand(n)

    target_sum = n / 2.0  # because integral over [0,2] equals 1

    # ------------------------------------------------------------
    # Repair routine: enforce 0≤h≤1 and sum constraint
    # ------------------------------------------------------------
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0.0:
            h = np.full_like(h, target_sum / len(h))
        else:
            h = h * (target_sum / s)
            h = np.clip(h, 0.0, 1.0)
            # iterative fine‑tuning of the sum
            for _ in range(5):
                s = h.sum()
                diff = target_sum - s
                if abs(diff) < 1e-12:
                    break
                free = (h > 0.0) & (h < 1.0)
                if free.any():
                    delta = diff / free.sum()
                    h[free] += delta
                else:
                    i = np.random.randint(0, len(h))
                    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
                h = np.clip(h, 0.0, 1.0)
        return h

    h0 = repair(h0)

    # ------------------------------------------------------------
    # Objective: compute C5 for a given h
    # ------------------------------------------------------------
    def compute_c5(h):
        try:
            # pre‑imported helper (if available)
            return float(evaluate_erdos_solution(h))
        except Exception:
            dx = 2.0 / len(h)
            conv = np.correlate(h, 1.0 - h, mode='full')
            return np.max(conv) * dx

    # ------------------------------------------------------------
    # Neighbor generation
    # ------------------------------------------------------------
    def generate_neighbor(h):
        h_new = h.copy()
        nloc = len(h_new)
        if random.random() < 0.7:
            # small Gaussian tweaks on a few entries
            num_mod = random.randint(1, 3)
            idx = np.random.choice(nloc, size=num_mod, replace=False)
            sigma = 0.2
            h_new[idx] += np.random.randn(num_mod) * sigma
        else:
            # simple swap (helps binary‑like vectors)
            i, j = np.random.choice(nloc, size=2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        return h_new

    # ------------------------------------------------------------
    # Simulated annealing main loop
    # ------------------------------------------------------------
    current = h0.copy()
    current_score = compute_c5(current)
    best = current.copy()
    best_score = current_score

    start = time.time()
    elapsed = 0.0

    temp = 0.1          # initial temperature
    cooling = 0.9999    # exponential cooling factor

    # loop until time budget is exhausted or target reached
    while elapsed < budget_s:
        neighbor = generate_neighbor(current)
        neighbor = repair(neighbor)
        neigh_score = compute_c5(neighbor)

        # record best ever
        if neigh_score < best_score:
            best = neighbor.copy()
            best_score = neigh_score

        # Metropolis acceptance for the current state
        delta = neigh_score - current_score
        if delta < 0.0 or random.random() < math.exp(-delta / max(temp, 1e-12)):
            current = neighbor
            current_score = neigh_score

        # temperature schedule
        temp *= cooling

        # time check
        elapsed = time.time() - start

        # early termination if we already beat the target bound
        if best_score < 0.3808:
            break

    # final safety repair and evaluation
    best = repair(best)
    final_score = compute_c5(best)

    # return as required: list of values, bound, number of points
    return best.tolist(), float(final_score), n
