def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start_time = time.time()
    rng = np.random.default_rng(seed)
    # define n
    n = 200
    # initial shape: triangular
    x = (np.arange(n) + 0.5) * (2.0 / n)  # midpoints
    h = np.where(x <= 1.0, x, 2.0 - x)
    # scale sum to S = n/2 via projection onto simplex with bounds
    S = n / 2.0
    # implement projection function
    def project(v):
        # binary search lambda
        lo = -np.max(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > S:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    h = project(h)

    # compute initial c5
    dx = 2.0 / n
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # maybe incorporate initial_h_values if available for different n?
    # But we may try also to use it as a candidate: if 'initial_h_values' defined:

        # ensure length matches n? If not, resample to n via interpolation.
        if init_h.shape[0] != n:

        init_c5 = compute_c5(init_h)
        if init_c5 < best_c5:

        # also use as starting point for optimization
        h = init_h.copy()
        current_c5 = init_c5
    else:
        current_c5 = best_c5

    # parameters
    alpha = 0.5  # step size scaling
    # schedule for step size decrease
    iter_count = 0
    # main loop, limited by budget time
    while time.time() - start_time < budget_s * 0.95:

        corr_full = np.correlate(h, 1.0 - h, mode='full')

        if shift >= 0:

        g[i] += 1.0 - h[j]

        step = alpha * g

        h_new = project(h_new)
        new_c5 = compute_c5(h_new)

        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5

            if rng.random() < 0.05:

                i1, i2 = i_rand[0], i_rand[1]
                if i1 != i2:

                    if delta_max > 1e-8:

                        current_c5 = compute_c5(h)
                        if current_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = current_c5

                    best_c5 = cand_c5

import time, numpy as np

def project_onto_bounds_and_sum(v, S):
    # binary search lam to satisfy sum clamp
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # adjust tiny residual if needed
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff proportionally to non-saturated entries
        mask = (x > 0) & (x < 1)
        if mask.any():
            # equally distribute
            x[mask] += diff / mask.sum()
            # Clip again
            x = np.clip(x, 0.0, 1.0)
    # final enforce sum exactly: may still have tiny diff but okay.
    return x

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
    else:
        d = -shift
        i = np.arange(d, n)
        j = i - d
    g[i] += 1.0 - h[j]
    g[j] -= h[i]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()

    # number of sample points (adjustable, <1000)
    n = 200
    target_sum = n / 2.0
    dx = 2.0 / n

    # ---------- helper functions ----------
    def project(v):
        """Project v onto the set {0<=x<=1, sum(x)=target_sum}."""
        lo = -np.max(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
        # tiny correction if needed
        diff = target_sum - proj.sum()
        if abs(diff) > 1e-12:
            mask = (proj > 0) & (proj < 1)
            if mask.any():
                proj[mask] += diff / mask.sum()
                proj = np.clip(proj, 0.0, 1.0)
        return proj

    def compute_c5(h):
        """C5 bound for a given h (uses the same definition as the evaluator)."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * dx
