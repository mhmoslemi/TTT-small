def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max() * dx)

def mismatched_indices(h, s):
    N = len(h)
    if s >= 0:
        idx = np.arange(N - s)
        i = idx[(h[idx] == 1) & (h[idx + s] == 0)]

import time, math
import numpy as np

def _compute_c5(h):
    """Return C5 bound for binary step h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement
    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max() * dx), int(c.max())

def _mismatch_indices(h, shift):
    """Indices i where h[i]==1 & h[i+shift]==0 (i+shift in range)
    and indices j where h[j]==0 & h[j+shift]==1."""
    N = h.shape[0]
    if shift >= 0:
        rng = np.arange(N - shift)
        i = rng[(h[rng] == 1) & (h[rng + shift] == 0)]
        j = rng[(h[rng] == 0) & (h[rng + shift] == 1)]
    else:
        d = -shift
        rng = np.arange(d, N)
        i = rng[(h[rng] == 1) & (h[rng - d] == 0)]
        j = rng[(h[rng] == 0) & (h[rng - d] == 1)]
    return i, j
