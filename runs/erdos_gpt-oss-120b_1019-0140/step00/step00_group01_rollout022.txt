import time
import numpy as np
import cvxpy as cp

def _compute_c5(h_vals: np.ndarray) -> float:
    """Calculate the C5 value for a given discretized h."""
    n = h_vals.size
    dx = 2.0 / n
    corr = np.correlate(h_vals, 1.0 - h_vals, mode='full')
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Find a step function h on [0,2] (discretised) that minimizes the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # ---- Choose number of points (even for binary variables) ----
    n = 48                     # 48 points → dx = 2/48, half‑ones constraint is integral
    target_sum = n // 2        # exact number of 1's for a binary solution

    # ---- Helper to keep time budget ----
    def _time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    best_h = None
    best_c5 = np.inf

    # --------------------------------------------------------------
    # 1. Try a Mixed‑Integer Linear Program (binary formulation)
    # --------------------------------------------------------------
    if _time_left() > 5.0:  # give the MILP a small window
        # Variables
        h = cp.Variable(n, integer=True)               # binary 0/1
        y = cp.Variable((n, n), nonneg=True)           # auxiliary linearisation
        t = cp.Variable(nonneg=True)                    # max overlap count

        constraints = []

        # h binary constraints
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == target_sum)

        # y diagonal zero
        for i in range(n):
            constraints.append(y[i, i] == 0)

        # Linking constraints for all i != j
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                constraints.append(y[i, j] <= h[i])
                constraints.append(y[i, j] <= 1 - h[j])
                constraints.append(y[i, j] >= h[i] - h[j])

        # Max‑overlap constraints for each lag L (both signs)
        for L in range(-n + 1, n):
            if L == 0:
                continue
            expr = cp.sum([y[i, i + L] for i in range(n) if 0 <= i + L < n])
            constraints.append(expr <= t)

        # Problem
        prob = cp.Problem(cp.Minimize(t), constraints)

        # Solver attempts (CBC preferred, fall back to ECOS_BB / GLPK_MI)
        solved = False
        try:
            prob.solve(solver=cp.CBC, maxSeconds=_time_left())
            solved = True
        except Exception:
            try:
                prob.solve(solver=cp.ECOS_BB, max_iters=50000)
                solved = True
            except Exception:
                try:
                    prob.solve(solver=cp.GLPK_MI)
                    solved = True
                except Exception:
                    solved = False

        if solved and h.value is not None:
            h_raw = np.array(h.value).flatten()
            h_bin = np.rint(np.clip(h_raw, 0, 1)).astype(int)

            # Adjust to exact sum if rounding drifted
            diff = target_sum - h_bin.sum()
            if diff != 0:
                # deviation from integer guess
                dev = np.abs(h_raw - np.round(h_raw))
                order = np.argsort(dev)
                for idx in order:
                    if diff == 0:
                        break
                    if diff > 0 and h_bin[idx] == 0:
                        h_bin[idx] = 1
                        diff -= 1
                    elif diff < 0 and h_bin[idx] == 1:
                        h_bin[idx] = 0
                        diff += 1

            # Evaluate C5
            c5_val = _compute_c5(h_bin)
            best_h = h_bin
            best_c5 = c5_val

    # --------------------------------------------------------------
    # 2. If MILP failed or did not meet target, run a simple local search
    # --------------------------------------------------------------
    if best_h is None or best_c5 > 0.3809:
        # Starting point: use initial_h_values if compatible, else random binary
        try:
            init = np.array(initial_h_values, dtype=float)
            if init.size == n:
                # Round & fix sum to obtain a feasible binary start
                start_bin = np.rint(np.clip(init, 0, 1)).astype(int)
                cur_sum = start_bin.sum()
                if cur_sum != target_sum:
                    # Randomly flip bits to reach exact sum
                    diff = target_sum - cur_sum
                    zeros = np.where(start_bin == 0)[0]
                    ones = np.where(start_bin == 1)[0]
                    if diff > 0:
                        flip = np.random.choice(zeros, diff, replace=False)
                        start_bin[flip] = 1
                    elif diff < 0:
                        flip = np.random.choice(ones, -diff, replace=False)
                        start_bin[flip] = 0
                h_current = start_bin
            else:
                raise ValueError
        except Exception:
            # Random binary vector with exact half ones
            h_current = np.zeros(n, dtype=int)
            ones = np.random.choice(n, target_sum, replace=False)
            h_current[ones] = 1

        best_h = h_current.copy()
