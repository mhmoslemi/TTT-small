def max_correlation(h):
    n = len(h)
    dx = 2.0 / n
    a = h
    b = 1 - h
    # compute convolution in full length (2n-1)
    # correlation = convolution of a and reversed b.
    # reversed b: b_rev = b[::-1]
    # use FFT with size = pow2 >= 2n
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)[:2*n-1]
    return np.max(conv) * dx
```

    v = np.clip(v, 0, 1)

def local_search(h, time_limit):
    n = len(h)
    start = time.time()
    cur_h = h.copy()
    cur_val = evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # temperature parameters
    T0 = 0.01  # starting temperature
    while time.time() - start < time_limit:
        # random pair
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # random direction
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 1e-12:
            continue
        # step size: fraction up to 0.5 of max_delta
        delta = max_delta * np.random.rand() * 0.5
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        # evaluate
        new_val = evaluate_c5(new_h)
        # accept if better or with probability
        if new_val < cur_val:
            accept = True
        else:
            t = T0 * (1 - (time.time() - start) / time_limit)
            if t > 0 and np.exp(-(new_val - cur_val) / t) > np.random.rand():
                accept = True
            else:
                accept = False
        if accept:
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
    return best_h, best_val

import numpy as np
import time

def project_to_bounds_and_sum(v, target):
    """Project vector v onto the box [0,1]^n intersected with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    # If already satisfied, return.
    cur = np.sum(v)
    if abs(cur - target) <= 1e-12:
        return v
    # Binary search for lambda such that sum(clip(v - lam)) = target.
    lo, hi = -1.0, 2.0  # wide enough to cover all possibilities
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def upscale(h, n):
    """Linear interpolation of vector h (length m) to length n on [0,2)."""
    m = len(h)
    if n == m:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, h)

def evaluate_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k)) dx for discretised h."""
    n = len(h)
    dx = 2.0 / n
    # Use FFT to compute full correlation efficiently.
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, n=size)
    B = np.fft.rfft((1.0 - h)[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)[:2 * n - 1]
    return float(np.max(conv) * dx)

def local_search(h, time_limit):
    """Hill‑climbing with occasional annealed acceptance."""
    n = len(h)
    start = time.time()
    cur_h = h.copy()
    cur_val = evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    T0 = 0.01  # initial temperature for occasional uphill moves

    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Randomly decide direction of transfer.
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 1e-12:
            continue
        # Choose a step up to 90 % of the feasible amount.
        delta = max_delta * np.random.rand() * 0.9
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        # Evaluate new candidate.
        new_val = evaluate_c5(new_h)
        if new_val < cur_val:
            accept = True
        else:
            # Simulated‑annealing acceptance.
            elapsed = time.time() - start
            t = T0 * (1.0 - elapsed / time_limit)
            accept = (t > 0.0) and (np.exp(-(new_val - cur_val) / t) > np.random.rand())
        if accept:
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    np.random.seed(seed)
    start_time = time.time()

    # Candidate discretisations (all below 1000 points).
    candidate_ns = [45, 90, 180, 360, 720]
    candidate_ns = [n for n in candidate_ns if n <= 1000]

    # Obtain a starting construction.
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n0 = len(init_h)
        init_h = project_to_bounds_and_sum(init_h, n0 / 2.0)
    else:
        # Simple uniform fallback.
        n0 = candidate_ns[0]
        init_h = np.full(n0, 0.5, dtype=np.float64)
        init_h = project_to_bounds_and_sum(init_h, n0 / 2.0)

    best_h = init_h.copy()
    best_val = evaluate_c5(best_h)

    # Progressive refinement across resolutions.
    for idx, n in enumerate(candidate_ns):
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 0.0:
            break
        # Allocate roughly equal remaining time among outstanding resolutions.
        alloc = remaining / (len(candidate_ns) - idx)

        # Prepare initial vector for this resolution.
        if n == len(best_h):
            h0 = best_h.copy()
        else:
            h0 = upscale(best_h, n)

        h0 = project_to_bounds_and_sum(h0, n / 2.0)

        # Run local optimisation (spend most of the allocated slice).
        h_opt, val_opt = local_search(h0, alloc * 0.9)

        if val_opt < best_val:
            best_h = h_opt
            best_val = val_opt

    # Compute final bound (use provided helper if present).
    try:

    except Exception:
        c5_bound = evaluate_c5(best_h)
