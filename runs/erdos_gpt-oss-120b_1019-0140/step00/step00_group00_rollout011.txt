def projected_gradient_descent(h, rng, time_end, step=0.01):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() < time_end:
        # compute full correlation
        conv = np.correlate(h, 1 - h, mode='full')
        # find lag index of max
        k_idx = np.argmax(conv)
        # compute relative shift k: we need zero shift corresponds to index n-1 (since full length = 2n-1)
        shift = k_idx - (n - 1)
        # compute gradient as described
        grad = np.ones_like(h) * dx  # derivative from sum_i h_i term * dx
        # subtract contributions from -h_i * h_{i+shift}
        # derivative: -dx*(h_{i+shift} + h_{i-shift})
        # but careful: gradient of - sum_i h_i * h_{i+shift} w.r.t h_j: - (h_{j+shift}) - (h_{j-shift})
        # So update gradient: grad -= dx * (h shifted forward + h shifted backward)
        # shift positive: forward shift
        # we treat out-of-bounds as 0
        if shift >= 0:
            # h_{j+shift} defined for j+shift < n
            grad[:n-shift] -= dx * h[shift:]
            # h_{j-shift} defined for j >= shift
            grad[shift:] -= dx * h[:n-shift]
        else:
            s = -shift
            # shift negative: h_{j+shift} = h_{j - s}
            grad[s:] -= dx * h[:n-s]
            # h_{j-shift} = h_{j + s}
            grad[:n-s] -= dx * h[s:]
        # take a step: h_new = h - step * grad
        # Clip to [0,1]
        h_new = h - step * grad
        np.clip(h_new, 0, 1, out=h_new)
        # adjust sum to target sum via projection onto hyperplane with box constraints (project_to_bounds_and_sum)
        target_sum = n/2
        h_new = project_to_bounds_and_sum(h_new, target_sum)
        # compute new C5
        c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        # If improvement, update best
        if c5_new < best_c5:
            best_c5 = c5_new

                h = h_new
        # adjust step size adaptively maybe decrease
        # after some iterations, reduce step
        # not necessary

    target_sum = n/2
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    step = 0.5

def optimize_at_resolution(n, rng, time_end, start_time, init_h=None):
    target_sum = n/2
    best_h = None
    best_c5 = np.inf
    # allocate time for multiple restarts
    while time.time() < time_end - 0.5:

        if init_h is not None and rng.random() < 0.5:

            h0 = rng.random(n)

def compute_c5(h):

    dx = 2.0 / n

def project_to_bounds_and_sum(y, target):
    # Euclidean projection of y onto box [0,1] and hyperplane sum=y_target
    # Use bisection on lambda such that x = clip(y - lam, 0, 1) has sum target
    lo = -1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
