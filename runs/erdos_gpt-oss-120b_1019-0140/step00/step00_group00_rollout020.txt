            if total_cap < 1e-12:
                break
            # distribute proportionally to capacity
            inc = diff * cap / total_cap
            h = h + inc
            # clip again
            h = np.clip(h, 0.0, 1.0)
        else:

            cap = h
            total_cap = np.sum(cap)
            if total_cap < 1e-12:
                break
            dec = (-diff) * cap / total_cap
            h = h - dec
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        attempts += 1
    return h
