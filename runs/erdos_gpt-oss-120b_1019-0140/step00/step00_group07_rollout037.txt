def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0
    x = (np.arange(n) + 0.5) * dx
    P = 7  # using harmonics 1..7
    freqs = np.arange(1, P+1).reshape(-1,1)  # (P,1)
    cos_mat = np.cos(freqs * np.pi * x)  # shape (P,n)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project_to_sum(v):
        # v is assumed within [0,1]; we project onto hyperplane sum=v_target
        lo, hi = -1.0, 1.0
        for _ in range(50):
            lam = (lo + hi) / 2.0
            w = np.clip(v + lam, 0.0, 1.0)
            s = w.sum()
            if s < target_sum:
                lo = lam
            else:
                hi = lam
        w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
        return w

    a_best = np.zeros(P)

import time
import numpy as np

def project_to_sum(v, target):
    """Project vector v onto the box [0,1] while enforcing sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h, dx):
    """Return the overlap constant for a given step function h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def initial_candidates(P):
    """A few hand‑crafted coefficient vectors respecting the L1≤0.5 budget."""
    cand = []

    a = np.zeros(P)
    a[0] = -0.5                      # pure sin²
    cand.append(a)

    a = np.zeros(P)
    a[0] = -0.2
    if P >= 3:
        a[2] = 0.15                 # cos 3πx
    if P >= 5:
        a[4] = 0.15                 # cos 5πx
    cand.append(a)

    a = np.zeros(P)
    a[0] = -0.25
    if P >= 3:
        a[2] = 0.125
    if P >= 5:
        a[4] = 0.125
    cand.append(a)

    a = np.zeros(P)
    a[0] = -0.2
    if P >= 3:
        a[2] = 0.1
    if P >= 5:
        a[4] = 0.1
    if P >= 7:
        a[6] = 0.1
    cand.append(a)

    return cand


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # discretisation
    n = 200                     # number of samples (feel free to adjust)
    dx = 2.0 / n
    target_sum = n / 2.0

    x = (np.arange(n) + 0.5) * dx                # mid‑point grid
    P = 7                                        # number of cosine harmonics
    freqs = np.arange(1, P + 1).reshape(-1, 1)   # (P,1)
    cos_mat = np.cos(freqs * np.pi * x)          # (P,n)

    # ----------- start from a few deterministic candidates -----------------
    best_score = np.inf
    best_h = None
    best_a = None

    for a in initial_candidates(P):
        h = 0.5 + a @ cos_mat
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_score:
            best_score, best_h, best_a = c5, h.copy(), a.copy()

    start = time.time()
    # -------------------------- random search -----------------------------
    while time.time() - start < budget_s * 0.7:
        # total L1 budget (≤0.5)
        total_amp = rng.random() * 0.5

        # enforce a negative contribution on the first harmonic (most important)
        a1_mag = rng.random() * total_amp
        a1 = -a1_mag
        remaining = total_amp - a1_mag

        a = np.zeros(P)
        a[0] = a1

        if remaining > 0 and P > 1:
            mags = rng.random(P - 1)
            mags = mags / mags.sum() * remaining
            signs = rng.choice([-1, 1], size=P - 1)
            a[1:] = signs * mags

        h = 0.5 + a @ cos_mat
        # quick numeric safety check (the L1 bound guarantees feasibility, but guard against round‑off)
        if h.min() < -1e-12 or h.max() > 1 + 1e-12:
            continue

        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h, dx)

        if c5 < best_score:
            best_score, best_h, best_a = c5, h.copy(), a.copy()
            if best_score < 0.3808:
                break

    # ----------------------- local refinement ----------------------------
    a_curr = best_a.copy()
    h_curr = best_h.copy()
    score_curr = best_score
    step = 0.1

    while time.time() - start < budget_s * 0.95:
        delta = rng.normal(scale=step, size=P)
        a_new = a_curr + delta
        # enforce the L1 budget again
        l1 = np.abs(a_new).sum()
        if l1 > 0.5:
            a_new = a_new * (0.5 / l1)

        h_new = 0.5 + a_new @ cos_mat
        if h_new.min() < -1e-12 or h_new.max() > 1 + 1e-12:
            step *= 0.9999
            continue

        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)

        if c5_new < score_curr:
            a_curr, h_curr, score_curr = a_new, h_new, c5_new
            if c5_new < best_score:
                best_score, best_h, best_a = c5_new, h_new.copy(), a_new.copy()
                if best_score < 0.3808:
                    break

        step *= 0.9999
        if step < 1e-6:
            step = 1e-6

    # final result
    return (best_h.tolist(), float(best_score), n)
