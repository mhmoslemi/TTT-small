```
def propose(h, step):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # compute max feasible delta
    if step <= 0:
        return None
    max_pos = min(step, 1 - h[i], h[j])
    max_neg = -min(step, h[i], 1 - h[j])
    if max_pos <= 0 and max_neg >= 0:
        return None
    delta = np.random.uniform(max_neg, max_pos)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    return new
```
