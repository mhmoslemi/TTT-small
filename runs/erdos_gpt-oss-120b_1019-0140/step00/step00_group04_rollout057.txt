import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    return np.max(corr) * dx, corr

def worst_shift_info(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)  # shift can be negative
    # compute contributions per i
    if shift >= 0:
        i_range = range(0, n - shift)
        j_offset = shift
    else:
        i_range = range(-shift, n)
        j_offset = shift
    contributions = np.zeros(n)
    for i in i_range:
        j = i + shift
        contributions[i] = h[i] * (1 - h[j])
    # contributions outside range remain 0.
    i_from = np.argmax(contributions)
    i_to = np.argmin(contributions)
    return shift, i_from, i_to, contributions, corr

def repair_h(h):
    # Clip to [0,1] and adjust sum exactly
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    # adjust first non boundary element
    for i in range(n):
        if diff > 0 and h[i] < 1.0:
            incr = min(diff, 1.0 - h[i])
            h[i] += incr
            diff -= incr
        elif diff < 0 and h[i] > 0.0:
            decr = min(-diff, h[i])
            h[i] -= decr
            diff += decr
        if abs(diff) < 1e-12:
            break
    return h

    best_h = cur_h.copy()

        max_idx = np.argmax(cur_corr)

        # compute region contributions
        if shift >= 0:
            region_len = n - shift
            h_i = cur_h[:region_len]

def repair_h(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    total = np.sum(h)
    diff = target - total
    if abs(diff) > 1e-12:
        # distribute diff to entries not at bounds
        # find indices where h < 1 if diff>0 else h>0
        if diff > 0:
            idx = np.where(h < 1)[0]
            # add diff equally among idx up to bound
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:

import numpy as np
import time

def compute_c5(h):
    """Return (C5 value, correlation array) for step vector h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return np.max(corr) * dx, corr

def repair_h(h):
    """Clip to [0,1] and enforce sum(h)=n/2 exactly (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    total = np.sum(h)
    diff = target - total
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def local_search(start_h, time_limit, rng):
    """Simulated-annealing style local search on fixed dimension."""
    start = time.time()
    n = len(start_h)
    cur_h = start_h.copy()
    cur_cost, cur_corr = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_cost = cur_cost

    # temperature schedule (linear decay)
    start_temp = max(1e-4, cur_cost * 0.1)

    iters_without_improve = 0
    iteration = 0

    while time.time() - start < time_limit:
        iteration += 1
        # progressive temperature
        elapsed = time.time() - start
        temp = start_temp * (1.0 - elapsed / time_limit)

        # identify worst shift using current correlation
        max_idx = int(np.argmax(cur_corr))
        shift = max_idx - (n - 1)          # shift ∈ [-(n-1), n-1]

        # region where the shift pairs indices
        if shift >= 0:
            region_len = n - shift
            region_start = 0
            h_i = cur_h[:region_len]
            h_j = cur_h[shift:shift + region_len]
        else:
            s = -shift
            region_len = n - s
            region_start = s
            h_i = cur_h[s:s + region_len]
            h_j = cur_h[:region_len]

        # contribution of each i to the worst shift
        contrib = h_i * (1.0 - h_j)
        i_from_local = int(np.argmax(contrib))
        i_from = region_start + i_from_local

        # choose i_to where partner value h_j is largest (so (1-h_j) small)
        i_to_local = int(np.argmax(h_j))
        i_to = region_start + i_to_local

        # avoid degenerate move
        if i_from == i_to:
            # pick second best in h_j
            sorted_idx = np.argsort(-h_j)  # descending
            for idx in sorted_idx:
                if idx != i_from_local:
                    i_to_local = int(idx)
                    i_to = region_start + i_to_local
                    break

        max_reduce = cur_h[i_from]
        max_increase = 1.0 - cur_h[i_to]

        if max_reduce > 1e-12 and max_increase > 1e-12:
            # move a fraction of the feasible amount
            delta = min(max_reduce, max_increase) * (0.05 + 0.45 * rng.random())
            new_h = cur_h.copy()
            new_h[i_from] -= delta
            new_h[i_to] += delta
        else:
            # fallback: random feasible pair
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_reduce = cur_h[i]
            max_increase = 1.0 - cur_h[j]
            if max_reduce <= 1e-12 or max_increase <= 1e-12:
                continue
            delta = min(max_reduce, max_increase) * (0.05 + 0.45 * rng.random())
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_cost, new_corr = compute_c5(new_h)

        # acceptance rule (simulated annealing)
        if new_cost < cur_cost or rng.random() < np.exp(-(new_cost - cur_cost) / max(temp, 1e-12)):
            cur_h = new_h
            cur_cost = new_cost
            cur_corr = new_corr
            if new_cost < best_cost - 1e-12:
                best_h = new_h.copy()
                best_cost = new_cost
                iters_without_improve = 0
            else:
                iters_without_improve += 1
        else:
            iters_without_improve += 1

        # occasional perturbation to escape stagnation
        if iters_without_improve >= 2000:
            perturb = rng.normal(scale=0.01, size=n)
            cur_h = best_h + perturb
            cur_h = repair_h(cur_h)
            cur_cost, cur_corr = compute_c5(cur_h)
            iters_without_improve = 0

    return best_h, best_cost

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # obtain initial vector (use provided parent if available)
    try:

    # final repair and evaluation

    final_cost, _ = compute_c5(best_h)
    n_points = len(best_h)

    return best_h.tolist(), float(final_cost), n_points
