def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)
    # reverse b for convolution
    b_rev = b[::-1]
    size = 1 << ((2*n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(A * B, n=size)[:2*n-1]
    return conv.max() * dx

def anneal(h_initial, time_budget, rng):
    h = h_initial.copy()
    n = len(h)
    dx = 2.0 / n
    size = 1 << ((2*n - 1).bit_length())
    # initial compute
    current_c5 = compute_c5(h, size, dx)
    best_c5 = current_c5
    best_h = h.copy()
    # maintain indices
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    # Create list but we need to modify.
    # We'll treat ones and zeros as ndarray, and we will update via assignment.
    start = time.time()
    # Temperature schedule constants
    T_initial = 0.5  # adjust
    T_final = 1e-4
    it = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        # progress ratio
        progress = elapsed / time_budget
        T = T_initial * (1 - progress) + T_final * progress
        # random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        idx_one = rng.integers(len(ones))
        i_one = ones[idx_one]
        idx_zero = rng.integers(len(zeros))
        i_zero = zeros[idx_zero]
        # perform swap
        h[i_one] = 0
        h[i_zero] = 1
        # update ones and zeros arrays in place
        ones[idx_one] = i_zero
        zeros[idx_zero] = i_one
        # compute new c5
        new_c5 = compute_c5(h, size, dx)
        delta = new_c5 - current_c5
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            # accept
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i_one] = 1
            h[i_zero] = 0
            ones[idx_one] = i_one
            zeros[idx_zero] = i_zero
        it += 1
    return best_h, best_c5

import numpy as np
import math

def _c5_of_h(h, size, dx):
    """Maximum overlap integral for a binary (or [0,1]) vector h."""
    a = h.astype(np.float64)
    b = (1.0 - a)[::-1]
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)[:2 * len(h) - 1]
    return float(conv.max()) * dx

def _fix_binary_sum(h, target_ones, rng):
    """Force a binary vector to have exactly target_ones ones."""
    cur = int(h.sum())
    if cur == target_ones:
        return h
    diff = target_ones - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        chosen = rng.choice(zeros, diff, replace=False)
        h[chosen] = 1
    else:
        diff = -diff
        ones = np.flatnonzero(h == 1)
        chosen = rng.choice(ones, diff, replace=False)
        h[chosen] = 0
    return h

def _anneal(h0, time_budget, rng):
    """Simple simulated‑annealing on a binary vector with fixed sum."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    size = 1 << ((2 * n - 1).bit_length())
    cur_c5 = _c5_of_h(h, size, dx)
    best_c5 = cur_c5
    best_h = h.copy()

    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)

    t0 = time.time()
    T_start = 0.8    # start temperature
    T_end = 1e-4     # final temperature

    while True:
        now = time.time()
        if now - t0 >= time_budget:
            break
        # linear cooling
        prog = (now - t0) / time_budget
        T = T_start * (1 - prog) + T_end * prog

        # pick a random 1‑index and a random 0‑index and swap
        i_one = ones[rng.integers(len(ones))]
        i_zero = zeros[rng.integers(len(zeros))]

        # perform swap
        h[i_one] = 0
        h[i_zero] = 1
        # update index lists in place
        idx_one = np.where(ones == i_one)[0][0]
        idx_zero = np.where(zeros == i_zero)[0][0]
        ones[idx_one] = i_zero
        zeros[idx_zero] = i_one

        new_c5 = _c5_of_h(h, size, dx)
        delta = new_c5 - cur_c5
        if delta <= 0.0 or rng.random() < math.exp(-delta / T):

            h[i_one] = 1
            h[i_zero] = 0
            ones[idx_one] = i_one
            zeros[idx_zero] = i_zero

            src_n = len(src)
            xs = np.linspace(0, 2, src_n, endpoint=False)
            xt = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(xt, xs, src)
            h0 = (interp >= 0.5).astype(np.int8)
        else:
            h0 = np.zeros(n, dtype=np.int8)
            idx = rng.choice(n, n // 2, replace=False)
            h0[idx] = 1

        # enforce exact half‑ones
        h0 = _fix_binary_sum(h0, n // 2, rng)

        # run annealing
        h_opt, c5_opt = _anneal(h0, alloc, rng)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # early exit if sufficiently good
        if best_c5 <= 0.3808:
            break

    # safety fallback
    if best_h is None:
        n = 128
        best_h = np.zeros(n, dtype=np.int8)
        best_h[: n // 2] = 1
        best_n = n
        size = 1 << ((2 * n - 1).bit_length())
        best_c5 = _c5_of_h(best_h, size, 2.0 / n)

    return (best_h.tolist(), float(best_c5), int(best_n))
