import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a binary (or [0,1]) step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h) using zero padding
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def improve_h(h: np.ndarray, rng: np.random.Generator, time_limit: float):
    """
    Greedy/annealing local search that repeatedly swaps a 1 with a 0.
    The swap is chosen from the positions that participate in the current
    worst (largest) overlap, aiming to reduce that value.
    """
    n = h.shape[0]
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # small probability to accept a worsening move (helps escape local minima)
    prob_worsen = 0.001

    while time.time() - start < time_limit:
        # correlation for the current configuration
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)          # shift corresponding to the max value

        # indices that contribute to this shift
        if shift >= 0:
            idx = np.arange(0, n - shift, dtype=int)
            hi = cur_h[idx]
            hj = cur_h[idx + shift]
        else:
            s = -shift
            idx = np.arange(s, n, dtype=int)
            hi = cur_h[idx]
            hj = cur_h[idx - s]

        # positions where a 1 is paired with a 0 (a mismatch that inflates the overlap)
        mism = idx[(hi == 1) & (hj == 0)]

        if mism.size == 0:
            # no obvious mismatch; perform a blind opposite-value swap
            i = rng.integers(n)
            for _ in range(10):
                j = rng.integers(n)
                if cur_h[i] != cur_h[j]:
                    break
            else:
                continue
        else:
            i = int(rng.choice(mism))
            zeros = np.where(cur_h == 0)[0]
            if zeros.size == 0:
                break
            j = int(rng.choice(zeros))

        # propose the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)

        if new_c5 < cur_c5 or (new_c5 > cur_c5 and rng.random() < prob_worsen):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert the change
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_c5

    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Candidate sample sizes (must be even to satisfy the area constraint)
    # ------------------------------------------------------------------
    candidate_ns = [80, 100, 120, 150, 180, 200, 240, 300]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    # ------------------------------------------------------------------
    # Primary phase: devote the bulk of the budget to improving the
    # initial construction (if it exists).  Otherwise improve a random
    # length‑80 start.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30:
        # spend up to 80 % of the remaining time on this phase
        alloc = min(0.8 * remaining, remaining - 5.0)
        best_h, best_c5 = improve_h(best_h, rng, alloc)
        remaining = budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Secondary phase: many short independent searches on various lengths
    # ------------------------------------------------------------------
    while remaining > 5.0:
        n = int(rng.choice(candidate_ns))
        # generate a fresh binary vector with exactly n/2 ones
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

        # each trial gets a modest time slice (up to 5 s or half of what is left)
        slice_time = min(5.0, remaining * 0.5)
        h_opt, c5_opt = improve_h(h0, rng, slice_time)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        remaining = budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    n_points = int(best_h.shape[0])
    # Ensure the area constraint (tiny floating‑point adjustments are harmless)
    # The construction is binary, so the constraint holds exactly.
    return best_h.astype(float), float(best_c5), n_points
