    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        cur_h = init.copy()
    except NameError:
        n = 200
        cur_h = np.full(n, 0.5)
    target_sum = n / 2.0
    # Ensure sum exactly target (account for floating errors)
    cur_sum = cur_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust first element
        cur_h[0] += target_sum - cur_sum
    # Clip to [0,1]
    cur_h = np.clip(cur_h, 0.0, 1.0)

    def compute_c5(vec):
        nloc = vec.shape[0]
        conv = np.correlate(vec, 1.0 - vec, mode='full')
        dx = 2.0 / nloc
        return np.max(conv) * dx

    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    max_temp = 0.05
    min_temp = 1e-6
    target_bound = 0.3808

    # Preallocate some variables for speed
    n_pts = n
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # cooling
        progress = elapsed / (budget_s * 0.95)
        temp = max_temp * (1 - progress) + min_temp * progress

        # propose move
        i = rng.integers(n_pts)
        j = rng.integers(n_pts - 1)
        if j >= i:
            j += 1
        # random direction
        if rng.random() < 0.5:
            source, target = i, j
        else:
            source, target = j, i
        max_transfer = min(cur_h[source], 1.0 - cur_h[target])
        if max_transfer <= 0:
            continue
        # transfer amount
        delta = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[source] -= delta
        new_h[target] += delta

        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 <= 0 or (temp > 0 and math.exp(-delta_c5 / temp) > rng.random()):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                if best_c5 <= target_bound:
                    break
        # occasionally reset cur_h to best_h to avoid drift
        if rng.random() < 0.001:
            cur_h = best_h.copy()

    # clip final just in case small violations
    best_h = np.clip(best_h, 0.0, 1.0)

        best_h[0] = min(max(best_h[0], 0.0), 1.0)

        delta = rng.random() * max_transfer
        new = cur.copy()
        new[src] -= delta
        new[trg] += delta
        return new

    def targeted_move(cur, k_max, rng):
        """Bias a move toward improving the worst shift (k_max)."""
        if k_max >= 0:
            i = rng.integers(0, n - k_max)
            i2 = i + k_max
        else:
            km = -k_max
            i = rng.integers(km, n)
            i2 = i - km
        inc_cap = min(1.0 - cur[i], 1.0 - cur[i2])
        if inc_cap <= 0.0:
            return None
        # try a few random source pairs
        for _ in range(10):
            s1 = rng.integers(0, n)
            s2 = rng.integers(0, n)
            if s1 == s2 or s1 in (i, i2) or s2 in (i, i2):
                continue
            dec_cap = min(cur[s1], cur[s2])
            if dec_cap <= 0.0:
                continue
            delta = min(inc_cap, dec_cap) * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] += delta
            new[i2] += delta
            new[s1] -= delta
            new[s2] -= delta
            return new
        return None

    # ----- initial evaluation ------------------------------------------------
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # temperature schedule
    max_temp = 0.05
    min_temp = 1e-6
    target_bound = 0.3808

    # main optimisation loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # recompute current correlation and worst shift
        conv_cur = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = np.max(conv_cur) * dx
        k_max = int(np.argmax(conv_cur) - (n - 1))

        # linear temperature decay
        progress = elapsed / (budget_s * 0.95)
        temp = max_temp * (1.0 - progress) + min_temp * progress

        # decide move type
        if rng.random() < 0.3:
            h_new = targeted_move(h, k_max, rng)
        else:
            h_new = propose_transfer(h, rng)

        if h_new is None:
            continue

        new_c5 = compute_c5(h_new)
        delta_c5 = new_c5 - cur_c5
        accept = False
        if delta_c5 <= 0.0:
            accept = True
        else:
            if temp > 0.0 and math.exp(-delta_c5 / temp) > rng.random():
                accept = True
        if accept:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                if best_c5 <= target_bound:
                    break

        # occasional reset to best to avoid drift
        if rng.random() < 0.001:
            h = best_h.copy()
            cur_c5 = best_c5

    # final sanity adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h[0] += diff
        best_h[0] = min(max(best_h[0], 0.0), 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), n)
