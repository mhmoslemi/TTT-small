def repair_h(h):

    target = n/2.0
    # Clip
    h = np.clip(h, 0, 1)
    # Ensure sum using iterative redistribution
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        # Determine indices that can move in direction of diff
        if diff > 0:
            movable = h < 1
            # capacity = sum(1 - h[idx])
            capacity = np.sum(1 - h[movable])
            if capacity == 0:

            if capacity == 0:

import numpy as np, time, scipy.optimize as opt, scipy.interpolate

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Simple scaling loop
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # need increase
            idx = h < 1
            if not np.any(idx):
                break
            capacity = np.sum(1 - h[idx])
            # distribute uniformly proportionally
            inc = diff * (1 - h[idx]) / capacity
            h[idx] += inc
        else:
            # need decrease
            idx = h > 0
            if not np.any(idx):
                break
            capacity = np.sum(h[idx])
            dec = -diff * h[idx] / capacity
            h[idx] -= dec
        h = np.clip(h, 0, 1)
    # final adjust scalar
    s = h.sum()
    if abs(s - target) > 1e-6:
        # adjust any element not at bound
        idx = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if idx.size > 0:
            i = idx[0]
            h[i] += target - s
            h[i] = np.clip(h[i], 0, 1)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    # grid points from 0 to 2 exclusive
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # use np.interp for linear interpolation
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    # repair (ensure sum)
    h_new = repair_h(h_new)
    return h_new

def try_optimize(h_start, maxiter=200):
    n = len(h_start)
    target = n / 2.0
    # define constraints
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    # bounds
    bounds = [(0.0, 1.0)] * n
    # objective function
    def obj(x):
        return compute_c5(x)
    # perform optimization
    res = opt.minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=[eq_cons], options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    # repair
    x_opt = repair_h(res.x)
    return x_opt, compute_c5(x_opt), res.success
