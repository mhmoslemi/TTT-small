def project_bounds(v, target, lower=0., upper=1., max_iter=100):
    lo = lower - np.max(v)
    hi = upper - np.min(v)
    for _ in range(max_iter):
        lam = (lo+hi)/2
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if abs(s - target) < 1e-9:
            return w
        if s > target:
            hi = lam
        else:
            lo = lam
    return w
```

def project(v, target):
    lower, upper = 0.0, 1.0
    # Clip first to keep inside bounds at start of binary search; but we can keep as v.
    lo = -10.0
    hi = 10.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + lo, lower, upper)
    # adjust small diff
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # distribute diff equally among free vars (not at bounds)
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, lower, upper)
        else:
            # all at bounds, can't adjust; just spread diff among all and clip again
            w += diff / len(w)
            w = np.clip(w, lower, upper)
    return w

import numpy as np
import time
import random
import math

def _project_to_bounds(v: np.ndarray, target: float, lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """Projection of v onto the hyperplane sum(v)=target with box bounds."""
    lo, hi = -10.0, 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + lo, lower, upper)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, lower, upper)
    return w


def _c5_score(h: np.ndarray) -> float:
    """Exact C5 score as used by the evaluation."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _worst_shift(h: np.ndarray) -> int:
    """Index shift (relative to zero) that attains the maximal correlation."""
    n = h.size
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr_full))
    return idx - (n - 1)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst shift correlation w.r.t. h."""
    n = h.size
    dx = 2.0 / n
    if shift >= 0:
        shifted = np.concatenate([h[shift:], np.zeros(shift)])
    else:
        k = -shift
        shifted = np.concatenate([np.zeros(k), h[: n - k]])
    return (1.0 - shifted) * dx


def _mutate(h: np.ndarray, sigma: float = 0.2, frac: float = 0.2, target: float = None) -> np.ndarray:
    """Gaussian mutation on a random subset of entries followed by projection."""
    n = h.size
    k = max(1, int(frac * n))
    idx = np.random.choice(n, size=k, replace=False)
    h_new = h.copy()
    h_new[idx] += np.random.normal(scale=sigma, size=k)
    if target is None:
        target = n / 2.0
    return _project_to_bounds(h_new, target)

    """Search for a step function h on [0,2] with a low C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------

    else:
        n = 70
        h0 = np.random.rand(n)

    target_sum = n / 2.0
    h0 = _project_to_bounds(h0, target_sum)

    # ----------------------------------------------------------------------
    # population based stochastic optimisation
    pop_size = 10
    population = []

    # seed with the provided construction (if any) and diverse random vectors
    def add_to_pop(vec):
        score = _c5_score(vec)
        population.append((score, vec.copy()))
    add_to_pop(h0)
