import numpy as np
import time
import math
import random

def _evaluate_h(h):
    """Return C5 bound for step function values h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _pairwise_move(h, step):
    """Propose a new h by moving a small amount from one index to another."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    # feasible delta respecting bounds
    max_up_i = 1.0 - h[i]
    max_down_i = -h[i]
    max_up_j = 1.0 - h[j]
    max_down_j = -h[j]
    max_pos = min(step, max_up_i, -max_down_j)
    max_neg = -min(step, -max_down_i, max_up_j)
    if max_pos <= 0 and max_neg >= 0:
        return None
    delta = np.random.uniform(max_neg, max_pos)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def _jitter(h, scale, target_sum):
    """Apply a small random perturbation and re‑project onto the constraints."""
    n = h.size
    delta = np.random.uniform(-scale, scale, size=n)
    new_h = np.clip(h + delta, 0.0, 1.0)
    s = np.sum(new_h)
    if s > 0:
        new_h = np.clip(new_h * (target_sum / s), 0.0, 1.0)
    # final residual correction on a non‑border element if possible
    residual = target_sum - np.sum(new_h)
    if abs(residual) > 1e-12:
        idx = np.where((new_h > 1e-12) & (new_h < 1.0 - 1e-12))[0]
        if idx.size:
            new_h[idx[0]] = np.clip(new_h[idx[0]] + residual, 0.0, 1.0)
        else:
            new_h[0] = np.clip(new_h[0] + residual, 0.0, 1.0)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a small C5 bound.

    Returns:
        h_values (list of float): the sampled function values.
        c5_bound (float): the achieved max overlap integral.
        n_points (int): number of samples used.
    """

    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        init = initial_h_values  # provided by harness, may be None
        has_init = init is not None
    except NameError:
        has_init = False

        n = h.size
    else:
        n = 70
        # uniform random start, then scale to satisfy sum = n/2
        h = np.random.rand(n)
        target = n / 2.0
        s = np.sum(h)
        if s > 0:
            h = h * (target / s)
        # tiny fix for possible bound breaches after scaling
        diff = target - np.sum(h)
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)

    target_sum = n / 2.0
    # Ensure exact sum (within floating tolerance)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # adjust first non‑boundary entry
        for i in range(n):
            new_val = np.clip(h[i] + diff, 0.0, 1.0)
            diff_adj = new_val - h[i]
            h[i] = new_val
