def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose n_points
    n_points = 200
    # build initial h
    h = build_initial_h(n_points)
    # compute current C5
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c5 = c_curr
    # parameters
    step_scale = 0.2
    max_no_improve = 2000
    no_improve = 0
    while time.time() - start < budget_s:
        # schedule step_scale reduction over time
        elapsed = time.time() - start
        # linearly decay step_scale from 0.2 to 0.001
        step_scale = 0.2 * (1 - elapsed / budget_s) + 0.001
        # propose move:
        i, j = rng.integers(0, n_points, size=2, endpoint=False)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            continue
        delta = rng.uniform(-step_scale, step_scale)
        delta = max(min(delta, upper), lower)
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        # compute C5
        c_new = compute_c5(h_new)
        # acceptance criteria
        if c_new < c_curr:
            h = h_new
            c_curr = c_new
            no_improve = 0
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        else:
            # accept with probability depending on temperature
            temp = max(budget_s - elapsed, 1e-6)
            prob = np.exp(-(c_new - c_curr) / (0.01 * temp))
            if rng.random() < prob:
                h = h_new
                c_curr = c_new
                no_improve = 0
            else:
                no_improve += 1
        # occasional random perturbation if stuck
        if no_improve >= max_no_improve:
            # perturb best_h slightly
            perturb = rng.normal(scale=0.02, size=n_points)
            h = np.clip(best_h + perturb, 0.0, 1.0)
            # adjust sum to target
            target = n_points / 2
            # ensure sum exact: adjust by constant offset then clip
            diff = target - h.sum()
            h += diff / n_points
            np.clip(h, 0.0, 1.0, out=h)

import numpy as np
import time

def _shift_zeros(arr: np.ndarray, shift: int) -> np.ndarray:
    """Shift with zero padding (no wrap). Positive shift moves values to the left."""
    n = arr.size
    if shift == 0:
        return arr
    if shift > 0:
        # arr[i+shift] for i < n-shift, else 0
        return np.concatenate((arr[shift:], np.zeros(shift, dtype=arr.dtype)))
    else:
        s = -shift
        return np.concatenate((np.zeros(s, dtype=arr.dtype), arr[:n - s]))

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Iteratively clip to [0,1] and adjust to meet the sum constraint."""
    for _ in range(15):
        np.clip(v, 0.0, 1.0, out=v)
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        v += diff / v.size
    np.clip(v, 0.0, 1.0, out=v)
    # tiny final correction if needed
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # distribute remaining mass on non‑boundary entries
        idx = np.where((v > 0) & (v < 1))[0]
        if idx.size:
            inc = diff / idx.size
            v[idx] += inc
            np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _initial_h(n_points: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible starting vector, using the supplied global if present."""
    target = n_points / 2.0
    if "initial_h_values" in globals():
        src = np.asarray(initial_h_values, dtype=float)
        if src.size > 1:
            x_src = np.linspace(0.0, 2.0, src.size, endpoint=False)
            x_tgt = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(x_tgt, x_src, src)
            np.clip(h, 0.0, 1.0, out=h)
            return _project_to_constraints(h, target)
    # fallback: half ones, half zeros
    h = np.zeros(n_points, dtype=float)
    idx = rng.choice(n_points, n_points // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose a moderate resolution
    n_points = 200
    target_sum = n_points / 2.0

    h = _initial_h(n_points, rng)
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # parameters for the descent
    iters_no_improve = 0
    MAX_STALL = 5000

    while time.time() - start < budget_s:
        elapsed = time.time() - start
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        k = idx_max - (n_points - 1)   # shift giving the max overlap

        # gradient of the worst term (subgradient)
        h_plus = _shift_zeros(h, k)
        h_minus = _shift_zeros(h, -k)
        grad = 1.0 - h_plus - h_minus

        # learning rate decays linearly
        lr = 0.08 * (1.0 - elapsed / budget_s) + 0.001

        h_new = h - lr * grad
        h_new = _project_to_constraints(h_new, target_sum)

        c5_new = _c5_value(h_new)

        # accept if improvement, else keep current (no SA for simplicity)
        if c5_new < best_c5 - 1e-7:
