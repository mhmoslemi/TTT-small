import numpy as np
import random

import math

def compute_c5(h: np.ndarray) -> float:
    """Return the max overlap integral C5 for step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def max_shift_gradient(h: np.ndarray):
    """Compute gradient of the current maximal overlap term and the shift value."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr)) - (n - 1)          # shift that attains the maximum
    grad = np.zeros_like(h)

    if k >= 0:
        nk = n - k
        if nk > 0:
            grad[:nk] += 1.0 - h[k:]
            grad[k:] -= h[:nk]
    else:
        k_abs = -k
        nk = n - k_abs
        if nk > 0:
            grad[k_abs:] += 1.0 - h[:nk]
            grad[:nk] -= h[k_abs:]
    return grad, k

        # occasional random shake to escape shallow minima
        if it % 5000 == 0:
            noise = (np.random.rand(n) - 0.5) * 0.02
            pert = cur_h + noise
            np.clip(pert, 0.0, 1.0, out=pert)
            pert = project_to_box_and_sum(pert, n / 2.0)
            pert_score = compute_c5(pert)
            if pert_score < cur_score:
                cur_h = pert
                cur_score = pert_score
                if pert_score < best_score:
                    best_h = pert.copy()
                    best_score = pert_score

    # final projection and evaluation
    best_h = project_to_box_and_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, int(n)
