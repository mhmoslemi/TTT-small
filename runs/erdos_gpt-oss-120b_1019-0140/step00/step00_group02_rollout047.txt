def project_to_bounds_and_sum(v, S):
    # v is numpy array
    n = len(v)
    # Clip to [0,1] initial
    w = np.clip(v, 0, 1)
    # Compute sum
    s = w.sum()
    if abs(s - S) < 1e-9:
        return w
    # Determine if need to increase or decrease
    if s > S:
        # Need to reduce sum: reduce values where w > 0
        # We'll treat as water-filling: Let free = indices where w > 0
        # We'll subtract equal amount from them, but ensure not below 0
        # iterative:
        diff = s - S
        while diff > 1e-12:
            free_idx = np.where(w > 0)[0]
            if len(free_idx) == 0:
                break
            # amount to subtract per free item:
            delta = diff / len(free_idx)
            # subtract delta from each free, but not below 0
            w_new = w.copy()
            w_new[free_idx] = np.maximum(0, w[free_idx] - delta)
            # recompute diff
            diff = w_new.sum() - S
            if np.allclose(w_new, w):
                break
            w = w_new
        return w
    else:
        # need to increase sum: similar
        diff = S - s
        while diff > 1e-12:
            free_idx = np.where(w < 1)[0]
            if len(free_idx) == 0:
                break
            delta = diff / len(free_idx)
            w_new = w.copy()
            w_new[free_idx] = np.minimum(1, w[free_idx] + delta)
            diff = S - w_new.sum()
            if np.allclose(w_new, w):
                break
            w = w_new
        return w

def project_to_bounds_and_sum(v, S, eps=1e-9):
    # v is np array (float)
    # start with lower bound such that sum > S? Need to ensure monotonic.
    # Actually for lambda = -inf, v_i - lambda = inf -> clip to 1, sum = n.
    # For lambda = +inf, v_i - lambda = -inf-> clip to 0, sum =0.
    # So we can start lo=-1e6, hi=1e6 and binary search for sum.
    lo = -1e6
    hi = 1e6
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    # After loop, x = clip(v - hi,0,1) satisfies sum ~ S (allow minor diff)
    x = np.clip(v - hi, 0, 1)
    # may adjust minor diff: if sum deviates due to rounding, adjust by adding diff to entries not at bounds
    diff = S - x.sum()
    if abs(diff) > eps:

        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
    return x

        else:

import numpy as np
import time
import math
import random

def _project_to_sum(v, target):
    """Project vector v onto the set {x: 0<=x_i<=1, sum(x)=target}
    using a binary search on the Lagrange multiplier."""
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_score(h, dx):
    """Return the Erdős C5 value for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _simulated_annealing(initial_h, n, time_limit):
    """Simple SA optimizing the C5 score for a fixed resolution n."""
    target_sum = n * 0.5
    dx = 2.0 / n
    cur = initial_h.copy()
    cur_score = _c5_score(cur, dx)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    T = 0.3          # initial temperature
    T_min = 1e-6
    decay = 0.9995   # exponential cooling

    while True:
        if time.time() - start > time_limit:
            break
        sigma = T * 0.5               # noise scale linked to temperature
        noise = np.random.normal(scale=sigma, size=n)
        cand = _project_to_sum(cur + noise, target_sum)
        cand_score = _c5_score(cand, dx)

        if cand_score < cur_score or random.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        T = max(T * decay, T_min)

    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible step function h, its C5 bound, and number of points."""
    np.random.seed(seed)
    random.seed(seed)

    # Base vector: use supplied initial_h_values if present, otherwise random.
