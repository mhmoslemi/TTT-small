def compute_c5(h):

    return np.max(conv) * dx

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # determine n and initial h
    init = globals().get('initial_h_values')
    if init is not None:
        h = np.array(init, dtype=np.float64).copy()
        n = h.size
    else:
        n = 200
        # start with binary random half ones
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
        # if n odd adjust
        if n % 2 == 1:
            # add fractional 0.5 to a random index
            idx = rng.integers(0, n)
            h[idx] = 0.5
    target_sum = n / 2.0
    # ensure sum exactly target_sum (due to binary it is exact)
    # compute C5 function
    def compute_c5(arr):
        conv = np.correlate(arr, 1 - arr, mode='full')
        return np.max(conv) * (2.0 / len(arr))
    best_h = h.copy()
    best_cost = compute_c5(h)
    current_h = h.copy()
    current_cost = best_cost

    T0 = 0.05  # start temperature
    # iteration counters
    iteration = 0
    last_improve_iter = 0
    # set thresholds
    restart_limit = 50000  # if no improvement for this many iterations restart
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.1:
            break
        T = T0 * max(0.0, 1.0 - elapsed / (budget_s - 0.1))
        # decide move type
        move_type = rng.integers(0, 5)  # more random
        if move_type == 0:
            # random transfer
            i, j = rng.choice(n, size=2, replace=False)
            # compute max transfer amount
            max_transfer = min(current_h[i], 1.0 - current_h[j])
            if max_transfer <= 0:
                continue
            # step fraction proportional to temperature
            step_frac = 0.1 + 0.9 * (T / T0)  # between 0.1 and 1.0
            amount = rng.random() * max_transfer * step_frac
            new_h = current_h.copy()
            new_h[i] -= amount
            new_h[j] += amount
        elif move_type == 1:
            # single index perturb with compensating index
            i = rng.integers(0, n)
            new_val = rng.random()
            delta = new_val - current_h[i]
            # find a partner j that can absorb delta
            # attempt few times
            found = False
            for _ in range(5):
                j = rng.integers(0, n)
                if j == i:
                    continue
                if 0 <= current_h[j] - delta <= 1:
                    found = True
                    break
            if not found:
                continue
            new_h = current_h.copy()
            new_h[i] = new_val
            new_h[j] -= delta
        elif move_type == 2:
            # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            # targeted transfer based on worst shift
            # compute worst shift index
            conv_arr = np.correlate(current_h, 1 - current_h, mode='full')
            idx_max = np.argmax(conv_arr)
            shift = idx_max - (n - 1)
            # compute product for valid indices
            indices = np.arange(n)
            if shift >= 0:
                valid = indices >= shift
                counterpart = indices - shift
            else:
                valid = indices < n + shift  # since shift negative, i-s = i+|shift|
                counterpart = indices - shift  # shift negative
            product = np.zeros(n)
            product[valid] = current_h[indices[valid]] * current_h[counterpart[valid]]
            # choose i with high h but low product, j with low h but high product
            # compute scores
            score_i = current_h - product  # large when hi high, product low
            score_j = product - current_h  # large when product high, hi low
            # mask for feasible transfers
            mask_i = (score_i > 0) & (current_h > 0)
            mask_j = (score_j > 0) & (current_h < 1)
            if not np.any(mask_i) or not np.any(mask_j):
                # fallback to random transfer
                i, j = rng.choice(n, size=2, replace=False)
                max_transfer = min(current_h[i], 1.0 - current_h[j])
                if max_transfer <= 0:
                    continue
                amount = rng.random() * max_transfer * 0.5
                new_h = current_h.copy()
                new_h[i] -= amount
                new_h[j] += amount
            else:
                # select i and j with prob proportional to score
                prob_i = score_i[mask_i] / score_i[mask_i].sum()
                prob_j = score_j[mask_j] / score_j[mask_j].sum()
                i = rng.choice(np.where(mask_i)[0], p=prob_i)
                j = rng.choice(np.where(mask_j)[0], p=prob_j)
                max_transfer = min(current_h[i], 1.0 - current_h[j])
                if max_transfer <= 0:
                    continue
                amount = rng.random() * max_transfer * 0.5
                new_h = current_h.copy()
                new_h[i] -= amount
                new_h[j] += amount
        # evaluate new candidate
        new_cost = compute_c5(new_h)
        # acceptance
        delta_cost = new_cost - current_cost
        if delta_cost < 0 or rng.random() < np.exp(-delta_cost / (T + 1e-12)):
            current_h = new_h
            current_cost = new_cost
            if new_cost < best_cost - 1e-12:
                best_cost = new_cost
                best_h = new_h.copy()
                last_improve_iter = iteration
        iteration += 1
        # restart if stagnated
        if iteration - last_improve_iter > restart_limit and elapsed < budget_s * 0.9:
            # reinitialize random binary solution
            current_h = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=n // 2, replace=False)
            current_h[ones] = 1.0
            if n % 2 == 1:
                idx = rng.integers(0, n)
                current_h[idx] = 0.5
            current_cost = compute_c5(current_h)
            # maybe also keep best_h unaffected
            last_improve_iter = iteration
    return best_h, best_cost, n

def project_to_bounds_sum(v, target):
    # v shape (n,)
    # objective: find x = clip(v - lam, 0, 1) such that sum(x) = target
    low = np.min(v) - 1.0
    high = np.max(v)

        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - ((low + high) / 2.0), 0.0, 1.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    import time
    import numpy as np

    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s - 0.1  # safety margin

    # ------------------------------------------------------------------
    # Problem size and initial feasible point
    # ------------------------------------------------------------------
    init = globals().get('initial_h_values')
    if init is not None:
        h = np.array(init, dtype=np.float64).copy()
        n = h.size
    else:
        n = 200  # fallback size
        # random feasible point via projection onto [0,1] simplex
        def _project(v, target):
            low = np.min(v) - 1.0
            high = np.max(v)
            for _ in range(60):
                lam = (low + high) / 2.0
                x = np.clip(v - lam, 0.0, 1.0)
                s = x.sum()
                if s > target:
                    low = lam
                else:
                    high = lam
            return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

        v = rng.random(n)
        h = _project(v, n / 2.0)

    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def compute_corr_and_cost(arr):
        """Return full correlation (h with 1-h) and the C5 value."""
        conv = np.correlate(arr, 1.0 - arr, mode='full')
        cost = np.max(conv) * dx
        return conv, cost

    # projection used after gradient‑style updates (not used in SA moves,
    # kept for possible future extensions)
    def _project(v):
        low = np.min(v) - 1.0
        high = np.max(v)
        for _ in range(60):
            lam = (low + high) / 2.0
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                low = lam
            else:
                high = lam
        return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    corr, cost = compute_corr_and_cost(h)
    best_h = h.copy()
    best_cost = cost

    current_h = h.copy()
    current_corr = corr
    current_cost = cost

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    T0 = 0.05                # initial temperature
    restart_limit = 200000   # iterations without improvement before restart
    last_improve_iter = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimization loop
    # ------------------------------------------------------------------
    while True:
        # time check
        if time.time() - start_time >= time_limit:
            break

        # current worst shift (index of maximum correlation)
        idx_max = int(np.argmax(current_corr))
        shift = idx_max - (n - 1)   # integer shift corresponding to worst C5

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = rng.integers(0, 5)   # 0..4

        if move_type == 0:
            # simple random transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(current_h[i], 1.0 - current_h[j])
            if max_transfer <= 0.0:
                continue
            # step size scales with temperature (larger early, smaller later)
            step_frac = 0.1 + 0.9 * (np.exp(-iteration / 50000.0))
            amount = rng.random() * max_transfer * step_frac
            new_h = current_h.copy()
            new_h[i] -= amount
            new_h[j] += amount

        elif move_type == 1:
            # perturb a single entry and compensate elsewhere
            i = rng.integers(0, n)
            new_val = rng.random()
            delta = new_val - current_h[i]
            # find a partner j that can absorb delta
            ok = False
            for _ in range(5):
                j = rng.integers(0, n)
                if j == i:
                    continue
                if 0.0 <= current_h[j] - delta <= 1.0:
                    ok = True
                    break
            if not ok:
                continue
            new_h = current_h.copy()
            new_h[i] = new_val
            new_h[j] -= delta

        elif move_type == 2:
            # swap two entries (does not change sum)
            i, j = rng.choice(n, size=2, replace=False)
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]

        else:
            # targeted transfer based on the current worst shift
            idx = np.arange(n)
            if shift >= 0:
                valid = idx >= shift
                counterpart = idx - shift
            else:
                valid = idx < n + shift
                counterpart = idx - shift    # shift is negative → addition
            product = np.zeros(n, dtype=np.float64)
            product[valid] = current_h[idx[valid]] * current_h[counterpart[valid]]

            score_i = current_h - product          # high when hi large, product low
            score_j = product - current_h          # high when counterpart large, hi small
            mask_i = (score_i > 0) & (current_h > 1e-12)
            mask_j = (score_j > 0) & (current_h < 1.0 - 1e-12)

            if mask_i.any() and mask_j.any():
                probs_i = score_i[mask_i]
                probs_i = probs_i / probs_i.sum()
                i = rng.choice(np.where(mask_i)[0], p=probs_i)

                probs_j = score_j[mask_j]
                probs_j = probs_j / probs_j.sum()
                j = rng.choice(np.where(mask_j)[0], p=probs_j)

                max_transfer = min(current_h[i], 1.0 - current_h[j])
                if max_transfer <= 0.0:
                    continue
                amount = rng.random() * max_transfer * 0.5
                new_h = current_h.copy()
                new_h[i] -= amount
                new_h[j] += amount
            else:
                # fallback to simple random transfer
                i, j = rng.choice(n, size=2, replace=False)
                max_transfer = min(current_h[i], 1.0 - current_h[j])
                if max_transfer <= 0.0:
                    continue
                amount = rng.random() * max_transfer * 0.5
                new_h = current_h.copy()
                new_h[i] -= amount
                new_h[j] += amount

        # --------------------------------------------------------------
        # Evaluate the candidate
        # --------------------------------------------------------------
        new_corr, new_cost = compute_corr_and_cost(new_h)

        # --------------------------------------------------------------
        # Acceptance decision (simulated annealing)
        # --------------------------------------------------------------
        delta = new_cost - current_cost
        elapsed = time.time() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / (time_limit))
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            current_h = new_h
            current_corr = new_corr
            current_cost = new_cost

            if new_cost < best_cost - 1e-12:
                best_cost = new_cost
                best_h = new_h.copy()
                last_improve_iter = iteration

        iteration += 1

        # --------------------------------------------------------------
        # Restart if stagnant for a long time
        # --------------------------------------------------------------
        if iteration - last_improve_iter > restart_limit and elapsed < time_limit * 0.9:
            # Re‑initialise a fresh feasible point
            v = rng.random(n)
            current_h = _project(v)
            current_corr, current_cost = compute_corr_and_cost(current_h)
            last_improve_iter = iteration

    return best_h, best_cost, n
