def worst_lag_and_gradient(h):
    N = h.shape[0]
    dx = 2.0 / N
    # compute correlation vector
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argmax(corr)  # index of max correlation
    lag = k_idx - (N - 1)   # convert to lag (range -(N-1) .. (N-1))
    # compute gradient
    grad = np.zeros_like(h)
    if lag >= 0:
        k = lag
        # first contribution
        grad[:N-k] += (1.0 - h[k:])  # 1 - h[i+k]
        # second contribution
        grad[k:] -= h[:N-k]
    else:
        kp = -lag
        # first contribution: indices from kp..N-1
        grad[kp:] += (1.0 - h[:N - kp])
        # second contribution: indices 0..N-kp-1
        grad[:N - kp] -= h[kp:]
    grad *= dx
    # The objective value:
    val = corr[k_idx] * dx

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def worst_lag_and_gradient(h: np.ndarray):
    """
    Compute the lag attaining the current C5, its value,
    and the gradient of that worst term w.r.t. h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)                     # lag in [-(n-1), …, n-1]
    grad = np.zeros_like(h)

    if lag >= 0:
        k = lag
        if k < n:
            grad[: n - k] += 1.0 - h[k:]
            grad[k:]      -= h[: n - k]
    else:
        kp = -lag
        if kp < n:
            grad[kp:]       += 1.0 - h[: n - kp]
            grad[: n - kp]  -= h[kp:]

    grad *= dx
    val = corr[idx] * dx
    return lag, float(val), grad


def project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1]^n intersected with the
    hyperplane sum(v)=target_sum.  Uses water-filling (binary search on the
    shift λ).
    """
    lo = v.min() - 1.0          # guarantees projection = all 1's (sum=n)
    hi = v.max()                # guarantees projection = all 0's (sum=0)

    for _ in range(60):         # enough for double precision
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def initialize_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial feasible h of length n.
    If a global variable `initial_h_values` exists, upscale it by linear
    interpolation; otherwise start from a random vector.
    """
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.shape[0]
        if n0 == n:
            h = base.copy()
        else:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x, x0, base)
    else:
        h = rng.random(n)

    target = n / 2.0
    h = project_to_feasible(h, target)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # Primary discretisation (balance between speed and expressivity)
    n = 180
    target_sum = n / 2.0

    best_h = None
    best_c5 = float("inf")

    # Number of independent starts; adapt to remaining time.
    restart = 0
    while time.time() < deadline and restart < 8:
        restart += 1

        # ----- initialization -----
        h = initialize_h(n, rng)
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()

        # ----- projected sub‑gradient descent -----
        step = 0.5
        no_improve = 0
        it = 0
        while time.time() < deadline:
            lag, val, grad = worst_lag_and_gradient(h)

            # gradually shrink step size
            step = max(0.01, step * 0.9999)

            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            new_c5 = compute_c5(h_new)

            if new_c5 < cur_c5 - 1e-12:
                h, cur_c5 = h_new, new_c5
                no_improve = 0
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h_new.copy()
            else:
                # occasional uphill move for escaping plateaus
                if rng.random() < 0.005:
                    h, cur_c5 = h_new, new_c5
                no_improve += 1

            it += 1
            if no_improve > 3000:

    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # Final safety projection (ensures exact feasibility)
    best_h = project_to_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
