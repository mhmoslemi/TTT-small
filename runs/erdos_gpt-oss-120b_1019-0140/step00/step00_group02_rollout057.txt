def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by correlation."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _worst_shift_and_pair(h):
    """
    Find shift k giving the maximal overlap integral and a pair (src,tgt)
    that can reduce it by moving mass from src to tgt.
    Returns (shift, src, tgt, eps_max).
    """
    n = h.size
    f = np.correlate(h, 1.0 - h, mode="full")
    zero_idx = n - 1
    # ignore the trivial shift 0 (its value is always 0)
    f_no0 = f.copy()
    f_no0[zero_idx] = -np.inf
    max_idx = int(np.argmax(f_no0))
    shift = max_idx - zero_idx  # may be negative
    # Determine source/target indices based on sign of shift
    if shift > 0:
        # contributors: h[i] * (1 - h[i+shift])
        hi = h[: n - shift]
        hj = h[shift:]
        contrib = hi * (1.0 - hj)
        i_rel = int(np.argmax(contrib))
        src = i_rel
        tgt = i_rel + shift
    elif shift < 0:
        sh = -shift
        # contributors for negative shift: h[i+sh] * (1 - h[i])
        hi = h[: n - sh]
        hj = h[sh:]
        contrib = hj * (1.0 - hi)
        i_rel = int(np.argmax(contrib))
        src = i_rel + sh
        tgt = i_rel
        shift = -sh  # make shift positive for later use
    else:
        # shift == 0 should not happen (ignored), fallback to random pair
        src = random.randrange(n)
        tgt = random.randrange(n)
        while tgt == src:
            tgt = random.randrange(n)
        shift = 0
    eps_max = min(float(h[src]), 1.0 - float(h[tgt]))
    return shift, src, tgt, eps_max

    cur_sum = float(np.sum(h))
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum

    no_improve = 0
    max_no_improve = 3000          # trigger a restart after this many fails
    step_factor = 0.3               # fraction of eps_max to actually move
    iteration = 0

    while True:
        if time.time() - start >= budget_s * 0.95:
            break

        # locate worst shift and a promising pair to transfer mass
        shift, src, tgt, eps_max = _worst_shift_and_pair(h)

        if eps_max <= 1e-14:
            # cannot move mass on this pair – count as a failure
            no_improve += 1
            if no_improve >= max_no_improve:
                # random restart (binary half‑ones)
                h = np.zeros(n, dtype=float)
                h[random.sample(range(n), n // 2)] = 1.0
                cur_score = _compute_c5(h, dx)
                no_improve = 0
            continue

        # transfer a fraction of the available mass
        epsilon = eps_max * (step_factor * random.random() + 0.1)  # 0.1‑0.4 of eps_max
        h[src] -= epsilon
        h[tgt] += epsilon

        # evaluate new configuration
        new_score = _compute_c5(h, dx)

        if new_score <= cur_score + 1e-12:
            # accept move
            cur_score = new_score
            no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
        else:
            # reject – revert
            h[src] += epsilon
            h[tgt] -= epsilon
            no_improve += 1
            if no_improve >= max_no_improve:
                # restart from current best with a small random perturbation
                h = best_h.copy()
                # perturb a few random positions
                for _ in range(max(1, n // 20)):
                    i = random.randrange(n)
                    delta = (random.random() - 0.5) * 0.2  # up to ±0.1
                    new_val = h[i] + delta
                    new_val = min(max(new_val, 0.0), 1.0)
                    # adjust another index to keep sum
                    j = random.randrange(n)
                    while j == i:
                        j = random.randrange(n)
                    diff = new_val - h[i]
                    if diff > 0:
                        # move mass from j
                        move = min(diff, h[j])
                        h[i] = new_val
                        h[j] -= move
                        diff -= move
                    else:
                        # move mass to j
                        move = min(-diff, 1.0 - h[j])
                        h[i] = new_val
                        h[j] += move
                cur_score = _compute_c5(h, dx)
                no_improve = 0

        iteration += 1

    # final evaluation (use helper if available)
    try:
        final_c5 = float(evaluate_erdos_solution(best_h.tolist()))
    except Exception:
        final_c5 = _compute_c5(best_h, dx)

    return (best_h.tolist(), final_c5, n)
