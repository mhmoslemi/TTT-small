def compute_corr(h):
    n = h.shape[0]
    # compute correlation via FFT: conv = ifft(fft(h) * fft(g_rev))
    g = 1 - h
    # reverse g
    g_rev = g[::-1]
    size = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(g_rev, size)
    conv = np.fft.irfft(H * G, size)[:2*n-1]
    return conv

def project_to_simplex(v, target):
    import cvxpy as cp
    n = v.shape[0]
    x = cp.Variable(n)
    constraints = [x >= 0, x <= 1, cp.sum(x) == target]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if x.value is None:
        # fallback: clip and adjust sum manually
        x_val = np.clip(v, 0, 1)
        # Adjust sum
        cur = x_val.sum()
        diff = target - cur
        # Add diff equally to all (maybe saturate)
        # Not needed if fails rarely
        return x_val
    return np.array(x.value).flatten()

Goal: find x that solves min||x - v||_2 s.t. sum x = target, 0 ≤ x ≤ 1.

If sum_x < target -> need to increase some components that are not at upper bound. If sum_x > target -> need to decrease some components not at lower bound. Equivalent to solving a constrained allocation problem: we have a residual delta = target - sum_x.

If delta > 0 (need to increase), then we can distribute delta among those indices where x_i < 1. The typical solution is to add a constant λ to each such index, but limited by the bound 1.

Indeed problem reduces to: find λ such that sum_i clip(v_i + λ, 0, 1) = target. This is because shifting each variable equally while respecting bounds yields optimum for L2 projection onto a hyperplane and box.

Procedure: binary search for λ.

def project_to_bounds_sum(v, target, max_iter=50):
    # v is np array. target is desired sum.
    # we want to find lambda such that sum(clip(v + lambda, 0, 1)) = target.
    lo = -1.0  # worst-case shift negative enough to bring all to 0
    hi = 1.0   # positive enough to bring all to 1
    # Actually we can set lo = -np.max(v) to bring all to zero, hi = 1 - np.min(v) to bring all to one.
    lo = -np.max(v) - 1  # ensures clip to 0
    hi = 1 - np.min(v) + 1 # ensures clip to 1
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # Might be slight sum error due to rounding. Adjust by adding to a random free index.
    # If sum rounding off, we can correct by adding small epsilon to some indices not at bounds.
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # adjust diff by adding to some free element
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            # distribute equally
            add = diff / free.size
            # Clip each update to maintain bounds
            for i in free:
                new_val = x[i] + add
                x[i] = np.clip(new_val, 0.0, 1.0)
    return x

def project_to_bounds_sum(v, target, max_iter=60):
    # solve for lambda such that sum(clip(v + lambda, 0,1)) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # adjust small residual:
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # adjust free entries
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            adjust = diff / free.size
            for i in free:
                x[i] = np.clip(x[i] + adjust, 0.0, 1.0)
    return x

import numpy as np
import time

def _project_to_bounds_sum(v, target, max_iter=60):
    # Find lambda such that sum(clip(v + lambda, 0, 1)) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size > 0:
            adjust = diff / free.size
            for i in free:
                x[i] = np.clip(x[i] + adjust, 0.0, 1.0)
    return x

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(g[::-1], size)
    conv = np.fft.irfft(H * G, size)[:2 * n - 1]
    return np.max(conv) * dx
