def compute_c5(h):

        h = np.interp(x_new, x_old, base)

        indices = np.argsort(-h)  # descending

        h[indices] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Simulated annealing parameters
    T0 = 0.01 * best_c5  # initial temperature
    T_end = 1e-6 * best_c5
    # Number of iterations based on budget; we can set a max iterations high but break if time runs out.
    max_iter = 2000000  # large upper bound

    # track indices of ones and zeros to enable swaps quickly
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    # annealing schedule: exponential decay over total iterations
    for it in range(max_iter):
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:  # leave some margin
            break
        # temperature schedule
        t = max(T0 * ((T_end / T0) ** (it / max_iter)), T_end)

        # propose swap
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # swap
        h[i], h[j] = h[j], h[i]
        # compute new cost
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5
        # acceptance criteria
        if new_c5 < best_c5 or random.random() < np.exp(-delta / t):
            # accept move
            # update sets
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

            h[i], h[j] = h[j], h[i]

import numpy as np, random, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute full correlation
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Determine size n
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        n = base.shape[0]
        # Ensure even n
        if n % 2 != 0:
            # pad with zero to make even
            base = np.append(base, 0.0)
            n += 1
        # Convert to binary: pick top half values
        indices = np.argsort(-base)  # descending
        h = np.zeros(n, dtype=float)
        h[indices[:n//2]] = 1.0
    else:
        n = 200
        if n % 2 != 0:
            n += 1
        h = np.zeros(n, dtype=float)
        ones_idx = np.random.choice(n, n//2, replace=False)
        h[ones_idx] = 1.0
    
    # Keep current and best
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    
    # Prepare sets of indices
    ones = set(np.where(cur_h == 1)[0].tolist())
    zeros = set(np.where(cur_h == 0)[0].tolist())
    
    # Parameters for simulated annealing
    target = 0.3808
    # initial temperature set as fraction of current cost
    T0 = max(1e-3, cur_c5 * 0.2)
    T_end = 1e-6
    
    # function to get temperature based on elapsed fraction
    def temperature(elapsed, total):
        # exponential decay
        frac = elapsed / total
        if frac >= 1:
            return T_end
        return T0 * (T_end / T0) ** frac
    
    # main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:
            break
        # Get temperature
        T = temperature(elapsed, budget_s * 0.9)
        # Propose swap
        # Randomly select a one and a zero
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        # Acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            # accept move
            cur_c5 = new_c5
            # Update sets
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                # early exit if we beat target
                if best_c5 <= target:
                    break
        else:
            # reject, revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    
    return (best_h.tolist(), float(best_c5), int(len(best_h)))

import time
import random
import math
import numpy as np

def _c5_of(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h); mode='full' gives all shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _random_binary(n):
    """Create a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # --- problem size -------------------------------------------------
    # Target a modest size (fast evaluation, still expressive)
    N = 200
    if N % 2:
        N += 1

    # --- initial vector ------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # up‑sample to the chosen size N
        n0 = base.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, N, endpoint=False)
        interpolated = np.interp(x, x0, base)
        # turn into a feasible binary vector (exactly N/2 ones)
        idx = np.argsort(-interpolated)  # descending
        h = np.zeros(N, dtype=float)
        h[idx[:N // 2]] = 1.0
    else:
        h = _random_binary(N)

    # best solution found so far
    cur_h = h.copy()
    cur_c5 = _c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # sets of indices for fast sampling
    ones_set = set(np.where(cur_h == 1.0)[0].tolist())
    zeros_set = set(np.where(cur_h == 0.0)[0].tolist())

    # simulated‑annealing schedule
    TARGET = 0.3808
    T0 = max(1e-3, cur_c5 * 0.2)
    T_end = 1e-6

    def temperature(elapsed, total):
        frac = min(1.0, elapsed / total)
        # exponential decay from T0 to T_end
        return T0 * (T_end / T0) ** frac

    # main optimisation loop
    steps_without_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97:
            break

        # temperature based on overall progress
        T = temperature(elapsed, budget_s * 0.9)

        # propose a swap between a 1‑position and a 0‑position
        i = random.sample(ones_set, 1)[0]
        j = random.sample(zeros_set, 1)[0]

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = _c5_of(cur_h)

        # acceptance rule
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            # accept
            cur_c5 = new_c5
            # update index sets
            ones_set.remove(i)
            zeros_set.remove(j)
            ones_set.add(j)
            zeros_set.add(i)
            steps_without_improve = 0
            # record global best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                if best_c5 <= TARGET:
                    break
        else:
            # reject – revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            steps_without_improve += 1

        # occasional restart if stagnation persists
        if steps_without_improve >= 5000:
            cur_h = _random_binary(N)
            cur_c5 = _c5_of(cur_h)
            ones_set = set(np.where(cur_h == 1.0)[0].tolist())
            zeros_set = set(np.where(cur_h == 0.0)[0].tolist())
            steps_without_improve = 0

    return (best_h.tolist(), float(best_c5), int(N))
