import numpy as np
from scipy.signal import correlate

def compute_fitness(individual, dx=2/45):
    """Returns (C5, individual)"""
    try:
        h_ar = np.array(individual, dtype=float)
        correlation = correlate(h_ar, 1 - h_ar, mode='full')
        best = np.max(correlation) if len(correlation) > 0 else 0  # Handled by correlate
        return best * dx  # C5 value
    except Exception:
        return float('inf')  # Penalize infinity despite being invalid

def repair_constraint(individual, target_sum=22.5):
    """Ensures the sum of h is 22.5 via scaling + clipping"""
    h_ar = np.array(individual, dtype=float)
    current_sum = np.sum(h_ar)
    if abs(current_sum - target_sum) < 1e-5:
        return individual
    # Scale approach with clipping
    factor = (target_sum - current_sum)
    # Additive scaling to retain existing values + correction
    h_ar = h_ar + factor * (22.5 / current_sum)  # Shrink/grow entire vector
    np.clip(h_ar, 0, 1, out=h_ar)  # Clip to [0,1]
    # Post-correction: enforce sum
    while not np.isclose(np.sum(h_ar), target_sum, atol=1e-5):
        # Random value push
        idx = np.random.randint(0, len(h_ar))
        h_ar[idx] += np.sign(target_sum - np.sum(h_ar)) * 0.01  # Gradual adjustment
        np.clip(h_ar, 0, 1, out=h_ar)
    return h_ar.tolist()

def crossover(parent1, parent2):
    num = len(parent1)
    # Generate random pivot index
    pivot_idx = np.random.randint(0, num)
    new_parent = []
    for i in range(num):
        if i < pivot_idx:
            new_parent.append(parent1[i])
        else:
            new_parent.append(parent2[i] if np.random.random() < 0.5 else parent1[i])
    return new_parent
