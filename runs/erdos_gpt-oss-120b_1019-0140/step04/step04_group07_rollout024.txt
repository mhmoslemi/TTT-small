def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Helper: projection
    def project(v, s):
        # Euclidean projection onto sum and bounds.
        # Use algorithm based on lam. (Implementation as in previous code.)
        n = v.size
        lo, hi = 0.0, 1.0
        # quick cases
        if s <= lo * n:
            return np.full_like(v, lo)
        if s >= hi * n:
            return np.full_like(v, hi)
        # find initial lam bounds
        lam_lo = np.min(v) - hi
        lam_hi = np.max(v) - lo
        def f(lam):
            return np.clip(v - lam, lo, hi).sum()
        # ensure f(lam_lo) >= s >= f(lam_hi) (since f decreasing?)
        # Actually f(lam) is decreasing in lam.
        # We'll adjust if needed:
        while f(lam_lo) < s:
            lam_lo -= (hi - lo) * 2  # expand
        while f(lam_hi) > s:
            lam_hi += (hi - lo) * 2
        for _ in range(60):
            lam_mid = (lam_lo + lam_hi) / 2.0
            fm = f(lam_mid)
            if abs(fm - s) < 1e-12:
                break
            if fm > s:
                lam_lo = lam_mid
            else:
                lam_hi = lam_mid
        return np.clip(v - lam_mid, lo, hi)

    def compute_c5_fft(h):
        n = h.shape[0]
        m = 2 * n
        # using real FFT for speed
        H = np.fft.rfft(h, m)
        G = np.fft.rfft(1.0 - h, m)
        corr = np.fft.irfft(H * G, m)
        corr = corr[:2*n-1]
        return np.max(corr) * (2.0 / n)

    def upsample(arr, new_n):
        m = arr.shape[0]
        if m == new_n:
            return arr.copy()
        x_old = np.linspace(0.0, 1.0, m, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

    def thue_morse(n):
        # generate as floats 0/1
        seq = np.fromfunction(lambda i: (np.unpackbits(i.astype(np.uint64).view(np.uint8))[::-1][:].sum() & 1), (n,), dtype=int) # too complex.
        # Simpler: use binary count via bitwise:
        idx = np.arange(n, dtype=np.int64)
        # count bits:
        bits = ((idx[:, None] & (1 << np.arange(64))) > 0).sum(axis=1)  # slow.
        # Instead use np.vectorize with bin count: maybe okay for n <= 512.
        # Simpler: use list comprehension.
        seq = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
        return seq

    def random_balanced(n):
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
        return h

    # Starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    # Ensure size divisible by many powers of two: base length = base.shape[0]; we want it as is.

    # Project to sum constraint
    base = project(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # Define heuristic functions
    def simulated_annealing(h, time_allowed):
        # use given time_allowed seconds
        end_time = time.time() + time_allowed
        cur = h.copy()
        cur_score = compute_c5_fft(cur)
        best_loc = cur.copy()
        best_loc_score = cur_score
        # temperature schedule
        T = 0.01 * cur_score  # initial temp ~ 1% of score
        # if score small may be zero; ensure not zero
        if T < 1e-4:
            T = 1e-4
        n = cur.size
        while time.time() < end_time:
            # pick i != j
            i, j = rng.choice(n, size=2, replace=False)
            # move from i to j
            max_delta = min(cur[i], 1.0 - cur[j])
            if max_delta <= 0:
                # try opposite direction
                max_delta = min(cur[j], 1.0 - cur[i])
                if max_delta <= 0:
                    continue
                # swap i and j roles
                i, j = j, i
            # delta size scaled by temperature - we can random fraction
            delta = max_delta * min(1.0, rng.random() * (T / cur_score + 0.1))
            # ensure delta > 0
            if delta <= 0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            new_score = compute_c5_fft(new)
            diff = new_score - cur_score
            if diff <= 0 or rng.random() < np.exp(-diff / T):
                cur = new
                cur_score = new_score
                if new_score < best_loc_score:
                    best_loc = new.copy()
                    best_loc_score = new_score
            # cooling
            T *= 0.9995
            # break if temperature tiny
            if T < 1e-6:
                T = 1e-6
        return best_loc, best_loc_score

    def gradient_descent(h, max_iters, step0):

        n = h.size
        s = n / 2.0
        step = step0
        cur = h.copy()
        cur_score = compute_c5_fft(cur)
        best_h_local = cur.copy()
        best_score_local = cur_score
        momentum = np.zeros_like(cur)
        beta = 0.9
        for it in range(max_iters):
            # compute correlation
            # find worst shifts
            # correlation as linear convolution
            n = cur.size
            m = 2 * n
            H = np.fft.rfft(cur, m)
            G = np.fft.rfft(1.0 - cur, m)
            corr = np.fft.irfft(H * G, m)[:2*n-1]
            max_corr = np.max(corr)
            # compute shifts achieving max
            shifts = np.where(np.abs(corr - max_corr) < 1e-12)[0] - (n - 1)
            # compute gradient: for each shift, derivative wrt h_i is (1 - h_{i+shift}) + (1 - h_{i-shift})? Actually derivative of corr for shift s:
            # corr_s = sum_i h_i * (1 - h_{i+s})
            # d/dh_i of corr_s = (1 - h_{i+s}) - h_{i-s} (if shift negative)
            # but we want gradient of max corr approximated as average over worst shifts.
            grad = np.zeros_like(cur)
            for sft in shifts:
                if sft >= 0:
                    # term1: (1 - h[i+sft]) for i in 0..n-sft-1 contributes to gradient of h[i]
                    grad[:n - sft] += (1.0 - cur[sft:])
                    # term2: -h[i - sft] for i in sft..n-1 contributes to gradient of h[i]
                    grad[sft:] -= cur[:n - sft]
                else:
                    s = -sft
                    # shift negative: corr_s = sum_i h_{i+s} * (1 - h_i)
                    # derivative wrt h_i: - (1 - h_{i+s})? Actually differentiate: sum_j h_{j+s}*(1 - h_j)
                    # For term h_i appears as (1 - h_i) factor when i is in second position, and as h_i in first position when i-s in range.
                    # Equivalent to case above with swapped.
                    grad[s:] += (1.0 - cur[:n - s])
                    grad[:n - s] -= cur[s:]
            if len(shifts) > 0:
                grad /= len(shifts)
            momentum = beta * momentum + (1 - beta) * grad
            new = cur - step * momentum
            new = project(new, s)
            new_score = compute_c5_fft(new)
            if new_score <= cur_score:
                cur = new
                cur_score = new_score
                step = min(step0, step * 1.02)
                if new_score < best_score_local:
                    best_h_local = new.copy()
                    best_score_local = new_score
            else:
                step *= 0.5
                if step < 1e-6:
                    step = step0 * 0.1
            # early exit

def simulated_annealing(h, time_allowed, rng):
    # compute number of iterations based on time_allowed? Could run while time left > 0.
    start = time.time()
    end = start + time_allowed
    cur = h.copy()
    cur_score = compute_c5_fft(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.size
    # initial temperature as maybe 0.01 * cur_score
    T = max(1e-4, cur_score * 0.01)
    while time.time() < end:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta == 0:
            max_delta = min(cur[j], 1.0 - cur[i])
            if max_delta == 0:
                continue
            i, j = j, i
        # sample delta proportionally to temperature
        # heat scaling: larger T allows larger delta, but we also limit by max_delta.
        # Use factor = min(1.0, rng.random() * (T / cur_score + 0.1))
        factor = min(1.0, rng.random() * (T / cur_score + 0.1))
        delta = max_delta * factor
        if delta <= 0:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5_fft(new)
        diff = new_score - cur_score
        if diff <= 0 or rng.random() < np.exp(-diff / T):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        # cooling
        T *= 0.9995
        if T < 1e-6:
            T = 1e-6
    return best, best_score

def compute_c5_fft(h):
    n = h.shape[0]
    m = 2 * n
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * G, m)
    # linear correlation result length 2n-1
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

    n = v.size

        return np.clip(v - lam, lo, hi).sum()

        lam_low -= (hi - lo) * 2
    while f(lam_high) > s:

        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5_fft(h):
    """Fast C5 evaluation via FFT (linear correlation)."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * G, m)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from size m to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)


def random_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(h, time_allowed, rng):
    """Continuous annealing that preserves sum and bounds."""
    start = time.time()
    end = start + time_allowed
    cur = h.copy()
    cur_score = compute_c5_fft(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.size
    T = max(1e-4, cur_score * 0.5)          # initial temperature

    while time.time() < end:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta == 0.0:
            # try opposite direction
            max_delta = min(cur[j], 1.0 - cur[i])
            if max_delta == 0.0:
                continue
            i, j = j, i
        # temperature‑scaled step size
        factor = min(1.0, rng.random() * (T / cur_score + 0.1))
        delta = max_delta * factor
        if delta <= 0.0:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5_fft(new)
        diff = new_score - cur_score
        if diff <= 0.0 or rng.random() < np.exp(-diff / T):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        T *= 0.9995
        if T < 1e-6:
            T = 1e-6
    return best, best_score


def local_pair_moves(h, max_moves, rng):
    """Iterative pairwise mass transfers accepting only improvements."""
    best = h.copy()
    best_score = compute_c5_fft(best)
    n = best.size
    for _ in range(max_moves):
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best[i], 1.0 - best[j])
        if max_delta == 0.0:
            continue
        delta = max_delta * rng.random()
        best[i] -= delta
        best[j] += delta
        new_score = compute_c5_fft(best)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ---- main entry point ---------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # ---- initial construction -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5_fft(best_h)

    # ---- multi‑scale exploration ---------------------------------------------
    scales = [64, 128, 256, 512]
    for n in scales:
        if time_left() <= 0.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # annealing on current scale
        anneal_sec = min(10.0, time_left() * 0.25)
        if anneal_sec > 1.0:
            h_ann, sc_ann = simulated_annealing(base, anneal_sec, rng)
            if sc_ann < best_score:
                best_h, best_score = h_ann.copy(), sc_ann
            base = h_ann

        # local pairwise moves
        moves = int(min(30000, max(2000, time_left() * 200)))
        if moves > 0:
            h_pair, sc_pair = local_pair_moves(base, moves, rng)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        # occasional random balanced restart
        if time_left() > 5.0:
            for _ in range(2):
                if time_left() < 3.0:
                    break
                rand_h = random_balanced(n, rng)
                # brief annealing on random start
                sec = min(5.0, time_left() * 0.2)
                h_rand, sc_rand = simulated_annealing(rand_h, sec, rng)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand
                base = h_rand

    # ---- final refinement ------------------------------------------------------
    if time_left() > 5.0:
        sec = min(30.0, time_left() * 0.5)
        h_final, sc_final = simulated_annealing(best_h, sec, rng)
        if sc_final < best_score:
            best_h, best_score = h_final.copy(), sc_final

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
