def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l): return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)
```

### compute_c5_fast

```
def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)
```

Works. For n=512, dx=2/n.

import time, numpy as np, scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def gradient_of_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        n_s = n - shift
        term1 = np.zeros(n); term1[:n_s] = h[shift:]
        term2 = np.zeros(n); term2[shift:] = h[:n_s]
    else:
        s = -shift
        n_s = n - s
        term1 = np.zeros(n); term1[s:] = h[:n_s]
        term2 = np.zeros(n); term2[:n_s] = h[s:]
    return 1.0 - term1 - term2

def compute_gradient_max_shift(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    mx = corr.max()
    idxes = np.where(corr == mx)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += gradient_of_shift(h, s)
    grad /= len(shifts)
    return grad

def gradient_descent_momentum(h0, time_left, step0=0.5, beta=0.9, max_iter=200000):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h, best_s = h.copy(), compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.0:
        grad = compute_gradient_max_shift(h)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        s_new = compute_c5_fast(h_new)
        if s_new < best_s:
            best_s, best_h = s_new, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_s

def slsqp_optimize_candidate(h0, max_iter=800):
    n = h0.size
    x0 = np.concatenate([h0, [compute_c5_fast(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def eq_sum(x):
        return np.sum(x[:n]) - n / 2.0
    def ineq(x):
        h, t = x[:n], x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
