def binary_local_search(h0, time_left, max_swaps, init_temp, decay):
    n = h0.size
    k = n // 2
    # binarize
    idx = np.argpartition(-h0, k)[:k]
    h = np.zeros_like(h0, dtype=float)
    h[idx] = 1.0
    curr_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = curr_score
    T = init_temp
    for _ in range(max_swaps):
        if time_left() <= 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - curr_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(T*decay, 1e-12)
    return best_h, best_score

def adam_optimize_continuous(h0, time_left, max_iters, lr=0.2):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iters+1):
        if time_left() <= 0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= target:
                return best_h, best_score
        h = h_new

import time
import numpy as np
import scipy.optimize as opt

TARGET = 0.3808

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):
    return upsample(arr, new_n)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def adam_optimize_continuous(h0, time_left, max_iters, lr=0.2):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iters + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_scaled_fft(h)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            if best_score <= TARGET:
                return best_h, best_score
        h = h_new
    return best_h, best_score

def binary_local_search(h0, rng, time_left, max_swaps, init_temp=0.05, decay=0.9999):
    n = h0.size
    k = n // 2
    idx = np.argpartition(-h0, k)[:k]
    h = np.zeros_like(h0, dtype=float)
    h[idx] = 1.0
    curr_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = curr_score
    T = init_temp
    for _ in range(max_swaps):
        if time_left() <= 0.0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - curr_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * decay, 1e-12)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    best_h = None
    best_score = np.inf

    # Determine schedule of resolutions
    resolutions = [128, 256, 512]
    # If enough time, attempt a finer grid
    if time_left() > 300:
        resolutions.append(1024)

    for n in resolutions:
        if time_left() < 5.0:
            break

        # Prepare initial vector at this resolution
        if best_h is None:
            if 'initial_h_values' in globals():
                init = np.asarray(initial_h_values, dtype=float)
                if init.size != n:
                    init = upsample(init, n) if init.size < n else downsample(init, n)
            else:
                init = thue_morse(n)
            h = project_onto_box_sum(init, n / 2.0)
        else:
            h = upsample(best_h, n)
            h = project_onto_box_sum(h, n / 2.0)

        # Continuous Adam optimization
        max_iter_cont = int(min(8000, max(2000, time_left() * 20)))
        h_cont, score_cont = adam_optimize_continuous(h, time_left, max_iter_cont, lr=0.3)
        if score_cont < best_score:
            best_score, best_h = score_cont, h_cont.copy()
            if best_score <= TARGET:
                return best_h, best_score, best_h.size

        # Binary local search (hill climbing with simulated annealing)
        remaining = time_left()
        max_swaps = int(min(50000, max(5000, remaining * 2000)))
        h_bin, score_bin = binary_local_search(h_cont, rng, time_left, max_swaps,
                                               init_temp=0.04, decay=0.99995)
        if score_bin < best_score:
            best_score, best_h = score_bin, h_bin.copy()
            if best_score <= TARGET:
                return best_h, best_score, best_h.size

    # Final safety projection and evaluation
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        # Fallback: use Thue-Morse at 512 points
        n = 512
        best_h = thue_morse(n)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
