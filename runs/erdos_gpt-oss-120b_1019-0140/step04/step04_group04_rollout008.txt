import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr(h):
    n = h.shape[0]
    f = np.fft.rfft(h, n * 2)
    g = np.fft.rfft(1.0 - h, n * 2)
    corr = np.fft.irfft(f * np.conj(g), n * 2)
    return corr[:2 * n - 1]

def compute_c5(h):
    n = h.shape[0]
    return np.max(compute_corr(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    if not shifts.size:
        return np.zeros_like(h)
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += 1.0 - term1 - term2
    grad /= shifts.size
    return grad

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def round_to_binary(h):
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    out = np.zeros_like(h)
    out[idx] = 1.0
    return out

def binary_local_improve(h_bin, rng, max_trials=500):
    n = h_bin.size
    best = h_bin.copy()
    best_score = compute_c5(best)
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return best, best_score
    for _ in range(max_trials):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        sc = compute_c5(cand)
        if sc < best_score:
            best, best_score = cand, sc
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
    return best, best_score

def adam_optimize(h0, n_iter, lr, rng, time_left, best_h, best_score):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for t in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = compute_corr(h)
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        if t % 2000 == 0:
            h += rng.normal(scale=0.01, size=n)
        h = project_onto_box_sum(h, s)
        score = np.max(compute_corr(h)) * (2.0 / n)
        if score < best_score:
            best_h, best_score = h.copy(), score
        if best_score <= 0.3808:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = random_binary_balanced(64, rng)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h, best_score = base.copy(), compute_c5(base)

    # resolution schedule
    resolutions = [64, 128, 256, 512]
    target_iters = {64: 5000, 128: 10000, 256: 20000, 512: 40000}

    for n in resolutions:
        if time_left() < 5.0:
            break
        if best_h.shape[0] != n:
            best_h = upsample(best_h, n)
            best_h = project_onto_box_sum(best_h, n / 2.0)

        # binary local improvement of current best
        h_bin, sc_bin = binary_local_improve(round_to_binary(best_h), rng, max_trials=300)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin

        max_iters_possible = int(time_left() * 2500)
        n_iter = min(target_iters[n], max_iters_possible)
        if n_iter <= 0:
            continue

        lr = 0.5 if n <= 256 else 0.3
        best_h, best_score = adam_optimize(best_h, n_iter, lr, rng, time_left, best_h, best_score)

        # another binary refinement
        h_bin, sc_bin = binary_local_improve(round_to_binary(best_h), rng, max_trials=500)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin

        if best_score <= 0.3808:
            break

    # final stochastic refinement
    while time_left() > 1.0:
        h_rand = random_binary_balanced(best_h.size, rng)
        sc_rand = compute_c5(h_rand)
        if sc_rand < best_score:
            best_h, best_score = h_rand, sc_rand
        i = rng.integers(best_h.size)
        j = rng.integers(best_h.size)
        if best_h[i] > 0 and best_h[j] < 1:
            delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.5)
            cand = best_h.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = compute_c5(cand)
            if sc < best_score:
                best_h, best_score = cand, sc
        if rng.random() < 0.2:
            pert = rng.normal(scale=0.015, size=best_h.size)
            cand = best_h + pert
            cand = project_onto_box_sum(cand, best_h.size / 2.0)
            sc = compute_c5(cand)
            if sc < best_score:
                best_h, best_score = cand, sc
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
