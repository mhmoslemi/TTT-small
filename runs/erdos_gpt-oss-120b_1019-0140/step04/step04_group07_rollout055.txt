def simulated_annealing_binary(h0, rng, time_left):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # set temperature schedule
    # maximum possible change in C5 per swap is at most 2 * dx (since a swap can affect up to two contributions for each shift?), but safe bound: 2*dx.
    dx = 2.0 / n
    max_delta = 2*dx
    T0 = max_delta * 0.5  # initial temperature
    T_min = max_delta * 1e-4
    # compute total iterations allowed approx by time budget
    start = time.time()
    it = 0
    while time_left() > 0:
        it += 1
        # schedule temperature linearly decreasing
        fraction = min(1.0, it * 1.0 / 500000)  # adjust the denominator for total iteration goal
        T = T0 * (1 - fraction) + T_min * fraction

        # pick two positions with opposite values
        # pick i where cur_h[i]==1, j where cur_h[j]==0
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0

        # optionally early exit if best_score already below target
        # but we will continue for more improvement
    return best_h, best_score
```

def gradient_descent_refine(h0, rng, time_left, max_iters=20000):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    step = 0.1
    for it in range(max_iters):
        if time_left() <= 0:
            break
        # compute correlation and worst shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        # compute gradient (average of gradients for worst shifts)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                # term1 indices where h_i appears as first factor
                grad[:n-shift] += (1.0 - h[shift:])
                # term2 indices where h_i appears as second factor (negative)
                grad[shift:] -= h[:n-shift]
            else:
                s_shift = -shift
                grad[s_shift:] += (1.0 - h[:n - s_shift])
                grad[:n - s_shift] -= h[s_shift:]
        grad /= len(shifts) if len(shifts)>0 else 1
        # update
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        # adapt step
        if new_score < compute_c5(h):
            step = min(step * 1.1, 0.5)
            h = h_new
        else:
            step *= 0.5
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        total = proj.sum()
        if abs(total - s) < 1e-12:
            return proj
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2 * n - 1]
    return np.max(corr) * dx

def max_corr_shifts(h):
    """Return max correlation value and list of shifts attaining it."""
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)[:2 * n - 1]
    max_corr = np.max(corr)
    idx = np.where(corr >= max_corr - 1e-12)[0]
    shifts = idx - (n - 1)
    return max_corr, shifts

def simulated_annealing_binary(h0, rng, start, budget):
    """Binary simulated annealing with swap moves."""
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    dx = 2.0 / n
    max_delta = 2 * dx
    T0 = max_delta * 0.5
    T_min = max_delta * 1e-4
    it = 0
    # leave ~10 s for later refinement
    while True:
        if time.time() - start >= budget - 10:
            break
        it += 1
        fraction = min(1.0, it / 500000.0)
        T = T0 * (1 - fraction) + T_min * fraction
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        accept = (delta <= 0) or (rng.random() < np.exp(-delta / T))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_score

def gradient_descent_refine(h0, rng, start, budget):
    """Projected (sub)gradient descent on the worst shifts."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    step = 0.1
    while True:
        if time.time() - start >= budget - 5:
            break
        max_corr, shifts = max_corr_shifts(h)
        grad = np.zeros_like(h)
        for shift in shifts:
            if shift >= 0:
                grad[:n - shift] += (1.0 - h[shift:])
                grad[shift:] -= h[:n - shift]
            else:
                sft = -shift
                grad[sft:] += (1.0 - h[:n - sft])
                grad[:n - sft] -= h[sft:]
        if shifts.size > 0:
            grad /= shifts.size
        h_new = h - step * grad
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.5
            if step < 1e-6:
                step = 0.1
    return best_h, best_score

def local_pair_moves(h0, rng, start, budget):
    """Small random mass transfers to fine‑tune."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.shape[0]
    while True:
        if time.time() - start >= budget:
            break
        i, j = rng.integers(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h = h_new
            best_score = score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    # base vector
    if 'initial_h_values' in globals():

    else:
        base = rng.random(64)
    base = upsample(base, n)
    # binary start
    h_bin = (base >= 0.5).astype(float)
    target = n // 2
    cur_sum = int(h_bin.sum())
    diff = cur_sum - target
    if diff > 0:
        idx = np.where(h_bin == 1)[0]
        to_zero = rng.choice(idx, size=diff, replace=False)
        h_bin[to_zero] = 0
    elif diff < 0:
        idx = np.where(h_bin == 0)[0]
        to_one = rng.choice(idx, size=-diff, replace=False)
        h_bin[to_one] = 1
    if h_bin.sum() != target:
        h_bin = project_onto_box_sum(h_bin, target)
    # simulated annealing (binary)
    h_sa, score_sa = simulated_annealing_binary(h_bin, rng, start, budget_s)
    # gradient descent refinement (continuous)
    if time.time() - start < budget_s - 5:
        h_gd, score_gd = gradient_descent_refine(h_sa.astype(float), rng, start, budget_s)
    else:
        h_gd, score_gd = h_sa, score_sa

        h_final, score_final = local_pair_moves(h_gd, rng, start, budget_s)
    else:
        h_final, score_final = h_gd, score_gd
    # final projection and evaluation
    h_final = project_onto_box_sum(h_final, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
