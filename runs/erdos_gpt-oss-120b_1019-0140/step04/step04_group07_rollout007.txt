def compute_corr_fft(h):
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - h
    # reverse b for convolution: correlation = conv(a, b_rev)
    b_rev = b[::-1]
    m = 2*n - 1
    # compute FFT length as next power of two for speed
    nfft = 1 << ((m-1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b_rev, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[:m]
    # conv length m
    # correlation output should be same ordering as np.correlate: shift k from -(n-1) to (n-1)
    # np.correlate returns array where index i corresponds to shift i-(n-1).
    return conv
```

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n/2
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    temp = 0.01  # initial temperature for occasional worse acceptance
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        # compute max C5 (i.e., max corr)
        max_corr = np.max(corr)
        # find shifts achieving max (worst)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        # shift values are shift_idxs - (n-1)
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp):
            # accept
            h = h_new
            cur_score = new_score
            temp *= 0.9995  # decrease temperature slightly
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # optional jitter after certain iterations
        if (it+1) % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_corr_fft(h):
    """Full correlation of h with (1-h) using FFT, matching np.correlate."""
    n = h.shape[0]
    a = h.astype(np.float64)
    b = (1.0 - h).astype(np.float64)
    b_rev = b[::-1]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b_rev, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[:m]
    return conv[::-1]  # reverse to obtain correlation ordering

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Gradient of max‑correlation term for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= len(shifts)
    return grad

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum, adaptive step, occasional uphill moves."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    temp = 0.02
    decay = 0.9995
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = compute_corr_fft(h)
        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shifts = shift_idxs - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                accept = True
        if accept:
            h = h_new
            cur_score = new_score
            temp = max(temp * decay, 1e-8)
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if (it % 2000) == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    """Transfer mass from points that most contribute to the worst shift."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0.5:
        corr = compute_corr_fft(best_h)
        max_corr = np.max(corr)
        idxs = np.where(corr >= max_corr - 1e-12)[0]
        shift = rng.choice(idxs) - (n - 1)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.9
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    """Swap small amounts of mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0.3:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            s = compute_c5(h_new)
            if s < best_score:
                best_h, best_score = h_new, s
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            s2 = compute_c5(h_new2)
            if s2 < best_score:
                best_h, best_score = h_new2, s2
    return best_h, best_score

def simulated_annealing_moves(h0, rng, time_left):
    """Random mass transfers with temperature schedule."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    T = 0.02
    decay = 0.9995
    while time_left() > 0.1:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h[i] -= delta
        h[j] += delta
        new_score = compute_c5(h)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        T = max(T * decay, 1e-8)
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.005, size=n)
            h = project_onto_box_sum(h + jitter, n / 2.0)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

        # Gradient descent at current scale
        if time_left() > 5:
            max_iters = int(min(200000, max(5000, time_left() * 50)))
            h_gd, sc_gd = gradient_descent_adaptive(base, max_iters,
                                                    step0=1.0, beta=0.9,
                                                    rng=rng, time_left=time_left)
            if sc_gd < best_score:
                best_h, best_score = h_gd.copy(), sc_gd
            base = h_gd

        # Targeted worst‑shift improvement
        if time_left() > 5:
            h_ms, sc_ms = improve_via_max_shift(base, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        # Local pairwise moves
        if time_left() > 5:
            h_pair, sc_pair = local_pair_moves(base, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        # Simulated annealing style random moves
        if time_left() > 5:
            h_sa, sc_sa = simulated_annealing_moves(base, rng, time_left)
            if sc_sa < best_score:
                best_h, best_score = h_sa.copy(), sc_sa
            base = h_sa

    # Final intensive refinement on the overall best
    if time_left() > 5:
        max_iters = int(min(300000, max(50000, time_left() * 80)))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters,
                                                      step0=0.5, beta=0.9,
                                                      rng=rng, time_left=time_left)

    # One more round of directed improvements
    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
