def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # parameters
    TARGET = 0.3808
    best_h = None
    best_score = np.inf
    best_n = None

    # function definitions: thue_morse, project_onto_box_sum, upsample, compute_c5_fft, compute_corr_full_fft, compute_grad_max_shifts, gradient_descent

    # Stage 1: try binary Thue-Morse at multiple n
    for n in [128, 256, 512, 1024, 2048]:
        if time_left() <= 1.0:
            break
        h = thue_morse(n) # binary
        # Can optionally random perturb small to break ties? Keep binary now
        score = compute_c5_fft(h.astype(float))
        if score < best_score:
            best_score, best_h, best_n = score, h.astype(float).copy(), n
            if best_score <= TARGET:
                break

    # Stage 2: hill climbing with swaps
    if best_h is not None and time_left() > 5.0:
        n = best_n
        # precompute indices arrays
        zero_idx = np.where(best_h == 0)[0]
        one_idx = np.where(best_h == 1)[0]
        # If either empty, break (not expected)
        while time_left() > 2.0:
            # choose random indices
            i = rng.choice(zero_idx)
            j = rng.choice(one_idx)
            # swap
            best_h[i], best_h[j] = best_h[j], best_h[i]
            # update zero_idx/one_idx quickly: we can just recompute after each swap, or for simplicity recompute each iteration (cost O(n)). Might be okay for n up to 2048.
            # Evaluate
            new_score = compute_c5_fft(best_h)
            if new_score < best_score - 1e-12:
                best_score = new_score
                # update idx arrays
                zero_idx = np.where(best_h == 0)[0]
                one_idx = np.where(best_h == 1)[0]
                if best_score <= TARGET:
                    break
            else:
                # revert swap
                best_h[i], best_h[j] = best_h[j], best_h[i]
                # maybe don't update idx
                # continue
        # after hill climbing, we can also try random binary swaps with small fraction changes and projection using gradient descent.

    # Stage 3: real-valued gradient descent on current best (maybe after upsampling)
    if best_h is not None and time_left() > 15.0:
        # upscale to higher resolution if possible
        # Use next scale factor if time permits
        for n_target in [best_n*2, best_n*4]:
            if time_left() <= 5.0:
                break
            if n_target > 4096: # limit size
                continue
            # upsample current best to new n_target
            h_upsampled = upsample(best_h, n_target)
            # project onto box sum exactly
            h_upsampled = project_onto_box_sum(h_upsampled, n_target/2.0)
            # set iteration budget proportionally to n_target
            max_iter = int(200000 * (n_target / 512))
            # ensure we have time left for this many iterations; we can also restrict by time
            # Let's perform gradient descent until time runs out or max_iter
            h_opt, sc_opt = gradient_descent(
                h_upsampled,
                n_iter=max_iter,
                step0=0.6,
                beta=0.9,
                rng=rng,
                time_left=time_left
            )
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                best_n = n_target
                if best_score <= TARGET:
                    break

    # As fallback, if best_h is None (should not happen), use thue_morse(128) etc.

    # Final projection
    if best_h is not None:

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
