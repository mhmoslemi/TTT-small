def compute_smooth_gradient(h, p=8, top_frac=0.05):
    n = h.size
    f_vec = np.correlate(h, 1.0 - h, mode='full')
    weights = p * np.maximum(f_vec, 0.0) ** (p - 1)
    top_n = max(5, int(top_frac * weights.size))
    # get indices of top weights
    idx = np.argpartition(weights, -top_n)[-top_n:]
    grad = np.zeros_like(h)
    for ind in idx:
        w = weights[ind]
        k = int(ind) - (n - 1)
        # shift arrays
        if k != 0:
            # shift left by k (positive) else shift right
            if k > 0:
                h_shift_k = np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
                h_shift_minus_k = np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))
            else:
                kk = -k
                h_shift_k = np.concatenate((np.zeros(kk, dtype=h.dtype), h[:n - kk]))
                h_shift_minus_k = np.concatenate((h[kk:], np.zeros(kk, dtype=h.dtype)))
        else:
            h_shift_k = h
            h_shift_minus_k = h
        grad += w * (1.0 - h_shift_k - h_shift_minus_k)
    return grad

def gradient_descent_smooth(h0, max_iters, step0, beta, p, top_frac, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    step = step0
    prev_score = best_score
    # compute period for jitter
    jitter_interval = max(200, max_iters // 10)
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        grad = compute_smooth_gradient(h, p, top_frac)
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        # adapt step
        if new_score <= prev_score:
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        prev_score = new_score
        h = h_new
        # jitter
        if it % jitter_interval == 0:
            # add small noise
            h = project_onto_box_sum(h + rng.normal(scale=0.005, size=n), s)
            prev_score = compute_c5(h)
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # Transfer delta
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random()
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
                continue  # continue to next iteration
        # try opposite direction
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random()
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    # After loop return
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1) as float."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def shift_array(arr, k):
    """Shift arr by k positions (positive = left), fill with zeros."""
    n = arr.size
    if k > 0:
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    if k < 0:
        k = -k
        out = np.empty_like(arr)
        out[k:] = arr[: n - k]
        out[:k] = 0.0
        return out
    return arr.copy()


def compute_smooth_gradient(h, p=8, top_frac=0.05):
    """Gradient of the p‑norm smooth approximation of max_k f_k."""
    n = h.size
    f_vec = np.correlate(h, 1.0 - h, mode='full')
    weights = p * np.maximum(f_vec, 0.0) ** (p - 1)
    top_n = max(5, int(top_frac * weights.size))
    idx = np.argpartition(weights, -top_n)[-top_n:]
    grad = np.zeros_like(h)
    for ind in idx:
        w = weights[ind]
        k = int(ind) - (n - 1)
        grad += w * (1.0 - shift_array(h, k) - shift_array(h, -k))
    return grad


def gradient_descent_smooth(h0, max_iters, step0, beta, p, top_frac, rng, time_remaining):
    """Projected gradient descent using smooth max gradient."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    momentum = np.zeros_like(h)
    step = step0
    prev_score = best_score
    jitter_interval = max(200, max_iters // 10)

    for it in range(1, max_iters + 1):
        if time_remaining() <= 0:
            break
        grad = compute_smooth_gradient(h, p=p, top_frac=top_frac)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        if new_score <= prev_score:
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        prev_score = new_score
        h = h_new

        if it % jitter_interval == 0:
            h = project_onto_box_sum(h + rng.normal(scale=0.005, size=n), s)
            prev_score = compute_c5(h)

    return best_h, best_score


def local_pair_moves(h, rng, time_remaining):
    """Transfer small mass between random positions, accept improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_remaining() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random()
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
                continue
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random()
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def improve_via_max_shift(h, rng, time_remaining):
    """Move mass directly from points that contribute most to the worst lag."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_remaining() > 0:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[: n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[: n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score
    return best_h, best_score


def binary_round(h):
    """Round to a balanced binary vector (exactly half ones)."""
    n = h.size
    idx = np.argpartition(-h, n // 2 - 1)[: n // 2]
    h_bin = np.zeros_like(h)
    h_bin[idx] = 1.0
    return h_bin


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_remaining():
        return budget_s - (time.time() - start)

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_remaining() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # gradient descent with smooth max
        max_iters = int(min(15000, max(2000, time_remaining() * 10 / n)))
        p = 6 if n <= 256 else 8
        step0 = 0.8 if n <= 256 else 0.5
        h_gd, sc_gd = gradient_descent_smooth(base, max_iters, step0, 0.9,
                                               p, 0.05, rng, time_remaining)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd
        base = h_gd

        # targeted max‑shift moves
        if time_remaining() > 2:
            h_ms, sc_ms = improve_via_max_shift(base, rng, time_remaining)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms
            base = h_ms

        # local pair‑wise moves
        if time_remaining() > 2:
            h_pair, sc_pair = local_pair_moves(base, rng, time_remaining)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair
            base = h_pair

        # random binary restarts with brief refinement
        while time_remaining() > 5:
            h_rand = random_binary_balanced(n, rng)
            rand_iters = int(min(4000, max(500, time_remaining() * 3 / n)))
            h_rand_ref, sc_rand = gradient_descent_smooth(h_rand, rand_iters,
                                                          0.6, 0.9, 6, 0.07,
                                                          rng, time_remaining)
            if sc_rand < best_score:
                best_h, best_score = h_rand_ref.copy(), sc_rand
            # short local search
            if time_remaining() > 2:
                h_rand2, sc_rand2 = local_pair_moves(h_rand_ref, rng, time_remaining)
                if sc_rand2 < best_score:
                    best_h, best_score = h_rand2.copy(), sc_rand2
            # exit if not enough time for another restart
            if time_remaining() < 4:
                break

    # final intensive refinement on the global best
    if time_remaining() > 5:
        final_iters = int(min(60000, max(20000, time_remaining() * 30)))
        best_h, best_score = gradient_descent_smooth(best_h, final_iters,
                                                     0.3, 0.9, 10, 0.04,
                                                     rng, time_remaining)

    # attempt binary rounding and final local optimization
    if time_remaining() > 2:
        h_bin = binary_round(best_h)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_h, best_score = h_bin, sc_bin
        if time_remaining() > 2:
            best_h, best_score = local_pair_moves(best_h, rng, time_remaining)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
