        h = h_new
        it += 1

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_corr_raw(h):
    n = h.shape[0]
    h_rev = h[::-1]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    return conv[:2 * n - 1]

def smooth_max_and_grad(h, alpha):
    n = h.shape[0]
    corr = compute_corr_raw(h)                # length 2n-1
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))     # stable weights
    sum_w = w.sum()
    wg = w / sum_w                            # normalized weights, sum to 1
    # one_weight: for each i, sum_{s=-i}^{n-1-i} wg_s
    prefix = np.concatenate([[0.0], np.cumsum(wg)])
    i_arr = np.arange(n)
    a = n - 1 - i_arr
    b = 2 * n - 2 - i_arr
    one_weight = prefix[b + 1] - prefix[a]
    # term1 = sum_s wg_s * h[i + s]; term2 = sum_s wg_s * h[i - s]
    len_conv = 3 * n - 2
    N = 1 << ((len_conv).bit_length())
    fft_h = np.fft.rfft(h, n=N)
    fft_wg = np.fft.rfft(wg, n=N)
    fft_wg_rev = np.fft.rfft(wg[::-1], n=N)
    conv_fwd = np.fft.irfft(fft_wg * fft_h, n=N)[:len_conv]      # h[i - s]
    conv_rev = np.fft.irfft(fft_wg_rev * fft_h, n=N)[:len_conv]  # h[i + s]
    term2 = conv_fwd[n - 1:n - 1 + n]
    term1 = conv_rev[n - 1:n - 1 + n]
    grad = one_weight - term1 - term2
    smooth_max = max_corr + np.log(sum_w) / alpha
    return smooth_max, grad

def smooth_gradient_descent(h0, time_left, max_iters=20000, alpha=20.0, lr0=0.5, beta=0.9):
    n = h0.size
    target = n / 2.0
    h = project_onto_box_sum(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iters and time_left() > 0.0:
        smooth_val, grad = smooth_max_and_grad(h, alpha)
        mom = beta * mom + (1.0 - beta) * grad
        eta = lr0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c5

def thue_morse_binary(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def random_real_balanced(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_swap_anneal(h0, rng, time_left, max_iters=200000):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_c5 = h.copy(), cur
    T, T_min, decay = 0.2, 1e-5, 0.99998
    it = 0
    while it < max_iters and time_left() > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_c5:
                best_c5, best_h = new, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_c5
