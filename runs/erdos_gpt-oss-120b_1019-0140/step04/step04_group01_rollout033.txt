import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto { lo<=x_i<=hi, sum x_i = s }."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search on lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 for step function h (dx = 2/n)."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max_{k} corr_k w.r.t. h, averaging over given shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def upsample_interpolate(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def binary_anneal(h_init, rng, time_left_func,
                 max_iters=500000, T0=0.2, decay=0.99995, Tmin=1e-5):
    """Simulated annealing on binary vectors via swaps."""
    h = h_init.copy()
    n = h.size
    score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = score
    T = T0
    it = 0
    while it < max_iters and time_left_func() > 0.02:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = compute_c5_fft(h)
        delta = new_score - score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
        it += 1
    return best_h, best_score

        h_current = random_binary_balanced(n, rng)

    best_h = h_current.copy()
    best_score = compute_c5_fft(best_h)

    # ----- Phase 1: continuous RMSProp -----
    max_iter_cont = 50000
    rms = np.zeros_like(best_h)
    lr0 = 0.5
    decay_rms = 0.9
    eps = 1e-8
    it = 1
    while it <= max_iter_cont and time_left() > 5.0:
        corr = compute_corr_scaled_fft(h_current)
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h_current, shifts)

        rms = decay_rms * rms + (1 - decay_rms) * grad * grad
        adjusted = grad / (np.sqrt(rms) + eps)
        lr = lr0 / np.sqrt(it)
        h_new = h_current - lr * adjusted
        h_new = project_onto_box_sum(h_new, n / 2.0)

        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

        h_current = h_new
        it += 1

    # ----- Phase 2: binary conversion & annealing at n=512 -----
    # Convert best continuous to binary by thresholding top half entries
    idx_top = np.argpartition(-best_h, n // 2)[:n // 2]
    h_bin = np.zeros_like(best_h)
    h_bin[idx_top] = 1.0
    bin_score = compute_c5_fft(h_bin)
    if bin_score < best_score:
        best_score, best_h = bin_score, h_bin.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # Anneal binary
    h_bin_opt, bin_opt_score = binary_anneal(best_h, rng, time_left,
                                            max_iters=400000, T0=0.3,
                                            decay=0.9999, Tmin=1e-6)
    if bin_opt_score < best_score:
        best_score, best_h = bin_opt_score, h_bin_opt.copy()
        if best_score <= target:
            return best_h, best_score, best_h.size

    # ----- Phase 3: upscale to n=1024 and refine -----
    if time_left() > 15.0:
        n2 = 1024
        h_cont_up = upsample_interpolate(best_h, n2)
        h_cont_up = project_onto_box_sum(h_cont_up, n2 / 2.0)

        # binary conversion at higher resolution
        idx_top2 = np.argpartition(-h_cont_up, n2 // 2)[:n2 // 2]
        h_bin2 = np.zeros_like(h_cont_up)
        h_bin2[idx_top2] = 1.0
        bin2_score = compute_c5_fft(h_bin2)
        if bin2_score < best_score:
            best_score, best_h = bin2_score, h_bin2.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

        # anneal at higher resolution
        h_bin2_opt, bin2_opt_score = binary_anneal(best_h, rng, time_left,
                                                  max_iters=600000,
                                                  T0=0.3, decay=0.9999,
                                                  Tmin=1e-6)
        if bin2_opt_score < best_score:
            best_score, best_h = bin2_opt_score, h_bin2_opt.copy()
            if best_score <= target:
                return best_h, best_score, best_h.size

    # ----- Phase 4: random binary restarts with limited hill climbing -----
    while time_left() > 6.0:
        # generate random balanced binary vector
        h_rand = random_binary_balanced(best_h.size, rng)
        # quick greedy hill climb (no temperature) for a few swaps
        score = compute_c5_fft(h_rand)
        improved = True
        climbs = 0
        while improved and time_left() > 1.0 and climbs < 5000:
            improved = False
            ones = np.where(h_rand > 0.5)[0]
            zeros = np.where(h_rand <= 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_rand[i] = 0.0
            h_rand[j] = 1.0
            new_score = compute_c5_fft(h_rand)
            if new_score < score:
                score = new_score
                improved = True
                if score < best_score:
                    best_score, best_h = score, h_rand.copy()
            else:
                h_rand[i] = 1.0
                h_rand[j] = 0.0
            climbs += 1
        if best_score <= target:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5_fft(best_h)
    return best_h, best_score, best_h.size
