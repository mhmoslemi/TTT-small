def binary_anneal(h0, rng, time_left, max_iters=2000000, init_T=0.5, decay=0.99995, T_min=1e-6):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = init_T
    it = 0
    n = h.size
    # Precompute indices of ones and zeros each iteration.
    while it < max_iters and time_left() > 0.0:
        # choose random i in ones, j in zeros
        # we can store ones and zeros list but update after each swap might be expensive.
        # simpler: sample random i; if h[i]==1 then find random j where h[j]==0 (loop). Could be slower but okay for small n.

        i = rng.integers(n)
        # Ensure i is a 1; else find next
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=50):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    # start with bounds that guarantee f(lam_low) >= s >= f(lam_high)
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > s:
        lam_high += (hi - lo) * 10
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fast(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    # convolution of reversed h with (1-h)
    conv = np.fft.irfft(
        np.fft.rfft(h[::-1], n=N) * np.fft.rfft(1.0 - h, n=N),
        n=N,
    )
    corr = conv[: 2 * n - 1]
    return corr.max() * (2.0 / n)

def thue_morse_binary(n):
    # generate Thue-Morse bits for length n (n must be power of two)
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def resample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_anneal(h0, rng, time_left, max_iters=2000000,
                 init_T=0.5, decay=0.99996, T_min=1e-6):
    h = h0.copy()
    cur = compute_c5_fast(h)
    best_h, best_s = h.copy(), cur
    T = init_T
    n = h.shape[0]
    it = 0
    while it < max_iters and time_left() > 0.0:
        # pick a 1 and a 0
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5_fast(h)
        delta = new - cur
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            if new < best_s:
                best_s, best_h = new, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_s
