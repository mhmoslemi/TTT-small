def binary_annealing(h0, rng, time_left, max_iters=10_000_000, T0=0.5, decay=0.99998, Tmin=1e-6, reheat_interval=200_000, reheat_factor=2):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5_fft(h)
    best_h = h.copy()
    best_score = cur_score
    T = T0
    it = 0
    last_improve_it = 0
    while it < max_iters and time_left() > 0.0:
        # sample swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = compute_c5_fft(h)
        delta = new_score - cur_score
        # accept if improvement or with prob
        if delta < 0 or (T > Tmin and rng.random() < np.exp(-delta / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                last_improve_it = it
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # update temperature
        T = max(Tmin, T * decay)
        # reheat if no improvement for long
        if it - last_improve_it > reheat_interval:
            T = min(T0, T * reheat_factor)  # increase temperature up to T0?
            last_improve_it = it
        it += 1
    return best_h, best_score
```

def compute_self_corr_fft(h):
    n = h.shape[0]
    M = 2*n

    corr = np.fft.irfft(H * np.conj(H), n=M)[:2*n-1]

import time
import numpy as np

def resample(arr, new_n):
    """Interpolate arr (size old_n) to new_n points on [0,1)."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_self_corr_fft(h):
    """Self‑correlation of binary/continuous h, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    corr = np.fft.irfft(H * np.conj(H), n=M)[: 2 * n - 1]
    return corr * (2.0 / n)

def binary_annealing(
    h0,
    rng,
    time_left_func,
    max_iters=10**9,
    T0=0.5,
    decay=0.999985,
    Tmin=1e-6,
    reheat_interval=50000,
    reheat_factor=2.0,
    target=None,
):
    """Simulated annealing on balanced binary vectors."""
    h = h0.astype(float).copy()
    n = h.shape[0]
    self_corr = compute_self_corr_fft(h)
    cur_score = 1.0 - np.min(self_corr)  # C5 = 1 - min self‑correlation
    best_h = h.copy()
    best_score = cur_score
    T = T0
    it = 0
    last_improve = 0
    while it < max_iters and time_left_func() > 0.0:
        # worst shift for current configuration
        kw = np.argmin(self_corr) - (n - 1)  # integer shift
        shift_plus = (np.arange(n) + kw) % n

        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]

        # candidates that improve the worst shift
        partner_one = h[shift_plus[ones_idx]]
        cand_i = ones_idx[partner_one == 0.0]

        partner_zero = h[shift_plus[zeros_idx]]
        cand_j = zeros_idx[partner_zero == 1.0]

        if cand_i.size > 0 and cand_j.size > 0:
            i = rng.choice(cand_i)
            j = rng.choice(cand_j)
        else:
            # fallback random swap
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        # evaluate new configuration
        new_self_corr = compute_self_corr_fft(h)
        new_score = 1.0 - np.min(new_self_corr)
        delta = new_score - cur_score

        accept = False
        if delta < 0.0:
            accept = True
        elif T > Tmin and rng.random() < np.exp(-delta / T):
            accept = True

        if accept:
            self_corr = new_self_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                last_improve = it
            if target is not None and new_score <= target:
                return best_h, best_score
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        # temperature schedule
        T = max(Tmin, T * decay)
        if it - last_improve > reheat_interval:
            T = min(T0, T * reheat_factor)
            last_improve = it

        it += 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    n = 512
    # obtain starting vector
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            init = resample(init, n)
    else:
        init = rng.random(n)

    # balanced binary rounding (exactly n/2 ones)
    idx = np.argpartition(-init, n // 2)[: n // 2]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0

    # initial score
    init_self_corr = compute_self_corr_fft(h_bin)
    init_score = 1.0 - np.min(init_self_corr)
    best_h = h_bin.copy()
    best_score = init_score

    target = 0.3808
    if best_score <= target:
        return best_h, best_score, n

    # main annealing phase
    best_h, best_score = binary_annealing(
        best_h,
        rng,
        time_left,
        T0=0.6,
        decay=0.99999,
        Tmin=1e-7,
        reheat_interval=40000,
        reheat_factor=2.0,
        target=target,
    )

    # final verification
    final_corr = compute_self_corr_fft(best_h)
    final_score = 1.0 - np.min(final_corr)
    best_score = final_score
    best_h = best_h.astype(float)

    return best_h, best_score, n
