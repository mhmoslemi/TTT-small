def adam_descent(h0, n_iter, lr0, beta1, beta2, eps, rng, time_left):

    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    for t in range(1, n_iter+1):
        if time_left() <= 0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # Occasionally add noise to escape flat regions
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def local_swap_search(h, best_score, rng, time_left, max_attempts=5000):
    n = h.size
    while max_attempts > 0 and time_left() > 0:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # Compute max possible delta
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random() * 0.2  # up to 20% of max
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score = sc
            h = h_new
            # Reset attempts
            max_attempts = 5000
        max_attempts -= 1
    return h, best_score

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = None

    resolutions = [128, 256, 512, 1024]
    TARGET = 0.3808
    for n in resolutions:
        if time_left() < 5.0: break
        # Upsample base if needed
        if base.size != n:
            base = upsample(base, n)

        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds enclose root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    return np.fft.ifft(fh * np.conj(f1mh)).real

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad

def adam_descent(h0, n_iter, lr0, beta1, beta2, eps, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(best_h)
    for t in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        if t % 5000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def local_swap_search(h, best_score, rng, time_left, max_attempts=5000):
    n = h.size
    attempts = max_attempts
    while attempts > 0 and time_left() > 0.0:
        i, j = rng.integers(n, size=2)
        if i == j:
            attempts -= 1
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            attempts -= 1
            continue
        delta = max_delta * rng.random() * 0.2
        if delta <= 0.0:
            attempts -= 1
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score = sc
            h = h_new
            attempts = max_attempts  # reset counter after improvement
        else:
            attempts -= 1
    return h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # initial base vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128)
    base = project_onto_box_sum(base, base.size / 2.0)

    best_h = None
    best_score = np.inf
    TARGET = 0.3808
    resolutions = [128, 256, 512, 1024]

    for n in resolutions:
        if time_left() < 5.0:
            break
        if base.size != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # choose parameters per size
        if n == 128:
            restarts, n_iter = 8, 150000
        elif n == 256:
            restarts, n_iter = 6, 150000
        elif n == 512:
            restarts, n_iter = 4, 150000
        else:  # 1024
            restarts, n_iter = 2, 100000

        for r in range(restarts):
            if time_left() < 3.0:
                break
            # initial point
            if rng.random() < 0.5:
                h0 = random_binary_balanced(n, rng).astype(float)
            else:
                h0 = base + rng.normal(scale=0.05, size=n)
                h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, n / 2.0)

            h_opt, sc_opt = adam_descent(
                h0, n_iter=n_iter, lr0=0.3,
                beta1=0.9, beta2=0.999, eps=1e-8,
                rng=rng, time_left=time_left)

            h_opt, sc_opt = local_swap_search(
                h_opt, sc_opt, rng, time_left, max_attempts=5000)

            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_h is not None:
            base = best_h.copy()
            if best_score <= TARGET:
                break

    if best_h is None:
        n_fallback = 512
        best_h = project_onto_box_sum(thue_morse(n_fallback), n_fallback / 2.0)
        best_score = compute_c5_fft(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
