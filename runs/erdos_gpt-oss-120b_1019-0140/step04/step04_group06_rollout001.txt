def mutate(child, mutation_rate, rng):
    if mutation_rate <= 0:
        return child
    mask = rng.random(child.size) < mutation_rate
    child[mask] = 1 - child[mask]  # flip bits
    return child

Repair:

```
def repair_binary(v, target_ones, rng):
    # Ensure exactly target_ones ones
    current_ones = v.sum()
    diff = int(target_ones - current_ones)
    if diff > 0:  # need to add ones
        zero_indices = np.where(v == 0)[0]
        if len(zero_indices) > 0:
            chosen = rng.choice(zero_indices, size=diff, replace=False)
            v[chosen] = 1
    elif diff < 0:  # need to remove ones
        one_indices = np.where(v == 1)[0]
        if len(one_indices) > 0:
            chosen = rng.choice(one_indices, size=-diff, replace=False)
            v[chosen] = 0
    return v
```

def compute_c5(h):

    return np.max(corr) * (2.0 / n)

def genetic_algorithm(initial_h, n, time_left):
    rng = np.random.default_rng()
    target_ones = n // 2
    popsize = 50
    tournament_k = 5
    mutation_rate = 0.03
    # Initialize population
    population = []
    scores = []
    cache = {}
    # Use initial_h as one of the individuals: convert to binary if not already.
    # initial_h may be continuous; we can binarize by threshold 0.5 then repair.
    init_bin = np.where(initial_h >= 0.5, 1, 0).astype(np.int8)
    init_bin = repair_binary(init_bin, target_ones, np.random.default_rng())
    # Fill rest random
    while len(population) < popsize:
        if len(population) == 0:
            individual = init_bin.copy()
        else:
            # random binary balanced
            individual = np.zeros(n, dtype=np.int8)
            ones_idx = rng.choice(n, size=target_ones, replace=False)
            individual[ones_idx] = 1
        key = individual.tobytes()
        if key in cache:
            score = cache[key]
        else:
            score = compute_c5(individual)
            cache[key] = score
        population.append(individual)
        scores.append(score)
    # Main loop
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]
    # time loop
    while time_left() > 0.5:
        new_pop = [best_h]  # keep elite
        new_scores = [best_score]
        # generate rest popsize-1 individuals
        while len(new_pop) < popsize:
            # tournament selection for two parents
            idx1 = tournament_select(scores, tournament_k, rng)
            idx2 = tournament_select(scores, tournament_k, rng)
            p1 = population[idx1]
            p2 = population[idx2]
            child = crossover(p1, p2, rng)
            child = mutate(child, mutation_rate, rng)
            child = repair_binary(child, target_ones, rng)
            key = child.tobytes()
            if key in cache:
                sc = cache[key]
            else:
                sc = compute_c5(child)
                cache[key] = sc
            new_pop.append(child)
            new_scores.append(sc)
            if sc < best_score:
                best_score = sc
                best_h = child.copy()
        # replace
        population = new_pop
        scores = new_scores
    return best_h, best_score, cache

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

    tournament_k = 5
    mutation_rate = 0.04

    pop = []

            ind[ones_idx] = 1

        scores.append(sc)
    best_idx = int(np.argmin(scores))

    while time_left() > 0.5:
        new_pop = [best_h]

        while len(new_pop) < popsize:
            idx1 = tournament_select(scores, tournament_k, rng)
            idx2 = tournament_select(scores, tournament_k, rng)
            child = crossover(pop[idx1], pop[idx2], rng)
            child = mutate(child, mutation_rate, rng)
            child = repair_binary(child, target_ones, rng)
            key = child.tobytes()

            if sc < best_score:
                best_score = sc
                best_h = child.copy()
        pop = new_pop
        scores = new_scores
    return best_h, best_score, cache

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = h - eta * momentum
        if it % 500 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def repair_binary(v, target_ones, rng):
    current = int(v.sum())
    diff = target_ones - current
    if diff > 0:
        zero_idx = np.where(v == 0)[0]
        if zero_idx.size:
            chosen = rng.choice(zero_idx, size=diff, replace=False)
            v[chosen] = 1
    elif diff < 0:
        one_idx = np.where(v == 1)[0]
        if one_idx.size:
            chosen = rng.choice(one_idx, size=-diff, replace=False)
            v[chosen] = 0
    return v

def tournament_select(scores, k, rng):
    participants = rng.choice(len(scores), size=k, replace=False)
    best = participants[0]
    best_score = scores[best]
    for i in participants[1:]:
        if scores[i] < best_score:
            best, best_score = i, scores[i]
    return best

def crossover(p1, p2, rng):
    mask = rng.random(p1.size) < 0.5
    return np.where(mask, p1, p2)

def mutate(v, rate, rng):
    if rate <= 0:
        return v
    mask = rng.random(v.size) < rate
    v[mask] = 1 - v[mask]
    return v

def genetic_algorithm(initial_h, n, time_left, rng):
    target_ones = n // 2
    popsize = 50
    tournament_k = 5
    mutation_rate = 0.04
    # initialise population
    pop = []
    scores = []
    cache = {}
    init_bin = np.where(initial_h >= 0.5, 1, 0).astype(np.int8)
    init_bin = repair_binary(init_bin, target_ones, rng)
    while len(pop) < popsize:
        if len(pop) == 0:
            ind = init_bin.copy()
        else:
            ind = np.zeros(n, dtype=np.int8)
            ind[rng.choice(n, size=target_ones, replace=False)] = 1
        key = ind.tobytes()
        if key in cache:
            sc = cache[key]
        else:
            sc = compute_c5(ind)
            cache[key] = sc
        pop.append(ind)
        scores.append(sc)
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # evolve
    while time_left() > 0.5:
        new_pop = [best_h]
        new_scores = [best_score]
        while len(new_pop) < popsize:
            i1 = tournament_select(scores, tournament_k, rng)
            i2 = tournament_select(scores, tournament_k, rng)
            child = crossover(pop[i1], pop[i2], rng)
            child = mutate(child, mutation_rate, rng)
            child = repair_binary(child, target_ones, rng)
            key = child.tobytes()
            if key in cache:
                sc = cache[key]
            else:
                sc = compute_c5(child)
                cache[key] = sc
            new_pop.append(child)
            new_scores.append(sc)
            if sc < best_score:
                best_score, best_h = sc, child.copy()
        pop, scores = new_pop, new_scores
    return best_h, best_score, cache

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    base = project_onto_box_sum(base, base.size / 2.0)
    best_h = None
    best_score = float('inf')

    for n in (64, 128, 256):
        if time_left() < 5.0:
            break
        cur_h = upsample(best_h if best_h is not None else base, n)
        cur_h = project_onto_box_sum(cur_h, n / 2.0)

        # allocate time for GA
        rem = time_left()
        ga_budget = rem * 0.6
        ga_start = time.time()
        ga_time_left = lambda: max(0.0, ga_budget - (time.time() - ga_start))
        h_ga, sc_ga, _ = genetic_algorithm(cur_h, n, ga_time_left, rng)
        if sc_ga < best_score:
            best_score, best_h = sc_ga, h_ga.copy()

        # gradient refinement
        if time_left() > 1.0:
            gd_budget = time_left() * 0.4
            gd_start = time.time()
            gd_time_left = lambda: max(0.0, gd_budget - (time.time() - gd_start))
            h_gd, sc_gd = grad_descent(best_h.astype(float), n_iter=50000,
                                       step0=0.5, beta=0.9, rng=rng,
                                       time_left=gd_time_left)
            if sc_gd < best_score:
                best_score, best_h = sc_gd, h_gd.copy()

        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    # final random restarts if time permits
    while time_left() > 2.0:
        n = best_h.size
        h_rand = np.zeros(n, dtype=np.int8)
        h_rand[rng.choice(n, size=n // 2, replace=False)] = 1
        if time_left() < 1.0:
            break
        gd_budget = time_left() * 0.5
        gd_start = time.time()
        gd_time_left = lambda: max(0.0, gd_budget - (time.time() - gd_start))
        h_gd, sc_gd = grad_descent(h_rand.astype(float), n_iter=50000,
                                   step0=0.5, beta=0.9, rng=rng,
                                   time_left=gd_time_left)
        if sc_gd < best_score:
            best_score, best_h = sc_gd, h_gd.copy()
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h.astype(float), best_score, best_h.size
