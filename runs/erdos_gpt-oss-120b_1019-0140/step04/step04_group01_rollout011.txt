def anneal_binary(h0, rng, time_left_func, max_iters=None):
    h = h0.copy()
    n = h.size
    corr = compute_corr_scaled_fft(h)
    cur_score = np.max(corr)
    best_h = h.copy()
    best_score = cur_score
    # temperature schedule
    T0 = 0.2
    Tmin = 1e-6
    # maybe set decay per iteration
    it = 0
    # Determine max_iters based on time_left or given.
    while time_left_func() > 0.0:
        it += 1
        # Update temperature
        # linear decay: T = T0 * (1 - it / max_iters) if max_iters else T0 * exp(-decay * it)
        # We'll use exponential with decay = 0.9999 per iteration
        T = max(Tmin, T0 * (0.9999 ** it))
        # Determine candidate shift with max corr (ties)
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        # Convert idx to shift
        shift_idx = rng.choice(idxes)
        k = shift_idx - (n - 1)
        # Determine mismatched i range
        start_i = max(0, -k)
        end_i = min(n, n - k)
        if start_i >= end_i:
            # No overlap? continue
            continue
        # Compute mismatches mask
        i_range = np.arange(start_i, end_i)
        j_range = i_range + k
        # h_i and h_j as arrays
        hi = h[i_range]
        hj = h[j_range]
        mask = (hi == 1) & (hj == 0)
        mismatched_i = i_range[mask]
        if mismatched_i.size == 0:
            # Could try opposite direction: mismatches where hi==0 and hj==1? So swapping would also reduce mismatches for opposite shift.
            # For now skip.
            # Optionally random swap.
            # do random swap to explore.
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(mismatched_i)
            j = i + k
        # perform swap
        h[i], h[j] = h[j], h[i]  # swap values
        # compute new correlation
        corr_new = compute_corr_scaled_fft(h)
        new_score = np.max(corr_new)
        # Accept condition
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            corr = corr_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # optional early break if best_score <= target (global target maybe)
        # Not necessary as target may be unknown globally.

    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto {lo <= x_i <= hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(l):
        return np.clip(v - l, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5_fft(h):
    """Maximum overlap integral C5 using FFT, scaled by dx=2/n."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_corr_scaled_fft(h):
    """Full correlation array scaled by dx."""
    n = h.shape[0]
    M = 2 * n
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[:2 * n - 1]
    return corr * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):

    grad = np.zeros_like(h)
    for shift in shifts:

            term1 = np.zeros_like(h)

            term2[:n - s] = h[s:]

    h[idx] = 1.0
    return h

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def downsample(arr, new_n):

def anneal_binary(h0, rng, time_left_func, T0=0.2, Tmin=1e-6, decay=0.9999):
    """Simulated annealing on binary h (0/1) preserving sum."""
    h = h0.copy()
    n = h.size
    corr = compute_corr_scaled_fft(h)
    cur_score = np.max(corr)
    best_h = h.copy()
    best_score = cur_score
    it = 0
    while time_left_func() > 0.0:
        it += 1
        T = max(Tmin, T0 * (decay ** it))
        max_corr = np.max(corr)
        idxes = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shift_idx = rng.choice(idxes)
        k = shift_idx - (n - 1)

        start_i = max(0, -k)
        end_i = min(n, n - k)
        if start_i >= end_i:
            # no overlap, pick random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i_range = np.arange(start_i, end_i)
            j_range = i_range + k
            hi = h[i_range]
            hj = h[j_range]
            mask = (hi == 1) & (hj == 0)
            mismatched_i = i_range[mask]
            if mismatched_i.size == 0:
                # fallback to random swap
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
            else:
                i = rng.choice(mismatched_i)
                j = i + k

        # swap
        h[i], h[j] = h[j], h[i]
        corr_new = compute_corr_scaled_fft(h)
        new_score = np.max(corr_new)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_score = new_score
            corr = corr_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return best_h, best_score
