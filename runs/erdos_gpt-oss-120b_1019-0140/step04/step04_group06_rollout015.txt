Implementation steps for a gradient descent iteration:

- Input h (np.ndarray of shape (n,)), n integer.

- Compute dx = 2.0 / n

- Compute conv_hh = np.correlate(h, h, mode='full')  # length 2n-1

- Compute g = 1 - conv_hh * dx

- Compute weights = np.exp(alpha * g) ; to avoid overflow, subtract max(g) for stability: weights = np.exp(alpha * (g - g.max())).

- Compute weight_norm = weights.sum(); w = weights / weight_norm

- Compute w_arr = w (still length 2n-1) in order of shifts.

- Compute conv_back = np.convolve(w_arr, h, mode='full') ; length 3n-2.

- Compute w_rev_arr = w_arr[::-1]

- Compute conv_rev = np.convolve(w_rev_arr, h, mode='full') ; length 3n-2.

- Slice index offset = n - 1

- backward = conv_back[offset:offset+n]

- forward = conv_rev[offset:offset+n]

- grad = 1. - (forward + backward)

def gradient_descent_smooth(start_h, budget_s, start_time):
    h = start_h.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    # schedule: total iterations = something like int( min(20000, time_left * (10000/n)) )
    # we also adapt alpha
    # Let's define alpha_max = 200 (for sharpness). start alpha = 10.
    # we will linearly increase alpha over iterations.
    total_iters = int(min(20000, max(2000, (budget_s - (time.time() - start_time)) * (5000.0 / n))))
    for it in range(1, total_iters+1):
        # check time
        if time.time() - start_time > budget_s: break
        # compute gradient
        dx = 2.0 / n
        conv_hh = np.correlate(h, h, mode='full')
        # g size 2n-1
        g = 1.0 - conv_hh * dx
        # compute alpha schedule
        alpha = 10.0 + (200.0 - 10.0) * (it / total_iters)
        # compute weights
        # shift for numerical stability
        maxg = np.max(g)
        w = np.exp(alpha * (g - maxg))
        w /= w.sum()
        # compute convolutions
        # using np.convolve
        conv_back = np.convolve(w, h, mode='full')
        w_rev = w[::-1]
        conv_rev = np.convolve(w_rev, h, mode='full')
        offset = n - 1
        backward = conv_back[offset:offset+n]
        forward = conv_rev[offset:offset+n]
        grad = 1.0 - (forward + backward)
        # step size schedule
        eta = step0 / np.sqrt(it)
        # update with momentum (optional)
        # maybe use simple: h_new = h - eta * grad
        h_new = h - eta * grad
        # projection
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda such that sum(clip(v - lam, lo, hi)) = s
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    arr = np.empty(n, dtype=float)
    for i in range(n):
        arr[i] = (bin(i).count('1') & 1)
    return arr

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def gradient_descent_smooth(h0, time_left, max_iters, step0, rng):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        # correlation of h with itself
        conv_hh = np.correlate(h, h, mode='full')
        g = 1.0 - conv_hh * dx      # g for each shift
        # smooth max weight
        maxg = g.max()
        alpha = 10.0 + (200.0 - 10.0) * (it / max_iters)
        w = np.exp(alpha * (g - maxg))
        w /= w.sum()
        # forward and backward sums via convolution
        conv_back = np.convolve(w, h, mode='full')
        w_rev = w[::-1]
        conv_rev = np.convolve(w_rev, h, mode='full')
        offset = n - 1
        backward = conv_back[offset:offset + n]
        forward = conv_rev[offset:offset + n]
        grad = 1.0 - (forward + backward)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        # occasional random jitter
        if rng.random() < 0.005:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
