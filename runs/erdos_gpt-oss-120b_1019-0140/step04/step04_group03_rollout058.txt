def simulated_annealing_binary(h, cur_score, rng, max_time):
    n = h.size
    best_h = h.copy()
    best_score = cur_score
    h_cur = h.copy()
    cur_score = cur_score
    start = time.time()
    total_steps = int(min(20000, max(5000, (max_time) * 500)))  # steps per second factor
    # Temperature schedule: exponential from T0 to T_final.
    T0 = 0.01 * cur_score  # maybe scale
    T_final = 1e-6
    for step in range(total_steps):
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # linear schedule:
        T = T0 * (1 - step/total_steps) + T_final
        # choose two indices with opposite bits
        i = rng.integers(n)
        # find j of opposite value
        # optimization: if h_cur[i]==1, pick j with 0; else if 0 pick j with 1.
        if h_cur[i] == 1:
            # choose j from zero positions
            zero_indices = np.where(h_cur == 0)[0]
            if zero_indices.size == 0:
                continue
            j = rng.choice(zero_indices)
        else:
            one_indices = np.where(h_cur == 1)[0]
            if one_indices.size == 0:
                continue
            j = rng.choice(one_indices)
        # swap
        h_new = h_cur.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]  # swap
        new_score = compute_c5_fft(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
    return best_h, best_score

def update_overlap_swap(O, h, i, j):
    # pre-swap: h[i]=1, h[j]=0 (or vice versa)
    # Let's assume we swap values, so after swap they are flipped.
    n = h.shape[0]
    offset = n - 1
    # Determine which is 1 and which is 0 before swap.
    if h[i] == h[j]:
        return O, h  # no change
    # Determine L (index of 1), G (index of 0)
    if h[i] > h[j]:
        L = i
        G = j
        val_L = 1  # before
        val_G = 0
    else:
        L = j
        G = i
        val_L = 1
        val_G = 0
    # Update O for all shifts s != 0; skip s=0.
    # Use vectorized approach:
    s = np.arange(-n+1, n)
    # Indices for L+s
    idxLp = L + s
    idxLm = L - s
    idxGp = G + s
    idxGm = G - s
    # Compute masks for valid indices
    maskLp = (idxLp >= 0) & (idxLp < n)
    maskLm = (idxLm >= 0) & (idxLm < n)
    maskGp = (idxGp >= 0) & (idxGp < n)
    maskGm = (idxGm >= 0) & (idxGm < n)
    # For s=0, we will zero out contributions later.
    # Gather h values at those positions
    # Since h is binary before swap, these are 0 or 1.
    # For s!=0, delta = -(h[idxLp] + h[idxLm]) + (h[idxGp] + h[idxGm])
    # Compute delta array
    delta = np.zeros_like(O, dtype=int)
    # For each mask, subtract or add.
    # subtract contributions from L.
    delta[maskLp] -= h[idxLp[maskLp]]
    delta[maskLm] -= h[idxLm[maskLm]]
    # add contributions from G.
    delta[maskGp] += h[idxGp[maskGp]]
    delta[maskGm] += h[idxGm[maskGm]]
    # fix for s=0: currently idxLp and idxLm both equal L, subtract twice, idxGp and idxGm both equal G, add twice. Since h[L]=1, h[G]=0 pre-swap, delta[center] = -2*1 + 2*0 = -2. Should be 0. So we set delta[center] = 0.
    delta[offset] = 0
    # Apply delta
    O += delta
    # Now swap h entries
    h[i], h[j] = h[j], h[i]
    return O, h

def update_overlap_swap(O, h, i, j):
    n = h.shape[0]
    offset = n - 1
    # Determine L (1) and G (0) before swap
    if h[i] == h[j]:
        return O, h
    if h[i] > h[j]:
        L_idx = i
        G_idx = j
    else:
        L_idx = j
        G_idx = i
    # create shift array
    s = np.arange(-n+1, n, dtype=int)
    # compute masks
    idxL_plus = L_idx + s
    idxL_minus = L_idx - s
    idxG_plus = G_idx + s
    idxG_minus = G_idx - s
    maskL_plus = (idxL_plus >= 0) & (idxL_plus < n)
    maskL_minus = (idxL_minus >= 0) & (idxL_minus < n)
    maskG_plus = (idxG_plus >= 0) & (idxG_plus < n)
    maskG_minus = (idxG_minus >= 0) & (idxG_minus < n)
    # delta array
    delta = np.zeros_like(O, dtype=int)
    # subtract contributions from L
    delta[maskL_plus] -= h[idxL_plus[maskL_plus]]
    delta[maskL_minus] -= h[idxL_minus[maskL_minus]]
    # add contributions from G
    delta[maskG_plus] += h[idxG_plus[maskG_plus]]
    delta[maskG_minus] += h[idxG_minus[maskG_minus]]
    # correct for shift 0 (index offset)
    delta[offset] = 0
    O += delta
    # swap h
    h[i], h[j] = h[j], h[i]
    return O, h

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(v - lam_mid, lo, hi)
        s = proj.sum()
        if abs(s - target_sum) < 1e-12:
            return proj
        if s > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, L)
    B = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(np.conj(H) * B).real
    return corr[: 2 * n - 1]

def compute_c5_fft(h):
    n = h.shape[0]
    return np.max(compute_corr_fft(h)) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            sz = n - shift
            term1 = np.empty(n, dtype=h.dtype)
            term1[:sz] = h[shift:]
            term1[sz:] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[shift:] = h[:sz]
            term2[:shift] = 0.0
        else:
            s = -shift
            sz = n - s
            term1 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:sz]
            term1[:s] = 0.0
            term2 = np.empty(n, dtype=h.dtype)
            term2[:sz] = h[s:]
            term2[sz:] = 0.0
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_momentum_custom(h0, n_iter, step0, beta, rng, max_time):
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time.time() - start >= max_time:
            break
        corr = compute_corr_fft(h)
        maxc = np.max(corr)
        idx = np.where(corr >= maxc - 1e-12)[0]
        shifts = idx - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:

        h_new = project_onto_box_sum(h_new, target)
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

            if cur_ones != n // 2:
                idx_desc = np.argsort(best_h)[::-1]
                binary_h[:] = 0.0
                binary_h[idx_desc[:n // 2]] = 1.0
            remaining = budget_s - (time.time() - start)
            if remaining > 5.0:
                h_ann, sc_ann = simulated_annealing_binary(binary_h, rng, remaining)
                if sc_ann < best_score:
                    best_score, best_h = sc_ann, h_ann.copy()
            base = best_h.copy()

        if idx < len(scales) - 1:
            base = best_h if best_h is not None else base

    # final polishing
    if best_h is not None and budget_s - (time.time() - start) > 5.0:
        n = best_h.size
        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, max(50000, remaining * (50000.0 / max(64, n)))))
        best_h, best_score = gradient_descent_momentum_custom(
            best_h, n_iter=max_iters, step0=0.2, beta=0.9,
            rng=rng, max_time=remaining)
        best_h = project_onto_box_sum(best_h, n / 2.0)
        best_score = compute_c5_fft(best_h)

    if best_h is None:
        n = n_final
        best_h = project_onto_box_sum(thue_morse(n), n / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, best_h.size
