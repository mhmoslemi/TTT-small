def compute_c5_fft(h):
    n = h.shape[0]
    # zero pad to length 2n
    size = 1 << (2*n -1).bit_length()  # next power of two
    # but we need full convolution length 2n-1
    f = np.fft.rfft(h, n=size)
    g = np.fft.rfft(1 - h, n=size)
    conv = np.fft.irfft(f * g, n=size)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def annealing_search(h_initial, total_time, rng):
    start = time.time()
    best_h = h_initial.copy()
    best_score = compute_c5(best_h)
    current_h = best_h.copy()
    current_score = best_score

    n = len(h_initial)
    # precompute indices of ones and zeros
    ones = np.flatnonzero(current_h == 1)
    zeros = np.flatnonzero(current_h == 0)

    # initial temperature: set such that acceptance probability ~0.8 for typical diff.
    # estimate typical diff as perhaps 0.001? We'll set temp0 = 0.001.
    temp = 0.001
    # cooling schedule: multiply by 0.9995 per iteration
    cooling = 0.9995

    iter = 0
    while time.time() - start < total_time:
        iter += 1
        # select random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        current_h[i] = 0
        current_h[j] = 1
        # update ones/zeros arrays efficiently
        # we can swap in the arrays:
        # find indices of i in ones and j in zeros to replace
        # but easier: recompute ones/zeros each iteration (cost O(n) but n=512, okay)
        # but we can maintain via sets.
        # Just recompute for simplicity.
        new_score = compute_c5(current_h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            current_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score

        # temperature cooling
        temp *= cooling
        # occasionally reset temperature to avoid freeze too early
        if iter % 10000 == 0:
            temp = max(temp, 0.001)
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Return C5 for binary/continuous step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Euclidean projection of v onto the box [lo,hi]^n with prescribed sum s."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def binary_from_continuous(v):
    """Return binary vector with exactly half entries equal to 1."""
    n = v.shape[0]
    k = n // 2
    idx = np.argpartition(v, -k)[-k:]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ------------------------------------------------------------------ #
    # Build initial binary solution (length 512)
    target_n = 512
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback random balanced binary of length 64
        init = np.zeros(64, dtype=float)
        init[:32] = 1.0
        rng.shuffle(init)
    # upsample to target size and binarize
    if init.shape[0] != target_n:
        init = upsample(init, target_n)
    h = binary_from_continuous(init)
    n = target_n

    # initial correlation and scores
    c = np.correlate(h, 1.0 - h, mode="full")   # unscaled
    cur_score = np.max(c) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score

    # pre‑computed shift array
    t = np.arange(-(n - 1), n)                 # length 2n‑1

    # simulated annealing parameters
    temp_start = 0.01
    temp_end = 1e-6

    # main annealing loop
    while time_left() > 5.0:
        # pick a random swap (1↔0)
        while True:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and h[i] != h[j]:
                break
        # ensure i holds a 1, j holds a 0
        if h[i] == 0.0:
            i, j = j, i
        delta_i, delta_j = -1.0, 1.0
        d = j - i

        # --- overlap (linear) contribution ---
        overlap_i = ((t >= 0) & (i <= n - 1 - t)) | ((t < 0) & (i >= -t))
        overlap_j = ((t >= 0) & (j <= n - 1 - t)) | ((t < 0) & (j >= -t))
        term_linear = delta_i * overlap_i.astype(float) + delta_j * overlap_j.astype(float)

        # --- product contribution ---
        idx_i_plus = i + t
        valid_i_plus = (idx_i_plus >= 0) & (idx_i_plus < n)
        prod_i_plus = np.where(valid_i_plus, h[idx_i_plus], 0.0)

        idx_i_minus = i - t
        valid_i_minus = (idx_i_minus >= 0) & (idx_i_minus < n)
        prod_i_minus = np.where(valid_i_minus, h[idx_i_minus], 0.0)

        idx_j_plus = j + t
        valid_j_plus = (idx_j_plus >= 0) & (idx_j_plus < n)
        prod_j_plus = np.where(valid_j_plus, h[idx_j_plus], 0.0)

        idx_j_minus = j - t
        valid_j_minus = (idx_j_minus >= 0) & (idx_j_minus < n)
        prod_j_minus = np.where(valid_j_minus, h[idx_j_minus], 0.0)

        term_product = (delta_i * (prod_i_plus + prod_i_minus) +
                        delta_j * (prod_j_plus + prod_j_minus))

        # cross term for distinct indices
        cross_mask = (t == d) | (t == -d)
        term_product += (delta_i * delta_j) * cross_mask.astype(float)

        # self‑square term needed for t==0
        self_mask = (t == 0)
        term_product += (delta_i**2 + delta_j**2) * self_mask.astype(float)

        delta_c = term_linear - term_product
        new_c = c + delta_c

        def ineq_fun(x):
            h = x[:n]
            t_val = x[-1]
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            return t_val - np.max(corr)

        def eq_sum(x):
            return np.sum(x[:n]) - n / 2.0

        x0 = np.concatenate([best_h.astype(float), [best_score]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [{'type': 'eq', 'fun': eq_sum},
                {'type': 'ineq', 'fun': ineq_fun}]
        res = opt.minimize(lambda x: x[-1], x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        if res.success:
            h_opt = res.x[:n]
            h_opt = project_onto_box_sum(h_opt, n / 2.0)
            score_opt = compute_c5(h_opt)
            if score_opt < best_score:
                best_h = h_opt
                best_score = score_opt

    # final clean‑up
    best_h = best_h.astype(float)
    best_score = float(best_score)
    return best_h, best_score, best_h.size
