def weighted_shift_sum(h, w):
    n = h.size
    # w length = 2n-1
    # forward sum: sum_k w_k * h_{i+k}
    w_rev = w[::-1]
    size = n + 2*n - 2  # len(conv) = len(w)+len(h)-1 = 3n-2
    H = np.fft.rfft(h, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_fwd = np.fft.irfft(H * W, n=size)
    # backward sum: sum_k w_k * h_{i-k}
    Wb = np.fft.rfft(w, n=size)
    conv_bwd = np.fft.irfft(H * Wb, n=size)
    # extract central n entries at indices (len(w)-1) to (len(w)-1 + n-1)
    offset = len(w) - 1
    fwd = conv_fwd[offset:offset + n]
    bwd = conv_bwd[offset:offset + n]
    return fwd, bwd
```

def simulated_annealing(h, rng, time_left, total_budget, start_temp=0.01, end_temp=1e-5):
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_h = h.copy()
    cur_score = best_score
    while time_left() > 0:
        # compute temperature based on elapsed fraction
        elapsed = total_budget - time_left()
        temp = start_temp * (1 - elapsed / total_budget) + end_temp * (elapsed / total_budget)
        # propose move
        i, j = rng.choice(cur_h.size, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection onto box [lo,hi] with sum constraint."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Exact max overlap for current grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to new length."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_balanced_binary(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Gradient of correlation for given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size:
        grad /= shifts.size
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 2500 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def simulated_annealing(h0, rng, time_left, total_budget,
                        start_temp=0.005, end_temp=1e-6):
    """SA with pairwise mass transfers."""
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = h0.copy()
    cur_score = best_score
    n = h0.size

    while time_left() > 0:
        # linear temperature schedule
        elapsed = total_budget - time_left()
        fraction = elapsed / total_budget
        temp = end_temp + (start_temp - end_temp) * (1.0 - fraction)

        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temp):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s
    time_left = lambda: max(0.0, total_budget - (time.time() - start))

    # initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [128, 256, 512]
    # add 1024 if enough time remains after previous phases
    if time_left() > 300:
        scales.append(1024)

    for n in scales:
        if time_left() < 5:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent phase
        remaining = time_left()
        max_iters = int(min(40000, max(2000, remaining * (8000.0 / n))))
        h_gd, sc_gd = gradient_descent_adaptive(
            base, max_iters, step0=0.8, beta=0.9, rng=rng, time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd
        base = h_gd

        # Simulated annealing phase (use up to 20% of remaining time, capped at 30s)
        if time_left() > 5:
            sa_budget = min(time_left() * 0.2, 30.0)
            sa_end = time.time() + sa_budget
            sa_time_left = lambda: max(0.0, sa_end - time.time())
            h_sa, sc_sa = simulated_annealing(
                base, rng, sa_time_left, total_budget,
                start_temp=0.005, end_temp=1e-6)
            if sc_sa < best_score:
                best_h, best_score = h_sa.copy(), sc_sa
            base = h_sa

        # Short post‑SA gradient refinement
        if time_left() > 5:
            max_iters = int(min(15000, max(1000, time_left() * (5000.0 / n))))
            h_ref, sc_ref = gradient_descent_adaptive(
                base, max_iters, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
            if sc_ref < best_score:
                best_h, best_score = h_ref.copy(), sc_ref
            base = h_ref

        # Random balanced restarts (few, cheap)
        for _ in range(2):
            if time_left() < 4:
                break
            h_rand = random_balanced_binary(n, rng)
            iters = int(min(3000, max(500, time_left() * (2000.0 / n))))
            h_rand_ref, sc_rand = gradient_descent_adaptive(
                h_rand, iters, step0=0.5, beta=0.9, rng=rng, time_left=time_left)
            if sc_rand < best_score:
                best_h, best_score = h_rand_ref.copy(), sc_rand
            base = h_rand_ref

    # Final intensive refinement on the global best
    if time_left() > 5:
        iters = int(min(80000, max(30000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            best_h, iters, step0=0.3, beta=0.9, rng=rng, time_left=time_left)

    # One more annealing round
    if time_left() > 3:
        sa_budget = min(time_left() * 0.3, 20.0)
        sa_end = time.time() + sa_budget
        sa_time_left = lambda: max(0.0, sa_end - time.time())
        h_sa, sc_sa = simulated_annealing(
            best_h, rng, sa_time_left, total_budget,
            start_temp=0.003, end_temp=1e-6)
        if sc_sa < best_score:
            best_h, best_score = h_sa, sc_sa

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
