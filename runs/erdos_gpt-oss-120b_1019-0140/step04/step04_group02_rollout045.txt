import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure bracket contains solution
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑overlap term (averaged over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    """Projected momentum gradient descent on max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    mom = np.zeros_like(h)
    no_imp = 0
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        mom = beta * mom + (1.0 - beta) * grad
        eta = step0 / (np.sqrt(it) + 1e-12)
        h_new = h - eta * mom
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.015, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score - 1e-12:
            best_score, best_h = score, h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= 50000:
            break
        h = h_new
    return best_h, best_score


def binary_hill_climb(h, score, rng, time_left):
    """Swap‑based hill climbing on balanced binary vectors."""
    n = h.size
    # ensure exactly n/2 ones
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    no_imp = 0
    while time_left() > 0.1 and no_imp < 20000:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc_new = compute_c5_fft(h_new)
        if sc_new < score - 1e-12:
            h, score = h_new, sc_new
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h <= 0.5)
            no_imp = 0
        else:
            no_imp += 1
    return h, score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    # Starting vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128).astype(float)

    TARGET = 0.3808
    best_h = None
    best_score = np.inf

    # Multi‑resolution gradient descent
    for n in (128, 256, 512):
        if time_left() < 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # several restarts: perturbed base and random binaries
        restarts = 4
        for r in range(restarts):
            if time_left() < 0.5:
                break
            if r == 0:
                h0 = base + rng.normal(scale=0.03, size=n)
            else:
                h0 = random_binary_balanced(n, rng).astype(float)
                h0 += rng.normal(scale=0.01, size=n)
            h0 = project_onto_box_sum(h0, n / 2.0)

            step0 = 0.6 if n <= 256 else 0.4
            h_opt, sc_opt = gradient_descent(
                h0, n_iter=120000, step0=step0, beta=0.9,
                rng=rng, time_left=time_left)

            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break
        if best_h is not None:
            base = best_h.copy()
        if best_score <= TARGET:
            break

    # Final intensive refinement at highest resolution
    if best_h is not None and time_left() > 5.0:
        h_ref, sc_ref = gradient_descent(
            best_h, n_iter=300000, step0=0.3, beta=0.9,
            rng=rng, time_left=time_left)
        if sc_ref < best_score:
            best_score, best_h = sc_ref, h_ref.copy()

    # Project to binary and perform aggressive hill climbing
    if best_h is not None and time_left() > 3.0:
        n = best_h.size
        bin_h = np.zeros(n, dtype=float)
        idx = np.argsort(-best_h)[:n // 2]
        bin_h[idx] = 1.0
        bin_score = compute_c5_fft(bin_h)
        bin_h, bin_score = binary_hill_climb(bin_h, bin_score, rng, time_left)
        if bin_score < best_score:
            best_score, best_h = bin_score, bin_h.copy()

    # Final tiny random perturbations (continuous) if time permits
    while best_h is not None and time_left() > 0.5:
        h_new = best_h + rng.normal(scale=0.005, size=best_h.size)
        h_new = project_onto_box_sum(h_new, best_h.size / 2.0)
        sc_new = compute_c5_fft(h_new)
        if sc_new < best_score - 1e-12:
            best_score, best_h = sc_new, h_new.copy()
        else:
            break

    # Safety final projection
    if best_h is not None:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
