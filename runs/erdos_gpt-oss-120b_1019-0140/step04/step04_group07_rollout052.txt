def annealing_moves(h, rng, time_left, init_temp=0.01, decay=0.995):
    best = h.copy()
    best_score = compute_c5(best)
    cur = h.copy()
    cur_score = best_score
    T = init_temp
    while time_left() > 0.2:
        i, j = rng.choice(cur.size, size=2, replace=False)
        # propose delta
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        # propose random delta up to max_delta * 0.5
        delta = max_delta * rng.random() * 0.5
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5(new)
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            cur, cur_score = new, new_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        T *= decay
        if T < 1e-7:
            T = init_temp
    return best, best_score
```

def compute_c5(h):
    n = h.shape[0]
    if n > 256:
        # compute full correlation using FFT
        rev = (1.0 - h)[::-1]
        N = 2 * n - 1
        fh = np.fft.rfft(h, n=N)
        frev = np.fft.rfft(rev, n=N)
        corr = np.fft.irfft(fh * frev, n=N)
    else:
        corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_subgradient(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            k = shift
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            k = -shift
            grad[k:] += 1.0 - h[:n - k]
            grad[:n - k] -= h[k:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad
```

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    # tolerance for near max shifts
    tol = 1e-8
    for it in range(max_iters):
        if time_left() <= 0:
            break
        corr = full_correlation(h)
        max_corr = np.max(corr)
        # find shifts achieving near max
        # Using tolerance relative to max_corr perhaps
        shifts = np.where(corr >= max_corr - tol)[0] - (n - 1)
        grad = compute_subgradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # occasional jitter
        if (it + 1) % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def full_correlation(h):
    """Linear correlation of h with (1-h) using FFT for speed when large."""
    n = h.shape[0]
    if n > 256:
        rev = (1.0 - h)[::-1]
        N = 2 * n - 1
        fh = np.fft.rfft(h, n=N)
        frev = np.fft.rfft(rev, n=N)
        corr = np.fft.irfft(fh * frev, n=N)
    else:
        corr = np.correlate(h, 1.0 - h, mode='full')
    return corr


def compute_c5(h):
    """Maximum overlap integral on the discrete grid."""
    n = h.shape[0]
    return np.max(full_correlation(h)) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from current length to new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_subgradient(h, shifts):
    """Subgradient of max correlation w.r.t. h for given active shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            k = shift
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            k = -shift
            grad[k:] += 1.0 - h[:n - k]
            grad[:n - k] -= h[k:]
    if shifts.size > 0:
        grad /= shifts.size
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with momentum and adaptive step."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    tol = 1e-8

    for it in range(max_iters):
        if time_left() <= 0:
            break

        corr = full_correlation(h)
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - tol)[0] - (n - 1)
        grad = compute_subgradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if (it + 1) % 2000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()

    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Random mass transfers between two positions, accepting any improvement."""
    n = h.size
    cur = h.copy()
    cur_score = compute_c5(cur)
    best_h = cur.copy()
    best_score = cur_score

    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)

        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            sc = compute_c5(new)
            if sc < best_score:
                best_h, best_score = new.copy(), sc
            if sc < cur_score:
                cur, cur_score = new, sc

        max_delta2 = min(cur[j], 1.0 - cur[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            new2 = cur.copy()
            new2[j] -= delta2
            new2[i] += delta2
            sc2 = compute_c5(new2)
            if sc2 < best_score:
                best_h, best_score = new2.copy(), sc2
            if sc2 < cur_score:
                cur, cur_score = new2, sc2

def annealing_moves(h, rng, time_left, init_temp=0.02, decay=0.995):
    """Simulated‑annealing style mass swaps."""
    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur = h.copy()
    cur_score = best_score
    T = init_temp

    while time_left() > 0.2:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            T = max(T * decay, 1e-8)
            continue
        delta = max_delta * rng.random() * 0.5
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - new_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_score = new, new_score
            if new_score < best_score:
                best_h, best_score = new.copy(), new_score

        T = max(T * decay, 1e-8)

    return best_h, best_score

    start = time.time()

    for n in scales:

                                                step0=0.8 if n <= 256 else 0.5,
                                                beta=0.9, rng=rng, time_left=time_left)
        if sc_gd < best_score:

        # Annealing moves

            base = h_an

    # Final intensive refinement on the global best

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
