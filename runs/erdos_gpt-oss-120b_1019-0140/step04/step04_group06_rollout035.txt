def smoothed_max_and_grad(h, alpha):
    n = h.shape[0]
    # correlation c (full)
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = c.max()
    # stable softmax
    shifted = c - max_c
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = exp_alpha.sum()
    w = exp_alpha / sum_exp   # sum to 1
    # compute gradient of c: for each shift k, weight w_k yields contribution.
    grad = np.ones_like(h)  # from sum_k w_k * 1 = 1
    # compute contributions from h_{j+k} and h_{j-k}
    # for each k
    n_shift = n - 1
    # loop over k idx
    for idx, wk in enumerate(w):
        if wk == 0:
            continue
        k = idx - n_shift
        if k >= 0:
            # grad[j] -= wk * h[j+k]
            grad[:n - k] -= wk * h[k:]
            # grad[j] -= wk * h[j-k] => for j>=k, j-k>=0
            grad[k:] -= wk * h[:n - k]
        else:
            k2 = -k
            # k negative: we have same logic swapping
            grad[k2:] -= wk * h[:n - k2]
            grad[:n - k2] -= wk * h[k2:]
    # gradient of smoothed max (unscaled) is -grad? Wait we defined grad = 1 - sum contributions.
    # Actually grad = 1 - Σ w_k * (h_{j+k} + h_{j-k}) so it's the derivative of c with sign? Let's rederive.
    # We had ∂c_k/∂h_j = 1 - (h_{j+k} + h_{j-k})
    # Then ∂f/∂h_j = Σ_k w_k * ∂c_k/∂h_j = Σ_k w_k * (1 - (h_{j+k}+h_{j-k})) = 1 - Σ_k w_k * (h_{j+k}+h_{j-k}) = grad.
    # So grad computed is indeed ∂f/∂h_j.
    # Now scaled objective multiplies by dx, gradient also scaled.
    dx = 2.0 / n
    grad_scaled = grad * dx
    # compute smoothed max value: f = max_c + (1/alpha) * np.log(sum_exp)
    f = max_c + (1.0 / alpha) * np.log(sum_exp)
    return f * dx, grad_scaled, c

def adam_optimize(h0, time_left, max_iter=50000, alpha_start=0.5, alpha_end=20.0):
   h = h0.copy()
   n = h.shape[0]
   total = n/2.0
   m = np.zeros_like(h)
   v = np.zeros_like(h)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   best_h = h.copy()
   best_score = compute_c5(h)
   for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        # schedule alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        # compute smoothed objective and gradient
        f, grad, c = smoothed_max_and_grad(h, alpha)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.01 * m_hat / (np.sqrt(v_hat) + eps)  # learning rate scale
        h_new = h - step
        # project onto constraints
        h_new = project_onto_box_sum(h_new, total)
        # compute actual C5
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
   return best_h, best_score
```

def binary_ga_optimize(n, time_left, init_h=None):
    # n must be even
    half = n // 2
    # initial population: include init_h if provided and binary-like
    pop = []
    if init_h is not None:
        # convert to binary by threshold 0.5 maybe
        bin0 = (init_h >= 0.5).astype(float)
        # adjust to exact half ones
        diff = bin0.sum() - half
        if diff > 0:
            # turn off diff ones at random among ones
            ones_idx = np.where(bin0 == 1)[0]
            remove = np.random.choice(ones_idx, size=int(diff), replace=False)
            bin0[remove] = 0
        elif diff < 0:
            zeros_idx = np.where(bin0 == 0)[0]
            add = np.random.choice(zeros_idx, size=int(-diff), replace=False)
            bin0[add] = 1
        pop.append(bin0)
    # fill rest of population with random balanced binary
    while len(pop) < 20:
        h = np.zeros(n, dtype=float)
        idx = np.random.choice(n, size=half, replace=False)
        h[idx] = 1.0
        pop.append(h)
    # evaluate
    scores = np.array([compute_c5(h) for h in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    # GA loop
    while time_left() > 0.5:  # need time for one generation
        # select two parents via tournament or random
        i, j = np.random.choice(len(pop), size=2, replace=False)
        p1, p2 = pop[i], pop[j]
        # crossover
        mask = np.random.rand(n) < 0.5
        child = np.where(mask, p1, p2).astype(float)
        # enforce half ones
        ones = int(child.sum())
        if ones > half:
            # turn off some ones
            idx_ones = np.where(child == 1)[0]
            turn_off = np.random.choice(idx_ones, size=ones-half, replace=False)
            child[turn_off] = 0
        elif ones < half:
            idx_zeros = np.where(child == 0)[0]
            turn_on = np.random.choice(idx_zeros, size=half-ones, replace=False)
            child[turn_on] = 1
        # mutation: swap a few bits
        for _ in range(2):
            if np.random.rand() < 0.2:
                # pick a 1 and a 0 to swap
                ones_idx = np.where(child == 1)[0]
                zeros_idx = np.where(child == 0)[0]
                if len(ones_idx) > 0 and len(zeros_idx) > 0:
                    i1 = np.random.choice(ones_idx)
                    i0 = np.random.choice(zeros_idx)
                    child[i1], child[i0] = 0.0, 1.0
        # evaluate child
        child_score = compute_c5(child)
        # replace worst if child is better
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            pop[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()

def smoothed_max_and_grad(h, alpha):
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = c.max()
    shifted = c - max_c
    exp_alpha = np.exp(alpha * shifted)
    sum_exp = exp_alpha.sum()
    w = exp_alpha / sum_exp
    # gradient
    grad = np.ones_like(h)
    n_shift = n - 1
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - n_shift
        if k >= 0:
            grad[:n - k] -= wk * h[k:]
            grad[k:] -= wk * h[:n - k]
        else:
            k2 = -k
            grad[k2:] -= wk * h[:n - k2]
            grad[:n - k2] -= wk * h[k2:]
    dx = 2.0 / n
    return (max_c + (1.0 / alpha) * np.log(sum_exp)) * dx, grad * dx, c

    h = h0.copy()

    total = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h, best_score = h.copy(), compute_c5(h)
    for it in range(1, max_iter+1):
        if time_left() <= 0:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        _, grad, _ = smoothed_max_and_grad(h, alpha)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        h_new = h - step
        h_new = project_onto_box_sum(h_new, total)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    # binary search for lambda
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

import time, math, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, h.dtype)
            term2 = np.zeros(n, h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def grad_descent(h0, n_iter, step0, beta, rng, time_left):

    total = n / 2.0
    best_h, best_score = h.copy(), compute_c5(h)

        if time_left() <= 0:

        h_new = project_onto_box_sum(h_new, total)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score, best_h = sc, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_ga_optimize(n, time_left, init_h=None, rng=None):
    half = n // 2
    pop = []
    if init_h is not None:
        bin0 = (init_h >= 0.5).astype(float)
        diff = int(round(bin0.sum() - half))
        if diff > 0:
            ones_idx = np.where(bin0 == 1)[0]
            remove = rng.choice(ones_idx, size=diff, replace=False)
            bin0[remove] = 0
        elif diff < 0:
            zeros_idx = np.where(bin0 == 0)[0]
            add = rng.choice(zeros_idx, size=-diff, replace=False)
            bin0[add] = 1
        pop.append(bin0)
    while len(pop) < 20:
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
        pop.append(h)
    scores = np.array([compute_c5(h) for h in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])
    while time_left() > 0.5:
        i, j = rng.choice(len(pop), size=2, replace=False)
        p1, p2 = pop[i], pop[j]
        mask = rng.random(n) < 0.5
        child = np.where(mask, p1, p2).astype(float)
        ones = int(child.sum())
        if ones > half:
            ones_idx = np.where(child == 1)[0]
            turn_off = rng.choice(ones_idx, size=ones - half, replace=False)

        elif ones < half:

            scores[worst_idx] = child_score
            if child_score < best_score:

    return best_h, best_score
