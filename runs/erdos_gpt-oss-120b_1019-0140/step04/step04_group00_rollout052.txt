def gradient_descent(h0, time_left, step0=0.5, beta=0.9, max_iter=200000, tol=1e-8):
    h = project_onto_box_sum(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    mom = np.zeros_like(h)
    it = 1
    while it <= max_iter and time_left() > 0.01:
        grad = gradient_max_shifts(h)
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target_sum)
        c5_new = compute_c5_fast(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c5

def gradient_max_shifts(h):
    n = h.shape[0]
    # compute correlation using FFT
    # N = 1<<...; compute full correlation.
    h_rev = h[::-1]
    N = 1 << ((2*n-1).bit_length())
    fft_rev = np.fft.rfft(h_rev, n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2*n-1] * (2.0 / n)  # multiply dx
    mx = corr.max()
    # find shift indices
    idxes = np.where(np.abs(corr - mx) < 1e-12)[0]  # tolerance maybe 1e-12 safe
    grads = np.zeros_like(h)
    for idx in idxes:
        shift = idx - (n - 1)  # convert index to shift.
        grads += gradient_of_shift_correct(h, shift)
    if len(idxes) > 0:
        grads /= len(idxes)
    return grads

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    rng = np.random.default_rng(seed)

    # set n
    n = 512  # maybe try 512
    target_sum = n / 2.0
    target = 0.3808

    # function to resample arrays to n points
    def resample(arr, new_n):
        old_n = arr.shape[0]
        if old_n == new_n:
            return arr.copy()
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
        return np.interp(x_new, x_old, arr)

    candidates = []

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        init = resample(init, n) if init.size != n else init.copy()
        candidates.append(project_onto_box_sum(init, target_sum))

    # Thue-Morse binary scaled to n, using repetition of sequence to fill.
    # generate base thue-morse up to n
    def thue_morse_binary(m):
        # generate using bit parity
        return np.fromiter(((bin(i).count('1') & 1) for i in range(m)), dtype=float, count=m)

    candidates.append(thue_morse_binary(n))

    # random balanced binary
    for _ in range(2):
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n//2, replace=False)] = 1.0
        candidates.append(h)

import time
import math
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=50):
    """Euclidean projection of v onto the set {x: lo<=x<=hi, sum(x)=target_sum}."""
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(l):
        return np.clip(v - l, lo, hi).sum()

    # expand bounds until they bracket the solution
    while f(lam_low) < target_sum:
        lam_low -= (hi - lo) * 10
    while f(lam_high) > target_sum:
        lam_high += (hi - lo) * 10

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5_fast(h):
    """Maximum overlap C5 for a step function h (array)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    return corr.max()


def gradient_of_shift_correct(h, shift):
    """Sub‑gradient of the overlap for a single shift."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        n_s = n - shift
        if n_s > 0:
            grad[:n_s] += 1.0 - h[shift:]
            grad[shift:] -= h[:n_s]
    else:
        s = -shift
        n_s = n - s
        if n_s > 0:
            grad[:n_s] -= h[s:]
            grad[s:] += 1.0 - h[:n_s]
    return grad


def gradient_and_max(h):
    """Return sub‑gradient (average over maximizing shifts) and the max C5."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fft_rev = np.fft.rfft(h[::-1], n=N)
    fft_one = np.fft.rfft(1.0 - h, n=N)
    conv = np.fft.irfft(fft_rev * fft_one, n=N)
    corr = conv[:2 * n - 1] * (2.0 / n)
    mx = corr.max()
    # indices attaining the maximum (with a tiny tolerance)
    idxes = np.where(np.abs(corr - mx) <= 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxes:
        shift = idx - (n - 1)
        grad += gradient_of_shift_correct(h, shift)
    if idxes.size:
        grad /= idxes.size
    return grad, mx


def thue_morse_binary(m):
    """Thue–Morse sequence as 0/1 array."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(m)), dtype=float, count=m)


def resample(arr, new_n):
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def gradient_descent_momentum(h0, target_sum, target, time_left,
                             max_iter=200000, beta=0.9, step0=0.5):
    """Momentum gradient descent on the max‑overlap objective."""
    h = project_onto_box_sum(h0, target_sum)
    mom = np.zeros_like(h)
    grad, mx = gradient_and_max(h)
    best_h = h.copy()
    best_c5 = mx
    it = 1
    while it <= max_iter and time_left() > 0.02:
        mom = beta * mom + (1 - beta) * grad
        eta = step0 / math.sqrt(it)
        h_new = project_onto_box_sum(h - eta * mom, target_sum)
        grad, mx = gradient_and_max(h_new)
        if mx < best_c5:
            best_c5 = mx
            best_h = h_new.copy()
            if best_c5 <= target:
                break
        h = h_new
        it += 1
    return best_h, best_c5

        if c5_bin < c5_opt:
            h_opt, c5_opt = h_bin, c5_bin

        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        if best_c5 <= target:
            return best_h, best_c5, n

    # Additional random restarts while time permits
    while time_left() > 12.0:
        v = rng.random(n)
        v = project_onto_box_sum(v, target_sum)
        h_opt, c5_opt = gradient_descent_momentum(
            v, target_sum, target, time_left)

        idx = np.argpartition(-h_opt, n // 2)[:n // 2]
        h_bin = np.zeros_like(h_opt)
        h_bin[idx] = 1.0
        c5_bin = compute_c5_fast(h_bin)
        if c5_bin < c5_opt:
            h_opt, c5_opt = h_bin, c5_bin

        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
        if best_c5 <= target:
            break

    # Final safeguard projection and evaluation
    best_h = project_onto_box_sum(best_h, target_sum)
    best_c5 = compute_c5_fast(best_h)
    return best_h, best_c5, n
