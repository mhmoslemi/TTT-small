def anneal(h0, time_left, rng):
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    total = time_left()
    start = time.time()
    while time_left() > 0:
        elapsed = time.time() - start
        # temp schedule:
        t_start = 1e-2
        t_end = 1e-6
        temp = t_start * (t_end / t_start) ** (elapsed / total)
        # propose move:
        i, j = rng.choice(len(h), 2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.3  # up to 30% of max
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / temp):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
        # else discard
    return best_h, best_score
```

def upsample(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure proper bracketing
    while f(lam_low) < s:
        lam_low -= (hi - lo) * 2
    while f(lam_high) > s:
        lam_high += (hi - lo) * 2

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Maximum overlap integral on the discrete grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation to a larger grid."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def anneal(h0, time_left, rng):
    """Simulated‑annealing style refinement with decreasing temperature."""
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    total = time_left()
    start = time.time()
    while time_left() > 0:
        elapsed = time.time() - start
        t_start, t_end = 1e-2, 1e-6
        temp = t_start * (t_end / t_start) ** (elapsed / total)

        i, j = rng.choice(h.size, 2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.3
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / temp):
            h = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h, best_score = h.copy(), cur_score
    return best_h, best_score

        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]

    for n in scales:
        if time_left() <= 2.0:
            break

        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent on current scale
        max_iters = int(min(15000, max(2000, time_left() * (2000.0 / n))))
        if max_iters > 0 and time_left() > 1.0:
            step0 = 1.0 if n <= 256 else 0.5
            base, sc = gradient_descent_adaptive(base, max_iters, step0, 0.95, rng, time_left)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Targeted max‑shift improvement
        if time_left() > 1.0:
            base, sc = improve_via_max_shift(base, rng, time_left)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Local pair‑wise moves
        if time_left() > 1.0:
            base, sc = local_pair_moves(base, rng, time_left)
            if sc < best_score:
                best_h, best_score = base.copy(), sc

        # Random balanced restarts with brief refinement
        for _ in range(3):
            if time_left() < 2.0:
                break
            h_rand = random_binary_balanced(n, rng)
            max_iters_rand = int(min(3000, max(300, time_left() * (800.0 / n))))
            if max_iters_rand > 0:
                h_rand, sc_rand = gradient_descent_adaptive(h_rand, max_iters_rand, 0.5, 0.9,
                                                            rng, time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand
            if time_left() > 1.0:
                h_rand, sc_rand = local_pair_moves(h_rand, rng, time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand

    # Final intensive refinement on the global best
    if time_left() > 4.0:
        n = best_h.size
        max_iters_final = int(min(60000, max(20000, time_left() * 20)))
        best_h, best_score = gradient_descent_adaptive(best_h, max_iters_final, 0.3, 0.9,
                                                      rng, time_left)

    # Final annealing pass if any time remains
    if time_left() > 2.0:
        best_h, best_score = anneal(best_h, time_left, rng)

    # final projection for safety
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, int(best_h.size)
