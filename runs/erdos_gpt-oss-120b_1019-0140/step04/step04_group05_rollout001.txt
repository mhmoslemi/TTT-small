def hill_climb(h0, time_budget, rng):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    # step factor schedule maybe linearly decreasing
    step_factor = 0.5
    # iteration counter
    it = 0
    max_iterations = 1000000
    while time.time() - start < time_budget and it < max_iterations:
        it += 1
        # compute max correlation and shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        if max_val * (2.0 / n) < best_score - 1e-9:
            # improvement
            best_score = max_val * (2.0 / n)
            best_h = h.copy()
        # If cannot improve further (max_val*dx approx best_score), break?
        # Actually we need to try to reduce max_val.
        # Determine shift
        k = max_idx - (n - 1)
        # Determine overlapping indices
        if k >= 0:
            i_range = np.arange(n - k)
            i2_range = i_range + k
        else:
            i_range = np.arange(-k, n)
            i2_range = i_range + k
        # compute contributions
        contrib = h[i_range] * (1.0 - h[i2_range])
        # find index with max contribution
        idx_local = np.argmax(contrib)
        i = i_range[idx_local]
        i2 = i2_range[idx_local]
        # compute delta
        max_delta = min(h[i], 1.0 - h[i2])
        if max_delta <= 0:
            # nothing to transfer, break
            break
        delta = max_delta * step_factor
        # apply
        h[i] -= delta
        h[i2] += delta
        # maybe adjust step_factor occasionally
        if it % 1000 == 0:
            # recompute improvement; if not improving, shrink step_factor
            cur_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
            if cur_score >= best_score - 1e-6:
                step_factor = max(step_factor * 0.9, 0.01)
    # final projection (in case numeric)
    h = project_onto_box_sum(h, n/2.0)
    final_score = compute_c5(h)
    if final_score < best_score:
        best_h = h.copy()
        best_score = final_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))
    # Determine target n
    target_n = 512
    # Initialize base solution
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != target_n:
            # upsample
            base = upsample(base, target_n)
    else:
        base = random_balanced_binary(target_n, rng).astype(float)
    base = project_onto_box_sum(base, target_n/2.0)
    best_h = base.copy()
    best_score = compute_c5(best_h)
    # Allocate time to initial hill climb
    alloc = time_left() * 0.6
    if alloc > 0:
        cand_h, cand_score = hill_climb(best_h, alloc, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
    # Multi-start restarts
    # Determine number of restarts based on remaining time
    while time_left() > 5.0:
        # allocate portion for each restart
        remaining = time_left()
        # decide per restart about 5% of total or 30 seconds whichever smaller
        per = min(30.0, remaining * 0.2)
        # generate random start
        start_h = random_balanced_binary(target_n, rng).astype(float)
        start_h = project_onto_box_sum(start_h, target_n/2.0)
        cand_h, cand_score = hill_climb(start_h, per, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
            # If we beat target, break early
            if best_score <= 0.3808:
                break
    # Final projection to ensure constraints
    best_h = project_onto_box_sum(best_h, target_n/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, target_n

import time
import numpy as np

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def hill_climb(h0, time_budget, rng):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    start = time.time()
    step_factor = 0.5
    it = 0
    max_it = 2000000
    while time.time() - start < time_budget and it < max_it:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        max_val = float(corr[max_idx])
        # update best if improvement
        cur_score = max_val * (2.0 / n)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        # shift
        k = max_idx - (n - 1)
        if k >= 0:
            i_range = np.arange(n - k)
            i2_range = i_range + k
        else:
            i_range = np.arange(-k, n)
            i2_range = i_range + k
        # contributions
        contrib = h[i_range] * (1.0 - h[i2_range])
        # choose max contribution
        idx_local = int(np.argmax(contrib))
        i = int(i_range[idx_local])
        i2 = int(i2_range[idx_local])
        max_delta = min(h[i], 1.0 - h[i2])
        if max_delta <= 0:

        h[i] -= delta
        h[i2] += delta

    final_score = compute_c5(h)
    if final_score < best_score:
        best_h = h.copy()
        best_score = final_score
