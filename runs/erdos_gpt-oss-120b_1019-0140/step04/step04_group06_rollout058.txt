        # SA schedule
        T0, T1 = 0.1, 1e-5
        max_iters = int(min(5e5, time_left() * 2000))
        for it in range(max_iters):
            if time_left() <= 0.0:
                break
            frac = it / max_iters
            T = T0 * ((T1 / T0) ** frac)

            # propose a single 1↔0 swap
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            new_c5 = compute_c5(h_new)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < math.exp(-delta / T):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                if best_c5 <= target_c5:
                    break

            # occasional multi-swap perturbation
            if rng.random() < 0.01:
                for _ in range(rng.integers(1, 4)):
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h[i] = 0.0
                    h[j] = 1.0
                cur_c5 = compute_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            # early exit if target reached
            if best_c5 <= target_c5:
                break

        # if SA didn't improve enough, random restart
        if best_c5 > target_c5 and time_left() > 2.0:
            h = np.zeros(n, dtype=np.float64)
            h[:n // 2] = 1.0
            rng.shuffle(h)
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # 4. final clean‑up: enforce exact binary and sum constraint
    n_final = best_h.shape[0]
    best_h = (best_h > 0.5).astype(np.float64)
    cur_sum = int(best_h.sum())
    target_sum = n_final // 2
    if cur_sum != target_sum:
        if cur_sum > target_sum:
            idx = np.where(best_h == 1)[0]
            to_zero = rng.choice(idx, size=cur_sum - target_sum, replace=False)
            best_h[to_zero] = 0.0
        else:
            idx = np.where(best_h == 0)[0]
            to_one = rng.choice(idx, size=target_sum - cur_sum, replace=False)
            best_h[to_one] = 1.0

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
