def sa_optimize(h0, time_limit, rng, init_temp=0.01, p_direct=0.7, target=0.3808):
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    iter = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # compute temperature proportional to remaining time
        temp = init_temp * max(0.0, 1.0 - elapsed / time_limit)
        # decide move type
        if rng.random() < p_direct:
            # directed move
            # compute worst shift if not stored
            # compute correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k = abs(max_idx - (n - 1))
            if k == 0:
                # fallback random
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i: j += 1
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0: continue
                delta = max_delta * 0.01
                h[i] -= delta
                h[j] += delta
            else:

import time
import numpy as np
import cvxpy as cp

def project_onto_box_sum(v, total, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = total}. """
    n = v.size
    if total <= lo * n:
        return np.full_like(v, lo)
    if total >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        s = f(lam_mid)
        if abs(s - total) < 1e-12:
            break
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(m):
    seq = np.empty(m, dtype=float)
    for i in range(m):
        seq[i] = (bin(i).count('1') & 1)
    return seq

def sa_optimize(h0, time_limit, rng, init_temp=0.02, p_direct=0.78,
                target=0.3808):
    """Simulated annealing with directed moves aimed at reducing the worst shift."""
    h = h0.copy()
    n = h.size
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    eps = 1e-12

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        remaining = time_limit - elapsed
        temp = init_temp * max(0.0, remaining / time_limit)

        # occasional random restart
        if rng.random() < 0.001:
            h = random_balanced_binary(n, rng)
            cur_score = compute_c5(h)
            continue

        # decide move type
        if rng.random() < p_direct:
            # directed move based on worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            k = abs(np.argmax(corr) - (n - 1))
            if k == 0:
                # fallback to random pair
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * (temp / init_temp) * 0.5
                if delta <= 0:
                    delta = max_delta * 0.01
                h[i] -= delta
                h[j] += delta
                move = ('rand', i, j, delta)
            else:
                # neighbor sum for shift k
                ns = np.zeros(n)
                ns[:n - k] += h[k:]
                ns[k:] += h[:n - k]
                low_idx = np.where(h > eps)[0]
                high_idx = np.where(h < 1.0 - eps)[0]
                if low_idx.size == 0 or high_idx.size == 0:
                    continue
                i_low = low_idx[np.argmin(ns[low_idx])]
                i_high = high_idx[np.argmax(ns[high_idx])]
                max_delta = min(h[i_low], 1.0 - h[i_high])
                if max_delta <= 0:
                    continue
                delta = max_delta * (temp / init_temp) * 0.5
                if delta <= 0:
                    delta = max_delta * 0.01
                h[i_low] -= delta
                h[i_high] += delta
                move = ('direct', i_low, i_high, delta)
        else:
            # random mass transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta * (temp / init_temp) * 0.5
            if delta <= 0:
                delta = max_delta * 0.01
            h[i] -= delta
            h[j] += delta
            move = ('rand', i, j, delta)

        new_score = compute_c5(h)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                if best_score <= target:
                    break
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                cur_score = new_score
            else:
                # revert move
                typ, a, b, d = move
                if typ == 'direct' or typ == 'rand':
                    h[a] += d
                    h[b] -= d
        # occasional re-projection to maintain exact sum
        if (rng.integers(1000) == 0):

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
