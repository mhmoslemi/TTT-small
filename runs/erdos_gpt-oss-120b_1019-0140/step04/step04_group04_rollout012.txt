def adam_optimize(h0, time_left, max_iters):
    h = h0.copy()
    n = h.size
    s = n/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # initial learning rate
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # compute gradient at current max shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat)+eps)
        # decaying lr schedule
        step = step / np.sqrt(it)
        h_new = h - step
        # occasional noise injection
        if it % 2000 == 0:
            h_new += np.random.default_rng().normal(scale=0.01, size=n)
        # project
        h_new = project_onto_box_sum(h_new, s)
        # evaluate
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def project_onto_box_sum(v, s, lo=0., hi=1.):
    # Clip v to [lo,hi] then adjust to meet sum s using water-filling
    # done by sorting (v - lo) and (hi - v) contributions? Let's write algorithm:
    n = v.size
    # start with clipping
    x = np.clip(v, lo, hi)
    excess = x.sum() - s
    if abs(excess) < 1e-10:
        return x
    # If sum too high, we need to reduce some values above lo.
    # If sum too low, we need to increase some values below hi.
    # Use sorting to compute threshold.
    if excess > 0:
        # reduce excess from components > lo
        # sort descending x
        sorted_idx = np.argsort(-x)  # descending
        x_sorted = x[sorted_idx]
        cum = np.cumsum(x_sorted - lo)
        # find k where cum[k-1] >= excess
        k = np.searchsorted(cum, excess) + 1  # number of elements to adjust
        # threshold = (cum[k-1] - excess) / k??? Wait we need to subtract amount from top k.
        # Actually we need to reduce each of the top k values by same amount delta so that total reduction = excess.
        delta = (cum[k-1] - excess) / k
        # set new top k = x_sorted[:k] - delta, rest unchanged
        x_new = x_sorted.copy()
        x_new[:k] = x_sorted[:k] - delta
        # reorder back
        x = np.empty_like(x)
        x[sorted_idx] = x_new
    else:
        # need to increase sum (excess negative)
        deficit = -excess
        # increase low values
        sorted_idx = np.argsort(x)  # ascending
        x_sorted = x[sorted_idx]
        cum = np.cumsum(hi - x_sorted)
        k = np.searchsorted(cum, deficit) + 1
        delta = (cum[k-1] - deficit) / k
        x_new = x_sorted.copy()
        x_new[:k] = x_sorted[:k] + delta
        x = np.empty_like(x)
        x[sorted_idx] = x_new
    return x
```

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # adjust bounds if needed
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad
```

def adam_optimize(h0, time_left, max_iters):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    rng = np.random.default_rng()
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        score = max_corr * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # identify shifts achieving max (tolerance)
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        # decay step size
        step = step / np.sqrt(it)
        # occasional perturbation
        if it % 2000 == 0:
            h = h + rng.normal(scale=0.01, size=n)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
    # final evaluation
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score
```

def slsqp_optimize_candidate(h0, max_iter=500):
    n = h0.size
    s = n / 2.0
    def objective(x):
        return x[-1]
    def eq_sum(x):
        return np.sum(x[:n]) - s
    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': eq_sum},
            {'type': 'ineq', 'fun': ineq_fun}]
    res = opt.minimize(objective, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, s)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt
```

import time
import numpy as np
import scipy.optimize as opt

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def thue_morse_arr(m):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(m)), dtype=float, count=m)

def adam_optimize(h0, start_time, budget_s, max_iters):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    rng = np.random.default_rng()
    for it in range(1, max_iters + 1):
        if budget_s - (time.time() - start_time) <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        score = max_corr * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        step = step / np.sqrt(it)
        if it % 2000 == 0:
            h = project_onto_box_sum(h + rng.normal(scale=0.01, size=n), s)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        h = h_new
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    return best_h, best_score

def slsqp_obj(x):
    return x[-1]

def slsqp_eq_sum(x, n, s):
    return np.sum(x[:n]) - s

def slsqp_ineq(x, n):
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - corr

def slsqp_optimize_candidate(h0, max_iter=800):
    n = h0.size
    s = n / 2.0
    x0 = np.concatenate([h0, [compute_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'eq', 'fun': slsqp_eq_sum, 'args': (n, s)},
            {'type': 'ineq', 'fun': slsqp_ineq, 'args': (n,)}]
    res = opt.minimize(slsqp_obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, s)
    t_opt = compute_c5(h_opt)
    return h_opt, t_opt

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    n = 512
    candidates = []
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)
        candidates.append(base)
    rand_cont = rng.random(n)
    rand_cont = project_onto_box_sum(rand_cont, n / 2.0)
    candidates.append(rand_cont)
    rand_bin = np.zeros(n, dtype=float)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
    candidates.append(rand_bin)
    candidates.append(thue_morse_arr(n))
    best_h = None
    best_score = np.inf
    remaining = len(candidates)
    for cand in candidates:
        if budget_s - (time.time() - start) <= 5:
            break
        iters = int(min(20000, max(0, (budget_s - (time.time() - start)) * 2000 / (remaining + 1))))
        h_opt, sc_opt = adam_optimize(cand, start, budget_s, iters)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        remaining -= 1
    if best_h is not None and budget_s - (time.time() - start) > 10:
        iters = int(max(0, (budget_s - (time.time() - start)) * 2000))
        h_opt, sc_opt = adam_optimize(best_h, start, budget_s, iters)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
    if best_h is not None and budget_s - (time.time() - start) > 5:
        try:
            h_opt, sc_opt = slsqp_optimize_candidate(best_h, max_iter=800)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
        except Exception:
            pass
    if best_h is None:
        best_h = project_onto_box_sum(np.zeros(n), n / 2.0)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
