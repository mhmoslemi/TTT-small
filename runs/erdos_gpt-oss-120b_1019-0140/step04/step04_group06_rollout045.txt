def adam_optimize(h0, time_left):
    # hyperparameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h0)
    v = np.zeros_like(h0)
    h = h0.copy()
    t = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    while time_left() > 0:
        t += 1
        grad = compute_gradients_for_shifts(h, max_shifts(h))
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # projection
        h_new = project_onto_box_sum(h_new, h_new.size/2.0)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)
    # Determine scales
    scales = [64, 128, 256, 512]
    best_h = None
    best_score = float('inf')
    for s_idx, n in enumerate(scales):
        # upsample base to n
        if base.shape[0] != n:
            base = upsample(base, n)
        # project to feasible
        base = project_onto_box_sum(base, n/2.0)
        # Do a few restarts of Adam
        restarts = 5 if n < 512 else 3
        for r in range(restarts):
            if time_left() < 1.0:
                break
            # random binary balanced initialization or perturbed base
            if r == 0:
                h0 = base.copy()
            else:
                h0 = random_binary_balanced(n, rng)
                # add small noise
                h0 = h0 + 0.02 * (rng.random(n) - 0.5)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n/2.0)
            h_opt, sc_opt = adam_optimize(h0, time_left)
            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
        # After restarts, set base = best_h for next scale
        if best_h is not None and best_h.size == n:
            base = best_h.copy()
    # Final local search refinement
    if best_h is None:
        best_h = base.copy()
        best_score = compute_c5(best_h)
    n_final = best_h.size
    # Pairwise tweaks
    while time_left() > 2.0:
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.05)
        if delta <= 0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        # break if target reached
        if best_score <= 0.3808:
            break
    best_h = project_onto_box_sum(best_h, best_h.size/2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size

import time, numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def max_shifts(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = corr.max()
    idx = np.where(corr == maxc)[0]
    return (idx - (n - 1)).tolist()

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros_like(h)
            term2 = np.zeros_like(h)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if shifts:
        grad /= len(shifts)
    return grad

def adam_optimize(h0, time_left, rng):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    beta1_pow = beta1
    beta2_pow = beta2
    t = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    while time_left() > 0.5:
        t += 1
        shifts = max_shifts(h)
        grad = compute_gradients_for_shifts(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if t % 500 == 0:
            h_new += 0.01 * (rng.random(n) - 0.5)
            h_new = project_onto_box_sum(h_new, target_sum)
        h = h_new
        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64)

    scales = [64, 128, 256, 512]
    best_h = None
    best_score = float('inf')

    for n in scales:
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # allocate restarts proportionally to remaining time
        max_restarts = 12 if n == 512 else (8 if n == 256 else (6 if n == 128 else 4))
        restarts = 0
        while time_left() > 5.0 and restarts < max_restarts:
            restarts += 1
            if restarts == 1:
                h0 = base.copy()
            else:
                h0 = random_binary_balanced(n, rng)
                h0 += 0.02 * (rng.random(n) - 0.5)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = project_onto_box_sum(h0, n / 2.0)
            h_opt, sc_opt = adam_optimize(h0, time_left, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_h = h_opt.copy()
        if best_h is not None and best_h.size == n:
            base = best_h.copy()

    if best_h is None:
        best_h = base.copy()
        best_score = compute_c5(best_h)

    n_final = best_h.size
    # final random pairwise tweaks
    while time_left() > 2.0:
        i, j = rng.choice(n_final, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.01, 0.05)
        if delta <= 0.0:
            continue
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        if best_score <= 0.3808:
            break

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
