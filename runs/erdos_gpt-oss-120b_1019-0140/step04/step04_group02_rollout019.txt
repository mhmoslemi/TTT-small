    global global_slsqp_n, global_slsqp_dx
    global_slsqp_n = n

    # Compute initial t
    corr = np.correlate(h0, 1.0 - h0, mode='full')

    x0 = np.concatenate([h0, [t0]])

def adam_optimize(h0, n_iter, alpha, beta1, beta2, eps, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n/2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    corr = compute_corr_full_fft(h)
    max_c = np.max(corr)
    best_score = max_c * (2.0 / n)
    for t in range(1, n_iter+1):
        if time_left() <= 0:
            break
        # compute gradient (same as before)
        grad = compute_grad_max_shifts(h, corr, max_c)
        # Adam updates
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
        # Project onto box sum
        h = project_onto_box_sum(h, s)
        # Recompute corr and max_c
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        score = max_c * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

import time
import numpy as np
import scipy.optimize as opt

# Global for SLSQP constraints
_slsqp_n = 0

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # ensure brackets
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def upsample(arr, new_n):
    """Linear interpolation from arr (size m) to new_n points."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_corr_full_fft(h):
    """Full cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr


def compute_c5_fft(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    max_c = np.max(corr)
    return max_c * (2.0 / n)


def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    """Gradient of the max‑correlation term (averaged over all max shifts)."""
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def adam_optimize(h0, n_iter, alpha, beta1, beta2, eps, rng, time_left):
    """Adam‑style projected optimization for the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    corr = compute_corr_full_fft(h)
    max_c = np.max(corr)
    best_h = h.copy()
    best_score = max_c * (2.0 / n)

    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        grad = compute_grad_max_shifts(h, corr, max_c)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        h = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
        # occasional noise to escape plateaus
        if (it % 2000) == 0:
            h += rng.normal(scale=0.02, size=n)
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target_sum)

        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        score = max_c * (2.0 / n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def slsqp_obj(x):
    return x[-1]


def slsqp_eq_sum(x):
    n = _slsqp_n
    return np.sum(x[:n]) - n / 2.0


def slsqp_ineq(x):
    n = _slsqp_n
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr


def slsqp_optimize(h0, max_iter, rng, time_left):
    """Refine h0 via SLSQP on the explicit max‑constraint formulation."""
    global _slsqp_n
    n = h0.shape[0]
    _slsqp_n = n

    corr = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = np.max(corr) * 1.01  # slight slack to stay feasible
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, n / 2.0)]
    cons = [{'type': 'eq', 'fun': slsqp_eq_sum},
            {'type': 'ineq', 'fun': slsqp_ineq}]

    res = opt.minimize(slsqp_obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x[:n]
    h_opt = project_onto_box_sum(h_opt, n / 2.0)
    c5 = compute_c5_fft(h_opt)
    return h_opt, c5


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start))

    # Base construction
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(128).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    TARGET = 0.3808
    best_h = None
    best_score = np.inf

    # Resolution cascade
    for n in (128, 256, 512):
        if time_left() < 5.0:
            break
        if base.shape[0] != n:
            base = upsample(base, n)
        base = project_onto_box_sum(base, n / 2.0)

        # Choose iteration count and restarts adaptively
        if n == 128:
            restarts, n_iter = 8, 250000
        elif n == 256:
            restarts, n_iter = 5, 200000
        else:
            restarts, n_iter = 3, 150000

        for r in range(restarts):
            if time_left() < 2.0:
                break
            # Random perturbation around base
            h0 = base + rng.normal(scale=0.05, size=n)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, n / 2.0)

            alpha = 0.6 * (128.0 / n)  # step size scaled with resolution
            h_opt, sc_opt = adam_optimize(
                h0, n_iter=n_iter, alpha=alpha,
                beta1=0.9, beta2=0.999, eps=1e-8,
                rng=rng, time_left=time_left)

            if sc_opt < best_score:
                best_score, best_h = sc_opt, h_opt.copy()
                if best_score <= TARGET:
                    break

            # Light SLSQP refinement on the 128‑point stage only
            if n == 128 and time_left() > 5.0:
                try:
                    h_sls, sc_sls = slsqp_optimize(h_opt, max_iter=500,
                                                  rng=rng, time_left=time_left)
                    if sc_sls < best_score:
                        best_score, best_h = sc_sls, h_sls.copy()
                        if best_score <= TARGET:
                            break
                except Exception:
                    pass

        if best_h is not None:
            base = best_h.copy()
        if best_score <= TARGET:
            break

    # Final random binary and small mass‑transfer search
    while time_left() > 4.0:
        if best_h is None:
            break
        n = best_h.size

        # Random balanced binary candidate
        h_bin = random_binary_balanced(n, rng)
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()
            if best_score <= TARGET:
                break

        # Small random mass transfer
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_h[i], 1.0 - best_h[j]) * rng.uniform(0.05, 0.2)
        if delta > 1e-12:
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                if best_score <= TARGET:
                    break

    # Attempt binary rounding and greedy swap improvement
    if best_h is not None:
        n = best_h.size
        target = n // 2
        h_bin = (best_h >= 0.5).astype(float)
        ones = int(h_bin.sum())
        if ones > target:
            idx_one = np.where(h_bin == 1)[0]
            to_zero = rng.choice(idx_one, size=ones - target, replace=False)
            h_bin[to_zero] = 0.0
        elif ones < target:
            idx_zero = np.where(h_bin == 0)[0]
            to_one = rng.choice(idx_zero, size=target - ones, replace=False)
            h_bin[to_one] = 1.0
        sc_bin = compute_c5_fft(h_bin)
        if sc_bin < best_score:
            best_score, best_h = sc_bin, h_bin.copy()

        # Greedy binary swap hill‑climb
        swaps = 2000
        for _ in range(swaps):
            if time_left() <= 0.0:
                break
            idx_one = np.where(best_h == 1)[0]
            idx_zero = np.where(best_h == 0)[0]
            if idx_one.size == 0 or idx_zero.size == 0:
                break
            i = rng.choice(idx_one)
            j = rng.choice(idx_zero)
            h_new = best_h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            sc_new = compute_c5_fft(h_new)
            if sc_new < best_score:
                best_score, best_h = sc_new, h_new.copy()
                # continue with updated arrays
        # final clean projection
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_score = compute_c5_fft(best_h)
        n_ret = best_h.size
    else:
        # Fallback construction
        n_ret = 512
        best_h = project_onto_box_sum(thue_morse(n_ret), n_ret / 2.0)
        best_score = compute_c5_fft(best_h)

    return best_h, best_score, n_ret
