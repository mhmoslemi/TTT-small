def aggressive_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = h.shape[0]
    while time_left() > 0.1:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_corr = np.max(corr)
        # find all shift indices where correlation close to max
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shift = shift_idxs[rng.integers(shift_idxs.size)] - (n - 1)
        # compute contribution for this shift
        if shift >= 0:
            contrib = cur_h[:n-shift] * (1.0 - cur_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = cur_h[s:] * (1.0 - cur_h[:n-s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # if cannot move, try a random small jitter and continue
            delta = 0.0
        else:
            # move a large fraction of allowed delta
            delta = max_delta * (0.5 + 0.5 * rng.random())
            cur_h[i] -= delta
            cur_h[j] += delta
        new_score = compute_c5(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert move
            cur_h[i] += delta
            cur_h[j] -= delta
            # occasional random jitter to escape
            if rng.random() < 0.01:
                idx = rng.integers(n)
                delta_j = min(cur_h[idx], 1.0 - cur_h[idx]) * rng.random() * 0.5
                # move small amount to random other position
                idx2 = rng.integers(n)
                max_d2 = min(cur_h[idx], 1.0 - cur_h[idx2])
                if max_d2 > 0:
                    d = max_d2 * rng.random() * 0.1
                    cur_h[idx] -= d
                    cur_h[idx2] += d
    return best_h, best_score

def simulated_annealing(h, rng, time_left):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = h.shape[0]
    start = time.time()
    # define schedule: temperature decreases linearly from t0 to 0 over allowed time.
    tot = time_left()
    while time_left() > 0:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / tot
        temperature = (1 - frac) * cur_score * 0.1  # scale
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.3
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        delta_score = new_score - cur_score
        if delta_score <= 0 or rng.random() < np.exp(-delta_score / (temperature+1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_score
```

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def upsample(arr, new_n):
    """Linear interpolation from a smaller array to a larger size."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def random_binary_balanced(n, rng):
    """Balanced binary vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size and momentum."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    return best_h, best_score


def local_pair_moves(h, rng, time_left):
    """Transfer a small mass between random positions, accepting improvements."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:
                best_h, best_score = h_new, score
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            score2 = compute_c5(h_new2)
            if score2 < best_score:
                best_h, best_score = h_new2, score2
    return best_h, best_score


def aggressive_max_shift(h, rng, time_left):
    """Large‑delta moves targeting the worst shift."""
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()

    n = h.shape[0]
    while time_left() > 0.1:

        max_corr = np.max(corr)
        shift_idxs = np.where(corr >= max_corr - 1e-12)[0]
        shift = shift_idxs[rng.integers(shift_idxs.size)] - (n - 1)
        if shift >= 0:
            contrib = cur_h[:n - shift] * (1.0 - cur_h[shift:])
            src = np.argmax(contrib)
            i = src
            j = src + shift
        else:
            s = -shift
            contrib = cur_h[s:] * (1.0 - cur_h[:n - s])
            src = np.argmax(contrib) + s
            i = src
            j = src - s
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # small random perturbation to escape stagnation
            if rng.random() < 0.02:
                p = rng.integers(n)
                q = rng.integers(n)
                max_d = min(cur_h[p], 1.0 - cur_h[q])
                if max_d > 0:
                    d = max_d * rng.random() * 0.2
                    cur_h[p] -= d
                    cur_h[q] += d
            continue
        delta = max_delta * (0.5 + 0.5 * rng.random())
        cur_h[i] -= delta
        cur_h[j] += delta
        new_score = compute_c5(cur_h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
            if rng.random() < 0.01:

                    cur_h[q] += d

        if delta_score <= 0 or rng.random() < np.exp(-delta_score / temperature):

                best_h = cur_h.copy()
        else:
            cur_h[i] += delta
