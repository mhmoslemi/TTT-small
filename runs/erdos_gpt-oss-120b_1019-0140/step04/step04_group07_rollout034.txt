def anneal_binary(h, corr, time_left):
    n = h.shape[0]
    cur_h = h.copy()
    cur_corr = corr.copy()
    cur_score = np.max(cur_corr) * (2.0/n)
    best_h = cur_h.copy()
    best_score = cur_score
    T = 0.001
    decay = 0.9995
    step = 0

    f = 1.0 - cur_h
    while time_left() > 1.0:
        # random swap of opposite bits
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        # compute dh
        dh_i = 1 - 2*cur_h[i]
        dh_j = -dh_i
        # copy corr
        new_corr = cur_corr.copy()
        # update new_corr
        # f and cur_h are needed for both i and j. use them
        new_corr[i:i+n] += dh_i * f
        new_corr[j:j+n] += dh_j * f
        new_corr[i:i+n] -= dh_i * cur_h
        new_corr[j:j+n] -= dh_j * cur_h
        center = n - 1
        d_shift = i - j
        new_corr[center] -= 2
        new_corr[center + d_shift] += 2
        new_corr[center - d_shift] += 2

        new_score = np.max(new_corr) * (2.0 / n)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]  # swap
            # Update f as well (since f depends on h)
            f[i] = 1.0 - cur_h[i]
            f[j] = 1.0 - cur_h[j]
            cur_corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        # update temperature
        T *= decay
        step += 1
    return best_h, best_score

import time
import math
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen bounds if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo

    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def thue_morse(n):
    """Thue–Morse binary sequence of length n (0/1)."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)


def compute_gradients_for_shifts(h, shifts):
    """Average gradient of the max‑correlation term over given integer shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term1[n - shift:] = 0.0
            term2[shift:] = h[:n - shift]
            term2[:shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n, dtype=h.dtype)
            term2 = np.empty(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term1[:s] = 0.0
            term2[:n - s] = h[s:]
            term2[n - s:] = 0.0
        grad += (1.0 - term1 - term2)
    if shifts.size > 0:
        grad /= float(shifts.size)
    return grad


def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    """Projected gradient descent with adaptive step size."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0

    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

    return best_h, best_score


def anneal_binary(h, corr, time_left, rng):
    """Simulated annealing on a binary (0/1) vector using O(n) updates."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_f = 1.0 - cur_h
    cur_corr = corr.copy()
    cur_score = np.max(cur_corr) * (2.0 / n)
    best_h = cur_h.copy()
    best_score = cur_score

    T = 0.001
    decay = 0.9995
    center = n - 1

    while time_left() > 1.0:
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        dh_i = 1.0 - 2.0 * cur_h[i]   # +1 if 0->1, -1 if 1->0
        dh_j = -dh_i

        new_corr = cur_corr.copy()
        # term A: conv(delta_h, f)
        new_corr[i:i + n] += dh_i * cur_f
        new_corr[j:j + n] += dh_j * cur_f
        # term B: conv(h, delta_h) using reversed h
        h_rev = cur_h[::-1]
        new_corr[i:i + n] -= dh_i * h_rev
        new_corr[j:j + n] -= dh_j * h_rev
        # term D: conv(delta_h, delta_h)
        d_shift = i - j
        new_corr[center] -= 2.0
        new_corr[center + d_shift] += 2.0
        new_corr[center - d_shift] += 2.0

        new_score = np.max(new_corr) * (2.0 / n)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            # accept swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            cur_f[i] = 1.0 - cur_h[i]
            cur_f[j] = 1.0 - cur_h[j]
            cur_corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        T *= decay
    return best_h, best_score

        max_iters = int(min(80000, max(20000, time_left() * 30)))
        best_h, best_score = gradient_descent_adaptive(
            h_float, max_iters, step0=0.3, beta=0.9, rng=rng, time_left=time_left
        )

    # final projection and score recomputation
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
