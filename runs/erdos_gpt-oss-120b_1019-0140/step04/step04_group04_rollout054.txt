import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=80):
    """Project v onto [lo,hi]^n with exact sum s (water-filling)."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    lam_mid = 0.0
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    """Linear interpolation of arr to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    """Return first n values of Thue–Morse sequence as floats 0/1."""
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def compute_gradients_for_shifts(h, shifts):
    """Subgradient of max_{k} ∫ h(1‑h(..)) for the given max shifts."""
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[:n - shift] = h[shift:]
            term2[shift:] = h[:n - shift]
        else:
            s = -shift
            term1 = np.zeros(n, dtype=h.dtype)
            term2 = np.zeros(n, dtype=h.dtype)
            term1[s:] = h[:n - s]
            term2[:n - s] = h[s:]
        grad += (1.0 - term1 - term2)
    if len(shifts):
        grad /= len(shifts)
    return grad

def gradient_descent_custom(h0, n_iter, step0, beta, rng, time_left):
    """Momentum gradient descent on the C5 objective with projection."""
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        idxes = np.where(corr >= max_corr - 1e-12)[0]
        shifts = idxes - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 1000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
        h = h_new
    return best_h, best_score

def binary_hill_climb(h_bin, max_swaps, rng, time_left):
    """Greedy hill climb on binary (0/1) vectors by swapping a 1 with a 0."""
    n = h_bin.size
    best = h_bin.copy()
    best_score = compute_c5(best)
    ones = np.where(best == 1)[0].tolist()
    zeros = np.where(best == 0)[0].tolist()
    for _ in range(max_swaps):
        if time_left() <= 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = best.copy()
        new[i] = 0
        new[j] = 1
        new_score = compute_c5(new)
        if new_score < best_score:
            best = new
            best_score = new_score
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
    return best, best_score

            base = h_gd

        # binary rounding and hill climbing
        k = n // 2
        idx_top = np.argpartition(base, -k)[-k:]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx_top] = 1.0
        bin_score = compute_c5(h_bin)
        if bin_score < best_score:
            best_h, best_score = h_bin.copy(), bin_score

        max_swaps = int(min(60000, time_left() * 2000))
        if max_swaps > 0:
            h_hc, sc_hc = binary_hill_climb(h_bin, max_swaps, rng, time_left)
            if sc_hc < best_score:
                best_h, best_score = h_hc.copy(), sc_hc
            base = best_h  # propagate improved binary solution

            # a couple of random restarts if time permits
            attempts = 0
            while time_left() > 2.0 and attempts < 2:
                rand_bin = np.zeros(n, dtype=float)
                rand_bin[rng.choice(n, size=k, replace=False)] = 1.0
                swaps = int(min(30000, time_left() * 2000))
                h_rand, sc_rand = binary_hill_climb(rand_bin, swaps, rng, time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand.copy(), sc_rand
                    base = best_h
                attempts += 1

    # final continuous refinement if time remains
    if time_left() > 2.0:
        n = best_h.size
        max_iter = int(min(20000, time_left() * 2000))
        if max_iter > 0:
            h_fin, sc_fin = gradient_descent_custom(
                best_h, max_iter, step0=0.2, beta=0.95, rng=rng, time_left=time_left)
            if sc_fin < best_score:
                best_h, best_score = h_fin.copy(), sc_fin

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
