def gradient_descent(h0, time_left, max_iter, step0, beta):

Initialize:

h = h0.copy()
n = h.size
s = n / 2.0

best_score = compute_c5(h)
momentum = np.zeros_like(h)

    if time_left() <= 0:
        break
    shifts = worst_shifts(h)
    grad = compute_gradient(h, shifts)
    momentum = beta * momentum + (1 - beta) * grad
    eta = step0 / np.sqrt(it)
    h_new = h - eta * momentum
    # maybe add small Gaussian noise occasionally to escape local minima

    score = compute_c5(h_new)

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=70):
    """Clamp to [lo,hi] while enforcing sum(v)=s via water‑filling (binary search)."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_lo = np.min(v) - hi
    lam_hi = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # expand bounds if needed
    if f(lam_lo) < s:
        lam_lo = np.min(v) - 10 * hi
    if f(lam_hi) > s:
        lam_hi = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_lo + lam_hi) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def worst_shifts(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    idx = np.where(corr >= maxc - 1e-12)[0]
    return idx - (n - 1)

def compute_gradient(h, shifts):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for sh in shifts:
        if sh >= 0:
            # contribution from h[i+sh]
            grad[:n - sh] += 1.0 - h[sh:]
            # contribution from h[i-sh]
            grad[sh:] += 1.0 - h[:n - sh]
        else:
            s = -sh
            grad[s:] += 1.0 - h[:n - s]
            grad[:n - s] += 1.0 - h[s:]
    if shifts:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, time_left, max_iter, step0, beta, rng, target=0.3808):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = np.zeros_like(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        shifts = worst_shifts(h)
        grad = compute_gradient(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            if best_score <= target:
                break
        h = h_new
    return best_h, best_score

def adam_refine(h0, time_left, max_iter, alpha, beta1, beta2, eps, rng, target=0.3808):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(1, max_iter + 1):
        if time_left() <= 0:
            break
        shifts = worst_shifts(h)
        grad = compute_gradient(h, shifts)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        step = alpha * m / (np.sqrt(v) + eps)
        h_new = h - step
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5(h_new)
        if score < best_score:
            best_score, best_h = score, h_new.copy()
            if best_score <= target:
                break
        h = h_new
    return best_h, best_score
