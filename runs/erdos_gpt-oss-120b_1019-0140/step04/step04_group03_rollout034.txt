        if elapsed >= time_budget - 0.1: break
        corr = np.correlate(h, 1.0 - h, mode='full')

def binary_hill_climb(h_bin, cur_score, rng, start_time, budget_s, max_steps=None):
    n = h_bin.size
    target_sum = int(np.sum(h_bin))
    # Precompute index sets
    ones = np.flatnonzero(h_bin > 0.5)
    zeros = np.flatnonzero(h_bin < 0.5)
    steps = max_steps or 20000
    for step in range(steps):
        if budget_s - (time.time() - start_time) <= 0.0:
            break
        # choose random i from ones, j from zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h_new = h_bin.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h_bin = h_new
            cur_score = new_score
            # update ones, zeros
            ones = np.append(ones[ones != i], j)
            zeros = np.append(zeros[zeros != j], i)
        # else maybe continue
    return h_bin, cur_score

def adam_softmax_descent(h0, max_iters, lr0, lr_min, alpha0, alpha_max, target_sum, start_time, time_budget):
    import time
    n = h0.size
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    for t in range(1, max_iters+1):
        if time_budget - (time.time() - start_time) <= 0.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        alpha = alpha0 + (alpha_max - alpha0) * (t - 1) / max_iters
        w = np.exp(alpha * (corr - max_corr))
        # cutoff
        cutoff = 1e-12
        idxs = np.where(w > cutoff)[0]
        w_vals = w[idxs]
        w_sum = w_vals.sum()
        grad = np.full(n, w_sum, dtype=float)
        for idx_shift, w_s in zip(idxs, w_vals):
            shift = idx_shift - (n - 1)
            if shift >= 0:
                sz = n - shift
                grad[:sz] -= w_s * h[shift:]
                grad[shift:] -= w_s * h[:sz]
            else:
                s = -shift
                sz = n - s
                grad[s:] -= w_s * h[:sz]
                grad[:s] -= w_s * h[sz:]
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr0 * (1 - (t - 1) / max_iters) + lr_min
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        # optional perturbation
        if t % 5000 == 0:
            h_new += np.random.default_rng().normal(scale=0.01, size=n)
        h_new = project_onto_box_sum(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if target_sum <= lo * n:
        return np.full_like(v, lo)
    if target_sum >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    f_low = np.clip(v - lam_low, lo, hi).sum()
    if f_low < target_sum:
        lam_low = np.min(v) - 10 * hi
    f_high = np.clip(v - lam_high, lo, hi).sum()
    if f_high > target_sum:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = np.clip(v - lam_mid, lo, hi).sum()
        if abs(fm - target_sum) < 1e-12:
            break
        if fm > target_sum:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n):
