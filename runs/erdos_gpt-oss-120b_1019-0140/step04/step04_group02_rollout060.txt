def adam_projected(h0, n_iter, step0, betas, eps, rng, time_left):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    for t in range(1, n_iter+1):
        if time_left() <= 0: break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        m = betas[0] * m + (1 - betas[0]) * grad
        v = betas[1] * v + (1 - betas[1]) * (grad ** 2)
        m_hat = m / (1 - betas[0] ** t)
        v_hat = v / (1 - betas[1] ** t)
        h_new = h - step0 * m_hat / (np.sqrt(v_hat) + eps)
        # optional noise every some steps
        if t % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=h.shape)
        h_new = project_onto_box_sum(h_new, len(h)/2)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    # ensure bounds contain root
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def thue_morse(n):
    return np.fromiter(((bin(i).count('1') & 1) for i in range(n)),
                       dtype=float, count=n)

def random_binary_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def compute_corr_full_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.fft(h, L)
    f1mh = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    return corr

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_full_fft(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_max_shifts(h, corr, max_c, tol=1e-12):
    n = h.shape[0]
    idxes = np.where(corr >= max_c - tol)[0]
    shifts = idxes - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            m = -k
            grad[m:] += 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad

def gradient_descent(h0, n_iter, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.shape[0]
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5_fft(h)
    momentum = np.zeros_like(h)
    for it in range(1, n_iter + 1):
        if time_left() <= 0.0:
            break
        corr = compute_corr_full_fft(h)
        max_c = np.max(corr)
        grad = compute_grad_max_shifts(h, corr, max_c)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        h_new = h - eta * momentum
        if it % 2000 == 0:
            h_new += rng.normal(scale=0.02, size=n)
        h_new = project_onto_box_sum(h_new, s)
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def local_search(h_start, rng, time_left):
    h_best = h_start.copy()
    best_score = compute_c5_fft(h_best)
    n = h_best.size
    s_val = n / 2.0
    # small mass transfer moves
    while time_left() > 2.0:
        # random pair
        i, j = rng.choice(n, size=2, replace=False)
        # determine max transferable amount
        delta = min(h_best[i], 1.0 - h_best[j]) * rng.uniform(0.02, 0.1)
        if delta < 1e-12:
            continue
        h_new = h_best.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # projection not needed as we keep sum unchanged and bounds
        score = compute_c5_fft(h_new)
        if score < best_score:
            best_score = score
            h_best = h_new
        # occasional random perturbation
        if rng.random() < 0.005:
            h_tmp = h_best + rng.normal(scale=0.01, size=n)
            h_tmp = project_onto_box_sum(h_tmp, s_val)
            sc_tmp = compute_c5_fft(h_tmp)
            if sc_tmp < best_score:
                best_score = sc_tmp
                h_best = h_tmp
    return h_best, best_score

def binary_local_search(h_bin, rng, time_left):
    h_best = h_bin.copy()
    best_score = compute_c5_fft(h_best)
    n = h_best.size
    # maintain index lists
    ones = np.flatnonzero(h_best > 0.5)
    zeros = np.flatnonzero(h_best < 0.5)
    while time_left() > 1.5 and len(ones) > 0 and len(zeros) > 0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new = h_best.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        sc = compute_c5_fft(h_new)
        if sc < best_score:
            best_score = sc
            h_best = h_new
            # update lists
            ones = np.flatnonzero(h_best > 0.5)
            zeros = np.flatnonzero(h_best < 0.5)
        else:
            # occasional acceptance? skip
            pass
    return h_best, best_score
