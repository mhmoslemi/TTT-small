def binary_hill_climb(h0, rng, time_budget):
    # h0 is a binary vector (0/1), sum = n/2.
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h0.size
    ones_idx = np.where(cur_h > 0.5)[0]
    zeros_idx = np.where(cur_h < 0.5)[0]
    start = time.time()
    # Define temperature schedule: start T = cur_score * 0.01, decay linearly
    while time.time() - start < time_budget:
        # pick random i from ones, j from zeros
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        # acceptance
        # compute delta
        delta = new_score - cur_score
        elapsed = time.time() - start
        T = max(1e-6, (cur_score * 0.02) * (1 - elapsed / time_budget))  # simple schedule
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
            # update indices
            ones_idx = np.where(cur_h > 0.5)[0]
            zeros_idx = np.where(cur_h < 0.5)[0]
        else:
            # reject, revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score
```

Concerns: computing ones_idx and zeros_idx each loop is O(n) time; maybe maintain sets, but fine because n small. Could instead simply update after accepted swap: after swap, replace i with j in ones_idx; but overhead not big.

def round_to_binary(h):
    n = h.size

    # get indices of largest k values
    idx = np.argpartition(-h, k-1)[:k]
    binary = np.zeros_like(h)
    binary[idx] = 1.0

def gradient_descent_adaptive(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    momentum = np.zeros_like(h)
    step = step0
    jitter_interval = max(1000, max_iters // 10)  # maybe
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find shifts achieving max_corr within tolerance
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)

        if new_score <= cur_score:
            h = h_new
            cur_score = new_score
            step = min(step0, step * 1.02)
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1

        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.02, size=n)
            h = project_onto_box_sum(h + jitter, s)
            cur_score = compute_c5(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()

def binary_hill_climb(h0, rng, time_budget):
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h0.size
    start = time.time()
    while time.time() - start < time_budget:
        # pick random 1 and 0 indices
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        elapsed = time.time() - start
        T = max(1e-8, (best_score * 0.02) * (1 - elapsed / time_budget))
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

                best_h, best_score = h_new2, score2
    return best_h, best_score


def round_to_binary(h):
    """Return a binary vector with exactly half entries set to 1 (largest values)."""
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    binary = np.zeros_like(h)
    binary[idx] = 1.0
    return binary


def binary_hill_climb(h0, rng, time_budget):
    """Simulated annealing on binary vectors (swap a 1 with a 0)."""
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = h0.size
    start = time.time()
    while time.time() - start < time_budget:
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        delta = new_score - cur_score
        elapsed = time.time() - start
        T = max(1e-8, (best_score * 0.02) * (1 - elapsed / time_budget))
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # Initial vector
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = thue_morse(64).astype(float)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)

    best_h = base.copy()
    best_score = compute_c5(best_h)

    scales = [64, 128, 256, 512, 1024]
    for idx, n in enumerate(scales):
        if time_left() <= 5:
            break
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        remaining_scales = len(scales) - idx
        allocated = max(5.0, time_left() * 0.85 / remaining_scales)

        # Gradient descent
        if allocated > 5 and time_left() > 5:
            max_iters = int(min(20000, max(2000, allocated * (2000.0 / n))))
            step0 = 0.8 if n <= 256 else 0.5
            gd_h, gd_score = gradient_descent_adaptive(
                base, max_iters, step0=step0, beta=0.9,
                rng=rng, time_left=time_left)
            if gd_score < best_score:
                best_h, best_score = gd_h.copy(), gd_score
            base = gd_h

        # Local pair moves
        if time_left() > 5:
            lp_h, lp_score = local_pair_moves(base, rng, time_left)
            if lp_score < best_score:
                best_h, best_score = lp_h.copy(), lp_score
            base = lp_h

        # Random binary restarts
        if time_left() > 20:
            restarts = int(max(1, time_left() // 25))
            for _ in range(restarts):
                if time_left() < 5:
                    break
                h_rand = random_binary_balanced(n, rng)
                max_iters_rand = int(min(5000, max(500, time_left() * (2000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_adaptive(
                    h_rand, max_iters_rand, step0=0.5, beta=0.9,
                    rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand
                if time_left() > 5:
                    h_rand_pair, sc_rand_pair = local_pair_moves(h_rand_ref, rng, time_left)
                    if sc_rand_pair < best_score:
                        best_h, best_score = h_rand_pair.copy(), sc_rand_pair
                base = best_h

        # Binary hill climbing on larger scales
        if n >= 512 and time_left() > 10:
            binary_start = round_to_binary(best_h)
            bin_time = min(30.0, time_left())
            bh_h, bh_score = binary_hill_climb(binary_start, rng, bin_time)
            if bh_score < best_score:
                best_h, best_score = bh_h.copy(), bh_score
            base = best_h

    # Final refinement
    if time_left() > 5:
        max_iters_final = int(min(40000, max(10000, time_left() * (2000.0 / best_h.size))))
        best_h, best_score = gradient_descent_adaptive(
            best_h, max_iters_final, step0=0.3, beta=0.9,
            rng=rng, time_left=time_left)

    # Final binary hill climbing if time permits
    if time_left() > 5:
        binary_start = round_to_binary(best_h)
        bh_h, bh_score = binary_hill_climb(binary_start, rng, time_left())
        if bh_score < best_score:
            best_h, best_score = bh_h.copy(), bh_score

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
