def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    s = h.size/2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    velocity = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # compute subgradient
        grad = compute_subgradient(h)
        # momentum
        velocity = beta * velocity + grad
        h_new = h - step * velocity
        h_new = project_onto_box_sum(h_new, s)

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    # optionally limit iterations
    while time_left() > 0:
        i, j = rng.choice(n, size=2, replace=False)
        # try moving from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            score = compute_c5(h_new)
            if score < best_score:

def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    s = h.size / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    velocity = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters+1):
        if time_left() <= 0:
            break
        # compute corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        # Nesterov momentum: v = beta * v + step * grad; h_new = h - v
        velocity = beta * velocity + grad
        h_new = h - step * velocity
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        # accept if improved
        if new_score <= best_score:
            best_h = h_new.copy()
            best_score = new_score
        if new_score < compute_c5(h):
            h = h_new
            # increase step gradually
            step = min(step0, step * 1.02)
        else:
            # reject, reduce step
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        # optional jitter
        if it % 5000 == 0:
            h = project_onto_box_sum(h + rng.normal(scale=0.005, size=h.shape), s)
    return best_h, best_score

import time
import numpy as np

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    n = v.size
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    def f(lam):
        return np.clip(v - lam, lo, hi).sum()
    if f(lam_low) < s:
        lam_low = np.min(v) - 10 * hi
    if f(lam_high) > s:
        lam_high = np.max(v) - 10 * lo
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def upsample(arr, new_n):
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradients_for_shifts(h, shifts):
    n = h.size
    grad = np.zeros_like(h)
    for shift in shifts:
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:] - h[:n - shift])
            grad[shift:] += (1.0 - h[:n - shift] - h[shift:])
        else:
            s = -shift
            grad[s:] += (1.0 - h[:n - s] - h[s:])
            grad[:n - s] += (1.0 - h[s:] - h[:n - s])
    grad /= len(shifts)
    return grad

def gradient_descent_nesterov(h0, max_iters, step0, beta, rng, time_left):
    h = h0.copy()
    s = h.size / 2.0
    cur_score = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    velocity = np.zeros_like(h)
    step = step0
    for it in range(1, max_iters + 1):
        if time_left() <= 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (h.size - 1)
        grad = compute_gradients_for_shifts(h, shifts)
        velocity = beta * velocity + grad
        h_new = h - step * velocity
        h_new = project_onto_box_sum(h_new, s)
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_h, best_score = h_new.copy(), new_score
        if new_score < cur_score:
            h, cur_score = h_new, new_score
            step = min(step0, step * 1.02)
        else:
            step *= 0.5
            if step < 1e-6:
                step = step0 * 0.1
        if it % 5000 == 0:
            h = project_onto_box_sum(h + rng.normal(scale=0.005, size=h.shape), s)
            cur_score = compute_c5(h)
    return best_h, best_score

def local_pair_moves(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    it = 0
    max_it = 20000
    while time_left() > 0 and it < max_it:
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            s = compute_c5(h_new)
            if s < best_score:
                best_h, best_score = h_new, s
        max_delta2 = min(best_h[j], 1.0 - best_h[i])
        if max_delta2 > 0:
            delta2 = max_delta2 * rng.random() * 0.5
            h_new2 = best_h.copy()
            h_new2[j] -= delta2
            h_new2[i] += delta2
            s2 = compute_c5(h_new2)
            if s2 < best_score:
                best_h, best_score = h_new2, s2
        it += 1
    return best_h, best_score

def improve_via_max_shift(h, rng, time_left):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    n = best_h.size
    it = 0
    max_it = 5000
    while time_left() > 0 and it < max_it:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_corr = np.max(corr)
        shifts = np.where(corr >= max_corr - 1e-12)[0] - (n - 1)
        if shifts.size == 0:
            break
        shift = rng.choice(shifts)
        if shift >= 0:
            contrib = best_h[:n - shift] * (1.0 - best_h[shift:])
            i = np.argmax(contrib)
            j = i + shift
        else:
            s = -shift
            contrib = best_h[s:] * (1.0 - best_h[:n - s])
            i = np.argmax(contrib) + s
            j = i - s
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta > 0:
            delta = max_delta * rng.random() * 0.5
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            s_new = compute_c5(h_new)
            if s_new < best_score:
                best_h, best_score = h_new, s_new
        it += 1
    return best_h, best_score

def binary_projection(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = rng.random(64)
    base = project_onto_box_sum(base, base.shape[0] / 2.0)
    best_h, best_score = base.copy(), compute_c5(base)

    scales = [64, 128, 256, 512, 1024]
    for n in scales:
        if time_left() <= 5:
            break
        # Upsample to current scale
        if best_h.size != n:
            base = upsample(best_h, n)
        else:
            base = best_h.copy()
        base = project_onto_box_sum(base, n / 2.0)

        # Gradient descent at this scale
        max_iters = int(min(20000, max(2000, time_left() * (10000.0 / n))))
        h_gd, sc_gd = gradient_descent_nesterov(base, max_iters,
                                                step0=0.5, beta=0.9,
                                                rng=rng, time_left=time_left)
        if sc_gd < best_score:
            best_h, best_score = h_gd.copy(), sc_gd

        # Targeted max-shift improvement
        if time_left() > 2:
            h_ms, sc_ms = improve_via_max_shift(best_h, rng, time_left)
            if sc_ms < best_score:
                best_h, best_score = h_ms.copy(), sc_ms

        # Local pairwise moves
        if time_left() > 2:
            h_pair, sc_pair = local_pair_moves(best_h, rng, time_left)
            if sc_pair < best_score:
                best_h, best_score = h_pair.copy(), sc_pair

        # Random continuous restarts with brief refinement
        if time_left() > 5:
            for _ in range(2):
                if time_left() < 2:
                    break
                h_rand = rng.random(n)
                h_rand = project_onto_box_sum(h_rand, n / 2.0)
                max_iters_r = int(min(5000, max(500, time_left() * (2000.0 / n))))
                h_rand_ref, sc_rand = gradient_descent_nesterov(h_rand, max_iters_r,
                                                               step0=0.5, beta=0.9,
                                                               rng=rng, time_left=time_left)
                if sc_rand < best_score:
                    best_h, best_score = h_rand_ref.copy(), sc_rand

    # Final intensive refinement
    if time_left() > 2:
        max_iters = int(min(50000, max(10000, time_left() * 20)))
        best_h, best_score = gradient_descent_nesterov(best_h, max_iters,
                                                      step0=0.2, beta=0.9,
                                                      rng=rng, time_left=time_left)

    # Try binary projection
    if time_left() > 2:
        bin_h = binary_projection(best_h)
        bin_score = compute_c5(bin_h)
        if bin_score < best_score:
            best_h, best_score = bin_h, bin_score
        else:
            # small refinement of binary candidate
            max_iters_bin = int(min(8000, time_left() * 5))
            bin_h_ref, bin_score_ref = gradient_descent_nesterov(bin_h, max_iters_bin,
                                                                step0=0.2, beta=0.9,
                                                                rng=rng, time_left=time_left)
            if bin_score_ref < best_score:
                best_h, best_score = bin_h_ref, bin_score_ref

    # One more round of targeted and pair moves
    if time_left() > 2:
        best_h, best_score = improve_via_max_shift(best_h, rng, time_left)
    if time_left() > 2:
        best_h, best_score = local_pair_moves(best_h, rng, time_left)

    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    best_score = compute_c5(best_h)
    return best_h, best_score, best_h.size
